import time, math, random
import numpy as np
from scipy.special import logsumexp  # maybe not needed

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_to_sum(x, target):
    # use water-filling iterative
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # if diff is zero, just return
    if abs(diff) < 1e-12:
        return x
    # simple iterative fill: while diff != 0
    # find indices not at bound
    for _ in range(100):
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        # clip again
        # after update, some may exceed bounds; clip them and recompute diff
        np.clip(x, 0.0, 1.0, out=x)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    # final small correction
    if abs(diff) > 1e-12:
        # distribute diff to any non-bound variable if possible
        for i in range(len(x)):
            if 0.0 < x[i] < 1.0:
                x[i] = min(1.0, max(0.0, x[i] + diff))
                break
    return x

def random_binary_state(n, target_sum):
    # return h array with binary plus possible half if needed
    half = target_sum - int(target_sum)
    int_target = int(np.floor(target_sum))
    ones_needed = int_target
    # half_needed = 1 if half>0 else 0
    # zeros = n - ones_needed - half_needed
    # choose random positions for ones
    indices = np.arange(n)
    np.random.shuffle(indices)
    ones = np.zeros(n, dtype=bool)
    ones[indices[:ones_needed]] = True
    half_idx = None
    if half > 1e-12:
        # pick half position from remaining
        remaining = indices[ones_needed:]
        half_idx = remaining[0]  # pick first
    return ones, half_idx

def binary_h_from_state(ones, half_idx, half_val):
    h = np.zeros_like(ones, dtype=np.float64)
    h[ones] = 1.0
    if half_idx is not None:
        h[half_idx] = half_val
    return h
```

def run(seed=42, budget_s=110, **kwargs):
    import time, math, random
    import numpy as np
    random.seed(seed)
    np.random.seed(seed)

    # Determine number of grid points
    n = 200
    if n % 2 != 0:
        n += 1
    target_sum = n // 2   # integer

    # Helper: compute C5
    def compute_c5(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(corr.max() * dx)

    # Initialize random binary vector with exactly target_sum ones
    def random_binary():
        # choose target_sum distinct indices for ones
        idx = np.arange(n)
        np.random.shuffle(idx)
        ones_idx = idx[:target_sum]
        h = np.zeros(n, dtype=np.float64)
        h[ones_idx] = 1.0
        return h

    # Start with a random binary vector
    cur_h = random_binary()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    start = time.time()
    total = budget_s - 0.5   # leave margin
    base_temp = 0.3          # initial temperature

    # For stagnation detection
    last_improve_time = start

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= total:
            break
        # compute temperature decreasing linearly
        frac = elapsed / total
        temp = base_temp * (1.0 - frac)

        # get indices of ones and zeros
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h < 0.5)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            # Should not happen
            cur_h = random_binary()
            cur_c = compute_c5(cur_h)
            continue

        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # propose swap
        cand_h = cur_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0

        cand_c = compute_c5(cand_h)

        # acceptance criterion
        if cand_c < cur_c or (temp > 0.0 and math.exp((cur_c - cand_c) / temp) > np.random.rand()):
            cur_h = cand_h
            cur_c = cand_c
            if cand_c < best_c:
                best_h = cand_h.copy()
                best_c = cand_c
                last_improve_time = now

        # restart if no improvement for a while (e.g., 5 seconds)
        if now - last_improve_time > 5.0:
            # perturb current state heavily to escape
            cur_h = random_binary()
            cur_c = compute_c5(cur_h)
            last_improve_time = now

    # final check: enforce exact sum (should be already)
    # but due to floating we can correct
    if abs(best_h.sum() - target_sum) > 1e-12:
        # adjust by simple projection (shouldn't happen)
        # use earlier project_to_sum function
        def project_to_sum(x, target):
            x = np.clip(x, 0.0, 1.0)
            diff = target - x.sum()
            for _ in range(50):
                free = (x > 0.0) & (x < 1.0)
                cnt = free.sum()
                if cnt == 0:
                    break
                delta = diff / cnt
                x[free] += delta
                np.clip(x, 0.0, 1.0, out=x)
                diff = target - x.sum()
                if abs(diff) < 1e-12:
                    break
            if abs(diff) > 1e-12:
                idx = np.where((x > 0.0) & (x < 1.0))[0]
                if idx.size:
                    i = idx[0]
                    x[i] = min(1.0, max(0.0, x[i] + diff))
            return x
        best_h = project_to_sum(best_h, target_sum)

    best_c = compute_c5(best_h)
    return (best_h.tolist(), best_c, n)

import time, math, random
import numpy as np

def _c5_value(h):
    """Fast C5 evaluation for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero‑padding (mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _random_binary(n):
    """Return a binary (0/1) vector of length n with exactly n/2 ones."""
    half = n // 2
    idx = np.arange(n)
    np.random.shuffle(idx)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:half]] = 1.0
    return h

def run(seed=42, budget_s=110, **kwargs):
    # ----------------------------------------------------------------------
    # preparation
    # ----------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # Choose a convenient grid size (even, > 1k points are not needed)
    n = 200                     # 200 points give a reasonable resolution
    if n % 2 != 0:               # enforce even length for exact half‑mass binary
        n += 1
    target_sum = n // 2          # exactly half the points are ones

    # ----------------------------------------------------------------------
    # quick random sampling to obtain a decent starting point
    # ----------------------------------------------------------------------
    start_time = time.time()
    total_budget = budget_s - 0.5          # keep a small safety margin
    best_h = _random_binary(n)
    best_c = _c5_value(best_h)

    # allocate ~10 % of the time to pure random exploration
    while time.time() - start_time < 0.10 * total_budget:
        cand = _random_binary(n)
        cand_c = _c5_value(cand)
        if cand_c < best_c:
            best_c = cand_c
            best_h = cand.copy()

    # ----------------------------------------------------------------------
    # simulated‑annealing on binary vectors (swap moves)
    # ----------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c = best_c

    # temperature schedule parameters
    base_temp = 0.4          # start fairly hot
    last_improve = time.time()

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= total_budget:
            break

        # linear cooling
        frac = elapsed / total_budget
        temp = base_temp * (1.0 - frac)

        # indices of 1's and 0's (boolean masks are cheap for n=200)
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h < 0.5)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            # should never happen, but guard against pathological states
            cur_h = _random_binary(n)
            cur_c = _c5_value(cur_h)
            continue

        # propose a swap of a random 1 and a random 0
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        cand_h = cur_h.copy()
        cand_h[i] = 0.0
        cand_h[j] = 1.0

        cand_c = _c5_value(cand_h)

        # Metropolis acceptance
        if cand_c < cur_c or (temp > 0.0 and
                              math.exp((cur_c - cand_c) / temp) > np.random.rand()):
            cur_h = cand_h
            cur_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand_h.copy()
                last_improve = now

        # if we have been idle for a few seconds, do a full random restart
        if now - last_improve > 5.0:
            cur_h = _random_binary(n)
            cur_c = _c5_value(cur_h)
            last_improve = now

    # final sanity projection (binary case is already exact)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = _c5_value(best_h)

    return (best_h.tolist(), best_c, n)
