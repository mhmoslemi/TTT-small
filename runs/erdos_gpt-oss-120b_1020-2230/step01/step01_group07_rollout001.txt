import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function h."""
    # Correlation of h with (1‑h); mode='full' gives all integer shifts.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def _random_feasible_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random vector in [0,1]^n with sum exactly n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=np.float64)

    # integer part of the sum
    ones = int(math.floor(target))
    idx_ones = rng.choice(n, size=ones, replace=False)
    h[idx_ones] = 1.0

    # fractional remainder (0 when n is even)
    remainder = target - ones
    if remainder > 0.0:
        zero_idx = rng.choice(np.where(h == 0)[0])
        h[zero_idx] = remainder
    return h
