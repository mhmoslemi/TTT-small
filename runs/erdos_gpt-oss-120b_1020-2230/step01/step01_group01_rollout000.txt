def project_to_box_sum(y, target):
    # lam binary search
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo+hi)/2.0, 0, 1)

    """
    Euclidean projection of y onto the box [0,1]^n intersected with the hyperplane sum = target.

    lo, hi = -2.0, 2.0

        # ------------------------------------------------------------------
        # evaluate candidate
        new_score = _compute_c5(new_h)
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < -1e-12 or rng.random() < math.exp(-delta_score / T):
            h = new_h
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional restart to escape stagnation
        if no_improve >= max_no_improve:
            h = best_h + rng.normal(scale=0.03, size=n)
            h = _project_onto_box_and_sum(h, target_sum)
            cur_score = _compute_c5(h)
            no_improve = 0

        # early exit if we already beat the target bound
        if best_score <= 0.3808:
            break

    return best_h, float(best_score), n
