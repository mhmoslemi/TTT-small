import numpy as np, time, math

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for the discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _shift_vec(v: np.ndarray, lag: int) -> np.ndarray:
    """Shift with zero padding (non‑periodic)."""
    n = v.shape[0]
    if lag == 0:
        return v.copy()
    if lag > 0:
        out = np.empty_like(v)
        out[:lag] = 0.0
        out[lag:] = v[: n - lag]
        return out
    else:
        k = -lag
        out = np.empty_like(v)
        out[-k:] = 0.0
        out[: n - k] = v[k:]
        return out


def _project_box_sum(v: np.ndarray, target: float, iters: int = 60) -> np.ndarray:
    """
    Euclidean projection of `v` onto the box [0,1]^n intersected with
    the hyperplane sum(x)=target.
    Implements a simple water‑filling binary search for the Lagrange multiplier.
    """
    low = np.min(v) - 1.0          # guarantees all entries become 1 after shift
    high = np.max(v)               # guarantees all entries become 0 after shift
    for _ in range(iters):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _upsample(h_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a coarse grid onto a finer uniform grid."""
    n_small = h_small.shape[0]
    xs = np.linspace(0.0, 2.0, n_small, endpoint=False)
    xl = np.linspace(0.0, 2.0, n_large, endpoint=False)
    return np.interp(xl, xs, h_small)
