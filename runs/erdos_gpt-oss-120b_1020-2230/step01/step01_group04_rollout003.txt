def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def solve_binary_milp(n, time_budget):
    dx = 2.0 / n
    x = cp.Variable(n, boolean=True)
    # Use upper-triangular y to avoid redundant variables
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n // 2,
                   t >= 0]
    # linking constraints
    for i in range(n):
        constraints.append(y[i,i] == x[i])
        for j in range(i+1, n):
            constraints.append(y[i,j] <= x[i])
            constraints.append(y[i,j] <= x[j])
            constraints.append(y[i,j] >= x[i] + x[j] - 1)
            # No need for lower triangle; enforce y[j,i] = y[i,j] maybe not needed
    # for each shift k (0..n-1)

    # objective

import time, math, numpy as np, cvxpy as cp

def _c5_of_h(h):
    """Exact evaluation used by the harness."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _binary_milp(n, time_limit):
    """Solve the binary version of the problem with a MILP.
    Returns (h, t) where h is a {0,1} vector (float) and t is the optimal
    max‑overlap value, or (None, None) on failure / timeout."""
    dx = 2.0 / n
    # binary variables for the function values
    x = cp.Variable(n, boolean=True)
    # binary variables for products x_i * x_j for i<=j
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()                     # the maximal overlap we want to minimise

    constraints = []

    # total mass = n/2  (integral equals 1)
    constraints.append(cp.sum(x) == n // 2)

    # link y to x (exact AND for binary variables)
    for i in range(n):
        constraints.append(y[i, i] == x[i])            # diagonal: y_ii = x_i
        for j in range(i + 1, n):
            constraints.append(y[i, j] <= x[i])
            constraints.append(y[i, j] <= x[j])
            constraints.append(y[i, j] >= x[i] + x[j] - 1)

    # constraints for each shift k (0 … n-1)
    for k in range(n):
        L = n - k                           # length of the overlapping region
        if L == 0:
            continue
        # mass in the overlap
        m_k = cp.sum(x[:L])
        # product in the overlap (only upper‑triangular part is needed)
        prod_terms = [y[i, i + k] for i in range(L)]
        p_k = cp.sum(cp.hstack(prod_terms)) if prod_terms else 0
        # max‑overlap of this shift
        constraints.append(t >= (m_k - p_k) * dx)

    problem = cp.Problem(cp.Minimize(t), constraints)

    # try a few solvers; most environments have CBC, otherwise fall back
    solvers = [cp.CBC, cp.GLPK_MI, cp.SCIP, cp.ECOS_BB]
    for solver in solvers:
        try:
            problem.solve(solver=solver, max_seconds=time_limit, verbose=False)
            if problem.status in ("optimal", "optimal_inaccurate"):
                break
        except Exception:
            continue

    if problem.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    h = np.rint(x.value).astype(float)      # round possible numerical noise
    t_val = float(t.value)
    return h, t_val


def _local_improve(h, best_c5, deadline):
    """Very cheap hill‑climbing: move a random amount from one coordinate to another.
    Stops when time runs out or the target bound is reached."""
    n = h.shape[0]
    dx = 2.0 / n
    target = 0.3808
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        # amount we can move without breaking bounds
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move          # random step size
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = _c5_of_h(h_new)
        if c5_new < best_c5 - 1e-9:               # accept only genuine improvement
            h, best_c5 = h_new, c5_new
            if best_c5 <= target:
                break
    return h, best_c5


def run(seed=42, budget_s=110, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a modest size (even) that gives the optimiser some flexibility.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        # we ignore the parent construction – the binary MILP works best on a fresh grid
        n = len(initial_h_values)
        if n % 2 == 1:
            n += 1
    else:
        n = 48                      # even, comfortably small for the MILP

    # -------------------------------------------------------------
    # 1) Try a global MILP on the binary version of the problem.
    # -------------------------------------------------------------
    time_elapsed = time.time() - start
    milp_budget = max(1.0, budget_s * 0.8 - time_elapsed)   # leave some slack for polishing
    h, t_val = _binary_milp(n, milp_budget)

    if h is not None:
        best_h = h
        best_c5 = t_val
    else:
        # fallback: trivial feasible vector (first half 1, second half 0)
        best_h = np.array([1.0] * (n // 2) + [0.0] * (n // 2), dtype=float)
        best_c5 = _c5_of_h(best_h)

    # -------------------------------------------------------------
    # 2) Cheap local search to polish the solution.
    # -------------------------------------------------------------
    deadline = start + budget_s * 0.96
    best_h, best_c5 = _local_improve(best_h, best_c5, deadline)

    # -------------------------------------------------------------
    # 3) Final mass‑balance correction (numerical drift).
    # -------------------------------------------------------------
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if not np.isclose(cur_sum, target_sum):
        diff = target_sum - cur_sum
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        # if still off, distribute uniformly – this is a tiny adjustment
        cur_sum = best_h.sum()
        if not np.isclose(cur_sum, target_sum):
            best_h += (target_sum - cur_sum) / n
            best_h = np.clip(best_h, 0.0, 1.0)

    # -------------------------------------------------------------
    # 4) Evaluate with the provided helper (or the internal one).
    # -------------------------------------------------------------
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = _c5_of_h(best_h)

    return best_h, float(final_c5), int(n)
