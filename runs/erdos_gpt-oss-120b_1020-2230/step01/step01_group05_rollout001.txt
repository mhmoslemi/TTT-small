import numpy as np
import time
import random

def _compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_box_simplex(v, target):
    """
    Euclidean projection of v onto {0 ≤ x_i ≤ 1, sum(x)=target}.
    Simple bisection on the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    Heuristic optimisation for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).

    np.random.seed(seed)
    random.seed(seed)
    rng = np.random.default_rng(seed)

    # low in the middle) which is known to give a low overlap.
    dx = 2.0 / n
    x = np.arange(n) * dx + dx * 0.5         # mid‑point sampling
    tri = np.minimum(x, 2.0 - x)             # triangular spike (0→1→0)
    h = 1.0 - tri                            # complementary U‑shape, values in [0,1]

    # if a parent construction exists, blend it in (interpolated)
    if init_global is not None and init_global.size > 0:
        x_old = np.linspace(0.0, 2.0, init_global.size, endpoint=False)

        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        lag = idx_max - (n - 1)               # integer lag (may be negative)

        # ----- propose a new candidate
        if lag == 0:

            i, j = rng.integers(0, n, size=2, replace=False)

        it += 1

    best_h = _project_box_simplex(best_h, target_sum)
    best_c = _compute_c5(best_h)

    return best_h, best_c, n
