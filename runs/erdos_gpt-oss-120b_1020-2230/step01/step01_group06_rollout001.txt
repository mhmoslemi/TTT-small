def construct_h(n, w_frac):
    L = 2.0  # length
    w = w_frac * L   # width of transition zone
    L1 = (L - w) / 2  # left side 1 region length

def _project_to_sum(x, target_sum):
    # clip
    x = np.clip(x, 0.0, 1.0)
    # water-filling
    diff = target_sum - x.sum()
    iter = 0
    while abs(diff) > 1e-10 and iter < 1000:
        # free indices: those not at bound
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # allocate diff to any index within bounds arbitrarily
            # if diff >0, add to first not at 1; else subtract from first >0
            if diff > 0:
                idx = np.where(x < 1.0 - 1e-12)[0]
                if idx.size > 0:
                    x[idx[0]] = min(1.0, x[idx[0]] + diff)
            else:
                idx = np.where(x > 1e-12)[0]
                if idx.size > 0:
                    x[idx[0]] = max(0.0, x[idx[0]] + diff)
            break
        delta = diff / cnt
        # apply delta, may violate bounds, so clip
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
        iter += 1
    return x

def _compute_c5_fft(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[:2*n-1]
    # conv length 2n-1 corresponds to full mode
    return float(conv.max() * (2.0 / n))

import time, math, random
import numpy as np

# ---------- basic utilities ----------
def _project_to_sum(x, target_sum, eps=1e-12, max_iter=200):
    """Project vector x onto the box [0,1] with exact sum = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(max_iter):
        if abs(diff) < eps:
            break
        free = (x > eps) & (x < 1.0 - eps)
        cnt = free.sum()
        if cnt == 0:
            # adjust a single coordinate if needed
            if diff > 0:
                idx = np.where(x < 1.0 - eps)[0]
            else:
                idx = np.where(x > eps)[0]
            if idx.size:
                i = idx[0]
                x[i] = np.clip(x[i] + diff, 0.0, 1.0)
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
    return x

def _full_corr(h):
    """Full linear correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * G, n=L)[: 2 * n - 1]
    return conv

def _c5_value(h):
    """Compute C5 for a step function h on [0,2] (discrete version)."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(_full_corr(h).max() * dx)

def _construct_linear_ramp(n, w_frac):
    """Build a monotone decreasing shape: plateaus of 1 and 0 with a linear ramp.
    w_frac ∈ [0,1] is the transition width as a fraction of the interval length (2)."""
    L = 2.0
    w = w_frac * L
    left = (L - w) / 2.0
    dx = L / n
    xs = (np.arange(n) + 0.5) * dx   # mid‑point sampling
    h = np.where(xs <= left, 1.0,
                 np.where(xs >= left + w, 0.0,
                          1.0 - (xs - left) / w))
    target = n / 2.0
    return _project_to_sum(h, target)
