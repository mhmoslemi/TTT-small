def project(v, target):
    lower, upper = 0.0, 1.0
    lo = v.min() - upper
    hi = v.max() - lower
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2.0), lower, upper)

import time, math
import numpy as np

def _c5_value(h):
    """C5 value for a discretized h (unscaled correlation)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / h.size)

def _project_sum_bounds(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Project v onto [lower,upper] with exact sum = target."""
    lo = v.min() - upper
    hi = v.max() - lower
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, lower, upper)

def run(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- problem size -------------------------------------------------
    n = 199                       # finer discretisation (odd → half‑integer total)
    target = n / 2.0              # integral of h must be 1 ⇒ sum = n/2

    # --- initialise ----------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        # linear interpolation to the new grid
        x_old = np.linspace(0, 2, init.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        h = _project_sum_bounds(h, target)
    else:
        # random feasible start
        h = rng.random(n)
        h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        h = _project_sum_bounds(h, target)

    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # --- parameters ----------------------------------------------------
    alpha = 25.0                   # soft‑max sharpness
    step0 = 0.6                    # initial gradient step
    sa_T0 = 0.03                   # simulated annealing temperature
    max_sa_iter = 2000

    # --- main gradient‑descent loop ------------------------------------
    it = 0
    while time.time() - start < budget_s - 1.0:
        # correlation (overlaps)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        # soft‑max weights (focus on the worst shift)
        m = corr.max()
        w_raw = np.exp(alpha * (corr - m))
        w = w_raw / w_raw.sum()               # shape (2n‑1,)

        # termA = sum of w over a sliding window of length n
        cum = np.cumsum(w)
        termA = cum[n - 1:] - np.concatenate(([0.0], cum[:-n]))

        # convolution terms for the product part of the gradient
        pad = np.zeros(n - 1, dtype=np.float64)
        h_pad = np.concatenate([pad, cur_h, pad])
        conv = np.correlate(h_pad, w, mode="valid")          # sum wℓ·h_{i+ℓ}
        conv_rev = np.correlate(h_pad, w[::-1], mode="valid")  # sum wℓ·h_{i‑ℓ}
        grad = termA - (conv + conv_rev)

        # diminishing step size
        elapsed = time.time() - start
        T = max(0.0, 1.0 - elapsed / budget_s)
        step = step0 * T

        # gradient step and projection
        new_h = cur_h - step * grad
        new_h = _project_sum_bounds(new_h, target)

        new_val = _c5_value(new_h)

        # simple Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (1e-4 + T)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()

        # occasional random jitter to escape plateaus
        if it % 70 == 0:
            idx = rng.choice(n, size=min(7, n), replace=False)
            delta = rng.uniform(-0.15, 0.15, size=idx.shape)
            cur_h[idx] = np.clip(cur_h[idx] + delta, 0.0, 1.0)
            cur_h = _project_sum_bounds(cur_h, target)
        it += 1

    # --- short simulated‑annealing refinement ---------------------------
    # start from the best found vector
    cur_h = best_h.copy()
    cur_val = best_val
    elapsed = time.time() - start
    rem = budget_s - elapsed - 0.3
    sa_iters = int(min(max_sa_iter, rem / 0.0003))  # approximate timing

    for i in range(sa_iters):
        # two‑coordinate mass transfer preserving the sum
        a, b = rng.choice(n, size=2, replace=False)
        ha, hb = cur_h[a], cur_h[b]
        delta_low = max(-ha, hb - 1.0)
        delta_high = min(1.0 - ha, hb)
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)
        new_h = cur_h.copy()
        new_h[a] = ha + delta
        new_h[b] = hb - delta

        new_val = _c5_value(new_h)

        # annealing temperature decays linearly
        T = sa_T0 * (1.0 - i / sa_iters)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()

    # final safety projection and value recomputation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_bounds(best_h, target)
    final_val = _c5_value(best_h)

    return best_h, float(final_val), int(n)
