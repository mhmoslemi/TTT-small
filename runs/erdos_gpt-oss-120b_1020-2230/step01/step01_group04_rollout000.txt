import numpy as np, time, math

def _compute_c5(h):
    """Maximum overlap integral approximated on the sample grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _project_to_target(x, target):
    """Shift all entries by a common amount and clamp to [0,1] so that sum==target."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        v = np.clip(x + mid, 0.0, 1.0)
        if v.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + hi, 0.0, 1.0)


def _random_feasible(n, target, rng):
    """Generate a random point in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return _project_to_target(v, target)


def run(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible vector (use provided one if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    else:
        n = 42
        h0 = _random_feasible(n, n / 2.0, rng)

    target = n / 2.0
    h0 = _project_to_target(h0, target)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing helper
    # ------------------------------------------------------------------
    start_time = time.time()
    total_end = start_time + budget_s * 0.96   # tiny margin for housekeeping
    start_temp = 2.0
    final_temp = 1e-4
    p_restart = 0.003

    current_h = best_h.copy()
    current_c5 = best_c5

    # ------------------------------------------------------------------
    # 3. Main SA loop (works for the current discretisation size)
    # ------------------------------------------------------------------
    def _sa_step(cur_h, cur_c5, n_now, targ):
        """Perform a single SA step and return updated (h,c5)."""
        # temperature based on global time
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.96)
        temp = start_temp * (final_temp / start_temp) ** frac

        # occasional restart around the best solution
        if rng.random() < p_restart:
            pert = best_h.copy()
            for _ in range(3):
                i, j = rng.choice(n_now, 2, replace=False)
                max_tr = min(pert[i], 1.0 - pert[j])
                if max_tr <= 0:
                    continue
                delta = max_tr * rng.random() * 0.5
                pert[i] -= delta
                pert[j] += delta
            return pert, _compute_c5(pert)

        r = rng.random()
        if r < 0.20:
            # pure swap – sum preserved automatically
            i, j = rng.choice(n_now, 2, replace=False)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        elif r < 0.55:
            # random transfer between two positions
            i, j = rng.choice(n_now, 2, replace=False)
            max_tr = min(cur_h[i], 1.0 - cur_h[j])
            if max_tr <= 0:
                return cur_h, cur_c5
            delta = max_tr * rng.random()
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        else:
            # directed transfer based on the worst shift
            full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx = np.argmax(full_corr)
            lag = idx - (n_now - 1)

            # derivative d_i = 1 - (h_{i+lag}+h_{i-lag})
            d = np.ones(n_now)
            if lag > 0:
                d[: n_now - lag] -= cur_h[lag:]
                d[lag:] -= cur_h[: n_now - lag]
            elif lag < 0:
                d2 = -lag
                d[d2:] -= cur_h[: n_now - d2]
                d[: n_now - d2] -= cur_h[d2:]
            else:  # lag == 0
                d = 1.0 - 2.0 * cur_h

            pos = np.where(d > 1e-9)[0]
            neg = np.where(d < -1e-9)[0]
            if pos.size == 0 or neg.size == 0:
                # fallback to random transfer
                i, j = rng.choice(n_now, 2, replace=False)
                max_tr = min(cur_h[i], 1.0 - cur_h[j])
                if max_tr <= 0:
                    return cur_h, cur_c5
                delta = max_tr * rng.random()
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                i = rng.choice(pos)
                j = rng.choice(neg)
                max_tr = min(cur_h[i], 1.0 - cur_h[j])
                if max_tr <= 0:
                    # fallback
                    i, j = rng.choice(n_now, 2, replace=False)
                    max_tr = min(cur_h[i], 1.0 - cur_h[j])
                    if max_tr <= 0:
                        return cur_h, cur_c5
                    delta = max_tr * rng.random()
                else:
                    # move a reasonable fraction of the allowed amount
                    delta = max_tr * rng.random() * 0.9
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        new_c5 = _compute_c5(new_h)

        # Metropolis acceptance
        diff = cur_c5 - new_c5
        if diff > 0 or rng.random() < math.exp(diff / temp):
            return new_h, new_c5
        else:
            return cur_h, cur_c5

    # ------------------------------------------------------------------
    # 4. Run SA until time out or target reached
    # ------------------------------------------------------------------
    while time.time() < total_end:
        current_h, current_c5 = _sa_step(current_h, current_c5, n, target)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            if best_c5 <= 0.3808:
                break

    # ------------------------------------------------------------------
    # 5. Optional refinement with a finer discretisation (n -> 2n)
    # ------------------------------------------------------------------
    remaining = total_end - time.time()
    if remaining > 5.0 and n == 42:
        # up‑sample by simple duplication; integral stays the same
        best_h = np.repeat(best_h, 2)
        n = best_h.shape[0]
        target = n / 2.0
        best_h = _project_to_target(best_h, target)

        current_h = best_h.copy()
        current_c5 = _compute_c5(current_h)

        # run SA on the finer grid for the leftover time
        while time.time() < total_end:
            current_h, current_c5 = _sa_step(current_h, current_c5, n, target)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                if best_c5 <= 0.3808:
                    break

    # ------------------------------------------------------------------
    # 6. Final feasibility correction
    # ------------------------------------------------------------------
    cur_sum = best_h.sum()
    if not np.isclose(cur_sum, target, atol=1e-12):
        # tiny adjustment on the first entry
        best_h[0] = np.clip(best_h[0] + (target - cur_sum), 0.0, 1.0)

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided helper (fallback to our own if missing)
    # ------------------------------------------------------------------
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(n)
