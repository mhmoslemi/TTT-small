import time, math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def compute_c5_binary(h: np.ndarray, dx: float) -> float:
    """C5 for a binary (0/1) step function using the autocorrelation."""
    # autocorrelation of h with itself, full length
    overlap = np.correlate(h, h, mode="full")
    min_overlap = overlap.min()
    return 1.0 - min_overlap * dx


def compute_c5_cont(h: np.ndarray, dx: float) -> float:
    """C5 for a general [0,1] step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_initial_h(n: int, target_sum: float):
    """
    Produce a feasible binary start:
    - if a parent `initial_h_values` exists and repeats nicely, use it;
    - otherwise a random half‑filled vector.
    """
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        m = parent.size
        if n % m == 0:
            factor = n // m
            h = np.repeat(parent, factor)
            # parent is expected to satisfy the sum constraint already
            return h
    except Exception:
        pass

    # fallback: random binary with exactly target_sum ones
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, int(target_sum), replace=False)
    h[idx] = 1.0
    return h

    if abs(cur_sum - target_sum) > 1e-12:

        end_time = start_time + budget_s - 0.05

        while time.time() < end_time:
            # linear temperature schedule over the whole remaining budget
            now = time.time()
            frac = (now - start_time) / budget_s
            T = T0 * (1.0 - frac) + T1 * frac

            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = cont_h[i], cont_h[j]

            # pick a direction and a feasible transfer amount
            if np.random.rand() < 0.5:
                max_tr = min(hi, 1.0 - hj)
                if max_tr <= 0:
                    continue
                delta = np.random.rand() * max_tr
                new_h = cont_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                max_tr = min(hj, 1.0 - hi)
                if max_tr <= 0:
                    continue
                delta = np.random.rand() * max_tr
                new_h = cont_h.copy()
                new_h[i] += delta
                new_h[j] -= delta

            new_c = compute_c5_cont(new_h, dx)

            if new_c + 1e-12 < cont_c:
                cont_h, cont_c = new_h, new_c
                if new_c < best_c:
                    best_c, best_h = new_c, new_h.copy()
            else:
                if np.random.rand() < math.exp(-(new_c - cont_c) / max(T, 1e-12)):
                    cont_h, cont_c = new_h, new_c

    # --------------------------------------------------------------
    # 4. Final exactness fix (enforce sum and bounds)
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute the remaining mass greedily
        order = np.random.permutation(n)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                cap = 1.0 - best_h[i]
                add = min(cap, diff)
                best_h[i] += add
                diff -= add
            else:
                cap = best_h[i]
                sub = min(cap, -diff)
                best_h[i] -= sub
                diff += sub

    final_c = compute_c5_cont(best_h, dx)

    return best_h, final_c, n
