def _project_to_sum(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    for _ in range(10):
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            break
    # If still diff not zero, adjust a single variable
    if abs(diff) > 1e-12:
        # try to add to first free
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            i = idx[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

import time
import math
import random
import numpy as np

def _project_to_sum(x, target):
    """Project x onto the box [0,1] and the hyperplane sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # simple water‑filling on the free variables
    for _ in range(20):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # any tiny residual is forced onto the first free entry
    if abs(diff) > 1e-12:
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            i = idx[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            # fallback: adjust a random coordinate
            i = random.randrange(len(x))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def _compute_c5(h):
    """Return the current C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _grad_for_shift(h, shift):
    """Sub‑gradient of the worst‑shift term C_shift = Σ h_i(1‑h_{i+shift})."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    i_start = max(0, -shift)
    i_end = min(n, n - shift)
    i = np.arange(i_start, i_end, dtype=int)
    j = i + shift
    # contribution from the linear part ( +1 )
    grad[i] += 1.0 - h[j]
    # contribution from the quadratic part ( -h_i )
    grad[j] += -h[i]
    return grad


def run(seed=42, budget_s=110, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # problem size – 200 points give a good trade‑off between accuracy and speed
    n = 200
    target_sum = n / 2.0

    # ----- initial construction (two symmetric blocks) -----
    block_len = n // 4
    h = np.zeros(n, dtype=np.float64)
    h[:block_len] = 1.0
    h[-block_len:] = 1.0
    h = _project_to_sum(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    # parameters for the sub‑gradient descent
    max_no_improve = 2000          # after this many useless steps we randomise
    no_improve_cnt = 0
    iteration = 0

    while True:
        now = time.time()
        if now - start >= budget_s - 0.5:
            break

        # evaluate the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # gradient of that worst term
        grad = _grad_for_shift(cur_h, shift)

        # step size: linear cooling + normalisation
        elapsed = now - start
        frac = elapsed / (budget_s - 0.5)
        base_step = 0.6 * (1.0 - frac) + 0.02   # from ~0.62 down to ~0.02
        norm = np.linalg.norm(grad) + 1e-12
        step = base_step / norm

        # propose a new point
        cand = cur_h - step * grad
        cand = _project_to_sum(cand, target_sum)
        cand_c = _compute_c5(cand)

        if cand_c < cur_c - 1e-12:          # improvement
            cur_h, cur_c = cand, cand_c
            no_improve_cnt = 0
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c
        else:
            no_improve_cnt += 1
            # occasional random jitter to escape plateaus
            if no_improve_cnt % 500 == 0:
                jitter = (np.random.rand(n) - 0.5) * 0.04
                cand2 = cur_h + jitter
                cand2 = _project_to_sum(cand2, target_sum)
                cand2_c = _compute_c5(cand2)
                if cand2_c < cur_c:
                    cur_h, cur_c = cand2, cand2_c
                    no_improve_cnt = 0
                    if cand2_c < best_c:
                        best_h, best_c = cand2.copy(), cand2_c

        # random restart if we are stuck for a long time
        if no_improve_cnt >= max_no_improve:
            cur_h = np.random.rand(n)
            cur_h = _project_to_sum(cur_h, target_sum)
            cur_c = _compute_c5(cur_h)
            no_improve_cnt = 0

        iteration += 1

    # final projection to guarantee feasibility
    best_h = _project_to_sum(best_h, target_sum)
    best_c = _compute_c5(best_h)

    return (best_h.tolist(), best_c, n)
