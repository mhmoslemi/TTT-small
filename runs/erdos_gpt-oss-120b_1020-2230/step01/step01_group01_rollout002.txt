def compute_c5(h):
    # compute correlation h with (1-h) via FFT (circular). Then max * dx.
    n = h.shape[0]
    # Using np.correlate is O(n^2), fine for n~200 maybe but many iterations -> large overhead.
    # Using FFT: correlation is inverse FFT of fft(h) * conj(fft(1-h)). Equivalent to convolution.
    # Compute conv = ifft(fft(h)* conj(fft(1-h))) = autocorr of h with 1-h (circular).
    # Then max value.
    f = np.fft.fft(h)
    g = np.fft.fft(1.0 - h)
    conv = np.real(np.fft.ifft(f * np.conj(g)))
    max_val = conv.max()
