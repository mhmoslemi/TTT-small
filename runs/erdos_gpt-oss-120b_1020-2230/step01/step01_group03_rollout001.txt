def gradient_softmax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1-h, mode='full') * dx  # length 2n-1
    maxf = f.max()
    exp_terms = np.exp(alpha * (f - maxf))  # stable
    w = exp_terms / exp_terms.sum()
    grad = np.zeros_like(h)
    centre = n - 1
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - centre
        if s >= 0:
            m = n - s
            # vectorized:
            # contributions:
            # g[:m] += weight*(1 - h[s:])  
            # g[s:] += weight*(-h[:m])
            grad[:m] += weight * (1 - h[s:])  # shape m
            grad[s:] += weight * (-h[:m])
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += weight * (1 - h[:m])
            grad[:m] += weight * (-h[s2:])
    grad *= dx
    return grad, f.max()

import numpy as np
import time
import math

def _project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto {0<=x<=1, sum(x)=target_sum} by bisection."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def _grad_softmax_max(h: np.ndarray, alpha: float):
    """
    Gradient (and current max value) of the smoothed max
    max_s f_s(h) where f_s = dx * sum_i h_i (1 - h_{i+s}).
    Uses a softmax weighting with parameter alpha.
    """
    n = h.shape[0]
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode="full") * dx  # shape 2n-1
    max_f = f.max()
    # stable softmax
    exp_terms = np.exp(alpha * (f - max_f))
    w = exp_terms / exp_terms.sum()
    grad = np.zeros_like(h)
    centre = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - centre
        if s >= 0:
            m = n - s
            # contributions for shift s
            grad[:m] += weight * (1.0 - h[s:])      # ∂/∂h_i
            grad[s:] += weight * (-h[:m])           # ∂/∂h_{i+s}
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += weight * (1.0 - h[:m])
            grad[:m]   += weight * (-h[s2:])
    grad *= dx
    return grad, float(max_f)


def _random_feasible(n: int, target_sum: float) -> np.ndarray:
    """Binary start (exactly target_sum ones) with a small random perturbation."""
    h = np.zeros(n, dtype=float)
    ones = int(target_sum)
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    # add modest noise to escape corners
    h += (np.random.rand(n) - 0.5) * 0.2
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum_bounds(h, target_sum)
    return h


def _local_refine(h: np.ndarray, target_sum: float, deadline: float):
    """Simple simulated‑annealing style hill climbing (mass swaps)."""
    n = h.shape[0]
    cur = h.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val
    T = 0.01
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * np.random.rand() * 0.5
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _c5_value(new)
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
        T = max(T * 0.9999, 1e-6)
    return best, best_val

        alpha = 3.0  # start softening

        for it in range(max_iters_per_run):
            if time.time() - start_time > budget_s * 0.85:
                break
            grad, cur_val = _grad_softmax_max(h, alpha)

            # record best seen so far
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()

            # Adam update (gradient descent on smoothed max)
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            h = _project_sum_bounds(np.clip(h - step, 0.0, 1.0), target_sum)

            # slowly sharpen the softmax
            alpha = min(alpha * 1.030, 40.0)

        # occasional tiny refinement of the current best (optional, cheap)
        if time.time() - start_time > budget_s * 0.75:
            break

    # ------------------------------------------------------------------
    # 3. Final local refinement on the best candidate found
    # ------------------------------------------------------------------
    remaining = start_time + budget_s - time.time()
    if remaining > 2.0:
        deadline = time.time() + remaining * 0.95
        best_h, _ = _local_refine(best_h, target_sum, deadline)

    # ------------------------------------------------------------------
    # 4. Final projection and exact evaluation
    # ------------------------------------------------------------------
    best_h = _project_sum_bounds(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, float(best_val), n
