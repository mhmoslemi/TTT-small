def project_sum_bounds(v, target_sum):
    # using bisection same as before
    low, high = -1.0, 2.0 # maybe lower than -10, 10 (excess)
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) / 2.0, 0.0, 1.0)

def c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

import numpy as np
import time
import math

def _project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target_sum} by bisection on a Lagrange multiplier."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact evaluation of the Erdős overlap functional on the discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def _subgradient_and_max(h: np.ndarray):
    """Sub‑gradient of the max‑overlap functional and its value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_val = vals.max()
    # indices attaining the max (within a tiny tolerance)
    act = np.where(vals >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    centre = n - 1
    for idx in act:
        s = idx - centre          # shift amount
        if s >= 0:                # h_i interacts with h_{i+s}
            m = n - s
            grad[:m] += (1.0 - h[s:])   # derivative w.r.t. the first factor
            grad[s:] += -h[:m]          # derivative w.r.t. the second factor
        else:                     # negative shift: treat symmetrically
            s2 = -s
            m = n - s2
            grad[s2:] += (1.0 - h[:m])
            grad[:m] += -h[s2:]
    grad *= dx
    return grad, float(max_val)

    # ------------------------------------------------------------------

    n0 = init.shape[0]
