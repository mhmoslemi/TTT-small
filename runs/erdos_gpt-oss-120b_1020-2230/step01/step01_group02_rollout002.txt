def mass_transfer_move(h, max_delta):
    n = len(h)
    i, j = np.random.randint(n, size=2)
    while i == j or h[i] <= 1e-12 and h[j] >= 1 - 1e-12:
        i, j = np.random.randint(n, size=2)
    # move mass from i to j
    delta_max = min(max_delta, h[i], 1 - h[j])
    if delta_max <= 1e-12:
        return None
    delta = np.random.uniform(0, delta_max)
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

def _mass_transfer_move(h, max_delta):
    n = len(h)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    # Decide direction: move from i to j
    # Ensure we can move some mass
    delta_max = min(max_delta, h[i], 1.0 - h[j])
    if delta_max <= 1e-12:
        # try opposite direction
        delta_max = min(max_delta, h[j], 1.0 - h[i])

        i, j = j, i
    delta = np.random.uniform(0, delta_max)
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return np.clip(x, 0.0, 1.0)

def _gradient_C_at_shift(h, shift):
    """Gradient of C(s)=Σ h_i(1‑h_{i+shift}) with respect to h."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += 1.0 - h[shift:]
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g

def _mass_transfer_move(h, max_delta):
    """Move a small amount of mass from i to j while keeping the sum."""
    n = len(h)
    if max_delta <= 0:
        return None
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    delta_max = min(max_delta, h[i], 1.0 - h[j])
    if delta_max <= 1e-12:
        # try the opposite direction
        delta_max = min(max_delta, h[j], 1.0 - h[i])
        if delta_max <= 1e-12:
            return None
        i, j = j, i
    delta = np.random.uniform(0.0, delta_max)
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

def _swap_move(h):
    """Swap a 1 and a 0 (preserves the sum exactly)."""
    ones = np.where(h > 0.9999)[0]
    zeros = np.where(h < 1e-12)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    cand = h.copy()
    cand[i] = 0.0
    cand[j] = 1.0
    return cand

def _random_feasible_vector(n, target):
    """Generate a random feasible vector (binary + at most one fraction)."""
    h = np.zeros(n, dtype=np.float64)
    ones = int(math.floor(target))
    if ones > 0:
        idx = np.random.choice(n, ones, replace=False)
        h[idx] = 1.0
    frac = target - ones
    if frac > 1e-12:
        zero_idx = np.random.choice(np.where(h == 0)[0])
        h[zero_idx] = frac
    return h

def run(seed=42, budget_s=110, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        n = 95
        target = n / 2.0
        h0 = np.zeros(n, dtype=np.float64)
        ones_needed = int(math.floor(target))
        # create an alternating pattern and shuffle a bit
        offset = np.random.randint(2)
        idx = np.arange(offset, n, 2)[:ones_needed]
        h0[idx] = 1.0
        if n % 2:
            zero_idx = np.random.choice(np.where(h0 == 0)[0])
            h0[zero_idx] = target - ones_needed  # usually 0.5
    dx = 2.0 / n
    target = n / 2.0
    h0 = _project_feasible(h0, target)

    cur_h = h0.copy()
    cur_val = _compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # optimisation loop (simulated annealing + local moves)
    it = 0
    since_last_improve = 0
    while time.time() - start < budget_s - 0.1:
        elapsed = time.time() - start
        progress = elapsed / budget_s

        # temperature & step sizes
        T = max(1e-5, 0.01 * (1.0 - progress) + 0.0005)
        sigma = 0.5 * (1.0 - progress) + 0.01
        step_g = 0.3 * (1.0 - progress) + 0.01
        max_delta = sigma

        r = np.random.rand()
        cand = None

        if r < 0.2:
            cand = _swap_move(cur_h)
        elif r < 0.5:
            cand = _mass_transfer_move(cur_h, max_delta)
        elif r < 0.8:
            cand = cur_h + sigma * np.random.randn(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)
        else:
            # gradient step on the worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            max_idx = corr.argmax()
            shift = max_idx - (n - 1)
            grad = _gradient_C_at_shift(cur_h, shift)
            cand = cur_h - step_g * grad
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)

        if cand is None:
            continue

        val = _compute_c5(cand, dx)
        delta = val - cur_val

        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_h = cand
            cur_val = val

        if val < best_val:
            best_val = val
            best_h = cand.copy()
            since_last_improve = 0
        else:
            since_last_improve += 1

        # restart if we have been stuck for a while
        if since_last_improve > 3000:
            if np.random.rand() < 0.5:
                cur_h = best_h.copy()
                cur_val = best_val
            else:
                cur_h = _random_feasible_vector(n, target)
                cur_val = _compute_c5(cur_h, dx)
            since_last_improve = 0

        it += 1

    # ------------------------------------------------------------------
    # final local refinement (tiny gradient steps)
    for _ in range(200):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        shift = corr.argmax() - (n - 1)
        grad = _gradient_C_at_shift(best_h, shift)
        improved = False
        for step in (0.05, 0.02, 0.01, 0.005):
            cand = best_h - step * grad
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)
            val = _compute_c5(cand, dx)
            if val + 1e-12 < best_val:
                best_h = cand
                best_val = val
                improved = True
                break
        if not improved:
            break

    best_h = _project_feasible(best_h, target)
    final_val = _compute_c5(best_h, dx)

    return best_h, final_val, n
