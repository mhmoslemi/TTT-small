import numpy as np
import time
import math

def _project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def _subgradient_and_max(h: np.ndarray):
    """Sub‑gradient of the max‑overlap functional and its current value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_val = vals.max()
    act = np.where(vals >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    centre = n - 1
    for idx in act:
        s = idx - centre
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])          # d/ dh_i component
            grad[s:] += -h[:m]                # d/ dh_{i+s} component
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += (1.0 - h[:m])
            grad[:m] += -h[s2:]
    grad *= dx
    return grad, float(max_val)

    # ------------------------------------------------------------------

    h = np.interp(x, x0, h0)                     # linear interpolation

    h += 0.02 * (np.random.rand(n) - 0.5)

    h = _project_sum_bounds(h, target_sum)

    best_h = h.copy()

    # ------------------------------------------------------------------
