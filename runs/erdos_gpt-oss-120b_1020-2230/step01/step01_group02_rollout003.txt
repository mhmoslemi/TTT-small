def _project_feasible(x, target, max_iter=100):
    """Project x onto {0≤x_i≤1, sum(x)=target} by water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x

def _gradient_C_at_shift(h, shift):
    """Gradient of C(s)=Σ h_i (1‑h_{i+shift}) w.r.t. h."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += 1.0 - h[shift:]
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g

def run(seed=42, budget_s=110, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
    else:
        n = 100                         # even size gives integer target
        h = np.zeros(n, dtype=np.float64)
        k = n // 2
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
    dx = 2.0 / n
    target = n / 2.0
    h = _project_feasible(h, target)

    current_h = h.copy()
    current_val = _compute_c5(current_h, dx)
    best_h = current_h.copy()
    best_val = current_val

    # --------------------------------------------------------------
    # Simulated‑annealing / local‑search loop
    init_temp = best_val * 5.0 + 1e-3   # a generous start temperature
    min_temp = 1e-6
    it = 0
    while time.time() - start < budget_s - 0.5:
        elapsed = time.time() - start
        progress = elapsed / budget_s
        # temperature cooling (linear)
        T = max(min_temp, init_temp * (1.0 - progress))

        # correlation of the current state
        corr = np.correlate(current_h, 1.0 - current_h, mode='full')
        max_idx = corr.argmax()
        max_shift = max_idx - (n - 1)

        made_move = False

        # ----------------------------------------------------------
        # Targeted swap that reduces the worst shift directly
        if max_shift >= 0:
            rng = np.arange(n - max_shift)
            i1 = rng
            i2 = rng + max_shift
        else:
            s = -max_shift
            i1 = np.arange(s, n)
            i2 = i1 - s

        bad = i1[(current_h[i1] > 0.5) & (current_h[i2] < 0.5)]
        good = i1[(current_h[i1] < 0.5) & (current_h[i2] > 0.5)]

        if bad.size and good.size:
            i = np.random.choice(bad)
            j = np.random.choice(good)
            cand = current_h.copy()
            cand[i], cand[j] = cand[j], cand[i]          # swap values
            cand_val = _compute_c5(cand, dx)
            delta = cand_val - current_val
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h, current_val = cand, cand_val
                made_move = True
                if cand_val < best_val:
                    best_val, best_h = cand_val, cand.copy()

        # ----------------------------------------------------------
        # If swap not possible or not taken, fall back to
        #   1) random Gaussian perturbation  OR
        #   2) gradient step on the worst shift
        if not made_move:
            if np.random.rand() < 0.5:
                # Gaussian perturbation
                sigma = 0.12 * (1.0 - progress) + 0.02
                cand = current_h + sigma * np.random.randn(n)
                cand = np.clip(cand, 0.0, 1.0)
                cand = _project_feasible(cand, target)
            else:
                # Gradient move
                grad = _gradient_C_at_shift(current_h, max_shift)
                step = 0.25 * (1.0 - progress) + 0.01
                cand = current_h - step * grad
                cand = np.clip(cand, 0.0, 1.0)
                cand = _project_feasible(cand, target)

            cand_val = _compute_c5(cand, dx)
            delta = cand_val - current_val
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h, current_val = cand, cand_val
                if cand_val < best_val:
                    best_val, best_h = cand_val, cand.copy()

        it += 1

    # final safety projection and evaluation
    best_h = _project_feasible(best_h, target)
    final_val = _compute_c5(best_h, dx)

    return best_h, final_val, n
