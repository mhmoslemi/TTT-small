def _compute_c5(h):
    n = len(h)

    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

import time, math, numpy as np

def _compute_c5(h):
    """Maximum overlap for a step function sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _milp_feasible(bound, n, target, time_limit):
    """
    Try to find a binary vector h with sum(target) and
    max overlap <= bound using a mixed‑integer linear program.
    Returns the feasible vector (as float) or None.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    # decision variables
    h = cp.Variable(n, boolean=True)
    constraints = [cp.sum(h) == target]

    # product variables for each lag
    for lag in range(1, n):
        m = n - lag                     # length of overlapping region
        y = cp.Variable(m, boolean=True)

        # linearisation of y = h[i] * h[i+lag]
        constraints.append(y <= h[:m])
        constraints.append(y <= h[lag:])
        constraints.append(y >= h[:m] + h[lag:] - 1)

        # overlap expressions (left and right shifts)
        overlap_left = cp.sum(h[:m]) - cp.sum(y)   # shift +lag
        overlap_right = cp.sum(h[lag:]) - cp.sum(y)  # shift -lag

        constraints.append(overlap_left <= bound)
        constraints.append(overlap_right <= bound)

    prob = cp.Problem(cp.Minimize(0), constraints)

    # try solving, letting the solver use the remaining time budget
    try:
        prob.solve(solver='CBC', verbose=False, max_iters=200000)
    except Exception:
        return None

    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        hv = np.array(h.value).astype(float).ravel()
        hv = np.clip(hv, 0.0, 1.0)
        # tiny correction to satisfy the exact sum (numerical drift)
        cur = hv.sum()
        if not np.isclose(cur, target):
            hv[0] = np.clip(hv[0] + (target - cur), 0.0, 1.0)
        return hv
    return None
