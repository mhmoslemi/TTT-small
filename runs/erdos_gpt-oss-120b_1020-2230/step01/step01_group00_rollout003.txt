def initial_shape(name, n):
    xs = (np.arange(n) + 0.5) * (2.0 / n)  # midpoints
    if name == 'sin2':
        h = np.sin(np.pi * xs / 2) ** 2
    elif name == 'cos':
        h = (1 + np.cos(np.pi * xs)) / 2
    elif name == 'triangular':
        # linear from 0 to 1 to 0 over [0,2]
        h = 1 - np.abs(xs - 1)  # ends at 0 at x=0,2; 1 at x=1
    else:
        ...
    # scaling to sum target
    target = n / 2.0
    s = h.sum()
    factor = target / s
    h *= factor
    # clip and correct any rounding errors
    h = np.clip(h, 0.0, 1.0)
    # fix sum if mis
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # adjust first non-boundary entries
        for i in range(n):
            if diff > 0 and h[i] < 1:
                add = min(diff, 1 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def simulated_annealing(h, start_time, budget, start_T=0.05, end_T=1e-5):
    n = len(h)
    best_h = h.copy()
    best_c = max_overlap(best_h)
    curr_h = h.copy()
    curr_c = best_c
    # compute number of iterations: maybe estimate based on time
    while time.time() - start_time < budget:
        elapsed = time.time() - start_time
        frac = elapsed / budget
        T = start_T * (1 - frac) + end_T * frac

        # choose move type: 70% transfer, 30% swap
        if np.random.rand() < 0.7:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            hi = curr_h[i]; hj = curr_h[j]
            max_ij = min(hi, 1 - hj)
            max_ji = min(hj, 1 - hi)
            if max_ij > 0 and (max_ji == 0 or np.random.rand() < 0.5):
                # move from i to j
                delta = np.random.rand() * max_ij
                new_h = curr_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            elif max_ji > 0:
                delta = np.random.rand() * max_ji
                new_h = curr_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                continue
        else:
            # swap values
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            new_h = curr_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        new_c = max_overlap(new_h)
        delta_c = new_c - curr_c
        if delta_c < 0 or np.random.rand() < math.exp(-delta_c / (T+1e-12)):
            curr_h, curr_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h, new_c

    return best_h, best_c

import time, math
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return C5 bound for a step function h."""
    n = h.size
    dx = 2.0 / n
    # np.correlate computes linear (non‑circular) correlation
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _adjust_sum(h: np.ndarray, target: float) -> None:
    """Modify h in‑place so that its sum equals target, respecting [0,1] bounds."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    # simple forward sweep
    for i in range(h.size):
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # any tiny leftover is ignored (numerical noise)


def _random_binary(n: int) -> np.ndarray:
    """Return a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def _random_continuous(n: int) -> np.ndarray:
    """Return a random vector in [0,1] with sum exactly n/2."""
    h = np.random.rand(n)
    target = n / 2.0
    s = h.sum()
    if s == 0:
        h[:] = 0.5
        s = h.sum()
    h *= target / s
    # enforce strict bounds and exact sum
    np.clip(h, 0.0, 1.0, out=h)
    _adjust_sum(h, target)
    return h


def _simulated_annealing(start_h: np.ndarray, time_limit: float) -> tuple:
    """
    Perform SA on a given start vector.
    time_limit is the absolute timestamp at which the run must stop.
    Returns (best_h, best_c).
    """
    n = start_h.size
    target = n / 2.0
    best_h = start_h.copy()
    best_c = _max_overlap(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    start = time.time()
    total = max(time_limit - start, 1e-9)

    while True:
        now = time.time()
        if now >= time_limit:
            break
        # temperature schedule (linear decay)
        frac = (now - start) / total
        T = 0.08 * (1.0 - frac) + 1e-6 * frac  # start relatively hot

        # choose move type
        if np.random.rand() < 0.7:          # transfer mass
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            max_ij = min(hi, 1.0 - hj)
            max_ji = min(hj, 1.0 - hi)
            if max_ij > 0 and (max_ji == 0 or np.random.rand() < 0.5):
                delta = np.random.rand() * max_ij
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            elif max_ji > 0:
                delta = np.random.rand() * max_ji
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                continue
        else:                               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        # evaluate
        new_c = _max_overlap(new_h)
        delta_c = new_c - cur_c
        if delta_c < -1e-12:
            cur_h, cur_c = new_h, new_c
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        else:
            if T > 0 and np.random.rand() < math.exp(-delta_c / T):
                cur_h, cur_c = new_h, new_c

    # final tidy‑up
    _adjust_sum(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = _max_overlap(best_h)
    return best_h, best_c

            # perturb current best a little
            cur = best_h.copy()
            # perform a few random transfers to diversify
            for _ in range(5):
                i, j = np.random.randint(n, size=2)
                hi, hj = cur[i], cur[j]
                max_ij = min(hi, 1.0 - hj)
                if max_ij <= 0:
                    continue
                delta = max_ij * np.random.rand()
                cur[i] -= delta
                cur[j] += delta
        else:
            # fresh random start (binary)
            cur = _random_binary(n)

        # run SA on this start vector
        local_h, local_c = _simulated_annealing(cur, time.time() + alloc)

        if local_c < best_c:
            best_h, best_c = local_h, local_c

    # ------------------------------------------------------------------
    # 3. Final correction (ensure exact constraints)
    # ------------------------------------------------------------------
    target = n / 2.0
    _adjust_sum(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = _max_overlap(best_h)

    return best_h, final_c, n
