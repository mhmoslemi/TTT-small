import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.special import hermite
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define Hermite polynomials up to H_32 for potential use
    H = [lambda z: hermite(n)(z) for n in range(33)]
    
    # Initial coefficient vectors (4 coefficients)
    initial_coeff = np.array([1.0, 0.5, 0.5, 0.5])
    
    best_coeff = initial_coeff.copy()
    best_score = float('inf')
    
    def objective(c):
        try:
            # Construct Q(z) = sum c_k H_{4k}(z)
            Q = lambda z: sum(c[k] * hermite(4*k)(z) for k in range(len(c)))
            
            # Ensure Q(0) = 0
            Q0 = Q(0)
            if Q0 != 0:
                c = c * Q(0) / Q0
            
            # Ensure positive at infinity by flipping sign if needed
            if Q(np.inf) < 0:
                c = -c
            
            # Remove z^2 factor and find r_max
            def Q_z2(z):
                return Q(z) / z**2
            
            # Find largest positive real root where sign changes
            r_max = 0
            for z in np.linspace(0.1, 100, 1000):
                if np.isclose(Q_z2(z), 0):
                    continue
                if np.sign(Q_z2(z)) != np.sign(Q_z2(z + 1e-6)):
                    r_max = z + 1e-6
                    break
            
            if r_max == 0:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            nonlocal best_score, best_coeff
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial simplex with 5 points for 4 coefficients
    initial_simplex = np.array([
        [1.0, 0.5, 0.5, 0.5],
        [1.5, 0.5, 0.5, 0.5],
        [1.0, 1.0, 0.5, 0.5],
        [1.0, 0.5, 1.0, 0.5],
        [1.0, 0.5, 0.5, 1.0]
    ])
    
    result = minimize(
        objective,
        initial_simplex[0],
        method='Nelder-Mead',
        bounds=[(-2, 2) for _ in range(4)],
        options={'xatol': 1e-8, 'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Refine with Bayesian optimization
    from skopt import BayesSearchCV
    from skopt.space import Real
    from skopt.utils import use_named_args
    
    search_space = Real(-2, 2, name='c0') + Real(-2, 2, name='c1') + Real(-2, 2, name='c2') + Real(-2, 2, name='c3')
    
    optimizer = BayesSearchCV(
        minimize(objective, bounds=search_space, n_initial_points=20),
        search_space,
        n_iter=50,
        random_state=seed,
        scoring='neg_mean_squared_error'
    )
    
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        optimizer.fit({})
    
    # Final refinement with L-BFGS-B
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            optimizer.best_params_[0],
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(4)],
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
