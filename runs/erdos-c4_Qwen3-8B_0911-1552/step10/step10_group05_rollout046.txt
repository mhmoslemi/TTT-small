from scipy.optimize import Bounds, minimize
import time
from scipy.special import hermite

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_score = float('inf')
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct Q(z) = sum c_k H_{4k}(z)
            coeffs = np.array(c)

            Q = np.zeros_like(z)
            for k in range(len(coeffs)):
                H = hermite(4*k, z)
                Q += coeffs[k] * H

            Q[0] = 0
            # Check sign change
            sign_changes = np.where(np.diff(np.sign(Q)) != 0)[0]
            if len(sign_changes) == 0:
                return float('inf')
            r_max = z[sign_changes[-1]]
            U = r_max**2 / (2 * np.pi)
            if U < best_score:
                best_score = U
                best_coeff = c.copy()
            return U
        except:
            return float('inf')
    
    # Initial grid search with adaptive bounds
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):

                        break

                        1.0 + i / grid_points - 2,

                    ])
                    result = minimize(
