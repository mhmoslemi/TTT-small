from scipy.optimize import minimize
import time

from sympy.polys.polytools import resultant

def run(seed=42, budget_s=100, **kwargs):

    z = symbols('z')

    def hermite(n):
        return sum([(-1)**k * 2**k * diff(exp(-z**2), z, k) * z**(n - k) / factorial(n - k) for k in range(n + 1)])

        leading_term = Q.as_coefficients_dict().get(z**16, 0)
        if leading_term < 0:
            Q = -Q
        
        # Remove z^2 factor
        Q = Q / z**2
        
        # Find real roots of Q(z)
        roots = real_roots(Q)
        real_roots_list = [r for r in roots if r.is_real]
        
        # Find r_max as the largest positive real root where sign changes
        r_max = None
        for r in sorted(real_roots_list):

                if left * right < 0:
