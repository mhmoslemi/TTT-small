import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, diff, solve, roots, S

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_c4_coefficients(c):
        m = len(c)
        z = symbols('z')
        q = sum(c[k] * HermiteH(4*k, z) for k in range(m))
        q = q.subs(z, 0)
        if q != 0:
            q = q - q / q
        q = q - q.subs(z, 0)
        q = q / z**2
        r_max = None
        for root in roots(q, z):
            if root.is_real and root > 0:
                multiplicity = diff(q, z).subs(z, root)
                if multiplicity != 0:
                    r_max = root
        if r_max is not None:
            return r_max**2 / (2 * np.pi)
        return float('inf')
    
    def objective(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    def generate_initial_guesses():
        guesses = []
        for _ in range(10):
            c = np.random.uniform(-1, 1, size=8)
            c = np.abs(c) * np.random.choice([1, -1], size=len(c))
            guesses.append(c)
        return guesses
    
    initial_guesses = generate_initial_guesses()
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*len(guess),
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
            best_score = result.fun
    
    return best_coeff
