from scipy.optimize import Bounds, minimize
import time
from sympy import symbols, hermite, simplify, solve, roots, Interval

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    best_coeff_list = []
    
    def objective(c):
        nonlocal best_score, best_coeff, best_coeff_list

            # Symbolic computation
            z = symbols('z')
            expr = sum(c[k] * hermite(4*k, z) for k in range(len(c)))

            expr = expr - expr.subs(z, 0)

            leading_term = expr.as_coefficients_dict().get(z**(4*len(c)), 0)
            if leading_term < 0:
                expr = -expr
            
            # Remove z^2 factor
            expr = expr / z**2
            
            # Find real roots
            roots_expr = roots(expr, z, domain='Real')
            real_roots = [float(r) for r in roots_expr.keys() if r.is_real]
            
            # Sort roots and find r_max
            real_roots.sort()
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            
            if r_max is None:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()

            return score
        except:
            return float('inf')

    grid_points = 16
