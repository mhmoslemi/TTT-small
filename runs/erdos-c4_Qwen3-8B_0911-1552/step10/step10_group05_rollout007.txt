import numpy as np
from scipy.optimize import Bounds, minimize
import time
from sympy import symbols, Poly, diff, hermite, roots

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_score = float('inf')
    best_coeff = np.array([1.0, 1.0, 1.0])
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Find leading term for positivity at infinity
            leading_term = Poly(Q, z).leading_term()
            if leading_term.get('z') % 2 != 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q / z**2
            
            # Find sign change roots
            roots_list = roots(Q, z)
            real_roots = [r for r in roots_list.keys() if r.is_real]
            
            # Sort and find r_max
            real_roots.sort()
            r_max = None
            for r in real_roots:
                if r > 0 and (Q.subs(z, r + 1e-10) * Q.subs(z, r - 1e-10) < 0):
                    r_max = r
                    break
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial Bayesian search
    initial_guesses = np.random.uniform(-2, 2, (10, 8))
    best_guess = initial_guesses[0]
    best_score = float('inf')
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-4, 4)]*len(guess),
            options={'maxiter': 100, 'ft': 1e-12}
        )
        if 'success' in result and result.success:
            best_guess = result.x
            best_score = result.fun
    
    # Final refinement
    result = minimize(
        objective,
        best_guess,
        method='L-BFGS-B',
        bounds=[(-4, 4)]*len(best_guess),

    )
    
    if 'success' in result and result.success:
        best_coeff = result.x
