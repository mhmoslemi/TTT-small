from scipy.optimize import minimize
import time
from sympy import symbols, Hermite, Rational, Poly, solve, Eq, simplify
from sympy.polys.polytools import factor, discriminant

def run(seed=42, budget_s=100, **kwargs):

            if Q.as_leading_coefficient() < 0:
                Q = -Q

            Q = Q.as_poly().remove_term(2)

            roots = solve(Q, z, domain='Real')
