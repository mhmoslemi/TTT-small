from scipy.optimize import Bounds, minimize
import time
from sympy import symbols, HermitePolynomial, roots, Poly, Rational
from sympy.polys.polyfuncs import hermite

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational coefficients
            exact_c = [Rational(int(round(x*10**10)))/10**10 for x in c]
            
            # Construct polynomial
            z = symbols('z')
            Q = sum(c_k * hermite(4*k, z) for k, c_k in enumerate(exact_c))

            Q = Q - Q.subs(z, 0)
            
            # Check sign at infinity
            sign_inf = Q.as_leading_term(z).is_positive
            if not sign_inf:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_poly().monic()
            
            # Find real roots
            roots_list = roots(Q, z, domain='Real')
            real_roots = [r for r in roots_list if r.is_real]
            
            # Find r_max
            real_roots_sorted = sorted(real_roots)
            r_max = None
            for r in real_roots_sorted:
                if r > 0 and Q.subs(z, r).as_numer_denom()[0].is_zero:
                    continue
                if r > 0:
                    r_max = r
                    break
            
            if r_max is None:
                return float('inf')
            
            score = (r_max**2) / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):

                            for o in range(grid_points):

                                    if time.time() - start_time > budget_s * 0.7:
                                        break
                                    guess = np.array([i/grid_points, j/grid_points, k/grid_points, l/grid_points, m/grid_points, n/grid_points, o/grid_points, p/grid_points])
                                    result = minimize(
                                        objective,
                                        guess,

        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
