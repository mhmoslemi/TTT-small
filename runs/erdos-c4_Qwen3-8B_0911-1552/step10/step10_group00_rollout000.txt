import time
from sympy import symbols, Eq, solve, Poly, Rational, roots, S
from scipy.optimize import minimize

                return float('inf')

            roots_list = roots(Q, z, multiple=True)
            real_roots = [r for r in roots_list if r.is_real]
            real_roots = sorted(real_roots)
            # Find r_max
            r_max = None

            return 1

            return 2*z
        else:
            return 2*z*eval_hermite(n-1, z) - 2*(n-1)*eval_hermite(n-2, z)
