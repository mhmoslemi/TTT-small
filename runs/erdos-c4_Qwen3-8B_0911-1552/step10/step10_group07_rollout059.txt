import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, expand, diff, HermiteH
from sympy.polys.polytools import roots

def hermite_basis(n):
    z = symbols('z')
    return [HermiteH(k, z) for k in range(n+1)]

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    z = symbols('z')
    Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
    Q = Q.subs(z, 0)
    Q = Q - Q.as_coefficients_dict().get(0, 0)
    Q = Q / z**2
    roots = []
    for root in roots(Q, z):
        if root.is_real and root > 0 and Q.subs(z, root).evalf() != 0:
            roots.append(root)
    if not roots:
        return float('inf')
    r_max = max(roots)
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    population_size = 32
    generations = 100
    mutation_rate = 0.3
    best_coeff = np.random.rand(8) * 2 - 1  # Initial diverse population
    
    def fitness(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, len(parent1))
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(c):
        for i in range(len(c)):
            if np.random.rand() < mutation_rate:
                c[i] += np.random.normal(0, 0.1)
                c[i] = np.clip(c[i], -2, 2)
        return c
    
    population = [np.random.rand(8) * 2 - 1 for _ in range(population_size)]
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
        fitnesses = [fitness(ind) for ind in population]
        parents = [population[i] for i in np.argsort(fitnesses)[:population_size//2]]
        next_population = parents[:]
        while len(next_population) < population_size:
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            next_population.append(child)
        population = next_population
    
    best_coeff = min(population, key=fitness)
    return best_coeff
