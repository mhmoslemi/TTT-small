import numpy as np
from scipy.optimize import Bounds, minimize
from deap import base, creator, tools, algorithms
import random
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Define the objective function
    def evaluate_c4_coefficients(c):
        return evaluate_c4_coefficients(c)
    
    # Genetic algorithm setup
    creator.create("FitnessMin", float, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(-2, 2))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=8)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def fitness(ind):
        try:
            score = evaluate_c4_coefficients(ind)
            return (score,)
        except:
            return (float('inf'),)
    
    toolbox.register("evaluate", fitness)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.5, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initial population
    pop = toolbox.population(n=50)
    
    # Run genetic algorithm
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        offspring = tools.selBest(pop, len(pop))
        offspring = algorithms.varAnd(offspring, toolbox, cxpb=0.5, mutpb=0.5)
        fits = map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = tools.selBest(offspring + pop, len(pop))
    
    # Refine best individual with L-BFGS-B
    best_ind = min(pop, key=lambda x: x.fitness.values[0])
    result = minimize(
        lambda c: evaluate_c4_coefficients(c),
        best_ind,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*8,
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    
    return result.x
