    samples = np.random.uniform(-2, 2, size=(20, 3))
    scores = np.array([evaluate_score(sample) for sample in samples])
    
    # Bayesian optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Predictive mean and variance
        y_pred, y_var = gp_model.predict(samples, return_variance=True)
        # Acquisition function: Expected improvement
        mu = y_pred
        sigma = np.sqrt(y_var)
        best = np.min(scores)
        improvement = mu - best
        acq = np.sqrt(sigma) * np.erf(improvement / np.sqrt(2)) * np.exp(-improvement**2 / 2)
        # Select next point
        next_idx = np.argmax(acq)
        next_point = samples[next_idx]
        next_score = evaluate_score(next_point)
        # Update model
        gp_model.fit(samples, scores)
        # Update best
        if next_score < best_score:
            best_score = next_score
            best_coeff = next_point.copy()

        samples = np.vstack((samples, next_point))
        scores = np.append(scores, next_score)
