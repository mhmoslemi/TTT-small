import numpy as np
from scipy.optimize import minimize
from deap import base, creator, algorithms, tools
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    num_coeff = 4
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(-1, 1, num_coeff))
    toolbox.register("individual", tools.initIterate, list, toolbox.attr_float)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", lambda ind: evaluate_c4_coefficients(ind))
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    def eval_func(ind):
        return evaluate_c4_coefficients(ind), 

    population = toolbox.population(n=20)
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)

    # Initial genetic algorithm run
    for _ in range(10):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.5)
        fits = toolbox.map(eval_func, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = tools.selBest(population + offspring, len(population))

    # Refine with L-BFGS on best individuals
    best_ind = min(population, key=lambda x: x.fitness.values[0])
    result = minimize(
        lambda x: evaluate_c4_coefficients(x),
        best_ind,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*num_coeff,
        options={'gtol': 1e-8, 'maxiter': 100}
    )
    return result.x
