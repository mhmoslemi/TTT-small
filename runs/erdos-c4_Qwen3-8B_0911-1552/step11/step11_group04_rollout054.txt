import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, expand, HermitePolynomial, solve, S
from sympy.polys import Poly

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_score = float('inf')
    best_coeff = None
    
    def evaluate_with_symPy(coeffs):
        z = symbols('z')
        Q = sum(c * HermitePolynomial(4*k, z) for k, c in enumerate(coeffs))
        Q = Q.as_expr()
        Q = Q.subs(z, 0)
        Q = Q.as_poly().all_coeffs()[::-1]
        r_roots = solve(Q, z)
        real_roots = [r for r in r_roots if r.is_real]
        real_roots = [r.evalf() for r in real_roots]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2*np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_with_symPy(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    for m in range(3, 7):
        grid_points = 16
        for i in range(grid_points):
            for j in range(grid_points):
                for k in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i/grid_points, 1.0 + j/grid_points, 1.0 + k/grid_points])
                    if m > 3:
                        guess = np.concatenate([guess, np.array([1.0 + i/grid_points])])
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*len(guess),
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='L-BUFFER',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
