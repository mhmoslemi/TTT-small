import numpy as np
from scipy.optimize import Bounds, minimize
import time
from numpy.polynomial.hermite import hermgauss
from mpmath import mp, nroots, poly

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    mp.dps = 50  # High precision for root isolation

    def evaluate_with_mp(c):
        try:
            # Construct polynomial
            poly = np.zeros(len(c) + 1)
            for k in range(len(c)):
                poly[k] = c[k]
            poly[len(c)] = -1  # Ensure Q(0) = 0

            # Compute Hermite polynomials at points
            n = len(c)
            x = hermgauss(n)
            H = np.array([hermgauss(n, k) for k in range(n+1)])
            Q = np.dot(H, c)
            Q = Q / (x ** 2)
            
            # Find real roots
            r = nroots(Q)
            real_roots = [r_i.real for r_i in r if abs(r_i.imag) < 1e-10]
            real_roots = sorted([r for r in real_roots if r > 0])
            
            # Find r_max
            for i in range(len(real_roots)-1, -1, -1):
                if real_roots[i] > 0:
                    r_max = real_roots[i]
                    break
            else:
                r_max = 1.0
            
            score = r_max ** 2 / (2 * np.pi)
            return score
        except:
            return float('inf')

    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_with_mp(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')

    # Initial grid search with 2 coefficients
    for i in range(2):
        for j in range(2):
            if time.time() - start_time > budget_s * 0.7:
                break

            result = minimize(
                objective,
                guess,
                method='Nelder-Mead',
                bounds=[(-2, 2)]*len(guess),
                options={'xatol': 1e-8, 'ftol': 1e-8}
            )

    # Final stage with 3 coefficients
    result = minimize(

        method='L-BFGS-B',

    )
