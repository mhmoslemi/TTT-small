def qmcHalton(d, n):
    sample = np.zeros((n, d))
    for i in range(d):
        base = 2 + i
        for j in range(n):
            sample[j, i] = 0
            k = j
            while k > 0:
                sample[j, i] += (1.0 / base ** (k.bit_length() - 1))
                k = k // base
    return sample
