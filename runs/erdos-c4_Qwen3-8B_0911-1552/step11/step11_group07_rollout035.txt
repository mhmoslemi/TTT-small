import numpy as np
import time
from scipy.optimize import minimize
from symengine import symbols, hermite, solve, simplify, Poly, roots
from sympy import N

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])  # 4 coefficients
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            z = symbols('z')
            H = [hermite(k) for k in range(4*len(c)+1)]
            Q = sum(c[k] * H[4*k] for k in range(len(c)))
            
            # Ensure Q(0) = 0
            Q0 = Q.subs(z, 0)
            if Q0 != 0:
                Q -= Q0
            
            # Flip sign for positivity at infinity
            leading_term = Q.as_poly().get_term_with_highest_degree()
            if leading_term.coeff != 1:
                Q = -Q
            
            # Remove z^2 factor
            Q_poly = Q.as_poly()
            if Q_poly.degree() >= 2:
                Q_poly = Q_poly / z**2
            
            # Find real roots
            roots_list = roots(Q_poly)
            real_roots = [r for r in roots_list if r.is_real]
            
            # Isolate largest positive root
            positive_roots = [r for r in real_roots if r > 0]
            if positive_roots:
                r_max = max(positive_roots)
                U = r_max**2 / (2 * np.pi)
                if U < best_score:
                    best_score = U
                    best_coeff = c.copy()
            return best_score
        except:
            return float('Recommend more exploration')
    
    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = minimize(
                        objective,
                        guess,
                        method='L-BFGS-B',
                        bounds=[(-2, 2)]*len(guess),
                        options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
