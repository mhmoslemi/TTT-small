import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, simplify, solve, S

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            c_sym = [S(c[0]), S(c[1]), S(c[2])]
            z = symbols('z')
            Q = sum(c_sym[k] * HermiteH(4*k, z) for k in range(len(c_sym)))
            
            # Enforce Q(0) = 0 and positive at infinity
            Q = Q - Q.as_coefficients_dict()[z**0]  # Remove constant term
            Q = Q / z**2  # Remove z² factor
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            
            # Find r_max
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            
            if r_max is None:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Grid search on log-scale
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                if time.time() - start_time > budget_s * 0.6:
                    break
                guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points])
                result = minimize(
                    objective,
                    guess,
                    method='L-BFGS-B',
                    bounds=[(-2, 2)]*len(guess),
                    options={'maxiter': 200, 'ftol': 1e-10}
                )
                if 'success' in result and result.success:
                    best_coeff = result.x
    
    # Refine with Nelder-Mead
    result = minimize(
        objective,
        best_coeff,
        method='Nelder-Mead',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'xatol': 1e-8, 'ftol': 1e-8}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
