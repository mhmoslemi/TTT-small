from scipy.optimize import minimize, Bounds
from sympy import symbols, Hermite, expand, simplify, Eq, solve, N
from functools import lru_cache
import time

def run(seed=42, budget_s=100, **kwargs):

    z = symbols('z')

            return (Q.subs(z, z_val) / z**2).evalf()

                r_max = r
