import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, Poly, roots, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    n_coeff = 4
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            c_list = list(c)
            z = symbols('z')
            H = [hermite(4*k, z) for k in range(n_coeff)]
            Q = sum(c_list[k] * H[k] for k in range(n_coeff))
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Make Q positive at infinity
            leading_coeff = 0
            for k in range(n_coeff):
                term = c_list[k] * hermite(4*k, z)
                leading_term = term.expand().as_poly(z).coeff_monomial(z**(4*k))
                leading_coeff += leading_term
            
            if leading_coeff > 0:
                Q = -Q
            
            # Remove z^2 factor
            Q_poly = Q.as_poly(z)
            Q_poly = Q_poly / z**2
            
            # Find real roots
            roots_list = roots(Q_poly, z, domain='Real')
            r_max = -np.inf
            for root, multiplicity in roots_list.items():
                if multiplicity == 1 and root > 0:
                    r_max = max(r_max, root)
            
            if r_max == -np.inf:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 4
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = minimize(
                        objective,
                        guess,
                        method='L-BFGS-B',
                        bounds=[(-2, 2)]*n_coeff,
                        options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-9}
                    )
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*n_coeff,
        options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    
    return best_coeff
