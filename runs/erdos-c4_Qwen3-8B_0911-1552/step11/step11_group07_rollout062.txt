import numpy as np
from scipy.optimize import minimize, Bounds
import time
from scipy.optimize._numdiff import approx_derivative
from sympy import symbols, Hermite, simplify, solve, Eq, diff, Rational, pi

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            z = symbols('z')
            expr = sum(c[k] * Hermite(4*k, z) for k in range(len(c)))
            # Enforce Q(0) = 0
            expr = expr - expr.subs(z, 0)
            # Find leading term for sign at infinity
            leading_term = expr.as_series(z, 100).as_leading_term()
            if leading_term.has(z):
                expr = expr - leading_term
            # Ensure positive at infinity
            if expr.as_series(z, 100).as_coefficients_dict().get(z**100, 0) < 0:
                expr = -expr
            # Remove z^2 factor
            expr = expr / z**2
            # Find real roots
            roots = solve(expr, z)
            real_roots = [r for r in roots if r.is_real]
            r_max = max(real_roots) if real_roots else 1.0
            score = (r_max**2) / (2 * pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with 4 coefficients
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.6:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 
                                     1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*len(guess),
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Final optimization with L-BFGS-B
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
