import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, simplify, solve, rts

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial low-rank guess
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def exact_eval(c):
        try:
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
            Q = Q - Q.as_ordered_terms()[0] * z**2
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            r_max = max(real_roots) if real_roots else 1.0
            return r_max**2 / (2*np.pi)
        except:
            return float('infinite')
    
    def numerical_eval(c):
        try:
            from scipy.optimize import root_scalar
            from scipy.optimize import fsolve
            def Q(z):
                return sum(c[k] * hermite(4*k, z) for k in range(len(c))) - z**2 * hermite(0, z)
            def f(z):
                return Q(z)
            try:
                root = root_scalar(f, bracket=[0.1, 10], method='brentq')
                r_max = root.root
                return r_max**2 / (2*np.pi)
            except:
                return float('infinite')
        except:
            return float('infinite')
    
    def objective(c):
        nonlocal best_score, best_coeff
        score = numerical_eval(c)
        if score < best_score:
            best_score = score
            best_coeff = c.copy()
        return score
    
    # Initial grid search
    grid_points = 16
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*len(guess),
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
