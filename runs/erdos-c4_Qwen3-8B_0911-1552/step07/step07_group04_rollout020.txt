import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, HermiteH, simplify, solve, Poly, roots

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Precompute Hermite polynomials up to H_32
    z = symbols('z')
    hermites = [simplify(HermiteH(k, z)) for k in range(0, 33)]
    
    def evaluate_coefficients(c):
        m = len(c)
        terms = [c[k] * hermites[4*k] for k in range(m)]
        Q = sum(terms)
        Q = Q.subs(z, 0)
        if Q != 0:
            Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        try:
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([float(r) for r in real_roots if r > 0])
            if not real_roots:
                return float('inf')
            r_max = real_roots[-1]
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Grid search with adaptive refinement
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            if time.time() - start_time > budget_s * 0.7:
                break
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=[(-2, 2), (-2, 2)],
                options={'eps': 1e-8, 'maxiter': 100}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='Nelder-Mead',
            bounds=[(-2, 2), (-2, 2)],
            options={'xatol': 1e-8, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
