import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, hermite, diff, Eq, roots, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * hermite(4*k, z) for k, c in enumerate(coeffs))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = roots(Q, z, domain='Real')
        real_roots = [r for r in roots if r.is_real and r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    def objective(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    # Initial guesses based on symmetric and antisymmetric Hermite polynomials
    initial_guesses = []
    for n in range(1, 5):
        ch = np.array([Rational(1, 2**k) for k in range(n)])
        initial_guesses.append(ch)
        initial_guesses.append(-ch)
    initial_guesses.append(np.array([1.0, 0.0, 0.0, 0.0]))
    initial_guesses.append(np.array([0.0, 1.0, 0.0, 0.0]))
    initial_guesses.append(np.array([0.0, 0.0, 1.0, 0.0]))
    initial_guesses.append(np.array([0.0, 0.0, 0.0, 1.0]))
    
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='BFGS',
            bounds=Bounds([-2, 2], len(guess)),
            options={'gtol': 1e-12, 'maxiter': 500}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
            best_score = result.fun
    
    return best_coeff
