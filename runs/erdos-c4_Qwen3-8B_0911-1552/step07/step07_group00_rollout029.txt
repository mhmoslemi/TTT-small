from scipy.optimize import differential_evolution
import time
from sympy import symbols, hermite, Poly, solve, Eq

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def evaluate(c):
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))

    def objective(c):
        try:
            return evaluate(c)

            return float('inf')
    
    bounds = [(-2, 2) for _ in range(8)]

    grid_points = 10

                break

            result = differential_evolution(
