import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, hermite, Eq, solve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Define symbolic variables
    z = symbols('z')
    H = [hermite(k, z) for k in range(4*len(best_coeff)+1)]
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct polynomial
            poly = sum(c[k] * H[4*k] for k in range(len(c)))
            
            # Enforce Q(0) = 0
            poly = poly - poly.subs(z, 0)
            
            # Check sign at infinity
            sign_at_infinity = poly.as_leading_coefficient().as_coefficient()
            if sign_at_infinity < 0:
                poly = -poly
                
            # Remove z^2 factor
            poly = poly.as_poly().remove_redundant()
            
            # Find real roots
            roots = solve(poly, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            
            # Find r_max
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
                if score < best_score:
                    best_score = score
                    best_coeff = c.copy()
            return score if r_max is not None else float('inf')
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            if time.time() - start_time > budget_s * 0.7:
                break
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=[(-2, 2), (-2, 2)],
                options={'eps': 1e-8, 'maxiter': 100}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Use current best as initial guess
        result = minimize(
            objective,
            best_coeff,
            method='Nelder-Mead',
            bounds=[(-2, 2), (-2, 2)],
            options={'xatol': 1e-8, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
