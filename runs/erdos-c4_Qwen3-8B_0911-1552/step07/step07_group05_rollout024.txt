import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, Hermite, expand, Poly, solve, roots
from sympy.polys.polytools import factor_list
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define symbolic variables
    z = symbols('z')
    H = [Hermite(n) for n in range(4 * 8 + 1)]
    
    # Function to evaluate the polynomial and find r_max
    def evaluate_c4_coefficients(coeff):
        m = len(coeff)
        Q = sum(coeff[k] * H[4*k] for k in range(m))
        Q = Q.subs(z, 0)
        Q = Q - Q.as_coeff_add()[0][0][0]
        Q = Q / z**2
        r = solve(Q, z)
        real_roots = [x.evalf() for x in r if x.is_real]
        real_roots = [x for x in real_roots if x > 0]
        real_roots.sort()
        r_max = real_roots[-1] if real_roots else 1
        return r_max**2 / (2 * np.pi)
    
    # Initial grid search with adaptive refinement
    grid_points = 10
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    for i in range(grid_points):
        for j in range(grid_points):
            if time.time() - start_time > budget_s * 0.7:
                break
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            score = evaluate_c4_coefficients(guess)
            if score < best_score:
                best_score = score
                best_coeff = guess.copy()
    
    # Refine using Nelder-Mead
    result = minimize(
        lambda c: evaluate_c4_coefficients(c),
        best_coeff,
        method='Nelder-Mead',
        bounds=[(-2, 2), (-2, 2)],
        options={'xatol': 1e-8, 'ftol': 1e-8}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    # Final refinement with L-BFGS-B
    result = minimize(
        lambda c: evaluate_c4_coefficients(c),
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2), (-2, 2)],
        options={'maxiter': 500, 'ftol': 1e-10}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
