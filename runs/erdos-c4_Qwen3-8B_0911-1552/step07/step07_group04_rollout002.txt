import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, Poly, real_roots, simplify

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')

    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            expr = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
            expr = expr.subs(z, 0)
            if expr != 0:
                expr = expr - expr.subs(z, 0)
            expr = expr / z**2
            expr = expr.as_poly()
            roots = real_roots(expr)
            valid_roots = [r for r in roots if r > 0 and expr.subs(z, r) * expr.subs(z, r + 1e-8) < 0]
            if valid_roots:
                r_max = max(valid_roots)
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')

    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            if time.time() - start_time > budget_s * 0.7:
                break
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=[(-2, 2), (-2, 2)],
                options={'eps': 1e-8, 'maxiter': 100}
            )
            if 'success' in result and result.success:
                best_coeff = result.x

    # Adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='Nelder-Mead',
            bounds=[(-2, 2), (-2, 2)],
            options={'xatol': 1e-8, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x

    return best_coeff
