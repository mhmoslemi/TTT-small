import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, HermiteH, simplify, solve, Function, diff, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational coefficients
            c_rational = [Rational(int(round(1000 * x))) for x in c]
            
            # Define symbols and construct Q(z)
            z = symbols('z')
            Q = sum(c_rational[k] * HermiteH(4*k, z) for k in range(len(c_rational)))
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Find leading coefficient for positivity at infinity
            leading_coeff = Q.as_coefficients_dict().get(z**(4*len(c_rational)), 0)
            if leading_coeff < 0:
                Q = -Q
            
            # Remove z² factor and find sign change
            Q = Q / z**2
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = [r for r in real_roots if not r.is_zero]
            real_roots = sorted([float(r) for r in real_roots])
            
            # Find r_max
            r_max = None
            for i in range(len(real_roots)-1):
                if Q.subs(z, real_roots[i]) * Q.subs(z, real_roots[i+1]) < 0:
                    r_max = real_roots[i+1]
                    break
            if r_max is None:
                r_max = real_roots[-1] if real_roots else 1.0
            
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('ineval')
    
    # Initial grid search with adaptive refinement
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            if time.time() - start_time > budget_s * 0.7:
                break
            result = minimize(
                objective,
                guess,
                method='Nelder-Mead',
                bounds=[(-2, 2), (-2, 2)],
                options={'xatol': 1e-8, 'ftol': 1e-8}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Use current best as initial guess
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2), (-2, 2)],
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
