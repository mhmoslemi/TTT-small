from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, HermiteH, solve, Eq, Abs, simplify

            if score < best_score:

        except:
            return float('inf')

        if len(c) > max_coeff_count:
            return float('inf')
        return objective(c)

    def get_hermite_polynomials(n):
        z = symbols('z')
        hermites = [HermiteH(k, z) for k in range(n+1)]
        return hermites

    def compute_Q(c, z):
        hermites = get_hermite_polynomials(len(c))
        Q = sum(c[k] * hermites[k] for k in range(len(c)))
        return Q

    def find_r_max(Q, z):
        z_val = symbols('z')
        Q_sub = Q.subs(z, z_val)
        roots = solve(Q_sub, z_val)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if r != 0:
                return r
        return float('inf')

    def evaluate_c4_coefficients(c):
        hermites = get_hermite_polynomials(len(c))
        z = symbols('z')
        Q = sum(c[k] * hermites[k] for k in range(len(c)))
        Q = Q.subs(z, z)
        Q = Q.as_ordered_factors()
        Q = Q.as_coefficients_dict()
        Q = Q * z**2
        Q = simplify(Q)
        Q = Q.as_poly()
        r_max = find_r_max(Q, z)
        if r_max == float('inf'):
            return float('inf')
        return r_max**2 / (2 * np.pi)

        return c + np.random.normal(0, scale, size=len(c))

        if len(c) > max_coeff_count:
            return float('inf')

            return evaluate_c4_coefficients(c)
        except:
            return float('inf')

            check_candidate,
            c,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ft0': 1e-10}

                best_score = result.fun
