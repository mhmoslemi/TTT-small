import numpy as np
from scipy.optimize import Bounds, minimize
from sympy import symbols, Hermite, simplify, solve, N
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            H = [Hermite(n) for n in range(4*len(c)+1)]
            Q = sum(c[k] * H[4*k] for k in range(len(c)))
            Q = Q.subs(z, z).as_poly()
            
            # Enforce Q(0) = 0
            Q = Q - Q.as_poly().coeff_monomial(z**0)
            
            # Ensure positivity at infinity
            if Q.as_poly().主导项().coeff < 0:
                Q = -Q
            
            # Remove z^2 factor
            if Q.as_poly().degree() >= 2:
                Q = Q / z**2
            
            # Find r_max
            roots = solve(Q, z)
            r_max = None
            for root in roots:
                if root.is_real and root > 0:
                    if r_max is None or root > r_max:
                        r_max = root
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with more points and refinement
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*len(guess),
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Final refinement using gradient descent
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*len(best_coeff),
        options={'maxiter': 1000, 'ftol': 1e-10}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
