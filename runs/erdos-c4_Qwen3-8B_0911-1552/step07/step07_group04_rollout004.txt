from scipy.optimize import minimize
import time
from sympy import symbols, Eq, solve, horner, HermiteH, simplify
from sympy.polys import Poly
from fractions import Fraction

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.array([Fraction(1), Fraction(1)] * 4)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational numbers
            c = np.array([Fraction(int(x)) for x in c])
            # Construct Q(z) = sum c_k H_{4k}(z)
            z = symbols('z')
            Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
            # Ensure Q(0) = 0
            if Q.subs(z, 0) != 0:
                Q -= Q.subs(z, 0)
            # Ensure positive at infinity
            if Q.as_leading_term(z) < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q.as_poly().remove_term(2)
            # Find roots
            roots = solve(Q, z)
            # Find r_max
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([float(r) for r in real_roots if r > 0])
            if not real_roots:
                return float('inf')
            r_max = real_roots[-1]
            # Compute upper bound
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Grid search on 8 coefficients
    grid_points = 10
    for i in range(grid_points):
