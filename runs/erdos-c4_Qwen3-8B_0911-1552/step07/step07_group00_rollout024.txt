import numpy as np
from scipy.optimize import differential_evolution, root_scalar
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.uniform(-2, 2, 8)
    best_score = float('inf')
    
    # Logarithmic grid search
    grid_points = 10
    grid = np.logspace(-2, 2, grid_points)
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    for m in range(grid_points):
                        for n in range(grid_points):
                            for o in range(grid_points):
                                for p in range(grid_points):
                                    if time.time() - start_time > budget_s * 0.9:
                                        break
                                    guess = np.array([grid[i], grid[j], grid[k], grid[l], grid[m], grid[n], grid[o], grid[p]])
                                    # Fast numerical check
                                    try:
                                        Q = np.polynomial.hermite_e.HermiteE([0]*8)
                                        Q = Q*guess
                                        def f(z):
                                            return np.polyval(Q, z)
                                        # Find r_max numerically
                                        def find_r_max(f):
                                            try:
                                                r = root_scalar(lambda x: f(x)/x**2, bracket=[1, 100], method='brentq')
                                                return r.root**2 / (2 * np.pi)
                                            except:
                                                return float('inf')
                                        current_score = find_r_max(f)
                                        if current_score < best_score:
                                            best_score = current_score
                                            best_coeff = guess
                                    except:
                                        pass
    return best_coeff
