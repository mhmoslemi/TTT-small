from scipy.interpolate import interp1d
from scipy.optimize import minimize
from sympy import symbols, hermite, diff, simplify, Poly
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    def get_hermite_coefficients(m):
        x = symbols('x')
        hermites = [hermite(k, x) for k in range(m)]
        return hermites
    
    def construct_Q(c, m):
        x = symbols('x')
        Q = sum(c[k] * hermite(4*k, x) for k in range(m))
        return Q
    
    def find_r_max(Q):
        x = symbols('x')
        Q_z2 = Q / x**2
        try:
            roots = np.roots([diff(Q_z2, x).as_poly().all_coeffs(), Q_z2.as_poly().all_coeffs()])
            real_roots = np.real(roots[np.isclose(np.imag(roots), 0)])
            real_roots = real_roots[np.isfinite(real_roots)]
            if len(real_roots) == 0:
                return float('inf')
            r_max = real_roots[np.argsort(real_roots)][-1]
            return r_max
        except:
            return float('inf')
    
    def evaluate_c4_coefficients(c):
        m = len(c)
        Q = construct_Q(c, m)
        Q = Q.subs(x, 0)
        Q = Q.as_poly()
        if Q.coeffs()[0] == 0:
            raise ValueError("Q(0) must be zero")
        Q = Q / x**2
        Q = Q.as_poly()
        if Q.coeffs()[0] < 0:
            Q = -Q
        r_max = find_r_max(Q)
        if r_max == float('inf'):
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    def numerical_root_finder(Q, x, max_iter=200):
        def f(z):
            return Q.subs(x, z).as_poly().all_coeffs()[0] + Q.subs(x, z).as_poly().all_coeffs()[1] * z + Q.subs(x, z).as_poly().all_coeffs()[2] * z**2
        try:
            root = root_scalar(f, bracket=[-100, 100], method='brentq', xtol=1e-10)
            return root.root
        except:
            return float('inf')
    
    def numerical_objective(c):
        m = len(c)
        x = symbols('x')
        Q = sum(c[k] * hermite(4*k, x) for k in range(m))
        Q = Q.subs(x, 0)
        if Q.as_poly().coeffs()[0] != 0:
            Q = Q / x**2
            Q = Q.as_poly()
            if Q.as_poly().coeffs()[0] < 0:
                Q = -Q
        try:
            r_max = numerical_root_finder(Q, x)
            if r_max == float('inf'):
                return float('inf')
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def get_hermite_basis_coeffs(m):
        x = symbols('x')
        hermites = [hermite(k, x) for k in range(m)]
        return hermites
    
    bases = [get_hermite_basis_coeffs(m) for m in range(1, 9)]
    
    def get_hermite_basis(m):
        x = symbols('x')
        basis = [hermite(k, x) for k in range(m)]
        return basis
    
    def evaluate(c):
        m = len(c)
        x = symbols('x')
        Q = sum(c[k] * hermite(4*k, x) for k in range(m))
        Q = Q.subs(x, 0)
        if Q.as_poly().coeffs()[0] != 0:
            Q = Q / x**2
            Q = Q.as_poly()
            if Q.as_poly().coeffs()[0] < 0:
                Q = -Q
        try:
            r_max = numerical_root_finder(Q, x)
            if r_max == float('inf'):
                return float('inf')
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def get_initial_guesses():
        guesses = []
        for m in range(1, 6):
            for ratio in [0.5, 1.0, 2.0, 0.707, 0.8, 1.414, 1.2, 0.95]:
                coeffs = np.array([ratio**k / np.sqrt(2**k) for k in range(m)])
                guesses.append(coeffs)
                guesses.append(-coeffs)
        return guesses

    for guess in initial_guesses:

            break
        result = differential_evolution(

            bounds=[(-2, 2) for _ in range(len(guess))],
            strategy='best1bin',
            popsize=15,
            maxiter=40,
            mutation=(0.5, 1),
            recombination=0.95,
            tol=1e-12
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
