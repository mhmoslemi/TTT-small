from scipy.optimize import minimize

    start_time = time.time()

    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:

                best_coeff = c.copy()

            return float('inf')

            method='L-BFGS-B',
