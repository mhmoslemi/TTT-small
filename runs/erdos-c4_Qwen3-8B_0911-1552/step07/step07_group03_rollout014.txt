import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, Eq, solve, HermitePoly, simplify, Poly
from sympy.polys.dense import dmp_to_dict

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')

    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational coefficients
            c_rational = np.array([round(x, 10) for x in c])
            c_rational = c_rational / np.max(np.abs(c_rational))
            
            # Build Hermite polynomial
            z = symbols('z')
            Q = sum(c_rational[k] * HermitePoly(4*k, z) for k in range(len(c_rational)))
            
            # Enforce Q(0) = 0 and positive at infinity
            Q = Q - Q.subs(z, 0)
            Q = Q + (Q.subs(z, 1000) - Q.subs(z, -1000)) * z**2 / 2000
            
            # Remove z^2 factor
            Q = Q / z**2
            
            # Find real roots with exact arithmetic
            roots = solve(Q, z, domain='Real')
            real_roots = [r.evalf() for r in roots if r.is_real]
            
            # Find r_max
            real_roots = [r for r in real_roots if r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            
            # Compute upper bound
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c_rational.copy()
            return score
        except:
            return float('inf')

    # Generate structured initial guesses with rational ratios
    initial_guesses = []
    for i in range(1, 10):

            for k in range(1, 10):

    for guess in initial_guesses:

            break

            guess,
            method='L-BFGS-B',

            options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
