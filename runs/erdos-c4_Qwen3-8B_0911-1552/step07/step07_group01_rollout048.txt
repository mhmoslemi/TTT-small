from scipy.optimize import minimize
from sympy import symbols, hermite, solve, Rational, diff, roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def objective(c):

        try:
            z = symbols('z')

            real_roots = [r[0] for r in real_roots]
            real_roots.sort()
            if real_roots:
                r_max = real_roots[-1]
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
