import numpy as np
from scipy.optimize import differential_evolution
from sympy import symbols, hermite, simplify, diff, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define symbolic variables
    z = symbols('z')
    H = [hermite(i, z) for i in range(5)]  # H_0 to H_4
    
    # Precompute symbolic expressions for H_{4k}(z)
    H4k = [hermite(4 * k, z) for k in range(5)]
    
    # Generate initial guesses based on symmetric properties
    initial_guesses = []
    for c0 in [Rational(1, 2), Rational(1, 1), Rational(1, 4), Rational(2, 1)]:  # 1/2, 1, 1/4, 2
        for c1 in [Rational(1, 2), Rational(1, 1), Rational(1, 4), Rational(2, 1)]:  # Same as above
            c = np.array([float(c0), float(c1)], dtype=np.float64)
            initial_guesses.append(c)
    
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct Q(z)
            Q = sum(c[k] * H4k[k] for k in range(len(c)))
            
            # Enforce Q(0) = 0 by subtracting the constant term
            Q = Q - Q.as_expr().subs(z, 0)
            
            # Find the largest positive real root of Q(z)/z²
            # Remove z² factor
            Q = Q.as_expr().as_poly(z)
            if Q.degree() < 2:
                return float('inf')
            Q = Q / z**2
            
            # Find roots and identify r_max
            roots = np.roots(Q.all_coeffs())
            real_roots = roots[np.isreal(roots)]
            positive_roots = real_roots[real_roots > 0]
            
            # Check for multiplicity
            r_max = None
            for r in positive_roots:
                # Check multiplicity
                der = diff(Q, z).as_expr().subs(z, r)
                if der != 0:
                    r_max = r
                    break
            
            if r_max is None:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            
            return score
        except:
            return float('inf')
    
    # Run optimization with initial guesses
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = differential_evolution(
            objective,
            bounds=[(-2, 2) for _ in range(len(guess))],
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-12
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
