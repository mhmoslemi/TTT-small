import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from sympy import symbols, Eq, solve, Rational, nsolve, sqrt, simplify

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational approximations
            c_rats = [Rational(int(round(1000000 * x)) / 1000000) for x in c]
            
            # Define symbolic variables
            z = symbols('z')
            H = lambda n: np.polysyn(n, z, 4, 0, exact=True)
            
            # Compute Hermite polynomials
            Q = sum(c_rats[k] * H(4*k) for k in range(len(c_rats)))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Flip sign if needed to make positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.expand()
            if Q.as_poly().degree >= 2:
                Q = Q / z**2
            
            # Find largest positive real root with even multiplicity
            roots = solve(Q, z, domain='Real')
            r_max = None
            
            for root in roots:
                if root.is_real and root > 0:
                    multiplicity = Q.as_poly().leading_term().exponents.count(root)
                    if multiplicity % 2 == 1:
                        r_max = max(r_max, root) if r_max is not None else root
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
                if score < best_score:
                    best_score = score
                    best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 10
    grid_values = np.linspace(-2, 2, grid_points)
    for c0 in grid_values:
        for c1 in grid_values:
            if np.abs(c0) > 1e-6 or np.abs(c1) > 1e-6:
                c = np.array([c0, c1])
                score = objective(c)
                if score < best_score:
                    best_score = score
                    best_coeff = c.copy()
    
    # Run differential evolution
    if time.time() - start_time < budget_s * 0.7:
        result = differential_evolution(
            objective,
            bounds=[(-2, 2), (-2, 2)],
            strategy='best1bin',
            popsize=20,
            maxiter=100,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True
        )
        best_coeff = result.x
    
    # Refine with L-BFGS
    result = minimize(
        objective,
        x0=best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2), (-2, 2)],
        options={'maxiter': 100}
    )
    best_coeff = result.x
    
    return best_coeff
