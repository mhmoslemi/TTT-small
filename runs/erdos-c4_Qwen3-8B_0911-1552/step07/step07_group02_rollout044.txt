import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, solve, roots, Rational, Eq, diff
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([Rational(1, 4), Rational(1, 4), Rational(1, 4), Rational(1, 4)])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct Q(z) with Hermite polynomials
            z = symbols('z')
            H = [hermite(i, z) for i in range(4*len(c)+1)]
            Q = sum(c[i]*H[4*i + 1] for i in range(len(c)))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positivity at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_ordered_factors()[0]
            
            # Find real roots
            roots_list = roots(Q, z, domain='Real')
            real_roots = [r for r in roots_list.keys() if r.is_real]
            
            # Find r_max
            r_max = 0
            for r in real_roots:
                if r > r_max:
                    r_max = r
            
            if r_max == 0:
                return float('inf')
            
            # Compute U(c)
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initialize with structured grid
    initial_guesses = []
    for a in [Rational(1, 2), Rational(1, 4), Rational(1, 8), Rational(1, 16)]: 
        for b in [Rational(1, 2), Rational(1, 4), Rational(1, 8), Rational(1, 16)]:

                for d in [Rational(1, 2), Rational(1, 4), Rational(1, 8), Rational(1, 16)]: 
                    initial_guesses.append([a, b, c, d])
                    initial_guesses.append([-a, b, c, d])
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break

            lambda x: objective([Rational(x[0]), Rational(x[1]), Rational(x[2]), Rational(x[3])]),

            options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = [float(g) for g in guess]
    
    return best_coeff
