import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Poly, solve, Hermite, simplify
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            H = [Hermite(n)(z) for n in range(4*len(c)+1)]
            Q = sum(c[k] * H[4*k] for k in range(len(c)))
            
            # Enforce Q(0) = 0 and positive at infinity
            if Q.subs(z, 0) != 0:
                Q = Q - Q.subs(z, 0)
            Q = Q.as_leading_term(z).as_coeff_monomial()[1] * Q
            
            # Find r_max
            Q_z2 = Q / z**2
            roots = solve(Q_z2, z)
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            r_max = max(real_roots) if real_roots else 1.0
            
            # Compute upper bound
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            if time.time() - start_time > budget_s * 0.7:
                break
            guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points])
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=[(-2, 2), (-2, 2)],
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='COBYLA',
        bounds=[(-2, 2), (-2, 2)],
        options={'maxiter': 100, 'ftol': 1e-12}
    )
    if 'success' in result and result.success:
        best_coeff = result.x
    
    return best_coeff
