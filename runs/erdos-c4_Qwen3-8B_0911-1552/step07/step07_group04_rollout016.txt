from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, Eq, solve, hermite, simplify, Poly, roots
from fractions import Fraction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    coeff_count = 4
    
    def evaluate_c4_coefficients(c):
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))

        if len(real_roots) < 1:
            return float('inf')
        r_max = real_roots[-1]
        score = r_max**2 / (2*np.pi)
        return score
    
    def objective(c):
        try:
            score = evaluate_c4_coefficients(c)
            return score
        except:
            return float('inf')
    
    def grid_search():
        nonlocal best_coeff, best_score
        grid_points = 10
        for i in range(grid_points):
            for j in range(grid_points):
                for k in range(grid_points):
                    for l in range(grid_points):
                        guess = np.array([1.0 + i/grid_points, 1.0 + j/grid_points, 1.0 + k/grid_points, 1.0 + l/grid_points])

                            return

                            objective,
                            guess,

            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
