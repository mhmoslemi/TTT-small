import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, diff, hermite, roots, simplify

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.uniform(-2, 2, size=8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            z = symbols('z')

            Q = Q.subs(z, 0)

            if Q != 0:
                Q = Q - Q.subs(z, 0)

                Q = -Q
            # Remove z^2 factor
            Q = Q.expand()
            if Q.find(z, 2) != -1:
                Q = Q / z**2
            # Find roots
            roots_expr = roots(Q, z, real=True)
            r_max = None
            for root in roots_expr:
                if root.is_real and root > 0:
                    r_max = max(r_max, root) if r_max is not None else root
            # Check for even multiplicity
            if r_max is not None:
                r_max = r_max ** 2 / (2 * np.pi)
            else:
                r_max = float('inf')
            score = r_max
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with adaptive refinement
    grid_points = 10
    for i in range(grid_points):

                objective,
                guess,
                method='Nelder-Mead',

        best_coeff,

    )
