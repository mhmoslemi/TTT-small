from scipy.optimize import differential_evolution, minimize
import time
from sympy import symbols, Hermite, simplify, solve, diff, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            H = [Hermite(k, z) for k in range(8)]
            Q = sum(c[k]*H[k] for k in range(8))
            
            # Ensure Q(0) = 0
            Q0 = Q.subs(z, 0)
            if Q0 != 0:
                Q = Q - Q0

            leading_term = Q.as_leading_term(z)
            sign_inf = 1 if leading_term.as_poly().all_coeffs()[-1] > 0 else -1
            Q = sign_inf * Q
            
            # Remove z^2 factor
            Q = Q.subs(z, z/np.sqrt(2))
            Q = Q.as_ordered_terms()
            Q = Q[1] + Q[0] if len(Q) > 1 else Q[0]
            
            # Find r_max
            roots = solve(Q, z)
            r_max = 0
            for root in roots:
                if root.is_real:
                    val = float(root)
                    if val > 0 and val > r_max:
                        r_max = val
            if r_max == 0:
                return float('inf')
            
            # Compute U(c)
            score = (r_max**2) / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()

        except:
            return float('inf')
    
    # Initial grid search
