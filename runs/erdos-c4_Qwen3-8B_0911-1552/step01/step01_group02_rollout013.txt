import numpy as np
import sympy as sp
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define Hermite polynomials
    z = sp.symbols('z')
    H = [sp.hermite(n) for n in range(4*8 + 1)]
    
    def evaluate(coeffs):
        Q = sum(c * H[4*k] for k, c in enumerate(coeffs))
        Q = Q.as_poly()
        Q = Q / z**2
        roots = sorted([r.evalf() for r in Q.all_roots()])
        sign_changes = 0
        for i in range(len(roots)):
            if i > 0 and roots[i-1] < 0 and roots[i] > 0:
                sign_changes += 1
        if sign_changes == 0:
            return float('inf')
        r_max = roots[-1]
        return r_max**2 / (2 * np.pi)
    
    # Initial grid search on log-scale
    log_coeffs = np.log(np.random.rand(8) + 1e-6)
    best_val = float('inf')
    best_coeffs = np.zeros(8)
    
    for _ in range(20):
        coeffs = np.exp(log_coeffs)
        val = evaluate(coeffs)
        if val < best_val:
            best_val = val
            best_coeffs = coeffs
    
    # Refine with conjugate gradient
    def objective(x):
        return evaluate(x)
    
    res = minimize(objective, best_coeffs, method='L-BFGS-B', options={'ftol': 1e-8})
    return res.x.tolist()
