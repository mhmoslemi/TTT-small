import sympy as sp

        H[n] = sp.hermite(n)

            return np.inf

            if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:

        return np.inf
    
    def objective(coeffs):

        r_max = find_r_max(Q)
        if r_max == np.inf:
            return np.inf
        return r_max**2 / (2 * np.pi)
    
    def genetic_search():

        generations = 100
