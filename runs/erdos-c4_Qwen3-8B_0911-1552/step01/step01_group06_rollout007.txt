import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Precompute orthogonalized Hermite polynomials up to H_31 (4*8-1)
    def hermite_orthogonalized(n):
        from scipy.special import hermite
        return hermite(n, poly=True)
    
    # Precompute orthogonalized basis
    basis = []
    for k in range(8):
        basis.append(hermite_orthogonalized(4*k))
    
    # Gram-Schmidt orthogonalization
    gram = np.zeros((8,8))
    for i in range(8):
        for j in range(i+1):
            gram[i,j] = np.vdot(basis[i], basis[j])
    
    # Inverse-gram matrix for optimization
    inv_gram = np.linalg.pinv(gram)
    
    def objective(x):
        # Convert to orthogonalized coefficients
        x_orth = np.dot(inv_gram, x)
        # Reshape to get coefficients for H_{4k}
        coeffs = x_orth
        # Evaluate Q(z) = sum c_k H_{4k}(z)
        def Q(z):
            val = 0
            for k in range(8):
                val += coeffs[k] * hermite_orthogonalized(4*k)(z)
            return val
        
        # Enforce Q(0) = 0 by adjusting last coefficient
        coeffs[-1] -= Q(0)
        
        # Find r_max as the largest positive real root of Q(z)/z² changing sign
        from scipy.optimize import root_scalar
        def f(z):
            return Q(z) / (z**2 + 1e-10)
        
        try:
            r = root_scalar(lambda z: f(z), bracket=(0, 100), method='brentq').root
            if r < 0:
                r = None
        except:
            r = None
        
        if r is not None:
            return r**2 / (2*np.pi)
        return np.inf
    
    def optimize():
        # Initial points on a grid
        points = []
        for i in range(10):
            x = np.random.randn(8)

        points.append(np.random.randn(8))
        
        best_x = None
        best_val = np.inf
        
        for x in points:

                best_x = result.x
                best_val = result.fun
