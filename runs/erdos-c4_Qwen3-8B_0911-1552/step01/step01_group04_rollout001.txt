import time
from scipy.optimize import differential_evolution

        return Q

        sign_changes = np.sign(np.real(Q[1:] - Q[:-1]))

                r_max = z[r]
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff

            if score < best_score:
