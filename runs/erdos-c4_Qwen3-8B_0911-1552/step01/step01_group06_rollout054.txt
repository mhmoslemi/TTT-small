import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
import time
from sympy import symbols, Eq, solve, hermite, simplify
from numpy.polynomial.hermite_e import hermweight

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def get_hermite_weights(n):
        z = symbols('z')
        H = hermite(n)(z)
        return simplify(hermweight(z) * H)
    
    def construct_Q(coeffs):
        z = symbols('z')
        Q = sum(c * hermite(4*k)(z) for k, c in enumerate(coeffs))
        Q = Q.subs(z, z)
        return Q
    
    def get_r_max(Q):
        from sympy import roots, N, sign
        
        z = symbols('z')
        expr = Q
        roots_expr = roots(expr, z)
        real_roots = [r for r in roots_expr if r.as_real_imag()[1] == 0]
        real_roots = [N(r) for r in real_roots]
        
        def sign_change(r):
            if r < 0:
                return 0
            try:
                val = expr.subs(z, r)
                return np.sign(val)
            except:
                return 0
        
        sorted_roots = sorted(real_roots)
        sign_prev = None
        r_max = None
        for r in sorted_roots:
            if r > 0:
                s = sign_change(r)
                if sign_prev is not None and sign_prev * s < 0:
                    r_max = r
                sign_prev = s
        if r_max is not None:
            return r_max
        return None
    
    def evaluate_c4_coefficients(coeffs):
        Q = construct_Q(coeffs)
        Q = Q.as_numer_denom()[0]
        Q = Q / z**2
        r_max = get_r_max(Q)
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    def surrogate(x):
        x = np.copy(x)
        x /= np.sum(np.abs(x))
        return evaluate_c4_coefficients(x)
    
    def grid_search():
        best = np.inf
        best_x = np.zeros(8)
        for i in range(8):
            for j in range(8):
                coeffs = np.zeros(8)
                coeffs[i] = 1
                coeffs[j] = 1
                val = surrogate(coeffs)
                if val < best:
                    best = val
                    best_x = coeffs.copy()
        return best_x
    
    def optimize():
        x0 = np.random.randn(8)
        result = minimize(surrogate, x0, method='L-BFGS-B', options={'gtol': 1e-8, 'ftol': 1e-8, 'maxiter': 100})
        return result.x
    
    best_x = grid_search()
    best_val = surrogate(best_x)
    
    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        x = np.random.randn(8)
        x /= np.sum(np.abs(x))
        x = optimize()
        val = surrogate(x)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x
