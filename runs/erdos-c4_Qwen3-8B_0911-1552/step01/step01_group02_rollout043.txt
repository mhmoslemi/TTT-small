from scipy.optimize import Bounds, minimize
from sympy import symbols, HermitePolynomial, diff, simplify

def run(seed=42, budget_s=100, **kwargs):

    def hermite_coeffs(n):
        z = symbols('z')
        return [diff(HermitePolynomial(n, z), z, k) / np.sqrt(2**k) for k in range(n+1)]
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(c * HermitePolynomial(4*k, z) for k, c in enumerate(coeffs))

        roots = [r for r in Q.as_numer_denom()[0].as_poly().all_roots() if r.real > 0 and r.imag == 0]
        r_max = max(roots) if roots else 1
        return r_max**2 / (2 * np.pi)

    initial = np.array([1.0 / (2**k) for k in range(8)])

    def objective(x):
        return evaluate_c4_coefficients(x)
    
    res = minimize(objective, initial, method='L-BFGS-B', bounds=Bounds(0, 1))
    best = res.x

    for _ in range(20):
