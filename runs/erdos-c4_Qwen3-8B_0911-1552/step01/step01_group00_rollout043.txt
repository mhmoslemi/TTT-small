import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def objective(x):
        c = x
        # Compute Q(z) = sum c_k H_{4k}(z)
        # We'll use scaled Hermite polynomials for numerical stability
        # Define Hermite polynomials H_0(z), H_4(z), H_8(z), H_12(z)
        def H0(z): return 1
        def H4(z): return 8*z**4 - 8*z**2 + 1
        def H8(z): return 16*z**8 - 48*z**6 + 32*z**4 - 12*z**2 + 1
        def H12(z): return 32*z**12 - 144*z**10 + 288*z**8 - 240*z**6 + 60*z**4 - 4*z**2 + 1
        
        # Evaluate Q(z)
        def Q(z):
            return c[0]*H0(z) + c[1]*H4(z) + c[2]*H8(z) + c[3]*H12(z)
        
        # Enforce Q(0) = 0 and ensure positivity at infinity
        # We'll scale c[3] to ensure positivity at infinity
        
        # Find r_max: largest positive real root where Q(z)/z^2 changes sign
        # We'll use a numerical root finding approach
        
        # Try to find r_max numerically
        try:
            def f(z):
                return Q(z)/(z**2)
            # Use root_scalar to find positive root
            result = root_scalar(f, bracket=[0, 10], method='brentq')
            r_max = result.root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Use differential evolution for global search
    initial_guess = np.random.rand(4)  # Use 4 coefficients (H0, H4, H8, H12)
    result = minimize(objective, initial_guess, method='L-BFGS-B', options={'ftol': 1e-8, 'gtol': 1e-8})
    
    # Output the best found coefficients
    return result.x
