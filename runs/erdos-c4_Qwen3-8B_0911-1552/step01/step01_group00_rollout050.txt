from scipy.optimize import Bounds, minimize
from sympy import symbols, HermiteH, simplify, solve, Eq, diff
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    initial_coefficients = np.random.rand(C4_MAX_COEFF_COUNT)
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        H = [HermiteH(4*k, z) for k in range(len(coeffs))]
        Q = sum(coeffs[k] * H[k] for k in range(len(coeffs)))
        Q = Q - Q.as_coefficients_dict()[H[0]] * H[0]  # Enforce Q(0)=0
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [r.evalf() for r in real_roots]
        real_roots = [r for r in real_roots if r > 0]
        real_roots = sorted(real_roots)
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return evaluate_c4_coefficients(x)
    
    result = minimize(objective, initial_coefficients, method='L-BFGS-B', bounds=Bounds(-10, 10))
    return result.x
