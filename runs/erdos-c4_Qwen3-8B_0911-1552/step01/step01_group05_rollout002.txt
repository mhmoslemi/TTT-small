import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def evaluate_c4_coefficients(c):
        if len(c) > 8:
            return float('inf')
        return evaluate_c4_coefficients(c)
    
    def objective(c):
        return evaluate_c4_coefficients(c)
    
    # Grid search with structured points
    grid_points = np.linspace(0.01, 1.0, 20)
    best = np.array([0.5]*8)
    best_val = float('inf')
    
    for i in range(8):
        for j in range(8):
            c = np.array([grid_points[i] if k == i else grid_points[j] if k == j else 0.5 for k in range(8)])
            val = objective(c)
            if val < best_val:
                best = c
                best_val = val
    
    # Gradient descent refinement
    def objective_with_grad(c):
        return evaluate_c4_coefficients(c), np.zeros_like(c)
    
    res = minimize(objective_with_grad, best, method='L-BFGS-B', bounds=[(0, 100) for _ in range(8)], options={'gtol': 1e-8, 'ftol': 1e-8})
    if res.success:
        best = res.x
    
    return best
