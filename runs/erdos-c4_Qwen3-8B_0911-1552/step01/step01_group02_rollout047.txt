import numpy as np
from scipy.optimize import minimize
from sympy import symbols, simplify, hermite, roots, Poly, Rational, S
from itertools import product
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    # Generate symbolic variables
    z = symbols('z')

    H_dict = {i: hermite(i, z) for i in range(4*8 + 1)}

        Q = sum(c * H_dict[4*i] for i, c in enumerate(coeffs))

        # Find roots of Q(z)/z^2
        Q_poly = Poly(Q, z)
        roots = Q_poly.all_roots()
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        
        # Find r_max
        if not real_roots:
            return float('inf')
        
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    # Grid search with adaptive perturbation
    def grid_search():
        base = np.logspace(-2, 2, 20)
        best = np.full(8, 1.0)
        best_val = evaluate(best)
