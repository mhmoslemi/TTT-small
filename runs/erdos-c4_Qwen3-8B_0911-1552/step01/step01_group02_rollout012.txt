    gp.fit(initial_points, values)

        return evaluate(x)
    
    def surrogate(x):
        return gp.predict(x.reshape(1, -1))[0]
    
    points = initial_points
    values = values
    
    for _ in range(20):
        # Predictive mean and variance
        mean = surrogate(points)
        var = np.sqrt(gp.kernel_.diag())
        
        # Acquisition function: Expected improvement
        best = np.min(mean)
        improvement = mean - best
        acq = improvement * np.exp(-0.5 * (improvement / var)**2)
        
        next_point = points[np.argmax(acq)]
        next_point += np.random.normal(0, 0.1, 8)

        new_val = evaluate(next_point)
        points = np.vstack([points, next_point])
        values = np.append(values, new_val)
        gp.fit(points, values)
    
    # Final refinement with local search
    best_idx = np.argmin(values)
    best_coeffs = points[best_idx]
