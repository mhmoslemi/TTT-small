import numpy as np
from scipy.optimize import minimize, fsolve
import time
from sympy import symbols, hermite, Eq, solve, N

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    z = symbols('z')
    H = [hermite(4*k, z) for k in range(8)]
    
    def exact_eval(c):
        Q = sum(c[k] * H[k] for k in range(8))
        Q = Q.subs(z, 0)
        Q = Q.as_poly().all_coeffs()[::-1]
        return Q
    
    def get_r_max(c):
        Q = sum(c[k] * H[k] for k in range(8))
        Q = Q.as_poly()
        roots = solve(Q, z)
        real_roots = [float(r.evalf()) for r in roots if r.is_real and r > 0]
        real_roots.sort()
        if not real_roots:
            return 1.0
        return real_roots[-1]
    
    def objective(c):
        r_max = get_r_max(c)
        return r_max**2 / (2 * np.pi)
    
    def perturb(c, alpha=0.1):
        return c + np.random.randn(8) * alpha
    
    def genetic_search(pop_size=32, generations=10):
        pop = np.random.rand(pop_size, 8)
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in pop])
            elites = pop[np.argsort(fitness)[:5]]
            next_pop = np.vstack([elites] + [perturb(np.random.choice(elites)) for _ in range(pop_size - 5)])
            pop = next_pop
        return pop[np.argmin(fitness)]
    
    best = genetic_search()
    for _ in range(10):
        improved = False
        for i in range(8):
            for delta in [-1e-3, 1e-3]:
                candidate = best.copy()
                candidate[i] += delta
                if objective(candidate) < objective(best):
                    best = candidate
                    improved = True
        if not improved:
            break
    
    return best
