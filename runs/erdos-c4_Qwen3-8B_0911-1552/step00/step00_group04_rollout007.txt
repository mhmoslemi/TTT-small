import time
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):

            return Q

            return np.sign(integrand(z))

        except:
            return float('inf')
    
    def surrogate_func(c):
        return eval_upper_bound(c)
    
    initial_guess = np.random.uniform(-1, 1, 8)
    
    start_time = time.time()
