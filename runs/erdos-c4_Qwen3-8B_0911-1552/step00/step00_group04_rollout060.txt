import numpy as np
import time
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, simplify, Rational, pi
from fractions import Fraction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    coefficients = np.random.rand(8)
    best_coeff = coefficients.copy()
    best_score = np.inf
    
    def numerical_surrogate(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_poly().all_coeffs()[0] * z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for i in range(len(real_roots) - 1):
            if real_roots[i+1] - real_roots[i] < 1e-8:
                continue
            r_max = real_roots[i+1]
            return r_max**2 / (2 * np.pi)
        return np.inf

    def evaluate(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_poly().all_coeffs()[0] * z**2
        Q = Q.as_poly()
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for i in range(len(real_roots) - 1):
            if real_roots[i+1] - real_roots[i] < 1e-8:
                continue
            r_max = real_roots[i+1]
            return r_max**2 / (2 * pi)
        return np.inf

    def objective(coeffs):
        return numerical_surrogate(coeffs)

    def objective_exact(coeffs):
        return evaluate(coeffs)

    def search():
        nonlocal best_coeff, best_score
        for _ in range(100):
            result = minimize(objective, x0=best_coeff, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
            if result.success and result.fun < best_score:
                best_score = result.fun
                best_coeff = result.x
            if time.time() - start_time > budget_s:
                break

    start_time = time.time()
    search()
    
    final_result = best_coeff
    return tuple(final_result)
