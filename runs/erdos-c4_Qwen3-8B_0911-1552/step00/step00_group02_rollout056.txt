from scipy.optimize import minimize
from scipy.optimize._differentiable_functions import _check_unknowns

def run(seed=42, budget_s=100, **kwargs):
    # Define Hermite polynomials
    def H(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        else:
            return 2*z*H(n-1, z) - 2*(n-1)*H(n-2, z)
    
    # Evaluate Q(z)
    def Q(z, c):
        m = len(c)
        total = 0
        for k in range(m):
            total += c[k] * H(4*k, z)
        return total

    def find_r_max(c):
        m = len(c)
        eps = 1e-8
        for _ in range(100):
            try:
                z = 1.0
                while True:
                    q = Q(z, c)
                    if np.abs(q) < eps:
                        break
                    if q > 0:
                        z += 1
                    else:
                        z -= 1
                # Newton step
                q = Q(z, c)
                q_prime = np.gradient(q, z, axis=0)
                z_new = z - q / q_prime
                if np.abs(z_new - z) < 1e-6:
                    break
                z = z_new
            except:
                break
        # Check roots
        roots = []
        z = 0
        while z < 100:
            q = Q(z, c)
            if np.abs(q) < 1e-8:
                roots.append(z)
            z += 0.1
        # Find largest positive root
        for r in roots:
            if r > 0:

        return 100
    
    # Objective function
    def obj(c):

        q0 = Q(0, c)

        c = np.append(c, c_m)

        if Q(100, c) < 0:

        r_max = find_r_max(c_scaled)
        return r_max**2 / (2 * np.pi)

    result = minimize(obj, initial_guess, method='L-BFGS-B', options={'maxiter': 100})
