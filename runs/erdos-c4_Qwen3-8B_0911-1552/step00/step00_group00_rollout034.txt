Use gradient descent on coefficient ratios, with numerical root finding to estimate r_max. Refine via exact polynomial checks, prioritizing lower U(c). Limit to 8 coefficients and use symmetry to reduce search space.

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.integrate import quad
from sympy import symbols, HermiteH, simplify, solve, roots, S
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    coeffs = np.random.rand(8) - 0.5
    best = coeffs.copy()
    best_val = float('inf')
    time_left = budget_s
    
    def evaluate_c4(coeffs):
        nonlocal time_left
        start_time = time()
        z = symbols('z')
        poly = sum(coeffs[k] * HermiteH(4*k, z) for k in range(len(coeffs)))
        poly = poly.subs(z, 0)
        c_m = -poly / (HermiteH(4*len(coeffs), 0))
        poly = sum(coeffs[k] * HermiteH(4*k, z) for k in range(len(coeffs))) + c_m * HermiteH(4*len(coeffs), z)
        poly = poly - poly.subs(z, 0)
        poly = poly / z**2
        try:
            roots = solve(poly, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            r_max = max(real_roots)
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def objective(x):
        nonlocal time_left
        if time_left <= 0:
            return best_val
        time_left -= 0.1
        x = np.clip(x, -10, 10)
        val = evaluate_c4(x)
        return val
    
    for _ in range(100):
        if time_left <= 0:
            break
        res = minimize(objective, best, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        if res.success:
            best = res.x
            best_val = res.fun
        else:
            break
    
    return best
