import random
from scipy.optimize import root_scalar
from scipy.special import hermite
from itertools import combinations

    def evaluate_coeffs(coeffs):

            poly[4*k] += coeffs[k]

            return sum(coeffs[k] * hermite(4*k)(z) for k in range(m))

        Q0 = Q(0)
        if Q0 != 0:
            coeffs = [c if k != m-1 else c * -Q0/Q0 for k, c in enumerate(coeffs)]
        
        # Remove z² factor and find r_max
        r_max = None
        try:
            def f(z):
                return Q(z) / (z**2)
            
            # Binary search for sign change
            def sign_change(l, r):
                fl = f(l)
                fr = f(r)
                return fl * fr < 0
            
            l, r = 0, 1000
            while r - l > 1e-6:
                mid = (l + r) / 2
                if sign_change(l, mid):
                    r = mid
                else:
                    l = mid
            r_max = l
            
            # Find largest positive root
            def f_zero(z):
                return Q(z) / (z**2)
            
            res = root_scalar(f_zero, bracket=[0, 1000], method='brentq')
            r_max = res.root
        except:
            r_max = 1000
        
        return r_max ** 2 / (2 * np.pi)
    
    def fitness(coeffs):

            return evaluate_coeffs(coeffs)

            return float('inf')

        new_coeffs = individual[:]
        for i in range(len(new_coeffs)):
            if random.random() < mutation_rate:
                new_coeffs[i] += np.random.normal(0, 0.1)
        return new_coeffs
    
    def crossover(ind1, ind2):
        m = len(ind1)
        return [ind1[i] if i % 2 == 0 else ind2[i] for i in range(m)]

        return [np.random.normal(0, 1, 8) for _ in range(size)]
