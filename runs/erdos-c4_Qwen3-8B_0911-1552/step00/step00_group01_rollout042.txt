import numpy as np
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, solve, Rational, N
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_bound = float('inf')
    
    def evaluate(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        # Ensure Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positive at infinity
        if Q.as_infinite().is_negative:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find r_max
        try:
            # Use numerical root finding
            def f(x):
                return N(Q.subs(z, x))
            result = root_scalar(f, bracket=(0, 100), method='brentq')
            r_max = result.root
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    # Grid search with refinement
    for _ in range(100):
        c = np.random.rand(8)
        current = evaluate(c)
        if current < best_bound:
            best_bound = current
            best_coeffs = c
        if _ % 10 == 0:
            print(f'Iteration {_}: Best bound = {best_bound}')
    
    return best_coeffs
