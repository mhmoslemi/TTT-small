Use gradient descent with a fast numerical surrogate to find good coefficient candidates, then refine using exact rational arithmetic via SymPy to verify and compute the upper bound.

import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Rational, N, sin, cos, pi
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    
    def surrogate(c):
        c = c.copy()
        c[-1] = 0
        t = np.linspace(-10, 10, 10000)
        Q = np.zeros_like(t)
        for k in range(8):
            H = HermiteH(4*k, t)
            Q += c[k] * H
        Q /= t**2
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = t[sign_changes[-1]] if sign_changes.size else 10
        return r_max**2 / (2 * np.pi)
    
    def exact_bound(c):
        c = np.round(c / 1e10) * 1e10
        t = symbols('t')
        Q = sum(c[k] * HermiteH(4*k, t) for k in range(8))
        Q = Q - Q.subs(t, 0)
        Q = Q / t**2
        roots = solve(Q, t)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if np.isclose(Q.subs(t, r), 0):
                continue
            sign = Q.subs(t, r + 1e-6).evalf()
            if sign > 0:
                return r**2 / (2 * pi)
        return 1000
    
    def optimize():
        res = minimize(lambda c: -surrogate(c), x0=c, bounds=[(-10, 10) for _ in range(8)], method='L-BFGS-B', options={'maxiter': 100})
        best_c = res.x
        best_bound = exact_bound(best_c)
        return best_c
    
    start = time()
    result = optimize()
    if time() - start < budget_s:
        return result
    else:
        return result
