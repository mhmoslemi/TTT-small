from scipy.optimize import minimize
import sympy as sp

def run(seed=42, budget_s=100, **kwargs):

    best_value = float('inf')
    
    def objective(coeffs):
        nonlocal best_value, best_coeffs
        c = coeffs
        z = sp.symbols('z')
        Q = sum(c[k] * sp.hermite(h=4*k, arg=z) for k in range(len(c)))

        roots = sp.roots(Q)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if real_roots else 1.0
        value = r_max**2 / (2*np.pi)
        if value < best_value:
            best_value = value
            best_coeffs = c.copy()
        return value
    
    for _ in range(10):
        initial = np.random.rand(8)
        res = minimize(objective, initial, method='L-BFGS-B', bounds=[(None, None) for _ in range(8)])
        if res.fun < best_value:
            best_coeffs = res.x
    
    return tuple(best_coeffs)
