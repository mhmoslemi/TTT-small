from scipy.optimize import minimize
from scipy.optimize import minimize_scalar
import sympy as sp

def run(seed=42, budget_s=100, **kwargs):

    # Define symbolic variables
    z = sp.symbols('z')
    H = [sp.hermite(z, n) for n in range(8)]
    H = [sp.expand(H[i]) for i in range(8)]

        Q = sum(c[k] * H[k] for k in range(8))

        Q = Q - Q.subs(z, 0)

            Q = -Q

        Q = Q / z**2

            roots = sp.roots(Q)
            r_max = None

            return float('inf')
