import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, Poly, solve, N
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def evaluate(coeffs):
        z = symbols('z')
        Q = 0
        for k, c in enumerate(coeffs):
            Q += c * HermiteH(4*k, z)
        Q = Q.as_expr()
        # Enforce Q(0) = 0
        Q = Q - Q.as_poly().subs(z, 0)
        # Ensure positive at infinity
        if Q.as_poly().coeffs()[-1] < 0:
            Q = -Q
        # Remove z² factor
        Q = Q / z**2
        # Find real roots
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([N(r) for r in real_roots if r > 0])
        # Find largest positive real root with odd multiplicity
        r_max = None
        for r in real_roots:
            poly = Poly(Q.subs(z, r), z)
            if poly.degree() == 0:
                continue
            if poly.all_coeffs()[-1] == 0:
                continue
            if poly.all_coeffs()[-2] == 0:
                continue
            if poly.all_coeffs()[-1] * poly.all_coeffs()[-2] < 0:
                r_max = r
                break
        if r_max is not None:
            return r_max**2 / (2 * np.pi)
        return float('inf')
    
    def objective(coeffs):
        return evaluate(coeffs)
    
    # Initial guess
    init_coeffs = np.random.rand(8)
    result = minimize(objective, init_coeffs, method='L-BFGS-B', options={'ftol': 1e-6})
    best_coeffs = result.x
    
    # Refine with exact checks
    start_time = time.time()
    def refine(coeffs):
        best = float('inf')
        for i in range(len(coeffs)):
            for j in range(i+1, len(coeffs)):
                coeffs_new = coeffs.copy()
                coeffs_new[i] = coeffs[i] + 1e-3
                coeffs_new[j] = coeffs[j] - 1e-3
                val = evaluate(coeffs_new)
                if val < best:
                    best = val
        return best
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        best_coeffs = np.random.normal(0, 1, 8)
    
    return best_coeffs
