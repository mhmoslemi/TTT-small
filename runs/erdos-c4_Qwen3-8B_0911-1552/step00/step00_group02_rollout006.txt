import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    x = sp.symbols('x')
    H = [sp.hermite_poly(k) for k in range(16)]  # precompute up to H_15 (indices 0..15)

    def evaluate_coefficients(coeffs):
        # Build polynomial Q(z)
        Q = sum(c * H[4*k] for k, c in enumerate(coeffs))
        # Ensure Q(0) = 0
        Q = Q - Q.subs(x, 0)
        # Ensure positivity at infinity
        leading_term = sp.LT(Q, x)
        if leading_term.coeff(x) < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / x**2
        # Find largest real root with sign change
        roots = sp.solvers.solve(Q, x)
        real_roots = [r.evalf() for r in roots if r.is_real]
        # Filter roots with sign change
        sign_changes = []
        for r in real_roots:
            if r > 0:
                sign_changes.append(r)
        if not sign_changes:
            return float('inf')
        r_max = max(sign_changes)
        return r_max**2 / (2 * np.pi)

    # Initial guess
    initial_guess = np.random.rand(C4_MAX_COEFF_COUNT)

    def objective(coeffs):
        # Convert to rational numbers for exact evaluation
        coeffs = np.array(coeffs, dtype=np.float64)
        coeff_rational = [sp.Rational(int(round(c * 10**10))) for c in coeffs]
        return evaluate_coefficients(coeff_rational)

    # Trust region optimization
    def optimize():
        result = minimize(
            lambda x: objective(x),
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(-5, 5) for _ in range(C4_MAX_COEFF_COUNT)],
            options={'maxfev': 500, 'disp': False}
        )
        return result.x

    # Run optimization within budget
    start_time = time()
    best_coeffs = optimize()
    elapsed = time() - start_time
    if elapsed > budget_s - 1:
        # If time is tight, try a few more steps with reduced tolerance
        result = minimize(
            lambda x: objective(x),
            x0=best_coeffs,
            method='L-BFGS-B',
            bounds=[(-5, 5) for _ in range(C4_MAX_COEFF_COUNT)],
            options={'maxfev': 100, 'disp': False}
        )
        best_coeffs = result.x
    return best_coeffs.tolist()
