import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.special import hermite
from functools import partial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_value = float('inf')
    for _ in range(5):  # Random restarts
        coeffs = np.random.rand(8) - 0.5
        def obj(x):
            nonlocal best_value
            coeffs = x
            m = len(coeffs)
            def Q(z):
                val = 0.0
                for k in range(m):
                    val += coeffs[k] * hermite(4*k, z)
                val -= coeffs[-1] * hermite(4*m, 0)  # Ensure Q(0) = 0

            # Find r_max
            try:
                r = root_scalar(lambda z: Q(z), bracket=[1, 1000], method='brentq')
                r_max = r.root

                return r_max**2 / (2 * np.pi)
            except:
                return 1e6
        res = minimize(obj, coeffs, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(8)], options={'maxiter': 50})
        if res.fun < best_value:
            best_value = res.fun
            best_coeff = res.x
    return best_coeff
