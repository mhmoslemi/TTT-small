from scipy.optimize import minimize
from sympy.polys.polytools import hermite
from sympy import symbols, Eq, solve, Rational, N

def run(seed=42, budget_s=100):

        r_max = real_roots[-1] if real_roots else 1
        bound = r_max**2 / (2 * np.pi)
        return bound

        return np.sum(coeffs**2) + 0.1 * np.sum(coeffs)
