from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve, simplify
from time import time

def run(seed=42, budget_s=100, **kwargs):

            return best_bound

        bound = r_max**2 / (2 * np.pi)

            best_coeffs = x.copy()
