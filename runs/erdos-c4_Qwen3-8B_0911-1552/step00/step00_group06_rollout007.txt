import numpy as np
import scipy.optimize as opt
from sympy import symbols, Eq, solve, simplify
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    coeffs = np.random.rand(C4_MAX_COEFF_COUNT)
    
    def objective(coeffs):
        def calc_r_max(c):
            z = symbols('z')
            Q = sum(c[k] * eval_hermite(4*k, z) for k in range(len(c)))
            Q = Q - Q.subs(z, 0)
            Q = Q / z**2
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            real_roots = sorted(real_roots)
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            return r_max
            
        def eval_hermite(n, z):
            return np.polys.Hermite(n)(z)
        
        r = calc_r_max(coeffs)
        if r is None:
            return float('inf')
        return r**2 / (2 * np.pi)
    
    start_time = time.time()
    for _ in range(100):
        grad = np.gradient(objective, coeffs, method='central', axis=0)
        coeffs -= 0.1 * grad
        if time.time() - start_time > budget_s:
            break
    
    return coeffs.tolist()

def eval_hermite(n, z):
    if n == 0:
        return 1
    elif n == 1:
        return 2*z
    elif n == 2:
        return 4*z**2 - 2
    elif n == 3:
        return 8*z**3 - 12*z
    elif n == 4:
        return 16*z**4 - 48*z**2 + 12
    else:
        raise ValueError("Hermite polynomial for n > 4 not implemented")
