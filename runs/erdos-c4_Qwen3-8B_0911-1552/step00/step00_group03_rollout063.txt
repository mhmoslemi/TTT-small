import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    def surrogate(c):
        def Q(z):
            return sum(c[k] * np.polyspecial.hermite(4*k, z) for k in range(len(c)))
        def f(z):
            return Q(z) / (z**2)
        try:
            sol = root_scalar(f, bracket=[1, 100], method='bvls')
            r_max = sol.root
            return r_max**2 / (2 * np.pi)
        except:
            return np.inf
    result = minimize(lambda c: surrogate(c), x0=np.random.rand(8), method='L-BFGS-B', options={'maxiter': 100})
    return result.x
