import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint
from sympy import symbols, diff, hermite, solve, Eq
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_score = float('inf')
    
    def evaluate(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        c_m = -Q / hermite(4*m, 0)
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m)) + c_m * hermite(4*m, z)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
        real_roots = [r for r in real_roots if r not in [r for r in real_roots if real_roots.count(r) > 1]]
        real_roots.sort()
        if not real_roots:
            r_max = 1000
        else:
            r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def cost(coeffs):
        return evaluate(coeffs)
    
    def objective(coeffs):
        return -cost(coeffs)
    
    def constraint(coeffs):
        return sum(coeffs) - 1
    
    def get_candidate_coefficients():
        return np.random.uniform(-1, 1, 8)

            best_coeff = res.x.copy()

            break
    
    return best_coeff.tolist()
