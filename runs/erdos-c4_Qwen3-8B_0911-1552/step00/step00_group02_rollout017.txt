import numpy as np
import sympy as sp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    best_c = c
    best_bound = float('inf')
    
    def objective(c):
        c = np.array(c)
        z = sp.symbols('z')
        Q = sum(c[k] * sp.hermite(h=4*k, z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[sp.hermite(h=0, z)] * z**2
        roots = sp.solvers.roots(Q, real=True)
        r_max = max(roots)
        return r_max**2 / (2 * np.pi)
    
    def evaluate(c):
        nonlocal best_c, best_bound
        try:
            val = objective(c)
            if val < best_bound:
                best_bound = val
                best_c = c
        except:
            pass
    
    for _ in range(100):
        t0 = time.time()
        result = minimize(lambda x: objective(x), x0=best_c, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)])
        if time.time() - t0 > budget_s * 0.7:
            break
        evaluate(result.x)
    
    return best_c
