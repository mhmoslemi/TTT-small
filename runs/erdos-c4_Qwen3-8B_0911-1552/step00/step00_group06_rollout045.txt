import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        c = np.array(coeffs)
        z = symbols('z')
        expr = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        expr = expr - expr.subs(z, 0)
        expr = expr.as_numerical()
        roots = solve(expr, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [float(r) for r in real_roots if r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def approximate_bound(coeffs):
        z = symbols('z')
        expr = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        expr = expr - expr.subs(z, 0)
        expr = expr.as_numerical()
        roots = solve(expr, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [float(r) for r in real_roots if r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return r_max**2 / (2 * np.pi)
    
    def objective(x):
        return approximate_bound(x)
    
    def exact_eval(x):
        return evaluate_coefficients(x)
    
    def refine_coeffs(coeffs, step=0.01):
        best_coeff = coeffs.copy()
        best_val = exact_eval(best_coeff)
        for i in range(len(coeffs)):
            for d in [-step, step]:
                new_coeff = coeffs.copy()
                new_coeff[i] += d
                val = exact_eval(new_coeff)
                if val < best_val:
                    best_coeff = new_coeff
                    best_val = val
        return best_coeff
    
    initial_guess = np.random.rand(8)
    best_coeff = initial_guess
    best_val = exact_eval(best_coeff)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_coeff, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        if res.success:
            best_coeff = res.x
            best_val = exact_eval(best_coeff)
        best_coeff = refine_coeffs(best_coeff)
    
    return best_coeff
