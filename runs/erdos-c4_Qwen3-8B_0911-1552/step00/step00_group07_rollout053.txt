import numpy as np
import sympy as sp
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def objective(c):
        return r_max_squared(c) / (2 * np.pi)
    
    def r_max_squared(c):
        z = sp.symbols('z')
        Q = sum(c[k] * sp.hermite(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0).as_coefficients_dict()[1] * Q
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = sp.roots(Q)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2
    
    def numerical_surrogate(c):
        z = sp.symbols('z')
        Q = sum(c[k] * sp.hermite(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        Q = Q.evalf()
        roots = np.roots([Q.as_poly().coeffs[i] for i in range(len(Q.as_poly().coeffs))])
        real_roots = [r for r in roots if np.isclose(r, np.round(r))]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        return real_roots[-1]**2
    
    best_coeff = np.random.rand(8)
    best_val = objective(best_coeff)
    
    for _ in range(100):
        time_start = time()
        res = minimize(lambda c: numerical_surrogate(c), best_coeff, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        best_coeff = res.x
        if time() - time_start > budget_s * 0.7:
            break
    
    # Verify with exact calculation
    final_val = objective(best_coeff)
    return best_coeff
