Use a gradient-based search with Sobol sequence to sample coefficient vectors. For each vector, compute the Hermite polynomial, find the root r_max via Newton-Raphson, and compute the upper bound. Track the minimum bound found within time budget.

import numpy as np
from scipy.optimize import minimize, newton
from scipy.stats import qmc
from functools import lru_cache

def hermite_poly(z, coeffs):
    return sum(c * np.polyspecial.hermitenorm(4*k, z) for k, c in enumerate(coeffs))

def evaluate_Q(z, coeffs):
    return hermite_poly(z, coeffs)

def find_r_max(coeffs):
    def Q(z):
        return evaluate_Q(z, coeffs)
    try:
        root = newton(lambda z: Q(z), 10)
        return root
    except:
        return np.inf

def compute_upper_bound(coeffs):
    try:
        r_max = find_r_max(coeffs)
        if np.isinf(r_max):
            return np.inf
        return r_max**2 / (2 * np.pi)
    except:
        return np.inf

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    dim = 8
    samples = 32
    iters = 10
    
    sobol = qmc.Sobol(d=dim)
    points = sobol.random(samples)
    best_coeffs = np.random.rand(dim)
    best_bound = np.inf

    for _ in range(iters):
        candidate = points[np.random.choice(samples)]
        coeffs = np.random.rand(dim)
        coeffs = (coeffs - 0.5) * 2.0
        current_bound = compute_upper_bound(coeffs)
        if current_bound < best_bound:
            best_bound = current_bound
            best_coeffs = coeffs.copy()
    return best_coeffs
