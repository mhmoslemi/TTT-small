import numpy as np
import sympy as sp
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from time import time

def run(seed=42, budget_s=100, **kwargs):
    # Precompute Hermite polynomials H_0 to H_32
    z = sp.symbols('z')
    h = [sp.hermite(n)(z) for n in range(33)]
    h = [sp.expand(hn) for hn in h]

    # Define the evaluator for given coefficients
    def evaluate(coeffs):

        Q = Q - Q.subs(z, 0)

            Q *= -1

        Q = Q / z**2
        # Find r_max
        def f(r):
            return Q.subs(z, r)
        try:
            r_max = root_scalar(f, bracket=[1, 100], method='brentq').root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    # Numerical surrogate for optimization
    def num_surrogate(coeffs):
        def fun(x):

        return fun

    init_coeffs = np.random.rand(8)

    start_time = time()
