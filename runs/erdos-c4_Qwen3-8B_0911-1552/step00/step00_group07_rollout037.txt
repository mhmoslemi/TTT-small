import numpy as np
import random
from scipy.optimize import bisect
from numpy.polynomial.hermite_e import hermeval
from numpy.polynomial.hermite_e import hermeframe

def evaluate_c4_coefficients(coeffs):
    m = len(coeffs)
    z = np.linspace(-100, 100, 100000)
    Q = np.zeros_like(z)
    for k in range(m):
        H = hermeframe([0]*m, [0]*m)
        H[k] = 1
        Q += coeffs[k] * hermeval(z, H)
    Q /= z**2
    try:
        r_max = bisect(lambda x: Q[np.abs(x) < 1000], 0, 100)
    except:
        r_max = 100
    return r_max**2 / (2 * np.pi)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    pop_size = 50
    generations = 10
    population = [np.random.rand(8) for _ in range(pop_size)]
    best = population[0]
    
    for gen in range(generations):
        if gen > 0:
            population = sorted(population, key=lambda x: evaluate_c4_coefficients(x))
            best = population[0]
        if gen % 2 == 0:
            new_pop = []
            for _ in range(pop_size // 2):
                p1 = random.choice(population)
                p2 = random.choice(population)
                child = (p1 + p2) / 2
                new_pop.append(child)
            population = new_pop + population[:pop_size // 2]
        else:
            population = [np.random.rand(8) for _ in range(pop_size)]
        if gen == generations - 1:
            break
    
    return best
