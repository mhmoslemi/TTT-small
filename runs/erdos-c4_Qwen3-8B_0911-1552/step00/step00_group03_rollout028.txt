Use gradient descent with adaptive step sizes to minimize U(c) by numerically approximating r_max for candidate coefficient vectors. Combine with exact rational checks for promising candidates using SymPy to ensure accuracy. Prioritize low U(c) values through iterative refinement.

import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from functools import wraps
import time

def evaluate_c4_coefficients(coeffs):
    import sympy as sp
    z = sp.symbols('z')
    H = [sp.hermite_h(i, z) for i in range(len(coeffs))]
    Q = sum(c * h for c, h in zip(coeffs, H))
    Q = Q - Q.as_poly().coeffs()[0] * z**2
    roots = sp.roots(Q.as_poly())
    real_roots = [r for r in roots if r.is_real]
    real_roots = sorted([float(r) for r in real_roots if r > 0])
    if not real_roots:
        return float('inf')
    r_max = real_roots[-1]
    return r_max**2 / (2 * np.pi)

def fast_roots(candidates):
    import numpy as np
    def f(x, c):
        H = [np.polynomial.hermite.hermite(i) for i in range(len(c))]
        Q = np.sum(c[i] * H[i](x) for i in range(len(c)))
        return Q / x**2
    
    res = []
    for c in candidates:
        try:
            r = opt.root_scalar(lambda x: f(x, c), bracket=[1e-5, 100], method='brentq', xtol=1e-8).root
            res.append(r**2 / (2 * np.pi))
        except:
            res.append(float('inf'))
    return res

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    def objective(coeffs):
        return fast_roots([coeffs])[0]
    
    def exact_eval(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def grad_func(coeffs):
        eps = 1e-6
        grads = []
        for i in range(len(coeffs)):
            p = coeffs.copy()
            p[i] += eps
            p_neg = coeffs.copy()
            p_neg[i] -= eps
            grads.append((exact_eval(p) - exact_eval(p_neg)) / (2 * eps))
        return np.array(grads)
    
    def score(coeffs):
        return exact_eval(coeffs)
    
    def inner_search(coeffs):
        result = minimize(
            fun=objective,
            x0=coeffs,
            method='L-BFGS-B',
            jac=grad_func,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        return result.x
    
    best_coeffs = np.random.uniform(-1, 1, 8)
    best_score = score(best_coeffs)
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        coeffs = inner_search(best_coeffs)
        current_score = score(coeffs)
        if current_score < best_score:
            best_score = current_score
            best_coeffs = coeffs
    
    return best_coeffs.tolist()
