<strategy>
Use gradient descent with random restarts to optimize the coefficient vector. For each candidate, compute the upper bound via Q(z)/z² sign change and r_max. Use a numerical surrogate for fast evaluation and exact checks for verification. Keep the best verified coefficients within budget.
</strategy>

```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_upper = float('inf')
    start_time = time.time()
    
    def evaluate_upper(coeffs):
        try:
            import sympy
            z = sympy.symbols('z')
            c = np.array(coeffs)
            terms = [c[k] * sympy.hermite(4*k, z) for k in range(len(coeffs))]
            Q = sum(terms)
            Q = Q - Q.subs(z, 0)
            Q = Q.as_numerical()
            
            roots = np.roots([Q.coeff(z, i) for i in range(len(Q)+1)])
            real_roots = np.sort([r for r in roots if np.isreal(r)])
            
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            if r_max is None:
                return float('inf')
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(coeffs):
        return evaluate_upper(coeffs)
    
    def update_best(new_coeff, new_upper):
        nonlocal best_coeff, best_upper
        if new_upper < best_upper:
            best_coeff = new_coeff
            best_upper = new_upper
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        init = np.random.uniform(-1, 1, 8)
        res = minimize(objective, init, method='L-BFGS-B', bounds=Bounds(-1, 1))
        if res.success:
            upper = res.fun
            update_best(res.x, upper)
    
    return best_coeff
```
