import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, Poly, roots, Eq, solve, N
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def objective(c):
        return np.sum(c**2)  # Surrogate for optimization
    
    def check_coefficients(c):
        z = symbols('z')
        terms = [c[k] * HermiteH(4*k, z) for k in range(len(c))]
        Q = sum(terms)
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2  # Remove z² factor
        roots = roots(Q, z, domain='Real')
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def search():
        initial = np.random.rand(8)
        result = minimize(objective, initial, method='L-BFGS-B', bounds=[(None, None) for _ in range(8)], options={'maxiter': 100})
        best = result.x
        min_val = float('inf')
        best_coeff = best
        for _ in range(20):
            perturb = np.random.normal(0, 0.1, 8)
            trial = best + perturb
            trial = np.clip(trial, -10, 10)
            score = check_coefficients(trial)
            if score < min_val:
                min_val = score
                best_coeff = trial
        return best_coeff
    
    start = time()
    coefficients = search()
    while time() - start < budget_s:
        coefficients = np.random.rand(8) * 2 - 1
    return coefficients
