import numpy as np
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, simplify, root, Eq, solve, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = None
    best_score = float('inf')
    start_time = time.time()

    def evaluate_coefficients(c):
        nonlocal best, best_score
        if time.time() - start_time > budget_s:
            return
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))

        Q = Q.subs(z, 0)

            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find r_max: largest positive root where Q changes sign
        try:
            roots = solve(Eq(Q, 0), z)
            real_roots = [r for r in roots if r.is_real and r > 0]
            real_roots.sort()
            # Check for sign change
            for i in range(len(real_roots) - 1):
                r1, r2 = real_roots[i], real_roots[i+1]
                if Q.subs(z, r1).evalf() * Q.subs(z, r2).evalf() < 0:
                    r_max = (r1 + r2) / 2
                    score = r_max**2 / (2 * np.pi)
                    if score < best_score:
                        best_score = score
                        best = c[:]
                    break
        except:
            pass

    # Initial grid search with random perturbations
    for _ in range(100):
