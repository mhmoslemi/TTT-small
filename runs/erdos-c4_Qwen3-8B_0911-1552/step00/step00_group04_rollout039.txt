import numpy as np
import scipy.optimize as opt
from sympy import symbols, HermiteH, Poly, solve, S, Rational
from time import time

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    c = np.random.rand(8) - 0.5
    best = c
    best_score = float('inf')
    start = time()

    def evaluate_coeffs(coeffs):
        coeffs = np.array(coeffs, dtype=float)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(len(coeffs)))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity
        if Q.as_leading_term(z) < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().div(z**2)[1]
        # Find real roots
        roots = solve(Q, z, domain=S.Reals)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([float(r) for r in real_roots])
        # Find r_max
        r_max = None
        for r in real_roots:
            if r > 0:
                r_max = r
                break
        if r_max is None:
            return float('inf')
        return r_max ** 2 / (2 * np.pi)

    def numerical_surrogate(coeffs):
        return np.sum(coeffs ** 2) + 0.1 * np.sum(np.abs(coeffs))

    def refine_coeffs(c):
        res = opt.minimize(lambda x: numerical_surrogate(x), c, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        return res['x']

    for _ in range(10):
        if time() - start > budget_s - 1:
            break
        c = refine_coeffs(c)
        score = evaluate_coeffs(c)
        if score < best_score:
            best_score = score
            best = c
        c += np.random.normal(0, 0.1, 8)

    return best.tolist()
