import numpy as np
import time
from scipy.optimize import root_scalar
from sympy import symbols, diff, Eq, solve
from functools import partial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    time_start = time.time()
    
    def objective(coeffs):
        nonlocal best_coeff, best_bound
        c = coeffs
        z = symbols('z')
        H = [1]
        for k in range(1, 4*len(c)+1):
            H.append(diff((z**k).expand(), z, k))
        Q = sum(c[k] * H[4*k] for k in range(len(c)))
        # Ensure Q(0) = 0
        Q0 = Q.subs(z, 0)
        if Q0 != 0:
            Q = Q - Q0
        # Ensure positivity at infinity
        leading_term = Q.as_coefficients_dict().get(z**(4*len(c)), 0)
        if leading_term < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q.expand()
        try:
            poly = Q.as_poly()
            degree = poly.degree()
            if degree < 2:
                return float('inf')
            # Normalize by z^2
            Q = Q / z**2
            def f(x):
                return Q.subs(z, x).evalf()
            # Find r_max: largest positive root where sign changes
            def sign_change(x):
                return np.sign(f(x))
            try:
                root = root_scalar(f, bracket=(0, 1000), method='brentq', xtol=1e-8)
                r = root.root
                # Check sign change
                sign_before = sign_change(r - 1)
                sign_after = sign_change(r + 1)
                if sign_before * sign_after > 0:
                    return float('inf')
                r_max = r
            except:
                return float('inf')
            U = r_max**2 / (2 * np.pi)
            return U
        except:
            return float('inf')
    
    def perturb(coeffs, step):
        return coeffs + np.random.normal(0, step, len(coeffs))
    
    def search():
        nonlocal best_coeff, best_bound
        for _ in range(100):
            if time.time() - time_start > budget_s - 1:
                break
            coeffs = np.random.rand(8)
            step = 0.1
            for _ in range(100):
                if time.time() - time_start > budget_s - 1:
                    break
                new_coeffs = perturb(coeffs, step)
                bound = objective(new_coeffs)
                if bound < best_bound:
                    best_bound = bound
                    best_coeff = new_coeffs
                step *= 0.95
            if time.time() - time_start > budget_s - 1:
                break
    
    search()
    return best_coeff
