import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, simplify, solve, N, Rational, Eq, diff
import time

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    
    def evaluate_U(c):
        z = symbols('z')
        H = [Hermite(i) for i in range(4*len(c)+1)]
        Q = sum(c[k] * H[4*k] for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [N(r) for r in real_roots]
        
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                r_max = r
                break
        else:
            r_max = max(real_roots) if real_roots else 1
        
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        return evaluate_U(c)
    
    def bounds(n):
        return [(-1, 1) for _ in range(n)]
    
    best_coeff = None
    best_value = float('inf')
    start_time = time.time()
    
    for n in range(1, 9):
        if time.time() - start_time > budget_s - 5:
            break
        initial_guess = np.random.uniform(-1, 1, n)
        result = minimize(objective, initial_guess, bounds=bounds(n), method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-8})
        
        if result.fun < best_value:
            best_value = result.fun
            best_coeff = result.x
    
    return best_coeff
