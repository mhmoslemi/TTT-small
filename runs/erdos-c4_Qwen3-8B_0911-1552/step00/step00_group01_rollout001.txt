import numpy as np
from scipy.optimize import root_scalar
from sympy import symbols, hermite, simplify

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    n_coeff = 8
    
    def evaluate(coeffs):
        z = symbols('z')
        H = [hermite(k, z) for k in range(n_coeff)]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = simplify(Q)
        # Ensure Q(0) = 0 and positive at infinity
        if Q.subs(z, 0) != 0:
            Q = Q - Q.subs(z, 0)
        # Check sign at infinity
        sign_inf = Q.as_leading_term(z).as_coeff_mul(z)[0].sign
        if sign_inf == -1:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().as_dict()
        if 2 in Q:
            Q = Q.copy()
            del Q[2]
            Q = sum(k * Q[k] * z**k for k in Q)
        else:
            Q = sum(k * Q[k] * z**k for k in Q)
        # Find r_max (largest real root where sign changes)
        def f(z_val):
            return Q.subs(z, z_val).evalf()
        try:
            roots = np.roots([Q.as_dict()[k] for k in sorted(Q.keys())[::-1]])
            real_roots = np.real(roots[np.isclose(np.imag(roots), 0)])
            real_roots = np.sort(real_roots)
            for i in range(len(real_roots)-1):
                if f(real_roots[i]) * f(real_roots[i+1]) < 0:
                    r_max = real_roots[i+1]
                    return r_max
            return None
        except:
            return None
    
    for _ in range(1000):
        coeffs = np.random.randn(n_coeff)
        coeffs = (coeffs - np.min(coeffs)) / (np.max(coeffs) - np.min(coeffs))
        coeffs = (coeffs * 10) + np.random.rand(n_coeff) * 5 - 2.5
        r_max = evaluate(coeffs)
        if r_max is not None:
            bound = r_max ** 2 / (2 * np.pi)
            if bound < best_bound:
                best_bound = bound
                best_coeff = coeffs
        if _ % 10 == 0:
            print(f"Iter {_:03d}: Best bound: {best_bound:.10f}")
    
    return best_coeff
