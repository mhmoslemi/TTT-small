import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, diff
import time

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    c = np.random.rand(8)
    best = c.copy()
    best_bound = float('inf')
    
    def objective(c):
        nonlocal best, best_bound
        try:
            z = symbols('z')
            H = [HermiteH(4*k, z) for k in range(len(c))]

                Q *= -1
            # Remove z² factor
            Q = Q / z**2
            # Find sign change roots
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            # Filter even multiplicity and find largest root
            r_max = -np.inf
            for r in real_roots:
                if r > 0:
                    # Check sign change
                    left = Q.subs(z, r - 1e-6).evalf()
                    right = Q.subs(z, r + 1e-6).evalf()
                    if left * right < 0:
                        r_max = max(r_max, r)
            if r_max == -np.inf:
                return float('inf')  # No sign change
            bound = r_max**2 / (2*np.pi)
            if bound < best_bound:
                best_bound = bound
                best = c.copy()
            return bound
        except:
            return float('inf')
    
    def surrogate(c):
        return np.sum(c**2)
    
    for _ in range(100):

            break
        best = result.x
        # Use exact evaluation for best candidate
        best_bound = objective(best)
    
    return best
