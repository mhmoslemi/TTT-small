import numpy as np
from scipy.optimize import fsolve
from scipy.optimize import minimize
from sympy import symbols, HermiteH, simplify, Poly, solve, Eq
import time

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    best_coeffs = None
    best_value = float('inf')

    def evaluate_coeffs(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q.as_poly()
        # Enforce Q(0) = 0
        Q = Q.subs(z, 0)
        Q = Q - Q.subs(z, 0)
        # Ensure positivity at infinity
        leading_term = Q.LM().as_expr()
        if leading_term.has(z**2):
            Q = Q - Q.as_poly().coeff_monomial(z**2) * z**2
        else:
            Q = Q - Q.as_poly().LC() * z**2
        # Simplify for root finding
        Q = simplify(Q)
        # Find r_max
        try:
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            if r_max is None:
                return float('inf')
            value = (r_max**2) / (2 * np.pi)
            return value
        except:
            return float('inf')

    def objective(x):
        return evaluate_coeffs(x)

    def perturb(x):
        return x + np.random.normal(0, 0.1, len(x))

    for _ in range(20):
        start_time = time.time()
        # Initial guess
        coeffs = np.random.uniform(-1, 1, 8)
        res = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
        if res.success:
            current_value = res.fun
            if current_value < best_value:
                best_value = current_value
                best_coeffs = res.x

        for _ in range(5):
            perturbed = perturb(res.x)
            res_perturbed = minimize(objective, perturbed, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)])
            if res_perturbed.success:
                current_value = res_perturbed.fun
                if current_value < best_value:
                    best_value = current_value
                    best_coeffs = res_perturbed.x
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_coeffs
