import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Poly, hermite, solve, diff, N
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    c = np.random.rand(8)
    best = c
    best_score = float('inf')
    time_start = time()
    
    def objective(x):
        nonlocal best, best_score
        if time() - time_start > budget_s:
            return best_score
        
        x = x.copy()
        z = symbols('z')
        H = [hermite(k, z) for k in range(8)]
        Q = sum(x[k] * H[k] for k in range(8))
        Q = Q - (Q.subs(z, 0))
        
        while Q.as_poly().lead_coeff != 1:
            Q = Q / Q.as_poly().lead_coeff
        
        Q = Q.as_poly()
        roots = solve(Q, z)
        
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        
        if not real_roots:
            return float('inf')
        
        r_max = max(real_roots)
        
        if r_max < 1e-8:
            return float('inf')
        
        score = r_max**2 / (2 * np.pi)
        if score < best_score:
            best = x
            best_score = score
        return score
    
    result = minimize(objective, x0=np.random.rand(8), method='L-BFGS-B', tol=1e-10, options={'max_iter': 1000})
    return best
