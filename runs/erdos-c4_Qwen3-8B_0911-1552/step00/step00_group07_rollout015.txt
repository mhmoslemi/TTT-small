def hermite_eval(coeffs, z):

    H = [Hermite(n)(z_sym) for n in range(len(coeffs))]
    return sum(c * h for c, h in zip(coeffs, H))

def compute_r_max(coeffs):

        return 1

def objective(coeffs):
