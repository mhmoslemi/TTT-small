import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import OptimizeResult
from sympy import symbols, HermiteH, Eq, solve, simplify, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    time_remaining = budget_s

    def evaluate_surrogate(c):
        c = np.array(c)
        coeffs = c
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(coeffs)))
        Q = Q.subs(z, 0)
        for k in range(len(coeffs)-1, -1, -1):
            if Q.has(HermiteH(4*k, z)):
                Q = Q.subs(HermiteH(4*k, z), 0)
        Q = Q.as_numer_denom()[0]
        Q = Q / z**2
        # Find real roots of Q(z)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        # Check sign changes
        r_max = None
        for i in range(len(real_roots)-1):
            r1 = real_roots[i]
            r2 = real_roots[i+1]
            if r1 < r2:
                r_max = r2
                break
        if r_max is not None:
            return r_max**2 / (2 * np.pi)
        return float('inf')

    def objective(c):
        nonlocal time_remaining
        start_time = time.time()
        if time_remaining <= 0:
            return float('inf')
        time_remaining -= time.time() - start_time
        try:
            bound = evaluate_surrogate(c)
            return bound
        except:
            return float('inf')

            if Q.has(HermiteH(4*k, z)):

        Q = Q / z**2

                break
        if r_max is not None:

            break
        if res.success:
            candidate = res.x
            bound = exact_check(candidate)
            if bound < best_bound:

                best_coeff = candidate

    return best_coeff
