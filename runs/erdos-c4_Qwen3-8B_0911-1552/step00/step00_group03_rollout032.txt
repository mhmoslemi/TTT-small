import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Hermite, simplify, Poly, roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    N = 8
    bounds = [(-10, 10) for _ in range(N)]
    
    def objective(c):
        c = np.array(c)
        z = symbols('z')
        Q = sum(c[k] * Hermite(4*k)(z) for k in range(N))
        Q = Q - Q.as_coefficients_dict()[z**0] * z**2
        try:
            roots = roots(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = np.sort([float(r) for r in real_roots if r > 0])
            if len(real_roots) == 0:
                return float('inf')
            r_max = real_roots[-1]
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def evaluate_c4_coefficients(coeffs):
        c = np.array(coeffs)
        z = symbols('z')
        Q = sum(c[k] * Hermite(4*k)(z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**0] * z**2
        try:
            roots = roots(Q, z, domain='Real')
            real_roots = [float(r) for r in roots if r.is_real and r > 0]
            if not real_roots:
                return float('inf')
            r_max = max(real_roots)
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    def inner_search():
        result = minimize(objective, x0=np.random.uniform(-10, 10, N), bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        return result.x

    start_time = time.time()
    best_coeffs = inner_search()

            break

        new_obj = objective(new_coeffs)
        if new_obj < objective(best_coeffs):
            best_coeffs = new_coeffs
    return best_coeffs.tolist()
