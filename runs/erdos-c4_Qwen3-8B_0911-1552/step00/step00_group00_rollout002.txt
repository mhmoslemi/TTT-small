import numpy as np
import sympy
from scipy.optimize import newton
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = None
    best_bound = float('inf')
    start_time = time.time()
    
    def evaluate_coefficients(coeffs):
        z = sympy.symbols('z')
        H = [sympy.hermite(n, z) for n in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q.subs(z, 0)
        c_m = -Q / (sympy.hermite(len(coeffs), z)).subs(z, 0)
        coeffs = list(coeffs) + [c_m]
        
        Q = sum(c * sympy.hermite(k, z) for k, c in enumerate(coeffs))
        Q = Q / z**2
        roots = sympy.roots(Q, z, domain=sympy.S.Reals)
        real_roots = [r for r in roots if roots[r].is_integer and r > 0]
        real_roots = sorted(real_roots)
        
        if not real_roots:
            return float('inf')
        
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    def fast_surrogate(coeffs):
        z = np.linspace(-10, 10, 10000)
        H = [sympy.hermite(k, z) for k in range(len(coeffs))]
        Q = sum(c * h for c, h in zip(coeffs, H))
        Q = Q / z**2
        sign_changes = np.where(np.sign(Q[1:-1]) != np.sign(Q[2:]))[0]
        if not sign_changes.size:
            return float('inf')
        r_max = np.abs(z[sign_changes[-1]])
        return r_max**2 / (2 * np.pi)
    
    for _ in range(100):
        coeffs = np.random.uniform(-1, 1, 8)
        bound = fast_surrogate(coeffs)
        if bound < best_bound:
            best_bound = bound
            best = coeffs.copy()
            print(f"Found better: {best_bound}")
    
    while time.time() - start_time < budget_s - 1:
        if best is None:
            break
        coeffs = best + np.random.normal(0, 0.1, 8)
        coeffs = np.clip(coeffs, -1, 1)
        bound = evaluate_coefficients(coeffs)
        if bound < best_bound:
            best_bound = bound
            best = coeffs.copy()
            print(f"Improved bound: {best_bound}")
    
    return best
