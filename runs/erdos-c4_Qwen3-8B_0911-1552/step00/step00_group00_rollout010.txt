import numpy as np
import random
from scipy.optimize import minimize
from sympy import symbols, HermiteH, simplify, solve, N, Poly
from scipy.optimize import root_scalar

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeffs = None
    best_value = float('inf')
    
    def cost(coeffs):
        c = coeffs
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.subs(z, 0)  # Enforce Q(0)=0
        Q = Q.as_poly().as_expr()
        Q = Q / z**2 if z**2 in Q.as_poly().free_symbols else Q

        def f(z_val):
            return Q.subs(z, z_val)

        try:
            res = root_scalar(f, bracket=[1, 100], method='brentq')
            r_max = res.root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(coeffs):
        return cost(coeffs)
    
    for _ in range(20):
        initial = np.random.uniform(-1, 1, 8)
        result = minimize(objective, initial, method='L-BFGS-B', bounds=[(-1, 1) for _ in range(8)])
        if result.fun < best_value:
            best_value = result.fun
            best_coeffs = result.x
    
    return best_coeffs
