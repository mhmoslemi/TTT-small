import numpy as np
import scipy.optimize as opt
import sympy as sp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    c = np.random.rand(C4_MAX_COEFF_COUNT)
    
    def objective(c):
        c = np.array(c, dtype=float)
        z = sp.symbols('z')
        H = [sp.hermt(n) for n in range(4*C4_MAX_COEFF_COUNT)]
        Q = sum(c[k] * H[4*k] for k in range(len(c)))

        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return np.inf
        r_max = real_roots[-1]
        return r_max**2 / (2 * np.pi)
    
    # Numerical optimization with bounds
    result = opt.minimize(
        fun=objective,
        x0=c,
        bounds=[(-10, 10) for _ in range(C4_MAX_COEFF_COUNT)],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxfun': int(budget_s / 0.1)}
    )
    
    return result.x.tolist()
