import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from sympy import symbols, HermiteH, simplify, solve
from itertools import product
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = None
    best_score = float('inf')
    
    def evaluate_coefficients(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * HermiteH(4*k, z) for k in range(m))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2
        roots = solve(Q, z)
        real_roots = [float(r) for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                r_max = r
                break
        else:
            r_max = 0
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        nonlocal best, best_score
        score = evaluate_coefficients(coeffs)
        if score < best_score:
            best_score = score
            best = coeffs.copy()
        return score
    
    def search():
        for _ in range(50):
            coeffs = np.random.uniform(-1, 1, 8)
            res = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(8)], options={'maxiter': 100})
            if res.success and res.fun < best_score:
                best_score = res.fun
                best = res.x
    
    start = time.time()
    search()
    elapsed = time.time() - start
    if elapsed > budget_s:
        print(f"Time exceeded. Returning best so far: {best}")
    else:
        print(f"Time used: {elapsed:.2f}s. Returning best so far: {best}")
    return best
