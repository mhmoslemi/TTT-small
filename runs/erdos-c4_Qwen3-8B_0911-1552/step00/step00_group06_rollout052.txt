from scipy.optimize import minimize, root_scalar
from scipy.special import hermite
import time

def run(seed=42, budget_s=100, **kwargs):

            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Initial guess with random coefficients
    initial_guess = np.random.rand(8)
    
    # Use gradient descent with restarts to minimize U(c)
    best_coeff = initial_guess
    best_score = obj_func(best_coeff)
