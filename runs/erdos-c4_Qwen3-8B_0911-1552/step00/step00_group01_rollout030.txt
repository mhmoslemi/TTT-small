Use gradient descent on coefficient ratios to minimize U(c) by numerically finding r_max via root isolation, with exact polynomial checks to ensure validity. Use 8 coefficients and leverage symmetry to reduce search space.

import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Rational, hermite, roots, Poly, S
from time import time

def run(seed=42, budget_s=100):
    np.random.seed(seed)
    c = np.random.rand(8)
    best = c.copy()
    best_val = float('inf')
    
    def objective(x):
        nonlocal best, best_val
        c = x
        try:
            # Construct Q(z) and enforce Q(0)=0
            z = symbols('z')
            Herm = [hermite(k, z) for k in range(8)]
            Q = sum(c[k]*Herm[k] for k in range(8))
            Q = Q - Q.subs(z, 0)
            
            # Ensure Q(z)/z² is positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q *= -1
            
            # Remove z² factor
            Q = Q.as_poly().as_ordered_basis()
            Q = Q - Q.as_poly().coeff_monomial(z**2)*z**2
            Q = Q.as_expr()
            
            # Find r_max
            roots_list = roots(Q)
            candidates = [float(r) for r in roots_list if r.is_real and r > 0]
            if not candidates:
                return float('inf')
            
            r_max = max(candidates)
            bound = r_max**2 / (2 * np.pi)
            
            if bound < best_val:
                best_val = bound
                best = c.copy()
            return bound
        except:
            return float('inf')
    
    start = time()
    while time() - start < budget_s:
        res = minimize(objective, best, method='L-BFGS-B', bounds=[(0, 10) for _ in range(8)], tol=1e-6)
        best = res.x
        if res.fun < best_val:
            best_val = res.fun
    
    return best
