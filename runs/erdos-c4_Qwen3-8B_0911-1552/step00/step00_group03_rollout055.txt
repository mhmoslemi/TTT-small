from scipy.optimize import minimize
from sympy import symbols, diff, hermite, simplify, roots, Poly
from time import time

def run(seed=42, budget_s=100, **kwargs):

    c = np.random.rand(8) - 0.5
    
    def surrogate(c):
        c = c.copy()
        c[-1] = 0
