import numpy as np
from scipy.optimize import minimize
from scipy.special import hermite
from itertools import product
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def evaluate_coefficients(coeffs):
        m = len(coeffs)
        H4k = [hermite(4*k, 0) for k in range(m)]
        Q = np.array([np.sum(coeffs[k] * H4k[k] for k in range(m))])
        return Q
    
    def r_max_function(coeffs):
        m = len(coeffs)
        x = np.linspace(-10, 10, 10000)
        Q = np.zeros_like(x)

        Q /= x**2
        sign_changes = np.where(np.diff(np.sign(Q)) != 0)[0]
        r_max = 0
        for idx in sign_changes:
            if x[idx] > r_max:
                r_max = x[idx]
        return r_max
    
    def score(coeffs):
        r_max = r_max_function(coeffs)
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        return score(coeffs)
    
    def gradient(coeffs):
        m = len(coeffs)
        x = np.linspace(-10, 10, 10000)
