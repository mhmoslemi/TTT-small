import numpy as np
from scipy.optimize import minimize
from scipy.optimize import root_scalar
from scipy.integrate import quad
from sympy import symbols, Hermite, simplify, Poly, solve
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    
    def evaluate_bound(c):
        z = symbols('z')

        return r_max**2 / (2 * np.pi)

        def f(x):
            return poly.subs(z, x)
        try:
            r_max = root_scalar(f, bracket=[1, 100], method='brentq').root
            return r_max**2 / (2 * np.pi)

            return float('inf')

        return fast_surrogate(c)

        return evaluate_bound(c)
