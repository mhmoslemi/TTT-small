import numpy as np
import random
from scipy.optimize import minimize
from sympy import symbols, Hermite, expand, solve, simplify, Eq, diff
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeff = None
    best_bound = float('inf')
    start_time = time()
    budget_remaining = budget_s

    def evaluate(coeffs):
        nonlocal best_bound
        if time() - start_time > budget_remaining:
            return
        try:
            c = np.array(coeffs)
            m = len(c)
            z = symbols('z')
            Q = sum(c[k] * Hermite(4*k, z) for k in range(m))
            Q = Q.as_expr()
            Q = Q - Q.as_poly().all_coeffs()[-1] * z**2
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real and r > 0]
            r_max = max(real_roots) if real_roots else 1.0
            bound = r_max**2 / (2 * np.pi)
            if bound < best_bound:
                best_bound = bound
                best_coeff = coeffs
            return bound
        except:
            return float('inf')

    def numerical_surrogate(coeffs):
        c = np.array(coeffs)
        m = len(c)
        z = np.linspace(-100, 100, 100000)
        Q = np.zeros_like(z)
        for k in range(m):
            H = np.polynomial.hermite_e.hermegrid2d(4*k, z, 100)
            Q += c[k] * H
        Q /= z**2
        sign_changes = np.where(np.sign(Q[1:] - Q[:-1]) != 0)[0]
        r_max = z[sign_changes[0]] if sign_changes.size > 0 else 1.0
        return r_max**2 / (2 * np.pi)

    for _ in range(10):
        coeffs = np.random.uniform(-1, 1, 8)
        res = minimize(numerical_surrogate, coeffs, method='L-BFGS-B', bounds=[(-1, 1)]*8, options={'ftol':1e-8, 'gtol':1e-8})
        if time() - start_time > budget_remaining - 1:
            break
        evaluate(res.x)

    return best_coeff
