import numpy as np
import random
from scipy.optimize import minimize
from sympy import symbols, Eq, solve, hermite, Roots, real_roots
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    C4_MAX_COEFF_COUNT = 8
    start_time = time()
    
    def eval_bound(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q - Q.as_poly().coeffs[-1] * z**2
        roots = real_roots(Q)
        r_max = 0
        for r in roots:
            if r > 0:
                r_max = max(r_max, r)
        return r_max ** 2 / (2 * np.pi)
    
    def fast_approx_bound(coeffs):
        z = np.linspace(-10, 10, 1000)
        m = len(coeffs)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * np.polynomial.hermite.hermval(z, np.zeros(4*k + 1))
        Q = Q - Q[np.abs(z) > 10] * 100
        sign_changes = np.where(np.diff(np.sign(Q)))[0]
        r_max = 0
        for idx in sign_changes:
            r_max = max(r_max, z[idx])
        return r_max ** 2 / (2 * np.pi)
    
    def objective(coeffs):
        return fast_approx_bound(coeffs)
    
    best_coeffs = np.random.rand(C4_MAX_COEFF_COUNT)
    best_value = objective(best_coeffs)
    
    for _ in range(100):
        if time() - start_time > budget_s - 1:
            break
        res = minimize(objective, best_coeffs, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(C4_MAX_COEFF_COUNT)])
        if res.success:
            best_coeffs = res.x
            best_value = res.fun
    
    return best_coeffs
