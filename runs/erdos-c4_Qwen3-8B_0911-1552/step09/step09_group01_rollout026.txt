import numpy as np
from scipy.optimize import Bounds, minimize
import time
from sympy import symbols, hermite, Poly, solve, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')

    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational coefficients
            c_rational = [Rational(int(round(x * 10**10))) for x in c]
            # Construct polynomial
            z = symbols('z')
            Q = sum(c_rational[k] * hermite(4*k, z) for k in range(len(c_rational)))
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Make Q positive at infinity
            if Q.as_leading_term(z) < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q.as_ordered_terms()[1]  # Skip leading z^2 term
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            # Find r_max
            r_max = None
            for i in range(len(real_roots)-1):
                if real_roots[i+1] - real_roots[i] > 1e-5:
                    r_max = real_roots[i+1]
                    break
            if r_max is None:
                r_max = real_roots[-1] if real_roots else 1
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial random search
    for _ in range(20):
        if time.time() - start_time > budget_s * 0.8:
            break
        guess = np.random.uniform(-2, 2, size=8)
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*8,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-9}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    # Refinement with Nelder-Mead
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        guess = np.random.uniform(-2, 2, size=8)
        result = minimize(
            objective,
            guess,
            method='Nelder-Mead',
            bounds=[(-2, 2)]*8,
            options={'xatol': 1e-8, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
