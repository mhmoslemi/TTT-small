from time import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time()

    best_score = float('inf')

        nonlocal best_score, best_coeff

            z = symbols('z')
            Q = sum(c[k] * H(4*k, z) for k in range(len(c)))

            Q = Q - Q.subs(z, 0)
            Q /= z**2
            # Find real roots
            roots_real = []
            for r in roots(Q, z):
                if r.is_real:
                    roots_real.append(r)
            # Remove roots of even multiplicity
            filtered_roots = []
            for r in roots_real:
                if isinstance(r, tuple):
                    root, mult = r
                    if mult % 2 == 1:
                        filtered_roots.append(root)
                else:
                    filtered_roots.append(r)
            # Compute r_max
            if not filtered_roots:
                return float('inf')
            r_max = max(filtered_roots)
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except Exception as e:
            return float('inf')
    
    # Helper function to evaluate Hermite polynomials
    def H(n, z):
        if n == 0:
            return 1
        elif n == 1:
            return 2*z
        else:
            return 2*z*H(n-1, z) - 2*(n-1)*H(n-2, z)

        return objective(c)
