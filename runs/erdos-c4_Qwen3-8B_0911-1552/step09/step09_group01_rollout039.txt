import numpy as np
import sympy as sp
from scipy.optimize import Bounds, minimize
import time
from skopt import BayesianOptimization

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct polynomial
            z = sp.symbols('z')
            Q = sum(c[k] * sp.hermiteH(4*k, z) for k in range(len(c)))

            Q = Q.subs(z, 0)
            const_term = Q.as_coefficients_dict().get(z**0, 0)
            if const_term != 0:
                Q -= const_term
            
            # Ensure positive at infinity
            if Q.as_leading_term(z) < 0:
                Q *= -1
            
            # Remove z^2 factor
            Q = Q.expand()
            if z**2 in Q.as_dict():
                Q = Q / z**2
            
            # Find real roots
            roots = sp.solvers.solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real and r != 0]
            real_roots = sorted([r for r in real_roots if r > 0])
            
            # Remove roots of even multiplicity
            unique_roots = []
            for r in real_roots:
                if not unique_roots or abs(r - unique_roots[-1]) > 1e-10:
                    unique_roots.append(r)
            
            if not unique_roots:
                return float('inf')
            
            r_max = unique_roots[-1]
            score = r_max**2 / (2 * np.pi)

                best_score = score
                best_coeff = c.copy()

        except:
            return float('inf')
    
    # Bayesian optimization with 8 coefficients
    pbounds = {f'c_{i}': (-2, 2) for i in range(8)}

        f=objective,
        pbounds=pbounds,
        random_state=seed,
        n_random_starts=10,
        acq_func='EI',
        acq_optimizer='random'
    )
    
    # Run optimization within budget

    while time.time() - start_time < budget_s:
        try:
            opt.maximize(init_points=2, n_iter=1)
        except:
            break
