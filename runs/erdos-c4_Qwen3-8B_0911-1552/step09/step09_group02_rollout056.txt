import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, Hermite, simplify, solve, Poly

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    budget = budget_s * 0.8
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            z = symbols('z')
            Q = 0
            for k in range(len(c)):
                Q += c[k] * Hermite(4*k, z)
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positivity at infinity
            leading_term = Q.as_coefficients_dict().get(z**(4*len(c)-2), 0)
            if leading_term < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q / z**2
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            
            # Find r_max as the largest positive real root
            r_max = 0
            for r in real_roots:
                if r > 0:
                    r_max = max(r_max, r)
            
            if r_max == 0:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    if time.time() - start_time > budget:
                        break
                    result = minimize(
                        objective,
                        guess,
                        method='L-BFGS-B',
                        bounds=[(-2, 2)]*4,
                        options={'eps': 1e-8, 'maxiter': 100}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget:
            break
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*4,
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
