import numpy as np
from scipy.optimize import minimize
from sympy import symbols, Eq, solve, expand, HermiteH, symbols, simplify
from time import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start = time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_c4_coefficients(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**2]*z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        if not real_roots:
            return float('inf')
        r_max = max(real_roots)
        return (r_max**2) / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Grid search with rational parameterization
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time() - start > budget_s * 0.7:
                        break
                    c = np.array([i/grid_points, j/grid_points, k/grid_points, l/grid_points])
                    result = minimize(
                        objective,
                        c,
                        method='L-BFGS-B',
                        bounds=[(-2, 2) for _ in range(4)],
                        options={'maxfun': 100, 'ftol': 1e-10}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Refinement with gradient descent
    for _ in range(50):
        if time() - start > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(4)],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
