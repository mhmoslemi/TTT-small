from scipy.optimize import Bounds, minimize
from sympy import symbols, expand, diff, hermite, solve, Eq, S
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])

            roots = solve(Q, z)

            real_roots = sorted([r for r in real_roots if r != 0])
            
            # Find r_max
            r_max = None
            for i in range(len(real_roots) - 1):

                    r_max = real_roots[i]
                    break
            
            if r_max is None:

            score = r_max ** 2 / (2 * np.pi)
