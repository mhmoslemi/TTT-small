import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, Hermite, expand, solve, roots, simplify
from fractions import Fraction
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational coefficients for symbolic computation
            c_rational = [Fraction(int(round(x * 10**10))) for x in c]
            
            z = symbols('z')
            Q = sum(c_rational[k] * Hermite(4*k, z) for k in range(len(c_rational)))
            Q = Q.as_poly()
            
            # Enforce Q(0) = 0
            if Q.subs(z, 0) != 0:
                Q += -Q.subs(z, 0)
            
            # Normalize to remove z^2 factor and find r_max
            Q = Q.as_expr()
            Q = Q / z**2 if z**2 in Q.as_expr().as_ordered_factors() else Q
            
            # Find real roots of Q(z) = 0
            roots_expr = solve(Q, z)
            real_roots = [r.evalf() for r in roots_expr if r.is_real]
            
            # Find the largest positive root
            positive_roots = [r for r in real_roots if r > 0]
            if positive_roots:
                r_max = max(positive_roots)
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with more points and refinement
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    for m in range(grid_points):
                        for n in range(grid_points):
                            for p in range(grid_points):
                                for q in range(grid_points):
                                    if time.time() - start_time > budget_s * 0.7:
                                        break
                                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 
                                                     1.0 + l / grid_points, 1.0 + m / grid_points, 1.0 + n / grid_points, 
                                                     1.0 + p / grid_points, 1.0 + q / grid_points])
                                    result = minimize(
                                        objective,
                                        guess,
                                        method='Nelder-Mead',
                                        bounds=[(-2, 2)]*8,
                                        options={'xatol': 1e-8, 'ftol': 1e-8}
                                    )
                                    if 'success' in result and result.success:
                                        best_coeff = result.x

    X = np.array([[1.0]*8])
    y = np.array([objective(x) for x in X])

    gp_kernel = C(1.0, (1e-5, 1e5)) * RBF(1.0, (1e-5, 1e5))
    gp = GaussianProcessRegressor(kernel=gp_kernel, n_restarts_optimizer=25)
    gp.fit(X, y)
    
    # Optimize using Bayesian optimization
    for _ in range(20):
        if time.time() - start_time > budget_s * 0.9:
            break
        next_x = np.random.uniform(-2, 2, size=8)
        result = minimize(
            objective,
            next_x,
            method='Nelder-Mead',
            bounds=[(-2, 2)]*8,
            options={'xatol': 1e-8, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
