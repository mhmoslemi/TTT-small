from scipy.optimize import minimize
import time
from sympy import symbols, diff, hermite, roots, simplify

def hermite_poly(n):
    z = symbols('z')
    return hermite(n, z)

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.array([1.0]*8)

    def objective(c):
        nonlocal best_score, best_coeff

            z = symbols('z')

            Q = Q - Q.subs(z, 0)

            if Q.as_leading_term(z) < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.expand()
            if Q.as_poly().degree() >= 2:
                Q = Q / z**2
            
            # Find real roots
            roots_list = roots(Q, z, domain='Real')
            real_roots = [r for r in roots_list if r.is_real]
            
            # Find r_max
            real_roots = [float(r) for r in real_roots]
            real_roots.sort()
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()

        except:
            return float('inf')
