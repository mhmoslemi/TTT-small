from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, hermite, simplify, Poly, roots, S

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.rand(8) * 2 - 1  # Initial guess
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Create symbolic variables
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
            
            # Enforce Q(0) = 0
            Q0 = Q.subs(z, 0)
            if Q0 != 0:
                Q = Q - Q0
            
            # Enforce positivity at infinity by flipping sign if needed
            leading_term = Q.as_poly().leaders[-1]
            if leading_term.coeff(z**4*len(c)) < 0:
                Q = -Q
            
            # Remove z^2 factor and find largest positive root
            Q = Q.as_poly().div(z**2)[1]
            roots_list = roots(Q, z, domain=S.Reals)
            r_max = 0
            for r in roots_list:
                if r.is_real and r > 0 and roots_list[r].is_odd:
                    r_max = max(r_max, r)
            
            if r_max == 0:
                return float('inf')
            
            U_c = r_max**2 / (2 * np.pi)
            return U_c
        except:
            return float('inf')
    
    # Initial grid search with adaptive refinement
    grid_points = 5
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    for m in range(grid_points):
                        for n in range(grid_points):
                            for o in range(grid_points):
                                for p in range(grid_points):
                                    for q in range(grid_points):
                                        guess = np.array([i/grid_points, j/grid_points, k/grid_points, l/grid_points, m/grid_points, n/grid_points, o/grid_points, q/grid_points])
                                        if time.time() - start_time > budget_s * 0.7:
                                            break
                                        result = minimize(
                                            objective,
                                            guess,

                                        )

        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
