import numpy as np
import sympy
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Enforce Q(0) = 0 and positive at infinity
            z = sympy.symbols('z')
            H = [sympy.hermite(n, z) for n in range(4*len(c))]
            Q = sum(c[k]*H[4*k] for k in range(len(c)))
            
            # Remove z^2 factor
            Q = Q / z**2
            
            # Find real roots
            roots = sympy.solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            
            # Find r_max
            r_max = 0
            for r in real_roots:
                if r > 0 and r > r_max:
                    r_max = r
            
            if r_max == 0:
                return float('inf')
            
            score = r_max**2 / (2*np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with 4 coefficients
    grid = np.linspace(-2, 2, 10)
    best = None
    for c1 in grid:
        for c2 in grid:
            for c3 in grid:
                for c4 in grid:
                    if time.time() - start_time > budget_s * 0.8:
                        break
                    c = np.array([c1, c2, c3, c4])
                    result = minimize(
                        objective,
                        c,
                        method='L-BFGS-B',
                        bounds=[(-2, 2)]*4,
                        options={'maxiter': 50, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best = result.x
    return best if best is not None else np.array([1.0, 1.0, 1.0, 1.0])
