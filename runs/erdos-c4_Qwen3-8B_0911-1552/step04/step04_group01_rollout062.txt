import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, Poly, solve, simplify

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_hermite_coefficients(coeffs):
        m = len(coeffs)
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
        Q = Q - Q.subs(z, 0)
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([float(r) for r in real_roots if r > 0])
        for r in real_roots:
            if np.isclose(Q.subs(z, r), 0):
                continue
            if np.sign(Q.subs(z, r - 1e-6)) != np.sign(Q.subs(z, r + 1e-6)):
                r_max = r
                break
        else:
            r_max = 1e6
        return r_max**2 / (2 * np.pi)
    
    def objective(coeffs):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_hermite_coefficients(coeffs)
            if score < best_score:
                best_score = score
                best_coeff = coeffs.copy()
            return score
        except:
            return float('inf')
    
    initial_guesses = [
        np.array([1.0, 1.0, 1.0, 1.0]),
        np.array([1.0, -1.0, 1.0, -1.0]),
        np.array([1.0, 0.5, 0.5, 0.25]),
        np.array([1.0, 1.0, 0.5, 0.25]),
        np.array([1.0, 0.5, 0.25, 0.125]),
        np.array([1.0, 1.0, 1.0, 0.5]),
        np.array([1.0, 0.707, 0.5, 0.353]),
        np.array([1.0, 0.6, 0.4, 0.2])
    ]
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(len(guess))],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
