from scipy.optimize import minimize
from sympy import symbols, hermite, roots, N

        try:
            Q = 0

                Q += c[k] * hermite(4*k, z)

            r_roots = roots(Q, z)
            real_roots = [N(r) for r in r_roots if r.is_real and r > 0]
            if not real_roots:
                return float('inf')
            r_max = max(real_roots)
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(c):
        return exact_eval(c)
    
    def get_optimal_coefficients(n_coeff):
        freq_weights = np.array([1.0/(4*k + 1) for k in range(n_coeff)])
        c0 = freq_weights
        
        def refine_guess(c):
            for _ in range(20):
                c = c * np.random.uniform(0.8, 1.2)
                c = c / np.linalg.norm(c)
                return c
        
        c = refine_guess(freq_weights)
        result = minimize(objective, c, method='L-BFGS-B', bounds=[(-10, 10) for _ in range(n_coeff)], tol=1e-10, options={'maxiter': 50})
