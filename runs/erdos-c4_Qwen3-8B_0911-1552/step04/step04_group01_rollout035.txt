import numpy as np
import sympy as sp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.rand(8) - 0.5
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational representation
            coeffs = np.array(c, dtype='float64')
            coeffs = np.round(coeffs, 10)
            
            # Build Hermite polynomial
            z = sp.symbols('z')
            Q = sp.Integer(0)
            for k in range(len(coeffs)):
                Q += coeffs[k] * sp.hermite(h=4*k, z=z)
            
            # Enforce Q(0) = 0 and positive at infinity
            Q = Q - Q.as_coeff_add()[0][0]
            Q = Q.as_poly().as_expr()
            Q = Q / z**2
            
            # Find real roots
            roots = sp.solvers.solve(Q, z, domain=sp.S.Reals)
            roots = [r.evalf() for r in roots if r.is_real and r > 0]
            
            if not roots:
                return float('inf')
            
            r_max = max(roots)
            score = r_max**2 / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial guesses with Chebyshev-inspired values
    initial_guesses = []
    for i in range(8):
        initial_guesses.append(np.array([np.cos(np.pi * i / 8) for _ in range(8)]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(8)],
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
