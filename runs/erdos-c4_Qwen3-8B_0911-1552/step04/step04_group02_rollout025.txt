from scipy.optimize import minimize
import time
from sympy import symbols, hermite, Poly, roots, N, simplify
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):

            poly = simplify(Q_z_over_z2)

            real_roots = [N(root) for root in roots(p) if root.is_real and root > 0]
            real_roots = sorted(real_roots)
            r_max = 1.0
            for r in reversed(real_roots):
                r_max = r
                break
            return r_max**2 / (2 * np.pi)
