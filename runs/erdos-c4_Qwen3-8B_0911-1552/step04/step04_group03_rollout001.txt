import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, diff, hermite, Poly, solve, N

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_coeffs = np.array([1.0])
    best_score = float('inf')
    
    def symbolic_check(coeffs):
        try:
            m = len(coeffs)
            z = symbols('z')
            Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
            Q = Q.as_ordered_terms()
            Q = Q.as_poly(z)
            Q = Q * z**-2
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = [N(r) for r in real_roots if r.is_real and r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def score(c):
        return symbolic_check(c)
    
    for m in range(2, 8):
        initial = np.random.uniform(-1, 1, m)
        result = minimize(score, x0=initial, method='L-BFGS-B', bounds=Bounds(-2, 2), options={'ftol': 1e-8, 'maxfun': 100})
        if score(result.x) < best_score:
            best_coeffs = result.x
            best_score = score(result.x)
    
    return best_coeffs
