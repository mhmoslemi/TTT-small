import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import Bounds
from scipy.stats import qmc
from functools import lru_cache
import sympy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = np.array([1.0])
    best_score = float('inf')

    z = sympy.symbols('z')
    H = [sympy.hermite(n, z) for n in range(0, 32, 4)]

    def exact_check(coeffs):
        try:
            Q = sum(c * H[k] for k, c in enumerate(coeffs))
            Q = Q.as_poly(z)
            Q = Q * z**-2
            roots = sympy.roots(Q)
            real_roots = [r for r in roots if r.is_real and r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    def numerical_surrogate(coeffs):
        try:
            Q = sum(c * H[k] for k, c in enumerate(coeffs))
            Q = Q.as_poly(z)
            Q = Q * z**-2
            roots = np.roots(Q.all_coeffs())
            real_roots = [r for r in roots if np.isclose(r.imag, 0) and r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')

    for m in range(1, 8):
        lhc = qmc.LatinHypercube(d=m, c=seed)
        samples = lhc.random(n=20)
        samples = np.clip(samples * 2, -2, 2)

        def objective(x):
            return numerical_surrogate(x)

        res = minimize(objective, x0=samples[0], method='L-BFGS-B', bounds=Bounds(-2, 2), options={'ftol': 1e-8, 'maxiter': 50})
        candidates = samples[1:]
        for x in candidates:
            if np.linalg.norm(x) > np.linalg.norm(res.x):
                continue
            score = numerical_surrogate(x)
            if score < best_score:
                best_score = score
                best = x

    return best
