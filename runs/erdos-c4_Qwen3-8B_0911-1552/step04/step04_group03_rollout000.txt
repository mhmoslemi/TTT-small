from scipy.optimize import minimize, Bounds
from sympy import symbols, diff, hermite, Poly, solve, N, Rational

def run(seed=42, budget_s=100, **kwargs):

    best_score = float('inf')
    
    def symbolic_check(coeffs):
        try:
            m = len(coeffs)
            z = symbols('z')
            Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
            Q = Q.as_ordered_terms()
            Q = Q.as_poly(z)
            Q = Q * z**-2

            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def score(c):
        return symbolic_check(c)
    
    def surrogate(x):
        return score(x)
    
    def generate_initial(m):
        return np.random.uniform(-1, 1, m)
    
    def optimize_coeffs(m):
        initial = generate_initial(m)
        res = minimize(surrogate, x0=initial, method='COBYLA', bounds=Bounds(-2, 2), options={'maxiter': 300, 'disp': False})

    for m in range(1, 8):
