import time
from sympy import symbols, diff, expand, HermiteH

def evaluate_c4_coefficients(coefficients):
    z = symbols('z')
    Q = sum(c * HermiteH(4 * k, z) for k, c in enumerate(coefficients))
    Q = Q.as_poly().as_expr()
    Q = Q.subs(z, 0)
    if Q != 0:
        Q = Q - Q.as_poly().coeffs()[-1]
    Q = Q / z**2
    roots = np.roots([c for c in Q.as_poly().all_coeffs() if c != 0])
    real_roots = [r for r in roots if np.isclose(r.imag, 0)]
    real_roots = sorted([r.real for r in real_roots])
    for r in real_roots:
        if r > 0:
            return r**2 / (2 * np.pi)
    return np.inf
