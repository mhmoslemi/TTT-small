import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = np.array([1.0])
    best_score = float('inf')
    
    def evaluate(coeffs):
        m = len(coeffs)
        z = np.linspace(-100, 100, 100000)
        Q = np.zeros_like(z)
        for k in range(m):
            Q += coeffs[k] * np.polynomial.hermite.hermval(z, np.zeros(m))
        Q /= z**2
        sign_changes = np.where(np.sign(Q[1:] - Q[:-1]) != 0)[0]
        r_max = np.max(z[sign_changes + 1])
        return r_max**2 / (2 * np.pi)
    
    def score(c):
        try:
            return evaluate(c)
        except:
            return float('inf')
    
    def surrogate(x):
        return score(x)
    
    for m in [4]:
        grid = np.linspace(-1, 1, 50)
        grid = np.vstack([grid**i for i in range(m)]).T
        for pt in grid:
            res = minimize(surrogate, x0=pt, method='L-BFGS-B', bounds=[(-2, 2) for _ in range(m)], options={'ftol': 1e-10, 'maxiter': 100})
            current_score = score(res.x)
            if current_score < best_score:
                best = res.x
                best_score = current_score
    return best
