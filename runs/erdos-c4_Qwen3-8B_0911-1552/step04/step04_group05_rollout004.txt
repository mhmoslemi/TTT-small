from scipy.optimize import differential_evolution, minimize
from scipy.optimize import Bounds
from functools import partial
from scipy.optimize import minimize_scalar
from numpy.polynomial.hermite_e import hermeval, hermefity
from scipy.optimize import root_scalar

def run(seed=42, budget_s=100, **kwargs):

    def score(c):
        try:
            # Construct polynomial
            z = np.linspace(-10, 10, 1000)
            Q = np.zeros_like(z)
            for k in range(len(c)):
                H = hermeval(z, [0]*k + [1])
                Q += c[k] * H
            
            # Ensure Q(0) = 0 and positive at infinity
            if np.abs(Q[0]) > 1e-10:
                Q -= Q[0]
            
            # Find r_max
            def f(t):
                return Q[np.abs(t) < 100] / t**2
            
            try:
                root = root_scalar(lambda t: f(t), bracket=[1, 100], method='brentq')
                r_max = root.root
                return r_max**2 / (2 * np.pi)
            except:
                return float('inf')
        except:
            return float('inf')
    
    # Initial exploration with differential evolution
    bounds = Bounds(-2, 2)
    result = differential_evolution(score, bounds=bounds, strategy='best1bin',

    best = result.x
    best_score = score(best)
    
    # Local refinement with L-BFGS-B
    def surrogate(x):
        return score(x)
    
    res = minimize(surrogate, x0=best, method='L-BFGS-B', bounds=bounds, options={'ft0l': 1e-8, 'maxiter': 50})
    best = res.x
    best_score = score(best)
