import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import minimize_scalar
from scipy.optimize import TNC
from sympy import symbols, diff, hermite, Poly, solve, N, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best = np.array([1.0])
    best_score = float('inf')
    
    def symbolic_check(coeffs):
        try:
            m = len(coeffs)
            z = symbols('z')
            Q = sum(coeffs[k] * hermite(4*k, z) for k in range(m))
            Q = Q.as_ordered_terms()
            Q = Q.as_poly(z)
            Q = Q * z**-2
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = [N(r) for r in real_roots if r.is_real and r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def score(c):
        return symbolic_check(c)
    
    def surrogate(x):
        return score(x)
    
    for m in range(4, 5):
        initial = np.random.uniform(-1, 1, m)
        res = minimize(surrogate, x0=initial, method='L-BFGS-B', bounds=Bounds(-2, 2), options={'ftol': 1e-8, 'maxiter': 50})
        current_score = score(res.x)
        if current_score < best_score:
            best = res.x
            best_score = current_score
    
    return best
