import numpy as np
from sympy import symbols, Hermite, solve, roots, simplify
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    z = symbols('z')
    H = [Hermite(k) for k in range(8)]
    
    def build_Q(coeffs):
        Q = sum(c * H[i] for i, c in enumerate(coeffs))
        Q = Q.subs(z, z).as_poly()
        return Q
    
    def find_r_max(Q):
        Q = Q.as_expr()
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        for r in real_roots:
            if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                return r
        return 1000
    
    def evaluate_c4_coefficients(coeffs):
        Q = build_Q(coeffs)
        r_max = find_r_max(Q)
        return (r_max**2) / (2 * np.pi)
    
    def objective(coeffs):
        return evaluate_c4_coefficients(coeffs)
    
    def f(x):
        return objective(x)
    
    def generate_grid(n, low, high):
        return np.linspace(low, high, n)
    
    def grid_search(low, high, n):
        best = np.inf
        best_coeff = None
        for c0 in generate_grid(n, low, high):
            for c1 in generate_grid(n, low, high):
                for c2 in generate_grid(n, low, high):
                    for c3 in generate_grid(n, low, high):
                        c = np.array([c0, c1, c2, c3])
                        val = f(c)
                        if val < best:
                            best = val
                            best_coeff = c
        return best_coeff
    
    initial_guess = np.random.uniform(-5, 5, 4)
    result = minimize(f, initial_guess, method='L-BFGS-B', bounds=[[-5, 5] for _ in range(4)])
    best = result.x
    return best
