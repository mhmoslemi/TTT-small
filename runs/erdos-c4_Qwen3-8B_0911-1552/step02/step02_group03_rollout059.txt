from scipy.optimize import minimize

    def hermite_eval(z, coeffs):
        m = len(coeffs)
        res = 0
        for k in range(m):
            res += coeffs[k] * np.polynomial.hermite.hermval(z, [1.0])**(4*k)
        return res
    
    def compute_r_max(coeffs):
        from scipy.optimize import brentq
        def Q(z):
            return hermite_eval(z, coeffs)
        
        try:
            r_pos = brentq(lambda z: Q(z)/z**2, 1, 100)
            return r_pos**2 / (2 * np.pi)
        except:
            return np.inf
    
    def objective(coeffs):
        try:
            return compute_r_max(coeffs)
        except:
            return np.inf
    
    result = minimize(objective, initial_coeff, method='L-BFGS-B', bounds=[(-5,5)]*8, options={'maxiter': 100})
    return result.x
