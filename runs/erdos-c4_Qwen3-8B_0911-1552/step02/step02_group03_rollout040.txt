from sympy import symbols, hermite, solve, Rational
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    z = symbols('z')
    H = [hermite(k, z) for k in range(8)]
    
    def compute_Q(c):
        Q = sum(c[k] * H[k] for k in range(len(c)))
        return Q
    
    def find_r_max(Q):
        Q = Q.as_poly()
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        real_roots.sort()
        r_max = None
        for i in range(len(real_roots)-1):
            if real_roots[i+1] - real_roots[i] > 1e-8:
                r_max = real_roots[i]
                break
        return r_max
    
    def evaluate_coefficients(c):
        Q = compute_Q(c)
        Q = Q.as_poly()
        if Q.LC() < 0:
            Q = -Q
        r_max = find_r_max(Q)
        if r_max is None:
            return np.inf
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        return evaluate_coefficients(c)
    
    initial_guess = np.random.uniform(-1, 1, 8)
    
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[[-1, 1] for _ in range(8)], options={'maxiter': 100})
    
    return result.x
