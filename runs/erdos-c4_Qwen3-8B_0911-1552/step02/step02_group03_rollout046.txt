import sympy as sp
from scipy.optimize import minimize
from scipy.optimize import NonlinearConstraint

        return val

        def Q(z):
            return hermite_eval(coeffs, z)

            return Q(z) / z**2

        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        
        r_max = None

            if Q_z2(r).evalf() * Q_z2(r + 1e-5).evalf() < 0:
                r_max = r
                break
        return r_max

        except:
