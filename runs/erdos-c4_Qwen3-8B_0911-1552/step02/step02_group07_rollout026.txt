import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, simplify, roots, solve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def eval_bound(c):
        try:
            from evaluate_c4_coefficients import evaluate_c4_coefficients
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    def objective(c):
        return eval_bound(c)
    
    def find_r_max(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
        Q = Q - Q.subs(z, 0)
        roots_list = roots(Q, z, domain='Real')
        real_roots = [float(r) for r in roots_list.keys() if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        r_max = real_roots[-1] if real_roots else 1.0
        return r_max
    
    def compute_bound(c):
        r_max = find_r_max(c)
        return r_max**2 / (2 * np.pi)
    
    def grid_search(num_points=50):
        c_min, c_max = -2, 2
        c_values = np.linspace(c_min, c_max, num_points)
        best_c = np.zeros(8)
        best_bound = float('inf')
        for i in range(8):
            for j in range(8):
                c = np.zeros(8)
                c[i] = c_values[i]
                c[j] = c_values[j]
                bound = compute_bound(c)
                if bound < best_bound:
                    best_bound = bound
                    best_c = c
        return best_c
    
    initial_guess = grid_search()
    
    def refine(bounds, max_iter=10):
        for _ in range(max_iter):
            x0 = np.random.uniform(bounds[:, 0], bounds[:, 1])
            res = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'ft': 1e-10, 'maxiter': 20})
            if res.success:
                new_bounds = np.array([(res.x[i] - 0.5, res.x[i] + 0.5) for i in range(len(res.x))])
                bounds = new_bounds
    
    best_c = initial_guess
    refine(np.array([[-2, 2] for _ in range(8)]), max_iter=5)
    
    return best_c[:4]
