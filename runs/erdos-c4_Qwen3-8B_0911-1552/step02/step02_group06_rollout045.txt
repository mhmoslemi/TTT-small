import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, solve, Eq, Rational, simplify
from sympy.polys.polytools import factor

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Known low bounds from literature
    low_bounds = np.array([Rational(1, 4), Rational(1, 4), Rational(1, 4), Rational(1, 4)])
    high_bounds = np.array([1, 1, 1, 1])
    
    def evaluate_with_symbolics(c):
        z = symbols('z')
        Q = sum(c[k] * HermiteH(4*k, z) for k in range(4))
        Q = Q - Q.as_coeff_add()[0]  # Remove constant term
        Q = Q / z**2
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real and r > 0 and r != 0]
        unique_real_roots = np.unique(real_roots)
        if len(unique_real_roots) == 0:
            return float('inf')
        r_max = max(unique_real_roots)
        return r_max**2 / (2*np.pi)
    
    def objective(c):
        try:
            return evaluate_with_symbolics(c)
        except:
            return float('inf')
    
    def surrogate(x):
        return objective(x)
    
    # Multi-start optimization
    results = []
    for _ in range(10):
        init = np.random.uniform(low_bounds, high_bounds)
        res = minimize(surrogate, x0=init, method='L-BFGS-B', bounds=[[-2, 2] for _ in range(4)], options={'ftol': 1e-8, 'maxiter': 100})
        results.append((res.fun, res.x))
    
    # Select best result
    results.sort()
    best_coeff = results[0][1]
    return best_coeff
