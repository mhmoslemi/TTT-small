import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def eval_bound(c):
        try:
            res = evaluate_c4_coefficients(c)
            return res[0]
        except:
            return float('inf')
    
    def objective(c):
        return eval_bound(c)
    
    # Generate a structured grid in 4D space with 10 points in each dimension
    grid_points = []
    for c0 in np.linspace(-2, 2, 10):
        for c1 in np.linspace(-2, 2, 10):
            for c2 in np.linspace(-2, 2, 10):
                for c3 in np.linspace(-2, 2, 10):
                    grid_points.append([c0, c1, c2, c3])
    
    best = grid_points[0]
    best_value = objective(best)
    
    for p in grid_points:
        val = objective(p)
        if val < best_value:
            best = p
            best_value = val
    
    # Local optimization with BFGS
    result = minimize(objective, best, method='L-BFGS-B', bounds=[[-2, 2] for _ in range(4)], options={'ftol': 1e-10, 'maxiter': 100})
    
    return result.x
