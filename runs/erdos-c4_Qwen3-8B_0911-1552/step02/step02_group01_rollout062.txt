import numpy as np
from scipy.optimize import differential_evolution, Bounds
from scipy.optimize import minimize
import time
from sympy import symbols, Hermite, lcm, solve, Interval, Root
from sympy.calculus.util import minimum, maximum

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.uniform(-1, 1, 8)
    best_score = float('inf')
    
    def evaluate_exact(c):
        try:
            z = symbols('z')
            Q = sum(c[k] * Hermite(4*k, z) for k in range(len(c)))
            Q = Q.subs(z, 0)
            Q = Q.as_leading_coefficient()
            Q = Q / z**2
            r_max = maximum(Root(Q, Interval(0, 100)))
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_exact(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial low-dimensional structured search
    for c0 in np.linspace(-1, 1, 10):
        for c1 in np.linspace(-1, 1, 10):
            for c2 in np.linspace(-1, 1, 10):
                for c3 in np.linspace(-1, 1, 10):
                    for c4 in np.linspace(-1, 1, 10):
                        for c5 in np.linspace(-1, 1, 10):
                            for c6 in np.linspace(-1, 1, 10):
                                for c7 in np.linspace(-1, 1, 10):
                                    c = np.array([c0, c1, c2, c3, c4, c5, c6, c7])
                                    score = evaluate_exact(c)
                                    if score < best_score:
                                        best_score = score
                                        best_coeff = c.copy()
    
    # Differential evolution with structured initialization
    bounds = Bounds([-1, 1], [-1, 1])
    result = differential_evolution(
        objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=30,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Gradient descent refinement
    if time.time() - start_time < budget_s * 0.8:
        result_refined = minimize(
            objective,
            best_coeff,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if 'success' in result_refined and result_refined.success:
            best_coeff = result_refined.x
    
    return best_coeff
