import numpy as np
import sympy as sp
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    H = [sp.hermite(h) for h in range(33)]  # Precompute Hermite polynomials
    
    def evaluate_coefficients(c):
        Q = sum(c[k] * H[4*k] for k in range(len(c)))

        return r_max**2 / (2 * np.pi)
    
    def fitness(c):
        try:
            return evaluate_coefficients(c)

            return float('inf')
    
    def objective(c):
        return fitness(c)
