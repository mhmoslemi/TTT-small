import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import Bounds, differential_evolution
from sympy import symbols, Rational, sqrt, hermite, expand, solve, Eq
from sympy.polys.polyroots import roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define symbolic variables
    z = symbols('z')
    H = [hermite(i, z) for i in range(4*8 + 1)]
    
    # Function to evaluate Q(z) and find r_max
    def evaluate_c4_coefficients(c):
        coeffs = [Rational(c[i]) for i in range(len(c))]
        Q = sum(coeffs[k] * H[4*k] for k in range(len(c)))
        Q = Q - Q.as_coeff_monomial().coeff * z**2
        var = symbols('z')
        Q = Q.subs(z, var)
        roots_list = roots(Q, var)
        r_max = 0
        for r in roots_list:
            if r.is_real and r > 0:
                if roots_list[r].is_positive:
                    r_max = max(r_max, r)
        return r_max**2 / (2 * np.pi)
    
    # Hybrid search
    best_coeff = [-1.0, 1.0]
    best_score = float('inf')
    
    # First stage: Latin hypercube + differential evolution
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    bounds = Bounds([-1, 1], [-1, 1])
    result = differential_evolution(
        objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Second stage: Bayesian optimization with Gaussian process
    from sklearn.gaussian_process import GaussianProcessRegressor
    from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
    
    class BayesianOptimizer:
        def __init__(self, bounds, objective, max_iter=100):
            self.bounds = bounds
            self.objective = objective
            self.max_iter = max_iter
            self.X = np.array([[-1, -1], [1, 1]])
            self.y = np.array([objective([-1, -1]), objective([1, 1])])
            self.kernel = C(1.0, (0.1, 10.0)) * RBF(1.0, (0.1, 10.0))
            self.gpr = GaussianProcessRegressor(kernel=self.kernel, n_restarts_optimizer=10)
        
        def optimize(self):
            for _ in range(self.max_iter):
                self.gpr.fit(self.X, self.y)
                mean, std = self.gpr.predict(self.X, return_std=True)
                next_point = np.random.uniform(self.bounds[0], self.bounds[1])
                self.X = np.vstack((self.X, next_point))
                self.y = np.append(self.y, self.objective(next_point))
                self.gpr.fit(self.X, self.y)
    
    if time.time() - start_time < budget_s * 0.6:
        opt = BayesianOptimizer(bounds, objective)
        opt.optimize()
    
    # Third stage: Gradient descent refinement
    def loss(c):
        return evaluate_c4_coefficients(c)
    
    from scipy.optimize import minimize
    
    result = minimize(
        loss,
        best_coeff,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if result.success:
        best_coeff = result.x
    
    return best_coeff
