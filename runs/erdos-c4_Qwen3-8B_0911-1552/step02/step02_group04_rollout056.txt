def run(seed=42, budget_s=100, **kwargs):

    # Define Hermite polynomials
    z = symbols('z')
    H = [hermite(n, z) for n in range(16)]

        m = len(c)
        Q = sum(c[k] * H[4*k] for k in range(m))

            r_max = valid_roots[-1] if valid_roots else 1
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(c):
        return numerical_evaluate(c)
