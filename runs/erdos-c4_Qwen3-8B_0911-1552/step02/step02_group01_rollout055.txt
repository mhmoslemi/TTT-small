Use a structured initialization with low-dimensional simplex for Hermite coefficients, then apply Bayesian optimization with gradient descent for refinement, leveraging analytical bounds on roots for early termination.

import numpy as np
from scipy.optimize import Bounds, minimize, minimize_scalar
from scipy.optimize import differential_evolution
from scipy.integrate import quad
from sympy import symbols, hermite, solve, diff, simplify
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define symbolic variables
    z = symbols('z')
    
    # Precompute Hermite polynomials up to H_4k for k=0 to 7
    H = [hermite(i, z) for i in range(0, 32, 4)]
    
    # Objective function for evaluating coefficients
    def evaluate_coefficients(c):
        Q = sum(c[i] * H[i] for i in range(len(c)))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Check sign at infinity
        sign_at_inf = Q.as_poly().leading_term().coeff
        if sign_at_inf > 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q.as_poly().div(z**2)[0]
        # Find roots
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        # Filter by sign change
        real_roots_sorted = sorted(real_roots)
        if not real_roots_sorted:
            return float('inf')
        r_max = real_roots_sorted[-1]
        # Compute upper bound
        return (r_max**2) / (2 * np.pi)
    
    # Define bounds for coefficients
    bounds = Bounds([-1, 1], [-1, 1])
    
    # Initial guess with structured initialization
    c0 = np.array([0.5, -0.5, 0.3, -0.3, 0.2, -0.2, 0.1, -0.1])
    c0 = c0 * np.random.randn(len(c0))
    
    # Bayesian optimization with gradient descent refinement
    def objective(c):
        try:
            return evaluate_coefficients(c)
        except:
            return float('inf')
    
    # First stage: Differential evolution
    result = differential_evolution(
        objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-4,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Second stage: L-BFGS-B with early termination
    if time.time() - start_time < budget_s * 0.7:
        result_refined = minimize(
            objective,
            result.x,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result_refined and result_refined.success:
            result.x = result_refined.x
    
    return result.x
