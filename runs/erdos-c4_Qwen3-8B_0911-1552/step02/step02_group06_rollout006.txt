import numpy as np
from scipy.optimize import minimize
from sympy import symbols, diff, HermiteH, solve, S

def run(seed=42, budget_s=100, **kwargs):

    initial = np.random.uniform(-1, 1, 8)
    best = initial.copy()
    best_score = float('inf')
    
    def score(c):
        try:
            c = np.array(c, dtype=np.float64)
            c = np.pad(c, (0, 8 - len(c)), 'constant')
            c = c[:8]
            
            z = symbols('z')
            Q = sum(c[k] * HermiteH(4*k, z) for k in range(8))
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)

            if Q.as_powerseries(z, 10).lead_term().as_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q / z**2
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([float(r) for r in real_roots if r > 0])
            
            # Find r_max
            r_max = 0
            for r in real_roots:
                if Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                    r_max = r
                    break
            
            if r_max == 0:
                return float('inf')
                
            return r_max**2 / (2*np.pi)
        except:

    # Multi-start optimization
