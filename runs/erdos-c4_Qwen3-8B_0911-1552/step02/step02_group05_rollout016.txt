import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from sympy import symbols, Hermite, Poly, solve, Eq

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    def eval_bound(c):
        try:
            import sympy
            z = symbols('z')
            H = [Hermite(n) for n in range(len(c))]
            Q = sum(c[k] * H[k] for k in range(len(c)))
            Q = Q.subs(z, z).as_poly()
            Q = Q - Q.subs(z, 0)
            Q = Q / z**2
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real and r > 0]
            real_roots = [r for r in real_roots if not r.is_zero]
            real_roots = sorted(real_roots)
            for i in range(len(real_roots)):
                if real_roots[i].is_real and real_roots[i] > 0:
                    r_max = real_roots[i]
                    break
            else:
                r_max = 1
            return r_max**2 / (2*np.pi)
        except:
            return float('inf')
    
    def fitness(c):
        return eval_bound(c)
    
    def objective(c):
        return -fitness(c)
    
    def generate_initial_points(n, dim):
        points = []
        for _ in range(n):
            c = np.random.uniform(-5, 5, dim)
            points.append(c)
        return points
    
    def refine_points(points, dim):
        refined = []
        for p in points:
            for i in range(dim):
                for delta in [-1, 1]:
                    new_p = p.copy()
                    new_p[i] += delta * 0.1
                    refined.append(new_p)
        return refined
    
    def find_best(points):
        best = points[0]
        best_fitness = fitness(best)
        for p in points:
            f = fitness(p)
            if f < best_fitness:
                best = p
                best_fitness = f
        return best
    
    initial_points = generate_initial_points(10, 8)
    best = find_best(initial_points)
    
    for _ in range(5):
        points = refine_points(initial_points, 8)
        best = find_best(points)
        initial_points = points
    
    res = minimize(objective, best, method='L-BFGS-B', bounds=[[-5, 5] for _ in range(8)], options={'ftol': 1e-8, 'maxiter': 50})
    
    return res.x
