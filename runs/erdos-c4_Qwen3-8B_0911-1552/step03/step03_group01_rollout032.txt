import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from skopt import BayesSearchCV
from skopt.space import Real, Categorical
from skopt.utils import use_file_object
from sympy import symbols, Hermite, simplify, solve, Eq, solve_real

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    c = np.random.uniform(-1, 1, 8)
    
    def objective(x):
        try:
            coeffs = x
            symbols_list = symbols('z')
            z = symbols_list[0]
            Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(len(coeffs)))
            
            # Ensure Q(0) = 0
            Q0 = Q.subs(z, 0)
            if Q0 != 0:
                Q -= Q0
            
            # Flip sign for positive at infinity
            if Q.as_coefficients_dict().get(z**np.inf, 0) < 0:
                Q *= -1
            
            # Remove z^2 factor
            Q = Q.as_poly().as_expr()
            Q = Q / z**2
            
            # Find real roots
            roots = solve(Q, z, domain='Real')
            real_roots = [r for r in roots if r.is_real]
            
            # Find r_max
            real_roots = sorted(real_roots)
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            else:
                r_max = 1
            
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Bayesian optimization
    space = [Real(-1, 1, name=f'c_{i}') for i in range(8)]
    opt = BayesSearchCV(
        estimator=None,
        search_space=space,
        objective=objective,
        n_initial_points=16,
        n_iter=40,
        random_state=seed,
        n_jobs=1
    )
    
    with use_file_object("/dev/null"):
        opt.fit(np.random.uniform(-1, 1, (16, 8)))
    
    return opt.best_params_.values
