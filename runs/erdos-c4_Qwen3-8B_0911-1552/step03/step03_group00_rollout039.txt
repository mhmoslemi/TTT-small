import numpy as np
from scipy.optimize import minimize
from scipy.stats import qloggamma
from functools import lru_cache
import time
import sympy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the Hermite polynomials up to H_24
    H = [lambda z: z]
    for n in range(1, 25):
        H_n = sympy.hermite(n, z)
        H.append(lambda z: H_n)
    
    def evaluate_c4_coefficients(c):
        # Symbolic calculation of Q(z)
        z = sympy.symbols('z')
        Q = sum(c[k] * H[4*k](z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict()[z**2] * z**2  # Remove z^2 factor
        roots = sorted(sympy.roots(Q, z, domain=sympy.S.Reals))
        for root in roots:
            if root.is_real and root > 0:
                r_max = root
                break
        else:
            r_max = 1.0
        return r_max**2 / (2 * np.pi)
    
    def numerical_approximation(c):
        # Numerical approximation of r_max using Newton-Raphson
        z = np.linspace(0, 10, 1000)
        Q = sum(c[k] * np.polynomial.hermite.hermval(z, [0]*k + [1] + [0]*(24 -k)) for k in range(len(c)))
        Q = Q - Q[0] * z**2  # Remove z^2 factor
        roots = []
        for i in range(len(z)-1):
            if Q[i] * Q[i+1] < 0:
                roots.append(z[i])
        if roots:
            r_max = np.max(roots)
        else:
            r_max = 1.0
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        try:
            return numerical_approximation(c)
        except:
            return float('inf')
    
    # Grid search with scaling
    scales = np.logspace(-2, 2, 10)
    best_coeff = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    best_score = float('inf')
    
    def grid_search():
        nonlocal best_score, best_coeff
        for scale in scales:
            c = np.array([0.5 * scale for _ in range(6)])
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
    
    grid_search()
    
    # Bayesian optimization
    class BayesianOpt:
        def __init__(self, n_points=20):
            self.n_points = n_points
            self.points = []
            self.values = []
        
        def suggest(self):
            if len(self.values) < self.n_points:
                return np.random.uniform(-1, 1, 6)
            else:
                weights = np.exp(-np.array(self.values))
                return np.random.choice(self.points, p=weights/np.sum(weights))
        
        def update(self, point, value):
            self.points.append(point)
            self.values.append(value)
    
    optimizer = BayesianOpt(n_points=20)
    
    while time.time() - start_time < budget_s * 0.8:
        point = optimizer.suggest()
        score = objective(point)
        optimizer.update(point, score)
        if score < best_score:
            best_score = score
            best_coeff = point
    
    # L-BFGS refinement
    if time.time() - start_time < budget_s:
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-1, 1) for _ in range(6)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
