from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from scipy.optimize import Bounds

    for _ in range(20):
        c = np.random.uniform(-1, 1, 8)
        s = score(c)
        if s < best_score:
            best = c
            best_score = s
    
    # Bayesian optimization
    kernel = ConstantKernel(1.0) * RBF(length_scale=1.0)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
    X = np.array([best])
    y = np.array([best_score])
    
    for _ in range(50):
        gp.fit(X, y)
        next_point = gp.predict(X, return_std=True)
        next_point = next_point[0]
        next_point = np.clip(next_point, -1, 1)
        next_point = np.random.normal(next_point, 0.1)
        next_point = np.clip(next_point, -1, 1)
        new_score = score(next_point)
        X = np.vstack((X, next_point))
        y = np.append(y, new_score)
        if new_score < best_score:
            best = next_point
            best_score = new_score
    
    # Local refinement with L-BFGS-B
    def surrogate(x):
        return score(x)
    
    res = minimize(surrogate, x0=best, method='L-BFGS-B', bounds=Bounds(-1, 1), options={'ftol': 1e-8, 'maxiter': 50})
    best = res.x
