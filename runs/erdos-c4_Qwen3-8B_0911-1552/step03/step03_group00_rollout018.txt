import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.random.rand(8) * 2 - 1
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Particle swarm optimization with adaptive scaling
    class PSO:
        def __init__(self, n_particles, dimensions, bounds, inertia=0.5, cognitive=1.2, social=1.2):
            self.n_particles = n_particles
            self.dimensions = dimensions
            self.bounds = bounds
            self.inertia = inertia
            self.cognitive = cognitive
            self.social = social
            self.particles = np.random.rand(n_particles, dimensions) * (bounds[1] - bounds[0]) + bounds[0]
            self.velocities = np.random.rand(n_particles, dimensions) * 0.1
            self.pbest = self.particles.copy()
            self.gbest = self.pbest[0].copy()
        
        def update_velocity(self, global_best):
            self.velocities = self.inertia * self.velocities + \
                             self.cognitive * np.random.rand(self.n_particles, self.dimensions) * (self.pbest - self.particles) + \
                             self.social * np.random.rand(self.n_particles, self.dimensions) * (global_best - self.particles)
        
        def update_positions(self):
            self.particles = np.clip(self.particles + self.velocities, self.bounds[0], self.bounds[1])
        
        def run(self, objective, max_iter=100):
            for _ in range(max_iter):
                fitness = np.array([objective(p) for p in self.particles])
                self.pbest = np.where(fitness < np.array([objective(p) for p in self.pbest]), self.particles, self.pbest)
                global_best_idx = np.argmin(fitness)
                self.gbest = self.pbest[global_best_idx].copy()
                self.update_velocity(self.gbest)
                self.update_positions()
            return self.gbest
    
    pso = PSO(n_particles=30, dimensions=8, bounds=(-1, 1))
    if time.time() - start_time < budget_s * 0.7:
        pso_result = pso.run(objective, max_iter=50)
        best_coeff = pso_result
    
    # Gradient descent refinement
    if time.time() - start_time < budget_s * 0.9:
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-1, 1)]*8,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
