from scipy.optimize import minimize, root_scalar
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):

    def evaluate_Q(z, c):

        return q / (z**2)
    
    def find_r_max(c):
        try:
            def f(z):
                return evaluate_Q(z, c)

            r_max = result.root

    def objective(c):

        except:

        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, tol=1e-10, options={'maxiter': 100})

    best_val = float('inf')
