from scipy.optimize import differential_evolution
import time
from sympy import symbols, hermite, simplify

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()

    z = symbols('z')

        H.append(simplify(hermite(n, z)))

            poly = poly.as_poly().as_dict()

            if not poly:
                return float('inf')

            roots = solveset(poly, z, domain='Reals')

                return float('inf')
