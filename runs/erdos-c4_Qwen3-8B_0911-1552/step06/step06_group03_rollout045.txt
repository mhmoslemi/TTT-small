import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, simplify, diff, solve, Eq, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Define symbolic variables
    z = symbols('z')
    H = [hermite(k, z) for k in range(8)]
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Build polynomial
            Q = sum(c[k] * H[k] for k in range(len(c)))
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Check positivity at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_poly().removeTerms()
            Q = Q.as_expr()
            Q = Q.subs(z, z**2)
            
            # Find largest positive real root
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real and r > 0]
            real_roots = sorted(real_roots)
            
            # Check for sign changes
            r_max = None
            for r in real_roots:
                if r.is_rational:
                    r = r.as_numer_denom()[0] / r.as_numer_denom()[1]
                if r_max is None or r > r_max:
                    r_max = r
            
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial guesses with different ratios
    initial_guesses = []
    for n in range(1, 9):
        for ratio in [0.5, 1.0, 2.0]:
            ch = np.array([ratio**k / np.sqrt(2**k) for k in range(n)])
            initial_guesses.append(ch)
            initial_guesses.append(-ch)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        try:
            result = minimize(objective, guess, method='L-BFGS-B', bounds=[(-2, 2) for _ in range(len(guess))], tol=1e-10)
            if result.success:
                best_coeff = result.x
        except:
            continue
    
    return best_coeff
