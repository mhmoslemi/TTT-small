import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def objective(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    # Use a more efficient parameterization with 4 coefficients
    # and perform a more guided search with adaptive bounds
    
    def adaptive_bounds(x):
        return [(-1.5, 1.5) if i < 3 else (-1, 1) for i, x in enumerate(x)]
    
    # Initial grid search with adaptive bounds
    grid_points = 8
    best_coeff = np.array([0.5, 0.5, 0.5, 0.5])
    best_score = float('inf')
    
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.8:
                        break
                    guess = np.array([
                        0.5 + i / grid_points,
                        0.5 + j / grid_points,
                        0.5 + k / grid_points,
                        0.5 + l / grid_points
                    ])
                    result = differential_evolution(
                        objective,
                        bounds=adaptive_bounds(guess),
                        strategy='best1bin',
                        popsize=20,
                        maxiter=50,
                        tol=1e-8,
                        mutation=(0.5, 1),
                        recombination=0.9,
                        polish=True,
                        disp=False
                    )
                    if result.fun < best_score:
                        best_score = result.fun
                        best_coeff = result.x
    
    return best_coeff
