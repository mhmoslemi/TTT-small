import numpy as np
from scipy.optimize import differential_evolution
from sympy import symbols, hermite, roots, simplify
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    z = symbols('z')
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        try:
            m = len(c)
            Q = sum(c[k] * hermite(4*k, z) for k in range(m))
            Q = Q.subs(z, 0)
            Q = Q.as_poly().all_coeffs()[0] * Q
            Q = Q - Q.as_poly().all_coeffs()[0] * Q
            Q = Q / z**2
            roots = roots(Q, z, domain='Real')
            real_roots = [r for r in roots if r.is_real and r > 0]
            r_max = max(real_roots) if real_roots else 1.0
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Initial grid search
    grid_points = 8
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.8:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points,
                                     1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = differential_evolution(
                        objective,
                        bounds=[(-2, 2)]*4,
                        strategy='best1bin',
                        popsize=20,
                        maxiter=50,
                        tol=1e-8,
                        mutation=(0.5, 1),
                        recombination=0.9,
                        polish=True,
                        disp=False
                    )
                    if result.fun < best_score:
                        best_score = result.fun
                        best_coeff = result.x
    
    return best_coeff
