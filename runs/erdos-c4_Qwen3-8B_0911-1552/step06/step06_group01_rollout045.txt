import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Precompute Hermite polynomials up to H_32
    z = symbols('z')
    H = [hermite(n, z) for n in range(0, 32)]
    
    def objective(c):
        try:
            # Construct Q(z)
            Q = sum(c[k] * H[4*k] for k in range(len(c)))
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Ensure positivity at infinity by checking leading term
            leading_term = Q.as_poly().coeffs()[-1]
            if leading_term < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q.expand()
            if Q.as_poly().degree() < 2:
                return float('inf')
            Q = Q / z**2
            # Find real roots
            roots = np.roots(Q.as_poly().all_coeffs()[::-1])
            # Find largest positive real root with odd multiplicity
            real_roots = np.sort(roots.real)
            r_max = None
            for r in real_roots:
                if r > 0:
                    # Check multiplicity
                    count = 0
                    for i in range(len(real_roots)):
                        if abs(real_roots[i] - r) < 1e-10:
                            count += 1
                    if count % 2 == 1:
                        r_max = r
                        break
            if r_max is None:
                return float('inf')
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Grid search with rational coefficients
    for m in range(1, 9):
        for n in range(1, m+1):
            for signs in range(2**n):
                coeffs = np.array([Rational(1, 2**k) for k in range(n)])
                if signs % 2 == 1:
                    coeffs *= -1
                if m > n:
                    coeffs = np.concatenate([coeffs, np.zeros(m - n)])
                if time.time() - start_time > budget_s * 0.7:
                    break
                result = minimize(
                    objective,
                    coeffs,
                    method='L-BFGS-B',
                    bounds=[(-2, 2) for _ in range(m)]
                )
                if 'success' in result and result.success:
                    best_coeff = result.x
    
    return best_coeff
