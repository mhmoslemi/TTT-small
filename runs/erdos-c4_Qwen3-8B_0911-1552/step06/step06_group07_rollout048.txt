import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from public import evaluate_c4_coefficients

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial guesses with diverse starting points
    initial_guesses = [
        np.array([1.0, 1.0, 1.0, 1.0]),
        np.array([-1.0, 1.0, 1.0, 1.0]),
        np.array([1.0, -1.0, 1.0, 1.0]),
        np.array([1.0, 1.0, -1.0, 1.0]),
        np.array([1.0, 1.0, 1.0, -1.0]),
        np.array([2.0, 1.0, 1.0, 1.0]),
        np.array([1.0, 2.0, 1.0, 1.0]),
        np.array([1.0, 1.0, 2.0, 1.0]),
        np.array([1.0, 1.0, 1.0, 2.0]),
        np.array([0.5, 1.0, 1.0, 1.0]),
        np.array([1.0, 0.5, 1.0, 1.0]),
        np.array([1.0, 1.0, 0.5, 1.0]),
        np.array([1.0, 1.0, 1.0, 0.5])
    ]
    
    # Run differential evolution
    if time.time() - start_time < budget_s * 0.6:
        result = differential_evolution(
            objective,
            bounds=[(-2, 2) for _ in range(4)],
            strategy='best1bin',
            popsize=20,
            maxiter=30,
            tol=1e-10,
            mutation=(0.5, 1),

        )

    from skopt import BayesSearchCV
    from skopt.space import Real
    from skopt.utils import use_named_args
    
    space = [Real(-2, 2) for _ in range(4)]
    
    @use_named_args
    def refined_objective(**params):
        c = np.array([params[f'x{i}'] for i in range(4)])
        return objective(c)
    
    opt_res = BayesSearchCV(
        estimator=None,
        search_space=space,
        Function=refined_objective,
        n_iter=20,
        random_state=seed
    ).fit({})
    
    best_coeff = opt_res.x
    
    return best_coeff
