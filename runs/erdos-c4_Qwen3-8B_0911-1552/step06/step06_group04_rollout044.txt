from scipy.optimize import differential_evolution, minimize
from scipy.special import hermite
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    def evaluate_Q(coeffs, z):
        m = len(coeffs)
        Q = 0
        for k in range(m):
            Q += coeffs[k] * hermite(4*k, z)
        return Q
    
    def compute_bound(coeffs):
        try:
            # Numerical surrogate for root finding
            def f(z):
                return evaluate_Q(coeffs, z) / (z**2)
            
            from scipy.optimize import root_scalar
            result = root_scalar(f, bracket=[1, 100], method='brentq')
            r_max = result.root
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    def objective(c):
        return compute_bound(c)
    
    # Initial grid search on log scale
    grid_points = 8
    best_coeff = np.logspace(-2, 2, grid_points)
    best_score = float('inf')
    
    for i in range(grid_points):
        if time.time() - start_time > budget_s * 0.8:
            break
        guess = np.logspace(-2, 2, 8)
        guess[i] = np.logspace(-2, 2, grid_points)[i]
        try:
            result = differential_evolution(
                objective,
                bounds=[(-2, 2) for _ in range(8)],
                strategy='best1bin',
                popsize=20,
                maxiter=50,
                tol=1e-8,
                mutation=(0.5, 1),
                recombination=0.9,
                polish=True,
                disp=False
            )
            if result.fun < best_score:
                best_score = result.fun

        except:
            pass
