from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time
    
    def objective(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    # Grid search with random perturbations
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    for _ in range(100):
        if time.time() - start_time >= budget_s:
            break

        score = objective(c)
        if score < best_score:
            best_score = score
            best_coeff = c.copy()
    
    # Bayesian optimization refinement
    def refine_objective(c):
        return objective(c)
    
    result = minimize(
        refine_objective,
        x0=best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)]*8
    )
    best_coeff = result.x
    
    return best_coeff
