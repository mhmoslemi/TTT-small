from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def objective(c):
        try:
            return evaluate_c4_coefficients(c)
        except:
            return float('inf')
    
    # Initial grid search with more coefficients
    grid_points = 5
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    result = differential_evolution(
                        objective,
                        bounds=[(-2, 2)] * len(guess),
                        strategy='best1bin',
                        popsize=20,
                        maxiter=50,
                        tol=1e-8,
                        mutation=(0.5, 1),
                        recombination=0.9,
                        polish=True,
                        disp=False
                    )
                    if result.fun < best_score:
                        best_score = result.fun
                        best_coeff = result.x
    
    # Local refinement
    result = minimize(
        objective,
        x0=best_coeff,
        method='L-BFGS-B',
        bounds=[(-2, 2)] * len(best_coeff),
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    return result.x
