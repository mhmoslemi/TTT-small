Use orthogonal basis for Hermite polynomials, scale coefficients to enforce Q(0)=0, and optimize with bounded trust regions to explore lower bounds efficiently.

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Enforce Q(0) = 0 by subtracting c_m * H_0(0)
            # H_0(z) = 1, so c_m = -c_0
            c = np.copy(c)
            c = np.concatenate((c, [-c[0]]))
            # Evaluate Q(z) = sum c_k H_{4k}(z)
            # Due to computational limits, only use first 3 coefficients
            # Approximate with first few Hermite polynomials
            def hermite(n, z):
                return np.polynomial.hermite_e.hermegrid2d(n, z)
            z = np.linspace(-10, 10, 1000)
            Q = np.zeros_like(z)
            for k in range(3):
                Q += c[k] * hermite(4*k, z)
            # Remove z^2 factor manually
            Q /= z**2
            # Find sign change root
            def sign_change(z):
                return np.sign(Q)
            # Use binary search to find r_max
            low, high = -10, 10
            while sign_change(low) == sign_change(high):
                low *= 2
                high *= 2
            r_max = 0
            while high - low > 1e-8:
                mid = (low + high) / 2
                if sign_change(mid) != sign_change(high):
                    r_max = mid
                    low = mid
                else:
                    high = mid
            if r_max > 0:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            if score < best_score:
                best_score = score
                best_coeff = c[:3]
            return score
        except:
            return float('inf')
    
    # Generate structured initial guesses with rational ratios
    initial_guesses = []
    for i in range(1, 5):
        for j in range(1, 5):
            initial_guesses.append(np.array([float(i)/float(j), 1.0, 1.0]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2)] * 3,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff[:3]
