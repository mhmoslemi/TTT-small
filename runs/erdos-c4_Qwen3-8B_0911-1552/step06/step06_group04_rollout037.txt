import numpy as np
from scipy.optimize import differential_evolution
import time
from sympy import symbols, hermite, Poly, solve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def objective(c):
        try:
            # Construct the polynomial
            z = symbols('z')
            H = [hermite(i, z) for i in range(len(c))]
            Q = sum(c[i] * H[i] for i in range(len(c)))
            Q = Q - Q.as_coefficients_dict()[z**2] * z**2
            
            # Find real roots
            roots = solve(Q, z, domain='Real')
            real_roots = [r.evalf() for r in roots if r.is_real]
            
            # Find r_max
            r_max = 0
            for r in real_roots:
                if r > 0 and Q.subs(z, r).evalf() * Q.subs(z, r + 1e-6).evalf() < 0:
                    r_max = max(r_max, r)
            
            if r_max == 0:
                return float('inf')
            
            return r_max**2 / (2 * np.pi)
        except:
            return float('inf')
    
    # Initial grid search with 3 coefficients
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    
    # Generate initial guesses
    initial_guesses = []
    for i in range(-2, 3):
        for j in range(-2, 3):
            for k in range(-2, 3):
                initial_guesses.append(np.array([i, j, k]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = differential_evolution(
            objective,
            bounds=[(-2, 2), (-2, 2), (-2, 2)],
            strategy='best1bin',
            popsize=20,
            maxiter=50,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.9,
            polish=True,
            disp=False
        )
        if result.fun < best_score:
            best_score = result.fun
            best_coeff = result.x
    
    return best_coeff
