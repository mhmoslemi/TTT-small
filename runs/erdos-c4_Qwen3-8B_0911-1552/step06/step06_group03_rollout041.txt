import numpy as np
from scipy.optimize import differential_evolution
from sympy import symbols, hermite, simplify, Eq, diff, solve, roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Fast numerical surrogate for initial screening
            c = np.array(c, dtype=float)
            m = len(c)
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(m))
            Q = Q - Q.as_leading_coefficient()*z**2
            r = roots(Q, z, domain='Real')
            r = np.sort([float(rk) for rk in r if rk.is_real and rk > 0])
            if len(r) == 0:
                return float('inf')
            r_max = r[-1]
            score = r_max**2 / (2*np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Structured grid search with symbolic verification
    initial_guesses = []
    for n in range(1, 5):
        for ratio in np.logspace(-2, 2, 5):
            ch = np.array([ratio**k / np.sqrt(2**k) for k in range(n)])
            initial_guesses.append(ch)
            initial_guesses.append(-ch)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = differential_evolution(
            objective,
            bounds=[(-2, 2) for _ in range(len(guess))],
            strategy='best1bin',
            popsize=20,
            maxiter=30,
            tol=1e-10,
            mutation=(0.5, 1),
            recombination=0.9
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
