from scipy.optimize import minimize
from sympy import symbols, hermite, simplify, Eq, diff, solve, N

import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def objective(c):
        m = len(c)
        z = symbols('z')

        return objective(c)
