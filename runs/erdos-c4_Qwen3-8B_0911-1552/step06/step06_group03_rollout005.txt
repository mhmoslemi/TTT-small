import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, hermite, solve, simplify, Eq, diff
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Define symbolic variables
    z = symbols('z')
    
    def evaluate_c4_coefficients(c):
        m = len(c)
        Q = sum(c[k] * hermite(4*k, z) for k in range(m))
        # Enforce Q(0) = 0
        Q = Q - Q.subs(z, 0)
        # Enforce positivity at infinity
        leading_term = Q.as_leading_term(z)
        if leading_term.as_coeff_mul()[0] < 0:
            Q = -Q
        # Remove z^2 factor
        Q = Q / z**2
        # Find real roots
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = sorted(r for r in real_roots if r > 0)
        # Check for sign change
        r_max = None
        for i in range(len(real_roots) - 1):
            if Q.subs(z, real_roots[i] + 1e-5) * Q.subs(z, real_roots[i] - 1e-5) < 0:
                r_max = real_roots[i]
                break
        if r_max is None:
            return float('inf')
        return r_max**2 / (2 * np.pi)
    
    # Initial guess with structured ratios
    initial_guesses = []
    ratios = [0.5, 1.0, 2.0, 3.0, 1/np.sqrt(2), np.sqrt(2), 1/np.sqrt(3), np.sqrt(3)]
    for n in range(1, 5):
        for r in ratios:
            ch = np.array([r**k / np.sqrt(2**k) for k in range(n)])
            initial_guesses.append(ch)
            initial_guesses.append(-ch)
    
    # Bayesian optimization
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial Bayesian search
    bounds = [(-2, 2) for _ in range(8)]
    result = minimize(
        objective,
        x0=np.array([1.0]*8),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'disp': False}
    )
    best_coeff = result.x
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.8:
        result = minimize(
            objective,
            x0=best_coeff,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'disp': False}
        )
        best_coeff = result.x
    
    return best_coeff
