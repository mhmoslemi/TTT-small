import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, diff, hermite, solve, N

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    # Define symbolic variables
    z = symbols('z')
    H = lambda n: hermite(n, z)
    
    def evaluate_coefficients(c):
        m = len(c)
        Q = sum(c[k] * H[4*k](z) for k in range(m))
        Q = Q.subs(z, 0).as_numer_denom()[0]  # Ensure Q(0) = 0
        if Q.as_poly().eval(0) != 0:
            Q += -Q.as_poly().eval(0)
        # Flip sign so Q(z)/z^2 is positive at infinity
        if Q.as_poly().lead_coeff().is_negative:
            Q *= -1
        Q = Q / z**2
        # Find real roots
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [N(r) for r in real_roots]
        real_roots = sorted([r for r in real_roots if r > 0])
        # Find r_max
        r_max = 0
        for i in range(len(real_roots) - 1):
            if real_roots[i] * real_roots[i+1] < 0:
                r_max = real_roots[i]
                break
        if r_max == 0:
            r_max = real_roots[-1] if real_roots else 0
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('0.5')
    
    # Grid search with more points and refinement
    grid_points = 15
    grid = np.linspace(-2, 2, grid_points)
    for i in range(grid_points):
        for j in range(grid_points):
            guess = np.array([grid[i], grid[j]])
            if time.time() - start_time > budget_s * 0.8:
                break
            result = minimize(
                objective,
                guess,
                method='L-BFGS-B',
                bounds=[(-2, 2), (-2, 2)],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    return best_coeff
