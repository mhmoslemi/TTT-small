import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, solve, Eq, diff, roots, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic computation for validation
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Make Q positive at infinity
            if Q.as_coefficients_dict().get(z**np.inf, 0) < 0:
                Q = -Q
            # Remove z² factor
            Q = Q / z**2
            # Find r_max
            roots_val = np.real(roots(Q, domain='Real'))
            r_max = 0
            for root in roots_val:
                if root > 0 and np.isclose(Q.subs(z, root), 0):
                    r_max = max(r_max, root)
            if r_max == 0:
                return float('inf')
            score = r_max**2 / (2*np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    def generate_initial_guesses(n_coeff):
        guesses = []
        for i in range(n_coeff):
            guess = np.zeros(n_coeff)
            guess[i] = 1.0
            guesses.append(guess)
        for i in range(n_coeff):
            guess = np.zeros(n_coeff)
            guess[i] = np.random.uniform(-2, 2)
            guesses.append(guess)
        return guesses
    
    n_coeff = 4
    initial_guesses = generate_initial_guesses(n_coeff)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(n_coeff)],
            options={'maxiter': 200, 'eps': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
