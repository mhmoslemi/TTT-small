import numpy as np
import time
from scipy.optimize import minimize, Bounds
from sympy import symbols, Hermite, simplify, roots, sqrt, pi

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate(c):
        try:
            m = len(c)
            z = symbols('z')
            Q = sum(c[k] * Hermite(4*k, z) for k in range(m))
            Q = Q.subs(z, 0)
            if Q != 0:
                Q += -Q
            Q = simplify(Q)
            Q = Q.as_poly().all_coeffs()[::-1]
            Q = Q[0] * z**2 + Q[1] * z + Q[2]
            Q = Q / z**2
            roots = roots(Q, z)
            real_roots = [r for r in roots.keys() if r.is_real]
            real_roots = sorted([float(r) for r in real_roots if r > 0])
            if not real_roots:
                return float('inf')
            r_max = real_roots[-1]
            return r_max**2 / (2 * pi)
        except:
            return float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        score = evaluate(c)
        if score < best_score:
            best_score = score
            best_coeff = c.copy()
        return score
    
    bounds = [(-2, 2) for _ in range(8)]
    
    # Initial structured grid search
    for i in range(1, 10):
        for j in range(1, 10):
            for k in range(1, 10):
                if time.time() - start_time > budget_s * 0.7:
                    break
                c = np.array([i/10, j/10, k/10])
                result = minimize(
                    objective,
                    c,
                    method='L-BFGS-B',
                    bounds=bounds[:len(c)],
                    options={'maxiter': 100}
                )
    
    # Final refinement
    result = minimize(
        objective,
        best_coeff,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'ft': 1e-10}
    )
    
    return best_coeff
