from sklearn.gaussian_process.kernels import RBF, ConstantKernel

        return np.array(np.meshgrid(*[grid]*n_vars)).reshape(n_vars, -1).T
    
    # Initial grid search
    n_vars = 8
    n_points = 5
    points = generate_grid(n_vars, n_points)

    kernel = ConstantKernel(1.0, (1e-5, 1e5)) * RBF(1.0, (0.1, 10.0))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
    gp.fit(points, values)
    
    best_coeff = points[np.argmin(values)]
    best_score = np.min(values)
    
    # Bayesian optimization
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Predictive mean and variance
        mu, sigma = gp.predict(points, return_std=True)
        # Acquisition function: Expected Improvement
        best = mu.max()
        improvement = mu - best
        alpha = 1.0 / (sigma + 1e-8)
        acq = np.exp(-0.5 * (improvement / sigma)**2) * improvement * alpha
        
        # Select next point
        next_point = points[np.argmax(acq)]

        score = evaluate_c4_coefficients(next_point)

            best_score = score
            best_coeff = next_point
        
        # Update GP
        gp.fit(np.vstack([points, next_point]), np.concatenate([values, [score]]))
    
    return best_coeff
