import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial structured grid search with 8 coefficients
    grid_points = 5
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    for m in range(grid_points):
                        for n in range(grid_points):
                            for o in range(grid_points):
                                for p in range(grid_points):
                                    guess = np.array([1.0 + i/grid_points, 1.0 + j/grid_points, 1.0 + k/grid_points, 
                                                     1.0 + l/grid_points, 1.0 + m/grid_points, 1.0 + n/grid_points, 
                                                     1.0 + o/grid_points, 1.0 + p/grid_points])
                                    if time.time() - start_time > budget_s * 0.7:
                                        break
                                    result = minimize(
                                        objective,
                                        guess,
                                        method='Nelder-Mead',
                                        bounds=[(-2, 2)]*8,
                                        options={'xatol': 1e-8, 'ftol': 1e-8}
                                    )
                                    if 'success' in result and result.success:
                                        best_coeff = result.x
    
    # Local search with L-BFGS-B
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*8,
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
