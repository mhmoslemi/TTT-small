from scipy.optimize.minimize import minimize
from sympy import symbols, Hermite, simplify, Eq, solve
import time

            Q = Q / z**2

            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]

            r_max = None
            for r in real_roots:
                if r > 0:

                        break
