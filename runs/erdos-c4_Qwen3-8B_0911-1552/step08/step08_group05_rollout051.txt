import numpy as np
from scipy.optimize import OptimizeResult
import time
from sympy import symbols, hermite, expand, solve, nsolve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to rational coefficients
            c_rational = np.array([float(x) for x in c])
            z = symbols('z')
            Q = sum(c_rational[k] * hermite(4*k, z) for k in range(8))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positive at infinity
            if Q.as_coefficients_dict().get(z**np.inf, 0) < 0:
                Q = -Q
                
            # Remove z^2 factor
            Q = Q.as_poly().removefactor()
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            
            # Find r_max
            real_roots = sorted([float(r) for r in real_roots if r > 0])
            if not real_roots:
                return float('inf')
            r_max = real_roots[-1]
            
            # Compute U(c)
            U = r_max**2 / (2*np.pi)
            
            if U < best_score:
                best_score = U
                best_coeff = c.copy()
            return U
        except:
            return float('inf')
    
    # Global search with simulated annealing
    initial_guess = np.random.uniform(-2, 2, 8)
    result = OptimizeResult(x=initial_guess, fun=objective(initial_guess))
    best_coeff = result.x
    
    # Refinement with Newton's method
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = OptimizeResult(x=best_coeff, fun=objective(best_coeff))
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
