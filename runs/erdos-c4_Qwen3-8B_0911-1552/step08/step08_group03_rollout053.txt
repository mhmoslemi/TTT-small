import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, expand, hermite, solve, apart

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    m = 4  # Number of free coefficients

    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(m))
            Q = Q.as_expr()
            
            # Ensure Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positivity at infinity by flipping sign if needed
            leading_term = Q.as_poly().leaders[0]
            if leading_term.coeffs()[0] < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_poly()
            if Q.LC() < 0:
                Q = -Q
            Q = Q.as_poly().div(z**2)[1]
            
            # Find real roots
            roots = solve(Q, z, domain='Real')
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted(real_roots)
            
            # Find r_max
            r_max = 0
            for i in range(len(real_roots) - 1):
                if real_roots[i] > 0 and real_roots[i+1] > 0:
                    r_max = max(r_max, real_roots[i])
            
            if r_max > 0:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial structured grid search
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*m,
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*m,
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
