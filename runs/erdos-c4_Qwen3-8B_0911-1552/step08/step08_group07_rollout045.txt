import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, simplify, solve, Eq

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic calculation
            z = symbols('z')
            H = [hermite(i, z) for i in range(4*len(c)+1)]
            Q = sum(c[i] * H[4*i] for i in range(len(c)))
            
            # Enforce Q(0) = 0
            Q0 = Q.subs(z, 0)
            if Q0 != 0:
                Q = Q - Q0
            
            # Ensure positivity at infinity
            if Q.as_infinite_limit(z) <= 0:
                Q = -Q
            
            # Remove z² factor
            Q = Q.as_poly().remove_terms()
            Q = Q.as_poly().div(z**2)[1]
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            
            # Find r_max
            r_max = None
            for i in range(len(real_roots) - 1):
                if real_roots[i] * real_roots[i+1] < 0:
                    r_max = real_roots[i]
                    break
            
            if r_max is None:
                r_max = real_roots[-1] if real_roots else 1.0
            
            score = r_max**2 / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            
            return score
        except:
            return float('inf')
    
    # Initial grid search with adaptive refinement
    grid_points = 10
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    result = minimize(
                        objective,
                        guess,
                        method='L-BFGS-B',
                        bounds=[(-2, 2) for _ in range(len(guess))],
                        options={'eps': 1e-8, 'maxiter': 100}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Local search with gradient-based optimization

        if time.time() - start_time > budget_s * 0.9:
            break

        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(len(best_coeff))],
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
