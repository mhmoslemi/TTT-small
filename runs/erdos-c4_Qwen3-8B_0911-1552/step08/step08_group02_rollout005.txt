import numpy as np
from scipy.optimize import minimize, Bounds
import time
from sympy import symbols, HermiteH, solve, S
from functools import lru_cache

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_with_symbolics(c):
        z = symbols('z')
        Q = sum(c[i] * HermiteH(4*i, z) for i in range(len(c)))
        Q = Q - Q.as_leading_term(z**2)
        roots = solve(Q, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        r_max = max(real_roots) if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_with_symbolics(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    def random_search():
        for _ in range(100):
            if time.time() - start_time > budget_s * 0.8:
                break
            c = np.random.uniform(-2, 2, size=8)
            result = minimize(
                objective,
                c,
                method='L-BFGS-B',
                bounds=[(-2, 2) for _ in range(8)],
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    random_search()
    
    return best_coeff
