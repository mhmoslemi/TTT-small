from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, simplify, Poly, solve, Eq
from sympy.polys.polytools import factor

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.array([1.0]*8)
    best_score = float('inf')
    
    # Symbolic variables
    z = symbols('z')
    H = [HermiteH(k, z) for k in range(8)]

    def evaluate_c4_coefficients(c):
        poly = sum(c[k] * H[k] for k in range(8))

            return float('inf')

        leading_coeff = poly.as_poly().all_coeffs()[-1]
        if leading_coeff < 0:
            poly = -poly
        # Remove z^2 factor
        poly = poly.as_poly().div(z**2)[1]
        # Find real roots
        roots = solve(poly, z, domain='Real')
        real_roots = [r for r in roots if r.is_real]
        # Find r_max
        if not real_roots:
            return float('inf')
        real_roots = sorted(real_roots)
        r_max = real_roots[-1] if real_roots[-1] > 0 else real_roots[-2]
        # Return U(c) = r_max^2 / (2π)
        return (r_max**2) / (2 * np.pi)

    grid_points = 10
