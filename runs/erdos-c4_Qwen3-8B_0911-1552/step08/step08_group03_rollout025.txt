import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, HermiteH, roots, S

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    z = symbols('z')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            m = len(c)
            Q = sum(c[k] * HermiteH(4*k, z) for k in range(m))
            Q = Q - Q.as_coefficients_dict()[HermiteH(0, z)] * HermiteH(0, z)
            Q = Q.as_poly().as_expr()
            roots_list = roots(Q, z, domain=S.Reals)
            r_max = 0
            for root in roots_list:
                if root.is_real and root > 0:
                    if roots_list[root].is_odd:
                        r_max = max(r_max, root)
            if r_max == 0:
                return float('inf')
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search with 8 coefficients
    grid_points = 5
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    if time.time() - start_time > budget_s * 0.7:
                        break
                    result = minimize(
                        objective,
                        guess,
                        method='Nelder-Mead',
                        bounds=[(-2, 2)]*len(guess),
                        options={'xatol': 1e-8, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    # Local search with adaptive refinement
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            best_coeff,
            method='L-BFGS-B',
            bounds=[(-2, 2)]*len(best_coeff),
            options={'eps': 1e-8, 'maxiter': 100}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
