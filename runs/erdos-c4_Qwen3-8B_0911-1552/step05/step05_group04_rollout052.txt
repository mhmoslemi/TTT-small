import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, Hermite, roots, simplify, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Use exact rational arithmetic for validation
            z = symbols('z')
            coeffs = [Rational(int(c[i])) for i in range(len(c))]
            Q = sum(coeffs[k] * Hermite(4*k, z) for k in range(len(c)))
            Q = Q.subs(z, 0)
            Q = Q.as_numer_denom()[0]
            Q = Q.as_poly()
            if Q.LC() < 0:
                Q = -Q
            Q = Q.as_poly()
            roots_expr = roots(Q, z)
            r_max = None
            for root in roots_expr:
                if root.is_real and root > 0:
                    if r_max is None or root > r_max:
                        r_max = root
            if r_max is None:
                return float('inf')
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Generate structured initial guesses with symmetry
    initial_guesses = []
    for a in np.linspace(-2, 2, 5):
        for b in np.linspace(-2, 2, 5):
            initial_guesses.append(np.array([a, b]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2), (-2, 2)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
