import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, solve, simplify, Eq, diff, N, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    m = 4
    
    def construct_Q(c):
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        Q = Q - Q.as_coefficients_dict().get(z**2, 0)*z**2
        return Q
    
    def find_r_max(Q):
        z = symbols('z')
        expr = Q.subs(z, z)
        roots = solve(expr, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = [float(r) for r in real_roots]
        real_roots = sorted([r for r in real_roots if r > 0])
        
        def sign_change(start, end):
            val_start = Q.subs(z, start).evalf()
            val_end = Q.subs(z, end).evalf()
            return np.sign(val_start) != np.sign(val_end)
        
        for i in range(len(real_roots)-1):
            if sign_change(real_roots[i], real_roots[i+1]):
                return real_roots[i+1]
        return float('inf')
    
    def evaluate_c4_coefficients(c):
        Q = construct_Q(c)
        r_max = find_r_max(Q)
        if r_max == float('inf'):
            return float('inf')
        return r_max**2 / (2*np.pi)
    
    def objective(c):
        try:
            score = evaluate_c4_coefficients(c)
            return score
        except:
            return float('inf')
    
    def bounded_objective(c):
        return objective(c)

        nonlocal best_coeff, best_score

            for c1 in np.linspace(-2, 2, 5):

                            best_score = score
                            best_coeff = c.copy()

            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )

            best_coeff = result.x
    
    constrained_optimizer()
    
    return best_coeff
