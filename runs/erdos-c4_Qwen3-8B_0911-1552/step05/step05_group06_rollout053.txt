import numpy as np
from scipy.optimize import minimize, Bounds
from sympy import symbols, hermite, simplify, Eq, diff, solve, rts
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Symbolic construction
            z = symbols('z')
            H = [hermite(k, z) for k in range(len(c))]
            Q = sum(c[k] * H[k] for k in range(len(c)))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Enforce positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z² factor
            Q = Q.as_poly().removefactor()
            
            # Find real roots
            roots = rts(Q.as_poly())
            real_roots = [r for r in roots if r.is_real]
            
            # Get r_max
            if not real_roots:
                return float('inf')
            r_max = max(real_roots)
            
            # Compute upper bound
            score = r_max**2 / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Try different coefficient counts
    for count in range(1, 9):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Initial guesses based on symmetry and orthogonality
        initial_guess = np.random.uniform(-1, 1, count)
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(count)],
            options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
