from scipy.optimize import minimize, differential_evolution
from scipy.special import eval_legendre
import time

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    best_coeff = np.zeros(8)
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:

                best_coeff = c.copy()

        except:
            return float('inf')

    initial_guesses = [np.array([eval_legendre(k, 0.5) for k in range(8)])]

            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',

            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    # Bayesian optimization
    if time.time() - start_time < budget_s * 0.9:
        result = differential_evolution(
            objective,
            bounds=[(-2, 2) for _ in range(8)],
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9
        )
        best_coeff = result.x
    
    return best_coeff
