import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, Rational, NSum, NIntegrate, NRoots
from sympy.polys import Poly

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([Rational(1), Rational(1), Rational(1)])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to sympy Rational
            sympy_c = [Rational(int(round(x))) for x in c]
            # Construct polynomial
            z = symbols('z')
            Q = sum(c_k * hermite(4*k, z) for k, c_k in enumerate(sympy_c))
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            # Ensure positivity at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            # Remove z^2 factor
            Q = Q / z**2
            # Find real roots
            roots = NRoots(Q)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([r for r in real_roots if r > 0])
            # Find r_max
            r_max = None
            for i in range(len(real_roots)-1):
                if Q.subs(z, real_roots[i]) * Q.subs(z, real_roots[i+1]) < 0:
                    r_max = real_roots[i]
                    break
            if r_max is not None:
                score = r_max**2 / (2 * np.pi)
            else:
                score = float('inf')
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Parameterized family with symmetry
    def parameterize(t):
        return np.array([Rational(1 + t), Rational(1 - t), Rational(1)])
    
    # Grid search
    t_values = np.linspace(-1, 1, 50)
    for t in t_values:
        if time.time() - start_time > budget_s * 0.7:
            break
        c = parameterize(t)
        result = minimize(
            objective,
            c,
            method='L-BFGS-B',
            bounds=[(-2, 2), (-2, 2), (-2, 2)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
