import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve, Eq, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to exact rational representation
            c = np.array(c, dtype=np.float64)
            c = np.round(c, 10)
            c_rat = [Rational(int(round(x * 10**10))) for x in c]
            
            # Construct Hermite polynomial
            z = symbols('z')
            Q = sum(c_rat[k] * hermite(4*k, z) for k in range(len(c_rat)))
            
            # Enforce Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_poly().as_ordered_terms()[0]
            Q = Q / z**2
            
            # Find real roots
            roots = solve(Q, z, domain='Real')
            real_roots = [float(r) for r in roots if r.is_real]
            
            # Find r_max
            real_roots.sort()
            r_max = None
            for r in real_roots:
                if r > 0:
                    r_max = r
                    break
            
            if r_max is None:
                return float('inf')
            
            score = r_max**2 / (2 * np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Generate structured initial guesses
    initial_guesses = []
    for i in range(1, 4):
        for j in range(1, 4):
            initial_guesses.append(np.array([float(i), float(j)]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2), (-2, 2)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
