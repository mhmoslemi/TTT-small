import numpy as np
from scipy.optimize import minimize
from sympy import symbols, HermiteH, simplify, Poly, roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Convert to symbolic variables
            z = symbols('z')
            poly = sum(c[k] * HermiteH(4*k, z) for k in range(len(c)))
            
            # Enforce Q(0) = 0
            poly = poly - poly.subs(z, 0)
            
            # Find r_max
            try:
                roots_poly = roots(poly / z**2, z)
                real_roots = [r.evalf() for r in roots_poly if r.is_real]
                real_roots = sorted([r for r in real_roots if r > 0])
                if not real_roots:
                    return float('inf')
                r_max = real_roots[-1]
                score = r_max**2 / (2 * np.pi)
                if score < best_score:
                    best_score = score
                    best_coeff = c.copy()
                return score
            except:
                return float('inf')
        except:
            return float('inf')
    
    # Diverse initial guesses
    initial_guesses = []
    for i in range(5):
        for j in range(5):
            initial_guesses.append(np.array([1.0 + i - 2, 1.0 + j - 2]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.9:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-3, 3) for _ in range(len(guess))],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
