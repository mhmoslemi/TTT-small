<strategy>Use a grid search over a small number of coefficients with exact rational arithmetic to find promising candidates, then refine with optimization to find the best verified upper bound on C₄.</strategy>

import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, Poly, solve, Eq, simplify, diff, Rational
from sympy.polys.polytools import factor, roots
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_c4_coefficients(coeffs):
        z = symbols('z')
        Q = sum(coeffs[k] * hermite(4*k, z) for k in range(len(coeffs)))
        Q = Q - Q.as_independent(z)[1].as_coefficients_dict()[z**2] * z**2
        roots = solve(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r != 0])
        for r in real_roots:
            if r > 0:
                r_max = r
                break
        else:
            r_max = None
        if r_max is None:
            return float('inf')
        return (r_max**2) / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Try with 4 coefficients using a grid search
    grid_points = np.linspace(-2, 2, 100)
    best_coeffs = None
    best_score = float('inf')
    
    for c0 in grid_points:
        for c1 in grid_points:
            for c2 in grid_points:
                for c3 in grid_points:
                    if time.time() - start_time > budget_s * 0.8:
                        break
                    coeffs = np.array([c0, c1, c2, c3])
                    score = evaluate_c4_coefficients(coeffs)
                    if score < best_score:
                        best_score = score
                        best_coeffs = coeffs
    
    if best_coeffs is not None:
        return best_coeffs
    else:
        # Fallback to optimization with 4 coefficients
        initial_guess = np.array([1.0, 1.0, 1.0, 1.0])
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(4)],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if 'success' in result and result.success:
            return result.x
        else:
            return np.array([1.0, 1.0, 1.0, 1.0])
