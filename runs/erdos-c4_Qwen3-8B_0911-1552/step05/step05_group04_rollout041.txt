import numpy as np
from skopt import BayesSearchCV
from skopt.utils import use_text
from skopt.space import Real
from skopt callback import VerboseCallback
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    param_space = [Real(-2, 2, prior='log-uniform') for _ in range(8)]
    
    def evaluate_c4_coefficients(c):
        return evaluate_c4_coefficients(c)
    
    opt = BayesSearchCV(
        lambda x: evaluate_c4_coefficients(x),
        search_space=param_space,
        n_iter=50,
        n_jobs=1,
        cv=1,
        scoring="neg_mean_squared_error",
        random_state=seed,
        verbose=0
    )
    
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
        opt.fit(np.random.uniform(-2, 2, size=(1, 8)))
    
    best = opt.best_params_
    return np.array([best[f"real_{i}"] for i in range(8)])
