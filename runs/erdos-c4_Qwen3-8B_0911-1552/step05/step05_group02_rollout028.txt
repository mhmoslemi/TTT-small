import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, Hermite, simplify, solve, real_roots, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    best_time = 0

    def objective(c):
        nonlocal best_score, best_coeff, best_time
        if time.time() - start_time > budget_s - 10:
            return float('inf')
        try:
            # Symbolic construction
            z = symbols('z')
            H = [Hermite(n, z) for n in range(len(c))]
            Q = sum(c[i] * H[i] for i in range(len(c)))
            
            # Force Q(0) = 0
            Q = Q - Q.subs(z, 0)
            
            # Ensure positive at infinity
            if Q.as_leading_coefficient() < 0:
                Q = -Q
            
            # Remove z^2 factor
            Q = Q.as_poly().div(z**2)[1]
            
            # Find real roots
            roots = real_roots(Q)
            r_max = None
            for r in roots:
                if r > 0:
                    r_max = max(r_max, r) if r_max is not None else r
            
            if r_max is None:
                return float('inf')
            
            # Compute upper bound
            score = (r_max**2) / (2 * np.pi)
            
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
                best_time = time.time() - start_time
            return score
        except:
            return float('inf')
    
    # Initial grid search with 4 coefficients
    grid_points = 5
    for i in range(grid_points):
        for j in range(grid_points):
            for k in range(grid_points):
                for l in range(grid_points):
                    guess = np.array([1.0 + i / grid_points, 1.0 + j / grid_points, 1.0 + k / grid_points, 1.0 + l / grid_points])
                    if time.time() - start_time > budget_s - 10:
                        break
                    result = minimize(
                        objective,
                        guess,
                        method='L-BFGS-B',
                        bounds=[(-2, 2) for _ in range(len(guess))],
                        options={'maxiter': 50, 'ftol': 1e-8}
                    )
                    if 'success' in result and result.success:
                        best_coeff = result.x
    
    return best_coeff
