import numpy as np
from scipy.optimize import minimize
from sympy import symbols, hermite, simplify, Eq, diff, solve, Rational
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        if len(c) != 4:
            return float('inf')
        try:
            # Symbolic construction
            z = symbols('z')
            Q = sum(c[k] * hermite(4*k, z) for k in range(4))
            Q = Q.subs(z, 0)
            # Ensure Q(0) = 0 and positive at infinity
            if Q != 0:
                return float('inf')
            # Remove z^2 factor and find r_max
            Q = Q.as_ordered_factors()[0]
            if not Q.has(z):
                return float('inf')
            Q = Q.expand()
            Q = Q / z**2
            # Find largest real root where sign changes
            roots = solve(Q, z)
            real_roots = [r.evalf() for r in roots if r.is_real]
            real_roots = [r for r in real_roots if r > 0]
            real_roots = sorted(real_roots)
            r_max = real_roots[-1] if len(real_roots) > 0 else 1.0
            score = r_max**2 / (2*np.pi)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Initial grid search on log scale
    log_grid = np.logspace(-4, 4, 20)
    for c0 in log_grid:
        for c1 in log_grid:
            for c2 in log_grid:
                for c3 in log_grid:
                    c = np.array([c0, c1, c2, c3])
                    if time.time() - start_time > budget_s * 0.6:
                        break
                    score = objective(c)
                    if score < best_score:
                        best_coeff = c.copy()
    
    # Refinement with Nelder-Mead
    if time.time() - start_time < budget_s * 0.8:
        result = minimize(
            objective,
            best_coeff,
            method='Nelder-Mead',
            bounds=[(-2, 2) for _ in range(4)],
            options={'maxiter': 50}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
