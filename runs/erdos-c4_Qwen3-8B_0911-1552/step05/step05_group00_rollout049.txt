import numpy as np
from scipy.optimize import minimize
from scipy.special import hermite
from sympy import symbols, Eq, solve, Rational, simplify
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 0.5])
    best_score = float('inf')
    
    def evaluate_c4_coefficients(c):
        m = len(c)
        z = symbols('z')
        poly = sum(c[k] * hermite(4*k, z) for k in range(m))
        poly = poly.subs(z, 0)
        if poly != 0:
            poly = poly - poly.as_numer_denom()[0]
        poly = poly / z**2
        roots = solve(poly, z)
        real_roots = [r.evalf() for r in roots if r.is_real]
        real_roots = [r for r in real_roots if r > 0]
        r_max = max(real_roots) if real_roots else 1.0
        return r_max**2 / (2 * np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Use fewer coefficients with tighter bounds
    for m in range(1, 5):
        if time.time() - start_time > budget_s * 0.9:
            break
        for _ in range(50):
            if time.time() - start_time > budget_s * 0.9:
                break
            c = np.random.uniform(-1, 1, m)
            if np.sign(evaluate_c4_coefficients(c)) < 0:
                c *= -1
            # Optimize with Powell's method for better precision
            result = minimize(
                objective,
                c,
                method='Powell',
                bounds=[(-1.5, 1.5) for _ in range(m)],
                options={'ftol': 1e-10, 'maxiter': 100}
            )
            if 'success' in result and result.success:
                best_coeff = result.x
    
    return best_coeff
