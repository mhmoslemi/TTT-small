from scipy.optimize import minimize
from sympy import symbols, hermite, solve, Eq, diff, sqrt, Rational, N
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            # Construct polynomial
            z = symbols('z')
            H = [hermite(k, z) for k in range(len(c))]
            Q = sum(c[k] * H[k] for k in range(len(c)))
            
            # Enforce Q(0) = 0 and adjust sign
            const = Q.as_coefficients_dict().get(z**0, 0)
            if const != 0:
                Q = Q - const
                
            # Ensure positivity at infinity
            leading_term = Q.as_poly().leading_term()
            if leading_term.coeff != 1:
                Q = Q / leading_term.coeff
                
            # Remove z^2 factor
            Q = Q.as_poly().remove_order(2)
            
            # Find real roots
            roots = solve(Q, z)
            real_roots = [r for r in roots if r.is_real]
            real_roots = sorted([float(r) for r in real_roots])
            
            # Identify sign change across largest positive root
            r_max = None
            for i in range(len(real_roots)-1):
                if real_roots[i] < 0 and real_roots[i+1] > 0:
                    r_max = real_roots[i+1]
                    break
            
            if r_max is None:
                return float('inf')
            
            # Compute upper bound
            score = r_max**2 / (2 * np.pi)
