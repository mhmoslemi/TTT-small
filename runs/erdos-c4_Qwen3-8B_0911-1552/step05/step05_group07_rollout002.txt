import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = None
    best_score = float('inf')
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Use a grid-based search with random perturbations for better coverage
    points = np.linspace(-2, 2, 50)
    candidates = []
    for i in range(len(points)):
        for j in range(len(points)):
            for k in range(len(points)):
                if i != j or j != k:
                    candidates.append([points[i], points[j], points[k]])
    
    # Add some random perturbations
    for i in range(len(candidates)):
        if np.random.rand() < 0.3:
            perturb = np.random.uniform(-0.5, 0.5, 3)
            candidates[i] = [c + p for c, p in zip(candidates[i], perturb)]
    
    # Evaluate candidates
    for guess in candidates:
        if time.time() - start_time > budget_s * 0.8:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(len(guess))],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
