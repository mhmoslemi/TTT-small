from sympy import symbols, Hermite, solve, Eq, sqrt, simplify
from fractions import Fraction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    m_values = [2, 3, 4]
    best_score = float('inf')
    best_coeff = None
    
    for m in m_values:
        for c0 in [Fraction(1), Fraction(2), Fraction(1/2)]:
            for c1 in [Fraction(1), Fraction(2), Fraction(1/2)]:
                if m == 2:
                    c = [float(c0), float(c1)]
                else:
                    c = [float(c0)] + [float(c1)] + [float(Fraction(1))]

                Q = sum(c[k] * Hermite(4*k, z) for k in range(len(c)))
                Q = Q - Q.subs(z, 0)
                Q = Q / z**2

                real_roots = [r.evalf() for r in roots if r.is_real]
                real_roots = sorted([r for r in real_roots if r > 0])
                
                if real_roots:
                    r_max = real_roots[-1]
                    score = r_max**2 / (2*np.pi)
                
                if score < best_score:
                    best_score = score
                    best_coeff = c
    
    return best_coeff
