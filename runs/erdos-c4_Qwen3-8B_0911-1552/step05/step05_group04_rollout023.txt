import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, hermite, simplify, solve, roots, Rational
from sympy.polys.polytools import resultant

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0, 1.0, 1.0])
    best_score = float('inf')
    
    def evaluate_c4_coefficients(c):
        from sympy import symbols, hermite, simplify, solve, roots, Rational
        z = symbols('z')
        Q = sum(c[k] * hermite(4*k, z) for k in range(len(c)))
        Q = Q.subs(z, 0)
        Q = Q.as_coefficients()
        Q = Q.as_poly()
        if Q.lterms == 0:
            return float('inf')
        Q = Q * z**2
        roots_list = roots(Q, z, domain='R')
        real_roots = [r for r in roots_list if r.is_real]
        real_roots = sorted(real_roots)
        r_max = None
        for r in real_roots:
            if r > 0:
                r_max = r
                break
        if r_max is None:
            return float('inf')
        return (r_max**2) / (2*np.pi)
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_c4_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    # Generate structured initial guesses with symmetry
    initial_guesses = []
    for a in [0.5, 1.0, 1.5]:
        for b in [0.5, 1.0, 1.5]:
            for c in [0.5, 1.0, 1.5]:
                for d in [0.5, 1.0, 1.5]:
                    initial_guesses.append(np.array([a, b, c, d]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2) for _ in range(len(guess))],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
