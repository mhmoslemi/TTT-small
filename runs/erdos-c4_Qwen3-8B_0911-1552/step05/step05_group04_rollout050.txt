import numpy as np
from scipy.optimize import minimize
import time
from sympy import symbols, Hermite, simplify, roots, Rational

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_coeff = np.array([1.0, 1.0])
    best_score = float('inf')
    max_iter = 50
    
    def evaluate_coefficients(c):
        m = len(c)
        z = symbols('z')
        Q = sum(c[k] * Hermite(4*k, z) for k in range(m))
        Q = Q.subs(z, 0)
        Q = Q.as_independent(z)[1]
        Q = Q.as_poly()
        Q = Q.remove_primes()
        Q = Q.as_poly()
        if Q.lterms() == 0:
            return float('inf')
        roots = roots(Q, z)
        real_roots = [r for r in roots if r.is_real]
        real_roots = sorted([r for r in real_roots if r > 0])
        if not real_roots:
            return float('inf')
        r_max = real_roots[-1]
        score = r_max**2 / (2 * np.pi)
        return score
    
    def objective(c):
        nonlocal best_score, best_coeff
        try:
            score = evaluate_coefficients(c)
            if score < best_score:
                best_score = score
                best_coeff = c.copy()
            return score
        except:
            return float('inf')
    
    initial_guesses = []
    for a in [0.5, 1, 2]:
        for b in [0.5, 1, 2]:
            initial_guesses.append(np.array([a, b]))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s * 0.7:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(-2, 2), (-2, 2)],
            options={'maxiter': max_iter, 'ftol': 1e-10}
        )
        if 'success' in result and result.success:
            best_coeff = result.x
    
    return best_coeff
