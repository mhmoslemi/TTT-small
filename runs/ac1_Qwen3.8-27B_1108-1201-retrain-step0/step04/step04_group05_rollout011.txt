`df/da = (1/beta) * sum_k exp(beta*(c[k]-c_max)) / sum_j exp(beta*(c[j]-c_max)) * dc[k]/da`
`= sum_k w[k] * dc[k]/da`

where `w[k] = exp(beta*(c[k]-c_max)) / sum_j exp(beta*(c[j]-c_max))`.

And I showed that `dc[k]/da[j] = 2*a[k-j]` (for valid indices).

So `g[j] = 2 * sum_k w[k] * a[k-j]` (summed over valid k).

Now, in the FFT domain, if I define `w_full` as the array of length 2n-1 with `w_full[k] = w[k]`, and `a_pad` as `a` zero-padded to length N:

`g[j] = 2 * sum_{k=j}^{j+n-1} w_full[k] * a[k-j]`

`g[j] = 2 * sum_{m=0}^{n-1} w_full[j+m] * a[m]`

This is `2 * (w_full ⊛ a)[j]` where ⊛ denotes cross-correlation.

So `(w_full ⊛ a)[j] = ifft(conj(fft(w_full_pad)) * fft(a_pad))[j]`

"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE subgradient + spectral shaping.
Combines smooth gradient methods with frequency-domain peak reduction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c of length 2n-1, A of length N)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_weights(c, beta):
    """Compute LSE weights from convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[c.argmax()] = 1.0
        return weights, c_max
    weights = exp_bc / total
    return weights, c_max


def lse_gradient_from_weights(w, a, N):
    """Compute gradient of weighted sum of convolution given weights."""
    n = len(a)
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    W = fft(w_pad)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return g


def spectral_flatten(a, N, alpha, method='median'):
    """Flatten the power spectrum, keeping DC fixed."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    
    # Keep DC fixed
    P_dc = P[0]
    P_rest = P[1:]
    
    if method == 'median':
        target = np.median(P_rest)
    elif method == 'mean':
        target = P_rest.mean()
    else:
        target = np.percentile(P_rest, 25)
    
    # Reduce the peaks in the spectrum
    P[1:] = P_rest - alpha * (P_rest - target)
    P[1:] = np.maximum(P[1:], 0)
    P[0] = P_dc
    
    phase = np.angle(A)
    A_new = np.sqrt(P) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def spectral_flatten_aggressive(a, N, alpha, top_fraction=0.1):
    """More aggressive spectral flattening: only reduce the top portion."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    
    P_dc = P[0]
    P_rest = P[1:]
    
    # Only reduce the top |top_fraction| of the rest
    threshold = np.percentile(P_rest, (1 - top_fraction) * 100)
    excess = np.maximum(P_rest - threshold, 0)
    P[1:] = P_rest - alpha * excess
    P[1:] = np.maximum(P[1:], 0)
    P[0] = P_dc
    
    phase = np.angle(A)
    A_new = np.sqrt(P) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def make_init_spectral(n, N, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    # Random phases
    phase = rng.uniform(0, 2 * np.pi, N)
    # Flat magnitude (with DC set appropriately)
    # For a flat spectrum with DC = S, the average of |A[k]|^2 for k>0
    # should be about S^2/N (roughly)
    # Let's just use a geometric distribution for the magnitude
    mag = np.ones(N)
    mag[0] = 1.0  # DC will be set later
    A = mag * np.exp(1j * phase)
    a_full = np.real(ifft(A))
    a = np.abs(a_full[:n]) + 0.01
    return a


def make_init_random(n, rng):
    """Random non-negative initialization."""
    a = rng.random(n) + 0.01
    return a


def make_init_chirp(n, rng):
    """Chirp-based initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt**2)) + 0.1 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def make_init_multi_freq(n, rng):
    """Multi-frequency initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for freq in [2, 5, 11, 17, 23]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.maximum(a, 0.01)

    """Compute the objective from the convolution."""
    return c.max()

    a = project_simplex(a_init, S)
    y = a.copy()

    start_time = time.monotonic()

        it += 1
        
        c, A = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        # Top-k peaks
        tk = min(top_k + it // 100, 512)
        top_idx = np.argsort(c)[::-1][:tk]
        top_vals = c[top_idx]
        
        # Weights proportional to excess
        c_median = np.median(c)
        weights = np.maximum(top_vals - c_median, 0)
        if weights.sum() < 1e-30:
            weights = np.ones(tk) / tk
        else:
            weights = weights / weights.sum()
        
        # Build weight array
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        # Compute gradient
        g = lse_gradient_from_weights(w_full, y, N)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break
        
        # Step size (diminishing)
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * g / gnorm, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y * (S / y_sum)
        a = x_new
        t = t_new
        
        # Check x_new too
        if it % 10 == 0:
            c_x, _ = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
    
    return best_a, best_cmax
