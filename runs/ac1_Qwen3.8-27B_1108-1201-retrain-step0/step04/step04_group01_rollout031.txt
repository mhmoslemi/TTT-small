import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_objective_and_grad(a, beta, S):
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    obj = cm + np.log(tot) / beta
    w = e / tot
    # Gradient: 2 * (w correlated with a)
    # w has length 2n-1, a has length n
    # grad[k] = 2 * sum_j w[j] * a[j-k] (with appropriate bounds)
    # This is cross-correlation of w and a
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad

def fista_backtrack(a_init, beta, S, max_time, rng):

    y = a.copy()
    t = 1.0

    L = 4.0 * beta * y.max() * S  # Conservative Lipschitz estimate

def softmax_objective(x, beta, n, S):
    """Objective for L-BFGS: LSE of convolution of softmax(x) scaled to sum S."""
    # Softmax
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum() * S
    # Compute LSE of convolution
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    obj = cm + np.log(np.exp(scaled).sum()) / beta
    return obj

def softmax_gradient(x, beta, n, S):
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum() * S
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    w = e / e.sum()
    # Gradient of LSE w.r.t. a
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # Chain rule: a = softmax(x) * S
    # da_i/dx_j = S * (delta_ij * softmax_i - softmax_i * softmax_j)
    # = S * softmax_i * (delta_ij - softmax_j)
    # = a_i * (delta_ij - a_j/S)
    # So grad_x_j = sum_i grad_a_i * a_i * (delta_ij - a_j/S)
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i / S)
    dot = np.dot(grad_a, a) / S
    grad_x = a * (grad_a - dot)
    return grad_x

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

Strategy: Log-domain L-BFGS optimization with multi-beta annealing
and perturbation-based escaping from local minima.

Key ideas:
1. Reparameterize a = softmax(x) * S to handle positivity and sum constraint
2. Use LSE-smoothed max as the objective (smooth approximation)
3. Anneal beta from low to high for progressive sharpening
4. Perturb and re-optimize to escape local minima
5. Try multiple sequence lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_and_grad_from_a(a, beta):
    """Compute LSE value and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad_a


def softmax_from_x(x, S):
    """Convert x to a via softmax, scaled to sum S."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum() * S
    return a


def lse_objective_x(x, beta, S):
    """Objective in x-space."""
    a = softmax_from_x(x, S)
    val, _ = lse_and_grad_from_a(a, beta)
    return val


def lse_gradient_x(x, beta, S):
    """Gradient in x-space using chain rule through softmax."""
    n = len(x)
    a = softmax_from_x(x, S)
    _, grad_a = lse_and_grad_from_a(a, beta)
    # da_i/dx_j = a_i * (delta_ij - a_j/S)
    # grad_x_j = sum_i grad_a_i * da_i/dx_j = a_j * (grad_a_j - sum_k grad_a_k * a_k / S)
    dot = np.dot(grad_a, a) / S
    grad_x = a * (grad_a - dot)
    return grad_x


def run_lbfgs(x_init, beta, S, maxiter=200, gtol=1e-8):
    """Run L-BFGS on the x-space objective."""
    result = minimize(
        fun=lambda x: lse_objective_x(x, beta, S),
        x0=x_init,
        jac=lambda x: lse_gradient_x(x, beta, S),
        method='L-BFGS-B',
        options={
            'maxiter': maxiter,
            'gtol': gtol,
            'maxfun': maxiter * 5,
            'ftol': 1e-12
        }
    )
    return result.x, result.fun, result.nit, result.success


def get_objective(a):
    """Compute the actual objective value."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def a_to_x(a, S):
    """Convert a to x-space (log domain)."""
    return np.log(np.maximum(a, 1e-10))
