def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    # Simple iterative projection
    for _ in range(50):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-30:
            return np.full(n, S / n)
        v = v * (S / s)
        if abs(s - S) / max(S, 1e-30) < 1e-12:
            break
    # Ensure exact sum
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-30:

    return v


def project_simplex_fast(v, S):
    """Fast Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def self_convolve(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute objective: 2*n*max(c)/sum(a)^2."""
    n = len(a)
    c = self_convolve(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))

    P_dc = P[0]
    P_nonDC = P[1:]

    # Find threshold at clip_quantile of non-DC bins
    threshold = np.quantile(P_nonDC, clip_quantile)

    # Clip above threshold
    excess = np.maximum(P_nonDC - threshold, 0.0)
    P_clipped = np.minimum(P_nonDC, threshold)

    # Redistribute excess to below-threshold bins proportionally
    below_mask = P_clipped < threshold
    n_below = np.sum(below_mask)
    if n_below > 0:
        total_excess = excess.sum()
        # Distribute proportionally to current values (or uniformly if all zero)
        below_vals = P_clipped[below_mask]
        below_sum = below_vals.sum()
        if below_sum > 1e-30:
            P_clipped[below_mask] += total_excess * below_vals / below_sum
        else:
            P_clipped[below_mask] += total_excess / n_below

    # Apply alpha mixing (soft update)
    P_new_nonDC = (1.0 - alpha) * P_nonDC + alpha * P_clipped
    P_new = np.concatenate([[P_dc], P_new_nonDC])

    # Reconstruct
    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex_fast(a_new_full, S)
    return a_new


def spectral_exponential_compress(a, alpha, floor_ratio=0.1):
    """
    Exponential compression of power spectrum.
    P_new = P^alpha for P > median, P^(1/alpha) for P < median.
    """
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    P_dc = P[0]
    P_nonDC = P[1:]
    p_med = np.median(P_nonDC)

    if p_med < 1e-30:
        return a

    ratio = P_nonDC / p_med
    # Compress: above median use ratio^alpha, below use ratio^(1/alpha)
    # This pushes toward 1 (flat)
    new_ratio = np.where(ratio >= 1.0,
                         np.power(ratio, alpha),
                         np.power(ratio, 1.0 / alpha))
    # Clip to avoid extremes
    new_ratio = np.clip(new_ratio, floor_ratio, 1.0 / floor_ratio)
    P_new_nonDC = p_med * new_ratio

    P_new = np.concatenate([[P_dc], P_new_nonDC])
    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex_fast(a_new_full, S)
    return a_new


def gradient_step(a, top_k=10, step_size=None):
    """
    Subgradient step targeting the top-k peaks of the autocorrelation.
    Returns new a after one step.
    """
    n = len(a)
    S = a.sum()
    c = self_convolve(a)

    # Find top-k peak positions
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    # Weighted subgradient
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]

    # Gradient of sum(w[k]*c[k]) w.r.t. a[i] = 2*(w * a)[i] (approximately)
    # Using FFT: grad = 2 * IFFT(FFT(w) * FFT(a))
    fsize = 1 << ((2 * n - 1) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]

    # Adaptive step size
    if step_size is None:
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            return a
        step_size = 0.1 * S / (n * gnorm)

    a_new = a - step_size * grad
    a_new = project_simplex_fast(a_new, S)
    return a_new


def coordinate_descent_polish(a, max_iters=5000, pair_sample=100):
    """
    Coordinate descent: try swapping small amounts between pairs of coordinates.
    For each pair (i,j), find optimal delta to minimize the peak.
    """
    n = len(a)
    S = a.sum()
    a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(12345)

    # Precompute for efficiency
    fsize = 1 << (2 * n - 2).bit_length()

    for it in range(max_iters):
        # Pick random pairs
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Current contributions to peaks
        c = self_convolve(a)
        c_max = c.max()
        peak_pos = np.argmax(c)

        # Try a range of delta values
        # If we move delta from j to i: a[i]+=delta, a[j]-=delta
        # c[k] changes by: delta*(a[k-i]+a[i-k] - a[k-j]-a[j-k])
        # For the peak position, this is quadratic in delta

        base = a.sum()
        # Compute derivative at peak
        di = a[peak_pos - i] if 0 <= peak_pos - i < n else 0.0
        di2 = a[i - peak_pos] if 0 <= i - peak_pos < n else 0.0
        dj = a[peak_pos - j] if 0 <= peak_pos - j < n else 0.0
        dj2 = a[j - peak_pos] if 0 <= j - peak_pos < n else 0.0

        # dc[peak]/da[i] = di + di2, dc[peak]/da[j] = dj + dj2
        # Moving delta from j to i: dc[peak] = delta*(dj+dj2 - di-di2) + delta^2*(correction)
        grad_i = di + di2
        grad_j = dj + dj2

        # If grad_i < grad_j, moving from i to j reduces peak
        # If grad_j > grad_i, moving from j to i reduces peak
        if grad_j > grad_i:
            # Move from j to i
            max_delta = a[j] * 0.5
            # Try several fractions
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = min(frac * max_delta, 1e-3 * S)
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if a_try[j] < 0:
                    continue
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try * (S / s)
                c_try = self_convolve(a_try)
                if c_try.max() < c_max - 1e-12:
                    a = a_try
                    break
        elif grad_i > grad_j:
            # Move from i to j
            max_delta = a[i] * 0.5
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = min(frac * max_delta, 1e-3 * S)
                a_try = a.copy()
                a_try[j] += delta
                a_try[i] -= delta
                if a_try[i] < 0:
                    continue
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try * (S / s)
                c_try = self_convolve(a_try)
                if c_try.max() < c_max - 1e-12:
                    a = a_try
                    break

        # Periodic full score check
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    return a, compute_score(a)


def multi_peak_gradient_descent(a, n_iter=2000, top_k=20, rng=None):
    """
    Multi-peak subgradient descent with Nesterov acceleration.
    """
    n = len(a)
    S = a.sum()
    if rng is None:
        rng = np.random.default_rng(42)

    x = a.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        c = self_convolve(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]

        fsize = 1 << ((2 * n - 1) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * A))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: O(1/sqrt(t))
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.01 * it))
        x_new = project_simplex_fast(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-15:
            y = y * (S / s)
        x = x_new
        t = t_new

        # Periodic noise injection
        if it % 500 == 250 and rng.random() < 0.3:
            y = project_simplex_fast(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_a, best_score


def annealed_spectral_optimize(a, n_iter, rng, method="waterfill"):
    """
    Spectral optimization with annealed parameters.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        # Anneal: start aggressive, end gentle
        alpha = 0.8 * (1.0 - 0.9 * progress) + 0.02

        if method == "waterfill":
            a_new = spectral_waterfill(a, alpha, clip_quantile=0.85 - 0.3 * progress)
        else:
            a_new = spectral_exponential_compress(a, alpha, floor_ratio=0.2 + 0.3 * progress)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        # Noise for exploration
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex_fast(a + rng.normal(0, 0.0005 * S / np.sqrt(n), n), S)

    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) * 0.5)
    inits.append(tt + 0.1)
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * tt))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.sin(2 * np.pi * 7 * tt))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt ** 2)))
    inits.append(rng.random(n) + 0.05)
    inits.append(np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.1)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1)

    # Smooth noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(0.2 + 0.6 * smooth)

    # Fractional sine
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * np.sqrt(tt + 0.1)))

    return inits


def optimize_length(n, budget, rng, target=1.50, init_seq=None):
    """Full optimization pipeline for a fixed sequence length."""
    S = np.sqrt(2.0 * float(n))
    deadline = time.monotonic() + budget
    global_best = np.inf
    global_a = None

    def update(a):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        a = a / a.sum() * S
        sc = compute_score(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
        return sc

    # Build initialization list
    inits = []
    if init_seq is not None and len(init_seq) == n:
        arr = np.clip(init_seq.astype(np.float64), 0.0, 1000.0)
        if arr.sum() > 0.01:
            inits.append(arr / arr.sum() * S)

    for arr in make_initializations(n, rng):
        arr = np.clip(arr, 0.01, 10.0)
        inits.append(arr / arr.sum() * S)

    n_inits = len(inits)
    per_init = budget / (n_inits * 2.0)

    for init_idx, a0 in enumerate(inits):
        elapsed = time.monotonic()
        remaining = deadline - elapsed
        if remaining < 10:
            break

        a = a0.copy()
        sc = update(a)
        if sc <= target:
            break

        alloc = min(per_init, remaining - 10)

        # Stage 1: Spectral water-filling
        t1 = alloc * 0.35
        n_ss = max(100, int(t1 / (n * 20 * 1e-9)))
        n_ss = min(n_ss, 3000)
        a1, sc1 = annealed_spectral_optimize(a, n_ss, rng, method="waterfill")
        update(a1)

        # Stage 2: Spectral exponential (different flavor)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t2 = min(alloc * 0.25, remaining - 15)
            n_s2 = max(50, int(t2 / (n * 20 * 1e-9)))
            n_s2 = min(n_s2, 2000)
            a2, sc2 = annealed_spectral_optimize(a1, n_s2, rng, method="exponential")
            update(a2)
            a1 = a2

        # Stage 3: Multi-peak gradient descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t3 = min(alloc * 0.30, remaining - 15)
            # Estimate iterations
            n_gd = max(200, int(t3 / 0.01))
            n_gd = min(n_gd, 5000)
            a3, sc3 = multi_peak_gradient_descent(a1, n_gd, top_k=min(30, n // 200), rng=rng)
            update(a3)
            a1 = a3

        # Stage 4: Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t4 = min(alloc * 0.20, remaining - 5)
            n_cd = max(100, int(t4 / 0.005))
            n_cd = min(n_cd, 3000)
            a4, sc4 = coordinate_descent_polish(a1, max_iters=n_cd)
            update(a4)

        # Print progress for first few inits
        if init_idx < 5:
            print("    init %d: best=%.6f (remaining %.0fs)" % (init_idx, global_best, deadline - time.monotonic()))

        if global_best <= target:
            break

    # Final polish on best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # More aggressive gradient descent
            t_final = min(remaining * 0.5, 60.0)
            n_f = max(500, int(t_final / 0.01))
            a_f, sc_f = multi_peak_gradient_descent(global_a, n_f, top_k=min(50, n // 100), rng=rng)
            update(a_f)

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 5:
                n_cd2 = max(500, int(remaining * 0.5 / 0.005))
                a_cd, sc_cd = coordinate_descent_polish(global_a if global_a is not None else a_f, max_iters=n_cd2)
                update(a_cd)

    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses spectral water-filling + gradient descent + coordinate polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, deadline - time.monotonic()))

    global_best = np.inf
    global_a = None

    # Try to get provided sequence
    init_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256:
            init_seq = given
            print("Using provided sequence of length %d" % len(given))
    except Exception:
        pass

    # Multi-length strategy
    # Prioritize the provided length, then try a few others
    lengths = []
    if init_seq is not None:
        lengths.append(len(init_seq))

    for L in [8192, 16384, 4096, 32768]:
        if L not in lengths:
            lengths.append(L)

    # Sort: provided length first, then by preference
    def length_priority(L):
        if init_seq is not None and L == len(init_seq):
            return (0, 0)
        if L == 8192:
            return (1, 0)
        if L == 16384:
            return (1, 1)
        if L == 4096:
            return (2, 0)
        return (2, 1)

    lengths.sort(key=length_priority)

    # Allocate budget: favor 8192 and provided length
    total_budget = min(budget_s - 50, 2300)

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Allocate based on length preference
        if n == 8192 or (init_seq is not None and n == len(init_seq)):
            alloc = min(total_budget * 0.40, remaining - 60)
        elif n == 16384:
            alloc = min(total_budget * 0.25, remaining - 60)
        elif n == 4096:
            alloc = min(total_budget * 0.15, remaining - 60)
        else:
            alloc = min(total_budget * 0.20, remaining - 60)

        alloc = max(alloc, 30)
        if alloc > remaining - 30:
            alloc = remaining - 30
        if alloc < 20:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        t0 = time.monotonic()

        a_res, score = optimize_length(n, alloc, rng, target=1.50, init_seq=init_seq)

        elapsed = time.monotonic() - t0
        print("  n=%d done in %.0fs, score=%.6f" % (n, elapsed, score))

        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  New global best: %.6f" % global_best)

        if global_best <= 1.5027:
            print("  Target achieved!")
            break

    # If we still have time, do additional polish on the best
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        print("Final polish with %.0fs remaining" % remaining)
        n_best = len(global_a)
        S = np.sqrt(2.0 * n_best)
        global_a = global_a / global_a.sum() * S

        # Extended gradient descent
        n_gd = int(min(remaining * 0.4, 120.0) / 0.01)
        a_gd, sc_gd = multi_peak_gradient_descent(global_a, n_gd, top_k=min(40, n_best // 150), rng=rng)
        if sc_gd < global_best:
            global_best = sc_gd
            global_a = a_gd

        # Extended coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_cd = int(min(remaining * 0.5, 60.0) / 0.005)
            a_cd, sc_cd = coordinate_descent_polish(global_a, max_iters=n_cd)
            if sc_cd < global_best:
                global_best = sc_cd
                global_a = a_cd

    # Fallback
    if global_a is None:
        print("FALLBACK: using uniform sequence")
        n_fb = 4096
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = compute_score(global_a)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: evaluate_sequence = %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")
            global_a = np.ones(4092, dtype=np.float64)
            ver = evaluate_sequence([float(x) for x in global_a])
            global_best = ver if np.isfinite(ver) else 2.0
    except Exception as e:
        print("Verification error: %s" % str(e))

    print("Final: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
