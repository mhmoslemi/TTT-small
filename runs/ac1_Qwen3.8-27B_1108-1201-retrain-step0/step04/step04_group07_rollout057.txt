"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using p-norm continuation with
projected Nesterov gradient, combined with spectral flattening and
subgradient polishing. The p-norm continuation smooths the non-smooth max
objective, enabling efficient gradient-based optimization. Multiple sequence
lengths are explored with adaptive time allocation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_pnorm_gradient(a, w):
    """
    Compute gradient of (1/p)*sum_k c[k]^p where c = convolve(a,a).
    w[k] = c[k]^(p-1).
    grad[i] = 2 * sum_j a[j] * w[i+j] (for valid indices).
    Computed via: 2 * convolve(a, w[::-1])[n-1:2n-1][::-1]
    """
    n = len(a)
    w_rev = w[::-1]
    full = np.convolve(a, w_rev)
    # grad[i] = 2 * full[2n-2-i], so grad = 2 * full[n-1:2n-1][::-1]
    grad = 2.0 * full[n - 1:2 * n - 1][::-1]
    return grad


def pnorm_continuation(a_init, n, S, max_time, rng, p_schedule=None, max_inner_iters=5000):
    """
    p-norm continuation method: minimize (1/p)*sum c[k]^p as a surrogate for max(c).
    Uses projected Nesterov gradient with adaptive step size.
    """
    start = time.monotonic()
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    if p_schedule is None:
        p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]

    best_a = a.copy()
    best_score = compute_score(a)
    N = 1 << (2 * n - 1).bit_length()

    for p in p_schedule:
        if time.monotonic() - start > max_time * 0.9:
            break

        # Initialize Nesterov state
        y = a.copy()
        t_k = 1.0
        lr_base = S / np.sqrt(n) * 0.1
        stall_count = 0
        prev_obj = None

        for it in range(max_inner_iters):
            if time.monotonic() - start > max_time * 0.9:
                break

            # Compute convolution and gradient
            c = fft_convolve_self(y)
            c = np.maximum(c, 0.0)  # numerical safety
            w = c ** (p - 1)
            grad = compute_pnorm_gradient(y, w)

            # Objective for monitoring
            obj = (np.sum(c ** p) / p) ** (1.0 / p) if p > 1 else c.max()
            score = 2.0 * n * c.max() / (y.sum() ** 2)
            if score < best_score:
                best_score = score
                best_a = y.copy()

            # Stall detection
            if prev_obj is not None and abs(obj - prev_obj) < 1e-12 * max(prev_obj, 1e-30):
                stall_count += 1
            else:
                stall_count = 0
            prev_obj = obj

            if stall_count > 50:
                # Perturb to escape
                y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                t_k = 1.0
                stall_count = 0
                continue

            # Gradient norm for adaptive step
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Nesterov step
            lr = lr_base / (1.0 + 0.01 * it)
            x_new = project_simplex(y - lr * grad / gnorm * gnorm, S)
            # Simplified: step = y - lr * grad
            x_new = project_simplex(y - lr * grad, S)

            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))
            beta = (t_k - 1.0) / t_new
            y = x_new + beta * (x_new - a)
            y = np.maximum(y, 0.0)
            # Re-project to ensure feasibility
            y_sum = y.sum()
            if y_sum < 1e-15:
                y = a.copy()
            else:
                y = y * (S / y_sum)
            a = x_new
            t_k = t_new

    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k=50, use_adaptive_lr=True):
    """
    Subgradient method with Nesterov acceleration for max(convolve(a,a)).
    Uses top-k active lags for the subgradient.
    """
    start = time.monotonic()
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_k = 1.0
    best_score = compute_score(a)
    best_a = a.copy()
    lr = S / np.sqrt(n) * 0.5
    it = 0
    prev_max_c = None
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        c_max = c.max()
        score = 2.0 * n * c_max / (y.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = y.copy()

        # Stall detection
        if prev_max_c is not None and abs(c_max - prev_max_c) < 1e-10 * prev_max_c:
            stall += 1
        else:
            stall = 0
        prev_max_c = c_max

        if stall > 200:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_k = 1.0
            stall = 0
            continue

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Build weight vector and compute gradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]

        # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
        # = 2 * convolve(a, w[::-1])[n-1:2n-1][::-1]
        w_rev = w_full[::-1]
        fsize = 1 << (3 * n - 2).bit_length()
        A = fft(y, fsize)
        W = fft(w_rev, fsize)
        grad_full = np.real(ifft(A * W))
        grad = 2.0 * grad_full[n - 1:2 * n - 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        if use_adaptive_lr:
            lr = 2.0 * S / (n * (1.0 + 0.005 * it))

        x_new = project_simplex(y - lr * grad, S)

        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))
        beta = (t_k - 1.0) / t_new
        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y * (S / y_sum)
        a = x_new
        t_k = t_k  # Already updated via t_new
        t_k = t_new

    return best_a, best_score


def spectral_flatten_pass(a, N, n_steps, alpha_init, rng, target_obj=1.5):
    """Spectral flattening: iteratively push |FFT(a)|^2 toward uniform."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_steps):
        progress = it / max(n_steps - 1, 1)
        alpha = max(0.01, alpha_init * (1.0 - 0.95 * progress))

        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        p_median = np.median(P)
        if p_median < 1e-30:
            break

        ratio = P / p_median
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * (1.0 - alpha * (1.0 - np.power(np.maximum(ratio[above], 1e-10), -alpha)))
        P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - np.power(np.maximum(ratio[~above], 1e-10), alpha)))
        P_new = np.maximum(P_new, 0.0)

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new_full, S)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if best_score <= target_obj:
            break

        # Periodic noise
        if it % 500 == 250 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_score


def peak_targeted_optimize(a_init, n, S, max_time, rng):
    """
    Targeted optimization: identify peak-contributing pairs and locally
    adjust them to reduce the peak.
    """
    start = time.monotonic()
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = compute_score(a)

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        score = 2.0 * n * c_max / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Find top contributing lags
        n_peaks = min(5, 2 * n - 1)
        peak_lags = np.argsort(c)[::-1][:n_peaks]

        # For each peak lag k, the contribution is sum_i a[i]*a[k-i]
        # Reduce pairs that contribute most
        for k in peak_lags:
            k = int(k)
            # Find pairs (i, k-i) that contribute to this lag
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            contribution = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    contribution[i] += a[i] * a[j]
                    if i != j:
                        contribution[j] += a[i] * a[j]

            # Reduce high-contribution positions slightly
            max_contrib = contribution.max()
            if max_contrib > 1e-30:
                reduction = 0.05 * contribution / max_contrib
                a_new = a * (1.0 - reduction)
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * (S / s_new)
                    a_new = project_simplex(a_new, S)
                    # Boost low positions to maintain sum
                    a = a_new

        # Add small random perturbation
        a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_score


def make_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * tt))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.5))
    inits.append(0.5 + 0.2 * np.sin(2 * np.pi * 5 * tt) + 0.15 * np.sin(2 * np.pi * 11 * tt))
    inits.append(rng.random(n) * 0.8 + 0.1)

    # Smooth random
    noise = rng.standard_normal(n)
    ks = max(3, n // 32)
    kernel = np.ones(ks) / ks
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(0.2 + 0.6 * smooth)

    # Chirp
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt ** 2)))

    return inits


def optimize_length(n, budget_s, rng, a_init=None):
    """Full optimization pipeline for a single length."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    best_a = None
    best_score = np.inf

    def update(a):
        nonlocal best_a, best_score
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        return sc

    t_start = time.monotonic()

    # Build candidate list of initializations
    candidates = []
    if a_init is not None and len(a_init) == n:
        candidates.append(np.clip(a_init, 0.0, 1000.0).astype(np.float64))
    inits = make_diverse_inits(n, rng)
    candidates.extend(inits)

    # Deduplicate and limit
    max_candidates = 6
    candidates = candidates[:max_candidates]

    per_cand_time = budget_s / (len(candidates) + 1)  # +1 for final polish

    for ci, a0 in enumerate(candidates):
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed
        if remaining < 5:
            break

        a = a0.copy()
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
        a = np.maximum(a, 0.0)

        # Stage 1: Spectral flattening (fast initial improvement)
        t1 = min(per_cand_time * 0.3, remaining * 0.4)
        if t1 > 2:
            n_ss = max(100, int(t1 / max(0.001, N * 20e-9)))
            n_ss = min(n_ss, 3000)
            a, sc = spectral_flatten_pass(a, N, n_ss, alpha_init=0.7, rng=rng, target_obj=1.5)
            update(a)

        # Stage 2: p-norm continuation
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed
        t2 = min(per_cand_time * 0.35, remaining * 0.35)
        if t2 > 3:
            a, sc = pnorm_continuation(a, n, S, t2, rng)
            update(a)

        # Stage 3: Subgradient polish
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed
        t3 = min(per_cand_time * 0.3, remaining * 0.3)
        if t3 > 3:
            tk = min(100, max(20, n // 80))
            a, sc = subgradient_polish(a, n, S, t3, rng, top_k=tk)
            update(a)

        if n <= 16384:
            print("  [cand %d, n=%d] score=%.6f (elapsed %.0fs/%.0fs)" %
                  (ci, n, best_score, time.monotonic() - t_start, budget_s))

    # Final polish on best
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = max(0, budget_s - elapsed)
        if remaining > 5:
            # Further subgradient with more top-k
            tk = min(200, max(50, n // 40))
            a_f, _ = subgradient_polish(best_a, n, S, min(remaining * 0.5, 120.0), rng, top_k=tk)
            update(a_f)

            # Gentle spectral pass
            elapsed = time.monotonic() - t_start
            remaining = max(0, budget_s - elapsed)
            if remaining > 3:
                n_sp = max(50, int(remaining * 0.5 / max(0.001, N * 20e-9)))
                a_f, _ = spectral_flatten_pass(best_a, N, n_sp, alpha_init=0.1, rng=rng, target_obj=1.5)
                update(a_f)

    return best_a, best_score
