import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= S:
        # Might still need projection if some are negative
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-15:
            return np.full(n, S / n)
        return v * (S / s)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, top_k):
    """
    Compute subgradient of max_k c[k] where c = conv(a,a).
    Returns (gradient, c, top_lag_indices, weights).
    """
    n = len(a)
    c = fft_convolve_self(a)
    # Use top_k lags with weights proportional to values above the median
    sort_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sort_idx]
    # Weight by excess above the minimum of the top set
    vmin = top_vals.min()
    excess = top_vals - vmin
    if excess.sum() < 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        weights = excess / excess.sum()
    # Compute gradient: g[i] = 2 * sum_j weights_j * a[k_j - i]
    g = np.zeros(n)
    for jj in range(top_k):
        k = int(sort_idx[jj])
        wj = weights[jj]
        # a[k-i] for valid i: need 0 <= k-i < n, so i in [k-n+1, k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # a[k-i] for i in [i_lo, i_hi]: indices k-i_hi to k-i_lo
        idx_start = k - i_hi
        idx_end = k - i_lo + 1
        g[i_lo:i_hi + 1] += 2.0 * wj * a[idx_start:idx_end]
    return g, c, sort_idx, weights


def spectral_shape_step(a, N, n, top_k, alpha, c=None):
    """
    One spectral shaping step. Modifies power spectrum to reduce peak lags.
    Returns new a (projected to simplex).
    """
    S = a.sum()
    if c is None:
        c = fft_convolve_self(a)
    c_max = c.max()
    sort_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sort_idx]
    vmin = top_vals.min()
    excess = top_vals - vmin
    if excess.sum() < 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        weights = excess / excess.sum()
    # Compute effective cosine pattern
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for jj in range(top_k):
        k = int(sort_idx[jj])
        cos_eff += weights[jj] * np.cos(2.0 * np.pi * m * k / N)
    # Only suppress where cos_eff > 0
    cos_eff = np.maximum(cos_eff, 0.0)
    # Adaptive alpha
    alpha_eff = alpha * (1.0 + 0.5 * (c_max - c.mean()) / max(c.mean(), 1e-10))
    alpha_eff = min(alpha_eff, 5.0)
    # Modify spectrum
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_new = P * np.exp(-alpha_eff * cos_eff)
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def nesterov_subgradient_opt(a_init, n, S, max_time, rng, k_init=5, k_max=256):
    """
    Nesterov-accelerated projected subgradient on max-conv objective.
    """
    a = a_init.copy()
    y = a.copy()

            best_local = y.copy()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Adaptive step size with decay
        step = base_step / (1.0 + 0.01 * it)
        # Try a few step multipliers
        best_x = y.copy()
        best_obj = cur
        for mult in [1.0, 0.5, 2.0, 0.25]:
            x_try = project_simplex(y - step * mult * g / max(gnorm, 1e-15), S)
            c_try = fft_convolve_self(x_try)
            obj_try = c_try.max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_x = x_try

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))

def make_init_structured(n, rng, method="sin"):
    """Structured initialization."""
    t = np.arange(n) / n
    if method == "sin":
        a = (0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t)
             + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 4 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = (0.3 + 0.4 * np.exp(-0.5 * ((t - 0.3) * 5) ** 2)
             + 0.3 * np.exp(-0.5 * ((t - 0.7) * 5) ** 2) + 0.05 * rng.standard_normal(n))
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.4) + 0.08 * rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = max(8, n // 128)
        piece_len = n // n_pieces
        heights = 0.2 + 0.8 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, 0.2 + 0.8 * rng.random(n - len(a))])
        a = a[:n]
    elif method == "gaussian_mix":
        centers = rng.uniform(0.1, 0.9, 5)
        widths = rng.uniform(0.05, 0.2, 5)
        amps = rng.uniform(0.3, 1.0, 5)
        a = np.zeros(n)
        for cc, ww, aa in zip(centers, widths, amps):
            a += aa * np.exp(-0.5 * ((t - cc) / ww) ** 2)
        a += 0.05 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, 3.0)


def make_init_msequence(n, rng):
    """
    Create a non-negative sequence inspired by m-sequence autocorrelation properties.
    Use a pseudo-random binary pattern mapped to [0, 1] with smoothing.
    """
    # Generate a good pseudo-random bit pattern
    bits = rng.integers(0, 2, n)
    # Smooth with a small kernel to reduce sharp transitions
    kernel = np.array([0.2, 0.6, 0.2])
    a = np.convolve(bits.astype(float), kernel, mode='same')
    a = np.clip(a, 0.05, 1.5)
    return a


def normalize_sequence(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_sequence(a, n):
    """Compute the score 2*n*max_conv/sum^2."""
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:

    return 2.0 * n * c.max() / (s * s)

        candidate_ns.insert(0, prev_n)

    print("Budget: %.0fs, Candidate lengths: %s, Previous seq: %s" %
          (budget_s, candidate_ns, "yes (%d)" % prev_n if prev_seq is not None else "no"))

    # ========== Phase 1: Multi-start exploration (25% of time) ==========
    print("=== Phase 1: Multi-start exploration ===")
    t_phase1 = time.monotonic()
    phase1_time = (deadline - t_phase1) * 0.25
    phase1_deadline = t_phase1 + phase1_time

    n_starts_per_len = 4
    time_per_start = phase1_time / (len(candidate_ns) * n_starts_per_len)
    time_per_start = max(time_per_start, 10.0)  # Minimum 10s per start

    for n in candidate_ns:
        if time.monotonic() > phase1_deadline - 30:
            break
        N = 1 << (2 * n - 1).bit_length()
        S = np.sqrt(2.0 * float(n))

        # Build initializations for this length
        inits = []
        if prev_seq is not None and prev_n == n:
            inits.append(("prev", prev_seq.copy()))
        inits.append(("rand", make_init_random(n, rng)))
        inits.append(("sin", make_init_structured(n, rng, "sin")))
        inits.append(("mseq", make_init_msequence(n, rng)))
        inits.append(("piece", make_init_structured(n, rng, "piecewise")))
        inits.append(("gauss", make_init_structured(n, rng, "gaussian_mix")))
        inits = inits[:n_starts_per_len]

        for name, arr in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            a_norm = normalize_sequence(arr, n)
            sc0 = score_sequence(a_norm, n)
            if sc0 < global_best:
                global_best = sc0
                global_a = a_norm.copy()
                global_n = n
                print("  [n=%d, %s] init: %.6f" % (n, name, global_best))

            # Quick spectral optimization
            t0 = time.monotonic()
            a_sw, sc_sw = spectral_loop(a_norm, n, N, S,
                                        min(time_per_start * 0.6, 40.0),
                                        rng, target=1.48, top_k_init=5, alpha_init=0.5)
            if try_update(a_sw, n):
                print("  [n=%d, %s] spectral: %.6f" % (n, name, global_best))

            # Quick subgradient
            rem = time_per_start - (time.monotonic() - t0)
            if rem > 5:
                a_ng, sc_ng = nesterov_subgradient_opt(global_a if global_n == n else a_norm,
                                                       n, S, min(rem * 0.5, 20.0),
                                                       rng, k_init=5, k_max=64)
                if try_update(a_ng, n):
                    print("  [n=%d, %s] subgrad: %.6f" % (n, name, global_best))

    print("=== Phase 1 done. Best: %.6f (n=%d) ===" % (global_best, global_n))

    if global_a is None:
        global_n = 8192
        global_a = normalize_sequence(make_init_random(global_n, rng), global_n)
        global_best = score_sequence(global_a, global_n)

    # ========== Phase 2: Intensive optimization (50% of time) ==========
    print("=== Phase 2: Intensive optimization (n=%d) ===" % global_n)
    t_phase2 = time.monotonic()
    phase2_time = (deadline - t_phase2) * 0.50
    phase2_deadline = t_phase2 + phase2_time

    n_best = global_n
    N_best = 1 << (2 * n_best - 1).bit_length()
    S_best = np.sqrt(2.0 * float(n_best))
    a_curr = global_a.copy()
    stall_count = 0
    it_count = 0

    while time.monotonic() < phase2_deadline:
        it_count += 1
        rem = phase2_deadline - time.monotonic()
        if rem < 3:
            break

        # Alternate between methods with increasing aggressiveness
        phase_in_cycle = it_count % 4

        if phase_in_cycle == 0:
            # Spectral shaping
            tb = min(rem * 0.25, 40.0)
            a_new, sc = spectral_loop(a_curr, n_best, N_best, S_best, tb,
                                      rng, target=1.45,
                                      top_k_init=10, alpha_init=0.4 / (1 + 0.1 * stall_count))
            if try_update(a_new, n_best):
                a_curr = a_new.copy()
                stall_count = 0
                print("  [P2 iter %d] spectral: %.6f" % (it_count, global_best))
            else:
                stall_count += 1

        elif phase_in_cycle == 1:
            # Subgradient
            tb = min(rem * 0.25, 40.0)
            k_init = 10 + stall_count * 5
            a_new, sc = nesterov_subgradient_opt(a_curr, n_best, S_best, tb,
                                                 rng, k_init=min(k_init, 50),
                                                 k_max=min(128 + stall_count * 10, 512))
            if try_update(a_new, n_best):
                a_curr = a_new.copy()
                stall_count = 0
                print("  [P2 iter %d] subgrad: %.6f" % (it_count, global_best))
            else:
                stall_count += 1

        elif phase_in_cycle == 2:
            # Spectral with different parameters
            tb = min(rem * 0.2, 30.0)
            a_new, sc = spectral_loop(a_curr, n_best, N_best, S_best, tb,
                                      rng, target=1.44,
                                      top_k_init=20 + stall_count * 5,
                                      alpha_init=0.2)
            if try_update(a_new, n_best):
                a_curr = a_new.copy()
                stall_count = 0
                print("  [P2 iter %d] spectral2: %.6f" % (it_count, global_best))
            else:
                stall_count += 1

        else:
            # Subgradient with different params
            tb = min(rem * 0.2, 30.0)
            a_new, sc = nesterov_subgradient_opt(a_curr, n_best, S_best, tb,
                                                 rng, k_init=50, k_max=300)
            if try_update(a_new, n_best):
                a_curr = a_new.copy()
                stall_count = 0
                print("  [P2 iter %d] subgrad2: %.6f" % (it_count, global_best))
            else:
                stall_count += 1

        # Perturbation when stalled
        if stall_count >= 4 and rem > 10:
            # Try different perturbation strengths
            for sigma in [0.02, 0.05, 0.1, 0.2]:
                a_pert = project_simplex(a_curr + rng.normal(0, sigma * S_best / np.sqrt(n_best), n_best), S_best)
                # Quick spectral from perturbed
                a_pert2, _ = spectral_loop(a_pert, n_best, N_best, S_best,
                                           min(rem * 0.1, 10.0),
                                           rng, target=1.45, top_k_init=10, alpha_init=0.5)
                if try_update(a_pert2, n_best):
                    a_curr = a_pert2.copy()
                    stall_count = 0
                    print("  [P2 iter %d] perturb(%.2f): %.6f" % (it_count, sigma, global_best))
                    break
                if try_update(a_pert, n_best):
                    a_curr = a_pert.copy()
                    stall_count = 0
                    break
            stall_count = 0

        # Also try the previous sequence's length if different
        if prev_seq is not None and prev_n != n_best and it_count % 5 == 0:
            n_alt = prev_n
            N_alt = 1 << (2 * n_alt - 1).bit_length()
            S_alt = np.sqrt(2.0 * float(n_alt))
            a_alt = normalize_sequence(prev_seq, n_alt)
            a_alt_sw, _ = spectral_loop(a_alt, n_alt, N_alt, S_alt,
                                        min(rem * 0.1, 15.0),
                                        rng, target=1.45, top_k_init=10, alpha_init=0.4)
            if try_update(a_alt_sw, n_alt):
                a_curr = a_alt_sw.copy()
                n_best = n_alt
                N_best = N_alt
                S_best = S_alt
                stall_count = 0
                print("  [P2 iter %d] switched to n=%d: %.6f" % (it_count, n_best, global_best))

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # ========== Phase 3: Fine polishing (25% of time) ==========
    print("=== Phase 3: Fine polishing ===")
    t_phase3 = time.monotonic()
    phase3_time = deadline - t_phase3

    # Coordinate descent polish
    seg1 = min(phase3_time * 0.35, 120.0)
    if seg1 > 3:
        a_cd, sc_cd = coordinate_descent_polish(global_a, global_n, S_best, seg1,
                                                rng, n_top=50, n_rounds=300)
        if try_update(a_cd, global_n):
            print("  [P3] coord_descent: %.6f" % global_best)

    # Fine subgradient
    rem3 = deadline - time.monotonic()
    seg2 = min(rem3 * 0.30, 100.0)
    if seg2 > 3:
        a_fine, sc_fine = nesterov_subgradient_opt(global_a, global_n, S_best, seg2,
                                                   rng, k_init=100, k_max=512)
        if try_update(a_fine, global_n):
            print("  [P3] fine_subgrad: %.6f" % global_best)

    # Very fine spectral
    rem4 = deadline - time.monotonic()
    seg3 = min(rem4 * 0.30, 80.0)
    if seg3 > 3:
        a_sf, sc_sf = spectral_loop(global_a, global_n, N_best, S_best, seg3,
                                    rng, target=1.40, top_k_init=50, alpha_init=0.1)
        if try_update(a_sf, global_n):
            print("  [P3] fine_spectral: %.6f" % global_best)

    # Final coordinate descent
    rem5 = deadline - time.monotonic()
    if rem5 > 3:
        a_cd2, _ = coordinate_descent_polish(global_a, global_n, S_best,
                                             min(rem5, 60.0),
                                             rng, n_top=80, n_rounds=500)
        if try_update(a_cd2, global_n):
            print("  [P3] final_coord: %.6f" % global_best)

    # One last subgradient burst
    rem6 = deadline - time.monotonic()
    if rem6 > 3:
        a_last, _ = nesterov_subgradient_opt(global_a, global_n, S_best,
                                             min(rem6, 30.0),
                                             rng, k_init=200, k_max=512)
        if try_update(a_last, global_n):
            print("  [P3] last_subgrad: %.6f" % global_best)

    # ========== Verification ==========
    print("=== Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified score: %.6f" % ver)
            print("Internal score: %.6f" % global_best)
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("Fallback score: %.6f" % ver)
    else:
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
