"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Top-k subgradient FISTA with adaptive step size (primary method)
2. LSE smoothing as a secondary gradient (for different landscape regions)
3. Local search via targeted mass redistribution
4. Multi-length exploration (n = 4096, 8192, 16384)
5. Structured perturbations to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using the exact simplex projection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(a, c, k=20, threshold_frac=0.5):
    """Exact subgradient from top-k peaks, weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    sorted_idx = np.argsort(c)[::-1][:k]
    # Weight by excess above a threshold
    thresh = c.max() * threshold_frac
    weights = np.maximum(c[sorted_idx] - thresh, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        # Uniform weight on top-k
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    
    w_full = np.zeros(len(c))
    w_full[sorted_idx] = weights
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        M = c.max()
        if M < best:
            best = M
            best_a = y.copy()
            last_improve = it
        
        if mode == 'topk':
            g = subgradient_topk(y, c, k=k_peaks)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Top-k exact subgradient FISTA with adaptive step size
2. Cyclical LSE smoothing with varying beta schedules
3. Targeted local search via peak-contributing element analysis
4. Multi-length exploration (n = 4096, 8192)
5. Structured perturbations and restarts to escape local minima
Key difference from prior: uses exact subgradients (not just LSE) as primary
signal, with a "peak conflict" local search that directly identifies and
resolves contributing element pairs.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(a, c, k=16, thresh_frac=0.5):
    """Exact subgradient from top-k peaks weighted by excess above threshold."""
    n = len(a)
    k = min(k, len(c))
    sorted_idx = np.argsort(c)[::-1][:k]
    thresh = c.max() * thresh_frac
    weights = np.maximum(c[sorted_idx] - thresh, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[sorted_idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_subgrad_fista(a_init, S, max_time, rng, k_peaks=16, mode='topk',
                      eta_init=0.5, eta_decay=0.99995):
    """FISTA with top-k subgradient or LSE gradient."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = eta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        M = c.max()
        if M < best:
            best = M
            best_a = y.copy()
            last_improve = it

        if mode == 'topk':
            g = subgradient_topk(y, c, k=k_peaks)
        else:
            beta = min(8000, 20.0 * np.exp(0.008 * it))
            g = lse_gradient(y, c, beta)
            eta = eta_init * np.exp(-0.003 * it)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        if mode == 'topk':
            step = eta
            eta *= eta_decay
        else:
            beta_used = min(8000, 20.0 * np.exp(0.008 * it))
            step = 1.0 / (2.0 * beta_used * 0.12)

        x_new = project_simplex(y - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

def run_peak_conflict_search(a_init, S, max_time, rng):
    """
    Targeted local search: identify peak-contributing element pairs and
    redistribute mass to reduce the peak while maintaining sum constraint.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for k_peak in peak_indices:
            k_peak = int(k_peak)
            # Find contributing elements
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            if i_start > i_end:
                continue

            # Compute products a[i]*a[k_peak-i]
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_peak - i_range
            valid = (j_range >= 0) & (j_range < n)
            i_range = i_range[valid]
            j_range = j_range[valid]
            if len(i_range) == 0:
                continue

            products = a[i_range] * a[j_range]
            if products.max() < 1e-15:
                continue

            # Find the largest contributor
            idx_max = np.argmax(products)
            i_large = i_range[idx_max]
            j_large = j_range[idx_max]

            # Try to reduce the product by moving mass
            # Move a small amount from the larger element to a "safe" position
            if a[i_large] > a[j_large]:
                src = i_large
                src_partner = j_large
            else:
                src = j_large
                src_partner = i_large

            # Find safe target: position that doesn't contribute to top peaks
            # Score positions by their total contribution to top peaks
            scores = np.zeros(n)
            for kp in peak_indices[:3]:
                kp = int(kp)
                s_i_start = max(0, kp - n + 1)
                s_i_end = min(n - 1, kp)
                for ipos in range(s_i_start, s_i_end + 1):
                    jpos = kp - ipos
                    if 0 <= jpos < n:
                        scores[ipos] += a[ipos] * a[jpos]

            # Target: a position with low score that is not src or src_partner
            candidate_mask = scores < np.median(scores)
            candidate_mask[src] = False
            if src_partner < n:
                candidate_mask[src_partner] = False
            candidates = np.where(candidate_mask)[0]
            if len(candidates) == 0:
                candidates = np.setdiff1d(np.arange(n), [src, src_partner])
            if len(candidates) == 0:
                continue

            target = candidates[rng.integers(len(candidates))]

            # Try different delta sizes
            for frac in [0.05, 0.02, 0.01, 0.005]:
                delta = frac * a[src]
                if delta < 1e-12:
                    continue
                a_new = a.copy()
                a_new[src] -= delta
                a_new[target] += delta
                if a_new[src] < 0:
                    continue
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new * (S / s)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < c.max() * (1 - 1e-9):
                    a = a_new
                    c = c_new
                    improved = True
                    break

        # If no improvement, try small random perturbation
        if not improved and it % 50 == 0:
            n_perturb = rng.integers(5, 30)
            pert_idx = rng.choice(n, n_perturb, replace=False)
            a_new = a.copy()
            a_new[pert_idx] *= (1.0 + rng.normal(0, 0.1, n_perturb))
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * (S / s)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < c.max():
                    a = a_new

    return best_a, best


def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())

    # Windows
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(1.0 + 0.5 * np.sin(4.0 * np.pi * tt))
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * 3 * tt) + 0.2 * np.cos(2.0 * np.pi * 7 * tt + 0.3))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Ramp (asymmetric)
    ramp = np.linspace(0.2, 1.0, n)
    inits.append(ramp)
    inits.append(2.0 - ramp)

    # Two-peaked
    inits.append(np.exp(-20.0 * (tt - 0.25) ** 2) + np.exp(-20.0 * (tt - 0.75) ** 2))

    # Block structure
    nblock = 8
    block_len = n // nblock
    heights = rng.uniform(0.3, 1.2, nblock)
    inits.append(np.repeat(heights, block_len)[:n])

    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses top-k subgradient FISTA, cyclical LSE, peak-conflict local search,
    and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
            print("  Given objective: %.6f" % objective_value(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Main length
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)

    # Prepare initializations
    inits = []
    if given is not None:
        if len(given) == n_main:
            inits.append(given.copy())
        elif len(given) > n_main:
            ratio = len(given) // n_main
            small = given[:n_main * ratio].reshape(n_main, ratio).mean(axis=1)
            inits.append(small)
        else:
            inits.append(upscale_sequence(given, n_main))

    gen_inits = generate_inits(n_main, rng)[:6]
    inits.extend(gen_inits)

    # === PHASE 1: Diverse exploration (25% time) ===
    t_phase1 = (deadline - time.monotonic()) * 0.25
    print("\n=== Phase 1: Exploration (%.0fs) ===" % t_phase1)

    best_a_n = None
    best_val_n = np.inf
    t_per_init = t_phase1 / len(inits)

    for i, init in enumerate(inits):
        if time.monotonic() > deadline - 90:
            break
        t_alloc = min(t_per_init, deadline - time.monotonic() - 45)
        if t_alloc < 5:
            continue

        n_i = len(init)
        S_i = np.sqrt(2.0 * n_i)

        if i % 4 == 0:
            a_opt, _ = run_subgrad_fista(init, S_i, t_alloc, rng,
                                          k_peaks=16, mode='topk',
                                          eta_init=0.5, eta_decay=0.99995)
        elif i % 4 == 1:
            a_opt, _ = run_cyclical_lse(init, S_i, t_alloc, rng)
        elif i % 4 == 2:
            a_opt, _ = run_subgrad_fista(init, S_i, t_alloc, rng,
                                          k_peaks=32, mode='lse',
                                          eta_init=0.3, eta_decay=0.9999)
        else:
            a_opt, _ = run_subgrad_fista(init, S_i, t_alloc, rng,
                                          k_peaks=8, mode='topk',
                                          eta_init=0.8, eta_decay=0.99999)

        val_check = fft_convolve_self(a_opt).max()
        obj = 2.0 * n_i * val_check / (a_opt.sum() ** 2)
        print("  Init %d: n=%d, obj=%.6f" % (i, n_i, obj))

        if val_check < best_val_n:
            best_val_n = val_check
            best_a_n = a_opt.copy()

    if best_a_n is None:
        best_a_n = np.ones(n_main) * S_main / n_main
        best_val_n = fft_convolve_self(best_a_n).max()

    obj_n = 2.0 * len(best_a_n) * best_val_n / (best_a_n.sum() ** 2)
    if obj_n < global_best:
        global_best = obj_n
        global_a = best_a_n.copy()
        print("  Phase 1 best: obj=%.6f" % global_best)

    # === PHASE 2: Intensive refinement (45% time) ===
    t_phase2 = (deadline - time.monotonic()) * 0.45
    print("\n=== Phase 2: Refinement (%.0fs) ===" % t_phase2)

    n_cur = len(global_a)
    S_cur = np.sqrt(2.0 * n_cur)

    # 2a: Subgradient FISTA with small k (sharp peaks)
    t_2a = t_phase2 * 0.30
    a_2a, _ = run_subgrad_fista(global_a, S_cur, t_2a, rng,
                                 k_peaks=6, mode='topk',
                                 eta_init=0.4, eta_decay=0.99997)
    val_2a = fft_convolve_self(a_2a).max()
    obj_2a = 2.0 * n_cur * val_2a / (a_2a.sum() ** 2)
    if obj_2a < global_best:
        global_best = obj_2a
        global_a = a_2a.copy()
        print("  After subgrad (k=6): obj=%.6f" % global_best)

    # 2b: Peak conflict local search
    t_2b = t_phase2 * 0.25
    a_2b, _ = run_peak_conflict_search(global_a, S_cur, t_2b, rng)
    val_2b = fft_convolve_self(a_2b).max()
    obj_2b = 2.0 * n_cur * val_2a if False else 2.0 * len(a_2b) * val_2b / (a_2b.sum() ** 2)
    if obj_2b < global_best:
        global_best = obj_2b
        global_a = a_2b.copy()
        print("  After peak conflict: obj=%.6f" % global_best)

    # 2c: Cyclical LSE for broader search
    t_2c = t_phase2 * 0.25
    a_2c, _ = run_cyclical_lse(global_a, S_cur, t_2c, rng)
    val_2c = fft_convolve_self(a_2c).max()
    obj_2c = 2.0 * len(a_2c) * val_2c / (a_2c.sum() ** 2)
    if obj_2c < global_best:
        global_best = obj_2c
        global_a = a_2c.copy()
        print("  After cyclical LSE: obj=%.6f" % global_best)

    # 2d: Another subgradient pass with different params
    t_2d = min(t_phase2 * 0.20, deadline - time.monotonic() - 30)
    if t_2d > 10:
        a_2d, _ = run_subgrad_fista(global_a, S_cur, t_2d, rng,
                                     k_peaks=24, mode='topk',
                                     eta_init=0.2, eta_decay=0.99998)
        val_2d = fft_convolve_self(a_2d).max()
        obj_2d = 2.0 * len(a_2d) * val_2d / (a_2d.sum() ** 2)
        if obj_2d < global_best:
            global_best = obj_2d
            global_a = a_2d.copy()
            print("  After subgrad (k=24): obj=%.6f" % global_best)

    # === PHASE 3: Multi-length exploration (15% time) ===
    t_phase3 = (deadline - time.monotonic()) * 0.15
    print("\n=== Phase 3: Multi-length (%.0fs) ===" % t_phase3)

    other_lengths = [4096, 16384]
    for n_alt in other_lengths:
        if time.monotonic() > deadline - 30:
            break
        S_alt = np.sqrt(2.0 * n_alt)
        t_alt = min(t_phase3, deadline - time.monotonic() - 20)
        if t_alt < 15:
            continue

        # Upscale or downscale current best
        if n_alt > len(global_a):
            init_alt = upscale_sequence(global_a, n_alt)
        else:
            ratio = len(global_a) // n_alt
            if ratio > 0 and len(global_a) % ratio == 0:
                init_alt = global_a[:n_alt * ratio].reshape(n_alt, ratio).mean(axis=1)
            else:
                init_alt = upscale_sequence(global_a, n_alt)

        a_alt, _ = run_subgrad_fista(init_alt, S_alt, t_alt * 0.6, rng,
                                      k_peaks=12, mode='topk',
                                      eta_init=0.4, eta_decay=0.99995)
        val_alt = fft_convolve_self(a_alt).max()
        obj_alt = 2.0 * n_alt * val_alt / (a_alt.sum() ** 2)
        print("  n=%d: obj=%.6f" % (n_alt, obj_alt))

        if obj_alt < global_best:
            global_best = obj_alt
            global_a = a_alt.copy()
            print("  New global best at n=%d: obj=%.6f" % (n_alt, global_best))

    # === PHASE 4: Final polishing (remaining time) ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 4: Final polish (%.0fs) ===" % remaining)

    n_final = len(global_a)
    S_final = np.sqrt(2.0 * n_final)

    if remaining > 20:
        # Very targeted subgradient with k=3
        t_f1 = remaining * 0.35
        a_f1, _ = run_subgrad_fista(global_a, S_final, t_f1, rng,
                                     k_peaks=3, mode='topk',
                                     eta_init=0.3, eta_decay=0.99998)
        val_f1 = fft_convolve_self(a_f1).max()
        obj_f1 = 2.0 * n_final * val_f1 / (a_f1.sum() ** 2)
        if obj_f1 < global_best:
            global_best = obj_f1
            global_a = a_f1.copy()
            print("  After final subgrad (k=3): obj=%.6f" % global_best)

    remaining2 = deadline - time.monotonic()
    if remaining2 > 15:
        # Peak conflict search for final tweaks
        t_f2 = remaining2 * 0.4
        a_f2, _ = run_peak_conflict_search(global_a, S_final, t_f2, rng)
        val_f2 = fft_convolve_self(a_f2).max()
        obj_f2 = 2.0 * len(a_f2) * val_f2 / (a_f2.sum() ** 2)
        if obj_f2 < global_best:
            global_best = obj_f2
            global_a = a_f2.copy()
            print("  After final peak conflict: obj=%.6f" % global_best)

    remaining3 = deadline - time.monotonic()
    if remaining3 > 10:
        # Final LSE polish with high beta
        t_f3 = remaining3 * 0.5
        a_f3, _ = run_cyclical_lse(global_a, S_final, t_f3, rng)
        val_f3 = fft_convolve_self(a_f3).max()
        obj_f3 = 2.0 * len(a_f3) * val_f3 / (a_f3.sum() ** 2)
        if obj_f3 < global_best:
            global_best = obj_f3
            global_a = a_f3.copy()
            print("  After final LSE: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Final: length=%d, obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
