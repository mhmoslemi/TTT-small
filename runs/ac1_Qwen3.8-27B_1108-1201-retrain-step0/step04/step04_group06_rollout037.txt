import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def flip_lag(a, k):
    """Compute flip_k(a)[j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    result = np.zeros(n)
    # j ranges from max(0, k-n+1) to min(k, n-1)
    j_lo = max(0, k - n + 1)
    j_hi = min(k, n - 1)
    if j_lo <= j_hi:
        result[j_lo:j_hi+1] = a[k - j_hi:k - j_lo:-1]
    return result

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    """One subgradient step using top-k lags."""
    n = len(a)
    c = fft_convolve_self(a)
    # Get top-k lags
    top_indices = np.argsort(c)[::-1][:top_k]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = fft(a, fsize)

def smoothed_max_grad(a, beta):

    Compute gradient of smoothed max of conv(a,a).

    Returns (grad, smoothed_max_value, c_full).

    exp_c = np.exp(beta * (c - c_max))

def spectral_flatten(a, S, N, flatten_frac=0.1, rng=None):

def structured_init(n, rng, method="rand"):

    elif method == "flat_spec":

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient descent with
exact gradient formula (2*flip_k(a)), backtracking line search, efficient
coordinate descent with O(n) convolution updates, spectral domain global moves,
and multi-length exploration. Key innovation: closed-form gradient of each

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_convolution_gradient(a, top_k, weights=None):
    """
    Compute weighted subgradient of max_k f_k(a) where f_k(a) = sum_i a[i]*a[k-i].
    Gradient of f_k at a is: (grad)[j] = 2*a[k-j] for valid k-j, else 0.
    Uses top_k lags with given weights.
    Returns (gradient, c_full, c_max).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()

    if weights is None:
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        w_sum = top_vals.sum()
        if w_sum < 1e-30:
            return np.zeros(n), c, c_max
        weights = top_vals / w_sum

    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        k = int(k)
        wt = weights[i]
        if abs(wt) < 1e-15:
            continue
        j_lo = max(0, k - n + 1)

        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wt * 2.0 * a[k - j_hi:k - j_lo:-1]

    return grad, c, c_max

            active = np.where(sm_weights > 1e-6)[0]
            grad = np.zeros(n)
            for k in active:
                k = int(k)
                wt = sm_weights[k]
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                if j_lo <= j_hi:
                    grad[j_lo:j_hi + 1] += wt * 2.0 * y[k - j_hi:k - j_lo:-1]
            cur_max = c_max_val
        else:
            grad, c, cur_max = compute_convolution_gradient(y, top_k)

        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step_try = step
        best_a_try = None
        best_val_try = cur_max
        for ls in range(10):
            x_new = project_simplex(y - step_try * grad / gnorm, S)
            c_new = fft_convolve_self(x_new)
            val_new = c_new.max()
            if val_new < best_val_try:
                best_val_try = val_new
                best_a_try = x_new.copy()
            if val_new < cur_max * (1 - 0.001 * ls):
                break
            step_try *= 0.5

        if best_a_try is not None:
            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t))
            momentum = (momentum_t - 1.0) / t_new
            y = best_a_try + momentum * (best_a_try - a)
            y = np.maximum(y, 0.0)
            a = best_a_try
            momentum_t = t_new

            if best_val_try < best_score:
                best_score = best_val_try
                best_local = best_a_try.copy()
            # Increase step size if we found improvement
            step = min(step * 1.05, 2.0 * S)
        else:
            # No improvement, decrease step
            step *= 0.7
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            a = y.copy()
            momentum_t = 1.0

        # Periodic spectral move
        if it % 300 == 0 and rng is not None:
            a_pad = np.zeros(N)
            a_pad[:n] = a
            A = fft(a_pad)
            P = np.abs(A) ** 2
            for frac in [0.05, 0.1, 0.15]:
                P_flat = (1 - frac) * P + frac * P.mean()
                A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
                a_spec = np.real(ifft(A_new))[:n]
                a_spec = project_simplex(a_spec, S)
                c_spec = fft_convolve_self(a_spec)
                if c_spec.max() < best_score:
                    best_score = c_spec.max()
                    best_local = a_spec.copy()
                    a = a_spec.copy()
                    y = a.copy()
                    momentum_t = 1.0
                    break

        # Large perturbation for exploration
        if it % 1500 == 0 and rng is not None:
            sigma = 0.05 * S / np.sqrt(n)
            a_pert = project_simplex(a + rng.normal(0, sigma, n), S)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_score:
                best_score = c_pert.max()
                best_local = a_pert.copy()
                a = a_pert.copy()
                y = a.copy()
                momentum_t = 1.0

        if verbose and it % 500 == 0:
            print("    it=%d, score=%.6f, step=%.4f" % (it, best_score, step))

    return best_local, best_score


def coord_desc_efficient(a_init, S, max_time, n_top=40, rng=None):
    """
    Efficient coordinate descent using O(n) convolution updates per coordinate change.
    When a[j] changes by delta, the convolution changes by 2*delta*a[k-j] (shifted).
    """
    n = len(a_init)
    a = a_init.copy()
    size = 2 * n - 1
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    fractions = [-0.5, -0.3, -0.15, 0.15, 0.3, 0.5]

    for it in range(200):
        if time.monotonic() - start > max_time:
            break

        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find peak lags
        peak_lags = np.argsort(c)[::-1][:8]

        any_improved = False
        for k_peak in peak_lags:
            k_peak = int(k_peak)
            j_lo = max(0, k_peak - n + 1)
            j_hi = min(k_peak, n - 1)
            if j_lo > j_hi:
                continue

            js = np.arange(j_lo, j_hi + 1)
            contribs = a[js] * a[k_peak - js]
            top_js = js[np.argsort(contribs)[::-1]][:n_top]

            for j in top_js:
                j = int(j)
                orig_val = a[j]
                if orig_val < 1e-10:
                    continue

                best_delta = 0.0
                best_cmax = c_max

                # Precompute effect vector: effect[k] = a[k-j] for valid k-j
                # k ranges from j to j+n-1
                k_start = j
                k_end = min(j + n, size)
                len_eff = k_end - k_start

                for frac in fractions:
                    delta = frac * orig_val
                    if orig_val + delta < 0:
                        continue
                    # New c = c + 2*delta*effect (for k in [k_start, k_end))
                    # + delta^2 at k=2j
                    # Compute new max efficiently
                    # c_new[k] = c[k] + 2*delta*a[k-j] for k in [k_start, k_end)
                    increment = 2.0 * delta * a[:len_eff]
                    # We need max of (c + increment) over all k
                    # But increment is zero outside [k_start, k_end)
                    # So new_max = max(max(c[k_start:k_end] + increment, increment...), max(c outside range))
                    inner_max = np.max(c[k_start:k_end] + increment)
                    outer_max = 0.0
                    if k_start > 0:
                        outer_max = max(outer_max, np.max(c[:k_start]))
                    if k_end < size:
                        outer_max = max(outer_max, np.max(c[k_end:]))
                    new_cmax = max(inner_max, outer_max)
                    # Add delta^2 correction at k=2j
                    if 2 * j < size:
                        new_cmax = max(new_cmax, c[2 * j] + 2.0 * delta * a[0] + delta * delta)

                    if new_cmax < best_cmax - 1e-12:
                        best_cmax = new_cmax
                        best_delta = delta

                if best_delta != 0.0:
                    a[j] += best_delta
                    any_improved = True

        if not any_improved:
            # Perturb to escape
            if rng is not None:
                n_pert = min(200, n)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, size=n_pert)
                a = project_simplex(a, S)

        # Recompute and renormalize
        a = project_simplex(a, S)
        c = fft_convolve_self(a)

    return best_a, best_score


def spectral_optimize(a_init, S, max_time, rng=None):
    """
    Pure spectral domain optimization: iteratively flatten the power spectrum
    while maintaining non-negativity via projection.
    """
    n = len(a_init)
    N = 1 << (2 * n - 1).bit_length()
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2

        # Flatten: reduce peaks, boost valleys
        P_mean = P.mean()
        P_max = P.max()
        P_min = P.min()

        if P_max / max(P_mean, 1e-30) > 1.01:
            # Aggressive flattening
            alpha = 0.1 + 0.2 * (P_max / P_mean - 1.0)
            alpha = min(alpha, 0.5)
            P_new = (1 - alpha) * P + alpha * P_mean
        else:
            alpha = 0.05
            P_new = (1 - alpha) * P + alpha * P_mean

        # Also apply a "water-filling" in time domain via the autocorrelation
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        if c_max > 0:
            # Reduce power at frequencies that contribute to the peak lag
            k_peak = int(np.argmax(c))
            m = np.arange(N)
            cos_peak = np.cos(2.0 * np.pi * m * k_peak / N)
            P_new = P_new * (1.0 - 0.05 * np.maximum(cos_peak, 0.0))

        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a_new.copy()
            a = a_new.copy()
        elif it % 100 == 0 and rng is not None:
            # Perturb and retry
            a_pert = project_simplex(a_new + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_score:
                best_score = c_pert.max()
                best_local = a_pert.copy()
                a = a_pert.copy()

    return best_local, best_score


def structured_init(n, rng, method="rand"):
    """Generate structured initializations."""
    if method == "rand":
        a = 0.3 + 0.7 * rng.random(n)
    elif method == "sin":
        t = np.arange(n) / n
        a = (0.4 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t)
             + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        t = np.arange(n) / n
        a = 0.4 + 0.4 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "ramp":
        t = np.arange(n) / n
        a = 0.3 + 0.7 * t + 0.1 * rng.standard_normal(n)
    elif method == "flat_spec":
        N = 1 << (2 * n - 1).bit_length()
        phases = rng.uniform(0, 2 * np.pi, N)
        A = np.exp(1j * phases)
        A[0] = 1.0
        a = np.abs(np.real(ifft(A))[:n])
    elif method == "binomial":
        # Approximate binomial distribution (raised cosine shape)
        t = np.arange(n) / n
        a = np.cos(np.pi * (t - 0.5)) ** 2 + 0.1 * rng.standard_normal(n)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.4) + 0.1 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.5 * rng.random(n)
    a = np.clip(a, 0.001, None)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive subgradient with exact gradient formula, efficient coordinate
    descent with O(n) convolution updates, spectral optimization, and
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, n):
        nonlocal global_best, global_a
        a_candidate = safe_normalize(a_candidate, n)
        c = fft_convolve_self(a_candidate)
        score = 2.0 * n * c.max() / (a_candidate.sum() ** 2)
        if score < global_best:
            global_best = score
            global_a = a_candidate.copy()
            return True
        return False

    # Load previous best if available
    prev_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 100:
            given = np.clip(given, 0, 1000)
            if given.sum() > 0:
                prev_seq = given
                print("Loaded previous sequence: n=%d" % len(prev_seq))
    except Exception:
        pass

    # Choose sequence lengths to try
    lengths = [8192]
    if budget_s > 1500:
        lengths.append(16384)

    time_per_length = (deadline - time.monotonic()) / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * float(n))
        print("\n=== n=%d, S=%.4f ===" % (n, S))

        # Build initializations
        inits = []

        if prev_seq is not None:
            if len(prev_seq) == n:
                inits.append(("prev", prev_seq.copy()))
            else:
                # Interpolate to target length
                x_old = np.linspace(0, 1, len(prev_seq))
                x_new = np.linspace(0, 1, n)
                from scipy.interpolate import interp1d
                f_interp = interp1d(x_old, prev_seq, kind='cubic', fill_value='extrapolate')
                resampled = np.maximum(f_interp(x_new), 0)
                inits.append(("prev_interp", resampled))

        for method in ["rand", "sin", "chirp", "binomial", "flat_spec", "alternating"]:
            inits.append((method, structured_init(n, rng, method)))

        n_inits = len(inits)
        time_per_init = time_per_length / n_inits

        # Score all initializations
        for name, arr in inits:
            a_norm = safe_normalize(arr, n)
            c0 = fft_convolve_self(a_norm)
            sc0 = 2.0 * n * c0.max() / S ** 2
            if sc0 < global_best:
                global_best = sc0
                global_a = a_norm.copy()
                print("  [%s] init: %.6f" % (name, global_best))

        # Run optimization on each initialization
        for name, arr in inits:
            if time.monotonic() > deadline - 60:
                break

            a_norm = safe_normalize(arr, n)
            rem = deadline - time.monotonic() - 60
            t_budget = min(time_per_init, rem * 0.6)

            if t_budget < 3:
                continue

            # Phase A: Adaptive subgradient (40% of budget)
            t_A = min(t_budget * 0.4, 60.0)
            a_sub, sc_sub = adaptive_subgradient(a_norm, S, t_A, top_k=20, rng=rng)
            if update_best(a_sub, n):
                print("  [%s] subgrad: %.6f" % (name, global_best))
            a_curr = a_sub.copy()

            # Phase B: Coordinate descent (30% of budget)
            t_B = min(t_budget * 0.3, 45.0)
            a_cd, sc_cd = coord_desc_efficient(a_curr, S, t_B, n_top=50, rng=rng)
            if update_best(a_cd, n):
                print("  [%s] coord: %.6f" % (name, global_best))
            a_curr = a_cd.copy()

            # Phase C: Spectral optimization (20% of budget)
            t_C = min(t_budget * 0.2, 30.0)
            a_spec, sc_spec = spectral_optimize(a_curr, S, t_C, rng=rng)
            if update_best(a_spec, n):
                print("  [%s] spectral: %.6f" % (name, global_best))
            a_curr = a_spec.copy()

            # Phase D: Final subgradient polish (10%)
            t_D = min(t_budget * 0.1, 20.0)
            if t_D > 2:
                a_fin, sc_fin = adaptive_subgradient(a_curr, S, t_D, top_k=30, rng=rng)
                if update_best(a_fin, n):
                    print("  [%s] final subgrad: %.6f" % (name, global_best))

    # Final intensive polish on the best
    if global_a is not None and time.monotonic() < deadline - 15:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        print("\n=== Final polish (n=%d, current best=%.6f) ===" % (n_best, global_best))

        a_work = global_a.copy()

        # Iterative: subgrad -> coord -> subgrad -> coord
        n_iters = 6
        time_per_polish = (deadline - time.monotonic() - 10) / n_iters

        for pi in range(n_iters):
            if time.monotonic() > deadline - 10:
                break
            rem_t = min(time_per_polish, (deadline - time.monotonic() - 10) * 0.9)
            if rem_t < 3:
                break

            # Alternate between subgradient and coordinate descent
            if pi % 2 == 0:
                a_new, _ = adaptive_subgradient(a_work, S_best, rem_t * 0.5,
                                                top_k=30, rng=rng)
            else:
                a_new, _ = coord_desc_efficient(a_work, S_best, rem_t * 0.5,
                                                n_top=60, rng=rng)
            if update_best(a_new, n_best):
                a_work = global_a.copy()
                print("  polish[%d]: %.6f" % (pi, global_best))

            # Small spectral move between
            rem_t2 = (deadline - time.monotonic() - 10) * 0.3
            if rem_t2 > 3:
                a_s, _ = spectral_optimize(a_work, S_best, rem_t2, rng=rng)
                if update_best(a_s, n_best):
                    a_work = global_a.copy()

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        # Ensure all values are valid
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("\nFinal verified score: %.6f" % ver)
            else:
                print("\nVerification failed, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception:
            print("\nVerification error, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
