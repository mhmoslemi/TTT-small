"""
Minimize 2*n*max_conv/sum^2 using a hybrid bundle method with LP subproblems,
spectral shaping kicks, multi-lag coordinate descent, and adaptive restarts.

Key innovations:
1. Bundle method: accumulate linearizations (cuts) of the objective and solve
   an LP at each step to find the next iterate. This is more principled than
   pure subgradient and provides better search directions.
2. Multi-lag coordinate descent: simultaneously reduce contributions from the
   top-k peak lags rather than just one.
3. Adaptive spectral kicks: when the bundle method stalls, apply targeted
   spectral modifications to escape local minima.
4. Multi-n search: try several sequence lengths and keep the best.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, peak_lag):
    """Compute subgradient of conv(a,a)[peak_lag] with respect to a."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    # conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d(a[j]) = a[k-j] + a[j] (if j != k-j) or 2*a[j] (if j = k-j)
    # More simply: it's the cross-correlation of a with the unit vector at lag k
    # which gives: grad[j] = a[k-j] (if valid) + a[j] (if valid and j != k-j)
    # Actually the cleanest way: grad = 2 * cyclic_shift(a, k) truncated
    # Let's use FFT: grad[j] = sum over the "other" index
    # conv(a,a)[k] = <a, shift_k(a)> where (shift_k(a))[i] = a[k-i]
    # d/d(a[j]) = shift_k(a)[j] + a[j] (accounting for symmetry)
    # = a[k-j] + a[j] if both valid, with care at boundaries
    
    # Simplest correct computation:
    grad = np.zeros(n)
    for j in range(n):
        other = peak_lag - j
        if 0 <= other < n:
            grad[j] += a[other]
        # The term a[j]*a[other] contributes a[other] to d/d(a[j])
        # If other == j, we have a[j]^2, derivative is 2*a[j], but we already added a[j] once
        # Actually: conv[k] = sum_i a[i]*a[k-i]. d/d(a[j]):
        #   term where i=j: a[k-j] (if k-j valid)
        #   term where k-i=j, i=k-j: a[j] (if k-j valid and k-j != j)
        #   if k-j == j: only one term a[j]*a[j], derivative 2*a[j]
        if 0 <= other < n and other != j:
            grad[j] += a[j]
        elif 0 <= other < n and other == j:
            grad[j] += a[j]  # total: a[j] + a[j] = 2*a[j]
    return grad

def spectral_shape(a, N, n, top_k, alpha, rng, target):
    """Spectral shaping: reduce power at frequencies that contribute to peak lags."""
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target:
        return a, c_max

    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max

    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total

    weights = excess / excess_sum

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    alpha_eff = alpha * (1.0 + 0.5 * (c_max - target) / max(c_max, 1e-10))
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    if rng.random() < 0.15:
        phase_noise = rng.normal(0, 0.03 * alpha_eff, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def bundle_lp_step(a_curr, bundle_a, bundle_g, bundle_c, S, n, damping=1.0):
    """
    Solve the bundle LP and return the new point.
    
    min t
    s.t. t >= c_i + g_i^T (a - a_i) for all i in bundle
         a >= 0
         sum(a) = S
    """
    m = len(bundle_c)
    if m == 0:
        return a_curr.copy()
    
    # Variables: [a[0], ..., a[n-1], t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: g_i^T a - t <= g_i^T a_i - c_i
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i in range(m):
        A_ub[i, :n] = bundle_g[i]
        A_ub[i, n] = -1.0
        b_ub[i] = bundle_g[i] @ bundle_a[i] - bundle_c[i]
    
    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            a_new = res.x[:n]
            a_new = np.maximum(a_new, 0.0)
            # Rescale to ensure sum = S (numerical correction)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * (S / s)
            # Damped step
            a_result = (1 - damping) * a_curr + damping * a_new
            a_result = project_simplex(a_result, S)
            return a_result
    except Exception:
        pass
    
    return a_curr.copy()

def nesterov_subgradient(a_init, n, S, max_time, rng, top_k=10):
    """Nesterov accelerated subgradient with adaptive step size."""
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step_base = 2.0 * S / (n + 1)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Compute weighted subgradient over top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Combine subgradients
        grad = np.zeros(n)
        for i, k in enumerate(top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for ii in range(lo, hi + 1):
                jj = k - ii
                if 0 <= jj < n:
                    if ii == jj:
                        grad[ii] += weights[i] * 2.0 * a[ii]
                    else:
                        grad[ii] += weights[i] * a[jj]
                        grad[jj] += weights[i] * a[ii]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size (backtracking)
        step = step_base / (1.0 + 0.01 * it)
        direction = -grad / gnorm
        
        # Try the step
        x_new = project_simplex(y + step * direction, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        
        # Occasional restart
        if it % 1000 == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_sequence(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    S = a.sum()
    return float(2.0 * n * c_max / (S ** 2))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses bundle method with LP, spectral shaping, Nesterov subgradient,
    and multi-lag coordinate descent.
    """
    global rng_local
    rng = np.random.default_rng(seed)
    rng_local = rng
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    # Try multiple lengths
    n_candidates = [8192, 4096]
    
    global_best = np.inf
    global_a = None
    global_n = 0
    
    def try_update(a_candidate, n):
        nonlocal global_best, global_a, global_n
        a_candidate = safe_normalize(a_candidate, n)
        sc = score_sequence(a_candidate)
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            global_n = n
            return True
        return False
    
    # Build initializations from height_sequence_1
    init_arrays = {}
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        for n in n_candidates:
            if len(given) >= n:
                init_arrays[n] = given[:n].copy()
            else:
                # Pad or interpolate
                arr = np.zeros(n)
                arr[:len(given)] = given
                arr[len(given):] = given[-1] if len(given) > 0 else 1.0
                init_arrays[n] = arr
    except Exception:
        pass
    
    # Add structured initializations
    t = np.linspace(0, 1, 8192)
    sin_init = 0.5 + 0.25 * np.sin(2 * np.pi * 3 * t) + 0.15 * np.sin(2 * np.pi * 7 * t) + 0.1 * np.sin(2 * np.pi * 11 * t)
    alternating_init = np.where(np.arange(8192) % 2 == 0, 0.8 + 0.1 * rng.standard_normal(4096), 0.4 + 0.1 * rng.standard_normal(4096))
    for n in n_candidates:
        if n not in init_arrays:
            init_arrays[n] = np.clip(sin_init[:n], 0.01, None)
    
    print("Starting optimization with budget=%.0fs" % budget_s)
    
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        N = 1 << (2 * n - 1).bit_length()
        S = np.sqrt(2.0 * float(n))
        
        print("\n=== n=%d, S=%.4f, N=%d ===" % (n, S, N))
        
        a = safe_normalize(init_arrays[n], n)
        sc = score_sequence(a)
        print("  Initial score: %.6f" % sc)
        try_update(a, n)
        
        N_lags = len(fft_convolve_self(a))
        
        # === Phase 1: Quick spectral shaping (5% time per n) ===
        phase1_time = min(60.0, (deadline - time.monotonic()) * 0.05)
        t0 = time.monotonic()
        best_local_a = a.copy()
        best_local_sc = sc
        
        while time.monotonic() - t0 < phase1_time:
            top_k = min(20, 5 + int((time.monotonic() - t0) / phase1_time * 15))
            alpha = 0.4 / (1.0 + 0.1 * (time.monotonic() - t0))
            a_new, _ = spectral_shape(a, N, n, top_k, alpha, rng, target=1.0)
            sc_new = score_sequence(a_new)
            if sc_new < best_local_sc:
                best_local_sc = sc_new
                best_local_a = a_new.copy()
                a = a_new
            else:
                a = a_new  # still move forward
            if sc_new < global_best:
                global_best = sc_new
                global_a = a.copy()
                global_n = n
                print("  [P1-spect] %.6f" % global_best)
        
        a = best_local_a
        sc = best_local_sc
        print("  After spectral shaping: %.6f" % sc)
        
        # === Phase 2: Bundle method with LP (45% time per n) ===
        phase2_time = min(600.0, (deadline - time.monotonic()) * 0.45)
        t0 = time.monotonic()
        
        # Initialize bundle
        bundle_a = [a.copy()]
        bundle_g = [np.zeros(n)]
        bundle_c = [sc]
        
        # Compute initial subgradient
        c = fft_convolve_self(a)
        peak = int(np.argmax(c))
        bundle_g[0] = compute_subgradient_fast(a, peak)
        
        best_a = a.copy()
        best_sc = sc
        stall_count = 0
        it = 0
        max_bundle = 80 if n <= 4096 else 50
        
        while time.monotonic() - t0 < phase2_time:
            it += 1
            
            # Solve LP
            damping = min(1.0, 0.3 + 0.7 * it / 50)
            a_new = bundle_lp_step(a, bundle_a, bundle_g, bundle_c, S, n, damping)
            
            # Evaluate
            c_new = fft_convolve_self(a_new)
            sc_new = c_new.max()
            score = 2.0 * n * sc_new / S ** 2
            
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                stall_count = 0
                if it % 20 == 0:
                    print("  [P2-bundle] it=%d, max_conv=%.6f, score=%.6f" % (it, best_sc, 2.0*n*best_sc/S**2))
            else:
                stall_count += 1
            
            a = a_new
            
            # Update bundle
            peak = int(np.argmax(c_new))
            g = compute_subgradient_fast(a, peak)
            bundle_a.append(a.copy())
            bundle_g.append(g)
            bundle_c.append(sc_new)
            
            # Manage bundle size
            if len(bundle_c) > max_bundle:
                # Keep the best and most recent
                best_idx = int(np.argmin(bundle_c))
                keep = list(range(len(bundle_c)))
                keep.remove(best_idx)
                # Remove oldest half of remaining
                remove = keep[:len(keep)//2]
                for idx in sorted(remove, reverse=True):
                    bundle_a.pop(idx)
                    bundle_g.pop(idx)
                    bundle_c.pop(idx)
            
            # If stalled, apply spectral kick
            if stall_count >= 15:
                a_kick, _ = spectral_shape(a, N, n, 10, 0.5, rng, target=1.0)
                a_kick = project_simplex(a_kick, S)
                sc_kick = fft_convolve_self(a_kick).max()
                if sc_kick < best_sc:
                    best_sc = sc_kick
                    best_a = a_kick.copy()
                    a = a_kick
                    bundle_a = [a.copy()]
                    bundle_g = [compute_subgradient_fast(a, int(np.argmax(fft_convolve_self(a))))]
                    bundle_c = [sc_kick]
                    print("  [P2-kick] %.6f" % best_sc)
                else:
                    # Perturb and reset bundle
                    a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
                    bundle_a = [a.copy()]
                    c_tmp = fft_convolve_self(a)
                    bundle_g = [compute_subgradient_fast(a, int(np.argmax(c_tmp)))]
                    bundle_c = [c_tmp.max()]
                stall_count = 0
            
            # Periodic Nesterov burst
            if it % 50 == 25:
                a_ng, sc_ng = nesterov_subgradient(a, n, S, min(5.0, (t0 + phase2_time) - time.monotonic()), rng, top_k=5)
                if sc_ng < best_sc:
                    best_sc = sc_ng
                    best_a = a_ng.copy()
                    a = a_ng
                    bundle_a = [a.copy()]
                    bundle_g = [compute_subgradient_fast(a, int(np.argmax(fft_convolve_self(a))))]
                    bundle_c = [sc_ng]
                    print("  [P2-nesterov] %.6f" % best_sc)
        
        a = best_a
        sc = best_sc
        print("  After bundle method: %.6f" % (2.0*n*sc/S**2))
        
        # Update global
        if 2.0 * n * sc / S ** 2 < global_best:
            global_best = 2.0 * n * sc / S ** 2
            global_a = a.copy()
            global_n = n
        
        # === Phase 3: Multi-lag coordinate descent (15% time per n) ===
        phase3_time = min(120.0, (deadline - time.monotonic()) * 0.15)
        if phase3_time > 5:
            t0 = time.monotonic()
            a_cd, sc_cd = multi_lag_coord_descent(a, n, S, phase3_time, n_top=15, n_lags=3)
            score_cd = 2.0 * n * sc_cd / S ** 2
            if score_cd < global_best:
                global_best = score_cd
                global_a = a_cd.copy()
                global_n = n
                print("  [P3-coord] %.6f" % global_best)
            a = a_cd
            sc = sc_cd
            print("  After coordinate descent: %.6f" % (2.0*n*sc/S**2))
        
        # === Phase 4: Final Nesterov polish (10% time per n) ===
        phase4_time = min(60.0, (deadline - time.monotonic()) * 0.10)
        if phase4_time > 5:
            a_ng, sc_ng = nesterov_subgradient(a, n, S, phase4_time, rng, top_k=20)
            score_ng = 2.0 * n * sc_ng / S ** 2
            if score_ng < global_best:
                global_best = score_ng
                global_a = a_ng.copy()
                global_n = n
                print("  [P4-nesterov] %.6f" % global_best)
    
    # === Final phase: intensive polish on the global best ===
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        n_best = global_n
        S_best = np.sqrt(2.0 * float(n_best))
        N_best = 1 << (2 * n_best - 1).bit_length()
        a = global_a.copy()
        
        print("\n=== Final polish (remaining %.0fs, n=%d) ===" % (remaining, n_best))
        
        # Alternate between methods
        methods = ['bundle', 'spectral', 'nesterov', 'coord']
        seg_time = remaining / 8.0
        for round_num in range(4):
            if time.monotonic() > deadline:
                break
            seg = min(seg_time, deadline - time.monotonic())
            if seg < 3:
                break
            
            # Bundle
            if seg > 5:
                t0 = time.monotonic()
                bundle_a = [a.copy()]
                c_tmp = fft_convolve_self(a)
                bundle_g = [compute_subgradient_fast(a, int(np.argmax(c_tmp)))]
                bundle_c = [c_tmp.max()]
                it = 0
                max_b = 40
                while time.monotonic() - t0 < seg * 0.5:
                    it += 1
                    a_new = bundle_lp_step(a, bundle_a, bundle_g, bundle_c, S_best, n_best, 1.0)
                    c_new = fft_convolve_self(a_new)
                    sc_new = c_new.max()
                    score = 2.0 * n_best * sc_new / S_best ** 2
                    if sc_new < (global_a.sum()**2 / (2.0 * n_best) * global_best):
                        a = a_new
                        bundle_a.append(a.copy())
                        bundle_g.append(compute_subgradient_fast(a, int(np.argmax(c_new))))
                        bundle_c.append(sc_new)
                        if len(bundle_c) > max_b:
                            bundle_a = bundle_a[::2] + [bundle_a[-1]]
                            bundle_g = bundle_g[::2] + [bundle_g[-1]]
                            bundle_c = bundle_c[::2] + [bundle_c[-1]]
                    else:
                        a = a_new
                        bundle_a.append(a.copy())
                        bundle_g.append(compute_subgradient_fast(a, int(np.argmax(c_new))))
                        bundle_c.append(sc_new)
                        if len(bundle_c) > max_b:
                            bundle_a = bundle_a[-max_b:]
                            bundle_g = bundle_g[-max_b:]
                            bundle_c = bundle_c[-max_b:]
                    if score < global_best:
                        global_best = score
                        global_a = a.copy()
                        print("  [Final-bundle] %.6f" % global_best)
            
            # Spectral
            seg2 = deadline - time.monotonic()
            if seg2 > 3:
                for _ in range(20):
                    if time.monotonic() > deadline - 2:
                        break
                    a_new, _ = spectral_shape(a, N_best, n_best, 15, 0.3, rng, target=1.0)
                    sc_new = fft_convolve_self(a_new).max()
                    score = 2.0 * n_best * sc_new / S_best ** 2
                    if score < global_best:
                        global_best = score
                        global_a = a_new.copy()
                        a = a_new
                        print("  [Final-spectral] %.6f" % global_best)
            
            # Nesterov
            seg3 = deadline - time.monotonic()
            if seg3 > 3:
                a_ng, sc_ng = nesterov_subgradient(a, n_best, S_best, min(seg3 * 0.3, 30), rng, top_k=30)
                score = 2.0 * n_best * sc_ng / S_best ** 2
                if score < global_best:
                    global_best = score
                    global_a = a_ng.copy()
                    a = a_ng
                    print("  [Final-nesterov] %.6f" % global_best)
    
    # Final coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_a is not None:
        n_best = global_n
        S_best = np.sqrt(2.0 * float(n_best))
        a_cd, sc_cd = multi_lag_coord_descent(global_a, n_best, S_best, min(remaining - 2, 30), n_top=20, n_lags=4)
        score = 2.0 * n_best * sc_cd / S_best ** 2
        if score < global_best:
            global_best = score
            global_a = a_cd.copy()
            print("  [Final-coord] %.6f" % global_best)
    
    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    print("Returning sequence of length %d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
