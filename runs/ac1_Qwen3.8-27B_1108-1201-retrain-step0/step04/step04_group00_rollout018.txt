import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_value_and_gradient(a, beta):
    """Compute LSE-smoothed max convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    w = e / tot
    # LSE value: cm + log(tot)/beta
    lse_val = cm + np.log(tot) / beta
    # Gradient
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, w, c

def normalized_lse(a, beta):
    """Normalized LSE objective and gradient."""
    S = a.sum()
    if S < 1e-30:
        return 1e10, np.zeros_like(a)
    lse_val, grad, w, c = lse_value_and_gradient(a, beta)
    obj = lse_val / (S * S)
    grad_obj = (grad * S - 2.0 * lse_val / S) / S
    return obj, grad_obj

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def greedy_init(n, S):
    """Generate the greedy initialization."""
    t = S * S / (2 * n - 1)
    a = np.zeros(n)
    a[0] = S / np.sqrt(2 * n - 1)
    for k in range(1, n):
        # c[k] should equal t
        # c[k] = 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i]
        cross = 0.0
        for i in range(1, k):
            cross += a[i] * a[k-i]
        a[k] = (t - cross) / (2 * a[0])
        a[k] = max(0, a[k])
    return a

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

    """Run L-BFGS-B on the normalized LSE objective."""
    n = len(a_init)
    x0 = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000.0)
    
    def fun_and_grad(x):
        S = x.sum()
        if S < 1e-30:
            return 1e10, np.zeros(n)
        c = fft_convolve_self(x)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        lse_val = cm + np.log(tot) / beta
        obj = lse_val / (S * S)
        
        w = e / tot

        W = fft(w, fsize)
        A = fft(x, fsize)
        grad_c = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        grad_obj = (grad_c * S - 2.0 * lse_val / S) / S
        return obj, grad_obj
    
    start = time.monotonic()
    result = minimize(fun_and_grad, x0, method='L-BFGS-B',
                      bounds=[(0, 1000)] * n,
                      options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                               'maxls': 50})

    return result.x, result.fun, elapsed

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-method optimization combining:
1. Greedy recursive initialization (equalizes early convolution lags)
2. L-BFGS-B second-order optimization on smoothed objective (continuation in beta)
3. FISTA with adaptive LSE for high-beta polishing
4. Active-set top-k gradient refinement
5. Multi-scale exploration (n=8192 and 16384)
Key differentiator: L-BFGS-B exploits curvature info for faster convergence
near the optimum, and the greedy init provides a structurally informed starting point.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def greedy_init(n, S):
    """Generate greedy initialization that equalizes early convolution lags."""
    t = S * S / (2.0 * n - 1)
    a = np.zeros(n)
    a[0] = S / np.sqrt(2.0 * n - 1)
    max_k = min(n, 200)
    for k in range(1, max_k):
        cross = 0.0
        for i in range(1, k):
            cross += a[i] * a[k - i]
        a[k] = (t - cross) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0
        if a[k] == 0 and k > 10:
            break
    # Fill remaining with small values to ensure sum > 0
    current_sum = a.sum()
    if current_sum < 0.01 * S:
        a = np.ones(n) * S / n
    return a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_lbfgs(a_init, beta, max_iter=1500, time_limit=120.0):
    """Run L-BFGS-B on the normalized LSE-smoothed objective."""
    n = len(a_init)
    x0 = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000.0)
    
    call_count = [0]
    
    def fun_and_grad(x):
        call_count[0] += 1
        S = x.sum()
        if S < 1e-30:
            return 1e10, np.zeros(n)
        c = fft_convolve_self(x)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            return 1e10, np.zeros(n)
        lse_val = cm + np.log(tot) / beta
        obj = lse_val / (S * S)
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(x, fsize)
        grad_c = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        grad_obj = (grad_c * S - 2.0 * lse_val / S) / S
        # Clip gradient for stability
        gnorm = np.linalg.norm(grad_obj)
        if gnorm > 1e6:
            grad_obj = grad_obj / gnorm * 1e6
        return obj, grad_obj

    start = time.monotonic()
    try:
        result = minimize(fun_and_grad, x0, method='L-BFGS-B',
                          bounds=[(0.0, 1000.0)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                   'maxls': 100, 'gtol': 1e-8})
    except Exception:
        result = None
    elapsed = time.monotonic() - start
    if result is not None and result.x is not None:
        return result.x, elapsed
    return x0, elapsed


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.005,
              beta_max=10000.0, step_scale=1.0):
    """FISTA with LSE smoothing and periodic restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        # Process in batches for efficiency
        batch_size = 20
        for _ in range(batch_size):
            if time.monotonic() - start >= max_time:
                break
            it += 1
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))
            step = step_scale / max(beta, 1.0)

            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-14:
                best_conv = cur_max
                best_a = y.copy()
                last_improve = it

            cm = c.max()
            shifted = beta * (c - cm)
            shifted -= shifted.max()
            e = np.exp(shifted)
            tot = e.sum()
            if tot < 1e-300:
                w = np.ones(len(c)) / len(c)
            else:
                w = e / tot
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A_fft = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                return best_a, best_conv

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

        # Periodic restart with noise when stuck
        stuck = it - last_improve
        if stuck > 100:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 3000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """Top-k active set polishing with adaptive step."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        for _ in range(10):
            if time.monotonic() - start >= max_time:
                break
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-14:
                best_conv = cur_max
                best_a = y.copy()
                last_improve = it

            stuck = it - last_improve
            # Adaptive top-k
            if stuck < 100:
                k = 1
            elif stuck < 500:
                k = 3
            elif stuck < 2000:
                k = min(8, 3 + stuck // 200)
            else:
                k = min(32, 8 + stuck // 500)

            idx = np.argsort(c)[::-1][:k]
            w_full = np.zeros(len(c))
            w_full[idx] = 1.0 / k
            fsize = 1 << (len(w_full) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A_fft = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                return best_a, best_conv

            step = S / (gnorm * (1.0 + 0.005 * it))
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

        # Noise when very stuck
        stuck = it - last_improve
        if stuck > 3000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv
