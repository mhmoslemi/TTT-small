Minimize 2*n*max(self_conv)/sum^2 using multi-phase LSE-smoothed optimization
with Nesterov acceleration, adaptive beta scheduling, and L-BFGS refinement.

import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, beta):
    """Compute LSE-smoothed gradient of max(b) where b = a*a."""
    n = len(a)
    size = 2 * n - 1
    c = fft_convolve_self(a)
    c_max = c.max()
    # Numerical stability
    e = np.exp(np.clip(beta * (c - c_max), -700, 700))
    w = e / e.sum()
    # Gradient: grad[i] = sum_k w[k] * a[k-i]
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    c_max_val = c_max + np.log(e.sum()) / beta
    return grad, c_max_val, c

    if s < 1e-15:

    return a * (S / s)

    S = np.sqrt(2.0 * n)
    a = normalize(a_init.copy(), S)
    y = a.copy()

    S = np.sqrt(2.0 * n)

    # This ensures a >= 0 and sum(a) = S automatically
    eps = 1e-10
    z = np.log(np.maximum(a, eps))  # initial z

        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum() * S
        # Compute LSE objective
        c = fft_convolve_self(a)
        c_max = c.max()
        e = np.exp(np.clip(beta * (c - c_max), -700, 700))
        return c_max + np.log(e.sum()) / beta
    
    def gradient(z):
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum() * S
        # LSE gradient w.r.t. a
        c = fft_convolve_self(a)
        c_max = c.max()
        e = np.exp(np.clip(beta * (c - c_max), -700, 700))
        w = e / e.sum()

        # = a[j] * grad_a[j] - (a[j]/S) * sum_i grad_a[i] * a[i]
        dot = np.dot(grad_a, a)
        grad_z = a * grad_a - (a / S) * dot
        return grad_z
    
    from scipy.optimize import minimize
    start = time.monotonic()

    a_opt = a_opt / a_opt.sum() * S
    a_opt = np.maximum(a_opt, 0)
    
    obj = compute_objective(a_opt)
    return a_opt, obj

"""
Minimize 2*n*max(self_conv)/sum^2 using multi-phase optimization:
Phase 1: Nesterov-accelerated LSE-smoothed projected gradient with BB step size
Phase 2: L-BFGS refinement on LSE-smoothed objective (log parameterization)
Phase 3: Active-set refinement (equalize peak constraints)
Phase 4: Fine polishing with near-exact subgradient
Multi-start with diverse initializations and length exploration.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad_and_val(a, beta):
    """Compute LSE-smoothed value and gradient of max(b) where b = a*a."""
    n = len(a)
    size = 2 * n - 1
    c = fft_convolve_self(a)
    c_max = c.max()
    x = np.clip(beta * (c - c_max), -700, 700)
    e = np.exp(x)
    e_sum = e.sum()
    w = e / e_sum
    lse_val = c_max + np.log(e_sum) / beta
    # Gradient: grad[i] = sum_k w[k] * a[k-i] (cross-correlation)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val, c


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a, S):
    """Clip to non-negative and rescale to sum = S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def nesterov_lse_bb(a_init, n, max_time, rng, beta_start=5.0, beta_end=1000.0,
                     momentum_decay=False):
    """
    Nesterov-accelerated projected gradient with LSE smoothing and BB step size.
    """
    S = np.sqrt(2.0 * n)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    beta = beta_start
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S**2
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    stall_count = 0

    # Initial step size estimate
    grad, _, _ = lse_grad_and_val(y, beta)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, best_obj
    step = 1.0 / (beta * S * 2.0)
    step = min(step, S / (n * gnorm + 1e-15) * 0.5)

    # BB step size state
    s_prev = None
    y_grad_prev = None

    max_betas = [beta_start]
    
    while time.monotonic() - start < max_time:
        it += 1
        # Update beta (gradual increase)
        progress = min(1.0, it / 50000.0)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        # Compute gradient at y
        grad, lse_val, c_y = lse_grad_and_val(y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Nesterov step
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        x_new = y - step * grad
        x_new = project_simplex(x_new, S)

        # BB step size update
        if s_prev is not None and y_grad_prev is not None:
            sy = np.dot(s_prev, grad - y_grad_prev)
            ss = np.dot(s_prev, s_prev)
            if ss > 1e-15 and abs(sy) > 1e-15:
                bb_step = ss / abs(sy)
                # Blend with previous step
                step = 0.5 * step + 0.5 * min(bb_step, 5.0 * S / (n * gnorm + 1e-15))
            s_prev = x_new - a
            y_grad_prev = grad

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Evaluate true objective every few iterations
        if it % 3 == 0:
            c_a = fft_convolve_self(a)
            obj = 2.0 * n * c_a.max() / S**2
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                last_improve = it
                stall_count = 0
            else:
                stall_count += 1

        # Adaptive step size
        if stall_count > 0:
            step *= (1.0 - 0.01 * min(stall_count, 50))
        step = max(step, 1e-10 * S / n)
        step = min(step, 2.0 * S / (n * gnorm + 1e-15))

        # Periodic perturbation to escape local minima
        if stall_count > 5000 and rng.random() < 0.1:
            scale = 0.003 * S / np.sqrt(n) * (1.0 + 0.1 * (stall_count // 5000))
            pert = a.copy()
            # Frequency-domain perturbation
            A = fft(pert)
            noise = rng.normal(0, 0.02, len(A)) + 1j * rng.normal(0, 0.02, len(A))
            A = A * (1.0 + noise / (1.0 + np.abs(A) * 100))
            pert = np.real(ifft(A))
            pert = np.maximum(pert, 0)
            a = project_simplex(pert + rng.normal(0, scale, n), S)
            y = a.copy()
            t_mom = 1.0
            step = 0.3 * S / (n * gnorm + 1e-15)
            stall_count = 0
            s_prev = None

        if it % 20000 == 0 and it > 0:
            print("    nesterov it=%d, best=%.8f, beta=%.1f, step=%.4e" % (it, best_obj, beta, step))

    return best_a, best_obj


def lbfgs_refine(a_init, n, max_time, rng, beta=200.0):
    """
    L-BFGS-B refinement using log-sum parameterization.
    a[i] = S * exp(z[i] - max(z)) / sum(exp(z - max(z)))
    """
    S = np.sqrt(2.0 * n)
    a = normalize(a_init.copy(), S)
    eps = 1e-12
    z = np.log(np.maximum(a, eps * S / n))

    n_evals = [0]
    start = time.monotonic()
    best_z_obj = np.inf

    def objective(z):
        n_evals[0] += 1
        if time.monotonic() - start > max_time * 0.9:
            return np.inf
        z_shifted = z - z.max()
        a_local = np.exp(z_shifted)
        a_sum = a_local.sum()
        if a_sum < 1e-300:
            return 1e10
        a_local = a_local / a_sum * S
        c = fft_convolve_self(a_local)
        c_max = c.max()
        x = np.clip(beta * (c - c_max), -700, 700)
        e = np.exp(x)
        return c_max + np.log(e.sum()) / beta

    def gradient(z):
        n_evals[0] += 1
        if time.monotonic() - start > max_time * 0.9:
            return np.zeros(n)
        z_shifted = z - z.max()
        a_local = np.exp(z_shifted)
        a_sum = a_local.sum()
        if a_sum < 1e-300:
            return np.zeros(n)
        a_local = a_local / a_sum * S
        c = fft_convolve_self(a_local)
        c_max = c.max()
        x = np.clip(beta * (c - c_max), -700, 700)
        e = np.exp(x)
        e_sum = e.sum()
        w = e / e_sum
        size = 2 * n - 1
        fsize = 1 << (size + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a_local, fsize)
        grad_a = np.real(ifft(W * np.conj(A)))[:n]
        # Chain rule for log-sum parameterization
        dot = np.dot(grad_a, a_local)
        grad_z = a_local * grad_a - (a_local / S) * dot
        return grad_z

    from scipy.optimize import minimize
    result = minimize(objective, z, jac=gradient, method='L-BFGS-B',
                      options={'maxfun': 200000, 'maxiter': 10000,
                               'ftol': 1e-15, 'gtol': 1e-10,
                               'iprint': 0, 'maxls': 50})

    z_final = result.x
    z_shifted = z_final - z_final.max()
    a_opt = np.exp(z_shifted)
    a_opt = a_opt / a_opt.sum() * S
    a_opt = np.maximum(a_opt, 0.0)

    obj = compute_objective(a_opt)
    print("    L-BFGS: %d evaluations, obj=%.8f" % (n_evals[0], obj))
    return a_opt, obj


def active_set_refine(a_init, n, max_time, rng):
    """
    Active-set refinement: identify constraints near the peak and try to
    equalize them using weighted subgradient.
    """
    S = np.sqrt(2.0 * n)
    a = normalize(a_init.copy(), S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    n = len(a)

    # Adaptive tolerance for "active" constraints
    tol_frac = 0.01  # constraints within 1% of max are "active"

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        max_b = c.max()
        cur_obj = 2.0 * n * max_b / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Find active constraints
        threshold = max_b * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
        n_active = len(active)
        if n_active == 0:
            break

        # Weighted subgradient: weight each active constraint by its "tightness"
        violations = c[active] - threshold
        weights = violations / (violations.sum() + 1e-300) + 1.0 / n_active
        weights = weights / weights.sum()

        # Compute weighted gradient
        size = 2 * n - 1
        fsize = 1 << (size + n - 1).bit_length()
        W_full = np.zeros(size)
        W_full[active] = weights
        W = fft(W_full, fsize)
        A = fft(a, fsize)
        grad = np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: decrease over iterations
        step = 0.5 / (it + 10.0) * S / (n * gnorm / (S / n) + 1e-15)
        step = min(step, S / (n * gnorm + 1e-15) * 0.1)

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2

        if obj_new < cur_obj:
            a = a_new
        else:
            # Try smaller step
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / S**2
            if obj_new < cur_obj:
                a = a_new
            else:
                # Perturb
                if it - last_improve > 5000:
                    scale = 0.002 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, scale, n), S)
                    last_improve = it

        # Gradually tighten the active set tolerance
        if it % 5000 == 0:
            tol_frac = max(0.001, tol_frac * 0.8)

        if it % 10000 == 0 and it > 0:
            print("    active_set it=%d, obj=%.8f, n_active=%d, tol=%.4f" % (it, best_obj, n_active, tol_frac))

    return best_a, best_obj


def multi_scale_nesterov(a_init, n, max_time, rng):
    """
    Multi-scale: optimize at coarser resolution, then refine at full resolution.
    """
    S = np.sqrt(2.0 * n)
    a = normalize(a_init.copy(), S)

    # Coarse optimization (use every 4th element)
    coarse_ratio = 4
    n_coarse = n // coarse_ratio
    if n_coarse >= 64:
        a_coarse = np.array([a[i * coarse_ratio:(i+1) * coarse_ratio].mean()
                            for i in range(n_coarse)])
        S_coarse = np.sqrt(2.0 * n_coarse)
        a_coarse = normalize(a_coarse, S_coarse)

        t_coarse = max_time * 0.3
        a_coarse_opt, _ = nesterov_lse_bb(a_coarse, n_coarse, t_coarse, rng,
                                          beta_start=3.0, beta_end=200.0)
        # Upsample back to full resolution
        a_up = np.interp(np.arange(n), np.arange(n_coarse), a_coarse_opt)
        a = normalize(a_up, S)

    # Full resolution refinement
    t_full = max_time * 0.7
    a_opt, obj = nesterov_lse_bb(a, n, t_full, rng, beta_start=10.0, beta_end=500.0)
    return normalize(a_opt, S), compute_objective(a_opt)


def generate_diverse_inits(n, rng, given=None):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None and len(given) == n:
        inits.append(("given", given.copy()))
        # Perturbed versions
        for i in range(3):
            noise = rng.normal(0, 0.1, n)
            inits.append(("given_pert%d" % i, np.maximum(given + noise * given.std(), 0)))

    # Structured initializations
    inits.append(("cos1", 1.0 + 0.5 * np.cos(2 * np.pi * tt)))
    inits.append(("cos2", 1.0 + 0.3 * np.cos(4 * np.pi * tt) + 0.2 * np.cos(6 * np.pi * tt)))
    inits.append(("cos3", 1.0 + 0.2 * np.cos(2 * np.pi * tt) + 0.3 * np.cos(2 * np.pi * 3 * tt)
                 + 0.1 * np.cos(2 * np.pi * 7 * tt)))
    inits.append(("sin2", np.maximum(np.sin(np.pi * tt)**2 + 0.1, 0.01)))
    inits.append(("sin4", np.maximum(np.sin(np.pi * tt)**4 + 0.1, 0.01)))
    inits.append(("edge", 1.0 + 1.5 * (tt**6 + (1-tt)**6)))
    inits.append(("center", 1.0 + 2.0 * np.exp(-8.0 * (tt - 0.5)**2)))
    inits.append(("rand", np.maximum(rng.random(n) * 0.5 + 0.5, 0.01)))
    inits.append(("rand2", np.maximum(rng.exponential(1.0, n) * 0.5 + 0.3, 0.01)))

    return inits

                a_init_alt = np.interp(np.linspace(0, 1, n_alt, endpoint=False),
                                       np.linspace(0, 1, len(global_a), endpoint=False),
                                       global_a)
            else:
                tt = np.linspace(0, 1, n_alt, endpoint=False)
                a_init_alt = 1.0 + 0.3 * np.cos(2 * np.pi * tt)

            rng3 = np.random.default_rng(seed + 8888)
            a_opt, obj = nesterov_lse_bb(a_init_alt, n_alt, t_alloc, rng3,
                                         beta_start=5.0, beta_end=500.0)
            a_opt = normalize(a_opt, np.sqrt(2.0 * n_alt))
            obj = compute_objective(a_opt)
            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  Phase 5 result: %.8f (n=%d)" % (global_best, n_alt))
            else:
                print("  Phase 5: no improvement (%.8f vs %.8f, n=%d)" % (obj, global_best, n_alt))

    # ===================== PHASE 6: Final fine polishing =====================
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t_alloc = remaining * 0.9
        print("\n=== Phase 6: Final polishing, n=%d, time=%.0fs ===" % (n, t_alloc))

        a_polish = global_a.copy()
        polish_rng = np.random.default_rng(seed + 1313)
        start_p = time.monotonic()
        step = 0.05
        it = 0
        best_polish = global_best
        stall = 0

        while time.monotonic() - start_p < t_alloc:
            it += 1
            c = fft_convolve_self(a_polish)
            cur = 2.0 * n * c.max() / S**2

            # Very sharp LSE (essentially weighted active set)
            beta = 1000.0
            c_max = c.max()
            x = np.clip(beta * (c - c_max), -700, 700)
            e = np.exp(x)
            w = e / e.sum()

            size = 2 * n - 1
            fsize = 1 << (size + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a_polish, fsize)
            grad = np.real(ifft(W * np.conj(A)))[:n]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            a_try = project_simplex(a_polish - step * grad / gnorm, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2

            if obj_try < cur - 1e-14:
                a_polish = a_try
                step = min(step * 1.01, 1.0)
                if obj_try < best_polish:
                    best_polish = obj_try
                    stall = 0
                else:
                    stall += 1
            else:
                step *= 0.95
                stall += 1

            if stall > 2000:
                scale = 0.001 * S / np.sqrt(n)
                a_polish = project_simplex(a_polish + polish_rng.normal(0, scale, n), S)
                step = 0.05
                stall = 0

            if it % 5000 == 0:
                print("    polish it=%d, obj=%.8f" % (it, best_polish))

        a_polish = normalize(a_polish, S)
        obj = compute_objective(a_polish)
        if obj < global_best:
            global_best = obj
            global_a = a_polish.copy()
            print("  Phase 6 result: %.8f" % global_best)

    # ===================== Final verification =====================
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected the result!")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256.0)
            global_best = 2.0

    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256.0)
        global_best = 2.0

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    print("\n=== Final: n=%d, obj=%.8f, total_time=%.0fs ===" % (
        len(final_list), global_best, time.monotonic() - t0))
    return final_list
