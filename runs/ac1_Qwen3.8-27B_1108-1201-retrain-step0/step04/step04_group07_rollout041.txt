Minimize 2*n*max(convolve(a,a))/sum(a)^2 using phase-aware spectral optimization
with Nesterov-accelerated subgradient descent.

Key innovations:
1. Phase optimization: adjust the phase of FFT(a) to reduce constructive interference
   in the autoconvolution, going beyond simple magnitude flattening.
2. Top-k weighted subgradient with adaptive step size and Nesterov acceleration.
3. Hybrid strategy: alternate between spectral (frequency-domain) steps and
   time-domain subgradient steps.
4. Multi-length search with progressive refinement.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))

    """Compute the full linear convolution of a with itself."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def subgradient_polish(a_init, N, max_time, rng, top_k=50, step_init=0.1, step_end=0.01):
    """
    Subgradient method with Nesterov acceleration for minimizing the peak
    of the autoconvolution.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    a = project_simplex(a, S)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = get_convolution(y, N)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
        
        # Compute gradient of weighted top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient: grad[j] = 2 * sum_l w_l * a[k_l - j]
        grad = np.zeros(n)
        for l in range(len(top_idx)):
            k = int(top_idx[l])
            w = weights[l]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                indices = np.arange(j_start, j_end + 1)
                grad[indices] += 2.0 * w * a[k - indices]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        progress = min(1.0, it / 10000.0)
        step = step_init * (step_end / step_init) ** progress * S / gnorm
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize
        s = y.sum()
        if s < 1e-15:
            y = a.copy()
        else:
            y = y * (S / s)
        a = x_new
        t = t_new
        
        # Periodic noise for exploration
        if it % 1000 == 500 and rng.random() < 0.15:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score


def magnitude_flatten_step(a, N, alpha, rng=None):
    """Reduce non-DC magnitude of FFT(a)."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A).copy()
    phase = np.angle(A)

    a_new = project_simplex(a_new, S)

def make_structured_init(n, rng, seed_offset=0):

    freqs = [3, 7, 13, 21, 37]

    # Bump
    inits.append(np.exp(-0.5 * ((tt - 0.5) * 3.0)**2) + 0.1)

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient with 
Nesterov acceleration, spectral re-shaping, and phase-aware refinement.

Key improvements:
1. Nesterov-accelerated subgradient with backtracking line search for robust step sizing.
2. Top-k adaptive weighting (k grows over iterations to smooth the landscape).
3. Periodic spectral re-shaping to escape local minima.
4. Phase-aware frequency domain adjustments to reduce autoconvolution peaks.
5. Multi-length search (8192, 16384) with progressive refinement.
6. Starts from the provided near-optimal solution for maximum head start.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_score_from_conv(c, n, S):
    """Compute score from precomputed convolution."""
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def full_pipeline_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def get_conv(a, N):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_subgradient(c, a, top_k):
    """Compute weighted subgradient of max convolution peak."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return None, 0.0, top_vals.max()
    weights = top_vals / total

    # grad[j] = 2 * sum_l w_l * a[k_l - j] for valid indices
    grad = np.zeros(n)
    for l in range(len(top_idx)):
        k = int(top_idx[l])
        w = weights[l]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            grad[indices] += 2.0 * w * a[k - indices]
    return grad, top_vals.max(), top_vals.max()


def spectral_reshape(a, N, strength, rng):
    """
    Reduce high-magnitude components of FFT(a) to diversify the landscape.
    This acts as a controlled perturbation in frequency domain.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A).copy()
    phase = np.angle(A)

    # Reduce non-DC magnitudes proportionally
    if N > 1:
        # Keep DC, reduce others
        reduction = np.ones(N)
        reduction[0] = 1.0
        reduction[1:] = 1.0 - strength * 0.15
        # Extra reduction for high frequencies (Nyquist region)
        half = N // 2
        reduction[half:] *= (1.0 - strength * 0.05)
        mag *= reduction

    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def phase_adjust_step(a, N, step, top_k, rng):
    """
    Adjust the phase of FFT(a) to reduce constructive interference in autoconvolution.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]

    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c.max()
    weights = top_vals / total

    # For each frequency m, the contribution to c[t] is:
    # (1/N) * |A[m]|^2 * cos(2*theta[m] + 2*pi*m*t/N)
    # To minimize weighted peak, adjust theta[m] so that the contribution is negative

    m_arr = np.arange(N)
    # Compute the "effective phase target" for each frequency
    # B[m] = sum_l w_l * exp(2j*pi*m*t_l/N)
    # We want cos(2*theta[m] + angle(B[m])) to be negative
    # So target: 2*theta[m] + angle(B[m]) = pi => theta[m] = (pi - angle(B[m]))/2

    B = np.zeros(N, dtype=complex)
    for l in range(len(top_idx)):
        t = int(top_idx[l])
        B += weights[l] * np.exp(2j * np.pi * m_arr * t / N)

    theta = np.angle(A)
    theta_target = (np.pi - np.angle(B)) / 2.0

    # Handle wraparound: find shortest angular difference
    diff = (theta_target - theta + np.pi) % (2 * np.pi) - np.pi

    mag = np.abs(A)
    # Only adjust for non-DC components (DC phase is irrelevant for real a)
    theta_new = theta.copy()
    theta_new[1:] += step * diff[1:]

    A_new = mag * np.exp(1j * theta_new)
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c.max()


def subgradient_nesterov(a_init, N, max_time, rng, top_k_init, top_k_max,
                          alpha_init, target, n_top_plateau):
    """
    Nesterov-accelerated subgradient with adaptive top_k and backtracking.
    """
    n = len(a_init)
    S = a_init.sum()
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best_score = full_pipeline_score(a)
    best_local = a.copy()
    it = 0
    plateau_count = 0
    alpha = alpha_init

    while time.monotonic() - start < max_time:
        it += 1
        c = get_conv(y, N)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
            plateau_count = 0
        else:
            plateau_count += 1

        # Adaptive top_k: start small, grow to focus on the hardest peaks
        k_eff = min(top_k_max, top_k_init + it // n_top_plateau)

        # Compute subgradient
        grad, gnorm, c_max = compute_subgradient(c, a, k_eff)
        if grad is None or gnorm < 1e-14:
            break

        # Backtracking line search
        step_base = alpha * S / gnorm
        accepted = False
        for attempt in range(6):
            step = step_base / (2 ** attempt)
            x_new = project_simplex(y - step * grad, S)
            c_new = get_conv(x_new, N)
            sc_new = 2.0 * n * c_new.max() / (x_new.sum() ** 2)
            if sc_new < sc + 1e-10:
                # Nesterov update
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
                y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
                y_new = np.maximum(y_new, 0.0)
                s_check = y_new.sum()
                if s_check < 1e-15:
                    y_new = x_new.copy()
                else:
                    y_new = y_new * (S / s_check)
                y = y_new
                a = x_new
                t_val = t_new
                accepted = True
                break

        if not accepted:
            # Try a smaller step
            step = step_base / 64.0
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            s_check = y.sum()
            if s_check > 1e-15:
                y = y * (S / s_check)
            else:
                y = x_new.copy()
            a = x_new
            t_val = t_new

        # Decay alpha slowly
        alpha = alpha_init / (1.0 + 0.0001 * it)

        # Periodic restart with noise
        if plateau_count > 5000 and rng.random() < 0.05:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            plateau_count = 0
            alpha = alpha_init * 0.5

        # Spectral reshape periodically to escape local minima
        if it % 5000 == 2500 and rng.random() < 0.3:
            a_shaped = spectral_reshape(a, N, 0.5, rng)
            sc_shaped = full_pipeline_score(a_shaped)
            if sc_shaped < best_score:
                a = a_shaped
                y = a_shaped.copy()
                t_val = 1.0

    return best_local, best_score


def phase_refine_loop(a_init, N, max_time, rng, target):
    """Phase-aware refinement loop."""
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_score = full_pipeline_score(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Anneal step size
        progress = min(1.0, it / 2000.0)
        step = 0.3 * (1.0 - 0.9 * progress) + 0.02
        top_k = 10 + it // 500

        sc = full_pipeline_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if best_score <= target:
            break

    return best_a, best_score

def make_initializations(n, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Smooth noise (low-pass filtered)
    noise = rng.standard_normal(n)
    A = fft(noise, n)
    # Keep only low frequencies
    bandwidth = max(1, n // 16)
    mask = np.zeros(n)
    mask[:bandwidth] = 1.0
    mask[-bandwidth:] = 1.0
    A[1:-1] *= mask[1:-1]
    smooth = np.real(ifft(A, n))
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(0.15 + 0.7 * smooth)

    # Multi-sine with random phases
    sig = np.zeros(n)
    for k in range(5):
        f = 2 * (k + 1) + 1
        phi = rng.random() * 2 * np.pi
        sig += np.sin(2 * np.pi * f * tt + phi) / (k + 1)
    sig = (sig - sig.min()) / max(sig.max() - sig.min(), 1e-10)
    inits.append(0.1 + 0.8 * sig)

    # Edge-weighted (exploit linear convolution edge effects)
    edge = 1.0 + 0.3 * np.sin(np.pi * tt) ** 2
    inits.append(0.2 * edge + 0.2 * rng.random(n))

    # Random with smoothing
    r = rng.random(n)
    kernel = np.ones(max(3, n // 32)) / (n // 32)
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = (r_smooth - r_smooth.min()) / max(r_smooth.max() - r_smooth.min(), 1e-10)
    inits.append(0.1 + 0.7 * r_smooth)

    return inits


def upsample_sequence(a, n_new):
    """Upsample a sequence to a new length using interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    return a_new

            inits = make_initializations(n, rng)

        # Run optimization on each initialization (but share budget)
        n_inits = len(inits)
        per_init_budget = alloc / max(n_inits, 1)

        for ii, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            here_budget = min(per_init_budget, remaining - 30)
            if here_budget < 15:
                continue

            a_init = np.clip(init, 0, 1000).astype(np.float64)
            s = a_init.sum()
            if s < 0.01:
                a_init = np.ones(n) * np.sqrt(2.0 * n) / n
                s = np.sqrt(2.0 * n)

            print("  Init %d/%d (sum=%.2f)" % (ii + 1, n_inits, s))
            a_best, sc = optimize_at_length(n, a_init, here_budget, rng, target=1.502)

            if sc < global_best:
                global_best = sc
                global_a = a_best.copy()
                print("  ** Global best updated: %.6f at n=%d" % (global_best, n))

    # Final intensive polish on the best found
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(global_a)
            N_best = 1 << (2 * n_best - 1).bit_length()
            print("\n=== Final polish (n=%d, %.0fs) ===" % (n_best, remaining))
            a_f, sc_f = subgradient_nesterov(global_a, N_best, remaining * 0.6, rng,
                                              top_k_init=100, top_k_max=500,
                                              alpha_init=0.08, target=1.50,
                                              n_top_plateau=1000)
            if sc_f < global_best:
                global_best = sc_f
                global_a = a_f.copy()
                print("  Polish improved to: %.6f" % global_best)

            # One more phase pass
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_p, sc_p = phase_refine_loop(global_a, N_best, remaining, rng, target=1.50)
                if sc_p < global_best:
                    global_best = sc_p
                    global_a = a_p.copy()
                    print("  Phase pass improved to: %.6f" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Fallback uniform: score=2.0")

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nVerified: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate returned inf, fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver)

    print("FINAL: n=%d, score=%.6f, elapsed=%.0fs" % (
        len(global_a), global_best, time.monotonic() - t_global))
    return final_list
