"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses hybrid FISTA + exact subgradient + adaptive LP best-response with
deflation targeting, multi-n exploration, and structured restarts.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lags(a, c, lags, n):
    """Compute subgradient direction: average gradient at specified lags."""
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    w = np.zeros(2 * n - 1)
    for k in lags:
        w[k] = 1.0
    w /= len(lags)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_lp_best_response(f, n, S, M, binding_lags, solver='SCS', time_limit=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in binding_lags, g >= 0.
    """
    top_k = len(binding_lags)
    if top_k == 0:
        return None
    
    rows = np.zeros((top_k, n))
    for i, k in enumerate(binding_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if time.monotonic() - start > time_limit:
        return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    
    return None


def find_binding_lags(c, M, frac=0.01):
    """Find lags where c[k] is within frac*M of the max."""
    threshold = M - frac * M
    return np.where(c >= threshold)[0]
    # return np.argsort(c)[::-1][:top_k]


def run_fista_subgrad(a_init, n, S, max_time, rng, 
                       beta_init=10.0, beta_rate=0.002, beta_max=1000.0,
                       subgrad_every=500, subgrad_n_lags=5):
    """FISTA with periodic exact subgradient steps."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it
        
        # Choose between LSE gradient and exact subgradient
        if it % subgrad_every == 0 and it > 100:
            # Use exact subgradient at top lags
            worst_lags = np.argsort(c)[::-1][:subgrad_n_lags]
            grad = subgradient_at_lags(y, c, worst_lags, n)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        step = 1.0 / (2.0 * beta)
        
        if it - last_improve_it > 3000:
            # Stuck: add noise and reset
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic repulsion
        if it % 500 == 0 and it > 500:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(30.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-0.05 * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 1e-15)
                y = project_simplex(y, S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_lp_iteration(f, n, S, c, M, binding_lags, solver='SCS'):
    """One LP iteration with bisection for step size."""
    g = solve_lp_best_response(f, n, S, M, binding_lags, solver=solver)
    if g is None:
        return f.copy(), M
    
    def eval_step(t):
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max()
    
    # Try full replacement first
    val1 = eval_step(1.0)
    if val1 <= M * 1.001:
        f_new = project_simplex(np.maximum(g, 0), S)
        return f_new, fft_convolve_self(f_new).max()
    
    # Golden section search for optimal step
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = eval_step(c1)
    f2 = eval_step(c2)
    
    for _ in range(15):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_step(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_step(c2)
    
    best_t = (lo + hi) / 2
    best_val = min(f1, f2)
    
    if best_val < M * 0.999:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), M

def deflation_phase(a, n, S, max_time, rng, n_lags=10, lr=0.01):
    """
    Targeted deflation: repeatedly reduce the worst lags using exact subgradient.
    """
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find worst lags
        worst = np.argsort(c)[::-1][:n_lags]
        grad = subgradient_at_lags(a, c, worst, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = lr / max(gnorm, 1e-10) * S / np.sqrt(n)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new, S)
        
        # Check if improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Reduce step and try again
            step *= 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new, S)
            a = a_new
        
        # Periodic noise to escape
        if it % 500 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def interpolate_sequence(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)
