"""
Minimize 2*n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses continuation FISTA with backtracking, multi-scale refinement,
and active-set polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value_and_weights(c, beta):
    """Compute LSE value and weights for the active set."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    return lse_val, w

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_subgradient(y, c, k, tol_frac):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.argsort(c)[::-1][:k]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1 (or normalize first)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def fista_continuation(a_init, max_time, rng, beta_start=1.0, beta_end=2000.0,
                        n_stages=8, backtrack=True):
    """
    FISTA with continuation on beta parameter.
    Divides time among n_stages, each with increasing beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Log-spaced beta values
    betas = np.geomspace(beta_start, beta_end, n_stages)
    time_per_stage = max_time / n_stages

    y = a.copy()
    t_mom = 1.0
    best_conv = np.inf
    best_a = a.copy()
    start = time.monotonic()

    for stage, beta in enumerate(betas):
        stage_deadline = start + (stage + 1) * time_per_stage
        it = 0
        step = S / (beta + 1.0)  # Initial step size
        last_improve = 0
        stuck_count = 0

        while time.monotonic() < stage_deadline:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-12:
                best_conv = cur_max
                best_a = y.copy()
                last_improve = it
                stuck_count = 0
            else:
                stuck_count += 1

            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            if backtrack:
                # Backtracking line search
                step_try = step
                x_new = project_simplex(y - step_try * grad, S)
                c_new = fft_convolve_self(x_new)
                lse_new, _ = lse_value_and_weights(c_new, beta)
                lse_old, _ = lse_value_and_weights(c, beta)
                while lse_new > lse_old - 1e-4 * step_try * gnorm * gnorm and step_try > 1e-10:
                    step_try *= 0.5
                    x_new = project_simplex(y - step_try * grad, S)
                    c_new = fft_convolve_self(x_new)
                    lse_new, _ = lse_value_and_weights(c_new, beta)
                step = min(step_try * 1.5, S / (0.5 * gnorm))
            else:
                x_new = project_simplex(y - step * grad, S)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            # Periodic restart within stage
            if it % 5000 == 0:
                t_mom = 1.0
                if stuck_count > 2000:
                    noise_scale = 0.001 * S / np.sqrt(n)
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_mom = 1.0
                    stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def active_polish(a_init, max_time, rng):
    """
    High-precision polishing using exact subgradient with adaptive active set.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step0 = S  # Initial step scale

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set: start with exact (k=1), widen when stuck
        if stuck < 100:
            k = 1
            tol_frac = 0.0
        elif stuck < 500:
            k = 2
            tol_frac = 0.001
        elif stuck < 2000:
            k = 4
            tol_frac = 0.005
        elif stuck < 10000:
            k = min(16, 4 + stuck // 500)
            tol_frac = 0.01
        else:
            k = min(64, 16 + stuck // 1000)
            tol_frac = 0.02

        grad = topk_subgradient(y, c, k, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step size
        step = step0 / (gnorm * np.sqrt(1.0 + 0.01 * it))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when stuck
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

def normalize(a):
    n = len(a)
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses continuation FISTA with backtracking line search, multi-scale refinement,
active-set polishing with adaptive subgradient, and random perturbation restarts.
Key improvements: backtracking for robustness, continuation on beta for smooth-to-sharp
transition, extended polishing phase, and diverse restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a):
    """Normalize to sum = sqrt(2n)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_objective(c, beta):
    """Compute LSE of c with parameter beta."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(y, c, k, tol_frac=0.0):
    """Subgradient using top-k or tolerance-based active set."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.argsort(c)[::-1][:min(k, len(c))]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_backtracking(a_init, max_time, rng, beta_start=2.0, beta_end=2000.0,
                       n_stages=6, use_backtrack=True, step_init=None):
    """
    FISTA with continuation on beta and optional backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a)

    betas = np.geomspace(beta_start, beta_end, n_stages)
    time_per_stage = max_time / n_stages

    y = a.copy()
    t_mom = 1.0
    best_conv = np.inf
    best_a = a.copy()
    start = time.monotonic()

    for stage, beta in enumerate(betas):
        stage_deadline = start + (stage + 1) * time_per_stage
        it = 0
        if step_init is not None:
            step = step_init
        else:
            step = S / (beta + 1.0)
        last_improve = it
        stuck_count = 0

        while time.monotonic() < stage_deadline:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-12:
                best_conv = cur_max
                best_a = y.copy()
                last_improve = it
                stuck_count = 0
            else:
                stuck_count += 1

            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            if use_backtrack and beta > 5:
                # Backtracking line search
                lse_old = lse_objective(c, beta)
                step_try = min(step, 2.0 * S / gnorm)
                x_new = project_simplex(y - step_try * grad, S)
                c_new = fft_convolve_self(x_new)
                lse_new = lse_objective(c_new, beta)
                count = 0
                while lse_new > lse_old - 1e-4 * step_try * gnorm * gnorm and step_try > 1e-12 and count < 8:
                    step_try *= 0.5
                    x_new = project_simplex(y - step_try * grad, S)
                    c_new = fft_convolve_self(x_new)
                    lse_new = lse_objective(c_new, beta)
                    count += 1
                step = min(step_try * 1.2, 2.0 * S / gnorm)
            else:
                x_new = project_simplex(y - step * grad, S)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            # Periodic restart / noise
            if stuck_count > 3000 and it % 2000 == 0:
                noise_scale = max(1e-5, 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0))
                if noise_scale > 1e-7:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_mom = 1.0
                    stuck_count = 0

            if it % 10000 == 0:
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def active_polish(a_init, max_time, rng, step_decay=0.005):
    """
    High-precision polishing using exact subgradient with adaptive active set.
    Uses decreasing step size and momentum.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step0 = S
    momentum = 0.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            momentum = 0.0

        stuck = it - last_improve

        # Adaptive active set
        if stuck < 50:
            k = 1
            tol_frac = 0.0
        elif stuck < 300:
            k = 2
            tol_frac = 0.001
        elif stuck < 1000:
            k = 4
            tol_frac = 0.003
        elif stuck < 5000:
            k = min(16, 4 + stuck // 400)
            tol_frac = 0.008
        else:
            k = min(32, 16 + stuck // 2000)
            tol_frac = 0.015

        grad = subgradient_topk(y, c, k, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: Polyak-like with decay
        step = step0 / (gnorm * (1.0 + step_decay * it))
        step = min(step, 3.0 * S / gnorm)

        # Adaptive momentum
        if stuck > 2000:
            momentum = min(0.3, 0.05 + 0.01 * (stuck // 2000))
        else:
            momentum = 0.0

        x_new = project_simplex(y - step * grad, S)

        if momentum > 0:
            # Simple momentum: blend with previous direction
            direction = x_new - y
            y_next = project_simplex(y + momentum * direction - step * grad, S)
            if y_next.sum() > 0:
                x_new = y_next

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when very stuck
        if stuck > 8000 and it % 2000 == 0:
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def smooth_perturbation(a, rng, scale):
    """Apply a smooth random perturbation and renormalize."""
    n = len(a)
    # Generate smooth noise via random low-frequency sinusoids
    noise = np.zeros(n)
    n_freqs = min(20, n // 4)
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // n_freqs))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 1.0)
        noise += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    noise *= scale / (np.linalg.norm(noise) + 1e-15)
    result = a + noise
    result = np.maximum(result, 0)
    return normalize_seq(result)

                                        beta_start=1.0, beta_end=1000.0,
                                        n_stages=5, use_backtrack=True)
            a2 = normalize_seq(a2)
            obj2 = compute_obj(a2)
            print("  Upsampled n=%d: obj=%.8f (t=%.0fs)" % (n2, obj2, time.monotonic() - t2))
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_best_a = a2.copy()
            # Polish the 2n version
            remaining2a = p2_budget * 0.45 - (time.monotonic() - t2)
            if obj2 < global_best_obj + 0.001:
                a2p, _ = active_polish(a2, min(remaining2a, p2_budget * 0.15), rng)
                obj2p = compute_obj(a2p)
                print("  Upsampled polished: obj=%.8f" % obj2p)
                if obj2p < global_best_obj:
                    global_best_obj = obj2p
                    global_best_a = a2p.copy()

        # Try downsampling to n/2
        n_half = n0 // 2
        if n_half >= 2048:
            a_half = downsample(global_best_a, factor=2)
            a_half = normalize_seq(a_half)
            a_half, _ = fista_backtracking(a_half, p2_budget * 0.2, rng,
                                            beta_start=5.0, beta_end=3000.0,
                                            n_stages=4, use_backtrack=True)
            a_half = normalize_seq(a_half)
            obj_half = compute_obj(a_half)
            print("  Downsampled n=%d: obj=%.8f (t=%.0fs)" % (n_half, obj_half, time.monotonic() - t2))
            if obj_half < global_best_obj:
                global_best_obj = obj_half
                global_best_a = a_half.copy()

        # Polish best at its current size
        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 10:
            a2b, _ = fista_backtracking(global_best_a, remaining2, rng,
                                         beta_start=200.0, beta_end=10000.0,
                                         n_stages=3, use_backtrack=True)
            obj2b = compute_obj(a2b)
            print("  Phase 2 final: obj=%.8f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Active-set polishing (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.30
        n_best = len(global_best_a)
        print("\n=== Phase 3: Active-set polish at n=%d (budget %.0fs) ===" % (n_best, p3_budget))
        t3 = time.monotonic()

        a3, _ = active_polish(global_best_a, p3_budget * 0.4, rng)
        obj3 = compute_obj(a3)
        print("  Active polish 1: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 10:
            a3b, _ = active_polish(global_best_a, remaining3, rng, step_decay=0.003)
            obj3b = compute_obj(a3b)
            print("  Active polish 2: obj=%.8f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Random perturbation + re-optimize (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.15
        n_best = len(global_best_a)
        print("\n=== Phase 4: Random restarts (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        n_restarts = 3
        time_per_restart = p4_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 5:
                break
            # Generate perturbed version
            scale = 0.02 * np.sqrt(2.0 * n_best) / np.sqrt(n_best)
            a_pert = smooth_perturbation(global_best_a, rng, scale)

            # Quick FISTA on perturbed version
            a_r, _ = fista_backtracking(a_pert, time_per_restart * 0.5, rng,
                                         beta_start=50.0, beta_end=2000.0,
                                         n_stages=3, use_backtrack=True)
            obj_r = compute_obj(a_r)

            # Polish if promising
            if obj_r < global_best_obj + 0.01:
                a_rp, _ = active_polish(a_r, time_per_restart * 0.4, rng)
                obj_rp = compute_obj(a_rp)
                if obj_rp < global_best_obj:
                    global_best_obj = obj_rp
                    global_best_a = a_rp.copy()
                    print("  Restart %d: obj=%.8f (improvement!)" % (r + 1, obj_rp))
                else:
                    print("  Restart %d: obj=%.8f (no improvement)" % (r + 1, obj_rp))
            else:
                print("  Restart %d: obj=%.8f (not promising)" % (r + 1, obj_r))

    # === Phase 5: Final intensive polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p5_budget = remaining * 0.90
        n_best = len(global_best_a)
        print("\n=== Phase 5: Final polish at n=%d (budget %.0fs) ===" % (n_best, p5_budget))
        t5 = time.monotonic()

        # Very high beta FISTA
        a5, _ = fista_backtracking(global_best_a, p5_budget * 0.3, rng,
                                    beta_start=1000.0, beta_end=50000.0,
                                    n_stages=3, use_backtrack=True)
        obj5 = compute_obj(a5)
        print("  High-beta FISTA: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        remaining5 = p5_budget - (time.monotonic() - t5)
        if remaining5 > 5:
            a5b, _ = active_polish(global_best_a, remaining5, rng, step_decay=0.002)
            obj5b = compute_obj(a5b)
            print("  Final active: obj=%.8f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("Total time used: %.0fs" % (time.monotonic() - (deadline - budget_s + 35.0)))
    return final_list
