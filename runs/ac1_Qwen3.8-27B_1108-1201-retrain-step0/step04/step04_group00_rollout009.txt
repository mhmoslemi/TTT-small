"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-initialization FISTA with backtracking, LP-guided direction
finding at reduced scale, and multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def active_subgradient(y, c, frac=0.0):
    n = len(y)
    max_val = c.max()
    if frac > 0:
        active = np.where(c >= max_val * (1.0 - frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def fista_backtracking(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                        beta_max=5000.0, step_scale=1.0, use_lse=True,
                        active_frac=0.0):
    """FISTA with backtracking line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    # Backtracking parameters
    c1 = 1e-4  # Armijo parameter
    rho_bt = 0.5  # Step size reduction factor

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Gradient computation
        if use_lse:
            grad = lse_gradient(y, c, beta)
        else:
            grad = active_subgradient(y, c, frac=active_frac)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step_trial = step
        x_candidate = project_simplex(y - step_trial * grad, S)
        c_candidate = fft_convolve_self(x_candidate)
        
        # Check if this is a sufficient decrease
        # We want max(c) to decrease
        for _ in range(20):
            if c_candidate.max() <= cur_max - c1 * step_trial * gnorm**2 + 1e-12:
                break
            step_trial *= rho_bt
            x_candidate = project_simplex(y - step_trial * grad, S)
            c_candidate = fft_convolve_self(x_candidate)
        
        step = step_trial * 1.5  # Grow step for next iteration (within limits)
        step = min(step, step_scale / max(beta, 1.0) * 3)

        x_new = x_candidate
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 2000:
            t_mom = 1.0
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 5000.0)
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-initialization FISTA with backtracking line search,
LP-guided direction finding at reduced scale, adaptive active-set tracking,
and multi-length exploration. Key innovations over baseline:
1. Asymmetric initializations to escape symmetric basins
2. Proper backtracking on LSE-smoothed objective
3. LP-based direction at reduced scale for global guidance
4. Adaptive switching between LSE and exact subgradient modes
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute the log-sum-exp smoothed value of max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    return cm + np.log(e.sum()) / beta


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_smoothed_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals - c.min(), 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_optimize(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                   beta_max=5000.0, mode="lse", step_scale=1.0,
                   backtrack=True, restart_interval=3000):
    """
    FISTA with adaptive LSE smoothing and optional backtracking.
    mode: 'lse', 'active', 'topk', 'adaptive'
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    c1 = 1e-4
    rho_bt = 0.5

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive mode switching
        if mode == "adaptive":
            if stuck > 3000:
                grad = active_subgradient(y, c, tol_frac=0.02)
            elif stuck > 1000:
                k = min(32, 4 + stuck // 200)
                grad = topk_smoothed_gradient(y, c, k, power=1.0)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "active":
            tol = 0.0 if stuck < 500 else min(0.05, 0.01 + stuck / 20000.0)
            grad = active_subgradient(y, c, tol_frac=tol)
        elif mode == "topk":
            k = max(4, min(64, n // 64))
            if stuck > 2000:
                k = max(2, k // 4)
            grad = topk_smoothed_gradient(y, c, k, power=1.0)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search
        if backtrack:
            # Compute current LSE value for comparison
            cur_lse = lse_value(c, beta)
            step_trial = step
            x_trial = project_simplex(y - step_trial * grad, S)
            for _ in range(15):
                if step_trial < 1e-10:
                    break
                c_trial = fft_convolve_self(x_trial)
                lse_trial = lse_value(c_trial, beta)
                if lse_trial <= cur_lse - c1 * step_trial * gnorm ** 2 + 1e-10:
                    break
                step_trial *= rho_bt
                x_trial = project_simplex(y - step_trial * grad, S)
            step = min(step_trial * 1.8, step_scale / max(beta, 1.0) * 5.0)
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            # Shrink step when stuck
            if stuck > 3000:
                step *= 0.4
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart with noise
        if stuck >= restart_interval and it % (restart_interval // 10) == 0:
            decay = np.exp(-stuck / (restart_interval * 3))
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

        # Hard restart if completely stuck
        if stuck > restart_interval * 4:
            t_mom = 1.0
            step = step_scale / max(beta_init, 1.0)
            noise_scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def solve_lp_direction(a, M, n_slim=None):
    """
    Solve the LP to find optimal direction.
    Primal: max sum(b_j) s.t. (a*b)[k] <= M, b >= 0
    For large n, solve at reduced size.
    Returns the direction b (normalized to sum = sqrt(2n)).
    """
    n = len(a)
    if n_slim is not None and n_slim < n:
        # Downsample a to n_slim
        blocks = n // n_slim
        if blocks < 1:
            blocks = 1
        a_small = a[:blocks * n_slim].reshape(n_slim, blocks).mean(axis=1)
        S_small = np.sqrt(2.0 * n_slim)
        a_small = a_small * (S_small / a_small.sum())
        c_small = fft_convolve_self(a_small)
        M_small = c_small.max()
        b_small = _solve_lp_core(a_small, M_small, n_slim)
        # Upsample b_small back to n
        n_full = len(a)
        t_old = np.linspace(0, 1, n_slim, endpoint=False)
        t_full = np.linspace(0, 1, n_full, endpoint=False)
        b_full = np.interp(t_full, t_old, b_small)
        S_full = np.sqrt(2.0 * n_full)
        b_full = b_full * (S_full / max(b_full.sum(), 1e-15))
        return b_full
    else:
        S = np.sqrt(2.0 * n)
        a_norm = a * (S / a.sum())
        c = fft_convolve_self(a_norm)
        M_actual = c.max()
        b = _solve_lp_core(a_norm, M_actual, n)
        return b


def _solve_lp_core(a, M, n, max_iter=1000):
    """
    Solve the LP using scipy linprog (HiGHS) for moderate n,
    or use a simple active-set heuristic for large n.
    """
    if n > 2048:
        # Too large for direct LP, use iterative heuristic
        return _lp_heuristic(a, M, n, max_iter)

    # Build constraint matrix A_eq: (a*b)[k] = sum_j a[j]*b[k-j] <= M
    # Number of constraints: 2n-1
    n_constraints = 2 * n - 1

    # A constraint matrix: A[k, j] = a[j] if 0 <= k-j <= n-1, else 0
    A_mat = np.zeros((n_constraints, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        A_mat[k_start:k_end + 1, j] = a[j]

    # c vector for objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)

    # Solve: min c_obj @ b s.t. A_mat @ b <= M, b >= 0
    result = linprog(c_obj, A_ub=A_mat, b_ub=np.full(n_constraints, M),
                     bounds=[(0, None)] * n, method='highs',
                     options={'maxiter': max_iter})

    if result.success:
        b = result.x
    else:
        # Fallback: uniform
        b = np.ones(n) / n
    S = np.sqrt(2.0 * n)
    b = np.maximum(b, 0)
    if b.sum() < 1e-15:
        b = np.ones(n) / n
    b = b * (S / b.sum())
    return b


def _lp_heuristic(a, M, n, max_iter=500):
    """
    Heuristic for LP direction: iterative coordinate ascent.
    Start with b = a, then iteratively increase b_j where it
    doesn't violate the constraints the most.
    """
    b = a.copy()
    S = np.sqrt(2.0 * n)
    b = b * (S / b.sum())

    for _ in range(min(max_iter, 100)):
        # Compute current convolution
        c = fft_convolve_self(b)
        # Find which constraints are most violated
        violations = M - c
        # Try to increase coordinates where it helps most
        # Simple: add small amount to coordinates with low "load"
        loads = np.zeros(n)
        for k in range(len(c)):
            if c[k] > M * 0.5:
                for j in range(max(0, k - n + 1), min(k + 1, n)):
                    loads[j] += a[j] * (c[k] - M * 0.5) / M

        # Increase b where load is low
        load_min = loads.min()
        mask = loads < load_min + 0.1 * (loads.max() - load_min + 1e-10)
        if mask.sum() > 0:
            b[mask] += 0.01 * S / mask.sum()
            b = np.maximum(b, 0)
            b = b * (S / b.sum())

    return b


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []

    # 1. Cosine (symmetric)
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t))

    # 2. Half-cosine (asymmetric - mass on left)
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))

    # 3. Exponential decay
    inits.append(np.exp(-3.0 * t))

    # 4. Two-lobe asymmetric
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * 2 * t - np.pi / 4))

    # 5. Random smooth (low-pass filtered noise)
    noise = rng.normal(0, 1, n)
    fsize = 1 << (n - 1).bit_length()
    N = fft(noise, fsize)
    # Keep only low frequencies
    cutoff = n // 8
    N[cutoff:-cutoff] = 0
    smooth = np.real(ifft(N, fsize))[:n]
    smooth = np.maximum(smooth, 0)
    if smooth.sum() > 0:
        inits.append(smooth)
    else:
        inits.append(np.ones(n))

    # 6. Power law
    inits.append((1.0 - t) ** 1.5 + 0.01)

    # 7. Triple lobe
    inits.append(0.4 + 0.6 * np.abs(np.cos(3 * np.pi * t)))

    return inits

        for nl in sorted(candidate_lengths):
            if time.monotonic() > deadline - 15:
                break
            if nl == n_best:
                a_try = global_best_a.copy()
            elif nl > n_best:
                # Upsample
                factor = nl // n_best
                if factor < 2:
                    # Padd with interpolated values
                    t_old = np.linspace(0, 1, n_best, endpoint=False)
                    t_new = np.linspace(0, 1, nl, endpoint=False)
                    a_try = np.interp(t_new, t_old, global_best_a)
                else:
                    a_try = upsample(global_best_a, factor=factor)
                    a_try = a_try[:nl]
            else:
                # Downsample
                factor = n_best // nl
                if factor >= 2:
                    a_try = downsample(global_best_a, factor=factor)
                else:
                    t_old = np.linspace(0, 1, n_best, endpoint=False)
                    t_new = np.linspace(0, 1, nl, endpoint=False)
                    a_try = np.interp(t_new, t_old, global_best_a)

            S_nl = np.sqrt(2.0 * nl)
            a_try = np.maximum(a_try, 0)
            a_try = a_try * (S_nl / max(a_try.sum(), 1e-15))

            obj_try = compute_obj(a_try)
            if obj_try < global_best_obj and nl != n_best:
                print("  n=%d: initial obj=%.6f (improvement!)" % (nl, obj_try))
                global_best_obj = obj_try
                global_best_a = a_try.copy()

            # Short FISTA polish at this length
            remaining_len = p4_budget - (time.monotonic() - t_phase4)
            if remaining_len > 5:
                budget_this = min(per_len_budget, remaining_len)
                a_pol, _ = fista_optimize(a_try, budget_this, rng,
                                           beta_init=30.0, beta_rate=0.004,
                                           beta_max=10000.0, mode="adaptive",
                                           step_scale=0.7, backtrack=True)
                obj_pol = compute_obj(a_pol)
                if obj_pol < global_best_obj:
                    print("  n=%d after polish: obj=%.6f" % (nl, obj_pol))
                    global_best_obj = obj_pol
                    global_best_a = a_pol.copy()
                else:
                    print("  n=%d: obj=%.6f (no improvement)" % (nl, obj_pol))

    # === Phase 5: Final tight polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p5_budget = remaining * 0.90
        n_best = len(global_best_a)
        print("\n=== Phase 5: Final polish, n=%d, budget=%.0fs ===" % (n_best, p5_budget))
        t_phase5 = time.monotonic()

        # Very high beta for fine-tuning
        a_final, _ = fista_optimize(global_best_a, p5_budget * 0.6, rng,
                                     beta_init=500.0, beta_rate=0.001,
                                     beta_max=50000.0, mode="active",
                                     step_scale=0.2, backtrack=False,
                                     restart_interval=10000)
        obj_final = compute_obj(a_final)
        print("  High-beta final: obj=%.6f" % obj_final)
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_best_a = a_final.copy()

        remaining5 = p5_budget - (time.monotonic() - t_phase5)
        if remaining5 > 5:
            # Top-k focused polish
            a_final2, _ = fista_optimize(global_best_a, remaining5, rng,
                                          beta_init=1000.0, beta_rate=0.001,
                                          beta_max=50000.0, mode="topk",
                                          step_scale=0.15, backtrack=False,
                                          restart_interval=15000)
            obj_final2 = compute_obj(a_final2)
            print("  Topk final: obj=%.6f" % obj_final2)
            if obj_final2 < global_best_obj:
                global_best_obj = obj_final2
                global_best_a = a_final2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = 0.54 + 0.46 * np.cos(2 * np.pi * np.linspace(0, 1, n_fb, endpoint=False))
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = 0.54 + 0.46 * np.cos(2 * np.pi * np.linspace(0, 1, n_fb, endpoint=False))
        final_list = [float(x) for x in global_best_a]

    print("Total time used: %.0fs" % (time.monotonic() - (deadline - budget_s + 20.0)))
    return final_list
