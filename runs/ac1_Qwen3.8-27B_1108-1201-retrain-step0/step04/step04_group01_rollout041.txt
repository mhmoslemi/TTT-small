import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    diff = c - cm
    if beta * diff.max() > 500:
        beta = 500.0 / max(1e-15, diff.max())
    terms = np.exp(beta * diff)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution w.r.t. a."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    if beta * diff.max() > 500:
        beta = 500.0 / max(1e-15, diff.max())
    scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: (grad b_k)_j = 2 * a_{k-j} (with zero-padding)
    # So grad LSE = sum_k w_k * 2 * shift_a(k)
    # Using FFT: this is 2 * (IFFT(FFT(w) * conj(FFT(a))))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def subgradient_fft(a, w):
    """Compute subgradient given weights w on convolution lags.
    grad[j] = 2 * sum_k w[k] * a[k-j] (with zero-padding for invalid indices)
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def bb_step_size(a_old, a_new, g_old, g_new):
    """Barzili-Borwein step size."""
    s = a_new - a_old
    y = g_new - g_old
    ys = np.dot(y, s)
    ss = np.dot(s, s)
    yy = np.dot(y, y)
    if abs(ys) < 1e-30:
        return 1.0 / max(yy, 1e-30)
    # BB1: ss/ys
    if ys > 0:
        return ss / ys
    # BB2: ys/yy
    elif yy > 0:
        return ys / yy
    return 1e-3


def run_fista_adaptive(a_init, S, max_time, rng, beta_init=20.0, 
                        beta_rate=0.003, beta_max=15000.0,

    """FISTA with adaptive LSE smoothing and BB-inspired restarts."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Check for timeout within the loop
        if time.monotonic() - start > max_time:
            break
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: 1/L where L ≈ 2*beta*||a||_inf (Lipschitz of LSE gradient)
        # Be more aggressive: use 1/(2*beta*S/n) since ||a||_inf ≤ S
        L_est = 2.0 * beta * S / n
        step = 1.0 / L_est
        
        # Limit step to avoid oscillation

        step = min(step, max_step)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 20000.0)
            if noise_scale > 1e-10:

                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:

def generate_tilted_inits(n, rng):
    """Generate tilted initializations (key: non-palindromic shapes)."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Various tilts
    inits.append(1.0 + 0.5 * tt)  # linear ramp up
    inits.append(1.0 + 0.5 * (1 - tt))  # linear ramp down
    inits.append(1.0 + 0.7 * np.sin(np.pi * tt))  # bump in middle
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * tt))  # cosine tilt
    
    # Exponential tilts
    inits.append(np.exp(2.0 * tt - 1.0))  # exponential ramp
    inits.append(np.exp(-2.0 * tt + 1.0))  # exponential decay
    
    # Multi-frequency
    for freq in [2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(1.0 + amp * np.sin(2 * np.pi * freq * tt))
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * tt + 0.5))
    
    # Quadratic
    inits.append(1.0 + 0.5 * (2*tt - 1)**2)
    inits.append(1.0 - 0.3 * (2*tt - 1)**2)
    
    # Random smooth
    for _ in range(3):
        f = rng.integers(1, 8)
        phase = rng.uniform(0, 2*np.pi)
        inits.append(1.0 + 0.5 * np.sin(2*np.pi*f*tt + phase))
    
    # Ensure non-negative
    return [np.maximum(x, 0.01) for x in inits]


def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    a_trimmed = a[:new_n * factor]
    return a_trimmed.reshape(new_n, factor).mean(axis=1)


def upscale(a_small, n_large):
    """Upscale by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def parametric_optimize(n, S, max_time, rng):
    """Optimize using a parametric (Fourier) representation."""
    # a_i = S/n * (1 + sum_j c_j * cos(2πj*i/n + φ_j))
    # For simplicity, use a fixed number of modes
    n_modes = 16
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Precompute basis functions
    # basis[i, j] = cos(2π*j*i/n + 0)  (we'll optimize amplitudes and phases)
    # Actually, let's use real and imaginary parts:
    # a_i = S/n * (1 + sum_j (r_j * cos(2πj*i/n) + s_j * sin(2πj*i/n)))
    
    freqs = np.arange(1, n_modes + 1)
    # Shape: (n, 2*n_modes)
    basis = np.zeros((n, 2 * n_modes))
    for j, f in enumerate(freqs):
        angles = 2 * np.pi * f * tt
        basis[:, 2*j] = np.cos(angles)
        basis[:, 2*j+1] = np.sin(angles)
    
    # Optimize coefficients theta (2*n_modes parameters)
    # a(theta) = S/n * (1 + basis @ theta)
    # Constraint: a_i >= 0 for all i
    
    theta = rng.normal(0, 0.1, 2 * n_modes)
    start = time.monotonic()
    best_obj = np.inf
    best_a = None
    
    lr = 0.01 / n_modes
    
    for it in range(100000):
        if time.monotonic() - start > max_time:
            break
        
        a = S / n * (1.0 + basis @ theta)
        a = np.maximum(a, 1e-10)  # Ensure positive
        # Renormalize
        a = a / a.sum() * S
        
        if a.sum() < 0.01:
            break
        
        c = fft_convolve_self(a)
        max_b = c.max()
        obj = 2.0 * n * max_b / (S * S)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Compute gradient of obj w.r.t. theta
        # obj = 2n * max_b / S^2
        # d obj / d theta = 2n / S^2 * d max_b / d theta
        # d max_b / d theta ≈ d b_{k*} / d theta (at peak lag k*)
        # d b_k / d a_j = 2 * a_{k-j}
        # d a_j / d theta = S/n * basis[j, :]
        # So d b_k / d theta = 2 * sum_j a_{k-j} * S/n * basis[j, :]
        
        k_star = np.argmax(c)
        # Gradient of b_{k*} w.r.t. a
        grad_b = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad_b[j] = 2.0 * a[idx]
        
        # Chain rule: grad_theta = grad_b @ (S/n * basis)
        grad_theta = (S / n) * (basis.T @ grad_b)
        
        # Add a small regularization to keep a positive
        a_raw = S / n * (1.0 + basis @ theta)
        neg_mask = a_raw < 0
        if np.any(neg_mask):
            # Penalize negative values
            grad_theta += 0.1 * (basis.T @ np.where(neg_mask, -a_raw, 0))
        
        # Gradient step
        theta -= lr * grad_theta / (np.linalg.norm(grad_theta) + 1e-10)
        
        # Decay learning rate
        if it % 1000 == 0:
            lr *= 0.99
        
        if it % 10000 == 0:
            print("    Param iter %d: obj=%.6f" % (it, best_obj))
    
    return best_a, best_obj

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
