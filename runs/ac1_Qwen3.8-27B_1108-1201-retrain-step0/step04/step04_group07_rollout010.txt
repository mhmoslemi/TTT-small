"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using subgradient with Nesterov momentum,

Key improvements over previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.ones(n) * S / n
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def compute_subgradient(a, k, temperature=None):
    """
    Compute subgradient of max(convolve(a,a)) using top-k peaks.
    Returns (gradient, c, c_max).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    
    if temperature is not None and temperature > 0:
        # Softmax weighting
        shifted = top_vals - c_max
        exp_shifted = np.exp(shifted / temperature)
        weights = exp_shifted / exp_shifted.sum()
    else:
        # Proportional weighting
        total = top_vals.sum()
        if total < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = top_vals / total
    
    g = np.zeros(n)
    for i in range(k):
        kk = int(top_idx[i])
        wk = weights[i]
        if wk < 1e-15:
            continue
        # dc[kk]/da[i] = a[kk-i] if 0<=kk-i<n + a[i-kk] if 0<=i-kk<n
        # Term 1: a[kk-i] for i in [kk-n+1, kk] intersect [0, n-1]
        lo1 = max(0, kk - n + 1)
        hi1 = min(kk, n - 1)
        if lo1 <= hi1:
            indices = np.arange(lo1, hi1 + 1)
            a_indices = kk - indices
            g[indices] += wk * a[a_indices]
        # Term 2: a[i-kk] for i in [kk, kk+n-1] intersect [0, n-1]
        lo2 = max(kk, 0)
        hi2 = min(kk + n - 1, n - 1)
        if lo2 <= hi2:
            indices = np.arange(lo2, hi2 + 1)
            a_indices = indices - kk
            g[indices] += wk * a[a_indices]
    
    return g, c, c_max


def spectral_flatten(a, alpha, N):
    """
    One step of spectral flattening.
    Pushes |FFT(a)|^2 toward uniform.
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    p_mean = P.mean()
    if p_mean < 1e-30:
        return a.copy()
    
    ratio = P / p_mean
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * (ratio[above] ** (1.0 - alpha))
    P_new[~above] = P[~above] * (ratio[~above] ** (1.0 + alpha * 0.5))
    P_new = np.maximum(P_new, 0.0)
    
    phase = np.angle(A)
    mag = np.sqrt(P_new)
    A_new = mag * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, S)
    return a_new


def lp_peak_shave(a, K, max_time=10.0):
    """
    Solve an LP to find a direction d that reduces the top-K peaks.
    Returns the direction d and the predicted new peak.
    """
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    
    # Get top K lags
    top_idx = np.argsort(c)[::-1][:K]
    c_max = c[top_idx[0]]
    
    # Build LP: minimize t
    # s.t. c[kj] + 2 * sum_i a[kj-i]*d[i] <= t  for each top lag kj
    #      d[i] >= -a[i] for all i
    #      sum(d) = 0
    #
    # Variables: x = [d[0], ..., d[n-1], t]
    
    # Objective: minimize t -> c_obj = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each top lag kj: sum_i a[(kj-i) mod valid] * d[i] - t <= c_max - c[kj] + epsilon
    # Actually: c[kj] + 2*conv[a,d][kj] <= t
    # => 2*conv[a,d][kj] - t <= t - c[kj]  ... no
    # => 2*conv[a,d][kj] - t <= -c[kj] + t  ... that's circular
    
    # Let me reformulate:
    # minimize t
    # s.t. 2*conv[a,d][kj] - t <= -c[kj]  for each kj (wait, that's not right either)
    
    # Correct formulation:
    # We want: c[kj] + 2*conv[a,d][kj] <= t  for all kj in top K
    # => 2*conv[a,d][kj] - t <= -c[kj]  ... no
    # => 2*conv[a,d][kj] - t <= t - c[kj]  ... no, t is a variable
    
    # Standard form: A_ub @ x <= b_ub
    # For constraint: 2*sum_i A[i,kj]*d[i] - t <= -c[kj]
    # where A[i,kj] = a[kj-i] (if 0<=kj-i<n) + a[i-kj] (if 0<=i-kj<n)
    # Wait, that's dc[kj]/da[i], not conv[a,d][kj].
    
    # conv[a,d][kj] = sum_i a[kj-i]*d[i] (for valid indices)
    # So the constraint is: c[kj] + 2*sum_i a[kj-i]*d[i] <= t
    # => 2*sum_i a[kj-i]*d[i] - t <= -c[kj]
    
    # But wait, this is only for the "forward" part. The full convolution contribution
    # to lag kj from perturbation d is:
    # conv[a+d, a+d][kj] - conv[a,a][kj] ≈ 2*conv[a,d][kj] (first order)
    # conv[a,d][kj] = sum_i a[kj-i]*d[i] (with a[x]=0 for x out of [0,n-1])
    
    # So constraint: 2 * sum_i a[kj-i]*d[i] - t <= -c[kj]
    
    # Build A_ub matrix (K + n) x (n + 1)
    # First K rows: peak constraints
    # Next n rows: d[i] >= -a[i] => -d[i] <= a[i]
    
    n_constraints = K + n
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for row in range(K):
        kk = int(top_idx[row])
        # 2*sum_i a[kj-i]*d[i] - t <= -c[kj]
        # Coefficients for d[i]: 2*a[kj-i] (if 0<=kj-i<n)
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        for i in range(lo, hi + 1):
            A_ub[row, i] = 2.0 * a[kk - i]
        # Coefficient for t: -1
        A_ub[row, n] = -1.0
        # RHS: -c[kk]
        b_ub[row] = -c[kk]
    
    # Non-negativity: -d[i] <= a[i]
    for i in range(n):
        A_ub[K + i, i] = -1.0
        b_ub[K + i] = a[i]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: d[i] is free (handled by A_ub), t is free
    bounds = [(None, None)] * (n + 1)
    
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': max_time})
    except Exception:
        return None, c_max
    
    if time.monotonic() - t_start > max_time + 1:
        return None, c_max
    
    if not res.success:
        return None, c_max
    
    d = res.x[:n]
    t_opt = res.x[n]
    
    # Scale d to be a reasonable step
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return None, c_max
    
    return d, t_opt

def subgradient_nesterov(a_init, n, max_time, rng, top_k=50, target=1.50):
    """
    Subgradient method with Nesterov acceleration for minimizing max(convolve(a,a)).
    Normalizes so sum(a) = 1.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S  # Normalize to sum = 1
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    
    # Adaptive step size
    step_init = 0.1
    step_decay = 0.9995
    
    N = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        g, c, c_max = compute_subgradient(y, top_k, temperature=0.01 * c_max)
        
        cur_score = 2.0 * n * c_max
        if cur_score < best_score:
            best_score = cur_score
            best_local = y.copy()
        
        if best_score <= target:
            break
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Step size with decay
        step = step_init * (step_decay ** it)
        step = max(step, 1e-6)
        
        # Nesterov momentum update
        grad_normalized = g / gnorm
        x_new = project_simplex(y - step * grad_normalized, 1.0)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum
        a = x_new
        t = t_new
        
        # Periodic noise to escape local minima
        if it % 200 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.005, n)
            y = project_simplex(y + noise, 1.0)
            t = 1.0
        
        # Periodic spectral flattening
        if it % 500 == 250:
            y = spectral_flatten(y, alpha=0.1, N=N)
    
    return best_local, best_score

        else:
            w = 1.0
        weights.append(w)
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]
    
    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_global_start
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        alloc = min(budgets[i], remaining - 40)
        if alloc < 30:
            break
        
        print("=== Optimizing n=%d with budget %.0fs ===" % (n, alloc))
        a_result, score = full_optimize_length(n, alloc, rng, target=1.50)
        
        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  New global best: %.6f at n=%d" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))
    
    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence, score=%.4f" % global_best)
    
    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback verification: %.6f" % ver)
    
    print("Final answer: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
