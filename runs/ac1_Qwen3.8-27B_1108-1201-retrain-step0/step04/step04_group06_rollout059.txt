    theta_val = (cumsum[rho - 1] - S_val) / rho
    return np.maximum(v - theta_val, 0.0)


def full_conv(a):
    n = len(a)
    A = fft(a, F_SIZE)
    c = np.real(ifft(A * A, F_SIZE))
    return c[:2 * n - 1]


def compute_score(a):
    """Score = max of self-convolution (when normalized to sum=sqrt(2n))."""
    c = full_conv(a)
    return c.max()


def spectral_step(a, top_k, alpha, rng, use_phase_noise=False):
    """One spectral shaping step. Returns new_a."""
    n = len(a)
    S_local = a.sum()
    A = fft(a, F_SIZE)
    P = np.abs(A) ** 2
    c = np.real(ifft(A * A, F_SIZE))[:2 * n - 1]
    c_max = c.max()
    
    # Find top-k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    excess = np.maximum(top_vals - c_max * 0.95, 0.0)  # peaks near the max
    if excess.sum() < 1e-30:
        excess = top_vals
    w = excess / excess.sum()
    
    # Compute effective cosine
    m = np.arange(F_SIZE)
    cos_eff = np.zeros(F_SIZE)
    for i in range(top_k):
        cos_eff += w[i] * np.cos(2.0 * np.pi * m * int(top_idx[i]) / F_SIZE)
    
    # Exponential shaping
    P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))
    
    if use_phase_noise and rng is not None:
        phase = np.angle(A) + rng.normal(0, 0.03 * alpha, F_SIZE)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    
    a_new = np.real(ifft(A_new, F_SIZE))[:n]
    a_new = project_simplex(a_new, S_local)
    return a_new, c_max


def spectral_loop(a_init, max_time, rng, target=1.50, top_k_init=5, alpha_init=0.5):
    """Iterative spectral peak reduction with backtracking."""
    n = len(a_init)
    a = a_init.copy()
    S_local = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    stall = 0
    it = 0
    
    while time.monotonic() < max_time:
        it += 1
        # Adaptive parameters
        top_k = min(64, top_k_init + it // 30)
        alpha = alpha_init / (1.0 + 0.03 * it)
        
        # Try multiple alphas (backtracking)
        best_trial_score = best_score
        best_trial_a = None
        for alpha_trial in [alpha, alpha * 0.5, alpha * 2.0]:
            a_trial, _ = spectral_step(a, top_k, alpha_trial, rng, use_phase_noise=(it % 50 == 0))
            sc = compute_score(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial.copy()
        
        if best_trial_a is not None:
            a = best_trial_a
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 200:
            # Perturb
            a = project_simplex(a + rng.normal(0, 0.003 * S_local / np.sqrt(n), n), S_local)
            stall = 0
        
        if best_score <= target:
            break
    
    return best_a, best_score


def nesterov_loop(a_init, max_time, rng, top_k_init=20, top_k_max=256):
    """Nesterov-accelerated projected subgradient descent."""
    n = len(a_init)
    S_local = a_init.sum()
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        it = int(t)
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = full_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        w = top_vals / top_vals.sum()
        
        # Gradient in time domain
        g = np.zeros(n)
        for i in range(top_k):
            k = int(top_idx[i])
            wi = w[i]
            # ∂c[k]/∂a[i] = 2*a[k-i] (valid range)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i2 in range(lo, hi + 1):
                j2 = k - i2
                if 0 <= j2 < n:
                    g[i2] += wi * 2.0 * y[j2]
        
        # Combine gradients (since we have top-k, g is already weighted sum)
        # Actually, the subgradient of max(c) is ∂c[k*]/∂a for the argmax k*
        # For top-k: it's the weighted sum of individual gradients
        # g[i] = sum_k w_k * 2 * y[k-i] (for valid k-i)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S_local / (n * (1.0 + 0.001 * t))
        x_new = project_simplex(y - step * g / max(gnorm, 1e-15), S_local)
        
        if it % 10 == 0:
            cx = full_conv(x_new)
            if cx.max() < best_score:
                best_score = cx.max()
                best_a = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S_local / s_y)
        else:
            y = x_new.copy()
        a = x_new
        t = t_new
        
        if it % 1000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.003 * S_local / np.sqrt(n), n), S_local)
            t = 1.0
    
    return best_a, best_score


def coord_descent(a_init, max_time, rng, n_top=30, n_rounds=500):
    """Coordinate descent with exact incremental convolution updates."""
    n = len(a_init)
    S_local = a_init.sum()
    a = a_init.copy()
    c = full_conv(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.4, 1.6]
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        peak_lag = int(np.argmax(c))
        
        # Find top-contributing coordinates at peak lag
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_lag - i
            contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_coords:
            i = int(idx)
            if time.monotonic() - start > max_time:
                break
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue
            
            orig_val = a[i]
            best_frac = 1.0
            best_cmax = c_max
            
            # Precompute the "other" max (max of unaffected lags)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Build the vector of affected contributions
            # For lag k in [k_lo, k_hi]: Δc[k] = 2*(a[i]-orig)*a[k-i] (+ (a[i]-orig)^2 if k==2i)
            affected = np.arange(k_lo, k_hi + 1)
            other_idx = np.ones(len(c), dtype=bool)
            other_idx[k_lo:k_hi + 1] = False
            other_max = c[other_idx].max() if other_idx.any() else 0.0
            
            for frac in multipliers:
                new_val = orig_val * frac
                delta = new_val - orig_val
                # New c for affected lags
                c_affected = c[k_lo:k_hi + 1].copy()
                for k in range(k_lo, k_hi + 1):
                    j2 = k - i
                    if 0 <= j2 < n:
                        c_affected[k - k_lo] += 2.0 * delta * a[j2]
                    if k == 2 * i:
                        c_affected[k - k_lo] += delta * delta
                
                new_cmax = max(c_affected.max(), other_max)
                if new_cmax < best_cmax:
                    best_cmax = new_cmax
                    best_frac = frac
            
            a[i] = orig_val * best_frac
            c = full_conv(a)  # Recompute for exactness
            if best_frac != 1.0:
                improved = True
        
        # Rescale
        s = a.sum()
        if s > 1e-15:
            a = a * (S_local / s)
            c = full_conv(a)
        
        if not improved and rng.random() < 0.3:
            top_coords2 = np.argsort(contribs)[::-1][:min(5, n_top)]
            for idx2 in top_coords2:
                a[int(idx2)] *= rng.uniform(0.5, 1.5)
            s = a.sum()
            if s > 1e-15:
                a = a * (S_local / s)
            c = full_conv(a)
    
    return best_a, best_score

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S_local = np.sqrt(2.0 * float(n))

    return a * (S_local / s)


def project_simplex(v, S_val):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S_val) / idx
    valid = theta < w
    if not np.any(valid):
        return np.full(n, S_val / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta_val = (cumsum[rho - 1] - S_val) / rho
    return np.maximum(v - theta_val, 0.0)


def full_conv(a):

            break

    return best_a, best_score


def coord_descent(a_init, max_time, rng, n_top=25, n_rounds=5000):
    n = len(a_init)
    S_local = a_init.sum()
    a = a_init.copy()
    c = full_conv(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.1])

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break

        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        peak_lag = int(np.argmax(c))
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)
        j_arr = peak_lag - np.arange(lo, hi + 1)
        contribs = np.zeros(n)
        contribs[lo:hi + 1] = a[lo:hi + 1] * a[j_arr]

        top_coords = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for idx in top_coords:
            i = int(idx)
            if time.monotonic() - start > max_time:
                break
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue
            orig_val = a[i]
            if orig_val < 1e-12:
                continue

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            other_max = 0.0
            if k_lo > 0:
                other_max = max(other_max, c[:k_lo].max())
            if k_hi + 1 < 2 * n - 1:
                other_max = max(other_max, c[k_hi + 1:].max())

            best_frac = 1.0
            best_cmax = c_max

            j_range = np.arange(k_lo, k_hi + 1) - i
            valid_mask = (j_range >= 0) & (j_range < n)
            a_valid = a[j_range[valid_mask]]
            c_base = c[k_lo:k_hi + 1]

            for frac in multipliers:
                delta = orig_val * (frac - 1.0)
                c_aff = c_base + 2.0 * delta * np.where(valid_mask, a_valid, 0.0)
                two_i = 2 * i
                if k_lo <= two_i <= k_hi:
                    c_aff[two_i - k_lo] += delta * delta
                new_cmax = max(c_aff.max(), other_max)
                if new_cmax < best_cmax:
                    best_cmax = new_cmax
                    best_frac = frac

            if best_frac != 1.0:
                a[i] = orig_val * best_frac
                improved = True

        s = a.sum()
        if s > 1e-15:
            a = a * (S_local / s)
        c = full_conv(a)

        if not improved and rng.random() < 0.15:
            pert_n = min(5, len(top_coords))
            for idx2 in top_coords[:pert_n]:
                a[int(idx2)] *= rng.uniform(0.4, 1.6)
            s = a.sum()
            if s > 1e-15:
                a = a * (S_local / s)
            c = full_conv(a)

    return best_a, best_score


def random_freq_perturb(a, sigma, rng):
    n = len(a)
    S_local = a.sum()
    A = fft(a, F_SIZE)
    noise = sigma * (rng.standard_normal(F_SIZE) + 1j * rng.standard_normal(F_SIZE))
    A_new = A + noise
    a_new = np.real(ifft(A_new, F_SIZE))[:n]
    a_new = project_simplex(a_new, S_local)
    return a_new


def structured_init(n, rng, method="mixed"):
    t = np.arange(n) / n
    if method == "sin":
        a = (0.5 + 0.3 * np.sin(2 * np.pi * 3 * t)
             + 0.2 * np.sin(2 * np.pi * 7 * t + 0.5)
             + 0.1 * np.sin(2 * np.pi * 13 * t + 1.0)
             + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "two_peak":
        a = 0.3 + 0.5 * np.exp(-20.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-20.0 * (t - 0.7) ** 2)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.15 * rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = 128
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, n // n_pieces)
        if len(a) < n:
            a = np.concatenate([a, 0.3 + 0.7 * rng.random(n - len(a))])
        a = a[:n]
    elif method == "cos_window":
        a = 0.3 + 0.5 * np.cos(np.pi * t) ** 2 + 0.2 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, 3.0)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = MAIN_N
    S_local = S_VAL

    print("n=%d, S=%.4f, F_SIZE=%d, budget=%.0fs" % (n, S_local, F_SIZE, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_cand):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_cand, n)
        sc = compute_score(a_cand)
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            return True
        return False

    # Build initializations
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
            print("  Loaded height_sequence_1 (len=%d)" % given.size)
    except Exception:
        pass

    for method in ["sin", "chirp", "two_peak", "alternating", "piecewise", "cos_window"]:
        inits.append((method, structured_init(n, rng, method)))
    inits.append(("rand", 0.3 + 0.7 * rng.random(n)))

    print("  %d initializations" % len(inits))

    # === Phase 1: Quick spectral from each init (25% time) ===
    print("=== Phase 1: Spectral optimization ===")
    t_phase1 = time.monotonic()
    phase1_budget = min((deadline - t_phase1) * 0.25, 480.0)
    per_init = phase1_budget / max(1, len(inits))

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        a_norm = safe_normalize(arr, n)
        sc0 = compute_score(a_norm)
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: %.6f" % (name, global_best))

        tb = min(per_init, 80.0)
        if tb < 2:
            continue
        a_sw, sc_sw = spectral_loop(a_norm, time.monotonic() + tb, rng,
                                    target=1.50, top_k_init=5, alpha_init=0.6)
        if update_best(a_sw):
            print("  [%s] spectral: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)
    if global_a is None:
        global_a = safe_normalize(structured_init(n, rng, "sin"), n)
        global_best = compute_score(global_a)

    # === Phase 2: Alternating spectral + coordinate (40% time) ===
    print("=== Phase 2: Alternating optimization ===")
    t_phase2 = time.monotonic()
    phase2_end = t_phase2 + min((deadline - t_phase2) * 0.40, 900.0)

    a_curr = global_a.copy()
    stall_count = 0

    while time.monotonic() < phase2_end:
        rem = phase2_end - time.monotonic()
        if rem < 5:
            break

        # Spectral burst
        tb = min(rem * 0.35, 40.0)
        if tb > 2:
            a_sw, _ = spectral_loop(a_curr, time.monotonic() + tb, rng,
                                    target=1.50, top_k_init=8, alpha_init=0.4)
            if update_best(a_sw):
                a_curr = a_sw.copy()
                stall_count = 0
                print("  [P2] spectral: %.6f" % global_best)
            else:
                # Still use the result as new starting point
                a_curr = a_sw.copy()
                stall_count += 1

        # Coordinate descent
        rem = phase2_end - time.monotonic()
        if rem > 3:
            tb = min(rem * 0.35, 30.0)
            a_cd, _ = coord_descent(a_curr, tb, rng, n_top=20, n_rounds=2000)
            if update_best(a_cd):
                a_curr = a_cd.copy()
                stall_count = 0
                print("  [P2] coord: %.6f" % global_best)
            else:
                a_curr = a_cd.copy()
                stall_count += 1

        # Exploration
        rem = phase2_end - time.monotonic()
        if rem > 15 and stall_count >= 3:
            for sigma in [0.03, 0.08, 0.15, 0.25]:
                a_p = random_freq_perturb(a_curr, sigma, rng)
                if update_best(a_p):
                    a_curr = a_p.copy()
                    stall_count = 0
                    print("  [P2] perturb(%.2f): %.6f" % (sigma, global_best))
                    break
                a_sw2, _ = spectral_loop(a_p, time.monotonic() + 3, rng,
                                         target=1.50, top_k_init=5, alpha_init=0.5)
                if update_best(a_sw2):
                    a_curr = a_sw2.copy()
                    stall_count = 0
                    print("  [P2] perturb+spec(%.2f): %.6f" % (sigma, global_best))
                    break
            stall_count = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # === Phase 3: Intensive coordinate descent polish (25% time) ===
    print("=== Phase 3: Coordinate descent polish ===")
    t_phase3 = time.monotonic()
    rem3 = min(deadline - t_phase3) * 0.25
    a_cp, _ = coord_descent(global_a, rem3, rng, n_top=35, n_rounds=20000)
    if update_best(a_cp):
        print("  [P3] coord: %.6f" % global_best)
    else:
        print("  [P3] no improvement, keeping %.6f" % global_best)

    # === Phase 4: Final exploration + polish (10% time) ===
    print("=== Phase 4: Final exploration ===")
    t_phase4 = time.monotonic()
    rem4 = max(0.0, deadline - t_phase4)

    # Try fine spectral with very small alpha
    if rem4 > 5:
        tb = min(rem4 * 0.3, 30.0)
        a_fs, _ = spectral_loop(global_a, time.monotonic() + tb, rng,
                                target=1.50, top_k_init=20, alpha_init=0.15)
        if update_best(a_fs):
            print("  [P4] fine spectral: %.6f" % global_best)

    # Final coordinate descent
    rem4 = max(0.0, deadline - time.monotonic())
    if rem4 > 3:
        tb = min(rem4 * 0.5, 60.0)
        a_fd, _ = coord_descent(global_a, tb, rng, n_top=40, n_rounds=5000)
        if update_best(a_fd):
            print("  [P4] final coord: %.6f" % global_best)

    # One more spectral pass from best
    rem4 = max(0.0, deadline - time.monotonic())
    if rem4 > 5:
        tb = min(rem4 * 0.5, 30.0)
        a_final, _ = spectral_loop(global_a, time.monotonic() + tb, rng,
                                   target=1.50, top_k_init=15, alpha_init=0.2)
        if update_best(a_final):
            print("  [P4] final spectral: %.6f" % global_best)

    # Final matrix coordinate descent
    rem4 = max(0.0, deadline - time.monotonic())
    if rem4 > 3:
        a_fm, _ = coord_descent(global_a, rem4, rng, n_top=50, n_rounds=10000)
        if update_best(a_fm):
            print("  [P4] matrix coord: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("Verification FAILED, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
