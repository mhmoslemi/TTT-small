def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a):
    """Compute full linear self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = compute_conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def lse_gradient(a, beta, N_grad):
    """
    Compute gradient of LSE(beta, conv(a,a)) with respect to a.
    Uses FFT-based cross-correlation for O(n log n) computation.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using LSE-smoothed projected gradient
descent with Nesterov acceleration, adaptive temperature, and multi-length search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.maximum(v, 0.0) * (S / max(v.sum(), 1e-30))
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def self_convolve(a, N=None):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def score_fn(a):
    """Compute objective."""
    n = len(a)
    c = self_convolve(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def lse_grad_and_conv(a, beta, N_grad=None):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*conv))) and the conv values.
    grad[j] = (2/beta) * sum_m p[m] * a[m-j] (valid)
    """
    n = len(a)
    conv = self_convolve(a)
    
    # Numerically stable softmax
    conv_max = conv.max()
    exp_vals = np.exp(beta * (conv - conv_max))
    p = exp_vals / exp_vals.sum()
    
    # Gradient via cross-correlation
    # grad[j] = (2/beta) * sum_{k=0}^{n-1} p[j+k] * a[k]
    # = (2/beta) * (a_rev * p)[j + n - 1]
    if N_grad is None:
        N_grad = 1 << (3*n - 2).bit_length()
    
    a_rev = a[::-1]
    A_rev = fft(a_rev, N_grad)
    P = fft(p, N_grad)
    conv_fp = np.real(ifft(A_rev * P))
    
    grad = (2.0 / beta) * conv_fp[n-1 : 2*n-1]
    return grad, conv, p

        if true_score < best_score:
