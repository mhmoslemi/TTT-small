```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
```

This is efficient: one FFT of `p` and one of `a`, then multiply and IFFT.

```python
"""
Minimize 2*n*max_conv/sum^2 using Nesterov-accelerated LSE optimization
with spectral initialization and adaptive peak targeting.

Key innovations:
1. Iterative spectral flattening initialization
2. Nesterov (FISTA) acceleration with adaptive step size
3. Adaptive beta annealing (simulated annealing in weight space)
4. Targeted perturbation for escape from local minima
5. Multi-resolution coarse-to-fine optimization
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute auto-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_gradient(a, c, beta, N_grad):
    """
    Compute gradient of LSE-smoothed max of c.
    Returns (gradient, lse_value, c_max, p) where p = softmax(beta*(c-c_max)).
    """
    n = len(a)
    c_max = c.max()
    exp_vals = np.exp(beta * (c - c_max))
    total = exp_vals.sum()
    if total < 1e-300:
        # Fallback: uniform weights
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_vals / total
    lse_val = c_max + np.log(total) / beta
    
    # Gradient: 2 * cross_correlation(p, a)[:n]
    p_pad = np.zeros(N_grad)
    p_pad[:len(c)] = p
    a_pad = np.zeros(N_grad)
    a_pad[:n] = a
    
    grad = 2.0 * np.real(ifft(fft(p_pad) * np.conj(fft(a_pad))))[:n]
    return grad, lse_val, c_max, p


def topk_gradient(a, c, top_k, N_grad):
    """
    Compute gradient using only top-k peaks (with softmax weighting among them).
    More stable than LSE at high effective beta.
    """
    n = len(a)
    c_max = c.max()
    
    # Get top k indices
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    top_vals = c[top_idx]
    # Softmax over top-k
    max_top = top_vals.max()
    exp_top = np.exp(50.0 * (top_vals - max_top))  # High effective beta for top-k
    total = exp_top.sum()
    if total < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = exp_top / total
    
    # Construct sparse weight vector
    p = np.zeros(len(c))
    for i, idx in enumerate(top_idx):
        p[int(idx)] = weights[i]
    
    p_pad = np.zeros(N_grad)
    p_pad[:len(c)] = p
    a_pad = np.zeros(N_grad)
    a_pad[:n] = a
    
    grad = 2.0 * np.real(ifft(fft(p_pad) * np.conj(fft(a_pad))))[:n]
    return grad


def spectral_init(n, N, rng, n_iters=10):
    """
    Iterative spectral flattening initialization.
    Creates a sequence whose auto-convolution is as flat as possible.
    """
    a = rng.random(n) + 0.1
    a = a / a.sum() * np.sqrt(2.0 * n)
    
    for _ in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        # Flatten the power spectrum
        P_flat = P ** 0.5  # Compress dynamic range
        # Preserve the DC component (related to sum)
        phase = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(np.abs(a_new) + 0.01, np.sqrt(2.0 * n))
    
    return a


def nesterov_fista(a_init, n, S, N, N_grad, max_time, rng,
                    beta_init=1.0, beta_max=2000.0, beta_rate=1.002,
                    top_k_switch=100.0, top_k=16):
    """
    FISTA (Nesterov accelerated projected gradient) on LSE-smoothed objective.
    Switches to top-k gradient when beta > top_k_switch.
    """
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    beta = beta_init
    step = S / (n * beta) * 2.0
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    stagnant = 0
    
    c = compute_conv(y, N)
    cur_max = c.max()
    if cur_max < best_cmax:
        best_cmax = cur_max
        best_a = y.copy()
    
    while time.monotonic() - start_time < max_time:
        it += 1
        elapsed = time.monotonic() - start_time

        if it % 10 == 0:
            beta = min(beta_max, beta * beta_rate)

        c_y = compute_conv(y, N)
        cur_max_y = c_y.max()
        
        if beta > top_k_switch:
            grad = topk_gradient(y, c_y, top_k, N_grad)
        else:
            grad, _, _, _ = lse_gradient(y, c_y, beta, N_grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size

        step = 2.0 * S / (n * max(beta, 1.0))

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)

                y_new = project_simplex(y_new + rng.normal(0, noise_scale, n), S)
                t = 1.0
                stagnant = 0
        
        a = x_new
        y = y_new
        t = t_new
    
    return best_a, best_cmax


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def targeted_perturb(a, c, n, S, N, rng, scale=0.005):
    """
    Perturb a in the direction that reduces the top peak.
    The top peak is at index m*, and grad of c[m*] is 2*shift(a, m*).
    We subtract a small multiple of this gradient (projected).
    """
    m_star = int(np.argmax(c))
    # Gradient of c[m_star] w.r.t. a: 2*a[m_star - j] for valid j
    # This is a shifted version of a
    grad = np.zeros(n)
    for j in range(n):
        idx = m_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    
    # Normalize and apply
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a
    
    direction = -grad / gnorm  # Descent direction for the peak
    step = scale * S
    a_new = project_simplex(a + step * direction, S)
    return a_new


def coarse_fine_optimize(a_init, n, S, N, N_grad, max_time, rng):

    Multi-resolution optimization:
    1. Downsample to n/4, optimize

    S_coarse = np.sqrt(2.0 * n_coarse)
    N_coarse = 1 << (2 * n_coarse - 1).bit_length()
    N_grad_coarse = 1 << (3 * n_coarse - 2).bit_length()
    
    # Downsample by averaging
    a_coarse = a_init[:n_coarse * 4].reshape(n_coarse, 4).mean(axis=1)
    a_coarse = safe_normalize(a_coarse, n_coarse, S_coarse)
    
    # Optimize at coarse resolution (use 30% of time)
    coarse_time = 0.3 * max_time
    a_c, sc_c = nesterov_fista(a_coarse, n_coarse, S_coarse, N_coarse, N_grad_coarse,

    # Upsample
    a_up = np.repeat(a_c, 4)[:n]
    a_up = safe_normalize(a_up, n, S)
    
    # Refine at full resolution
    fine_time = 0.7 * max_time
    a_f, sc_f = nesterov_fista(a_up, n, S, N, N_grad, fine_time, rng,

    return a_f, sc_f
```

```python
"""
Minimize 2*n*max_conv/sum^2 using refined L-BFGS on LSE-smoothed objective
with adaptive peak targeting, spectral initialization, and targeted perturbation.

Key improvements over previous approach:
1. Better spectral initialization via iterative power spectrum flattening
2. Adaptive top-k targeting: focus on the few peaks that matter most
3. Improved line search with curvature estimation
4. Longer final polishing with very high effective beta (top-2 targeting)
5. Targeted perturbation using peak gradient for escape from local minima
6. Better time budget allocation favoring the fine-tuning phases

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_conv(a, N):
    """Compute full auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_lse_grad(a, c, beta, N_grad):
    """
    Compute gradient of LSE-smoothed max of c.
    Returns (grad, lse_val, c_max).
    """
    n = len(a)
    c_max = c.max()
    exp_vals = np.exp(beta * (c - c_max))
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_vals / total
    lse_val = c_max + np.log(total) / beta

    p_pad = np.zeros(N_grad)
    p_pad[:len(c)] = p
    a_pad = np.zeros(N_grad)
    a_pad[:n] = a
    grad = 2.0 * np.real(ifft(fft(p_pad) * np.conj(fft(a_pad))))[:n]
    return grad, lse_val, c_max


def compute_topk_grad(a, c, top_k, N_grad, k_beta=100.0):
    """
    Compute gradient using softmax weighting over top-k peaks.
    More targeted than full LSE at high effective beta.
    Returns (grad, c_max).
    """
    n = len(a)
    c_max = c.max()
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    max_top = top_vals.max()
    exp_top = np.exp(k_beta * (top_vals - max_top))
    total = exp_top.sum()
    if total < 1e-300:
        weights = np.ones(k) / k
    else:
        weights = exp_top / total

    p = np.zeros(len(c))
    for i in range(k):
        p[int(top_idx[i])] = weights[i]

    p_pad = np.zeros(N_grad)
    p_pad[:len(c)] = p
    a_pad = np.zeros(N_grad)
    a_pad[:n] = a
    grad = 2.0 * np.real(ifft(fft(p_pad) * np.conj(fft(a_pad))))[:n]
    return grad, c_max


def lbfgs_two_loop(s_hist, y_hist, g, m_max=10):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        sy = np.dot(s_i, y_i)
        beta_i = sy / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lse_lbfgs_block(a_init, n, S, N, N_grad, max_time, rng,
                    beta_init=5.0, beta_max=200.0, beta_rate=1.01,
                    m_lbfgs=10, use_topk=False, top_k=8, k_beta=100.0):
    """
    L-BFGS on LSE-smoothed (or top-k) objective with adaptive beta.
    Returns (best_a, best_cmax).
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    c = compute_conv(a, N)
    c_max = c.max()
    if use_topk:
        g, c_max = compute_topk_grad(a, c, top_k, N_grad, k_beta)
    else:
        g, _, c_max = compute_lse_grad(a, c, beta, N_grad, N_grad)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    # Initial step size estimate
    gnorm = np.linalg.norm(g)
    step_scale = 2.0 * S / (n * max(beta, 1.0)) if gnorm > 0 else 1.0

    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Update beta periodically
        if it % 50 == 0:
            beta = min(beta_max, beta * beta_rate)

        # Trim history
        if it % 25 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            direction = -g / max(gnorm, 1e-18)
            dir_norm = 1.0

        # Line search with backtracking
        alpha_ls = step_scale
        a_new = a
        c_new = c
        cmax_new = c_max
        found = False

        for ls in range(20):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1.0 + 1e-10) or ls == 19:
                a_new = candidate
                c_new = c_ls
                cmax_new = cm_ls
                if cm_ls <= c_max * (1.0 + 1e-10):
                    found = True
                    step_scale = alpha_ls * 1.5
                    break
            alpha_ls *= 0.5

        if not found:
            step_scale = max(alpha_ls * 0.5, 1e-10 * S)

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        # Compute new gradient
        if use_topk:
            g_new, c_new_max = compute_topk_grad(a_new, c_new, top_k, N_grad, k_beta)
        else:
            g_new, _, c_new_max = compute_lse_grad(a_new, c_new, beta, N_grad, N_grad)

        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c = c_new
        c_max = c_new_max
        gnorm = np.linalg.norm(g)

        # Periodic perturbation to escape local minima
        if it % 300 == 0 and rng.random() < 0.25:
            noise_scale = 0.004 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            c = compute_conv(a, N)
            c_max = c.max()
            if use_topk:
                g, c_max = compute_topk_grad(a, c, top_k, N_grad, k_beta)
            else:
                g, _, c_max = compute_lse_grad(a, c, beta, N_grad, N_grad)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

    return best_a, best_cmax


def spectral_flatten_init(n, N, rng, n_iters=20):
    """
    Iterative spectral power flattening for initialization.
    Creates sequence with as-flat auto-convolution as possible.
    """
    S = np.sqrt(2.0 * n)
    a = rng.random(n) + 0.1
    a = a / a.sum() * S

    for _ in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        # Compress dynamic range of power spectrum
        P_min = P.min()
        P_max = P.max()
        if P_max > P_min + 1e-10:
            P_norm = (P - P_min) / (P_max - P_min)
            P_new = P_min + (P_max - P_min) * (P_norm ** 0.4)
        else:
            P_new = P
        # Preserve DC
        P_new[0] = P[0]
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(np.abs(a_new) + 0.01, S)

    return a


def targeted_perturb(a, c, n, S, rng, scale=0.01):
    """
    Perturb in the descent direction of the maximum peak.
    For peak at lag m*, grad of c[m*] w.r.t. a is 2*shifted(a).
    """
    m_star = int(np.argmax(c))
    grad = np.zeros(n)
    # grad[j] = 2 * a[m_star - j] if 0 <= m_star - j < n
    lo = max(0, m_star - (n - 1))
    hi = min(n - 1, m_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m_star - j]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a

    direction = -grad / gnorm
    step = scale * S
    a_new = project_simplex(a + step * direction, S)
    return a_new


def nesterov_block(a_init, n, S, N, N_grad, max_time, rng,
                   beta_init=50.0, beta_max=3000.0, beta_rate=1.003,
                   top_k=4, k_beta=200.0):
    """
    Nesterov-accelerated projected gradient with top-k targeting.
    For fine polishing with very high effective beta.
    """
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t_mom = 1.0
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    stagnant = 0

    c = compute_conv(y, N)
    cur_max = c.max()
    best_cmax = cur_max
    best_a = y.copy()

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 20 == 0:
            beta = min(beta_max, beta * beta_rate)

        c_y = compute_conv(y, N)
        cur_y = c_y.max()

        # Use top-k gradient
        grad, _ = compute_topk_grad(y, c_y, top_k, N_grad, k_beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = 2.0 * S / (n * max(beta, 10.0))

        # Projected gradient step
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)

        # Track best
        if it % 3 == 0:
            c_x = compute_conv(x_new, N)
            new_max = c_x.max()
            if new_max < best_cmax:
                best_cmax = new_max
                best_a = x_new.copy()
                stagnant = 0
            else:
                stagnant += 1

            if stagnant > 30:
                # Targeted perturbation
                a = targeted_perturb(a, c_x, n, S, rng, scale=0.008)
                y = a.copy()
                t_mom = 1.0
                stagnant = 0

        a = x_new
        y = y_new
        t_mom = t_new

    return best_a, best_cmax


def make_init_rand(n, rng, S):
    return safe_normalize(rng.random(n) + 0.05, n, S)


def make_init_chirp(n, rng, S):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.35 * np.sin(2 * np.pi * (4 * tt + 1.5 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return safe_normalize(np.clip(a, 0.01, None), n, S)


def make_init_multifreq(n, rng, S):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n) * 0.8
    for freq in [3, 7, 13, 23, 41]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return safe_normalize(np.clip(a, 0.01, None), n, S)


def make_init_binary(n, rng, S):
    a = (rng.random(n) > 0.35).astype(np.float64) * (0.4 + 0.6 * rng.random(n))
    return safe_normalize(a, n, S)


def make_init_fourier(n, N, rng, S):
    """Fourier-based initialization with controlled power spectrum."""
    P = rng.exponential(1.0, N)
    P[0] = 0.0
    P = np.maximum(P, 0.01)
    # Flatten
    P = P ** 0.5
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    A[0] = 0
    a_full = np.real(ifft(A))[:n]
    return safe_normalize(np.abs(a_full) + 0.01, n, S)

        # Switch to top-k mode for later rounds
        use_topk = round_idx >= 3
        tk = min(16, 4 + round_idx)

        a_new, sc_new = lse_lbfgs_block(a_curr, n, S, N, N_grad, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         beta_rate=1.012, m_lbfgs=12,
                                         use_topk=use_topk, top_k=tk, k_beta=80.0)

        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            ratio = 2.0 * n * global_best / S ** 2
            print("  [P2 r%d] maxconv=%.8f ratio=%.6f (topk=%s)" % (round_idx, global_best, ratio, tk if use_topk else "LSE"))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count >= 2:
                c_curr = compute_conv(a_curr, N)
                a_curr = targeted_perturb(a_curr, c_curr, n, S, rng, scale=0.01)
                stall_count = 0

    # --- Phase 3: Nesterov high-beta polishing ---
    print("=== Phase 3: Nesterov high-beta polish ===")
    t3 = time.monotonic()
    phase3_budget = min(600.0, (deadline - t3) * 0.30)

    a_ng, sc_ng = nesterov_block(global_a, n, S, N, N_grad, phase3_budget, rng,
                                  beta_init=100.0, beta_max=5000.0,
                                  beta_rate=1.003, top_k=4, k_beta=200.0)
    _update_best(a_ng, "nesterov")

    # --- Phase 4: Ultra-targeted (top-2) L-BFGS ---
    print("=== Phase 4: Ultra-targeted top-2 ===")
    t4 = time.monotonic()
    phase4_budget = min(300.0, (deadline - t4) * 0.20)

    a_u, sc_u = lse_lbfgs_block(global_a, n, S, N, N_grad, phase4_budget, rng,
                                 beta_init=200.0, beta_max=3000.0,
                                 beta_rate=1.005, m_lbfgs=8,
                                 use_topk=True, top_k=2, k_beta=300.0)
    _update_best(a_u, "top2_LBFGS")

    # --- Phase 5: Final Nesterov with top-2 ---
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("=== Phase 5: Final top-2 Nesterov ===")
        a_f, sc_f = nesterov_block(global_a, n, S, N, N_grad, min(remaining - 5, 200.0), rng,
                                    beta_init=500.0, beta_max=10000.0,
                                    beta_rate=1.003, top_k=2, k_beta=300.0)
        _update_best(a_f, "top2_final")

    # --- Final small perturbation search ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Final perturbation search ===")
        for trial in range(10):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.002 * S / np.sqrt(n)
            a_p = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            _update_best(a_p, "perturb%d" % trial)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("  Rejected by evaluator, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, best maxconv=%.8f, ratio=%.6f" % (
        len(global_a), global_best, 2.0 * n * global_best / S ** 2))
    return [float(x) for x in global_a]
```
