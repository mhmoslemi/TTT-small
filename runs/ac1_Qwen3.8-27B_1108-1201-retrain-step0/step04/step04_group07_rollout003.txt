import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))

def spectral_magnitude_flatten(a, N, alpha, rng):
    """
    Flatten the magnitude spectrum |A(k)|^2 toward uniform.
    Pushes high-magnitude bins down and low-magnitude bins up.
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2

    # Don't touch DC (k=0)
    P_non_dc = P[1:]
    p_median = np.median(P_non_dc)
    if p_median < 1e-30:
        return a

    ratio = P_non_dc / p_median
    # Compress: new_ratio = ratio^alpha for ratio > 1
    # Expand: new_ratio = ratio^(1/alpha) for ratio < 1
    P_new_non_dc = P_non_dc.copy()
    above = ratio >= 1.0
    P_new_non_dc[above] = P_non_dc[above] * np.power(ratio[above], alpha - 1.0)

    P_new[1:] = P_new_non_dc

    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, S)
    return a_new

    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_rand", (0.3 + 0.7 * smooth) * (S / (n * 0.65))))

    # Multi-frequency
    freqs = [1, 3, 5, 7, 11]
    multi = np.zeros(n)
    amps = [1.0, 0.5, 0.3, 0.2, 0.15]
    for f, amp in zip(freqs, amps):
        multi += amp * np.cos(2 * np.pi * f * tt + rng.random() * 2 * np.pi)
    multi = np.abs(multi)
    ms = multi.sum()
    if ms > 0:
        inits.append(("multi_freq", multi * (S / ms)))

    # Windowed: raised cosine
    raised = 0.5 * (1 + np.cos(2 * np.pi * tt))
    rs = raised.sum()
    if rs > 0:
        inits.append(("raised_cos", raised * (S / rs)))

    # Anti-periodic: alternating high/low with smooth transitions
    alt = 0.5 + 0.4 * np.sin(2 * np.pi * n / 2 * tt)
    alt = np.maximum(alt, 0.01)
    as_ = alt.sum()
    inits.append(("alt", alt * (S / as_)))

    # Random
    rand = rng.random(n) + 0.1
    rs = rand.sum()
    inits.append(("random", rand * (S / rs)))

    return inits


def optimize_single_length(n, budget_s, rng, target=1.50):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    global_best = np.inf
    global_a = None
    t_start = time.monotonic()

    inits = make_initializations(n, rng, S)
    n_inits = len(inits)

    # Allocate time: give more time to better initializations
    # First pass: quick optimization of all inits
    # Second pass: extended optimization of the best ones

    # Phase 1: Quick screening
    per_init_quick = max(3.0, budget_s * 0.3 / n_inits)
    quick_results = []

    for name, arr in inits:
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s * 0.35:
            break
        remaining = per_init_quick
        a_screen, sc_screen = subgradient_optimize(arr, n, S, remaining, rng,
                                                    top_k_init=min(50, n // 100),
                                                    peak_target=target)
        quick_results.append((sc_screen, a_screen.copy(), name))
        if sc_screen < global_best:
            global_best = sc_screen
            global_a = a_screen.copy()
        print("  [quick %s n=%d] score=%.6f (t=%.0fs)" % (name, n, sc_screen, time.monotonic() - t_start))

    # Phase 2: Extended optimization of top results
    quick_results.sort(key=lambda x: x[0])
    top_k_results = min(4, len(quick_results))
    remaining_budget = budget_s - (time.monotonic() - t_start) - 5.0
    if remaining_budget > 10:
        per_init_ext = remaining_budget / (top_k_results * 2)

        for i in range(top_k_results):
            elapsed = time.monotonic() - t_start
            rem = budget_s - elapsed - 5.0
            if rem < 5:
                break
            alloc = min(per_init_ext, rem)
            _, a_start, name = quick_results[i]
            print("  [ext %s n=%d] budget=%.0fs" % (name, n, alloc))
            a_ext, sc_ext = subgradient_optimize(a_start, n, S, alloc, rng,
                                                  top_k_init=min(200, n // 50),
                                                  peak_target=target)
            if sc_ext < global_best:
                global_best = sc_ext
                global_a = a_ext.copy()
            print("  [ext %s n=%d] score=%.6f (t=%.0fs)" % (name, n, sc_ext, time.monotonic() - t_start))

    # Phase 3: Final polish with spectral methods
    if global_a is not None:
        rem = budget_s - (time.monotonic() - t_start) - 3.0
        if rem > 3:
            a = global_a.copy()
            # Gentle spectral flattening
            for _ in range(min(50, int(rem / 0.1))):
                if time.monotonic() - t_start > budget_s - 3:
                    break
                c = fft_convolve_self(a)
                peak_lag = int(np.argmax(c))
                alpha = 0.5
                phase_str = 0.1
                a = spectral_joint_optimize(a, N, peak_lag, alpha, phase_str, rng)
                sc = compute_score(a)
                if sc < global_best:
                    global_best = sc
                    global_a = a.copy()

            # Final subgradient polish
            rem2 = budget_s - (time.monotonic() - t_start) - 2.0
            if rem2 > 2:
                a_final, sc_final = subgradient_optimize(global_a, n, S, rem2, rng,
                                                         top_k_init=min(300, n // 30),
                                                         peak_target=target)
                if sc_final < global_best:
                    global_best = sc_final
                    global_a = a_final.copy()

    return global_a, global_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive subgradient + spectral phase/magnitude optimization.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    deadline = t_global_start + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Determine lengths to try
    lengths_to_try = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 2048:
            lengths_to_try.append(len(given))
    except Exception:
        pass

    for L in [4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Sort: provided length first, then by size
    provided_len = lengths_to_try[0] if lengths_to_try else 8192
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: (x != provided_len, abs(x - 8192)))

    print("Lengths to try: %s" % lengths_to_try)

    global_best = np.inf
    global_a = None

    # Allocate budget
    total_budget = budget_s - 40  # reserve 40s for verification and fallback
    # Weight allocation: favor the provided length and 8192
    weights = []
    for L in lengths_to_try:
        if L == provided_len:
            w = 1.5
        elif L == 8192:
            w = 1.2
        elif L <= 4096:
            w = 0.6
        else:
            w = 0.8
        weights.append(w)
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_global_start
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(budgets[i], remaining - 20)
        if alloc < 15:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        a_result, score = optimize_single_length(n, alloc, rng)

        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  *** New global best: %.6f at n=%d ***" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

    # If we have time, do a final focused optimization on the global best
    if global_a is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            n_best = len(global_a)
            S = global_a.sum()
            print("=== Final focused optimization, n=%d, budget=%.0fs ===" % (n_best, rem))
            a_final, sc_final = subgradient_optimize(global_a, n_best, S, rem, rng,
                                                     top_k_init=min(500, n_best // 20),
                                                     peak_target=1.49)
            if sc_final < global_best:
                global_best = sc_final
                global_a = a_final.copy()
                print("  *** Final best: %.6f ***" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence")

    # Verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: evaluate_sequence = %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluation returned inf, using fallback")
            global_a = np.ones(8192, dtype=np.float64) * 1.0
            ver = evaluate_sequence([float(x) for x in global_a])
            global_best = ver
    except Exception as e:
        print("Verification error: %s" % str(e))
        global_a = np.ones(8192, dtype=np.float64)

    print("Final: length=%d, reported_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
