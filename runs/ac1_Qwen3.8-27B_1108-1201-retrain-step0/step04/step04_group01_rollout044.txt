"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Adaptive multi-phase optimization with:
1. Softmax parameterization for constraint-free optimization
2. Top-lag targeted gradient with FISTA acceleration

4. Structured perturbation for local minima escape
5. Coordinate-wise refinement for final polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def top_lag_gradient(a, c, k, weights=None):
    """Gradient focusing on top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    if weights is None:
        weights = np.ones(k)
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def softmax_parameterization_gradient(g, a, c):
    """Convert gradient from a-space to x-space for softmax parameterization."""
    # dx_i = a_i * (g_i - (1/c) * sum_j a_j * g_j)
    cross = np.dot(a, g) / c
    return a * (g - cross)


def run_adaptive_fista(a_init, n, S, max_time, rng, mode="adaptive",
                       beta_init=50.0, beta_rate=0.003, beta_max=15000.0):
    """FISTA with adaptive mode switching."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_acc = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    patience = 500

    while time.monotonic() - start < max_time:
        # Check timeout frequently
        if time.monotonic() - start > max_time:
            break

        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive beta
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Choose mode based on progress
        if progress < 0.3:
            # Early: use moderate smoothing for fast convergence
            effective_mode = "lse"
            effective_beta = beta
        elif progress < 0.7:
            # Middle: use top-lag targeting
            effective_mode = "top_lag"
            effective_beta = beta
        else:
            # Late: very sharp smoothing
            effective_mode = "lse"
            effective_beta = min(beta_max, beta * 2)

        if effective_mode == "lse":
            grad = lse_gradient(y, c, effective_beta)
            step = 1.0 / (2.0 * effective_beta * 0.35)
        else:
            k = min(32, n // 16)
            # Weight by excess above mean
            mean_b = c.mean()
            top_c = np.sort(c)[::-1][:k]
            excess = np.maximum(top_c - mean_b, 0)
            wsum = excess.sum()
            if wsum < 1e-30:
                excess = np.ones(k)
                wsum = k
            weights = excess / wsum
            grad = top_lag_gradient(y, c, k, weights=weights)
            step = 1.0 / (0.5 * effective_beta * 0.35)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        if it - last_improve > patience:
            # Escape: structured perturbation
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                # Structured: perturb in pairs to preserve sum
                n_perturb = max(1, n // 100)
                idx1 = rng.choice(n, n_perturb, replace=False)
                idx2 = rng.choice(n, n_perturb, replace=False)
                delta = rng.normal(0, noise_scale, n_perturb)
                y[idx1] += delta
                y[idx2] -= delta
                y = project_simplex(y, S)
                t_acc = 1.0
                last_improve = it
            else:
                break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
        y = x_new + ((t_acc - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_acc = t_new

    return best_a, best


def run_softmax_opt(a_init, n, S, max_time, rng):
    """Optimize in softmax parameterization space."""
    a = normalize_to_sum(a_init, S)
    # Initialize x from a
    a_pos = np.maximum(a, 1e-10)
    x = np.log(a_pos / a_pos.sum() * n)  # log of normalized values * n
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    lr = 0.01
    it = 0
    last_improve = 0
    patience = 200
    momentum = 0.9
    vx = np.zeros(n)

    while time.monotonic() - start < max_time:
        if time.monotonic() - start > max_time:
            break
        it += 1

        # Convert x to a
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = S * exp_x / exp_x.sum()

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it

        if it - last_improve > patience:
            lr *= 0.8
            if lr < 1e-8:
                break
            # Perturb x
            x += rng.normal(0, 0.01, n)
            vx = np.zeros(n)
            last_improve = it

        # Compute gradient in x-space
        beta = 50.0 + 10.0 * it
        g_a = lse_gradient(a, c, beta)
        g_x = softmax_parameterization_gradient(g_a, a, S)
        gnorm = np.linalg.norm(g_x)
        if gnorm < 1e-14:
            break

        # Momentum step
        vx = momentum * vx - lr * g_x
        x = x + vx
        # Keep x bounded
        x = np.clip(x, x.max() - 50, x.max())

    return best_a, best


def run_coordinate_refinement(a_init, n, S, max_time, rng):
    """Coordinate-wise refinement for final polishing."""
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    
    sweep_count = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        sweep_count += 1
        improved = False
        
        # Randomized coordinate sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            # Try perturbing a[i]
            # Compute how much reducing a[i] helps the peak
            # At lag i: conv(a,a)_i includes a[0]*a[i] and a[i]*a[0] = 2*a[0]*a[i] (for i < n)
            # More generally, the contribution of a[i] to conv(a,a)_k is a[i]*a[i-k] + a[i-k]*a[i] = 2*a[i]*a[i-k]
            # (with boundary effects)
            
            # Simple: try decreasing a[i] and increasing a[j] for some j
            # Find the j that benefits most
            if a[i] < 1e-10:
                continue
            
            # Compute current peak
            c = fft_convolve_self(a)
            peak = c.max()
            
            # Try reducing a[i] slightly
            delta = a[i] * 0.01
            a[i] -= delta
            # Redistribute to the element with smallest contribution to peak
            # Simple heuristic: add to random element
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            a[j] += delta
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            
            c_new = fft_convolve_self(a)
            new_peak = c_new.max()
            
            if new_peak < peak - 1e-15:
                improved = True
                if new_peak < best:
                    best = new_peak
                    best_a = a.copy()
                no_improve = 0
            else:
                # Revert
                a[j] -= delta
                a[i] += delta
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        
        if sweep_count % 5 == 0:
            print("  Coord refinement sweep %d: peak=%.6f" % (sweep_count, best))

    return best_a, best

Strategy: Since the problem is convex (max of convex quadratic forms over a simplex),
we focus on fast convergence using:
1. FISTA with adaptive LSE smoothing (increasingly sharp approximation of max)

3. Adaptive step size with progress-based adjustment

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def top_k_lag_gradient(a, c, k, use_excess_weighting=True):
    """Gradient focusing on top-k lags with optional excess weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    if use_excess_weighting:
        mean_b = c[k:].mean()
        weights = np.maximum(c[idx] - mean_b, 1e-10)
    else:
        weights = np.ones(k)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k)
        wsum = k
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_convex(a_init, n, S, max_time, rng,
                     beta_init=20.0, beta_rate=0.005, beta_max=50000.0,
                     step_factor=0.25, restart_every=5000):
    """
    FISTA for the convex problem min max_k conv(a,a)_k s.t. a in simplex.
    Uses adaptive beta and progress-aware step sizing.
    """
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_acc = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    best_norm = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        if elapsed > max_time:
            break

        progress = min(1.0, elapsed / max_time)
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Compute gradient
        if beta > 500:
            # Very high beta: use top-1 or top-2 lag directly
            k_top = 1 if beta > 5000 else 2
            grad = top_k_lag_gradient(y, c, k_top, use_excess_weighting=True)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Track gradient norm for convergence detection
        if gnorm < best_norm * 0.1:
            best_norm = gnorm

        # Adaptive step size
        # Lip estimate: for LSE, L ~ 2*beta*2*S (rough)
        # Use step_factor to be conservative
        step = step_factor / (2.0 * beta * max(1.0, S * 0.01))
        # Cap step to avoid overshooting
        max_step = S / (2.0 * np.sqrt(n))
        step = min(step, max_step)

        # Reduce step if stuck
        stalled = it - last_improve
        if stalled > 3000:
            step *= 0.5
        elif stalled > 1000:
            step *= 0.75

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
        y = x_new + ((t_acc - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_acc = t_new

        # Periodic restart to avoid momentum issues
        if it % restart_every == 0:
            t_acc = 1.0
            y = a.copy()

    return best_a, best


def run_top_lag_polish(a_init, n, S, max_time, rng):
    """
    Final polishing: focus exclusively on the single top lag.
    Uses exact subgradient of the max function (top-1 lag).
    """
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_acc = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        if elapsed > max_time:
            break

        progress = min(1.0, elapsed / max_time)
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Top-1 lag gradient (exact subgradient of max)
        grad = top_k_lag_gradient(y, c, 1, use_excess_weighting=True)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Decreasing step for subgradient method
        base_step = S / (n ** 0.5 * (1.0 + 0.01 * it) ** 0.5)
        step = base_step * (1.0 - 0.5 * progress)

        if it - last_improve > 2000:
            step *= 0.5
            if step < 1e-10:
                break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
        y = x_new + ((t_acc - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_acc = t_new

    return best_a, best


def run_multi_beta_fista(a_init, n, S, max_time, rng):
    """
    FISTA with multiple beta values tried in sequence.
    Each phase uses a different beta, allowing the algorithm to
    transition from fast convergence (low beta) to precise peak targeting (high beta).
    """
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    # Phases with increasing beta
    phases = [
        (30.0, 0.25),   # beta=30, 25% of time
        (100.0, 0.20),  # beta=100, 20%
        (300.0, 0.20),  # beta=300, 20%
        (1000.0, 0.15), # beta=1000, 15%
        (5000.0, 0.10), # beta=5000, 10%
        (20000.0, 0.10) # beta=20000, 10%
    ]

    for beta_val, time_frac in phases:
        t_phase = max_time * time_frac
        if time.monotonic() - start > max_time - t_phase * 0.5:
            break

        a, val = run_fista_convex(a, n, S, t_phase, rng,
                                   beta_init=beta_val, beta_rate=0.0,
                                   beta_max=beta_val, step_factor=0.3)
        if val < best:
            best = val
            best_a = a.copy()
            obj = 2.0 * n * val / (a.sum() ** 2)
            print("    Phase beta=%.0f: obj=%.6f" % (beta_val, obj))

    return best_a, best


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Exploits convexity with FISTA + adaptive LSE + top-lag polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given_arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given_arr.size >= 64 and np.sum(np.clip(given_arr, 0, 1000)) > 0:
            given = np.clip(given_arr, 0, 1000)
            print("Using given sequence of length %d" % len(given))
            obj_given = objective_value(given)
            print("Given sequence objective: %.6f" % obj_given)
    except Exception as e:
        print("No given sequence available: %s" % str(e))

    global_best = np.inf
    global_a = None

    n = 8192
    if given is not None:
        n = len(given)

    S = np.sqrt(2.0 * n)
    print("Optimizing with n=%d, S=%.4f" % (n, S))

    # Prepare main initialization
    if given is not None and len(given) == n:
        a_main = normalize_to_sum(given, S)
    else:
        # Fallback: use a reasonable initialization
        tt = np.linspace(0, 1, n, endpoint=False)
        a_main = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.cos(2 * np.pi * 7 * tt)
        a_main = np.maximum(a_main, 0.01)
        a_main = normalize_to_sum(a_main, S)

    c_init = fft_convolve_self(a_main)
    print("Initial max_b=%.6f, mean_b=%.6f, ratio=%.4f" %
          (c_init.max(), c_init.mean(), c_init.max() / c_init.mean()))
    obj_init = 2.0 * n * c_init.max() / (a_main.sum() ** 2)
    print("Initial objective: %.6f" % obj_init)

    # === Phase 1: Multi-beta FISTA (60% of time) ===
    t1 = (deadline - time.monotonic()) * 0.50
    print("\n=== Phase 1: Multi-beta FISTA (%.0fs) ===" % t1)
    a1, val1 = run_multi_beta_fista(a_main, n, S, t1, rng)
    global_a = a1.copy()
    global_best = 2.0 * n * val1 / (a1.sum() ** 2)
    print("After Phase 1: obj=%.6f" % global_best)

    # === Phase 2: Continue FISTA with adaptive beta (25% of time) ===
    t2 = (deadline - time.monotonic()) * 0.50
    if t2 > 20:
        print("\n=== Phase 2: Adaptive FISTA (%.0fs) ===" % t2)
        a2, val2 = run_fista_convex(global_a, n, S, t2, rng,
                                     beta_init=50.0, beta_rate=0.003,
                                     beta_max=30000.0, step_factor=0.25)
        if val2 < fft_convolve_self(global_a).max():
            global_a = a2.copy()
            global_best = 2.0 * n * val2 / (a2.sum() ** 2)
            print("After Phase 2: obj=%.6f" % global_best)

    # === Phase 3: Top-lag polishing (10% of time) ===
    t3 = max(10, (deadline - time.monotonic()) * 0.6)
    if t3 > 5:
        print("\n=== Phase 3: Top-lag polish (%.0fs) ===" % t3)
        a3, val3 = run_top_lag_polish(global_a, n, S, t3, rng)
        if val3 < fft_convolve_self(global_a).max():
            global_a = a3.copy()
            global_best = 2.0 * n * val3 / (a3.sum() ** 2)
            print("After Phase 3: obj=%.6f" % global_best)

    # === Phase 4: Final high-beta FISTA (remaining time) ===
    t4 = deadline - time.monotonic()
    if t4 > 10:
        print("\n=== Phase 4: Final polish (%.0fs) ===" % t4)
        a4, val4 = run_fista_convex(global_a, n, S, t4, rng,
                                     beta_init=10000.0, beta_rate=0.0,
                                     beta_max=10000.0, step_factor=0.5,
                                     restart_every=3000)
        if val4 < fft_convolve_self(global_a).max():
            global_a = a4.copy()
            global_best = 2.0 * n * val4 / (a4.sum() ** 2)
            print("After Phase 4: obj=%.6f" % global_best)

    # Verify with evaluate_sequence
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification ===")
        print("Internal best: %.6f" % global_best)
        print("evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            global_best = evaluate_sequence([float(x) for x in global_a])
        else:
            global_best = ver
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        global_best = evaluate_sequence([float(x) for x in global_a])

    print("\nFinal: length=%d, obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
