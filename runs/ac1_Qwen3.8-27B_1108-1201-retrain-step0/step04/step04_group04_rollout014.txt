def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):

    else:

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lp_iteration(a, M, n, solver=cp.ECOS):
    """One LP iteration: find best b direction. Returns b or None."""
    if not HAS_CVXPY:
        return None
    
    b = cp.Variable(n, nonneg=True)
    
    # Build sparse constraint matrix A where A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max >= j_min:
            js = list(range(j_min, j_max + 1))
            ks = [k] * len(js)
            avals = [a[k - j] for j in js]
            rows.extend(ks)
            cols.extend(js)
            vals.extend(avals)
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve()
        except:
            return None
    
    if b.value is None:
        return None
    return np.asarray(b.value).flatten()

                if s < 1e-10:
                    continue
                a_trial = a_trial * (S / s)
                c_trial = fft_convolve_self(a_trial)
                val = c_trial.max()
                if val < best_val:
                    best_val = val
                    best_t = t
        
        if best_t < 0.001:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a * (S / s)
        
        c = fft_convolve_self(a)
        M = c.max()
        it += 1
        
        if M < best:
            best = M
            best_a = a.copy()
        
        if it % 10 == 0:
            elapsed = time.monotonic() - start
            print("    LP iter %d: M=%.6f, t=%s, elapsed=%.0fs" % (it, M, best_t, elapsed))
    
    return best_a, best


def coordinate_descent_fast(a_init, n, S, max_time, rng):
    """Efficient coordinate descent using precomputed contributions."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    
    # Precompute: for each position i, the "partner" indices for each lag
    # d_k for transferring from m to j:
    # dc_k = a[j+k] + a[j-k] - a[m+k] - a[m-k] (with bounds, 0 outside [0,n-1])
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        n_pairs = min(30 + it, 150)
        
        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 2:
                break
            
            # Choose pair
            if rng.random() < 0.4:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, n - 1)
                m = (j + rng.integers(1, 3)) % n
            
            if j == m:
                continue
            if a[j] < 1e-12 and a[m] < 1e-12:
                continue
            
            # Compute d vector
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                v = 0.0
                if 0 <= j + k < n:
                    v += a[j + k]
                if 0 <= j - k < n:
                    v += a[j - k]
                if 0 <= m + k < n:
                    v -= a[m + k]
                if 0 <= m - k < n:
                    v -= a[m - k]
                d[k] = v
            
            # Quadratic correction for k=0
            # c[0] changes by 2*(a[j]-a[m])*delta + 2*delta^2
            # (because c[0] = a[0]^2, and only a[j] and a[m] change)
            # Wait, c[0] = a[0]^2. So if j=0 or m=0, c[0] changes quadratically.
            # More generally, c[k] = sum_i a[i]*a[k-i].
            # If we change a[j] by +delta and a[m] by -delta:
            # dc[k] = delta * (a[k-j] + a[j-k] - a[k-m] - a[m-k]) + delta^2 * (delta_{k,j+m} ...)
            # The quadratic term appears when j+k = m or m+k = j etc.
            # For simplicity, ignore the quadratic term (it's usually small).
            
            # Find optimal delta by golden section search
            lo = -a[j] if d @ c > 0 else min(-a[j], 0.1 * S)
            hi = a[m] if d @ c > 0 else max(a[m], -0.1 * S)
            # Actually, delta must be in [-a[j], a[m]]
            lo = -a[j]
            hi = a[m]
            
            if hi - lo < 1e-15:
                continue
            
            # Evaluate f(delta) = max(c + delta * d)
            # Use a few sample points to find the optimum
            best_d = 0.0
            best_f = (c).max()
            
            # Sample-based search
            n_samples = 20
            deltas = np.linspace(lo, hi, n_samples)
            best_f = np.inf
            for dlt in deltas:
                val = (c + dlt * d).max()
                if val < best_f:
                    best_f = val
                    best_d = dlt
            
            # Refine with golden section
            if best_f < c.max() - 1e-10:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_g = max(lo, best_d - (hi - lo) / n_samples)
                b_g = min(hi, best_d + (hi - lo) / n_samples)
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                
                def f_eval(dlt):
                    return (c + dlt * d).max()
                
                fc = f_eval(c_g)
                fd = f_eval(d_g)
                for _ in range(30):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = f_eval(c_g)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = f_eval(d_g)
                
                best_d = (a_g + b_g) / 2.0
                best_f = f_eval(best_d)
            
            if best_f < best - 1e-12:
                a[j] += best_d
                a[m] -= best_d
                c += best_d * d
                # Recompute c[0] exactly if j==0 or m==0
                if j == 0 or m == 0:
                    c[0] = a[0] ** 2
                best = c.max()
                best_a = a.copy()
                improved = True
        
        if it % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        if not improved and it > 10:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best


def multiplicative_weight_update(a_init, n, S, max_time, rng, eta_init=0.1):
    """Entropic mirror descent / multiplicative weight update."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Compute gradient (approximate using LSE with moderate beta)
        beta = 50.0 + 10.0 * it
        cm = c.max()
        e = np.exp(beta * (c - cm))
        w = e / e.sum()
        
        # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # Use FFT
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Multiplicative update: a[i] *= exp(-eta * grad[i])
        # Then normalize
        log_factor = -eta * grad / (np.max(np.abs(grad)) + 1e-15)
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-15)  # keep positive
        a_new = a_new / a_new.sum() * S
        
        # Anneal eta
        eta = max(0.01, eta_init * np.exp(-it / 1000.0))
        
        a = a_new
        
        # Periodic noise
        if it % 500 == 0:
            noise = 0.01 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return best_a, best


def resample_sequence(a, n_from, n_to):
    """Resample a sequence from one length to another."""
    if n_from == n_to:
        return a.copy()
    t_old = np.linspace(0, 1, n_from, endpoint=False)
    t_new = np.linspace(0, 1, n_to, endpoint=False)
    return np.interp(t_new, t_old, a)
