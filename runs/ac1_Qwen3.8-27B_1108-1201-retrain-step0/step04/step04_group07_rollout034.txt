def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))

    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def compute_direct_gradient(a, peak_indices, weights):
    """
    Compute subgradient of max(c) using direct formula.
    grad[i] = 2 * sum_j w_j * a[kj - i] for valid indices.
    O(n * len(peak_indices)) time.
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(len(peak_indices)):
        k = int(peak_indices[i])
        w = weights[i]
        # grad[i] += 2 * w * a[k-i] for 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # For i in [i_start, i_end]: a[k-i] where k-i ranges from k-i_end to k-i_start
            idx = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w * a[idx]
    return grad


def fista_optimize(a_init, n, S, max_time, top_k=10, step_scale=0.5, rng=None,
                   target_obj=1.49, spectral_every=5000, print_interval=5.0):
    """
    Accelerated subgradient (Fista) for minimizing max(c) with fixed sum.
    Uses direct gradient computation for speed.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_objective(a)
    best_local = a.copy()
    last_print = start

    c = fft_convolve_self(y)
    peak_val = c.max()

    # Estimate Lipschitz constant
    L = 2.0 * S * step_scale
    step = S / (L + 1e-10)

    it = 0
    while time.monotonic() - start < max_time:
        it += 1

        # Compute convolution and find peaks
        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)

        if sc < best_score:
            best_score = sc
            best_local = y.copy()

        # Periodic print
        now = time.monotonic()
        if now - last_print > print_interval:
            print("    Fista it=%d score=%.6f peak=%.4f/%.4f time=%.0fs" %
                  (it, best_score, cur_max, S * S / n, now - start))
            last_print = now

        if best_score <= target_obj:
            break

        # Find top-k peaks with soft weighting
        peak_idx_arr = np.argsort(c)[::-1][:top_k]
        peak_vals = c[peak_idx_arr]
        # Exponential soft weighting
        val_range = peak_vals.max() - peak_vals.min()
        if val_range > 1e-30:
            norm_vals = (peak_vals - peak_vals.min()) / val_range
            weights = np.exp(5.0 * norm_vals)
        else:
            weights = np.ones(top_k) / top_k
        weights /= weights.sum()

        # Compute gradient directly
        grad = compute_direct_gradient(y, peak_idx_arr, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break

        # Adaptive step size (decrease slowly)
        step = S / (L * (1.0 + 0.001 * it))

        # Proximal step
        x_new = project_simplex(y - step * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * (S / s_y)
        a = x_new
        t = t_new

        # Periodic spectral shaping to escape local minima
        if it % spectral_every == 0 and rng is not None:
            y = spectral_shape_step(y, n, S, alpha=0.3, rng=rng)

        # Occasional random perturbation
        if it % 10000 == 5000 and rng is not None:
            noise_level = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_level, n), S)
            t = 1.0

    return best_local, best_score


def spectral_shape_step(a, n, S, alpha=0.3, rng=None):
    """
    One step of targeted spectral shaping.
    Reduces frequency components that contribute to the convolution peak.
    """
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    A = fft(a, fsize)
    P = np.abs(A) ** 2
    c = np.real(ifft(A * A))[:size]
    c_max = c.max()

    # Identify which frequency pairs contribute to the peak
    peak_lag = int(np.argmax(c))

    # The contribution of frequency m to c[peak_lag] (circular approximation)
    # is |A[m]| * |A[peak_lag - m]| * cos(phase_diff)
    # Simplify: reduce |A[m]| for m where |A[m]| * |A[peak_lag mod fsize - m]| is large

    # Compute "contribution spectrum"
    contrib = np.abs(A) * np.roll(np.abs(A), peak_lag)
    # Normalize
    contrib_max = contrib.max()
    if contrib_max < 1e-30:
        return a

    # Reduce high-contribution frequencies
    reduction = 1.0 - alpha * (contrib / contrib_max)
    reduction = np.maximum(reduction, 0.3)  # Don't reduce by more than 70%

    # Apply to magnitude, preserve phase
    new_mag = np.abs(A) * reduction
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)

    # Preserve DC component (sum constraint)
    A_new[0] = A[0]

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_flatten_aggressive(a, n, S, n_iter=200, alpha_init=0.5, rng=None):
    """
    Aggressive spectral flattening: push |A|^2 toward uniform.
    Good for initial diversification.
    """
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_objective(a)

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        alpha = alpha_init * (1.0 - 0.7 * progress)

        A = fft(a, fsize)
        P = np.abs(A) ** 2
        P[0] = 0  # Don't touch DC

        p_median = np.median(P[P > 1e-30]) if np.any(P > 1e-30) else 1.0
        ratio = P / p_median
        ratio = np.where(ratio > 0, ratio, 1.0)

        # Compression above, expansion below
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * np.power(ratio[above], -alpha)
        P_new[~above] = P[~above] * np.power(ratio[~above], 1.0 - alpha * 0.5)
        P_new = np.maximum(P_new, 0.0)
        P_new[0] = P[0]

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        sc = compute_objective(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc

Minimize 2*n*max(convolve(a,a))/sum(a)^2 using accelerated subgradient methods
with direct gradient computation, spectral shaping, and multi-length search.

Key ideas:
- The full self-convolution peak can be below S^2/n, enabling objectives below 2.

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
