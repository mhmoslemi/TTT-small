```python
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. Log-softmax parameterized Adam optimizer with combined LSE + L2-flatness objective
2. FFT-based gradient computation (O(n log n) per step)
3. Multiple diverse initializations from parent and structured starts
4. Periodic spectral shaping kicks to escape local minima
5. Final subgradient + coordinate descent polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_gradient(a, c, beta, fsize):
    """Compute gradient of LSE(c; beta) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    # g_j = 2 * sum_k w_k * a_{k-j} (cross-correlation of w with a)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def compute_flatness_gradient(a, c, C, fsize):
    """Compute gradient of mean((c - C)^2) w.r.t. a."""
    n = len(a)
    d = c - C
    D = fft(d, fsize)
    A = fft(a, fsize)
    grad = 4.0 * np.real(ifft(D * np.conj(A)))[:n]
    return grad


def softmax_from_logspace(x):
    """Convert log-space parameters to probability vector."""
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def adam_step(x, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam update step. Returns updated x, m, v."""
    m = beta1 * m + (1 - beta1) * x
    v = beta2 * v + (1 - beta2) * (x ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v
```

`b[n-1] = sum_i a[i] * a[n-1-i]`. By Cauchy-Schwarz: `b[n-1] <= sqrt(sum a[i]^2 * sum a[n-1-i]^2) = sum a[i]^2`. And `sum a[i]^2 >= S^2/n`.

So `b[n-1] <= sum a[i]^2`, and `sum a[i]^2 >= S^2/n`. But `b[n-1]` could be much less than `sum a[i]^2`!

For example, if `a = [S/n, S/n, ..., S/n]` (uniform): `b[n-1] = n * (S/n)^2 = S^2/n = 2`. And `sum a[i]^2 = 2`. So they're equal for the uniform case.

But if `a` is concentrated at the ends: `a = [S/2, 0, ..., 0, S/2]`: `b[n-1] = (S/2)^2 + (S/2)^2 = S^2/2 = n`. And `sum a[i]^2 = S^2/2 = n`. Still equal.

Hmm, what about `a = [S/2, S/2, 0, ..., 0]`: `b[0] = S^2/4`, `b[1] = S^2/2`, `b[n-1] = 0` (since `a[0]*a[n-1] + a[1]*a[n-2] + ... = 0` for n > 2). So `max_b = S^2/2 = n` (at lag 1).

OK so the point is: `b[n-1]` is NOT bounded below by `S^2/n`. The lower bound is on `sum a[i]^2`, which is distributed across various lags.

So what IS the lower bound on `max b[k]`?

`sum_k b[k] = S^2 = 2n`. There are `2n-1` terms. So `max b[k] >= S^2/(2n-1) = 2n/(2n-1) ≈ 1`.

This is the simple averaging argument. And it's achievable in principle (if all `b[k]` are equal to `2n/(2n-1)`), but in practice with non-negativity constraints, we can't achieve it.

So the target of 1.5027 means max_b ≈ 1.5027 when S = sqrt(2n). That's about 50% above the theoretical minimum of 1.0.

Now I understand the problem better. Let me proceed with the implementation.

Key insight for the algorithm: We want to make `b[k]` as flat as possible (close to `2n/(2n-1) ≈ 1` for all k). The main challenge is the non-negativity constraint on `a`.

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
1. Log-softmax parameterized Adam with FFT-based gradients
2. Combined LSE + L2-flatness objective for smooth optimization
3. Spectral domain shaping for diversity
4. Subgradient + coordinate descent for final polish
5. Multiple diverse initializations and adaptive length selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def convolve_grad_fft(c_or_w, a, fsize):
    """Compute 2 * cross-correlation of c_or_w (len 2n-1) with a (len n).
    Result: g[j] = 2 * sum_k c_or_w[k] * a[k-j] for valid indices."""
    W = fft(c_or_w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def logspace_to_a(x, S):
    """Convert log-space x to a = S * softmax(x)."""
    x_max = x.max()
    e = np.exp(x - x_max)
    p = e / e.sum()
    return S * p


def a_to_logspace(a, S):
    """Convert a to log-space: x = log(a/S + eps)."""
    p = np.maximum(a / S, 1e-30)
    return np.log(p)


def run_adam_opt(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=500.0,
                 alpha_l2=0.3, lr_start=0.05, lr_end=0.005,
                 noise_period=5000, noise_scale=0.1):
    """Adam optimizer in log-space with combined LSE + L2 objective."""
    x = a_to_logspace(a_init, S).copy()
    x -= x.mean()  # center
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    fsize = 1 << (3 * n - 2).bit_length()
    C = S * S / (2 * n - 1)
    
    start = time.monotonic()
    best = np.inf
    best_a = a_init.copy()
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        a = logspace_to_a(x, S)
        c = fft_convolve_self(a)
        cur_max = c.max()
        score = 2.0 * n * cur_max / (S * S)
        
        if score < best:
            best = score
            best_a = a.copy()
            last_improve = t
        
        # Adaptive beta
        frac = min(1.0, t / (max_time * 100))  # ramp over first ~10% of time
        # Actually let's base it on iterations
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 20000.0)
        
        # Compute LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            w = np.full(len(c), 1.0 / len(c))
        else:
            w = e / tot
        
        g_lse = convolve_grad_fft(w, a, fsize)
        
        # Compute L2 flatness gradient
        g_flat = convolve_grad_fft(c - C, a, fsize) * 2.0  # factor of 2 for (1/2)*sum
        
        # Combined gradient w.r.t. a
        alpha_lse = 1.0 - alpha_l2
        g = alpha_lse * g_lse + alpha_l2 * g_flat
        
        # Convert to gradient w.r.t. x (log-space)
        # df/dx_j = (a_j/S) * (g_j - dot(g,a)/S)
        dot_ga = np.dot(g, a)
        grad_x = (a / S) * (g - dot_ga / S)
        
        # Adam update
        lr = lr_start + (lr_end - lr_start) * min(1.0, t / 50000.0)
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Center x
        x -= x.mean()
        
        # Periodic noise injection
        if t % noise_period == 0:
            decay = np.exp(-t / (2 * noise_period))
            x += rng.normal(0, noise_scale * decay, n)
            x -= x.mean()
            m = np.zeros(n)
            v = np.zeros(n)
        
        # Check for divergence
        a_check = logspace_to_a(x, S)
        if not np.all(np.isfinite(a_check)):
            x = a_to_logspace(best_a, S)
            x -= x.mean()
            m = np.zeros(n)
            v = np.zeros(n)
    
    return best_a, best


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Subgradient method with Polyak step size for final polish."""
    a = a_init.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    C = S * S / (2 * n - 1)  # lower bound for max_b
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        score = 2.0 * n * cur / (S * S)
        
        if score < best:
            best = score
            best_a = a.copy()
        
        # Subgradient at argmax
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        grad = np.zeros(n)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_star - idx]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak step: (f(x) - f*) / ||g||^2
        gap = cur - C  # f(x) - lower_bound
        step = gap / (gnorm * gnorm)
        step = min(step, 1.0 / gnorm)  # cap
        
        x_new = a - step * grad
        a = project_simplex(x_new, S)
        
        # Occasionally add noise
        if rng.random() < 0.01:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best


def run_coord_descent(a_init, n, S, max_time, rng):
    """Paired coordinate descent for final polish."""
    a = a_init.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        n_try = min(30 + it // 5, 100)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break
            
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue
            
            # Compute gradient of max(b) for transfer from m to j
            # b_k changes by: delta * (a_{k-j} + a_{j-k} - a_{k-m} - a_{m-k})
            # for the linear part (ignoring k=0 quadratic)
            
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val -= a[mk]
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val
            
            # Binary search for optimal delta in [-a[j], a[m]]
            lo_d = -a[j]
            hi_d = a[m]
            
            # Try a few candidate deltas
            for delta_frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                delta = lo_d + delta_frac * (hi_d - lo_d)
                c_trial = c + delta * d
                c_trial[0] = c[0] + 2 * (a[j] - a[m]) * delta + 2 * delta * delta
                cur_trial = c_trial.max()
                if cur_trial < best - 1e-13:
                    a[j] += delta
                    a[m] -= delta
                    c = c_trial
                    best = cur_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if it % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        if not improved and it > 10:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
    
    return best_a, best


def spectral_kick(a, n, S, rng, n_steps=500):
    """Spectral domain shaping to escape local minima."""
    fsize = 1 << (2 * n - 1).bit_length()
    a_work = a.copy()
    
    for i in range(n_steps):
        a_pad = np.zeros(fsize)
        a_pad[:n] = a_work
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Flatten: P_new = P^alpha / mean(P^alpha)
        alpha = 0.3 + 0.5 * (i / n_steps)
        P_flat = np.power(np.maximum(P, 1e-30), alpha)
        P_flat /= P_flat.mean()
        P_new = 0.5 * P / P_mean + 0.5 * P_flat
        
        mag_new = np.sqrt(P_new)
        mag_new *= np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]
        
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_work = np.ones(n) * S / n
        else:
            a_work = a_raw * (S / s)
    
    return a_work
```
