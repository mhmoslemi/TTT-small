with FFT-based gradient computation and spectral shaping moves.
"""
import numpy as np

from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Compute self-convolution via FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c

def compute_subgrad(a, A, peaks, weights, N, n):
    """Compute weighted subgradient of max(conv) at given peaks."""
    w = np.zeros(N)
    for i, p in enumerate(peaks):
        wp = int(p)
        if 0 <= wp < N:
            w[wp] = weights[i]
    W = fft(w)
    G = 2.0 * np.conj(A) * W
    grad = np.real(ifft(G))[:n]
    return grad

def spectral_shaping(a, A, N, n, S, top_k, alpha):
    """Spectral shaping: modify spectrum to flatten convolution."""
    c = np.real(ifft(A * A))[:2*n-1]
    # Compute target: reduce peaks
    c_mean = c.mean()
    c_std = c.std()
    threshold = c_mean + 0.5 * c_std
    c_target = np.minimum(c, threshold)
    # Pad to N
    c_target_full = np.zeros(N)
    c_target_full[:2*n-1] = c_target
    C_target = fft(c_target_full)
    # Update A: blend current and target
    # A_new^2 should be closer to C_target
    # A_new = A * (C_target / |A|^2)^alpha ... this is a magnitude update
    mag_A = np.abs(A)
    mag_A = np.maximum(mag_A, 1e-10)
    C_mag = np.abs(C_target)
    # Blend in log domain

    mag_new = np.exp(log_blend)
    phase_A = np.angle(A)
    A_new = mag_new * np.exp(1j * phase_A)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated projected subgradient
with FFT-based exact subgradient computation and target-tracking strategy.

Key ideas:
1. Exact subgradient via FFT (no LSE approximation bias) - gradient of c_p is 2*a shifted by p.
2. Target-tracking: maintain target T, focus subgradient on peaks above T, lower T when achieved.
3. Adaptive step size with progress monitoring and backtracking.
4. Periodic spectral shaping moves for global exploration.
5. Nesterov momentum for accelerated convergence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv FFT(a, N):
    """Compute self-convolution via FFT. Returns real convolution (length 2n-1)."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    n = len(a)
    return c[:2 * n - 1], A


def compute_subgrad_fft(A, peaks, weights, N, n):
    """
    Compute weighted sum of subgradients of c_p for given peaks.
    Result: sum_i weights[i] * grad_a(c_{peaks[i]})
    Formula: 2 * IFFT(conj(A) * FFT(w))[:n] where w[peaks[i]] = weights[i]
    """
    w = np.zeros(N)
    for i in range(len(peaks)):
        p = int(peaks[i])
        if 0 <= p < N:
            w[p] = weights[i]
    W = fft(w)
    G = 2.0 * np.conj(A) * W
    grad = np.real(ifft(G))[:n]
    return grad


def nesterov_subgrad(a_init, n, S, N, max_time, rng,
                     top_k_init=10, top_k_max=200,
                     step_init=None, use_target=True):
    """
    Nesterov-accelerated projected subgradient method for minimizing max(conv).
    Returns (best_a, best_cmax).
    """
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    A = fft(a, N)

    if step_init is None:
        step_init = 1.0 * S / (n * 0.5)

    step = float(step_init)
    best_a = a.copy()
    best_cmax = np.inf
    target = np.inf

    start_time = time.monotonic()
    it = 0
    no_improve = 0
    prev_best = np.inf

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        # Compute convolution
        A = fft(y, N)
        c = np.real(ifft(A * A))[:2 * n - 1]
        cur_max = c.max()

        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Perturb to escape local min
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            a = y.copy()
            t_val = 1.0
            no_improve = 0
            continue

        # Determine target
        c_mean = c.mean()
        c_std = c.std()
        if use_target:
            if target == np.inf:
                target = cur_max
            # Lower target if we've beaten it
            if cur_max < target * 0.98:
                target = cur_max
            # Target should be above mean
            target = max(target, c_mean + 0.05 * c_std)
        else:
            target = c_mean + 0.1 * c_std

        # Find peaks above target
        # Use a soft threshold: weight peaks by their excess
        excess = c - target
        n_above = np.sum(excess > 0)

        if n_above == 0:
            # All below target - lower target and continue
            target = c_mean + 0.01 * c_std
            continue

        # Get top-k peaks (those with highest excess)
        if n_above > top_k:
            # Get indices of top-k excess values
            top_k_minus_1 = top_k - 1
            part_idx = np.argpartition(excess, -top_k_minus_1)[-top_k:]
        else:
            part_idx = np.where(excess > 0)[0]

        if len(part_idx) == 0:
            target = c_mean
            continue

        # Weights proportional to excess
        wts = excess[part_idx]
        wts = np.maximum(wts, 1e-10)
        wts = wts / wts.sum()

        # Compute subgradient
        grad = compute_subgrad_fft(A, part_idx, wts, N, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        # Use 2/(L*t) style but with practical adaptation
        step_curr = step / (1.0 + 0.002 * it)

        # Nesterov update
        x_new = project_simplex(y - step_curr * grad / gnorm, S)

        # Ensure x_new is valid
        if not np.all(np.isfinite(x_new)):
            x_new = a.copy()
            break

        # Update Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        # Project y back (momentum can push off the simplex)
        y = project_simplex(y, S)

        a = x_new
        t_val = t_new

        # Adaptive step: if no improvement for a while, reduce step
        if no_improve > 200:
            step *= 0.9
        elif no_improve == 0:
            step = min(step * 1.005, step_init * 2.0)

        # Periodic spectral move
        if it % 2000 == 0 and rng.random() < 0.2:
            A_curr = fft(x_new, N)
            c_curr = np.real(ifft(A_curr * A_curr))[:2 * n - 1]
            # Spectral shaping: reduce peak magnitude in frequency domain
            C = np.abs(A_curr) ** 2
            C_mean = C.mean()
            C_max = C.max()
            if C_max > C_mean * 1.5:
                # Compress the spectrum
                alpha_sp = 0.1
                C_new = C_mean + (C - C_mean) * (1 - alpha_sp)
                C_new = np.maximum(C_new, 0)
                phase = np.angle(A_curr)
                A_new = np.sqrt(C_new) * np.exp(1j * phase)
                a_sp = np.real(ifft(A_new))[:n]
                a_sp = project_simplex(a_sp, S)
                # Check if spectral move helps
                A_check = fft(a_sp, N)
                c_check = np.real(ifft(A_check * A_check))[:2 * n - 1]
                if c_check.max() < cur_max:
                    a = a_sp
                    y = a_sp.copy()
                    t_val = 1.0
                    A = A_check

    return best_a, best_cmax


def spectral_block(a_init, n, S, N, max_time, rng, alpha=0.15):
    """
    Spectral shaping block: iteratively modify the spectrum to flatten the convolution.
    """
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()

    while time.monotonic() - start_time < max_time:
        A = fft(a, N)
        c = np.real(ifft(A * A))[:2 * n - 1]
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Spectral compression
        A_mag = np.abs(A)
        A_mag = np.maximum(A_mag, 1e-10)
        A_mag_sq = A_mag ** 2
        mean_sq = A_mag_sq.mean()
        if A_mag_sq.max() < mean_sq * 1.1:
            # Spectrum is already fairly flat
            break

        # Soft-threshold the spectrum magnitude squared
        threshold = mean_sq * 1.5
        A_mag_sq_new = np.where(A_mag_sq > threshold,
                                threshold + (A_mag_sq - threshold) * (1 - alpha),
                                A_mag_sq)
        A_mag_new = np.sqrt(np.maximum(A_mag_sq_new, 0))
        phase = np.angle(A)
        A_new = A_mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)

    return best_a, best_cmax


def lse_refine(a_init, n, S, N, max_time, rng, beta_start=20.0, beta_end=200.0):
    """
    LSE-smoothed gradient refinement (complementary to exact subgradient).
    Uses L-BFGS for local convergence.
    """
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    beta = beta_start
    s_hist = []
    y_hist = []
    m_lbfgs = 10

    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Ramp beta
        if it % 100 == 0:
            beta = min(beta_end, beta * 1.1)

        # Compute LSE gradient
        A = fft(a, N)
        c = np.real(ifft(A * A))[:2 * n - 1]
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        # Gradient via FFT
        W = fft(w, N)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # L-BFGS direction
        direction = _lbfgs_direction(s_hist, y_hist, grad)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Backtracking line search
        alpha_ls = 2.0 * S / (n * beta)
        a_new = a
        c_new = c
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction / dir_norm, S)
            A_c = fft(candidate, N)
            c_ls = np.real(ifft(A_c * A_c))[:2 * n - 1]
            if c_ls.max() <= c_max or ls == 9:
                a_new = candidate
                c_new = c_ls
                if c_ls.max() <= c_max:
                    break
            alpha_ls *= 0.5

        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best_a = a_new.copy()

        # Update L-BFGS history
        A_new = fft(a_new, N)
        c_nw = np.real(ifft(A_new * A_new))[:2 * n - 1]
        exp_bc_new = np.exp(beta * (c_nw - c_nw.max()))
        total_new = exp_bc_new.sum()
        w_new = exp_bc_new / total_new if total_new > 1e-300 else np.zeros(2*n-1)
        W_new = fft(w_new, N)
        Af_new = fft(a_new, N)
        grad_new = 2.0 * np.real(ifft(W_new * np.conj(Af_new)))[:n]

        s_k = a_new - a
        y_k = grad_new - grad
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new

    return best_a, best_cmax


def _lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g if gn > 1e-18 else np.zeros(n)

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

    print("n=%d, S=%.4f, N=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_cand, label=""):
        nonlocal global_best, global_a
        a_c = project_simplex(a_cand, S)
        A = fft(a_c, N)
        c = np.real(ifft(A * A))[:2 * n - 1]
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_c.copy()
            if label:
                print("  [%s] %.6f" % (label, sc))
        return sc

    # Initialize from previous best
    a_init = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            a_init = np.clip(given, 0, 1000).astype(np.float64)
            if a_init.sum() > 0.01:
                a_init = a_init * (S / a_init.sum())
            else:
                a_init = np.ones(n) * S / n
            print("Loaded previous best, initial score: ", end="")
            score = update_best(a_init, "prev_best")
            print("%.6f" % score)
    except Exception:
        pass

    if a_init is None:
        a_init = np.ones(n) * S / n
        update_best(a_init, "uniform")

    # === Phase 1: Nesterov subgradient (main) ===
    print("=== Phase 1: Nesterov subgradient ===")
    t1 = time.monotonic()
    phase1_budget = min(1400.0, (deadline - t1) * 0.55)

    a_curr = global_a.copy()
    stall_count = 0
    last_best = global_best
    block_id = 0

    while time.monotonic() < t1 + phase1_budget:
        block_id += 1
        block_time = min(60.0, t1 + phase1_budget - time.monotonic())
        if block_time < 5:
            break

        # Adaptive parameters based on stall
        top_k = 10 + min(50, stall_count * 3)
        step_size = 1.0 * S / (n * (0.5 + 0.2 * stall_count))

        a_new, sc_new = nesterov_subgrad(
            a_curr, n, S, N, block_time, rng,
            top_k_init=top_k, top_k_max=top_k + 100,
            step_init=step_size, use_target=True)

        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            print("  [P1-%d] %.6f" % (block_id, global_best))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count >= 3:
                # Perturb
                noise_scale = 0.005 + 0.005 * stall_count
                a_curr = project_simplex(
                    a_curr + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
                stall_count = 0
                print("  [P1-%d] Perturbed (scale=%.4f)" % (block_id, noise_scale))

    # === Phase 2: LSE refinement ===
    print("=== Phase 2: LSE refinement ===")
    t2 = time.monotonic()
    remaining = deadline - t2
    phase2_budget = min(remaining * 0.4, 600.0)

    if phase2_budget > 20:
        a_lse, sc_lse = lse_refine(
            global_a, n, S, N, phase2_budget, rng,
            beta_start=20.0, beta_end=300.0)
        update_best(a_lse, "LSE")

        # Second LSE pass with higher beta
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_lse2, sc_lse2 = lse_refine(
                global_a, n, S, N, min(remaining - 5, 300.0), rng,
                beta_start=50.0, beta_end=500.0)
            update_best(a_lse2, "LSE2")

    # === Phase 3: Spectral shaping ===
    print("=== Phase 3: Spectral shaping ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_sp, sc_sp = spectral_block(
            global_a, n, S, N, min(remaining * 0.5, 200.0), rng, alpha=0.1)
        update_best(a_sp, "spectral")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_sp2, sc_sp2 = spectral_block(
                global_a, n, S, N, min(remaining - 3, 100.0), rng, alpha=0.05)
            update_best(a_sp2, "spectral2")

    # === Phase 4: Final Nesterov polish with large top-k ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fin, sc_fin = nesterov_subgrad(
            global_a, n, S, N, min(remaining - 5, 400.0), rng,
            top_k_init=50, top_k_max=500,
            step_init=S / (n * 3.0), use_target=True)
        update_best(a_fin, "final_nest")

    # === Phase 5: Final LSE polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_pol, sc_pol = lse_refine(
            global_a, n, S, N, min(remaining - 3, 200.0), rng,
            beta_start=100.0, beta_end=1000.0)
        update_best(a_pol, "final_LSE")

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("FINAL: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
