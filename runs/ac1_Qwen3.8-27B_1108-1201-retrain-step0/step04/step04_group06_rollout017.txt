"""
Minimize 2*n*max_conv/sum^2 using split-step FFT (IRAM) optimization,
improved coordinate descent, and multi-resolution search.

Key innovations:
1. Split-step FFT (IRAM-like): Alternates between spectral flattening and
   simplex projection - fundamentally different from spectral water-filling.
2. Adaptive multi-temperature flattening with phase randomization.
3. Improved coordinate descent with importance sampling of peak lags.
4. Multi-resolution: try different sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def split_step_iterate(a, N, n, beta, rng, n_inner=1):
    """
    One round of split-step: flatten spectrum, project to simplex.
    beta in (0,1]: 1 = no change, 0 = fully flat.
    Returns new a.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Soft flattening: mag_new = mag^beta (which flattens for beta < 1)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    mag_normalized = mag / mag_mean
    mag_new = mag_mean * np.power(mag_normalized, beta)
    
    # Optional: random phase perturbation
    if rng is not None and rng.random() < 0.1:
        phase += rng.normal(0, 0.1 * (1 - beta), N)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def split_step_optimize(a_init, n, N, max_time, rng, target=1.0,
                        beta_start=0.7, beta_end=0.95):
    """
    Iterative split-step optimization.
    Returns (best_a, best_score).
    """
    a = a_init.copy()
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf
    it = 0
    stall = 0
    
    while time.monotonic() < max_time:
        it += 1
        # Adaptive beta: start aggressive, end gentle
        progress = min(1.0, it / 500.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        a_new = split_step_iterate(a, N, n, beta, rng)
        c = fft_convolve_self(a_new)
        score = c.max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if it % 200 == 0:
                print("    SS iter %d: %.6f" % (it, best_score))
        else:
            a = a_new.copy()
            stall += 1
            if stall > 200:
                # Phase randomization to escape
                a_pad = np.zeros(N)
                a_pad[:n] = a
                A = fft(a_pad)
                phase = rng.uniform(0, 2*np.pi, N)
                mag = np.abs(A)
                A_new = mag * np.exp(1j * phase)
                a_new = np.real(ifft(A_new))[:n]
                a = project_simplex(a_new, S)
                stall = 0
        
        if best_score <= target:
            break
    
    return best_a, best_score


def coordinate_descent_improved(a_init, n, S, max_time, rng, n_top=30):
    """
    Improved coordinate descent: for each coordinate, do a proper line search
    by checking the effect on the top peak lags.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Find top peak lags
        top_lags = np.argsort(c)[::-1][:20]
        
        improved = False
        for lag in top_lags[:10]:
            k = int(lag)
            # Find coordinates that contribute to this lag
            low = max(0, k - n + 1)
            high = min(k, n - 1)
            
            if high < low:
                continue
            
            # Compute contributions
            contribs = np.zeros(n)
            for i in range(low, high + 1):
                j = k - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            
            # Pick top contributing coordinates
            top_coords = np.argsort(contribs)[::-1][:n_top]
            
            for idx in top_coords[:15]:
                i = int(idx)
                if contribs[i] < 1e-15:
                    continue
                j = k - i
                if j < 0 or j >= n or i == j:
                    continue
                
                orig_val = a[i]
                best_frac = 1.0
                best_local = c_max
                
                # Try several multipliers
                for frac in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3]:
                    delta = orig_val * (frac - 1.0)
                    # Quick update of affected lags
                    # Lag k changes by: delta * a[j] (from the a[i]*a[j] term)
                    # Plus contributions to other lags
                    new_c_max = c_max
                    # Check if reducing a[i] helps lag k
                    new_k_val = c[k] + delta * a[j]
                    if new_k_val < 0:
                        new_k_val = 0
                    # The max might shift to another lag
                    # For speed, just check if this lag improves significantly
                    if new_k_val < best_local and new_k_val >= 0:
                        # Check a few other top lags
                        ok = True
                        for lag2 in top_lags[:5]:
                            lag2 = int(lag2)
                            if lag2 == k:
                                continue
                            # Check if a[i] contributes to lag2
                            low2 = max(0, lag2 - n + 1)
                            high2 = min(lag2, n - 1)
                            if low2 <= i <= high2:
                                j2 = lag2 - i
                                if 0 <= j2 < n:
                                    delta2 = delta * a[j2]
                                    if c[lag2] + delta2 > c_max * 1.01:
                                        ok = False
                                        break
                        if ok:
                            best_local = new_k_val
                            best_frac = frac
                
                a[i] = orig_val * best_frac
                if best_frac != 1.0:
                    improved = True
            
            # Re-clip and renormalize periodically
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a * (S / s)
        
        if not improved:
            # Perturb
            idx = rng.integers(n)
            a[idx] *= rng.uniform(0.5, 1.5)
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
    
    return best_a, best_score


def nesterov_smoothed(a_init, n, S, max_time, rng, tau=0.5):
    """
    Nesterov-accelerated gradient descent on smoothed log-sum-exp objective.
    f(a) = log(sum_k exp(c_k/tau)) where c = a*a
    """
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        s = a.sum()

    y = a.copy()
    momentum = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute gradient of log-sum-exp
        c_scaled = c / tau
        c_max_val = c_scaled.max()
        exp_c = np.exp(c_scaled - c_max_val)
        weights = exp_c / exp_c.sum()
        
        # Gradient: dw_k/d(a_i) for each k, then combine
        # c_k = sum_i a_i * a[k-i], so dc_k/d(a_i) = a[k-i] + a[i-k] (if valid)
        # grad_i = sum_k weights_k * (dc_k/d(a_i)) / tau
        # This is expensive to compute directly, so use FFT trick
        
        # Alternative: use the fact that the gradient of the weighted sum of
        # autocorrelations can be computed via cross-correlation
        w_full = np.zeros(2 * n - 1)
        w_full[:] = weights
        # Gradient of sum_k w_k * c_k w.r.t. a:
        # = 2 * (w_full * a) convolution (symmetrized)
        # Actually: d/d(a_i) [sum_k w_k * sum_j a_j * a[k-j]] 
        # = sum_k w_k * (a[k-i] + a[i-k] if valid)
        # = 2 * cross-correlation of w_full with a (for the symmetric part)
        
        # Use FFT: the gradient is 2 * ifft(fft(w_full_extended) * conj(fft(a_extended)))
        fsize = 1 << (2 * n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = x_new + ((t - 1.0) / t_new) * (x_new - a)
        momentum = np.maximum(momentum, 0.0)
        a = x_new
        t = t_new
        
        # Occasional perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            momentum = project_simplex(momentum + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-30:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def split_step_one(a, N, n, beta, rng):
    """
    One IRAM/split-step iteration.
    Flatten |FFT(a)| with exponent beta, IFFT, project to simplex.
    Phase is preserved (only magnitude modified).
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a
    
    # Power compression: mag_new = mag_mean * (mag/mag_mean)^beta
    # beta=1: no change; beta=0: perfectly flat
    ratio = mag / mag_mean
    mag_new = mag_mean * np.power(ratio, beta)
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

def nesterov_smoothed(a_init, n, S, max_time, rng, tau=0.4):
    """
    Nesterov-accelerated projected gradient on log-sum-exp smoothed objective.
    Minimizes log(sum_k exp(c_k/tau)) ~ max_k c_k as tau -> 0.
    """

        a = np.ones(n) * S / n
    else:

    t_nesterov = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Log-sum-exp gradient
        c_scaled = c / tau
        c_shifted = c_scaled - c_scaled.max()
        exp_c = np.exp(c_shifted)
        w = exp_c / exp_c.sum()
        
        # Gradient via FFT cross-correlation
        fsize = 1 << (2 * n - 1).bit_length()
        W = fft(w, fsize)
        A_y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A_y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

    deadline = time.monotonic() + max(0.0, budget_s - 25.0)
    
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 16384
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    # Build initializations
    inits = make_initializations(n, rng, S)
    print("  %d initializations" % len(inits))
    
    # Score all initializations
    for name, arr in inits:
        c = fft_convolve_self(arr)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = arr.copy()
        print("  [%s]: %.6f" % (name, sc))
    
    if global_a is None:
        global_a = np.ones(n) * (S / n)
        global_best = S * S / n  # = 2.0 for uniform
    
    print("Initial best: %.6f" % global_best)
    
    # === Phase 1: Split-step from multiple starts (35% time) ===
    print("=== Phase 1: Split-step multi-start ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.35
    
    for name, arr in inits:
        if time.monotonic() > t_phase1 + phase1_budget - 10:
            break
        rem = t_phase1 + phase1_budget - time.monotonic()
        if rem < 8:
            break
        time_budget = min(rem * 0.8, 50.0)
        
        a_sw, sc_sw = split_step_optimize(arr, n, N, time.monotonic() + time_budget,
                                          rng, target=1.45,
                                          beta_start=0.3, beta_end=0.9,
                                          restart_every=200)
        if sc_sw < global_best:
            global_best = sc_sw
            global_a = a_sw.copy()
            print("  [%s] split-step: %.6f" % (name, global_best))
        else:
            print("  [%s] split-step: %.6f (no improve)" % (name, sc_sw))
    
    print("=== Phase 1 done. Best: %.6f ===" % global_best)
    
    # === Phase 2: Alternating methods (35% time) ===
    print("=== Phase 2: Alternating optimization ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.35
    
    a_curr = global_a.copy()
    stall_global = 0
    round_num = 0
    
    while time.monotonic() < t_phase2 + phase2_budget:
        round_num += 1
        rem = t_phase2 + phase2_budget - time.monotonic()
        if rem < 10:
            break
        
        # Method 1: Split-step
        t_budget = min(rem * 0.3, 25.0)
        a_try, sc_try = split_step_optimize(a_curr, n, N, time.monotonic() + t_budget,
                                            rng, target=1.45,
                                            beta_start=0.6, beta_end=0.95,
                                            restart_every=150)
        if sc_try < global_best:
            global_best = sc_try
            global_a = a_try.copy()
            a_curr = a_try.copy()
            stall_global = 0
            print("  [P2 r%d] split-step: %.6f" % (round_num, global_best))
        else:
            stall_global += 1
        
        # Method 2: Nesterov
        rem = t_phase2 + phase2_budget - time.monotonic()
        if rem < 8:
            break
        t_budget = min(rem * 0.3, 25.0)
        tau = 0.2 + 0.4 * (1.0 / (1.0 + 0.1 * round_num))
        a_try, sc_try = nesterov_smoothed(a_curr, n, S, time.monotonic() + t_budget,
                                          rng, tau=tau)
        if sc_try < global_best:
            global_best = sc_try
            global_a = a_try.copy()
            a_curr = a_try.copy()
            stall_global = 0
            print("  [P2 r%d] nesterov: %.6f" % (round_num, global_best))
        else:
            stall_global += 1
        
        # Method 3: Coordinate descent
        rem = t_phase2 + phase2_budget - time.monotonic()
        if rem < 8:
            break
        t_budget = min(rem * 0.25, 20.0)
        a_try, sc_try = coord_descent_multi(a_curr, n, S, time.monotonic() + t_budget,
                                            rng, n_top=30)
        if sc_try < global_best:
            global_best = sc_try
            global_a = a_try.copy()
            a_curr = a_try.copy()
            stall_global = 0
            print("  [P2 r%d] coord: %.6f" % (round_num, global_best))
        else:
            stall_global += 1
        
        # Escalation: if stuck, try aggressive restart
        if stall_global >= 4:
            rem = t_phase2 + phase2_budget - time.monotonic()
            if rem < 8:
                break
            # Aggressive spectral perturbation
            a_pert_pad = np.zeros(N)
            a_pert_pad[:n] = a_curr
            A_pert = fft(a_pert_pad)
            mag_pert = np.abs(A_pert)
            phase_pert = rng.uniform(-np.pi, np.pi, N)
            A_new = mag_pert * np.exp(1j * phase_pert)
            a_pert = np.real(ifft(A_new))[:n]
            a_pert = project_simplex(a_pert, S)
            # Quick split-step from perturbed state
            t_budget = min(rem * 0.2, 15.0)
            a_try, sc_try = split_step_optimize(a_pert, n, N, time.monotonic() + t_budget,
                                                rng, target=1.45,
                                                beta_start=0.3, beta_end=0.85,
                                                restart_every=100)
            if sc_try < global_best:
                global_best = sc_try
                global_a = a_try.copy()
                a_curr = a_try.copy()
                print("  [P2 r%d] restart: %.6f" % (round_num, global_best))
            stall_global = 0
    
    print("=== Phase 2 done. Best: %.6f ===" % global_best)
    
    # === Phase 3: Final polish (30% time) ===
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3
    
    if global_a is not None and remaining > 10:
        # Long coordinate descent
        t_budget = min(remaining * 0.3, 120.0)
        a_cd, sc_cd = coord_descent_multi(global_a, n, S, time.monotonic() + t_budget,
                                          rng, n_top=50)
        if sc_cd < global_best:
            global_best = sc_cd
            global_a = a_cd.copy()
            print("  [P3] coord: %.6f" % global_best)
        
        # Fine split-step
        rem = deadline - time.monotonic()
        if rem > 10:
            t_budget = min(rem * 0.25, 60.0)
            a_ss, sc_ss = split_step_optimize(global_a, n, N, time.monotonic() + t_budget,
                                              rng, target=1.45,
                                              beta_start=0.85, beta_end=0.97,
                                              restart_every=200)
            if sc_ss < global_best:
                global_best = sc_ss
                global_a = a_ss.copy()
                print("  [P3] fine SS: %.6f" % global_best)
        
        # Final Nesterov with small tau
        rem = deadline - time.monotonic()
        if rem > 10:
            t_budget = min(rem * 0.25, 60.0)
            a_ng, sc_ng = nesterov_smoothed(global_a, n, S, time.monotonic() + t_budget,
                                            rng, tau=0.15)
            if sc_ng < global_best:
                global_best = sc_ng
                global_a = a_ng.copy()
                print("  [P3] nesterov fine: %.6f" % global_best)
        
        # One more coordinate descent
        rem = deadline - time.monotonic()
        if rem > 5:
            a_cd2, sc_cd2 = coord_descent_multi(global_a, n, S, time.monotonic() + rem * 0.4,
                                                rng, n_top=60)
            if sc_cd2 < global_best:
                global_best = sc_cd2
                global_a = a_cd2.copy()
                print("  [P3] final coord: %.6f" % global_best)
    
    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf (rejected)")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    
    print("Returning sequence of length %d, internal score=%.6f" % (len(final_list), global_best))
    return final_list
