"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses multi-phase optimization: FISTA with top-LSE smoothing, excess annealing,
peak shaving, and local search. Tries multiple sequence lengths.

import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho = np.max(np.where(cond)[0]) + 1 if np.any(cond) else n
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def objective(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def topk_lse(c, k, beta):
    """Smooth max focusing on top-k elements."""
    topk_idx = np.argsort(c)[::-1][:k]
    c_topk = c[topk_idx]
    cm = c_topk.max()
    scaled = beta * (c_topk - cm)
    scaled = np.minimum(scaled, 500)
    val = cm + np.log(np.exp(scaled).sum()) / beta
    return val

def topk_lse_grad(a, c, k, beta):
    """Gradient of top-k LSE."""
    n = len(a)
    topk_idx = np.argsort(c)[::-1][:k]
    c_topk = c[topk_idx]
    cm = c_topk.max()
    scaled = beta * (c_topk - cm)
    scaled = np.minimum(scaled, 500)
    e = np.exp(scaled)
    w_topk = e / e.sum()
    
    # Build full weight vector
    w = np.zeros(len(c))
    w[topk_idx] = w_topk
    
    # Gradient: 2 * convolve(w, a) (truncated to length n)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fista_topk_lse(a_init, n, S, max_time, rng, k_init=8, k_growth=2, beta_init=50, beta_max=5000):
    """FISTA with top-k LSE objective."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t = 1.0
    start = time.time()
    best = fft_convolve(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    
    while time.time() - start < max_time:
        it += 1
        k = min(k_init + k_growth * (it // 1000), n)
        beta = min(beta_max, beta_init * (1 + 0.01 * it))
        
        c = fft_convolve(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad = topk_lse_grad(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Backtracking step size
        step = 1.0 / (2 * beta * 0.1)  # initial guess based on Lipschitz estimate
        # Actually, the Lipschitz constant of the topk_lse gradient is at most 2 * beta * ||a||_2 * 2 (from the convolution structure)
        # More precisely, the Hessian of c[k] is a 2x2 block matrix, so the Lipschitz constant is bounded by 4 * max|a| * S or something
        # Let's use a conservative estimate
        step = 0.5 * S / (n * beta * 0.01)
        step = min(step, 1.0 / (beta * 0.5))
        
        # FISTA update
        z = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z + ((t - 1.0) / t_new) * (z - a)
        y = np.maximum(y, 0.0)
        a = z
        t = t_new
        
        # Restart if stuck
        if no_improve > 500:
            noise = 0.005 * S / np.sqrt(n) * np.exp(-no_improve / 2000.0)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
            no_improve = 0
    
    return best_a, best

def excess_anneal(a_init, n, S, max_time, rng):
    """Anneal threshold T downward, minimizing excess above T."""
    a = project_simplex(a_init.copy(), S)
    start = time.time()
    c0 = fft_convolve(a)
    best = c0.max()
    best_a = a.copy()
    
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    
    y = a.copy()
    t = 1.0
    it = 0
    
    while time.time() - start < max_time:
        it += 1
        progress = min(1.0, (time.time() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.6)
        T = max(T, T_end)
        
        c = fft_convolve(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Gradient of sum(max(c - T, 0)^2)
        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            # All below threshold, do a small regularization step
            T_new = c.max() - 0.005 * (c.max() - c.mean() + 1e-10)
            excess = np.maximum(c - max(T_new, c.mean()), 0.0)
        
        w = 2.0 * excess
        wsum = w.sum()
        if wsum < 1e-30:
            break
        # Don't normalize w - use it as-is for gradient
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Lipschitz constant estimate: 4 * max|a| * 2 * n (rough)
        step = 1.0 / (4.0 * np.max(np.abs(a)) * n * 0.1)
        step = min(step, 1.0 / (0.5 * S * n))
        
        z = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z + ((t - 1.0) / t_new) * (z - a)
        y = np.maximum(y, 0.0)
        a = z
        t = t_new
        
        # Perturbation
        if it % 1000 == 0:
            noise = 0.002 * S / np.sqrt(n) * (1 - progress)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)

    return best_a, best
