        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Annealing noise
        if stuck > 2000:
            noise_scale = 0.004 * S / np.sqrt(n) * np.exp(-stuck / 6000.0)
            if noise_scale > 1e-7 and it % 10 == 0:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                avg_grad = np.zeros(n)
                avg_count = 0

        if it % 5000 == 0:
            t_mom = 1.0
            if stuck > 5000:
                # Aggressive restart
                noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
                y = project_simplex(y + rng.normal(0, max(noise_scale, 2e-6), n), S)
                t_mom = 1.0
                avg_grad = np.zeros(n)
                avg_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_phase3(a_init, max_time, rng):
    """Phase 3: Intensive polishing with top-k and exact subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    stuck = 0
    step_size = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        # Adaptive k based on stuck count
        if stuck < 50:
            k = 1
            tol_frac = 0.0
        elif stuck < 200:
            k = 2
            tol_frac = 0.0
        elif stuck < 1000:
            k = min(4, 2 + stuck // 100)
            tol_frac = 0.005
        elif stuck < 5000:
            k = min(16, 4 + stuck // 500)
            tol_frac = 0.01
        else:
            k = min(64, 16 + stuck // 500)
            tol_frac = 0.02

        if k == 1 and tol_frac == 0.0:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k=k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step with merit backtracking
        step = step_size / (1.0 + 0.005 * it)
        for _ in range(6):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-14:
                break
            step *= 0.4

        y = x_new.copy()
        a = x_new.copy()

        # Noise injection for escape
        if stuck > 3000 and it % 50 == 0:
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_power_sum_phase(a_init, max_time, rng):
    """Alternative: minimize power-sum of top values."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        # Power-sum gradient: focus on top values with increasing power
        k = min(32, max(4, n // 128))
        power = min(4.0, 1.0 + 0.5 * (it / 1000.0) + 0.1 * stuck / 100.0)
        grad = topk_gradient(y, c, k=k, power=power)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (gnorm + 1e-10) * 0.8
        step = min(step, 1.0)

        x_new = project_simplex(y - step * grad, S)

        # Check merit
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur_max * 1.005:
            step *= 0.3
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 2000 and it % 15 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 5000.0)
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

        if it % 4000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: LSE FISTA -> transition to exact subgradient -> intensive polish
    -> power-sum alternative -> final refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    margin = 30.0
    deadline = time.monotonic() + max(0.0, budget_s - margin)
    t_total_start = time.monotonic()

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence from provided construction
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    obj0 = compute_obj_raw(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    def update_best(a, label=""):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj_raw(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  NEW BEST: obj=%.6f (%s)" % (obj, label))
        return obj

    # === Phase 1: FISTA with LSE (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p1_budget = remaining * 0.25
        print("\n=== Phase 1: FISTA-LSE at n=%d (budget %.0fs) ===" % (n0, p1_budget))
        t1 = time.monotonic()

        a1, _ = run_phase1(global_best_a, p1_budget, rng)
        update_best(a1, "Phase1")

        print("  Phase1 done (t=%.0fs), best=%.6f" % (time.monotonic() - t1, global_best_obj))

    # === Phase 2: Transition + exact subgradient (35% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.35
        print("\n=== Phase 2: Transition+Exact at n=%d (budget %.0fs) ===" % (n0, p2_budget))
        t2 = time.monotonic()

        a2, _ = run_phase2(global_best_a, p2_budget, rng)
        update_best(a2, "Phase2")

        print("  Phase2 done (t=%.0fs), best=%.6f" % (time.monotonic() - t2, global_best_obj))

    # === Phase 3: Intensive polish (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Intensive polish at n=%d (budget %.0fs) ===" % (n0, p3_budget))
        t3 = time.monotonic()

        a3, _ = run_phase3(global_best_a, p3_budget, rng)
        update_best(a3, "Phase3")

        print("  Phase3 done (t=%.0fs), best=%.6f" % (time.monotonic() - t3, global_best_obj))

    # === Phase 3b: Power-sum alternative (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p3b_budget = remaining * 0.10
        print("\n=== Phase 3b: Power-sum at n=%d (budget %.0fs) ===" % (n0, p3b_budget))
        t3b = time.monotonic()

        a3b, _ = run_power_sum_phase(global_best_a, p3b_budget, rng)
        update_best(a3b, "Phase3b-power")

        print("  Phase3b done (t=%.0fs), best=%.6f" % (time.monotonic() - t3b, global_best_obj))

    # === Phase 4: Multi-scale upsample (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and n0 <= 8192:
        p4_budget = remaining * 0.10
        n2 = n0 * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_budget))
        t4 = time.monotonic()

        a2_init = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())

        # Short FISTA on upsampled
        a4, _ = run_phase2(a2_init, p4_budget, rng)
        obj4 = compute_obj_raw(a4)
        print("  Upsampled obj=%.6f" % obj4)

        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()
            print("  NEW BEST from upsample: %.6f" % obj4)
        else:
            # Try downsampling back and comparing
            a4_ds = downsample(a4, factor=2)
            S_ds = np.sqrt(2.0 * n0)
            a4_ds = a4_ds * (S_ds / a4_ds.sum())
            update_best(a4_ds, "Phase4-downsampled")

        print("  Phase4 done (t=%.0fs), best=%.6f" % (time.monotonic() - t4, global_best_obj))

    # === Phase 5: Final intensive polish on best (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p5_budget = remaining * 0.85
        n_best = len(global_best_a)
        print("\n=== Phase 5: Final polish at n=%d (budget %.0fs) ===" % (n_best, p5_budget))
        t5 = time.monotonic()

        a5, _ = run_phase3(global_best_a, p5_budget, rng)
        update_best(a5, "Phase5-final")

        print("  Phase5 done (t=%.0fs), best=%.6f" % (time.monotonic() - t5, global_best_obj))

    # Final verification
    print("\n=== Final Summary ===")
    print("Total time: %.0fs" % (time.monotonic() - t_total_start))
    print("Best objective: %.6f (n=%d)" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Verify sum > 0.01
        if sum(final_list) < 0.01:
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]

        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
        else:
            print("WARNING: verification failed, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb

    return final_list
