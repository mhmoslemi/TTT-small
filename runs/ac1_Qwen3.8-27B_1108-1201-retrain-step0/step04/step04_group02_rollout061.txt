    2*n*max(conv(a,a)) / (sum(a))^2.

Strategy
--------
1.  Start from the supplied previous-best sequence when available, and also
    build several analytic/random initializations.
2.  Optimize at one or two nearby lengths (including an upsampled length).
3.  Use a hybrid local method:
    - log-space multiplicative descent with active-set/softmax subgradients;
    - LP-based active-constraint direction finding using HiGHS, with a

4.  Keep the best sequence by an internal FFT-based objective, and return it.

MAX_SEQUENCE_LENGTH = 16384
DENSE_LP_BUDGET = 4_000_000


def next_pow2(x):
    """Smallest power of two >= x, at least 1."""
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_with_sum(a, S):
    """Clip, sanitize, and rescale a to sum exactly S."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(int(max(1, S)), dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) * (float(S) / a.size)
    return a * (float(S) / s)


def objective_value(a):
    """Fast objective value using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float(2.0 * n * cm / (S * S))


def fft_convolve_self(a):
    """Full linear self-convolution of a by FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    fsize = next_pow2(2 * n - 1)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def fft_convolve_cross(a, b):
    """Full linear convolution of a and b by FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    fsize = next_pow2(n + m - 1)
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:n + m - 1]


def cross_correlation_weights(a, w):
    """
    If w has length 2*n-1, return an array g of length n with
        g[i] = 2 * sum_j w[i+j] * a[j]
    (valid convolution weights-only convolution).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.array([], dtype=np.float64)
    fsize = next_pow2(w.size + n - 1)
    W = fft(w, fsize)
    AR = fft(a[::-1], fsize)
    conv = np.real(ifft(W * AR))
    return 2.0 * conv[n - 1:2 * n - 1]


def topk_log_grad(a, c, k):
    """
    Log-space gradient proxy for the top-k largest convolution coefficients.
    This is proportional to a_i * d log C_i, and differs from the exact
    gradient only by a constant factor for each peak, which is harmless.
    """
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    if n == 0 or c.size == 0:
        return np.zeros(0, dtype=np.float64)
    k = max(1, min(k, c.size))
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n, dtype=np.float64)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        if hi >= lo:
            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    return a * g


def normalize_direction(g):
    """Center by median and rescale to unit RMS. Return None if degenerate."""
    g = np.asarray(g, dtype=np.float64)
    if g.size == 0:
        return None
    g = np.where(np.isfinite(g), g, 0.0)
    g = g - np.median(g)
    denom = float(np.linalg.norm(g))
    if denom < 1e-15:
        return None
    return g / (denom / np.sqrt(g.size))


def mirror_step(a, direction, eta, S):
    """Multiplicative update in log-space, then renormalize to sum S."""
    a = np.asarray(a, dtype=np.float64)
    direction = np.asarray(direction, dtype=np.float64)
    step = eta * direction
    step = np.clip(step, -20.0, 20.0)
    b = a * np.exp(-step)
    return normalize_with_sum(b, S)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum x = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return np.array([], dtype=np.float64)
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cum - float(S)) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cum[rho - 1] - float(S)) / rho
    return np.maximum(v - theta, 0.0)


def random_log_perturb(a, S, rng, scale):
    """Log-Gaussian perturbation plus a low-frequency modulation."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    v = np.log(np.maximum(a, 1e-300))
    v = v + rng.normal(0.0, scale, n)
    if n > 16:
        t = np.arange(n, dtype=np.float64)
        freq = float(rng.uniform(1.0, 7.0))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        v = v + 0.5 * scale * np.sin(2.0 * np.pi * freq * t / float(n) + phase)
    v = v - np.max(v)
    b = np.exp(np.clip(v, -50.0, 50.0))
    return normalize_with_sum(b, S)


def resample_length(a, m):
    """Periodic linear resampling from length n to length m."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if n <= 0:
        return np.zeros(m, dtype=np.float64)
    if n == 1:
        return np.full(m, float(a[0]), dtype=np.float64)
    x = np.arange(m, dtype=np.float64) * (float(n) / float(m))
    left = np.floor(x).astype(np.int64) % n
    right = (left + 1) % n
    frac = x - np.floor(x)
    return (1.0 - frac) * a[left] + frac * a[right]


def power_shape(n, left_alpha=0.5, right_alpha=0.0, gamma=0.0, strength=1.0):
    """A one-sided singular shape optionally multiplied by a exponential tilt."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    val = np.ones(n, dtype=np.float64)
    if left_alpha != 0.0:
        val *= np.power(t + 1.0e-12, -float(left_alpha))
    if right_alpha != 0.0:
        val *= np.power((1.0 - t) + 1.0e-12, -float(right_alpha))
    if gamma != 0.0:
        val *= np.exp(float(gamma) * t)
    val *= float(strength)
    return np.maximum(val, 1.0e-12)


def random_shape(n, rng):
    """One of several plausible nonnegative shapes for a random start."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    choice = int(rng.integers(0, 7))
    if choice == 0:
        alpha = float(rng.uniform(0.35, 0.55))
        val = power_shape(n, left_alpha=alpha)
    elif choice == 1:
        alpha = float(rng.uniform(0.35, 0.55))
        val = power_shape(n, right_alpha=alpha)
    elif choice == 2:
        alpha = float(rng.uniform(0.30, 0.55))
        beta = float(rng.uniform(0.0, 0.35))
        val = power_shape(n, left_alpha=alpha, right_alpha=beta)
    elif choice == 3:
        alpha = float(rng.uniform(0.40, 0.58))
        gamma = float(rng.uniform(-2.0, 2.0))
        val = power_shape(n, left_alpha=alpha, gamma=gamma)
    elif choice == 4:
        z = 2.0 * t - 1.0
        c0 = float(rng.normal(0.0, 0.25))
        c1 = float(rng.normal(0.0, 0.45))
        c2 = float(rng.normal(0.0, 0.45))
        c3 = float(rng.normal(0.0, 0.45))
        logv = c0 + c1 * z + c2 * z * z + c3 * z * z * z
        val = np.exp(np.clip(logv, -3.0, 3.0))
    elif choice == 5:
        alpha = 0.5
        r = float(rng.uniform(0.0, 0.55))
        val = np.power(t + 1e-12, -alpha) * (1.0 - r * t)
        val = np.maximum(val, 1e-8)
    else:
        alpha = float(rng.uniform(0.45, 0.65))
        beta = float(rng.uniform(-0.2, 0.45))
        gamma = float(rng.uniform(-1.0, 1.0))
        val = power_shape(n, left_alpha=alpha, right_alpha=beta, gamma=gamma)
    val = np.where(np.isfinite(val), val, 1.0)
    val = np.maximum(val, 1.0e-12)
    if float(val.max()) <= 0.0:
        val = np.ones(n, dtype=np.float64)
    return val


def choose_initial(given, n, rng):
    """Pick a good seed of exact length n from known and random candidates."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    S = float(np.sqrt(2.0 * n))
    candidates = []
    if given is not None and np.asarray(given).size > 0:
        candidates.append(resample_length(given, n))
    n_random = 3 if n > 8192 else 6
    for _ in range(n_random):
        candidates.append(random_shape(n, rng))
    # A few deterministic sensible shapes.
    candidates.append(power_shape(n, left_alpha=0.5))
    candidates.append(power_shape(n, left_alpha=0.45, gamma=0.7))
    candidates.append(power_shape(n, right_alpha=0.5))

    best = None
    best_obj = np.inf
    for cand in candidates:
        cand = np.asarray(cand, dtype=np.float64)
        if cand.size != n:
            cand = resample_length(cand, n)
        a = normalize_with_sum(cand, S)
        obj = objective_value(a)
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best = a
    if best is None:
        best = np.ones(n, dtype=np.float64) * (S / float(n))
    return best


def default_lp_max_K(n):
    """Maximum number of active LP rows to keep dense matrix memory small."""
    n = max(1, int(n))
    k = DENSE_LP_BUDGET // n
    return int(max(16, min(512, k)))


def build_lp_matrix(p, K):
    """
    Dense matrix A for rows (p * g)[k] <= M, plus a final sum(g) row.
    K contains 0 <= k < 2*n-1.
    """
    p = np.asarray(p, dtype=np.float64)
    K = np.asarray(K, dtype=np.int64)
    n = p.size
    Klen = K.size
    A = np.zeros((Klen + 1, n), dtype=np.float64)
    for r in range(Klen):
        k = int(K[r])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[r, lo:hi + 1] = p[k - hi:k - lo + 1][::-1]
    A[Klen, :] = 1.0
    return A


def lp_direction(a, c, budget, rng, max_K=64):
    """
    LP-based active-constraint direction.

    Solve max sum(g) subject to selected convolution constraints
    (p*g)[k] <= M0 and sum(g) <= 4. The solution is normalized back to sum(a).
    Violated omitted constraints are added iteratively when possible.
    """
    if linprog is None or budget < 3.0:
        return None
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None
    S = float(a.sum())
    if S <= 0.0:
        return None
    p = a / S
    S2 = S * S
    M0 = float(np.max(c)) / S2
    if not np.isfinite(M0) or M0 <= 0.0:
        return None

    full_len = 2 * n - 1
    order = np.argsort(c)[::-1]
    if order.size == 0:
        return None

    max_K = int(max(16, min(max_K, default_lp_max_K(n))))
    K = order[:min(max_K, order.size)].astype(np.int64)

    # Add a little diversity among the upper half of the spectrum of c.
    if K.size < max_K and full_len > K.size:
        half = int(max(K.size + 1, min(2 * K.size, full_len // 2)))
        rand_size = int(min(max_K - K.size, max(0, half - K.size)))
        if rand_size > 0:
            rand_idx = rng.choice(half, size=rand_size, replace=False)
            K = np.unique(np.concatenate((K, rand_idx.astype(np.int64))))
            if K.size > max_K:
                K = K[:max_K]
    K = np.sort(K)

    sum_upper = 4.0
    start_day = time.monotonic()
    max_iters = max(1, int(max(1, budget) // 3) + 1)

    best_g = None
    best_sum = 0.0
    best_viol = np.inf

    for _ in range(max_iters):
        elapsed = time.monotonic() - start_day
        if elapsed > max(0.0, budget - 1.0):
            break

        K = np.asarray(K, dtype=np.int64)
        if K.size == 0:
            break

        A = build_lp_matrix(p, K)
        b = np.concatenate((np.full(K.size, M0, dtype=np.float64),
                            np.array([sum_upper], dtype=np.float64)))
        jobjective = -np.ones(n, dtype=np.float64)

        solve_limit = max(1.0, budget - elapsed - 1.0)
        res = None
        try:
            res = linprog(jobjective, A_ub=A, b_ub=b,
                          bounds=(0.0, sum_upper),
                          method='highs',
                          options={'presolve': True, 'time_limit': solve_limit})
        except Exception:
            try:
                res = linprog(jobjective, A_ub=A, b_ub=b,
                              bounds=(0.0, sum_upper),
                              method='highs',
                              options={'presolve': True})
            except Exception:
                res = None

        if res is None or not hasattr(res, 'x') or res.x is None:
            break
        g = np.asarray(res.x, dtype=np.float64)
        if g.size != n or not np.all(np.isfinite(g)):
            break
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 0.0:
            break

        # Check omitted full convolution constraints.
        tg = fft_convolve_cross(p, g)
        if tg.size != full_len:
            break
        viol = float(np.max(np.asarray(tg, dtype=np.float64) - M0))
        viol_rel = viol / M0 if M0 > 0 else 0.0
        if not np.isfinite(viol_rel):
            viol_rel = 1e6

        if gs > best_sum and viol_rel < best_viol:
            best_sum = gs
            best_viol = viol_rel
            best_g = g / gs

        status = getattr(res, 'status', -1)
        if status == 0 and viol_rel < 1e-5:
            break

        # Add violated constraints if there is room.
        if K.size < max_K:
            viol_idx = np.where(tg > M0 * (1.0 + 1.0e-7))[0]
            if viol_idx.size > 0:
                if viol_idx.size > max(16, K.size):
                    score = np.asarray(tg[viol_idx], dtype=np.float64)
                    local = np.argsort(score)[::-1]
                    add_size = int(min(max_K - K.size, max(8, (max_K - K.size) // 2)))
                    add_idx = viol_idx[local[:add_size]]
                else:
                    add_idx = viol_idx
                K = np.unique(np.concatenate((K, add_idx.astype(np.int64))))
                if K.size > max_K:
                    K = K[:max_K]
        else:
            base_size = int(max(8, max_K // 4))
            base = order[:base_size].astype(np.int64)
            viol_order = np.argsort(-np.asarray(tg, dtype=np.float64))[:max(1, max_K - base_size)]
            K = np.unique(np.concatenate((base, viol_order.astype(np.int64))))
            if K.size > max_K:
                K = K[:max_K]

        if best_g is not None and time.monotonic() - start_day > 0.75 * budget:
            break

    if best_g is not None and best_sum >= 0.999:
        return best_g * S
    return None


def optimize_sequence(a_init, time_limit, rng, name="seq"):
    """
    Local hybrid optimizer for a fixed length.
    Returns (best_sequence, best_objective).
    """
    a_init = np.asarray(a_init, dtype=np.float64)
    n = a_init.size
    if n <= 0:
        fallback = np.ones(1024, dtype=np.float64)
        S = np.sqrt(2.0 * 1024)
        return fallback * (S / 1024.0), objective_value(fallback * (S / 1024.0))

    S = float(np.sqrt(2.0 * n))
    a = normalize_with_sum(a_init, S)
    c = fft_convolve_self(a)
    c = np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)
    cur = float(2.0 * n * np.max(c) / (S * S))
    if not np.isfinite(cur) or cur > 1e6:
        a = np.ones(n, dtype=np.float64) * (S / float(n))
        c = fft_convolve_self(a)
        c = np.maximum(c, 0.0)
        cur = float(2.0 * n * np.max(c) / (S * S))

    best_obj = cur
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    eta = 0.22
    failed = 0
    last_lp = t0
    lp_calls = 0

    lp_enabled = (linprog is not None and time_limit > 60.0)
    if not lp_enabled:
        max_lp_calls = 0
    else:
        max_lp_calls = int(max(2, min(30, time_limit / 45.0)))
    lp_max_K = default_lp_max_K(n)

    print("[%s] n=%d start obj=%.8f budget=%.0fs" % (name, n, cur, time_limit))

    while True:
        elapsed = time.monotonic() - t0
        rem = float(time_limit) - elapsed
        if rem < 2.0:
            break

        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()

        do_lp = False
        if lp_enabled and lp_calls < max_lp_calls:
            if it == 0 or (it % 90 == 0) or failed > 12 or (time.monotonic() - last_lp > 35.0):
                do_lp = True

        if do_lp:
            last_lp = time.monotonic()
            lp_calls += 1
            max_budget = max(rem * 0.25, 6.0)
            budget_lp = min(45.0, max(3.0, max_budget))
            g = lp_direction(a, c, budget_lp, rng, max_K=lp_max_K)
            it += 10

            if g is not None:
                base = a
                best_t = 0.0
                best_mix_a = None
                best_mix_c = None
                best_mix_obj = cur
                for t in (0.1, 0.25, 0.5, 0.75, 1.0):
                    m = (1.0 - t) * base + t * g
                    m = normalize_with_sum(m, S)
                    cm = fft_convolve_self(m)
                    cm = np.maximum(np.where(np.isfinite(cm), cm, 0.0), 0.0)
                    ob = float(2.0 * n * np.max(cm) / (S * S))
                    if np.isfinite(ob) and ob < best_mix_obj - 1e-12:
                        best_mix_obj = ob
                        best.mix_t = t
                        best_mix_a = m
                        best_mix_c = cm
                if best_mix_a is not None:
                    a = best_mix_a
                    c = best_mix_c
                    cur = best_mix_obj
                    eta = min(1.0, max(eta, 0.18))
                    failed = 0
                    if cur < best_obj - 1e-12:
                        best_obj = cur
                        best_a = a.copy()
                    if cur < 1.52 or True:
                        pass
                else:
                    failed += 1
            else:
                failed += 1

        else:
            # Dense exact active-set gradient most of the time, smoothed LSE occasionally.
            if it % 4 >= 2:
                # Smooth.softmax gradient
                beta = 10.0 + min(250.0, 0.05 * it)
                cm = float(np.max(c))
                with np.errstate(under='ignore', over='ignore'):
                    e = np.exp(beta * (c - cm))
                tot = float(np.sum(e))
                if not np.isfinite(tot) or tot <= 0.0:
                    lg = topk_log_grad(a, c, 4)
                else:
                    w = e / tot
                    cross = cross_correlation_weights(a, w)
                    lg = a * cross
            else:
                lg = topk_log_grad(a, c, 4)

            dirv = normalize_direction(lg)
            if dirv is None:
                break

            improved = False
            e_try = eta
            for _ in range(4):
                b = mirror_step(a, dirv, e_try, S)
                cb = fft_convolve_self(b)
                cb = np.maximum(np.where(np.isfinite(cb), cb, 0.0), 0.0)
                ob = float(2.0 * n * np.max(cb) / (S * S))
                if np.isfinite(ob) and ob < cur - 1e-12:
                    a = b
                    c = cb
                    cur = ob
                    eta = min(1.2, e_try * 1.12)
                    improved = True
                    if cur < best_obj - 1e-12:
                        best_obj = cur
                        best_a = a.copy()
                    break
                e_try *= 0.55

            it += 1
            if not improved:
                failed += 1
                eta = max(0.02, eta * 0.78)
                if failed > 12:
                    scale = 0.03 if failed < 22 else 0.09
                    a = random_log_perturb(a, S, rng, scale)
                    c = fft_convolve_self(a)
                    c = np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)
                    cur = float(2.0 * n * np.max(c) / (S * S))
                    if not np.isfinite(cur) or cur > 1e6:
                        a = np.ones(n, dtype=np.float64) * (S / float(n))
                        c = fft_convolve_self(a)
                        cur = float(2.0 * n * np.max(c) / (S * S))
                    eta = 0.2
                    failed = 0
            else:
                failed = 0

        if it % 800 == 0:
            print("[%s] it=%d obj=%.8f best=%.8f rem=%.0fs" %
                  (name, it, cur, best_obj, rem))

    best_a = normalize_with_sum(best_a, S)
    cbest = fft_convolve_self(best_a)
    obest = float(2.0 * n * np.max(np.maximum(cbest, 0.0)) / (S * S))
    if not np.isfinite(obest):
        best_a = a
        best_a = normalize_with_sum(best_a, S)
        cbest = fft_convolve_self(best_a)
        obest = float(2.0 * n * np.max(np.maximum(cbest, 0.0)) / (S * S))
    print("[%s] final obj=%.8f elapsed=%.0fs" % (name, obest, time.monotonic() - t0))
    return best_a, obest


def load_given_sequence():
    """Load the previous best sequence from the global variable if present."""
    try:
        seq = height_sequence_1
    except Exception:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size < 64:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() <= 0.0:
        return None
    if arr.size > 200000:
        arr = resample_length(arr, MAX_SEQUENCE_LENGTH)
    if arr.size > MAX_SEQUENCE_LENGTH:
        arr = resample_length(arr, MAX_SEQUENCE_LENGTH)
    return arr


def make_tasks(given):
    """Choose sequence lengths and rough time fractions."""
    if given is not None:
        Lg = int(given.size)
        L1 = int(max(1024, min(MAX_SEQUENCE_LENGTH, Lg)))
        if L1 < 2048:
            L1 = 2048
        if L1 < MAX_SEQUENCE_LENGTH:
            L2 = int(min(MAX_SEQUENCE_LENGTH, max(16384, 2 * L1)))
        else:
            L2 = int(max(2048, L1 // 2))
        L3 = int(max(1024, min(8192, L1 // 2)))
        raw = [(L1, 0.42), (L2, 0.32), (L3, 0.26)]
    else:
        raw = [(8192, 0.45), (16384, 0.30), (4096, 0.25)]

    tasks = []
    used = set()
    for n, frac in raw:
        try:
            n = int(n)
        except Exception:
            continue
        if n < 512:
            n = 1024
        if n > MAX_SEQUENCE_LENGTH:
            n = MAX_SEQUENCE_LENGTH
        if n in used:
            continue
        used.add(n)
        tasks.append((n, float(frac)))
    return tasks


def propose_candidate(seed=42, budget_s=2400):
    """
    Search and return a list of non-negative heights.
    This function is invoked by the scoring harness.
    """
    budget_s = float(max(60.0, budget_s))
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, budget_s - 60.0)

    given = load_given_sequence()
    given_obj = np.inf
    if given is not None:
        try:
            given_obj = objective_value(given)
        except Exception:
            given_obj = np.inf

    tasks = make_tasks(given)
    total_frac = 0.0
    for _, fr in tasks:
        total_frac += fr
    if total_frac <= 0.0:
        tasks = [(8192, 1.0)]
        total_frac = 1.0

    global_a = None
    global_obj = np.inf

    original_available = max(30.0, deadline - time.monotonic())
    print("Tasks: %s" % [(int(n), float(f)) for n, f in tasks])

    for task_idx, (n_len, frac) in enumerate(tasks):
        remaining = deadline - time.monotonic()
        if remaining < 60.0:
            break
        t_alloc = original_available * (float(frac) / total_frac)
        t_alloc = min(t_alloc, max(20.0, remaining - 60.0))
        if t_alloc < 40.0:
            t_alloc = max(30.0, remaining - 60.0)
        if t_alloc < 30.0:
            break

        # Prefer the current global best as the source for resampling if it is
        # at least as good as the given sequence.
        if global_a is not None and global_obj <= given_obj:
            source = global_a
        else:
            source = given

        init = choose_initial(source, int(n_len), rng)
        name = "task%d_n%d" % (task_idx, int(n_len))
        print("=== %s: allocating %.0fs ===" % (name, t_alloc))
        a_opt, obj_opt = optimize_sequence(init, t_alloc, rng, name=name)
        if obj_opt < global_obj:
            global_obj = obj_opt
            global_a = a_opt.copy()
            print("New global best: %.8f at n=%d" % (global_obj, int(len(a_opt))))

    # Final intensive polish on the best candidate found.
    remaining = deadline - time.monotonic()
    if remaining > 80.0 and global_a is not None:
        final_len = int(len(global_a))
        init = global_a
        t_final = max(30.0, remaining - 70.0)
        a_opt, obj_opt = optimize_sequence(init, t_final, rng, name="final_polish")
        if obj_opt < global_obj:
            global_obj = obj_opt
            global_a = a_opt.copy()
            print("Final polish improved to: %.8f" % global_obj)

    # Robust fallbacks.
    if global_a is None:
        n_fb = 8192
        rng_fb = np.random.default_rng(12345)
        global_a = random_shape(n_fb, rng_fb)
        global_a = normalize_with_sum(global_a, np.sqrt(2.0 * n_fb))
        global_obj = objective_value(global_a)

    # If the global best is unexpectedly long, downsample to keep eventual
    # sparse/direct convolution evaluation tractable.
    if int(len(global_a)) > MAX_SEQUENCE_LENGTH:
        global_a = resample_length(global_a, MAX_SEQUENCE_LENGTH)
        global_a = normalize_with_sum(global_a, np.sqrt(2.0 * MAX_SEQUENCE_LENGTH))

    # Final sanitization and validation by the internal objective.
    global_a = np.asarray(global_a, dtype=np.float64)
    global_a = np.clip(global_a, 0.0, 1000.0)
    global_a = np.where(np.isfinite(global_a), global_a, 0.0)
    if global_a.sum() < 0.01:
        n_fb = 1024
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2.0 * n_fb) / float(n_fb))

    final_obj = objective_value(global_a)
    if not np.isfinite(final_obj):
        n_fb = 1024
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2.0 * n_fb) / float(n_fb))
        final_obj = objective_value(global_a)

    print("Returning sequence length=%d objective=%.8f total_time=%.0fs" %
          (int(len(global_a)), final_obj, time.monotonic() + 0.0))

    return [float(x) for x in global_a]
