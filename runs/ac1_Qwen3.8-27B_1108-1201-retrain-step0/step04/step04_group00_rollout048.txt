def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    shifted = shifted - shifted.max()

    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_hybrid(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                     beta_max=50000.0, use_mw=False, anneal_period=2000,
                     anneal_factor=0.5):
    """
    FISTA with hybrid strategy: LSE + periodic exact subgradient +
    optional multiplicative weights update. Time-annealing for beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-15, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta with periodic annealing
        base_beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        if anneal_period > 0 and it % anneal_period == 0:
            # Periodic reheating
            beta = base_beta * anneal_factor
            # Inject noise
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
        elif stuck_count > 1000 and it % 500 == 0:
            beta = base_beta * 0.3
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 3000.0)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
        else:
            beta = base_beta

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient: mix of LSE and exact subgradient
        if stuck_count > 2000 and it % 3 == 0:
            tol = min(0.05, 0.005 * (1.0 + stuck_count / 5000.0))
            grad = exact_subgradient(y, c, tol_frac=tol)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = 1.0 / max(beta, 1.0)
        if stuck_count > 5000:
            step *= 0.5

        if use_mw:
            # Multiplicative weights update
            eta = step / max(gnorm, 1e-10)
            eta = min(eta, 0.5)
            log_update = -eta * grad
            log_update -= log_update.max()
            new_a = a * np.exp(log_update)
            ns = new_a.sum()
            if ns < 1e-30:
                new_a = project_simplex(y - step * grad, S)
            else:
                new_a = S * new_a / ns
            x_new = new_a
        else:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = y * (S / max(y.sum(), 1e-30))
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_set_polish(a_init, max_time, rng):
    """
    Polishing with exact subgradients and adaptive active set.
    Uses decreasing step sizes and periodic noise.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 100:
            k = 1
            tol_frac = 0.0
        elif stuck < 500:
            k = 3
            tol_frac = 0.003
        elif stuck < 2000:
            k = min(16, 4 + stuck // 200)
            tol_frac = 0.008
        elif stuck < 5000:
            k = min(32, 16 + stuck // 500)
            tol_frac = 0.015
        else:
            k = min(64, 32 + stuck // 1000)
            tol_frac = 0.03

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polychotomic step size
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / max(gnorm, 1e-10))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when very stuck
        if stuck > 3000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def local_coordinate_search(a_init, max_time, rng, n_trials=200):
    """
    Direct local search: perturb individual coordinates or small groups
    and accept if improvement.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    c_best = fft_convolve_self(a)

    trial = 0
    while time.monotonic() - start < max_time and trial < n_trials:
        trial += 1
        # Pick a random coordinate or small group
        group_size = rng.integers(1, min(5, n // 4) + 1)
        idx = rng.integers(0, n, group_size)

        # Try multiplicative perturbation
        for _ in range(10):
            a_try = best_a.copy()
            deltas = rng.normal(0, 0.05, group_size)
            a_try[idx] = best_a[idx] * np.exp(deltas)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try * (S / s_try)

            c_try = fft_convolve_self(a_try)
            obj_try = c_try.max()

            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                c_best = c_try.copy()
                break

    return best_a


def upsample(a, factor=2):
    """Upsample using interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: exploration, FISTA hybrid, active-set polish, local search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a_candidate):
        nonlocal global_best_obj, global_best_a
        if a_candidate is not None and len(a_candidate) > 0:
            obj = compute_obj(a_candidate)
            if np.isfinite(obj) and obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a_candidate.copy()
                print("  Improved! obj=%.6f (n=%d, t=%.0fs elapsed)" %
                      (obj, len(a_candidate), time.monotonic() - (deadline - budget_s + 30.0)))

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Initial from height_sequence_1: n=%d, obj=%.6f" % (n0, obj0))
        update_best(a_start)
    else:
        n0 = 8192
        a_start = np.ones(n0) * (np.sqrt(2.0 * n0) / n0)
        update_best(a_start)

    # === Phase 1: Multi-size exploration (15% time) ===
    p1_budget = (deadline - time.monotonic()) * 0.15
    print("\n=== Phase 1: Multi-size exploration (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    sizes_to_try = []
    if n0 <= 8192:
        sizes_to_try = [n0, n0 * 2]
    elif n0 <= 16384:
        sizes_to_try = [n0 // 2, n0, n0 * 2]
    else:
        sizes_to_try = [n0 // 2, n0]

    # Add 8192 if not in list
    if 8192 not in sizes_to_try:
        sizes_to_try.append(8192)

    # Sort by size and deduplicate
    sizes_to_try = sorted(set(s for s in sizes_to_try if 1024 <= s <= 32768))

    time_per_size = p1_budget / max(len(sizes_to_try), 1)
    for sz in sizes_to_try:
        if time.monotonic() >= deadline:
            break
        remaining = min(time_per_size, deadline - time.monotonic())
        if remaining < 5:
            continue

        # Initialize from best (upsampled or downsampled)
        if len(global_best_a) == sz:
            a_sz = global_best_a.copy()
        elif len(global_best_a) < sz:
            factor = sz // len(global_best_a)
            if factor == 1:
                a_sz = global_best_a.copy()
            else:
                a_sz = upsample(global_best_a, factor=min(factor, 4))
                if len(a_sz) != sz:
                    # Resample
                    t_old = np.linspace(0, 1, len(a_sz), endpoint=False)
                    t_new = np.linspace(0, 1, sz, endpoint=False)
                    a_sz = np.interp(t_new, t_old, a_sz)
        else:
            a_sz = downsample(global_best_a, 1)
            if len(a_sz) != sz:
                t_old = np.linspace(0, 1, len(a_sz), endpoint=False)
                t_new = np.linspace(0, 1, sz, endpoint=False)
                a_sz = np.interp(t_new, t_old, a_sz)

        S_sz = np.sqrt(2.0 * sz)
        a_sz = a_sz * (S_sz / a_sz.sum())

        # Short FISTA
        a_opt, _ = run_fista_hybrid(a_sz, remaining, rng,
                                     beta_init=5.0, beta_rate=0.01,
                                     beta_max=2000.0, use_mw=False,
                                     anneal_period=0)
        update_best(a_opt)

    print("Phase 1 done: best obj=%.6f" % global_best_obj)

    # === Phase 2: Intensive FISTA (35% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.35
        n_best = len(global_best_a)
        print("\n=== Phase 2: Intensive FISTA at n=%d (budget %.0fs) ===" % (n_best, p2_budget))
        t2 = time.monotonic()

        # Run with different parameter settings
        configs = [
            dict(beta_init=5.0, beta_rate=0.008, beta_max=5000.0, use_mw=False, anneal_period=3000, anneal_factor=0.5),
            dict(beta_init=20.0, beta_rate=0.004, beta_max=10000.0, use_mw=True, anneal_period=2000, anneal_factor=0.4),
            dict(beta_init=50.0, beta_rate=0.002, beta_max=30000.0, use_mw=False, anneal_period=1500, anneal_factor=0.3),
        ]

        time_per_config = p2_budget / len(configs)
        for cfg in configs:
            if time.monotonic() >= deadline:
                break
            remaining_cfg = min(time_per_config, deadline - time.monotonic())
            if remaining_cfg < 5:
                continue
            # Small perturbation for diversity
            a_pert = global_best_a + rng.normal(0, 0.01 * np.sqrt(2.0 * len(global_best_a)) / np.sqrt(len(global_best_a)), len(global_best_a))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * (np.sqrt(2.0 * len(a_pert)) / a_pert.sum())

            a_opt, _ = run_fista_hybrid(a_pert, remaining_cfg, rng, **cfg)
            update_best(a_opt)

    # === Phase 3: Active-set polishing (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p3_budget = remaining * 0.30
        n_best = len(global_best_a)
        print("\n=== Phase 3: Active-set polish at n=%d (budget %.0fs) ===" % (n_best, p3_budget))
        t3 = time.monotonic()

        # Split time into two runs
        a_opt, _ = run_active_set_polish(global_best_a, p3_budget * 0.6, rng)
        update_best(a_opt)

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 5:
            # Second polish with perturbation
            a_pert = global_best_a + rng.normal(0, 0.005 * np.sqrt(2.0 * n_best) / np.sqrt(n_best), n_best)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * (np.sqrt(2.0 * n_best) / a_pert.sum())
            a_opt2, _ = run_active_set_polish(a_pert, remaining3, rng)
            update_best(a_opt2)

    # === Phase 4: Local coordinate search (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p4_budget = remaining * 0.50
        n_best = len(global_best_a)
        print("\n=== Phase 4: Local coordinate search (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a_opt = local_coordinate_search(global_best_a, p4_budget, rng,
                                         n_trials=500)
        update_best(a_opt)

    # === Phase 5: Try different sizes with the optimized shape (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_best_a is not None:
        n_best = len(global_best_a)
        print("\n=== Phase 5: Size exploration (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()

        # Try 2x upsample
        if n_best <= 16384:
            n_2x = n_best * 2
            a_2x = upsample(global_best_a, factor=2)
            S_2x = np.sqrt(2.0 * n_2x)
            a_2x = a_2x * (S_2x / a_2x.sum())
            obj_2x_init = compute_obj(a_2x)
            print("  2x upsample init: obj=%.6f" % obj_2x_init)

            if obj_2x_init < global_best_obj * 1.05:  # Close enough to be worth trying
                time_left = deadline - time.monotonic()
                if time_left > 10:
                    a_opt, _ = run_fista_hybrid(a_2x, time_left * 0.5, rng,
                                                 beta_init=5.0, beta_rate=0.006,
                                                 beta_max=5000.0, use_mw=False,
                                                 anneal_period=0)
                    update_best(a_opt)
                    time_left = deadline - time.monotonic()
                    if time_left > 5:
                        a_opt2, _ = run_active_set_polish(global_best_a, time_left, rng)
                        update_best(a_opt2)

        # Try half size
        if n_best >= 4096:
            n_half = n_best // 2
            a_half = downsample(global_best_a, factor=2)
            S_half = np.sqrt(2.0 * n_half)
            a_half = a_half * (S_half / a_half.sum())
            obj_half = compute_obj(a_half)
            print("  Half downsample: obj=%.6f" % obj_half)
            if obj_half < global_best_obj:
                update_best(a_half)
                time_left = deadline - time.monotonic()
                if time_left > 10:
                    a_opt, _ = run_active_set_polish(a_half, time_left * 0.7, rng)
                    update_best(a_opt)

    # === Final: One more polishing pass ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_best_a is not None:
        print("\n=== Final polish (budget %.0fs) ===" % remaining)
        a_opt, _ = run_active_set_polish(global_best_a, remaining * 0.8, rng)
        update_best(a_opt)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Verify sum > 0.01
        if sum(final_list) < 0.01:
            final_list = [1.0] * 8192
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Internal objective: %.6f" % global_best_obj)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            final_list = [float(x) for x in np.ones(8192) * (np.sqrt(2 * 8192) / 8192)]
    else:
        n_fb = 8192
        final_list = [float(x) for x in np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)]
        print("Using fallback uniform sequence")

    return final_list
