"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE smoothing with FISTA, backtracking, and multi-peak tracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_conv(a):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def adjust_sum(a, S):
    """Rescale a to have sum S, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)

def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE(beta * max(c)) approximation.
    Returns (gradient, max_c, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = c - cm
    # For numerical stability
    e = np.exp(beta * shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    lse_val = cm + np.log(tot) / beta
    
    # grad LSE = sum_k w_k * grad(result[k])
    # grad(result[k])[m] = 2*a[k-m] if 0 <= k-m < n
    # So grad = 2 * sum_k w_k * a[k-m] (valid range)
    # This is 2 * (w * a) at the appropriate lag... 
    # Actually: grad[m] = 2 * sum_k w[k] * a[k-m] (where 0<=k-m<n)
    # = 2 * (w reversed convolved with a) at position m... 
    # Let's think: for each m, grad[m] = 2 * sum_k w[k] * a[k-m]
    # Let j = k-m, so k = j+m: grad[m] = 2 * sum_j w[j+m] * a[j] (where 0<=j<n, 0<=j+m<2n-1)
    # This is 2 * (cross-correlation of w and a) at lag m
    
    # More efficiently: 
    # grad[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * sum_j a[j] * w[j+m]  (substituting j = k-m)
    # = 2 * (a * w)[m] where * is cross-correlation
    
    # Using FFT: this is the inverse FFT of A * conj(W)
    fsize = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(A * np.conj(W)))[:n]
    
    return grad, cm, lse_val

def run_adaptive_fista(a_init, n, S, max_time, rng):
    """FISTA with adaptive LSE, backtracking, and momentum resets."""
    a = adjust_sum(a_init, S)
    y = a.copy()
    t_km1 = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    consecutive_increases = 0
    step_scale = 1.0
    
    # Adaptive parameters
    beta = 50.0
    beta_growth = 1.03  # multiplicative growth per 100 iters
    beta_max = 20000.0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Periodic beta increase
        if it % 100 == 0:
            beta = min(beta * beta_growth, beta_max)
        
        # Compute convolution and gradient
        c = fft_conv(y)
        cm = c.max()
        cur_obj = cm  # Our objective (with S normalization)
        
        if cur_obj < best:
            best = cur_obj
            best_a = y.copy()
            consecutive_increases = 0
        else:
            consecutive_increases += 1
        
        # Reset momentum if stuck
        if consecutive_increases > 2000:
            t_km1 = 1.0
            y = a.copy()
            step_scale *= 0.5
            consecutive_increases = 0
        
        # Compute LSE gradient
        grad, cm, _ = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = step_scale / (2.0 * beta)
        # Additional reduction if not improving
        if consecutive_increases > 500:
            step *= 0.5
        elif consecutive_increases > 200:
            step *= 0.75
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_km1 * t_km1))
        y = x_new + ((t_km1 - 1.0) / t_k) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum is approximately S
        y = adjust_sum(y, S)
        a = x_new
        t_km1 = t_k
        
        # Periodic noise to escape local minima
        if it % 10000 == 0 and it > 5000:
            decay = np.exp(-(it - 5000) / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_km1 = 1.0
                a = y.copy()
    
    # Final check
    c_final = fft_conv(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def run_exact_subgradient(a_init, n, S, max_time, rng):
    """Pure subgradient method at the active peak with peak tracking."""
    a = adjust_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step_size = S / (2.0 * np.sqrt(n))  # Initial step
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_conv(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Find all "active" peaks (within epsilon of max)
        threshold = cm - 0.01 * cm  # 1% below max
        active = np.where(c >= threshold)[0]
        
        # Compute averaged subgradient over active peaks
        grad = np.zeros(n)
        for k in active:
            # grad_k[m] = 2*a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            m_vals = np.arange(lo, hi + 1)
            a_vals = a[k - m_vals]
            grad[m_vals] += 2.0 * a_vals
        
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step: decrease if not improving
        step = step_size * (2.0 / (it + 1.0))  # L_i approximation
        a_new = project_simplex(a - step * grad, S)
        a = a_new
        
        # Periodic restart
        if it % 5000 == 0:
            small_noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a = project_simplex(a + small_noise, S)
    
    c_final = fft_conv(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def adaptive_polish(a_init, n, S, max_time, rng):

    a = adjust_sum(a_init, S)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE-smoothed FISTA with backtracking, KKT-based polishing,
and diverse restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v, axis=0)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def adjust_sum(a, S):
    """Rescale non-negative a to have sum exactly S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) * S / len(a)
    return a * (S / s)


def lse_gradient_fft(a, c, beta):
    """Compute LSE gradient using FFT cross-correlation.
    Returns (grad, max_c).
    grad[m] = 2 * sum_j a[j] * w[j+m] where w = softmax(beta*(c-max_c))
    """
    n = len(a)
    cm = c.max()
    shifted = c - cm
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0 / beta, 0.0)
    e = np.exp(beta * shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot

    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * w[j+m]
    # This is 2 * cross-correlation of a and w at lag m
    # Using FFT: ifft(fft(w) * conj(fft(a))) gives the full cross-correlation
    fsize = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    full = np.real(ifft(W * np.conj(A)))
    # The valid part starts at index len(w) - n = (2n-1) - n = n-1
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad, cm


def exact_peak_gradient(a, c, num_peaks=3, rel_tol=0.01):
    """Compute weighted subgradient using top peaks."""
    n = len(a)
    cm = c.max()
    threshold = cm * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    if len(active) > num_peaks:
        # Use top num_peaks
        top_vals = c[active]
        order = np.argsort(top_vals)[::-1][:num_peaks]
        active = active[order]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            m_vals = np.arange(lo, hi + 1)
            grad[m_vals] += 2.0 * a[k - m_vals]
    grad /= len(active)
    return grad, cm


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_start=30.0,
                       beta_end=8000.0, ramp_frac=0.3):
    """FISTA with adaptive LSE smoothing and backtracking."""
    a = adjust_sum(a_init, S)
    y = a.copy()
    t_km1 = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step_scale = 1.0
    last_improve_it = 0
    beta = beta_start

    ramp_iters = max(100, int(ramp_frac * max_time * 50))  # approximate

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: exponential ramp then hold
        if it < ramp_iters:
            frac = it / ramp_iters
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end
            # Slow increase after ramp
            if it % 5000 == 0:
                beta = min(beta * 1.1, beta_end * 3)

        # Compute convolution
        c = fft_conv(y)
        cm = c.max()

        if cm < best - 1e-9:
            best = cm
            best_a = y.copy()
            last_improve_it = it

        # Compute gradient
        stall = it - last_improve_it
        if beta > 500 and rng.random() < min(0.3, stall / 20000.0):
            # Use exact peak gradient for precision
            grad, cm = exact_peak_gradient(y, c, num_peaks=3, rel_tol=0.005)
        else:
            grad, cm = lse_gradient_fft(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with adaptive scaling
        if stall < 500:
            step = step_scale / (1.5 * beta)
        elif stall < 2000:
            step = step_scale * 0.5 / (1.5 * beta)
        else:
            step = step_scale * 0.25 / (1.5 * beta)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_km1 * t_km1))
        y = x_new + ((t_km1 - 1.0) / t_k) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_km1 = t_k

        # Backtracking: if no improvement for a while, reduce step
        if stall > 3000 and it % 1000 == 0:
            step_scale *= 0.7
            t_km1 = 1.0
            y = a.copy()

        # Periodic noise injection to escape local minima
        if it % 8000 == 0 and it > 4000:
            elapsed_frac = (time.monotonic() - start) / max_time
            decay = max(0.0, 1.0 - elapsed_frac * 1.5)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0.0, noise_scale, n), S)
                a = y.copy()
                t_km1 = 1.0

        # Periodic momentum reset
        if it % 5000 == 0:
            t_km1 = 1.0
            y = a.copy()

    # Final evaluation
    c_final = fft_conv(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

    alpha = 0.05

            best_a = a.copy()

        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:

        if gnorm < 1e-14:
            break

        a_pad[:n] = a
        F = fft(a_pad)
        P = np.abs(F) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Adaptive flattening strength
        progress = min(1.0, it / 500.0)
        alpha = 0.2 + 0.5 * progress

        # Target: flatten the power spectrum with peak suppression
        P_ratio = P / np.maximum(P_mean, 1e-30)
        # Suppress peaks, boost valleys
        P_new = P * np.power(P_ratio, -alpha * 0.5)
        # Blend
        P_new = (1.0 - alpha) * P + alpha * P_new
        # Renormalize
        P_new = P_new * (P.sum() / P_new.sum())

        # Reconstruct with same phase
        phase = F / np.maximum(np.abs(F), 1e-30)
        mag_new = np.sqrt(np.maximum(P_new, 0))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project to simplex
        a_new = np.maximum(a_raw, 0.0)
        a_new = adjust_sum(a_new, S)

        c = fft_conv(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def generate_starts(n, rng, parent):
    """Generate diverse starting sequences."""
    S = np.sqrt(2.0 * n)
    starts = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        starts.append(("parent", adjust_sum(parent, S)))

    starts.append(("uniform", np.ones(n)))
    starts.append(("cosine", np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * t)), 0.001)))
    starts.append(("gaussian", np.exp(-6.0 * (t - 0.5) ** 2 / 0.3)))
    starts.append(("sinc", np.abs(np.sinc(t - 0.5)) ** 2 + 0.01))
    starts.append(("random", rng.random(n) + 0.1))

    # Multi-frequency with random phases
    mf = np.ones(n)
    for f in [2, 5, 11, 23]:
        mf += 0.3 * np.cos(2.0 * np.pi * f * t + rng.random() * 2 * np.pi)
    starts.append(("multifreq", np.maximum(mf, 0.01)))

    # Palindromic version of parent
    if parent is not None:
        half = parent[:n // 2]
        if n % 2 == 0:
            pal = np.concatenate([half, half[::-1]])
        else:
            pal = np.concatenate([half, parent[n//2:n//2+1], half[::-1]])
        starts.append(("palindromic", adjust_sum(np.maximum(pal, 0), S)))

    return starts


def propose_candidate(seed=42, budget_s=2100):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    1. Adaptive LSE-FISTA as main optimizer (60% time)
    2. KKT-based polishing (20% time)
    3. Spectral refinement + diverse restarts (20% time)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 40.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 1024:
            parent = np.clip(parent, 0, 1000.0)
            if parent.sum() < 0.01:
                parent = None
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "len=%d" % len(parent) if parent is not None else "None"))

    global_best = np.inf
    global_a = None

    # Evaluate parent
    if parent is not None:
        p = adjust_sum(parent, S)
        c_p = fft_conv(p)
        sc_p = c_p.max()
        global_best = sc_p
        global_a = p.copy()
        print("  Parent score: %.7f" % sc_p)

    def try_update(a_cand, label):
        nonlocal global_best, global_a
        a_norm = adjust_sum(a_cand, S)
        c_chk = fft_conv(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] score=%.7f (elapsed %.0fs)" % (label, score, elapsed))
        return score

    # === Phase 1: Main FISTA optimization (55% of budget) ===
    print("\n=== Phase 1: FISTA Optimization ===")
    p1_time = (deadline - time.monotonic()) * 0.55

    if parent is not None:
        main_start = adjust_sum(parent, S)
    else:
        main_start = np.ones(n)

    a1, _ = run_fista_adaptive(main_start, n, S, p1_time, rng,
                                beta_start=30.0, beta_end=8000.0, ramp_frac=0.25)
    try_update(a1, "fista_main")

    # === Phase 2: KKT Polish (20% of budget) ===
    print("\n=== Phase 2: KKT Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        p2_time = remaining * 0.35
        a2, _ = run_kkt_polish(global_a, n, S, p2_time, rng)
        try_update(a2, "kkt_polish")

    # === Phase 3: Further FISTA from polished result (15%) ===
    print("\n=== Phase 3: Post-KKT FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        p3_time = remaining * 0.4
        a3, _ = run_fista_adaptive(global_a, n, S, p3_time, rng,
                                    beta_start=200.0, beta_end=15000.0, ramp_frac=0.2)
        try_update(a3, "fista_post_kkt")

    # === Phase 4: Spectral refinement + diverse starts (remaining) ===
    print("\n=== Phase 4: Spectral + Diverse ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_a is not None:
        # Spectral refinement
        p4_spec = remaining * 0.3
        if p4_spec > 15:
            a4, _ = run_spectral_refinement(global_a, n, S, p4_spec, rng)
            try_update(a4, "spectral")

        # Quick FISTA from best spectral result
        remaining = deadline - time.monotonic()
        if remaining > 20 and global_a is not None:
            p4_fista = remaining * 0.4
            a5, _ = run_fista_adaptive(global_a, n, S, p4_fista, rng,
                                        beta_start=500.0, beta_end=20000.0, ramp_frac=0.3)
            try_update(a5, "final_fista")

    # === Phase 5: Final KKT polish ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        a6, _ = run_kkt_polish(global_a, n, S, remaining * 0.8, rng)
        try_update(a6, "final_kkt")

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        print("  WARNING: No valid sequence found, using fallback")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("  Verification score: %.7f" % ver)
    else:
        print("  Verification failed (inf), using fallback")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Returned: length=%d, score=%.7f, time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
