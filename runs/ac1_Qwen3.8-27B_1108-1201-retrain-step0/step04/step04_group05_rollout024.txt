import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def gs_hio(a, N, S, n_iter, rng, hio_lambda=1.2):
    """GS/HIO algorithm for flat spectrum initialization."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        return a
    a = a * (S / s)
    
    target_mag = np.full(N, S / np.sqrt(N))
    
    best_a = a.copy()
    best_cmax = np.inf
    
    for it in range(n_iter):
        A = fft(a, N)
        phase = np.angle(A)
        
        # GS step: replace magnitude with target
        A_flat = target_mag * np.exp(1j * phase)
        a_gs = np.real(ifft(A_flat))[:n]
        
        # HIO feedback
        if it % 3 == 0:  # Do HIO every 3rd iteration
            a_in = np.maximum(a, 0)
            a_new = a_in + hio_lambda * (a_gs - a_in)
        else:
            a_new = a_gs
        
        # Non-negativity + rescale
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s_new)
        a = a_new
        
        if it % 50 == 0:
            c = compute_conv(a, N)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()
    
    c = compute_conv(a, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = a.copy()
    
    return best_a

def topk_descent(a, N, S, max_time, rng, k_init=200, k_min=1, lr_init=None):
    """Top-k subgradient descent with adaptive k."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a * (S / s)
    
    if lr_init is None:
        lr_init = 2.0 * S / (n * k_init)
    
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    # Adaptive learning rate with Nesterov-like momentum
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() - start_time < max_time:
        it += 1
        k = max(k_min, k_init - it // 10)
        
        c = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient via FFT
        w_full = np.zeros(N)
        for i, kx in enumerate(top_idx):
            w_full[int(kx)] = weights[i]
        W = fft(w_full)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adaptive step size
        lr = 2.0 * S / (n * max(k, 1) * (1 + 0.01 * it))
        
        # Nesterov momentum
        momentum = mu * momentum + grad
        step_dir = momentum
        
        a_new = project_simplex(a - lr * step_dir / max(gnorm, 1e-16), S)
        
        # Occasional perturbation
        if it % 500 == 0 and rng.random() < 0.3:
            a_new = project_simplex(a_new + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
        
        a = a_new
    
    return best_a, best_cmax

def lse_refine(a, N, S, max_time, rng, beta_init=20.0, beta_max=500.0):
    """LSE-based refinement (improved version)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a * (S / s)
    
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    # Simple gradient-based with line search
    while time.monotonic() - start_time < max_time:
        it += 1
        
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # LSE gradient
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        w_full = np.zeros(N)
        w_full[:len(w)] = w
        W = fft(w_full)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Line search
        lr = 2.0 * S / (n * max(beta, 1.0))
        best_lr = 0
        for ls in range(20):
            candidate = project_simplex(a - lr * grad / gnorm, S)
            c_ls = compute_conv(candidate, N)
            if c_ls.max() <= c_max:
                best_lr = lr
                a = candidate
                break
            lr *= 0.5
        
        if best_lr == 0:
            a = project_simplex(a - lr * grad / gnorm, S)
        
        # Beta ramping
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.1)
        
        # Perturbation
        if it % 800 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

def coord_descend_peaks(a, N, S, max_time, rng, n_rounds=100):
    """Coordinate descent targeting specific peaks."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start_time > max_time:
            break
        
        c = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()
        
        # Find the peak lag
        peak_lag = int(np.argmax(c))
        peak_val = c[peak_lag]
        
        # For lag k, b[k] = sum_i a[i]*a[i+k] (approximately)
        # To reduce b[k], we can try to reduce a[i] where a[i]*a[i+k] is large
        # and increase a[j] where it doesn't contribute to this peak
        
        # Simple approach: find the indices that contribute most to this peak
        # and try to reduce them
        if peak_lag == 0:
            # b[0] = sum(a^2), reduce by making a more uniform
            mean_a = a.sum() / n
            a = project_simplex(a - 0.01 * S / n * (a - mean_a), S)
        else:
            # For lag k > 0, b[k] = sum_{i=0}^{n-1-k} a[i] * a[i+k]
            # Find the pairs (i, i+k) with largest product
            products = a[:n - peak_lag] * a[peak_lag:n]
            # Reduce the largest contributors
            top_prod_idx = np.argsort(products)[::-1][:min(50, len(products)//10)]
            for idx in top_prod_idx:
                # Reduce a[idx] and a[idx+k] slightly
                reduction = 0.001 * S / n
                a[idx] = max(0, a[idx] - reduction)
                a[idx + peak_lag] = max(0, a[idx + peak_lag] - reduction)
            # Compensate by adding to random positions
            add_indices = rng.choice(n, size=len(top_prod_idx), replace=False)
            a[add_indices] += 2.0 * reduction
        
        a = project_simplex(a, S)
    
    return best_a, best_cmax

Minimize 2*n*max_conv/sum^2 using GS/HIO spectral initialization + adaptive top-k 
subgradient descent + LSE refinement.

Key improvements over previous approach:
1. GS/HIO (Gerchberg-Saxton / Hybrid Input-Output) for fast spectral initialization
2. Adaptive top-k subgradient descent with momentum for main optimization
3. Periodic spectral reshaping as global moves to escape local minima
4. LSE refinement with adaptive beta for final polishing
5. Better time allocation across phases

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def gs_hio_iter(a, N, S, n_iter, rng, hio_lambda=1.2, target_power=None):
    """GS/HIO algorithm for flat spectrum initialization.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    if target_power is None:
        target_mag = np.full(N, S / np.sqrt(N))
    else:
        target_mag = np.sqrt(target_power)
    
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    
    for it in range(n_iter):
        A = fft(a, N)
        phase = np.angle(A)
        
        # GS step: replace magnitude with target
        A_flat = target_mag * np.exp(1j * phase)
        a_gs = np.real(ifft(A_flat))[:n]
        
        # HIO feedback (every other iteration)
        if it % 2 == 1:
            a_in = np.maximum(a, 0.0)
            a_new = a_in + hio_lambda * (a_gs - a_in)
        else:
            a_new = a_gs
        
        # Non-negativity + rescale
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s_new)
        a = a_new
        
        if (it + 1) % 100 == 0:
            c = compute_conv(a, N)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()
    
    c = compute_conv(a, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = a.copy()
    
    return best_a, best_cmax


def topk_descent(a, N, S, max_time, rng, k_init=200, k_min=1, use_momentum=True):
    """Top-k subgradient descent with adaptive k and optional momentum.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0
    
    momentum = np.zeros(n) if use_momentum else None
    mu = 0.85
    
    while time.monotonic() - start_time < max_time:
        it += 1
        # Adaptive k: start large, decrease over time
        k = max(k_min, int(k_init * max(0.01, 1.0 - 0.002 * it)))
        
        c = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i in range(k):
            w_full[int(top_idx[i])] = weights[i]
        W = fft(w_full, N)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adaptive step size with backtracking line search
        lr = 2.0 * S / (n * max(k, 5) * (1.0 + 0.005 * it))
        
        if use_momentum:
            momentum = mu * momentum + grad
            step_dir = momentum
            dir_norm = np.linalg.norm(step_dir)
            if dir_norm < 1e-16:
                step_dir = grad
                dir_norm = gnorm
        else:
            step_dir = grad
            dir_norm = gnorm
        
        # Backtracking line search
        best_lr = lr
        a_old = a.copy()
        for ls in range(10):
            candidate = project_simplex(a - lr * step_dir / dir_norm, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= cur_max:
                a = candidate
                best_lr = lr
                break
            lr *= 0.5
            best_lr = lr
        else:
            a = project_simplex(a - best_lr * step_dir / dir_norm, S)
        
        # Periodic perturbation to escape local minima
        if it % 1000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            if use_momentum:
                momentum = np.zeros(n)
    
    return best_a, best_cmax


def lse_refine(a, N, S, max_time, rng, beta_init=20.0, beta_max=500.0):
    """LSE-smoothed gradient descent with backtracking.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # LSE gradient
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        w_full = np.zeros(2 * n - 1)
        w_full[:] = w
        W = fft(w_full, N)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        lr = 2.0 * S / (n * max(beta, 1.0))
        for ls in range(15):
            candidate = project_simplex(a - lr * grad / gnorm, S)
            c_ls = compute_conv(candidate, N)
            if c_ls.max() <= c_max:
                a = candidate
                break
            lr *= 0.5
        else:
            a = project_simplex(a - lr * grad / gnorm, S)
        
        # Beta ramping (slow increase for smooth exploration)
        if it % 50 == 0:
            beta = min(beta_max, beta * 1.05)
        
        # Periodic perturbation
        if it % 1000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax


def spectral_reshape(a, N, S, top_k, alpha, rng):
    """Spectral shaping: reduce power at frequencies that contribute to peak lags.
    Returns reshaped sequence."""
    n = len(a)
    c = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i in range(top_k):
        kx = int(top_idx[i])
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * kx / N)
    
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

"""
Minimize 2*n*max_conv/sum^2 using GS/HIO spectral initialization + adaptive top-k 
subgradient descent + LSE refinement + spectral shaping.

Key improvements over previous approach:
1. GS/HIO (Gerchberg-Saxton / Hybrid Input-Output) for fast spectral initialization
   - ~10x faster per iteration than L-BFGS, enabling more global exploration
2. Adaptive top-k subgradient descent with momentum for main optimization
3. Periodic spectral reshaping as global moves to escape local minima
4. LSE refinement with adaptive beta for final polishing
5. Asymmetric initializations to reduce center correlation
6. Better time allocation and stall detection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def gs_hio_iter(a, N, S, n_iter, rng, hio_lambda=1.2):
    """GS/HIO algorithm for flat spectrum initialization.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    target_mag = np.full(N, S / np.sqrt(N))

    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()

    for it in range(n_iter):
        A = fft(a, N)
        phase = np.angle(A)

        A_flat = target_mag * np.exp(1j * phase)
        a_gs = np.real(ifft(A_flat))[:n]

        if it % 2 == 1:
            a_in = np.maximum(a, 0.0)
            a_new = a_in + hio_lambda * (a_gs - a_in)
        else:
            a_new = a_gs

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s_new)
        a = a_new

        if (it + 1) % 100 == 0:
            c = compute_conv(a, N)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()

    c = compute_conv(a, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = a.copy()

    return best_a, best_cmax


def topk_descent(a, N, S, max_time, rng, k_init=200, k_min=1, use_momentum=True):
    """Top-k subgradient descent with adaptive k and optional momentum.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0

    momentum = np.zeros(n) if use_momentum else None
    mu = 0.85

    while time.monotonic() - start_time < max_time:
        it += 1
        k = max(k_min, int(k_init * max(0.01, 1.0 - 0.001 * it)))

        c = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i in range(k):
            w_full[int(top_idx[i])] = weights[i]
        W = fft(w_full, N)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        lr = 2.0 * S / (n * max(k, 5) * (1.0 + 0.003 * it))

        if use_momentum:
            momentum = mu * momentum + grad
            step_dir = momentum
            dir_norm = np.linalg.norm(step_dir)
            if dir_norm < 1e-16:
                step_dir = grad
                dir_norm = gnorm
        else:
            step_dir = grad
            dir_norm = gnorm

        a_updated = False
        for ls in range(12):
            candidate = project_simplex(a - lr * step_dir / dir_norm, S)
            c_ls = compute_conv(candidate, N)
            if c_ls.max() <= cur_max:
                a = candidate
                a_updated = True
                break
            lr *= 0.5
        if not a_updated:
            a = project_simplex(a - lr * step_dir / dir_norm, S)

        if it % 2000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            if use_momentum:
                momentum = np.zeros(n)

    return best_a, best_cmax


def lse_refine(a, N, S, max_time, rng, beta_init=20.0, beta_max=500.0):
    """LSE-smoothed gradient descent with backtracking.
    Returns (best_a, best_cmax)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        w_full = np.zeros(2 * n - 1)
        w_full[:] = w
        W = fft(w_full, N)
        Af = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        lr = 2.0 * S / (n * max(beta, 1.0))
        a_updated = False
        for ls in range(15):
            candidate = project_simplex(a - lr * grad / gnorm, S)
            c_ls = compute_conv(candidate, N)
            if c_ls.max() <= c_max:
                a = candidate
                a_updated = True
                break
            lr *= 0.5
        if not a_updated:
            a = project_simplex(a - lr * grad / gnorm, S)

        if it % 50 == 0:
            beta = min(beta_max, beta * 1.05)

        if it % 1500 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def spectral_reshape(a, N, S, top_k, alpha):
    """Spectral shaping: reduce power at frequencies contributing to peak lags."""
    n = len(a)
    c = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i in range(top_k):
        kx = int(top_idx[i])
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * kx / N)

    A = fft(a, N)
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def make_initializations(n, rng, N):
    """Create diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("rand", rng.random(n) + 0.01))

    a = 0.5 + 0.4 * np.sin(2.0 * np.pi * (7.0 * tt + tt * tt)) + 0.05 * rng.standard_normal(n)
    inits.append(("chirp", np.clip(a, 0.01, None)))

    a = 1.0
    for freq in [3, 7, 13, 21, 37, 53]:
        a += 0.12 * np.sin(2.0 * np.pi * freq * tt + rng.random() * 2.0 * np.pi)
    inits.append(("multifreq", np.clip(a, 0.01, None)))

    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2.0 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    inits.append(("fourier_pow", np.abs(a_full) + 0.01))

    phase = rng.uniform(0, 2.0 * np.pi, N)
    A = np.sqrt(float(n)) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    inits.append(("fourier_flat", np.abs(a_full) + 0.01))

    a = (rng.random(n) > 0.5).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    inits.append(("binary", a))

    a = np.linspace(0.1, 2.0, n) + 0.05 * rng.standard_normal(n)
    inits.append(("ramp", np.clip(a, 0.01, None)))

    a = np.exp(-((tt - 0.25) ** 2) / 0.01) + np.exp(-((tt - 0.75) ** 2) / 0.01)
    inits.append(("twohump", a + 0.01))

    a = np.exp(-((tt - 0.1) ** 2) / 0.005) + 0.3 * np.ones(n)
    inits.append(("frontload", a))

    return inits


def update_best(state, a_candidate, n, S, N, label=""):
    """Update the best solution in state dict."""
    a_cand = safe_normalize(a_candidate, n, S)
    c = compute_conv(a_cand, N)
    sc = float(c.max())
    if sc < state['best']:
        state['best'] = sc
        state['a'] = a_cand.copy()
        if label:
            print("  [%s] obj=%.6f" % (label, 2.0 * n * sc / (S * S)))
    return sc
