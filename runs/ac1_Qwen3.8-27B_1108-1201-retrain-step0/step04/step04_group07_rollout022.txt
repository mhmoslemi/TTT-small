"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using subgradient descent with
Nesterov acceleration, adaptive top-k, and periodic spectral reshaping.

Key improvements over previous approach:
1. Polyak-inspired step size using the theoretical lower bound 2n/(2n-1).
2. Nesterov acceleration for faster convergence.
3. Adaptive top-k that grows over time for smoother gradients.
4. Periodic spectral reshaping (geometric interpolation toward flat spectrum).
5. Momentum-based stepping with backtracking line search.
6. Multi-restart strategy with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_subgradient(a, top_idx, weights):
    """Compute the subgradient of max_k c_k at point a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w_val = weights[i]
        if k == 0:
            g += 2.0 * w_val * a
        else:
            if n - k > 0:
                g[:n - k] += w_val * a[k:]
                g[k:] += w_val * a[:n - k]
    return g


def spectral_reshape(a, strength, rng=None):
    """
    Reshape the spectrum toward flatness using geometric interpolation.
    Returns new a after projection.
    """
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    A = fft(a, fsize)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Target: flat spectrum with correct total power
    total_power = S * S  # sum of |A|^2 = S^2 by Parseval
    P_flat = np.full(fsize, total_power / fsize)

    # Geometric interpolation: P_new = P^(1-s) * P_flat^s
    # For bins where P is 0, use P_flat
    P_new = np.zeros_like(P)
    positive = P > 1e-30
    P_new[positive] = np.power(P[positive], 1.0 - strength) * np.power(P_flat[positive], strength)
    P_new[~positive] = P_flat[~positive] * strength

    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def subgradient_nesterov(a_init, n, S, budget_s, rng, top_k_init=5, top_k_max=100,
                         step_init=1.0, target_lb=None):
    """
    Subgradient descent with Nesterov acceleration.
    """
    if target_lb is None:
        target_lb = 2.0 * n / (2.0 * n - 1.0)

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    step = step_init
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    patience = 0
    best_at_patience = best_score

    fsize = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - start < budget_s:
        it += 1
        c = fft_convolve_self(y)
        f_val = c.max()
        obj = 2.0 * n * f_val / (S * S)
        if obj < best_score:
            best_score = obj
            best_local = y.copy()
            patience = 0
        else:
            patience += 1

        # Compute top-k
        k_cur = min(top_k_max, top_k_init + it // 20)
        top_idx = np.argsort(c)[::-1][:k_cur]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Compute subgradient
        g = compute_subgradient(y, top_idx, weights)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Step size: Polyak-inspired with damping
        gap = f_val - target_lb
        if gap > 0:
            polyak_step = gap * S / (gnorm * gnorm)
            polyak_step = min(polyak_step, 3.0 * S / gnorm)
            step = 0.5 * step + 0.5 * polyak_step
        else:
            step = step * 0.95

        # Nesterov update
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t_prev - 1.0) / t

        y_new = y + momentum * (y - a_prev)
        x_new = project_simplex(y_new - step * g / max(S, 1e-10), S)

        # Backtracking: check if we improved
        a_prev = a
        a = x_new

        # Periodic spectral reshape
        if it % 500 == 250 and rng.random() < 0.3:
            a = spectral_reshape(a, 0.3, rng)

        # Periodic noise injection to escape local minima
        if patience > 2000 and rng.random() < 0.2:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            a_prev = a.copy()
            y = a.copy()
            t = 1.0
            step = step_init
            patience = 0

        # Early termination if very close to lower bound
        if best_score <= target_lb * 1.001:
            break

    return best_local, best_score


def local_refinement(a_init, n, S, budget_s, rng):
    """
    Fine-grained local refinement using small subgradient steps.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_score = compute_score(a)
    best_local = a.copy()
    start = time.monotonic()
    target_lb = 2.0 * n / (2.0 * n - 1.0)

    while time.monotonic() - start < budget_s:
        c = fft_convolve_self(a)
        f_val = c.max()
        obj = 2.0 * n * f_val / (S * S)

        # Use all lags above 90th percentile for a smooth gradient
        threshold = np.percentile(c, 90)
        active_idx = np.where(c >= threshold)[0]
        active_vals = c[active_idx]
        total = active_vals.sum()
        if total < 1e-30:
            break
        weights = active_vals / total

        g = compute_subgradient(a, active_idx, weights)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Small adaptive step
        gap = f_val - target_lb
        if gap > 0:
            step = 0.1 * gap * S / (gnorm * gnorm)
            step = min(step, 0.5 * S / gnorm)
        else:
            step = 0.01 * S / max(gnorm, 1e-10)

        a_new = project_simplex(a - step * g / max(S, 1e-10), S)
        sc_new = compute_score(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_local = a_new.copy()
            a = a_new
        else:
            step *= 0.8

    return best_local, best_score


def make_structured_init(n, rng, seed_type="varying_cosine"):
    """Create a structured initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)

    if seed_type == "varying_cosine":
        # Sum of cosines with varying frequencies (inspired by Zadoff-Chu)
        a = np.ones(n)
        for harmonic in range(1, 16):
            freq = harmonic * (1 + 0.1 * rng.standard_normal())
            a += 0.15 * np.cos(2 * np.pi * freq * tt + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.0)
    elif seed_type == "weighted_binary":
        # Binary pattern with smooth weights
        b = (rng.random(n) > 0.5).astype(float)
        # Smooth it
        kernel_size = max(3, n // 128)
        kernel = np.ones(kernel_size) / kernel_size
        b_smooth = np.convolve(b, kernel, mode='same')
        a = 0.3 + 0.7 * b_smooth
    elif seed_type == "chirp_mix":
        # Mix of chirps
        a = np.ones(n) * 0.5
        a += 0.3 * np.sin(2 * np.pi * (3 * tt + 5 * tt * tt))
        a += 0.2 * np.sin(2 * np.pi * (7 * tt + 2 * tt * tt))
        a = np.maximum(a, 0.0)
    elif seed_type == "smooth_random":
        noise = rng.standard_normal(n)
        kernel_size = max(3, n // 64)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(noise, kernel, mode='same')
        smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
        a = 0.2 + 0.6 * smooth
    else:
        a = np.ones(n) * 0.5

    return a


def optimize_length(n, budget_s, rng, provided_seq=None):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        a_s = a.sum()
        if a_s < 0.01:
            return
        a_norm = a * (S / a_s)
        sc = compute_score(a_norm)
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()
        return sc

    # Build list of initializations
    inits = []
    if provided_seq is not None and len(provided_seq) == n:
        inits.append(("provided", np.clip(provided_seq, 0, 1000).astype(np.float64).copy()))

    inits.append(("structured1", make_structured_init(n, rng, "varying_cosine")))
    inits.append(("structured2", make_structured_init(n, rng, "weighted_binary")))
    inits.append(("structured3", make_structured_init(n, rng, "chirp_mix")))
    inits.append(("structured4", make_structured_init(n, rng, "smooth_random")))
    inits.append(("random", rng.random(n) * 0.8 + 0.1))

    n_inits = len(inits)
    t_start = time.monotonic()
    main_budget = budget_s * 0.7
    per_init = main_budget / n_inits

    for name, arr in inits:
        elapsed = time.monotonic() - t_start
        remaining = main_budget - elapsed
        if remaining < 5:
            break

        a = np.clip(arr, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        # Quick score check
        sc = try_sequence(a)

        # Phase 1: Subgradient with Nesterov (60% of per-init budget)
        rem1 = min(per_init * 0.6, remaining - 5)
        if rem1 > 2:
            a1, sc1 = subgradient_nesterov(a, n, S, rem1, rng,
                                           top_k_init=5, top_k_max=80,
                                           step_init=1.0)
            sc = try_sequence(a1)

        # Phase 2: Local refinement (30%)
        rem2 = min(per_init * 0.3, remaining - 5)
        if rem2 > 2 and global_best < 2.0:
            a2, sc2 = local_refinement(a1 if 'a1' in dir() else a, n, S, rem2, rng)
            sc = try_sequence(a2)

        # Phase 3: Spectral reshape + subgradient (10%)
        rem3 = min(per_init * 0.1, remaining - 5)
        if rem3 > 2 and global_best < 2.0:
            a_spec = spectral_reshape(a1 if 'a2' in dir() and 'a2' is not None else a, 0.2, rng)
            a3, sc3 = subgradient_nesterov(a_spec, n, S, rem3, rng,
                                           top_k_init=20, top_k_max=80,
                                           step_init=0.5)
            sc = try_sequence(a3)

        elapsed = time.monotonic() - t_start
        print("  [%s n=%d] score=%.6f (elapsed %.0fs/%.0fs)" %
              (name, n, global_best, elapsed, budget_s))

    # Final polish
    if global_a is not None:
        final_budget = max(0, budget_s - (time.monotonic() - t_start))
        if final_budget > 5:
            a_f, sc_f = local_refinement(global_a, n, S, final_budget * 0.7, rng)
            try_sequence(a_f)

            rem = max(0, budget_s - (time.monotonic() - t_start))
            if rem > 3 and global_best < 2.0:
                a_f2, _ = subgradient_nesterov(global_a, n, S, rem, rng,
                                               top_k_init=30, top_k_max=100,
                                               step_init=0.3)
                try_sequence(a_f2)

    return global_a, global_best
