"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient
with log-sum-exp smoothing and multi-restart coevolution.

Key improvements over previous approach:
1. Log-sum-exp smoothed gradient for smoother optimization landscape.
2. Adaptive active set: grow top_k as we converge.
3. Barzilai-Borwein adaptive step size.
4. Multi-restart with structured perturbations.
5. Better spectral initialization with multiple rounds.

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_convolution(a, N):
    """Compute full self-convolution via FFT."""
    n = len(a)
    A = fft(a, N)
    C = np.real(ifft(A * A))
    return C[:2*n-1]

def compute_score_from_conv(C, n):
    """Compute objective from convolution and sequence length."""
    S2 = np.sum(C)  # = (sum(a))^2
    if S2 < 1e-10:
        return np.inf, C.max()
    return 2.0 * n * C.max() / S2, C.max()

def compute_score(a):
    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    C = compute_convolution(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * C.max() / (S * S))

def lse_smoothed_gradient(a, N, temperature):
    """
    Compute the gradient of the log-sum-exp smoothed max(C) with respect to a.
    Returns (gradient, C, lse_value).
    """
    n = len(a)
    C = compute_convolution(a, N)
    
    # Log-sum-exp: LSE = t * log(sum(exp(C/t)))
    C_max = C.max()
    shifted = (C - C_max) / temperature
    exp_shifted = np.exp(shifted)
    log_sum = C_max / temperature + np.log(exp_shifted.sum())
    lse_value = temperature * log_sum
    
    # Weights: w[k] = exp(C[k]/t) / sum(exp(C[k']/t)) = exp_shifted[k] / sum
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of LSE w.r.t. a[i]:
    # dLSE/da[i] = sum_k (C[k]/t) * exp(C[k]/t) / sum(exp(C[k']/t)) * dC[k]/da[i]
    # = sum_k w[k] * (C[k]/t) * dC[k]/da[i]
    # 
    # dC[k]/da[i] = a[k-i] + a[i-k] (for full self-convolution)
    # More precisely: dC[k]/da[i] = a[max(0,k-i)] if 0<=k-i<n else 0
    #                             + a[max(0,i-k)] if 0<=i-k<n else 0
    # But since C[k] = sum_j a[j]*a[k-j], we have:
    # dC[k]/da[i] = a[k-i] (from j=i term) + a[k-i] (from k-j=i, i.e., j=k-i term)
    # = 2*a[k-i] if i != k-i, or a[i] if i = k-i
    # For simplicity (the diagonal case is measure zero): dC[k]/da[i] = 2*a[k-i]
    #
    # So grad[i] = sum_k w[k] * (C[k]/t) * 2*a[k-i]
    # = 2 * sum_k (w[k] * C[k]/t) * a[k-i]
    # = 2 * cross_correlation(w * C/t, a)
    
    # Cross-correlation via FFT:
    # cross[i] = sum_k v[k] * a[k-i] where v[k] = w[k] * C[k] / t
    v = w * C / temperature
    # cross[i] = sum_k v[k] * a_padded[k-i]
    # = IFFT(conj(FFT(v_padded)) * FFT(a_padded))[i]  ... need to be careful
    
    # Actually, let's use a different formulation:
    # cross[i] = sum_{j=0}^{n-1} v[i+j] * a[j]  (where v is defined for 0..2n-2)
    # This is a valid inner product for each i.
    # In terms of convolution: cross = (flip(v) * a)[2n-2 - i] ... 
    
    # Simplest FFT approach:
    # Define v_full of length 2n-1, a_pad of length 2n-1 (a zero-padded)
    # cross[i] = sum_k v_full[k] * a_pad[k-i] 
    # This is the correlation, which in FFT is: IFFT(conj(FFT(v_full)) * FFT(a_pad))
    # But we need to be careful about the length.
    
    # Let's use size N (which is >= 2n-1):
    v_N = np.zeros(N)
    v_N[:2*n-1] = v
    a_N = np.zeros(N)
    a_N[:n] = a
    
    V = fft(v_N)
    A = fft(a_N)
    cross = np.real(ifft(np.conj(V) * A))[:n]
    
    grad = 2.0 * cross
    return grad, C, lse_value

def spectral_flatten(a, N, n_steps, alpha):
    """Spectral flattening step."""
    n = len(a)
    S = a.sum()
    for _ in range(n_steps):
        A = fft(a, N)
        P = np.abs(A) ** 2
        p_median = np.median(P)
        if p_median < 1e-30:
            break
        ratio = P / p_median
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * (1.0 - alpha * (1.0 - np.power(ratio[above], -alpha)))
        P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - np.power(ratio[~above], alpha)))
        P_new = np.maximum(P_new, 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
    return a

def adaptive_subgradient(a, N, n_steps, top_k_init, top_k_max, rng, target=1.50):
    """
    Adaptive subgradient method with growing active set.
    """
    n = len(a)
    S = a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    # Nesterov state
    y = a.copy()
    t = 1.0
    step_scale = 1.0
    
    for it in range(n_steps):
        # Adaptive top_k: grow from top_k_init to top_k_max
        progress = it / max(n_steps - 1, 1)
        top_k = int(top_k_init + (top_k_max - top_k_init) * progress)
        top_k = min(top_k, 2*n - 1)
        
        C = compute_convolution(y, N)
        cur_max = C.max()
        score = 2.0 * n * cur_max / (y.sum() ** 2)
        
        if score < best_score:
            best_score = score
            best = y.copy()
        
        if best_score <= target:
            break
        
        # Compute subgradient from top-k lags
        top_idx = np.argsort(C)[::-1][:top_k]
        top_vals = C[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient via FFT
        w_full = np.zeros(N)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, N)
        A = fft(y, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size (Barzilai-Borwein-like)
        step = step_scale / (1.0 + 0.1 * np.sqrt(it + 1))
        
        a_new = project_simplex(y - step * grad / gnorm, S)
        a_new = np.maximum(a_new, 0.0)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - y)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        t = t_new
        
        # Periodic noise
        if it % 200 == 100 and rng.random() < 0.3:
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
        
        # Adaptive step scaling
        if it % 100 == 50:
            local_score = compute_score(y)
            if local_score > best_score * 1.01:
                step_scale *= 0.8
            else:
                step_scale = min(step_scale * 1.05, 2.0)
    
    return best, best_score

def lse_gradient_descent(a, N, n_steps, t_init, t_final, rng, target=1.50):
    """
    Log-sum-exp smoothed gradient descent with annealing temperature.
    """
    n = len(a)
    S = a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    # Nesterov state
    y = a.copy()
    x = a.copy()
    t = 1.0
    
    for it in range(n_steps):
        progress = it / max(n_steps - 1, 1)
        # Exponential temperature annealing
        temperature = t_init * (t_final / t_init) ** progress
        
        grad, C, lse_val = lse_smoothed_gradient(y, N, temperature)
        
        score = 2.0 * n * C.max() / (y.sum() ** 2)
        if score < best_score:
            best_score = score
            best = y.copy()
        
        if best_score <= target:
            break
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: adapt based on temperature
        # When temperature is high, the gradient is smoother, so we can take larger steps
        step = temperature / (n * gnorm + 1e-10)
        step = min(step, 1.0 / np.sqrt(n))
        
        x_new = project_simplex(y - step * grad, S)
        x_new = np.maximum(x_new, 0.0)
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        t = t_new
        
        # Periodic noise
        if it % 500 == 250 and rng.random() < 0.2:
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best, best_score
