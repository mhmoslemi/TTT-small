def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)

    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:

    theta = (cumsum[rho - 1] - S) / rho

The aliasing would affect (w ⋆_N a)[j] by adding (w ⋆ a)[j+N], (w ⋆ a)[j+2N], etc. Since (w ⋆ a) has support [0, 3n-3], the aliased terms are (w ⋆ a)[j+N] which is non-zero only if j+N <= 3n-3, i.e., j <= 3n-3-N. For N = 2n-1 (next power of 2 >= 2n-1, which is typically 2n or 4n depending on n), we have j <= 3n-3-(2n-1) = n-2. So for j = n-2 and j = n-1, there might be aliasing if N = 2n-1.

To be safe, let's use N >= 3n-2 (next power of 2). For n=8192, that's N=32768.

Hmm, but the previous code used `fsize = 1 << (len(w) + n - 1).bit_length()` which for w of length 2n-1 gives `1 << (2n-1+n-1).bit_length() = 1 << (3n-2).bit_length()`. For n=8192: 3n-2 = 24574, bit_length = 15, so fsize = 32768. That's correct!

OK so the gradient formula in the previous code is correct. Let me now focus on the algorithm.

Given the correct gradient formula, the FISTA algorithm should work. The key question is the step size.

The Lipschitz constant of the gradient:
The gradient is grad(a) = 2 * (w ⋆ a) where w depends on a (through c = conv(a,a)).

The Jacobian of grad with respect to a is:
J[i,j] = ∂grad[j]/∂a[i] = 2 * (∂(w ⋆ a)[j]/∂a[i])

= 2 * (w[j-i] * 1[0<=j-i<2n-1] + sum_k (∂w[k]/∂a[i]) * a[k-j] * 1[0<=k-j<n])

The first term is the "linear" part (treating w as constant): 2 * w[j-i]
The second term involves ∂w/∂a, which is more complex.

For a rough bound: ||J|| <= 2 * ||w||_1 + 2 * ||∂w/∂a|| * ||a||

Since w is a probability vector, ||w||_1 = 1.
||∂w/∂a|| is related to beta (the LSE parameter) and the spread of c.

A rough bound: L ≈ 2 + 2 * beta * max(c) * ||a||_∞ (very rough)

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using adaptive LSE-FISTA
with correct gradient (cross-correlation based), adaptive step size via
backtracking, and multi-phase optimization. Key improvements: correct
Lipschitz estimation, backtracking line search, and peak-equalization
polishing phase.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """LSE smooth approximation of max(c)."""
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    return float(cm + np.log(np.exp(scaled).sum()) / beta)


def lse_gradient(a, c, beta, fsize_grad):
    """
    Gradient of LSE_max(conv(a,a)) wrt a.
    Uses cross-correlation: grad[j] = 2 * sum_k w[k] * a[k-j]
    """
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize_grad)
    A = fft(a, fsize_grad)
    return 2.0 * np.real(ifft(np.conj(W) * A))[:n]


def grad_lipschitz(a, c, beta):
    """Estimate Lipschitz constant of the LSE gradient."""
    n = len(a)
    a_max = a.max()
    # Conservative bound: L ~ 2 + 2*beta*max|c-var(c)|*a_max
    # Simpler: use 2*beta*max(a) + 2*sum(a)/n as rough estimate
    return 2.0 * beta * a_max + 2.0 * a.sum() / max(n, 1)


def fista_main(a_init, S, max_time, rng, beta_start=20.0, beta_end=5000.0,
               beta_ramp_iters=8000):
    """
    FISTA with adaptive LSE smoothing and backtracking line search.
    Returns (best_a, best_objective).
    """
    n = len(a_init)
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_grad = 1 << (3 * n - 2).bit_length()

    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    y = a.copy()
    t_kk = 1.0
    it = 0
    start_time = time.monotonic()

    c = fft_convolve_self(a)
    best_maxb = c.max()
    best_a = a.copy()
    last_improve = 0
    step = 1.0
    consecutive_increases = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta
        progress = min(1.0, it / max(beta_ramp_iters, 1))
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, 8000.0)

        # Compute current state
        c = fft_convolve_self(y)
        f_curr = lse_value(c, beta)
        grad = lse_gradient(y, c, beta, fsize_grad)
        gnorm = np.linalg.norm(grad)

        # Track best true objective
        if c.max() < best_maxb - 1e-12:
            best_maxb = c.max()
            best_a = y.copy()
            last_improve = it

        if gnorm < 1e-14:
            break

        # Adaptive step size with backtracking
        # Initial step from Lipschitz estimate
        L_est = grad_lipschitz(y, c, beta)
        step_candidate = 1.0 / max(L_est, 1e-10)
        # Use running step size (adapted from previous iterations)
        if it <= 10:
            step = step_candidate
        else:
            step = min(step * 1.02, step_candidate * 2.0)  # slowly increase
            step = max(step, step_candidate * 0.3)  # don't go too far

        # Backtracking line search (limited to 6 iterations)
        f_trial = f_curr
        x_trial = project_simplex(y - step * grad, S)
        for bt in range(6):
            c_trial = fft_convolve_self(x_trial)
            f_trial = lse_value(c_trial, beta)
            if f_trial <= f_curr + 1e-12:
                break
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            if step < 1e-10:
                break

        if f_trial > f_curr:
            consecutive_increases += 1
            # Reduce step more aggressively
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
        else:
            consecutive_increases = 0

        x_new = x_trial

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_kk * t_kk))
        y = x_new + ((t_kk - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_kk = t_new

        # Restart momentum periodically or when stuck
        if consecutive_increases > 5 or (it - last_improve > 1500):
            t_kk = 1.0
            # Small perturbation to escape local minima
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            consecutive_increases = 0
            last_improve = it

    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best_maxb:
        best_maxb = c_final.max()
        best_a = a.copy()

    obj = 2.0 * n * best_maxb / (S * S)
    return best_a, obj


def peak_equalize(a, S, n_peaks=5, n_iters=20000, rng=None):
    """
    Direct peak equalization: iteratively reduce the top peaks by
    transferring mass from high-contribution to low-contribution positions.
    """
    n = len(a)
    fsize_conv = 1 << (2 * n - 1).bit_length()
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_maxb = c.max()
    best_a = a.copy()

    for it in range(n_iters):
        c = fft_convolve_self(a)
        if c.max() < best_maxb - 1e-12:
            best_maxb = c.max()
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        threshold = c.mean()

        # Compute combined gradient direction
        grad = np.zeros(n)
        for k in top_idx:
            w = max(c[k] - threshold, 0.0)
            if w <= 0:
                continue
            # grad contribution from lag k: 2 * a[k-j] for valid j
            # i.e., for each j, contribution is 2*w*a[k-j] if 0<=k-j<n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                # grad[j] += 2 * w * a[k-j] for j in [lo, hi]
                # k-j ranges from k-lo to k-hi, i.e., from min(n-1,k) to max(0,k-n+1)
                js = np.arange(lo, hi + 1)
                a_vals = a[k - js]
                grad[js] += 2.0 * w * a_vals

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: try to reduce the peak by a fixed fraction
        # b[k*](t) ≈ c[k*] - 2*t*w*(avg contribution) + ...
        # Use a small step proportional to the excess
        max_excess = c[top_idx[0]] - threshold
        if max_excess < 1e-10:
            break
        step = 0.3 * max_excess / max(gnorm * gnorm / max(n, 1), 1e-10)
        step = min(step, 1.0 / max(gnorm, 1e-10) * 0.5)

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() >= c.max() * (1 + 1e-10):
            step *= 0.5
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() >= c.max():
                break
        a = x_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best_maxb:
        best_maxb = c_final.max()
        best_a = a.copy()

    obj = 2.0 * n * best_maxb / (S * S)
    return best_a, obj


def fgl_exact(a, S, max_time, rng, K=3):
    """
    Exact FGL: at each step, target the top-K peaks with exact subgradient
    and adaptive step size.
    """
    n = len(a)
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_maxb = c.max()
    best_a = a.copy()
    start_time = time.monotonic()
    it = 0
    y = a.copy()
    t_kk = 1.0
    step = 0.1
    stall = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = fft_convolve_self(y)
        if c.max() < best_maxb - 1e-12:
            best_maxb = c.max()
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Top-K peaks
        top_idx = np.argsort(c)[::-1][:K]
        threshold = c.mean() + 0.1 * (c.max() - c.mean())

        grad = np.zeros(n)
        for k in top_idx:
            w = max(c[k] - threshold, 0.0)
            if w <= 0:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                grad[js] += 2.0 * w * a[k - js]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        x_new = project_simplex(y - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() > c.max() * (1 + 1e-8):
            step *= 0.7
            x_new = project_simplex(y - step * grad, S)
        else:
            step = min(step * 1.05, 0.5)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_kk * t_kk))
        y = x_new + ((t_kk - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_kk = t_new

        if stall > 2000:
            t_kk = 1.0
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            stall = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best_maxb:
        best_maxb = c_final.max()
        best_a = a.copy()
    obj = 2.0 * n * best_maxb / (S * S)
    return best_a, obj
