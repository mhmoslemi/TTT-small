```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def solve_lp_direction(f, M):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    size = 2 * n - 1
    
    # Build the constraint matrix C (size x n) where C[k,j] = f[k-j] if 0 <= k-j < n
    # Use sparse format
    rows = []
    cols = []
    vals = []
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    C = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    # LP: max 1^T g s.t. C g <= M, g >= 0
    # In scipy form: min -1^T g s.t. C g <= M, g >= 0
    
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n)  # Fallback
```

   For coordinate j: (a*a)_k = a_j * a_{k-j} + C_k (where C_k = sum_{i != j} a_i a_{k-i}).
   sum(a) = a_j + S' (where S' = sum_{i != j} a_i).
   
   f_j(a_j) = max_k [a_j * a_{k-j} + C_k] / (a_j + S')^2
   
   This is a 1D function of a_j. We can minimize it with golden section search. Each evaluation is O(n) (compute the max over k). With 50 evaluations, that's O(50n) per coordinate. For n coordinates, total is O(50n^2) per full sweep.

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Projected gradient with exact 1D line search (minimax of quadratics),
cycling through gradient types (top-1, top-K, LSE), with periodic LP-inspired
direction finding and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def topk_subgrad(p, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(p)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * p[j - i_vals]
    return grad / k


def lse_subgrad(p, c, beta):
    """LSE-smoothed gradient of max."""
    n = len(p)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    P = fft(p, fsize)
    return 2.0 * np.real(ifft(W * np.conj(P)))[:n]


def line_search_quadratic(pp, pd, dd, t_max=1.0, n_evals=40):
    """
    Minimize max_k [pp[k] + 2*t*pd[k] + t^2*dd[k]] over t in [0, t_max].
    Uses golden section search.
    """
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    a = 0.0
    b = t_max
    
    def f(t):
        return np.max(pp + 2.0 * t * pd + t * t * dd)
    
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_evals):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = f(d)
    
    return 0.5 * (a + b)
