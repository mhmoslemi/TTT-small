"""

Multi-phase optimization: spectral flattening init, Nesterov-accelerated
subgradient with adaptive active set, peak-aware coordinate descent,
and local search with diverse restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    condition = w > (cumsum - S) / idx
    if np.any(condition):
        rho = int(np.max(np.where(condition)[0])) + 1
    else:
        rho = n
    tau = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - tau, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def grad_bk(a, k):
    """Compute gradient of b_k = sum_i a_i * a_{k-i} w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        js = np.arange(lo, hi + 1)
        g[js] += a[k - js]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        js2 = np.arange(lo2, hi2 + 1)
        g[js2] += a[js2 - k]
    return g


def active_set_gradient(a, c, n_active=8, gap_frac=0.05):
    """Compute weighted subgradient using top n_active lags."""
    n = len(a)
    M = c.max()
    threshold = M - gap_frac * M
    active_idx = np.where(c >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c)])
    if len(active_idx) > n_active:
        top_k = np.argsort(c[active_idx])[::-1][:n_active]
        active_idx = active_idx[top_k]
    weights = c[active_idx] - c + M  # weights proportional to how close to max
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, active_idx):
        g += w * grad_bk(a, int(k))
    return g


def lse_subgradient(a, c, beta, n):
    """Subgradient of LSE approximation to max(c)."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def spectral_flatten(a_init, n, S, max_time, rng, n_alphas=8):
    """Spectral flattening with multiple alpha values, protected DC."""
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()

    alphas = np.linspace(0.2, 0.9, n_alphas)
    it = 0
    current_alpha_idx = 0

    while time.monotonic() - start < max_time:
        it += 1
        alpha = alphas[current_alpha_idx % n_alphas]

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Protected flattening: don't reduce DC too much
        P_work = P.copy()
        P_dc = P_work[0]
        P_work = P_work[1:]
        P_mean_work = P_work.mean()

        # Power compression
        P_compressed = np.power(np.maximum(P_work, 1e-30), 1.0 - alpha)
        P_compressed = P_compressed / P_compressed.mean()

        # Blend
        P_new_work = (1.0 - alpha * 0.5) * P_work / P_mean_work + alpha * 0.5 * P_compressed
        # Restore DC (protect the sum)
        P_new = np.concatenate([[P_dc], P_new_work])
        # Scale to preserve total energy
        P_new = P_new * (P.sum() / max(P_new.sum(), 1e-30))

        mag_new = np.sqrt(P_new)
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_raw, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
            a = a_new
            # Stay at this alpha longer if improving
            current_alpha_idx = current_alpha_idx % n_alphas
        else:
            # Try next alpha
            current_alpha_idx += 1
            # Still update a but with damping
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * (S / s)

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fista_optimize(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=3000.0):
    """Nesterov-accelerated optimization with hybrid LSE/subgradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    c = fft_convolve_self(y)
    best = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        if it % 100 == 0 and time.monotonic() - start > max_time - 2:
            break

        if it < 500:
            beta = beta_init + (beta_max - beta_init) * (it / 500.0)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient: hybrid LSE + exact subgradient
        use_exact = (it > 200 and rng.random() < min(0.4, 0.001 * it))
        if use_exact:
            grad = active_set_gradient(y, c, n_active=6, gap_frac=0.03)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for subgradient: use a conservative rule
            step = S / (gnorm * (1.0 + 0.02 * np.log1p(it / 100.0)))
        else:
            grad, _ = lse_subgradient(y, c, beta, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            if it - last_improve > 800:
                step *= 0.7

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * (S / sy)
        a = x_new
        t_mom = t_new

        # Periodic noise for escaping local minima
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

        # Restart if stuck
        if it - last_improve > 3000:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coord_descent(a_init, n, S, max_time, rng):
    """Peak-aware coordinate descent with smart pair selection."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0

    # Identify peak lags for smart pair selection
    while time.monotonic() - start < max_time:
        it += 1
        if it % 50 == 0 and time.monotonic() - start > max_time - 3:
            break

        improved = False
        M = c.max()
        # Get top peaks
        top_k = np.argsort(c)[::-1][:10]

        n_try = min(80 + it, 300)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break

            # Smart pair selection: j from peak contributors, m from low contributors
            k_peak = int(top_k[rng.integers(len(top_k))])

            # Find j that contributes most to b_{k_peak}
            contrib_j = np.array([a[k_peak - j] if 0 <= k_peak - j < n else 0.0 +
                                  a[j - k_peak] if 0 <= j - k_peak < n else 0.0
                                  for j in range(n)])
            # Find m that contributes least
            total_contrib = np.zeros(n)
            for kp in top_k[:5]:
                kp = int(kp)
                for j in range(n):
                    cj = (a[kp - j] if 0 <= kp - j < n else 0.0) + \
                          (a[j - kp] if 0 <= j - kp < n else 0.0)
                    total_contrib[j] += cj * (c[kp] / c.sum())

            # Sample j from high contributors, m from low contributors
            if rng.random() < 0.7:
                j_candidates = np.argsort(contrib_j)[::-1][:max(5, n // 100)]
                m_candidates = np.argsort(total_contrib)[:max(5, n // 200)]
                j = int(j_candidates[rng.integers(len(j_candidates))])
                m = int(m_candidates[rng.integers(len(m_candidates))])
            else:
                j = int(rng.integers(n))
                m = int(rng.integers(n))

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute d_k = change in b_k per unit delta
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                vj = (a[k - j] if 0 <= k - j < n else 0.0) + \
                     (a[j - k] if 0 <= j - k < n else 0.0)
                vm = (a[k - m] if 0 <= k - m < n else 0.0) + \
                     (a[m - k] if 0 <= m - k < n else 0.0)
                d[k] = vj - vm

            # Golden section search for optimal delta
            lo = -a[j]
            hi = a[m]
            if hi - lo < 1e-15:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = lo, hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            # Evaluate max(c + delta*d) for candidate deltas
            fc = (c + c_g * d).max()
            fd = (c + d_g * d).max()

            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = (c + c_g * d).max()
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = (c + d_g * d).max()

            delta = (a_g + b_g) / 2.0
            delta = max(lo, min(hi, delta))

            if abs(delta) < 1e-15:
                continue

            # Check improvement
            c_trial = c + delta * d
            cur_trial = c_trial.max()
            if cur_trial < best - 1e-14:
                a[j] += delta
                a[m] -= delta
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True
                # Update top peaks
                M = best
                top_k = np.argsort(c)[::-1][:10]

        # Recompute c periodically to avoid drift
        if it % 30 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        if not improved and it > 10:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def local_search(a_init, n, S, max_time, rng):
    """Local search with random perturbations and re-optimization."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        # Try different perturbation scales
        scale = 0.005 * S / np.sqrt(n) * (0.5 + rng.random())
        a_trial = project_simplex(a + rng.normal(0, scale, n), S)
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < best:
            best = c_trial.max()
            best_a = a_trial.copy()
            a = a_trial.copy()
        # Also try structured perturbations
        if rng.random() < 0.3:
            # Perturb a random block
            block_size = int(n * rng.uniform(0.01, 0.1))
            start_idx = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            a_trial[start_idx:start_idx + block_size] *= (0.5 + rng.random())
            a_trial = project_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best:
                best = c_trial.max()
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best


def make_initializations(n, S, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * S / n))

    # Gaussian
    inits.append(("gaussian", normalize_sum(np.exp(-8.0 * (tt - 0.5) ** 2), n)))

    # Raised cosine (Tukey) window
    r = 0.3
    wk = np.ones(n)
    n_edge = int(r * n / 2)
    if n_edge > 0:
        edge = 0.5 * (1 + np.cos(np.pi * np.arange(n_edge) / n_edge))
        wk[:n_edge] = edge
        wk[-n_edge:] = edge[::-1]
    inits.append(("tukey", normalize_sum(wk, n)))

    # Sinc-like (clipped)
    inits.append(("sinc", normalize_sum(np.maximum(np.sinc(tt - 0.5) ** 2, 0.05), n)))

    # Cosine-tapered
    inits.append(("cos_taper", normalize_sum(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt - np.pi)), 0.01), n)))

    # Flat spectrum init: inverse FFT of constant magnitude
    fsize = 1 << (2 * n - 1).bit_length()
    A_flat = np.ones(fsize, dtype=np.complex128)
    # Add some randomness to break symmetry
    A_flat *= (1.0 + 0.1 * rng.random(fsize))
    A_flat[n // 2] = A_flat[n // 2].real  # Make it even
    a_flat = np.real(ifft(A_flat))[:n]
    inits.append(("flat_spec", normalize_sum(a_flat, n)))

    # Weighted random
    a_rand = np.abs(rng.standard_normal(n)) ** 0.5
    inits.append(("rand_sqrt", normalize_sum(a_rand, n)))

    # Exponential decay from center
    center = n // 2
    dist = np.abs(np.arange(n) - center)
    inits.append(("exp_center", normalize_sum(np.exp(-0.5 * dist / (n * 0.1)), n)))

    return inits


def propose_candidate(seed=42, budget_s=2100):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 256 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, budget=%.0fs, parent=%s" % (n, budget_s, "yes" if parent is not None else "no"))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] score=%.6f (elapsed %.0fs)" % (label, score, elapsed))
        return score

    # Build initializations
    if parent is not None:
        parent_norm = normalize_sum(parent, n)
        inits = [("parent", parent_norm)]
    else:
        inits = []

    for name, arr in make_initializations(n, S, rng):
        inits.append((name, arr))

    print("Generated %d initializations" % len(inits))

    # Phase 1: Spectral flattening on each init (quick pass)
    print("\n=== Phase 1: Spectral Flattening ===")
    p1_total = (deadline - time.monotonic()) * 0.25
    t_per_init = p1_total / len(inits)
    spect_results = []
    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - t_start) * 0.6 - 60:
            break
        t_alloc = min(t_per_init, deadline - time.monotonic() - 120)
        if t_alloc < 5:
            break
        a_sf, score = spectral_flatten(arr, n, S, t_alloc, rng)
        spect_results.append((score, name, a_sf))
        print("  SF %s: %.6f" % (name, score))

    # Sort by score
    spect_results.sort(key=lambda x: x[0])
    print("Best after spectral: %.6f (%s)" % (spect_results[0][0], spect_results[0][1]))

    # Update global best from spectral results
    for score, name, a_sf in spect_results[:3]:
        update_best(a_sf, "sf_" + name)

    # Phase 2: FISTA on top candidates
    print("\n=== Phase 2: FISTA Optimization ===")
    p2_total = (deadline - time.monotonic()) * 0.35
    n_fista = min(3, len(spect_results))
    # Weight first candidate more
    fista_weights = np.array([0.45, 0.30, 0.25][:n_fista])
    fista_weights /= fista_weights.sum()

    fista_results = []
    for i in range(n_fista):
        if time.monotonic() > deadline - (deadline - t_start) * 0.3 - 60:
            break
        t_alloc = p2_total * fista_weights[i]
        if t_alloc < 20:
            break
        score, name, a_start = spect_results[i]
        a_opt, score_opt = fista_optimize(a_start, n, S, t_alloc, rng,
                                           beta_init=30.0, beta_max=2000.0)
        fista_results.append((score_opt, name, a_opt))
        update_best(a_opt, "fista_" + name)

    fista_results.sort(key=lambda x: x[0])
    print("Best after FISTA: %.6f" % fista_results[0][0])

    # Phase 3: Coordinate descent polish
    print("\n=== Phase 3: Coordinate Descent ===")
    p3_total = (deadline - time.monotonic()) * 0.25
    if p3_total > 20:
        # CD on best FISTA result
        best_fista_score, best_fista_name, best_fista_a = fista_results[0]
        a_cd, score_cd = coord_descent(best_fista_a, n, S, p3_total * 0.6, rng)
        update_best(a_cd, "cd_best")

        # CD on second best if time
        remaining_p3 = p3_total * 0.4
        if len(fista_results) > 1 and remaining_p3 > 15:
            a_cd2, score_cd2 = coord_descent(fista_results[1][2], n, S, remaining_p3, rng)
            update_best(a_cd2, "cd_second")

    # Phase 4: Final polish - combined FISTA + CD on global best
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Alternate: short FISTA -> CD -> FISTA -> CD
            half_time = remaining / 2.0
            a_f1, _ = fista_optimize(global_a, n, S, half_time * 0.5, rng,
                                      beta_init=200.0, beta_max=5000.0)
            update_best(a_f1, "final_fista1")

            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_c1, _ = coord_descent(global_a, n, S, remaining2 * 0.7, rng)
                update_best(a_c1, "final_cd1")

            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                a_f2, _ = fista_optimize(global_a, n, S, remaining3 * 0.6, rng,
                                          beta_init=500.0, beta_max=8000.0)
                update_best(a_f2, "final_fista2")

            remaining4 = deadline - time.monotonic()
            if remaining4 > 8:
                a_c2, _ = coord_descent(global_a, n, S, remaining4 * 0.8, rng)
                update_best(a_c2, "final_cd2")

    # Phase 5: Local search
    print("\n=== Phase 5: Local Search ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_ls, _ = local_search(global_a, n, S, remaining * 0.7, rng)
            update_best(a_ls, "local_search")
            # Quick FISTA after local search
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_fs, _ = fista_optimize(global_a, n, S, remaining2 * 0.8, rng,
                                          beta_init=100.0, beta_max=3000.0)
                update_best(a_fs, "post_ls_fista")

    # Phase 6: Try alternate length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 180 and global_a is not None:
        print("\n=== Phase 6: Alternate Length ===")
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 65536:
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                nblocks = n // alt_n
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alt1 = (deadline - time.monotonic()) * 0.4
            if t_alt1 > 20:
                a_alt, _ = spectral_flatten(alt_start, alt_n, S_alt, t_alt1 * 0.5, rng)
                update_best(a_alt, "alt_sf")
                t_alt2 = deadline - time.monotonic() - (deadline - t_start) * 0.1 - 30
                if t_alt2 > 20:
                    a_alt2, _ = fista_optimize(a_alt, alt_n, S_alt, t_alt2, rng,
                                                beta_init=20.0, beta_max=1000.0)
                    update_best(a_alt2, "alt_fista")

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
