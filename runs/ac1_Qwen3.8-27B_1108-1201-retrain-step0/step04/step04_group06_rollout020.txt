"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences using adaptive spectral
optimization with simulated annealing, multi-scale phase-aware modifications,
and gradient-based coordinate descent. Key improvements over previous approach:
1. Simulated annealing in spectral domain for better global exploration
2. Adaptive peak-weighted shaping with dynamic target tracking
3. Phase optimization (not just amplitude) for more flexibility
4. Multi-resolution spectral operations for both exploration and exploitation
5. Sophisticated coordinate descent with quadratic line search
6. Periodic re-initialization from diverse spectral_templates

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        return np.maximum(v, 0.0)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_self_conv(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_self_conv_full(a, N):
    """Self-convolution via FFT with pre-computed size N."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def spectral_shape_step(a, N, n, c, c_max, top_k, alpha, temperature, rng, target):
    """
    One spectral shaping step with optional phase perturbation and temperature.
    Returns new sequence.
    """
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total_top = top_vals.sum()
    if total_top < 1e-30:
        return a

    # Weight by excess above target (with floor)
    excess = np.maximum(top_vals - target * 0.9, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total_top

    weights = excess / excess_sum

    # Compute effective cosine
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i in range(len(top_indices)):
        k = int(top_indices[i])
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * k / N)

    # Shape the power spectrum
    # Use a softer, more adaptive shaping
    alpha_eff = alpha * (0.5 + 0.5 * min(1.0, (c_max - target) / max(c_max * 0.1, 1e-10)))
    shape = np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    # Temperature-based randomization in log-spectral domain
    if temperature > 1e-8:
        log_P = np.log(P + 1e-30)
        log_P += rng.normal(0, temperature, N)
        P_new = np.exp(log_P) * shape
    else:
        P_new = P * shape

    # Ensure DC component is preserved (it determines the sum)
    P_new[0] = P[0]

    # Phase perturbation for exploration
    if temperature > 0.01 and rng.random() < 0.3:
        phase_noise = rng.normal(0, min(0.1, temperature * 0.5), N)
        phase_noise[0] = 0  # Keep DC phase
        phase_new = phase + phase_noise
    else:
        phase_new = phase

    A_new = np.sqrt(P_new) * np.exp(1j * phase_new)
    a_new = np.real(ifft(A_new))[:n]
    S = a.sum()
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_shape_multi(a, N, n, c, c_max, rng, target, base_alpha, temperature):

    Multi-scale spectral shaping: apply shaping at multiple alpha values
    and keep the best result.

            if cmax_try < best_cmax:

def gradient_polish(a, N, n, S, steps, rng, top_k=50, step_scale=0.1):
    """
    Gradient-based polish using subgradient of the max autocorrelation.
    Uses top-k subgradient for efficiency.
    """
    a = a.copy()
    best_a = a.copy()
    best_cmax = np.inf

    for step in range(steps):
        c = fft_self_conv_full(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        # Compute subgradient using top-k peaks
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        w = top_vals / top_vals.sum()

        # Gradient: for each peak lag k, grad += a shifted by k
        grad = np.zeros(n)
        A = fft(a, N)
        for i in range(len(top_indices)):
            k = int(top_indices[i])
            # Cross-correlation of a with delta at lag k
            # This is a shifted version of a
            # FFT approach: multiply spectrum by exp(-2*pi*i*k*m/N)
            m = np.arange(N)
            shift = np.exp(-2j * np.pi * k * m / N)
            xc = np.real(ifft(A * shift * w[i]))[:n]
            # Also the reverse shift
            shift_rev = np.exp(2j * np.pi * k * m / N)
            xc_rev = np.real(ifft(A * shift_rev * w[i]))[:n]
            grad += xc + xc_rev

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        lr = step_scale * S / (n * gnorm)
        lr = min(lr, 0.5 * S / n)  # Cap to avoid instability

        a_new = project_simplex(a - lr * grad, S)
        c_new = fft_self_conv_full(a_new, N)
        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best_a = a_new.copy()
        a = a_new

    return best_a, best_cmax


def coordinate_descent(a, n, S, max_time, rng, n_top=20, max_rounds=100):
    """
    Coordinate descent on top-contributing coordinates at the peak lag.
    Uses exact 1D optimization.
    """
    a = a.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    N = 1 << (2 * n - 1).bit_length()

    for rnd in range(max_rounds):
        if time.monotonic() - start > max_time:
            break

        c = fft_self_conv_full(a, N)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        peak_lag = int(np.argmax(c))

        # Find top-contributing coordinates at the peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]
        improved = False

        for idx in top_coords:
            i = int(idx)
            if time.monotonic() - start > max_time:
                break
            j = peak_lag - i
            if j < 0 or j >= n:
                continue

            orig_val = a[i]
            best_frac = 1.0
            best_cmax_local = c_max

            # Try several multipliers
            for frac in [0.2, 0.4, 0.6, 0.8, 1.2, 1.4, 0.1, 0.5, 1.5, 0.3]:
                a[i] = orig_val * frac
                # Quick recompute of affected lags
                c_affected = c.copy()
                delta = a[i] - orig_val
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j1 = k - i
                    if 0 <= j1 < n:
                        c_affected[k] += delta * a[j1]
                    j2 = i
                    k_off = k - j2
                    if 0 <= k_off < n and k_off != j1:
                        c_affected[k] += delta * a[k_off]
                new_cmax = c_affected.max()
                if new_cmax < best_cmax_local:
                    best_cmax_local = new_cmax
                    best_frac = frac

            a[i] = orig_val * best_frac
            if best_frac != 1.0:
                improved = True

        a = project_simplex(a, S)

        if not improved and rng.random() < 0.4:
            # Perturb top contributors
            for idx in top_coords[:min(5, len(top_coords))]:
                a[int(idx)] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)

    return best_a, best_score

def normalize_seq(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            break
        a_norm = normalize_seq(arr, n)
        c0 = fft_self_conv_full(a_norm, N)
        sc0 = c0.max()
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: cmax=%.6f" % (name, global_best))

    # Quick spectral shaping from best
    if global_a is not None:
        rem = deadline - time.monotonic()
        quick_budget = min(p1_budget - (time.monotonic() - t_p1), rem * 0.5)
        if quick_budget > 5:
            # Rapid shaping iterations
            a_curr = global_a.copy()
            for k in range(20):
                if time.monotonic() > t_p1 + p1_budget:
                    break
                c = fft_self_conv_full(a_curr, N)
                c_max = c.max()
                if c_max < global_best:
                    global_best = c_max
                    global_a = a_curr.copy()
                top_k = min(100, 10 + k * 5)
                alpha = 0.5 * (1.0 + 0.1 * k)
                a_try = spectral_shape_step(a_curr, N, n, c, c_max, top_k, alpha,
                                            0.02, rng, global_best * 0.95)
                c_try = fft_self_conv_full(a_try, N)
                if c_try.max() < c_max * 1.05:
                    a_curr = a_try
                else:
                    a_curr = spectral_shape_step(a_curr, N, n, c, c_max, top_k,
                                                 alpha * 0.5, 0.01, rng, global_best)
            c_final = fft_self_conv_full(a_curr, N)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_curr.copy()

    print("=== Phase 1 done. Best cmax: %.6f ===" % global_best)
    if global_a is None:
        global_a = normalize_seq(structured_init(n, rng, "sin"), n)
        global_best = fft_self_conv_full(global_a, N).max()

    # === Phase 2: Simulated annealing optimization (55% time) ===
    print("=== Phase 2: Simulated annealing ===")
    t_p2 = time.monotonic()
    p2_budget = (deadline - t_p2) * 0.55

    # Split into multiple SA runs with different parameters
    n_sa_runs = 3
    per_run = p2_budget / n_sa_runs

    for run in range(n_sa_runs):
        if time.monotonic() > deadline - 60:
            break
        rem = per_run - (time.monotonic() - t_p2) - run * per_run
        if rem < 5:
            # Use whatever time is left
            rem = deadline - 60 - time.monotonic()
            if rem < 5:
                break

        # For subsequent runs, perturb the best
        if run == 0:
            a_start = global_a.copy()
        else:
            # Perturb from best
            A = fft(global_a, N)
            sigma = 0.1 * S / np.sqrt(N)
            noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
            noise[0] = 0
            a_start = np.real(ifft(A + noise))[:n]
            a_start = project_simplex(a_start, S)

        a_sa, c_sa = simulated_annealing_spectral(a_start, N, n, S,
                                                   time.monotonic() + rem,
                                                   rng, target=global_best * 0.99)
        if try_update(a_sa):
            print("  [SA run %d] improved: cmax=%.6f" % (run, global_best))

    print("=== Phase 2 done. Best cmax: %.6f ===" % global_best)

    # === Phase 3: Gradient polish (20% time) ===
    print("=== Phase 3: Gradient polish ===")
    t_p3 = time.monotonic()
    p3_budget = (deadline - t_p3) * 0.20

    a_curr = global_a.copy()
    for k in range(10):
        if time.monotonic() > t_p3 + p3_budget:
            break
        rem = t_p3 + p3_budget - time.monotonic()
        if rem < 3:
            break
        n_steps = max(5, int(rem * 5))
        top_k = min(300, 20 + k * 20)
        a_try, c_try = gradient_polish(a_curr, N, n, S, n_steps, rng,
                                       top_k=top_k, step_scale=0.05)
        if try_update(a_try):
            a_curr = a_try.copy()
            print("  [Grad iter %d] improved: cmax=%.6f" % (k, global_best))
        else:
            # Perturb and retry
            a_curr = project_simplex(
                a_curr + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    print("=== Phase 3 done. Best cmax: %.6f ===" % global_best)

    # === Phase 4: Coordinate descent final polish (15% time) ===
    print("=== Phase 4: Coordinate descent ===")
    t_p4 = time.monotonic()
    p4_budget = (deadline - t_p4) * 0.15

    a_curr = global_a.copy()
    for k in range(5):
        if time.monotonic() > t_p4 + p4_budget:
            break
        rem = t_p4 + p4_budget - time.monotonic()
        if rem < 3:
            break
        a_try, c_try = coordinate_descent(a_curr, n, S, rem, rng,
                                          n_top=30 + k * 10,
                                          max_rounds=50)
        if try_update(a_try):
            a_curr = a_try.copy()
            print("  [CD iter %d] improved: cmax=%.6f" % (k, global_best))
        else:
            # Small perturbation
            a_curr = project_simplex(
                a_curr + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    print("=== Phase 4 done. Best cmax: %.6f ===" % global_best)

    # === Final: One more round of combined optimization ===
    if time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            print("=== Final optimization round (%.0fs) ===" % rem)
            # Short SA
            a_final, _ = simulated_annealing_spectral(global_a, N, n, S,
                                                       time.monotonic() + rem * 0.4,
                                                       rng, target=global_best * 0.995)
            try_update(a_final)
            # Gradient
            rem2 = deadline - time.monotonic() - 5
            if rem2 > 5:
                a_g, _ = gradient_polish(global_a, N, n, S, int(rem2 * 3), rng,
                                         top_k=200, step_scale=0.03)
                try_update(a_g)
            # Coordinate
            rem3 = deadline - time.monotonic() - 3
            if rem3 > 3:
                a_cd, _ = coordinate_descent(global_a, n, S, rem3, rng,
                                             n_top=50, max_rounds=100)
                try_update(a_cd)

    print("=== Final best cmax: %.6f ===" % global_best)

    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
