So the outer problem is to maximize over p, and the inner is to minimize over a. The alternating algorithm should:

2. Given a*, find p* that maximizes sum p_k c_k, which is p_k = 1 at the argmax (outer)

But step 2 with p concentrated on a single k is very "spiky" and leads to slow convergence. The exponential weights version (p_k ∝ exp(c_k/τ)) is a smoothed version that converges faster in practice.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using a combination of:
1. Hankel-based min-max duality (novel approach exploiting the structure)
2. Smoothed subgradient FISTA with softmax-based gradient
3. Adaptive spectral refinement
Key insight: The problem min_a max_k c_k s.t. sum(a)=S, a>=0 has a saddle-point
formulation via p in simplex: max_p min_a sum_k p_k c_k. We exploit this with
efficient O(n log n) Hankel matrix-vector products via FFT.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_self_fft(a, N):
    """Self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    c = convolve_self_fft(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def hankel_mv(p, a, N):
    """Compute Q*a where Q_{j,m} = p[j+m] (Hankel matrix) using FFT. O(n log n)."""
    n = len(a)
    P = fft(p, N)
    A = fft(a, N)
    result = np.real(ifft(P * np.conj(A)))[:n]
    return result


def softmax_weights(c, tau):
    """Compute softmax/normalized exponential weights."""
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    return exp_c / exp_c.sum()


def hankel_minmax_optimize(a_init, n, S, max_time, rng, inner_steps=15, tau_p_init=0.05, tau_p_final=0.005):
    """
    Hankel-based min-max dual algorithm.
    Alternates between:
    - Inner: projected gradient on min a^T Q(p) a s.t. sum(a)=S, a>=0
    - Outer: exponential weights update on p
    """
    a = project_simplex(a_init.copy(), S)
    N_conv = 1 << (2 * n - 1).bit_length()
    N_hankel = 1 << (3 * n - 2).bit_length()

    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    # Initialize p from current convolution
    c = convolve_self_fft(a, N_conv)
    c_pos = np.maximum(c, 1e-10)
    p = c_pos / c_pos.sum()

    outer_iter = 0
    stuck_count = 0
    prev_best = best_score

    while time.monotonic() - start < max_time:
        outer_iter += 1

        # Anneal temperature for p update
        progress = min(1.0, outer_iter / 500.0)
        tau_p = tau_p_init * (tau_p_final / tau_p_init) ** progress

        # Inner: projected gradient on min a^T Q(p) a s.t. sum(a)=S, a>=0
        # Step size: 1/(2*||Q||). Since ||Q|| <= 1 for probability p, use eta=0.45.
        eta = 0.45
        for _ in range(inner_steps):
            grad = 2.0 * hankel_mv(p, a, N_hankel)
            a_new = project_simplex(a - eta * grad, S)
            # Check if we made progress
            if np.any(a_new != a):
                a = a_new
            else:
                break

        # Compute convolution and update p
        c = convolve_self_fft(a, N_conv)
        c_max = c.max()
        score = 2.0 * n * c_max / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            # If stuck, add perturbation
            if stuck_count % 50 == 0:
                a = project_simplex(a + 0.01 * S / n * rng.standard_normal(n), S)

        # Update p: exponential weights (smoothed max)
        p = softmax_weights(c, tau_p)

        if outer_iter % 100 == 0:
            elapsed = time.monotonic() - start
            print("    HankelMM iter %d: score=%.6f, c_max=%.6f, tau_p=%.4f (%.0fs)" %
                  (outer_iter, score, c_max, tau_p, elapsed))

        # Early stop if we're doing very well
        if best_score < 1.500:
            break

    return best_a, best_score


def fista_smooth_max(a_init, n, S, max_time, rng, tau_init=0.1, tau_final=0.01, n_inner_fista=3000):
    """
    FISTA for smoothed max objective with annealed temperature.
    Objective: soft_max(conv(a), tau) = tau * log(sum exp(c_k/tau))
    Constraint: sum(a) = S, a >= 0
    Gradient: 2 * cross_correlation(softmax_weights(c), a)
    """
    a = project_simplex(a_init.copy(), S)
    N_conv = 1 << (2 * n - 1).bit_length()
    N_hankel = 1 << (3 * n - 2).bit_length()

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    # Initial Lipschitz estimate
    L = 4.0
    it = 0

    while time.monotonic() - start < max_time and it < n_inner_fista:
        it += 1
        progress = min(1.0, it / max(1, n_inner_fista))
        tau = tau_init * (tau_final / tau_init) ** progress

        # Compute gradient at y
        c_y = convolve_self_fft(y, N_conv)
        p_y = softmax_weights(c_y, tau)
        grad = 2.0 * hankel_mv(p_y, y, N_hankel)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Backtracking line search (simplified: use L with occasional adjustment)
        x_new = project_simplex(y - grad / L, S)

        # Periodically re-estimate L
        if it % 200 == 0:
            # L should be ~ 4 * max(c) / tau (rough bound)
            L_est = max(1.0, 4.0 * c_y.max() / tau)
            L = 1.5 * L_est

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y * (S / sy)
        a = x_new
        t = t_new

        # Evaluate periodically
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Periodic noise for escaping local minima
        if it % 500 == 250 and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + noise_scale * rng.standard_normal(n), S)
            a = y.copy()
            t = 1.0

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def spectral_refine(a_init, n, S, max_time, rng, target=1.0):
    """
    Spectral domain refinement: iteratively modify |FFT(a)|^2 to flatten
    the time-domain autocorrelation peak.
    """
    a = project_simplex(a_init.copy(), S)
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    alpha = 0.5
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal
        alpha = max(0.02, 0.5 * (0.2 ** (it / 200.0)))

        # Current spectrum
        A = fft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Current autocorrelation
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < 1e-15:
            break

        # Flatten: reduce high-magnitude bins
        p_median = np.median(P)
        if p_median < 1e-30:
            break
        ratio = P / p_median
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        # Compress above-median bins
        P_new[above] = P[above] * (1.0 - alpha * (1.0 - ratio[above] ** (-alpha)))
        # Slightly boost below-median bins
        below = ~above
        P_new[below] = P[below] * (1.0 + alpha * 0.5 * (1.0 - ratio[below] ** alpha))
        P_new = np.maximum(P_new, 0.0)

        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if it % 200 == 0:
            print("    Spectral iter %d: score=%.6f" % (it, best_score))

        if best_score < 1.495:
            break

    return best_a, best_score


def make_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("uniform", np.ones(n) * 0.5))
    inits.append(("random", rng.random(n) + 0.05))

    # Multi-sine
    inits.append(("multi_sin", 0.5 + 0.15 * np.sin(2 * np.pi * 3 * tt) +
                                0.1 * np.sin(2 * np.pi * 7 * tt) +
                                0.08 * np.sin(2 * np.pi * 13 * tt)))

    # Chirp
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt ** 2))))

    # M-sequence-like
    b = (rng.random(n) > 0.5).astype(float)
    b = b - b.mean() + 0.5
    inits.append(("mseq", np.clip(b, 0.05, 0.95)))

    # Smooth noise
    noise = rng.standard_normal(n)
    ks = max(3, n // 32)
    kernel = np.ones(ks) / ks
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth", 0.2 + 0.6 * smooth))

    return inits


def optimize_length(n, budget_s, rng, provided_seq=None, target=1.50):

    S = np.sqrt(2.0 * float(n))

            return
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Build list of starting points
    start_points = []

        ps = np.clip(provided_seq, 0, 1000).astype(np.float64)
        if ps.sum() > 0.01:

            start_points.append(("provided", ps))

        arr = np.clip(arr, 0, 1000)
