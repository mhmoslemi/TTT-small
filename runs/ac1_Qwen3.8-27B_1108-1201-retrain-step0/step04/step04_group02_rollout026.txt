"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using annealed LSE-FISTA
with anti-symmetric cosine initializations and peak-equalization polish.
Key insight: anti-symmetric sequences (a[i]*a[n-1-i] small) reduce the center
convolution peak below 1/n, enabling objectives below 2.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on theta."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c)))."""
    cm = c.max()
    if beta * (c.max() - c.min()) > 700:
        # Avoid overflow: just use the max
        n_max = np.sum(c > cm - 1.0 / beta)
        return cm + np.log(max(n_max, 1.0)) / beta
    e = np.exp(beta * (c - cm))
    return cm + np.log(e.sum()) / beta


def lse_grad(a, c, beta, n):
    """
    Gradient of LSE objective w.r.t. a.
    grad[j] = 2 * sum_k w_k * a[k-j] for valid k,j
    where w = softmax(beta * c).
    Uses top-k truncation for efficiency.
    """
    cm = c.max()
    if beta * (c.max() - c.min()) > 500:
        # Very peaked: use top-k
        weights = np.zeros_like(c)
        top_k = min(50, len(c))
        idx = np.argpartition(c, -top_k)[-top_k:]
        e_top = np.exp(beta * (c[idx] - cm))
        w_top = e_top / e_top.sum()
        weights[idx] = w_top
    else:
        e = np.exp(beta * (c - cm))
        weights = e / e.sum()

    # grad[j] = 2 * sum_k weights[k] * a[k-j] (if 0 <= k-j < n)
    # Use FFT: this is a cross-correlation
    # Define: g[j] = sum_k weights[k] * a[k-j] for 0<=k-j<n
    # = (weights cross-correlate a)[j]
    # Using FFT: ifft(fft(weights) * conj(fft(a))) gives circular cross-correlation
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * cross


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def fista_lse(a_init, n, max_time, rng, beta_start=20.0, beta_end=None, verbose=False):
    """
    FISTA with annealed LSE smoothing and backtracking line search.
    Works in probability space (sum = 1).
    """
    if beta_end is None:
        beta_end = max(200.0, n / 40.0)

    a = project_simplex(a_init.copy(), 1.0)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    best_lse = lse_value(fft_convolve_self(a), beta_start)
    it = 0
    lip = 2.0  # Initial Lipschitz estimate
    max_iter = 1000000

    while time.monotonic() - start < max_time and it < max_iter:
        it += 1

        # Anneal beta
        frac = min(1.0, it / max(1000, int(max_time * 5)))
        beta = beta_start + (beta_end - beta_start) * (1.0 - (1.0 - frac) ** 3)

        c_y = fft_convolve_self(y)
        grad = lse_grad(y, c_y, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Backtracking line search
        step = 1.0 / lip
        a_old = a.copy()
        improved = False
        for bt in range(30):
            s_try = step
            x_new = project_simplex(y - s_try * grad, 1.0)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, beta)
            # Armijo-type check with proximal term
            prox_term = 0.5 / s_try * np.sum((x_new - y) ** 2)
            lse_y = lse_value(c_y, beta)
            if lse_new <= lse_y + np.dot(grad, x_new - y) + prox_term:
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break

        if not improved:
            # Accept anyway if we've been stuck
            a_old = a.copy()
            x_new = project_simplex(y - step * grad, 1.0)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, beta)
            lip = lip * 2.0

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        a = x_new
        y = a + ((t - 1.0) / t_new) * (a - a_old)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            # Don't renormalize y (FISTA requires the exact momentum formula)
            # But clip to prevent issues
            y = np.clip(y, 0.0, 10.0)
        t = t_new

        # Update Lipschitz estimate
        if improved:
            lip = max(lip * 0.9, 1.0 / (step * 2.0))

        # Check actual objective periodically
        if it % 100 == 0 or (time.monotonic() - start) > max_time - 5:
            cur_obj = compute_objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

    # Final check
    cur_obj = compute_objective(a)
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()

    return best_a, best_obj, it


def peak_equalize(a, n, S, max_iter=200, rng=None):
    """
    Iteratively equalize the top peaks of the convolution.
    At each step, find the direction that most reduces the highest peak
    while keeping other top peaks from increasing.
    """
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_objective(a)

    for it in range(max_iter):
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top peaks
        top_k = min(10, len(c))
        peak_idx = np.argpartition(c, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1]]

        if len(peak_idx) < 2:
            break

        # If peaks are already well-equalized, stop
        c_top = c[peak_idx]
        if c_top.max() - c_top.min() < 1e-8 * c_top.max():
            break

        # Gradient of the top peak
        k_star = peak_idx[0]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] = 2.0 * a[k_star - i_vals]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step: try to reduce the peak without increasing others
        step = S / (n * gnorm * 50.0)
        best_step = 0

        for trial in range(10):
            s_try = step * (1.0 + 0.5 * trial)
            a_try = project_simplex(a - s_try * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < cur_obj:
                best_step = s_try
                a_new = a_try
            else:
                break

        if best_step == 0:
            # Try top-3 peak gradient
            grad3 = np.zeros(n)
            for k in peak_idx[:3]:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                i_vals = np.arange(lo, hi + 1)
                grad3[i_vals] += 2.0 * a[k - i_vals]
            gnorm3 = np.linalg.norm(grad3)
            if gnorm3 > 1e-18:
                s3 = S / (n * gnorm3 * 20.0)
                a_try = project_simplex(a - s3 * grad3, S)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n * c_try.max() / S**2
                if obj_try < cur_obj:
                    a = a_try
                    continue
            break

        a = a_new

    return best_a, best_obj


def gradient_descent_polish(a, n, S, max_time, rng):
    """
    Careful gradient descent with very small steps on the exact (non-smooth) objective.
    Uses top-peak subgradients with aggressive line search.
    """
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_objective(a)
    start = time.monotonic()
    step = 0.1
    consecutive_fail = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0

        # Use top-5 peak subgradient
        top_k = min(5, len(c))
        idx = np.argpartition(c, -top_k)[-top_k:]
        grad = np.zeros(n)
        for k in idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[k - i_vals]
        grad /= len(idx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Line search
        improved = False
        for trial in range(15):
            s_try = step * (0.5 ** trial)
            a_try = project_simplex(a - s_try * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < cur_obj - 1e-14:
                a = a_try
                improved = True
                break
        if improved:
            step = min(step * 1.3, 2.0)
        else:
            step *= 0.5
            consecutive_fail += 1
            if consecutive_fail > 30:
                # Perturb
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                step = 0.1
                consecutive_fail = 0
            if step < 1e-10:
                break

    return best_a, best_obj


def make_antisymmetric_cosine(n, alpha=1.0, beta2=0.0):
    """Create anti-symmetric cosine initialization: 1 + alpha*cos(pi*(i+0.5)/n) + beta2*cos(2*pi*(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * (i + 0.5) / n)
    if beta2 != 0.0:
        a = a + beta2 * np.cos(2.0 * np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

                t_long = min(time.monotonic() - t0, deadline - time.monotonic() - 60) * 0.4
                t_long = max(30.0, t_long)
                print("  Long FISTA on best init, time=%.0fs" % t_long)
                a_long, obj_long, n_iter = fista_lse(
                    best_init, n, t_long, rng,
                    beta_start=50.0, beta_end=max(1000.0, n / 10.0)
                )
                a_long_orig = a_long * S
                real_obj = compute_objective(a_long_orig)
                if real_obj < global_best:
                    global_best = real_obj
                    global_a = a_long_orig.copy()
                    print("  Long FISTA: obj=%.6f, iters=%d (elapsed %.0fs)" % (
                        global_best, n_iter, time.monotonic() - t0))

    print("=== After Phase 1: best=%.6f, elapsed=%.0fs ===" % (
        global_best, time.monotonic() - t0))

    # Phase 2: Peak equalization and gradient polish
    remaining = deadline - time.monotonic()
    print("=== Phase 2: Polish (remaining %.0fs) ===" % remaining)

    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Peak equalization
        t_pe = remaining * 0.25
        print("  Peak equalization, time=%.0fs" % t_pe)
        a_pe, obj_pe = peak_equalize(global_a / S_best, n_best, S_best, max_iter=500)
        a_pe_orig = a_pe * S_best
        real_obj = compute_objective(a_pe_orig)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_pe_orig.copy()
            print("  After peak equalize: %.6f" % global_best)

        # Gradient descent polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_gd = remaining2 * 0.4
            print("  Gradient polish, time=%.0fs" % t_gd)
            a_gd, obj_gd = gradient_descent_polish(global_a, n_best, S_best, t_gd, rng)
            real_obj = compute_objective(a_gd)
            if real_obj < global_best:
                global_best = real_obj
                global_a = a_gd.copy()
                print("  After gradient polish: %.6f" % global_best)

    # Phase 3: Another round of FISTA with different parameters
    remaining = deadline - time.monotonic()
    print("=== Phase 3: Advanced FISTA (remaining %.0fs) ===" % remaining)

    if global_a is not None and remaining > 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_prob = global_a / S_best

        # FISTA with very high beta (near-exact max)
        t_f3 = remaining * 0.5
        print("  High-beta FISTA, time=%.0fs" % t_f3)
        a_f, s_f, n_iter = fista_lse(
            a_prob, n_best, t_f3, rng,
            beta_start=200.0, beta_end=max(2000.0, n_best / 5.0)
        )
        a_f_orig = a_f * S_best
        real_obj = compute_objective(a_f_orig)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_f_orig.copy()
            print("  High-beta FISTA: %.6f (iters=%d)" % (global_best, n_iter))

        # Peak equalize again
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_pe2, obj_pe2 = peak_equalize(global_a / S_best, n_best, S_best, max_iter=300)
            a_pe2_orig = a_pe2 * S_best
            real_obj = compute_objective(a_pe2_orig)
            if real_obj < global_best:
                global_best = real_obj
                global_a = a_pe2_orig.copy()
                print("  After final peak equalize: %.6f" % global_best)

    # Phase 4: Try a different length if time allows
    remaining = deadline - time.monotonic()
    if remaining > 120:
        for n_try in [4096, 16384]:
            if n_try == (len(global_a) if global_a is not None else 0):
                continue
            if time.monotonic() > deadline - 60:
                break
            S_try = np.sqrt(2.0 * n_try)
            a_init = make_antisymmetric_cosine(n_try, 1.0) * S_try
            t_try = min(remaining * 0.4, 120.0)
            print("=== Trying n=%d for %.0fs ===" % (n_try, t_try))
            a_t, s_t, _ = fista_lse(
                a_init / S_try, n_try, t_try, rng,
                beta_start=50.0, beta_end=max(500.0, n_try / 20.0)
            )
            a_t_orig = a_t * S_try
            real_obj = compute_objective(a_t_orig)
            if real_obj < global_best:
                global_best = real_obj
                global_a = a_t_orig.copy()
                print("  n=%d: obj=%.6f" % (n_try, global_best))

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, obj=%.6f" % (len(final_list), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return final_list
