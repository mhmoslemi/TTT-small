```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Uses FISTA with proper Lipschitz-based step sizes on LSE-smoothed max,
adaptive beta, and multiple restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_and_gradient(a, c, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    mask = scaled > 500
    if np.any(mask):
        scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def run_fista(a_init, S, max_time, rng, beta_init=20.0, beta_end=5000.0,
              step_scale=1.0, use_fista=True, noise_interval=2000, noise_scale=0.005):
    """FISTA or GD with LSE smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.full(len(a), S / len(a))
    else:
        a = a * (S / s)
    
    n = len(a)
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_init * (beta_end / beta_init) ** (progress ** 0.5)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        lse_val, grad = lse_and_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: based on Lipschitz estimate
        # L ≈ 1 + 4 * ||y||_2^2
        L_est = 1.0 + 4.0 * np.sum(y ** 2)
        step = step_scale / L_est
        
        if use_fista:
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
            y = np.maximum(y, 0.0)
            x_prev = x_new
            t = t_new
        else:
            # Plain gradient descent with backtracking
            x_new = project_simplex(y - step * grad, S)
            y = x_new
        
        # Periodic noise
        if it % noise_interval == 0:
            decay = max(0.0, 1.0 - progress)
            ns = noise_scale * S / np.sqrt(n) * decay
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                if use_fista:
                    x_prev = y.copy()
                    t = 1.0
        
        # Stuck detection
        if it - last_improve > 5000:
            ns = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, ns, n), S)
            if use_fista:
                x_prev = y.copy()
                t = 1.0
            last_improve = it
    
    return best_a, best
```

```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Key insight: The LSE-smoothed max objective has Lipschitz constant L = 1 + 4*||a||_2^2
independent of the smoothing parameter beta. This allows much larger effective step sizes
(~0.1) compared to the naive 1/beta step, dramatically accelerating convergence.

Algorithm:
1. Multi-start FISTA with proper Lipschitz-based step sizes on LSE-smoothed max
2. Adaptive beta schedule (moderate to very high)
3. FISTA momentum with adaptive restarts
4. Iterative peak reduction (IPR) refinement
5. High-beta polishing phase
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.array(v, dtype=np.float64, copy=True)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Compute LSE value and gradient of (1/beta)*log(sum exp(beta*c))."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 350.0)  # prevent overflow
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    # Gradient via FFT: grad = 2 * ifft(W * conj(A))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def norm2_sq(v):
    """Squared L2 norm."""
    return np.dot(v, v)


def run_fista_lse(a_init, S, max_time, rng, beta_init=20.0, beta_end=5000.0,
                  step_scale=1.0, momentum=True, noise_interval=3000,
                  noise_scale=0.004, max_steps=None):
    """FISTA with LSE-smoothed objective and proper Lipschitz step sizes."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(len(a), S / len(a))
    else:
        a = a * (S / s)

    n = len(a)
    y = a.copy()
    x_prev = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    prev_lse = np.inf
    stuck_count = 0

    while time.monotonic() - start < max_time:
        if max_steps is not None and it >= max_steps:
            break
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Beta schedule: moderate to high
        beta = beta_init * (beta_end / beta_init) ** (progress ** 0.4)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        lse_val, grad = lse_gradient(y, c, beta)

        # Check for increase (potential instability)
        if lse_val > prev_lse + 1e-10:
            stuck_count += 1
        else:
            stuck_count = max(0, stuck_count - 1)
        prev_lse = lse_val

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Lipschitz-based step size
        L_est = 1.0 + 4.0 * norm2_sq(y)
        step = step_scale / L_est

        # Backtracking if stuck
        if stuck_count > 5:
            step *= 0.5
            if stuck_count > 20:
                # Reset momentum
                if momentum:
                    x_prev = y.copy()
                    t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)

        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
            y = np.maximum(y, 0.0)
            # Ensure sum constraint is approximately met
            y_sum = y.sum()
            if y_sum > 0:
                y = y * (S / y_sum)
            x_prev = x_new
            t_mom = t_new
        else:
            y = x_new

        # Periodic perturbation to escape local minima
        if it % noise_interval == 0:
            decay = max(0.0, 1.0 - progress * 0.8)
            ns = noise_scale * S / np.sqrt(n) * decay
            if ns > 1e-9:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                x_prev = y.copy()
                t_mom = 1.0

        # Stuck detection with larger perturbation
        if it - last_improve > 8000 and progress < 0.9:
            ns = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, ns, n), S)
            x_prev = y.copy()
            t_mom = 1.0
            last_improve = it

    return best_a, best


def run_ipr(a_init, S, max_time, rng, top_k=8):
    """Iterative Peak Reduction: directly reduce the top peaks."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(len(a), S / len(a))
    else:
        a = a * (S / s)

    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Get top-k peak lags
        k = min(top_k, len(c))
        peak_idx = np.argsort(c)[::-1][:k]
        mean_b = c[peak_idx].mean()
        weights = np.maximum(c[peak_idx] - mean_b, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights = weights / wsum

        # Weighted sum of gradients
        w_full = np.zeros(len(c))
        w_full[peak_idx] = weights
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Make zero-sum (preserve sum constraint)
        grad = grad - grad.mean()

        # Step size: based on gradient magnitude and feasibility
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: ensure non-negativity
        max_step = np.inf
        neg_mask = grad < -1e-15
        if np.any(neg_mask):
            max_step = np.min(a[neg_mask] / (-grad[neg_mask]))
        step = min(0.1 * S / gnorm, 0.5 * max_step if np.isfinite(max_step) else 0.1 * S / gnorm)
        step = max(step, 1e-10)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-15:
            a_new = a_new * (S / a_new_sum)
        a = a_new

    return best_a, best


def run_high_beta_polish(a_init, S, max_time, rng, beta_start=5000.0, beta_end=100000.0):
    """Final polishing with very high beta (near-exact max approximation)."""
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(len(a), S / len(a))
    else:
        a = a * (S / s)

    n = len(a)
    y = a.copy()
    x_prev = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        lse_val, grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        L_est = 1.0 + 4.0 * norm2_sq(y)
        step = 0.8 / L_est

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        x_prev = x_new
        t_mom = t_new

        # Occasional micro-perturbation
        if it % 5000 == 0 and progress < 0.7:
            ns = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            if ns > 1e-10:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                x_prev = y.copy()
                t_mom = 1.0

    return best_a, best


def generate_starts(n, rng, base=None):
    """Generate diverse starting sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base is not None:
        b = np.clip(base.astype(np.float64), 0, 1000)
        if len(b) == n:
            starts.append(b.copy())
        elif len(b) > n:
            ratio = len(b) // n
            if ratio > 0:
                small = b[:n * ratio].reshape(n, ratio).mean(axis=1)
                starts.append(small)
        else:
            t_old = np.linspace(0, 1, len(b))
            t_new = np.linspace(0, 1, n)
            starts.append(np.interp(t_new, t_old, b))

    starts.append(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)) + 0.01)
    starts.append(np.sin(np.pi * tt) ** 2 + 0.01)
    starts.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.09) + 0.01)
    starts.append(1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * tt) + 0.1 * np.cos(2.0 * np.pi * 7.0 * tt + 0.5))
    starts.append(1.0 + 0.2 * np.sin(2.0 * np.pi * 5.0 * tt + 1.0) + 0.15 * np.cos(2.0 * np.pi * 11.0 * tt))
    starts.append(rng.random(n) * 0.7 + 0.2)
    starts.append(np.exp(rng.normal(0, 0.4, n)))

    # Multi-scale
    for scale in [16, 64]:
        m = max(4, n // scale)
        coarse = rng.uniform(0.3, 1.5, m)
        fine = np.interp(tt, np.linspace(0, 1, m), coarse)
        starts.append(fine + 0.05)

    return [np.maximum(x, 0.01) for x in starts]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses FISTA with Lipschitz-correct step sizes (key improvement over 1/beta steps).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.1fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence: len=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)

    # Phase 1: Multi-start FISTA
    print("\n=== Phase 1: Multi-start FISTA exploration ===")
    starts = generate_starts(n_main, rng, base=given)

    phase1_total = min(500.0, (deadline - time.monotonic()) * 0.35)
    time_per_start = phase1_total / len(starts)

    for i, st in enumerate(starts):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.5 - 60:
            break
        t_alloc = min(time_per_start * 1.3, deadline - time.monotonic() - 120)
        if t_alloc < 10:
            break

        S_i = np.sqrt(2.0 * len(st))
        a_opt, val = run_fista_lse(st, S_i, t_alloc, rng,
                                    beta_init=15.0, beta_end=5000.0,
                                    step_scale=1.0, momentum=True,
                                    noise_interval=2000, noise_scale=0.004)
        obj = 2.0 * len(a_opt) * val / (a_opt.sum() ** 2)
        print("  Start %2d: obj=%.8f, max_b=%.8f" % (i, obj, val))

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("    -> new best: %.8f" % global_best)

    print("Phase 1 best: %.8f" % global_best)

    # Phase 2: Extended FISTA on best (higher beta range)
    print("\n=== Phase 2: Extended FISTA ===")
    n_best = len(global_a)
    S_best = np.sqrt(2.0 * n_best)

    remaining = deadline - time.monotonic()
    phase2_time = min(500.0, remaining * 0.3)

    a_opt, val = run_fista_lse(global_a, S_best, phase2_time, rng,
                                beta_init=50.0, beta_end=15000.0,
                                step_scale=1.0, momentum=True,
                                noise_interval=3000, noise_scale=0.003)
    obj = 2.0 * n_best * val / (a_opt.sum() ** 2)
    print("  Phase 2: obj=%.8f" % obj)
    if obj < global_best:
        global_best = obj
        global_a = a_opt.copy()

    # Phase 3: IPR refinement
    print("\n=== Phase 3: IPR refinement ===")
    remaining = deadline - time.monotonic()
    phase3_time = min(200.0, remaining * 0.15)

    a_ipr, val_ipr = run_ipr(global_a, S_best, phase3_time, rng, top_k=16)
    obj_ipr = 2.0 * n_best * val_ipr / (a_ipr.sum() ** 2)
    print("  IPR: obj=%.8f" % obj_ipr)
    if obj_ipr < global_best:
        global_best = obj_ipr
        global_a = a_ipr.copy()

    # Phase 4: High-beta FISTA polishing
    print("\n=== Phase 4: High-beta polishing ===")
    remaining = deadline - time.monotonic()
    phase4_time = min(400.0, remaining * 0.35)

    a_pol, val_pol = run_high_beta_polish(global_a, S_best, phase4_time, rng,
                                           beta_start=3000.0, beta_end=150000.0)
    obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
    print("  Polish: obj=%.8f" % obj_pol)
    if obj_pol < global_best:
        global_best = obj_pol
        global_a = a_pol.copy()

    # Phase 5: Final micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final micro-polish ===")
        phase5_time = remaining * 0.8
        a_fin, val_fin = run_fista_lse(global_a, S_best, phase5_time, rng,
                                        beta_init=10000.0, beta_end=300000.0,
                                        step_scale=0.7, momentum=True,
                                        noise_interval=5000, noise_scale=0.001)
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        print("  Final: obj=%.8f" % obj_fin)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()

    # Also try IPR on final result
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_ipr2, val_ipr2 = run_ipr(global_a, S_best, remaining * 0.6, rng, top_k=32)
        obj_ipr2 = 2.0 * n_best * val_ipr2 / (a_ipr2.sum() ** 2)
        print("  Final IPR: obj=%.8f" % obj_ipr2)
        if obj_ipr2 < global_best:
            global_best = obj_ipr2
            global_a = a_ipr2.copy()

    # Verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: evaluate_sequence = %.8f" % ver)

    if not np.isfinite(ver):
        print("  Fallback needed!")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Result: length=%d, obj=%.8f" % (len(global_a), ver if np.isfinite(ver) else global_best))
    return final_list
```
