"""
Minimize 2*n*max_conv/sum^2 using adaptive multi-scale spectral shaping with
cosine-basis peak suppression, multiplicative load balancing, coordinate descent,
and annealed exploration. Key improvements: faster inner loops, better adaptive
parameters, interleaved local/global search, and multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (css - S) / rho < u
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho_max = int(np.max(np.where(cond)[0])) + 1
    theta = (css[rho_max - 1] - S) / rho_max
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= S / s
    return result


def convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def spectral_shape_step(a, N, n, top_lags, top_weights, alpha, rng=None, phase_noise_scale=0.0):
    """
    One spectral shaping step: modify power spectrum to reduce specified lags.
    Returns new a (already projected to simplex with same sum).
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_lags):
        cos_eff += top_weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))

    if rng is not None and phase_noise_scale > 0:
        noise = rng.normal(0, phase_noise_scale, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def multiplicative_balance_step(a, c, n, top_k, gamma, temperature):
    """
    Multiplicative load-balancing: reduce positions that contribute to peak lags.
    c is the convolution, a is the current sequence.
    """
    S = a.sum()
    c_max = c.max()
    threshold = c_max - temperature * (c_max - c.min())
    active = c > threshold
    if not np.any(active):
        return a.copy()

    active_lags = np.where(active)[0]
    # Compute load at each position
    load = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(k_int, n - 1)
        for i in range(lo, hi + 1):
            j = k_int - i
            if 0 <= j < n:
                load[i] += a[j]

    load_max = load.max()
    if load_max < 1e-15:
        return a.copy()

    # Multiplicative update: reduce high-load positions
    ratio = load_max / (load + 1e-15)
    a_new = a * np.power(np.minimum(ratio, 10.0), gamma)
    a_new = project_simplex(a_new, S)
    return a_new


def coordinate_descent_step(a, n, S, peak_lag, n_top, multipliers, c_current):
    """
    Coordinate descent: adjust top-contributing coordinates at the peak lag.
    Returns (new_a, improved).
    """
    a = a.copy()
    c = c_current.copy()
    k = int(peak_lag)
    improved = False

    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    contribs = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n:
            contribs[i] = a[i] * a[j]

    top_idx = np.argsort(contribs)[::-1][:n_top]
    c_max = c.max()

    for idx in top_idx:
        i = int(idx)
        j = k - i
        if j < 0 or j >= n or i == j:
            continue
        orig = a[i]
        best_val = orig
        best_cmax = c_max

        for mult in multipliers:
            new_val = orig * mult
            if new_val < 0:
                continue
            # Update affected convolution values
            # Changing a[i] affects C[k'] for k' where i and k'-i are both valid
            # C[k'] changes by (new_val - orig) * a[k'-i] for each valid k'
            delta = new_val - orig
            k_lo = i  # minimum k' where a[i] * a[k'-i] contributes (k'-i=0)
            k_hi = i + n - 1  # maximum k' where a[i] * a[k'-i] contributes (k'-i=n-1)
            k_lo = max(0, k_lo)
            k_hi = min(2 * n - 2, k_hi)

            # The change to C[k']: delta * a[k'-i] for each k'
            # But also: if k'-i != i, then C[k'] also gets delta * a[k'-i] from the j=k'-i term
            # Actually C[k'] = sum_m a[m]*a[k'-m]. Changing a[i] by delta:
            # Delta C[k'] = delta * a[k'-i] (from the term m=i) + delta * a[k'-i] (from m=k'-i if k'-i != i)
            # = delta * a[k'-i] + (delta * a[i'] if k'-i != i else 0) where i' = k'-i
            # Wait, more carefully: C[k'] = sum_m a[m] a[k'-m]
            # dC[k']/da[i] = a[k'-i] + a[k'-i] if i != k'-i, else a[k'-i]
            # So dC[k'] = delta * a[k'-i] + (delta * a[k'-i] if i != k'-i else 0)
            # = delta * a[k'-i] * (2 if i != k'-i else 1)

            for k_p in range(k_lo, k_hi + 1):
                j_p = k_p - i
                if 0 <= j_p < n:
                    if j_p == i:
                        c[k_p] += delta * a[j_p]
                    else:
                        c[k_p] += 2.0 * delta * a[j_p]

            new_cmax = c.max()
            if new_cmax < best_cmax - 1e-12:
                best_cmax = new_cmax
                best_val = new_val

            a[i] = orig  # reset

        if best_val != orig:
            a[i] = best_val
            delta = best_val - orig
            for k_p in range(k_lo, k_hi + 1):
                j_p = k_p - i
                if 0 <= j_p < n:
                    if j_p == i:
                        c[k_p] += delta * a[j_p]
                    else:
                        c[k_p] += 2.0 * delta * a[j_p]
            improved = True

    a = project_simplex(a, S)
    return a, improved, c


def fast_spectral_loop(a, n, N, S, deadline, rng, target, top_k_start=10, top_k_max=200):
    """
    Fast iterative spectral shaping loop.
    Returns (best_a, best_score).
    """
    a = a.copy()
    c = convolve_self(a)
    best_a = a.copy()
    best_score = c.max()
    it = 0
    stall = 0
    last_improve_it = 0

    while time.monotonic() < deadline:
        it += 1
        top_k = min(top_k_max, top_k_start + it // 50)
        alpha = 0.3 / (1.0 + 0.01 * it)

        # Find top lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        a_new = spectral_shape_step(a, N, n, top_indices, weights, alpha, rng,
                                    phase_noise_scale=0.02 * alpha)
        c_new = convolve_self(a_new)
        score_new = c_new.max()

        if score_new < best_score - 1e-9:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            c = c_new
            stall = 0
            last_improve_it = it
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape
                a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
                c = convolve_self(a)
                if c.max() < best_score:
                    best_score = c.max()
                    best_a = a.copy()
                stall = 0

        if best_score <= target:
            break

    return best_a, best_score


def annealed_coord_descent(a, n, S, deadline, rng, n_top=30):
    """
    Annealed coordinate descent: progressively smaller steps.
    Returns (best_a, best_score).
    """
    a = a.copy()
    c = convolve_self(a)
    best_a = a.copy()
    best_score = c.max()
    multipliers = [0.5, 0.75, 1.25, 1.5, 0.25, 1.75, 0.875, 1.125]
    round_num = 0

    while time.monotonic() < deadline:
        round_num += 1
        peak_lag = int(np.argmax(c))
        c_max = c[peak_lag]

        if c_max < best_score - 1e-12:
            best_score = c_max
            best_a = a.copy()

        # Progressively fewer multipliers (finer search)
        n_mult = max(3, len(multipliers) - round_num // 10)
        mults = multipliers[:n_mult]

        a, improved, c = coordinate_descent_step(a, n, S, peak_lag, n_top, mults, c)

        if not improved and round_num % 10 == 0:
            # Perturb
            peak_lag2 = int(np.argmax(c))
            lo = max(0, peak_lag2 - n + 1)
            hi = min(peak_lag2, n - 1)
            contribs = np.zeros(n)
            for i in range(lo, hi + 1):
                j = peak_lag2 - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top = np.argsort(contribs)[::-1][:10]
            for idx in top:
                a[int(idx)] *= rng.uniform(0.7, 1.3)
            a = project_simplex(a, S)
            c = convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a.copy()

    return best_a, best_score


def balanced_spectral_loop(a, n, N, S, deadline, rng, target):
    """
    Spectral loop alternating between peak reduction and load balancing.
    """
    a = a.copy()
    c = convolve_self(a)
    best_a = a.copy()
    best_score = c.max()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = convolve_self(a)
        score = c.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if best_score <= target:
            break

        if it % 3 == 0:
            # Multiplicative balance step
            temp = max(0.05, (best_score - 1.0) * 0.3)
            a_new = multiplicative_balance_step(a, c, n, 50, 0.3, temp)
            c_check = convolve_self(a_new)
            if c_check.max() < best_score:
                a = a_new
        else:
            # Spectral shaping step
            top_k = min(100, 10 + it // 20)
            top_indices = np.argsort(c)[::-1][:top_k]
            top_vals = c[top_indices]
            total = top_vals.sum()
            if total < 1e-30:
                break
            weights = top_vals / total
            alpha = 0.2 / (1.0 + 0.005 * it)
            a_new = spectral_shape_step(a, N, n, top_indices, weights, alpha, rng,
                                        phase_noise_scale=0.01)
            c_check = convolve_self(a_new)
            if c_check.max() < best_score + 0.001:  # Allow small worsenings
                a = a_new

    return best_a, best_score
