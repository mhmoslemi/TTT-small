import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def conv_and_A(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]
    return c, A

def subgrad_toppk(a, N, top_k=32):
    n = len(a)
    c, A = conv_and_A(a, N)
    c_max = c.max()
    k = min(top_k, 2*n-1)

    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

    """LSE-smoothed gradient, optionally blended with top-k subgradient."""
    n = len(a)
    c, A = conv_and_A(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_bc / total
    w_full = np.zeros(N)
    w_full[:2*n-1] = w
    W = fft(w_full)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    lse_val = c_max + np.log(max(total, 1e-300)) / beta
    return grad, lse_val, c, c_max

def backtracking_step(a, direction, S, N, c_max_cur, max_iters=20):
    """Backtracking line search on the true max objective."""
    n = len(a)
    alpha = 2.0 * S / (n * np.linalg.norm(direction))
    if alpha < 1e-15:
        return a, c_max_cur
    best_a = a
    best_cmax = c_max_cur
    for i in range(max_iters):
        candidate = project_simplex(a + alpha * direction, S)
        c_cand, _ = conv_and_A(candidate, N)
        cm = c_cand.max()
        if cm <= c_max_cur * (1.0 - 1e-7) or cm < best_cmax:
            best_a = candidate
            best_cmax = cm
            if cm <= c_max_cur:
                return best_a, best_cmax
        alpha *= 0.5
    return best_a, best_cmax

def spectral_flatten(a, N, alpha=0.1, top_k=10):
    """Reduce spectral peaks by adjusting magnitudes."""
    n = len(a)
    S = a.sum()
    c, A = conv_and_A(a, N)
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P = np.abs(A) ** 2
    P_new = P * (1.0 - alpha * cos_eff)
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

import numpy as np

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = np.where(theta_candidates <= w)[0]
    if len(valid) == 0:
        rho = n
    else:
        rho = valid[-1] + 1
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv_fft(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]
    return c, A, a_pad

def get_subgradient(a, a_pad, A, c, top_k, N):
    n = len(a)
    total_len = 2*n - 1
    k = min(top_k, total_len)
    if k == total_len:
        w = c / c.sum()
    else:
        top_idx = np.argpartition(c, -k)[-k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            return np.zeros(n)
        w_full = np.zeros(total_len)
        w_full[top_idx] = top_vals / total
    W = fft(w_full, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

"""
Hybrid optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2 via:
- Top-k subgradient descent with Nesterov momentum (time domain)
- Frequency-domain peak defocusing (spectral water-filling)
- Adaptive multi-scale (top_k, step size, momentum) scheduling
- Multi-start with diverse inits (random, chirp, CAZAC-projected, exponential spectral)
- Multi-n exploration (4096, 8192, 16384) to find best tradeoff length
- Final high-precision polish with increasing top-k and shrinking steps
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = np.where(theta_candidates <= w)[0]
    if len(valid) == 0:
        rho = n
    else:
        rho = valid[-1] + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_fft(a, N):
    """Self-convolution via FFT. Returns (c, A, a_pad)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A, a_pad


def subgrad_topk(a, A, c, top_k, N):
    """Subgradient from top-k peaks of convolution."""
    n = len(a)
    total_len = 2 * n - 1
    k = min(top_k, total_len)
    if k < total_len:
        top_idx = np.argpartition(c, -k)[-k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            return np.zeros(n)
        w_full = np.zeros(total_len)
        w_full[top_idx] = top_vals / total
    else:
        w_full = c / c.sum()
    W = fft(w_full, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_subgrad(a, A, c, beta, N):
    """LSE-smoothed subgradient."""
    n = len(a)
    total_len = 2 * n - 1
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w_full = np.zeros(total_len)
        w_full[np.argmax(c)] = 1.0
    else:
        w_full = np.zeros(total_len)
        w_full[:total_len] = exp_bc / total
    W = fft(w_full, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = c_max + np.log(max(total, 1e-300)) / beta
    return grad, lse_val


def spectral_defocus(a, A, c, N, alpha):
    """Reduce peak of convolution by adjusting spectral magnitudes."""
    n = len(a)
    S = a.sum()
    m_star = int(np.argmax(c))
    k_arr = np.arange(N, dtype=np.float64)
    angles = np.angle(A)
    contrib = np.abs(A) ** 2 * np.cos(2.0 * angles + 2.0 * np.pi * k_arr * m_star / N)
    max_abs = np.max(np.abs(contrib))
    if max_abs < 1e-20:
        return a
    P = np.abs(A) ** 2
    P_new = P * (1.0 - alpha * contrib / max_abs)
    P_new = np.maximum(P_new, 1e-12)
    for k_idx in range(1, N // 2 + 1):
        P_new[N - k_idx] = P_new[k_idx]
    A_new = np.sqrt(P_new) * np.exp(1j * angles)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def make_init_random(n, rng):
    return rng.random(n) + 0.01


def make_init_chirp(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * t + 0.7 * t * t)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_cazac(n, rng, N):
    """CAZAC-like: flat magnitude spectrum, random phases, project to non-negative."""
    P = np.ones(N)
    P[0] = float(n)
    for k_idx in range(1, N // 2):
        P[N - k_idx] = P[k_idx]
    phase = rng.uniform(0, 2 * np.pi, N)
    phase[N // 2] = 0.0
    for k_idx in range(1, N // 2):
        phase[N - k_idx] = -phase[k_idx]
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    return np.abs(a) + 0.01


def make_init_exp_spec(n, rng, N):
    """Exponential spectral decay."""
    k_arr = np.arange(N, dtype=np.float64)
    P = np.exp(-3.0 * k_arr / N * 2.0)
    P[0] = float(n)
    for k_idx in range(1, N // 2):
        P[N - k_idx] = P[k_idx]
    phase = rng.uniform(0, 2 * np.pi, N)
    phase[N // 2] = 0.0
    for k_idx in range(1, N // 2):
        phase[N - k_idx] = -phase[k_idx]
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    return np.abs(a) + 0.01


def optimize_td(a_init, n, S, N, max_time, rng):
    """Time-domain top-k subgradient with Nesterov momentum."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    y = a.copy()
    t_param = 1.0
    v = np.zeros(n)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c, A, _ = compute_conv_fft(y, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = y.copy()

        top_k = min(256, 4 + it // 8)
        if it < 200:
            grad = subgrad_topk(y, A, c, top_k, N)
        else:
            beta = min(100.0, 5.0 + it * 0.1)
            grad, _ = lse_subgrad(y, A, c, beta, N)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 2.0 * S / (n * (1.0 + t_param))
        v = 0.9 * v + 0.1 * (grad / gnorm * step)
        y_new = project_simplex(y - v, S)

        c_new, _, _ = compute_conv_fft(y_new, N)
        if c_new.max() > c_max * 1.01:
            v = v * 0.5
            y_new = project_simplex(y - v, S)
            c_new, _, _ = compute_conv_fft(y_new, N)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        beta_mom = (t_param - 1.0) / t_new
        y_next = y_new + beta_mom * (y_new - y)
        y_next = np.maximum(y_next, 0.0)
        y = project_simplex(y_next, S)
        t_param = t_new

        if it % 300 == 150 and rng.random() < 0.4:
            y = project_simplex(y + 0.01 * S / np.sqrt(n) * rng.standard_normal(n), S)
            v = np.zeros(n)
            t_param = 1.0

    c_final, _, _ = compute_conv_fft(best_a, N)
    return best_a, c_final.max()


def optimize_spectral(a_init, n, S, N, max_time, rng):
    """Frequency-domain peak defocusing."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c, A, _ = compute_conv_fft(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        alpha = 0.03 + 0.15 * max(0.0, (c_max - 1.0) / max(c_max, 1e-10))
        a_cand = spectral_defocus(a, A, c, N, alpha)
        c_cand, _, _ = compute_conv_fft(a_cand, N)
        if c_cand.max() < c_max:
            a = a_cand
        else:
            a_cand2 = spectral_defocus(a, A, c, N, alpha * 0.2)
            c_cand2, _, _ = compute_conv_fft(a_cand2, N)
            if c_cand2.max() < c_max:
                a = a_cand2
            elif it % 100 == 50 and rng.random() < 0.3:
                a = project_simplex(a + 0.008 * S / np.sqrt(n) * rng.standard_normal(n), S)

    c_final, _, _ = compute_conv_fft(best_a, N)
    return best_a, c_final.max()


def final_polish(a_init, n, S, N, max_time, rng):
    """High-precision final polish: increasing top-k, shrinking steps."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c, A, _ = compute_conv_fft(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        top_k = min(2 * n - 1, 32 + it // 5)
        grad = subgrad_topk(a, A, c, top_k, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = S / (n * gnorm) * max(0.05, 0.5 / (1.0 + 0.03 * it))
        direction = -grad / gnorm
        a_cand = project_simplex(a + step * direction, N and S)
        c_cand, _, _ = compute_conv_fft(a_cand, N)
        if c_cand.max() < c_max:
            a = a_cand
        else:
            a_cand2 = project_simplex(a + step * 0.1 * direction, S)
            c_cand2, _, _ = compute_conv_fft(a_cand2, N)
            if c_cand2.max() < c_max:
                a = a_cand2

    c_final, _, _ = compute_conv_fft(best_a, N)
    return best_a, c_final.max()


def propose_candidate(seed=42, budget_s=2400):

    global_a = None
    global_n = 0

        a_norm = project_simplex(np.clip(a, 0, 1000), S)

        if obj < global_best:
