"""

Strategy: Adaptive multi-phase optimization:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Enforce palindromic symmetry."""
    return 0.5 * (a + a[::-1])


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - c.min() + 1e-10, 1e-10), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_subgradient(y, c, tol_frac=0.01):
    """Subgradient from lags within tolerance of max, weighted by excess."""
    n = len(y)
    max_val = c.max()
    min_val = c.min()
    range_val = max_val - min_val
    if range_val < 1e-15:
        active = np.array([int(np.argmax(c))])
        w = np.array([1.0])
    else:
        active = np.where(c >= max_val - tol_frac * range_val)[0]
        w = (c[active] - max_val + tol_frac * range_val)
        w = np.maximum(w, 1e-10)
        w = w / w.sum()
    w_full = np.zeros(len(c))
    w_full[active] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def upsample(a, factor=2):
    """Upsample using linear interpolation and renormalize."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_initialization(n, rng, style='cosine'):
    """Create diverse initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if style == 'cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif style == 'gaussian':
        center = n / 2
        sigma = n / 6
        a = np.exp(-0.5 * ((tt * n - center) / sigma) ** 2)
    elif style == 'raised_cos':
        a = np.maximum(0, np.cos(np.pi * (tt - 0.5))) ** 2
    elif style == 'bimodal':
        a = np.exp(-0.5 * ((tt * n - n * 0.25) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((tt * n - n * 0.75) / (n * 0.1)) ** 2)
    elif style == 'uniform_perturbed':
        a = np.ones(n) * 0.5 + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    a = np.maximum(a, 0)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    a = symmetrize(a)
    a = project_simplex(a, S)
    return a

            global_best_a = a_candidate.copy()
            print("  NEW BEST: obj=%.8f (n=%d, t=%.0fs)" %
                  (obj_c, len(a_candidate), time.monotonic() - total_start))

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Parent: n=%d, obj=%.8f" % (n0, obj0))
        update_best(a_start)
    else:
        n0 = 8192
        a_start = make_initialization(n0, rng, 'cosine')
        print("Fallback init: n=%d" % n0)
        update_best(a_start)

    remaining = deadline - time.monotonic()

    # === Phase 1: Multi-start FISTA at original size (35% time) ===
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: Multi-start FISTA at n=%d (budget %.0fs) ===" % (n0, p1_budget))

    styles = ['cosine', 'gaussian', 'raised_cos', 'bimodal', 'uniform_perturbed']
    n_starts = min(len(styles), max(2, int(p1_budget / 30)))

    for i in range(n_starts):
        if time.monotonic() > deadline - 60:
            break
        style = styles[i % len(styles)]
        if i == 0 and a_start is not None:
            a_init = a_start.copy()
        else:
            a_init = make_initialization(n0, rng, style)
        sub_budget = (deadline - 60 - time.monotonic()) / (n_starts - i)
        sub_budget = min(sub_budget, p1_budget / max(1, n_starts - i) * 1.5)
        print("  Start %d (%s): budget %.0fs" % (i, style, sub_budget))

        a_opt, obj_opt = run_fista_adaptive(a_init, sub_budget, rng,
                                             beta_init=15.0, beta_rate=0.004,
                                             beta_max=5000.0, k_init=1, k_max=48)
        update_best(a_opt)

    remaining = deadline - time.monotonic()

    # === Phase 2: Active-set polish (25% time) ===
    p2_budget = remaining * 0.25
    print("\n=== Phase 2: Active-set polish at n=%d (budget %.0fs) ===" %
          (len(global_best_a), p2_budget))

    a_polished, obj_polished = run_active_polish(global_best_a, p2_budget, rng)
    update_best(a_polished)

    remaining = deadline - time.monotonic()

    # === Phase 3: Coordinate descent (20% time) ===
    p3_budget = remaining * 0.20
    print("\n=== Phase 3: Coordinate descent at n=%d (budget %.0fs) ===" %
          (len(global_best_a), p3_budget))

    a_cd, obj_cd = coord_descent_polish(global_best_a, p3_budget, rng)
    update_best(a_cd)

    remaining = deadline - time.monotonic()

    # === Phase 4: Multi-scale (20% time) ===
    if remaining > 120 and n0 <= 16384:
        p4_budget = remaining * 0.30
        n2 = n0 * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_budget))

        a2_init = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())
        a2_init = symmetrize(a2_init)
        a2_init = project_simplex(a2_init, S2)

        # FISTA at 2n
        a2_opt, obj2 = run_fista_adaptive(a2_init, p4_budget * 0.6, rng,
                                           beta_init=10.0, beta_rate=0.004,
                                           beta_max=3000.0, k_init=1, k_max=32)
        update_best(a2_opt)

        # Polish at 2n
        rem4 = p4_budget - (time.monotonic() - (deadline - remaining))
        if rem4 > 30:
            a2_pol, obj2p = run_active_polish(a2_opt, min(rem4, remaining * 0.15), rng)
            update_best(a2_pol)

    remaining = deadline - time.monotonic()

    # === Phase 5: Final intensive polish (remaining time) ===
    if remaining > 60:
        p5_budget = remaining * 0.7
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_budget)

        # High-precision active polish
        a_final, obj_final = run_active_polish(global_best_a, p5_budget * 0.5, rng)
        update_best(a_final)

        rem5 = p5_budget - (time.monotonic() - (deadline - remaining))
        if rem5 > 20:
            # Coordinate descent on the best
            a_cd2, obj_cd2 = coord_descent_polish(global_best_a, rem5 * 0.6, rng)
            update_best(a_cd2)

            # One more round of active polish
            rem5b = rem5 - (time.monotonic() - (deadline - remaining - rem5))
            if rem5b > 10:
                a_fp, obj_fp = run_active_polish(global_best_a, rem5b, rng)
                update_best(a_fp)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - total_start))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected result")

    # Fallback
    n_fb = 8192
    a_fb = make_initialization(n_fb, rng, 'cosine')
    final_list = [float(x) for x in a_fb]
    print("Using fallback: obj=%.6f" % compute_obj(a_fb))
    return final_list
