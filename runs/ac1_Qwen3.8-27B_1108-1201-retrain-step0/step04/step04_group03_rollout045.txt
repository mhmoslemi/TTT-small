```python
def solve_lp_separation(f, n, S, M, max_rounds=10, time_limit=5):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0
    Using a separation oracle (cutting plane method)."""
    start = time.monotonic()
    
    # Start with top-50 constraints
    c = fft_convolve_self(f)
    active_idx = np.argsort(c)[::-1][:50].tolist()
    
    g_best = None
    best_sum = 0
    
    for round in range(max_rounds):
        if time.monotonic() - start > time_limit:
            break
        
        # Build constraint matrix for active constraints
        A = np.zeros((len(active_idx), n))
        for i, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            A[i, js] = f[k - js]
        
        # Solve LP
        g_var = cp.Variable(n)
        constraints = [A @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        prob.solve(solver='MOSEK', verbose=False)
        
        if g_var.value is None:
            break
        
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        
        if g_sum > best_sum:
            best_sum = g_sum
            g_best = g.copy()
        
        # Check separation: is there a violated constraint?
        # Compute (f*g)(k) for all k
        fg = np.convolve(f, g)
        violation = fg.max() - M
        
        if violation < 1e-6 * M:
            break  # No significant violation
        
        # Add the most violated constraint
        worst_k = int(np.argmax(fg))
        if worst_k not in active_idx:
            active_idx.append(worst_k)
        else:
            # Already in active set, add next most violated
            fg_sorted = np.argsort(fg)[::-1]
            for k in fg_sorted:
                if k not in active_idx:
                    active_idx.append(int(k))
                    break
    
    if g_best is None:
        return None
    return g_best
```

"""
Minimize 2*n*max(f*f)/sum(f)^2 using LP best-response iterations
with a separation oracle for exact LP solutions, combined with
polishing via coordinate descent and FISTA.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def build_constraint_row(f, k, n):
    """Build row k of the constraint matrix: A[k, j] = f[k-j] if valid."""
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    js = np.arange(lo, hi + 1)
    row[js] = f[k - js]
    return row


def solve_lp_best_response(f, n, M, max_rounds=15, time_budget=8.0, top_init=50):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k=0..2n-2, g >= 0
    Using a separation oracle (cutting plane) approach.
    Returns the optimal g, or None on failure.
    """
    start = time.monotonic()
    
    # Initialize with top-k most binding constraints
    c = fft_convolve_self(f)
    active_idx = list(np.argsort(c)[::-1][:top_init])
    
    g_sol = None
    best_sum = -1
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        # Build constraint matrix
        m = len(active_idx)
        A_rows = np.zeros((m, n))
        for i, k in enumerate(active_idx):
            A_rows[i] = build_constraint_row(f, k, n)
        
        # Solve LP
        g_var = cp.Variable(n)
        constraints = [A_rows @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break
        
        if g_var.value is None:
            break
        
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        
        if g_sum > best_sum:
            best_sum = g_sum
            g_sol = g.copy()
        
        # Separation check: find most violated constraint
        fg = fft_convolve(f, g)
        violations = fg - M
        max_viol = violations.max()
        
        if max_viol < 1e-4 * M:
            break  # Feasible within tolerance
        
        # Add most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active_idx:
            active_idx.append(worst_k)
        else:
            # Add second most violated
            sorted_k = np.argsort(violations)[::-1]
            added = False
            for k in sorted_k:
                k = int(k)
                if k not in active_idx:
                    active_idx.append(k)
                    added = True
                    break
            if not added:
                break
    
    return g_sol


def lp_iteration(f, n, S, time_budget_per_lp=5.0):
    """
    One LP best-response iteration.
    Returns (new_f, new_M, improved).
    """
    c = fft_convolve_self(f)
    M = c.max()
    
    g = solve_lp_best_response(f, n, M, time_budget=time_budget_per_lp)
    if g is None or g.sum() < 1e-10:
        return f.copy(), M, False
    
    # Normalize g to have sum S
    g = g * (S / g.sum())
    
    # Line search for optimal mixing parameter t
    best_t = 0.0
    best_val = M
    
    # Try various t values
    t_candidates = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0])
    
    for t in t_candidates:
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        s = f_test.sum()
        if s < 1e-15:
            continue
        f_test = f_test * (S / s)
        c_test = fft_convolve_self(f_test)
        val = c_test.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t == 0:
        return f.copy(), M, False
    
    f_new = (1 - best_t) * f + best_t * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new * (S / f_new.sum())
    
    return f_new, best_val, True


def coordinate_descent_polish(f, n, max_rounds=3, time_budget=60.0):

    """

            break
        
        full_conv = fft_convolve_self(f)
        improved = False
        
        for i in range(n):
            old = f[i]
            
            # Compute b_k = sensitivity of (f*f)(k) to f_i
            # For k=0: (f*f)(0) = sum_j f_j^2, so (f*f)(0) = (full_conv[0] - old^2) + x^2
            # For k!=0: (f*f)(k) = (full_conv[k] - old*b_k) + x*b_k
            # where b_k = f_{i+k}(if valid) + f_{i-k}(if valid, k!=0)
            
            b = np.zeros(2 * n - 1)
            # For k >= 1: b_k = f[i+k] (if i+k < n) + f[i-k] (if i-k >= 0)
            for k in range(1, 2 * n - 1):
                val = 0.0
                if i + k < n:
                    val += f[i + k]
                if i - k >= 0:
                    val += f[i - k]
                b[k] = val
            # b[0] is special: (f*f)(0) has x^2 term
            # (f*f)(0) with f_i = x: (full_conv[0] - old^2) + x^2
            
            # c_k = (f*f)_{-i}(k) = full_conv[k] - old * b_k (for k != 0)
            # c_0 = full_conv[0] - old^2
            
            # Objective to minimize: max( c_0 + x^2, max_{k!=0}(c_k + x*b_k) )
            # subject to x >= 0, and we want sum to remain S (so we'll renormalize)
            
            # For efficiency, let's just evaluate at a few candidate x values
            # The optimal x* satisfies: the max is minimized
            # Try x = 0, x = old, and the "balancing" value
            
            c_0 = full_conv[0] - old * old
            c_nonzero = full_conv[1:] - old * b[1:]
            
            def obj_at_x(x):
                val0 = c_0 + x * x
                val_rest = c_nonzero + x * b[1:]
                return max(val0, val_rest.max())
            
            # Find optimal x by ternary search (the function is convex in x >= 0)
            # First, find the range
            x_lo = 0.0
            x_hi = old * 2 + 1.0  # upper bound

            if obj_at_x(0.0) <= obj_at_x(x_hi * 0.01):

            if abs(x_opt - old) > 1e-10:

        s = f.sum()
