```python
"""
Minimize 2*n*max_conv/sum^2 using Frank-Wolfe (conditional gradient) method
with LP-based vertex updates and line search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def convolution_with(a, b):
    """Compute full convolution of a and b (different sequences)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]

def frank_wolfe_lp(a, active_lags, S):
    """
    Solve the Frank-Wolfe LP:
    minimize t
    subject to: sum_i x[i] * a[k-i] <= t for k in active_lags
                sum_i x[i] = S
                x[i] >= 0
    
    Returns x (the LP solution).
    """
    n = len(a)
    K = len(active_lags)
    
    # Variables: x[0], ..., x[n-1], t
    # Total: n + 1 variables
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: A_ub @ vars <= b_ub
    # For each active lag k: sum_i x[i] * conv_coeff[k, i] - t <= 0
    # where conv_coeff[k, i] = a[k-i] (with a[j]=0 for j outside [0,n-1])
    
    # Build constraint matrix
    # Row k: [a[k-0], a[k-1], ..., a[k-(n-1)], -1] <= 0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for idx, k in enumerate(active_lags):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, i] = a[j]
        A_ub[idx, -1] = -1.0  # -t
    
    # Equality constraint: sum x = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: x[i] >= 0, t unbounded (or t >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    # Solve LP
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    else:
        print("  LP failed:", result.message)
        return a.copy()  # fallback
```

```python
def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    """Full convolution of two sequences via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1

    B = fft(b, fsize)

def solve_lp_oracle(a, active_lags, S):
    """
    Solve the LP: minimize t
    subject to: (y*a_full)[k] <= t for k in active_lags
                sum(y) = S
                y >= 0
    Returns y (the Frank-Wolfe vertex).
    """
    n = len(a)
    K = len(active_lags)
    
    # Build constraint matrix for the LP
    # Variables: y[0], ..., y[n-1], t (total n+1)
    # For each active lag k: sum_i y[i]*a[k-i] - t <= 0
    # where a[j] = 0 for j outside [0, n-1]
```

```python
FISTA warm-up, LP-based FW iterations, and subgradient polishing.

    n = len(a)

    fsize = 1 << (size - 1).bit_length()

def fft_convolve(a, b):
    """Full convolution of two sequences via FFT."""
    n = len(a)

    """
    Solve LP: minimize t
    subject to: (y*a)[k] <= t for k in active_lags
                sum(y) = S
                y >= 0
    Returns y or None on failure.
    """
    n = len(a)
    K = len(active_lags)

    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)

    for k in active:
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            grad[i] += a[k - i]
    grad /= K_actual

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy()

    step = 1.0 / gnorm
    new_a = np.maximum(a - step * grad, 0.0)
    s = new_a.sum()
    if s > 1e-15:
        new_a = new_a * (S / s)
    return new_a


def fista_phase(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.005, beta_max=5000.0):
    """FISTA with LSE smoothing for warm-up."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = 1.0 / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

        # LSE gradient
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            grad = np.zeros(n)
        else:
            w = e / tot
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Projected FISTA update
        x_new = np.maximum(y - step * grad, 0.0)
        x_new = x_new * (S / (x_new.sum() + 1e-30))
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def fw_lp_phase(a_init, S, max_time, rng, K_init=20, K_max=100):
    """
    Frank-Wolfe with LP-based oracle.
    Returns (best_a, best_conv).
    """
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    K = K_init
    lp_fail_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive K
        if stuck > 50 and K < K_max:
            K = min(K_max, K + 5)
        elif stuck > 300 and K < K_max + 50:
            K = min(K_max + 50, K + 10)

        # Try LP oracle
        top_idx = np.argsort(c)[::-1][:K]
        active_lags = sorted(top_idx.tolist())

        t_start = time.monotonic()
        d = solve_lp_oracle(a, active_lags, S)
        lp_time = time.monotonic() - t_start

        if d is not None:
            lp_fail_count = 0
            c_dir = fft_convolve(d, a)
            t = line_search_conv(c, c_dir)

            if t > 1e-6:
                new_a = (1.0 - t) * a + t * d
            else:
                new_a = a.copy()
        else:
            lp_fail_count += 1
            # Fallback: subgradient step
            new_a = subgradient_step(a, S, K, tol_frac=0.02)

        # Additional fallback if no improvement
        if np.max(fft_convolve_self(new_a)) >= cur_max * (1.0 - 1e-10):
            if lp_fail_count > 5:
                # Use gradient step
                new_a = subgradient_step(a, S, min(K, 10), tol_frac=0.01)
                lp_fail_count = 0

        a = np.maximum(new_a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)

        # Periodic reporting
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_conv / (S * S)
            print("  FW-LP it=%d, K=%d, stuck=%d, max_conv=%.8f, obj=%.6f, lp_t=%.3fs, elapsed=%.0fs"
                  % (it, K, stuck, best_conv, obj, lp_time, elapsed))

        # Perturbation when very stuck
        if stuck > 500 and it % 100 == 0:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a = np.maximum(a + noise, 0.0)
            a = a * (S / a.sum())
            stuck = 0
            last_improve = it

    return best_a, best_conv


def polish_phase(a_init, S, max_time, rng):
    """
    Intensive polishing using exact subgradient with decreasing step.
    """
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()

        k_max = int(np.argmax(c))
        # Gradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(k_max, n - 1)
        for i in range(i_lo, i_hi + 1):
            grad[i] = a[k_max - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.01 * it))
        new_a = np.maximum(a - step * grad, 0.0)
        s = new_a.sum()
        if s > 1e-15:
            new_a = new_a * (S / s)
        a = new_a

        if it % 500 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_conv / (S * S)
            print("  Polish it=%d, max_conv=%.8f, obj=%.6f, elapsed=%.0fs"
                  % (it, best_conv, obj, elapsed))

    return best_a, best_conv


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA warm-up, LP-based Frank-Wolfe, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n)

    n = len(a_start)
    S = np.sqrt(2.0 * n)
    a = a_start.copy()
    a = a * (S / a.sum())

    obj0 = compute_obj(a)
    print("Initial: n=%d, obj=%.6f" % (n, obj0))
    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a.copy()

    # === Phase 1: FISTA warm-up (15% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.15
    print("\n=== Phase 1: FISTA warm-up (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    a1, conv1 = fista_phase(a, S, p1_budget, rng, beta_init=30.0, beta_rate=0.008, beta_max=5000.0)
    obj1 = compute_obj(a1)
    print("  FISTA: obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: LP-based Frank-Wolfe (50% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.50
    print("\n=== Phase 2: FW-LP (budget %.0fs) ===" % p2_budget)
    t2 = time.monotonic()

    a2, conv2 = fw_lp_phase(global_best_a, S, p2_budget, rng, K_init=15, K_max=80)
    obj2 = compute_obj(a2)
    print("  FW-LP: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
    if obj2 < global_best_obj:
        global_best_obj = obj2
        global_best_a = a2.copy()

    # Continue FW-LP with updated K
    remaining2 = deadline - time.monotonic()
    if remaining2 > 60:
        p2b_budget = remaining2 * 0.30
        print("\n=== Phase 2b: FW-LP continued (budget %.0fs) ===" % p2b_budget)
        t2b = time.monotonic()
        a2b, conv2b = fw_lp_phase(global_best_a, S, p2b_budget, rng, K_init=40, K_max=150)
        obj2b = compute_obj(a2b)
        print("  FW-LP (K=40): obj=%.6f (t=%.0fs)" % (obj2b, time.monotonic() - t2b))
        if obj2b < global_best_obj:
            global_best_obj = obj2b
            global_best_a = a2b.copy()

    # === Phase 3: Subgradient polish (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.25
        print("\n=== Phase 3: Polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a3, conv3 = polish_phase(global_best_a, S, p3_budget, rng)
        obj3 = compute_obj(a3)
        print("  Polish: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Final FW-LP round (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.80
        print("\n=== Phase 4: Final FW-LP (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, conv4 = fw_lp_phase(global_best_a, S, p4_budget, rng, K_init=20, K_max=60)
        obj4 = compute_obj(a4)
        print("  Final FW-LP: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            S_fb = np.sqrt(2.0 * n_fb)
            global_best_a = np.ones(n_fb) * (S_fb / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_best_a = np.ones(n_fb) * (S_fb / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
```
