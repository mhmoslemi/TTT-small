Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses power-sum FISTA with curriculum + subgradient polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def power_sum_value(b, p):
    vals = b ** p
    s = vals.sum()
    if s < 1e-300:
        return 0.0
    return s ** (1.0 / p)


def power_sum_gradient(a, b, p):
    n = len(a)
    F = power_sum_value(b, p)
    if F < 1e-300:
        return np.zeros(n)
    # cross-correlation: sum_j b[m+j] * a[j] for m=0..n-1
    fsize = 1 << (3 * n - 3).bit_length()
    B = fft(b, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(B * np.conj(A)))[:n]
    return 2.0 * cross / (F ** (p - 1))


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    best_conv = np.inf

    it = 0
    step = 1.0
    p = 2.0
    
    if p_schedule is None:
        p_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Power-sum FISTA with p-curriculum, exact subgradient polishing,
and multi-start exploration. Uses cross-correlation-based gradients for
the power-sum objective, which is smoother and more stable than LSE.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def power_sum_value(b, p):
    """Compute (sum b^p)^(1/p)."""
    if p <= 1.0 + 1e-10:
        return b.sum()
    maxb = b.max()
    if maxb < 1e-300:
        return 0.0
    scaled = b / maxb
    vals = scaled ** p
    s = vals.sum()
    if s < 1e-300:
        return 0.0
    return maxb * s ** (1.0 / p)


def power_sum_gradient(a, b, p):
    """
    Gradient of F_p(a) = (sum (conv(a,a))^p)^(1/p) w.r.t. a.
    Uses cross-correlation: grad[m] = 2/F^(p-1) * sum_j b[m+j]^... * a[j]
    Actually: grad[m] = 2 * F_p^{1-p} * sum_j b[m+j]^{p-1} * a[j]
    """
    n = len(a)
    F = power_sum_value(b, p)
    if F < 1e-300:
        return np.zeros(n)
    if p <= 1.0 + 1e-10:
        # p=1: F = sum(b), grad[m] = 2 * sum_j b[m+j] * a[j] / F... 
        # Actually for p=1: dF/da[m] = 2 * sum_j b[m+j]^0 * a[j] / F = 2*sum_j a[j] / F... 
        # No, that's wrong. For p=1: F = sum(b) = (sum a)^2, 
        # dF/da[m] = 2*sum(a) which is constant. Not useful.
        # Just use the general formula.
        pass
    # For p > 1:
    # grad[m] = 2 * F^{1-p} * sum_j b[m+j]^{p-1} * a[j]
    bp1 = b ** (p - 1)  # element-wise power
    # Cross-correlation: cross[m] = sum_j bp1[m+j] * a[j]
    fsize = 1 << (len(bp1) + n - 1 - 1).bit_length()
    B = fft(bp1, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(B * np.conj(A)))[:n]
    return 2.0 * np.float64(F) ** (1.0 - p) * cross


def exact_subgradient(a, b, tol_frac=0.0):
    """
    Subgradient of max(b) w.r.t. a using active lags within tol_frac of max.
    grad[m] = 2 * (1/|A|) * sum_{k in A} a[k-m] (reflected/shifted)
    """
    n = len(a)
    max_val = b.max()
    if tol_frac > 0:
        active = np.where(b >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(b))])
    # grad[m] = 2/|A| * sum_{k in A} a[k-m] for valid k-m
    # = 2/|A| * cross-correlation of indicator_A with a
    # where indicator_A[k] = 1 for k in active
    ind = np.zeros(len(b))
    ind[active] = 1.0 / len(active)
    fsize = 1 << (len(ind) + n - 1 - 1).bit_length()
    I = fft(ind, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(I * np.conj(A)))[:n]
    return 2.0 * cross


def topk_subgradient(a, b, k, power=1.0):
    """Subgradient using top-k lags with power weighting."""
    n = len(a)
    k = min(k, len(b))
    idx = np.argsort(b)[::-1][:k]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.maximum(b[idx], 1e-30) ** (power - 1.0)
        w = w / w.sum()
    ind = np.zeros(len(b))
    ind[idx] = w
    fsize = 1 << (len(ind) + n - 1 - 1).bit_length()
    I = fft(ind, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(I * np.conj(A)))[:n]
    return 2.0 * cross


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a, S=None):
    """Normalize to have sum S."""
    n = len(a)
    if S is None:
        S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def lse_gradient(a, b, beta):
    """LSE smoothing gradient for comparison."""
    n = len(a)
    cm = b.max()
    shifted = beta * (b - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_subgradient_polish(a_init, max_time, rng, mode="adaptive"):

    last_improve = 0
    step = S * 0.1

    while time.monotonic() - start < max_time:

            if stuck < 100:

def make_raised_cosine(n):
    """Create a raised-cosine shaped sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t))


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian bump sequence."""
    t = np.arange(n) / n - 0.5
    return np.exp(-0.5 * (t / sigma_frac) ** 2)


def make_ramp(n, power=1.0):
    """Create a ramp-like sequence (asymmetric)."""
    t = np.arange(n) / n
    return (1.0 - t) ** power


def upsample_interp(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        n = 8192
        a_start = make_raised_cosine(n)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # Also evaluate structured alternatives
    for name, maker in [("raised_cos", make_raised_cosine), ("gauss", make_gaussian), ("ramp", lambda n: make_ramp(n, 1.5))]:
        alt = maker(n0)
        obj_alt = compute_obj(alt)
        print("  %s: obj=%.8f" % (name, obj_alt))
        if obj_alt < global_best_obj:
            global_best_obj = obj_alt
            global_best_a = alt.copy()
            a_start = alt.copy()

    # === Phase 1: Power-sum FISTA (40% of time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.40
    print("\n=== Phase 1: Power-sum FISTA (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    a1, conv1 = run_fista_power(global_best_a, p1_budget * 0.6, rng,
                                 p_schedule=[2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256],
                                 max_iters_per_p=2000)
    obj1 = compute_obj(a1)
    print("  Power-sum FISTA: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 20:
        # Run LSE FISTA as alternative
        a1b, conv1b = run_lse_fista(global_best_a, remaining1, rng,
                                     beta_init=15.0, beta_rate=0.005, beta_max=8000.0)
        obj1b = compute_obj(a1b)
        print("  LSE FISTA: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Subgradient polish (30% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p2_budget = remaining * 0.30
        print("\n=== Phase 2: Subgradient polish (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, conv2 = run_subgradient_polish(global_best_a, p2_budget * 0.5, rng, mode="adaptive")
        obj2 = compute_obj(a2)
        print("  Adaptive subgrad: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 15:
            a2b, conv2b = run_subgradient_polish(global_best_a, remaining2, rng, mode="topk")
            obj2b = compute_obj(a2b)
            print("  Topk subgrad: obj=%.8f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Power-sum FISTA round 2 with different schedule (20% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Second power-sum FISTA (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        # Higher p schedule for more aggressive peak reduction
        a3, conv3 = run_fista_power(global_best_a, p3_budget * 0.7, rng,
                                     p_schedule=[4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048],
                                     max_iters_per_p=1500)
        obj3 = compute_obj(a3)
        print("  High-p FISTA: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 10:
            a3b, conv3b = run_subgradient_polish(global_best_a, remaining3, rng, mode="adaptive")
            obj3b = compute_obj(a3b)
            print("  Final subgrad: obj=%.8f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Multi-start exploration (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = remaining * 0.10
        print("\n=== Phase 4: Multi-start (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        n_best = len(global_best_a)

        # Try perturbations of current best
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            trial_budget = min(p4_budget / max(1, 3 - trial), 30.0)
            if trial_budget < 10:
                break
            # Create perturbed version
            noise_scale = 0.05 * np.sqrt(2.0 * n_best) / np.sqrt(n_best)
            perturbed = global_best_a + rng.normal(0, noise_scale, n_best)
            perturbed = np.maximum(perturbed, 0)
            perturbed = normalize(perturbed)

            a4, _ = run_fista_power(perturbed, trial_budget, rng,
                                     p_schedule=[8, 16, 32, 64, 128, 256],
                                     max_iters_per_p=1000)
            obj4 = compute_obj(a4)
            print("  Trial %d: obj=%.8f" % (trial, obj4))
            if obj4 < global_best_obj:
                global_best_obj = obj4
                global_best_a = a4.copy()

    # === Phase 5: Final intensive polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()

        # Very careful subgradient polish
        a5, conv5 = run_subgradient_polish(global_best_a, remaining * 0.6, rng, mode="adaptive")
        obj5 = compute_obj(a5)
        print("  Final adaptive: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        remaining5 = remaining - (time.monotonic() - t5)
        if remaining5 > 5:
            a5b, conv5b = run_lse_fista(global_best_a, remaining5, rng,
                                         beta_init=100.0, beta_rate=0.001, beta_max=20000.0)
            obj5b = compute_obj(a5b)
            print("  Final LSE: obj=%.8f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Try upsampled version if time permits and n is small enough ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and len(global_best_a) <= 16384:
        n2 = len(global_best_a) * 2
        print("\n=== Upsample to n=%d (budget %.0fs) ===" % (n2, remaining * 0.7))
        t6 = time.monotonic()
        a_up = upsample_interp(global_best_a, factor=2)
        a_up = normalize(a_up)
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.8f" % obj_up)

        a_up2, _ = run_fista_power(a_up, remaining * 0.4, rng,
                                    p_schedule=[4, 8, 16, 32, 64, 128],
                                    max_iters_per_p=2000)
        obj_up2 = compute_obj(a_up2)
        print("  After FISTA: obj=%.8f (t=%.0fs)" % (obj_up2, time.monotonic() - t6))
        if obj_up2 < global_best_obj:
            global_best_obj = obj_up2
            global_best_a = a_up2.copy()

        remaining6 = remaining * 0.7 - (time.monotonic() - t6)
        if remaining6 > 20:
            a_up3, _ = run_subgradient_polish(a_up2, remaining6, rng, mode="adaptive")
            obj_up3 = compute_obj(a_up3)
            print("  After polish: obj=%.8f" % obj_up3)
            if obj_up3 < global_best_obj:
                global_best_obj = obj_up3
                global_best_a = a_up3.copy()

    # Final verification
    print("\n=== Final Result ===")
    print("Best objective: %.8f (n=%d)" % (global_best_obj, len(global_best_a)))
    print("Total time: %.0fs" % (time.monotonic() - t0))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.8f" % ver)
        else:
            print("WARNING: evaluation failed, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
