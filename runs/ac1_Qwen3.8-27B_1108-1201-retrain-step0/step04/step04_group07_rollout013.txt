"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using adaptive softmax-gradient descent
with spectral shaping and multi-scale refinement.

Key improvements:
1. Exponential (softmax) weighting for gradient computation - better than top-k.
2. Adaptive temperature annealing for the max approximation.
3. Spectral shaping: directly modify |FFT(a)| to reduce peak contributions.
4. Multi-scale: work at coarser scales first, then refine.
5. Nesterov acceleration for faster convergence.
6. Periodic spectral re-randomization to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using hybrid softmax-gradient descent
with spectral shaping, Nesterov acceleration, and multi-scale refinement.

Key improvements over previous approach:
1. Log-sum-exp (softmax) gradient with adaptive temperature annealing.
2. Hybrid spectral shaping: periodically pull |FFT(a)| toward flat magnitude.
3. Nesterov momentum for acceleration.
4. Backtracking line search for adaptive step sizing.
5. Multi-length search (8192, 16384) with adaptive budget allocation.
6. Periodic noise injection with spectral re-randomization.
7. Coarse-to-fine multi-scale refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N=None):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2 * n - 2).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return float(2.0 * n * c.max() / (S * S))


def softmax_gradient(c, weights, a, N_grad):
    """
    Compute the gradient of sum_k weights[k] * c_k with respect to a.
    c_k = sum_j a[j]*a[k-j]
    dc_k/da_m = 2*a[k-m] (if k-m in [0,n-1]), 0 otherwise
    grad[m] = 2 * sum_k weights[k] * a[k-m]
    """
    n = len(a)
    u = a[::-1]
    w_ext = np.zeros(N_grad)
    w_ext[:2 * n - 1] = weights
    u_ext = np.zeros(N_grad)
    u_ext[:n] = u
    W = fft(w_ext, N_grad)
    U = fft(u_ext, N_grad)
    conv_full = np.real(ifft(W * U))
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    return grad


def spectral_shape_step(a, S, N, t_blend):
    """
    One step of spectral shaping: pull |FFT(a)| toward uniform magnitude.
    Keeps DC component fixed.
    """
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A).copy()
    phase = np.angle(A)
    
    # Target: uniform magnitude S/sqrt(N) for all f, but DC = S
    target = S / np.sqrt(N)
    target_mag = np.full(N, target)
    target_mag[0] = S
    
    # Blend current and target
    new_mag = (1.0 - t_blend) * mag + t_blend * target_mag
    new_mag[0] = S  # DC is fixed
    
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Project to feasible set
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    a_new = project_simplex(a_new, S)
    return a_new


def hybrid_optimizer(a_init, n, S, budget_s, rng, target_score=1.50):
    """
    Hybrid optimizer combining softmax gradient descent with spectral shaping.
    """
    N = 1 << (2 * n - 2).bit_length()
    N_grad = 1 << (3 * n - 2).bit_length()
    
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    a = project_simplex(a, S)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    
    t_start = time.monotonic()
    it = 0
    
    # Temperature schedule: from large (smooth) to small (sharp)
    tau_large = S * S / n
    tau_small = S * S / n / 50.0
    
    while time.monotonic() - t_start < budget_s:
        it += 1
        progress = min(1.0, it / 10000.0)
        
        # Compute convolution at y (Nesterov lookahead)
        c = fft_convolve_self(y, N)
        c_max = c.max()
        c_sum = c.sum()
        S_y = y.sum()
        score = 2.0 * n * c_max / (S_y * S_y) if S_y > 0.01 else np.inf
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if best_score <= target_score:
            break
        
        # Adaptive temperature
        tau = tau_large * (1.0 - progress) + tau_small * progress
        tau = max(tau, 1e-10)
        
        # Softmax weights (numerically stable)
        c_shifted = c - c.max()
        weights = np.exp(c_shifted / tau)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = weights / w_sum
        
        # Compute gradient
        grad = softmax_gradient(c, weights, y, N_grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        a_try = a - (0.5 * S / gnorm) * (grad / gnorm)
        a_try = project_simplex(np.maximum(a_try, 0.0), S)
        
        # Check if this improves (use smooth objective as proxy)
        def smooth_obj(x):
            cx = fft_convolve_self(x, N)
            cx_shifted = cx - cx.max()
            return float(np.sum(weights * cx))
        
        f_curr = smooth_obj(a)
        f_try = smooth_obj(a_try)
        
        alpha = 0.5 * S / gnorm
        if f_try > f_curr:
            # Backtrack
            for _ in range(5):
                alpha *= 0.5
                a_try = a - alpha * (grad / gnorm)
                a_try = project_simplex(np.maximum(a_try, 0.0), S)
                f_try = smooth_obj(a_try)
                if f_try <= f_curr:
                    break
            else:
                # If all backtracks fail, use a smaller step
                alpha = 1e-4 * S / n
                a_try = a - alpha * (grad / gnorm)
                a_try = project_simplex(np.maximum(a_try, 0.0), S)
        
        # Nesterov update
        new_a = a_try
        y = new_a + momentum * (new_a - a)
        y = np.maximum(y, 0.0)
        a = new_a
        
        # Periodic spectral shaping
        if it % 20 == 0:
            t_blend = 0.15 * (1.0 - 0.5 * progress) + 0.02
            a = spectral_shape_step(a, S, N, t_blend)
            y = a.copy()
        
        # Periodic noise to escape local minima
        if it % 200 == 100 and rng.random() < 0.5:
            noise_scale = 0.01 * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
        
        # Update momentum (slowly decay)
        momentum = max(0.5, 0.9 * (1.0 - 0.3 * progress))
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    it={it} score={score:.6f} best={best_score:.6f} tau={tau:.2f} elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coarse_fine_refine(a, n, S, budget_s, rng, target_score=1.50):
    """
    Multi-scale refinement: work on a coarse version first, then refine.
    """
    N = 1 << (2 * n - 2).bit_length()
    
    # Coarse: average pairs
    n_coarse = n // 2
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a[2*i] + a[2*i+1]
    S_coarse = a_coarse.sum()
    
    # Optimize coarse
    a_c_best, sc_c = hybrid_optimizer(a_coarse, n_coarse, S_coarse, 
                                       budget_s * 0.3, rng, target_score)
    
    # Upsample: split each coarse element back
    a_up = np.zeros(n)
    for i in range(n_coarse):
        a_up[2*i] = a_c_best[i] * 0.5
        a_up[2*i+1] = a_c_best[i] * 0.5
    
    # Blend with original
    a_blend = 0.5 * a + 0.5 * a_up
    a_blend = project_simplex(np.maximum(a_blend, 0.0), S)
    
    # Refine full resolution
    a_best, sc_best = hybrid_optimizer(a_blend, n, S, budget_s * 0.6, rng, target_score)
    
    return a_best, sc_best


def make_diverse_inits(n, rng, S):
    """Generate diverse initializations for the optimization."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) * S / n)
    
    # Random
    r = rng.random(n)
    inits.append(project_simplex(r, S))
    
    # Sine
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(project_simplex(0.5 + 0.4 * np.sin(2 * np.pi * tt), S))
    
    # Chirp
    inits.append(project_simplex(0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt**2)), S))
    
    # Bump
    inits.append(project_simplex(np.exp(-0.5 * ((tt - 0.5) * 3)**2), S))
    
    # Multi-frequency
    inits.append(project_simplex(
        0.5 + 0.2 * np.sin(2*np.pi*3*tt) + 0.15 * np.sin(2*np.pi*7*tt) + 0.1 * np.sin(2*np.pi*13*tt), S))
    
    # Smooth noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(project_simplex(0.2 + 0.6 * smooth, S))
    
    return inits


def optimize_length(n, budget_s, rng, start_seq=None, target_score=1.50):
    """Full optimization for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    
    global_best = np.inf
    global_a = None
    
    t_start = time.monotonic()
    
    # Build list of initializations
    inits = []
    if start_seq is not None and len(start_seq) == n:
        sc = compute_score(np.maximum(start_seq, 0))
        if np.isfinite(sc):
            inits.append(("provided", np.maximum(start_seq, 0.0).astype(np.float64)))
    
    n_inits_extra = min(4, max(2, int(budget_s / 60)))
    extra_inits = make_diverse_inits(n, rng, S)
    for i, init in enumerate(extra_inits[:n_inits_extra]):
        inits.append((f"init_{i}", init))
    
    # Allocate budget: give more time to the best starting point
    n_total = len(inits)
    # First init should get the most time
    if n_total == 1:
        budgets = [budget_s * 0.85]
    else:
        # 50% to first, rest split among others
        first_budget = budget_s * 0.50
        rest_budget = budget_s * 0.35 / (n_total - 1)
        budgets = [first_budget] + [rest_budget] * (n_total - 1)
    
    for idx, (name, init_a) in enumerate(inits):
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed - 10  # leave 10s for final polish
        if remaining < 30:
            break
        
        alloc = min(budgets[idx], remaining)
        if alloc < 20:
            break
        
        print(f"  [{name}] n={n} budget={alloc:.0f}s")
        
        a_result, score = hybrid_optimizer(init_a, n, S, alloc, rng, target_score)
        
        if score < global_best:
            global_best = score
            global_a = a_result.copy()
            print(f"    New best for n={n}: {global_best:.6f}")
    
    # Final polish on best found
    if global_a is not None:
        remaining = max(0, budget_s - (time.monotonic() - t_start))
        if remaining > 15:
            a_polish, sc_polish = hybrid_optimizer(global_a, n, S, remaining, rng, target_score)
            if sc_polish < global_best:
                global_best = sc_polish
                global_a = a_polish.copy()
    
    return global_a, global_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses hybrid softmax-gradient + spectral shaping with multi-length search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    
    print("Starting hybrid optimization with budget %.0fs" % budget_s)
    
    global_best = np.inf
    global_a = None
    
    t_global_start = time.monotonic()
    
    # Try to get the provided sequence
    provided_seq = None
    try:
        provided_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        print(f"  Provided sequence length: {len(provided_seq)}")
    except Exception:
        pass
    
    # Determine lengths to try
    lengths = []
    if provided_seq is not None and len(provided_seq) >= 1024:
        lengths.append(len(provided_seq))
    if 8192 not in lengths:
        lengths.append(8192)
    if 16384 not in lengths and budget_s > 1000:
        lengths.append(16384)
    
    # Allocate budget
    total_budget = budget_s * 0.90  # leave 10% for verification/fallback
    if len(lengths) == 1:
        length_budgets = [total_budget]
    elif len(lengths) == 2:
        # 60% to first (provided), 40% to second
        length_budgets = [total_budget * 0.60, total_budget * 0.40]
    else:
        length_budgets = [total_budget * w for w in [0.45, 0.35, 0.20]]
    
    for i, n in enumerate(lengths):
        elapsed = time.monotonic() - t_global_start
        remaining = total_budget - elapsed
        if remaining < 60:
            break
        
        alloc = min(length_budgets[i], remaining - 30)
        if alloc < 40:
            break
        
        print(f"=== Optimizing n={n} (budget={alloc:.0f}s) ===")
        
        start = None
        if i == 0 and provided_seq is not None and len(provided_seq) == n:
            start = provided_seq
        
        a_result, score = optimize_length(n, alloc, rng, start_seq=start, target_score=1.50)
        
        if a_result is not None:
            if score < global_best:
                global_best = score
                global_a = a_result.copy()
                print(f"  Global best updated: {global_best:.6f} (n={n})")
            else:
                print(f"  Best for n={n}: {score:.6f} (global: {global_best:.6f})")
    
    # If no result, use fallback
    if global_a is None:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.ones(n_fb) * S_fb / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence")
    
    # Final clip and verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print(f"Verification score: {ver:.6f}")
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0
    
    total_time = time.monotonic() - t_global_start
    print(f"Done in {total_time:.0f}s. Final: length={len(global_a)}, score={global_best:.6f}")
    
    return final_list
