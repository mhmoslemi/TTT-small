def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    cm = c.max()
    x = beta * (c - cm)
    x = np.clip(x, -500, 500)
    return cm + np.log(np.sum(np.exp(x))) / beta

def lse_subgradient(a, c, beta):

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 via:
1. Clean FISTA with LSE smoothing and backtracking
2. Simulated annealing local search
3. Coarse-to-fine multi-resolution
4. Multi-start with diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_max_conv(a):
    c = fft_convolve_self(a)
    return c.max(), c

def lse_value(c, beta):
    cm = c.max()
    x = np.clip(beta * (c - cm), -500, 500)
    return cm + np.log(np.sum(np.exp(x))) / beta

def lse_grad(a, c, beta):
    n = len(a)
    cm = c.max()
    x = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(x)
    w = e / e.sum()
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def objective_val(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def fista_lse(a0, S, time_budget, rng, beta_start=20.0, beta_end=5000.0, verbose=False):
    """FISTA with LSE smoothing and backtracking line search."""
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_val = objective_val(a)
    best_a = a.copy()
    alpha = 0.1 / (beta_start * 2.0)
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        val = c.max()
        s = y.sum()
        obj = 2.0 * n * val / (s * s)
        if obj < best_val:
            best_val = obj
            best_a = y.copy()
            if verbose and it % 500 == 0:
                print("  it=%d beta=%.0f obj=%.6f" % (it, beta, best_val))
        
        # Compute LSE gradient
        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        f_curr = lse_value(c, beta)
        alpha_new = alpha
        for bt in range(30):
            x_new = project_simplex(y - alpha_new * grad, S)
            c_new = fft_convolve_self(x_new)
            f_new = lse_value(c_new, beta)
            # Armijo condition
            if f_new <= f_curr - 0.01 * alpha_new * gnorm * np.linalg.norm(x_new - y):
                break
            alpha_new *= 0.5
        alpha = alpha_new
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Restart if stuck
        if it % 2000 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() >= val:
                y = a.copy()
                t = 1.0
                alpha = 0.5 * alpha
    
    # Final check
    c_final = fft_convolve_self(best_a)
    obj_final = 2.0 * n * c_final.max() / (best_a.sum() ** 2)
    if obj_final < best_val:
        best_val = obj_final
    return best_a, best_val

def simulated_annealing(a0, S, time_budget, rng, T_start=0.01, T_end=0.0001, verbose=False):
    """Simulated annealing with Gaussian perturbations on the simplex."""
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    start = time.monotonic()
    best_val = objective_val(a)
    best_a = a.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress
        
        # Perturb: add Gaussian noise and project
        sigma = T * S / np.sqrt(n)
        a_pert = a + rng.normal(0, sigma, n)
        a_pert = project_simplex(a_pert, S)
        
        val_pert = objective_val(a_pert)
        val_curr = objective_val(a)
        
        if val_pert < val_curr or rng.random() < np.exp(-(val_pert - val_curr) / max(T * 0.01, 1e-10)):
            a = a_pert
            if val_pert < best_val:
                best_val = val_pert
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print("  SA it=%d T=%.6f obj=%.6f" % (it, T, best_val))
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive: if stuck for too long, increase temperature briefly
        if no_improve > 500:
            T = max(T, T_start * 0.5)
            no_improve = 0
    
    return best_a, best_val

    return [np.maximum(x, 0.01) for x in inits]

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using:
1. Clean FISTA with LSE smoothing and backtracking
2. Structured perturbation restarts (kick strategy)
3. Simulated annealing local search
4. Multi-resolution coarse-to-fine
5. Multi-start with diverse initializations at n=8192 and n=16384
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Smooth max via log-sum-exp."""
    cm = c.max()
    x = np.clip(beta * (c - cm), -500.0, 500.0)
    return cm + np.log(np.sum(np.exp(x))) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of conv(a,a) w.r.t. a."""
    n = len(a)
    cm = c.max()
    x = np.clip(beta * (c - cm), -500.0, 500.0)
    e = np.exp(x)
    w = e / e.sum()
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_val(a):
    """Compute the objective 2*n*max_b/sum_a^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def fista_optimize(a0, S, time_budget, rng, beta_start=20.0, beta_end=5000.0,
                   kick_interval=3000, kick_strength=0.05, verbose=False):
    """
    FISTA with LSE smoothing, backtracking, and periodic structured kicks.
    Returns (best_sequence, best_objective).
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_val = objective_val(a)
    best_a = a.copy()
    alpha = 0.5 / (beta_start * 3.0)
    it = 0
    last_improve = 0

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (y.sum() ** 2)

        if cur_obj < best_val:
            best_val = cur_obj
            best_a = y.copy()
            last_improve = it
            if verbose and it % 1000 == 0:
                print("    it=%d beta=%.0f obj=%.6f" % (it, beta, best_val))

        # Kick strategy: structured perturbation when stuck
        if it - last_improve > kick_interval:
            kick_type = rng.integers(0, 4)
            if kick_type == 0:
                # Block flip
                block_size = min(500, n // 8)
                start_idx = rng.integers(0, n - block_size)
                y[start_idx:start_idx + block_size] = S / n
            elif kick_type == 1:
                # Low-frequency sinusoidal perturbation
                tt = np.linspace(0, 1, n)
                freq = rng.uniform(1.0, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                pert = kick_strength * S / n * np.sin(2 * np.pi * freq * tt + phase)
                y = project_simplex(y + pert, S)
            elif kick_type == 2:
                # Gaussian kick
                sigma = kick_strength * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, sigma, n), S)
            else:
                # Blend with uniform
                y = project_simplex(0.8 * y + 0.2 * S / n, S)
            t = 1.0
            alpha = 0.5 / (beta * 3.0)
            last_improve = it

        # Compute gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search on LSE
        f_curr = lse_value(c, beta)
        alpha_bw = alpha
        x_new = y  # fallback
        for bt in range(40):
            candidate = project_simplex(y - alpha_bw * grad, S)
            c_candidate = fft_convolve_self(candidate)
            f_candidate = lse_value(c_candidate, beta)
            decrease = 0.01 * alpha_bw * gnorm * np.linalg.norm(candidate - y)
            if f_candidate <= f_curr - decrease:
                x_new = candidate
                break
            alpha_bw *= 0.5
        alpha = max(alpha_bw, 1e-10)
        if x_new is y:
            x_new = project_simplex(y - alpha * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Small periodic restart for stability
        if it % 5000 == 0:
            y = a.copy()
            t = 1.0

    # Final evaluation
    c_final = fft_convolve_self(best_a)
    obj_final = 2.0 * n * c_final.max() / (best_a.sum() ** 2)
    if obj_final < best_val:
        best_val = obj_final
    return best_a, best_val


def simulated_annealing(a0, S, time_budget, rng, T_start=0.005, T_end=0.0002, verbose=False):
    """Simulated annealing with Gaussian perturbations."""
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    start = time.monotonic()
    best_val = objective_val(a)
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_start * (T_end / T_start) ** (progress ** 0.5)

        sigma = T * S / np.sqrt(n)
        a_pert = project_simplex(a + rng.normal(0, sigma, n), S)

        val_pert = objective_val(a_pert)
        val_curr = objective_val(a)

        if val_pert < val_curr:
            a = a_pert
            if val_pert < best_val:
                best_val = val_pert
                best_a = a.copy()
                if verbose and it % 200 == 0:
                    print("    SA it=%d obj=%.6f" % (it, best_val))
        else:
            accept_prob = np.exp(-(val_pert - val_curr) / max(T * 0.005, 1e-12))
            if rng.random() < accept_prob:
                a = a_pert

    return best_a, best_val


def structured_kick_search(a0, S, time_budget, rng, verbose=False):
    """
    Repeatedly kick the sequence with structured perturbations and re-optimize.
    This helps escape local minima.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    start = time.monotonic()
    best_val = objective_val(a)
    best_a = a.copy()

    kick_sizes = [200, 500, 1000, 2000]

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3:
            break

        # Choose kick type and size
        kick_size = kick_sizes[rng.integers(0, len(kick_sizes))]
        kick_size = min(kick_size, n // 4)
        start_idx = rng.integers(0, max(1, n - kick_size))

        # Create perturbed version
        a_kick = a.copy()
        pert_type = rng.integers(0, 3)
        if pert_type == 0:
            # Randomize a block
            a_kick[start_idx:start_idx + kick_size] = rng.uniform(0.1, 2.0, kick_size) * S / n
        elif pert_type == 1:
            # Swap two blocks
            s2 = rng.integers(0, max(1, n - kick_size))
            if abs(s2 - start_idx) > kick_size:
                tmp = a_kick[start_idx:start_idx + kick_size].copy()
                a_kick[start_idx:start_idx + kick_size] = a_kick[s2:s2 + kick_size]
                a_kick[s2:s2 + kick_size] = tmp
        else:
            # Blend with a different smooth pattern
            tt = np.linspace(0, 1, kick_size)
            a_kick[start_idx:start_idx + kick_size] = S / n * np.maximum(
                np.sin(np.pi * tt) ** 2, 0.05)

        a_kick = project_simplex(a_kick, S)

        # Quick FISTA from the kicked state
        t_alloc = min(remaining * 0.4, 60.0)
        a_opt, val = fista_optimize(a_kick, S, t_alloc, rng,
                                     beta_start=50.0, beta_end=2000.0,
                                     kick_interval=500, kick_strength=0.02)
        if val < best_val:
            best_val = val
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print("    Kick improvement: obj=%.6f" % best_val)

        # Also try the kicked state with SA
        t_sa = min(remaining * 0.15, 20.0)
        if t_sa > 3:
            a_sa, val_sa = simulated_annealing(a_kick, S, t_sa, rng)
            if val_sa < best_val:
                best_val = val_sa
                best_a = a_sa.copy()
                a = a_sa.copy()

    return best_a, best_val


def diverse_inits(n, rng, given=None):
    """Generate diverse initializations for length n."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if given is not None:
        if len(given) == n:
            inits.append(given.copy())
        else:
            t_given = np.linspace(0, 1, len(given))
            inits.append(np.interp(t, t_given, given))

    inits.append(np.ones(n) * 0.8)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.05))
    inits.append(np.exp(-3.0 * (t - 0.4) ** 2 / 0.15))
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * t) + 0.2 * np.cos(2 * np.pi * 5 * t + 0.5))
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    return [np.maximum(x, 0.01) for x in inits]


def upscale(a_small, n_large):
    """Interpolate sequence to larger length."""
    t_old = np.linspace(0, 1, len(a_small))
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)
