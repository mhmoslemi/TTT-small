"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient with
Nesterov acceleration, spectral power compression, and multi-length search.

Key ideas:
1. Proper subgradient computation for top-k lags with adaptive weighting
2. Spectral power compression (|A|^alpha) to flatten the autocorrelation
3. Nesterov acceleration for faster convergence
4. Adaptive step size based on recent objective improvement
5. Periodic spectral re-phasing to escape local minima
6. Multi-length search (4096, 8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value 2*n*max(c)/(sum(a))^2."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_topk_gradient(a, c, top_k, temperature=1.0):
    """
    Compute the gradient of the smoothed max of c using top-k lags.
    Uses softmax weighting over the top-k values for smoothness.
    Returns (gradient, c_max, top_idx).
    """
    n = len(a)
    # Find top-k indices
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    # Smooth weighting via softmax (temperature controls sharpness)
    if temperature > 0:
        exp_vals = np.exp(temperature * (top_vals - top_vals.max()))
        weights = exp_vals / exp_vals.sum()
    else:
        # Hard: uniform weight over top-k
        weights = np.ones(top_k) / top_k
    
    c_max = c.max()
    
    # Compute gradient: g[m] = 2 * sum_k w_k * a[k-m] for 0 <= k-m < n
    grad = np.zeros(n)
    for i in range(top_k):
        k = int(top_idx[i])
        w = 2.0 * weights[i]
        if 0 <= k < 2 * n - 1:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    
    return grad, c_max, top_idx


def spectral_step(a, N, alpha, S):
    """
    Spectral power compression step.
    Compute |A|^alpha to compress the power spectrum, then reconstruct.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Power compression: P_new = P^alpha (alpha < 1 compresses)
    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()
    
    # Normalize to avoid overflow
    P_norm = P / P_max
    P_new = np.power(P_norm, alpha) * P_max
    
    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Normalize to maintain sum
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    a_new = project_simplex(a_new, S)
    return a_new

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient with
Nesterov acceleration, spectral power compression, and multi-length search.

Key ideas:
1. Proper subgradient computation for top-k lags with softmax weighting for smoothness
2. Spectral power compression (|A|^alpha) to flatten the autocorrelation
3. Nesterov acceleration for faster convergence
4. Adaptive step size based on recent objective improvement
5. Periodic spectral re-phasing and noise injection to escape local minima
6. Multi-length search (4096, 8192, 16384)
7. Multiplicative update option for better non-negativity handling
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score_fast(a, c):
    """Compute score given precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_score(a):
    """Compute the objective value 2*n*max(c)/(sum(a))^2."""
    c = fft_convolve_self(a)
    return compute_score_fast(a, c)


def compute_topk_gradient(a, c, top_k, temperature):
    """
    Compute the gradient of the softmax-weighted top-k of c w.r.t. a.
    dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    Uses softmax weighting for smoothness.
    Returns (grad, c_max, top_idx).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    c_max = c.max()

    # Softmax weighting
    if temperature > 0 and len(top_vals) > 1:
        shifted = top_vals - top_vals[0]
        exp_vals = np.exp(temperature * shifted)
        weights = exp_vals / exp_vals.sum()
    else:
        weights = np.ones(len(top_vals)) / len(top_vals)

    # Compute gradient: g[m] = 2 * sum_k w_k * a[k-m] for 0 <= k-m < n
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = 2.0 * weights[i]
        if w < 1e-15:
            continue
        if 0 <= k < 2 * n - 1:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # For m in [lo, hi]: a[k-m] ranges from a[k-hi] down to a[k-lo]
                grad[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

    return grad, c_max, top_idx


def spectral_power_step(a, N, alpha, S):
    """
    Spectral power compression: |A|^alpha to flatten the power spectrum.
    alpha < 1 compresses (reduces peak), alpha > 1 expands.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()

    # Power compression with normalization
    P_norm = P / P_max
    P_new = np.power(P_norm, alpha)
    # Rescale to preserve total energy (sum of P = sum of |A|^2 related to sum(a)^2)
    # Actually, let's just rescale to preserve the DC component (sum of a)
    a_new = np.real(ifft(np.sqrt(P_new) * np.exp(1j * phase)))[:n]

    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_target_step(a, N, gamma, S):
    """
    Target spectrum approach: partially flatten |A|^2 toward the mean.
    |A|_new^2 = (1-gamma)*|A|^2 + gamma * mean(|A|^2)
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    P_mean = P.mean()
    P_new = (1.0 - gamma) * P + gamma * P_mean
    P_new = np.maximum(P_new, 0.0)

    a_new = np.real(ifft(np.sqrt(P_new) * np.exp(1j * phase)))[:n]

    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    a_new = project_simplex(a_new, S)
    return a_new

            if sc_noisy < best_score * 1.02:
                y = y_noisy
                a = y_noisy
                t = 1.0
            # Occasionally try multiplicative mode
            if rng.random() < 0.15:
                use_multiplicative = not use_multiplicative

        # Stall detection
        if it - last_improvement_it > stall_threshold:
            # Aggressive restart
            noise_scale = 0.08 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
            eta = min(eta * 1.3, 5.0)
            temperature = max(1.0, temperature * 0.8)
            last_improvement_it = it
            use_multiplicative = rng.random() < 0.5

        # Check target
        if best_score <= target:
            break

    return best_a, best_score


def make_initializations(n, rng, provided=None):
    """Generate diverse initializations."""
    inits = []

    if provided is not None and len(provided) == n:
        p = np.clip(provided, 0, 1000).astype(np.float64)
        if p.sum() > 0.01:
            inits.append(("provided", p))

    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * 0.5))

    # Sine
    inits.append(("sin1", 0.5 + 0.35 * np.sin(2.0 * np.pi * tt)))

    # Multi-sine (more complex structure)
    inits.append(("multi_sin", 0.5 + 0.12 * np.sin(2 * np.pi * 3 * tt) +
                                0.08 * np.sin(2 * np.pi * 7 * tt) +
                                0.05 * np.sin(2 * np.pi * 11 * tt) +
                                0.03 * np.sin(2 * np.pi * 17 * tt)))

    # Chirp
    inits.append(("chirp", 0.5 + 0.25 * np.sin(2 * np.pi * (2 * tt + 3 * tt * tt))))

    # Random
    inits.append(("random", rng.random(n) + 0.03))

    # Smooth noise (low-pass)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 256)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth", 0.15 + 0.7 * smooth))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive subgradient + spectral power compression + multi-length search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()

    print("Starting optimization with budget %.0fs" % budget_s)

    # Get provided sequence if available
    provided = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            provided = given
            print("Using provided sequence of length %d, score=%.6f" % (
                len(given), compute_score(np.clip(given, 0, 1000))))
    except Exception:
        pass

    # Determine which lengths to try
    lengths_to_try = []
    if provided is not None:
        lengths_to_try.append(len(provided))

    for L in [8192, 16384, 4096]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Sort: provided first, then mid-range
    if provided is not None:
        p_len = len(provided)
        lengths_to_try.sort(key=lambda x: (x != p_len, abs(x - 8192)))
    else:
        lengths_to_try.sort(key=lambda x: abs(x - 8192))

    global_best = np.inf
    global_a = None

    # Allocate budget: reserve for verification
    reserve = 30.0
    avail = budget_s - reserve

    # Weight allocation: favor provided length and 8192
    weights = []
    for L in lengths_to_try:
        if provided is not None and L == len(provided):
            weights.append(1.8)
        elif L == 8192:
            weights.append(1.3)
        elif L == 16384:
            weights.append(1.0)
        else:
            weights.append(0.8)
    total_w = sum(weights)
    length_budgets = [avail * w / total_w for w in weights]

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_start
        remaining = budget_s - reserve - elapsed
        if remaining < 20:
            break

        alloc = min(length_budgets[i], remaining)
        if alloc < 15:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))

        inits = make_initializations(n, rng, provided)
        n_inits = len(inits)

        # Budget per init (more for provided)
        per_init_w = [2.5 if nm == "provided" else 1.0 for nm, _ in inits]
        total_pi_w = sum(per_init_w)
        per_init_budgets = [alloc * w / total_pi_w for w in per_init_w]

        # Reserve for final polish
        polish_budget = min(40.0, alloc * 0.12)
        opt_alloc = alloc - polish_budget

        local_best = np.inf
        local_a = None

        for j, (name, arr) in enumerate(inits):
            if time.monotonic() - t_start > budget_s - reserve - polish_budget:
                break

            bi = min(per_init_budgets[j], max(5.0, opt_alloc - elapsed * 0.1))
            if bi < 5:
                break

            a_init = np.clip(arr, 0.0, 1000.0)
            s = a_init.sum()
            if s < 1e-15:
                a_init = np.ones(n) * np.sqrt(2.0 * n) / n
            else:
                a_init = a_init * (np.sqrt(2.0 * n) / s)

            print("  [%s] starting (budget %.0fs)..." % (name, bi))
            t0 = time.monotonic()
            a_opt, score = optimize_sequence(a_init, n, bi, rng)
            dt = time.monotonic() - t0
            print("  [%s] score=%.6f (%.0fs)" % (name, score, dt))

            if score < local_best:
                local_best = score
                local_a = a_opt.copy()
            if score < global_best:
                global_best = score
                global_a = a_opt.copy()
                print("  ** New global best: %.6f (n=%d, init=%s) **" % (score, n, name))

        # Final polish
        if local_a is not None and polish_budget > 5:
            rem_polish = min(polish_budget, max(0, budget_s - reserve - (time.monotonic() - t_start)))
            if rem_polish > 3:
                print("  [polish] budget=%.0fs" % rem_polish)
                a_pol, score_pol = optimize_sequence(local_a, n, rem_polish, rng)
                if score_pol < local_best:
                    local_best = score_pol
                    local_a = a_pol.copy()
                if score_pol < global_best:
                    global_best = score_pol
                    global_a = a_pol.copy()
                    print("  ** New global best after polish: %.6f **" % score_pol)

        print("  Best for n=%d: %.6f (global: %.6f)" % (n, local_best, global_best))

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence, score=%.4f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback verification: %.6f" % ver)

    print("Final: length=%d, score=%.6f, time=%.0fs" % (
        len(global_a), global_best, time.monotonic() - t_start))
    return [float(x) for x in global_a]
