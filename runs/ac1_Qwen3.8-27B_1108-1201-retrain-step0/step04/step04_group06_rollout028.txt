def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho

spectral water-filling, and simulated annealing restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_topk(a, c, top_k, temperature):
    """Compute weighted subgradient using top-k lags."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    w = np.exp(temperature * (top_vals - top_vals.max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        k = int(top_idx[idx_i])
        wi = w[idx_i]
        if wi < 1e-10:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            grad[js] += 2.0 * wi * a[k - js]
    return grad

    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        temp = t0 + dt * it
        top_k = min(kmax, k0 + dk * it)
        
        c = fft_convolve_self(y)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = y.copy()
        
        grad = compute_gradient_topk(y, c, top_k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

    if c.max() < best_score:
        best_score = c.max()
        best_a = a.copy()

    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Numerically stable softmax
    shifted = top_vals - top_vals.max()
    w = np.exp(temperature * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        k = int(top_idx[idx_i])
        wi = w[idx_i]
        if wi < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            grad[js] += 2.0 * wi * a[k - js]
    return grad


def fista_optimize(a_init, S, n, max_time, rng, t0=5.0, dt=0.005,
                   k0=16, dk=3, kmax=4096, step_scale=0.3):
    """
    FISTA with adaptive temperature and softmax-weighted subgradient.
    Returns (best_a, best_score).
    """
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        temp = t0 + dt * it
        top_k = min(kmax, k0 + dk * it)

        c = fft_convolve_self(y)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = y.copy()

        grad = compute_weighted_gradient(y, c, top_k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step: scale so that max element change is ~ step_scale * S/n
        max_grad = np.max(np.abs(grad))
        if max_grad > 1e-18:
            step = step_scale * S / (n * max_grad)
        else:
            step = step_scale / n

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        beta = (t_k - 1.0) / t_new
        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-15:
            y *= S / s
        a = x_new
        t_k = t_new

        # Occasional perturbation to escape local minima
        if it % 5000 == 0 and rng.random() < 0.08:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0

    # Final check
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_a = a.copy()

    return best_a, best_score


def spectral_shape_step(a, n, N, alpha, top_k, target, rng):
    """
    Single spectral shaping step using geometric-mean target.
    More conservative than exponential: P_new = P * (target / max(P_contribution, target))^alpha
    where P_contribution is the spectral contribution to peak lags.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target:
        return a, c_max

    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = top_vals.sum()
    if excess_sum < 1e-30:
        return a, c_max
    weights = excess / excess_sum

    # Compute spectral contribution to peak lags
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        k = int(k)
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * k / N)

    # Geometric mean shaping: reduce power where cos_eff > 0
    # P_new = P * (1 - alpha * cos_eff^+)^gamma
    # Use a softer version
    reduction = np.maximum(cos_eff, 0.0)
    P_new = P * np.power(1.0 - alpha * reduction, 1.0)
    P_new = np.maximum(P_new, 1e-30)

    # Preserve total power (sum of P = S^2 * N by Parseval for the padded signal)
    # Actually, preserve the DC component to maintain the sum
    P_new[0] = P[0]

    a_new = np.real(ifft(np.sqrt(P_new) * np.exp(1j * np.angle(A))))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def spectral_waterfill(a_init, n, N, max_time, rng, target=1.45,
                       top_k_init=8, alpha_init=0.4):
    """
    Iterative spectral water-filling loop.
    """
    a = a_init.copy()
    S = a.sum()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    stall = 0

    while time.monotonic() < max_time:
        it += 1
        top_k = min(64, top_k_init + it // 10)
        alpha = alpha_init / (1.0 + 0.03 * it)

        a_new, c_max = spectral_shape_step(a, n, N, alpha, top_k, target, rng)

        # Verify
        c_check = fft_convolve_self(a_new)
        actual_cmax = c_check.max()

        if actual_cmax < best_cmax:
            best_cmax = actual_cmax
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            a = a_new.copy()
            stall += 1
            if stall > 50:
                noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                stall = 0

        if best_cmax <= target:
            break

    return best_a, best_cmax


def pair_coordinate_descent(a_init, n, S, max_time, rng, n_top=20, n_rounds=200):
    """
    Pair coordinate descent: optimize pairs (i, j) that both contribute to the peak lag.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.4, 1.6]

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        c_max = c.max()
        peak_lag = int(np.argmax(c))

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find top-contributing coordinates at the peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:min(n_top * 2, high - low + 1)]

        improved = False
        for idx in range(len(top_coords)):
            if time.monotonic() - start > max_time:
                break
            i = int(top_coords[idx])
            j = peak_lag - i
            if j < 0 or j >= n or j == i:
                continue

            orig_i = a[i]
            orig_j = a[j]
            best_val = c_max
            best_mi = 1.0
            best_mj = 1.0

            for mi in multipliers:
                a[i] = orig_i * mi
                for mj in multipliers:
                    a[j] = orig_j * mj
                    # Quick eval: recompute affected lags
                    # Lags affected by changing a[i]: from i to i+n-1
                    # Lags affected by changing a[j]: from j to j+n-1
                    # Just recompute the peak lag and a few neighbors
                    affected = [peak_lag]
                    if peak_lag + 1 < 2 * n - 1:
                        affected.append(peak_lag + 1)
                    if peak_lag - 1 >= 0:
                        affected.append(peak_lag - 1)

                    new_max = 0.0
                    for k in affected:
                        val = 0.0
                        lo = max(0, k - n + 1)
                        hi = min(k, n - 1)
                        for pp in range(lo, hi + 1):
                            val += a[pp] * a[k - pp]
                        new_max = max(new_max, val)

                    if new_max < best_val:
                        best_val = new_max
                        best_mi = mi
                        best_mj = mj
                a[j] = orig_j
            a[i] = orig_i
            a[j] = orig_j

            if best_mi != 1.0 or best_mj != 1.0:
                a[i] = orig_i * best_mi
                a[j] = orig_j * best_mj
                improved = True

        # Rescale
        a = project_simplex(a, S)

        if not improved and rng.random() < 0.3:
            # Perturb top contributors
            for idx in range(min(5, len(top_coords))):
                i = int(top_coords[idx])
                a[i] *= rng.uniform(0.6, 1.4)
            a = project_simplex(a, S)

    return best_a, best_score


def diverse_init(n, rng, method="uniform"):

    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t) \
            + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = 0.3 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 4) ** 2) + 0.2 * np.exp(-0.5 * ((t - 0.7) * 4) ** 2)

        a = np.repeat(heights, piece_len)
        if len(a) < n
