"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    cm = c.max()
    if beta > 1e10:
        return cm
    diff = c - cm
    if (beta * diff).max() > 500:
        return cm
    terms = np.exp(beta * diff)
    return cm + np.log(terms.sum()) / beta

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    diff = c - cm
    scaled = beta * diff
    if scaled.max() > 500:
        return np.zeros(n)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def objective_direct(a):
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def gradient_descent_lse(a_init, S, max_time, rng, beta_start=5.0, beta_end=5000.0, 
                         n_beta_steps=None, use_fista=True, momentum_decay=0.95):
    """LSE-smoothed gradient descent with adaptive beta annealing."""
    n = len(a_init)
    a = np.clip(a_init, 0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    if n_beta_steps is None:
        n_beta_steps = 500
    
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal beta
        progress = min(1.0, (it - 1) / max(1, n_beta_steps))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
        
        # Compute gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: 1/L where L ≈ 2*beta*max|a| (conservative)
        # More practically: 1/(2*beta) works well
        step = 1.0 / (2.0 * beta)
        
        # Try the step
        a_new = project_simplex(a - step * grad, S)
        
        # Backtracking if objective increased significantly
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * (1 + 0.5) :
            # Too aggressive, reduce step
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
        
        a = a_new
        
        # Periodic restart with small noise
        if it % 1000 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best

def fourier_perturb(a, rng, S, perturbation_type="phase"):
    """Apply Fourier domain perturbation."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    
    if perturbation_type == "phase":
        # Randomize phase partially
        mag = np.abs(A)
        phase = np.angle(A)
        # Blend with random phase
        t = rng.uniform(0.1, 0.5)
        new_phase = phase + t * rng.normal(0, 1, fsize)
        A_new = mag * np.exp(1j * new_phase)
    elif perturbation_type == "magnitude_squeeze":
        # Squeeze magnitude toward uniform
        mag = np.abs(A)
        phase = np.angle(A)
        mean_mag = mag[1:].mean()
        t = rng.uniform(0.1, 0.4)
        mag_new = (1 - t) * mag + t * mean_mag
        mag_new[0] = mag[0]  # Preserve DC
        A_new = mag_new * np.exp(1j * phase)
    elif perturbation_type == "lowpass":
        # Add energy at low frequencies
        mag = np.abs(A)
        phase = np.angle(A)
        low_mask = np.zeros(fsize)
        low_mask[:fsize // 8] = 1
        low_mask[fsize - fsize // 8:] = 1
        t = rng.uniform(0.05, 0.2)
        mag_new = mag + t * low_mask * mag.mean()
        A_new = mag_new * np.exp(1j * phase)
    else:
        return a.copy()
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = np.full(n, S / n)
    return a_new

def fourier_crossover(a1, a2, rng, S):
    """Crossover in Fourier domain."""
    n = len(a1)
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    
    mag1 = np.abs(A1)
    mag2 = np.abs(A2)
    phase1 = np.angle(A1)
    phase2 = np.angle(A2)
    
    # Geometric mean of magnitudes, random phase blend
    mag_new = np.sqrt(mag1 * mag2)
    t = rng.uniform(0.3, 0.7)
    phase_new = t * phase1 + (1 - t) * phase2
    A_new = mag_new * np.exp(1j * phase_new)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = np.full(n, S / n)
    return a_new

def simulated_annealing(a_init, S, max_time, rng, T_start=0.1, T_end=0.001):
    """Simulated annealing with structured moves."""
    n = len(a_init)
    a = np.clip(a_init, 0, None).copy()
    s = a.sum()
    if s > 1e-15:
        a = a * (S / s)
    
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    current_obj = best
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        T = T_start * (T_end / T_start) ** progress
        
        # Choose a move
        move_type = rng.integers(0, 4)
        
        if move_type == 0:
            # Add sinusoidal perturbation
            freq = rng.uniform(0.5, n / 4)
            amp = T * S / np.sqrt(n)
            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n)
            a_try = a + amp * np.sin(2 * np.pi * freq * tt / n + phase)
        elif move_type == 1:
            # Fourier phase perturbation
            a_try = fourier_perturb(a, rng, S, "phase")
        elif move_type == 2:
            # Random swap of two elements
            i, j = rng.integers(0, n, 2)
            a_try = a.copy()
            delta = a_try[i] - a_try[j]
            a_try[i] = a_try[j]
            a_try[j] = a_try[i]
        else:
            # Small random noise
            noise = rng.normal(0, T * S / np.sqrt(n), n)
            a_try = a + noise
        
        a_try = np.clip(a_try, 0, None)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try * (S / s_try)
        else:
            a_try = a
        
        c_try = fft_convolve_self(a_try)
        new_obj = c_try.max()
        
        # Accept if better or with probability exp(-delta/T)
        if new_obj < current_obj:
            a = a_try
            current_obj = new_obj
            if new_obj < best:
                best = new_obj
                best_a = a.copy()
        else:
            delta = new_obj - current_obj
            if delta < T * 100:  # Scale delta by S^2 for temperature
                if rng.random() < np.exp(-delta / max(T * (S*S/(2*n)), 1e-30)):
                    a = a_try
                    current_obj = new_obj
    
    return best_a, best

def subgradient_peak_shave(a_init, S, max_time, rng, top_k=8):
    """Peak shaving using subgradient of max."""
    n = len(a_init)
    a = np.clip(a_init, 0, None).copy()
    s = a.sum()
    if s > 1e-15:
        a = a * (S / s)
    
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
        
        # Find top-k peaks
        idx = np.argsort(c)[::-1][:top_k]
        # Weight by excess
        weights = np.maximum(c[idx] - np.mean(c[idx]), 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        w_full = np.zeros(len(c))
        w_full[idx] = weights
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Decreasing step size
        step = S / (n ** 0.5 * (1 + 0.01 * it))
        
        a_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
        a = a_new
    
    return best_a, best

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix any numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10 * S:
        nz = result > 0
        result[nz] += diff / nz.sum()
        result = np.maximum(result, 0)
    return result


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    scaled = beta * diff
    if scaled.max() > 500:
        # Beta too large, reduce effectively
        return np.zeros(n)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute the objective value directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def normalize(a, S):
    """Clip to non-negative and rescale to sum = S."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def gradient_descent_lse(a_init, S, max_time, rng, beta_start=5.0, beta_end=5000.0,
                         n_total_iters=None, use_acceleration=True):
    """
    LSE-smoothed gradient descent with optional Nesterov acceleration.
    Uses backtracking line search for robustness.
    """
    n = len(a_init)
    a = normalize(a_init, S)
    y = a.copy()  # Nesterov lookahead
    t_k = 1.0
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Pre-compute fsize
    conv_size = 2 * n - 1
    fsize_grad = 1 << (conv_size + n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        if n_total_iters is None:
            progress = min(1.0, (time.monotonic() - start) / max_time)
        else:
            progress = min(1.0, it / n_total_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient at y (Nesterov lookahead point)
        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / (2.0 * beta)
        # Scale gradient to have norm ~ 1 for better conditioning
        grad_scaled = grad / gnorm * min(gnorm, S)
        
        for bt in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            new_max = c_new.max()
            # Armijo condition (approximate, since we're tracking max not LSE)
            if new_max <= cur_max + 1e-10 * S * S / (2 * n):
                break
            step *= 0.5
        else:
            # If backtracking fails, take a very small step
            step = max(step, 1e-10)
            x_new = project_simplex(y - step * grad, S)
        
        if use_acceleration:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            # Ensure y is on the simplex
            y = project_simplex(y, S)
            a = x_new
            t_k = t_new
        else:
            a = x_new
            y = a.copy()
        
        # Periodic restart
        if it - last_improve > max(500, it // 5):
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it // 2000)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def fourier_perturb(a, rng, S, mode="phase"):
    """Apply Fourier domain perturbation and project back."""
    n = len(a)
    conv_size = 2 * n - 1
    fsize = 1 << (conv_size - 1).bit_length()
    A = fft(a, fsize)
    
    if mode == "phase":
        mag = np.abs(A)
        phase = np.angle(A)
        t = rng.uniform(0.05, 0.3)
        # Only perturb non-DC phases
        noise = rng.normal(0, t, fsize)
        noise[0] = 0
        new_phase = phase + noise
        A_new = mag * np.exp(1j * new_phase)
    elif mode == "mag_squeeze":
        mag = np.abs(A)
        phase = np.angle(A)
        t = rng.uniform(0.05, 0.2)
        mean_mag = mag[1:].mean()
        mag_new = (1 - t) * mag + t * mean_mag
        mag_new[0] = mag[0]  # Keep DC
        # Handle negative side (by symmetry for real signals)
        A_new = mag_new * np.exp(1j * phase)
    elif mode == "band_add":
        mag = np.abs(A)
        phase = np.angle(A)
        # Add energy in a random frequency band
        center = rng.integers(fsize // 8, 7 * fsize // 8)
        bandwidth = fsize // 16
        band = np.zeros(fsize)
        lo = max(1, center - bandwidth)
        hi = min(fsize - 1, center + bandwidth)
        band[lo:hi] = 1
        band[fsize - hi:fsize - lo] = 1  # By symmetry
        t = rng.uniform(0.02, 0.1)
        mag_new = mag + t * band * mag[1:].mean()
        A_new = mag_new * np.exp(1j * phase)
    elif mode == "smooth":
        # Low-pass filter in frequency (smooths in time)
        mag = np.abs(A)
        phase = np.angle(A)
        # Apply a gentle low-pass
        freq = np.fft.fftfreq(fsize)
        cutoff = rng.uniform(0.3, 0.8)
        lp = np.exp(-20.0 * np.abs(freq) / cutoff)
        t = rng.uniform(0.1, 0.4)
        mag_new = (1 - t) * mag + t * mag * lp
        mag_new[0] = mag[0]
        A_new = mag_new * np.exp(1j * phase)
    else:
        return a.copy()
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.clip(a_new, 0.0, None)
    return normalize(a_new, S)


def fourier_crossover(a1, a2, rng, S):
    """Fourier domain crossover between two sequences."""
    n = len(a1)
    conv_size = 2 * n - 1
    fsize = 1 << (conv_size - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    
    mag1 = np.abs(A1)
    mag2 = np.abs(A2)
    phase1 = np.angle(A1)
    phase2 = np.angle(A2)
    
    # Geometric mean of magnitudes
    mag_new = np.sqrt(mag1 * mag2)
    
    # Blend phases
    t = rng.uniform(0.3, 0.7)
    phase_new = t * phase1 + (1 - t) * phase2
    # Handle phase wrapping
    phase_new = np.angle(np.exp(1j * phase_new))
    
    A_new = mag_new * np.exp(1j * phase_new)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.clip(a_new, 0.0, None)
    return normalize(a_new, S)


def peak_shave_iter(a, S, n_iters=100, top_k=8, step_scale=1.0):
    """Peak shaving using subgradient on top peaks."""
    n = len(a)
    conv_size = 2 * n - 1
    fsize_grad = 1 << (conv_size + n - 1).bit_length()
    
    for it in range(n_iters):
        c = fft_convolve_self(a)
        # Find top-k peaks
        k = min(top_k, conv_size)
        idx = np.argsort(c)[::-1][:k]
        # Weight by excess above mean of top peaks
        peak_vals = c[idx]
        weights = np.maximum(peak_vals - peak_vals.min(), 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        w_full = np.zeros(conv_size)
        w_full[idx] = weights
        
        W = fft(w_full, fsize_grad)
        A = fft(a, fsize_grad)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = step_scale * S / (np.sqrt(n) * (1 + 0.05 * it))
        a = project_simplex(a - step * grad, S)
    
    c = fft_convolve_self(a)
    return a, c.max()


def evolve_population(pop, S, rng, max_time, n=0):
    """Evolutionary search in Fourier domain."""
    if n == 0:
        n = len(pop[0][1])
    
    start = time.monotonic()
    best_obj, best_a = pop[0]
    
    gen = 0
    while time.monotonic() - start < max_time:
        gen += 1
        # Tournament selection: pick 2 parents
        p1_idx = rng.integers(0, len(pop))
        p2_idx = rng.integers(0, len(pop))
        
        # Create child
        child = fourier_crossover(pop[p1_idx][1], pop[p2_idx][1], rng, S)
        
        # Mutate
        mode = ["phase", "mag_squeeze", "band_add", "smooth"][rng.integers(0, 4)]
        child = fourier_perturb(child, rng, S, mode)
        
        # Short gradient refinement
        child, child_max = gradient_descent_lse(child, S, 3.0, rng,
                                                 beta_start=50.0, beta_end=500.0,
                                                 n_total_iters=50)
        child_obj = 2.0 * n * child_max / (child.sum() ** 2)
        
        # Replace worst if improved
        worst_idx = np.argmax([p[0] for p in pop])
        if child_obj < pop[worst_idx][0]:
            pop[worst_idx] = (child_obj, child)
        
        # Track global best
        if child_obj < best_obj:
            best_obj = child_obj
            best_a = child.copy()
    
    return pop, best_obj, best_a
