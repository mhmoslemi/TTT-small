def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def cross_correlation(f, w, n):
    """Compute r[i] = sum_j f[j] * w[j+i] for i=0,...,n-1 using FFT."""
    # (f * w)(i) = sum_j f[j] w[j+i]
    # This is the cross-correlation of f and w
    # FFT: ifft(FFT(f) * conj(FFT(w)))
    fsize = 1 << (n + len(w) - 1).bit_length()
    F = fft(f, fsize)
    W = fft(w, fsize)
    r = np.real(ifft(F * np.conj(W)))[:n]
    return r

"""
Minimize 2*n*max(f*f)/sum(f)^2 using log-space Adam with LSE smoothing.
Key innovations over previous approach:
1. Log-parametrization (f_i = exp(g_i)) for natural non-negativity handling
2. Adam optimizer with adaptive per-coordinate learning rates
3. Multi-phase strategy with spectral diversification and peak-shaving
4. Improved adaptive beta schedule and periodic restarts
5. LP polishing for final refinement
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_correlation(f, w, n):
    """Compute r[i] = sum_j f[j] * w[j+i] for i=0,...,n-1 using FFT."""
    fsize = 1 << (n + len(w) - 1).bit_length()
    F = fft(f, fsize)
    W = fft(w, fsize)
    r = np.real(ifft(F * np.conj(W)))[:n]
    return r


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def logspace_normalize(g, S):
    """Normalize log-space vector so that sum(exp(g)) = S."""
    g_max = g.max()
    exp_g = np.exp(g - g_max)
    log_sum = g_max + np.log(exp_g.sum())
    g = g - (log_sum - np.log(S))
    return g


def compute_lse_gradient(f, beta):
    """Compute LSE-approximated gradient of max(c) w.r.t. f."""
    n = len(f)
    c = fft_convolve_self(f)
    cm = c.max()
    exponents = beta * (c - cm)
    exponents = np.maximum(exponents, -700.0)
    e = np.exp(exponents)
    tot = e.sum()
    if tot < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = e / tot
    r = cross_correlation(f, w, n)
    grad = 2.0 * r
    return grad, c, cm


def run_logspace_adam(f_init, n, S, max_time, rng,
                       beta_start=3.0, beta_end=400.0,
                       lr_start=0.02, lr_end=0.0005,
                       diversify_interval=400,
                       spectral_interval=800):
    """
    Adam optimizer in log-space with LSE smoothing.
    Returns (best_f, best_value).
    """
    f = np.maximum(f_init, 1e-10).copy()
    f = f * (S / f.sum())
    g = np.log(f)
    g = logspace_normalize(g, S)

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0

    b1, b2, eps = 0.9, 0.999, 1e-8

    best_val = np.inf
    best_f = f.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    patience = 0

    est_iters = int(max_time * 40)
    total_est = max(100, est_iters)

    while time.monotonic() - start < max_time:
        it += 1
        t_adam += 1
        prog = min(1.0, it / total_est)

        # Adaptive hyperparameters
        beta = beta_start + (beta_end - beta_start) * (prog ** 1.5)
        lr = lr_start * (lr_end / lr_start) ** prog

        # Current state
        f_cur = np.exp(g)
        f_cur = np.maximum(f_cur, 1e-15)
        f_cur *= S / f_cur.sum()

        c = fft_convolve_self(f_cur)
        cur_val = 2.0 * n * c.max() / S**2

        if cur_val < best_val:
            best_val = cur_val
            best_f = f_cur.copy()
            last_improve = it
            patience = 0
        else:
            patience += 1

        # LSE gradient
        cm = c.max()
        exponents = beta * (c - cm)
        exponents = np.maximum(exponents, -700.0)
        e = np.exp(exponents)
        tot = e.sum()
        if tot < 1e-30:
            w = np.ones(len(c)) / len(c)
        else:
            w = e / tot

        r = cross_correlation(f_cur, w, n)
        grad_f = 2.0 * r
        grad_g = grad_f * f_cur

        # Adam update
        m_adam = b1 * m_adam + (1 - b1) * grad_g
        v_adam = b2 * v_adam + (1 - b2) * (grad_g ** 2)
        m_hat = m_adam / (1 - b1**t_adam)
        v_hat = v_adam / (1 - b2**t_adam)

        g = g - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Normalize in log-space
        g = logspace_normalize(g, S)

        # Periodic diversification when stuck
        if patience > diversify_interval * 3:
            noise_scale = 0.05 * max(0.1, 1.0 - prog)
            g = g + rng.normal(0, noise_scale, n)
            g = logspace_normalize(g, S)
            patience = 0

        # Periodic spectral jump
        if it % spectral_interval == 0 and patience > spectral_interval:
            f_cur = np.exp(g)
            f_cur = np.maximum(f_cur, 1e-15)
            f_cur *= S / f_cur.sum()
            fsize = 1 << (2 * n - 1).bit_length()
            F = fft(f_cur, fsize)
            mag = np.abs(F)
            phase = np.angle(F)
            # Compress magnitude spectrum
            alpha = 0.85
            mag_new = mag ** alpha
            F_new = mag_new * np.exp(1j * phase)
            f_new = np.real(ifft(F_new))[:n]
            f_new = np.maximum(f_new, 1e-15)
            s_new = f_new.sum()
            if s_new > 0.01:
                f_new *= S / s_new
                g = np.log(f_new)
                g = logspace_normalize(g, S)
                patience = 0

    return best_f, best_val


def peak_shaving(f, n, S, num_steps=200, rng=None):
    """
    Direct peak-shaving: identify peak, reduce contributing elements,
    increase non-contributing elements.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy()
    f = normalize_sum(f, n)
    best_val = 2.0 * n * (fft_convolve_self(f).max()) / S**2
    best_f = f.copy()

    for step in range(num_steps):
        c = fft_convolve_self(f)
        cm = c.max()
        cur_val = 2.0 * n * cm / S**2
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()

        # Find peak position
        p = np.argmax(c)

        # Compute blame: contribution of each f_i to c[p]
        # c[p] = sum_j f[j] * f[p-j]
        # blame[i] = f[i] * f[p-i] (if p-i in range)
        blame = np.zeros(n)
        for i in range(n):
            j = p - i
            if 0 <= j < n:
                blame[i] = f[i] * f[j]

        # Top blamed and bottom blamed
        k = max(5, n // 100)
        top_idx = np.argsort(blame)[::-1][:k]
        bot_idx = np.argsort(blame)[:k]

        # Transfer mass from top to bottom
        total_top = f[top_idx].sum()
        total_bot = f[bot_idx].sum()
        if total_top < 1e-10:
            continue
        frac = 0.05  # fraction to transfer
        transfer = total_top * frac
        # Reduce top proportionally
        for i in top_idx:
            f[i] *= (1 - frac)
        # Increase bottom proportionally
        if total_bot > 1e-10:
            for i in bot_idx:
                f[i] += transfer * (f[i] / total_bot)
        else:
            for i in bot_idx:
                f[i] += transfer / len(bot_idx)

        f = np.maximum(f, 0.0)
        f = normalize_sum(f, n)

    return best_f, best_val


def solve_lp_best_response(f, n, S, top_k=100, time_limit=8.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver='SCS', verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 3:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def lp_polish(f, n, S, max_time, top_k=80):
    """LP-based polishing: iteratively solve LP and take convex combinations."""
    f = f.copy()
    f = normalize_sum(f, n)
    best_val = 2.0 * n * (fft_convolve_self(f).max()) / S**2
    best_f = f.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        g = solve_lp_best_response(f, n, S, top_k=top_k, time_limit=5.0)
        if g is None:
            break

        c_g = fft_convolve_self(g)
        val_g = 2.0 * n * c_g.max() / S**2

        if val_g < best_val:
            best_val = val_g
            best_f = g.copy()
            f = g.copy()
        else:
            # Try convex combination
            for frac in [0.3, 0.2, 0.5, 0.1]:
                f_test = (1 - frac) * f + frac * g
                f_test = np.maximum(f_test, 0)
                f_test = normalize_sum(f_test, n)
                c_test = fft_convolve_self(f_test)
                val_test = 2.0 * n * c_test.max() / S**2
                if val_test < best_val:
                    best_val = val_test
                    best_f = f_test.copy()
                    f = f_test.copy()
                    break

    return best_f, best_val


def generate_structured_init(n, rng):
    """Generate a structured (non-trivial) initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []
    # Sum of a few sinusoids
    for freq in [2, 3, 5, 7]:
        f = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        inits.append(np.maximum(f, 0.01))
    # Gaussian bump
    inits.append(np.exp(-5.0 * (t - 0.3)**2) + 0.5 * np.exp(-5.0 * (t - 0.7)**2))
    # Random smooth
    coeffs = rng.normal(0, 1, min(10, n // 10))
    f = 1.0
    for k, c in enumerate(coeffs, 1):
        f += (c / k) * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    inits.append(np.maximum(f, 0.01))
    return inits
