def _fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Project v onto {x >= 0, sum(x) = S} using sorted algorithm."""

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def _normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def _lse_smoothed_obj_and_grad(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = _fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    obj = cm + np.log(tot) / beta
    w = e / tot
    
    # Gradient: d/da_i sum_k w_k * c_k
    # c_k = sum_j a_j * a_{k-j}
    # d c_k / d a_i = a_{k-i} + a_{i-k} (symmetric)
    # So grad_i = sum_k w_k * (a_{k-i} + a_{i-k})
    # = sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}
    # = (w * a)_i + (w * reverse(a))_i  [with appropriate shifts]

    # "cross-correlation" of w with a

    # And sum_k w_k * a_{i-k} = (w * a)_{i} [same thing by symmetry of the sum]

    # c_k = sum_j a_j * a_{k-j}
    # d c_k / d a_i: a_i appears when j=i (giving a_{k-i}) and when k-j=i, i.e., j=k-i (giving a_i * 1)
    # So d c_k / d a_i = a_{k-i} + a_{i-k}... no wait.
    # c_k = sum_j a_j * a_{k-j}
    # The term a_i appears as a_j when j=i: contributes a_{k-i}
    # The term a_i appears as a_{k-j} when k-j=i, i.e., j=k-i: contributes a_{k-i}... 
    # Wait no. a_{k-j} = a_i when k-j = i, so j = k-i. The term is a_{k-i} * a_i, and derivative w.r.t. a_i is a_{k-i}.
    # And a_j = a_i when j = i. The term is a_i * a_{k-i}, derivative w.r.t. a_i is a_{k-i}.
    # So d c_k / d a_i = a_{k-i} + a_{k-i} = 2 * a_{k-i} (when i != k-i, i.e., i != k/2)

    # Actually, more carefully: c_k = sum_j a_j * a_{k-j}

    # d c_k / d a_i = sum_j [d(a_j * a_{k-j})/d a_i]
    # = [d(a_i * a_{k-i})/d a_i] + [d(a_{k-i} * a_i)/d a_i] (when k-i != i)

    # This is the (standard) convolution. The derivative of the k-th component of a*a w.r.t. a_i is:

    # But we need to handle the case where a_{k-i} is out of bounds (then it's 0).

    # But 2*a_{k-i} = 2*a_i when k=2i, so the formula 2*a_{k-i} works in all cases!

    # So d c_k / d a_i = 2*a_i (from derivative of a_i^2), not 2*a_{k-i} (which would give the same thing since k-i=i).
    # So the formula is: d c_k / d a_i = 2 * a_{k-i} (with a_j = 0 for invalid j)
    # This works because when k-i = i (i.e., k=2i), we get 2*a_i which is correct.
    # When k-i != i, we get 2*a_{k-i} which accounts for both the j=i and j=k-i contributions.
    # When k-i is out of bounds, a_{k-i} = 0.
    
    # So: grad_i = sum_k w_k * 2 * a_{k-i}
    # = 2 * sum_k w_k * a_{k-i}
    # = 2 * (w * a)_i  [where * is cross-correlation, treating a as zero outside [0,n-1]]
    
    # The cross-correlation can be computed via FFT:
    # corr_i = sum_k w_k * a_{k-i}
    # This is the convolution of w with reverse(a), evaluated at position (2n-2) + i... 
    # Or more simply: it's the "autocorrelation-like" quantity.
    
    # Let's use: corr = ifft(fft(w, fsize) * conj(fft(a, fsize)))
    # Wait, that gives the cross-correlation in the standard sense.
    # Actually: ifft(FFT(w) * conj(FFT(a))) gives the circular cross-correlation.
    # For linear cross-correlation, we need appropriate padding.
    
    # corr_i = sum_k w_k * a_{k-i} for k = 0,...,2n-2 and a_j defined for j=0,...,n-1

    # Then corr_i = sum_k w_k * b_{k-i} = (w * reverse(b))_{i}...

    # grad = 2 * real(ifft(W * conj(A)))[:n]

    # with appropriate padding to fsize >= 3n-2 or so.
    
    # Actually, let me just verify with a small example mentally.
    # a = [a0, a1], n=2
    # c_0 = a0^2, c_1 = 2*a0*a1, c_2 = a1^2

    # grad_1 = 2*w1*a0 + 2*w2*a1

    # i=1: w0*a1(=0... wait, a_{0-1} doesn't exist)

    # corr_i = sum_k w_k * a_{k-i} = (w * a_reflected)_i where a_reflected is a reversed

    # (w * a_reflected)_i = sum_k w_k * a_reflected_{k-i} = sum_k w_k * a_{n-1-(k-i)} = sum_k w_k * a_{n-1-k+i}
    # Hmm, that's not quite right either.
    
    # OK let me just use the direct FFT formula:
    # Define the "linear cross-correlation" as:
    # (w ⋆ a)_i = sum_k w_k * conj(a_{k-i})  [for real: sum_k w_k * a_{k-i}]
    # This equals ifft(fft(w) * conj(fft(a))) in the circular sense, with proper zero-padding.
    
    # For the linear version, I need fsize >= (2n-1) + n - 1 = 3n-2.
    # With fsize = next_power_of_2(3n-2):
    # corr = real(ifft(fft(w, fsize) * conj(fft(a, fsize))))
    # The first n elements give corr_0, ..., corr_{n-1}.
    
    # Wait, I need to check the indexing. The circular cross-correlation computed by
    # ifft(fft(w, N) * conj(fft(a, N))) gives:
    # corr_i = sum_k w_k * a_{(k-i) mod N}
    # With zero-padding (N >= len(w) + len(a) - 1), this equals the linear cross-correlation
    # for the valid range of i.
    
    # For our case: w has length 2n-1, a has length n.
    # N >= 2n-1 + n - 1 = 3n-2.
    # corr_i = sum_{k=0}^{2n-2} w_k * a_{k-i} (with a_j = 0 for j not in [0,n-1])
    # This is valid for i = 0, ..., n-1 (the range we need).
    
    # Let me verify with the n=2 example:
    # w = [w0, w1, w2] (length 3), a = [a0, a1] (length 2)
    # N >= 3+2-1 = 4. Use N=4.
    # FFT(w, 4) = [w0+w1+w2, w0-i*w1+w2, w0-w1+w2, w0+i*w1+w2]
    # FFT(a, 4) = [a0+a1, a0-i*a1, a0+a1, a0+i*a1]
    # conj(FFT(a,4)) = [a0+a1, a0+i*a1, a0+a1, a0-i*a1]

    # ifft of product gives corr of length 4.
    # corr_0 should be w0*a0 + w1*a1 + w2*0 = w0*a0 + w1*a1
    # Let me just trust the formula and verify numerically in the code.
    
    fsize2 = 1 << (3 * n - 2).bit_length()
    W = fft(np.concatenate([w, np.zeros(fsize2 - (2*n-1))]), fsize2)
    A = fft(np.concatenate([a, np.zeros(fsize2 - n)]), fsize2)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    grad = 2.0 * corr
    
    return obj, grad, c

def _run_fista_v2(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=10000.0,
                  beta_ramp_iters=3000):
    """FISTA with high-beta LSE and backtracking."""
    a = _normalize_sum(a_init, n)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    fsize = 1 << (2 * n - 1).bit_length()
    
    # Backtracking parameters
    c1 = 1e-4
    rho = 0.5
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max
        
        # Compute smoothed obj and gradient at y
        obj_y, grad_y, c_y = _lse_smoothed_obj_and_grad(y, beta)
        cur_max = c_y.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = 1.0 / (2.0 * beta)  # Initial guess
        for _ in range(20):
            x_new = _project_simplex(y - step * grad_y, S)
            obj_x, _, _ = _lse_smoothed_obj_and_grad(x_new, beta)
            if obj_x <= obj_y + np.dot(grad_y, x_new - y) + 0.5 / step * np.dot(x_new - y, x_new - y):
                break
            step *= rho
        else:
            # If backtracking fails, use a small step
            step = 1e-6
            x_new = _project_simplex(y - step * grad_y, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        # Periodic restarts to escape local minima
        if it % 5000 == 0:
            t_k = 1.0
            y = a.copy()
            # Add small noise
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-8:
                y = _project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def _multiplicative_gd(a_init, n, S, max_time, rng, beta_init=100.0, beta_max=5000.0,
                       n_beta_ramp=2000):
    """Multiplicative (exponentiated) gradient descent for the simplex."""
    a = _normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if it < n_beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / n_beta_ramp)
        else:
            beta = beta_max
        
        obj, grad, c = _lse_smoothed_obj_and_grad(a, beta)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 1.0 / (beta + it * 0.01)
        
        # Multiplicative update
        a_new = a * np.exp(-step * grad)
        # Handle potential underflow
        a_new = np.maximum(a_new, 1e-30)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)
        
        a = a_new
        
        # Periodic noise
        if it % 2000 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50000.0)
            if noise_scale > 1e-8:
                a = _normalize_sum(a + rng.normal(0, noise_scale, n), n)
    
    return best_a, best

def _peak_shaving_cd(a_init, n, S, max_time, rng):
    """Coordinate descent that targets the peak lags specifically."""
    a = _normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    c = _fft_convolve_self(a)
    best = c.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = _fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top lags
        top_k = np.argsort(c)[::-1][:5]
        
        improved = False
        n_tries = min(100 + it, 500)
        
        for _ in range(n_tries):
            if time.monotonic() - start > max_time - 2:
                break
            
            # Choose a peak lag to target
            k_target = top_k[rng.integers(0, len(top_k))]
            
            # The peak at lag k_target: c[k_target] = sum_j a_j * a_{k_target - j}
            # We want to reduce this by transferring mass.
            # Find the "hottest" pair (j, k_target - j) and transfer from the larger to a "cold" position.
            
            # Find j that maximizes a_j * a_{k_target - j}
            j_valid = np.arange(max(0, k_target - n + 1), min(k_target + 1, n))
            if len(j_valid) == 0:
                continue
            j_other = k_target - j_valid
            products = a[j_valid] * a[j_other]
            j_hot = j_valid[np.argmax(products)]
            j_hot_other = k_target - j_hot
            
            # Find a "cold" position to receive the mass
            # We want a position m such that transferring from j_hot to m reduces c[k_target]
            # Transfer delta from j_hot to m:
            # Change in c[k_target]: -a_{k_target - j_hot} * delta + a_{k_target - m} * delta
            # = delta * (a_{k_target - m} - a_{k_target - j_hot})
            # We want this to be negative, so we want a_{k_target - m} < a_{k_target - j_hot}
            
            hot_val = a[j_hot_other] if 0 <= j_hot_other < n else 0.0
            
            # Random candidate m
            for _try_m in range(10):
                m = rng.integers(0, n)
                if m == j_hot:
                    continue
                m_other = k_target - m
                cold_val = a[m_other] if 0 <= m_other < n else 0.0
                
                if cold_val >= hot_val:
                    continue
                
                # Optimal delta: reduce c[k_target] while not increasing other lags too much
                # For simplicity, use a small fixed fraction
                delta = 0.1 * a[j_hot]
                delta = min(delta, a[j_hot])
                
                a_trial = a.copy()
                a_trial[j_hot] -= delta
                a_trial[m] += delta
                if a_trial[j_hot] < -1e-10:
                    continue
                
                # Quick check: is c[k_target] reduced?
                # c[k_target] changes by delta * (cold_val - hot_val) < 0
                # But other lags might increase. Let's do a full check periodically.
                
                if rng.random() < 0.1:
                    c_trial = _fft
