"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Diverse multi-start with adaptive active-set FISTA.
Phase 1: Try 8 diverse window initializations with short FISTA trials.
Phase 2: Extended optimization on top candidates with adaptive hyperparameters.
Phase 3: Intensive active-set polishing with structural restarts.
Phase 4: Final verification.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_set_gradient(y, c, tol_frac):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    threshold = max_val * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Weight by proximity to max
    weights = c[active] - (max_val * (1.0 - tol_frac))
    weights = np.maximum(weights, 1e-10)
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[active] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_adaptive(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                       beta_max=10000.0, use_active=False, step_scale=1.0):
    """
    FISTA with adaptive switching between LSE and active-set gradients.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive gradient selection
        if use_active and stuck_count > 200:
            # Switch to active set based on level of stuckness
            if stuck_count < 2000:
                tol_frac = 0.005
            elif stuck_count < 10000:
                tol_frac = 0.01
            else:
                tol_frac = 0.02
            grad = active_set_gradient(y, c, tol_frac)
        else:
            grad = lse_gradient(y, c, beta)
            # Grow beta when making progress
            if stuck_count == 0:
                beta = min(beta_max, beta * (1.0 + beta_rate * 10))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_scale / max(beta, 1.0) if not (use_active and stuck_count > 200) else S / (gnorm * (1.0 + 0.002 * it))
        step = min(step, S / max(gnorm, 1e-10) * 2.0)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Stuck handling: structural restart
        if stuck_count > 3000 and stuck_count % 1000 == 0:
            # Try different restart strategies
            strategy = (stuck_count // 1000) % 4
            if strategy == 0:
                # Small noise
                noise_scale = 0.001 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            elif strategy == 1:
                # Flip a random segment
                seg_len = n // 8
                start_idx = rng.integers(0, n - seg_len)
                y[start_idx: start_idx + seg_len] = rng.uniform(0, 2 * S / n, seg_len)
                y = project_simplex(y, S)
            elif strategy == 2:
                # Shift the "energy center"
                shift = rng.integers(-n // 16, n // 16 + 1)
                y = np.roll(y, shift)
                y = project_simplex(y, S)
            else:
                # Multiply by random window
                tt = np.linspace(0, 1, n)
                center = rng.uniform(0.2, 0.8)
                width = rng.uniform(0.1, 0.5)
                window = np.exp(-0.5 * ((tt - center) / width) ** 2)
                y = y * (0.5 + 0.5 * window)
                y = project_simplex(y, S)
            t_mom = 1.0

        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_polish(a_init, max_time, rng):
    """Intensive polishing using adaptive active-set gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 100:
            tol_frac = 0.0
        elif stuck < 500:
            tol_frac = 0.001
        elif stuck < 2000:
            tol_frac = 0.005
        elif stuck < 10000:
            tol_frac = 0.01
        else:
            tol_frac = 0.015

        grad = active_set_gradient(y, c, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step with occasional increase
        base_step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(base_step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when very stuck
        if stuck > 10000 and it % 2000 == 0:
            noise_scale = max(1e-6, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

        # Momentum restart
        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def make_initializations(n, rng):

    inits['sinc'] = np.sinc(x) + 0.3

    r = rng.uniform(0.1, 1.0, n)

        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0:
            results.append((name, seq))

            results.append((name, np.ones(n)))
