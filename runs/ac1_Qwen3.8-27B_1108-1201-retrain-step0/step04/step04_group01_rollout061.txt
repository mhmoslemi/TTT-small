```python
"""
Search for non-negative sequences minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

The main strategy is a time-boxed multi-scale local search:
- Start from the previously found sequence (global height_sequence_1) when available.
- Upsample/downsample between lengths 8192, 16384, and 32768.
- Use mirror-descent in the simplex with weighted peak subgradients.
- Use a cross-correlation mirror descent step similar in spirit to the LP
  alternating method: fixed f, optimize g to minimize max(f*g).
- Occasionally generate a low-dimensional LP direction using HiGHS.
- Keep the internally evaluated best sequence and return it.
"""
import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_to_sum(a, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([S], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, S / a.size, dtype=np.float64)
    a = a * (S / s)
    return a


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cw = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = w > (cw - S) / idx
    valid = np.where(cond)[0]
    if valid.size == 0:
        rho = n
    else:
        rho = int(valid[-1]) + 1
    theta = (cw[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def get_base_sequence():
    try:
        obj = height_sequence_1
    except Exception:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.sum() < 1e-12:
        return None
    return arr


def resample_sequence(base, n):
    if base is None:
        return np.ones(n, dtype=np.float64)
    base = np.asarray(base, dtype=np.float64).reshape(-1)
    m = base.size
    if m == n:
        return base.copy()
    if m == 0 or n == 0:
        return np.ones(n, dtype=np.float64)
    if m > n:
        block = m // n
        if block >= 1:
            take = n * block
            if take >= m - max(1, n // 100):
                return base[:take].reshape(n, block).mean(axis=1)
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, base)


def compute_conv(a, N):
    n = a.size
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return A, c


def compute_conv_max(a, N):
    _, c = compute_conv(a, N)
    return float(c.max())


def lse_value(c, beta):
    beta = max(1e-6, float(beta))
    m = float(c.max())
    x = beta * (c - m)
    x = np.clip(x, -700.0, 0.0)
    t = np.exp(x)
    return m + float(np.log(t.sum())) / beta


def weights_from_conv(c, mode, beta, k):
    n = c.size
    out = np.zeros(n, dtype=np.float64)
    if mode == "top":
        kk = int(max(1, min(k, n)))
        if kk >= n:
            out[:] = 1.0 / n
        else:
            idx = np.argpartition(c, n - kk)[n - kk:]
            out[idx] = 1.0 / kk
        return out
    if mode == "excess":
        cm = float(c.max())
        cs = float(c.mean())
        threshold = cm - 0.10 * max(1e-12, cm - cs)
        w = np.maximum(c - threshold, 0.0)
        s = float(w.sum())
        if s <= 1e-30:
            return weights_from_conv(c, "top", beta, k)
        w /= s
        return w
    # default: log-sum-exp softmax
    b = max(1e-3, float(beta))
    cm = float(c.max())
    x = b * (c - cm)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 1e-300:
        idx = int(np.argmax(c))
        out[idx] = 1.0
        return out
    return e / s


def gradient_from_weights(w, A, n, factor, N):
    W = fft(w, N)
    g = factor * np.real(ifft(W * np.conj(A)))[:n]
    return g


def mirror_update(a, g, eta, S):
    n = a.size
    gc = g - float(g.mean())
    gmax = float(np.max(np.abs(gc)))
    if gmax < 1e-16:
        return a.copy()
    ex = -float(eta) * gc
    ex = np.clip(ex, -30.0, 30.0)
    a_new = a * np.exp(ex)
    floor = 1e-14 * max(S / n, 1e-300)
    a_new = np.maximum(a_new, floor)
    s = float(a_new.sum())
    if s < 1e-300:
        return np.full(n, S / n, dtype=np.float64)
    a_new *= S / s
    return a_new


def multiplicative_noise(a, S, sigma, rng):
    n = a.size
    z = rng.normal(0.0, 1.0, n)
    fac = np.exp(np.clip(z * sigma, -5.0, 5.0))
    return normalize_to_sum(a * fac, S)


def run_self_mirror(a, N, S, max_time, beta0, beta_max, mode, eta_init, k, rng):
    max_time = float(max_time)
    if max_time <= 0:
        a = normalize_to_sum(a, S)
        return a, compute_conv_max(a, N)

    start = time.monotonic()
    a = normalize_to_sum(a, S)
    n = a.size
    A, c = compute_conv(a, N)
    m = float(c.max())
    best_a = a.copy()
    best_m = m

    beta = max(1e-1, float(beta0))
    eta = float(eta_init) if eta_init is not None else 0.05 / max(S, 1e-12)
    last_improve = start
    stagnant = 0

    while time.monotonic() - start < max_time:
        if m < best_m:
            best_m = m
            best_a = a.copy()
            last_improve = time.monotonic()
            stagnant = 0

        w = weights_from_conv(c, mode, beta, k)
        g = gradient_from_weights(w, A, n, 2.0, N)
        gc = g - float(g.mean())
        gmax = float(np.max(np.abs(gc)))
        now = time.monotonic()

        if gmax < 1e-14 or stagnant > 60 or now - last_improve > 18.0:
            sig = 0.05 if max_time > 30 else 0.09
            a = multiplicative_noise(a, S, sig, rng)
            A, c = compute_conv(a, N)
            m = float(c.max())
            stagnant = 0
            last_improve = time.monotonic()
            eta = 0.05 / max(S, 1e-12)
            continue

        max_eta = 0.20 / max(gmax, 1e-12)
        if eta > max_eta:
            eta = max_eta

        a_new = mirror_update(a, gc, eta, S)
        A_new, c_new = compute_conv(a_new, N)
        m_new = float(c_new.max())
        m_old = m

        accepted = False
        if mode == "lse":
            lold = lse_value(c, beta)
            lnew = lse_value(c_new, beta)
            if m_new < m_old - 1e-12:
                accepted = True
            elif lnew < lold - 1e-10:
                accepted = True
            elif lnew < lold * 0.995 and m_new <= m_old * 1.01:
                accepted = True
        else:
            accepted = (m_new < m_old - 1e-12)

        if accepted:
            a = a_new
            A = A_new
            c = c_new
            m = m_new
            if m_new < m_old - 1e-12:
                stagnant = 0
                eta = min(eta * 1.05, 0.25 / max(gmax, 1e-12))
            else:
                stagnant += 1
                if m_new > m_old:
                    eta *= 0.85
        else:
            stagnant += 2
            eta *= 0.7

        if mode == "lse":
            beta = min(float(beta_max), beta * 1.0007 + 1e-3)

    if m < best_m:
        best_m = m
        best_a = a.copy()
    return best_a, best_m


def run_projected_top(a, N, S, max_time, k, step_init, rng):
    max_time = float(max_time)
    if max_time <= 0:
        a = normalize_to_sum(a, S)
        return a, compute_conv_max(a, N)

    start = time.monotonic()
    a = normalize_to_sum(a, S)
    n = a.size
    A, c = compute_conv(a, N)
    m = float(c.max())
    best_a = a.copy()
    best_m = m
    step = max(0.01, float(step_init))
    last_improve = start
    stagnant = 0

    while time.monotonic() - start < max_time:
        if m < best_m:
            best_m = m
            best_a = a.copy()
            last_improve = time.monotonic()
            stagnant = 0

        w = weights_from_conv(c, "top", 1.0, k)
        g = gradient_from_weights(w, A, n, 2.0, N)
        d = g - float(g.mean())
        dnorm = float(np.linalg.norm(d))
        now = time.monotonic()

        if dnorm < 1e-13 or stagnant > 80 or now - last_improve > 20.0:
            a = multiplicative_noise(a, S, 0.04, rng)
            A, c = compute_conv(a, N)
            m = float(c.max())
            step = max(0.005, step_init * 0.5)
            stagnant = 0
            last_improve = time.monotonic()
            continue

        d = d / dnorm
        accepted = False
        for _ in range(5):
            y = project_simplex(a - step * d, S)
            Ay, cy = compute_conv(y, N)
            my = float(cy.max())
            if my < m - 1e-12:
                a = y
                A = Ay
                c = cy
                m = my
                step = min(0.12, step * 1.12)
                accepted = True
                stagnant = 0
                break
            step *= 0.62
            if step < 1e-7:
                break

        if not accepted:
            stagnant += 2
            if stagnant > 60:
                a = multiplicative_noise(a, S, 0.04, rng)
                A, c = compute_conv(a, N)
                m = float(c.max())
                step = max(0.005, step_init * 0.5)
                stagnant = 0
                last_improve = now

    if m < best_m:
        best_m = m
        best_a = a.copy()
    return best_a, best_m


def run_cross_mirror(f, g_init, N, S, max_time, beta0, beta_max, rng):
    max_time = float(max_time)
    if max_time <= 0:
        g = normalize_to_sum(g_init, S)
        return g, compute_conv_max(g, N)

    start = time.monotonic()
    f = normalize_to_sum(f, S)
    n = f.size
    F = fft(f, N)
    g = normalize_to_sum(g_init, S)

    G, h = (fft(g, N), np.real(ifft(F * fft(g, N)))[:2 * n - 1])
    m = float(h.max())
    best_g = g.copy()
    best_m = m

    beta = max(1.0, float(beta0))
    eta = 0.05 / max(S, 1e-12)
    last_improve = start
    stagnant = 0

    while time.monotonic() - start < max_time:
        if m < best_m:
            best_m = m
            best_g = g.copy()
            last_improve = time.monotonic()
            stagnant = 0

        w = weights_from_conv(h, "lse", beta, 64)
        grad = gradient_from_weights(w, F, n, 1.0, N)
        gc = grad - float(grad.mean())
        gmax = float(np.max(np.abs(gc)))
        now = time.monotonic()

        if gmax < 1e-14 or stagnant > 60 or now - last_improve > 15.0:
            g = multiplicative_noise(g, S, 0.05, rng)
            G = fft(g, N)
            h = np.real(ifft(F * G))[:2 * n - 1]
            m = float(h.max())
            stagnant = 0
            last_improve = time.monotonic()
            eta = 0.05 / max(S, 1e-12)
            continue

        max_eta = 0.20 / max(gmax, 1e-12)
        if eta > max_eta:
            eta = max_eta

        g_new = mirror_update(g, gc, eta, S)
        G_new = fft(g_new, N)
        h_new = np.real(ifft(F * G_new))[:2 * n - 1]
        m_new = float(h_new.max())
        m_old = m

        accepted = False
        lold = lse_value(h, beta)
        lnew = lse_value(h_new, beta)
        if m_new < m_old - 1e-12:
            accepted = True
        elif lnew < lold - 1e-10:
            accepted = True
        elif lnew < lold * 0.995 and m_new <= m_old * 1.01:
            accepted = True

        if accepted:
            g = g_new
            G = G_new
            h = h_new
            m = m_new
            if m_new < m_old - 1e-12:
                stagnant = 0
                eta = min(eta * 1.05, 0.25 / max(gmax, 1e-12))
            else:
                stagnant += 1
                if m_new > m_old:
                    eta *= 0.85
        else:
            stagnant += 2
            eta *= 0.7

        beta = min(float(beta_max), beta * 1.001 + 1e-3)

    if m < best_m:
        best_m = m
        best_g = g.copy()
    return best_g, best_m


def mix_and_pick(f, g, alphas, S, N):
    f = normalize_to_sum(f, S)
    g = normalize_to_sum(g, S)
    best = f.copy()
    Af = fft(f, N)
    cf = np.real(ifft(Af * Af))
    best_m = float(np.max(np.maximum(cf, 0.0)))

    for alpha in alphas:
        alpha = float(alpha)
        if alpha <= 0.0:
            continue
        if alpha >= 1.0:
            cand = g.copy()
        else:
            cand = normalize_to_sum((1.0 - alpha) * f + alpha * g, S)
        Ac = fft(cand, N)
        cc = np.real(ifft(Ac * Ac))
        mc = float(np.max(np.maximum(cc, 0.0)))
        if mc < best_m - 1e-12:
            best = cand
            best_m = mc
    return best, best_m


def lp_cross_direction(f, time_limit, nmax):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = f.size
    if n < 2 or n > nmax:
        return None
    N = next_pow2(3 * n - 2)
    c = compute_conv(f, N)[1]
    M = float(c.max()) * (1.0 + 1e-10)
    if not np.isfinite(M) or M <= 0:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    obj = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={
                "time_limit": float(max(1.0, time_limit)),
                "presolve": True,
            },
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None
    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    if x.sum() < 1e-30:
        return None
    return x


def lp_global_candidate(a_source, n_target, time_available, rng):
    time_available = float(time_available)
    if time_available < 25.0 or n_target < 512:
        return None
    small_n = 512 if time_available < 100.0 else 1024
    f_small = resample_sequence(a_source, small_n)
    S_small = np.sqrt(2.0 * small_n)
    f_small = normalize_to_sum(f_small, S_small)
    limit = min(40.0, max(10.0, time_available * 0.35))

    g_small = None
    try:
        g_small = lp_cross_direction(f_small, limit, 1024)
    except Exception:
        g_small = None

    if g_small is None:
        return None
    g_small = normalize_to_sum(g_small, S_small)
    g_big = resample_sequence(g_small, n_target)
    S_big = np.sqrt(2.0 * n_target)
    g_big = normalize_to_sum(g_big, S_big)
    return g_big


def run_lp_alternation_small(f_init, n, deadline, rng):
    deadline = float(deadline)
    start = time.monotonic()
    S = np.sqrt(2.0 * n)
    f = normalize_to_sum(f_init, S)
    N = next_pow2(3 * n - 2)
    best_f = f.copy()
    best_m = compute_conv_max(f, N)
    alphas = np.array([1.0, 0.65, 0.4, 0.2, 0.1], dtype=np.float64)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        limit = min(20.0, max(8.0, rem * 0.30))
        try:
            g = lp_cross_direction(f, limit, n)
        except Exception:
            break
        if g is None:
            break
        g = normalize_to_sum(g, S)

        cur_best = best_f.copy()
        cur_best_m = best_m
        for alpha in alphas:
            if alpha >= 1.0:
                cand = g.copy()
            else:
                cand = normalize_to_sum((1.0 - alpha) * best_f + alpha * g, S)
            cm = compute_conv_max(cand, N)
            if cm < cur_best_m - 1e-10:
                cur_best = cand
                cur_best_m = cm

        if cur_best_m < best_m - 1e-7:
            best_f = cur_best
            best_m = cur_best_m
            f = best_f.copy()
        else:
            break

    return best_f, best_m


def optimize_one_length(a_init, n, S, deadline, rng, allow_lp):
    N = next_pow2(3 * n - 2)
    best_a = normalize_to_sum(a_init, S)
    best_m = compute_conv_max(best_a, N)
    print("    n=%d initial objective=%.7f" % (n, best_m), flush=True)

    def update(cand, cm):
        nonlocal best_a, best_m
        if cm < best_m - 1e-12:
            best_a = cand
            best_m = cm
            print("      improved to %.7f" % best_m, flush=True)

    rem = deadline - time.monotonic()

    # 1) smooth full-spectrum mirror descent
    bud = min(120.0, rem * 0.25)
    if bud > 10.0:
        a2, m2 = run_self_mirror(
            best_a, N, S, bud, 6.0, 220.0, "lse", None, 96, rng
        )
        m2 = compute_conv_max(a2, N)
        update(a2, m2)

    rem = deadline - time.monotonic()
    cross_total = min(260.0, rem * 0.38)
    used_cross = 0.0
    while rem > 60.0 and used_cross < cross_total:
        cb = min(60.0, cross_total - used_cross)
        if cb < 18.0:
            break
        g0 = run_cross_mirror(best_a, best_a, N, S, cb, 18.0, 450.0, rng)[0]
        gp = g0
        if cb > 25.0:
            ppb = min(20.0, cb * 0.35)
            gp = run_self_mirror(
                g0, N, S, ppb, 25.0, 350.0, "excess", None, 96, rng
            )[0]

        alphas = np.array([1.0, 0.75, 0.5, 0.3, 0.15], dtype=np.float64)
        cand1, m1 = mix_and_pick(best_a, g0, alphas, S, N)
        update(cand1, m1)
        if gp is not g0:
            cand2, m2 = mix_and_pick(best_a, gp, alphas, S, N)
            update(cand2, m2)

        used_cross += cb
        rem = deadline - time.monotonic()

    if allow_lp:
        rem = deadline - time.monotonic()
        if rem > 120.0:
            lg = lp_global_candidate(best_a, n, rem * 0.35, rng)
            if lg is not None:
                alphas = np.array([1.0, 0.65, 0.45, 0.3], dtype=np.float64)
                cand, cm = mix_and_pick(best_a, lg, alphas, S, N)
                update(cand, cm)

    rem = deadline - time.monotonic()
    if rem > 25.0:
        bud = min(90.0, rem * 0.35)
        a2, m2 = run_self_mirror(
            best_a, N, S, bud, 60.0, 800.0, "excess", None, 128, rng
        )
        m2 = compute_conv_max(a2, N)
        update(a2, m2)

    rem = deadline - time.monotonic()
    if rem > 25.0:
        bud = min(60.0, rem * 0.35)
        a2, m2 = run_projected_top(best_a, N, S, bud, 20, 0.02, rng)
        m2 = compute_conv_max(a2, N)
        update(a2, m2)

    rem = deadline - time.monotonic()
    if rem > 15.0:
        bud = min(50.0, rem * 0.5)
        a2, m2 = run_self_mirror(
            best_a, N, S, bud, 800.0, 5000.0, "top", None, 8, rng
        )
        m2 = compute_conv_max(a2, N)
        update(a2, m2)

    best_a = np.clip(best_a, 0.0, None)
    return best_a, best_m


def propose_candidate(seed=42, budget_s=2400):
    start = time.monotonic()
    budget_s = max(30.0, float(budget_s))
    deadline = start + max(5.0, budget_s - 90.0)
    rng = np.random.default_rng(seed)

    base = get_base_sequence()
    sources = []
    if base is not None:
        sources.append(base)

    # Initial global fallback
    if base is not None and 512 <= base.size <= 32768:
        global_a = base.copy()
    elif base is not None:
        global_a = resample_sequence(base, 32768 if base.size > 32768 else 8192)
    else:
        global_a = np.ones(8192, dtype=np.float64)

    global_n = len(global_a)
    global_S = np.sqrt(2.0 * global_n)
    global_a = normalize_to_sum(global_a, global_S)
    global_N = next_pow2(3 * global_n - 2)
    global_obj = compute_conv_max(global_a, global_N)
    print("Initial global objective: %.7f (n=%d)" % (global_obj, global_n), flush=True)

    # Low-resolution LP alternation for a different, possibly global seed.
    total_available = deadline - time.monotonic()
    if total_available > 1600.0:
        small_n = 1024 if total_available > 2100.0 else 512
        f_small = resample_sequence(base if base is not None else global_a, small_n)
        S_small = np.sqrt(2.0 * small_n)
        f_small = normalize_to_sum(f_small, S_small)
        tiny_deadline = time.monotonic() + min(80.0, max(25.0, total_available * 0.08))
        lp_seed, lp_m = run_lp_alternation_small(f_small, small_n, tiny_deadline, rng)
        if lp_m < 2.0:
            sources.append(lp_seed)
            print("Low-res LP seed objective: %.7f (n=%d)" % (lp_m, small_n), flush=True)

    total_available = deadline - time.monotonic()
    if total_available < 120.0:
        out = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        return out

    if total_available < 700.0:
        lengths = [8192]
    elif total_available < 1500.0:
        lengths = [8192, 16384]
    else:
        lengths = [8192, 16384, 32768]

    # Avoid duplicating too much work if base is already very large.
    if base is not None and base.size > 16384:
        lengths = [16384, 32768]

    # Allocate more time to larger lengths, since they usually help.
    if len(lengths) == 3:
        fracs = np.array([0.20, 0.40, 0.40], dtype=np.float64)
    elif len(lengths) == 2:
        fracs = np.array([0.30, 0.70], dtype=np.float64)
    else:
        fracs = np.array([1.0], dtype=np.float64)

    for i, n in enumerate(lengths):
        time_budget = max(20.0, (deadline - time.monotonic()) * fracs[i])
        if i == len(lengths) - 1 and time_budget > 30.0:
            # Use whatever remains, but leave a tiny buffer.
            time_budget = min(time_budget, max(20.0, deadline - time.monotonic() - 5.0))

        # Choose the best source for this length.
        Nsel = next_pow2(3 * n - 2)
        init_a = None
        init_obj = np.inf
        for src in sources:
            cand = resample_sequence(src, n)
            cand = normalize_to_sum(cand, np.sqrt(2.0 * n))
            obj = compute_conv_max(cand, Nsel)
            if obj < init_obj:
                init_obj = obj
                init_a = cand
        if init_a is None:
            init_a = normalize_to_sum(np.ones(n), np.sqrt(2.0 * n))
            init_obj = compute_conv_max(init_a, Nsel)

        print("Optimizing length %d, available %.1fs, init obj %.7f"
              % (n, time_budget, init_obj), flush=True)

        S = np.sqrt(2.0 * n)
        phase_deadline = time.monotonic() + time_budget
        allow_lp = (n >= 4096 and time_budget > 250.0)
        a_opt, m_opt = optimize_one_length(
            init_a, n, S, phase_deadline, rng, allow_lp
        )

        if m_opt < global_obj - 1e-12:
            global_a = a_opt.copy()
            global_obj = m_opt
            print("New global best: %.7f (n=%d)" % (global_obj, n), flush=True)

        if len(sources) >= 4:
            sources.pop(0)
        sources.append(a_opt.copy())

    # Final sanity normalization/clipping.
    global_a = np.asarray(global_a, dtype=np.float64).reshape(-1)
    global_a = np.where(np.isfinite(global_a), global_a, 0.0)
    global_a = np.clip(global_a, 0.0, 1000.0)
    gs = float(global_a.sum())
    if gs < 1e-8:
        n = 8192
        global_a = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        gs = float(global_a.sum())
    global_a = global_a * (np.sqrt(2.0 * len(global_a)) / gs)
    global_a = np.clip(global_a, 0.0, 1000.0)

    final_list = [float(x) for x in global_a]
    try:
        if len(final_list) <= 4096:
            ver = evaluate_sequence(final_list)
            print("Final internal objective: %.7f, evaluate_sequence: %s"
                  % (global_obj, ver), flush=True)
        else:
            print("Final internal objective: %.7f (skipped local evaluate_sequence for n=%d)"
                  % (global_obj, len(final_list)), flush=True)
    except Exception as e:
        print("Final internal objective: %.7f (evaluate_sequence not called: %s)"
              % (global_obj, e), flush=True)

    return final_list
```
