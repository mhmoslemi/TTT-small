`da[k]/dz[j] = S * (softmax(z)[k] * (delta(k,j) - softmax(z)[j]))`

So `df/dz[j] = sum_k (df/da[k]) * S * softmax(z)[k] * (delta(k,j) - softmax(z)[j])`

`= S * a[j]/S * (df/da[j] - sum_k (df/da[k]) * a[k]/S)`
`= a[j] * (df/da[j] - (1/S) * sum_k (df/da[k]) * a[k])`

`db[k']/da[j] = a[k'-j] + a[j-k']` (with appropriate index bounds)

    # FFT(p_padded) = P, FFT(a_padded) = A
    # g_full = ifft(conj(P) * A)
    # 
    # g_full[m] = sum_k p_padded[k]*a_padded[(k-m) mod 8] ... (circular)
    # 
    # For m=0: g_full[0] = p0*a0 + p1*a1 + p2*a2 + ... = p0*a0 + p1*a1 ✓
    # For m=1: g_full[1] = p0*a[-1 mod 8] + p1*a0 + p2*a1 + ... = p1*a0 + p2*a1 ✓ (since a[7]=0)
    # 
    # Great! So the FFT formula works.
    
    # So the gradient of the smoothed objective w.r.t. a is:
    # grad_a[m] = 2 * ifft(conj(FFT(p)) * FFT(a))[m] for m = 0, ..., n-1
    # (with proper zero-padding)
    
    # And the gradient w.r.t. z (the log-softmax parameter):
    # a = S * softmax(z)
    # da[k]/dz[j] = a[k] * (delta(k,j) - a[j]/S)
    # 
    # df/dz[j] = sum_k (df/da[k]) * da[k]/dz[j]
    # = sum_k grad_a[k] * a[k] * (delta(k,j) - a[j]/S)
    # = a[j] * (grad_a[j] - sum_k grad_a[k]*a[k]/S)
    # = a[j] * (grad_a[j] - mean_weighted)
    # where mean_weighted = sum_k grad_a[k]*a[k] / S
    
    # Now, since we're using the LSE approximation, the objective value is:
    # f ≈ 2n/S^2 * (max_c + log(sum exp(beta*(c - max_c)))/beta)
    # For large beta, this is approximately 2n/S^2 * max_c.
    
    # For the actual objective value (not the smoothed one), we should use:
    # f_exact = 2n * max_c / S^2
    
    p_full = np.zeros(fsize)
    p_full[:2*n-1] = w
    A_full = np.zeros(fsize)
    A_full[:n] = a
    
    grad_a = 2.0 * np.real(ifft(np.conj(fft(p_full, fsize)) * fft(A_full, fsize)))[:n]
    
    # Gradient w.r.t. z
    mean_weighted = np.dot(grad_a, a) / S
    grad_z = a * (grad_a - mean_weighted)
    
    return f_val, grad_z

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized L-BFGS-B
with LSE smoothing, multiple initializations, adaptive sizes, and active-set polish.
Key innovation: reparameterize via softmax to handle simplex constraint smoothly,
use LSE with adaptive beta for the max penalty, and multi-scale search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.signal import gaussian_filter1d


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_obj_and_grad(z, S, n, beta):
    """
    LSE-smoothed objective and gradient for softmax parameterization.
    a = S * softmax(z), minimize 2n*max(conv(a,a))/S^2.
    """
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = S * exp_z / exp_z.sum()

    c = fft_convolve_self(a)
    max_c = c.max()

    # LSE-smoothed objective
    shifted = beta * (c - max_c)
    shifted = shifted - shifted.max()
    log_sum = shifted.max() + np.log(np.exp(shifted).sum())
    f_val = (2.0 * n / (S * S)) * (max_c + log_sum / beta)

    # Probability weights
    w = np.exp(shifted) / np.exp(shifted).sum()

    # Gradient of smoothed objective w.r.t. a:
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (with a[i]=0 outside [0,n-1])
    # FFT computation:
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    fsize = max(fsize, 1 << (2 * 2 * n - 1 - 1).bit_length())
    P = np.zeros(fsize)
    P[:2 * n - 1] = w
    A = np.zeros(fsize)
    A[:n] = a

    P_fft = fft(P)
    A_fft = fft(A)

    # grad_a[j] = 2 * ifft(conj(P_fft) * A_fft)[j] for j = 0..n-1
    grad_a = 2.0 * np.real(ifft(np.conj(P_fft) * A_fft))[:n]

    # Gradient w.r.t. z:
    # df/dz[j] = a[j] * (grad_a[j] - sum_k grad_a[k]*a[k]/S)
    mean_weighted = np.dot(grad_a, a) / S
    grad_z = a * (grad_a - mean_weighted)

    return f_val, grad_z


def exact_subgrad_conv(a, k_star):
    """Exact subgradient of max(conv(a,a)) at peak lag k_star."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        km = k_star - j
        if 0 <= km < n:
            grad[j] = 2.0 * a[km]
    return grad

        # Average subgradient over active lags
        grad = np.zeros(n)
        for ks in k_list:
            for j in range(n):
                km = ks - j
                if 0 <= km < n:
                    grad[j] += 2.0 * y[km]
        grad /= len(k_list)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Decreasing step
        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, 2.0 * S / (gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when stuck
        if stuck > 1000 and it % 200 == 0:
            ns = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            t_mom = 1.0

    return best_a


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search: L-BFGS with LSE + FISTA + active-set polish.
    Tries multiple sizes and initializations to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence from previous solution
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Initial: n=%d, obj=%.6f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # Helper: make initializations for a given n
    def make_inits(n):
        S = np.sqrt(2.0 * n)
        tt = np.linspace(0, 1, n, endpoint=False)
        inits = {}

        # From previous best
        if a_start is not None:
            if len(a_start) == n:
                inits['prev'] = a_start * (S / a_start.sum())
            else:
                # Interpolate to new size
                t_old = np.linspace(0, 1, len(a_start), endpoint=False)
                t_new = np.linspace(0, 1, n, endpoint=False)
                a_interp = np.interp(t_new, t_old, a_start)
                a_interp = np.maximum(a_interp, 0)
                inits['prev_interp'] = a_interp * (S / a_interp.sum())

        # Cosine
        a = (0.54 + 0.46 * np.cos(2 * np.pi * tt))
        inits['cos'] = a * (S / a.sum())

        # Hann window
        a = 0.5 * (1 - np.cos(2 * np.pi * tt))
        inits['hann'] = a * (S / a.sum())

        # Gaussian
        x = (np.arange(n) - n / 2) / (n / 4.0)
        a = np.exp(-0.5 * x**2)
        inits['gauss'] = a * (S / a.sum())

        # Flat-top
        a = np.ones(n)
        inits['flat'] = a * (S / a.sum())

        # Power law
        d = np.abs(np.arange(n) - n / 2) / (n / 2)
        a = np.maximum(1 - d**1.5, 0)
        if a.sum() > 0:
            inits['pow15'] = a * (S / a.sum())

        a = np.maximum(1 - d**3, 0)
        if a.sum() > 0:
            inits['pow3'] = a * (S / a.sum())

        # Random smooth
        for i in range(2):
            a_raw = np.abs(rng.normal(0, 1, n))
            a_smooth = gaussian_filter1d(a_raw, sigma=n / 40.0)
            inits['rand%d' % i] = a_smooth * (S / a_smooth.sum())

        return inits

    # === Phase 1: L-BFGS with LSE at multiple sizes ===
    sizes_to_try = []
    if a_start is not None:
        sizes_to_try.append(len(a_start))
    for s in [8192, 16384, 32768]:
        if s not in sizes_to_try:
            sizes_to_try.append(s)

    # Sort by size (start with original, then larger)
    sizes_to_try = sorted(set(sizes_to_try))[:5]

    print("\n=== Phase 1: L-BFGS optimization ===")
    t_phase1 = time.monotonic()
    p1_budget = (deadline - t_phase1) * 0.40

    for n in sizes_to_try:
        if time.monotonic() > deadline - (deadline - t_phase1) * 0.6:
            break

        S = np.sqrt(2.0 * n)
        inits = make_inits(n)
        n_per_init = max(200, int(p1_budget / (len(inits) * 2.0)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 120:
                break

            # Convert to softmax parameterization
            z0 = np.log(np.maximum(a0, 1e-300))

            # LSE with adaptive beta
            def make_func(beta_start, beta_end, beta_iters):
                state = {'it': 0}
                def func(z):
                    beta = beta_start + (beta_end - beta_start) * min(1.0, state['it'] / beta_iters)
                    state['it'] += 1
                    f, _ = lse_obj_and_grad(z, S, n, beta)
                    return f
                return func

            def func_wrapper(z, beta):
                f, _ = lse_obj_and_grad(z, S, n, beta)
                return f

            def func_grad_wrapper(z, beta):
                _, g = lse_obj_and_grad(z, S, n, beta)
                return g

            # Run L-BFGS with increasing beta
            t0 = time.monotonic()
            z_cur = z0.copy()
            best_z = z0.copy()
            best_f = np.inf

            for beta_round in [50, 200, 500, 1500, 4000]:
                if time.monotonic() > deadline - 100:
                    break
                f_round = 0

                def f_func(z, _beta=beta_round):
                    nonlocal f_round
                    f_round += 1
                    fv, _ = lse_obj_and_grad(z, S, n, _beta)
                    return fv

                def g_func(z, _beta=beta_round):
                    _, g = lse_obj_and_grad(z, S, n, _beta)
                    return g

                try:
                    result = minimize(f_func, z_cur, jac=g_func,
                                      method='L-BFGS-B',
                                      options={'maxiter': 300, 'gtol': 1e-7,
                                               'ftol': 1e-12, 'maxfun': 1000})
                    f_val = compute_obj(S * np.exp(result.x - result.x.max()))
                    f_val = f_val * (S * S) / (S * S)  # same thing
                    # Recompute properly
                    a_test = S * np.exp(result.x - result.x.max())
                    a_test = a_test / a_test.sum() * S
                    f_val = compute_obj(a_test)

                    if f_val < best_f:
                        best_f = f_val
                        best_z = result.x.copy()
                    z_cur = result.x.copy()
                except Exception as e:
                    pass

            # Extract best solution
            if best_f < np.inf:
                a_opt = S * np.exp(best_z - best_z.max())
                a_opt = a_opt / a_opt.sum() * S
                obj = compute_obj(a_opt)
                elapsed = time.monotonic() - t0
                print("  n=%d %-12s obj=%.6f (t=%.1fs)" % (n, name, obj, elapsed))

                if obj < global_best_obj:
                    global_best_obj = obj
                    global_best_a = a_opt.copy()
                    print("    *** NEW BEST! ***")

    # === Phase 2: FISTA polish on best solution ===
    print("\n=== Phase 2: FISTA polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_best_a is not None:
        p2_budget = remaining * 0.35
        t2 = time.monotonic()
        n_best = len(global_best_a)

        # Run FISTA with different beta parameters
        for beta_init in [10, 50, 200]:
            if time.monotonic() > deadline - 30:
                break
            t_sub = min(remaining - (time.monotonic() - t2), p2_budget / 3.0)
            a_out = run_fista_smooth(global_best_a, max(t_sub, 5),
                                     beta_init=beta_init, beta_rate=0.004,
                                     beta_max=5000.0)
            obj = compute_obj(a_out)
            print("  FISTA beta=%d: obj=%.6f (t=%.0fs)" % (beta_init, obj, time.monotonic() - t2))
            if obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a_out.copy()
                print("    *** NEW BEST! ***")

    # === Phase 3: Active-set polish ===
    print("\n=== Phase 3: Active-set polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_best_a is not None:
        p3_budget = remaining * 0.40
        t3 = time.monotonic()

        a_out = run_active_polish(global_best_a, p3_budget, rng)
        obj = compute_obj(a_out)
        print("  Active polish: obj=%.6f (t=%.0fs)" % (obj, time.monotonic() - t3))
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_out.copy()
            print("    *** NEW BEST! ***")

        # Another round with different seed
        remaining3 = deadline - time.monotonic()
        if remaining3 > 30:
            a_out2 = run_active_polish(global_best_a, remaining3 * 0.8,
                                       np.random.default_rng(777))
            obj2 = compute_obj(a_out2)
            print("  Active polish 2: obj=%.6f" % obj2)
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_best_a = a_out2.copy()
                print("    *** NEW BEST! ***")

    # === Phase 4: Try upsample if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_best_a is not None and len(global_best_a) <= 8192:
        print("\n=== Phase 4: Upsample to 2n ===")
        n_up = len(global_best_a) * 2
        t_up = np.linspace(0, 1, len(global_best_a), endpoint=False)
        t_up2 = np.linspace(0, 1, n_up, endpoint=False)
        a_up = np.interp(t_up2, t_up, global_best_a)
        S_up = np.sqrt(2.0 * n_up)
        a_up = a_up * (S_up / a_up.sum())

        p4_budget = min(remaining - 20, 120)
        a_up2 = run_fista_smooth(a_up, p4_budget, beta_init=10, beta_rate=0.005)
        obj_up = compute_obj(a_up2)
        print("  Upsampled n=%d: obj=%.6f" % (n_up, obj_up))
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up2.copy()
            print("    *** NEW BEST! ***")

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")

    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    print("Fallback: obj=%.6f" % compute_obj(a_fb))
    return [float(x) for x in a_fb]
