"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA with optimal beta schedule + lag-aware coordinate descent + spectral polish.
Key innovations: correct FFT-based gradient, peak-targeted CD pairing, multi-scale restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import math


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective_val(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def fista_gradient(w, y, n):
    """Compute gradient of LSE(max(c)) w.r.t. sequence.
    grad[j] = (w cross-correlate y)[j] + y[j]
    using zero-padded FFT."""
    fft_size = 1 << (3 * n).bit_length()
    W = fft(w, fft_size)
    Y = fft(y, fft_size)
    cross = np.real(ifft(W * np.conj(Y)))[:n]
    return cross + y


def run_fista(a_init, n, S, max_time, rng, beta_init=100.0, beta_max=10000.0,
              beta_ramp=3000, noise_interval=8000):
    """FISTA with adaptive LSE smoothing and correct FFT gradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp))
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        grad = fista_gradient(w, y, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        if it - last_improve > 3000:
            step *= 0.5
        if it - last_improve > 6000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % noise_interval == 0:
            decay = np.exp(-it / (noise_interval * 10.0))
            ns = 0.002 * S / np.sqrt(n) * decay
            if ns > 1e-9:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0
                a = y.copy()

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_fista_hybrid(a_init, n, S, max_time, rng):
    """FISTA that switches to exact subgradient near convergence."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 500.0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        # Switch to subgradient mode when close to convergence
        if it > 2000 and rng.random() < min(0.5, 0.01 * it / 2000):
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            grad = np.zeros(n)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * y[k_star - idx]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (gnorm * (1.0 + 0.05 * np.log1p(it / 100.0)))
            step = min(step, 2.0 / beta)
        else:
            e = np.exp(beta * (c - cm))
            tot = e.sum()
            if tot < 1e-300:
                break
            w = e / tot
            grad = fista_gradient(w, y, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            if it - last_improve > 2000:
                step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            decay = np.exp(-it / 100000.0)
            ns = 0.001 * S / np.sqrt(n) * decay
            if ns > 1e-9:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0
                a = y.copy()

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coordinate_descent_peak(a_init, n, S, max_time, rng):
    """Lag-aware coordinate descent targeting the peak."""
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best = c.max()
    best_a = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        c_max = c.max()
        if c_max < best - 1e-10:
            best = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 80:
            ns = 0.003 * S / np.sqrt(n) * np.exp(-stall / 20.0)
            if ns < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fft_convolve_self(a)
            stall = 0
            continue

        k_star = int(np.argmax(c))
        n_pairs = min(200 + it * 3, 2000)
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 1.0:
                break

            # Smart pair selection
            r = rng.random()
            if r < 0.25:
                # Target peak: find i where a[i]*a[k_star-i] is large
                valid_i = np.where(a > 0.05 * a.max())[0]
                if len(valid_i) < 2:
                    j = rng.integers(0, n)
                    m = rng.integers(0, n)
                else:
                    i = valid_i[rng.integers(0, len(valid_i))]
                    ki = k_star - i
                    if 0 <= ki < n and a[ki] > 0.01 * a.max():
                        j = i
                        m = ki
                    else:
                        j = rng.integers(0, n)
                        m = rng.integers(0, n)
            elif r < 0.5:
                # Local neighbors
                j = rng.integers(0, n - 1)
                m = (j + rng.integers(1, 6)) % n
            else:
                # Random
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute v[k] = 2*(a[k-j] - a[k-m]) for k in [0, 2n-2]
            v = np.zeros(2 * n - 1)
            v[j:j + n] += 2.0 * a
            v[m:m + n] -= 2.0 * a
            qj = (2 * j < 2 * n - 1)
            qm = (2 * m < 2 * n - 1)

            # 1D optimization: minimize max_k (c[k] + d*v[k] + [k==2j]*d^2 - [k==2m]*d^2)
            lo = -a[j]
            hi = a[m]
            if hi - lo < 1e-12:
                continue

            # Golden section search
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            aG = lo
            bG = hi
            cG = bG - gr * (bG - aG)
            dG = aG + gr * (bG - aG)

            # Evaluate f at a point
            def fval(dlt):
                # Only compute max of the affected part
                # For speed, use the fact that only v changes
                peak = c_max
                # Check the top few lags
                top_idx = np.argsort(c)[-20:]
                for ki in top_idx:
                    val = c[ki] + dlt * v[ki]
                    if qj and ki == 2 * j:
                        val += dlt * dlt
                    if qm and ki == 2 * m:
                        val -= dlt * dlt
                    if val > peak:
                        peak = val
                # Also check if any previously non-top lag becomes the max
                # (approximation: check a few more)
                if dlt > 0:
                    asc_idx = np.where(v > 0)[0][:10]
                    for ki in asc_idx:
                        if ki not in top_idx:
                            val = c[ki] + dlt * v[ki]
                            if qj and ki == 2 * j:
                                val += dlt * dlt
                            if qm and ki == 2 * m:
                                val -= dlt * dlt
                            if val > peak:
                                peak = val
                else:
                    desc_idx = np.where(v < 0)[0][:10]
                    for ki in desc_idx:
                        if ki not in top_idx:
                            val = c[ki] + dlt * v[ki]
                            if qj and ki == 2 * j:
                                val += dlt * dlt
                            if qm and ki == 2 * m:
                                val -= dlt * dlt
                            if val > peak:
                                peak = val
                return peak

            fC = fval(cG)
            fD = fval(dG)

            for _ in range(45):
                if fC < fD:
                    bG = dG
                    dG = cG
                    fD = fC
                    cG = bG - gr * (bG - aG)
                    fC = fval(cG)
                else:
                    aG = cG
                    cG = dG
                    fC = fD
                    dG = aG + gr * (bG - aG)
                    fD = fval(dG)

            d_opt = (aG + bG) / 2.0
            d_opt = max(lo, min(hi, d_opt))
            if abs(d_opt) < 1e-12:
                continue

            # Verify with full computation
            vals = c + d_opt * v
            if qj:
                vals[2 * j] += d_opt * d_opt
            if qm:
                vals[2 * m] -= d_opt * d_opt
            new_max = vals.max()

            if new_max < c_max - 1e-12:
                a[j] += d_opt
                a[m] -= d_opt
                if a[j] < 0:
                    a[j] = 0
                if a[m] < 0:
                    a[m] = 0
                c = vals
                c_max = new_max
                improved = True

        if it % 30 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def spectral_optimize(a_init, n, S, max_time, rng):
    """Spectral domain optimization: flatten power spectrum while maintaining non-negativity."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    alpha = 0.3

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Progressive flattening
        progress = min(1.0, it / 3000.0)
        alpha = 0.2 + 0.7 * progress

        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target

        # Also penalize peaks more aggressively
        P_peak_mask = P > 2.0 * P_mean
        P_new[P_peak_mask] *= (1.0 - 0.3 * alpha)

        mag_new = np.sqrt(P_new)
        mag_new *= np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        # Blend for stability
        blend = min(1.0, 0.1 + 0.01 * it)
        a = blend * a_new + (1.0 - blend) * a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a *= S / s

    return best_a, best


def make_variants(a, n, rng, S):
    """Generate diverse variants for multi-start."""
    variants = [a.copy()]
    tt = np.linspace(0, 1, n, endpoint=False)

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, n))

    # Reversed
    variants.append(a[::-1].copy())

    # Perturbations
    for scale in [0.02, 0.05, 0.1]:
        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        variants.append(project_simplex(a + noise, S))

    # Tapered versions (multiply by a window)
    for taper in [0.3, 0.5, 0.7]:
        window = np.ones(n)
        edge = int(n * taper / 2.0)
        if edge > 0:
            ramp = np.linspace(0, 1, edge)
            window[:edge] = ramp
            window[n - edge:] = ramp[::-1]
        v = a * window
        s = v.sum()
        if s > 1e-10:
            variants.append(v * (S / s))

    # Low-pass
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    for cutoff in [0.2, 0.4]:
        F_f = F * np.exp(-8.0 * (freqs / cutoff) ** 2)
        a_f = np.real(ifft(F_f))[:n]
        a_f = np.maximum(a_f, 0)
        s = a_f.sum()
        if s > 1e-10:
            variants.append(a_f * (S / s))

    return variants


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing 2n*max_conv/sum^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 128 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Start: n=%d, budget=%.0fs, parent=%s" % (n, budget_s, "yes" if parent is not None else "no"))

    global_best = np.inf
    global_a = np.ones(n) * S / n

    def update_best(a_cand, label):
        nonlocal global_best, global_a
        a_n = normalize_sum(a_cand, len(a_cand))
        c_chk = fft_convolve_self(a_n)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_n.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: max_conv=%.8f, obj=%.6f (t=%.0fs)" % (
                label, len(a_n), score, 2.0 * len(a_n) * score / (a_n.sum() ** 2), elapsed))
        return score

    # Build initial candidates
    starts = []
    if parent is not None:
        p = normalize_sum(parent, n)
        starts.append(("parent", p))
        for i, v in enumerate(make_variants(p, n, rng, S)):
            starts.append(("var_%d" % i, v))

    # Additional structured starts
    tt = np.linspace(0, 1, n, endpoint=False)
    starts.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.001)))
    starts.append(("gauss", np.exp(-10.0 * (tt - 0.5) ** 2)))
    starts.append(("cos2", np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * tt), 0.001)))
    starts.append(("tri", np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.001)))
    starts.append(("rand", rng.random(n) + 0.05))

    # Phase 1: FISTA on top candidates (40% of time)
    print("\n=== Phase 1: FISTA ===")
    p1_budget = (deadline - time.monotonic()) * 0.38
    n_starts = len(starts)
    # Weight parent heavily
    weights = np.ones(n_starts)
    weights[0] = 4.0  # parent
    if parent is not None:
        weights[1:5] = 1.5  # first few variants
    weights /= weights.sum()
    times = p1_budget * weights

    for i, (name, arr) in enumerate(starts):
        remaining_check = deadline - time.monotonic()
        if remaining_check < 60:
            break
        t_alloc = min(times[i], remaining_check * 0.4)
        if t_alloc < 3:
            continue
        arr_n = len(arr)
        S_i = np.sqrt(2.0 * arr_n)
        if i == 0:
            # Best get hybrid FISTA
            a_opt, _ = run_fista_hybrid(arr, arr_n, S_i, t_alloc, rng)
        else:
            a_opt, _ = run_fista(arr, arr_n, S_i, t_alloc, rng,
                                 beta_init=50.0, beta_max=5000.0, beta_ramp=2000)
        update_best(a_opt, "fista_" + name)

    print("After Phase 1: best max_conv=%.8f, obj=%.6f" % (
        global_best, 2.0 * n * global_best / (S ** 2)))

    # Phase 2: Coordinate descent on best (35% of time)
    print("\n=== Phase 2: Coordinate Descent ===")
    p2_budget = (deadline - time.monotonic()) * 0.35
    if p2_budget > 10:
        a_cd, _ = coordinate_descent_peak(global_a, n, S, p2_budget * 0.7, rng)
        update_best(a_cd, "cd_main")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_cd2, _ = coordinate_descent_peak(global_a, n, S, remaining2 * 0.5, rng)
            update_best(a_cd2, "cd_second")

    print("After Phase 2: best max_conv=%.8f, obj=%.6f" % (
        global_best, 2.0 * n * global_best / (S ** 2)))

    # Phase 3: Spectral + FISTA polish (20% of time)
    print("\n=== Phase 3: Spectral + Polish ===")
    p3_budget = (deadline - time.monotonic()) * 0.20
    if p3_budget > 10:
        a_sp, _ = spectral_optimize(global_a, n, S, p3_budget * 0.4, rng)
        update_best(a_sp, "spectral")

        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            a_fin, _ = run_fista_hybrid(global_a, n, S, remaining3 * 0.6, rng)
            update_best(a_fin, "polish_fista")

    print("After Phase 3: best max_conv=%.8f, obj=%.6f" % (
        global_best, 2.0 * n * global_best / (S ** 2)))

    # Phase 4: Final coordinate descent (remaining time)
    print("\n=== Phase 4: Final CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_final_cd, _ = coordinate_descent_peak(global_a, n, S, remaining * 0.8, rng)
        update_best(a_final_cd, "final_cd")

    # Phase 5: Try different length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 5: Alt length ===")
        # Try 2x or n
        for alt_n in [n * 2, n // 2]:
            if alt_n < 512 or alt_n > 131072:
                continue
            if remaining < 30:
                break
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt = np.interp(t_new, t_old, global_a)
            else:
                blocks = n // alt_n
                alt = global_a[:blocks * alt_n].reshape(alt_n, blocks).mean(axis=1)
            alt = normalize_sum(alt, alt_n)
            t_alloc = min(remaining * 0.3, 60.0)
            a_alt, _ = run_fista(alt, alt_n, S_alt, t_alloc, rng,
                                 beta_init=50.0, beta_max=3000.0)
            update_best(a_alt, "alt_n_%d" % alt_n)
            remaining = deadline - time.monotonic()

    # Final verification
    print("\n=== Final ===")
    final_a = np.clip(global_a, 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256) * np.sqrt(512.0) / 256

    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    print("Final: n=%d, score=%.6f, time=%.0fs" % (len(final_a), ver, time.monotonic() - t_start))

    if not np.isfinite(ver):
        print("  Fallback to uniform!")
        final_a = np.ones(256) * np.sqrt(512.0) / 256
        final_list = [float(x) for x in final_a]

    return final_list
