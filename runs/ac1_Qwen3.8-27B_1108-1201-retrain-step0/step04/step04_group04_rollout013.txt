```python
def lp_direction(f, M):
    """Find b >= 0 maximizing sum(b) subject to ||f*b||_inf <= M."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n where A @ b = f*b (full conv)
    # A[k, j] = f[k-j] if valid
    
    # Actually, let's use scipy's linprog
    # Minimize -sum(b) subject to A @ b <= M, b >= 0
    
    n_constraints = 2 * n - 1
    # A is n_constraints x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build A sparsely
    # For each k, the valid j range is [max(0, k-n+1), min(k, n-1)]
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((n_constraints, n), dtype=np.float64)
    for k in range(n_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = f[k - j]
    A = A.tocsr()
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(n_constraints, M)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if res.status == 0:
        return res.x, res.fun
    else:
        return None, None
```

```python
import time
```

```python
from scipy.sparse import csr_matrix

# ============================================================
# Core utility functions
# ============================================================

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm

    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ============================================================
# LP-based direction finding (state-of-the-art)
# ============================================================

def build_conv_matrix(n, f):
    """Build sparse matrix A where A @ b = convolve(f, b) (full)."""
    n_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_constraints):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    return A


def lp_direction(f, M, max_time_lp=60.0):
    """Find b >= 0 maximizing sum(b) subject to ||f*b||_inf <= M."""
    n = len(f)
    t0 = time.monotonic()

    if n > 16384:
        return None

    A = build_conv_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'presolve': True, 'time_limit': max_time_lp})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            return b
    except Exception:
        pass

    if time.monotonic() - t0 > max_time_lp:
        return None
    return None


def lp_iteration(f, S, M, t_mix=0.3):
    """One LP-based iteration: find direction, line search, update."""
    n = len(f)
    b = lp_direction(f, M)
    if b is None:
        return f, M, False

    sum_b = b.sum()
    if sum_b < 1e-10:
        return f, M, False

    g = b * (S / sum_b)

    # Line search
    best_M = M
    best_t = 0.0
    best_f = f.copy()

    n_trials = 20
    for i in range(n_trials):
        t = i / (n_trials - 1) if n_trials > 1 else 0.0
        f_try = (1.0 - t) * f + t * g
        f_try = project_simplex(f_try, S)
        c_try = fft_convolve_self(f_try)
        M_try = c_try.max()
        if M_try < best_M:
            best_M = M_try
            best_t = t
            best_f = f_try.copy()

    improved = best_M < M - 1e-12
    return best_f, best_M, improved

# FISTA optimizer
# ============================================================

def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=3000.0,
              beta_ramp_iters=2000, use_hybrid=True):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if use_hybrid and it > 500 and rng.random() < min(0.2, 0.0003 * it):
            grad = exact_subgradient(y, c)
        else:
            grad, _ = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * max(beta, 1.0))
        if it - last_improve > 2000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

# Directed coordinate descent
# ============================================================

def coordinate_descent_directed(a_init, n, S, max_time, rng):
    """Directed coordinate descent: always target the peak lag."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Find the peak lag
        k_star = int(np.argmax(c))

        # Compute gradient of b_{k_star} w.r.t. a
        # grad[j] = 2 * a[k_star - j] if valid
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * a[k_star - j]

        # Find best transfer: decrease a at high-grad positions, increase at low-grad
        # Try multiple candidate pairs
        n_candidates = min(30 + it, 100)
        best_delta_obj = 0.0
        best_pair = None
        best_delta = 0.0

        for _ in range(n_candidates):
            # Pick j (to increase) from low-gradient region
            # Pick m (to decrease) from high-gradient region
            if rng.random() < 0.7:
                # Directed: j from bottom third of grad, m from top third
                sorted_idx = np.argsort(grad)
                j_pool = sorted_idx[:n // 3]
                m_pool = sorted_idx[-n // 3:]
                j = int(rng.choice(j_pool))
                m = int(rng.choice(m_pool))
            else:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))

            if j == m or a[j] < 1e-10:
                continue

            # Compute change in b_k for transferring delta from m to j
            # b_k changes by delta * (grad_j - grad_m) for k = k_star
            # For other lags, approximate
            d_ks = grad[j] - grad[m]

            if d_ks >= 0:
                continue  # This transfer would increase the peak

            # Maximum delta: limited by a[m]
            max_delta = a[m] * 0.9
            # Choose delta to minimize the peak
            # Approximate: new peak ≈ c[k_star] + delta * d_ks
            # We want c[k_star] + delta * d_ks = c[k_star] * (1 - eps)
            # delta = c[k_star] * eps / |d_ks|
            target_reduction = 0.01 * c[k_star]
            delta = min(max_delta, target_reduction / max(abs(d_ks), 1e-15))

            if delta < 1e-15:
                continue

            # Try the transfer
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[m] < 0:
                continue

            c_trial = fft_convolve_self(a_trial)
            obj = c_trial.max()
            if obj < best - 1e-12:
                best_delta_obj = obj - best
                best_pair = (j, m)
                best_delta = delta

        if best_pair is not None:
            j, m = best_pair
            a[j] += best_delta
            a[m] -= best_delta
            if a[m] < 0:
                a[m] = 0.0
                a[j] -= (-a[m])
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * (S / s)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            improved = True

        if not improved and it > 10:
            # Add noise and recompute
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 200.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best

# Frequency-domain simulated annealing
# ============================================================

def freq_sa(a_init, n, S, max_time, rng, T_init=0.1, T_final=0.001, n_freq_perturb=10):
    """Simulated annealing in frequency domain."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    n_total = max(int(max_time / 0.05), 100)  # estimate total iterations
    T_current = T_init

    while time.monotonic() - start < max_time:
        it += 1
        # Cool down
        progress = min(1.0, it / n_total)
        T_current = T_init * (T_final / T_init) ** progress

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)

        # Perturb random frequency bins
        n_perturb = n_freq_perturb
        bins = rng.integers(0, fsize // 2, n_perturb * 2)
        F_trial = F.copy()
        for b_idx in bins[:n_perturb]:
            # Perturb magnitude
            mag_scale = 1.0 + rng.normal(0, 0.3)
            F_trial[b_idx] *= max(mag_scale, 0.1)
            F_trial[fsize - b_idx] = np.conj(F_trial[b_idx])

        a_trial = np.real(ifft(F_trial))[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = np.ones(n) * S / n
        else:
            a_trial = a_trial * (S / s)

        c_trial = fft_convolve_self(a_trial)
        obj_trial = c_trial.max()
        obj_current = c.max()

        delta = obj_trial - obj_current
        if delta < 0 or rng.random() < np.exp(-delta / max(T_current, 1e-10)):
            a = a_trial
            c = c_trial
            if obj_trial < best:
                best = obj_trial
                best_a = a.copy()

    return best_a, best

def spectral_flatten(a_init, n, S, max_time, rng, alpha_start=0.3, alpha_end=0.8):
    """Gerchberg-Saxton style spectral flattening with adaptive alpha."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    n_est = max(int(max_time / 0.02), 100)

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / n_est)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Flatten: P_new = (1-alpha)*P/P_mean + alpha*1
        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target
        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        a = a_new

    return best_a, best

def make_palindromic(a, n):
    """Make a sequence palindromic."""
    half_len = n // 2
    if n % 2 == 0:
        return np.concatenate([a[:half_len], a[:half_len][::-1]])
    else:
        return np.concatenate([a[:half_len], a[half_len:half_len+1], a[:half_len][::-1]])


def generate_diverse_starts(n, S, rng, parent=None):
    """Generate a set of diverse starting sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) == n:
        starts.append(("parent", parent.copy()))
        starts.append(("parent_pal", make_palindromic(parent, n)))
        for scale in [0.05, 0.1, 0.2, 0.3]:
            noise = rng.normal(0, scale, n)
            v = project_simplex(parent + noise, S)
            starts.append(("parent_noise_%.1f" % scale, v))

    starts.append(("uniform", np.ones(n)))
    starts.append(("gaussian", np.exp(-4.0 * (tt - 0.5) ** 2)))
    starts.append(("sinc", np.maximum(np.sinc((tt - 0.5) * 3) ** 2, 0.01)))
    starts.append(("cosine", 0.5 * (1 + np.cos(2 * np.pi * tt))))
    starts.append(("cosine2", 0.5 * (1 + np.cos(4 * np.pi * tt))))
    starts.append(("raised_cos", np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.001) ** 2))
    starts.append(("bimodal", np.maximum(np.exp(-20*(tt-0.25)**2) + np.exp(-20*(tt-0.75)**2), 0.001)))
    starts.append(("random", rng.random(n) + 0.1))
    starts.append(("triangular", 1.0 - np.abs(2*tt - 1)))

    # Normalize all
    normalized = []
    for name, arr in starts:
        arr_n = normalize_sum(arr, n)
        normalized.append((name, arr_n))

    return normalized

def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "available" if parent is not None else "none"))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [***] %s: score=%.6f (elapsed %.0fs)" % (label, score, elapsed))
        return score

    # ================================================================
    # Phase 0: Initialization and evaluation
    # ================================================================
    print("\n=== Phase 0: Initialization ===")
    starts = generate_diverse_starts(n, S, rng, parent)

    best_start = None
    best_start_score = np.inf
    for name, arr in starts:
        c = fft_convolve_self(arr)
        score = c.max()
        if score < best_start_score:
            best_start_score = score
            best_start = arr.copy()
        update_best(arr, "init_" + name)

    print("Best init: score=%.6f" % best_start_score)

    # ================================================================
    # Phase 1: LP-based optimization (key new element)
    # ================================================================
    print("\n=== Phase 1: LP-based Optimization ===")
    if n <= 16384:
        p1_budget = min(700.0, (deadline - time.monotonic()) * 0.35)
        if p1_budget > 30:
            a_lp = best_start.copy()
            c_lp = fft_convolve_self(a_lp)
            M_lp = c_lp.max()
            it = 0
            lp_start = time.monotonic()

            while time.monotonic() - lp_start < p1_budget:
                it += 1
                t0 = time.monotonic()
                a_new, M_new, improved = lp_iteration(a_lp, S, M_lp, t_mix=0.3)
                lp_time = time.monotonic() - t0

                if not improved:
                    # Try with a larger M (relax constraint)
                    M_new = M_lp * 1.05
                    a_new, M_new2, improved2 = lp_iteration(a_lp, S, M_new, t_mix=0.2)
                    if improved2:
                        M_new = M_new2
                    else:
                        # Stuck, add noise
                        noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                        a_lp = project_simplex(a_lp + noise, S)
                        c_lp = fft_convolve_self(a_lp)
                        M_lp = c_lp.max()
                        continue

                a_lp = a_new
                M_lp = M_new
                update_best(a_lp, "LP_iter_%d" % it)

                if it % 20 == 0:
                    print("  LP iter %d: M=%.6f, time/iter=%.1fs" % (it, M_lp, lp_time))

                # If LP is too slow, switch to FISTA
                if lp_time > 60:
                    print("  LP too slow (%.1fs), switching to FISTA" % lp_time)
                    break

            print("LP phase done: it=%d, best=%.6f" % (it, global_best))

    # ================================================================
    # Phase 2: FISTA polish
    # ================================================================
    print("\n=== Phase 2: FISTA Polish ===")
    p2_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    if p2_budget > 20:
        a_fista, score_fista = run_fista(global_a, n, S, p2_budget, rng,
                                          beta_init=50.0, beta_max=5000.0,
                                          beta_ramp_iters=1000)
        update_best(a_fista, "FISTA")

    # ================================================================
    # Phase 3: Directed coordinate descent
    # ================================================================
    print("\n=== Phase 3: Directed Coordinate Descent ===")
    p3_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    if p3_budget > 20:
        a_cd, score_cd = coordinate_descent_directed(global_a, n, S, p3_budget, rng)
        update_best(a_cd, "CoordDesc")

    # ================================================================
    # Phase 4: Frequency-domain SA
    # ================================================================
    print("\n=== Phase 4: Frequency SA ===")
    p4_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    if p4_budget > 20:
        a_sa, score_sa = freq_sa(global_a, n, S, p4_budget, rng,
                                 T_init=0.05, T_final=0.001, n_freq_perturb=15)
        update_best(a_sa, "FreqSA")

    # ================================================================
    # Phase 5: Spectral flattening on best
    # ================================================================
    print("\n=== Phase 5: Spectral Flattening ===")
    p5_budget = min(150.0, (deadline - time.monotonic()) * 0.12)
    if p5_budget > 10:
        a_sf, score_sf = spectral_flatten(global_a, n, S, p5_budget, rng,
                                          alpha_start=0.3, alpha_end=0.7)
        update_best(a_sf, "Spectral")

    # ================================================================
    # Phase 6: Final combined polish
    # ================================================================
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Quick FISTA
        t_f = remaining * 0.4
        a_fin, _ = run_fista(global_a, n, S, t_f, rng,
                             beta_init=200.0, beta_max=8000.0, beta_ramp_iters=500)
        update_best(a_fin, "FinalFISTA")

        # Quick CD
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_cd2, _ = coordinate_descent_directed(global_a, n, S, remaining2 * 0.8, rng)
            update_best(a_cd2, "FinalCD")

    # ================================================================
    # Phase 7: Try alternate length if time permits
    # ================================================================
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Phase 7: Alternate Length ===")
        alt_n = n * 2 if n <= 8192 else n // 2
        if 1024 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                nblocks = n // alt_n
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alt = remaining * 0.5
            a_alt, _ = run_fista(alt_start, alt_n, S_alt, t_alt, rng,
                                 beta_init=10.0, beta_max=2000.0)
            update_best(a_alt, "AltLen_%d" % alt_n)

    # ================================================================
    # Final verification
    # ================================================================
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Final score: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    total_time = time.monotonic() - t_start
    print("Done: length=%d, score=%.6f, time=%.0fs" % (len(global_a), ver, total_time))
    return final_list
```
