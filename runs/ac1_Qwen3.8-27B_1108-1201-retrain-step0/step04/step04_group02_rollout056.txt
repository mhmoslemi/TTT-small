    c = np.real(ifft(A * A))

def coordinate_descent(u, c, S, n, max_time, rng, verbose=True):
    """
    Exact coordinate descent for minimizing max_c subject to sum(u) = S, u >= 0.
    """
    start = time.monotonic()
    best_obj = c.max()
    best_u = u.copy()
    no_improve_count = 0
    
    iteration = 0
    while time.monotonic() - start < max_time:
        iteration += 1
        
        # Pick a coordinate (random with some cycling)
        if iteration % 1000 == 0:
            # Full sweep
            coords = np.arange(n)
        else:
            # Random single coordinate
            j = rng.integers(0, n)
            coords = [j]
        
        improved = False
        for j in coords:
            if time.monotonic() > start + max_time:
                break
            
            u_j = u[j]
            t = S - u_j  # sum of others
            if t < 1e-15:
                continue
            
            # Compute A and B
            # A[m] = c[m] - 2*u[j]*u[m-j]*1[valid, m≠2j] - u[j]^2*1[m=2j]
            # B[m] = u[m-j]*1[valid, m≠2j]
            
            m_2j = 2 * j
            size = 2 * n - 1
            
            if m_2j < size:
                # Valid range for m-j: 0 <= m-j < n => j <= m < j+n
                lo = max(0, j)
                hi = min(size - 1, j + n - 1)
                
                # B[m] = u[m-j] for m in [lo, hi], m ≠ 2j
                B = np.zeros(size)
                valid_indices = np.arange(lo, hi + 1)
                B[valid_indices] = u[valid_indices - j]
                if m_2j >= lo and m_2j <= hi:
                    B[m_2j] = 0  # Exclude the self-term
                
                # A[m] = c[m] - contribution of u[j]
                # Contribution: 2*u[j]*B[m] + u[j]^2*1[m=2j]
                A = c.copy()
                A[valid_indices] -= 2 * u_j * B[valid_indices]
                if m_2j < size:
                    A[m_2j] -= u_j * u_j  # subtract u[j]^2
            else:
                # m_2j out of range (shouldn't happen for j < n)
                continue
            
            # Solve 1D problem
            x_new = solve_1d_convex(A, B, t, S, m_2j, n)
            
            if abs(x_new - u_j) < 1e-12 * S:
                continue
            
            # Update u
            alpha = (S - x_new) / t
            u_old = u.copy()
            u[:] *= alpha
            u[j] = x_new
            
            # Recompute convolution
            c_new = fft_convolve_self(u)
            new_max = c_new.max()
            
            if new_max < best_obj:
                best_obj = new_max
                best_u = u.copy()
                no_improve_count = 0
            else:
                # Accept if it's a valid step (should always improve the 1D sub-problem)
                # But due to numerical issues or the non-convexity of the full problem,
                # it might not improve the global max
                pass
            
            c = c_new
            improved = True
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD iter {iteration}: best_obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_u, best_obj


def compute_objective(u):
    """Compute 2n * max_c / S^2."""
    n = len(u)
    S = u.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(u)
    return 2.0 * n * c.max() / (S * S)


def generate_initializations(n, rng, parent=None):
