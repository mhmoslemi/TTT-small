import numpy as np

from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A

def lse_value_grad(a, N, beta):

def _lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad

def _lbfgs_dir(s_hist, y_hist, g):
    """Two-loop L-BFGS recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i, y_i = s_hist[i], y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    r = q
    # Scale
    if m > 0:
        s_last, y_last = s_hist[-1], y_hist[-1]
        yy = np.dot(y_last, y_last)
        if yy > 1e-30:
            gamma = max(1e-10, np.dot(s_last, y_last) / yy)
        else:
            gamma = 1.0
        r = gamma * q
    for i in range(m):
        s_i, y_i = s_hist[i], y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def fourier_reinit(n, N, rng, S):
    """Generate a random sequence via Fourier domain."""
    # Random power spectrum with some structure
    P = np.ones(N)
    # Add some structure to avoid pure noise
    for freq in rng.choice(N, size=10, replace=False):
        P[freq] = rng.uniform(0.5, 3.0)
    # Also add a low-frequency component
    for freq in range(min(5, N // 2)):
        P[freq] += rng.uniform(0.1, 1.0)
        P[N - freq] = P[freq]
    
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    return project_simplex(np.abs(a) + 0.01, S)

"""
Minimize 2*n*max_conv/sum^2 using alternating direct-peak Adam and LSE-LBFGS.

Strategy:
1. Start from given sequence or diverse initializations.
2. Alternate between:
   - Adam on the gradient of the top-k peaks (direct, precise)
   - L-BFGS on LSE-smoothed objective (smooth, handles non-differentiability)
3. Use structured Fourier-domain perturbations to escape local minima.
4. Progressive refinement with increasing precision.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0 or n == 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = (theta_candidates <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (conv[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def peak_gradient(a, N, peak_indices):
    """
    Gradient of sum of conv(a,a) at specified peak indices.
    Uses FFT for efficiency.
    """
    n = len(a)
    w_full = np.zeros(2 * n - 1)
    for k in peak_indices:
        w_full[int(k)] = 1.0 / len(peak_indices)
    fsize = 1 << (3 * n).bit_length()
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def lse_gradient(a, N, beta):
    """Gradient of LSE-smoothed max of conv(a,a)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i, y_i = s_hist[i], y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    # Initial scaling
    if m > 0:
        s0, y0 = s_hist[-1], y_hist[-1]
        yy0 = np.dot(y0, y0)
        if yy0 > 1e-30:
            gamma = max(1e-12, np.dot(s0, y0) / yy0)
        else:
            gamma = 1.0
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i, y_i = s_hist[i], y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def adam_peak_block(a_init, n, S, N, max_time, rng, top_k=5, lr=0.01):
    """
    Adam optimizer directly on the top-k peak gradient.
    More precise than LSE for final refinement.
    """
    a = project_simplex(a_init, S)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    current_lr = lr
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c, _ = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()
        
        # Get top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        grad = peak_gradient(a, N, top_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        step = current_lr * m_hat / (np.sqrt(v_hat) + eps)
        step_norm = np.linalg.norm(step)
        if step_norm < 1e-20:
            break
        
        # Try the step
        a_new = project_simplex(a - step, S)
        c_new, _ = compute_conv(a_new, N)
        
        if c_new.max() <= cur_max + 1e-10:
            a = a_new
        else:
            # Try smaller step
            a_new2 = project_simplex(a - 0.5 * step, S)
            c_new2, _ = compute_conv(a_new2, N)
            if c_new2.max() <= cur_max + 1e-10:
                a = a_new2
            else:
                current_lr *= 0.8
        
        # Periodic perturbation
        if t % 200 == 0 and rng.random() < 0.1:
            # Fourier-domain perturbation
            _, A = compute_conv(a, N)
            noise_phase = rng.normal(0, 0.01, N)
            A_pert = A * np.exp(1j * noise_phase)
            a_pert = np.real(ifft(A_pert))[:n]
            a_pert = project_simplex(a_pert, S)
            c_pert, _ = compute_conv(a_pert, N)
            if c_pert.max() < cur_max:
                a = a_pert
                m = np.zeros(n)
                v = np.zeros(n)
            elif rng.random() < 0.05:
                a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
                m = np.zeros(n)
                v = np.zeros(n)
        
        if t % 500 == 0:
            current_lr *= 0.95
    
    return best_a, best_cmax


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta=50.0, m_mem=15):
    """L-BFGS on LSE-smoothed objective."""
    a = project_simplex(a_init, S)
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, c_max = lse_gradient(a, N, beta)
    best_cmax = c_max
    step_size = 1.0 / (np.linalg.norm(g) + 1e-10)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Backtracking line search
        alpha = step_size / dir_norm * 0.5
        a_new = a
        for ls in range(20):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            if c_ls.max() <= c_max + 1e-10:
                a_new = candidate
                break
            alpha *= 0.5
        else:
            a_new = candidate
        
        c_new, _ = compute_conv(a_new, N)
        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best_a = a_new.copy()
        
        g_new, _ = lse_gradient(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        if sy > 1e-14 * max(1.0, np.dot(s_k, s_k)):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_mem:
                s_hist = s_hist[1:]
                y_hist = y_hist[1:]
        
        a = a_new
        g = g_new
        c_max = c_new.max()

def alternating_block(a_init, n, S, N, max_time, rng,

def fourier_perturb(a, N, rng, S, scale=0.05):
    """Structured perturbation in Fourier domain."""
    n = len(a)
    _, A = compute_conv(a, N)
    # Perturb the magnitude (not phase) - this preserves chirality
    mag_pert = np.ones(N) * (1.0 + scale * rng.standard_normal(N))
    mag_pert = np.maximum(mag_pert, 0.1)
    # Make it symmetric for real signal
    for k in range(N // 2 + 1):
        mirror = N - k if k < N else 0
        if k < N - k:
            avg = (mag_pert[k] + mag_pert[N - k]) / 2
            mag_pert[k] = avg
            mag_pert[N - k] = avg
    A_new = A * mag_pert
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def make_init_random(n, rng, S):
    a = rng.random(n) + 0.05
    return project_simplex(a, S)


def make_init_chirp(n, rng, S):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * (4 * tt + 0.3 * tt ** 2))
    a += 0.05 * rng.standard_normal(n)
    return project_simplex(np.abs(a) + 0.01, S)


def make_init_spread(n, rng, S):
    """Initialize with spread-out energy."""
    a = np.ones(n) * 0.8
    # Add structured perturbation
    for _ in range(5):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.1 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_simplex(np.abs(a) + 0.01, S)
