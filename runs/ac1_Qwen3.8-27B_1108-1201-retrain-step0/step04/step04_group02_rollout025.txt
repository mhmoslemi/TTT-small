import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum_k exp(beta*c[k])) w.r.t. a.
    grad[j] = 2 * sum_k w[k]*a[k-j] where w = softmax(beta*c)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def make_u_shape(n, depth, rng):
    """Create a U-shaped initialization with random modulation."""
    t = np.linspace(0, 1, n)
    base = 1.0 + depth * (2.0 * t - 1.0) ** 2
    # Add some random harmonic content
    n_harm = rng.integers(1, 5)
    for _ in range(n_harm):
        freq = rng.uniform(1, 8)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        base += amp * np.sin(2 * np.pi * freq * t + phase)
    return np.clip(base, 0.01, None)


def make_initializations(n, rng, given=None):

    # U-shaped with varying depth

    # Cosine-based
    for freq in [1, 2, 3]:
        a = 1.0 + 0.5 * np.cos(np.pi * freq * (2.0 * t - 1.0))
        inits.append(("cos_%d" % freq, np.clip(a, 0.01, None)))

    # Gaussian at edges

    # Random smooth

        inits.append(("rand_smooth_%d" % i, a))

    inits.append(("rand", rng.random(n) + 0.1))

                     beta_start=3.0, beta_end=800.0,
                     anneal_iter=15000):
    """
    LSE-smoothed Nesterov gradient descent with adaptive step size
    and periodic perturbations to escape local minima.
    """
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    step_scale = 1.0

    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / S**2

    while time.monotonic() - start < max_time:
        it += 1

        # Anneal beta
        progress = min(1.0, it / max(1, anneal_iter))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient at Nesterov point y
        c_y = fft_convolve_self(y)

        # Blend gradient strategies
        if it < 100:
            grad = topk_subgradient(y, c_y, 1)
        elif it < 500:
            grad = 0.5 * topk_subgradient(y, c_y, 4) + 0.5 * lse_gradient(y, c_y, beta)
        else:
            grad = lse_gradient(y, c_y, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        # Lipschitz estimate for LSE gradient: ~2 * beta * ||a||_2
        L = 2.0 * beta * np.linalg.norm(a)
        step = 1.0 / max(L, 1e-10) * step_scale
        # Cap step to avoid太大 moves
        step = min(step, 2.0 * S / (n * gnorm + 1e-10))

        # Nesterov step
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Evaluate periodically
        if it % 200 == 0 or it == 1:
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / S**2
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve_it = it
                step_scale = min(step_scale * 1.02, 3.0)
            else:
                step_scale = max(step_scale * 0.95, 0.1)

        # Periodic perturbation when stuck
        if it - last_improve_it > 5000 and rng.random() < 0.05:
            noise_scale = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + noise_scale * rng.normal(0, 1, n), S)
            y = a.copy()
            t_mom = 1.0
            c = fft_convolve_self(a)
            step_scale = 1.0

        # Occasionally reset momentum
        if it % 20000 == 0:
            t_mom = 1.0

    # Final evaluation
    c_a = fft_convolve_self(a)
    cur_obj = 2.0 * n * c_a.max() / S**2
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()

    return best_a, best_obj


def run_hybrid_opt(a_init, n, S, max_time, rng):
    """
    Hybrid optimization: alternates between LSE Nesterov, top-k subgradient,
    and LP-inspired direction finding. More aggressive exploration.
    """
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    y = a.copy()
    t_mom = 1.0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        phase = it % 1000

        if phase < 400:
            # LSE Nesterov phase
            c_y = fft_convolve_self(y)
            beta = min(500.0, 5.0 + 0.05 * it)
            grad = lse_gradient(y, c_y, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            L = 2.0 * beta * np.linalg.norm(a)
            step = (1.0 / max(L, 1e-10)) * step_scale
            step = min(step, 2.0 * S / (n * gnorm + 1e-10))
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
        elif phase < 700:
            # Top-k subgradient with backtracking
            c = fft_convolve_self(a)
            k = min(32, 2 + it // 500)
            grad = topk_subgradient(a, c, k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = step_scale * S / (n * gnorm + 1e-10)
            step = min(step, 0.5 * S / n)
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            obj_cur = 2.0 * n * c.max() / S**2
            if obj_try < obj_cur:
                a = a_try
                step_scale = min(step_scale * 1.05, 3.0)
            else:
                step_scale = max(step_scale * 0.8, 0.1)
        else:
            # LP-inspired direction: reduce the most violated constraint
            c = fft_convolve_self(a)
            k_star = int(np.argmax(c))
            # Gradient at the peak
            grad_peak = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad_peak[i_vals] = 2.0 * a[k_star - i_vals]

            # Add small regularization from mean
            c_mean = c.mean()
            excess = max(0, c.max() - c_mean)
            if excess > 1e-10:
                # Normalize gradient
                gn = np.linalg.norm(grad_peak)
                if gn > 1e-15:
                    step = step_scale * 0.1 * S / (n * gn + 1e-10)
                    a_try = project_simplex(a - step * grad_peak, S)
                    c_try = fft_convolve_self(a_try)
                    obj_try = 2.0 * n * c_try.max() / S**2
                    obj_cur = 2.0 * n * c.max() / S**2
                    if obj_try < obj_cur:
                        a = a_try

        # Evaluate
        if it % 200 == 0:
            c_eval = fft_convolve_self(a)
            cur_obj = 2.0 * n * c_eval.max() / S**2
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

        # Restart if stuck
        if it - last_improve > 8000:
            strategy = rng.integers(0, 3)
            if strategy == 0:
                noise = 0.01 * S / np.sqrt(n) * rng.normal(0, 1, n)
                a = project_simplex(a + noise, S)
            elif strategy == 1:
                # Frequency domain perturbation
                A = fft(a)
                noise = (rng.normal(0, 0.1, len(A)) + 1j * rng.normal(0, 0.1, len(A)))
                A = A * (1.0 + noise)
                a = np.real(ifft(A))
                a = project_simplex(a, S)
            else:
                # Structured: modulate with random smooth function
                t_arr = np.arange(n) / n
                f = rng.uniform(2, 10)
                mod = 1.0 + 0.2 * np.sin(2 * np.pi * f * t_arr + rng.uniform(0, 6.28))
                a = project_simplex(a * mod, S)
            y = a.copy()
            t_mom = 1.0
            step_scale = 1.0
            last_improve = it  # reset timer

    # Final
    c_a = fft_convolve_self(a)
    cur_obj = 2.0 * n * c_a.max() / S**2
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()

    return best_a, best_obj


def local_steepest_descent(a_init, n, S, max_time, rng):
    """
    Coordinate-wise steepest descent: at each step, find the single coordinate
    that gives the best improvement and update it.
    """
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / S**2
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        best_j = -1
        best_val = cur_obj
        best_a_j = a[j] if j < n else 0

        # Sample a subset of coordinates for efficiency
        sample_size = min(n, 500)
        if n > sample_size:
            indices = rng.integers(0, n, sample_size)
        else:
            indices = np.arange(n)

        for j in indices:
            # Try a small range of values for a[j]
            current = a[j]
            # Candidate values: reduce, keep, increase
            deltas = [-0.5 * current, -0.2 * current, -0.05 * current,
                      0.05 * current, 0.2 * current, 0.5 * current]
            for d in deltas:
                a_try = a.copy()
                a_try[j] = max(0, current + d)
                # Renormalize
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try * (S / s)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n * c_try.max() / S**2
                if obj_try < best_val:
                    best_val = obj_try
                    best_j = j
                    best_a_j = current + d

        if best_j >= 0:
            a[best_j] = max(0, a[best_j])
            # Recompute
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / S**2
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Perturb to escape
            if no_improve > 50:
                noise = 0.01 * S / np.sqrt(n) * rng.normal(0, 1, n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
                cur_obj = 2.0 * n * c.max() / S**2
                no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    print("Starting optimization, given_len=%s" % ("None" if given is None else len(given)))
    t0 = time.monotonic()

    # Phase 1: Quick evaluation of initializations
    print("=== Phase 1: Quick evaluation ===")
    candidates = []  # (name, array, obj)

    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        obj_g = compute_objective(given)
        candidates.append(("given", given, obj_g))
        print("  Given: n=%d, obj=%.6f" % (n_g, obj_g))

    # Try several lengths with U-shaped inits
    lengths_to_try = []
    if given is not None:
        lengths_to_try.append(len(given))
    for L in [2048, 4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    for L in lengths_to_try[:4]:
        S_L = np.sqrt(2.0 * L)
        t_arr = np.linspace(0, 1, L)
        # U-shaped
        for depth in [0.5, 1.0, 2.0]:
            a_init = 1.0 + depth * (2.0 * t_arr - 1.0) ** 2
            obj_i = compute_objective(a_init)
            candidates.append(("u_%d_d%s" % (L, depth), a_init, obj_i))
        print("  n=%d: U-shape objs computed" % L)

    # Sort by objective
    candidates.sort(key=lambda x: x[2])
    print("  Top candidate: %s (obj=%.6f)" % (candidates[0][0], candidates[0][2]))

    # Phase 2: Main optimization on top candidates
    print("=== Phase 2: Main optimization ===")
    usable_time = max(60.0, deadline - time.monotonic())
    n_main = min(4, len(candidates))
    per_candidate = usable_time * 0.7 / n_main

    for i in range(n_main):
        if time.monotonic() > deadline - usable_time * 0.25:
            break
        name, arr, obj_est = candidates[i]
        n_c = len(arr)
        S_c = np.sqrt(2.0 * n_c)
        t_alloc = min(per_candidate, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            break

        print("  Optimizing %s (n=%d, init_obj=%.6f, time=%.0fs)" % (
            name, n_c, obj_est, t_alloc))

        # Run hybrid optimization
        a_opt, score = run_hybrid_opt(arr, n_c, S_c, t_alloc, rng)
        a_norm = normalize(a_opt, n_c)
        real_score = compute_objective(a_norm)

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("    -> obj=%.6f (NEW BEST)" % global_best)
        else:
            print("    -> obj=%.6f" % real_score)

    # Phase 3: Intensive polishing on the best
    remaining = deadline - time.monotonic()
    print("=== Phase 3: Polishing (remaining %.0fs) ===" % remaining)

    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Run LSE Nesterov on the best
        seg1 = remaining * 0.5
        a_pol, s_pol = run_lse_nesterov(global_a, n_best, S_best, seg1, rng,
                                         beta_start=10.0, beta_end=1000.0,
                                         anneal_iter=20000)
        a_pol = normalize(a_pol, n_best)
        s_pol = compute_objective(a_pol)
        if s_pol < global_best:
            global_best = s_pol
            global_a = a_pol.copy()
            print("  After LSE Nesterov: %.6f" % global_best)
        else:
            print("  After LSE Nesterov: %.6f (no improvement)" % s_pol)

        # Run hybrid again with different params
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_h2, s_h2 = run_hybrid_opt(global_a, n_best, S_best, remaining2 * 0.6, rng)
            a_h2 = normalize(a_h2, n_best)
            s_h2 = compute_objective(a_h2)
            if s_h2 < global_best:
                global_best = s_h2
                global_a = a_h2.copy()
                print("  After hybrid2: %.6f" % global_best)

        # Micro-optimization: very small steps
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_micro = global_a.copy()
            micro_start = time.monotonic()
            step = 0.05
            for _ in range(100000):
                if time.monotonic() - micro_start > remaining3 * 0.8:
                    break
                c_m = fft_convolve_self(a_micro)
                cur_m = 2.0 * n_best * c_m.max() / S_best**2
                k = 1
                grad_m = topk_subgradient(a_micro, c_m, k)
                gn_m = np.linalg.norm(grad_m)
                if gn_m < 1e-15:
                    break
                eff_step = step / (1.0 + 0.0001)
                a_try = project_simplex(a_micro - eff_step * grad_m, S_best)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n_best * c_try.max() / S_best**2
                if obj_try < cur_m - 1e-14:
                    a_micro = a_try
                    step = min(step * 1.01, 1.0)
                else:
                    step *= 0.9
                    if step < 0.001:
                        noise = 0.005 * S_best / np.sqrt(n_best)
                        a_micro = project_simplex(a_micro + noise * rng.normal(0, 1, n_best), S_best)
                        step = 0.05
            a_micro = normalize(a_micro, n_best)
            s_micro = compute_objective(a_micro)
            if s_micro < global_best:
                global_best = s_micro
                global_a = a_micro.copy()
                print("  After micro: %.6f" % global_best)
            else:
                print("  After micro: %.6f" % s_micro)

    # Phase 4: Try a different length if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60 and given is not None:
        # Try a different length by interpolating
        alt_len = len(given) * 2 if len(given) <= 8192 else len(given) // 2
        if alt_len >= 512 and alt_len <= 32768:
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(given))
            x_new = np.linspace(0, 1, alt_len)
            f_interp = interp1d(x_old, given, kind='cubic', fill_value="extrapolate")
            a_alt = np.clip(f_interp(x_new), 0.01, None)
            S_alt = np.sqrt(2.0 * alt_len)
            a_alt = normalize(a_alt, alt_len)
            obj_alt = compute_objective(a_alt)
            print("  Alt length n=%d, init obj=%.6f" % (alt_len, obj_alt))

            seg = min(remaining * 0.6, 200)
            a_alt_opt, s_alt = run_hybrid_opt(a_alt, alt_len, S_alt, seg, rng)
            a_alt_opt = normalize(a_alt_opt, alt_len)
            s_alt = compute_objective(a_alt_opt)
            if s_alt < global_best:
                global_best = s_alt
                global_a = a_alt_opt.copy()
                print("  Alt length improved: %.6f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
        global_a = normalize(global_a, np.sqrt(512.0))

    # Verify with evaluate_sequence
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver > global_best + 1e-6:
                print("Warning: verification differs from computed!")
        else:
            print("evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64)
            global_a = normalize(global_a, np.sqrt(512.0))
            final_list = [float(x) for x in global_a]
    except Exception as e:
        print("Verification error: %s" % str(e))

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    print("Returning n=%d, obj=%.6f, total_time=%.0fs" % (
        len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
