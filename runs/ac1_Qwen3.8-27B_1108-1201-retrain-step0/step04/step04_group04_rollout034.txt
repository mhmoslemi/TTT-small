"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Uses multi-resolution warm start, L-BFGS-B, FISTA, and coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_max(c, beta):
    """Compute log-sum-exp approximation to max(c)."""
    cm = c.max()
    return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta

def lse_gradient_of_max(c, a, beta):
    """
    Compute gradient of LSE(beta * c) with respect to a,
    where c = convolve(a, a).
    Returns grad such that grad[i] = d(LSE)/da[i].
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    w = e / tot
    # d(LSE)/da_i = sum_k w_k * dc_k/da_i
    # dc_k/da_i = a_{i+k} + a_{i-k} (with bounds), i.e., 2 * (w * a) at appropriate shifts
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * grad

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def run_fista(a_init, n, max_time, beta_init, beta_max, beta_ramp, rng):
    """FISTA with LSE smoothing."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_init + (beta_max - beta_init) * min(1.0, it / beta_ramp)
        if it > beta_ramp:
            beta = beta_max
            
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        if it % 500 == 0:
            grad = lse_gradient_of_max(c, y, beta)
        else:
            grad = lse_gradient_of_max(c, y, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 1.0 / (1.0 * beta + 1e-10)
        if it - last_improve > 2000:
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

def run_lbfgsb(a_init, n, max_time, beta_start, beta_end):
    """L-BFGS-B optimization of LSE-smoothed objective."""
    S = np.sqrt(2.0 * n)
    a0 = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a0.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    def objective_and_grad(x, beta):
        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-15:
            return 1e10, np.zeros_like(x)
        a = a * (S / s)
        c = fft_convolve_self(a)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        val = cm + np.log(tot) / beta
        # Gradient
        w = e / tot
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        # Account for normalization (approximate)
        grad = grad_a * (S / s)
        return val, grad
    
    # Multi-stage: increase beta over time
    remaining = max_time
    stages = [(beta_start, 0.3), (beta_start * 5, 0.25), (beta_start * 25, 0.25), (beta_end, 0.2)]
    for beta, frac in stages:
        t_stage = remaining * frac
        if t_stage < 2:
            continue
        t_start_stage = time.monotonic()
        max_iter = int(10000 * t_stage / max_time * 10)
        # Callback to check time
        def callback(xk):
            nonlocal best, best_a
            a_chk = np.maximum(xk, 0.0)
            s = a_chk.sum()
            if s > 1e-15:
                a_chk = a_chk * (S / s)
                c_chk = fft_convolve_self(a_chk)
                if c_chk.max() < best:
                    best = c_chk.max()
                    best_a = a_chk.copy()
            return time.monotonic() - t_start_stage > t_stage - 1
        
        res = minimize(
            lambda x: objective_and_grad(x, beta)[0],
            a0,
            jac=lambda x: objective_and_grad(x, beta)[1],
            method='L-BFGS-B',
            bounds=[(0, 1000)] * n,
            options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'ftol': 1e-15, 'gtol': 1e-12, 'callback': callback}
        )
        # Update best
        a_res = np.maximum(res.x, 0.0)
        s = a_res.sum()
        if s > 1e-15:
            a_res = a_res * (S / s)
            c_res = fft_convolve_self(a_res)
            if c_res.max() < best:
                best = c_res.max()
                best_a = a_res.copy()
        a0 = best_a.copy()
        remaining = max_time - (time.monotonic() - start)
        if remaining <= 0:
            break
        # Renormalize a0 for next stage
        a0 = normalize_sum(a0, n)
    
    return best_a, best

def coordinate_descent_v2(a_init, n, max_time, rng):
    """Improved coordinate descent with binding constraint awareness."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        # Find binding constraints (top 5 lags)
        top_k = np.argsort(c)[::-1][:5]
        
        # Try moves: random pairs + targeted pairs
        n_try = min(100 + it // 5, 400)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break
            
            if rng.random() < 0.4:
                # Targeted: try to reduce a binding constraint
                k_bad = top_k[rng.integers(0, len(top_k))]
                # Need to reduce c_{k_bad}
                # c_k = sum_i a_i * a_{i+k}
                # Reducing a pair (i, i+k) would help
                i = rng.integers(0, n - 1)
                if i + k_bad >= n:
                    continue
                j = i  # take from j = i
                m = i + k_bad  # or take from somewhere else
                # Actually, let's just do random pairs but weighted
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            
            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue
            
            # Compute the change in c for a transfer of delta from m to j
            # c_k(delta) = c_k + delta * (a_{j+k} + a_{j-k} - a_{m+k} - a_{m-k})
            # For k=0: c_0(delta) = c_0 + 2(a_j - a_m)*delta + 2*delta^2
            d = np.zeros(2 * n - 1, dtype=np.float64)
            for k_idx in range(2 * n - 1):
                vj = 0.0
                jm1 = j + k_idx
                jm2 = j - k_idx
                if 0 <= jm1 < n:
                    vj += a[jm1]
                if 0 <= jm2 < n:
                    vj += a[jm2]
                vm = 0.0
                mm1 = m + k_idx
                mm2 = m - k_idx
                if 0 <= mm1 < n:
                    vm += a[mm1]
                if 0 <= mm2 < n:
                    vm += a[mm2]
                d[k_idx] = vj - vm
            
            lo = -a[j]
            hi = a[m]
            
            # Bisection on the derivative to find min of max
            # Or just ternary search
            lo_b = lo
            hi_b = hi
            for _ in range(30):
                m1 = lo_b + (hi_b - lo_b) / 3.0
                m2 = hi_b - (hi_b - lo_b) / 3.0
                f1 = (c + m1 * d).max() + 2.0 * m1 * m1  # approx for k=0
                f2 = (c + m2 * d).max() + 2.0 * m2 * m2
                if f1 < f2:
                    hi_b = m2
                else:
                    lo_b = m1
            
            delta = (lo_b + hi_b) / 2.0
            delta = np.clip(delta, lo, hi)
            
            if abs(delta) < 1e-12:
                continue
                
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            
            c_trial = c + delta * d
            c_trial[0] = c[0] + 2.0 * (a[j] - a[m]) * delta + 2.0 * delta * delta
            
            cur_trial = c_trial.max()
            if cur_trial < best - 1e-10:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True
        
        if it % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        if not improved and it > 10:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 200.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        elif not improved and it > 50:
            break
    
    return best_a, best

def exact_subgradient_step(a, n, max_time, rng, step_decay=0.999):
    """Refinement using exact subgradients at the argmax."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    step = 0.1 * S / n  # initial step
    it = 0
    stale = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        
        if stale > 200:
            # Add noise and reset
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 1000.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stale = 0
            step = 0.1 * S / n
        
        # Exact subgradient at argmax
        k = int(np.argmax(c))
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k - idx]
        
        # Project step onto simplex tangent (subtract mean)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        a_new = project_simplex(a - step * grad / gnorm, S)
        
        # Check if improvement; if not, decrease step
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best:
            a = a_new
            step = min(step * 1.05, 0.5 * S / n)
        else:
            step *= step_decay
        
        if it % 1000 == 0:
            print("  subgrad it=%d, best=%.6f, step=%.2e" % (it, best, step))
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

def downsample(a, new_n):
    """Downsample by averaging blocks."""
    n = len(a)
    if new_n >= n:
        return a.copy()
    nblocks = n // new_n
    a_trim = a[:nblocks * new_n]
    return a_trim.reshape(new_n, nblocks).mean(axis=1)

def upsample(a, new_n):
    """Upsample by linear interpolation."""
    n = len(a)
    if new_n <= n:
        return downsample(a, new_n)
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)

"""

Combines multi-resolution warm start, L-BFGS-B (scipy), FISTA, coordinate descent,
and exact subgradient refinement to achieve score below 1.5027.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

# Module-level state for L-BFGS-B callbacks
_lbfgs_S = 1.0
_lbfgs_beta = 100.0
_lbfgs_fsize = 16384
_lbfgs_best = np.inf
_lbfgs_best_a = None


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient_of_max(c, a, beta):
    """
    Compute gradient of LSE(beta * c) wrt a, where c = convolve(a, a).
    Returns grad of shape len(a).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def _lbfgs_obj(x):
    """Objective for L-BFGS-B (module-level to avoid closure)."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    a_n = a * (_lbfgs_S / s)
    c = fft_convolve_self(a_n)
    beta = _lbfgs_beta
    cm = c.max()
    val = cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    if c.max() < _lbfgs_best:
        _lbfgs_best = c.max()
        _lbfgs_best_a = a_n.copy()
    return val


def _lbfgs_grad(x):
    """Gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.zeros_like(x)
    a_n = a * (_lbfgs_S / s)
    c = fft_convolve_self(a_n)
    beta = _lbfgs_beta
    cm = c.max()
    e = np.exp(beta * (c - cm))
    w = e / e.sum()
    n = len(a_n)
    fsize = _lbfgs_fsize
    W = fft(w, fsize)
    A = fft(a_n, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad * (_lbfgs_S / s)


def run_fista(a_init, n, max_time, beta_init, beta_max, beta_ramp, rng):
    """FISTA with LSE smoothing and adaptive step."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_accel = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_init + (beta_max - beta_init) * min(1.0, float(it) / beta_ramp)
        if it > beta_ramp:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient_of_max(c, y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (beta + 1e-10)
        if it - last_improve > 1500:
            step *= 0.7

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_accel * t_accel))
        momentum = (t_accel - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_accel = t_new

        # Periodic noise injection
        if it % 4000 == 0:
            decay = np.exp(-it / 40000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_accel = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_lbfgsb(a_init, n, max_time, beta):
    """L-BFGS-B optimization of LSE-smoothed objective."""
    global _lbfgs_S, _lbfgs_beta, _lbfgs_fsize, _lbfgs_best, _lbfgs_best_a
    S = np.sqrt(2.0 * n)
    fsize = 1 << (2 * n - 1).bit_length()
    a0 = normalize_sum(a_init, n)

    _lbfgs_S = S
    _lbfgs_beta = beta
    _lbfgs_fsize = fsize
    _lbfgs_best = np.inf
    _lbfgs_best_a = a0.copy()
    c0 = fft_convolve_self(a0)
    _lbfgs_best = c0.max()

    start = time.monotonic()
    # Estimate iterations: each iter ~ 3-5ms for n=8192
    est_time_per_iter = max(0.003, 0.001 * n / 8192.0)
    max_iter = int((max_time - 5.0) / est_time_per_iter)
    max_iter = min(max_iter, 10000)
    max_iter = max(max_iter, 50)

    try:
        res = minimize(
            _lbfgs_obj,
            a0,
            jac=_lbfgs_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                     'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50}
        )
    except Exception as e:
        print("  L-BFGS-B error: %s" % str(e))
        return a0, _lbfgs_best

    a_res = np.maximum(res.x, 0.0)
    s = a_res.sum()
    if s > 1e-15:
        a_res = a_res * (S / s)
        c_res = fft_convolve_self(a_res)
        if c_res.max() < _lbfgs_best:
            _lbfgs_best = c_res.max()
            _lbfgs_best_a = a_res.copy()

    return _lbfgs_best_a.copy(), _lbfgs_best


def coordinate_descent(a_init, n, max_time, rng):
    """Coordinate descent with binding-constraint awareness."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Find top binding lags
        top_lags = np.argsort(c)[::-1][:8]

        n_try = min(80 + it, 350)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose pair
            if rng.random() < 0.3:
                # Targeted: pick indices near a binding lag
                k_bad = int(top_lags[rng.integers(0, len(top_lags))])
                i = rng.integers(0, max(1, n - 1))
                j = i
                m_candidate = (i + k_bad) % n
                if m_candidate == j:
                    m = (m_candidate + 1) % n
                else:
                    m = m_candidate
            elif rng.random() < 0.5:
                # Local pair
                j = rng.integers(0, n - 3)
                m = (j + rng.integers(1, 6)) % n
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # Compute dc for transfer delta from m to j
            # d[k] = a[j+k] + a[j-k] - a[m+k] - a[m-k]
            d = np.zeros(2 * n - 1, dtype=np.float64)
            for k_idx in range(2 * n - 1):
                vj = 0.0
                idx1 = j + k_idx
                idx2 = j - k_idx
                if 0 <= idx1 < n:
                    vj += a[idx1]
                if 0 <= idx2 < n:
                    vj += a[idx2]
                vm = 0.0
                idx3 = m + k_idx
                idx4 = m - k_idx
                if 0 <= idx3 < n:
                    vm += a[idx3]
                if 0 <= idx4 < n:
                    vm += a[idx4]
                d[k_idx] = vj - vm

            lo = -a[j]
            hi = a[m]

            # Ternary search for optimal delta
            lo_b = lo
            hi_b = hi
            for _ in range(25):
                third = (hi_b - lo_b) / 3.0
                m1 = lo_b + third
                m2 = hi_b - third
                # f(delta) = max(c + delta*d) + quadratic correction at k=0
                f1 = (c + m1 * d).max() + 2.0 * m1 * m1
                f2 = (c + m2 * d).max() + 2.0 * m2 * m2
                if f1 <= f2:
                    hi_b = m2
                else:
                    lo_b = m1

            delta = (lo_b + hi_b) / 2.0
            delta = np.clip(delta, lo, hi)
            if abs(delta) < 1e-13:
                continue

            c_trial = c + delta * d
            c_trial[0] = c[0] + 2.0 * (a[j] - a[m]) * delta + 2.0 * delta * delta
            cur_trial = c_trial.max()

            if cur_trial < best - 1e-11:
                a[j] += delta
                a[m] -= delta
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Periodic recompute
        if it % 30 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 15:
                noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-no_improve / 30.0)
                if noise_scale < 1e-9:
                    break
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best


def exact_subgradient_refine(a_init, n, max_time, rng):
    """Refinement using exact subgradients at the argmax lag."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    step = 0.05 * S / np.sqrt(n)
    it = 0
    stale = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > 100:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stale / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stale = 0
            step = 0.05 * S / np.sqrt(n)
            continue

        # Exact subgradient at argmax
        k = int(np.argmax(c))
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k - idx]

        # Center the gradient (tangent to simplex)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = project_simplex(a - step * grad / gnorm, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best:
            a = a_new
            step = min(step * 1.03, 0.2 * S / np.sqrt(n))
        else:
            step *= 0.95
            if step < 1e-10 * S / n:
                step = 0.05 * S / np.sqrt(n)

        if it % 500 == 0:
            print("    subgrad it=%d, best=%.7f, step=%.2e" % (it, best, step))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def downsample(a, new_n):
    """Downsample by averaging blocks."""
    n = len(a)
    if new_n >= n:
        return a.astype(np.float64).copy()
    nblocks = n // new_n
    a_trim = a[:nblocks * new_n]
    return a_trim.reshape(new_n, nblocks).mean(axis=1)


def upsample(a, new_n):
    """Upsample by linear interpolation."""
    n = len(a)
    if new_n <= n:
        return downsample(a, new_n)
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)

            t_stage = remaining_p2 * frac
            if t_stage < 5:
                continue
            print("  L-BFGS-B beta=%d, time=%.0fs" % (beta, t_stage))
            a_lb, score = run_lbfgsb(global_a, n_target, t_stage, beta)
            _update(a_lb, "lbfgsb_b%d" % beta)

    print("  Phase 2 done: best=%.7f" % global_best)

    # === Phase 3: FISTA with high beta ===
    print("\n=== Phase 3: FISTA high-beta ===")
    if global_a is not None:
        remaining_p3 = (deadline - time.monotonic()) * 0.22
        t_f3 = remaining_p3 * 0.6
        a_f3, _ = run_fista(global_a, n_target, t_f3, 500, 8000, 500, rng)
        _update(a_f3, "fista_high_%d" % n_target)

        # Perturbed restart
        remaining_p3b = deadline - time.monotonic()
        if remaining_p3b > 10:
            perturb_scale = 0.02 * S / np.sqrt(n_target)
            a_pert = project_simplex(global_a + rng.normal(0, perturb_scale, n_target), S)
            t_f3b = remaining_p3b * 0.4
            a_f3b, _ = run_fista(a_pert, n_target, t_f3b, 200, 5000, 300, rng)
            _update(a_f3b, "fista_perturb")

    print("  Phase 3 done: best=%.7f" % global_best)

    # === Phase 4: Coordinate descent ===
    print("\n=== Phase 4: Coordinate descent ===")
    if global_a is not None:
        remaining_p4 = (deadline - time.monotonic()) * 0.20
        if remaining_p4 > 10:
            a_cd, _ = coordinate_descent(global_a, n_target, remaining_p4, rng)
            _update(a_cd, "coord_desc")

    print("  Phase 4 done: best=%.7f" % global_best)

    # === Phase 5: Exact subgradient refinement ===
    print("\n=== Phase 5: Exact subgradient ===")
    if global_a is not None:
        remaining_p5 = (deadline - time.monotonic()) * 0.15
        if remaining_p5 > 8:
            a_es, _ = exact_subgradient_refine(global_a, n_target, remaining_p5, rng)
            _update(a_es, "subgrad")

    print("  Phase 5 done: best=%.7f" % global_best)

    # === Phase 6: Final polish loop ===
    print("\n=== Phase 6: Final polish ===")
    if global_a is not None:
        for rnd in range(4):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            # Short FISTA
            t_f = remaining * 0.35
            a_pf, _ = run_fista(global_a, n_target, t_f, 1000, 12000, 300, rng)
            _update(a_pf, "polish_f%d" % rnd)
            # Short subgradient
            remaining2 = deadline - time.monotonic()
            if remaining2 < 5:
                break
            t_s = remaining2 * 0.4
            a_ps, _ = exact_subgradient_refine(global_a, n_target, t_s, rng)
            _update(a_ps, "polish_s%d" % rnd)
            # Short coord descent
            remaining3 = deadline - time.monotonic()
            if remaining3 < 5:
                break
            t_c = remaining3 * 0.4
            a_pc, _ = coordinate_descent(global_a, n_target, t_c, rng)
            _update(a_pc, "polish_c%d" % rnd)

    print("  Phase 6 done: best=%.7f" % global_best)

    # === Final verification ===
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.7f" % ver)
    else:
        print("Verification failed (inf), using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Returning length=%d, score=%.7f, time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
