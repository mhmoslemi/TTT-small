import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(y, w_vec, fsize):
    """Compute gradient of sum_k w[k]*c[k] wrt a using FFT cross-correlation."""
    W = fft(w_vec, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))
    return grad[:len(y)]


def nesterov_optimize(a0, n, deadline, rng, 
                       top_k_start=1, top_k_end=500,
                       T_anneal=5000,
                       step_scale=1.0,
                       perturb_interval=2000):
    """
    Nesterov-accelerated subgradient method.
    """
    S = np.sqrt(2.0 * float(n))
    a = a0 * (S / a0.sum())
    a = np.maximum(a, 0)
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t = 1.0
    it = 0
    best_a = a.copy()
    best_cmax = np.inf
    last_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Adaptive top-k: start small, grow over time
        progress = min(1.0, it / T_anneal)
        top_k = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        # Compute convolution
        c = fft_convolve_self(y)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = y.copy()
            last_improve = it
            if it % 2000 == 0:
                print("    iter %d: cmax=%.6f" % (it, c_max))
        
        # Find top-k lags
        if top_k == 1:
            peak = int(np.argmax(c))
            w_vec = np.zeros(2*n-1)
            w_vec[peak] = 1.0
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            w_vec = np.zeros(2*n-1)
            w_vec[top_idx] = 1.0 / top_k
        
        # Compute gradient
        grad = compute_grad(y, w_vec, fsize)
        
        # Step size: adaptive
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = step_scale * S / (gnorm * np.sqrt(1.0 + 0.1 * it))
        # Cap step to avoid overshooting
        step = min(step, 2.0 * S / n)
        
        # Projected update
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-15:
            y = y * (S / s)
        a = x_new
        t = t_new
        
        # Periodic perturbation to escape local minima
        if it % perturb_interval == 0:
            if rng.random() < 0.3:
                sigma = 0.005 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, sigma, n), S)
                t = 1.0
            # Reset if stalled
            if it - last_improve > perturb_interval * 3:
                sigma = 0.02 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, sigma, n), S)
                a = y.copy()
                t = 1.0
    
    return best_a, best_cmax

Actually, `sum_i a[i]*a[k-i] <= t` with `a[i] >= 0` can be written as a sum of products, which is not directly a SOCP constraint. We'd need to introduce auxiliary variables `x[i] = sqrt(a[i]*a[k-i])` and constraints `x[i]^2 <= a[i]*a[k-i]`, which are also not SOCP.

So SOCP doesn't directly apply. Let me stick with the SCA (linearized LP) approach.

Actually, there's another approach: **semidefinite programming (SDP)**. If we let `X = a * a^T` (outer product), then `c[k] = sum_i a[i]*a[k-i] = trace(X * P_k)` for some permutation matrix `P_k`. The constraint `X = a*a^T` can be replaced by `X >= 0` and `rank(X) = 1`. Dropping the rank constraint gives an SDP relaxation.

But this SDP would have an `(n x n)` matrix variable, which is infeasible for `n = 8192`.

OK, let me stick with the SCA + LP approach.

Let me now write the complete code. I'll aim for a clean implementation that combines:
1. SCA with LP (main optimization)
2. Nesterov subgradient (secondary optimization)
3. Coordinate descent (polish)
4. Multi-start (initialization)

Here's the code:

"""
Minimize max(conv(a,a)) subject to a>=0, sum(a)=sqrt(2n) using:
1. Successive Linear Approximation (SLA) with LP solves
2. Nesterov-accelerated subgradient descent
3. Coordinate descent polish
Combined with multi-start initialization and periodic perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def sla_lp_solve(a0, n, S, active_lags, max_time=5.0):
    """
    Solve the linearized LP for one SCA iteration.
    a0: current point (length n)
    active_lags: list of lag indices to constrain
    Returns: new a (length n) or None if failed
    """
    c0 = fft_convolve_self(a0)
    n_vars = n + 1  # a[0..n-1] and t
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    # Inequality constraints: for each active lag k
    # sum_j 2*a0[k-j] * a[j] - t <= c0[k]
    A_ub = np.zeros((len(active_lags), n_vars))
    b_ub = np.zeros(len(active_lags))
    
    for row, k in enumerate(active_lags):
        # Coefficients of a[j]: 2*a0[k-j] for valid j
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[row, j] = 2.0 * a0[kj]
        A_ub[row, n] = -1.0  # coefficient of t
        b_ub[row] = c0[k]
    
    # Equality constraint: sum(a) = S
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: a[j] >= 0, t free (but t >= 0 in practice)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = res.x[:n]
            a_new = np.maximum(a_new, 0)  # numerical safety
            return a_new
    except Exception:
        pass
    return None


def sla_optimize(a0, n, deadline, rng, 
                 n_active_start=5, n_active_end=200,
                 max_iter=500):
    """
    Successive Linear Approximation optimization.
    """
    S = np.sqrt(2.0 * float(n))
    a = a0 * (S / a0.sum())
    a = np.maximum(a, 0)
    
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        progress = min(1.0, it / (max_iter * 0.5))
        n_active = max(1, int(n_active_start + (n_active_end - n_active_start) * progress))
        
        # Compute current convolution
        c = fft_convolve_self(a)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            if it % 20 == 0:
                print("    SLA iter %d: cmax=%.6f" % (it, c_max))
        
        # Select active lags: top n_active
        if n_active == 1:
            active = [int(np.argmax(c))]
        else:
            active = list(np.argsort(c)[::-1][:n_active])
        
        # Solve LP
        a_new = sla_lp_solve(a, n, S, active)
        
        if a_new is not None:
            # Check if the new point is better
            c_new = fft_convolve_self(a_new)
            c_new_max = c_new.max()
            
            if c_new_max < c_max:
                a = a_new
            else:
                # LP didn't help; take a small step toward LP solution
                alpha = 0.5
                a = (1 - alpha) * a + alpha * a_new
                a = project_simplex(a, S)
                
                # Check again
                c_check = fft_convolve_self(a)
                if c_check.max() > c_max:
                    a = best_a.copy()  # revert
                    break  # try different active set or perturb
        else:
            # LP failed, perturb
            sigma = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, sigma, n), S)
        
        # Periodic perturbation
        if it % 50 == 0 and rng.random() < 0.1:
            sigma = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, sigma, n), S)
    
    return best_a, best_cmax


def nesterov_subgrad(a0, n, deadline, rng,
                      top_k_start=1, top_k_end=300,
                      total_anneal=10000):
    """Nesterov-accelerated subgradient method."""
    S = np.sqrt(2.0 * float(n))
    a = a0 * (S / a0.sum())
    a = np.maximum(a, 0)
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t_momentum = 1.0
    it = 0
    best_a = a.copy()
    best_cmax = np.inf
    last_improve = 0
    step_mult = 1.0
    
    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / total_anneal)
        top_k = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        c = fft_convolve_self(y)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = y.copy()
            last_improve = it
        
        # Compute subgradient
        if top_k == 1:
            peak = int(np.argmax(c))
            w_vec = np.zeros(2*n-1)
            w_vec[peak] = 1.0
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            w_vec = np.zeros(2*n-1)
            w_vec[top_idx] = 1.0 / top_k
        
        W = fft(w_vec, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = step_mult * S / (gnorm * np.sqrt(1.0 + 0.05 * it))
        step = min(step, 5.0 * S / n)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum**2))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-15:
            y = y * (S / s)
        a = x_new
        t_momentum = t_new
        
        # Perturbation
        if it % 3000 == 0:
            if it - last_improve > 3000:
                sigma = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, sigma, n), S)
                a = y.copy()
                t_momentum = 1.0
        
        if it % 5000 == 0:
            print("    NS iter %d: cmax=%.6f" % (it, best_cmax))
    
    return best_a, best_cmax
