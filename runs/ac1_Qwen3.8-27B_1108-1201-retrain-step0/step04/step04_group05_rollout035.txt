"""
Minimize 2*n*max_conv/sum^2 using continuation L-BFGS on LSE-smoothed objective.

Improvements over previous:
1. More aggressive continuation (beta ramping) with convergence-based switching
2. Better line search (strong Wolfe with cubic interpolation)
3. Adaptive top-k subgradient with BB step size for polishing
4. Smarter restarts using structured perturbations
5. Periodic exact-max correction to avoid LSE approximation bias

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    if rho == 0:
        return np.zeros(n)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_values(c, beta):
    """Compute LSE value and weights."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return c_max, np.zeros(len(c))
    w = exp_vals / total
    lse_val = c_max + np.log(total) / beta
    return lse_val, w


def compute_grad_from_weights(weights, A, n, N):
    """Compute gradient given weights on convolution positions."""
    # weights has length 2n-1, represents weights on conv positions
    w_full = np.zeros(N)
    w_full[:2*n-1] = weights
    W = fft(w_full)
    grad_freq = 2.0 * np.conj(A) * W
    grad = np.real(ifft(grad_freq))[:n]
    return grad


def lbfgs_two_loop(s_hist, y_hist, g, m_max=15):
    """L-BFGS two-loop recursion."""
    m = min(len(s_hist), m_max)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            return np.zeros(n)
        return -g / gn
    
    s_hist = s_hist[-m:]
    y_hist = y_hist[-m:]
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    # Initial Hessian approximation
    r = q
    if m > 0:
        s_last = s_hist[-1]
        y_last = y_hist[-1]
        sy = np.dot(s_last, y_last)
        yy = np.dot(y_last, y_last)
        if sy > 1e-30 and yy > 1e-30:
            gamma = sy / yy
        else:
            gamma = 1.0 / (np.dot(q, q) + 1e-30) * np.dot(q, q)
            gamma = max(gamma, 1e-8)
        r = gamma * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    # Ensure descent direction
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

            best_cmax = cmax_new
            best_a = a_new.copy()
            last_improve_it = it

        g = g_new

def _lse_grad(a, A, c, beta, n, N):
    """Compute LSE gradient."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, np.zeros(len(c))
    w = exp_vals / total
    grad = compute_grad_from_weights(w, A, n, N)
    f_val = c_max + np.log(total) / beta
    return grad, f_val, w


def toppk_subgrad_block(a_init, n, S, N, max_time, rng, top_k_init=5, top_k_max=200):
    """Top-k subgradient with Nesterov momentum and BB step size."""
    a = np.clip(a_init, 0, 1000)
    s_cur = a.sum()
    if s_cur < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s_cur)
    
    y = a.copy()
    t_param = 1.0
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    prev_step = None
    prev_grad = None
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        
        c, A = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        # Compute gradient from top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(2*n-1)
        weights[top_idx] = 1.0
        weights /= top_k
        
        grad = compute_grad_from_weights(weights, A, n, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # BB step size
        if prev_step is not None and prev_grad is not None:
            s_bb = y - prev_step
            ys = np.dot(s_bb, grad - prev_grad)
            ss_bb = np.dot(s_bb, s_bb)
            if ss_bb > 1e-30 and ys > 1e-30:
                step_bb = ss_bb / ys
                step = np.clip(step_bb, 1e-10, 10.0 * S / (n * gnorm))
            else:
                step = 2.0 * S / (n * gnorm * (1.0 + 0.01 * it))
        else:
            step = 2.0 * S / (n * gnorm)
        
        # Nesterov update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param ** 2))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = project_simplex(x_new, S)
        else:
            y = y * (S / s_y)  # re-normalize
        
        prev_step = x_new
        prev_grad = grad
        x_new = x_new
        t_param = t_new

def spectral_equalize(a, N, n, alpha, top_k=10):
    """Spectral equalization: flatten the power spectrum."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(A * A))[:2*n-1]
    
    # Find top peaks and create a "mask" to reduce them
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    weights = top_vals / total
    P_new = P.copy()
    # Reduce power at frequencies that contribute most to the peaks
    m_arr = np.arange(N)
    for i, k in enumerate(top_idx):
        cos_eff = np.cos(2.0 * np.pi * m_arr * int(k) / N)
        P_new = P_new * (1.0 - alpha * weights[i] * cos_eff)

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return a_new

Minimize 2*n*max_conv/sum^2 using continuation L-BFGS on LSE-smoothed objective

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64).copy()
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    if rho <= 0:
        return np.zeros(n)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, A, c, beta, n, N):
    """Compute LSE-smoothed gradient. Returns (grad, lse_val, w)."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, np.zeros(len(c))
    w = exp_vals / total
    f_val = c_max + np.log(total) / beta
    w_full = np.zeros(N)
    w_full[:2 * n - 1] = w
    W = fft(w_full)
    grad_freq = 2.0 * np.conj(A) * W
    grad = np.real(ifft(grad_freq))[:n]
    return grad, f_val, w


def grad_from_topk(c, A, n, N, top_k):
    """Compute gradient using uniform weights on top-k peaks."""
    if top_k >= len(c):

    w_full[:2 * n - 1] = w
    W = fft(w_full)

def lbfgs_two_loop(s_hist, y_hist, g, m_max):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = min(len(s_hist), m_max)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-30:
        if gn < 1e-30:
            return np.zeros(n)
        return -g / gn
    s_hist = s_hist[-m:]
    y_hist = y_hist[-m:]
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    gamma = 1.0
    if m > 0:
        s_last = s_hist[-1]
        y_last = y_hist[-1]
        sy = np.dot(s_last, y_last)
        yy = np.dot(y_last, y_last)
        if sy > 1e-30 and yy > 1e-30:
            gamma = sy / yy
        else:
            gamma = max(1e-8, np.dot(q, q) / (np.dot(g, g) + 1e-30))
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    P_new = P.copy()
    for i in range(top_k):
        k = int(top_idx[i])
        cos_eff = np.cos(2.0 * np.pi * m_arr * k / N)
        P_new = P_new * (1.0 - alpha * weights[i] * cos_eff)
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return a_new


def make_inits(n, rng, N):
    """Generate diverse initializations."""
    inits = []
    inits.append(("rand", rng.random(n) + 0.01))
    tt = np.linspace(0, 1, n, endpoint=False)
    chirp = np.sin(np.pi * n * tt * tt + rng.random() * np.pi)
    inits.append(("chirp", np.abs(chirp) + 0.01))
    a_mf = np.ones(n)
    for freq in [3, 7, 13, 23]:
        a_mf += 0.25 * np.sin(2.0 * np.pi * freq * tt + rng.random() * 2.0 * np.pi)
    inits.append(("multifreq", np.maximum(a_mf, 0.01)))
    b = (rng.random(n) > 0.5).astype(np.float64)
    from scipy.ndimage import uniform_filter1d
    b_sm = uniform_filter1d(b, size=7)
    inits.append(("binary_sm", b_sm + 0.01))
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    phase = rng.uniform(0, 2.0 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_f = np.real(ifft(A))[:n]
    inits.append(("fourier", np.abs(a_f) + 0.01))
    return inits


def try_update(a_candidate, n, S, N, global_best, global_a, label):
    """Try to update the global best. Returns (new_best, new_a)."""
    a_c = np.clip(a_candidate, 0, 1000)
    sc = a_c.sum()
    if sc < 0.01:
        return global_best, global_a
    a_c = a_c * (S / sc)
    c, _ = compute_conv(a_c, N)
    sc_max = c.max()
    if sc_max < global_best:
        if label:
            print("  [%s] max_conv=%.10f ratio=%.8f" % (label, sc_max, 2.0 * n * sc_max / (S * S)))
        return sc_max, a_c.copy()
    return global_best, global_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses multi-phase optimization: multi-start L-BFGS, continuation refinement,
    top-k subgradient polishing, and spectral equalization.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine sequence length
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 2048 <= given.size <= 16384:
            n = given.size
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.15)

    inits = make_inits(n, rng, N)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.insert(0, ("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        a_norm = normalize_to(arr, n, S)
        rem = max(2.0, per_init - 0.5)
        a_l, sc_l = lbfgs_block(a_norm, n, S, N, rem, rng, 3.0, 40.0, 8)
        global_best, global_a = try_update(a_l, n, S, N, global_best, global_a, name)

    if global_a is None:
        global_a = np.ones(n) * S / n
        c, _ = compute_conv(global_a, N)
        global_best = c.max()

    print("Phase 1 done. Best ratio: %.8f" % (2.0 * n * global_best / (S * S)))

    # --- Phase 2: Continuation L-BFGS (main optimization) ---
    print("=== Phase 2: Continuation L-BFGS ===")
    t2 = time.monotonic()
    phase2_budget = min(1300.0, (deadline - t2) * 0.55)

    a_curr = global_a.copy()
    stall = 0
    beta_cursor = 5.0
    p2_start_best = global_best

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(150.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        a_new, sc_new = lbfgs_block(a_curr, n, S, N, block_time, rng,
                                     beta_cursor, min(1000.0, beta_cursor * 25.0), 12)
        if sc_new < global_best - 1e-11:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            beta_cursor = min(1000.0, beta_cursor * 1.6)
            print("  [P2] ratio=%.8f (beta=%.0f)" % (2.0 * n * global_best / (S * S), beta_cursor))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                # Structured perturbation: modify a random subset
                subset_size = max(100, n // 8)
                idx = rng.choice(n, subset_size, replace=False)
                noise = rng.normal(0, 0.008 * S / np.sqrt(n), subset_size)
                a_curr[idx] += noise
                a_curr = project_simplex(a_curr, S)
                beta_cursor = max(3.0, beta_cursor * 0.4)
                stall = 0

    # --- Phase 3: Top-k subgradient polishing ---
    print("=== Phase 3: Top-k subgradient polishing ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.2)
    if phase3_budget > 15:
        a_ng, sc_ng = toppk_subgrad_block(global_a, n, S, N, phase3_budget, rng, 3, 80)
        global_best, global_a = try_update(a_ng, n, S, N, global_best, global_a, "topk")

        # Second subgradient pass with different params
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_ng2, sc_ng2 = toppk_subgrad_block(global_a, n, S, N, min(remaining * 0.4, 200.0),
                                                 rng, 5, 150)
            global_best, global_a = try_update(a_ng2, n, S, N, global_best, global_a, "topk2")

    # --- Phase 4: Spectral equalization + final L-BFGS ---
    print("=== Phase 4: Spectral + final polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        a_sp = global_a.copy()
        S_sp = a_sp.sum()
        n_steps = min(40, int(remaining / 8))
        for i in range(n_steps):
            a_sp = spectral_equalize(a_sp, N, n, alpha=0.08, top_k=6)
            a_sp = project_simplex(a_sp, S_sp)
            if time.monotonic() > deadline - 15:
                break
        global_best, global_a = try_update(a_sp, n, S, N, global_best, global_a, "spectral")

        # Final L-BFGS at high beta
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fin, sc_fin = lbfgs_block(global_a, n, S, N, remaining - 5, rng,
                                         50.0, 800.0, 10)
            global_best, global_a = try_update(a_fin, n, S, N, global_best, global_a, "final")

    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
        else:
            print("Verification FAILED, using fallback")
            global_a = np.ones(1024, dtype=np.float64)
    else:
        global_a = np.ones(1024, dtype=np.float64)

    print("Returning length %d, internal best ratio=%.8f" % (
        len(global_a), 2.0 * n * global_best / (S * S)))
    return [float(x) for x in global_a]
