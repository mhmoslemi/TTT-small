def spectral_flatness_optimization(a_init, n, S, N, max_time, rng):
    """Iteratively flatten the convolution by operating on the power spectrum."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    
    C_target = S**2 / (2*n - 1)
    eta = 1.0  # step size
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        A = fft(a, N)
        P = np.abs(A)**2
        phase = np.angle(A)
        
        # Compute current convolution
        c = np.real(ifft(P))[:2*n-1]
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Compute gradient of sum((c-C_target)^2) w.r.t. P
        e = c - C_target
        e_padded = np.zeros(N)
        e_padded[:2*n-1] = e
        grad_P = np.real(fft(e_padded))  # This gives the gradient (up to a constant)
        
        # Update P
        P_new = P - eta * grad_P / N
        P_new = np.maximum(P_new, 0)
        
        # Renormalize P to maintain Parseval: sum(P) = N * sum(a^2)
        # Actually, we'll just renormalize a later, so let's not worry about this.
        
        # Reconstruct A
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
        
        # Adaptive step size
        if it % 100 == 0:
            eta *= 0.95  # Slowly decrease step size
        
        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.3:
            phase_pert = rng.normal(0, 0.1, N)
            A_pert = np.sqrt(P_new) * np.exp(1j * (phase + phase_pert))
            a_pert = np.real(ifft(A_pert))[:n]
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a_pert = a_pert * (S / s_pert)
                c_pert = np.real(ifft(np.abs(fft(a_pert, N))**2))[:2*n-1]

                    best_cmax = c_pert.max()
                    best_a = a.copy()
    
    return best_a, best_cmax

One more thought: the "deflation" initialization. Here's how it works:

1. Start with a = ones(n) * S/n.
2. Compute conv(a,a). The peak is at the center (lag n-1) with value S²/n (since all elements are equal).
3. To reduce the peak, we need to make the sequence less "uniform." 
4. Specifically, we want to reduce the overlap at the center. This means making a[i] * a[n-1-i] smaller for the center pairing.
5. One way: make a "anti-symmetric" perturbation: a[i] = S/n + δ * v[i] where v[i] = -v[n-1-i] (anti-symmetric about the center).
6. This doesn't change the sum, and it reduces the center overlap.
7. But it might increase other peaks. So we need to iterate.

Actually, a simpler deflation:
1. Start with a = ones(n).
2. Find the peak lag k*.
3. For the positions that contribute to the peak (i and k*-i), reduce them and add to "safe" positions.
4. A "safe" position is one where the contribution to all peaks is small.
5. Repeat.

def deflation_init(n, S, N, n_iters=1000, rng=None):
    a = np.ones(n) * S / n
    if rng is None:
        rng = np.random.default_rng(0)
    
    for _ in range(n_iters):
        c = np.real(ifft(np.abs(fft(a, N))**2))[:2*n-1]
        k_star = int(np.argmax(c))
        
        # Find positions contributing to the peak
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        # Positions: i and k_star - i for valid i
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        
        # Compute contribution of each position to the peak
        # d(conv[k_star])/d(a[i]) = a[k_star - i] (if valid)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        
        # Move mass from high-contribution to low-contribution positions
        # Find the position with highest contribution
        i_max = int(np.argmax(contrib + 1e-10))  # +1e-10 to avoid issues
        # Find a "safe" position (low contribution to all peaks)
        # Approximate: position with lowest contribution to current peak
        i_min = int(np.argmin(contrib))
        
        if i_max == i_min:
            break
        
        # Move a small amount from i_max to i_min
        transfer = 0.01 * a[i_max]
        if transfer > a[i_max] * 0.5:
            transfer = a[i_max] * 0.5
        a[i_max] -= transfer
        a[i_min] += transfer
    
    # Normalize
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    return a

**Better initialization: "Fourier deflation"**

1. Choose a target "flat" power spectrum: P_target[m] = S²/N for all m (constant).
2. This corresponds to a delta in the time domain (a[0] = S, rest = 0), which is not non-negative in the right way.
3. Instead, choose a "spread" power spectrum: P_target[m] = S² * w[m] where w is a window (e.g., raised cosine).
4. Initialize the phases randomly.
5. Take IFFT to get a time-domain signal.
6. Project to non-negative.
7. Iterate: update the spectrum to be closer to the target, project the time domain.

This is essentially the GS algorithm with a non-trivial target. The target should be chosen to give a flat convolution.

For a flat convolution, we need IFFT(P) to be flat, which means P should be a delta. But a delta P gives a complex exponential in time domain. The closest non-negative real sequence to a complex exponential is... a rectified complex exponential, which is a "half-wave rectified" cosine or similar.

So the initialization should be: a[i] = max(S/n + A * cos(2π * m0 * i / N), 0) for some frequency m0.

This is a "single frequency" non-negative sequence. Its convolution will have a specific structure that might be close to flat.

Actually, one more thought. Let me reconsider the problem from a theoretical perspective.

We want to minimize: 2n * max_k conv(a,a)[k] / (sum a)²

Let's define the "peak-to-mean ratio" (PMR):
PMR = max_k conv(a,a)[k] / mean_k conv(a,a)[k]

where mean_k = (sum a)² / (2n-1) (since sum of all conv values = (sum a)²).

So the objective = 2n / (2n-1) * PMR ≈ PMR (for large n).

We want to minimize the PMR of the aperiodic autocorrelation (well, convolution, which is related).

This is the "dynamic range" of the autocorrelation, which is a well-studied quantity in radar and communications. The goal is to find signals with low peak-to-average power ratio (PAPR) in the autocorrelation domain.

Known results:
- For random signals, PMR ≈ O(log n) (for the envelope) or O(sqrt(log n)) (for the power).
- For "Barker" sequences (binary), the PMR is bounded by a constant.
- For "Zadoff-Chu" sequences (complex), the PMR is 1 (perfectly flat).
- For real non-negative sequences, the minimum PMR is not well-studied.

The previous solution achieved PMR ≈ 1.507 (for n=8192). This is quite good. To improve further, we need a more sophisticated approach.

One idea from the literature: "CAZAC" (Constant Amplitude Zero Autocorrelation) sequences have perfectly flat autocorrelation, but they're complex and have constant amplitude. For real non-negative sequences, we can approximate this by:
1. Taking real part of a CAZAC sequence.
2. Adding a DC offset to make it non-negative.
3. Optimizing the DC offset and possibly the phase rotation.

"""
Minimize 2*n*max_conv/sum^2 using L-BFGS on LSE + spectral phase randomization restarts.

Key innovations over previous approach:
1. Spectral phase randomization: when stuck, randomize FFT phases (preserving magnitudes)
   to escape local minima - fundamentally different from noise perturbation.
2. Chirp / Zadoff-Chu-inspired initializations that exploit known low-PAPR structures.
3. Multi-beta LSE: combine gradients from multiple smoothing levels for better conditioning.
4. Adaptive "peak tracking" that switches between focusing on the single max vs. top-K.
5. Deflation-based constructive initialization from the all-ones sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    beta_val = float(beta)
    if beta_val < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta_val * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta_val
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def multi_beta_grad(a, N, betas):
    """Weighted combination of LSE gradients at multiple beta levels."""
    n = len(a)
    total_grad = np.zeros(n)
    for i, beta in enumerate(betas):
        g, f_val, c, c_max = lse_grad(a, N, beta)
        w = (1.0 + i) / (len(betas) * (len(betas) + 1) / 2.0)
        total_grad += w * g
    total_grad /= len(betas)
    c, A = compute_conv_full(a, N)
    return total_grad, c, c.max()


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_phase_restart(a, N, rng):
    """Randomize FFT phases while preserving magnitudes, project to non-negative."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_partial_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of the FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0,
                    m_lbfgs=10, use_multi_beta=False):
    """L-BFGS on LSE-smoothed objective with adaptive beta."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    if use_multi_beta:
        betas = [beta, beta * 3, beta * 9]
        g, c, c_max = multi_beta_grad(a, N, betas)
    else:
        g, f_val, c, c_max = lse_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 150 == 0:
            beta = min(beta_max, beta * 1.1)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            if use_multi_beta:
                betas = [beta, beta * 3, beta * 9]

            else:
                g, f_val, c, c_max = lse_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if it % 40 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(12):

                if cm_ls <= c_max:

            best_cmax = cmax_new
            best_a = a_new.copy()

        y_k = g_new - g
        sy = np.dot(s_k, y_k)

            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:

                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 400 == 0 and rng.random() < 0.25:
            a = spectral_partial_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            if use_multi_beta:

def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov top-k subgradient for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        c, A = compute_conv_full(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        if it % 10 == 0:
            c_x, _ = compute_conv_full(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 1500 == 0 and rng.random() < 0.15:
            y = spectral_partial_restart(y, N, rng, frac=0.2)
            t = 1.0
    return best_a, best_cmax


def deflation_init(n, S, N, n_iters=2000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = np.real(ifft(np.abs(fft(a, N)) ** 2))[:2 * n - 1]
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def make_init_chirp(n, rng):

    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_binary_like(n, rng):
    a = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses L-BFGS on LSE with spectral phase randomization restarts,
    multi-beta gradients, and diverse initializations including
    Zadoff-Chu and deflation-based constructions.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c, _ = compute_conv_full(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(420.0, (deadline - t1) * 0.18)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("binary", make_init_binary_like(n, rng)))
    inits.append(("deflation", deflation_init(n, S, N, 1500, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = lbfgs_lse_block(a_norm, n, S, N, min(rem, 60.0), rng,
                                     beta_init=5.0, beta_max=80.0, m_lbfgs=8)
        update_best(a_l, name + "/LBFGS")

    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    # --- Phase 2: Focused L-BFGS with spectral restarts ---
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    print("=== Phase 2: Focused L-BFGS with spectral restarts ===")
    t2 = time.monotonic()
    phase2_budget = min(1000.0, (deadline - t2) * 0.45)

    a_curr = global_a.copy()
    stall = 0
    use_mb = True

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(150.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        beta_start = 8.0 + 3.0 * min(stall, 10)
        beta_end = min(400.0, 40.0 * (1 + 0.4 * stall))
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         m_lbfgs=12, use_multi_beta=use_mb)
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2 * n * sc_new / (S * S)
            print("  [P2] score=%.6f (beta %.0f-%.0f, multi_beta=%s)" % (score, beta_start, beta_end, use_mb))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.4)
                use_mb = not use_mb
                stall = 0
            elif stall >= 4:
                a_curr = spectral_phase_restart(a_curr, N, rng)
                stall = 0

    # --- Phase 3: Nesterov polishing ---
    print("=== Phase 3: Nesterov polishing ===")
    t3 = time.monotonic()
    remaining3 = min(400.0, (deadline - t3) * 0.25)
    if remaining3 > 20:
        a_ng, sc_ng = nesterov_toppk_block(global_a, n, S, N, remaining3, rng,
                                            top_k_init=30, top_k_max=400)
        update_best(a_ng, "nesterov")

    # --- Phase 4: Final high-beta L-BFGS ---
    print("=== Phase 4: Final high-beta L-BFGS ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 15
    if remaining4 > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining4, 400.0),
                                         rng, beta_init=30.0, beta_max=600.0,
                                         m_lbfgs=15, use_multi_beta=False)
        update_best(a_fin, "final_LBFGS")

    # --- Phase 5: Short Nesterov + L-BFGS alternation ---
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 30:
        half = remaining5 / 2.0
        a_ng2, sc_ng2 = nesterov_toppk_block(global_a, n, S, N, min(half, 200.0),
                                              rng, top_k_init=50, top_k_max=600)
        update_best(a_ng2, "nesterov2")
        if deadline - time.monotonic() > 20:
            a_fin2, sc_fin2 = lbfgs_lse_block(global_a, n, S, N, min(deadline - time.monotonic() - 5, 200.0),
                                               rng, beta_init=50.0, beta_max=800.0,
                                               m_lbfgs=15, use_multi_beta=False)
            update_best(a_fin2, "final2")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
