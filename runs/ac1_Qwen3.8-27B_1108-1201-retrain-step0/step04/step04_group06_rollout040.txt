"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using softmax-approximated
gradient descent with adaptive annealing, top-k refinement, and
coordinate descent polishing.

Key ideas:
1. Softmax approximation to max makes the objective smooth and differentiable
2. Adaptive beta (temperature) annealing: start smooth, end sharp
3. FFT-based gradient computation for efficiency
4. Top-k subgradient refinement for final pushes
5. Coordinate descent with bisection for local optimization
6. Multi-n exploration (4096, 8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        return np.ones(n) * (S / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) * (S / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_self_fft(a, N):
    """Compute self-convolution via FFT. N should be >= 2*len(a)-1."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def grad_softmax_obj(a, w, N):
    """
    Gradient of sum_k w[k]*c[k] w.r.t. a, where c = convolve(a,a).
    g[j] = 2 * sum_k w[k] * a[k-j] (approximately, ignoring k=2j correction).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    
    A = fft(a_pad)
    W = fft(w_pad)
    
    # Cross-correlation: (w * a)[j] = sum_k w[k] * a[k-j]
    # = IFFT(conj(FFT(w)) * FFT(a))
    g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return g


def score_from_conv(c, n):
    """Compute score from convolution and length."""
    max_c = c.max()
    return max_c  # We'll normalize separately


def softmax_gradient_step(a, S, n, N, beta, eta, rng=None):
    """
    One step of softmax-approximated gradient descent.
    Returns (new_a, new_score_estimate).
    """
    c = convolve_self_fft(a, N)
    c_max = c.max()
    
    # Compute softmax weights
    # Use c/c_max to avoid overflow
    c_norm = c / max(c_max, 1e-30)
    exp_c = np.exp(beta * (c_norm - 1.0))  # Subtract max for numerical stability
    w = exp_c / exp_c.sum()
    
    # Compute gradient
    g = grad_softmax_obj(a, w, N)
    
    # Step
    a_new = project_simplex(a - eta * g, S)
    
    return a_new, c_max


def softmax_anneal(a_init, n, N, S, max_time, rng, 
                    beta_start=5.0, beta_end=500.0,
                    n_iter_max=50000, eta=0.4):
    """
    Softmax gradient descent with beta annealing.
    Returns (best_a, best_cmax).
    """
    start = time.monotonic()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = convolve_self_fft(a, N).max()
    
    c = convolve_self_fft(a, N)
    c_max = c.max()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Anneal beta
        progress = min(1.0, it / (n_iter_max * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute weights and gradient
        c_norm = c / max(c_max, 1e-30)
        exp_c = np.exp(beta * (c_norm - 1.0))
        w = exp_c / exp_c.sum()
        
        g = grad_softmax_obj(a, w, N)
        g_norm = np.linalg.norm(g)
        
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        # Lipschitz constant ~ 2 * ||Q|| where ||Q|| <= 1 (since w sums to 1, w >= 0)
        # So eta_max = 1/L = 1/2
        # But in practice, we use a slightly smaller step for safety
        step = eta * S / max(g_norm, S * 0.01)
        step = min(step, 0.8)  # Cap step size
        
        a_new = project_simplex(a - step * g / g_norm * S, S)
        
        # Verify
        c_new = convolve_self_fft(a_new, N)
        c_max_new = c_new.max()
        
        if c_max_new < best_cmax:
            best_cmax = c_max_new
            best_a = a_new.copy()
            a = a_new.copy()
            c = c_new
            c_max = c_max_new
            stall = 0
            if it % 5000 == 0:
                elapsed = time.monotonic() - start
                print("      anneal iter %d: cmax=%.6f, beta=%.1f, time=%.0fs" % (it, best_cmax, beta, elapsed))
        else:
            a = a_new.copy()
            c = c_new
            c_max = c_max_new
            stall += 1
            if stall > 200:
                # Perturb
                a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                c = convolve_self_fft(a, N)
                c_max = c.max()
                stall = 0
        
        # Periodic re-computation of c (already done above)
        if it % 10000 == 0:
            elapsed = time.monotonic() - start
            if elapsed > max_time * 0.9:
                break
    
    return best_a, best_cmax


def topk_gradient_descent(a_init, n, N, S, max_time, rng,
                           k_start=10, k_end=200, eta=0.3):
    """
    Gradient descent using top-k subgradient.
    Returns (best_a, best_cmax).
    """
    start = time.monotonic()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = convolve_self_fft(a, N).max()
    
    c = convolve_self_fft(a, N)
    c_max = c.max()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_end, int(k_start + (k_end - k_start) * min(1.0, it / 20000.0)))
        
        # Top-k subgradient
        top_idx = np.argpartition(c, -k)[-k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        w = np.zeros(2 * n - 1)
        w[top_idx] = top_vals / total
        
        g = grad_softmax_obj(a, w, N)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        step = min(0.5, eta * S / max(g_norm, S * 0.01))
        a_new = project_simplex(a - step * g / g_norm * S, S)
        
        c_new = convolve_self_fft(a_new, N)
        c_max_new = c_new.max()
        
        if c_max_new < best_cmax:
            best_cmax = c_max_new
            best_a = a_new.copy()
            stall = 0
            if it % 5000 == 0:
                print("      topk iter %d: cmax=%.6f, k=%d" % (it, best_cmax, k))
        else:
            stall += 1
        
        a = a_new.copy()
        c = c_new
        c_max = c_max_new
        
        if stall > 100:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            c = convolve_self_fft(a, N)
            c_max = c.max()
            stall = 0
    
    return best_a, best_cmax


def coord_descent_polish(a_init, n, N, S, max_time, rng, n_rounds=500, n_top=20):
    """
    Coordinate descent: for each round, find the peak lag,
    identify top contributing coordinates, and do 1D optimization.
    """
    start = time.monotonic()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = convolve_self_fft(a, N).max()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        c = convolve_self_fft(a, N)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        peak_lag = int(np.argmax(c))
        
        # Find contributing coordinates at peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_coords:
            if time.monotonic() - start > max_time:
                break
            i = int(idx)
            j = peak_lag - i
            if j < 0 or j >= n:
                continue
            
            orig = a[i]
            if orig < 1e-15:
                continue
            
            # Binary search for best multiplier
            best_frac = 1.0
            best_val = c_max
            
            # Try a range of multipliers
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a[i] = orig * frac
                # Quick check: recompute only affected lags
                # Affected lags: k = i to i+n-1 (where c[k] involves a[i])
                # c[k] changes by (a[i] - orig) * a[k-i] + (if k-i != i) (a[i] - orig) * a[k-i]
                # Actually simpler: recompute c at affected lags
                delta = a[i] - orig
                # c[0] = a[0]^2 → if i==0, c[0] changes
                # c[k] = sum_i a[i]*a[k-i]
                # Terms involving a[i]: a[i]*a[k-i] and a[k-i]*a[i] (same if k-i=i)
                # So c[k] changes by delta * a[k-i] + (delta * a[k-i] if k-i != i else 0)
                # = delta * (a[k-i] + a[k-i]) = 2*delta*a[k-i] (if k != 2i)
                # = delta * a[k-i] (if k == 2i)
                
                # For speed, just check the peak lag and a few neighbors
                k_check = peak_lag
                dk = delta * a[j]  # change at lag peak_lag from pair (i, j)
                if i != j:
                    dk += delta * a[j]  # second term
                # Actually c[peak_lag] involves a[i]*a[j] where j = peak_lag - i
                # The change is: new_c[peak_lag] - old_c[peak_lag] = (a[i]+delta)*a[j] + (if i!=j) a[i]*a[j] + delta*a[j]
                # Wait no. c[peak_lag] = sum_m a[m]*a[peak_lag-m]
                # The terms involving a[i]: a[i]*a[peak_lag-i] = a[i]*a[j]
                # And if peak_lag-i != i: a[peak_lag-i]*a[i] = a[j]*a[i] (same term!)
                # So the contribution of a[i] to c[peak_lag] is:
                #   if i != j: a[i]*a[j] + a[j]*a[i] = 2*a[i]*a[j]
                #   if i == j: a[i]*a[i] = a[i]^2
                # The change:
                #   if i != j: 2*(a[i]+delta)*a[j] - 2*a[i]*a[j] = 2*delta*a[j]
                #   if i == j: (a[i]+delta)^2 - a[i]^2 = 2*delta*a[i] + delta^2
                
                if i != j:
                    new_c_peak = c_max + 2 * delta * a[j]
                else:
                    new_c_peak = c_max + 2 * delta * orig + delta * delta
                
                # Also need to check other lags, but for speed, just check peak
                # and a few neighbors
                worst = new_c_peak
                for k_offset in [-2, -1, 1, 2]:
                    k2 = peak_lag + k_offset
                    if 0 <= k2 < 2 * n - 1:
                        j2 = k2 - i
                        if 0 <= j2 < n and j2 != i:
                            new_ck2 = c[k2] + 2 * delta * a[j2]
                        elif 0 <= j2 < n:
                            new_ck2 = c[k2] + 2 * delta * a[j2] + delta * delta
                        else:
                            new_ck2 = c[k2]
                        worst = max(worst, new_ck2)
                
                if worst < best_val:
                    best_val = worst
                    best_frac = frac
            
            a[i] = orig * best_frac
            if best_frac != 1.0:
                improved = True
        
        if not improved:
            # Try perturbing
            if rng.random() < 0.5:
                for idx in top_coords[:5]:
                    a[int(idx)] *= rng.uniform(0.7, 1.3)
                a = project_simplex(a, S)
            else:
                break  # No improvement, stop
    
    return best_a, best_cmax


def make_structured_init(n, rng, method="flat_osc"):
    """Create structured initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    if method == "flat_osc":
        # Alternating with smooth modulation
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
    elif method == "multi_freq":
        a = 0.5 + 0.2 * np.sin(2*np.pi*3*t) + 0.2*np.sin(2*np.pi*5*t) + 0.1*np.sin(2*np.pi*8*t)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2*np.pi*(t + 3*t*t))
    elif method == "random_smooth":
        # Random low-pass filtered
        a = 0.5 + 0.3 * rng.standard_normal(n)
        # Simple smoothing (moving average)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif method == "bump_trains":
        # Train of bumps
        a = np.zeros(n)
        n_bumps = 16
        for b in range(n_bumps):
            center = int(n * b / n_bumps)
            width = n / (n_bumps * 3)
            a += 0.5 * np.exp(-0.5 * ((t * n - center) / width) ** 2)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    
    return np.clip(a, 0.01, 3.0)


def normalize_to_sum(a, n, S=None):
    """Clamp and normalize to sum S = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if S is None:
        S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    Uses softmax gradient descent with annealing, top-k refinement,
    and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    global_best_score = np.inf
    global_best_a = None
    
    def update_global(a_candidate, n):
        nonlocal global_best_score, global_best_a
        a_candidate = normalize_to_sum(a_candidate, n)
        N = 1 << (2 * n - 1).bit_length()
        c = convolve_self_fft(a_candidate, N)

        S = a_candidate.sum()

The algorithm:
1. Softmax gradient descent: approximate max(c) with softmax, compute
   smooth gradient, take projected steps with adaptive beta annealing.
2. Top-k refinement: use the top-k values of c as weights for a
   sharper (but less smooth) gradient, for final optimization.
3. Coordinate descent: iterate on peak lag, optimize individual
   coordinates via 1D search.
4. Multi-n exploration: try n=4096, 8192, 16384.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.astype(np.float64)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n, dtype=np.float64)
    r = rho[-1]
    theta = (cssv[r] - S) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_self(a, N):
    """Self-convolution of a via FFT of size N."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def grad_hankel(a, w, N):
    """
    Gradient of f(a) = sum_k w[k]*convolve(a,a)[k] w.r.t. a.
    g[j] = 2 * sum_i w[i+j] * a[i]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    A = fft(a_pad)
    W = fft(w_pad)
    g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return g
