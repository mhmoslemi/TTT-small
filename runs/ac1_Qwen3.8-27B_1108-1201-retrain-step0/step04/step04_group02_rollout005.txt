         tc = (cumsum - S) / idx
         valid = tc <= w
         if not np.any(valid):
             theta = cumsum[n-1] / n
         else:
             rho = int(np.max(np.where(valid)[0]) + 1)
             theta = (cumsum[rho - 1] - S) / rho
         return np.maximum(v - theta, 0.0)
     
     def lse_grad(y, c, beta):
         """Compute LSE gradient of max(a*a)."""
         # w_k = exp(beta * (c_k - c_max)) / sum(exp(beta * (c_j - c_max)))
         cm = c.max()
         # For numerical stability
         shifted = beta * (c - cm)
         # Clip to prevent overflow
         shifted = np.clip(shifted, -500, 500)
         e = np.exp(shifted)
         w = e / e.sum()
         n = len(y)
         # Gradient = sum_k w_k * a[k-i] for valid k-i
         # = cross-correlation of w and a (reversed)
         fsize = 1 << (len(w) + n - 1).bit_length()
         W = fft(w, fsize)
         Y = fft(y, fsize)
         grad = np.real(ifft(W * np.conj(Y)))[:n]
         return grad
     
     def nestevr_lse_optimize(a_init, n, S, max_time, rng):
         """FISTA with LSE smoothing and adaptive beta."""
         a = project_simplex(a_init.copy(), S)
         y = a.copy()
         t = 1.0
         start = time.monotonic()
         
         # Adaptive LSE beta: start small, increase over time
         beta_start = 5.0
         beta_end = 2000.0
         
         best_obj = np.inf
         best_a = a.copy()
         it = 0
         
         # Pre-allocate
         c = fft_convolve_self(y)
         
         while time.monotonic() - start < max_time:
             it += 1
             # Adaptive beta
             progress = min(1.0, (time.monotonic() - start) / max_time)
             beta = beta_start * (beta_end / beta_start) ** progress
             
             # Compute LSE gradient at y
             grad = lse_grad(y, c, beta)
             
             # Lipschitz constant
             a_norm_sq = np.dot(a, a)
             lip = 1.0 + beta * a_norm_sq
             step = 1.0 / np.sqrt(lip)
             
             # FISTA update
             x_new = project_simplex(y - step * grad, S)
             t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
             y = x_new + ((t - 1.0) / t_new) * (x_new - a)
             y = np.maximum(y, 0.0)
             a = x_new
             t = t_new
