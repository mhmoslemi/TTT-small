"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient with
Nesterov acceleration, spectral shaping, and multi-length search.

Key ideas:

- Nesterov-accelerated projected subgradient descent
- Adaptive peak count (K) and step size

- Multiple sequence lengths

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - S) / rho > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho_max = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_max - 1] - S) / rho_max
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, N):
    """Self-convolution via FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def compute_score(a):
    """Compute the objective."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    c = fft_convolve(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def compute_peak_gradient(a, k, n):
    """
    Compute gradient of c_k with respect to a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        # j in [lo, hi], k-j in [k-hi, k-lo], both in [0, n-1]
        g[lo:hi+1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def multi_peak_gradient(a, lags, weights, n):
    """Weighted sum of peak gradients."""
    g = np.zeros(n)
    for k, w in zip(lags, weights):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def spectral_compress(a, N, alpha):
    """
    Apply spectral power compression: P -> P^alpha.
    This flattens the power spectrum, reducing autocorrelation peaks.
    alpha < 1 compresses (flattens), alpha > 1 expands (sharpens).
    """
    A = fft(a, N)
    P = np.abs(A) ** 2
    # Normalize P so that sum(P) is preserved (approximately)
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    # Compression: P_new = P_mean * (P/P_mean)^alpha
    P_new = P_mean * np.power(P / P_mean, alpha)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)


def spectral_linear_compress(a, N, alpha):
    """
    Linear compression: P_new = (1-alpha)*P_mean + alpha*P.
    alpha in (0,1) compresses toward the mean.
    """
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    P_new = (1.0 - alpha) * P_mean + alpha * P
    phase = np.angle(A)
    A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)

    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)

    # Use a conservative initial step
    L_est = 2.0 * np.max(a) * n / S  # Rough Lipschitz estimate
    eta = 1.0 / max(L_est, 1e-10)
    
    start = time.monotonic()
    it = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        # Evaluate at y (Nesterov extrapolated point)
        c = fft_convolve(y, N)
        cur_peak = np.max(c)
        cur_score = 2.0 * n * cur_peak / (y.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        if best_score <= target:
            break
        
        # Find top K peaks
        # Adaptive K: start small, grow over time

        K = max(5, int(5 + 195 * progress))  # 5 to 200
        K = min(K, 2 * n - 1)

        c_median = np.median(c)

        top_vals = c[top_idx]
        # Weight: proportional to (val - median)^+ or val^2
        weights = np.maximum(top_vals - c_median, 0.0)
        w_sum = weights.sum()
        if w_sum < 1e-30:

        else:
            weights = weights / w_sum
