import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def autocorrelation_fft(a, N):
    """Compute full autocorrelation via FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2*len(a)-1]

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2*n-1).bit_length()
    c = autocorrelation_fft(a, N)
    return 2.0 * n * c.max() / (S*S)

def spectral_subgradient_polish(a_init, n, S, N, max_iter, rng, T_init=1.0, T_min=0.01, alpha=1.0, perturb_every=200):
    """
    Spectral subgradient method with adaptive temperature.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, max_iter+1):
        # Adaptive temperature
        T = max(T_min, T_init * (T_min/T_init)**(t/max_iter))
        
        # Compute autocorrelation
        c = autocorrelation_fft(a, N)
        
        # Softmax weighting
        c_shifted = c - c.max()  # for numerical stability
        w = np.exp(c_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w = w / w_sum
        
        # Compute gradient: 2 * (w * a) via FFT
        # w has length 2n-1, a has length n
        # (w * a)[j] = sum_k w[k] * a[j-k] (full convolution, length 2n-1+n-1 = 3n-2)
        # But we only need [0, n-1]
        w_pad = np.zeros(N)
        w_pad[:2*n-1] = w
        a_pad = np.zeros(N)
        a_pad[:n] = a
        W = fft(w_pad)
        A = fft(a_pad)
        grad = 2.0 * np.real(ifft(W * A))[:n]
        
        # Step size
        step = alpha / (1.0 + 0.05 * t)
        
        # Update
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        
        # Evaluate
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        # Periodic perturbation
        if t % perturb_every == 0:
            noise_scale = max(0.001, 0.01 * (T / T_init)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_score

"""
Minimize 2*n*max(autocorr(a))/sum(a)^2 using FFT-based spectral subgradient
method with adaptive temperature, spectral equalization, and multi-start search.

The key insight: the autocorrelation c = ifft(|FFT(a)|^2), so minimizing its
peak is equivalent to making the power spectrum as flat as possible. We use
a subgradient method where the gradient is computed exactly via FFT cross-
correlation, with an adaptive softmax temperature that gradually focuses on
the peaks.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def full_autocorr(a, N):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute objective 2*n*max(c)/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2 * n - 1).bit_length()
    c = full_autocorr(a, N)
    return float(2.0 * n * c.max() / (S * S))


def compute_gradient(w, a, N_grad):
    """
    Compute gradient of f = sum_k w[k]*c[k] w.r.t. a,
    where c = full_autocorr(a).
    Returns array of length n.
    """
    n = len(a)
    W = fft(w, N_grad)
    A = fft(a, N_grad)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    grad = 2.0 * corr
    # Diagonal correction: subtract w[2i]*a[i]
    diag_idx = 2 * np.arange(n)
    diag_idx = np.minimum(diag_idx, 2 * n - 2)
    grad -= w[diag_idx] * a
    return grad


def spectral_subgradient(a_init, n, S, N, N_grad, max_iter, rng,
                         T_init=1.0, T_min=0.005, base_step=1.0,
                         perturb_every=500, eq_alpha=0.0):
    """
    Spectral subgradient optimization.
    
    Args:
        a_init: initial sequence (will be normalized)
        n: length
        S: target sum (= sqrt(2n) for normalization)
        N: FFT size for autocorrelation (>= 2n-1)
        N_grad: FFT size for gradient (>= 3n-2)
        max_iter: number of iterations
        rng: random number generator
        T_init: initial temperature for softmax
        T_min: minimum temperature
        base_step: base step size
        perturb_every: how often to add random perturbation
        eq_alpha: spectral equalization strength (0 = off)
    
    Returns:
        (best_a, best_score)
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, max_iter + 1):
        # Adaptive temperature (exponential decay)
        T = max(T_min, T_init * (T_min / T_init) ** (t / max_iter))
        
        # Compute autocorrelation
        c = full_autocorr(a, N)
        
        # Softmax weighting (numerically stable)
        c_shifted = c - c.max()
        w = np.exp(c_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = compute_gradient(w, a, N_grad)
        
        # Step size (decreasing)
        step = base_step / (1.0 + 0.01 * t)
        
        # Update
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        
        # Optional: spectral equalization
        if eq_alpha > 0:
            A = fft(a_new, N)
            P = np.abs(A) ** 2
            # Target: flat spectrum (except DC)
            P_target = np.full(N, S * S / N)
            P_target[0] = S * S
            P_new = (1 - eq_alpha) * P + eq_alpha * P_target
            A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
            a_eq = np.real(ifft(A_new))[:n]
            a_new = 0.5 * a_new + 0.5 * project_simplex(a_eq, S)
        
        # Evaluate
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        # Periodic perturbation to escape local minima
        if t % perturb_every == 0:
            noise_scale = max(0.0005, 0.005 * (T / T_init)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_score
