```python
import numpy as np
import time

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))
```

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using soft-max mirror descent
with Nesterov acceleration, adaptive learning rates, and multi-start.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute 2*n*max(c)/S^2."""
    n = len(a)
    c = fft_convolve_self(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_softmax_gradient(a, w, n):
    """Compute gradient of soft-max objective. Returns 2*(w cross-corr a)[:n]."""
    size = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def mirror_descent_step(a, grad, lr, S, n):
    """Multiplicative (mirror descent) update with simplex projection."""
    # a_new = a * exp(-lr * grad) / normalize
    # Clip exponent to avoid overflow/underflow
    exp_arg = -lr * grad
    exp_arg = np.clip(exp_arg, -50, 50)
    a_temp = a * np.exp(exp_arg)
    a_new = project_simplex(a_temp, S)
    return a_new


def projected_gd_step(a, grad, lr, S):
    """Standard projected gradient descent step."""
    a_temp = a - lr * grad
    a_new = project_simplex(a_temp, S)
    return a_new


def optimize_sequence(a_init, n, S, time_budget, rng, method='mirror'):
    """
    Optimize a single sequence using soft-max gradient descent.
    Returns (best_a, best_score).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Nesterov state
    y = a.copy()
    t_val = 1.0
    
    # Parameters
    beta = 1.0  # soft-max sharpness
    lr = 0.5 / S  # initial learning rate
    lr_min = 0.01 / S
    lr_max = 2.0 / S
    lr_decay = 0.9999
    noise_interval = 2000  # iterations between noise injections
    noise_scale = 0.02
    check_interval = 200  # iterations between score checks
    
    it = 0
    start_time = time.monotonic()
    n_fft = 1 << (2 * n - 1).bit_length()
    
    while True:
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget:
            break
        
        it += 1
        
        # Compute convolution of y
        A_y = fft(y, n_fft)
        c = np.real(ifft(A_y * A_y))[:2 * n - 1]
        c_max = c.max()
        
        # Soft-max weights
        if beta > 0:
            exp_c = np.exp(beta * (c - c_max))
            w = exp_c / exp_c.sum()
        else:
            # Hard max: uniform weight on top-k
            top_k = max(1, min(20, 2 * n - 1))
            top_idx = np.argsort(c)[::-1][:top_k]
            w = np.zeros(2 * n - 1)
            w[top_idx] = 1.0 / top_k
        
        # Compute gradient
        grad = compute_softmax_gradient(y, w, n)
        
        # Adaptive learning rate: scale by gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            # Target: change in objective ~ 0.1 * c_max per step
            # delta ~ lr * ||grad||^2, so lr ~ 0.1 * c_max * S^2 / (n * ||grad||^2)
            # Simplified: lr ~ c_max / (||grad|| * ||a||)
            target_change = 0.05 * c_max
            lr_adaptive = target_change * S / (gnorm * max(gnorm, 1e-10))
            lr = max(lr_min, min(lr_max, 0.5 * lr + 0.5 * lr_adaptive))
        
        # Take step
        if method == 'mirror':
            a_new = mirror_descent_step(y, grad, lr, S, n)
        else:
            a_new = projected_gd_step(y, grad, lr, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = a_new + ((t_val - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y
        y_s = y.sum()
        if y_s < 1e-15:
            y = a_new.copy()
        else:
            y = y * (S / y_s)
        a = a_new
        t_val = t_new
        
        # Periodic score check
        if it % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # Reset Nesterov when we find improvement
                # (optional - can hurt convergence)
            
            # Anneal beta: increase over time for sharper focus on peak
            progress = elapsed / time_budget
            beta = 1.0 + 10.0 * progress * progress
            
            # Decay learning rate
            lr *= lr_decay
        
        # Periodic noise injection
        if it % noise_interval == 0 and rng.random() < 0.3:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_val = 1.0  # Reset momentum
        
        # Safeguard: if score is getting worse, revert
        if it % (check_interval * 5) == 0:
            sc_check = compute_score(a)
            if sc_check > best_score * 1.5:
                # Revert to best
                a = best_a.copy()
                y = best_a.copy()
                t_val = 1.0
                lr = lr_max * 0.5
    
    # Final score check
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_a, best_score
```

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using soft-max mirror descent
with heavy-ball momentum, adaptive learning rates, multi-start, and
periodic perturbations. Uses FFT-based gradient computation via
cross-correlation for O(n log n) per-iteration cost.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute 2*n*max(c)/S^2."""
    n = len(a)
    c = fft_convolve_self(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_gradient(a, w, n, n_fft):
    """
    Compute gradient of soft-max log-sum-exp objective.
    grad[j] = 2 * (w cross-corr a)[j] = 2 * sum_m w[j+m] * a[m]
    Computed via FFT: ifft(conj(FFT(w)) * FFT(a))[:n]
    """
    size = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def mirror_step(a, grad, lr, S, eps=1e-12):
    """
    Mirror descent (multiplicative) step on simplex.
    a_new = a * exp(-lr * grad / a) normalized to sum S.
    """
    ratio = grad / np.maximum(a, eps)
    exp_arg = -lr * ratio
    exp_arg = np.clip(exp_arg, -40.0, 40.0)
    a_temp = a * np.exp(exp_arg)
    s = a_temp.sum()
    if s < 1e-30:
        return a.copy()
    return a_temp * (S / s)


def projected_step(a, grad, lr, S):
    """Standard projected gradient step."""
    a_temp = a - lr * grad
    return project_simplex(a_temp, S)


def optimize_single(a_init, n, S, time_budget, rng, method='mirror',
                    beta_init=1.0, beta_max=8.0, lr_init=None,
                    momentum=0.85, noise_every=1500, noise_frac=0.015,
                    check_every=100, restart_prob=0.2):
    """
    Optimize a single sequence with soft-max gradient + mirror/projected steps.
    Returns (best_a, best_score).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = compute_score(a)
    best_time = time.monotonic()

    # Momentum (heavy ball)
    y = a.copy()
    mu = momentum

    # Parameters
    beta = beta_init
    if lr_init is None:
        lr = 2.0 / max(n, 10)
    else:

    lr_min = 0.1 / n
    lr_max = 10.0 / n

    it = 0
    t0 = time.monotonic()
    stagnant = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        it += 1

        # Compute convolution of momentum point y
        Ay = fft(y, n_fft)
        c = np.real(ifft(Ay * Ay))[:2 * n - 1]

        w = exp_c / exp_c.sum()

                a = project_simplex(a + 0.5 * noise, S)
                mu = momentum

    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    sc_y = compute_score(y)
    if sc_y < best_score:
        best_score = sc_y
        best_a = y.copy()

    return best_a, best_score


def make_inits(n, rng, S, n_inits=8):
    """Generate diverse starting points."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) * S / n)

        x = np.maximum(x, 0.01)
        inits.append(x * S / x.sum())

    # Chirp
    x = 0.5 + 0.3 * np.sin(2.0 * np.pi * (3 * tt + 2 * tt * tt))
    x = np.maximum(x, 0.01)
    inits.append(x * S / x.sum())

    # Random
    x = rng.random(n) + 0.05
    inits.append(x * S / x.sum())

    # Smooth random (low-pass)
    x = rng.standard_normal(n)
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    x = np.convolve(x, kernel, mode='same')
    x = (x - x.min()) / (x.max() - x.min() + 1e-10)
    x = 0.2 + 0.8 * x
    inits.append(x * S / x.sum())

    # Two-lobe
    x = np.exp(-0.5 * ((tt - 0.25) * 5) ** 2) + np.exp(-0.5 * ((tt - 0.75) * 5) ** 2) + 0.05
    inits.append(x * S / x.sum())

    # Alternating (checkerboard-like)
    x = np.ones(n)
    x[::2] = 1.5
    x[1::2] = 0.5
    inits.append(x * S / x.sum())

    # Triangle wave
    x = 1.0 - np.abs(2.0 * tt - 1.0)
    x = np.maximum(x, 0.05)
    inits.append(x * S / x.sum())

    return inits[:n_inits]


def refine_by_restarts(a_best, n, S, time_budget, rng, n_restarts=5):
    """Refine by perturbing the best and re-optimizing."""
    best_a = a_best.copy()
    best_score = compute_score(a_best)
    t0 = time.monotonic()
    per_restart = time_budget / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_budget:
            break

        # Perturb

        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        a_pert = np.maximum(a_best + noise, 0.0)
        sp = a_pert.sum()
        if sp < 1e-15:
            continue
        a_pert = a_pert * (S / sp)

        # Optimize

                                    method=method, beta_init=3.0, beta_max=12.0,
                                    momentum=0.9)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print("    Restart %d improved: %.6f" % (r, best_score))

    return best_a, best_score
```
