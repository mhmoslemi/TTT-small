"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def subgradient_at_lags(y, c, lags):
    """Compute subgradient as average over specified lags."""
    n = len(y)
    w = np.zeros(len(c))
    for k in lags:
        w[k] = 1.0
    w /= len(lags)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def fw_step(a, grad, S):
    """Frank-Wolfe step: move toward the vertex minimizing <grad, e_j>."""
    j = int(np.argmin(grad))
    # Line search: minimize f(a + t*(e_j - a)) for t in [0,1]
    # Use a simple grid search
    a_new = a.copy()
    e_j = np.zeros(len(a))
    e_j[j] = 1.0
    best_t = 0.0
    best_val = np.inf
    c_cur = fft_convolve_self(a)
    cur_max = c_cur.max()
    for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
        a_trial = (1 - t) * a + t * e_j
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        # We need sum = S, so renormalize
        a_trial = a_trial * (S / s)
        c_trial = fft_convolve_self(a_trial)
        val = c_trial.max()
        if val < best_val:
            best_val = val
            best_t = t
    return (1 - best_t) * a + best_t * e_j

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase hybrid optimizer combining FISTA with LSE smoothing,
Frank-Wolfe structural steps, adaptive active-set subgradient, and
coordinate "swap" operations for escaping local minima.
Key improvements: structural FW steps for large moves, swap-based local
search, and adaptive multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lags(y, c, lags):
    """Compute subgradient as average over specified lags."""
    n = len(y)
    w = np.zeros(len(c))
    for k in lags:
        w[k] = 1.0
    w /= len(lags)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k, tol_frac=0.0):
    """Subgradient from top-k largest convolution values."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        threshold = max_val * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
        if len(active) > k:
            idx = np.argsort(c)[::-1][:k]
        else:
            idx = active
    else:
        idx = np.argsort(c)[::-1][:k]
    return subgradient_at_lags(y, c, idx)


def swap_step(a, c, S, amount=0.01):
    """
    Structural swap: move mass from hot pairs to cold coordinates.
    Identifies the peak lag, finds the hottest contributing pair,
    and moves mass to a cold coordinate.
    Returns (new_a, improved).
    """
    n = len(a)
    k_star = int(np.argmax(c))
    # Find the j that maximizes a[j] * a[j + k_star]
    if k_star == 0:
        # For lag 0, hot coordinates are just the large ones
        # Move from the largest to the smallest
        j_hot = int(np.argmax(a))
    else:
        # Valid j range: 0 <= j < n - k_star for positive lags
        # Handle negative lags too
        if k_star < n:
            products = a[:n - k_star] * a[k_star:]
            j_hot = int(np.argmax(products))
        else:
            j_hot = int(np.argmax(a))
    
    j_cold = int(np.argmin(a))
    if j_hot == j_cold:
        return a.copy(), False
    
    # Move mass from hot to cold
    delta = amount * a[j_hot]
    if delta > a[j_hot] * 0.5:
        delta = a[j_hot] * 0.5
    if delta < 1e-15:
        return a.copy(), False
    
    a_new = a.copy()
    a_new[j_hot] -= delta
    a_new[j_cold] += delta
    # Renormalize
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False
    a_new = a_new * (S / s)
    
    c_new = fft_convolve_self(a_new)
    improved = c_new.max() < c.max()
    return a_new, improved


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]

                               use_swap=True)
            obj4 = compute_obj(a4)
            print("  Upsampled n=%d: obj=%.6f (%.0fs)" % (n2, obj4, time.monotonic() - t4))
            if obj4 < global_best_obj:
                global_best_obj, global_best_a = obj4, a4.copy()

            # Polish the upsampled
            rem4 = p4 - (time.monotonic() - t4)
            if rem4 > 20:
                a4b, _ = run_subgradient_polish(global_best_a, rem4, rng)
                obj4b = compute_obj(a4b)
                print("  Upsampled polish: obj=%.6f (%.0fs)" % (obj4b, time.monotonic() - t4))
                if obj4b < global_best_obj:
                    global_best_obj, global_best_a = obj4b, a4b.copy()
        else:
            # Already large, try downsample then re-optimize at original
            n_ds = len(global_best_a) // 2
            a_ds = downsample(global_best_a, 2)
            S_ds = np.sqrt(2.0 * n_ds)
            a_ds = a_ds * (S_ds / a_ds.sum())
            obj_ds = compute_obj(a_ds)
            print("  Downsampled n=%d: obj=%.6f" % (n_ds, obj_ds))
            if obj_ds < global_best_obj:
                global_best_obj, global_best_a = obj_ds, a_ds.copy()
                a_ds2, _ = run_subgradient_polish(global_best_a, p4 * 0.8, rng)
                obj_ds2 = compute_obj(a_ds2)
                if obj_ds2 < global_best_obj:
                    global_best_obj, global_best_a = obj_ds2, a_ds2.copy()

    # === Phase 5: Final intensive polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p5 = remaining * 0.85
        print("\n=== Phase 5: Final polish at n=%d (%.0fs) ===" % (len(global_best_a), p5))
        t5 = time.monotonic()

        # Alternate between subgradient and FISTA
        half = p5 * 0.5
        a5, _ = run_subgradient_polish(global_best_a, half, rng)
        obj5 = compute_obj(a5)
        print("  Final-sub: obj=%.6f (%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj, global_best_a = obj5, a5.copy()

        rem5 = p5 - (time.monotonic() - t5)
        if rem5 > 5:
            a5b, _ = run_fista(global_best_a, rem5, rng,
                                beta_init=500.0, beta_rate=0.001,
                                beta_max=100000.0, step_scale=0.15,
                                use_swap=True)
            obj5b = compute_obj(a5b)
            print("  Final-fista: obj=%.6f (%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj, global_best_a = obj5b, a5b.copy()

    # Final verification
    print("\n=== Final Result ===")
    if global_best_a is not None:
        print("Best internal obj: %.6f (n=%d)" % (global_best_obj, len(global_best_a)))
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
            print("Fallback obj: %.6f" % compute_obj(global_best_a))
    else:
        n_fb = 8192
        final_list = [float(x) for x in np.ones(n_fb)]
        print("Using uniform fallback, n=%d" % n_fb)

    print("Total time: %.0fs" % (time.monotonic() - (deadline - budget_s + 30.0)))
    return final_list
