def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)

After changing a[i] from x to x' and rescaling:
a_new[j] = a[j] * S / (S - x + x') for j ≠ i
a_new[i] = x' * S / (S - x + x')

Let f = S / (S - x + x') (the rescaling factor, which is > 1 when x' < x).

The new convolution:
b_new[k] = sum_j a_new[j] * a_new[k-j]

This is more complex because all elements change. The simple linear analysis above doesn't apply directly.

Given this complexity, I think a practical approach is:
1. Try several values of x' (e.g., x*0.1, x*0.3, x*0.5, x*0.7, 0)
2. For each, compute the exact new convolution and max
3. Keep the best

This is simpler and more robust than trying to derive a closed-form solution.

adaptive lag targeting, coordinate descent with incremental updates, and multi-scale

coordinate updates avoiding full FFT, adaptive step scheduling, and phase diversity.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_fft_padded(a, N):
    """Compute FFT of a padded to length N."""
    a_pad = np.zeros(N)
    n = len(a)
    a_pad[:n] = a
    return fft(a_pad)

    """

    Returns (best_a, best_score, best_b).
    """
    S = a.sum()

    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target:
        return a, c_max, c

    # Find top-k peak lags with excess weighting
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]

        excess = top_vals
        excess_sum = top_vals.sum()
    if excess_sum < 1e-30:
        return a, c_max, c
    weights = excess / excess_sum

    # Build effective gradient in frequency domain
    m = np.arange(N)
    grad_real = np.zeros(N)
    for i, k in enumerate(top_indices):
        grad_real += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Try multiple step sizes (line search)
    base_alpha = 0.3 * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.2)
    step_multipliers = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    best_a = a.copy()
    best_score = c_max
    best_c = c.copy()

    for sm in step_multipliers:
        alpha = base_alpha * sm
        P_new = P * np.exp(-alpha * np.maximum(grad_real, 0.0))
        # Normalize to maintain same total power (preserves sum approximately)
        P_sum_new = P_new.sum()
        if P_sum_new < 1e-30:
            continue
        P_new = P_new * (P.sum() / P_sum_new)
        A_mag = np.sqrt(P_new)
        # Keep phase
        A_new = A_mag * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * (S / s_new)

        # Compute actual score
        c_new = fft_convolve_self(a_new)
        score_new = c_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            best_c = c_new.copy()

    return best_a, best_score, best_c

def random_phase_perturb(a, N, n, S, sigma, rng):

    phase_noise = sigma * rng.standard_normal(N)
    A_new = A * np.exp(1j * phase_noise)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new * (S / s_new)
    return a_new
