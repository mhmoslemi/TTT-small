Actually, for the coordinate descent, the most efficient approach is:
1. Compute d_k = 2*a[k-j] - 2*a[k-m] (the linear change in c_k per unit δ)
2. c_trial[k] = c[k] + δ * d[k] + (quadratic correction)
3. Find δ that minimizes max(c_trial)

The quadratic correction is small (O(δ^2)) and can be ignored for a first pass. Then we do a line search.

For step 2, computing d_k for all k is O(n) (just lookups into a). Then the line search is a 1D optimization over δ ∈ [-a_j, a_m].

The function to minimize is f(δ) = max_k (c[k] + δ * d[k]). This is a 1D convex function (max of linear functions). We can find the minimum via:

- This is a linear programming problem in 1D, solvable by finding the intersection points

Actually, for 1D, we can just do a ternary/golden section search since f is convex. Or we can use the fact that the minimum of the max of lines occurs at a vertex of the upper envelope.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_score(a):
    c = fft_convolve_self(a)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def lse_grad(a, c, beta, fsize):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # gradient = 2 * conv(w, a) truncated to n
    # w has length 2n-1, a has length n
    # conv(w, a) has length 3n-2
    # But we only need the first n elements (or the relevant range)
    # Actually, the gradient of LSE w.r.t. a[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
    # This is 2 * (w * a)[j] where * is the "sliding dot product"
    # More precisely: grad[j] = 2 * sum_{k} w[k] * a[k-j] where a[x]=0 for x outside [0,n-1]
    # = 2 * (cross-correlation of w and a at lag j)
    # = 2 * ifft(fft(w) * conj(fft(a))) ... not quite
    # 
    # Let me think again. c[k] = sum_i a[i] * a[k-i]
    # grad_j = d/d a_j [sum_k w_k * c_k] = sum_k w_k * d c_k / d a_j = sum_k w_k * 2 * a[k-j]
    # (where a[k-j] = 0 if k-j outside [0, n-1])
    # = 2 * sum_{k: 0 <= k-j < n} w[k] * a[k-j]
    # Let l = k-j, then k = l+j:
    # = 2 * sum_{l: -j <= l < n-j} w[l+j] * a[l]
    # = 2 * sum_l a[l] * w[l+j]
    # This is 2 * (cross-correlation of a and w at lag j)
    # = 2 * ifft(conj(fft(a)) * fft(w)) ... 
    # Actually, cross-correlation R_aw[j] = sum_l a[l] * w[l+j]
    # DFT: R_aw = ifft(conj(A) * W) where A = fft(a), W = fft(w)
    # But we need to be careful about padding and the definition of cross-correlation.
    
    # Alternative: just use the formula directly with FFT

def spectral_equalize(a_init, n, S, max_time, rng):
    """Novel spectral equalization: reduce peak in frequency domain."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        
        if P_mean < 1e-30:
            break
        
        # Identify "hot" frequencies: where P is above mean
        # Reduce them, boost cold ones
        # This should make c = ifft(A^2) flatter
        
        # Method: P_new = P^alpha / mean(P^alpha) for alpha < 1 (compression)
        alpha = 0.5 + 0.3 * (it / max(it, 100))
        P_new = np.power(np.maximum(P, 1e-30), alpha)
        P_new = P_new / P_new.mean()
        
        # Blend
        blend = min(0.3, 0.1 + 0.01 * it / 100)
        P_final = (1 - blend) * P / P_mean + blend * P_new
        
        mag_new = np.sqrt(P_final)
        # Preserve total energy
        mag_new *= np.sqrt(P.sum() / max(P_final.sum(), 1e-30))
        
        A_new = phase * mag_new
        a_raw = np.real(ifft(A_new))[:n]
        
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best
