"""

Strategy: Multi-phase optimization combining:
1. FISTA with adaptive LSE smoothing and backtracking line search
2. Active-set coordinate descent (focus on top convolution lags)

Key improvements: backtracking FISTA, efficient active-set CD,

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix floating point: ensure exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference to non-zero elements
        nonzero = result > 1e-15
        if nonzero.sum() > 0:
            result[nonzero] += diff / nonzero.sum()
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    val = cm + np.log(tot) / beta
    return val, grad


def subgradient_at_argmax(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def fista_optimize(a_init, n, S, max_time, rng,
                   beta_init=5.0, beta_max=300.0,
                   beta_ramp=1000, use_backtrack=True):
    """FISTA with adaptive LSE and optional backtracking."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_size = 0.01

    c = fft_convolve_self(y)
    cur = c.max()
    best = cur
    best_a = y.copy()

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp))
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-10:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Compute gradient
        lse_val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if use_backtrack:
            # Lipschitz estimate for LSE gradient
            # L ≈ beta * (max_w^2 - mean_w^2) * 4 * max_a^2... approximate
            w_max = np.exp(beta * (c.max() - c.max()))  # = 1 at argmax
            # Simpler: use the Hessian bound ~ 2*beta*max_w * sum_a
            L = 4.0 * beta * a.sum() * (c.max() - c.min() + 1.0 / beta)
            step = 1.0 / max(L, 1e-6)
            step = min(step, 0.1)
        else:
            step = 1.0 / (2.0 * beta)
            if it - last_improve > 500:
                step *= 0.7

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = np.ones(n) * S / n
        else:
            y_new = y_new * (S / s)

        # Backtracking: check if we made progress
        if use_backtrack:
            c_trial = fft_convolve_self(y_new)
            if c_trial.max() > cur + 0.05 * (it < 100):
                # Reduce step
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
                y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
                y_new = np.maximum(y_new, 0.0)
                s = y_new.sum()
                if s > 1e-15:
                    y_new = y_new * (S / s)
                else:
                    y_new = np.ones(n) * S / n

        a = x_new
        t = t_new
        y = y_new

        # Periodic noise to escape local minima

            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:

                t = 1.0

        # Convergence check
        if it - last_improve > 3000 and beta >= beta_max:

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def active_set_cd(a_init, n, S, max_time, rng, top_k=10, n_pairs_per_iter=50):
    """Active-set coordinate descent focusing on top convolution lags."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    no_improve_iter = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Find top-k lags
        top_lags = np.argsort(c)[::-1][:top_k]
        current_max = c[top_lags[0]]

        for _ in range(n_pairs_per_iter):
            if time.monotonic() - start > max_time - 0.5:
                    break

            # Choose a target lag (weighted towards the max)
            k_target = top_lags[rng.integers(0, min(3, len(top_lags)))]

            # Choose pair (j, m): transfer from m to j
            # Good candidates: j where a[j] is small (room to grow)
            #                m where a[m] is large (room to shrink)
            if rng.random() < 0.7:
                # Targeted: pick m with large contribution to c_{k_target}
                # c_k = sum_i a_i * a_{i+k}, contribution of a_m is a_{m-k} + a_{m+k}
                lo = max(0, k_target - n + 1)
                hi = min(k_target, n - 1)
                contrib = np.zeros(n)
                idx = np.arange(lo, hi + 1)
                contrib[idx] = a[k_target - idx]
                # Also a[m + k] term
                idx2 = np.arange(max(0, -k_target), min(n - k_target, n))
                for i in idx2:
                    mi = i + k_target
                    if mi < n:
                        contrib[mi] += a[i]
                # Pick m from high-contribution indices
                m_candidates = np.argsort(contrib)[::-1][:50]
                m = m_candidates[rng.integers(0, len(m_candidates))]
                # Pick j from low-a indices (to balance)
                sort_idx = np.argsort(a)
                j_pool = sort_idx[:max(10, n // 10)]
                j = j_pool[rng.integers(0, len(j_pool))]
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute d[k] = change in c_k per unit delta (transfer from m to j)
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val -= a[mk]
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val

            # Quadratic correction for k=0
            quad_coeff = 2.0  # c_0 changes by 2*(a_j - a_m)*delta + 2*delta^2

            # Binary search for optimal delta in [-a[j], a[m]]
            lo_d = -a[j]
            hi_d = a[m]

            def obj_at_delta(dlt):
                # Only check top lags for speed
                vals = c[top_lags] + dlt * d[top_lags]
                vals[0] += quad_coeff * (a[j] - a[m]) * dlt + quad_coeff * dlt * dlt
                return vals.max()

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ga = lo_d
            gb = hi_d
            gc = gb - gr * (gb - ga)
            gd = ga + gr * (gb - ga)
            fc = obj_at_delta(gc)
            fd = obj_at_delta(gd)
            for _ in range(30):
                if fc < fd:
                    gb = gd
                    gd = gc
                    fd = fc
                    gc = gb - gr * (gb - ga)
                    fc = obj_at_delta(gc)
                else:
                    ga = gc
                    gc = gd
                    fc = fd
                    gd = ga + gr * (gb - ga)
                    fd = obj_at_delta(gd)

            delta = (ga + gb) / 2.0
            delta = max(lo_d, min(hi_d, delta))

            if abs(delta) < 1e-14:
                continue

            # Apply and check
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < -1e-14 or a_trial[m] < -1e-14:
                continue

            # Full check on top lags
            c_trial_top = c[top_lags] + delta * d[top_lags]
            c_trial_top[0] += 2.0 * (a[j] - a[m]) * delta + 2.0 * delta * delta
            trial_max = c_trial_top.max()

            if trial_max < best - 1e-12:
                a = a_trial
                c[top_lags] = c_trial_top
                # Update other lags approximately (will be corrected periodically)
                c = c + delta * d
                c[0] = c[0] - delta * d[0] + 2.0 * (a[j] - a[m]) * delta + 2.0 * delta * delta
                best = trial_max
                best_a = a.copy()
                improved = True

        # Periodic full recompute
        if it % 10 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve_iter += 1
            if no_improve_iter > 5:
                # Inject noise
                noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-no_improve_iter / 20.0)
                if noise_scale < 1e-9:
                    break
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()
                no_improve_iter = 0
        else:
            no_improve_iter = 0

    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng):
    """FFT-based spectral flattening."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    alpha = 0.3

    c = fft Convolve_self(a) if False else fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Flatten power spectrum
        progress = min(1.0, it / 500.0)
        alpha = 0.3 + 0.5 * progress

        P_new = (1.0 - alpha) * P / P_mean + alpha
        # Enhanced: inverse weighting for peaks
        peak_ratio = P / (P_mean + 1e-30)
        P_new = P_new * (1.0 + 0.2 * (1.0 / np.sqrt(np.maximum(peak_ratio, 0.1)) - 1.0))

        mag_new = np.sqrt(np.maximum(P_new, 0))
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_new = np.maximum(a_raw, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.new.copy() if hasattr(a, 'new') else a_new.copy()

        a = a_new

    return best_a, best


def subgradient_average(a_init, n, S, max_time, rng):
    """Subgradient method with averaging for final polish."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    avg = np.zeros(n)
    count = 0
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur = c.max()
        k_max = int(np.argmax(c))

        # Subgradient
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        grad = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: O(1/sqrt(t))
        step = 1.0 / np.sqrt(it + 1.0)
        # Scale to be reasonable
        step = step * S / (gnorm + 1e-10)
        step = min(step, 0.5 * S / n)

        a = project_simplex(a - step * grad, S)

        # Running average
        avg = avg + a
        count += 1
        a_avg = avg / count

        c_avg = fft_convolve_self(a_avg)
        if c_avg.max() < best:
            best = c_avg.max()
            best_a = a_avg.copy()

        if (it + 1) % 100 == 0:
            c_cur = fft_convolve_self(a)
            if c_cur.max() < best:
                best = c_cur.max()
                best_a = a.copy()

    return best_a, best


def make_palindromic(a, n):
    """Make a sequence palindromic."""
    half = a[:n // 2]
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget_s = float(budget_s)
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        parent = np.clip(parent, 0, 1000)
        if parent.size < 64 or parent.sum() < 0.01:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent_avail=%s" % (
        n, budget_s, parent is not None))

    global_best = np.inf
    global_a = None

    def check_and_update(a_cand, label):
        nonlocal global_best, global_a
        a_cand = normalize_sum(a_cand, len(a_cand))
        c_chk = fft_convolve_self(a_cand)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_cand.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: score=%.6f (elapsed %.0fs)" % (label, len(a_cand), score, elapsed))
        return score

    # Build initial candidates
    starts = []
    if parent is not None:
        p_norm = normalize_sum(parent, n)
        starts.append(("parent", p_norm))
        # Palindromic version
        pal = make_palindromic(p_norm, n)
        starts.append(("palindromic", normalize_sum(pal, n)))
        # Small perturbations
        for sc in [0.02, 0.05]:
            v = project_simplex(p_norm + rng.normal(0, sc * S / np.sqrt(n), n), S)
            starts.append(("perturb_%d" % int(sc * 100), v))

    # Structured starts
    tt = np.linspace(0, 1, n, endpoint=False)
    # Raised cosine window (smooth, zero at ends approximately)
    raised_cos = 0.5 * (1 + np.cos(2 * np.pi * tt))
    starts.append(("raised_cos", normalize_sum(np.maximum(raised_cos, 0.001), n)))
    # Gaussian
    gauss = np.exp(-4.0 * (tt - 0.5) ** 2)
    starts.append(("gaussian", normalize_sum(gauss, n)))
    # Sinc-squared
    sinc_sq = np.sinc(tt - 0.5) ** 2
    starts.append(("sinc_sq", normalize_sum(np.maximum(sinc_sq, 0.001), n)))
    # Random
    starts.append(("random", normalize_sum(rng.random(n) + 0.1, n)))
    # Two-peak
    two_peak = np.exp(-20 * (tt - 0.3) ** 2) + np.exp(-20 * (tt - 0.7) ** 2)
    starts.append(("two_peak", normalize_sum(two_peak, n)))

    # Evaluate all starts
    for name, arr in starts:
        c = fft_convolve_self(arr)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = arr.copy()
            print("  Start [%s]: score=%.6f" % (name, score))

    print("Best start: %.6f" % global_best)

    # Phase 1: FISTA on top candidates (40% time)
    print("\n=== Phase 1: FISTA ===")
    p1_budget = (deadline - time.monotonic()) * 0.40
    n_starts = len(starts)
    # Prioritize the best starts
    start_scores = []
    for name, arr in starts:
        c = fft_convolve_self(arr)
        start_scores.append(c.max())
    start_scores = np.array(start_scores)
    order = np.argsort(start_scores)
    times = p1_budget / max(1, n_starts)
    # Give more time to better starts
    weights = 1.0 / (start_scores[order] + 0.1)
    weights = weights / weights.sum()
    times = p1_budget * weights

    for rank, idx in enumerate(order):
        if time.monotonic() > deadline - (deadline - t_start) * 0.55 - 60:
            break
        name, arr = starts[idx]
        t_alloc = max(5.0, times[rank])
        if time.monotonic() + t_alloc > deadline - (deadline - t_start) * 0.50:
            t_alloc = deadline - (deadline - t_start) * 0.50 - time.monotonic()
            if t_alloc < 5:
                break
        print("  FISTA on [%s] with %.0fs" % (name, t_alloc))
        a_opt, score = fista_optimize(arr, n, S, t_alloc, rng,
                                       beta_init=5.0, beta_max=200.0,
                                       beta_ramp=500, use_backtrack=True)
        check_and_update(a_opt, "fista_" + name)

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: Active-set coordinate descent (30% time)
    print("\n=== Phase 2: Active-Set CD ===")
    if global_a is not None:
        p2_budget = (deadline - time.monotonic()) * 0.30
        if p2_budget > 15:
            # Split between CD and spectral
            t_cd = p2_budget * 0.7
            t_spec = p2_budget * 0.3
            a_cd, _ = active_set_cd(global_a, n, S, t_cd, rng, top_k=15, n_pairs_per_iter=40)
            check_and_update(a_cd, "active_cd")

            if global_a is not None and time.monotonic() < deadline - 30:
                a_spec, _ = spectral_flatten(global_a, n, S, t_spec, rng)
                check_and_update(a_spec, "spectral")

    print("After Phase 2: best=%.6f" % global_best)

    # Phase 3: Subgradient averaging (15% time)
    print("\n=== Phase 3: Subgradient Averaging ===")
    if global_a is not None:
        p3_budget = (deadline - time.monotonic()) * 0.15
        if p3_budget > 10:
            a_sub, _ = subgradient_average(global_a, n, S, p3_budget, rng)
            check_and_update(a_sub, "subgrad_avg")

    print("After Phase 3: best=%.6f" % global_best)

    # Phase 4: Final FISTA polish (10% time)
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        p4_budget = (deadline - time.monotonic()) * 0.10
        if p4_budget > 10:
            # Use a fresh FISTA with different parameters
            a_fin, _ = fista_optimize(global_a, n, S, p4_budget * 0.6, rng,
                                       beta_init=50.0, beta_max=500.0,
                                       beta_ramp=200, use_backtrack=True)
            check_and_update(a_fin, "final_fista")

            # Quick CD polish
            remaining = deadline - time.monotonic()
            if remaining > 5 and global_a is not None:
                a_cd2, _ = active_set_cd(global_a, n, S, remaining * 0.8, rng,
                                          top_k=5, n_pairs_per_iter=20)
                check_and_update(a_cd2, "final_cd")

    print("After Phase 4: best=%.6f" % global_best)

    # Phase 5: Try alternate length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Phase 5: Alternate Length ===")
        # Try n*1.5 or n//2
        for alt_n in [n * 2, n // 2]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            if not (1024 <= alt_n <= 65536):
                continue
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                nblocks = max(1, n // alt_n)
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alloc = remaining * 0.3
            a_alt, _ = fista_optimize(alt_start, alt_n, S_alt, t_alloc, rng,
                                       beta_init=5.0, beta_max=200.0, beta_ramp=300)
            check_and_update(a_alt, "alt_n_%d" % alt_n)

    # Final verification and fallback
    print("\n=== Final Verification ===")
    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: len=%d, score=%.6f, time=%.0fs" % (
        len(global_a), ver if np.isfinite(ver) else float('inf'),
        time.monotonic() - t_start))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]

    return final_list
