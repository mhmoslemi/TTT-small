def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_subgradient(y, c, k, power=1.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - c.min(), 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:

            w = w / wsum
    w_full = np.zeros(len(c))

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k, power=1.0):
    """Top-k subgradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - c.min() + 1e-30, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c):
    """Subgradient from the argmax lag only."""
    n = len(y)
    active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=8000.0, mode="lse", step_scale=1.0):
    """
    FISTA with adaptive LSE smoothing and noise injection.
    Returns (best_a, best_conv_max).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    stuck_count = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient computation with adaptive mode switching
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if stuck_count > 500:
                k = max(1, k // 4)
            grad = topk_subgradient(y, c, k, power=1.5)
        elif mode == "hybrid":
            if stuck_count > 200:
                k = min(64, 2 + stuck_count // 100)
                grad = topk_subgradient(y, c, k, power=1.0)
            else:
                grad = lse_gradient(y, c, beta)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive noise injection when stuck
        if stuck_count > 1000:
            step *= max(0.1, 1.0 - stuck_count / 5000.0)
            noise_scale = 0.004 * S / np.sqrt(n) * np.exp(-stuck_count / 4000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 500  # Partial reset to avoid too-frequent noise

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum restart
        if it % 4000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 40000.0)
            if decay > 0.03:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact/topk subgradient with adaptive k.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: start focused, widen when stuck
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        elif stuck < 10000:
            k = min(32, 8 + stuck // 500)
        else:
            k = min(128, 32 + stuck // 200)

        if k == 1:
            grad = exact_subgradient(y, c)
        else:
            grad = topk_subgradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: balance between aggressive and safe
        step = S / (gnorm * (1.0 + 0.008 * it))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when very stuck
        if stuck > 5000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def spectral_init(n, rng):
    """
    Generate a sequence with approximately flat power spectrum.
    Uses random phase with constant magnitude, then takes |IDFT|.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    # Random phase, constant magnitude
    phase = rng.uniform(0, 2 * np.pi, fsize)
    # Ensure Hermitian symmetry for real output
    spec = np.exp(1j * phase)
    # Make it conjugate symmetric
    for i in range(n // 2 + 1):
        spec[fsize - i] = np.conj(spec[i])
    spec[0] = 1.0
    if fsize % 2 == 0:
        spec[fsize // 2] = 1.0

    a = np.real(ifft(spec, fsize))[:n]
    a = np.abs(a)
    # Add small random component to avoid symmetries
    a = a + 0.1 * np.abs(rng.normal(0, 1, n))
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def diverse_init(n, rng, seed_offset=0):
    """Generate diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Cosine window
    a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    inits.append(a * (S / a.sum()))

    # Spectral init
    inits.append(spectral_init(n, rng))

    # Random positive
    a = np.abs(rng.normal(0, 1, n)) + 0.1
    inits.append(a * (S / a.sum()))

    # Exponential decay (two-sided)
    center = n // 2
    idx = np.arange(n)
    dist = np.minimum(np.abs(idx - center), n - np.abs(idx - center))
    a = np.exp(-0.5 * dist / (n * 0.1))
    inits.append(a * (S / a.sum()))

    # Bernoulli-like
    a = (rng.random(n) > 0.5).astype(float)
    a = a + 0.1 * rng.random(n)
    inits.append(a * (S / a.sum()))

    # Triangle
    a = 1.0 - 2.0 * np.abs(tt - 0.5)
    inits.append(a * (S / a.sum()))

    return inits


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)
