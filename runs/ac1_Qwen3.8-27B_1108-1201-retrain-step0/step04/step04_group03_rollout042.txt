"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Hybrid multiplicative repulsion + LP best-response with 
adaptive constraint selection, multi-length exploration, and mirror operations.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 1 << (2 * n - 1 + 128).bit_length()  # pad a bit
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)


def repulsion_update(f, lr, beta):
    """Multiplicative repulsion update."""
    n = len(f)
    c = fft_convolve_self(f)
    cm = c.max()
    # Softmax weights on the convolution peaks
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    # Compute r[j] = sum_k w[k] * f[k-j] via FFT
    # r = convolve(w, f_rev)[n-1 : 2n-1]
    f_rev = f[::-1]
    size = 1 << (2*n - 1 + n - 1 + 128).bit_length()  # need size >= 3n-2
    W = fft(w, size)
    F = fft(f_rev, size)
    conv = np.real(ifft(W * F))
    r = conv[n-1 : 2*n-1]
    # Multiplicative update
    f_max = f.max()
    if f_max < 1e-15:
        return f.copy()
    f_new = f * np.exp(-lr * r / f_max)
    f_new = np.maximum(f_new, 1e-15)
    S = f.sum()
    f_new = project_simplex(f_new, S)
    return f_new


def run_repulsion(f_init, S, max_time, rng, beta_init=5.0, beta_end=200.0, 
                  lr_init=0.5, lr_end=0.01, noise_every=200, noise_scale_init=0.01):
    """Run multiplicative repulsion iterations."""
    n = len(f_init)
    f = normalize_sum(f_init, S).copy()
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive parameters
        progress = min(1.0, it / max(1, int(max_time * 10)))  # rough progress
        beta = beta_init * (beta_end / beta_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress
        
        f = repulsion_update(f, lr, beta)
        
        # Check improvement
        if it % 50 == 0:
            c = fft_convolve_self(f)
            cur = c.max()
            if cur < best:
                best = cur
                best_f = f.copy()
        
        # Noise injection
        if it % noise_every == 0:
            noise_scale = noise_scale_init * (0.5 ** (it // (noise_every * 5)))
            if noise_scale > 1e-8:
                f = project_simplex(f + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
    
    # Final check
    c = fft_convolve_self(f)
    if c.max() < best:
        best = c.max()
        best_f = f.copy()
    
    return best_f, best


def solve_lp_topk(f, S, top_k=80, time_limit=15.0):
    """Solve LP with top-k constraints."""
    n = len(f)
    c = fft_convolve_self(f)
    M = c.max()
    
    # Select top-k most binding constraints
    # Also always include k=0, k=n-1, k=2n-2
    idx = np.argsort(c)[::-1][:top_k]
    # Ensure critical lags are included
    for critical in [0, n-1, 2*n-2]:
        if critical not in idx:
            idx = np.append(idx[:top_k-1], critical)
    
    # Build constraint matrix
    rows = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    # Solve LP
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        prob.solve(solver='SCS', verbose=False, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def lp_step(f, S, top_k=80):
    """One LP-based step with bisection for step size."""
    n = len(f)
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    
    g = solve_lp_topk(f, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old
    
    # Try different step sizes
    best_t = 0.0
    best_val = M_old
    for t in [1.0, 0.5, 0.3, 0.2, 0.1]:
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        if c_test.max() < best_val:
            best_val = c_test.max()
            best_t = t
    
    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), M_old


def mirror_improve(f, S):
    """Try mirror operations to improve."""
    n = len(f)
    f_rev = f[::-1].copy()
    c_orig = fft_convolve_self(f).max()
    
    candidates = [f.copy()]
    # Average with reverse
    f_avg = 0.5 * (f + f_rev)
    f_avg = project_simplex(f_avg, S)
    candidates.append(f_avg)
    # Try different mixtures
    for alpha in [0.3, 0.7]:
        f_mix = alpha * f + (1 - alpha) * f_rev
        f_mix = project_simplex(f_mix, S)
        candidates.append(f_mix)
    
    best_f = f.copy()
    best_val = c_orig
    for cand in candidates:
        c = fft_convolve_self(cand)
        if c.max() < best_val:
            best_val = c.max()
            best_f = cand.copy()
    
    return best_f, best_val


def run_combined(f_init, S, max_time, rng, use_lp=True, lp_every=100, repulsion_batch=200):

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining fast multiplicative repulsion updates,
LP best-response corrections with adaptive constraint selection, mirror exploitation,
and multi-length exploration. Uses adaptive beta schedules and cooling for hyperparameters.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_repulsion_vec(f, w):
    """Compute r[j] = sum_k w[k]*f[k-j] via FFT. w has length 2n-1."""
    n = len(f)
    f_rev = f[::-1]
    size = 1 << (3 * n - 2).bit_length()
    W = fft(w, size)
    F = fft(f_rev, size)
    conv = np.real(ifft(W * F))
    return conv[n - 1:2 * n - 1]


def repulsion_step(f, lr, beta):
    """One multiplicative repulsion update."""
    n = len(f)
    S = f.sum()
    c = fft_convolve_self(f)
    cm = c.max()
    if cm < 1e-30:
        return f.copy()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    r = compute_repulsion_vec(f, w)
    f_max = f.max()
    if f_max < 1e-15:
        return f.copy()
    scale = lr * r / f_max
    scale = np.clip(scale, -50.0, 50.0)
    f_new = f * np.exp(-scale)
    f_new = np.maximum(f_new, 1e-18)
    f_new = project_simplex(f_new, S)
    return f_new


def fista_lse_step(f, y, t_mom, beta):
    """One FISTA step with LSE smoothing of max(f*f)."""
    n = len(y)
    S = f.sum()
    c = fft_convolve_self(y)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f, y, t_mom
    w /= tot
    r = compute_repulsion_vec(y, w)
    step = 1.0 / (2.0 * beta)
    x_new = y - step * r
    x_new = project_simplex(x_new, S)
    t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
    y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - f)
    y_new = np.maximum(y_new, 0.0)
    return x_new, y_new, t_new


def run_repulsion_phase(f_init, S, max_time, rng, beta_start, beta_end,
                        lr_start, lr_end, noise_period=500, noise_scale=0.005,
                        use_fista=False, fista_ratio=0.5):
    """Run repulsion or FISTA iterations."""
    n = len(f_init)
    f = normalize_sum(f_init, S).copy()
    y = f.copy()
    t_mom = 1.0
    start = time.monotonic()

    best_f = f.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(10, max_time * 2))
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        if use_fista and (it % 2 == 0 or lr < 0.1):
            f, y, t_mom = fista_lse_step(f, y, t_mom, beta)

                f = repulsion_step(y, lr, beta)
        else:
            f = repulsion_step(f, lr, beta)
            y = f.copy()
            t_mom = 1.0

                best = cur

                    y = f.copy()
                    t_mom = 1.0

    return best_f, c.max()


def solve_lp_reduced(f, S, top_k=60, max_iters=3000):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    n = len(f)
    c = fft_convolve_self(f)
    M = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    # Ensure critical lags
    for crit in [0, n - 1, 2 * n - 2]:
        if crit not in top_idx:
            top_idx = np.concatenate([top_idx[:top_k - 1], [crit]])
    top_k = len(top_idx)
    A = np.zeros((top_k, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        A[i, js] = f[k - js]
    g_var = cp.Variable(n)
    cons = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), cons)
    try:
        prob.solve(solver='SCS', verbose=False, max_iters=max_iters)
    except Exception:
        return None
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            return None
        return g * (S / gs)
    return None


def lp_iteration(f, S, top_k=60):
    """One LP-based iteration with step size search."""
    n = len(f)
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    g = solve_lp_reduced(f, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old
    best_t = 0.0
    best_val = M_old
    for t in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
        ft = (1.0 - t) * f + t * g
        ft = np.maximum(ft, 0.0)
        ft = project_simplex(ft, S)
        ct = fft_convolve_self(ft)
        if ct.max() < best_val:
            best_val = ct.max()
            best_t = t
    if best_t > 0:
        fn = (1.0 - best_t) * f + best_t * g
        fn = np.maximum(fn, 0.0)
        fn = project_simplex(fn, S)
        return fn, best_val
    return f.copy(), M_old


def mirror_improve(f, S):
    """Try reflecting/averaging with reversed sequence."""
    f_rev = f[::-1].copy()
    candidates = [f.copy(), f_rev.copy(),
                  0.5 * (f + f_rev), 0.6 * f + 0.4 * f_rev,
                  0.4 * f + 0.6 * f_rev]
    best_f = f.copy()
    best_v = fft_convolve_self(f).max()
    for cand in candidates:
        cand = project_simplex(cand, S)
        cv = fft_convolve_self(cand).max()
        if cv < best_v:
            best_v = cv
            best_f = cand.copy()
    return best_f, best_v

def interpolate_to_length(a, new_n):
    """Interpolate sequence a to new length new_n."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

    return float(2.0 * n * c.max() / (a.sum() ** 2))
