    # g_j = 2 * sum_k w_k * (a_{j+k}*(j+k<n) + a_{j-k}*(j-k>=0))
    #     = 2 * sum_k w_k * (above)
    
    # This can be computed as: g = 2 * cross_correlation(a, w) (with appropriate definition)
    # Or using FFT: g = 2 * real(ifft(fft(a, N) * fft(w_full, N)))[:n]
    # where w_full is the weight vector (length 2n-1) and N >= 3n-2
    
    # Wait, let me verify with a simple example.
    # a = [a0, a1], n = 2
    # b = [a0^2, 2*a0*a1, a1^2] (linear convolution)
    # w = [w0, w1, w2]
    # f = w0*a0^2 + w1*2*a0*a1 + w2*a1^2
    # df/da0 = 2*w0*a0 + 2*w1*a1
    # df/da1 = 2*w1*a0 + 2*w2*a1
    # 
    # Now using the formula: g_j = 2 * sum_k w_k * (a_{j+k} + a_{j-k}) (with bounds)
    # g_0 = 2 * (w0*(a0) + w1*(a1 + a_{-1}=0) + w2*(a2=0 + a_{-2}=0))
    #     = 2 * (w0*a0 + w1*a1) ✓
    # g_1 = 2 * (w0*(a1 + a_{-1}=0) + w1*(a2=0 + a0) + w2*(a3=0 + a1))
    #     = 2 * (w0*a1 + w1*a0 + w2*a1) ✓
    
    # Great! So the formula is:
    # g_j = 2 * sum_k w_k * (a_{j+k} * (0<=j+k<n) + a_{j-k} * (0<=j-k<n))
    # 
    # This is 2 times the "auto-correlation-like" weighted sum.
    # Using FFT: g = 2 * real(ifft(fft(a, N) * fft(w_full, N)))[:n]
    # Wait, let me check: fft(a)*fft(w) gives the circular convolution of a and w (padded).
    # The circular convolution (a circ w)_j = sum_k a_k * w_{(j-k) mod N}
    # But what I want is sum_k w_k * (a_{j+k} + a_{j-k})
    # = sum_k w_k * a_{j+k} + sum_k w_k * a_{j-k}
    # = (w * a)_j + (w * a)_{-j} (in convolution notation)
    # Hmm, this is getting complicated. Let me just use the direct formula with FFT.
    
    # Actually, the simplest correct formula:
    # g = 2 * real(ifft(fft(a, N) * fft(w_full, N)))[:n]
    # where N >= n + (2n-1) - 1 = 3n-2
    # and w_full is the weight vector of length 2n-1.
    
    # Let me verify: the circular convolution (a * w_full) at index j (with zero padding to N)
    # = sum_{k=0}^{n-1} a[k] * w_full[(j-k) mod N]
    # Since N >= 3n-2 and w_full has support [0, 2n-2], and a has support [0, n-1],
    # (a * w_full)[j] = sum_{k=0}^{n-1} a[k] * w_full[j-k] (for j-k in [0, 2n-2])
    # = sum_{k: 0<=j-k<=2n-2} a[k] * w_full[j-k]
    # = sum_{k: j-(2n-2)<=k<=j} a[k] * w_full[j-k] (with a[k]=0 outside [0,n-1])
    # Let m = j-k, then k = j-m, and m ranges over [max(0, j-n+1), min(2n-2, j)]
    # = sum_m a[j-m] * w_full[m]
    
    # But what I want is g_j/2 = sum_k w_k * (a_{j+k} + a_{j-k})
    # = sum_k w_k * a_{j+k} + sum_k w_k * a_{j-k}
    # First term: sum_k w_k * a_{j+k} = sum_k' a[k'] * w_{j+k'-... } hmm
    # Let's substitute: in sum_k w_k * a_{j+k}, let k' = j+k, so k = k'-j
    # = sum_{k'} w_{k'-j} * a[k'] (where k'-j is in [0, 2n-2] and k' is in [0, n-1])
    # = sum_{k'=0}^{n-1} a[k'] * w_{k'-j}
    
    # Second term: sum_k w_k * a_{j-k}
    # Let k' = j-k, so k = j-k'
    # = sum_{k'} w_{j-k'} * a[k'] (where j-k' in [0, 2n-2] and k' in [0, n-1])
    # = sum_{k'=0}^{n-1} a[k'] * w_{j-k'}
    
    # So g_j/2 = sum_{k'=0}^{n-1} a[k'] * (w_{k'-j} + w_{j-k'})
    # This is the cross-correlation of a with the "symmetrized" weight w_sym[m] = w_m + w_{-m} (where w_{-m} = w_{-m} if -m is in range, else 0)
    
    # Hmm, this is getting complicated. Let me just use a different approach.
    
    # The cleanest way: use the fact that the gradient of sum_k p_k * b_k w.r.t. a
    # equals 2 * (p * a) truncated, where * denotes the appropriate convolution/correlation.
    
    # Actually, I realize the simplest approach is:
    # The function h(a) = sum_k p_k * (a*a)[k] can be written as a^T H a / 2 where H is a matrix.
    # H_{i,j} = p_{|i-j|} ... no, H_{i,j} = p_{i+j} (from the expansion sum_{i,j} a_i a_j p_{i+j})
    # Wait: h(a) = sum_k p_k * sum_{i+j=k} a_i a_j = sum_{i,j} a_i a_j p_{i+j}
    # So H_{i,j} = p_{i+j} and h(a) = a^T H a (no factor of 1/2 since we sum over all i,j)
    # The gradient is (H + H^T) a = 2H a (since H is symmetric: H_{i,j} = p_{i+j} = p_{j+i} = H_{j,i})
    # So grad = 2 * H * a where (Ha)_i = sum_j p_{i+j} * a_j
    
    # Now (Ha)_i = sum_{j=0}^{n-1} p_{i+j} * a_j
    # This is the cross-correlation of a and p (where p has length 2n-1, indices 0 to 2n-2)
    # (Ha)_i = sum_j a[j] * p[i+j]
    
    # To compute this with FFT:
    # Pad a to length N (N >= n + (2n-1) - 1 = 3n-2)
    # Pad p to length N
    # Compute c = ifft(fft(a_pad) * fft(p_pad)) → this gives circular convolution
    # c[i] = sum_j a[j] * p[(i-j) mod N]
    # What I want is sum_j a[j] * p[i+j]
    # = sum_j a[j] * p_[(i+j)]
    # Let me define p_rev[m] = p[-m mod (2n-1)] ... this is getting circular (pun intended).
    
    # OK let me just use a direct approach. For n up to 16384, I can compute the gradient
    # using the FFT trick:
    # grad = 2 * real(ifft(fft(a, N) * fft(p_rev, N)))[:n]
    # where p_rev[m] = p[2n-2-m] (reversal of p)
    # Let me verify: ifft(fft(a)*fft(p_rev)) = a * p_rev (linear convolution, since N is large enough)
    # (a * p_rev)[i] = sum_j a[j] * p_rev[i-j] = sum_j a[j] * p[2n-2-(i-j)] = sum_j a[j] * p[2n-2-i+j]
    # Hmm that's not quite what I want either.
    
    # You know what, let me just use the formula from the previous code which worked:
    # grad = 2 * real(ifft(fft(w, fsize) * conj(fft(a, fsize))))[:n]
    # where w is the weight vector on lags (length 2n-1) and fsize >= 2n-1 + n - 1 = 3n-2
    # Let me verify this gives the right answer.
    
    # ifft(fft(w) * conj(fft(a))) = cross_correlation(w, a) in the standard sense
    # cross_correlation(w, a)[i] = sum_j w[j] * conj(a[j-i]) = sum_j w[j] * a[j-i] (for real a)
    # = sum_j w[j] * a[j-i]
    # Let k = j-i: = sum_k w[k+i] * a[k]
    # = sum_k a[k] * w[k+i]
    # This is exactly (Ha)_i = sum_k a[k] * p[k+i] where p = w!
    
    # So: grad = 2 * real(ifft(fft(w_full, N) * conj(fft(a, N))))[:n]
    # where N >= len(w_full) + n - 1 = (2n-1) + n - 1 = 3n-2
    
    # Let me double check with the example:
    # a = [a0, a1], w = [w0, w1, w2]
    # cross_correlation(w, a)[0] = sum_k a[k] * w[k+0] = a0*w0 + a1*w1
    # cross_correlation(w, a)[1] = sum_k a[k] * w[k+1] = a0*w1 + a1*w2
    # grad = 2 * [a0*w0 + a1*w1, a0*w1 + a1*w2]
    # Expected: [2*(w0*a0 + w1*a1), 2*(w1*a0 + w2*a1)] ✓
    
    # So the correct formula is:
    # grad = 2 * real(ifft(fft(w, N) * conj(fft(a, N))))[:n]
    # where N >= 3n-2 (next power of 2 for efficiency)
    
    n = len(a)
    size_b = len(b)  # 2n-1
    fsize = 1 << (n + size_b - 1).bit_length()
    
    w_full = np.zeros(size_b)
    w_full[close] = weights
    A = fft(a, fsize)
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return grad


def softmax_weights(b, beta):
    """Compute softmax weights on the convolution lags."""
    max_b = b.max()
    scaled = beta * (b - max_b)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    exp_s = np.exp(scaled)
    total = exp_s.sum()
    if total < 1e-300:
        # All weights are essentially zero, use uniform on the max
        idx = np.argmax(b)
        w = np.zeros(len(b))
        w[idx] = 1.0
        return w
    return exp_s / total


def run_fista_smooth(a_init, S, max_time, rng, beta_start=50.0, beta_end=50000.0, 
                     mode='adaptive'):
    """FISTA with LSE-smoothed max, adaptive beta."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.full(len(a), S/len(a))
    
    n = len(a)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time
        
        # Adaptive beta
        if mode == 'adaptive':
            beta = beta_start * (beta_end / beta_start) ** (progress ** 0.5)
        else:
            beta = beta_start * (1 + 0.01 * it)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        if scaled.max() > 500:
            beta = 500.0 / max(1e-15, c.max() - cm if c.max() > cm else 1e-15)
            if beta < 1:
                beta = 1.0
            scaled = beta * (c - cm)
        e = np.exp(np.clip(scaled, -500, 0))
        tot = e.sum()
        w = e / tot
        
        fsize = 1 << (n + len(c) - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: for LSE-smoothed max, Lipschitz constant ≈ 2 * beta * ||y||_2^2 (rough)
        # Use a heuristic: step = c / (beta * max(c))
        step = 1.0 / (beta * cm * 0.5)
        
        # Adaptive: if stuck, reduce step and add noise
        if stall_count > 500:
            step *= 0.5
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stall_count = 0
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_exact_subgradient_descent(a_init, S, max_time, rng):
    """Exact subgradient descent with backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.full(len(a), S/len(a))
    
    n = len(a)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Adaptive tolerance for "close to max"
    tol_start = 0.05
    tol_end = 0.001
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        tol = tol_start + (tol_end - tol_start) * progress
        
        b = fft_convolve_self(a)
        cur_max = b.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Compute subgradient using softmax weights
        beta_sg = 1.0 / (tol * cur_max + 1e-10)
        w = softmax_weights(b, beta_sg)
        
        fsize = 1 << (n + len(b) - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step_start = 1.0 / (gnorm * 0.5)
        step = step_start
        improved = False
        for _ in range(20):
            a_trial = project_simplex(a - step * grad, S)
            b_trial = fft_convolve_self(a_trial)
            if b_trial.max() < cur_max - 1e-10:
                a = a_trial
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try a small random perturbation
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            if it % 100 == 0:
                # Occasionally try a larger perturbation
                noise_large = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a_trial = project_simplex(a + noise_large, S)
                b_trial = fft_convolve_self(a_trial)
                if b_trial.max() < fft_convolve_self(a).max():
                    a = a_trial
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def fourier_crossover(a1, a2, S, rng):
    """Create offspring using Fourier-domain crossover."""
    n = len(a1)
    fsize = 1 << (2 * n - 1).bit_length()
    
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    
    # Mix magnitudes and phases
    alpha = rng.uniform(0.2, 0.8)
    
    # Child 1: mag from a1, phase from a2 (and vice versa for child 2)
    child1_spec = np.abs(A1) ** alpha * np.abs(A2) ** (1-alpha) * np.exp(1j * (alpha * np.angle(A1) + (1-alpha) * np.angle(A2)))
    child2_spec = np.abs(A1) ** (1-alpha) * np.abs(A2) ** alpha * np.exp(1j * ((1-alpha) * np.angle(A1) + alpha * np.angle(A2)))
    
    child1 = np.real(ifft(child1_spec))[:n]
    child2 = np.real(ifft(child2_spec))[:n]
    
    # Project to feasible set
    child1 = project_simplex(child1, S)
    child2 = project_simplex(child2, S)
    
    return child1, child2


def coordinate_descent_polish(a_init, S, max_time, rng):
    """Coordinate descent: try adjusting pairs of coordinates."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    a = a * (S / s) if s > 1e-15 else np.full(len(a), S/len(a))
    
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # Precompute some structure
    while time.monotonic() - start < max_time:
        b = fft_convolve_self(a)
        cur_max = b.max()
        max_idx = np.argmax(b)
        
        # Find coordinates that contribute most to the peak
        # b[max_idx] = sum_i a[i] * a[i + max_idx] (approximately)
        # The "contributors" are positions i and i+max_idx where a[i]*a[i+max_idx] is large
        
        k = max_idx
        # Get the "profile" that contributes to b[k]
        if k == 0:
            contributors = np.arange(n)
            contrib_vals = a * a
        else:
            n_overlap = n - k
            contributors = np.arange(n_overlap)
            contrib_vals = a[:n_overlap] * a[k:k+n_overlap]
        
        # Find the top contributors
        top_k = min(64, len(contributors))
        top_idx = np.argsort(contrib_vals)[::-1][:top_k]
        
        # Try to reduce the peak by moving mass from high contributors
        # to low contributors
        moved = False
        for _ in range(10):
            if time.monotonic() - start > max_time:
                break
            
            # Pick a random pair from top contributors
            i = contributors[top_idx[rng.integers(top_k)]]
            # Pick a destination (random position not in top contributors)
            dest = rng.integers(n)
            if dest == i:
                continue
            
            # Try moving a small amount from i to dest
            delta = a[i] * rng.uniform(0.01, 0.1)
            if delta < 1e-10:
                continue
            
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[dest] += delta
            
            b_trial = fft_convolve_self(a_trial)
            if b_trial.max() < cur_max - 1e-12:
                a = a_trial
                cur_max = b_trial.max()
                moved = True
                if cur_max < best:
                    best = cur_max
                    best_a = a.copy()
        
        if not moved:
            # Try a random perturbation
            if rng.random() < 0.1:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a_trial = project_simplex(a + noise, S)
                b_trial = fft_convolve_self(a_trial)
                if b_trial.max() < cur_max:
                    a = a_trial
                    cur_max = b_trial.max()
                    if cur_max < best:
                        best = cur_max
                        best_a = a.copy()
            else:
                # Break out if no progress for a while
                break
    
    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
    1. FISTA with adaptive LSE smoothing for coarse optimization
    2. Exact subgradient descent with backtracking for fine tuning
    3. Fourier crossover for diversification
    4. Coordinate descent polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print("Starting hybrid optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass
    
    if given is None:
        given = np.ones(8192)
    
    n = len(given)
    S = np.sqrt(2.0 * n)
    
    # Normalize given
    given_norm = given * (S / given.sum())
    
    global_best = np.inf
    global_a = given_norm.copy()
    
    # Phase 1: FISTA with adaptive LSE (40% of time)
    time_phase1 = (deadline - time.monotonic()) * 0.4
    print("\n=== Phase 1: FISTA with adaptive LSE (%.0fs) ===" % time_phase1)
    
    a1, val1 = run_fista_smooth(given_norm, S, time_phase1, rng,
                                  beta_start=50.0, beta_end=50000.0)
    obj1 = 2.0 * n * val1 / S**2
    print("  Phase 1 result: max_b=%.6f, obj=%.6f" % (val1, obj1))
    
    if val1 < fft_convolve_self(global_a).max():
        global_a = a1.copy()
        global_best = obj1
    
    # Phase 2: Generate diverse candidates via Fourier crossover (20% of time)
    time_phase2 = (deadline - time.monotonic()) * 0.2
    print("\n=== Phase 2: Fourier crossover diversification (%.0fs) ===" % time_phase2)
    
    # Create a few diverse candidates
    candidates = [global_a.copy()]
    
    # Add perturbed versions
    for i in range(3):
        noise_scale = 0.02 * S / np.sqrt(n)
        perturbed = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
        candidates.append(perturbed)
    
    # Fourier crossover between best and perturbed
    if len(candidates) >= 2:
        for i in range(min(3, len(candidates) - 1)):
            c1, c2 = fourier_crossover(candidates[0], candidates[i+1], S, rng)
            candidates.append(c1)
            candidates.append(c2)
    
    # Quick FISTA on each candidate
    time_per_cand = time_phase2 / len(candidates)
    for i, cand in enumerate(candidates):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.3:
            break
        a_c, val_c = run_fista_smooth(cand, S, min(time_per_cand, time.monotonic() and (deadline - time.monotonic()) * 0.3), rng,
                                       beta_start=100.0, beta_end=30000.0)
        obj_c = 2.0 * n * val_c / S**2
        if val_c < fft_convolve_self(global_a).max():
            global_a = a_c.copy()
            global_best = obj_c
            print("  Candidate %d: max_b=%.6f, obj=%.6f" % (i, val_c, obj_c))
    
    # Phase 3: Exact subgradient descent (30% of time)
    time_phase3 = (deadline - time.monotonic()) * 0.45
    print("\n=== Phase 3: Exact subgradient descent (%.0fs) ===" % time_phase3)
    
    a3, val3 = run_exact_subgradient_descent(global_a, S, time_phase3, rng)
    obj3 = 2.0 * n * val3 / S**2
    print("  Phase 3 result: max_b=%.6f, obj=%.6f" % (val3, obj3))
    
    if val3 < fft_convolve_self(global_a).max():
        global_a = a3.copy()
        global_best = obj3
    
    # Phase 4: Coordinate descent polishing (15% of time)
    time_phase4 = deadline - time.monotonic() - 10
    if time_phase4 > 10:
        print("\n=== Phase 4: Coordinate descent polishing (%.0fs) ===" % time_phase4)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid optimization strategy:
1. FISTA with adaptive LSE smoothing (coarse-to-fine beta schedule)
2. Fourier-domain crossover for diversification
3. Exact subgradient descent with backtracking line search (fine tuning)
4. Coordinate descent polishing on peak-contributing coordinates
Key improvements over previous: better step-size calibration, exact subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_weight(c, beta):
    """Compute LSE-smoothed max and the softmax weight vector."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max() if (c - cm).max() > 0 else 1e-15)
        scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return cm, w
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return lse_val, w


def compute_gradient(a, w):
    """Compute gradient of weighted sum of convolution lags.
    grad = 2 * cross_correlation(w, a) truncated to len(a)
    where (cross_corr[w,a])[i] = sum_j a[j] * w[i+j]
    """
    n = len(a)
    fsize = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_lse(a_init, S, max_time, rng, beta_start=30.0, beta_end=100000.0):
    """FISTA with LSE-smoothed max and adaptive beta."""
    a = normalize_to_sum(a_init, S)
    n = len(a)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adaptive beta: geometric interpolation
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        lse_val, w = lse_value_and_weight(c, beta)
        grad = compute_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: Lipschitz of LSE-smoothed max is ~ 2*beta*||y||_2^2
        # Use: step = 1 / (2 * beta * ||y||_2^2) * safety
        lip_est = 2.0 * beta * np.dot(y, y)

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)

def fourier_crossover(a1, a2, S, rng):
    """Fourier-domain crossover producing two children."""
    n = len(a1)
    fsize = 1 << (2 * n - 1).bit_length()
    
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    
    alpha = rng.uniform(0.3, 0.7)
    
    # Child 1: blend magnitudes, blend phases
    mag1 = np.abs(A1) ** alpha * np.abs(A2) ** (1.0 - alpha)
    phase1 = alpha * np.angle(A1) + (1.0 - alpha) * np.angle(A2)
    child1_spec = mag1 * np.exp(1j * phase1)
    
    # Child 2: complementary blend
    mag2 = np.abs(A1) ** (1.0 - alpha) * np.abs(A2) ** alpha
    phase2 = (1.0 - alpha) * np.angle(A1) + alpha * np.angle(A2)
    child2_spec = mag2 * np.exp(1j * phase2)
    
    child1 = np.real(ifft(child1_spec))[:n]
    child2 = np.real(ifft(child2_spec))[:n]
    
    child1 = project_simplex(child1, S)
    child2 = project_simplex(child2, S)
    
    return child1, child2

            global_best = obj_c
            print("  Crossover candidate %d: obj=%.6f" % (i, obj_c))
    
    # Phase 3: Exact subgradient descent
    time_remaining = deadline - time.monotonic()
    t3 = time_remaining * 0.35
    print("\n=== Phase 3: Exact subgradient (%.0fs) ===" % t3)
    
    a3, val3 = run_exact_subgradient(global_a, S, t3, rng, top_k_start=2, top_k_end=16)
    obj3 = 2.0 * n * val3 / S**2
    print("  Phase 3: max_b=%.8f, obj=%.6f" % (val3, obj3))
    if val3 < fft_convolve_self(global_a).max():
        global_a = a3.copy()
        global_best = obj3
    
    # Phase 4: High-beta FISTA polishing
    time_remaining = deadline - time.monotonic()
    t4 = time_remaining * 0.30
    if t4 > 10:
        print("\n=== Phase 4: High-beta polish (%.0fs) ===" % t4)
        a4, val4 = run_fista_high_beta(global_a, S, t4, rng)
        obj4 = 2.0 * n * val4 / S**2
        print("  Phase 4: max_b=%.8f, obj=%.6f" % (val4, obj4))
        if val4 < fft_convolve_self(global_a).max():
            global_a = a4.copy()
            global_best = obj4
    
    # Phase 5: Coordinate descent
    time_remaining = deadline - time.monotonic()
    t5 = time_remaining * 0.50
    if t5 > 10:
        print("\n=== Phase 5: Coordinate descent (%.0fs) ===" % t5)
        a5, val5 = run_peak_coordinate_descent(global_a, S, t5, rng)
        obj5 = 2.0 * n * val5 / S**2
        print("  Phase 5: max_b=%.8f, obj=%.6f" % (val5, obj5))
        if val5 < fft_convolve_self(global_a).max():
            global_a = a5.copy()
            global_best = obj5
    
    # Final high-beta polish with remaining time
    time_remaining = deadline - time.monotonic()
    if time_remaining > 15:
        print("\n=== Final polish (%.0fs) ===" % time_remaining)
        af, valf = run_fista_high_beta(global_a, S, time_remaining * 0.7, rng)
        objf = 2.0 * n * valf / S**2
        if valf < fft_convolve_self(global_a).max():
            global_a = af.copy()
            global_best = objf
        
        time_remaining = deadline - time.monotonic()
        if time_remaining > 10:
            asg, vals = run_exact_subgradient(global_a, S, time_remaining * 0.8, rng,
                                               top_k_start=1, top_k_end=4)
            objs = 2.0 * n * vals / S**2
            if vals < fft_convolve_self(global_a).max():
                global_a = asg.copy()
                global_best = objs
    
    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192) * (np.sqrt(2 * 8192) / 8192)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nFinal verification: %.6f" % ver)
    print("Final objective (internal): %.6f, n=%d" % (global_best, n))
    
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Fallback.")
        global_a = np.ones(8192) * (np.sqrt(2 * 8192) / 8192)
        final_list = [float(x) for x in global_a]
    
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
