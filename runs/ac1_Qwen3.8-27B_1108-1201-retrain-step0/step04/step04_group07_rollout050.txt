def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:

    # Water-filling algorithm
    mu = np.sort(v)[::-1]

    rho = np.where(mu > (cumsum - S) / np.arange(1, n+1))[0]

    rho = rho[-1] + 1
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    # Compute convolution
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return float(2 * n * c.max() / (S * S))

    n = len(a)
    size = 2*n - 1

    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    mu = np.sort(v)[::-1]
    cumsum_mu = np.cumsum(mu)
    rho_candidates = np.where(mu > (cumsum_mu - S) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        return np.zeros(n)
    rho = rho_candidates[-1] + 1
    theta = (cumsum_mu[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return c


def grad_at_peak(a, c, top_k=10):
    """Compute subgradient of max(conv(a,a)) using top_k lags."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        partners = k - js
        grad[js] += a[js] + a[partners]
    return grad


def spectral_reshape(a, N, strength=0.1):
    """Compress |FFT(a)|^2 toward uniform to reduce spectral peaks."""
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    p_mean = P.mean()
    if p_mean < 1e-30:
        return a
    # Compress: P_new = P^(1-strength) * p_mean^strength (geometric mean with flat)
    # This reduces peaks and boosts valleys
    P_new = np.power(P + 1e-30, 1.0 - strength) * np.power(p_mean, strength)
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return a_new

        # Project

        # Update y for next iteration

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    mu = np.sort(v)[::-1]
    cumsum_mu = np.cumsum(mu)
    rho_candidates = np.where(mu > (cumsum_mu - S) / np.arange(1, n + 1))[0]
    if len(rho_candidates) == 0:
        return np.zeros(n)
    rho = rho_candidates[-1] + 1
    theta = (cumsum_mu[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
