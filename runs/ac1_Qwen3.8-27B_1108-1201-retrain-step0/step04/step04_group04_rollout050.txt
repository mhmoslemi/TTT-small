- Spectral method targets A^2 (self-convolution spectrum) not |A|^2 (autocorrelation)
- Coordinate descent uses sparse quadratic structure for efficiency
- Simulated annealing phase for escaping local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def grad_conv_lag(a, k):
    """Gradient of (a*a)[k] with respect to a. O(n) time."""
    n = len(a)
    g = np.zeros(n)
    # a[k-j] contribution: j from max(0,k-n+1) to min(k, n-1)
    lo1 = max(0, k - n + 1)
    hi1 = min(k, n - 1)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
    # a[j-k] contribution: j from max(k,0) to min(k+n-1, n-1)
    lo2 = max(k, 0)
    hi2 = min(k + n - 1, n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    return g


def grad_conv_weighted(a, c, beta, n_active=5):
    """Weighted subgradient using softmax over top lags."""
    n = len(a)
    cm = c.max()
    # Use top lags weighted by exp(beta * (c[k] - cm))
    top_idx = np.argsort(c)[::-1][:n_active]
    weights = np.exp(beta * (c[top_idx] - cm))
    weights /= weights.sum()
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        g += weights[idx] * grad_conv_lag(a, int(top_idx[idx]))
    return g


def run_nesterov_subgrad(a_init, n, S, max_time, rng, initial_step=None,
                         soft_beta=5.0, soft_every=2000, soft_active=5):
    """Nesterov-accelerated subgradient descent with exact/soft subgradients."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Estimate initial step size
    c = fft_convolve_self(y)
    g = grad_conv_lag(y, int(np.argmax(c)))
    gnorm = np.linalg.norm(g)
    if initial_step is None:
        step_size = S / (gnorm * np.sqrt(n)) if gnorm > 1e-14 else 0.01
    else:
        step_size = initial_step

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Adaptive step: decrease if no improvement
        step = step_size / (1.0 + 0.001 * (it - last_improve))
        step = max(step, 1e-8)

        # Choose gradient type
        if it % soft_every == 0 and it > soft_every:
            g = grad_conv_weighted(y, c, soft_beta, soft_active)
        else:
            k_star = int(np.argmax(c))
            g = grad_conv_lag(y, k_star)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Nesterov update
        x_new = project_simplex(y - step * g / gnorm * S, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * (S / s_y)
        a = x_new
        t = t_new

        # Periodic noise injection (annealing)
        if it % 5000 == 0:
            elapsed_frac = (time.monotonic() - start) / max_time
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 - elapsed_frac)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_peak_reduce(a_init, n, S, max_time, rng, n_inner=None):
    """GS-style peak reduction on A^2 (self-convolution spectrum)."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    if n_inner is None:
        n_inner = 500

    while time.monotonic() - start < max_time and it < n_inner:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        A = fft(a_pad)
        A_sq = A * A
        c = np.real(ifft(A_sq))[:2 * n - 1]

        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Peak reduction: clip c and convert back
        # Adaptive clipping level
        progress = min(1.0, it / max(n_inner, 1))
        clip_level = cur * (1.0 - 0.05 * (1.0 - progress))
        c_clipped = np.minimum(c, clip_level)
        # Also ensure non-negativity (convolution of non-neg seq should be non-neg)
        c_clipped = np.maximum(c_clipped, 0.0)

        # Convert back to A^2
        A_sq_new = fft(np.concatenate([c_clipped, np.zeros(fsize - len(c_clipped))]))
        # Take square root (preserving phase)
        phase = A_sq_new / np.maximum(np.abs(A_sq_new), 1e-30)
        mag = np.sqrt(np.abs(A_sq_new))
        A_new = phase * mag

        # Get new a
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Momentum blending
        blend = 0.3 + 0.4 * progress
        a = blend * a_new + (1 - blend) * a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def focused_coord_descent(a_init, n, S, max_time, rng, n_top_lags=20):
    """Coordinate descent focusing on pairs that reduce near-max lags."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Determine which lags are "hot" (near max)
        c_max = c.max()
        threshold = c_max * 0.95
        hot_lags = np.where(c >= threshold)[0]
        if len(hot_lags) == 0:
            hot_lags = np.array([int(np.argmax(c))])

        # Try a set of pairs
        n_try = min(30 + it // 5, 100)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break

            # Choose pair: bias towards coordinates that affect hot lags
            if rng.random() < 0.6:
                # Local pair
                j = rng.integers(0, n - 1)
                m = (j + rng.integers(1, min(10, n - 1))) % n
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m:
                continue

            # Compute d[k] for hot lags only (for speed)
            # Full d for all k would be O(n), but we only need hot lags for search
            # However, we need the full max, so let's compute d for all k
            # but use the structure: d[k] = a[k-j]+a[j-k]-a[k-m]-a[m-k]

            d = np.zeros(2 * n - 1)
            # a[k-j]: valid for k-j in [0,n-1] → k in [j, j+n-1]
            lo = j
            hi = min(j + n - 1, 2 * n - 2)
            if lo <= hi:
                d[lo:hi + 1] += a[0:hi - lo + 1]
            # a[j-k]: valid for j-k in [0,n-1] → k in [j-n+1, j]
            lo2 = max(0, j - n + 1)
            hi2 = j
            if lo2 <= hi2:
                d[lo2:hi2 + 1] += a[hi2 - lo2 + 1:0:-1] if hi2 > lo2 else a[0:1]
                # Fix: a[j-k] for k from lo2 to hi2
                # j-k ranges from j-lo2 to j-hi2
                d[lo2:hi2 + 1] = 0  # reset (we double-added above)
                # Redo properly
            # Let me just do it the straightforward way
            d = np.zeros(2 * n - 1)
            size = 2 * n - 1
            # a[k-j] for k in [0, size-1]: valid when 0<=k-j<n → j<=k<=j+n-1
            k_lo = max(0, j)
            k_hi = min(size - 1, j + n - 1)
            if k_lo <= k_hi:
                d[k_lo:k_hi + 1] += a[0:k_hi - k_lo + 1]
            # a[j-k] for k in [0, size-1]: valid when 0<=j-k<n → j-n+1<=k<=j
            k_lo2 = max(0, j - n + 1)
            k_hi2 = min(size - 1, j)
            if k_lo2 <= k_hi2:
                # j-k for k=k_lo2..k_hi2 gives values j-k_lo2 down to j-k_hi2
                d[k_lo2:k_hi2 + 1] += a[j - k_hi2:j - k_lo2 + 1]
            # -a[k-m]
            k_lo3 = max(0, m)
            k_hi3 = min(size - 1, m + n - 1)
            if k_lo3 <= k_hi3:
                d[k_lo3:k_hi3 + 1] -= a[0:k_hi3 - k_lo3 + 1]
            # -a[m-k]
            k_lo4 = max(0, m - n + 1)
            k_hi4 = min(size - 1, m)
            if k_lo4 <= k_hi4:
                d[k_lo4:k_hi4 + 1] -= a[m - k_hi4:m - k_lo4 + 1]

            # Quadratic term q: non-zero at k=2j (val=1), k=j+m (val=-2), k=2m (val=1)
            q_positions = {}
            if 0 <= 2 * j < size:
                q_positions[2 * j] = q_positions.get(2 * j, 0) + 1
            if 0 <= j + m < size:
                q_positions[j + m] = q_positions.get(j + m, 0) - 2
            if 0 <= 2 * m < size:
                q_positions[2 * m] = q_positions.get(2 * m, 0) + 1

            # Find optimal delta by golden section search
            lo_d = -a[j]
            hi_d = a[m]
            if lo_d >= hi_d:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            def eval_delta(dlt):
                vals = c + dlt * d
                for kp, qv in q_positions.items():
                    vals[kp] += dlt * dlt * qv
                return vals.max()

            fc = eval_delta(c_g)
            fd = eval_delta(d_g)
            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = eval_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = eval_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = max(lo_d, min(hi_d, delta))

            if abs(delta) < 1e-12:
                continue

            # Apply and check
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue

            c_trial = c + delta * d
            for kp, qv in q_positions.items():
                c_trial[kp] += delta * delta * qv

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-10:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Periodic full recompute
        if it % 10 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        # Noise if stuck
        if not improved and it > 3:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def simulated_annealing(a_init, n, S, max_time, rng):
    """Simulated annealing with structured perturbations."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    T = 0.01 * S / np.sqrt(n)  # initial temperature

    while time.monotonic() - start < max_time:
        it += 1
        # Cool
        T *= 0.9995
        if T < 1e-10:
            T = 0.001 * S / np.sqrt(n)  # reheat

        # Generate perturbation
        ptype = rng.random()
        a_trial = a.copy()

        if ptype < 0.3:
            # Random pair transfer
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j != m:
                delta = rng.normal(0, T)
                delta = max(-a[j], min(a[m], delta))
                a_trial[j] += delta
                a_trial[m] -= delta
        elif ptype < 0.6:
            # Block perturbation
            block_size = rng.integers(4, min(64, n // 4))
            pos = rng.integers(0, n - block_size)
            a_trial[pos:pos + block_size] += rng.normal(0, T, block_size)
        elif ptype < 0.8:
            # Spectral perturbation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            A_noise = A + rng.normal(0, T * 0.1, fsize) + 1j * rng.normal(0, T * 0.1, fsize)
            # Enforce conjugate symmetry
            A_noise[0] = np.real(A_noise[0])
            for k in range(1, fsize // 2 + 1):
                A_noise[fsize - k] = np.conj(A_noise[k])
            a_trial = np.real(ifft(A_noise))[:n]
        else:
            # Smooth perturbation
            a_trial += rng.normal(0, T * 0.5, n)
            # Low-pass filter
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a_trial, fsize)
            freqs = np.fft.fftfreq(fsize)
            A *= np.exp(-5.0 * (freqs / 0.2) ** 2)
            a_trial = np.real(ifft(A))[:n]

        # Project
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial * (S / s)

        # Evaluate
        c_trial = fft_convolve_self(a_trial)
        score_trial = c_trial.max()

        # Accept/reject
        if score_trial < best:
            a = a_trial
            best = score_trial
            best_a = a.copy()
        else:
            delta_score = score_trial - best
            if delta_score < T * 0.1:
                accept_prob = np.exp(-delta_score / max(T * 0.01, 1e-12))
                if rng.random() < accept_prob:
                    a = a_trial

    return best_a, best


def generate_structured_starts(n, rng, S):
    """Generate diverse structured starting sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    starts.append(np.ones(n) * S / n)

    # Cosine window
    starts.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.001))

    # Gaussian
    starts.append(np.exp(-4.0 * (tt - 0.5) ** 2))

    # Sinc-like
    starts.append(np.maximum(np.sinc(tt - 0.5) ** 2, 0.001))

    # Bimodal (two Gaussians)
    starts.append(np.exp(-30.0 * (tt - 0.25) ** 2) + np.exp(-30.0 * (tt - 0.75) ** 2))

    # Anti-symmetric perturbation of uniform
    anti = 1.0 + 0.3 * np.sin(4 * np.pi * tt)
    starts.append(np.maximum(anti, 0.01))

    # Higher frequency oscillation
    anti2 = 1.0 + 0.2 * np.cos(16 * np.pi * tt) * np.cos(2 * np.pi * tt)
    starts.append(np.maximum(anti2, 0.01))

    # Random
    starts.append(rng.random(n) + 0.1)

    # Triangle
    starts.append(1.0 - np.abs(2.0 * tt - 1.0))

    # Waterfall
    starts.append(np.exp(-3.0 * tt))

    # Normalize all
    normed = []
    for s in starts:
        s = np.clip(s, 0, 1000)
        ss = s.sum()
        if ss > 1e-15:
            normed.append(s * (S / ss))
        else:
            normed.append(np.ones(n) * S / n)
    return normed


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 512 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes(len=%d)" % len(parent) if parent is not None else "no"))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] score=%.6f (elapsed %.0fs)" % (label, score, elapsed))
        return score

    # === Phase 1: Quick evaluation of multiple starts (10% time) ===
    print("\n=== Phase 1: Multi-start evaluation ===")
    p1_budget = (deadline - time.monotonic()) * 0.10

    starts = []
    labels = []
    if parent is not None:
        starts.append(parent)
        labels.append("parent")
        # Palindromic version
        half = parent[:n // 2]
        if n % 2 == 0:
            pal = np.concatenate([half, half[::-1]])
        else:
            pal = np.concatenate([half, parent[n // 2:n // 2 + 1], half[::-1]])
        starts.append(pal)
        labels.append("palindromic")

    for i, s in enumerate(generate_structured_starts(n, rng, S)):
        starts.append(s)
        labels.append("init_%d" % i)

    # Quick FISTA on each start
    n_starts = len(starts)
    time_per_start = p1_budget / n_starts
    for i in range(n_starts):
        if time.monotonic() > deadline - (deadline - t_start) * 0.8 - 30:
            break
        a_opt, score = run_nesterov_subgrad(starts[i], n, S, time_per_start, rng)
        update_best(a_opt, "p1_" + labels[i])

    print("After Phase 1: best=%.6f" % global_best)

    # === Phase 2: Nesterov subgradient from best (30% time) ===
    print("\n=== Phase 2: Nesterov subgradient polish ===")
    if global_a is not None:
        p2_budget = (deadline - time.monotonic()) * 0.30
        # Run two variants: one with exact, one with soft
        t2a = p2_budget * 0.5
        t2b = p2_budget * 0.5
        a2a, _ = run_nesterov_subgrad(global_a, n, S, t2a, rng,
                                       soft_every=10000, soft_active=1)
        update_best(a2a, "p2_exact")
        a2b, _ = run_nesterov_subgrad(global_a, n, S, t2b, rng,
                                       soft_beta=10.0, soft_every=500, soft_active=5)
        update_best(a2b, "p2_soft")

    print("After Phase 2: best=%.6f" % global_best)

    # === Phase 3: Spectral peak reduction (20% time) ===
    print("\n=== Phase 3: Spectral peak reduction ===")
    if global_a is not None:
        p3_budget = (deadline - time.monotonic()) * 0.20
        a3, _ = spectral_peak_reduce(global_a, n, S, p3_budget, rng)
        update_best(a3, "p3_spectral")
        # Run again from the result
        if p3_budget > 60:
            a3b, _ = spectral_peak_reduce(a3, n, S, p3_budget * 0.5, rng, n_inner=300)
            update_best(a3b, "p3_spectral2")

    print("After Phase 3: best=%.6f" % global_best)

    # === Phase 4: Focused coordinate descent (25% time) ===
    print("\n=== Phase 4: Coordinate descent ===")
    if global_a is not None:
        p4_budget = (deadline - time.monotonic()) * 0.25
        a4, _ = focused_coord_descent(global_a, n, S, p4_budget, rng)
        update_best(a4, "p4_coord")

    print("After Phase 4: best=%.6f" % global_best)

    # === Phase 5: Simulated annealing for global escape (10% time) ===
    print("\n=== Phase 5: Simulated annealing ===")
    if global_a is not None:
        p5_budget = (deadline - time.monotonic()) * 0.10
        if p5_budget > 20:
            a5, _ = simulated_annealing(global_a, n, S, p5_budget, rng)
            update_best(a5, "p5_anneal")

    print("After Phase 5: best=%.6f" % global_best)

    # === Phase 6: Final polish with remaining time ===
    print("\n=== Phase 6: Final polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Alternate between subgradient and spectral
            n_rounds = 3
            t_per = remaining / n_rounds
            for r in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                a_r, _ = run_nesterov_subgrad(global_a, n, S, t_per * 0.5, rng,
                                               soft_every=500, soft_active=3,
                                               soft_beta=8.0)
                update_best(a_r, "p6_r%d_sg" % r)
                a_r2, _ = spectral_peak_reduce(global_a, n, S, t_per * 0.4, rng, n_inner=100)
                update_best(a_r2, "p6_r%d_sp" % r)
                a_r3, _ = focused_coord_descent(global_a, n, S, t_per * 0.1, rng)
                update_best(a_r3, "p6_r%d_cd" % r)

    # === Final verification ===
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification: inf - using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, total_time=%.0fs" % (len(global_a), time.monotonic() - t_start))
    return final_list
