"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses a combination of:
1. Flat spectrum iteration (Gerchberg-Saxton variant)
2. Coordinate descent with exact 1D optimization
3. LSE-smoothed gradient descent with adaptive beta
4. Multi-start from diverse initializations including the previous best

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        theta = (v.sum() - S) / n
        return np.maximum(v - theta, 0.0)
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)

def flat_spectrum_iter(a, n, S, n_iter, alpha, rng):
    """
    Gerchberg-Saxton style iteration to flatten the autoconvolution spectrum.
    alpha controls how aggressively we flatten |A(omega)|.
    """
    a = a.copy()
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    
    best_obj = compute_objective(a)
    best_a = a.copy()
    start = time.monotonic()
    
    for it in range(n_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten the magnitude: |A_new| = mag^alpha * (mean_mag)^(1-alpha)
        mean_mag = mag.mean()
        if mean_mag < 1e-15:
            break
        target_mag = mag ** alpha * (mean_mag ** (1 - alpha))
        
        # But keep A[0] positive (it's the sum)
        # Blend
        A_new = target_mag * np.exp(1j * phase)
        # Ensure A[0] is real and positive
        A_new[0] = np.abs(A_new[0])
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        
        # Rescale
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        
        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 1000 == 0:
            print("  FS iter %d: obj=%.6f" % (it, best_obj))
    
    return best_a, best_obj

def coordinate_descent(a, n, S, max_passes, rng):
    """
    Cyclic coordinate descent. For each coordinate, try several values
    and pick the best.
    """
    a = a.copy()
    best_obj = compute_objective(a)
    best_a = a.copy()
    start = time.monotonic()
    
    b = fft_convolve_self(a)
    
    for pase in range(max_passes):
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:
            # Profile: p[k] = a[k-i] if 0 <= k-i < n, else 0
            # p is non-zero for k in [i, i+n-1]
            lo = i
            hi = i + n - 1
            # b[k] = a[i]*a[k-i] + d[k] where d[k] = b[k] - a[i]*a[k-i]
            current_val = a[i]
            if current_val < 1e-15:
                continue
            
            # Profile
            p = a  # p[k-i] = a[k-i], so p is just a shifted version
            # The relevant b values: b[lo:hi+1] = current_val * a[0:n] + d[lo:hi+1]
            # d[lo:hi+1] = b[lo:hi+1] - current_val * a[0:n]
            d_subset = b[lo:hi+1] - current_val * a
            
            # Try several values
            candidates = [0.0, current_val * 0.5, current_val * 0.8, current_val, 
                         current_val * 1.2, current_val * 1.5]
            # Also try the value that would make the peak equal to the average
            # peak ≈ current_val * max(a) + max(d_subset)
            # We want x * max(a) + max(d_subset) = avg_b
            # x = (avg_b - max(d_subset)) / max(a)
            max_a = a.max()
            if max_a > 1e-15:
                avg_b = b.mean()
                x_target = (avg_b - d_subset.max()) / max_a
                if x_target > 0:
                    candidates.append(x_target)
            
            best_x = current_val
            best_j = 2.0 * n * (b.max()) / S**2  # current objective (sum doesn't change in this approximation)
            
            # Actually, sum changes: new_sum = S - current_val + x
            # Let's compute properly
            for x in candidates:
                if x < 0:
                    continue
                new_max_b = np.maximum(d_subset + x * a, 0).max()
                # Also need to check other b values not in [lo, hi]
                # Those don't change: b[0:lo] and b[hi+1:]
                other_max = 0
                if lo > 0:
                    other_max = max(other_max, b[0:lo].max())
                if hi + 1 < len(b):
                    other_max = max(other_max, b[hi+1:].max())
                total_max = max(new_max_b, other_max)
                
                new_sum = S - current_val + x
                if new_sum < 0.01:
                    continue
                j = 2.0 * n * total_max / (new_sum ** 2)
                if j < best_j:
                    best_j = j
                    best_x = x
            
            if best_x != current_val:
                a[i] = best_x
                # Update b: b[lo:hi+1] += (best_x - current_val) * a
                b[lo:hi+1] += (best_x - current_val) * a
                current_val = best_x
                improved = True
                if best_j < best_obj:
                    best_obj = best_j
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj

def lse_gradient(a, c, beta):
    """LSE-smoothed gradient of max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_lse_descent(a_init, n, S, max_time, rng):
    """LSE-smoothed gradient descent with adaptive beta."""
    a = a_init.copy()
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    beta = 5.0
    step = 1.0
    last_improve = 0
    
    c = fft_convolve_self(a)
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Adaptive beta: increase when stuck
        if it - last_improve > 500:
            beta = min(500.0, beta * 1.1)
        
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step
        eff_step = step * S / (n * gnorm + S)
        a_new = project_simplex(a - eff_step * grad, S)
        
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new < cur_obj:
            a = a_new
            c = c_new
            step = min(step * 1.03, 5.0)
        else:
            step *= 0.8
            if it - last_improve > 2000:
                # Restart with perturbation
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
                step = 1.0
                beta = 5.0
    
    return best_a, best_obj
