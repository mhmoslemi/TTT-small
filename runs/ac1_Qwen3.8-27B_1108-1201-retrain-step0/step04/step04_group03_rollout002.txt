"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Hybrid of adaptive multiplicative updates, direct peak-attack redistribution,
LSE-smoothed gradient descent with annealing, and frequency-domain spectral shaping.
Uses multi-scale exploration and Barzilai-Borwein step adaptation.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    if cm < 1e-300:
        e = np.zeros(len(c))
        e[np.argmax(c)] = 1.0
    else:
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            e = np.zeros(len(c))
            e[np.argmax(c)] = 1.0
        else:
            e = e / tot
    fsize = 1 << (len(e) + n - 1).bit_length()
    W = fft(e, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_attack(f, c, n, S, alpha=0.1, top_k=5):
    """
    Direct peak-attack redistribution.
    Reduces elements most responsible for the top-k peak lags,
    redistributes mass to elements with low responsibility.
    """
    f = f.copy()
    peak_idx = np.argsort(c)[::-1][:top_k]
    
    # Compute responsibility for each element across top-k lags
    resp = np.zeros(n)
    for k in peak_idx:
        for j in range(n):
            partner = k - j
            if 0 <= partner < n:
                resp[j] += f[j] * f[partner]
    
    max_resp = resp.max()
    if max_resp < 1e-15:
        return f
    
    # Reduce high-responsibility elements
    reduction = np.zeros(n)
    high_mask = resp > 0.3 * max_resp
    reduction[high_mask] = alpha * resp[high_mask] / max_resp * f[high_mask]
    reduced_mass = reduction.sum()
    
    if reduced_mass < 1e-15:
        return f
    
    # Redistribute to low-responsibility elements
    low_mask = ~high_mask
    if low_mask.sum() == 0:
        low_mask = np.ones(n, dtype=bool)
    low_weights = np.maximum(1.0 - resp[low_mask] / max_resp, 0.01)
    low_weights /= low_weights.sum()
    
    f -= reduction
    f[low_mask] += reduced_mass * low_weights
    f = np.maximum(f, 0.0)
    f = project_simplex(f, S)
    return f


def spectral_smooth(f, n, sigma=2.0):
    """Apply Gaussian low-pass filter in frequency domain, then re-clip."""
    S = f.sum()
    N = 1 << (n - 1).bit_length()
    F = fft(f, N)
    freqs = np.fft.fftfreq(N, d=1.0/N)
    # Gaussian low-pass: keep low frequencies, attenuate high
    mask = np.exp(-2.0 * np.pi**2 * sigma**2 * freqs**2)
    F_filtered = F * mask
    f_new = np.real(ifft(F_filtered, n))
    f_new = np.maximum(f_new, 0.0)
    f_new = project_simplex(f_new, S)
    return f_new


def multiplicative_update(f, c, n, S, beta=30.0, lr=0.05):
    """Mirror descent / multiplicative update step."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return f.copy()
    e /= tot
    fsize = 1 << (len(e) + n - 1).bit_length()
    W = fft(e, fsize)
    A = fft(f, fsize)
    force = np.real(ifft(W * np.conj(A)))[:n]
    max_force = max(force.max(), 1e-10)
    f_new = f * np.exp(-lr * force / max_force)
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def barzilai_borwein_step(grad_prev, grad_curr, x_prev, x_curr, prev_step):
    """Barzilai-Borwein step size."""
    s = x_curr - x_prev
    y = grad_curr - grad_prev
    ss = np.dot(s, s)
    ys = np.dot(y, s)
    if ss < 1e-30:
        return prev_step
    if ys > 1e-15:
        return ss / ys
    elif ys < -1e-15:
        return ss / (-ys)
    return prev_step

def try_interpolation(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if new_n == n:
        return f.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, f)

    print("Returning sequence of length %d, best score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
