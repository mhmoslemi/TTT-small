"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining improved FISTA with LSE smoothing,
exact coordinate descent sweeps, multi-scale exploration, and adaptive restarts.
Key improvements: adaptive beta based on convolution flatness, coordinate descent
with exact 1D minimization, diverse restarts, and sequence length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val < 1e-30:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    active = active[active < 2 * n - 1]
    if len(active) == 0:
        return np.zeros(n)
    w_full = np.zeros(2 * n - 1)
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_v2(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                 beta_max=5000.0, mode="adaptive", step_scale=1.0):
    """
    Improved FISTA with adaptive beta based on convolution flatness.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive beta: lower when peaked, higher when flat
        c_mean = c.mean()
        c_std = c.std()
        flatness = c_std / max(c_mean, 1e-10) if c_mean > 1e-10 else 1.0
        # When flat (low flatness), use higher beta to capture subtle structure
        adaptive_factor = max(0.3, min(3.0, 1.0 / (0.5 + flatness)))
        beta = min(beta_max, beta_init * np.exp(beta_rate * it) * adaptive_factor)
        step = step_scale / max(beta, 1.0)

        # Gradient computation with mode selection
        if mode == "adaptive":
            if stuck_count > 2000:
                grad = exact_subgradient(y, c, tol_frac=max(0.005, 0.001 * stuck_count / 1000))
            elif stuck_count > 500:
                k = min(32, max(4, stuck_count // 100))
                grad = topk_gradient(y, c, k, power=1.2)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with backtracking-lite
        if stuck_count > 5000:
            step *= 0.2
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it
        elif stuck_count > 2000:
            step *= 0.5

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 10000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 60000.0)
            if decay > 0.03:
                noise = rng.normal(0, 0.003 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def coordinate_descent(a_init, max_time, rng, n_sweeps=None):
    """
    Coordinate descent with exact 1D optimization for each coordinate.
    Uses the structure of the convolution to update efficiently.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S_target / a.sum())
    conv = fft_convolve_self(a)
    S = a.sum()
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    sweep = 0

    max_sweeps = n_sweeps if n_sweeps is not None else 10000

    while time.monotonic() - start < max_time and sweep < max_sweeps:
        sweep += 1
        improved = 0
        # Randomize sweep order for better mixing
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > max_time:
                break
            a_old = a[j]
            if a_old < 1e-12:
                continue

            # Find affected range: k in [j, j+n-1]
            k_start = j
            k_end = min(j + n, 2 * n - 1)

            # Compute the 1D problem
            # For k in [k_start, k_end-1]:
            #   new_conv[k] = conv[k] + 2*(x-a_old)*a[k-j] + (x-a_old)^2 if k==2j
            # sum_new = S + (x - a_old)
            # obj = 2*n*max(new_conv) / sum_new^2

            # Find good candidate x values
            # Strategy: try to reduce the max by moving x
            # The max is typically at some lag k*. We want to reduce it.
            # At the optimum, typically the max is "shared" between a few lags.

            # Simple approach: try a few candidate values
            # x = 0, x = a_old/2, x = a_old, x = a_old*1.5, x = a_old*0.5
            # Plus: x that would make conv[k*] equal to the second max

            cur_max = conv.max()
            # Find the lag where max occurs (in affected range if possible)
            max_lag = int(np.argmax(conv))

            candidates = [0.0, a_old * 0.5, a_old, a_old * 1.5]
            # Add a candidate based on gradient information
            if k_start <= max_lag < k_end and max_lag != 2 * j:
                # d(conv[max_lag])/d(a[j]) = 2*a[max_lag - j]
                # To reduce conv[max_lag], we want to decrease a[j] if a[max_lag-j] > 0
                # New conv[max_lag] = conv[max_lag] + 2*(x-a_old)*a[max_lag-j]
                # Set this equal to current max: already at max, so try reducing
                # x such that conv[max_lag] decreases: if a[max_lag-j] > 0, decrease x
                candidates.append(a_old * 0.7)
                candidates.append(a_old * 1.3)

            # Also try: x that makes sum_new^2 / max_new_conv optimal
            # This is hard to solve analytically, so we use a small grid
            if a_old > 1e-10:
                for frac in [0.3, 0.7, 1.0, 1.4]:
                    candidates.append(a_old * frac)

            best_x = a_old
            best_f = 2.0 * n * cur_max / (S * S)

            for x in candidates:
                if x < 0:
                    x = 0.0
                dx = x - a_old
                if abs(dx) < 1e-14:
                    continue

                # Compute new max efficiently
                # Only affected lags change
                ds = S + dx
                if ds < 1e-10:
                    continue

                # Compute max of affected lags
                # new_conv[k] = conv[k] + 2*dx*a[k-j] + dx^2*(1 if k==2j else 0)
                affected_slice = conv[k_start:k_end]
                offsets = np.arange(k_start, k_end) - j
                affected_new = affected_slice + 2.0 * dx * a[offsets]
                # Add quadratic term for k=2j if in range
                if k_start <= 2 * j < k_end:
                    idx_2j = 2 * j - k_start
                    affected_new[idx_2j] += dx * dx

                max_affected = np.max(affected_new)
                max_unaffected = cur_max  # upper bound (actual unaffected max might be lower)
                # But if the global max was in the affected range, we need to be more careful
                if k_start <= max_lag < k_end:
                    # The old max might have changed, compute new global max
                    new_global_max = max(max_affected, np.max(conv[:k_start]) if k_start > 0 else 0,
                                         np.max(conv[k_end:]) if k_end < 2 * n - 1 else 0)
                else:
                    new_global_max = max(cur_max, max_affected)

                f_new = 2.0 * n * new_global_max / (ds * ds)
                if f_new < best_f - 1e-15:
                    best_f = f_new
                    best_x = x

            if best_x != a_old:
                dx = best_x - a_old
                a[j] = best_x
                S += dx
                # Update convolution
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                offsets = np.arange(k_start, k_end) - j
                conv[k_start:k_end] += 2.0 * dx * a[offsets]
                if k_start <= 2 * j < k_end:
                    conv[2 * j] += dx * dx
                improved += 1

        # Periodic normalization to prevent drift
        if S > 0:
            a = a * (S_target / S)
            S = S_target
            conv = fft_convolve_self(a)

        if sweep % 50 == 0:
            cur_obj = compute_obj(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            print("    CD sweep %d: obj=%.8f, improved_coords=%d" % (sweep, cur_obj, improved))

        if improved == 0 and sweep > 5:
            # Stuck, add small noise
            a = project_simplex(a + rng.normal(0, 1e-4 * S / np.sqrt(n), n), S_target)
            conv = fft_convolve_self(a)
            S = a.sum()

    return best_a

    p1_budget = remaining * 0.30
    print("\n=== Phase 1: FISTA at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_v2(global_best_a, p1_budget * 0.6, rng,
                          beta_init=10.0, beta_rate=0.006,
                          beta_max=5000.0, mode="adaptive", step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  FISTA-1: obj=%.8f" % obj1)
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 10:
        a1b, _ = run_fista_v2(global_best_a, remaining1, rng,
                               beta_init=50.0, beta_rate=0.004,
                               beta_max=10000.0, mode="adaptive", step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-2: obj=%.8f" % obj1b)
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Coordinate descent (15% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15
    if p2_budget > 20:
        print("\n=== Phase 2: Coordinate descent at n=%d (budget %.0fs) ===" % (len(global_best_a), p2_budget))
        a2 = coordinate_descent(global_best_a, p2_budget, rng)
        obj2 = compute_obj(a2)
        print("  CD result: obj=%.8f" % obj2)
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Multi-scale exploration (30% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.30
    if p3_budget > 30:
        print("\n=== Phase 3: Multi-scale (budget %.0fs) ===" % p3_budget)
        a3 = multi_scale_optimize(global_best_a, p3_budget, rng)
        obj3 = compute_obj(a3)
        print("  Multi-scale best: obj=%.8f" % obj3)
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Smart restarts + re-optimization (20% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 15:
        print("\n=== Phase 4: Smart restarts (budget %.0fs) ===" % p4_budget)
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)
        n_restarts = max(1, int(p4_budget / 30))  # ~30s per restart
        t4_start = time.monotonic()

        for r in range(n_restarts):
            remaining_r = p4_budget - (time.monotonic() - t4_start)
            if remaining_r < 10:
                break
            # Generate perturbed candidate
            a_pert = smart_restart(global_best_a, rng, S_best)
            # Optimize it
            a_opt, _ = run_fista_v2(a_pert, remaining_r * 0.8, rng,
                                     beta_init=20.0, beta_rate=0.005,
                                     beta_max=5000.0, mode="adaptive", step_scale=0.7)
            obj_r = compute_obj(a_opt)
            print("  Restart %d: obj=%.8f" % (r + 1, obj_r))
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_opt.copy()

    # === Phase 5: Final intensive polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p5_budget = remaining * 0.9
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_budget)
        n_best = len(global_best_a)

        # High-beta FISTA
        a5, _ = run_fista_v2(global_best_a, p5_budget * 0.5, rng,
                              beta_init=200.0, beta_rate=0.002,
                              beta_max=50000.0, mode="adaptive", step_scale=0.3)
        obj5 = compute_obj(a5)
        print("  High-beta: obj=%.8f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        # Quick CD polish
        remaining5 = p5_budget - (time.monotonic() - (deadline - remaining))
        if remaining5 > 10:
            a5b = coordinate_descent(global_best_a, remaining5, rng, n_sweeps=200)
            obj5b = compute_obj(a5b)
            print("  Final CD: obj=%.8f" % obj5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver <= global_best_obj + 1e-8:
                return final_list
            else:
                # Use our computed best
                print("Using computed best: %.8f" % global_best_obj)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")

    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    return [float(x) for x in a_fb]
