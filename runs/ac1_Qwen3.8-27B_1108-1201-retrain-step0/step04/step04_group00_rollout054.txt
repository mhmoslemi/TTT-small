import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_gradient(y, c, k, power=1.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    n = len(a_init)
    S = 1.0  # Normalize to sum=1
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:

        if mode == "lse":

        elif mode == "hybrid":

        elif mode == "topk":

                t_mom = 1.0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

"""
Minimize 2*n*max(conv)/sum^2 over non-negative sequences.

Key insight: The optimal sequence is asymmetric (tilted), exploiting the fact that
for non-symmetric x, the center convolution value (x·rev(x)) can be less than ||x||^2,
allowing the max to be lowered below the symmetric (uniform) bound of 2.

Strategy: Multi-phase optimization with tilted initializations:
1. Evaluate multiple initialization strategies (linear tilt, exponential, previous best)
2. FISTA with adaptive LSE smoothing
3. Hybrid mode (LSE + top-k) for mid-phase
4. Intensive active-set polishing
5. Multi-length exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting on the largest lags."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "hybrid":
            if stuck_count > 800:
                k = min(128, 8 + stuck_count // 100)
                grad = topk_gradient(y, c, k, power=1.2)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 2000:
                k = max(2, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if stuck_count > 4000:
            noise_scale = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0))
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 6000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 60000.0)
            if decay > 0.05:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """Intensive active-set polishing with adaptive k and step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 200:
            k = 1
            tol_frac = 0.0
        elif stuck < 1000:
            k = 3
            tol_frac = 0.003
        elif stuck < 3000:
            k = min(16, 4 + stuck // 300)
            tol_frac = 0.008
        elif stuck < 8000:
            k = min(48, 16 + stuck // 500)
            tol_frac = 0.015
        else:
            k = min(96, 48 + stuck // 1000)
            tol_frac = 0.03

        if k == 1 and tol_frac == 0.0:
            grad = exact_subgradient(y, c, 0.0)
        elif tol_frac > 0:
            grad = exact_subgradient(y, c, tol_frac)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, S / (0.5 * gnorm))
        step = max(step, S / (100.0 * gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 6000 and it % 300 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 12000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def make_tilted_init(n, alpha, rng):
    """Create a tilted initialization. alpha in [0,1] controls tilt amount."""
    m = np.arange(n, dtype=np.float64)
    t = m / max(n - 1, 1)
    x = 1.0 + alpha * (2.0 * t - 1.0)
    x = np.maximum(x, 0.01)
    x = x / x.sum()
    S = np.sqrt(2.0 * n)
    return x * S


def make_exponential_init(n, beta, rng):
    """Create an exponential initialization."""
    m = np.arange(n, dtype=np.float64)
    t = m / max(n - 1, 1)
    x = np.exp(beta * t)
    x = x / x.sum()
    S = np.sqrt(2.0 * n)
    return x * S


def make_quadratic_init(n, alpha, rng):
    """Create a quadratic tilt initialization."""
    m = np.arange(n, dtype=np.float64)
    t = m / max(n - 1, 1)
    x = 1.0 + alpha * (2.0 * t - 1.0) + 0.3 * alpha * (2.0 * t - 1.0) ** 2
    x = np.maximum(x, 0.01)
    x = x / x.sum()
    S = np.sqrt(2.0 * n)
    return x * S

                               beta_max=20000.0, mode="topk", step_scale=0.4)
            obj2c = compute_obj(a2c)
            print("  FISTA-topk: obj=%.6f (t=%.0fs)" % (obj2c, time.monotonic() - t2))
            try_update(a2c)

    # === Phase 3: Upsample exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 300 and global_best_a is not None and len(global_best_a) <= 8192:
        p3_budget = min(remaining * 0.20, 500.0)
        n2 = len(global_best_a) * 2
        # Create upsampled initialization with slight tilt
        a_up = global_best_a.copy()
        n_old = len(a_up)
        n_new = n2
        t_old = np.linspace(0, 1, n_old, endpoint=False)
        t_new = np.linspace(0, 1, n_new, endpoint=False)
        a_up = np.interp(t_new, t_old, a_up)
        # Add slight random perturbation
        a_up = a_up + rng.normal(0, 0.001 * a_up.max(), n_new)
        a_up = np.maximum(a_up, 0)
        S2 = np.sqrt(2.0 * n_new)
        a_up = a_up * (S2 / a_up.sum())

        print("\n=== Phase 3: Upsample to n=%d (budget %.0fs) ===" % (n2, p3_budget))
        t3 = time.monotonic()
        a3, _ = run_fista(a_up, p3_budget * 0.7, rng, beta_init=8.0, beta_rate=0.005,
                          beta_max=8000.0, mode="lse", step_scale=1.0)
        obj3 = compute_obj(a3)
        print("  FISTA at 2n: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        try_update(a3)

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 20 and global_best_a is None or compute_obj(a3) < compute_obj(global_best_a):
            a3b, _ = run_fista(a3 if obj3 < compute_obj(global_best_a) else global_best_a,
                               remaining3, rng, beta_init=40.0, beta_rate=0.003,
                               beta_max=12000.0, mode="hybrid", step_scale=0.6)
            obj3b = compute_obj(a3b)
            print("  Hybrid at 2n: obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            try_update(a3b)

    # === Phase 4: Active-set polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and global_best_a is not None:
        p4_budget = remaining * 0.40
        n_best = len(global_best_a)
        print("\n=== Phase 4: Active-set polish at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        a4a, _ = run_active_polish(global_best_a, p4_budget * 0.4, rng)
        obj4a = compute_obj(a4a)
        print("  Active polish 1: obj=%.6f (t=%.0fs)" % (obj4a, time.monotonic() - t4))
        try_update(a4a)

        remaining4 = p4_budget - (time.monotonic() - t4)
        if remaining4 > 20:
            a4b, _ = run_active_polish(global_best_a, remaining4 * 0.5, rng)
            obj4b = compute_obj(a4b)
            print("  Active polish 2: obj=%.6f (t=%.0fs)" % (obj4b, time.monotonic() - t4))
            try_update(a4b)

        remaining4b = p4_budget - (time.monotonic() - t4)
        if remaining4b > 10:
            a4c, _ = run_active_polish(global_best_a, remaining4b, rng)
            obj4c = compute_obj(a4c)
            print("  Active polish 3: obj=%.6f (t=%.0fs)" % (obj4c, time.monotonic() - t4))
            try_update(a4c)

    # === Phase 5: Final intensive refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 50 and global_best_a is not None:
        p5_budget = remaining * 0.80
        n_best = len(global_best_a)
        print("\n=== Phase 5: Final refinement (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # Very high beta FISTA
        a5a, _ = run_fista(global_best_a, p5_budget * 0.35, rng,
                           beta_init=200.0, beta_rate=0.002, beta_max=50000.0,
                           mode="hybrid", step_scale=0.3)
        obj5a = compute_obj(a5a)
        print("  High-beta: obj=%.6f (t=%.0fs)" % (obj5a, time.monotonic() - t5))
        try_update(a5a)

        remaining5 = p5_budget - (time.monotonic() - t5)
        if remaining5 > 20:
            a5b, _ = run_active_polish(global_best_a, remaining5, rng)
            obj5b = compute_obj(a5b)
            print("  Final active: obj=%.6f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            try_update(a5b)

    # === Phase 6: Try a different length if we have time ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_best_a is not None:
        n_cur = len(global_best_a)
        # Try a smaller length (downsample)
        if n_cur > 4096:
            n_ds = n_cur // 2
            a_ds = global_best_a[:n_ds * 2].reshape(n_ds, 2).mean(axis=1)
            S_ds = np.sqrt(2.0 * n_ds)
            a_ds = a_ds * (S_ds / a_ds.sum())
            obj_ds = compute_obj(a_ds)
            print("\n=== Phase 6: Downsampled n=%d: obj=%.6f ===" % (n_ds, obj_ds))
            if obj_ds < global_best_obj:
                global_best_obj = obj_ds
                global_best_a = a_ds.copy()
                # Polish it
                a_ds2, _ = run_active_polish(a_ds, remaining * 0.3, rng)
                obj_ds2 = compute_obj(a_ds2)
                print("  After polish: obj=%.6f" % obj_ds2)
                try_update(a_ds2)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    total_time = time.monotonic() - (deadline - budget_s + 30.0)
    print("Total time: %.0fs" % total_time)
    return final_list
