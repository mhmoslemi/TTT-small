import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta, top_k=None):
    """Gradient of LSE-smoothed peak. If top_k given, use top-k weighted grad instead of full LSE."""
    n = len(a)
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    if top_k is not None and top_k < len(c):
        # Generalized LSE: weight only top-k lags
        threshold = np.sort(c)[::-1][top_k - 1]
        mask = c >= threshold - 1e-15
        exp_bc = np.zeros(len(c))
        exp_bc[mask] = np.exp(beta * (c[mask] - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = exp_bc / total
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = exp_bc / total
    lse_val = c_max + np.log(max(total, 1e-300)) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_flatten_step(a, N, alpha, target_flatness):
    """Modify spectrum to make A^2 flatter (target: constant in time domain)."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    A2 = A * A
    C = np.real(ifft(A2))[:2 * n - 1]
    C_target = S * S / (2 * n - 1)
    # Error in time domain
    error = C - C_target
    E = fft(error, N)
    # Modify A to reduce error: A_new = A - alpha * E / (2*A) (Newton step on A^2 = target)
    A_safe = np.where(np.abs(A) > 1e-10, A, 1e-10)
    A_modified = A - alpha * E / (2.0 * A_safe)
    # Ensure conjugate symmetry
    A_modified[N - np.arange(1, N // 2 + 1)] = np.conj(A_modified[np.arange(1, N // 2 + 1)])
    A_modified[0] = np.real(A_modified[0])
    if N % 2 == 0:
        A_modified[N // 2] = np.real(A_modified[N // 2])
    a_new = np.real(ifft(A_modified))[:n]
    return project_simplex(a_new, S)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def coordinate_descent_polish(a_init, n, S, N, max_time, rng):
    """Polish using coordinate descent on active lags."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv_full(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Find top lags
        top_k = min(50, n)
        top_idx = np.argsort(c)[::-1][:top_k]

        # For each active lag, try to reduce it by adjusting coordinates
        for k in top_idx[:10]:
            # Gradient of C[k] wrt a[j] = a[k-j] + a[j-k] (with bounds)
            j_range = np.arange(n)
            ij1 = k - j_range
            ij2 = j_range - k
            valid1 = (ij1 >= 0) & (ij1 < n)
            valid2 = (ij2 >= 0) & (ij2 < n)
            grad_k = np.zeros(n)
            grad_k[valid1] += a[ij1[valid1]]
            grad_k[valid2] += a[ij2[valid2]]

            # Try small step in negative gradient direction
            gnorm = np.linalg.norm(grad_k)
            if gnorm < 1e-15:
                continue
            step = S / (n * gnorm) * 0.1
            candidate = project_simplex(a - step * grad_k, S)
            c_ls, _ = compute_conv_full(candidate, N)
            if c_ls.max() < cur:
                a = candidate

        # Random coordinate perturbation
        if it % 10 == 0:
            idx = rng.integers(0, n, size=min(100, n))
            a[idx] += rng.normal(0, 0.01 * S / n, size=len(idx))
            a = project_simplex(a, S)

    return best_a, best_cmax


def make_init_rand(n, rng):

def make_init_cyclic(n, rng, period):
    """Create a sequence with a periodic structure."""
    base = 1.0 + 0.3 * np.sin(2 * np.pi * np.arange(n) / period)
    base += 0.1 * rng.standard_normal(n)

def make_init_fourier_flat(n, rng, N):
    """Initialize with flat |A|^2 spectrum."""
    # Target: A^2 should be approximately flat in time domain
    # This means A should have approximately constant magnitude
    # and specific phase structure
    mag = np.ones(N) * np.sqrt(n)
    mag[0] = float(n)  # DC component
    phase = rng.uniform(0, 2 * np.pi, N)
    # Enforce conjugate symmetry
    phase[N - np.arange(1, N // 2 + 1)] = -phase[np.arange(1, N // 2 + 1)]
    phase[0] = 0.0
    if N % 2 == 0:
        phase[N // 2] = 0.0
    A = mag * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    return np.abs(a) + 0.01


def make_init_chebyshev(n, rng):
    """Chebyshev-like initialization: energy spread."""
    tt = np.linspace(0, np.pi, n, endpoint=False)
    a = 1.0 + 0.2 * np.cos(3 * tt) + 0.1 * np.cos(7 * tt) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    S = a.sum()

        phase1_budget = min(90.0, (deadline - n_start) * 0.1)

        inits = []
        try:
            given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            if given.size == n:
                inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
        except Exception:
            pass
        inits.append(("rand", make_init_rand(n, rng)))
        inits.append(("cyclic5", make_init_cyclic(n, rng, 5)))
        inits.append(("cyclic7", make_init_cyclic(n, rng, 7)))
        inits.append(("fourier", make_init_fourier_flat(n, rng, N)))
        inits.append(("cheb", make_init_chebyshev(n, rng)))

        n_inits = len(inits)
        per_init = phase1_budget / n_inits
        best_a_n = None
        best_sc_n = np.inf

        for name, arr in inits:
            if time.monotonic() > deadline - 300:
                break
            a_norm = safe_normalize(arr, n, S)
            t0 = time.monotonic()
            rem = max(2.0, per_init - 1.0)

            # Quick L-BFGS
            a_l, sc_l = lbfgs_lse_block(a_norm, n, S, N, rem, rng,
                                         beta_init=5.0, beta_max=50.0, m_lbfgs=8)
            if sc_l < best_sc_n:
                best_sc_n = sc_l
                best_a_n = a_l.copy()
                print("    [%s] sc=%.6f obj=%.6f" % (name, sc_l, 2*n*sc_l/S**2))

        if best_a_n is None:
            best_a_n = np.ones(n) * S / n
            best_sc_n = 2.0

        # Phase 2: Focused L-BFGS (60% of remaining)
        print("  Phase 2: Focused L-BFGS")
        t2 = time.monotonic()
        phase2_budget = min(900.0, (deadline - t2) * 0.55)

        a_curr = best_a_n.copy()
        stall = 0
        last_best = best_sc_n

        while time.monotonic() < t2 + phase2_budget:
            block_time = min(180.0, (t2 + phase2_budget) - time.monotonic())
            if block_time < 10:
                break
            beta_start = 10.0 + 3.0 * stall
            beta_end = min(500.0, 80.0 * (1 + 0.3 * stall))
            use_toppk = stall > 2

            a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                             beta_init=beta_start, beta_max=beta_end,
                                             m_lbfgs=12, top_k_mode=use_toppk)
            if sc_new < best_sc_n - 1e-9:
                best_sc_n = sc_new
                best_a_n = a_new.copy()
                a_curr = a_new.copy()
                stall = 0
                print("    [P2] sc=%.6f obj=%.6f (beta %.0f-%.0f)" %
                      (best_sc_n, 2*n*best_sc_n/S**2, beta_start, beta_end))
            else:
                a_curr = a_new.copy()
                stall += 1
                if stall >= 3:
                    a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                    stall = 0

        # Phase 3: Spectral refinement (150s)
        print("  Phase 3: Spectral refinement")
        t3 = time.monotonic()
        phase3_budget = min(150.0, (deadline - t3) * 0.15)
        if phase3_budget > 5:
            a_sp, sc_sp = spectral_block(best_a_n, n, S, N, phase3_budget, rng)
            if sc_sp < best_sc_n:
                best_sc_n = sc_sp
                best_a_n = a_sp.copy()
                print("    [Spectral] sc=%.6f obj=%.6f" % (best_sc_n, 2*n*best_sc_n/S**2))
            # L-BFGS after spectral
            rem = min(60.0, (deadline - time.monotonic()) * 0.2)
            if rem > 5:
                a_l2, sc_l2 = lbfgs_lse_block(a_sp, n, S, N, rem, rng,
                                               beta_init=30.0, beta_max=200.0, m_lbfgs=10)
                if sc_l2 < best_sc_n:
                    best_sc_n = sc_l2
                    best_a_n = a_l2.copy()
                    print("    [Post-spectral LBFGS] sc=%.6f obj=%.6f" % (best_sc_n, 2*n*best_sc_n/S**2))

        # Phase 4: Coordinate descent polish (120s)
        print("  Phase 4: Coordinate descent polish")
        t4 = time.monotonic()
        phase4_budget = min(120.0, (deadline - t4) * 0.1)
        if phase4_budget > 5:
            a_cd, sc_cd = coordinate_descent_polish(best_a_n, n, S, N, phase4_budget, rng)
            if sc_cd < best_sc_n:
                best_sc_n = sc_cd
                best_a_n = a_cd.copy()
                print("    [CoordDesc] sc=%.6f obj=%.6f" % (best_sc_n, 2*n*best_sc_n/S**2))
            # Final L-BFGS
            rem = min(60.0, (deadline - time.monotonic()) * 0.15)
            if rem > 5:
                a_fin, sc_fin = lbfgs_lse_block(a_cd, n, S, N, rem, rng,
                                                 beta_init=50.0, beta_max=400.0, m_lbfgs=10)
                if sc_fin < best_sc_n:
                    best_sc_n = sc_fin
                    best_a_n = a_fin.copy()
                    print("    [Final] sc=%.6f obj=%.6f" % (best_sc_n, 2*n*best_sc_n/S**2))

        # Update global best
        obj_n = 2.0 * n * best_sc_n / (S * S)
        print("  n=%d final: sc=%.6f obj=%.6f" % (n, best_sc_n, obj_n))
        if obj_n < global_best_obj:
            global_best_obj = obj_n
            global_best_seq = best_a_n.copy()

    # Fallback
    if global_best_seq is None:
        global_best_seq = np.ones(256, dtype=np.float64)

    # Final verification
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)
    if global_best_seq.sum() < 0.01:
        global_best_seq = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_best_seq]
    ver = evaluate_sequence(final_list)
    print("Verification: %.6f (length=%d)" % (ver if np.isfinite(ver) else float('inf'), len(final_list)))

    if not np.isfinite(ver):
        print("  Rejected, using fallback")
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver)

    return final_list
