Minimize 2*n*max(convolve(a,a))/sum(a)^2 using log-sum-exp smoothed

Key ideas:
1. Use log-sum-exp as a smooth surrogate for the max in the objective,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (css - S) / idx > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho = int(np.max(idx[cond]))
    theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value 2*n*max(c)/S^2."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_gradient_lse(a, c, T, N):
    """
    Compute gradient of (1/T)*log(sum(exp(T*c[k]/S^2))) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    S2 = a.sum() ** 2
    d = c / S2
    # Numerically stable softmax
    d_max = d.max()
    log_w = T * (d - d_max)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w = w / w_sum

    # Cross-correlation via FFT: (w cross-corr a)[i] = sum_j w[j+i]*a[j]
    w_pad = np.zeros(N)
    w_pad[:len(c)] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    corr = np.real(ifft(np.conj(fft(a_pad, N)) * fft(w_pad, N)))
    grad = (2.0 / S2) * corr[:n]
    return grad


def compute_gradient_tophat(a, c, k):
    """
    Gradient of sum of top-k convolution values (normalized by S^2).
    """
    n = len(a)
    S2 = a.sum() ** 2
    # Find top-k lags
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0
    w = w / k

    N = 1 << (2 * n - 2).bit_length()
    w_pad = np.zeros(N)
    w_pad[:len(c)] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    corr = np.real(ifft(np.conj(fft(a_pad, N)) * fft(w_pad, N)))
    grad = (2.0 / S2) * corr[:n]
    return grad
