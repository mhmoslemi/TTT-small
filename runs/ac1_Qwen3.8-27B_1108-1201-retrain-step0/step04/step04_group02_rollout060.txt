    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def lse_gradient(a, c, beta):
    """Compute the gradient of the LSE-smoothed objective."""
    n = len(a)
    S = a.sum()
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    # Gradient of sum(w_k * c_k) w.r.t. a_i
    # c_k = sum_i a[i] * a[k-i]
    # d(c_k)/d(a_i) = a[k-i] if 0 <= k-i < n
    # So grad_i = 2*sum_k w_k * a[k-i] (factor of 2 because a appears twice in c_k)
    # Wait, actually c_k = sum_i a[i]*a[k-i], and d/d(a_i) of this is:
    #   a[k-i] (from the first factor) + a[i] (from the second factor, when k-j = i, i.e., j = k-i)
    # But since we sum over all i, the second term is sum_{j: k-j=i} a[j] = a[k-i]
    # So d(c_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n
    # 
    # Actually let me be more careful. c_k = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
    # d(c_k)/d(a_i) = a[k-i] (if 0<=k-i<n) + sum_j a[j]*d(a[k-j])/d(a_i)
    #                = a[k-i] + a[k-i] (from the term where j=i in the sum, giving a[k-i])
    # Wait no. Let me just use: c_k = sum_i a[i] * a[k-i]
    # d(c_k)/d(a_i) = a[k-i] + sum_j a[j] * d(a[k-j])/d(a_i)
    # The second sum: d(a[k-j])/d(a_i) = 1 if k-j = i, i.e., j = k-i.
    # So the second term is a[k-i] (if 0 <= k-i < n).
    # Total: d(c_k)/d(a_i) = 2*a[k-i] if 0 <= k-i < n
    #
    # So grad_i = 2*sum_k w_k * a[k-i] (where 0 <= k-i < n)
    # = 2*sum_{k: 0<=k-i<n} w_k * a[k-i]
    # = 2*sum_{m: 0<=m<n} w_{m+i} * a[m] (substituting m = k-i)
    #
    # This is a cross-correlation of w and a!
    # Specifically, grad = 2 * cross_corr(w, a) where cross_corr means:
    # (w ⋆ a)[i] = sum_m w[m+i] * a[m]
    #
    # To compute this efficiently via FFT:
    # cross_corr(w, a)[i] = sum_m w[m+i] * a[m]
    # This is the correlation of w with a, which can be computed as:
    # ifft(fft(w_reversed) * conj(fft(a)))... 
    # Actually, let's think of it as:
    # (w ⋆ a)[i] = sum_m w[m+i]*a[m]
    # Reverse w: w_rev[m] = w[M-1-m] where M is the length
    # Then (w ⋆ a)[i] = (w_rev * a)[M-1-i] (convolution)
    # Hmm, this is getting complicated. Let me just use the FFT trick:
    
    # Actually, the simplest way: grad = 2 * ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
    # Let me verify: ifft(fft(w)*conj(fft(a)))[k] = sum_m w[m] * a[m-k] (circular)
    # With zero padding, this becomes the linear cross-correlation.
    # (w cross a)[k] = sum_m w[m] * a[m-k] (with linear bounds)
    # We want: grad[i] = 2 * sum_m w[m+i] * a[m] = 2 * (w cross a)[-i]... 
    
    # Let me just use the direct formula:
    # grad[i] = 2 * sum_{k=max(i,0)}^{min(i+size-1, size-1)} w[k] * a[k-i] (if k-i < n)
    #         = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    
    # This is a "sliding dot product" of w with a, which is a cross-correlation.
    # Using FFT: cross_corr[i] = sum_j a[j] * w[j+i]
    # This equals ifft(fft(w_padded) * conj(fft(a_padded))) at the right indices.
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # ifft(W * conj(A))[k] = sum_j w[j] * a[(j-k) mod fsize]
    # With zero padding, for k in range(n): sum_j w[j] * a[j-k] (linear)
    # We want sum_j a[j] * w[j+i] = sum_j w[j+i] * a[j]
    # Let m = j+i: sum_m w[m] * a[m-i] = ifft(W*conj(A))[i] (with zero padding)
    
    corr = np.real(ifft(W * np.conj(A)))[:n]
    grad = 2.0 * corr
    
    # Now subtract the sum gradient term
    # obj = 2n * lse(c; beta) / S^2
    # d(obj)/d(a_i) = 2n/S^2 * d(lse)/d(a_i) + 2n*lse * d(S^{-2})/d(a_i)
    # = 2n/S^2 * grad[i] - 4n*lse/S^3
    # where lse = cm + log(tot)/beta
    lse_val = cm + np.log(max(tot, 1e-300)) / beta
    grad = 2.0 * n * grad / (S ** 2) - 4.0 * n * lse_val / (S ** 3)
    
    return grad, lse_val

def run_bb_lse(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=200.0):
    """
    Barzilai-Borwein method with LSE-smoothed subgradient.
    """
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    
    it = 0
    last_improve = 0
    it_at_last_improve = 0
    
    # State for BB
    a_prev = a.copy()
    g_prev = None
    bb_alpha = 1.0 / np.linalg.norm(fft_convolve_self(a).max() + 1.0)  # initial step
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Update beta (anneal)
        frac = min(1.0, it / max(1, it * (max_time / max_time))))  # just use it
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 5000.0)
        
        # Compute gradient
        g, lse_val = lse_gradient(a, c, beta)
        
        # BB step size
        if g_prev is not None:
            s = a - a_prev
            y = g - g_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            ss = np.dot(s, s)
            if sy > 1e-15 and yy > 1e-15:
                alpha_bb_s = sy / yy
                alpha_bb_l = sy / ss if ss > 1e-15 else alpha_bb_s
                # Use the geometric mean for stability
                bb_alpha = np.sqrt(max(alpha_bb_s, 1e-10) * max(alpha_bb_l, 1e-10))
            elif sy <= 1e-15:
                # Non-descent direction, use damped step
                bb_alpha *= 0.5
        
        a_prev = a.copy()
        g_prev = g.copy()
        
        # Line search (a few backtracks)
        alpha = bb_alpha
        improved = False
        for _ in range(5):
            a_try = project_simplex(a - alpha * g, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            cur_obj = 2.0 * n * c.max() / S**2
            if obj_try < cur_obj:
                a = a_try
                c = c_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            a = a_prev.copy()  # revert
            bb_alpha *= 0.3
        
        # Update best
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Stalling detection and restart
        if it - last_improve > 2000:
            # Try a perturbation
            stall_duration = it - last_improve
            if stall_duration > 5000:
                # Big restart
                noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
            elif stall_duration > 2000:
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
            g_prev = None
            a_prev = a.copy()
            bb_alpha = 0.1
            last_improve = it
        
        # Print progress
        if it % 5000 == 0:
            print("    BB-LSE: it=%d, obj=%.6f, beta=%.1f, alpha=%.6f" % (it, best_obj, beta, bb_alpha))
    
    return best_a, best_obj

def run_adaptive_topk(a_init, n, S, max_time, rng):
    """
    Adaptive top-k subgradient with Barzilai-Borwein-like step size.
    """
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    
    it = 0
    last_improve = 0
    step = 0.5
    a_prev = a.copy()
    g_prev = None
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive top-k
        k = min(128, max(1, 1 + it // 500))
        size_c = len(c)
        k = min(k, size_c)
        
        idx = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for j in idx:
            lo = max(0, j - n + 1)
            hi = min(j, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[j - i_vals]
        grad /= k
        
        # BB-like step
        if g_prev is not None:
            s = a - a_prev
            y = grad - g_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if sy > 1e-15 and yy > 1e-15:
                step = sy / yy
                step = np.clip(step, 1e-6, 10.0)
        
        a_prev = a.copy()
        g_prev = grad.copy()
        
        # Try the step
        improved = False
        for _ in range(5):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            cur_obj = 2.0 * n * c.max() / S**2
            if obj_try < cur_obj:
                a = a_try
                c = c_try
                improved = True
                break
            step *= 0.5
        
        if not improved:
            a = a_prev.copy()
            step *= 0.3
        
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 3000:
            noise_scale = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            g_prev = None
            a_prev = a.copy()
            step = 0.5
            last_improve = it
        
        if it % 5000 == 0:
            print("    TopK-BB: it=%d, obj=%.6f, k=%d" % (it, best_obj, k))
    
    return best_a, best_obj

def spectral_flatten_init(n, rng, n_trials=5):
    """
    Generate initializations using spectral flattening.
    Start with random phases, iteratively flatten the spectrum while
    maintaining non-negativity.
    """
    S = np.sqrt(2.0 * n)
    fsize = 1 << (2 * n).bit_length()
    
    inits = []
    for _ in range(n_trials):
        # Random initial sequence
        a = rng.random(n) + 0.1
        a = a / a.sum() * S
        
        for _ in range(20):
            A = fft(a, fsize)
            mag = np.abs(A)
            # Compress the dynamic range
            alpha = 0.5
            mag_new = mag ** alpha
            # Renormalize to keep the zero frequency correct
            mag_new = mag_new / mag_new[0] * mag[0]
            # Build new A with same phase
            phase = np.angle(A)
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.real(ifft(A_new, fsize))[:n]
            # Project to non-negative
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s * S
            a = a_new
        
        inits.append(a)
    
    return inits

def structured_perturb(a, n, S, rng):
    """Apply structured perturbations."""
    strategy = rng.integers(0, 6)
    t = np.arange(n) / n
    a_new = a.copy()
    
    if strategy == 0:
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * t + phase)
        a_new = a * mod
    elif strategy == 1:
        mod = 1.0 + 0.15 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
        a_new = a * mod
    elif strategy == 2:
        # Edge vs center
        edge = np.minimum(t, 1-t)
        mod = 1.0 + 0.3 * np.cos(np.pi * edge * 2)
        a_new = a * mod
    elif strategy == 3:
        # Random block
        nblocks = rng.integers(8, 32)
        block_mod = rng.uniform(0.7, 1.3, nblocks)
        mod = np.interp(t, np.linspace(0, 1, nblocks), block_mod)
        a_new = a * mod
    elif strategy == 4:
        # Frequency domain noise
        fsize = 1 << (2 * n).bit_length()
        A = fft(a, fsize)
        noise = rng.normal(0, 0.03, len(A)) + 1j * rng.normal(0, 0.03, len(A))
        A = A * (1 + noise)
        a_new = np.real(ifft(A, fsize))[:n]
    else:
        # Symmetric perturbation
        half = n // 2
        perturb = rng.normal(0, 0.05, half)
        p = np.concatenate([perturb, perturb[::-1]])
        if n % 2 == 0:
            a_new = a * (1 + p[:n])
        else:
            a_new = a * (1 + np.concatenate([p[:half], p[half-1:0:-1]]))
    
    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using:
- Barzilai-Borwein adaptive step size with LSE-smoothed subgradient
- Multi-length search (4096, 8192, 16384) with multi-scale initialization
- Spectral flattening for diverse initializations
- Adaptive peak weighting and annealed beta
- Structured perturbations for escaping local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta):
    """
    Compute the gradient of the LSE-smoothed objective.
    Returns (gradient, lse_value).
    """
    n = len(a)
    S = a.sum()
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot

    # cross[i] = sum_m w[m+i] * a[m]  (for m = 0..n-1)
    # Computed via: ifft(fft(w) * conj(fft(a)))
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]

    # d(LSE)/d(a_i) = 2 * cross[i]
    # d(f)/d(a_i) = (2n/S^2) * 2 * cross[i] - 4n * LSE / S^3
    lse_val = cm + np.log(max(tot, 1e-300)) / beta
    grad = (4.0 * n / (S ** 2)) * corr - (4.0 * n * lse_val) / (S ** 3)

    return grad, lse_val


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def weighted_subgradient(a, c, k):
    """Weighted subgradient where weights are based on peak prominence."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    # Weights: exponential based on how much above the (k+1)-th peak
    if k < len(c):
        threshold = c[np.argsort(c)[::-1][k]]
    else:
        threshold = 0
    weights = np.maximum(vals - threshold, 0.01)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for j, w in zip(idx, weights):
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * w * a[j - i_vals]
    return grad


def run_bb_lse(a_init, n, S, max_time, rng, beta_start=15.0, beta_end=200.0, anneal_steps=8000):
    """
    Barzilai-Borwein method with LSE-smoothed subgradient and annealed beta.
    """
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S ** 2
    best_a = a.copy()

    it = 0
    last_improve = 0
    a_prev = a.copy()
    g_prev = None
    bb_alpha = 0.1
    fail_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Anneal beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / anneal_steps)

        # Compute gradient
        g, lse_val = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # BB step size computation
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if sy > 1e-14 and yy > 1e-14:
                alpha_s = sy / yy
                alpha_l = sy / ss if ss > 1e-14 else alpha_s
                # Geometric mean for stability
                bb_alpha = np.sqrt(max(alpha_s, 1e-10) * max(alpha_l, 1e-10))
            elif sy <= 1e-14:
                bb_alpha *= 0.5

        # Clamp step size
        max_alpha = 2.0 * S / (n * gnorm + 1e-15)
        bb_alpha = min(bb_alpha, max_alpha)
        bb_alpha = max(bb_alpha, 1e-8)

        a_prev = a.copy()
        g_prev = g.copy()

        # Backtracking line search
        alpha = bb_alpha
        cur_obj = 2.0 * n * c.max() / S ** 2
        improved = False
        for _ in range(8):
            a_try = project_simplex(a - alpha * g, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2
            if obj_try < cur_obj:
                a = a_try
                c = c_try
                improved = True
                fail_count = 0
                break
            alpha *= 0.5

        if not improved:
            a = a_prev.copy()
            c = fft_convolve_self(a)
            bb_alpha *= 0.4
            fail_count += 1

        # Update best
        cur_obj = 2.0 * n * c.max() / S ** 2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Stalling detection
        stall = it - last_improve
        if stall > 3000:
            if stall > 8000:
                noise_scale = 0.05 * S / np.sqrt(n)
            elif stall > 4000:
                noise_scale = 0.02 * S / np.sqrt(n)
            else:
                noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            g_prev = None
            a_prev = a.copy()
            bb_alpha = 0.1
            last_improve = it
            fail_count = 0

        if it % 10000 == 0:
            print("      BB-LSE: it=%d, obj=%.6f, beta=%.1f, alpha=%.6f" % (it, best_obj, beta, bb_alpha))

    return best_a, best_obj


def run_topk_bb(a_init, n, S, max_time, rng, k_start=2, k_end=64):
    """
    Top-k subgradient with BB step size.
    """
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S ** 2
    best_a = a.copy()

    it = 0
    last_improve = 0
    step = 0.3
    a_prev = a.copy()
    g_prev = None

    while time.monotonic() - start < max_time:
        it += 1

        k = min(k_end, k_start + it // 500)
        g = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # BB step
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = g - g_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-14 and yy > 1e-14:
                step = sy / yy
                step = np.clip(step, 1e-8, 5.0)
            else:
                step *= 0.5

        max_step = 2.0 * S / (n * gnorm + 1e-15)
        step = min(step, max_step)

        a_prev = a.copy()
        g_prev = g.copy()

        # Line search
        alpha = step
        cur_obj = 2.0 * n * c.max() / S ** 2
        improved = False
        for _ in range(6):
            a_try = project_simplex(a - alpha * g, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2
            if obj_try < cur_obj:
                a = a_try
                c = c_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            a = a_prev.copy()
            step *= 0.4

        cur_obj = 2.0 * n * c.max() / S ** 2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stall = it - last_improve
        if stall > 2000:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            g_prev = None
            a_prev = a.copy()
            step = 0.3
            last_improve = it

        if it % 10000 == 0:
            print("      TopK-BB: it=%d, obj=%.6f, k=%d" % (it, best_obj, k))

    return best_a, best_obj


def spectral_flatten_init(n, rng, n_trials=4):
    """Generate initializations using iterative spectral flattening."""
    S = np.sqrt(2.0 * n)
    fsize = 1 << (2 * n).bit_length()
    inits = []

    for _ in range(n_trials):
        # Start with a smooth base + randomness
        t = np.arange(n) / n
        base = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t + rng.uniform(0, 6.28))
        a = np.maximum(base + rng.normal(0, 0.1, n), 0.01)
        a = a / a.sum() * S

        # Iteratively flatten the spectrum
        for _ in range(30):
            A = fft(a, fsize)
            mag = np.abs(A).copy()
            # Compress dynamic range: mag_new = mag^alpha
            alpha = 0.4
            mag_new = mag ** alpha
            # Preserve zero frequency (DC)
            if mag[0] > 1e-15:
                mag_new = mag_new / mag_new[0] * mag[0]
            phase = np.angle(A)
            A_new = mag_new * np.exp(1j * phase)
            a_new = np.real(ifft(A_new, fsize))[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a = a_new / s * S

        inits.append(a.copy())

    return inits


def structured_perturb(a, n, S, rng):
    """Apply a structured perturbation to escape local minima."""
    strategy = rng.integers(0, 7)
    t = np.arange(n) / n
    a_new = a.copy()

    if strategy == 0:
        freq = rng.uniform(0.5, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.15 * np.sin(2 * np.pi * freq * t + phase)
        a_new = a * mod
    elif strategy == 1:
        mod = 1.0 + 0.1 * np.cos(4 * np.pi * t) + 0.08 * np.cos(8 * np.pi * t)
        a_new = a * mod
    elif strategy == 2:
        edge = np.minimum(t, 1 - t)
        mod = 1.0 + 0.2 * np.cos(np.pi * edge * 3)
        a_new = a * mod
    elif strategy == 3:
        nblocks = rng.integers(8, 64)
        block_mod = rng.uniform(0.8, 1.2, nblocks)
        mod = np.interp(t, np.linspace(0, 1, nblocks), block_mod)
        a_new = a * mod
    elif strategy == 4:
        fsize = 1 << (2 * n).bit_length()
        A = fft(a, fsize)
        noise = rng.normal(0, 0.02, len(A)) + 1j * rng.normal(0, 0.02, len(A))
        A = A * (1 + noise)
        a_new = np.real(ifft(A, fsize))[:n]
    elif strategy == 5:
        # Mirror-symmetric perturbation
        half = n // 2
        p = rng.normal(0, 0.03, half)
        full_p = np.zeros(n)
        full_p[:half] = p
        if n % 2 == 0:
            full_p[half:] = p[::-1]
        else:
            full_p[half:] = p[:n-half][::-1]
        a_new = a * (1 + full_p)
    else:
        # Random sparse perturbation
        n_sparse = max(1, n // 100)
        idx = rng.choice(n, size=n_sparse, replace=False)
        a_new = a.copy()
        a_new[idx] *= rng.uniform(0.5, 1.5, n_sparse)

    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)


def multi_scale_init(given, n_target, rng):
    """Create an initialization at n_target from a smaller solution."""
    n_src = len(given)
    S = np.sqrt(2.0 * n_target)

    if n_src == n_target:
        return given.copy()

    # Interpolate
    t_src = np.linspace(0, 1, n_src, endpoint=False)
    t_tgt = np.linspace(0, 1, n_target, endpoint=False)
    a_interp = np.interp(t_tgt, t_src, given)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-10:
        a_interp = np.ones(n_target) / n_target
        s = 1.0
    a_interp = a_interp / s * S

    # Add small random perturbation
    a_interp += rng.normal(0, 0.01, n_target)
    a_interp = np.maximum(a_interp, 0)
    return project_simplex(a_interp, S)


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses BB step size, LSE smoothing, multi-scale initialization, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 50.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs, given_len=%s" % (
        budget_s, len(given) if given is not None else "None"))
    t0 = time.monotonic()

    # Define candidate lengths
    candidate_lengths = []
    if given is not None:
        candidate_lengths.append(len(given))
    for L in [4096, 8192, 16384]:
        if L not in candidate_lengths:
            candidate_lengths.append(L)

    # Sort by priority: given length first, then by size
    candidate_lengths.sort()

    print("Candidate lengths: %s" % candidate_lengths)

    # === Phase 1: Exploration across lengths (20% time) ===
    total_time = deadline - time.monotonic()
    phase1_time = total_time * 0.20
    per_length_time = phase1_time / len(candidate_lengths)

    length_results = {}  # n -> (best_a, best_obj)

    print("=== Phase 1: Exploration (%.0fs total) ===" % phase1_time)

    for L in candidate_lengths:
        if time.monotonic() > deadline - total_time * 0.55:
            break
        S = np.sqrt(2.0 * L)
        t_alloc = min(per_length_time, deadline - time.monotonic() - 10)
        if t_alloc < 15:
            continue

        # Generate initializations
        inits = []
        if given is not None and len(given) == L:
            inits.append(("given", given.copy()))
        elif given is not None:
            inits.append(("multiscale", multi_scale_init(given, L, rng)))

        # Spectral flattening
        try:
            spectral = spectral_flatten_init(L, rng, n_trials=2)
            for i, s in enumerate(spectral):
                inits.append(("spec%d" % i, s))
        except Exception:
            pass

        # Structured inits
        tt = np.linspace(0, 1, L, endpoint=False)
        inits.append(("cos", np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01)))
        inits.append(("rand", rng.random(L) + 0.05))

        best_for_L = None
        best_obj_L = np.inf
        per_init_time = t_alloc / max(1, len(inits))

        for name, arr in inits:
            if time.monotonic() > deadline - total_time * 0.50:
                break
            if per_init_time < 3:
                continue

            arr = np.maximum(arr, 0)
            s_arr = arr.sum()
            if s_arr < 0.01:
                continue
            arr = arr / s_arr * S

            # Quick BB-LSE optimization
            try:
                a_opt, obj_opt = run_bb_lse(
                    arr, L, S, per_init_time, rng,
                    beta_start=10, beta_end=80, anneal_steps=3000)
            except Exception:
                continue

            a_opt = project_simplex(a_opt, S)
            obj_real = compute_objective(a_opt)

            if obj_real < best_obj_L:
                best_obj_L = obj_real
                best_for_L = a_opt.copy()
                print("  [L=%d, %s] obj=%.6f" % (L, name, obj_real))

        if best_for_L is not None:
            length_results[L] = (best_for_L, best_obj_L)
            if best_obj_L < global_best:
                global_best = best_obj_L
                global_a = best_for_L.copy()

    print("=== After Phase 1: best=%.6f, elapsed=%.0fs ===" % (
        global_best, time.monotonic() - t0))

    # === Phase 2: Intensive optimization (50% time) ===
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.50

    if global_a is not None and phase2_time > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Phase 2: Intensive optimization (n=%d, %.0fs) ===" % (n_best, phase2_time))

        a_cur = global_a.copy()

        # Round 1: BB-LSE with moderate annealing
        t_alloc = min(phase2_time * 0.35, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            print("  Round 1: BB-LSE beta=[15,200]")
            a1, s1 = run_bb_lse(a_cur, n_best, S_best, t_alloc, rng,
                                beta_start=15, beta_end=200, anneal_steps=10000)
            a1 = project_simplex(a1, S_best)
            s1 = compute_objective(a1)
            if s1 < global_best:
                global_best = s1
                global_a = a1.copy()
                a_cur = a1.copy()
                print("    -> %.6f" % global_best)

        # Round 2: TopK-BB
        t_alloc = min(phase2_time * 0.25, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            print("  Round 2: TopK-BB")
            a2, s2 = run_topk_bb(a_cur, n_best, S_best, t_alloc, rng,
                                 k_start=2, k_end=48)
            a2 = project_simplex(a2, S_best)
            s2 = compute_objective(a2)
            if s2 < global_best:
                global_best = s2
                global_a = a2.copy()
                a_cur = a2.copy()
                print("    -> %.6f" % global_best)

        # Round 3: BB-LSE with high beta (precise peak targeting)
        t_alloc = min(phase2_time * 0.25, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            print("  Round 3: BB-LSE beta=[100,500]")
            a3, s3 = run_bb_lse(a_cur, n_best, S_best, t_alloc, rng,
                                beta_start=100, beta_end=500, anneal_steps=5000)
            a3 = project_simplex(a3, S_best)
            s3 = compute_objective(a3)
            if s3 < global_best:
                global_best = s3
                global_a = a3.copy()
                a_cur = a3.copy()
                print("    -> %.6f" % global_best)

        # Round 4: Another TopK with different params
        t_alloc = min(phase2_time * 0.15, deadline - time.monotonic() - 10)
        if t_alloc > 5:
            print("  Round 4: TopK-BB k=[1,16]")
            a4, s4 = run_topk_bb(a_cur, n_best, S_best, t_alloc, rng,
                                 k_start=1, k_end=16)
            a4 = project_simplex(a4, S_best)
            s4 = compute_objective(a4)
            if s4 < global_best:
                global_best = s4
                global_a = a4.copy()
                print("    -> %.6f" % global_best)

    # === Phase 3: Fine polishing (25% time) ===
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.25

    if global_a is not None and phase3_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Phase 3: Fine polishing (%.0fs) ===" % phase3_time)

        a_cur = global_a.copy()

        # Very high beta LSE (essentially exact subgradient with tiny smoothing)
        t_alloc = min(phase3_time * 0.5, deadline - time.monotonic() - 10)
        if t_alloc > 5:
            a_p1, s_p1 = run_bb_lse(a_cur, n_best, S_best, t_alloc, rng,
                                    beta_start=200, beta_end=800, anneal_steps=3000)
            a_p1 = project_simplex(a_p1, S_best)
            s_p1 = compute_objective(a_p1)
            if s_p1 < global_best:
                global_best = s_p1
                global_a = a_p1.copy()
                print("  High-beta LSE: %.6f" % global_best)

        # Micro-optimization with exact top-1 subgradient
        t_alloc = min(phase3_time * 0.5, deadline - time.monotonic() - 5)
        if t_alloc > 5:
            a_micro = global_a.copy()
            c_micro = fft_convolve_self(a_micro)
            step = 0.005
            micro_start = time.monotonic()
            best_micro = global_best
            it = 0
            last_imp = 0

            while time.monotonic() - micro_start < t_alloc:
                it += 1

                # Top-3 weighted subgradient
                g = weighted_subgradient(a_micro, c_micro, 3)
                gnorm = np.linalg.norm(g)
                if gnorm < 1e-18:
                    break

                a_try = project_simplex(a_micro - step * g, S_best)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n_best * c_try.max() / S_best ** 2
                cur_obj = 2.0 * n_best * c_micro.max() / S_best ** 2

                if obj_try < cur_obj - 1e-14:
                    a_micro = a_try
                    c_micro = c_try
                    step = min(step * 1.01, 0.5)
                    if obj_try < best_micro:
                        best_micro = obj_try
                        last_imp = it
                        if best_micro < global_best:
                            global_best = best_micro
                            global_a = a_micro.copy()
                else:
                    step *= 0.85
                    if step < 0.0003:
                        noise = rng.normal(0, 0.001 * S_best / np.sqrt(n_best), n_best)
                        a_micro = project_simplex(a_micro + noise, S_best)
                        c_micro = fft_convolve_self(a_micro)
                        step = 0.003
                        last_imp = it

                if it - last_imp > 3000:
                    noise = rng.normal(0, 0.003 * S_best / np.sqrt(n_best), n_best)
                    a_micro = project_simplex(a_micro + noise, S_best)
                    c_micro = fft_convolve_self(a_micro)
                    step = 0.005
                    last_imp = it

            a_micro = project_simplex(a_micro, S_best)
            s_micro = compute_objective(a_micro)
            if s_micro < global_best:
                global_best = s_micro
                global_a = a_micro.copy()
            print("  Micro: %.6f (it=%d)" % (global_best, it))

    # === Phase 4: Perturbation exploration (remaining time) ===
    remaining = deadline - time.monotonic()

    if global_a is not None and remaining > 20:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Phase 4: Perturbation search (%.0fs) ===" % remaining)

        a_cur = global_a.copy()
        n_pert = min(30, int(remaining / 4))

        for p in range(n_pert):
            if time.monotonic() > deadline - 15:
                break
            pert = structured_perturb(a_cur, n_best, S_best, rng)
            t_alloc = min(4.0, (deadline - time.monotonic() - 10) / max(1, n_pert - p))
            if t_alloc < 1:
                break
            try:
                a_p, s_p = run_bb_lse(pert, n_best, S_best, t_alloc, rng,
                                      beta_start=20, beta_end=100, anneal_steps=2000)
                a_p = project_simplex(a_p, S_best)
                s_p = compute_objective(a_p)
                if s_p < global_best:
                    global_best = s_p
                    global_a = a_p.copy()
                    a_cur = a_p.copy()
                    print("  Pert %d: %.6f" % (p, global_best))
            except Exception:
                continue

    # === Phase 5: Try other lengths if we have time ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        # Try a different length with the current best as template
        alt_lengths = [L for L in candidate_lengths if L != len(global_a)]
        for L in alt_lengths[:1]:
            if remaining < 30:
                break
            S_alt = np.sqrt(2.0 * L)
            a_alt = multi_scale_init(global_a, L, rng)
            t_alloc = min(remaining * 0.5, deadline - time.monotonic() - 10)
            if t_alloc > 10:
                print("=== Trying alternate length L=%d ===" % L)
                a_alt_opt, s_alt = run_bb_lse(a_alt, L, S_alt, t_alloc, rng,
                                              beta_start=10, beta_end=150, anneal_steps=5000)
                a_alt_opt = project_simplex(a_alt_opt, S_alt)
                s_alt = compute_objective(a_alt_opt)
                if s_alt < global_best:
                    global_best = s_alt
                    global_a = a_alt_opt.copy()
                    print("  Alt L=%d: %.6f" % (L, global_best))
            remaining = deadline - time.monotonic()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
