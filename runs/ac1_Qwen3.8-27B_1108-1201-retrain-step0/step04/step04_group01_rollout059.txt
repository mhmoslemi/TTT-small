def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def topk_lse(c, k, beta):
    """Top-k LSE: smooth max over the top k elements of c."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c_top - cm).max())
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    val = cm + np.log(e.sum()) / beta
    # Return value and weights (on the top-k indices)
    w = e / e.sum()
    return val, w, idx

def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain to explore."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()

    phase_perturb = rng.normal(0, amount, fsize)

    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten(a, alpha, iterations=10):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(iterations):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-20:
            break
        # We want to reduce peaks in the autocorrelation, which means
        # reducing concentration in the power spectrum
        # Soft threshold: reduce power where it's above mean
        ratio = power / max(mean_p, 1e-20)
        # Damp high-power frequencies, boost low-power
        new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
        new_power = np.maximum(new_power, 1e-20)
        # Preserve phase
        A_new = A * np.sqrt(np.maximum(new_power / power, 0))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        a = normalize(a_new, S)
    
    return a

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten_step(a, alpha):
    """One step of spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    ratio = power / max(mean_p, 1e-20)
    # Damp high-power, boost low-power frequencies
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_fista_refined(a_init, S, max_time, rng,
                      beta_init=20.0, beta_rate=0.003, beta_max=5000.0,
                      k_frac=0.08, use_momentum=True):
    """Refined FISTA with top-k LSE, backtracking, and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > 1500:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Periodic spectral perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            y = spectral_perturb(y, 0.08, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_adam(a_init, S, max_time, rng,
             lr_init=0.02, beta_adam=50.0, beta_rate=0.01):
    """Adam optimizer with adaptive LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    k = max(4, int(0.06 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = min(2000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate: decay over time
        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.7 * progress)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        # Periodic restart
        if t % 5000 == 0 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

        # Spectral flattening step
        if t % 3000 == 0 and rng.random() < 0.1:
            a = spectral_flatten_step(a, 0.05)

    return best_a, best

def run_multiplicative(a_init, S, max_time, rng):
    """Multiplicative (ISRA-style) update, naturally maintains positivity."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    k = max(4, int(0.05 * (2 * n - 1)))
    it = 0
    eta = 1.0 / (2.0 * 50.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(5000.0, 20.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)

        # Multiplicative update: a_j *= exp(-eta * g_j)
        # Use first-order: a_j *= (1 - eta * g_j) but clip to stay positive
        update = 1.0 - eta * grad
        update = np.maximum(update, 0.01)
        a_new = a * update
        a = project_simplex(a_new, S)

        # Decay eta
        eta = 1.0 / (2.0 * (20.0 + it * 0.05))

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

        t_a = time_limit * 0.35
        if t_a > 20:
            a_opt, val_opt = run_fista_refined(a_start, S, t_a, rng,
                                               beta_init=15.0, beta_rate=0.003,
                                               beta_max=3000.0, k_frac=0.08)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  FISTA: obj=%.6f" % obj)

        # Phase B: Adam (30% of time)
        t_b = time_limit * 0.30
        if t_b > 20 and time.monotonic() < deadline - 30:
            a_opt, val_opt = run_adam(best_a_local, S, t_b, rng,
                                      lr_init=0.015, beta_adam=40.0,
                                      beta_rate=0.008)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Adam: obj=%.6f" % obj)

        # Phase C: Polish (20% of time)
        t_c = time_limit * 0.20
        if t_c > 15 and time.monotonic() < deadline - 20:
            a_opt, val_opt = run_polish(best_a_local, S, t_c, rng)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Polish: obj=%.6f" % obj)

        # Phase D: Multiplicative (10% of time)
        t_d = time_limit * 0.10
        if t_d > 10 and time.monotonic() < deadline - 15:
            a_opt, val_opt = run_multiplicative(best_a_local, S, t_d, rng)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Multiplicative: obj=%.6f" % obj)

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Best for n=%d: obj=%.6f" % (n, obj_final))

        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_local.copy()
            print("  *** New global best: %.6f ***" % global_best)

        return best_a_local, best_local

    # Main run: n = 8192
    remaining = deadline - time.monotonic()
    if remaining > 60:
        try_optimization(n_primary, given, remaining * 0.55, "Main")

    # Second run: n = 16384 (if time permits)
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_secondary = 16384
        a_sec = resample_sequence(global_a, n_secondary) if global_a is not None else None
        try_optimization(n_secondary, a_sec, remaining * 0.30, "Secondary")

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs) ===" % remaining)

        # Very high beta FISTA
        t_fin = remaining * 0.5
        a_fin, val_fin = run_fista_refined(global_a, S_best, t_fin, rng,
                                           beta_init=500.0, beta_rate=0.001,
                                           beta_max=50000.0, k_frac=0.02,
                                           use_momentum=False)
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final FISTA: obj=%.6f" % global_best)

        # Polish
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 15:
            a_pol, val_pol = run_polish(global_a, S_best, remaining2, rng)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  Final Polish: obj=%.6f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
