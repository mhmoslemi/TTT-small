"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:

2. Targeted spectral flattening via modified Gerchberg-Saxton

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta, topk=None):
    """Gradient of LSE approximation, optionally using only top-k values."""
    n = len(a)
    cm = c.max()
    if topk is not None and topk < len(c):
        # Zero out all but top-k values
        c_use = c.copy()
        threshold = np.sort(c)[::-1][topk - 1]
        c_use[c_use < threshold] = cm - 1.0 / beta  # push below
    else:
        c_use = c
    e = np.exp(beta * (c_use - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient_topk(a, c, k=5):
    """Exact subgradient combining top-k argmax positions."""
    n = len(a)
    # Get top-k positions
    top_positions = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for k_pos in top_positions:
        k_pos = int(k_pos)
        lo = max(0, k_pos - n + 1)
        hi = min(k_pos, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k_pos - idx]
    return grad / len(top_positions)


def run_fista_v2(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=3000.0,
                 beta_ramp=2000, topk=5, use_hybrid=True):
    """Improved FISTA with adaptive beta and top-k subgradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    patience = 5000

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: ramp up, then adapt based on peakiness
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max
            # Periodic beta reduction to escape local minima
            if it % 5000 == 0:
                beta = beta_max * 0.5

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient
        if use_hybrid and it > 2000 and rng.random() < min(0.4, 0.001 * (it - 2000)):
            grad = exact_subgradient_topk(y, c, topk=topk)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Scale step for subgradient
            step = S / (gnorm * 1.5 * (1.0 + 0.005 * np.log1p(it / 500.0)))
        else:
            grad = lse_gradient(y, c, beta, topk=topk if it > 3000 else None)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            # Adaptive: reduce if stuck
            if it - last_improve > patience:
                step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_flatten_v3(a_init, n, S, max_time, rng):
    """Improved spectral flattening with adaptive target."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time - 5:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Compute current convolution peak ratio
        c = fft_convolve_self(a)
        c_max = c.max()
        c_mean = c.mean()
        peak_ratio = c_max / max(c_mean, 1e-30)

        # Adaptive alpha: more aggressive when peak is prominent
        progress = min(1.0, it / 1000.0)
        base_alpha = 0.3 + 0.5 * progress
        if peak_ratio > 3.0:
            alpha = min(0.9, base_alpha + 0.2)
        else:
            alpha = base_alpha * 0.7

        # Flatten with soft targeting
        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target

        # Additional peak suppression in frequency domain
        # Identify frequency bins that contribute to time-domain peaks
        # (this is approximate but helps)
        P_peak_factor = 1.0 - 0.3 * np.minimum(P / P_mean, 3.0) / 3.0
        P_new = P_new * P_peak_factor

        mag_new = np.sqrt(np.maximum(P_new, 0))
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def coord_descent_v2(a_init, n, S, max_time, rng):
    """Gradient-guided coordinate descent."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    # Find the max position
    k_max = int(np.argmax(c))

    while time.monotonic() - start < max_time - 5:
        it += 1
        improved = False
        c = fft_convolve_self(a)
        k_max = int(np.argmax(c))

        n_try = min(30 + it // 5, 100)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 5:
                break

            # Choose pair: prefer to move mass from high-convolution-contribution
            # positions to low-contribution positions
            # Contribution of position i to c[k_max] is a[k_max - i] (if valid)
            contrib = np.zeros(n)
            lo = max(0, k_max - n + 1)
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            contrib[idx] = a[k_max - idx]

            # Move from high contributor to low contributor
            if contrib.max() > 0:
                # Source: high contribution
                source_weights = contrib ** 2
                source_weights /= source_weights.sum() + 1e-30
                j = rng.choice(n, p=source_weights)
                # Destination: low contribution, non-zero a
                dest_weights = np.maximum(a - contrib / n, 0)
                dest_weights[j] = 0
                if dest_weights.sum() > 0:
                    dest_weights /= dest_weights.sum()
                    m = rng.choice(n, p=dest_weights)
                else:
                    m = rng.integers(0, n)
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # Compute direction
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val_j = 0.0
                if 0 <= j + k < n:
                    val_j += a[j + k]
                if 0 <= j - k < n:
                    val_j += a[j - k]
                val_m = 0.0
                if 0 <= m + k < n:
                    val_m += a[m + k]
                if 0 <= m - k < n:
                    val_m += a[m - k]
                d[k] = val_j - val_m

            lo_d = -a[j]
            hi_d = a[m]

            # Line search
            best_delta = 0.0
            best_val = c.max()
            for _ in range(20):
                mid = (lo_d + hi_d) / 2
                c_trial = c + mid * d
                c_trial[0] += 2 * (a[j] - a[m]) * mid + 2 * mid * mid
                val = c_trial.max()
                if val < best_val:
                    best_val = val
                    best_delta = mid
                # Binary search for the minimizer
                # (this is a 1D convex function, so we can binary search the derivative)
                # But for simplicity, use a ternary-like approach
                if mid > 0:
                    hi_d = mid
                else:
                    lo_d = mid

            if abs(best_delta) > 1e-10:
                a[j] += best_delta
                a[m] -= best_delta
                if a[j] < 0 or a[m] < 0:
                    a[j] = max(a[j], 0)
                    a[m] = max(a[m], 0)
                    s = a.sum()
                    if s > 0:
                        a *= S / s
                    c = fft_convolve_self(a)
                    if c.max() < best:
                        best = c.max()
                        best_a = a.copy()
                        improved = True
                else:
                    c = fft_convolve_self(a)
                    if c.max() < best - 1e-10:
                        best = c.max()
                        best_a = a.copy()
                        improved = True

        if not improved and it > 10:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-8:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def generate_starts(parent, n, rng, S):
    """Generate diverse starting sequences."""
    starts = []

    if parent is not None and len(parent) == n:
        starts.append(("parent", parent.copy()))
        # Perturbed parent
        for scale in [0.01, 0.03, 0.05, 0.1]:
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            v = project_simplex(parent + noise, S)
            starts.append(("parent_pert_%s" % scale, v))

        # Palindromic version
        pal = (parent + parent[::-1]) / 2
        starts.append(("parent_pal", normalize_sum(pal, n)))

        # Spectral modification
        fsize = 1 << (2 * n - 1).bit_length()
        F = fft(parent, fsize)
        freqs = np.fft.fftfreq(fsize)
        # Low-pass
        F_lp = F * np.exp(-5.0 * (freqs / 0.2) ** 2)
        a_lp = np.real(ifft(F_lp))[:n]
        a_lp = np.maximum(a_lp, 0)
        s = a_lp.sum()
        if s > 1e-10:
            starts.append(("parent_lp", a_lp * S / s))

    # Structured starts
    tt = np.linspace(0, 1, n, endpoint=False)
    starts.append(("uniform", np.ones(n) * S / n))
    starts.append(("gaussian", np.exp(-4.0 * (tt - 0.5) ** 2)))
    starts.append(("bimodal", np.exp(-50*(tt-0.25)**2) + np.exp(-50*(tt-0.75)**2)))
    starts.append(("cosine", np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01)))
    starts.append(("random", rng.random(n) + 0.05))
    # "Anti-convolution" - concentrated at edges
    starts.append(("edge", np.maximum(np.abs(0.5 - tt) ** 2, 0.01)))

    return [(name, normalize_sum(arr, n)) for name, arr in starts]
