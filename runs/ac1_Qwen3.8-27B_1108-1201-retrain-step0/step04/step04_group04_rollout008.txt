"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses FISTA with high-beta LSE, exact subgradient steps, and targeted coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(a, beta):
    """Compute LSE approximation to max(conv(a,a)) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, cm, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # LSE value
    lse_val = cm + np.log(tot) / beta
    return lse_val, cm, grad


def exact_subgradient_at_peak(a, k_peak):
    """Exact subgradient of b[k_peak] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            grad[i] += a[j]
    return 2.0 * grad  # factor of 2 because b[k] = sum_i a[i]*a[k-i] and both terms depend on a


def run_fista_high_beta(a_init, n, S, max_time, rng, beta_init=200.0, beta_max=8000.0):
    """FISTA with high initial beta."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Beta schedule: start high, increase slowly
        progress = min(1.0, it / 2000.0)
        beta = beta_init + (beta_max - beta_init) * progress
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new

        best = c_final.max()

        # We want c to be flatter.
        # Strategy: modify FFT(a_padded) to reduce the peak of IFFT of its square.
        
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        A = fft(a_pad)
        
        # The peak of c is at some lag. The imaginary part of A determines the "phase"
        # of the signal. We can adjust the phase to spread out the convolution.
        
        # Alternative: water-filling on the magnitude
        # |A|^2 determines the "energy" at each frequency.
        # The convolution c[k] = (1/fsize) * sum_m A[m]^2 * exp(2pi imk/fsize)
        # To make c flat, we want A[m]^2 to be concentrated at m=0.
        # But |A[0]|^2 = (sum a)^2 is fixed.
        # So we want to minimize |A[m]| for m != 0.
        # This is equivalent to making a as "smooth" as possible... which gives uniform a.
        # But uniform a gives score 2, which is worse than 1.5!
        
        # The issue is that we're using CIRCULAR convolution here, but the actual
        # problem uses LINEAR convolution. The linear convolution is the first 2n-1
        # values of the circular convolution of the zero-padded signal.
        # So we only care about the first 2n-1 values being flat.
        
        # A different approach: directly minimize the peak of c using a
        # "peak shaving" technique in the time domain.
        
        # Find the peak
        k_peak = int(np.argmax(c))
        # Reduce the contribution to the peak
        # The peak is c[k_peak] = sum_i a[i]*a[k_peak-i]
        # The largest contributors are where a[i]*a[k_peak-i] is largest.
        
        # Try to reduce the two largest factors
        products = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                products[i] = a[i] * a[j]
        
        # Find the pair with the largest product
        i_max = int(np.argmax(products))
        j_max = k_peak - i_max
        
        if 0 <= j_max < n and i_max != j_max and a[i_max] > 1e-10 and a[j_max] > 1e-10:
            # Try to transfer mass from the larger of the two to a random position
            if a[i_max] > a[j_max]:
                src, dst = i_max, j_max
            else:
                src, dst = j_max, i_max
            
            # Transfer a small amount from src to a random position
            # Actually, we want to reduce the product a[src]*a[dst]
            # Reducing a[src] helps, but we need to put the mass somewhere else
            # that doesn't increase other peaks too much.
            # Simple: move to a position that doesn't contribute to the peak.
            # Positions that don't contribute to peak k_peak: those j where k_peak-j is out of range
            # or where a[k_peak-j] is small.
            
            # For now, just do a small random perturbation focused on reducing the peak
            move_amt = 0.01 * a[src]
            dst_pos = int(rng.integers(0, n))
            if dst_pos != src and dst_pos != dst and a[dst_pos] >= 0:
                a_trial = a.copy()
                a_trial[src] -= move_amt
                a_trial[dst_pos] += move_amt
                if a_trial[src] >= 0:
                    c_trial = fft_convolve_self(a_trial)
                    if c_trial.max() < cur:
                        a = a_trial
        
        # Also do a spectral step: reduce high-frequency content
        # (but not too much, since we need structure to beat the uniform case)
        if it % 5 == 0:
            A_mag = np.abs(A)
            A_phase = A / np.maximum(A_mag, 1e-30)
            # Apply a mild low-pass to reduce the peak
            freqs = np.arange(fsize)
            # Distance from DC (and from Nyquist, since the signal is real)
            dist = np.minimum(freqs, fsize - freqs).astype(float)
            # Mild attenuation of high frequencies
            att = np.exp(-0.01 * (dist / fsize) ** 2 * it)  # very mild
            A_new = A * att
            a_new = np.real(ifft(A_new))[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * (S / s_new)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur:
                    a = a_new
    
    return best_a, best
