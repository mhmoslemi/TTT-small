"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using log-sum-exp smoothed subgradient
with Nesterov acceleration and FFT-based gradient computation.
"""
import numpy as np
from scipy.fft import fft, ifft
import time

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

def lse_gradient(a_eval, c_eval, beta, fsize):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a_eval)
    # Softmax weights (numerically stable)
    c_shifted = c_eval - c_eval.max()
    exp_vals = np.exp(beta * c_shifted)
    w = exp_vals / exp_vals.sum()
    
    # Cross-correlation: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
    W = fft(w, fsize)
    A = fft(a_eval, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr

def exact_subgradient(a_eval, c_eval, fsize):
    """
    Compute the exact subgradient of max_k c[k] at a_eval.
    If unique max at k*, grad[j] = 2*a[k*-j] for valid j.
    """
    n = len(a_eval)
    k_star = int(np.argmax(c_eval))
    # grad[j] = 2 * a[k_star - j] for 0 <= k_star - j < n
    grad = np.zeros(n)
    j_min = max(0, k_star - (n - 1))
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        grad[indices] = 2.0 * a_eval[k_star - indices]
    return grad

def subgradient_optimize(a_init, n, S, max_time, rng, 
                         beta_init=100.0, beta_max=2000.0,
                         use_nesterov=True, noise_period=1000, noise_prob=0.3):

    Main subgradient optimization loop with log-sum-exp smoothing.
    """
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_nest = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_grad = 1 << (3 * n - 1).bit_length()
    
    it = 0
    step_base = 2.0 * S / n
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute convolution and score at extrapolated point
        c = fft_convolve_self(y)
        c_max_val = c.max()

def spectral_refine(a_init, n, S, max_time, rng):
    """
    Spectral refinement: iteratively modify the power spectrum to reduce
    the peak autocorrelation, while maintaining non-negativity via projection.
    """
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        c = np.real(ifft(P))[:2*n-1]
        c_max = c.max()
        
        # Identify lag with peak
        k_peak = int(np.argmax(c))
        
        # Compute how much each frequency bin contributes to the peak
        # c[k] = sum_m P[m] * exp(2πi*m*k/N) / N (from IFFT)
        # To reduce c[k_peak], reduce P[m] where exp(2πi*m*k_peak/N) has positive real part
        # that aligns with making c[k_peak] large.
        # Actually, c[k] = ifft(P)[k] = (1/N) * sum_m P[m] * exp(2πi*m*k/N)
        # So the contribution of bin m to c[k_peak] is P[m] * exp(2πi*m*k_peak/N) / N
        # To reduce c[k_peak], we want to reduce P[m] where the real part of
        # exp(2πi*m*k_peak/N) is positive (constructive interference at the peak).
        
        # But we also want to maintain the total power (which relates to S).
        # Actually, sum(P) = N * sum(a^2), not directly S.
        # The DC component P[0] = S^2, which we want to keep.
        
        # Strategy: reduce bins that are high AND contribute to the peak
        contrib_to_peak = np.real(P * np.exp(2j * np.pi * np.arange(fsize) * k_peak / fsize))
        
        # Create a "harmful" mask: bins that are above median and have positive contribution to peak
        p_median = np.median(P)
        harmful = (P > p_median) & (contrib_to_peak > 0)
        
        # Reduce harmful bins, slightly boost others
        eta = 0.1  # reduction amount
        P_new = P.copy()
        P_new[harmful] = P_new[harmful] * (1.0 - eta)
        P_new[~harmful] = P_new[~harmful] * (1.0 + eta * harmful.sum() / max((~harmful).sum(), 1))
        
        # Ensure non-negative power
        P_new = np.maximum(P_new, 0.0)
        
        # Reconstruct with current phase
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
