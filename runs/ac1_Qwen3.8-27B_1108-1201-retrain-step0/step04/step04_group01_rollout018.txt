"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Diverse initialization with quick screening
2. FISTA with adaptive LSE smoothing (coarse)
3. Fourier-domain mutation for basin hopping
4. Mirror descent (KL-proximal) steps for exploration
5. High-beta polishing with peak shaving
6. Adaptive annealing to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=16):
    """Exact subgradient from the top-k lags, weighted by excess above mean."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((c_k - T)^p_+)."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj


def mirror_descent_step(a, grad, eta, S):
    """
    Mirror descent (KL-proximal) step.
    a_new_j = a_j * exp(-eta * grad_j) / Z, normalized to sum S.
    """
    # Scale grad to avoid overflow
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled_eta = eta / g_max
    # Clip to prevent overflow
    exponents = -np.clip(scaled_eta * grad, -50, 50)
    a_new = a * np.exp(exponents)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)


def fourier_mutate(a, rng, scale=0.1):
    """
    Fourier-domain mutation: perturb the phase of the DFT,
    then take IFFT and project back.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Perturb phase
    phase = np.angle(A)
    mag = np.abs(A)
    # Add random phase perturbation (larger at high freqs)
    freqs = np.arange(fsize)
    freq_weight = 1.0 / (1.0 + freqs / (0.1 * fsize))
    phase_noise = rng.normal(0, scale * freq_weight, fsize)
    new_phase = phase + phase_noise
    new_A = mag * np.exp(1j * new_phase)
    # Make sure DC component is preserved (to maintain sum)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, a.sum())


def fourier_crossover(a1, a2, rng, alpha=0.5):
    """
    Fourier-domain crossover: interpolate magnitudes in Fourier domain.
    """
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    # Interpolate: random mix of magnitude and phase
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    # Randomly choose phase from each
    r = rng.random(fsize)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]  # Preserve DC
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005, beta_max=5000.0,
              use_mirror_every=0):
    """
    FISTA with adaptive LSE smoothing.
    If use_mirror_every > 0, periodically take mirror descent steps.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (1.2 * beta * 0.3)

        # Stuck detection
        if it - last_improve > 2000:
            step *= 0.5
            if use_mirror_every > 0 and it % use_mirror_every == 0:
                y = mirror_descent_step(y, grad, step * 2, S)
            else:
                noise_scale = 0.003 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def run_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping: alternate between local optimization and
    Fourier-domain perturbations.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    # Parameters
    opt_time = 30  # seconds per local opt
    n_iterations = int(max_time / (opt_time + 5))

    for it in range(n_iterations):
        if time.monotonic() - start > max_time - opt_time:
            break

        # Local optimization
        a_opt, val = run_fista(a, S, opt_time, rng,
                                beta_init=50 + 20 * it,
                                beta_rate=0.003,
                                beta_max=8000 + 2000 * it)
        if val < best:
            best = val
            best_a = a_opt.copy()

        # Fourier-domain perturbation for next iteration
        scale = max(0.01, 0.3 * np.exp(-it / (n_iterations * 0.3)))
        a = fourier_mutate(a_opt, rng, scale=scale)

    return best_a, best


def run_adaptive_polish(a_init, S, max_time, rng):
    """
    Adaptive polishing: alternate between different gradient types
    with adaptive switching.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    y = a.copy()
    t_mom = 1.0
    it = 0
    current_mode = 0  # 0: lse, 1: peak, 2: excess
    mode_duration = 200  # iterations per mode

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        # Switch modes periodically
        if it % mode_duration == 0:
            current_mode = (current_mode + 1) % 3

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()

        # Compute gradient based on mode
        if current_mode == 0:  # LSE
            beta = 50.0 * (1 + progress * 200)
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (1.0 * beta * 0.3)
        elif current_mode == 1:  # Peak shaving
            k = min(8 + int(32 * progress), n // 8)
            grad = peak_shave_gradient(y, c, k=k)
            step = 2.0 * S / (n * (1 + 0.01 * it))
        else:  # Excess
            T = c.mean() + 0.3 * (c.max() - c.mean())
            grad, _ = excess_gradient(y, c, T, p=2.0)
            step = 1.0 / (0.3 * (50.0 * (1 + progress * 100)) * 0.3)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Mirror descent step occasionally
        if it % 3000 == 0 and progress > 0.3:
            y = mirror_descent_step(y, grad, step * 0.5, S)
            t_mom = 1.0

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))

    # Standard windows
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))  # Hanning
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))  # Hamming
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * tt) + 0.08 * np.cos(4 * np.pi * tt))  # Blackman
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.exp(-5.0 * (tt - 0.5) ** 2))  # Gaussian

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.cos(10 * np.pi * tt))
    inits.append(1.0 + 0.4 * np.sin(6 * np.pi * tt + 0.5) + 0.15 * np.cos(14 * np.pi * tt))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(1.0, 0.3, n)))

    # Block structures
    for nb in [4, 8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n])

    # Ramp + noise
    inits.append(np.maximum(tt + rng.normal(0, 0.1, n), 0.01))
    inits.append(np.maximum((1-tt) + rng.normal(0, 0.1, n), 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Diverse initialization with quick FISTA screening
2. Basin hopping with Fourier-domain phase mutations
3. Mirror descent (KL-proximal) steps for non-Euclidean exploration
4. Adaptive mode-switching polish (LSE / peak-shaving / excess)
5. KKT-guided final polishing with very high-beta LSE
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=16):
    """Exact subgradient from top-k lags weighted by excess above mean."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def kkt_subgradient(a, c, tol=1e-6):
    """
    Compute KKT-optimized subgradient: find non-negative weights on
    active lags (where c_k is within tol of max) that minimize the
    correlation with a (i.e., make the subgradient as 'aligned' as
    possible with the negative simplex normal cone).
    """
    n = len(a)
    cmax = c.max()
    active = np.where(c >= cmax - tol * cmax)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # For each active lag k, compute the subgradient direction 2*(e_k * a)
    # We want a convex combination that is "most negative" on supported coords
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    # Compute all subgradient directions simultaneously via matrix
    # grad_k = 2 * ifft(fft(e_k) * conj(A))[:n] where e_k has 1 at position k
    # This is: 2 * ifft(conj(A) * exp(-2j*pi*k/fsize))[:n] for each k
    # Simpler: just compute them one by one for the (few) active lags
    active = active[:20]  # Limit to 20 active lags for efficiency
    grads = np.zeros((len(active), n))
    for i, k in enumerate(active):
        w_full = np.zeros(len(c))
        w_full[k] = 1.0
        W = fft(w_full, fsize)
        grads[i] = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # Find weights lambda that minimize max_j (sum_k lambda_k * grad_k)_j
    # over j where a_j > 0 (these should be <= 0 at optimum)
    # Approximate: use the weights proportional to c_k (uniform over active as fallback)
    active_weights = np.ones(len(active)) / len(active)
    combined = np.zeros(n)
    for i in range(len(active)):
        combined += active_weights[i] * grads[i]
    return combined


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((c_k - T)^p_+)."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj


def mirror_descent_step(a, grad, eta, S):
    """
    KL-proximal (mirror descent) step:
    a_new_j proportional to a_j * exp(-eta * grad_j), normalized to sum S.
    """
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled = -np.clip(eta * grad / g_max, -30, 30)
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)


def fourier_phase_mutate(a, rng, scale=0.1):
    """
    Fourier-domain mutation: perturb the phase of the DFT,
    take IFFT, project to simplex.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    # Frequency-dependent noise: more perturbation at high freqs
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve DC (sum)
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """
    Fourier-domain crossover: random mix of magnitudes and phases.
    """
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r  # Random mixing coefficient per frequency
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, mirror_every=0, perturbation_period=5000):
    """
    FISTA with adaptive LSE smoothing and optional mirror descent steps.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: conservative for LSE (L ~ 4*beta roughly)
        step = 1.0 / (3.0 * beta)

        # Stuck detection: perturb or mirror step
        if it - last_improve > 2500:
            if mirror_every > 0 and it % mirror_every == 0:
                y = mirror_descent_step(y, grad, step * 0.3, S)
            elif it % 100 == 0:
                noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 10000.0)
                if noise_scale > 1e-10:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic small perturbation to avoid getting stuck
        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (10.0 * perturbation_period))
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping: alternate local FISTA with Fourier-domain mutations.
    Uses adaptive mutation scale that decreases over time.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(20.0, max_time / 15.0)
    n_iters = max(1, int(max_time / (opt_time + 5)))

    # Keep a population of diverse candidates
    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = it / max(1, n_iters)

        # Adaptive mutation scale: large early, small late
        base_scale = 0.4 * np.exp(-2.0 * progress) + 0.02
        scale = base_scale * (1.0 + 0.5 * rng.random())

        # Choose mutation type
        r = rng.random()
        if len(population) >= 2 and r < 0.3:
            # Crossover from population
            idx1 = rng.integers(len(population))
            idx2 = rng.integers(len(population))
            if idx1 != idx2:
                a = fourier_crossover(population[idx1][1], population[idx2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, scale)
        else:
            a = fourier_phase_mutate(a, rng, scale)

        # Local optimization
        a_opt, val = run_fista(a, S, opt_time, rng,
                                beta_init=20 + 50 * it,
                                beta_rate=0.003,
                                beta_max=5000 + 2000 * min(it, 20),
                                mirror_every=2000,
                                perturbation_period=8000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))

        # Maintain population diversity
        population.append((val, a_opt.copy()))
        if len(population) > 8:
            population.sort(key=lambda x: x[0])

def run_adaptive_polish(a_init, S, max_time, rng):
    """
    Adaptive mode-switching polish: cycle through LSE, peak-shaving,
    excess, and KKT modes with FISTA momentum.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    y = a.copy()
    t_mom = 1.0
    it = 0
    mode = 0
    mode_len = 300  # iterations per mode
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        # Cycle modes: LSE -> peak -> excess -> KKT
        if it % mode_len == 0:
            mode = (mode + 1) % 4

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        # Stuck handling
        if it - last_improve > 3000:
            if mode == 2:
                y = mirror_descent_step(y, c.max() * np.ones(n) - c.mean() * np.ones(n),
                                        0.01, S)
            else:
                noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
                y = project_simplex(y + rng.normal(0, max(noise_scale, 1e-12), n), S)
            t_mom = 1.0
            last_improve = it
            continue

        # Compute gradient based on current mode
        if mode == 0:  # LSE
            beta = 30.0 * (1.0 + 100.0 * progress)
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (3.0 * beta)
        elif mode == 1:  # Peak shaving
            k = min(4 + int(32 * progress), n // 16)
            grad = peak_shave_gradient(y, c, k=k)
            gnorm = np.linalg.norm(grad)
            step = 1.5 * S / (n * (1.0 + 0.005 * it) * max(gnorm / (S / n), 1e-10))
        elif mode == 2:  # Excess
            T = c.mean() + 0.2 * (c.max() - c.mean() + 1e-10)
            grad, _ = excess_gradient(y, c, T, p=2.0)
            gnorm = np.linalg.norm(grad)
            step = 1.0 / (0.5 * np.linalg.norm(grad) * 0.1 + 1.0) * S / n
        else:  # KKT-inspired
            grad = kkt_subgradient(y, c, tol=1e-5)
            gnorm = np.linalg.norm(grad)
            step = 1.0 * S / (n * max(np.linalg.norm(grad), 1e-10) * 0.5 + n * 0.01)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_kkt_polish(a_init, S, max_time, rng):
    """
    KKT-guided polishing: use the KKT subgradient direction
    with very small steps for final precision.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    it = 0
    t_mom = 1.0
    y = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()

        # KKT subgradient with tightening tolerance
        tol = max(1e-4, 1e-6 + (1e-4 - 1e-6) * progress)
        grad = kkt_subgradient(y, c, tol=tol)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small, decreasing steps
        step = 0.1 * S / (n * (1.0 + 0.01 * it))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))

    # Standard windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * tt) + 0.08 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / (0.25)) + 0.01)

    # Multi-frequency signals
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

    # Block structures
    for nb in [4, 8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse init + quick FISTA screening (~150s)
    2. Basin hopping with Fourier mutations (~500s)
    3. Adaptive mode-switching polish (~700s)
    4. KKT-guided final polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            return True
        return False

    # Phase 1: Quick screening
    print("\n=== Phase 1: Quick screening (~150s) ===")
    p1_start = time.monotonic()
    p1_budget = min(150.0, (deadline - p1_start) * 0.12)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(8, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.85:
            break
        t_alloc = min(per_init_time, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break
        a_opt, val = run_fista(inits[i], S, t_alloc, rng,
                                beta_init=5.0, beta_rate=0.008, beta_max=3000)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping with Fourier mutations
    print("\n=== Phase 2: Basin hopping (~500s) ===")
    p2_start = time.monotonic()
    p2_budget = min(500.0, (deadline - p2_start) * 0.30)

    # Run basin hopping from the top candidates
    bh_starts = min(3, len(candidates))
    for i in range(bh_starts):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = (deadline - time.monotonic() - 300) * (0.6 / (bh_starts - i))
        t_alloc = min(t_alloc, p2_budget / (bh_starts - i) + 50)
        if t_alloc < 60:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = run_basin_hopping(candidates[i][1], S, t_alloc, rng)
        update_global(a_bh, val_bh)
        print("  BH %d done: obj=%.6f" % (i, 2.0 * n * val_bh / (a_bh.sum() ** 2)))

    # Phase 3: Adaptive polish
    print("\n=== Phase 3: Adaptive polish (~700s) ===")
    p3_start = time.monotonic()
    p3_budget = min(700.0, (deadline - p3_start) * 0.40)

    if global_a is not None and p3_budget > 60:
        # Run multiple adaptive polish sub-phases with different seeds
        n_sub = 3
        sub_budget = p3_budget / n_sub
        for sub in range(n_sub):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(sub_budget, deadline - time.monotonic() - 120)
            if t_alloc < 30:
                break
            print("  Adaptive sub-phase %d/%d, t=%.0fs" % (sub + 1, n_sub, t_alloc))
            a_p, val_p = run_adaptive_polish(global_a, S, t_alloc, rng)
            update_global(a_p, val_p)
            print("  After sub-phase %d: obj=%.6f" % (sub + 1, global_best))

    # Phase 4: KKT-guided final polish
    print("\n=== Phase 4: KKT final polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10

    if global_a is not None and p4_budget > 30:
        # Very high beta FISTA first
        seg1 = p4_budget * 0.5
        a_hb, val_hb = run_fista(global_a, S, seg1, rng,
                                  beta_init=500.0, beta_rate=0.001, beta_max=50000.0)
        update_global(a_hb, val_hb)
        print("  After high-beta FISTA: obj=%.6f" % global_best)

        # KKT polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            a_kkt, val_kkt = run_kkt_polish(global_a, S, remaining, rng)
            update_global(a_kkt, val_kkt)
            print("  After KKT polish: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
