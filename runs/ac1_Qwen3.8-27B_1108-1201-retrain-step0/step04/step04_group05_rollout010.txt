from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_value_grad(a, N, beta):
    """LSE-smoothed max of conv(a,a) and its gradient. Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta

    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max

        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def peak_directed_perturb(a, N, S, rng, magnitude):
    """Create a perturbation that specifically targets the worst peaks."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    # Use top peaks to create a targeted gradient
    top_k = min(50, len(c))
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    # Compute gradient from top peaks
    fsize = 1 << (3 * n).bit_length()
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-18:
        return a
    # Perturb opposite to the peak gradient, plus some random noise
    noise = rng.normal(0, 0.3 * magnitude / np.sqrt(n), n)
    perturb = -magnitude * grad / gnorm + noise
    a_new = project_simplex(a + perturb, S)
    return a_new


def spectral_step(a, N, top_k, alpha):
    """One spectral shaping step: reduce power at peak-corresponding frequencies."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def optimize_lse_block(a_init, n, S, N, time_budget, rng, beta_start=10.0,
                       beta_end=300.0, m_lbfgs=10, max_betas=None):
    """
    L-BFGS on LSE-smoothed objective with adaptive beta ramping.
    Returns (best_a, best_cmax).
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_start
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_value_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    last_best = best_cmax
    no_improve_iter = 0

    while time.monotonic() - start_time < time_budget:
        it += 1

        # Adaptive beta increase
        if it % 100 == 0:
            progress = min(1.0, it / (time_budget * 3))
            beta = beta_start * (beta_end / max(beta_start, 1e-6)) ** progress
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_value_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                no_improve_iter = 0
            else:
                no_improve_iter += 100

        # Trim L-BFGS history
        if it % 30 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # Get L-BFGS direction
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-16:
            direction = -g
            dir_norm = np.linalg.norm(g)
            if dir_norm < 1e-16:
                break
            direction = direction / dir_norm

        # Armijo backtracking line search
        alpha_ls = step_scale
        g_dir = np.dot(g, direction)
        c_current = c_max
        accepted = False
        a_new = a
        best_ls_alpha = 0.0

        for ls in range(20):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv_full(candidate, N)
            cm_ls = c_ls.max()
            # Armijo: accept if improvement is sufficient
            if cm_ls <= c_current - 1e-12 * abs(c_current) or cm_ls < best_cmax:
                a_new = candidate
                if cm_ls <= c_current:
                    best_ls_alpha = alpha_ls
                    accepted = True
                    break
            # Even if not Armijo, track best
            if cm_ls < c_current:
                a_new = candidate
                best_ls_alpha = alpha_ls
            alpha_ls *= 0.5

        if not accepted and best_ls_alpha == 0.0:
            # Stuck - try a smaller pure gradient step
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            alpha_ls = step_scale * 0.1
            for ls in range(5):
                candidate = project_simplex(a - alpha_ls * g / gnorm, S)
                c_ls, _ = compute_conv_full(candidate, N)
                if c_ls.max() < c_current:
                    a_new = candidate
                    best_ls_alpha = alpha_ls
                    accepted = True
                    break
                alpha_ls *= 0.5
            if not accepted:
                no_improve_iter += 1
                if no_improve_iter > 200:
                    # Restart with perturbation
                    a = peak_directed_perturb(a, N, S, rng, 0.01 * S / np.sqrt(n))
                    s_hist = []
                    y_hist = []
                    g, f_val, c, c_max = lse_value_grad(a, N, beta)
                    no_improve_iter = 0
                    continue

        # Update best
        c_new, _ = compute_conv_full(a_new, N)
        cmax_new = c_new.max()
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            no_improve_iter = 0

        # Update L-BFGS history
        g_new, _, c_nw, cmax_nw = lse_value_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)

        if sy > 1e-12 * max(ss, 1.0):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > 2 * m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        else:
            # Clear history on non-curvature step
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs // 2:]
                y_hist = y_hist[-m_lbfgs // 2:]

        a = a_new
        g = g_new
        c = c_nw
        c_max = cmax_nw

        # Periodic perturbation for diversity
        if it % 800 == 400 and rng.random() < 0.25:
            magnitude = 0.005 * S / np.sqrt(n)
            a = peak_directed_perturb(a, N, S, rng, magnitude)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_value_grad(a, N, beta)

    return best_a, best_cmax


def compute_conv_full(a, N):
    """Full convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def spectral_polish(a_init, n, S, N, time_budget, rng, top_k=15, alpha=0.15):
    """Spectral shaping polishing loop."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < time_budget:
        it += 1
        c = compute_conv_full(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
        excess = max(0.0, cur - 1.0) / max(cur, 1e-10)
        alpha_curr = alpha * (0.3 + 0.7 * min(1.0, excess * 5))
        a = spectral_step(a, N, top_k, alpha_curr)
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def nesterov_polish(a_init, n, S, N, time_budget, rng, top_k_init=10, top_k_max=200):
    """Nesterov accelerated top-k subgradient polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t_val = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < time_budget:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        c = compute_conv_full(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = 1 << (3 * n).bit_length()
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_val = t_new
        if it % 1500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t_val = 1.0

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_ramp(n, rng):
    """Monotonic ramp - pairs large with small at center."""
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * t + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_cosine(n, rng):
    """Cosine window shape."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_bell(n, rng):
    """Gaussian bell shape."""
    t = np.linspace(-1, 1, n)
    a = np.exp(-2 * t * t) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_chirp(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * t + 0.5 * t * t)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_binary(n, rng):
    a = (rng.random(n) > 0.5).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    # Enforce symmetry for real signal
    for i in range(1, N // 2 + 1):
        P[N - i] = P[i]
        phase[N - i] = -phase[i]
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.02


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses adaptive L-BFGS on LSE with peak-directed restarts,
    spectral polishing, Nesterov acceleration, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine primary n
    n_primary = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and given.size <= 32768:
            n_primary = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n_primary))
    N = 1 << (2 * n_primary - 1).bit_length()

    print("Primary: n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n_primary, S, N, budget_s))

    global_best = np.inf
    global_a = None
    global_n = n_primary

    def update_best(a_candidate, n_val, label=""):
        nonlocal global_best, global_a, global_n
        a_cand = safe_normalize(a_candidate, n_val, np.sqrt(2.0 * n_val))
        N_val = 1 << (2 * n_val - 1).bit_length()
        c = compute_conv_full(a_cand, N_val)
        sc = c.max()
        # Compute actual R value
        sum_a = a_cand.sum()
        R = 2.0 * n_val * sc / (sum_a ** 2)
        if R < global_best:
            global_best = R
            global_a = a_cand.copy()
            global_n = n_val
            if label:
                print("  [%s] n=%d R=%.6f (maxconv=%.6f)" % (label, n_val, R, sc))
        return R

    # --- Phase 1: Multi-start exploration with diverse inits ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(250.0, (deadline - t1) * 0.12)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n_primary:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n_primary, rng)))
    inits.append(("ramp", make_init_ramp(n_primary, rng)))
    inits.append(("cosine", make_init_cosine(n_primary, rng)))
    inits.append(("bell", make_init_bell(n_primary, rng)))
    inits.append(("chirp", make_init_chirp(n_primary, rng)))
    inits.append(("fourier", make_init_fourier(n_primary, rng, N)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        a_norm = safe_normalize(arr, n_primary, S)
        rem = max(1.0, per_init - 0.5)
        a_l, sc_l = optimize_lse_block(a_norm, n_primary, S, N, rem, rng,
                                        beta_start=5.0, beta_end=50.0, m_lbfgs=8)
        update_best(a_l, n_primary, name + "/quick")

    print("Phase 1 done. Best R: %.6f" % global_best)

    # --- Phase 2: Main iterative L-BFGS optimization ---
    print("=== Phase 2: Main L-BFGS optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(900.0, (deadline - t2) * 0.4)

    if global_a is None:
        global_a = np.ones(n_primary) * S / n_primary
        global_best = 2.0

    a_curr = global_a.copy()
    stall = 0

    while time.monotonic() < t2 + phase2_budget and time.monotonic() < deadline - 300:
        block_time = min(150.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        beta_start = max(10.0, 5.0 + 3.0 * stall)
        beta_end = min(400.0, 30.0 + 20.0 * stall)
        a_new, sc_new = optimize_lse_block(a_curr, n_primary, S, N, block_time, rng,
                                            beta_start=beta_start, beta_end=beta_end,
                                            m_lbfgs=12)
        R_new = 2.0 * n_primary * sc_new / (a_new.sum() ** 2)
        if R_new < global_best - 1e-10:
            global_best = R_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2] R=%.6f (beta %.0f->%.0f)" % (R_new, beta_start, beta_end))
        else:
            # Try peak-directed restart
            a_curr = peak_directed_perturb(a_new, N, S, rng, 0.015 * S / np.sqrt(n_primary))
            stall += 1
            if stall >= 4:
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n_primary),
                                                             n_primary), S)
                stall = 0

    # --- Phase 3: Spectral shaping refinement ---
    print("=== Phase 3: Spectral shaping ===")
    t3 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t3) * 0.15)

    a_sp, sc_sp = spectral_polish(global_a, n_primary, S, N, min(phase3_budget, 200.0),
                                   rng, top_k=12, alpha=0.12)
    update_best(a_sp, n_primary, "spectral")

    a_sp2, sc_sp2 = spectral_polish(global_a, n_primary, S, N,
                                     min(120.0, (deadline - time.monotonic()) * 0.3),
                                     rng, top_k=6, alpha=0.06)
    update_best(a_sp2, n_primary, "spectral_fine")

    # --- Phase 4: Nesterov accelerated polishing ---
    print("=== Phase 4: Nesterov polish ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 200  # reserve 200s for final
    if remaining4 > 30:
        a_ng, sc_ng = nesterov_polish(global_a, n_primary, S, N, min(remaining4, 400.0),
                                       rng, top_k_init=10, top_k_max=150)
        update_best(a_ng, n_primary, "nesterov")

    # --- Phase 5: Final high-beta L-BFGS polishing ---
    print("=== Phase 5: Final high-beta polish ===")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 30:
        a_fin, sc_fin = optimize_lse_block(global_a, n_primary, S, N, min(remaining5 - 10, 300.0),
                                            rng, beta_start=80.0, beta_end=600.0, m_lbfgs=12)
        update_best(a_fin, n_primary, "final_LBFGS")

        # One more round
        remaining5b = deadline - time.monotonic()
        if remaining5b > 30:
            a_fin2, sc_fin2 = optimize_lse_block(global_a, n_primary, S, N,
                                                  min(remaining5b - 5, 200.0),
                                                  rng, beta_start=150.0, beta_end=800.0,
                                                  m_lbfgs=10)
            update_best(a_fin2, n_primary, "final_LBFGS2")

    # --- Phase 6: Try different n values ---
    print("=== Phase 6: Multi-n exploration ===")
    remaining6 = deadline - time.monotonic()
    if remaining6 > 60:
        for n_try in [4096, 16384]:
            if remaining6 < 40:
                break
            S_try = np.sqrt(2.0 * n_try)
            N_try = 1 << (2 * n_try - 1).bit_length()
            print("  Trying n=%d..." % n_try)
            # Create initial from global_a by reshape or new init
            if n_try < n_primary:
                a_try = global_a[:n_try].copy()
            elif n_try > n_primary:
                a_try = np.zeros(n_try)
                a_try[:n_primary] = global_a
                # Interpolate
                t_old = np.linspace(0, 1, n_primary)
                t_new = np.linspace(0, 1, n_try)
                a_try[n_primary:] = 0  # keep zeros for new part initially
                a_try = np.interp(t_new, t_old, global_a)
            a_try = safe_normalize(a_try, n_try, S_try)
            time_for_try = min(remaining6 * 0.4, 200.0)
            a_res, sc_res = optimize_lse_block(a_try, n_try, S_try, N_try, time_for_try,
                                                rng, beta_start=10.0, beta_end=100.0,
                                                m_lbfgs=10)
            update_best(a_res, n_try, "n=%d" % n_try)
            remaining6 = deadline - time.monotonic()

    # --- Final verification and return ---
    print("=== Final result ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
            if not np.isfinite(ver):
                print("  Rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("  Eval error: %s" % str(e))
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, best R=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
