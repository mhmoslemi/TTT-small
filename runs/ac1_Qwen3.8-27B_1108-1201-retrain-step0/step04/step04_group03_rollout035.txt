Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive multiplicative 
weight method with LSE smoothing, FISTA polishing, and multi-scale exploration.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, n, S=None):
    if S is None:
        S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def mult_weight_update(a, n, S, beta, eta):
    """Multiplicative weight update: a[j] *= exp(-eta * g[j]), then normalize."""
    c = fft_convolve_self(a)
    g = lse_gradient(a, c, beta)
    # Clip g to avoid overflow
    g_max = np.max(np.abs(g))
    if g_max < 1e-15:

    scale = eta / g_max  # normalize so the max change is ~eta
    a_new = a * np.exp(-scale * (g - g.mean()))
    a_new = np.maximum(a_new, 1e-15)
    return normalize(a_new, n, S)

"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, n, S=None):
    if S is None:
        S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topK_gradient(a, c, K):
    n = len(a)
    size = 2 * n - 1
    top_K = np.argsort(c)[::-1][:K]
    w = np.zeros(size)
    w[top_K] = 1.0 / K
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_val(a, n, c=None):
    if c is None:
        c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_mult_weight(a_init, n, S, max_time, rng,
                    beta_init=5.0, beta_max=300.0, eta_base=0.15,
                    beta_growth=1.002, stuck_threshold=3000):

    a = normalize_seq(a_init, n, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    # Polyak averaging
    polyak_sum = np.zeros(n)
    polyak_count = 0
    polyak_start = 200

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        obj = objective_val(a, n, c)

        if obj < best:

            best_a = a.copy()

            polyak_sum += a
            polyak_count += 1

                a_avg = polyak_sum / polyak_count
                a_avg = project_simplex(a_avg, S)
                obj_avg = objective_val(a_avg, n)
                if obj_avg < best:
                    best = obj_avg
                    best_a = a_avg.copy()
                    print(f"    Polyak avg it={it}: obj={best:.6f}")

        # Compute gradient

        g = lse_gradient(a, c, beta)

        a = project_simplex(a_new, S)

        # Stuck detection and restart

            # Reduce beta to broaden focus
            beta = max(beta_init, beta * 0.4)
            # Structured perturbation: move mass from hot to cold
            c_now = fft_convolve_self(a)

            polyak_sum = np.zeros(n)

    a = normalize_seq(a_init, n, S)

    best_a = a.copy()

    last_improve = 0

        # Compute top-K subgradient
        g = topK_gradient(a, c, K)
        g_centered = g - g.mean()
        g_scale = np.max(np.abs(g_centered))

            a = a_new

            last_improve = it
            cur_lr = lr

def run_fista_high_beta(a_init, n, S, max_time, rng, beta_init=100.0, beta_max=5000.0):
    """High-beta FISTA for final polishing."""
    a = normalize_seq(a_init, n, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()

    best_a = a.copy()

    last_improve = 0

        if obj < best:
            best = obj
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve > 3000:
            ns = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, ns, n), S)
            t = 1.0
            last_improve = it
            beta = max(50.0, beta * 0.7)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

        t = t_new

def run_coord_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: try moving mass from one position to another."""
    a = normalize_seq(a_init, n, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        obj = objective_val(a, n, c)
        if obj < best:
            best = obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 2000:
            break

        # Find the position that contributes most to the max
        k_star = np.argmax(c)
        # Contribution of position j to c[k_star] is proportional to a[k_star - j]
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for j in range(lo, hi + 1):
            contrib[j] = a[k_star - j]

        # Find the position with highest contribution
        j_src = np.argmax(contrib)
        # Find candidate destination: position with lowest contribution
        # to a "cold" lag
        cold_k = np.argmin(c)
        contrib_cold = np.zeros(n)
        lo2 = max(0, cold_k - n + 1)
        hi2 = min(cold_k, n - 1)
        for j in range(lo2, hi2 + 1):
            contrib_cold[j] = a[cold_k - j]
        j_dst = np.argmin(contrib_cold)

        if j_src == j_dst:
            continue

        # Try moving a small amount of mass
        for frac in [0.001, 0.005, 0.01, 0.02]:
            amt = a[j_src] * frac
            if amt < 1e-12:
                continue
            a_test = a.copy()
            a_test[j_src] = a[j_src] - amt
            a_test[j_dst] = a[j_dst] + amt
            if a_test[j_src] < 0:
                continue
            obj_test = objective_val(a_test, n)
            if obj_test < obj:
                a = a_test
                break

    return best_a, best


def freq_domain_init(n, rng):
    """Generate initialization using frequency domain shaping."""
    # Target: make |FFT(a)|^2 as close to a Dirichlet kernel as possible
    # (which corresponds to a flat autocorrelation)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    # Target spectrum: DFT of a rectangular window of length size
    # rect[k] = 1 for k=0..size-1, 0 otherwise
    # FFT(rect)[m] = sum_{k=0}^{size-1} exp(2*pi*i*m*k/fsize)
    # This is a Dirichlet kernel
    m = np.arange(fsize)
    target_spec = np.abs(np.sum(np.exp(2j * np.pi * m[None, :] * np.arange(size)[:, None] / fsize)))
    target_spec = np.abs(target_spec)  # shape (size, fsize) -> take first row? No...
    # Actually let's just compute FFT of rect
    rect = np.zeros(fsize)
    rect[:size] = 1.0
    target_A_sq = np.abs(fft(rect)) ** 2
    target_A = np.sqrt(target_A_sq)

    # Start with random phase
    phase = np.exp(2j * np.pi * rng.random(fsize))
    # Ensure conjugate symmetry for real signal
    phase[0] = 1.0
    if fsize % 2 == 0:
        phase[fsize // 2] = 1.0
    for i in range(1, (fsize + 1) // 2):
        phase[fsize - i] = np.conj(phase[i])

    A = target_A * phase
    a = np.real(ifft(A))[:n]
    a = np.abs(a)
    return a


def cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def gaussian_init(n):

            print(f"    Phase 1b result: {global_best:.6f}")

    # Phase 2: FISTA high-beta polishing
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 120:
        t_budget = remaining * 0.25
        print(f"  Phase 2: FISTA high-beta, budget={t_budget:.0f}s")
        a_fista, obj_f = run_fista_high_beta(global_a, n_main, S_main, t_budget, rng,
                                              beta_init=100.0, beta_max=5000.0)
        if obj_f < global_best:
            global_best = obj_f
            global_a = a_fista.copy()
            print(f"    Phase 2 result: {global_best:.6f}")

    # Phase 3: More MW with different hyperparameters
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 90:
        t_budget = remaining * 0.25
        print(f"  Phase 3: MW with different params, budget={t_budget:.0f}s")
        # Try higher initial beta (sharper focus)
        a_mw2, obj_mw2 = run_mult_weight(global_a, n_main, S_main, t_budget * 0.5, rng,
                                          beta_init=20.0, beta_max=500.0, eta_base=0.1,
                                          beta_growth=1.003, stuck_threshold=1500)
        if obj_mw2 < global_best:
            global_best = obj_mw2
            global_a = a_mw2.copy()
            print(f"    Phase 3a result: {global_best:.6f}")

        # Try lower initial beta (broader focus)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a_mw3, obj_mw3 = run_mult_weight(global_a, n_main, S_main, remaining2 * 0.7, rng,
                                              beta_init=2.0, beta_max=100.0, eta_base=0.2,
                                              beta_growth=1.005, stuck_threshold=2000)
            if obj_mw3 < global_best:
                global_best = obj_mw3
                global_a = a_mw3.copy()
                print(f"    Phase 3b result: {global_best:.6f}")

    # Phase 4: Coordinate descent refinement
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        t_budget = min(remaining * 0.2, 120.0)
        print(f"  Phase 4: Coord descent, budget={t_budget:.0f}s")
        a_cd, obj_cd = run_coord_descent(global_a, n_main, S_main, t_budget, rng)
        if obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
            print(f"    Phase 4 result: {global_best:.6f}")

    # Phase 5: Try different length
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 90:
        alt_n = 16384
        S_alt = np.sqrt(2.0 * alt_n)
        t_budget = remaining * 0.4
        print(f"  Phase 5: Alt length n={alt_n}, budget={t_budget:.0f}s")
        alt_init = interpolated_init(alt_n, global_a)
        alt_init = normalize_seq(alt_init, alt_n, S_alt)
        a_alt, obj_alt = run_mult_weight(alt_init, alt_n, S_alt, t_budget, rng,
                                          beta_init=5.0, beta_max=200.0, eta_base=0.15)
        if obj_alt < global_best:
            global_best = obj_alt
            global_a = a_alt.copy()
            print(f"    Phase 5 result: {global_best:.6f}")
        else:
            print(f"    Phase 5: {obj_alt:.6f} (no improvement)")

    # Phase 6: Final intensive FISTA polish
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        t_budget = remaining * 0.8
        print(f"  Phase 6: Final FISTA polish, n={n_final}, budget={t_budget:.0f}s")
        a_final, obj_final = run_fista_high_beta(global_a, n_final, S_final, t_budget, rng,
                                                  beta_init=200.0, beta_max=8000.0)
        if obj_final < global_best:
            global_best = obj_final
            global_a = a_final.copy()
            print(f"    Phase 6 result: {global_best:.6f}")

    # Final verification
    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    final_list = [float(x) for x in global_a]

    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(final_list)
        print(f"Final verification: {ver:.6f}")
        if not np.isfinite(ver):
            raise ValueError("Invalid")
    except Exception:
        print("Verification failed, using fallback")
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]

    # Ensure sum > 0.01
    if sum(final_list) < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_time
    print(f"Done in {elapsed:.0f}s. Returning sequence of length {len(final_list)}, score={global_best:.6f}")
    return final_list
