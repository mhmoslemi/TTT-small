def lbfgs_recursion(s_list, y_list, g, m=10):
    """Standard L-BFGS two-loop recursion."""
    n = len(g)
    s_list = s_list[-m:]
    y_list = y_list[-m:]
    m_actual = len(s_list)
    if m_actual == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            return np.zeros(n)
        return -g / gn
    
    q = g.copy()
    alphas = np.zeros(m_actual)
    for i in range(m_actual - 1, -1, -1):
        si = s_list[i]
        yi = y_list[i]
        yy = np.dot(yi, yi)
        if yy > 1e-30:
            alphas[i] = np.dot(si, q) / yy
            q = q - alphas[i] * yi
    
    # Initial Hessian approximation
    if m_actual > 0:
        s0 = s_list[-1]
        y0 = y_list[-1]
        sy = np.dot(s0, y0)
        yy0 = np.dot(y0, y0)
        gamma = sy / yy0 if yy0 > 1e-30 else 1.0
        r = gamma * q
    else:
        r = q
    
    for i in range(m_actual):
        si = s_list[i]
        yi = y_list[i]
        yy = np.dot(yi, yi)
        if yy > 1e-30:
            beta_i = np.dot(si, yi) / yy
            r = r + (alphas[i] - beta_i) * yi
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def spectral_shaping(a, N, top_k, alpha, S, n):
    """One step of spectral shaping."""
    c, A = convolve_self(a, N)
    P = np.abs(A) ** 2
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)

def peak_subgradient_step(a, N, S, n, top_k, step_size):
    """Subgradient step targeting top-k peaks."""
    c, A = convolve_self(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    # Gradient of weighted sum of peaks
    grad = np.zeros(n)
    for i, j in enumerate(top_idx):
        j = int(j)
        w = weights[i]
        for k in range(n):
            m = j - k
            if 0 <= m < n:
                grad[k] += 2.0 * w * a[m]
    a_new = project_simplex(a - step_size * grad, S)
    return a_new

def peak_subgradient_step_fft(a, N, S, n, top_k, step_size):
    """FFT-based subgradient step targeting top-k peaks."""
    c, A = convolve_self(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    # Weighted sum of peaks: sum_i w_i * c[j_i]
    # Its gradient: sum_i w_i * 2*a[j_i - k] = 2 * sum_i w_i * a[j_i - k]
    # = 2 * (w_delta cross-correlated with a) where w_delta is a sparse vector
    # Let's build w_delta and use FFT
    w_delta = np.zeros(N)
    for i, j in enumerate(top_idx):
        w_delta[int(j)] = weights[i]
    fsize = 1 << (3 * n - 2).bit_length()
    Wd = fft(w_delta, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(Wd * np.conj(Af)))[:n]
    a_new = project_simplex(a - step_size * grad, S)
    return a_new

def make_random_init(n, rng, S):
    return project_simplex(rng.random(n) * S, S)

def make_chirp_init(n, rng, S):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2*np.pi*7*tt + 0.2*np.sin(2*np.pi*3*tt))
    a = np.abs(a) + 0.01
    return project_simplex(a * S / a.sum(), S)

def make_fourier_init(n, rng, S, N):
    # Random phase with flat-ish magnitude
    P = np.ones(N) * (S / np.sqrt(N)) ** 2
    P[0] = S ** 2
    noise = rng.random(N) * 0.5 + 0.25
    P = P * noise
    phase = rng.uniform(0, 2*np.pi, N)
    # Ensure conjugate symmetry for real signal
    phase[0] = 0
    if N % 2 == 0:
        phase[N//2] = 0
    for k in range(1, N//2 + 1):
        if k != N - k:
            phase[N-k] = -phase[k] if k < N-k else phase[k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.001
    return project_simplex(a * S / a.sum(), S)

import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_self(a, N):
    """Compute full self-convolution of a via FFT. Returns (c of length 2n-1, A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """
    Compute LSE-smoothed max of conv(a,a) and its gradient.
    Returns (gradient of length n, lse_value).
    """
    n = len(a)
    c, A = convolve_self(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    # grad[k] = 2 * sum_j w[j] * a[j-k] (for valid j-k)
    # = 2 * cross_correlation(w, a)[k]
    # = 2 * ifft(fft(w) * conj(fft(a)))[k]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val


def lbfgs_direction(s_hist, y_hist, g, m_max=12):
    """L-BFGS two-loop recursion. Returns search direction."""
    n = len(g)
    m = min(m_max, len(s_hist))
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            return np.zeros(n)
        return -g / gn
    s_hist = s_hist[-m:]
    y_hist = y_hist[-m:]
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy > 1e-30:
            alphas[i] = np.dot(si, q) / yy
            q = q - alphas[i] * yi
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    sy_last = np.dot(s_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-10, sy_last / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy > 1e-30:
            beta_i = np.dot(si, yi) / yy
            r = r + (alphas[i] - beta_i) * yi
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def mueller_peak_reduction(a, N, S, n, max_iters, rng, threshold_frac=0.02):
    """
    Mueller-like iterative peak reduction.
    At each step: identify peaks above threshold, compute correction, apply.
    Returns (best_a, best_cmax).
    """
    a = a.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    for _ in range(max_iters):
        it += 1
        c, A = convolve_self(a, N)
        c_max = c.max()
        c_mean = c.mean()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        # Threshold: peaks that are significantly above mean
        threshold = c_mean + threshold_frac * c_max
        peak_mask = c > threshold
        n_peaks = peak_mask.sum()
        if n_peaks == 0:
            break

        # Compute weighted gradient from peaks
        excess = np.where(peak_mask, c - threshold, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            break
        w = excess / total_excess

        # Gradient: 2 * cross_correlation(w, a)
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        # Step size: adaptive
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        alpha = 2.0 * S / (n * max(beta_for_alpha(a, c), 5.0))

        # Apply with line search
        best_alpha = 0.0
        best_trial_cmax = c_max
        for trial in range(15):
            a_trial = project_simplex(a - alpha * grad / gnorm, S)
            c_trial, _ = convolve_self(a_trial, N)
            if c_trial.max() < best_trial_cmax:
                best_trial_cmax = c_trial.max()
                best_alpha = alpha
            if c_trial.max() < c_max * 0.999:
                break
            alpha *= 0.6

        if best_alpha == 0.0:
            break

        a = project_simplex(a - best_alpha * grad / gnorm, S)

        # Periodic perturbation
        if it % 200 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

        if time.monotonic() - start_time > max_iters * 0.05:
            break

    return best_a, best_cmax


def beta_for_alpha(a, c):
    """Estimate effective beta from current peak distribution."""
    c_max = c.max()
    c_sorted = np.sort(c)[::-1]
    top10 = c_sorted[:min(10, len(c_sorted))]
    if c_max < 1e-10:
        return 10.0
    ratio = top10.sum() / max(c.sum(), 1e-10)
    return 50.0 / max(ratio, 0.01)


def spectral_shaping(a, N, S, n, top_k, alpha, use_multi_scale=False):
    """
    Spectral shaping: reduce peak contribution in frequency domain.
    Modifies |A|^2 at frequencies that contribute to time-domain peaks.
    """
    c, A = convolve_self(a, N)
    P = np.abs(A) ** 2
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total

    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)

    if use_multi_scale:
        # Also apply a broader smoothing
        P_smooth = np.maximum(P * (1.0 - 0.3 * alpha * cos_eff), 0.0)
        P_new = 0.7 * P_new + 0.3 * P_smooth

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def spectral_block(a_init, N, S, n, max_time, rng, top_k=10, alpha_init=0.12):
    """Iterative spectral shaping with adaptive parameters."""

def subgradient_polish(a_init, N, S, n, max_time, rng, top_k=8):
    """Subgradient method targeting top-k peaks with adaptive step."""
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = convolve_self(a, N)
        c_max = c.max()
        if c_max < best_cmax - 1e-10:
            best_cmax = c_max
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Build sparse weight vector
        w_delta = np.zeros(N)
        for i, j in enumerate(top_idx):
            w_delta[int(j)] = weights[i]

        fsize = 1 << (3 * n - 2).bit_length()
        Wd = fft(w_delta, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(Wd * np.conj(Af)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break

        # Adaptive step: Nesterov-like
        step = 2.0 * S / (n * (1.0 + 0.01 * it)) * step_scale
        a_new = project_simplex(a - step * grad / gnorm, S)

        c_new, _ = convolve_self(a_new, N)
        if c_new.max() < c_max:
            a = a_new
            step_scale = min(2.0, step_scale * 1.01)
        else:
            # Rollback and reduce step
            step_scale = max(0.1, step_scale * 0.7)
            if step_scale < 0.15:
                break

        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            step_scale = 1.0

    return best_a, best_cmax


def make_init_random(n, rng, S):
    """Random non-negative initialization."""
    a = rng.random(n) + 0.01
    return project_simplex(a * S / a.sum(), S)


def make_init_chirp(n, rng, S):
    """Chirp-like initialization with multiple frequencies."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for freq, amp in [(3, 0.2), (7, 0.15), (13, 0.1), (21, 0.05)]:
        a += amp * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    a = np.abs(a) + 0.01
    return project_simplex(a * S / a.sum(), S)


def make_init_fourier(n, rng, S, N):
    """Fourier-domain initialization with controlled spectrum."""
    P = np.ones(N) * (S / np.sqrt(N)) ** 2
    P[0] = S ** 2
    # Log-uniform magnitude perturbation
    noise = 0.5 + 0.5 * rng.random(N)
    P = P * noise
    P = np.maximum(P, 1e-10)
    # Random phase with conjugate symmetry
    phase = rng.uniform(0, 2 * np.pi, N)
    phase[0] = 0.0
    for k in range(1, (N // 2) + 1):
        if k <= N - 1 - k:
            phase[N - k] = -phase[k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.005
    return project_simplex(a * S / a.sum(), S)


def make_init_binary(n, rng, S):
    """Binary-like initialization."""
    a = (rng.random(n) > 0.35).astype(np.float64)
    a = a * (0.5 + 0.5 * rng.random(n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return project_simplex(a * S / s, S)


def coarsen_and_upsample(a, n_target, method='avg'):
    """Coarsen a to shorter length, then upsample back."""
    n_orig = len(a)
    if n_orig == n_target:
        return a.copy()
    if n_orig > n_target:
        # Coarsen by averaging
        n_chunks = n_orig // n_target
        if n_chunks == 0:
            return a.copy()
        a_c = a[:n_chunks * n_target].reshape(n_chunks, n_target).mean(axis=0)
    else:
        # Upsample by repetition
        rep = n_target // n_orig + 1
        a_c = np.tile(a, rep)[:n_target]
    return a_c

        a_curr = best_phase1_a.copy() if best_phase1_a is not None else np.ones(n) * S / n
        stall = 0

        while time.monotonic() - t_phase2 < p2_budget:
            block_time = min(150.0, p2_budget - (time.monotonic() - t_phase2))
            if block_time < 10:
                break
            beta_start = 8.0 + 4.0 * stall
            beta_end = min(500.0, 40.0 * (1 + 0.3 * stall))
            a_new, sc_new = lbfgs_optimize(a_curr, n, S, N, block_time, rng,
                                            beta_start=beta_start,
                                            beta_end=beta_end, m_lbfgs=12,
                                            verbose=True)
            update_best(a_new, n, "P2-LBFGS")
            if sc_new < nonlocal_best['val'] - 1e-8:
                a_curr = a_new.copy()
                stall = 0
            else:
                a_curr = a_new.copy()
                stall += 1
                if stall >= 2:
                    a_curr = project_simplex(
                        a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                    stall = 0

        # --- Phase 3: Mueller peak reduction ---
        t_phase3 = time.monotonic()
        p3_budget = time_alloc * 0.15
        a_mu, sc_mu = mueller_peak_reduction(a_curr, N, S, n,
                                              max_iters=50000,
                                              rng=rng,
                                              threshold_frac=0.015)
        update_best(a_mu, n, "Mueller")
        if sc_mu < nonlocal_best['val']:
            a_curr = a_mu.copy()

        # --- Phase 4: Spectral shaping ---
        t_phase4 = time.monotonic()
        p4_budget = time_alloc * 0.15
        a_sp, sc_sp = spectral_block(a_curr, N, S, n, p4_budget, rng,
                                      top_k=8, alpha_init=0.10)
        update_best(a_sp, n, "Spectral")
        if sc_sp < nonlocal_best['val']:
            a_curr = a_sp.copy()

        # Fine spectral
        a_sp2, sc_sp2 = spectral_block(a_curr, N, S, n, min(100.0, p4_budget * 0.5),
                                        rng, top_k=4, alpha_init=0.05)
        update_best(a_sp2, n, "Spectral-fine")
        if sc_sp2 < nonlocal_best['val']:
            a_curr = a_sp2.copy()

        # --- Phase 5: Subgradient polish ---
        t_phase5 = time.monotonic()
        p5_budget = time_alloc * 0.10
        a_sg, sc_sg = subgradient_polish(a_curr, N, S, n, p5_budget, rng, top_k=6)
        update_best(a_sg, n, "SubGrad")
        if sc_sg < nonlocal_best['val']:
            a_curr = a_sg.copy()

        # --- Phase 6: Final high-beta L-BFGS ---
        remaining_for_n = deadline - time.monotonic()
        if remaining_for_n > 20:
            p6_budget = min(remaining_for_n - 10, 200.0)
            a_fin, sc_fin = lbfgs_optimize(a_curr, n, S, N, p6_budget, rng,
                                            beta_start=50.0, beta_max=800.0,
                                            m_lbfgs=15, verbose=True)
            update_best(a_fin, n, "FinalLBFGS")

    # --- Final verification ---
    if nonlocal_best['arr'] is not None:
        final_a = nonlocal_best['arr']
        final_n = nonlocal_best['n']
    else:
        final_a = np.ones(256, dtype=np.float64)
        final_n = 256

    final_a = np.clip(final_a, 0.0, 1000.0)
    final_list = [float(x) for x in final_a]

    # Verify with actual evaluator
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\n=== FINAL: n=%d, evaluated=%.6f ===" % (final_n, ver))
    else:
        print("\n=== WARNING: evaluation failed, using fallback ===")
        final_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in final_a]

    print("Returning sequence of length %d" % len(final_list))
    return final_list
