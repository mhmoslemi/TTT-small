def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    # = 2 * (a * w_flipped)[2n-2-m] where w_flipped = w[::-1]
    # because (a * w_flipped)[k] = sum_j a[j] * w_flipped[k-j] = sum_j a[j] * w[2n-2-k+j]
    # For k = 2n-2-m: sum_j a[j] * w[m+j]. Yes!
    
    w_flipped = w_full[::-1]
    M = 1 << (2*n - 1).bit_length()  # M >= 2n-1, same as for fft_convolve_self
    # Actually we need M >= n + (2n-1) - 1 = 3n-2 for the full linear convolution
    M2 = 1 << (3*n - 2).bit_length()
    A = fft(a, M2)
    W = fft(w_flipped, M2)
    conv_result = np.real(ifft(A * W))
    # conv_result[k] = sum_j a[j] * w_flipped[k-j] = sum_j a[j] * w[2n-2-(k-j)] = sum_j a[j] * w[2n-2-k+j]
    # We want sum_j a[j] * w[j+m], so 2n-2-k = m, i.e., k = 2n-2-m
    # grad[m] = 2 * conv_result[2n-2-m]
    grad = 2.0 * conv_result[2*n-2: n-1: -1]  # m=0: index 2n-2, m=1: index 2n-3, ..., m=n-1: index n-1
    
    # Verify: grad has length n
    assert len(grad) == n, f"grad length {len(grad)} != n={n}"
    
    return grad, c, weights


def spectral_flatten(a, n, N, target_ratio=0.8, rng=None):
    """
    Spectral flattening: equalize |A[m]|^2 towards a target level.
    The target is chosen so that the resulting conv is more uniform.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2*n-1]
    c_max = c.max()
    c_mean = c.mean()
    
    # Target: make P more uniform
    # Current P has P[0] = S^2 (the DC component)
    # We want P[m] for m!=0 to be closer to c_mean * N (approximate)
    
    # Water-filling approach: reduce the peaks in P
    # Target level: geometric mean of P (excluding DC)
    P_nonzero = P[1:N//2+1]  # Use symmetry
    target = np.median(P_nonzero) * target_ratio
    
    # Modify P: reduce values above target, boost values below
    P_new = P.copy()
    P_new[1:] = np.where(P[1:] > target, target, P[1:])
    # Keep DC
    P_new[0] = P[0]
    
    # Reconstruct with modified magnitude, keep original phase
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    
    return a_new


def annealed_smoothed_descent(a_init, n, N, max_time, rng, 
                               temp_start=1.0, temp_end=50.0,
                               n_top=50, step_scale=1.0):
    """
    Annealed smoothed gradient descent with simplex projection.
    Temperature increases from temp_start to temp_end.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    stall = 0
    
    # Total iterations estimate
    # Each iteration: one FFT conv + one FFT gradient ≈ 3 FFTs of size N or M2
    # For n=8192, N=16384, M2=32768 or 65536
    # Each FFT takes ~0.1ms, so each iteration ~1ms
    # In 60s: ~60000 iterations
    
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal temperature
        progress = min(1.0, it / max(1, int(max_time * 50)))  # ~50 iters per second
        temp = temp_start + (temp_end - temp_start) * (progress ** 0.5)
        
        # Compute gradient
        grad, c, weights = smoothed_gradient_fft(a, temp, n_top)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            stall = 0
            if it % 200 == 0:
                print(f"    ASD iter {it}: cmax={c_max:.6f}, temp={temp:.2f}")
        else:
            stall += 1
        
        # Step size: use a simple adaptive rule
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        a_norm = np.linalg.norm(a)
        step = step_scale * S * S / (n * gnorm) * (1.0 + 0.1 / (1 + 0.01 * it))
        
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        
        # Check if we made progress
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Try half step
            a_new2 = a - 0.5 * step * grad
            a_new2 = project_simplex(a_new2, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < c_max:
                a = a_new2
            else:
                a = a_new  # Accept anyway for exploration
                stall += 1
        
        # Random perturbation for exploration
        if stall > 50 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            stall = 0
        
        if best_cmax <= 1.45:
            break
    
    return best_a, best_cmax


def lbfgs_smoothed(a_init, n, max_time, rng, temp_start=5.0, temp_end=100.0):
    """
    L-BFGS optimization of the smoothed objective with simplex constraint.
    Uses the softmax-weighted gradient.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    
    # L-BFGS memory
    m_mem = 10
    s_hist = []  # step history
    y_hist = []  # gradient difference history
    
    while time.monotonic() - start < max_time:
        it += 1
        temp = temp_start + (temp_end - temp_start) * min(1.0, it / 5000)
        
        # Compute gradient
        grad, c, _ = smoothed_gradient_fft(a, temp, n_top=100)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # L-BFGS two-loop recursion for direction
        p = -grad.copy()
        q = p.copy()
        rho = []
        alpha = []
        
        for i in range(len(s_hist) - 1, -1, -1):
            s_i = s_hist[i]
            y_i = y_hist[i]
            rho_i = 1.0 / np.dot(y_i, s_i) if np.dot(y_i, s_i) > 1e-15 else 0.0
            rho.append(rho_i)
            alpha_i = rho_i * np.dot(s_i, q)
            alpha.append(alpha_i)
            q = q - alpha_i * y_i
        
        gamma = np.dot(s_hist[-1], y_hist[-1]) / np.dot(y_hist[-1], y_hist[-1]) if s_hist and np.dot(y_hist[-1], y_hist[-1]) > 1e-15 else 1.0 / (np.linalg.norm(grad) + 1e-15)
        r = gamma * q
        
        rho.reverse()
        alpha.reverse()
        for i in range(len(s_hist)):
            beta_i = rho[i] * np.dot(y_hist[i], r)
            r = r + (alpha[i] - beta_i) * s_hist[i]
        
        p = r  # Search direction (should be -gradient direction scaled)
        
        # Ensure descent direction
        if np.dot(p, grad) > 0:
            p = -grad
        
        # Line search (backtracking)
        step = 1.0 * S / np.linalg.norm(p)
        best_obj = c_max
        for ls in range(10):
            a_trial = project_simplex(a - step * p, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_obj * 1.001:
                s_new = a_trial - a
                grad_new, c_new, _ = smoothed_gradient_fft(a_trial, temp, n_top=100)
                y_new = grad_new - grad
                if np.dot(s_new, y_new) > 1e-15:
                    s_hist.append(s_new)
                    y_hist.append(y_new)
                    if len(s_hist) > m_mem:
                        s_hist.pop(0)
                        y_hist.pop(0)
                a = a_trial
                break
            step *= 0.5
        else:
            a = project_simplex(a - step * p, S)
        
        if it % 100 == 0:
            print(f"    LBF iter {it}: cmax={best_cmax:.6f}, temp={temp:.1f}")
        
        if best_cmax <= 1.45:
            break
    
    return best_a, best_cmax


def momentum_smoothed_descent(a_init, n, max_time, rng,
                               temp_start=2.0, temp_end=50.0,
                               momentum=0.9, n_top=50):
    """
    Momentum-based smoothed gradient descent with simplex projection.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        temp = temp_start + (temp_end - temp_start) * (1 - np.exp(-0.001 * it))
        
        grad, c, _ = smoothed_gradient_fft(a, temp, n_top)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            stall = 0
            if it % 500 == 0:
                print(f"    MSD iter {it}: cmax={c_max:.6f}, temp={temp:.2f}")
        else:
            stall += 1
        
        # Momentum update
        v = momentum * v - 0.01 * grad / (np.linalg.norm(grad) + 1e-15)
        v /= (np.linalg.norm(v) + 1e-15)
        
        step_size = 0.1 * S / np.sqrt(n)
        a_new = a + step_size * v
        a_new = project_simplex(a_new, S)
        
        # Only accept if not too bad
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c_max * 1.05:
            a = a_new
        
        # Perturbation
        if stall > 100 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            v *= 0.5
            stall = 0
    
    return best_a, best_cmax


def multi_lag_subgradient(a_init, n, max_time, rng, top_k=20):
    """
    Subgradient method using top-k active lags with adaptive weights.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            if it % 100 == 0:
                print(f"    MLS iter {it}: cmax={c_max:.6f}")
        
        # Get top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        # Weights: exponential of excess
        excess = top_vals - c_max
        w = np.exp(excess / max(c_max * 0.1, 1e-10))
        w /= w.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi+1] += w[i] * 2.0 * a[k-hi:k-lo:-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = 2.0 * S / (n * gnorm)
        a_new = project_simplex(a - step * grad, S)
        
        # Accept or reject
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Halve step
            a_new = project_simplex(a - 0.5 * step * grad, S)
            c_new = fft_convolve_self(a_new)
            a = a_new  # Accept anyway to explore
    
    return best_a, best_cmax


def spectral_preconditioner(a, n, N, n_iter=50, rng=None):
    """
    Spectral pre-conditioning: iteratively flatten the spectrum.
    """
    S = a.sum()
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_cmax = fft_convolve_self(a_curr).max()
    
    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a_curr
        A = fft(a_pad)
        P = np.abs(A) ** 2
        
        # Target: flatten P (excluding DC)
        P_target = np.full(N, P[0] / (2*n - 1))  # Rough target
        # Blend: P_new = (1-t)*P + t*P_target
        t = 0.3
        P[1:] = (1-t) * P[1:] + t * P_target[1:]
        
        # Keep phase
        A_new = np.sqrt(P) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        c = fft_convolve_self(a_new)
        if c.max() < best_cmax:
            best_cmax = c.max()
            best_a = a_new.copy()
            a_curr = a_new
    
    return best_a, best_cmax


def structured_init(n, rng, method="sin"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2*np.pi*3*t) + 0.2 * np.sin(2*np.pi*7*t) \
            + 0.1 * np.sin(2*np.pi*13*t) + 0.05 * rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2*np.pi*(2*t + 3*t*t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = 0.3 + 0.5 * np.exp(-0.5*((t-0.3)*4)**2) + 0.2 * np.exp(-0.5*((t-0.7)*4)**2)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.1 * rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = 64
        piece_len = n // n_pieces
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, rng.random(n - len(a)) * 1.0 + 0.3])
        a = a[:n]
    elif method == "white":
        a = rng.random(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, 2.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

Wait, I realize there might be an issue with my `smoothed_gradient_fft` function. Let me double-check the cross-correlation computation.

We want: grad[m] = 2 * sum_{j=0}^{n-1} a[j] * w[j+m] for m = 0, 1, ..., n-1

where w has length 2n-1 (indices 0 to 2n-2).

Let's define:
- x[j] = a[j] for j = 0, ..., n-1
- y[k] = w[k] for k = 0, ..., 2n-2

We want: g[m] = sum_j x[j] * y[j+m] for m = 0, ..., n-1

This is the cross-correlation of x and y at lag m: (x ⋆ y)[m] = sum_j x[j] * y[j+m]
