def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()

        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def fft_convolve_self(a):

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):

    if n == 0:
        return v

    s = result.sum()

def lse_subgradient(a, c, beta):
    """
    Compute subgradient of the LSE approximation to max(c).
    Returns: (grad, lse_value)
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (where valid)
    # = 2 * cross_correlation(w, a)[i]
    # cross_corr(w, a)[i] = sum_j w[i+j] * a[j] = ifft(fft(w) * conj(fft(a)))[:n]
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += a[j - i_vals]
    return grad / k

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Adaptive LSE (soft-max) subgradient with backtracking line search,
momentum smoothing, frequency-domain perturbations, multi-length search,
and iterative reweighting of active constraints.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = tc[rho - 1]
    else:
        theta = cumsum[-1] / n
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-30 and abs(s - S) > 1e-10 * S:
        result = result * (S / s)
        result = np.maximum(result, 0.0)
    return result


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_subgradient(a, c, beta):
    """
    Compute subgradient of LSE approximation to max(c).
    Returns (grad, lse_value).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += a[j - i_vals]
    return grad / k


def reweight_subgradient(a, c, active_weights):
    """
    Subgradient using reweighted active constraints.
    active_weights: array same length as c, summing to 1 over active set.
    """
    n = len(a)
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(active_weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def update_active_weights(c, old_weights, temperature):
    """Update weights on active constraints using softmax with temperature."""
    cm = c.max()
    e = np.exp(temperature * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros_like(c)
    return e / tot


def freq_perturb(a, rng, scale):
    """Perturb in frequency domain maintaining real-valued output."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    A_scale = np.abs(A).max() + 1e-10
    noise = (rng.normal(0, 1, fsize) + 1j * rng.normal(0, 1, fsize)) * scale * A_scale
    noise[0] = np.real(noise[0])

            s_pol2_real = compute_objective(a_pol2)
            if s_pol2_real < global_best:
                global_best = s_pol2_real
                global_a = a_pol2.copy()
                print("  Top-k polish: %.7f" % s_pol2_real)

    # === Final verification ===
    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = compute_objective(global_a)
        print("Using fallback (uniform n=256): obj=%.4f" % global_best)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nVerification: %.7f (length=%d)" % (ver, len(final_list)))
        if ver < global_best:
            global_best = ver
    else:
        print("\nWARNING: evaluate_sequence returned inf! Using safe fallback.")
        n_fb = 256
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verify: %s" % ver)

    print("\nFINAL: obj=%.7f, length=%d" % (global_best, len(final_list)))
    return final_list
