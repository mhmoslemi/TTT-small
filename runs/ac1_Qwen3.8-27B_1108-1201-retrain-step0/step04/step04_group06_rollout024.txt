import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_cross_convolve(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def build_lp_matrix(f, selected_lags, n):
    """
    Build the LP constraint matrix A where A[k, j] = f[k-j] if 0 <= k-j < n.
    Only for selected lags.
    """
    K = len(selected_lags)
    A = np.zeros((K, n))
    for i, k in enumerate(selected_lags):
        # A[i, j] = f[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            A[i, indices] = f[k - indices]
    return A

def solve_lp_direction(f, M, selected_lags, n):
    """
    Solve: max 1^T g s.t. A g <= M * 1, g >= 0
    Returns g (the LP solution).
    """
    K = len(selected_lags)
    A = build_lp_matrix(f, selected_lags, n)
    c_obj = -np.ones(n)  # minimize -sum(g)
    b = M * np.ones(K)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def find_best_step(f, g, t_max=1.0):
    """
    Find t in [0, t_max] that minimizes max_k ((1-t)f + t*g) * ((1-t)f + t*g)
    """
    f2 = fft_convolve_self(f)
    fg = fft_cross_convolve(f, g)
    g2 = fft_convolve_self(g)
    
    # (h*h)_k = (1-t)^2 f2_k + 2t(1-t) fg_k + t^2 g2_k
    # = f2_k + t(2(fg_k - f2_k) + t(g2_k - 2fg_k + f2_k))
    
    def score(t):
        h2 = (1-t)**2 * f2 + 2*t*(1-t) * fg + t**2 * g2
        return h2.max()
    
    # Ternary search
    lo, hi = 0.0, t_max
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if score(m1) < score(m2):
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    return t_opt, score(t_opt)

def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def structured_init(n, rng, method="sin"):
    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t) \

"""
Minimize 2*n*max_conv/sum^2 using LP-based directional optimization.

Key innovation: At each iteration, solve a linear program to find the
optimal direction g that maximizes sum(g) subject to (f*g)_k <= M for
the top-K peak lags. This gives a principled descent direction that
respects the autocorrelation constraints. Combined with optimal step
size selection via ternary search on the resulting quadratic peak.
"""
import numpy as np

from scipy.sparse import csr_matrix

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def build_sparse_lp(f, selected_lags, n):
    """Build sparse LP constraint matrix."""
    K = len(selected_lags)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(selected_lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            idx_f = k - j
            if 0 <= idx_f < n and f[idx_f] > 1e-15:
                rows.append(i)
                cols.append(j)
                vals.append(f[idx_f])
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    return A


def solve_lp_direction(f, M, selected_lags, n):
    """Solve max 1^T g s.t. A g <= M, g >= 0."""
    K = len(selected_lags)
    A = build_sparse_lp(f, selected_lags, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    
    if res.success:
        return res.x
    return None


def find_best_step(f2, fg, g2, t_max=1.0):
    """Find t minimizing max_k of (1-t)^2*f2_k + 2t(1-t)*fg_k + t^2*g2_k."""
    def score(t):
        h2 = (1.0 - t) ** 2 * f2 + 2.0 * t * (1.0 - t) * fg + t ** 2 * g2
        return h2.max()
    
    lo = 0.0
    hi = t_max
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if score(m1) < score(m2):
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    return t_opt, score(t_opt)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_exp_shape(a, N, n, top_k, alpha, target=1.45):
    """Spectral shaping step (from previous approach)."""
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    if c_max <= target:
        return a, c_max
    
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max
    
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total
    
    weights = excess / excess_sum
    
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    alpha_eff = alpha * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.3)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max

    cols_arr = np.concatenate(all_cols)

"""
Minimize 2*n*max_conv/sum^2 using LP-based directional optimization with
adaptive constraint selection, optimal step-size via ternary search on
quadratic peak envelope, spectral water-filling polish, and coordinate descent.

Key innovation: Each iteration solves a linear program finding the optimal
non-negative direction g maximizing sum(g) subject to (f*g)_k <= M for the
top-K peak lags. This gives a principled descent direction respecting
autocorrelation constraints. Step size is optimized exactly via ternary
search on the resulting piecewise-quadratic peak function.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def build_sparse_lp(f, selected_lags, n):
    """Build sparse LP constraint matrix where row i corresponds to lag k_i."""
    K = len(selected_lags)
    all_rows = []
    all_cols = []
    all_vals = []
    for i in range(K):
        k = int(selected_lags[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        f_idx = k - js
        fv = f[f_idx]
        mask = fv > 1e-12
        if not np.any(mask):
            continue
        cnt = int(np.sum(mask))
        all_rows.append(np.full(cnt, i, dtype=np.int32))
        all_cols.append(js[mask].astype(np.int32))
        all_vals.append(fv[mask])
    if not all_rows:
        return csr_matrix((K, n))
    rows_arr = np.concatenate(all_rows)
    cols_arr = np.concatenate(all_cols)
    vals_arr = np.concatenate(all_vals)
    A = csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(K, n))
    return A


def solve_lp_direction(f, M, selected_lags, n, time_limit=20):
    """Solve max 1^T g s.t. A g <= M, g >= 0. Returns g or None."""
    K = len(selected_lags)
    A = build_sparse_lp(f, selected_lags, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def find_best_step(f2, fg, g2, t_max=1.0):
    """Find t in [0,t_max] minimizing max_k of (1-t)^2 f2_k + 2t(1-t) fg_k + t^2 g2_k."""
    lo = 0.0
    hi = t_max
    # First check endpoints
    score_0 = f2.max()
    score_1 = g2.max()
    if score_0 <= score_1:
        hi = 0.0
        return 0.0, score_0
    # Ternary search
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h2_m1 = (1.0 - m1) ** 2 * f2 + 2.0 * m1 * (1.0 - m1) * fg + m1 ** 2 * g2
        h2_m2 = (1.0 - m2) ** 2 * f2 + 2.0 * m2 * (1.0 - m2) * fg + m2 ** 2 * g2
        if h2_m1.max() < h2_m2.max():
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    h2 = (1.0 - t_opt) ** 2 * f2 + 2.0 * t_opt * (1.0 - t_opt) * fg + t_opt ** 2 * g2
    return t_opt, h2.max()


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    a = np.clip(a, 0.0, 1000.0)
    c = fft_convolve_self(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (sum_a ** 2)


def spectral_exp_shape(a, N, n, top_k, alpha, target=1.45):
    """Spectral shaping step."""
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    if c_max <= target:
        return a, c_max
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total
    weights = excess / excess_sum
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    alpha_eff = alpha * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.3)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def coordinate_descent_round(f, n, S, c, multipliers, top_count, deadline):
    """One round of coordinate descent on top-contributing coordinates. Returns (f, improved)."""
    c_max = c.max()
    peak_lag = int(np.argmax(c))
    low = max(0, peak_lag - n + 1)
    high = min(peak_lag, n - 1)
    contribs = np.zeros(n)
    for i in range(low, high + 1):
        j = peak_lag - i
        if 0 <= j < n:
            contribs[i] = f[i] * f[j]
    top_coords = np.argsort(contribs)[::-1][:top_count]
    improved = False
    for idx in top_coords:
        if time.monotonic() > deadline:
            break
        i = int(idx)
        j = peak_lag - i
        if j < 0 or j >= n or i == j:
            continue
        orig_val = f[i]
        best_frac = 1.0
        best_cmax = c_max
        for frac in multipliers:
            new_val = orig_val * frac
            if new_val < 0:
                continue
            delta = new_val - orig_val
            # Approximate new peak: c_peak + 2*delta*f[j]
            new_peak = c_max + 2.0 * delta * f[j]
            if new_peak < best_cmax - 1e-12:
                best_cmax = new_peak
                best_frac = frac
        f[i] = orig_val * best_frac
        if best_frac != 1.0:
            improved = True
    f = project_simplex(f, S)
    return f, improved
