import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_subgrad(a, N, beta):
    """LSE-smoothed subgradient. Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def topk_subgrad(a, c, N, top_k):
    """Exact top-k subgradient. Returns gradient vector."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    fsize = 1 << (3 * n + top_k).bit_length()
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def spectral_equality(a, N, alpha, n_peaks):
    """Spectral equalization: reduce power at frequencies contributing to peaks."""
    n = len(a)
    S = a.sum()
    c, A = compute_conv(a, N)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:n_peaks]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = (top_vals - c_max * 0.5) ** 2
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = top_vals / total
    else:
        weights = weights / wsum
    P = np.abs(A) ** 2
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), P * 0.01)
    phase = np.angle(A)
    # Handle zero-magnitude bins
    mask = P > 1e-30
    A_new = np.zeros(N, dtype=np.complex128)
    A_new[mask] = np.sqrt(P_new[mask]) * np.exp(1j * phase[mask])
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def nesterov_block(a_init, n, S, N, max_time, rng, beta_start=10.0, beta_end=200.0,
                    top_k_switch=50, momentum=0.9):
    """Nesterov-accelerated subgradient with adaptive smoothing."""
    a = np.clip(a_init, 0.0, 1000.0)
    a = project_simplex(a, S)
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    best_a = a.copy()
    best_cmax = np.inf
    c, _ = compute_conv(a, N)
    best_cmax = c.max()
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv(y, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = y.copy()

        # Adaptive beta: start smooth, get sharper
        progress = min(1.0, it / max(1, max_time * 100))
        beta = beta_start + (beta_end - beta_start) * progress

        # Choose subgradient type
        c_sorted = np.sort(c)[::-1]
        peak_concentration = c_sorted[:5].sum() / max(c.sum(), 1e-10)
        if it > top_k_switch and peak_concentration > 0.3:
            # Use exact top-k (more aggressive)
            grad = topk_subgrad(y, c, N, 3)
        else:
            # Use LSE-smoothed (more stable)
            grad, _, _, _ = lse_subgrad(y, N, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (beta + 1.0) * (1.0 + 0.001 * it))
        v = momentum * v - step * grad / gnorm
        y_new = y + v
        x = project_simplex(y_new, S)

        # Check improvement
        c_x, _ = compute_conv(x, N)
        cmax_x = c_x.max()

        if cmax_x <= cur_max * 1.001:
            a = x
        else:
            # Line search: try smaller step
            a = project_simplex(y - 0.5 * step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        t = t_new

        # Periodic spectral equalization
        if it % 200 == 0:

            a_spec = spectral_equality(a, N, alpha_spec, 5)

            if c_spec.max() < best_cmax:

                best_cmax = c_spec.max()
                a = a_spec

            v = np.zeros(n)

def coordinate_descent(a_init, n, S, N, max_time, rng):
    """Cyclic coordinate descent on most influential coordinates."""
    a = np.clip(a_init, 0.0, 1000.0)
    a = project_simplex(a, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()

    c, _ = compute_conv(a, N)
    best_cmax = c.max()

    # Find influential coordinates (high magnitude)
    influence = np.abs(a)
    sorted_idx = np.argsort(influence)[::-1]
    n_soft = min(n, 2048)
    soft_mask = np.zeros(n, dtype=bool)
    soft_mask[sorted_idx[:n_soft]] = True

    it = 0
    while time.monotonic() - start_time < max_time:
        improved = False
        c, _ = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()

        # Find worst peak
        k_worst = int(np.argmax(c))

        # Try adjusting top-50 most influential coordinates
        for j in sorted_idx[:50]:
            if not soft_mask[j]:
                continue
            a_j = a[j]
            if a_j < 1e-10:
                continue
            # Try small reduction
            delta = 0.01 * a_j
            a_old_j = a[j]
            a[j] = a_j - delta
            c_try, _ = compute_conv(a, N)
            if c_try.max() < cur_max:
                cur_max = c_try.max()
                improved = True
            else:
                a[j] = a_old_j

        if not improved:
            # Try increasing small coordinates
            small_coords = sorted_idx[n_soft:]
            for j in small_coords[:20]:
                a_j = a[j]
                delta = 0.01 * S / n
                a_old_j = a[j]
                a[j] = a_j + delta
                c_try, _ = compute_conv(a, N)
                if c_try.max() < cur_max:
                    cur_max = c_try.max()
                    improved = True
                else:
                    a[j] = a_old_j

        if not improved:
            # Perturb and continue
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n_affected(a, n_soft)), S)
            it += 1

    return best_a, best_cmax

    a = np.zeros(n)
    half = n // 2
    a[:half] = 1.0 + 0.2 * rng.standard_normal(half)
    a[half:] = 0.1 + rng.random(n - half) * 0.2
    return np.clip(a, 0.0, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive Nesterov subgradient + spectral equalization + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Try multiple sequence lengths
    candidate_lengths = [8192]
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= given.size <= 32768:
            candidate_lengths.insert(0, len(given))
    except Exception:
        pass

    # Filter to reasonable lengths
    candidate_lengths = [l for l in candidate_lengths if 2048 <= l <= 16384]
    if not candidate_lengths:
        candidate_lengths = [8192]

    global_best = np.inf
    global_a = None
    global_n = None

    def update_best(a_candidate, n_val, label=""):
        nonlocal global_best, global_a, global_n
        S = np.sqrt(2.0 * float(n_val))
        N = 1 << (2 * n_val - 1).bit_length()
        a_cand = np.clip(a_candidate, 0.0, 1000.0)
        a_cand = project_simplex(a_cand, S)
        c, _ = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            global_n = n_val
            if label:
                print("  [%s n=%d] %.6f" % (label, n_val, global_best))
        return sc

    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.15)

    for n in candidate_lengths[:2]:  # Try at most 2 lengths
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        print("  Trying n=%d, N(FFT)=%d" % (n, N))

        inits = []
        try:
            given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            if given.size == n:
                inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
        except Exception:
            pass
        inits.append(("rand", make_init_rand(n, rng)))
        inits.append(("ramp", make_init_ramp(n, rng)))
        inits.append(("chirp", make_init_chirp(n, rng)))
        inits.append(("half", make_init_half(n, rng)))
        inits.append(("fourier", make_init_fourier(n, rng, N)))

        n_inits = len(inits)
        per_init = max(5.0, phase1_budget / (n_inits * len(candidate_lengths[:2])))

        for name, arr in inits:
            if time.monotonic() > deadline - 600:
                break
            a_norm = project_simplex(np.clip(arr, 0, 1000), S)
            t0 = time.monotonic()
            rem = max(2.0, per_init - (time.monotonic() - t0))
            a_opt, sc = nesterov_block(a_norm, n, S, N, rem, rng,
                                        beta_start=5.0, beta_end=50.0,
                                        top_k_switch=30, momentum=0.85)
            update_best(a_opt, n, name + "/nest")

    print("Phase 1 done. Best: %.6f" % global_best)

    if global_a is None:
        n = 8192
        S = np.sqrt(2.0 * float(n))
        global_a = np.ones(n) * S / n
        global_n = n
        global_best = 2.0

    # --- Phase 2: Intensive Nesterov optimization ---
    print("=== Phase 2: Intensive Nesterov ===")
    t2 = time.monotonic()
    phase2_budget = min(1400.0, (deadline - t2) * 0.55)
    n = global_n
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    a_curr = global_a.copy()
    stall_count = 0

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(180.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break

        # Vary parameters based on progress
        elapsed_frac = (time.monotonic() - t2) / phase2_budget
        beta_start = 5.0 + 30.0 * elapsed_frac
        beta_end = 50.0 + 200.0 * elapsed_frac
        top_k_sw = max(20, int(100 * (1 - elapsed_frac)))

        a_new, sc_new = nesterov_block(a_curr, n, S, N, block_time, rng,
                                        beta_start=beta_start, beta_end=beta_end,
                                        top_k_switch=top_k_sw, momentum=0.9)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            print("  [P2] %.6f (stall=%d)" % (global_best, 0))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count >= 2:
                # Structured perturbation
                c, A = compute_conv(a_curr, N)
                k_worst = int(np.argmax(c))
                # Create perturbation based on worst peak structure
                mirror = np.zeros(n)
                for i in range(n):
                    j = k_worst - i
                    if 0 <= j < n:
                        mirror[i] = a_curr[j]
                mirror = mirror / (np.linalg.norm(mirror) + 1e-10)
                a_curr = project_simplex(a_curr + 0.02 * (mirror - mirror.mean()), S)
                stall_count = 0

    # --- Phase 3: Spectral equalization refinement ---
    print("=== Phase 3: Spectral equalization ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.2)

    for npk in [5, 10, 20]:
        rem = max(5.0, phase3_budget / 3.0 - (time.monotonic() - t3))
        if rem < 5 or time.monotonic() > deadline - 100:
            break
        a_sp, sc_sp = spectral_block(global_a, n, S, N, rem, rng,
                                      n_peaks=npk, alpha_init=0.12)
        update_best(a_sp, n, "spec_k%d" % npk)

    # --- Phase 4: Final Nesterov polish ---
    print("=== Phase 4: Final polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        a_fin, sc_fin = nesterov_block(global_a, n, S, N, min(remaining - 10, 300.0),
                                        rng, beta_start=30.0, beta_end=300.0,
                                        top_k_switch=10, momentum=0.95)
        update_best(a_fin, n, "final")

    # --- Phase 5: Quick coordinate descent ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_cd, sc_cd = coordinate_descent(global_a, n, S, N, min(remaining - 15, 120.0), rng)
        update_best(a_cd, n, "coord")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, internal=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
