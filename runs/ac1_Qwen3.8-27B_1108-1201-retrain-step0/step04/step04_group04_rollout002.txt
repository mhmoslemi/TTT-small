"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
LP-based peak reduction with FISTA initialization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=2000.0,
              beta_ramp_iters=2000):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def compute_conv_gradient(a, k):
    """Compute dc[k]/da for a single lag k."""
    n = len(a)
    grad = np.zeros(n)
    # Term 1: a[k-j] for j where 0 <= k-j < n
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
    # Term 2: a[j-k] for j where 0 <= j-k < n
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        grad[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
    return grad


def lp_refine(a_init, n, S, max_time, rng, K_top=50, step_decay=0.95):
    """LP-based peak reduction."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    cur_max = c.max()
    best = cur_max
    best_a = a.copy()
    
    it = 0
    step_size = 1.0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 5:
            break
        
        # Find top K lags
        sorted_lags = np.argsort(c)[::-1]
        active_lags = sorted_lags[:K_top]
        
        # Compute gradient matrix J
        J = np.zeros((K_top, n))
        for idx, k in enumerate(active_lags):
            J[idx] = compute_conv_gradient(a, int(k))
        
        c_active = c[active_lags]
        
        # Build LP
        # Variables: [delta (n), t (1)]
        # Objective: min t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality: J @ delta - t <= -c_active
        # [J, -1] @ [delta, t] <= -c_active
        A_ub = np.hstack([J, -np.ones((K_top, 1))])
        b_ub = -c_active
        
        # Bounds: -a[j] <= delta[j] <= min(a[j]*2, S) (generous upper bound)
        lower_bound = -a
        upper_bound = np.minimum(2.0 * a, S)
        bounds = list(zip(lower_bound, upper_bound))
        bounds.append((0.0, np.inf))  # for t
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        # Solve
        t_lp = min(remaining * 0.8, 30.0)  # time limit for LP
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': t_lp, 'presolve': True})
        except Exception as e:
            print("  LP error at iter %d: %s" % (it, str(e)))
            break
        
        if res.status != 0:
            if res.status == 2:  # timeout
                print("  LP timeout at iter %d" % it)
                # Try with fewer active lags
                if K_top > 10:
                    K_top = max(10, K_top // 2)
                    continue
            else:
                print("  LP status %d at iter %d: %s" % (it, res.status, res.message))
                break
        
        delta = res.x[:n]
        t_val = res.x[n]
        
        if it <= 5 or it % 50 == 0:
            print("  LP iter %d: t_LP=%.6f, cur_max=%.6f, step=%.3f" % (it, t_val, cur_max, step_size))
        
        # Line search: find best alpha in {step_size, 0.5*step_size, 0.25*step_size, ...}
        best_alpha = 0
        best_new_max = cur_max
        
        for alpha in [step_size, 0.5*step_size, 0.25*step_size, 0.1*step_size, 0.05*step_size]:
            a_trial = a + alpha * delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_convolve_self(a_trial)
            new_max = c_trial.max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_alpha = alpha
                best_a_trial = a_trial.copy()
        
        if best_alpha > 0:
            a = best_a_trial
            cur_max = best_new_max
            if cur_max < best:
                best = cur_max
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
            step_size = min(1.0, step_size * 1.05)  # slowly increase step
        else:
            stagnant += 1
            step_size *= step_decay
        
        if stagnant > 50:
            # Add noise to escape
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            cur_max = c.max()
            stagnant = 0
            step_size = 0.5
            print("  Iter %d: noise injection, cur_max=%.6f" % (it, cur_max))
        
        c = fft_convolve_self(a)  # Update c each iteration
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
    
    return best_a, best


def coordinate_polish(a_init, n, S, max_time, rng):
    """Fast coordinate descent polish."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        n_try = min(100 + it, 500)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break
            
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue
            
            # Transfer delta from m to j
            # c[k] changes by: delta * (a[j+k] + a[j-k] - a[m+k] - a[m-k])
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                if 0 <= j + k < n:
                    val += a[j + k]
                if 0 <= j - k < n:
                    val += a[j - k]
                if 0 <= m + k < n:
                    val -= a[m + k]
                if 0 <= m - k < n:
                    val -= a[m - k]
                d[k] = val
            
            # Golden section search for optimal delta
            lo = -a[j] * 0.99
            hi = a[m] * 0.99
            
            if hi - lo < 1e-15:
                continue
            
            gr = (np.sqrt(5.0) + 1.0) / 2.0
            a_g, b_g = lo, hi
            c_g = b_g - (b_g - a_g) / gr
            d_g = a_g + (b_g - a_g) / gr
            
            def f_d(dlt):
                return (c + dlt * d).max()
            
            fc, fd = f_d(c_g), f_d(d_g)
            for _ in range(30):
                if fc < fd:
                    b_g, d_g = d_g, c_g
                    fd = fc
                    c_g = b_g - (b_g - a_g) / gr
                    fc = f_d(c_g)
                else:
                    a_g, c_g = c_g, d_g
                    fc = fd
                    d_g = a_g + (b_g - a_g) / gr
                    fd = f_d(d_g)
            
            delta = (a_g + b_g) / 2.0
            
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < 0 or a_trial[m] < 0:
                continue
            
            c_trial = c + delta * d
            # Fix c[0] (quadratic term)
            c_trial[0] = c[0] + 2*(a[j]-a[m])*delta + 2*delta*delta
            
            if c_trial.max() < best - 1e-10:
                a = a_trial
                c = c_trial
                best = c_trial.max()
                best_a = a.copy()
                improved = True
        
        if it % 20 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        if not improved and it > 10:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best
