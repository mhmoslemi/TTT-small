"""
Minimize 2*n*max(f*f)/(sum(f))^2 over non-negative sequences.
Uses FISTA with LSE smoothing, adaptive beta, periodic LP corrections,
and multiplicative repulsion updates.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max(f*f)/(sum(f))^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def multiplicative_repulsion(a, n, S, beta=50.0, lr=0.1):
    """Multiplicative repulsion update to reduce peak."""
    c = fft_convolve_self(a)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    a_new = a * np.exp(-lr * rep / max(a.max(), 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    a_new = project_simplex(a_new, S)
    return a_new


def solve_lp_best_response(f, n, S, top_k=200, time_limit=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)

The LP best response works as follows:
1. Given current f with max(f*f) = M.
2. Solve LP: max sum(g) s.t. (f*g)(k) <= M, g >= 0.
3. If sum(g) > sum(f), normalize g and check if max(g*g) < M.
4. If yes, update f = g.
5. Repeat.

The key question is: does this monotonically decrease max(f*f)? Not guaranteed, but often helpful.

Let me also think about a simpler "polishing" approach:
- Identify the top K lags where the convolution is closest to the max.
- Set up a QP: minimize max(c[k] for k in top_K) subject to sum(a) = S, a >= 0.
- This can be solved as: minimize t subject to sum_j a[j]*a[k-j] <= t for each k in top_K, sum(a) = S, a >= 0.

This is a QP (quadratic constraints) which cvxpy can handle with appropriate solvers.

Actually, the constraint sum_j a[j]*a[k-j] <= t is a quadratic constraint. cvxpy can handle convex quadratic constraints. But is sum_j a[j]*a[k-j] convex in a?

sum_j a[j]*a[k-j] = a^T Q_k a where Q_k is a symmetric matrix. For this to be a convex constraint (a^T Q_k a <= t), we need Q_k to be positive semidefinite.

As I noted before, Q_k for the "center" lag (k = n-1) is the reversal matrix, which has eigenvalues ±1. So it's NOT PSD, and the constraint is non-convex.

For other lags, Q_k might be PSD or not depending on the lag.

So we can't directly solve this as a convex QP. The LP relaxation (fixing a and solving for g) is the standard approach.

"""
Minimize 2*n*max(f*f)/(sum(f))^2 over non-negative sequences.
Strategy: FISTA with LSE smoothing, adaptive beta schedule, periodic LP best-response
corrections, multiplicative repulsion, and diverse restarts to escape local minima.
Final polishing with very high beta for accuracy.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max(f*f)/(sum(f))^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def lse_weight(c, beta):
    """Compute softmax weights for LSE approximation."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c), cm
    return e / tot, cm


def gradient_from_weights(a, w):
    """Compute gradient given precomputed weights."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def multiplicative_repulsion(a, n, S, beta=40.0, lr=0.05):
    """Multiplicative update to reduce peak convolution."""
    c = fft_convolve_self(a)
    w, _ = lse_weight(c, beta)
    rep = gradient_from_weights(a, w)
    a_new = a * np.exp(-lr * rep / max(a.max(), 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    a_new = project_simplex(a_new, S)
    return a_new


def solve_lp_best_response(f, n, S, top_k=200):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    try:
        prob.solve(solver='SCS', verbose=False, max_iters=100000)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None

def generate_diverse_inits(n, rng):
    """Generate several diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Sin^2
    inits.append(np.maximum(np.sin(np.pi * tt) ** 2, 0.01))
    # Gaussian bump
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2 / 0.1))
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    # Random
    inits.append(rng.random(n) + 0.05)
    # Log-normal
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    # Block structure
    nblocks = 17
    block_len = n // nblocks
    heights = rng.uniform(0.2, 2.0, nblocks)
    block_seq = np.repeat(heights, block_len)
    if len(block_seq) < n:
        block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
    inits.append(block_seq[:n])

    return [np.maximum(x, 0.01) for x in inits]


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

        a4, best4, _ = run_fista(global_a, n, S, t4, rng,
                                  beta_start=2000, beta_end=8000,
                                  repulsion_every=500, repulsion_lr=0.02,
                                  noise_every=5000, restart_after=8000)
        a4 = normalize_sum(a4, n)
        c4 = fft_convolve_self(a4)
        obj4 = c4.max()
        print("  Phase 4 best: %.6f" % obj4)
        if obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()

    # Phase 5: Try diverse restarts (remaining time - margin)
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        t5 = remaining * 0.6
        print("=== Phase 5: Diverse restarts (%.0fs) ===" % t5)
        inits = generate_diverse_inits(n, rng)
        # Also include perturbed version of current best
        inits.append(project_simplex(global_a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S))
        per_init = t5 / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 5:
                break
            ai, bi, _ = run_fista(init, n, S, per_init, rng,
                                   beta_start=200, beta_end=3000,
                                   repulsion_every=200, repulsion_lr=0.04,
                                   noise_every=1000, restart_after=2000)
            ai = normalize_sum(ai, n)
            ci = fft_convolve_self(ai)
            obji = ci.max()
            if obji < global_best:
                global_best = obji
                global_a = ai.copy()
                print("  Diverse init %d: %.6f" % (i, global_best))

    # Phase 6: Final LP polish on best
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Phase 6: Final LP polish (%.0fs) ===" % remaining)
        a6, best6, nlp6 = run_lp_phase(global_a, n, S, remaining * 0.7, rng, top_k=400)
        a6 = normalize_sum(a6, n)
        c6 = fft_convolve_self(a6)
        obj6 = c6.max()
        print("  Phase 6 best: %.6f (LP calls: %d)" % (obj6, nlp6))
        if obj6 < global_best:
            global_best = obj6
            global_a = a6.copy()

    # Phase 7: Final high-beta FISTA (last bit of time)
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        print("=== Phase 7: Final FISTA (%.0fs) ===" % remaining)
        a7, best7, _ = run_fista(global_a, n, S, remaining * 0.8, rng,
                                  beta_start=5000, beta_end=10000,
                                  repulsion_every=1000, repulsion_lr=0.01,
                                  noise_every=10000, restart_after=10000)
        a7 = normalize_sum(a7, n)
        c7 = fft_convolve_self(a7)
        obj7 = c7.max()
        print("  Phase 7 best: %.6f" % obj7)
        if obj7 < global_best:
            global_best = obj7
            global_a = a7.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
