def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

(f*f)(k) = sum_j (1/n)(1/n) = (number of valid j) / n^2

For k=0: (f*f)(0) = n * (1/n)^2 = 1/n
For k=n-1: (f*f)(n-1) = n * (1/n)^2 = 1/n (wait, no)

Actually, (f*f)(k) = sum_j f(j) f(k-j) where both j and k-j must be in [0, n-1].

For f(j) = 1/n for all j:
(f*f)(k) = sum_{j=max(0,k-n+1)}^{min(k,n-1)} (1/n)(1/n) = (min(k,n-1) - max(0,k-n+1) + 1) / n^2

For k = n-1: min(n-1, n-1) - max(0, n-1-n+1) + 1 = (n-1) - 0 + 1 = n
So (f*f)(n-1) = n/n^2 = 1/n

For k = 0: min(0, n-1) - max(0, 0-n+1) + 1 = 0 - 0 + 1 = 1
So (f*f)(0) = 1/n^2

The max is at k = n-1 (the center), where (f*f)(n-1) = 1/n.

Objective = 2n * (1/n) / (sum f)^2 = 2n * (1/n) / (1)^2 = 2.

So the uniform sequence gives objective 2. We need to do much better (target 1.5027).

What about a sequence that's concentrated in the middle? Like a Gaussian?

For a Gaussian f(j) = exp(-alpha * (j - (n-1)/2)^2 / n^2), the autocorrelation is also a Gaussian (with wider spread), and the peak is at the center. The ratio max/(sum)^2 depends on the width.

For a very narrow Gaussian (delta-like), the peak of the autocorrelation is ~1 (concentrated at lag 0), and sum ~ 1, so the ratio is ~ 2n * 1 / 1 = 2n. That's terrible.

For a very wide Gaussian (uniform-like), we get back to the uniform case with ratio ~2.

The optimal width is somewhere in between, and the non-uniformity of the envelope helps reduce the peak autocorrelation.

This makes sense: we want a sequence that's "spread out" but not too uniform, to reduce the peak of the autocorrelation.

Now, the key question is: what's the optimal shape? It's not a simple Gaussian or cosine - it's something more complex that optimally balances the tradeoff between spread and peak reduction.

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Shift to make投影 work
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta, n):
    cm = c.max()
    e = np.exp(beta * (c - cm))

        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def upsample(a, new_n):
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)

def solve_lp(f, n, S, k_constraints, solver='PDLP', time_limit=15.0):
    """Solve the LP best response with a given number of constraints."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Select constraints: top k by value + some random
    k = min(k_constraints, 2*n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    
    # Build constraint matrix
    A_rows = []
    rhs = []
    for k_idx in top_idx:
        k_idx = int(k_idx)
        lo = max(0, k_idx - n + 1)
        hi = min(k_idx, n - 1)
        js = np.arange(lo, hi + 1)
        row = np.zeros(n)
        row[js] = f[k_idx - js]
        A_rows.append(row)
        rhs.append(M)
    
    if not A_rows:
        return None
    
    A_mat = np.array(A_rows)
    b_vec = np.array(rhs)
    
    g = cp.Variable(n)
    constraints = [A_mat @ g <= b_vec, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                return None
    
    elapsed = time.monotonic() - start
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        if g_val.sum() > 1e-10:
            g_val = g_val * (S / g_val.sum())
            return g_val, elapsed
    return None, elapsed

def line_search(f, g, S, current_max, fsize=None):
    """Find best step size t for f_new = (1-t)*f + t*g."""
    best_t = 0.0
    best_val = current_max

        f_test = np.maximum(f_test, 0)

        c_test = fft_convolve_self(f_test)
        val = c_test.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        for _ in range(5):

        f_new = np.maximum(f_new, 0)
        return project_simplex(f_new, S), best_val
    
    return f.copy(), current_max

def run_fista_short(a, n, S, time_budget, rng, beta_init=20.0, beta_rate=0.003, beta_max=1000.0):
    """Run FISTA for a short time budget."""
    a = normalize_sum(a, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best

"""
Minimize 2*n*max(f*f)/(sum f)^2 using multi-resolution progressive LP + FISTA hybrid.
Strategy:
1. Multi-resolution warmup (256->8192) building up from small to large
2. Main phase: aggressive LP best-response with progressive constraint growth
3. Polish phase: high-precision FISTA + final LP with many constraints
Key improvements: PDLP solver, adaptive constraint selection, smart line search,
multi-resolution initialization, and periodic diversification.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta, n):
    """Gradient of LSE approximation of max."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_from_max(a, cmax):
    """Compute objective given array and max convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * cmax / (s * s)


def make_initial(n, kind='nuttall'):
    """Generate initial sequence of given kind."""
    t = np.linspace(0, 1, n, endpoint=False)
    if kind == 'nuttall':
        a0, a1, a2, a3, a4 = 0.3635819, 0.4891775, 0.1365995, 0.0116814, 0.0003746
        w = a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) - a3 * np.cos(6 * np.pi * t) + a4 * np.cos(8 * np.pi * t)
    elif kind == 'cosine':
        w = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    elif kind == 'cos2':
        w = 0.5 * (1.0 + np.cos(2 * np.pi * t)) ** 2
    elif kind == 'cos3':
        w = 0.5 * (1.0 + np.cos(2 * np.pi * t)) ** 3
    elif kind == 'flat':
        w = np.ones(n)
    elif kind == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    else:
        w = np.ones(n)
    return np.maximum(w, 0.005)


def upsample_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)


def build_constraint_matrix(f, n, lag_indices):
    """Build the constraint matrix A where row k is the convolution kernel for lag k."""
    k_count = len(lag_indices)
    A = np.zeros((k_count, n), dtype=np.float64)
    for i, k_idx in enumerate(lag_indices):
        k_idx = int(k_idx)
        lo = max(0, k_idx - n + 1)
        hi = min(k_idx, n - 1)
        js = np.arange(lo, hi + 1)
        A[i, js] = f[k_idx - js]
    return A


def solve_lp_best_response(f, n, S, lag_indices, solver='PDLP', time_limit=20.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in lag_indices, g >= 0.
    Returns (normalized_g, elapsed_time) or (None, elapsed_time).
    """
    c = fft_convolve_self(f)
    M = c.max()
    
    k_count = len(lag_indices)
    if k_count == 0:
        return None, 0.0
    
    A = build_constraint_matrix(f, n, lag_indices)
    b = np.full(k_count, M, dtype=np.float64)
    
    g = cp.Variable(n)
    constraints = [A @ g <= b, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    start = time.monotonic()
    solved = False
    for solver_try in [solver, 'SCS', 'ECOS']:
        try:
            prob.solve(solver=solver_try, verbose=False)
            solved = True
            break
        except Exception:
            continue
    
    elapsed = time.monotonic() - start
    
    if solved and g.value is not None:
        g_val = np.maximum(g.value, 0.0)
        g_sum = g_val.sum()
        if g_sum > 1e-10:
            g_val = g_val * (S / g_sum)
            return g_val, elapsed
    
    return None, elapsed


def select_lags(c, n, k_total, rng, top_fraction=0.8):
    """Select lags for LP constraints: mix of top by value and random."""
    size = len(c)
    k_total = min(k_total, size)
    
    k_top = int(k_total * top_fraction)
    k_rand = k_total - k_top
    
    top_idx = np.argsort(c)[::-1][:k_top]
    
    if k_rand > 0:
        remaining = np.setdiff1d(np.arange(size), top_idx)
        rand_idx = rng.choice(remaining, size=min(k_rand, len(remaining)), replace=False)
        idx = np.concatenate([top_idx, rand_idx])
    else:
        idx = top_idx
    
    return idx


def line_search_step(f, g, S, current_max, n):
    """Find best convex combination step. Returns (new_f, new_max)."""
    best_t = 0.0
    best_val = current_max
    
    candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
    
    for t in candidates:
        f_test = (1.0 - t) * f + t * g
        f_test = np.maximum(f_test, 0.0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        val = c_test.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best
    if best_t > 0.05:
        for _ in range(4):
            for dt in [best_t * 0.85, best_t * 1.15, best_t - 0.05, best_t + 0.05]:
                if 0.01 < dt < 1.0:
                    f_test = (1.0 - dt) * f + dt * g
                    f_test = np.maximum(f_test, 0.0)
                    f_test = project_simplex(f_test, S)
                    c_test = fft_convolve_self(f_test)
                    val = c_test.max()
                    if val < best_val:
                        best_val = val
                        best_t = dt
    
    if best_t > 0.001:
        f_new = (1.0 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), current_max


def run_fista(a_init, n, S, time_budget, rng, beta_init=20.0, beta_rate=0.003, beta_max=1000.0):
    """Run FISTA with LSE smoothing for a given time budget."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_max:
            best_max = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom_new if (t_mom_new := t_new) else t_new
    
    return best_a, best_max


def run_lp_phase(a_init, n, S, time_budget, rng, start_k=200, max_k=3000,
                 k_growth=1.15, fista_between=3.0, solver='PDLP'):
    """
    Main LP-based optimization phase with progressive constraint growth.
    Returns (best_a, best_max).
    """
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    start_time = time.monotonic()
    it = 0
    last_improve = 0
    k_current = start_k
    
    while time.monotonic() - start_time < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start_time)
        if remaining < 10:
            break
        
        # Select lags
        c_current = fft_convolve_self(a)
        lag_idx = select_lags(c_current, n, k_current, rng)
        
        # Solve LP
        g, lp_time = solve_lp_best_response(a, n, S, lag_idx, solver=solver,
                                             time_limit=min(remaining * 0.3, 25.0))
        
        improved = False
        if g is not None:
            f_new, new_max = line_search_step(a, g, S, best_max, n)
            if new_max < best_max * (1.0 - 1e-7):
                a = f_new
                best_max = new_max
                best_a = a.copy()
                last_improve = it
                improved = True
                
        # Grow constraint count if improving
        if improved:
            k_current = min(int(k_current * k_growth), max_k)
        elif it - last_improve > 5:
            # Reset k and try perturbation
            k_current = start_k
            if it - last_improve > 15:
                # Add noise to escape
                noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 100.0)
                a = project_simplex(a + rng.normal(0, max(noise_scale, 1e-8), n), S)
                c = fft_convolve_self(a)
                if c.max() < best_max:
                    best_max = c.max()
                    best_a = a.copy()
                last_improve = it
        
        # FISTA fill-in between LP calls
        remaining = time_budget - (time.monotonic() - start_time)
        fista_budget = min(fista_between, remaining * 0.1, lp_time * 0.3)
        if fista_budget > 0.3:
            a_f, max_f = run_fista(a, n, S, fista_budget, rng,
                                    beta_init=50.0, beta_rate=0.002, beta_max=3000.0)
            if max_f < best_max:
                a = a_f
                best_max = max_f
                best_a = a.copy()
                last_improve = it
    
    return best_a, best_max
