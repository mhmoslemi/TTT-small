        # Periodic restart with noise if stuck
        if it - last_improve > restart_threshold:
            noise_scale = 0.05 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-8:
                a = normalize_sum(a + rng.normal(0, noise_scale, n), S)
                a_prev = a.copy()
                momentum = 0.0
                last_improve = it
        
        # Gradually increase momentum
        momentum = min(0.85, momentum + 0.0005)
    
    # Final check
    b_final = fft_convolve_self(a)
    if b_final.max() < best:
        best = b_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_fista_improved(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=5000.0,
                       beta_ramp=2000):
    """Improved FISTA with better step size and hybrid gradients."""
    a = normalize_sum(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    grad_fsize = 1 << (3 * n - 2).bit_length()
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    lb = S * S / (2 * n - 1)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp) ** 1.2
        else:
            beta = beta_max
        
        b, A = compute_conv_and_fft(y, fsize)
        cur = b.max()
        
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        gap = cur - lb
        if gap < 1e-12:
            break
        
        # Gradient
        if it % 50 == 0 and rng.random() < 0.3:
            grad = exact_subgradient(b, y)
        else:
            e = np.exp(beta * (b - cur))
            tot = e.sum()
            if tot < 1e-300:
                w = np.zeros(len(b))
                w[np.argmax(b)] = 1.0
            else:
                w = e / tot
            w_pad = np.zeros(grad_fsize)
            w_pad[:len(w)] = w
            W = fft(w_pad, grad_fsize)
            A_g = fft(y, grad_fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A_g)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = 1.0 / (2.0 * beta + 1.0)
        # Polyak correction
        polyak_step = gap / (gnorm * gnorm + 1e-15)
        step = min(step, polyak_step * 1.5)
        step = max(step, 1e-10)
        
        # Project onto simplex {x >= 0, sum(x) = S}
        x_new = _project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        
        # Noise injection
        if it % 5000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = normalize_sum(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    b_final = fft_convolve_self(a)
    if b_final.max() < best:
        best = b_final.max()
        best_a = a.copy()
    return best_a, best


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_equalize(a_init, n, S, max_time, rng):
    """
    Spectral equalization: iteratively adjust |A| to flatten b = IFFT(A^2).
    Uses iterative proportion fitting in the amplitude domain.
    """
    a = normalize_sum(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    A = fft(a, fsize)
    phase = A / np.maximum(np.abs(A), 1e-30)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        A = phase * np.abs(A)
        B = A * A
        b = np.real(ifft(B))[:2 * n - 1]
        cur = b.max()
        
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Target: B should be as flat as possible
        # B_target[k] = B_mean * (1 + alpha * (B_mean / (|B[k]| + eps)))
        # Push down peaks, push up troughs
        B_abs = np.abs(B)
        B_mean = B_abs.mean()
        if B_mean < 1e-30:
            break
        
        # Soft flattening: new_|B| = |B|^alpha normalized
        alpha = 0.5 + 0.3 * min(1.0, it / 500.0)
        B_target = np.ones_like(B)
        
        # Blend: push B towards flat
        blend = min(0.3, 0.05 + 0.001 * it)
        B_new = (1.0 - blend) * B + blend * B_mean
        
        # Update A: A_new = sqrt(B_new) with phase preserved for A
        # A_new^2 = B_new, so A_new = sqrt(B_new) (choosing a branch)
        # We want A_new to be close to A, so use the phase of A
        B_new_mag = np.abs(B_new)
        B_new_phase = B_new / np.maximum(B_new_mag, 1e-30)
        
        # A_new has phase = B_new_phase / 2 (in log domain)
        # Use: A_new = sqrt(B_new) with proper branch
        A_new = np.sqrt(B_new_mag) * np.exp(0.5j * np.angle(B_new))
        
        # Ensure A_new is close to A (same "branch")
        # Adjust phase to minimize distance
        phase_diff = np.angle(A * np.conj(A_new))
        A_new = A_new * np.exp(-1j * phase_diff)
        
        # Enforce Hermitian symmetry (A is FFT of real signal)
        A_new = (A_new + np.conj(A_new[np.mod(-np.arange(fsize), fsize)])) / 2.0
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = normalize_sum(a_new, S)
        
        # Update phase for next iteration
        A_temp = fft(a_new, fsize)
        phase = A_temp / np.maximum(np.abs(A_temp), 1e-30)
        a = a_new
    
    return best_a, best


def spectral_flatten_power(a_init, n, S, max_time, rng):
    """
    Flatten |A|^2 (power spectrum) which indirectly helps flatten b.
    Uses iterative amplitude adjustment.
    """
    a = normalize_sum(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Progressive flattening
        progress = min(1.0, it / 1000.0)
        alpha = 0.2 + 0.5 * progress
        
        # P_new = P^alpha / mean(P^alpha)
        P_alpha = np.power(np.maximum(P, 1e-30), alpha)
        P_new = P_alpha / P_alpha.mean()
        
        # Blend
        blend = 0.3 + 0.4 * progress
        P_final = (1.0 - blend) * P / P_mean + blend * P_new
        
        mag_new = np.sqrt(P_final)
        # Preserve total energy
        mag_new = mag_new * np.sqrt(P.sum() / max(P_final.sum(), 1e-30))
        
        A_new = phase * mag_new
        a_new = np.real(ifft(A_new))[:n]
        a_new = normalize_sum(a_new, S)
        
        b = fft_convolve_self(a_new)
        cur = b.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best


def coordinate_descent_fft(a_init, n, S, max_time, rng, n_pairs=100):
    """
    Efficient coordinate descent using FFT for gradient computation.
    Tries pairwise transfers to reduce max(b).
    """
    a = normalize_sum(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    grad_fsize = 1 << (3 * n - 2).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    b = fft_convolve_self(a)
    best = b.max()
    
    # Identify "hot" lags (near the max)
    threshold = b.max() * 0.9
    hot_lags = np.where(b >= threshold)[0]
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        n_try = min(n_pairs + it, 300)
        
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break
            
            # Choose pair: bias towards moving mass from "expensive" to "cheap" positions
            r = rng.random()
            if r < 0.4:
                # Move from high-contribution to low-contribution
                # Contribution of a[j] to max(b) is approx 2*a[j]*max_neighbor
                j = rng.choice(np.where(a > 0.01 * a.mean())[0])
                m = rng.integers(0, n)
            elif r < 0.7:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, max(1, n - 10))
                m = j + rng.integers(1, min(10, n - 1))
                m = m % n
            
            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue
            
            # Compute the change in b for transferring delta from m to j
            # b[k] changes by: delta * (a[k-j] + a[j-k] - a[k-m] - a[m-k])
            # For the hot lags, compute the direction
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                ij = k - j
                if 0 <= ij < n:
                    val += a[ij]
                ij2 = j - k
                if 0 <= ij2 < n:
                    val += a[ij2]
                im = k - m
                if 0 <= im < n:
                    val -= a[im]
                im2 = m - k
                if 0 <= im2 < n:
                    val -= a[im2]
                d[k] = val
            
            # Find optimal delta by checking hot lags
            # b[k](delta) = b[k] + delta * d[k] + (quadratic term for k where j or m = k)
            lo = -a[j] * 0.9
            hi = a[m] * 0.9
            
            if lo >= hi:
                continue
            
            # Evaluate at a few candidate deltas
            best_delta = 0.0
            best_val = b.max()
            
            # Check: 0, and values that make hot lags equal
            candidates = [0.0]
            for k in hot_lags[:20]:
                if abs(d[k]) > 1e-15:
                    dk = (b.max() - b[k]) / d[k]
                    if lo < dk < hi:
                        candidates.append(dk)
            
            # Also golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = lo, hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            
            def eval_delta(dlt):
                vals = b + dlt * d
                # Quadratic correction for k=0 (if j or m is 0)
                # b[0] = a[0]^2, change = 2*a[0]*dlt + dlt^2 (if j=0) etc.
                # For simplicity, just use linear approximation
                return vals.max()
            
            fc = eval_delta(c_g)
            fd = eval_delta(d_g)
            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = eval_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = eval_delta(d_g)
            
            candidates.append((a_g + b_g) / 2.0)
            
            for dlt in candidates:
                dlt = max(lo, min(hi, dlt))
                if abs(dlt) < 1e-12:
                    continue
                new_max = eval_delta(dlt)
                if new_max < best_val - 1e-12:
                    best_val = new_max
                    best_delta = dlt
            
            if best_delta != 0.0:
                a[j] += best_delta
                a[m] -= best_delta
                if a[j] < 0 or a[m] < 0:
                    a[j] = max(0, a[j])
                    a[m] = max(0, a[m])
                    a = normalize_sum(a, S)
                else:
                    a = normalize_sum(a, S)
                b = fft_convolve_self(a)
                cur = b.max()
                if cur < best:
                    best = cur
                    best_a = a.copy()
                    improved = True
                threshold = b.max() * 0.9
                hot_lags = np.where(b >= threshold)[0]
        
        if not improved and it > 20:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = normalize_sum(a + rng.normal(0, noise_scale, n), S)
            b = fft_convolve_self(a)
            if b.max() < best:
                best = b.max()
                best_a = a.copy()
    
    return best_a, best


def generate_diverse_starts(n, S, rng, parent=None):
    """Generate diverse starting sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        starts.append(("parent", normalize_sum(parent, S)))
        starts.append(("parent_rev", normalize_sum(parent[::-1], S)))
    
    # Cosine window
    starts.append(("cosine", normalize_sum(0.5 * (1 + np.cos(2 * np.pi * tt)), S)))
    
    # Raised cosine
    starts.append(("raised_cos", normalize_sum(0.5 * (1 + np.cos(np.pi * 2 * tt)), S)))
    
    # Gaussian
    starts.append(("gaussian", normalize_sum(np.exp(-10 * (tt - 0.5) ** 2), S)))
    
    # Two bumps
    starts.append(("two_bump", normalize_sum(np.exp(-20 * (tt - 0.3) ** 2) + np.exp(-20 * (tt - 0.7) ** 2), S)))
    
    # Sinc-like
    starts.append(("sinc", normalize_sum(np.maximum(np.sinc(3 * (tt - 0.5)) ** 2, 0.001), S)))
    
    # Linear ramp (asymmetric)
    starts.append(("ramp", normalize_sum(np.maximum(tt, 0.01), S)))
    
    # Inverse ramp
    starts.append(('inv_ramp', normalize_sum(np.maximum(1 - tt, 0.01), S)))
    
    # Random smooth
    freqs = np.fft.fftfreq(n)
    random_spec = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
    random_spec = random_spec * np.exp(-5 * freqs ** 2)  # Low-pass
    a_rand = np.real(ifft(random_spec))[:n]
    starts.append(("rand_smooth", normalize_sum(np.maximum(a_rand, 0), S)))
    
    # Pure random
    starts.append(("random", normalize_sum(rng.random(n) + 0.01, S)))
    
    # Palindromic random
    half = rng.random(n // 2) + 0.01
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, [half[0]], half[::-1]])
    starts.append(("pal_random", normalize_sum(pal[:n], S)))
    
    return starts


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)
    
    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None
    
    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192
    
    S = np.sqrt(2.0 * n)
    
    print("Starting: n=%d, budget=%.0fs, parent_len=%s" % (
        n, budget_s, len(parent) if parent is not None else "None"))
    
    global_best = np.inf
    global_a = None
    global_n = n
    
    def update_best(a_candidate, label):
        nonlocal global_best, global_a, global_n
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        if a_norm.sum() < 0.01:
            return np.inf
        c_chk = fft_convolve_self(a_norm)
        score = 2.0 * len(a_norm) * c_chk.max() / (a_norm.sum() ** 2)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            global_n = len(a_norm)
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: score=%.6f (max_b=%.6f, sum=%.4f) [%ds]" % (
                label, len(a_norm), score, c_chk.max(), a_norm.sum(), int(elapsed)))
        return score
    
    # Generate diverse starts
    print("\nGenerating diverse starts...")
    starts = generate_diverse_starts(n, S, rng, parent)
    
    # Quick evaluation of all starts
    print("Evaluating starts:")
    start_scores = []
    for name, arr in starts:
        b = fft_convolve_self(arr)
        score = 2.0 * n * b.max() / (arr.sum() ** 2)
        start_scores.append((score, name, arr))
        print("  %s: score=%.6f" % (name, score))
    
    start_scores.sort(key=lambda x: x[0])
    
    # Phase 1: EMD on top starts (40% of budget)
    print("\n=== Phase 1: Entropic Mirror Descent ===")
    p1_budget = (deadline - time.monotonic()) * 0.35
    top_k = min(5, len(start_scores))
    p1_per = p1_budget / top_k
    
    for i in range(top_k):
        if time.monotonic() > deadline - (deadline - t_start) * 0.5 - 60:
            break

        if t_alloc < 10:
            continue
        print("  EMD on %s (budget %.0fs)..." % (name, t_alloc))
        a_opt, _ = run_emd(arr, n, S, t_alloc, rng,
                           beta_init=20.0, beta_max=3000.0,
                           beta_ramp=2000, momentum=0.6)
        update_best(a_opt, "emd_" + name)
    
    print("After Phase 1: best=%.6f" % global_best)
    
    # Phase 2: FISTA on best (20% of remaining)
    print("\n=== Phase 2: FISTA Polish ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        p2_budget = (deadline - time.monotonic()) * 0.20
        if p2_budget > 15:

                                            beta_init=50.0, beta_max=5000.0,

                    if remaining3 > 10:

                                                   remaining3 * 0.8, rng,
                                                   beta_init=500.0, beta_max=10000.0,
                                                   beta_ramp=500)

            if factor > 1:
                t_old = np.linspace(0, 1, global_n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                blocks = global_n // alt_n
                alt_start = global_a[:blocks * alt_n].reshape(alt_n, blocks).mean(axis=1)
            
            alt_start = normalize_sum(alt_start, S_alt)
            t_alt = remaining * 0.3
            if t_alt > 10:
                a_alt, _ = run_emd(alt_start, alt_n, S_alt, t_alt, rng,
                                   beta_init=20.0, beta_max=3000.0,
                                   beta_ramp=1000, momentum=0.6)
                update_best(a_alt, "alt_n_%d" % alt_n)
            
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
    
    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification FAILED, using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
    
    if not np.isfinite(ver):
        print("  Even fallback failed, using uniform")
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
    
    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(final_list), ver, time.monotonic() - t_start))
    return final_list
