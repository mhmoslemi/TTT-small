"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA with adaptive LSE, LP direction finding, multi-scale.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lp_find_direction(a, c, max_constraints=200, margin=1.0):
    """
    Find a direction g >= 0 that maximizes sum(g) subject to
    convolve(a, g)[k] <= c[k] * margin for top-k lags.
    Returns the direction or None.
    """
    n = len(a)
    M = c.max()
    
    # Select active constraints: top max_constraints lags
    # or lags within a fraction of max
    active = np.argsort(c)[::-1][:max_constraints]
    
    if len(active) < 1:
        return None
    
    # Build LP: maximize sum(g) subject to:
    # g >= 0
    # convolve(a, g)[k] <= M * margin for k in active
    # g[i] <= 1 (upper bound)
    
    # For each active lag k, the constraint is:
    # sum_j a[k-j] * g[j] <= M * margin
    # where j ranges over valid indices
    
    A_ub = np.zeros((len(active), n))
    b_ub = np.full(len(active), M * margin)
    
    for row_idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        a_indices = k - np.arange(j_min, j_max + 1)
        A_ub[row_idx, j_min:j_max+1] = a[a_indices]
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, 1.0)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    
    return None

    Improved FISTA with adaptive mode and optional LP direction finding.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    lp_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (y.sum() ** 2)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1
        
        # LP direction finding when stuck
        if use_lp and stuck_count > lp_interval and it - last_improve > lp_interval * 2:

                # Blend

                y_new = (1 - t_blend) * y + t_blend * g_norm

                a = y.copy()
                t_mom = 1.0
                stuck_count = 0
                lp_count += 1
                continue

            grad = lse_gradient(y, c, beta)
        elif mode == "exact":
            grad = exact_subgradient(y, c, tol_frac=0.0)
        elif mode == "adaptive":
            if stuck_count > 5000:
                grad = exact_subgradient(y, c, tol_frac=0.01)
            elif stuck_count > 1000:
                grad = lse_gradient(y, c, beta * 0.5)
            else:
                grad = lse_gradient(y, c, beta)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step

            step *= 0.1
        elif stuck_count > 3000:
            step *= 0.3

            a = y.copy()
            t_mom = 1.0
            stuck_count = max(0, stuck_count - 2000)
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart
        if it % 10000 == 0:
            t_mom = 1.0
    
    # Final check
    final_obj = compute_obj(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimization: adaptive FISTA with LP direction finding,
multi-scale upsample, and final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE-smoothed gradient."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from active lags."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    active = np.unique(active)
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lp_direction(a, c, max_active=50, margin=0.995, time_limit=5.0):
    """
    Find LP direction g >= 0 maximizing sum(g) s.t. conv(a,g)[k] <= c[k]*margin.
    Returns g or None.
    """
    n = len(a)
    M = c.max()
    
    # Select active lags: top max_active by value
    active = np.argsort(c)[::-1][:max_active]
    # Only include lags where c[k] is significant
    active = active[c[active] > 0.5 * M]
    
    if len(active) < 1:
        return None
    
    # Build constraint matrix
    A_ub = np.zeros((len(active), n))
    b_ub = np.zeros(len(active))
    
    for row_idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        a_idx = k - np.arange(j_min, j_max + 1)
        A_ub[row_idx, j_min:j_max+1] = a[a_idx]
        b_ub[row_idx] = c[k] * margin
    
    c_obj = -np.ones(n)
    bounds = [(0.0, 1.0)] * n
    
    try:
        t_start = time.monotonic()
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': time_limit, 'presolve': True})
        elapsed = time.monotonic() - t_start
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    
    return None
