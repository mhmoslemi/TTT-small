"""
Minimize 2*n*max_conv/sum^2 using hard top-k gradient descent with adaptive
momentum, Armijo line search, spectral equalization, and multi-start search.

Key improvements over previous LSE-LBFGS approach:
1. Hard top-k gradient (sharper descent near optimum, no soft approximation)

3. Adaptive Nesterov momentum with cosine decay
4. Periodic spectral equalization to flatten power spectrum variance
5. Peak balancing term to escape saddle points
6. Multi-scale n search (8192 and 16384)
7. Extended final polishing phase

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def topk_gradient(c, a, N, top_k, balance_lambda=0.0):
    """Compute gradient of sum of top-k convolution values (with optional variance balancing).
    
    Returns (gradient, top_indices).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    if top_vals.sum() < 1e-30:
        return np.zeros(n), top_idx
    
    # Uniform weights for top-k peaks
    # Gradient: grad = 2/k * sum_{p in top_k} shift(a, p)
    # In FFT: 2/k * real(ifft(sum_{p} exp(-2pi*i*freq*p/N) * conj(A)))... 
    # Actually use the cross-correlation formula:
    # grad = 2/k * real(ifft(W * conj(Af)))[:n] where W = fft(w)
    
    w_full = np.zeros(max(2 * n - 1, N))
    for p in top_idx:
        p_int = int(p)
        if 0 <= p_int < len(w_full):
            w_full[p_int] = 1.0 / top_k
    
    fsize = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fsize)
    w_padded[:len(w_full)] = w_full
    W = fft(w_padded)
    a_padded = np.zeros(fsize)
    a_padded[:n] = a
    Af = fft(a_padded)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    
    # Optional: add variance balancing term for top-k peaks
    if balance_lambda > 0 and top_k > 1:
        mean_top = top_vals.mean()
        # Gradient of variance: 2/k * sum (c_p - mean) * dc_p/da
        # This is similar but with weighted peaks
        w_bal = np.zeros(fsize)
        for i, p in enumerate(top_idx):
            p_int = int(p)
            if 0 <= p_int < len(w_bal):
                w_bal[p_int] = (top_vals[i] - mean_top) / top_k
        W_bal = fft(w_bal)
        grad_bal = 2.0 * np.real(ifft(W_bal * np.conj(Af)))[:n]
        grad += balance_lambda * grad_bal
    
    return grad, top_idx


def spectral_equalize(a, N, alpha):
    """One step of spectral equalization: flatten power spectrum to reduce conv variance."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    P_std = P.std()
    if P_std < 1e-15:
        return a
    # Reduce variance of power spectrum
    P_new = P * (1.0 - alpha * (P - P_mean) / P_std)
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def safe_normalize(a, n, S):
    """Clip to [0, 1000] and normalize to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    inits.append(("rand", make_init_rand(n1, rng)))
    inits.append(("chirp", make_init_chirp(n1, rng)))
    inits.append(("multifreq", make_init_multi_freq(n1, rng)))
    inits.append(("fourier", make_init_fourier(n1, rng, N1)))
    
    # Quick exploration: try each init briefly
    n_inits = len(inits)
    per_init_quick = min(30.0, part1_budget / (n_inits * 3))
    
    print("  Quick exploration: %d inits x %.1fs" % (n_inits, per_init_quick))
    best_init = None
    best_init_score = np.inf
    
    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        a_norm = safe_normalize(arr, n1, S1)
        a_test, sc_test = main_optimize(a_norm, n1, S1, N1, per_init_quick, rng,
                                         top_k_start=5, top_k_end=20,
                                         balance_start=0.0, balance_end=0.2,
                                         spec_interval=20)
        update_best(a_test, n1)
        if sc_test < best_init_score:
            best_init_score = sc_test
            best_init = a_test.copy()
    
    # Main optimization from best init (polyphase)
    remaining = (t1 + part1_budget) - time.monotonic()
    if remaining > 60 and best_init is not None:
        print("  Polyphase optimization (%.0fs):" % remaining)
        a_pp, sc_pp = polyphase_optimize(best_init, n1, S1, N1, remaining, rng)
        update_best(a_pp, n1)
    
    # === Part 2: n=16384 search (if time permits) ===
    remaining_for_p2 = deadline - time.monotonic()
    if remaining_for_p2 > 300:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        N2 = 1 << (2 * n2 - 1).bit_length()
        part2_budget = min(500.0, remaining_for_p2 * 0.5)
        
        print("=== Part 2: n=%d (budget %.0fs) ===" % (n2, part2_budget))
        
        inits2 = [("rand", make_init_rand(n2, rng)),
                   ("chirp", make_init_chirp(n2, rng)),
                   ("fourier", make_init_fourier(n2, rng, N2))]
        
        # If we have a good n=8192 result, upsample it
        if global_a is not None and global_n == n1:
            # Upsample by interpolation
            from scipy.interpolate import interp1d
            x_old = np.arange(n1)
            x_new = np.arange(n2)
            f_interp = interp1d(x_old, global_a, kind='cubic', fill_value='extrapolate')
            a_up = np.maximum(f_interp(x_new), 0)
            inits2.insert(0, ("upsampled", a_up))
        
        part2_per_init = part2_budget / (len(inits2) * 2)
        part2_per_init = max(20.0, part2_per_init)
        
        best_init2 = None
        best_init2_score = np.inf
        
        for name, arr in inits2:
            if time.monotonic() > deadline - 100:
                break
            a_norm = safe_normalize(arr, n2, S2)
            a_test, sc_test = main_optimize(a_norm, n2, S2, N2, part2_per_init, rng,
                                             top_k_start=5, top_k_end=25,
                                             balance_start=0.0, balance_end=0.2,
                                             spec_interval=30)
            update_best(a_test, n2)
            if sc_test < best_init2_score:
                best_init2_score = sc_test
                best_init2 = a_test.copy()
        
        # Polyphase on best n=16384 init
        remaining2 = deadline - time.monotonic()
        if remaining2 > 120 and best_init2 is not None:
            a_pp2, sc_pp2 = polyphase_optimize(best_init2, n2, S2, N2,
                                                min(remaining2 - 30, 400.0), rng)
            update_best(a_pp2, n2)
    
    # === Part 3: Final polishing on best result ===
    remaining_final = deadline - time.monotonic()
    if remaining_final > 60 and global_a is not None:
        n_f = global_n
        S_f = np.sqrt(2.0 * n_f)
        N_f = 1 << (2 * n_f - 1).bit_length()
        
        print("=== Part 3: Final polishing (n=%d, %.0fs) ===" % (n_f, remaining_final))
        a_fin, sc_fin = main_optimize(global_a, n_f, S_f, N_f,
                                       min(remaining_final - 20, 400.0), rng,
                                       top_k_start=1, top_k_end=5,
                                       balance_start=0.0, balance_end=0.1,
                                       spec_interval=200)
        update_best(a_fin, n_f)
        
        # Extra spectral equalization passes
        if time.monotonic() < deadline - 30:
            for sp in range(20):
                if time.monotonic() > deadline - 30:
                    break
                global_a = spectral_equalize(global_a, N_f, 0.05)
                c_chk, _ = compute_conv(global_a, N_f)
                if c_chk.max() < global_best:
                    global_best = c_chk.max()
                    print("  [spec_pass %d] %.6f" % (sp, global_best))
    
    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    n_ret = len(global_a)
    S_ret = np.sqrt(2.0 * n_ret)
    N_ret = 1 << (2 * n_ret - 1).bit_length()
    c_final, _ = compute_conv(global_a, N_ret)
    print("Returning length %d, internal max_conv=%.6f, R_equiv=%.6f" %
          (n_ret, c_final.max(), c_final.max() * 2 * n_ret / (global_a.sum()**2)))
    
    return [float(x) for x in global_a]
