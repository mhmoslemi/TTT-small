import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Full self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad(a, N, beta):
    """
    LSE-smoothed max of self-convolution and its gradient.
    Returns (lse_value, gradient, c, c_max).
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def spectral_shaping(a, N, top_k, alpha):
    """Reduce power at frequency bins corresponding to convolution peaks."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        kk = int(k) % N
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * kk / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)

        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 4096 and given.size <= 65536:
            n = len(given)
            given_arr = np.clip(given, 0.0, 1000.0)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    # Phase 1: Multi-start exploration
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(350.0, (deadline - t1) * 0.14)

    inits = []
    if given_arr is not None and len(given_arr) == n:
        inits.append(("given", given_arr.copy()))
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("multifreq", make_init_multifreq(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d initializations, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 800:
            break
        a_norm = project_simplex(arr.astype(np.float64), S)
        t0 = time.monotonic()
        rem = max(2.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc = nesterov_lse_block(a_norm, n, S, N, min(rem, per_init),
                                       rng, beta_init=5.0, beta_max=80.0,
                                       beta_ramp=1.04, step_factor=1.5,
                                       spectral_every=300, noise_every=800)
        # Update global best
        a_norm2 = project_simplex(a_l, S)
        c_chk = compute_conv(a_norm2, N)
        sc2 = c_chk.max()
        if sc2 < global_best:
            global_best = sc2
            global_a = a_norm2.copy()
            print("  [%s] cmax=%.6f" % (name, global_best))

    print("Phase 1 done. Best cmax=%.6f" % global_best)

    if global_a is None:
        global_a = project_simplex(make_init_rand(n, rng), S)
        global_best = compute_conv(global_a, N).max()

    # Phase 2: Main Nesterov+LSE optimization in blocks
    print("=== Phase 2: Main optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(1300.0, (deadline - t2) * 0.50)

    a_curr = global_a.copy()
    stall_count = 0
    block_idx = 0

    while time.monotonic() < t2 + phase2_budget:
        rem = (t2 + phase2_budget) - time.monotonic()
        if rem < 15:
            break
        block_time = min(250.0, rem)
        block_idx += 1

        # Increasing temperature schedule
        beta_start = 8.0 + 15.0 * block_idx
        beta_end = min(2000.0, 40.0 * (1.0 + block_idx) ** 1.2)
        step_f = max(0.4, 2.0 - 0.15 * block_idx)

        a_new, sc_new = nesterov_lse_block(a_curr, n, S, N, block_time, rng,
                                            beta_init=beta_start, beta_max=beta_end,
                                            beta_ramp=1.05, step_factor=step_f,
                                            spectral_every=500, noise_every=2000)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            est_R = 2.0 * n * global_best / S**2
            print("  [P2 b%d] cmax=%.6f (R=%.6f, beta %.0f-%.0f)"
                  % (block_idx, global_best, est_R, beta_start, beta_end))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count >= 3:
                a_curr = project_simplex(
                    a_curr + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)
                stall_count = 0

    # Phase 3: High-temperature polish
    print("=== Phase 3: High-temp polish ===")
    t3 = time.monotonic()
    phase3_budget = min(500.0, (deadline - t3) * 0.20)

    if phase3_budget > 10:
        a_pol, sc_pol = nesterov_lse_block(global_a, n, S, N, phase3_budget, rng,
                                            beta_init=80.0, beta_max=3000.0,
                                            beta_ramp=1.10, step_factor=0.7,
                                            spectral_every=600, noise_every=4000)
        if sc_pol < global_best:
            global_best = sc_pol
            global_a = a_pol.copy()
            est_R = 2.0 * n * global_best / S**2
            print("  [polish] cmax=%.6f (R=%.6f)" % (global_best, est_R))

    # Phase 4: Spectral shaping refinement
    print("=== Phase 4: Spectral refinement ===")
    t4 = time.monotonic()
    rem4 = min(150.0, max(0.0, (deadline - t4) * 0.08))

    if rem4 > 10:
        a_sp = global_a.copy()
        t_sp = time.monotonic()
        while time.monotonic() - t_sp < rem4:
            a_sp = spectral_shaping(a_sp, N, top_k=5, alpha=0.03)
            c_sp = compute_conv(a_sp, N)
            sc_sp = c_sp.max()
            if sc_sp < global_best:
                global_best = sc_sp
                global_a = a_sp.copy()
                est_R = 2.0 * n * global_best / S**2
                print("  [spectral] cmax=%.6f (R=%.6f)" % (global_best, est_R))

    # Phase 5: Peak shaving
    rem5 = max(0.0, deadline - time.monotonic() - 5.0)
    if rem5 > 20:
        print("=== Phase 5: Peak shaving (%.0fs) ===" % rem5)
        a_ps, sc_ps = peak_shave(global_a, N, min(rem5, 100.0), rng)
        if sc_ps < global_best:
            global_best = sc_ps
            global_a = a_ps.copy()
            est_R = 2.0 * n * global_best / S**2
            print("  [peak_shave] cmax=%.6f (R=%.6f)" % (global_best, est_R))

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Final verification: R = %.6f" % ver)
            else:
                print("Verification returned inf, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("Verification error: %s" % str(e))

    est_R_final = 2.0 * len(global_a) * global_best / (np.sum(global_a) ** 2) if np.sum(global_a) > 0 else float('inf')
    print("Returning length %d, cmax=%.6f, est R=%.6f" % (len(global_a), global_best, est_R_final))
    return [float(x) for x in global_a]
