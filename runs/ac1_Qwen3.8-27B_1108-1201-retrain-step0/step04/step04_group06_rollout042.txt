def mirror_step(a, g, eta, clip=50):
    # center g
    gc = g - np.mean(g)
    scale = np.max(np.abs(gc))
    if scale < 1e-15: return a

```python
"""
Search for non-negative sequences minimizing 2*n*max(a*a)/(sum a)^2.

The implementation works with sequences normalized to sum 1, so the score is
equivalent to 2*n*max(a*a).  The main engine is an active-set exponentiated
gradient / mirror descent method on the simplex, with two variants:
  1. full simplex updates, and
  2. updates on the anti-palindromic manifold a[i] + a[n-1-i] = 2/n,
     which is very useful for keeping the central convolution peak low.

Several structured starts are used, including the previous construction if
present in the global `height_sequence_1`, Thue-Morse anti-palindromic
constructs, Rudin-Shapiro-inspired constructs, random/smooth anti-palindromic
sequences, and a final spectral shaping pass.
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def _pow2_at_least(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_sum1(a):
    """Clip to nonnegative finite values and rescale to sum 1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def conv_self(a, N):
    """Linear self-convolution by zero-padded FFT. Returns length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    A = fft(a, N)
    b = np.real(ifft(A * A))[:2 * n - 1]
    return b


def internal_q(a, n, N):
    """Internal objective for sum-1 sequence: 2*n*max(conv)."""
    b = conv_self(a, N)
    if np.any(~np.isfinite(b)):
        return np.inf
    mx = float(np.max(b))
    if mx < 0.0:
        mx = 0.0
    return 2.0 * float(n) * mx


def anti_from_u(n, u):
    """Build anti-palindromic sequence of length n from left-half weights u."""
    m = n // 2
    c = 1.0 / float(m)
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    if u.size < m:
        u = np.pad(u, (0, m - u.size), mode="edge")
    else:
        u = u[:m]
    u = np.clip(u, 0.0, 1.0)
    a = np.empty(n, dtype=np.float64)
    a[:m] = c * u
    a[m:] = c * (1.0 - u[::-1])
    return a


def extract_u(a, n):
    """Extract anti-palindromic left-half weights u (ratio in each pair)."""
    m = n // 2
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    left = np.clip(a[:m], 0.0, 1000.0)
    right = np.clip(a[m:][::-1], 0.0, 1000.0)  # index order matching left pairs
    s = left + right
    u = np.where(s > 1e-15, left / np.where(s > 1e-15, s, 1.0), 0.5)
    return np.clip(u, 1e-12, 1.0 - 1e-12)


def project_anti(a):
    """Project to the exact anti-palindromic sum-1 manifold."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    return anti_from_u(a.size, extract_u(a, a.size))


def mirror_step(a, g, eta, clip=3.0):
    """
    Exponentiated gradient step on the simplex:
       a_new[i] proportional to a[i] * exp(-eta * center(g)[i])
    """
    gc = g - np.mean(g)
    sc = float(np.max(np.abs(gc)))
    if sc < 1e-15:
        return a
    x = -eta * gc / sc
    x = np.clip(x, -clip, clip)
    a_safe = np.maximum(a, 1e-300)
    loga = np.log(a_safe) + x
    loga -= float(loga.max())
    out = np.exp(loga)
    s = float(out.sum())
    if s <= 1e-300:
        return a
    return out / s


def logit_update_u(u, h, eta, clip=3.0):
    """Exponentiated gradient step for u in (0,1)^m using logit map."""
    eps = 1e-12
    uc = np.clip(u, eps, 1.0 - eps)
    hv = h - np.mean(h)
    sc = float(np.max(np.abs(hv)))
    if sc < 1e-15:
        return u
    x = -eta * hv / sc
    x = np.clip(x, -clip, clip)
    lt = np.log(uc) - np.log1p(-uc)
    lt += x
    lt = np.clip(lt, -50.0, 50.0)
    return 1.0 / (1.0 + np.exp(-lt))


def perturb_log(a, sigma, rng):
    """Small multiplicative random perturbation on the simplex."""
    a_safe = np.maximum(a, 1e-300)
    loga = np.log(a_safe) + sigma * rng.standard_normal(a.size)
    loga -= float(loga.max())
    out = np.exp(loga)
    s = float(out.sum())
    if s <= 1e-300:
        return a
    return out / s


def select_active(q, K, n):
    """Select active convolution lags and normalized weights."""
    L = 2 * n - 1
    K = int(max(1, min(K, L)))
    q = np.asarray(q, dtype=np.float64)
    if K >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(q, -K)[-K:].astype(np.int64)

    comp = (L - 1) - idx
    idx = np.unique(np.concatenate((idx, comp)))
    maxsize = 2 * K
    if idx.size > maxsize:
        order = np.argsort(q[idx])[::-1]
        idx = idx[order[:maxsize]]

    w = np.maximum(q[idx], 1e-12).astype(np.float64)
    sw = float(w.sum())
    if sw <= 1e-300:
        w = np.ones(idx.size, dtype=np.float64) / max(1, idx.size)
    else:
        w = w / sw
    return idx, w


def active_gradient(a, idx, w):
    """
    Gradient of weighted sum of selected convolution coefficients.

    For b_k = sum_i a_i a_{k-i},
       db_k / da_i = 2 a_{k-i} if 0 <= k-i < n, else 0.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    if idx.size == 0:
        return g

    a_rev = a[::-1]
    for ii in range(idx.size):
        wt = 2.0 * float(w[ii])
        if wt <= 1e-14:
            continue
        k = int(idx[ii])

        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            start = n - 1 - (k - lo)
            end = n - k + hi
            g[lo:hi + 1] += wt * a_rev[start:end]
    return g


def should_accept(q_new, q_cur, q_best, rng, T):
    """Annealed acceptance for local search."""
    if not np.isfinite(q_new):
        return False
    if q_new <= q_best + 1e-12:
        return True
    if q_new <= q_cur + 1e-12:
        return True
    diff = q_new - q_cur
    if diff < 0.1:
        T = max(1e-6, float(T))
        if rng.random() < np.exp(-diff / T):
            return True
    return False


def gradient_pass(a, mode, n, N, rng, time_budget,
                  K_min=2, K_max=20, eta0=0.5, target=1.5020):
    """
    Active-set mirror descent.

    mode = 'full' : update all n coordinates on the simplex.
    mode = 'anti' : restrict to anti-palindromic manifold.
    Returns (best_a, best_q).
    """
    start = time.monotonic()
    deadline = start + max(1.0, float(time_budget))

    a = normalize_sum1(a)
    if mode == "anti":
        u = extract_u(a, n)
    else:
        u = None

    best_a = a.copy()
    best_q = internal_q(best_a, n, N)

    eta = float(eta0)
    it = 0
    stall = 0
    m = n // 2

    while time.monotonic() < deadline:
        if mode == "anti":
            a = anti_from_u(n, u)

        b = conv_self(a, N)
        q = 2.0 * float(n) * b
        q = np.maximum(q, 0.0)
        if np.any(~np.isfinite(q)):
            break
        qmax = float(q.max())

        if qmax < best_q - 1e-10:
            best_q = qmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if best_q <= target:
            break

        # Adaptive number of active lags: more lags as we approach the target.
        if qmax > 1.80:
            K = K_min
        elif qmax > 1.60:
            K = max(K_min + 1, (K_min + K_max) // 4)
        elif qmax > 1.55:
            K = (K_min + K_max) // 2
        elif qmax > 1.52:
            K = int(0.75 * (K_min + K_max))
        else:
            K = K_max

        idx, w = select_active(q, K, n)
        g = active_gradient(a, idx, w)

        accepted = False
        q_try = qmax
        T = 0.003 + 0.02 * min(1.0, stall / 50.0)

        if mode == "anti":
            h = g[:m] - g[n - 1:n - 1 - m:-1]
            for e in (eta, 0.5 * eta):
                u_try = logit_update_u(u, h, e)
                a_try = anti_from_u(n, u_try)
                q_try = internal_q(a_try, n, N)
                if should_accept(q_try, qmax, best_q, rng, T):
                    u = u_try
                    a = a_try
                    accepted = True
                    break

            if not accepted and stall > 30 and rng.random() < 0.3:
                u_try = logit_update_u(u, rng.standard_normal(m), 0.2)
                a_try = anti_from_u(n, u_try)
                q_try = internal_q(a_try, n, N)
                if should_accept(q_try, qmax, best_q, rng, 0.05):
                    u = u_try
                    a = a_try
                    accepted = True

        else:
            for e in (eta, 0.5 * eta):
                a_try = mirror_step(a, g, e)
                q_try = internal_q(a_try, n, N)
                if should_accept(q_try, qmax, best_q, rng, T):
                    a = a_try
                    accepted = True
                    break

            if not accepted and stall > 30 and rng.random() < 0.3:
                a_try = perturb_log(a, 0.1, rng)
                q_try = internal_q(a_try, n, N)
                if should_accept(q_try, qmax, best_q, rng, 0.05):
                    a = a_try
                    accepted = True

        if accepted:
            if q_try < qmax:
                eta = min(1.5, eta * 1.08)
            else:
                eta = max(0.05, eta * 0.8)
        else:
            eta = max(0.02, eta * 0.5)
            if stall > 250:
                break

        it += 1
        if it % 800 == 0:
            print("  gradient pass %s iter=%d best=%.8f current=%.8f" %
                  (mode, it, best_q, qmax), flush=True)

    return best_a, best_q


def spectral_shape_pass(a, n, N, rng, time_budget, target=1.5020):
    """
    Optional spectral shaping pass.  Modifies the power spectrum to reduce
    active peak lags, reconstructs a real sequence, and projects to simplex.
    """
    start = time.monotonic()
    deadline = start + max(1.0, float(time_budget))
    a = normalize_sum1(a)
    best_a = a.copy()
    best_q = internal_q(best_a, n, N)
    alpha = 0.35
    it = 0
    stall = 0

    freq = np.arange(N, dtype=np.float64)
    twopi_over_N = 2.0 * np.pi / float(N)
    cos_cache_base = twopi_over_N * freq  # shape (N,)

    while time.monotonic() < deadline:
        b = conv_self(a, N)
        q = 2.0 * float(n) * np.maximum(b, 0.0)
        qmax = float(q.max())
        if qmax < best_q - 1e-10:
            best_q = qmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if best_q <= target:
            break

        idx, w = select_active(q, 8, n)
        A = fft(a, N)
        P = np.abs(A) ** 2

        ceff = np.zeros(N, dtype=np.float64)
        for ii in range(idx.size):
            k = int(idx[ii])
            wt = float(w[ii])
            ceff += wt * np.cos(cos_cache_base * k)

        Pnew = P * np.exp(-alpha * np.maximum(ceff, 0.0))
        if float(Pnew.max()) <= 0.0:
            break

        phase = np.angle(A)
        Anew = np.sqrt(np.maximum(Pnew, 0.0)) * np.exp(1j * phase)
        anew = np.real(ifft(Anew))[:n]
        anew = np.maximum(anew, 0.0)
        anew = normalize_sum1(anew)
        qnew = internal_q(anew, n, N)

        if should_accept(qnew, qmax, best_q, rng, 0.02):
            a = anew
            alpha = min(1.0, alpha * 1.05)
        else:
            stall += 1
            alpha = max(0.05, alpha * 0.7)

        if stall > 120:
            break

        it += 1
        if it % 500 == 0:
            print("  spectral pass iter=%d best=%.8f" % (it, best_q), flush=True)

    return best_a, best_q


def thue_u(n):
    """Thue-Morse weights for anti-palindromic sequence length n."""
    m = n // 2
    u = np.empty(m, dtype=np.float64)
    for i in range(m):
        u[i] = float(i.bit_count() & 1)
    return u


def rudin_shapiro_block(p):
    """Rudin-Shapiro +/-1 block of length p (power of two)."""
    r = np.array([1.0], dtype=np.float64)
    s = np.array([1.0], dtype=np.float64)
    while r.size < p:
        r, s = np.concatenate((r, s)), np.concatenate((r, -s))
    return r[:p]


def rs_u(n):
    """Rudin-Shapiro-inspired anti-palindromic left weights."""
    m = n // 2
    p = _pow2_at_least(m)
    r = rudin_shapiro_block(p)
    return 0.5 * (1.0 + r[:m])


def make_initial_candidates(n, rng):
    """Create diverse structured starting points."""
    cands = []

    # Previous state-of-the-art, if available and compatible.
    given_arr = None
    try:
        given_seq = height_sequence_1
    except Exception:
        given_seq = None
    if given_seq is not None:
        try:
            given_arr = np.asarray(given_seq, dtype=np.float64).reshape(-1)
        except Exception:
            given_arr = None

    if given_arr is not None and given_arr.size == n:
        g = np.clip(np.nan_to_num(given_arr, nan=0.0, posinf=1000.0, neginf=0.0), 0.0, 1000.0)
        s = float(g.sum())
        if s > 0.01:
            cands.append(("prev", g / s, "full"))
            cands.append(("prevanti", project_anti(g / s), "anti"))

    # Thue-Morse anti-palindromic.
    cands.append(("thue", anti_from_u(n, thue_u(n)), "anti"))

    # Rudin-Shapiro-inspired anti-palindromic.
    cands.append(("rs", anti_from_u(n, rs_u(n)), "anti"))

    # Random binary anti-palindromic.
    u = rng.integers(0, 2, size=n // 2).astype(np.float64)
    cands.append(("randbits", anti_from_u(n, u), "anti"))

    # Smooth anti-palindromic shapes.
    m = n // 2
    t = np.arange(m, dtype=np.float64) / float(m)
    for freq in (1, 2, 3, 5, 8, 13):
        ph1 = rng.random() * 2.0 * np.pi
        ph2 = rng.random() * 2.0 * np.pi
        u = 0.5 + 0.30 * np.sin(2.0 * np.pi * freq * t + ph1)
        u += 0.15 * np.sin(2.0 * np.pi * (2 * freq + 1) * t + ph2)
        u += 0.05 * rng.standard_normal(m)
        u = np.clip(u, 0.01, 0.99)
        cands.append(("smooth%d" % freq, anti_from_u(n, u), "anti"))

    # Uniform random anti-palindromic.
    u = rng.random(m)
    cands.append(("randuni", anti_from_u(n, u), "anti"))

    # Generic full simplex starts.
    a = 0.2 + 0.8 * rng.random(n)
    cands.append(("randfull", normalize_sum1(a), "full"))

    a = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * np.arange(n) / n)
    a += 0.1 * np.sin(2.0 * np.pi * 7.0 * np.arange(n) / n)
    a += 0.05 * rng.standard_normal(n)
    cands.append(("sinfull", normalize_sum1(a), "full"))

    # Fallback.
    cands.append(("uniform", np.ones(n, dtype=np.float64) / n, "full"))

    return cands


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search entry point.
    Returns a list of non-negative floats minimizing evaluate_sequence.
    """
    rng = np.random.default_rng(int(seed))
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(60.0, budget_s - 45.0)

    # Use the historical length when it is sane; otherwise use 8192.
    n = 8192
    try:
        given_seq = height_sequence_1
        gsize = np.asarray(given_seq, dtype=np.float64).size
        if 4096 <= gsize <= 16384 and (gsize % 2 == 0):
            n = int(gsize)
    except Exception:
        pass

    N = _pow2_at_least(2 * n - 1)

    print("Starting search: n=%d, FFT-N=%d, budget=%.1fs" % (n, N, budget_s), flush=True)

    cands = make_initial_candidates(n, rng)
    scored = []
    best_a = None
    best_q = np.inf

    for name, arr, kind in cands:
        arr = normalize_sum1(arr)
        q = internal_q(arr, n, N)
        if np.isfinite(q):
            scored.append((float(q), arr.copy(), kind, name))

    if not scored:
        best_a = np.ones(n, dtype=np.float64) / n
        best_q = internal_q(best_a, n, N)
    else:
        scored.sort(key=lambda x: x[0])
        best_a = scored[0][1].copy()
        best_q = scored[0][0]

    print("Best initial score=%.8f (%s)" % (best_q, scored[0][3]), flush=True)

    # ------------------------------------------------------------------
    # Phase 1: short passes from the best diverse starts.
    # ------------------------------------------------------------------
    # Keep a few distinct-looking starts: ranked by score, plus ensure at least one anti.
    top_starts = []
    seen_names = set()
    for q, arr, kind, name in scored:
        if name in seen_names:
            continue
        top_starts.append((arr, kind, name))
        seen_names.add(name)
        if len(top_starts) >= 7:
            break

    # Make sure one anti start is present.
    if not any(k == "anti" for _, k, _ in top_starts):
        for q, arr, kind, name in scored:
            if kind == "anti":
                top_starts.append((arr, kind, name))
                break

    for arr, kind, name in top_starts:
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        budget = min(60.0, 0.10 * rem)
        mode = "anti" if kind == "anti" else "full"
        if mode == "anti":
            arr0 = project_anti(arr)
        else:
            arr0 = normalize_sum1(arr)
        a_res, q_res = gradient_pass(
            arr0, mode, n, N, rng, budget,
            K_min=2, K_max=16, eta0=0.6
        )
        if q_res < best_q:
            best_q = q_res
            best_a = a_res.copy()
            print("Phase1 start %s improved to %.8f" % (name, best_q), flush=True)
        if best_q <= 1.5010:
            break

    # ------------------------------------------------------------------
    # Phase 2: alternating polishing of the current best.
    # ------------------------------------------------------------------
    alt = 0
    polish_count = 0
    while time.monotonic() < deadline - 60.0:
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        budget = min(120.0, 0.25 * rem)

        if alt % 2 == 0:
            start_a = project_anti(best_a)
            qproj = internal_q(start_a, n, N)
            if qproj <= best_q + 0.12:
                mode = "anti"
            else:
                mode = "full"
                start_a = best_a
        else:
            mode = "full"
            start_a = best_a

        # Occasionally perturb the start to escape a local minimum.
        if polish_count and polish_count % 5 == 4:
            start_a = perturb_log(start_a, 0.08, rng)

        a_res, q_res = gradient_pass(
            start_a, mode, n, N, rng, budget,
            K_min=2, K_max=28, eta0=0.5
        )
        if q_res < best_q:
            best_q = q_res
            best_a = a_res.copy()
            print("Phase2 polish %s improved to %.8f" % (mode, best_q), flush=True)

        alt += 1
        polish_count += 1

        if best_q <= 1.5010:
            break

    # ------------------------------------------------------------------
    # Phase 3: spectral shaping if there is time.
    # ------------------------------------------------------------------
    rem = deadline - time.monotonic()
    if rem > 90.0:
        budget = min(160.0, rem - 45.0)
        a_res, q_res = spectral_shape_pass(best_a, n, N, rng, budget)
        if q_res < best_q:
            best_q = q_res
            best_a = a_res.copy()
            print("Phase3 spectral improved to %.8f" % best_q, flush=True)

    # ------------------------------------------------------------------
    # Phase 4: final small-step full polish.
    # ------------------------------------------------------------------
    rem = deadline - time.monotonic()
    if rem > 30.0:
        budget = min(70.0, rem - 15.0)
        a_res, q_res = gradient_pass(
            best_a, "full", n, N, rng, budget,
            K_min=4, K_max=36, eta0=0.2
        )
        if q_res < best_q:
            best_q = q_res
            best_a = a_res.copy()
            print("Phase4 final polish improved to %.8f" % best_q, flush=True)

    # ------------------------------------------------------------------
    # Final sanity check and fallback.
    # ------------------------------------------------------------------
    if best_a is None or not np.all(np.isfinite(best_a)):
        best_a = np.ones(n, dtype=np.float64) / n

    best_a = np.clip(np.asarray(best_a, dtype=np.float64).reshape(-1), 0.0, 1000.0)
    s = float(best_a.sum())
    if s < 0.01:
        best_a = np.ones(n, dtype=np.float64) / n
        s = float(best_a.sum())

    # Return normalized to sum 1; objective is scale invariant and this is
    # safely within evaluate_sequence's accepted range.
    final = best_a / s
    final = np.clip(final, 0.0, 1000.0)

    try:
        val = evaluate_sequence([float(x) for x in final])
        if np.isfinite(val):
            print("Final evaluate_sequence: %.8f (internal best=%.8f)" % (val, best_q), flush=True)
            if val < best_q:
                best_q = float(val)
        else:
            print("Final evaluate_sequence not finite; using fallback uniform.", flush=True)
            nfb = 256
            final = np.ones(nfb, dtype=np.float64) / nfb
    except Exception as exc:
        print("Verification error: %s" % exc, flush=True)

    print("Returning length=%d, best_score=%.8f" % (len(final), best_q), flush=True)
    return [float(x) for x in final]
```
