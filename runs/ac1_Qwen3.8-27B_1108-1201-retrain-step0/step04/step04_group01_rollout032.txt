"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Uses adaptive subgradient descent with Polyak step sizing, exact peak subgradients,
local coordinate descent polishing, and multi-start perturbation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective_value(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_subgradient(a, c, K, mode="excess"):
    """Compute weighted subgradient at top-K peaks using FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    K = min(K, size)
    
    # Find top K peaks
    idx = np.argsort(c)[::-1][:K]
    
    if mode == "excess":
        threshold = c.mean() + 0.3 * (c.max() - c.mean())
        weights = np.maximum(c[idx] - threshold, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(K) / K
        else:
            weights = weights / wsum
    elif mode == "uniform":
        weights = np.ones(K) / K
    elif mode == "rank":
        # Weight by rank: top peak gets most weight
        ranks = np.arange(K, 0, -1, dtype=np.float64)
        weights = ranks / ranks.sum()
    else:
        weights = np.ones(K) / K
    
    # Build full weight vector
    w_full = np.zeros(size)
    w_full[idx] = weights
    
    # Compute cross-correlation: grad[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * ifft(conj(fft(a)) * fft(w_full))
    fsize = 1 << (size + n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w_full, fsize)
    cross = np.real(ifft(np.conj(A) * W))[:n]
    grad = 2.0 * cross
    return grad


def run_subgradient_phase(a_init, max_time, rng, K_start=1, K_end=32,
                          step_mode="polyak", noise_period=500):

    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    
    # Normalize so objective = max(conv)
    S_target = np.sqrt(2.0 * n)
    if S > 1e-10:
        a = a * S_target / S
        S = S_target
    
    start = time.monotonic()
    best_obj = compute_objective_value(a)
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    current_max_b = c.max()
    
    # Theoretical lower bound for max(conv) with sum = sqrt(2n) is ~1
    lower_bound = 1.0
    
    step = 1.0 / (2.0 * n)  # initial step
    it = 0
    no_improve_count = 0
    patience = 800
    max_patience = 3000
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive K
        K = int(K_start + (K_end - K_start) * min(1.0, progress * 1.5))
        
        c = fft_convolve_self(a)
        current_max_b = c.max()
        
        if current_max_b < best_obj:

                step = 1e-8
        else:

            step = 1.0 / (0.5 * n * (1.0 + 0.01 * it))

                no_improve_count = 0

        best_a = a.copy()
    
    # Rescale to target sum

    if s > 1e-10:

def run_local_polish(a_init, max_time, rng):
    """Local coordinate descent: try perturbations at hot spots."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    
    start = time.monotonic()
    best_obj = compute_objective_value(a)
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    current_max_b = c.max()
    
    # Find "hot spots": indices that contribute most to peaks
    top_peaks = np.argsort(c)[::-1][:16]
    
    # For each peak, find the contributing indices
    hot_indices = set()
    for k in top_peaks:
        # a[j] * a[k-j] contributes to c[k]
        # The "hot" indices are where the product is largest
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            products = a[lo:hi+1] * a[k-lo:k-hi+1][::-1]
            top_local = np.argsort(products)[::-1][:8]
            for ti in top_local:
                hot_indices.add(lo + ti)
    
    hot_list = list(hot_indices)
    if not hot_list:
        hot_list = list(range(n))
    
    rng.shuffle(hot_list)
    
    # Pulse amplitude for perturbations
    base_pulse = S / n * 0.1
    
    while time.monotonic() - start < max_time:
        # Recompute hot spots every 100 iterations
        c = fft_convolve_self(a)
        current_max_b = c.max()
        
        if current_max_b < best_obj:
            best_obj = current_max_b
            best_a = a.copy()
        
        # Try perturbing a few hot indices
        n_try = min(50, len(hot_list))
        for i in range(n_try):
            if time.monotonic() - start > max_time:
                break
            j = hot_list[i % len(hot_list)]
            
            # Try small increases and decreases
            for delta in [base_pulse, -base_pulse, base_pulse * 0.5, -base_pulse * 0.5]:
                a[j] = max(0, a[j] + delta)
                # Re-normalize
                s = a.sum()
                if s < 1e-10:
                    a[j] = max(0, a[j] - delta)
                    continue
                a = a * S / s
                
                c_new = fft_convolve_self(a)
                if c_new.max() < current_max_b - 1e-12:
                    current_max_b = c_new.max()
                else:
                    # Revert
                    a[j] = max(0, a[j] - delta * S / a.sum() if a.sum() > 0 else 0)
                    s = a.sum()
                    if s > 1e-10:
                        a = a * S / s
                    # Better: just revert explicitly
                    a[j] = max(0, a[j])
                    break
            
            # Recompute hot spots periodically
            if i % 20 == 0:
                c = fft_convolve_self(a)
                top_peaks = np.argsort(c)[::-1][:8]
                hot_indices = set()
                for k in top_peaks:
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    if hi >= lo:
                        seg_a = a[lo:hi+1]
                        seg_b = a[k-lo:k-hi+1][::-1]
                        products = seg_a * seg_b
                        top_local = np.argsort(products)[::-1][:4]
                        for ti in top_local:
                            hot_indices.add(lo + ti)
                hot_list = list(hot_indices)
                if hot_list:
                    rng.shuffle(hot_list)
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best_obj:
        best_obj = c_final.max()
        best_a = a.copy()
    
    return best_a, best_obj


def run_two_phase_refinement(a_init, max_time, rng):
    """Two-phase: coarse (low K) then fine (high K) subgradient."""
    n = len(a_init)
    S = a_init.sum()
    
    # Phase 1: Coarse (K=1,4) - 60% of time
    t1 = max_time * 0.6
    a1, obj1 = run_subgradient_phase(a_init, t1, rng, K_start=1, K_end=4,
                                      step_mode="polyak")
    
    # Phase 2: Fine (K=8,32) - 40% of time
    t2 = max_time * 0.4 - 2
    a2, obj2 = run_subgradient_phase(a1, t2, rng, K_start=8, K_end=32,
                                      step_mode="polyak")
    
    return a2, obj2


def try_different_length(a_base, n_target, max_time, rng):
    """Try optimizing at a different length by down/up-sampling."""
    n_base = len(a_base)
    
    if n_target == n_base:
        a = a_base.copy()
    elif n_target > n_base:
        # Upsample
        t_old = np.linspace(0, 1, n_base)
        t_new = np.linspace(0, 1, n_target)
        a = np.interp(t_new, t_old, a_base)
    else:
        # Downsample by averaging
        ratio = n_base // n_target
        a = a_base[:n_target * ratio].reshape(n_target, ratio).mean(axis=1)
    
    # Normalize
    S = np.sqrt(2.0 * n_target)
    s = a.sum()
    if s > 1e-10:
        a = a * S / s
    
    # Optimize
    a_opt, obj = run_subgradient_phase(a, max_time, rng, K_start=2, K_end=16,
                                        step_mode="polyak")
    
    # Compute actual objective
    actual_obj = compute_objective_value(a_opt)
    return a_opt, actual_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses adaptive subgradient with Polyak steps, local polish, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    print("Time: %.1f" % time.monotonic())
    
    # Load the given sequence
    a_given = None
    try:
        a_given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_given = np.clip(a_given, 0, 1000)
        if a_given.size >= 64 and a_given.sum() > 0:
            print("Using given sequence of length %d" % len(a_given))
        else:
            a_given = None
    except Exception:
        pass
    
    if a_given is None:
        # Fallback: random initialization
        a_given = np.ones(8192) * 0.01 + rng.random(8192)
        print("No given sequence, using random init of length 8192")
    
    n = len(a_given)
    
    # Normalize given sequence
    S_target = np.sqrt(2.0 * n)
    s = a_given.sum()
    a_start = a_given * S_target / s
    
    best_obj = compute_objective_value(a_start)
    best_a = a_start.copy()
    print("Initial objective: %.6f" % best_obj)
    
    # Phase 1: Main subgradient optimization (55% of time)
    t_phase1 = (deadline - time.monotonic()) * 0.55
    print("\n=== Phase 1: Adaptive subgradient (%.0fs) ===" % t_phase1)
    a1, obj1 = run_subgradient_phase(a_start, t_phase1, rng,
                                      K_start=1, K_end=32, step_mode="polyak")
    obj1_actual = compute_objective_value(a1)
    print("Phase 1 result: %.6f" % obj1_actual)
    if obj1_actual < best_obj:
        best_obj = obj1_actual
        best_a = a1.copy()
    
    # Phase 2: Local polish (15% of time)
    remaining = deadline - time.monotonic()
    t_phase2 = remaining * 0.25
    print("\n=== Phase 2: Local polish (%.0fs) ===" % t_phase2)
    a2, obj2 = run_local_polish(a1, t_phase2, rng)
    obj2_actual = compute_objective_value(a2)
    print("Phase 2 result: %.6f" % obj2_actual)
    if obj2_actual < best_obj:
        best_obj = obj2_actual
        best_a = a2.copy()
    
    # Phase 3: Multi-start perturbation (25% of time)
    remaining = deadline - time.monotonic()
    n_restarts = 3
    t_per_restart = remaining * 0.35 / n_restarts
    print("\n=== Phase 3: Multi-start (%.0fs total) ===" % (remaining * 0.35))
    
    for ri in range(n_restarts):
        if time.monotonic() > deadline - 10:
            break
        # Perturb the best solution
        noise_scale = 0.003 * best_a.sum() / np.sqrt(n) / (ri + 1)
        a_pert = best_a + rng.normal(0, noise_scale, n)
        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p > 1e-10:
            a_pert = a_pert * S_target / s_p
        
        # Brief optimization
        a_r, obj_r = run_subgradient_phase(a_pert, t_per_restart, rng,
                                            K_start=4, K_end=16, step_mode="polyak")
        obj_r_actual = compute_objective_value(a_r)
        print("  Restart %d: obj=%.6f" % (ri, obj_r_actual))
        if obj_r_actual < best_obj:
            best_obj = obj_r_actual
            best_a = a_r.copy()
            print("  New best! %.6f" % best_obj)
    
    # Phase 4: Final intensive polish (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 4: Final polish (%.0fs) ===" % remaining)
        t_final = remaining * 0.7
        a_f, obj_f = run_subgradient_phase(best_a, t_final, rng,
                                            K_start=8, K_end=64, step_mode="polyak")
        obj_f_actual = compute_objective_value(a_f)
        print("Final polish result: %.6f" % obj_f_actual)
        if obj_f_actual < best_obj:
            best_obj = obj_f_actual
            best_a = a_f.copy()
        
        # Quick local polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_lp, obj_lp = run_local_polish(best_a, remaining2 * 0.8, rng)
            obj_lp_actual = compute_objective_value(a_lp)
            print("Local polish: %.6f" % obj_lp_actual)
            if obj_lp_actual < best_obj:
                best_obj = obj_lp_actual
                best_a = a_lp.copy()
    
    # Final normalization and verification
    best_a = np.clip(best_a, 0.0, 1000.0)
    s_final = best_a.sum()
    if s_final < 0.01:
        best_a = np.ones(256) * (np.sqrt(512.0) / 256)
    
    final_list = [float(x) for x in best_a]
    
    # Verify
    ver = evaluate_sequence(final_list)
    print("\nVerification via evaluate_sequence: %.6f" % ver)
    
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        best_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in best_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver)
    
    print("Returning sequence of length %d, obj=%.6f" % (len(best_a), best_obj))
    return final_list
