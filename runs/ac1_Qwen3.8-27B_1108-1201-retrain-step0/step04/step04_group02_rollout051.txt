LP-based optimization for minimizing 2*n*max(a*a)/(sum a)^2 over non-negative sequences.
Core idea: iterate the LP "best response" map: given a, solve max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
The fixed point of this map is optimal. Combine with gradient refinement for larger n where LP is too expensive.
"""
import numpy as np
import time
import scipy.sparse
import scipy.optimize
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_conv_cross(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_obj(a):
    """Compute 2*n*max(a*a)/(sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv_self(a)
    return 2.0 * n * c.max() / (s * s)

    """Clamp to [0, inf) and rescale to sum S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:

    return a * (S / s)

    col_indices = []

            col_indices.extend(js.tolist())
            data.extend(a[k - js].tolist())
    T = scipy.sparse.csr_matrix((data, (row_indices, col_indices)), shape=(size, n))
    return T


def solve_lp_direction(a, M):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized) or None if solver fails.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b = np.full(size, M)
    c_obj = -np.ones(n)
    
    try:
        result = scipy.optimize.linprog(
            c=c_obj,
            A_ub=T,
            b_ub=b,
            method='highs',
            options={'presolve': True, 'time_limit': 60}
        )
    except Exception:
        return None
    
    if not result.success:
        return None
    
    g = result.x
    if g is None or not np.all(np.isfinite(g)):
        return None
    
    g = np.maximum(g, 0.0)
    if g.sum() < 1e-15:
        return None
    return g


def lp_optimize(a_init, n, S, max_time, rng, verbose=True):
    """
    Main LP iteration: repeatedly solve the LP to find optimal direction.
    """
    a = normalize_to_sum(a_init, n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if verbose:
        print("  LP start: obj=%.6f" % best_obj)
    
    start = time.monotonic()
    it = 0
    t_mix = 0.5
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        M = c.max()
        
        # Solve LP
        g = solve_lp_direction(a, M)
        if g is None:
            if verbose:
                print("  LP failed at iter %d, stopping." % it)
            break
        
        # Rescale g to sum S
        g = g * (S / g.sum())
        
        # Check convergence: is g close to a?
        diff = np.linalg.norm(g - a) / (S + 1e-30)
        if diff < 1e-5:
            if verbose:
                print("  LP converged at iter %d (diff=%.2e)" % (it, diff))
            break
        
        # Mix step
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stagnant = 0
            # Increase mixing when improving
            t_mix = min(t_mix * 1.05, 0.9)
        else:
            stagnant += 1
            t_mix = max(t_mix * 0.85, 0.2)
        
        a = a_new
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print("  it=%d, obj=%.6f, best=%.6f, diff=%.2e, t=%.2f, elapsed=%.0fs" % (
                it, obj_new, best_obj, diff, t_mix, elapsed))
        
        if stagnant > 30:
            # Perturb to escape
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = normalize_to_sum(a + noise, S)
            stagnant = 0
            t_mix = 0.5
    
    if verbose:
        print("  LP done: best=%.6f after %d iters, %.0fs" % (
            best_obj, it, time.monotonic() - start))
    
    return best_a, best_obj


def gradient_refine(a_init, n, S, max_time, rng, verbose=True):
    """
    Gradient-based refinement using top-k subgradient and LSE smoothing.
    Works for any n (no LP needed).
    """
    a = normalize_to_sum(a_init, n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    c = fft_conv_self(a)
    
    start = time.monotonic()
    it = 0
    step = 0.5
    consecutive_fail = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0
        
        # Choose gradient based on phase
        phase = it % 500
        if phase < 10:
            # Top-1 exact subgradient
            idx = np.argmax(c)
            lo = max(0, idx - n + 1)
            hi = min(idx, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad = np.zeros(n)
            grad[i_vals] = 2.0 * a[idx - i_vals]
        elif phase < 100:
            # Top-16 subgradient
            k = 16
            top_idx = np.argsort(c)[::-1][:k]
            grad = np.zeros(n)
            for j in top_idx:
                lo = max(0, j - n + 1)
                hi = min(j, n - 1)
                i_vals = np.arange(lo, hi + 1)
                grad[i_vals] += 2.0 * a[j - i_vals]
            grad /= k
        elif phase < 300:
            # LSE-smoothed gradient
            beta = min(200.0, 5.0 + 0.05 * it)
            cm = c.max()
            e = np.exp(beta * (c - cm))
            tot = e.sum()
            if tot < 1e-300:
                grad = np.zeros(n)
            else:
                w = e / tot
                fsize = 1 << (len(w) + n - 1).bit_length()
                W = fft(w, fsize)
                A = fft(a, fsize)
                grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        else:
            # Mixed: top-8
            k = 8
            top_idx = np.argsort(c)[::-1][:k]
            grad = np.zeros(n)
            for j in top_idx:
                lo = max(0, j - n + 1)
                hi = min(j, n - 1)
                i_vals = np.arange(lo, hi + 1)
                grad[i_vals] += 2.0 * a[j - i_vals]
            grad /= k
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step
        max_step = 2.0 * S / (n * gnorm + 1e-15)
        step = min(step, max_step)
        
        # Projected gradient step
        a_try = a - step * grad
        # Project onto {x >= 0, sum(x) = S}
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try * (S / s_try)
        
        c_try = fft_conv_self(a_try)
        obj_try = 2.0 * n * c_try.max() / (S * S)
        
        if obj_try < cur_obj - 1e-14:
            a = a_try
            c = c_try
            step = min(step * 1.03, max_step)
        else:
            consecutive_fail += 1
            step *= 0.75
            if consecutive_fail > 30:
                noise_scale = 0.003 * S / np.sqrt(n)
                a = normalize_to_sum(a + rng.normal(0, noise_scale, n), S)
                c = fft_conv_self(a)
                step = 0.5
                consecutive_fail = 0
    
    if verbose:
        print("  Grad refine: best=%.6f after %d iters, %.0fs" % (
            best_obj, it, time.monotonic() - start))
    
    return best_a, best_obj


def dual_lp_direction(a, M, max_iter=200):
    """
    Approximate the LP direction using dual multiplicative weights.
    Dual: min M * sum(lambda) s.t. (T^T lambda)[j] >= 1, lambda >= 0.
    (T^T lambda)[j] = sum_k a[k-j] * lambda[k] = cross-correlation.
    At optimum, g_j > 0 iff (T^T lambda)[j] = 1.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Initialize lambda
    lam = np.ones(size) / size
    
    # Compute T^T lambda via FFT cross-correlation
    # (T^T lambda)[j] = sum_k a[k-j] * lambda[k] for valid k
    # = sum_{m=0}^{n-1} a[m] * lambda[j+m] (substituting k = j+m)
    # This is the "cross-correlation" of a and lambda
    
    def compute_TT_lambda(lam):
        """Compute T^T @ lambda efficiently via FFT."""
        fsize = 1 << (n + size - 1).bit_length()
        A = fft(a, fsize)
        Lam = fft(lam, fsize)
        # We want C[j] = sum_m a[m] * lam[j+m] for j=0,...,n-1
        # This is the cross-correlation. Let's use:
        # C = ifft(conj(fft(a)) * fft(lam)) shifted appropriately
        # Actually: conv(a_rev, lam)[n-1+j] = sum_i a_rev[i] * lam[n-1+j-i]
        #         = sum_i a[n-1-i] * lam[n-1+j-i]
        # Let k = n-1-i: = sum_k a[k] * lam[j+k] = C[j]
        a_rev = a[::-1]
        A_rev = fft(a_rev, fsize)
        conv = np.real(ifft(A_rev * Lam))[:n + size - 1]
        C = conv[n - 1: 2 * n - 1]
        return C
    
    # Multiplicative weights on dual
    eta = 0.5
    for it in range(max_iter):
        C = compute_TT_lambda(lam)
        
        # Check feasibility
        min_C = C.min()
        if min_C >= 1.0 - 1e-6:
            break
        
        # Update: increase lambda where it helps violated constraints
        # Subgradient of (1 - C[j]) for violated j:
        # d(1-C[j])/d(lam[k]) = -a[k-j] (for valid k)
        # We want to increase lam where the gradient of the violation is most negative
        violated = np.maximum(1.0 - C, 0.0)
        
        if violated.max() < 1e-8:
            break
        
        # Compute subgradient of total violation w.r.t. lambda
        # d(sum_j violated[j])/d(lam[k]) = -sum_{j: violated} a[k-j] (for valid)
        # This is: -cross(violated, a) at each k
        fsize2 = 1 << (size + n - 1).bit_length()
        V = fft(violated, fsize2)
        A2 = fft(a, fsize2)
        # cross[k] = sum_j violated[j] * a[k-j]... 
        # Actually we need: sum_j violated[j] * a[k-j] for k-j in [0,n-1]
        # = conv(violated, a)[k]... let me think
        # conv(violated, a)[k] = sum_j violated[j] * a[k-j]
        # We need sum_j violated[j] * a[k-j] where 0 <= k-j < n
        # Since violated has length n and a has length n:
        # conv has length 2n-1, and conv[k] = sum_j violated[j] * a[k-j] for j=0..n-1, k-j=0..n-1
        conv_va = np.real(ifft(V * A2))[:2 * n - 1]
        
        # lambda[k] should be increased where conv_va[k] is large (more violated constraints involve it)
        lam *= np.exp(eta * conv_va / (conv_va.max() + 1e-30))
        lam /= lam.sum()
        
        eta *= 0.98
    
    # The support of g is where C[j] is close to 1
    C = compute_TT_lambda(lam)
    # Soft assignment: g_j proportional to "tightness"
    margin = np.maximum(C - 1.0, 0.0)
    g = np.exp(-10.0 * margin)
    g_sum = g.sum()
    if g_sum < 1e-15:
        g = np.ones(n)
    else:
        g = g / g_sum
    return g


def lp_with_dual_fallback(a_init, n, S, max_time, rng, verbose=True):
    """LP optimization with dual-based fallback if full LP is too slow."""
    a = normalize_to_sum(a_init, n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start = time.monotonic()
    it = 0
    t_mix = 0.5
    use_full_lp = n <= 4096
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        M = c.max()
        
        # Get direction
        if use_full_lp:
            g = solve_lp_direction(a, M)
            if g is None:
                use_full_lp = False  # Fall back to dual
            else:
                g = g * (S / g.sum())
        else:
            g = dual_lp_direction(a, M, max_iter=100)
            g = normalize_to_sum(g, S)
        
        diff = np.linalg.norm(g - a) / (S + 1e-30)
        if diff < 1e-5:
            if verbose:
                print("  Converged at iter %d" % it)
            break
        
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = normalize_to_sum(a_new, S)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stagnant = 0
            t_mix = min(t_mix * 1.05, 0.85)
        else:
            stagnant += 1
            t_mix = max(t_mix * 0.9, 0.3)
        
        a = a_new
        
        if verbose and it % 3 == 0:
            print("  it=%d, obj=%.6f, best=%.6f, t=%.2f, %.0fs" % (
                it, obj_new, best_obj, t_mix, time.monotonic() - start))
        
        if stagnant > 25:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a = normalize_to_sum(a + noise, S)
            stagnant = 0
            t_mix = 0.5
    
    if verbose:
        print("  Done: best=%.6f, %d iters, %.0fs" % (best_obj, it, time.monotonic() - start))
    
    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    # Sine-squared
    inits.append(np.sin(np.pi * tt) ** 2)
    # Gaussian
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    # Uniform
    inits.append(np.ones(n))
    # U-shaped
    inits.append(1.0 + 2.0 * (tt - 0.5) ** 2)
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    # Random
    inits.append(rng.random(n) + 0.1)
    # Power law
    inits.append(np.maximum(1.0 / (tt + 0.1), 0.1))
    # Bimodal
    inits.append(np.exp(-20*(tt-0.2)**2) + np.exp(-20*(tt-0.8)**2))
    
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(a*a)/(sum a)^2.
    Uses LP best-response iteration for small/medium n, gradient refinement for large n.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)
    
    global_best = np.inf
    global_a = None
    global_n = 0
    
    print("Starting search, budget=%.0fs" % budget_s)
    t0 = time.monotonic()
    
    # Try to get previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass
    
    # Candidate lengths to try (prioritize where LP is feasible)
    candidate_ns = []
    if given is not None:
        candidate_ns.append(len(given))
    
    for n_try in [1024, 2048, 4096, 8192, 16384]:
        if n_try not in candidate_ns:
            candidate_ns.append(n_try)
    
    # Sort by priority: prefer LP-feasible sizes first, then the given size, then larger
    def priority(n):
        if n in (1024, 2048, 4096):
            return 0
        if n == (len(given) if given is not None else -1):
            return 1
        if n <= 8192:
            return 2
        return 3
    
    candidate_ns.sort(key=priority)
    
    # Allocate time: give more time to LP-feasible sizes
    time_per_n = {}
    remaining_n = list(candidate_ns)
    
    # First pass: allocate time
    n_lp = [n for n in candidate_ns if n <= 4096]
    n_grad = [n for n in candidate_ns if n > 4096]
    
    # Give 60% to LP sizes, 40% to gradient sizes
    total_time = min(budget_s - 30, deadline - time.monotonic())
    lp_time = total_time * 0.6 / max(1, len(n_lp))
    grad_time = total_time * 0.4 / max(1, len(n_grad))
    
    for n in n_lp:
        time_per_n[n] = max(60.0, lp_time)
    for n in n_grad:
        time_per_n[n] = max(30.0, grad_time)
    
    # Process each candidate length
    for n in candidate_ns:
        if time.monotonic() > deadline - 30:
            break
        
        n = int(n)
        if n < 64:
            continue
        
        S = np.sqrt(2.0 * n)
        t_alloc = time_per_n.get(n, 60.0)
        
        # Check if we still have time
        if time.monotonic() + 30 > deadline:
            break
        
        print("\n=== n=%d, allocating %.0fs ===" % (n, t_alloc))
        
        # Generate initializations
        inits = []
        if given is not None and len(given) == n:
            inits.append(("given", given.copy()))
        elif given is not None:
            # Resize given to n via interpolation
            from numpy import interp
            old_n = len(given)
            new_idx = np.linspace(0, old_n - 1, n)
            resized = np.interp(new_idx, np.arange(old_n), given)
            inits.append(("resized_given", resized))
        
        gen_inits = generate_initializations(n, rng)
        for i, g in enumerate(gen_inits):
            inits.append(("init_%d" % i, g))
        
        # For each initialization, run optimization
        n_inits = len(inits)
        per_init_time = t_alloc / max(1, n_inits)
        
        # If we have a given sequence of this size, prioritize it
        reordered = []
        for name, arr in inits:
            if "given" in name or "resized" in name:
                reordered.insert(0, (name, arr))
            else:
                reordered.append((name, arr))
        inits = reordered
        
        best_for_n = None
        best_obj_for_n = np.inf
        
        for name, arr in inits:
            if time.monotonic() > deadline - 30:
                break
            
            arr_n = len(arr)
            if arr_n != n:
                continue
            
            S_i = np.sqrt(2.0 * arr_n)
            t_init = min(per_init_time, deadline - time.monotonic() - 30)
            if t_init < 10:
                break
            
            # Choose method based on n
            if n <= 4096:
                # LP-based
                a_opt, obj = lp_with_dual_fallback(
                    arr, arr_n, S_i, t_init, rng, verbose=(t_init > 30))
            else:
                # Gradient-based
                if time.monotonic() + min(t_init, 120) > deadline - 30:
                    t_init = max(10, deadline - time.monotonic() - 30)
                    if t_init < 10:
                        break
                a_opt, obj = gradient_refine(
                    arr, arr_n, S_i, t_init, rng, verbose=(t_init > 30))
            
            # Verify
            a_opt = normalize_to_sum(a_opt, arr_n)
            real_obj = compute_obj(a_opt)
            
            if real_obj < best_obj_for_n:
                best_obj_for_n = real_obj
                best_for_n = a_opt.copy()
                if "given" in name:
                    print("  [%s] obj=%.6f" % (name, real_obj))
        
        if best_for_n is not None and best_obj_for_n < global_best:
            global_best = best_obj_for_n
            global_a = best_for_n.copy()
            global_n = n
            print("New global best: obj=%.6f at n=%d (elapsed %.0fs)" % (
                global_best, n, time.monotonic() - t0))
    
    # === Refinement phase: focus on the best found ===
    remaining = deadline - time.monotonic()
    print("\n=== Refinement phase (%.0fs remaining) ===" % remaining)
    
    if global_a is not None and remaining > 30:
        n_best = global_n
        S_best = np.sqrt(2.0 * n_best)
        
        # Try further LP refinement
        if n_best <= 4096 and remaining > 60:
            a_ref, obj_ref = lp_with_dual_fallback(
                global_a, n_best, S_best, remaining * 0.5, rng, verbose=True)
            a_ref = normalize_to_sum(a_ref, n_best)
            obj_ref = compute_obj(a_ref)
            if obj_ref < global_best:
                global_best = obj_ref
                global_a = a_ref.copy()
                print("  LP refinement: %.6f" % global_best)
        
        remaining2 = deadline - time.monotonic()
        
        # Gradient polish
        if remaining2 > 30:
            a_grad, obj_grad = gradient_refine(
                global_a, n_best, S_best, min(remaining2 * 0.5, 300), rng, verbose=True)
            a_grad = normalize_to_sum(a_grad, n_best)
            obj_grad = compute_obj(a_grad)
            if obj_grad < global_best:
                global_best = obj_grad
                global_a = a_grad.copy()
                print("  Grad polish: %.6f" % global_best)
        
        # Micro-optimization: very small steps on the peak
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_micro = global_a.copy()
            S_m = S_best
            n_m = n_best
            step = 0.01
            start_m = time.monotonic()
            it_m = 0
            best_m = global_best
            
            while time.monotonic() - start_m < remaining3 * 0.5:
                it_m += 1
                c_m = fft_conv_self(a_micro)
                cur = 2.0 * n_m * c_m.max() / (S_m * S_m)
                
                if cur < best_m - 1e-13:
                    best_m = cur
                
                # Top-1 subgradient
                idx = np.argmax(c_m)
                lo = max(0, idx - n_m + 1)
                hi = min(idx, n_m - 1)
                i_vals = np.arange(lo, hi + 1)
                grad = np.zeros(n_m)
                grad[i_vals] = 2.0 * a_micro[idx - i_vals]
                
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    break
                
                eff_step = step / (1.0 + 0.005 * it_m)
                a_try = a_micro - eff_step * grad
                a_try = normalize_to_sum(a_try, S_m)
                
                c_try = fft_conv_self(a_try)
                obj_try = 2.0 * n_m * c_try.max() / (S_m * S_m)
                
                if obj_try < cur - 1e-14:
                    a_micro = a_try
                    step = min(step * 1.02, 0.3)
                else:
                    step *= 0.85
                    if step < 0.0005:
                        noise = rng.normal(0, 0.0005 * S_m / np.sqrt(n_m), n_m)
                        a_micro = normalize_to_sum(a_micro + noise, S_m)
                        step = 0.01
            
            a_micro = normalize_to_sum(a_micro, n_m)
            final_m = compute_obj(a_micro)
            if final_m < global_best:
                global_best = final_m
                global_a = a_micro.copy()
                print("  Micro: %.6f" % global_best)
    
    # === Final: try a few more different lengths quickly ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Quick try with a different length
        alt_n = 2048 if global_n != 2048 else 4096
        S_alt = np.sqrt(2.0 * alt_n)
        
        # Use the best sequence as a template (resize)
        if global_a is not None and len(global_a) != alt_n:
            from numpy import interp
            old_n = len(global_a)
            new_idx = np.linspace(0, old_n - 1, alt_n)
            alt_init = np.interp(new_idx, np.arange(old_n), global_a)
        else:
            alt_init = np.ones(alt_n)
        
        a_alt, obj_alt = lp_with_dual_fallback(
            alt_init, alt_n, S_alt, min(remaining - 20, 200), rng, verbose=True)
        a_alt = normalize_to_sum(a_alt, alt_n)
        obj_alt = compute_obj(a_alt)
        
        if obj_alt < global_best:
            global_best = obj_alt
            global_a = a_alt.copy()
            global_n = alt_n
            print("  Alt n=%d: %.6f" % (alt_n, global_best))
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("\nVerification: obj=%.6f" % ver)
                if ver < global_best:
                    global_best = ver
            else:
                print("evaluate_sequence returned non-finite, using fallback")
                global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
                global_best = 2.0
        except Exception as e:
            print("Verification error: %s" % str(e))
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0
    
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    print("\nFinal: n=%d, obj=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
