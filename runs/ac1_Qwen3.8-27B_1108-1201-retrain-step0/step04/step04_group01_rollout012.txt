```python
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
```

        # Additional: pure random perturbation every few perturbations
        if it % (perturb_period * 3) == 0 and progress < 0.8:
            amp = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, amp, n), S)
            t_k = 1.0
            stall = 0
    
    # Final evaluation
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_max:
        best_max = c_final.max()
    
    obj_final = 2.0 * n * best_max / (S * S)
    if verbose:
        print(f"  {label}: {it} iters, obj={obj_final:.8f}, time={time.monotonic()-start:.1f}s")
    
    return best_a, obj_final
