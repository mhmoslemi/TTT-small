def project_simplex(v, S):

        A_new = A_new * np.exp(1j * phase_noise)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def local_coordinate_search(a_init, S, max_time, rng, n_pairs=20):
    """
    Local coordinate search: find pairs (i, j) that contribute most to peak lags
    and adjust them.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    
    for rnd in range(10000):
        if time.monotonic() - start > max_time:
            break
        
        c = convolve_self(a)
        cmax = c.max()
        peak_lag = int(np.argmax(c))
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            if rnd % 50 == 0:
                print(f"    LCS round {rnd}: cmax={cmax:.8f}")
        
        # Find top-contributing pairs at peak lag
        pairs = []
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        
        pairs.sort(key=lambda x: -x[0])
        
        improved = False
        for val, i, j in pairs[:n_pairs]:
            if time.monotonic() - start > max_time:
                break
            if i == j:
                continue
            
            # Try reducing the larger of a[i], a[j]
            if a[i] >= a[j]:
                orig = a[i]
                for frac in [0.7, 0.5, 0.8]:
                    a[i] = orig * frac
                    c_check = convolve_self(a)
                    if c_check.max() < cmax:
                        cmax = c_check.max()
                        improved = True
                        break
                    a[i] = orig * (1.2 if frac < 1 else 0.8)  # Try increasing too
                    c_check = convolve_self(a)
                    if c_check.max() < cmax:
                        cmax = c_check.max()
                        improved = True
                        break
                    a[i] = orig
            else:
                orig = a[j]
                for frac in [0.7, 0.5, 0.8]:
                    a[j] = orig * frac
                    c_check = convolve_self(a)
                    if c_check.max() < cmax:
                        cmax = c_check.max()
                        improved = True
                        break
                    a[j] = orig * (1.2 if frac < 1 else 0.8)
                    c_check = convolve_self(a)
                    if c_check.max() < cmax:
                        cmax = c_check.max()
                        improved = True
                        break
                    a[j] = orig
        
        if not improved and rnd % 10 == 0:
            # Perturb
            idx = rng.choice(n, size=min(5, n), replace=False)
            for i in idx:
                a[i] *= rng.uniform(0.5, 1.5)
        
        a = project_simplex(a, S)
    
    return best_a, best_cmax


def nesterov_subgradient(a_init, S, max_time, rng, top_k=30):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = convolve_self(y)
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = y.copy()
        
        grad = compute_topk_subgradient(y, c, top_k=top_k, exp_temperature=0.05)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = y - step * grad / max(gnorm, 1e-15)
        x_new = project_simplex(x_new, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 1000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_a, best_cmax


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_topk_subgradient_fast(a, c, top_k=20, exp_temperature=0.1):
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Exponential weighting by rank
    weights = np.array([np.exp(-i * exp_temperature) for i in range(top_k)])
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        k = int(k)
        wk = 2.0 * weights[i]
        if k < n:
            grad[:k+1] += wk * a[:k+1][::-1]
        else:
            lo = k - n + 1
            grad[lo:] += wk * a[lo:][::-1]
    
    return grad
