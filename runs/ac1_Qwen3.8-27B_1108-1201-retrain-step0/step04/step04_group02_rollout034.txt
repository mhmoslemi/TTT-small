def lp_direction(f, M, n, tol=1e-6, max_iter=100):
    """
    Find direction g that approximately solves:
    max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    
    Using dual greedy ascent.
    Returns g normalized to sum = 1.
    """
    size = 2 * n - 1
    y = np.zeros(size)
    
    for it in range(max_iter):
        # Compute (f ⊛ y) at each j
        # (f ⊛ y)_j = sum_k f[k] * y[k-j] (with 0 <= k < n, 0 <= k-j < size)

        Y = fft(y, fsize)
        cross = np.real(ifft(F * np.conj(Y)))[:n]
        # cross[j] = sum_k f[k] * y[k-j] = (f ⊛ y)_j

        if cross.min() >= 1.0 - tol:

        direction = np.zeros(size)

def optimize_lp_direction(f, n, S, max_time, rng):
    """
    Main optimization loop using LP direction finding via multiplicative weights.
    """
    start = time.monotonic()
    best_obj = compute_objective(f)
    best_f = f.copy()
    t = 0.3  # mixing parameter
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        M = c.max()
        cur_obj = 2.0 * n * M / S**2
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Find LP direction
        g = lp_direction_mw(f, M, n, eta=0.5, n_iter=5)
        
        # Normalize g to have sum S
        g = g * (S / g.sum())
        
        # Take a step
        t = max(0.05, 0.5 / (1 + 0.01 * it))
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new * (S / f_new.sum())  # re-normalize
        
        c_new = fft_convolve_self(f_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new < cur_obj:
            f = f_new
        else:
            # Try smaller step
            t *= 0.5
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new * (S / f_new.sum())
            c_new = fft_convolve_self(f_new)
            obj_new = 2.0 * n * c_new.max() / S**2
            if obj_new < cur_obj:
                f = f_new
            else:
                # Give up on this direction, perturb
                f = f + rng.normal(0, 0.01 * S / np.sqrt(n), n)
                f = np.maximum(f, 0)
                f = f * (S / f.sum())
    
    return best_f, best_obj

"""
Minimize 2*n*max_conv/sum^2 using LP-based direction finding via multiplicative weights.
Key approach: At each iteration, solve (approximately) the LP
  max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
using exponentiated gradient / multiplicative weights, then take a convex combination
step. Combined with adaptive step size, periodic perturbations, and multi-length search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def fft_conv_cross(a, b):
    """Full linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    return np.real(ifft(fft(a, fsize) * fft(b, fsize)))[:size]


def cross_corr(c, f, n_f):
    """Compute h[j] = sum_k c[k] * f[k-j] for j=0..n_f-1 via FFT."""
    total = len(f) + len(c) - 1
    fsize = 1 << (total - 1).bit_length()
    return np.real(ifft(fft(c, fsize) * np.conj(fft(f, fsize))))[:n_f]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_conv_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def normalize_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lp_direction_mw(f, M, n, eta=1.0, n_iter=15):
    """
    Find g approximately solving: max sum(g) s.t. (f*g)[k] <= M, g >= 0
    using multiplicative weights (exponentiated gradient).
    Returns g normalized to sum 1.
    """
    g = np.ones(n) / n
    log_g = np.zeros(n) - np.log(n)

    for t in range(n_iter):
        # Compute c = f*g (cross-convolution)
        c = fft_conv_cross(f, g)
        # Compute h[j] = sum_k c[k] * f[k-j]
        h = cross_corr(c, f, n)
        # Multiplicative weights update
        log_g -= eta * h / M
        # Normalize for numerical stability
        log_g -= log_g.max()
        g = np.exp(log_g)
        gs = g.sum()
        if gs < 1e-300:
            g = np.ones(n) / n
            log_g = np.zeros(n) - np.log(n)
        else:
            g /= gs
            log_g = np.log(np.maximum(g, 1e-300))

    return g


def lse_subgradient(a, c, beta, n):
    """LSE-smoothed gradient of max, using FFT for efficiency."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # gradient[j] = 2 * sum_k w[k] * a[k-j] = 2 * cross_corr(w, a)[j]
    return 2.0 * cross_corr(w, a, n)


def topk_subgradient(a, c, k, n):
    """Subgradient using top-k peaks of convolution."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k

                inits.append(("given", given.copy()))
            else:
                # Resample given to this length
                idx = (np.arange(n) * len(given) / n).astype(int)
                inits.append(("given_resampled", given[idx]))

        # Generate structured inits
        gen_inits = generate_initializations(n, rng)
        for i, g in enumerate(gen_inits[:5]):
            inits.append(("init_%d" % i, g))

        # Allocate time among initializations
        # Give most time to the first (given) initialization
        n_inits = len(inits)
        if n_inits == 0:
            continue

        # First init gets 60% of the time, rest split equally
        first_time = alloc * 0.60
        rest_time = alloc * 0.40 / max(1, n_inits - 1)

        for ii, (name, arr) in enumerate(inits):
            if time.monotonic() > deadline - 10:
                break

            if ii == 0:
                t_alloc = first_time
            else:
                t_alloc = rest_time

            if t_alloc < 5:
                continue

            print("  [%s] n=%d, alloc=%.1fs" % (name, n, t_alloc))

            # Run LP optimization
            a_opt, score = run_lp_optimization(arr, n, S, t_alloc, rng)
            a_opt = normalize_sum(a_opt, n)
            real_score = compute_objective(a_opt)

            if real_score < global_best:
                global_best = real_score
                global_a = a_opt.copy()
                print("  [%s] n=%d: obj=%.6f (elapsed %.0fs)" % (
                    name, n, global_best, time.monotonic() - t0))

            # If we have time, also try FISTA from this result
            remaining = deadline - time.monotonic()
            if remaining > 30 and real_score < global_best + 0.01:
                fista_time = min(remaining * 0.3, 120.0)
                a_f, s_f = refine_with_fista(a_opt, n, S, fista_time, rng)
                a_f = normalize_sum(a_f, n)
                s_f = compute_objective(a_f)
                if s_f < global_best:
                    global_best = s_f
                    global_a = a_f.copy()
                    print("  [%s after FISTA] n=%d: obj=%.6f" % (name, n, global_best))

    # Final polish on the best sequence
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Final polish (%.0fs remaining) ===" % remaining)

        polish_time = min(remaining * 0.7, 180.0)
        a_p, s_p = run_lp_optimization(global_a, n_best, S_best, polish_time, rng, polish=True)
        a_p = normalize_sum(a_p, n_best)
        s_p = compute_objective(a_p)
        if s_p < global_best:
            global_best = s_p
            global_a = a_p.copy()
            print("  After polish: %.6f" % global_best)

    # Very final micro-optimization
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a = global_a.copy()
        c = fft_conv_self(a)
        micro_end = time.monotonic() + remaining * 0.5

        while time.monotonic() < micro_end:
            M = c.max()
            cur = 2.0 * n_best * M / S_best ** 2
            g = lp_direction_mw(a, M, n_best, eta=1.5, n_iter=5)
            gs = g.sum()
            if gs < 1e-15:
                break
            g = g * (S_best / gs)
            a_try = 0.999 * a + 0.001 * g
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S_best / max(a_try.sum(), 1e-15))
            c_try = fft_conv_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best ** 2
            if obj_try < cur:
                a = a_try
                c = c_try
                if obj_try < global_best:
                    global_best = obj_try
                    global_a = a.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using uniform fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
