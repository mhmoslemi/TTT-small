import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

    """

    phase = np.angle(A)
    
    c = np.real(ifft(P))[:2*n-1]

    # Threshold: keep only the top fraction of power bins at full, 
    # reduce others proportionally

    # A better approach: scale down bins above the median
    p_median = np.median(P[P > 0])
    if p_median < 1e-30:

    """

        S = a.sum()
    else:
        a = a * (S / a.sum())

        it += 1

            best_score = sc
            best_a = a.copy()
        
        # Find lags contributing to peak

        if s < 1e-15:

        else:
            a = a * (S / s)

def fienup_optimize(a_init, fsize, budget_s, S=None):

    Fienup-style error reduction: alternate between
    (1) constraining the power spectrum (reduce peak autocorrelation)
    (2) projecting onto non-negative simplex
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    if S is None:
        S = a.sum()
    else:
        a = a * (S / max(a.sum(), 1e-15))
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t_start = time.monotonic()
    it = 0
    gamma = 0.5  # interpolation parameter
    
    while time.monotonic() - t_start < budget_s:
        it += 1
        
        # Step 1: Frequency domain - reduce peak
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        c = np.real(ifft(P))[:2*n-1]
        c_max = c.max()
        
        # Find which frequency bins contribute most to the peak
        # The peak of c is at some lag k*, and c[k*] = sum_j |A[j]|^2 * exp(2*pi*j*k*/N) / N
        # So bins where exp(2*pi*j*k*/N) aligns with the peak direction contribute most
        
        # Simpler approach: just cap the top bins
        p_sorted = np.sort(P)[::-1]
        p_cap = p_sorted[min(10, len(P)-1)]  # cap at 10th largest
        
        # Soft cap
        P_capped = np.minimum(P, p_cap)
        
        # This directly reduces the peak of ifft(P)
        A_capped = np.sqrt(P_capped) * np.exp(1j * phase)
        a_freq = np.real(ifft(A_capped))[:n]
        
        # Step 2: Interpolate and project
        a_interp = (1 - gamma) * a + gamma * a_freq
        a = project_simplex(a_interp, S)
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Anneal gamma
        progress = it / max(it + 1000, 2)
        
        if it % 500 == 0:
            print("    Fienup iter %d: score=%.6f" % (it, best_score))
        
        if best_score < 1.50:
            break
    
    return best_a, best_score

def peak_cancellation(a, time_budget, S):
    """Targeted peak cancellation at the current peak lag."""
    n = len(a)
    sz = 1 << ((2*n - 2).bit_length())
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        it += 1
        c = convolve_self(a)
        k_peak = int(np.argmax(c))
        c_max = c[k_peak]
        sc = 2.0*n*c_max/(S*S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the top contributing pairs
        # b[k_peak] = sum_j a[j] a[k_peak - j]
        # Find j where a[j]*a[k_peak-j] is largest
        products = np.zeros(n)
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                products[j] = a[j] * a[k2]
        
        top_j = np.argsort(products)[::-1][:5]
        
        # Reduce the largest contributors
        total_reduce = 0.0
        for j in top_j:
            j = int(j)
            k2 = k_peak - j
            if 0 <= k2 < n:
                # Reduce a[j] and a[k2] slightly
                reduce_amt = 0.01 * a[j]
                a[j] -= reduce_amt
                a[k2] -= reduce_amt
                total_reduce += 2 * reduce_amt
        
        # Redistribute to random positions (or the smallest ones)
        if total_reduce > 0:
            # Add to positions with smallest values
            min_idx = np.argsort(a)[:len(top_j)]
            per_add = total_reduce / len(min_idx)
            for idx in min_idx:
                a[int(idx)] += per_add
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (S / s)
        a = project_simplex(a, S)
    
    return best_a, best_sc
