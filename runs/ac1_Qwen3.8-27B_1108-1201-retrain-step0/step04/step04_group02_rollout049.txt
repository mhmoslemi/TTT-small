# Helper functions at top level

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A*A))
    return c[:size]

def project_simplex(v, S):
    # Project v onto {x >= 0, sum(x) = S}
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

    return 2.0 * n * max_b / (sum_a**2)

            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial_a = a_try
                best_trial_step = s_try

        if best_trial_obj < cur_obj - 1e-14:
            a = best_trial_a
            step = best_trial_step * 1.05
        else:
            stall += 1
            step *= 0.7
            if stall > 50:
                # Perturb to escape
                noise_scale = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 0.3
                stall = 0
            if stall > 200:
                # Larger perturbation
                noise_scale = 0.02 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 0.5
                stall = 0

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.2))
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * 3 * tt) * np.cos(2 * np.pi * 7 * tt))
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.sin(10 * np.pi * tt + 1.0))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses LSE continuation FISTA + peak squeeze, multi-length.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs" % budget_s)
    t0 = time.monotonic()

    # Decide lengths to try
    lengths_to_try = [8192, 16384]
    if given is not None and 128 <= len(given) <= 32768:
        # Also try the given length
        glen = len(given)
        if glen not in lengths_to_try:
            lengths_to_try.insert(0, glen)

    # Allocate time across lengths
    # Give more time to 8192 (proven) and 16384 (more DOF)
    time_fracs = []
    for ln in lengths_to_try:
        if ln == 8192:
            time_fracs.append(0.45)
        elif ln == 16384:
            time_fracs.append(0.40)
        else:
            time_fracs.append(0.15)
    total_frac = sum(time_fracs)
    time_fracs = [f / total_frac for f in time_fracs]

    for li, n in enumerate(lengths_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        n_time = min(remaining - 10, budget_s * time_fracs[li])
        if n_time < 20:
            break

        S = np.sqrt(2.0 * n)
        print("\n=== Length n=%d, time=%.0fs ===" % (n, n_time))

        # Build initialization list
        inits = []
        if given is not None and len(given) == n:
            inits.append(("given", given.copy()))
        elif given is not None:
            # Resample given to length n
            from scipy.fft import fft as _fft
            g = given.copy()
            ng = len(g)
            # Zero-pad or truncate
            if ng < n:
                g = np.zeros(n)
                g[:ng] = given
            else:
                g = given[:n].copy()
            inits.append(("given_resized", g))

        gen_inits = generate_initializations(n, rng)
        for i, g in enumerate(gen_inits[:4]):
            inits.append(("init_%d" % i, g))

        # Perturb the given if available
        if given is not None:
            g = normalize_sum(given if len(given) == n else np.pad(given, (0, max(0, n - len(given)))), n)
            for pi in range(2):
                pert = project_simplex(g + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
                inits.append(("pert_%d" % pi, pert))

        n_inits = len(inits)
        per_init_time = n_time * 0.6 / max(n_inits, 1)

        # Phase A: LSE continuation FISTA on each init
        for i, (name, arr) in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(per_init_time, rem - 10)
            if t_alloc < 5:
                continue

            arr = normalize_sum(arr, n)
            print("  [%s] starting, time_alloc=%.0fs" % (name, t_alloc))
            a_opt, score = run_lse_fista_continuation(arr, n, S, t_alloc, rng)
            a_opt = normalize_sum(a_opt, n)
            real_score = compute_objective(a_opt)

            if real_score < global_best:
                global_best = real_score
                global_a = a_opt.copy()
                print("  [%s] NEW BEST: obj=%.6f" % (name, global_best))

        # Phase B: Peak squeeze on the best found
        rem = deadline - time.monotonic()
        if rem > 15 and global_a is not None and len(global_a) == n:
            squeeze_time = min(rem * 0.4, n_time * 0.3)
            print("  Phase B: peak squeeze, time=%.0fs" % squeeze_time)
            a_sq, s_sq = run_peak_squeeze(global_a, n, S, squeeze_time, rng)
            a_sq = normalize_sum(a_sq, n)
            s_sq = compute_objective(a_sq)
            if s_sq < global_best:
                global_best = s_sq
                global_a = a_sq.copy()
                print("  After squeeze: obj=%.6f" % global_best)

    # Final LSE continuation on the overall best with remaining time
    rem = deadline - time.monotonic()
    if rem > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final LSE continuation, time=%.0fs ===" % rem)
        a_fin, s_fin = run_lse_fista_continuation(global_a, n_best, S_best, rem * 0.5, rng)
        a_fin = normalize_sum(a_fin, n_best)
        s_fin = compute_objective(a_fin)
        if s_fin < global_best:
            global_best = s_fin
            global_a = a_fin.copy()
            print("  Final LSE: obj=%.6f" % global_best)

        # Final peak squeeze
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_sq2, s_sq2 = run_peak_squeeze(global_a, n_best, S_best, rem2 * 0.5, rng)
            a_sq2 = normalize_sum(a_sq2, n_best)
            s_sq2 = compute_objective(a_sq2)
            if s_sq2 < global_best:
                global_best = s_sq2
                global_a = a_sq2.copy()
                print("  Final squeeze: obj=%.6f" % global_best)

    # Final verification with evaluate_sequence
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Verification: %.6f" % ver)
                if ver < global_best:
                    global_best = ver
            else:
                print("evaluate_sequence rejected!")
                global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
                global_best = 2.0
        except Exception as e:
            print("Verification error: %s" % str(e))

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("\nReturning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
