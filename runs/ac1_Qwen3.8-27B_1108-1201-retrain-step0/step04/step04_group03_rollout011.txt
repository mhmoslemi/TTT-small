"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Combined Gerchberg-Saxton spectral shaping + FISTA + direct peak targeting.
Key innovations over previous approach:
1. Phase-preserving GS iterations to escape local minima
2. Direct "blame" distribution for peak reduction
3. Adaptive method switching (FISTA <-> GS based on stagnation)
4. Asymmetric structured initializations
5. Multi-scale warmup (smaller n first, then upscale)
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft, next_fast_len


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = next_fast_len(len(w) + n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def gs_step(f, S, N=None, gamma=0.7, blend=0.8):
    """
    Gerchberg-Saxton style spectral shaping step.
    Compresses the power spectrum dynamic range while preserving phase.
    gamma: compression exponent (0 < gamma < 1, lower = more compression)
    blend: how much of the new f to use (1.0 = full replacement)
    """
    n = len(f)
    if N is None:
        N = next_fast_len(2 * n - 1)
    
    F = fft(f, N)
    phase = np.angle(F)
    P = np.abs(F) ** 2
    
    # Geometric mean compression: P_new = P^gamma * mean(P)^(1-gamma)
    P_mean = P.mean()
    if P_mean < 1e-300:
        return f.copy()
    
    log_P = np.log(np.maximum(P, 1e-300))
    log_P_mean = np.log(P_mean)
    log_P_new = gamma * log_P + (1 - gamma) * log_P_mean
    P_new = np.exp(log_P_new)
    
    # Reconstruct with original phase
    F_new = np.sqrt(P_new) * np.exp(1j * phase)
    f_new = np.real(ifft(F_new))[:n]
    
    # Project to {x >= 0, sum = S}
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    f_new = f_new * (S / s)
    
    # Blend with original to avoid discontinuities
    f_out = blend * f_new + (1 - blend) * f
    f_out = np.maximum(f_out, 0.0)
    f_out = project_simplex(f_out, S)
    return f_out


def gs_step_adaptive(f, S, N=None):
    """
    Adaptive GS step: chooses gamma based on the current peak structure.
    If peaks are sharp (high dynamic range in correlation), use more compression.
    """
    n = len(f)
    c = fft_convolve_self(f)
    c_max = c.max()
    c_mean = c.mean()
    dynamic_range = c_max / max(c_mean, 1e-15)
    
    # More compression when dynamic range is high
    gamma = np.clip(1.0 - 0.3 * np.log10(max(dynamic_range, 1)), 0.4, 0.9)
    blend = 0.6
    
    return gs_step(f, S, N=N, gamma=gamma, blend=blend)


def direct_peak_reduce(f, S, top_k=10, lr=0.1):
    """
    Direct peak reduction: identify top-k peaks in autocorrelation,
    compute 'blame' on each element, and multiplicatively reduce blamed elements.
    """
    n = len(f)
    c = fft_convolve_self(f)
    c_max = c.max()
    
    # Find top-k peaks
    idx = np.argsort(c)[::-1][:top_k]
    
    # Compute blame: for each peak at lag k, element i is blamed for f[i]*f[k-i]
    blame = np.zeros(n)
    for k in idx:
        weight = c[k] / c_max  # relative importance of this peak
        # Contribution of element i to c[k] is f[i]*f[k-i] (if both valid)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        # blame[i] += weight * f[k-i] for i in [lo, hi]
        # c[k] = sum_{j=lo}^{hi} f[j] * f[k-j]
        # d(c[k])/d(f[i]) = f[k-i] if k-i is in [0,n-1], plus f[i-k] if i-k is in [0,n-1]
        # For full convolution: d(c[k])/d(f[i]) = f[k-i] (if 0<=k-i<n) + f[i-k] (if 0<=i-k<n)
        parts = []
        if 0 <= k - hi < n:
            parts.append(f[k - hi:k - lo + 1])
        if 0 <= k - lo < n:
            parts.append(f[k - lo:k - hi + 1][::-1])
        # Simpler: just compute gradient directly
        grad_k = np.zeros(n)
        for j in range(lo, hi + 1):
            grad_k[j] += f[k - j]
            if j != k - j:
                grad_k[k - j] += f[j]
        blame += weight * grad_k
    
    # Multiplicative reduction
    max_blame = blame.max()
    if max_blame < 1e-15:
        return f.copy()
    
    f_new = f * np.exp(-lr * blame / max_blame)
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def solve_lp_best_response(f, n, S, top_k=200, time_limit=15.0):
    """
    Solve the LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max()
    
    idx = np.argsort(c)[::-1][:top_k]
    
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    solvers = ['SCS', 'ECOS', 'PDLP']
    g = None
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                break
        except Exception:
            continue
        if time.monotonic() - start > time_limit:
            break
    
    if g is not None and g.sum() > 1e-10:
        g = g * (S / g.sum())
        return g
    return None

        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)
        
        # Adaptive: if very stuck, add noise
        if stuck > 5000:
            noise_scale = 0.003 * S / np.sqrt(n) * min(1.0, stuck / 10000.0)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_fista = 1.0
            last_improve_it = it
            # Reset beta to re-explain
            beta = max(10.0, beta * 0.5)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t_fista = t_new
    
    return best_f, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses combined FISTA + GS + peak reduction + LP approach.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to use given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    main_n = 8192
    S = np.sqrt(2.0 * main_n)
    
    print("Strategy: GS+FISTA+PeakReduce+LP, main_n=%d, budget=%.0fs" % (main_n, budget_s))
    
    # === Phase 0: Quick warmup at smaller n (if time allows) ===
    remaining = deadline - time.monotonic()
    warmup_n = 2048
    S_warm = np.sqrt(2.0 * warmup_n)
    
    if given is not None:
        warmup_init = interpolate_sequence(given, warmup_n)
        warmup_init = normalize_sum(warmup_init, warmup_n)

        warmup_init = generate_asymmetric_init(warmup_n, rng)[0]
        warmup_init = normalize_sum(warmup_init, warmup_n)
    
    if remaining > 300:
        warmup_time = min(60.0, remaining * 0.1)
        print("=== Phase 0: Warmup at n=%d (%.0fs) ===" % (warmup_n, warmup_time))
        a_warm, s_warm = run_optimization(warmup_init, warmup_n, S_warm, warmup_time, rng,
                                           use_gs=True, use_lp=False, use_peak_reduce=True)
        obj_warm = 2.0 * warmup_n * fft_convolve_self(a_warm).max() / (a_warm.sum() ** 2)
        print("  Warmup best: %.6f" % obj_warm)
        # Use as initialization for main optimization
        main_init = interpolate_sequence(a_warm, main_n)
    elif given is not None:
        main_init = interpolate_sequence(given, main_n)
    else:
        main_init = generate_asymmetric_init(main_n, rng)[0]
    
    main_init = normalize_sum(main_init, main_n)
    
    # === Phase 1: Main optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase1_time = remaining * 0.5
        print("=== Phase 1: Main optimization at n=%d (%.0fs) ===" % (main_n, phase1_time))
        a1, s1 = run_optimization(main_init, main_n, S, phase1_time, rng,
                                   use_gs=True, use_lp=True, use_peak_reduce=True,
                                   gs_every=40, peak_every=150, lp_every=400)
        c1 = fft_convolve_self(a1)
        obj1 = 2.0 * main_n * c1.max() / (a1.sum() ** 2)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
        print("  Phase 1 best: %.6f" % global_best)
    
    # === Phase 2: Refine with different GS parameters ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        phase2_time = remaining * 0.3
        print("=== Phase 2: GS-heavy refinement (%.0fs) ===" % phase2_time)
        
        # Use more aggressive GS
        f2 = global_a.copy()
        y2 = f2.copy()
        t2 = 1.0
        start2 = time.monotonic()
        best2 = global_best
        beta2 = 50.0
        it2 = 0
        
        while time.monotonic() - start2 < phase2_time:
            it2 += 1
            c2 = fft_convolve_self(y2)
            cur2 = c2.max()
            obj2 = 2.0 * main_n * cur2 / (y2.sum() ** 2)
            if obj2 < best2:
                best2 = obj2
                f2 = y2.copy()
            
            # Alternate FISTA and GS
            if it2 % 3 == 0:
                # GS step
                gamma = 0.5 + 0.2 * np.random.random()
                blend = 0.6 + 0.3 * np.random.random()
                y2_new = gs_step(y2, S, gamma=gamma, blend=blend)
                c2_new = fft_convolve_self(y2_new)
                if c2_new.max() < cur2 * 1.01:
                    y2 = y2_new
                    t2 = 1.0
            else:
                # FISTA step
                beta2 = min(2000.0, beta2 * 1.001)
                grad2 = lse_gradient(y2, c2, beta2)
                step2 = 1.0 / (2.0 * beta2)
                x2 = project_simplex(y2 - step2 * grad2, S)
                t2_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t2 ** 2))
                y2 = x2 + ((t2 - 1.0) / t2_new) * (x2 - f2)
                y2 = np.maximum(y2, 0.0)
                f2 = x2
                t2 = t2_new
            
            # Periodic peak reduction
            if it2 % 300 == 0:
                y2 = direct_peak_reduce(y2, S, top_k=10, lr=0.08)
                t2 = 1.0
        
        obj2_final = 2.0 * main_n * fft_convolve_self(f2).max() / (f2.sum() ** 2)

            if g_lp is None:

            improved = False
            for frac in [1.0, 0.8, 0.6, 0.4, 0.2]:
                test = (1 - frac) * global_a + frac * g_lp
                test = np.maximum(test, 0)
                test = project_simplex(test, S)
                c_test = fft_convolve_self(test)
                obj_test = 2.0 * main_n * c_test.max() / (test.sum() ** 2)
                if obj_test < global_best:
                    global_best = obj_test
                    global_a = test.copy()
                    improved = True
                    print("  LP %d: %.6f (frac=%.1f)" % (n_lp, global_best, frac))
                    break
            if not improved:
                # Try peak reduction after LP
                global_a = direct_peak_reduce(global_a, S, top_k=15, lr=0.05)
                c_check = fft_convolve_self(global_a)
                obj_check = 2.0 * main_n * c_check.max() / (global_a.sum() ** 2)
                if obj_check < global_best:
                    global_best = obj_check
                    print("  LP %d + peak: %.6f" % (n_lp, global_best))
                else:
                    # LP not helping, break
                    if n_lp > 2:
                        break
    
    # === Phase 4: Final FISTA polish with high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("=== Phase 4: Final FISTA polish (%.0fs) ===" % remaining)
        f4 = global_a.copy()
        y4 = f4.copy()
        t4 = 1.0
        start4 = time.monotonic()
        beta4 = 500.0
        it4 = 0
        best4 = global_best
        
        while time.monotonic() - start4 < remaining - 5:
            it4 += 1
            c4 = fft_convolve_self(y4)
            cur4 = c4.max()
            obj4 = 2.0 * main_n * cur4 / (y4.sum() ** 2)
            if obj4 < best4:
                best4 = obj4
                f4 = y4.copy()
            
            beta4 = min(5000.0, beta4 * 1.0005)
            grad4 = lse_gradient(y4, c4, beta4)
            step4 = 1.0 / (2.0 * beta4)
            x4 = project_simplex(y4 - step4 * grad4, S)
            t4_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t4 ** 2))
            y4 = x4 + ((t4 - 1.0) / t4_new) * (x4 - f4)
            y4 = np.maximum(y4, 0.0)
            f4 = x4
            t4 = t4_new
        
        if best4 < global_best:
            global_best = best4
            global_a = f4.copy()
        print("  Phase 4 best: %.6f" % global_best)
    
    # === Phase 5: Try different n if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        alt_n = 16384
        S_alt = np.sqrt(2.0 * alt_n)
        print("=== Phase 5: Try n=%d (%.0fs) ===" % (alt_n, remaining))
        alt_init = interpolate_sequence(global_a, alt_n)
        alt_init = normalize_sum(alt_init, alt_n)
        
        a5, s5 = run_optimization(alt_init, alt_n, S_alt, min(remaining - 30, 600.0), rng,
                                   use_gs=True, use_lp=True, use_peak_reduce=True,
                                   gs_every=60, peak_every=200, lp_every=600)
        c5 = fft_convolve_self(a5)
        obj5 = 2.0 * alt_n * c5.max() / (a5.sum() ** 2)
        if obj5 < global_best:
            global_best = obj5

        else:
            print("  Phase 5 (n=%d): %.6f (no improvement)" % (alt_n, obj5))
    
    # Final verification
    if global_a is None:
