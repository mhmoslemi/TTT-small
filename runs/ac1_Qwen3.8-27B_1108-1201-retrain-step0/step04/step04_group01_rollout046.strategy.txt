    # Sequence configuration: powers of two
    n_vals = [1024, 8192, 32768]
    total_fitness = np.inf
    
    # Candidates initialization strategies
    candidates = []
    for n_len in n_vals:
        # Constant base
        candidates.append(np.ones(n_len) * 0.01)
        candidates.append(np.ones(n_len) * (np.sqrt(2 * n_len) / n_len))
        
        # Multi-modal: random + exponential + polynomial
        arr = rng.random(n_len)
        arr = (arr + rng.exponential(size=n_len)).clip(0, 1000)
        candidates.append(arr)
        
        # Polynomial initialization with controlled range
        if n_len > 10:
            t = np.linspace(0, 1, n_len)
            poly_degree = min(3, np.log(n_len) / np.log(10))
            poly_base = np.power(np.abs(t - 0.5), poly_degree, where=(t > 0) & (t < 1)).clip(0, 1000)
            poly_vals = np.pad(poly_base * (1 + 0.1 * rng.random(remaining)), (0, n_len))
            candidates.append(poly_vals)

    # Parent best candidate list
    best_candidates = [None, None, None]
    best_fitness = [np.inf] * 3
    active_length = None
    strategy_steps = ["shared_exponential", "individual_peak_shave", "dimension_refinement"]
    current_step = 0
    
    # Fitness list tracking
    fitness_history = [np.inf] * len(candidates)
    diversity_pockets = {}
    
    # Rough estimate steps: tournament coefficient varied by step
    step_duration = {
        0: 0.1,    # Coarse search
        1: 0.05,   # Detailed inspection
        2: 0.025   # Precision refinement
    }
    
    # Main EVOLUTION loop
    for idx in range(100):  # Control the total number of generations
        if remaining <= 0:
            break
            
        new_fitness = []
        for i in range(len(candidates)):
            # Check if candidate still within bounds
            try:
                if is_stale(candidates[i]):
                    continue
                fitness = evaluate_sequence(candidates[i])
            except Exception as e:
                print(f"Evaluation error for candidate {i}: {e}")
                fitness = np.inf
                
            new_fitness.append(fitness)

        # Update best_candidates with the current step's result
        for i, f in enumerate(new_fitness):
            if (not best_candidates[i]) or (new_fitness[i] < best_fitness[i]):
                best_candidates[i] = candidates[i]
                best_fitness[i] = new_fitness[i]
                diversity_pockets[i] = (i % 2 == 0)  # Track odd/even patterns for diversity
                
        # For lower fitness in previous steps, consider refinement
        if idx > 0 and best_fitness[0] > best_fitness[1]:
            current_step = 1
            
        if idx > 0 and best_fitness[1] > best_fitness[2]:
            current_step = 2
            
        remaining -= step_duration[current_step]
        
        # Update the sequence for the next generation based on current_step
        next_generation = best_candidates[current_step].copy()
        if current_step == 0:
            # Shared exponential decay
            next_generation = np.exp(next_generation) * 0.999
            next_generation = next_generation.clip(0, 1000)
        elif current_step == 1:
            # Individual improvement
            next_generation = refine_individual(next_generation)
        else:
            # Global integration
            next_generation = integrate_patterns(next_generation, best_candidates)
            
    # Determine which length yielded the best result
    target_sequence = best_candidates[np.argmin(best_fitness)]
    if target_sequence is None:
        target_sequence = np.ones(8192) * (np.sqrt(2*8192) / 8192)
        
    target_sequence = target_sequence.clip(0, 1000).tolist()
    return target_sequence

def is_stale(candidate):
    # Check if candidate contains NaN, INF, or zeros only
    for v in candidate:
        if np.isnan(v) or np.isinf(v) or v == 0.0:
            return True
    candidate_sum = np.sum(candidate)
    return candidate_sum < 0.01 or candidate_sum > 100
