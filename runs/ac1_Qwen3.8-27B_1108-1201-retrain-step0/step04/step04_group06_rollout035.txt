def bidirectional_spectral_step(a, N, n, top_k, alpha, rng, current_max_b):
    """

def peak_chasing_step(a, n, step_size, n_peaks=5):
    """
    Time-domain peak chasing: for each peak lag, compute the gradient direction
    and take a step to reduce it. Uses the fact that for non-negative a,
    reducing a[i] only decreases b(k) for k involving i.
    """
    S = a.sum()
    c = fft_convolve_self(a)
    c_max = c.max()

    # Find top peak lags
    top_lags = np.argsort(c)[::-1][:n_peaks]

    a_curr = a.copy()
    for lag in top_lags:
        k = int(lag)
        # The gradient of b(k) w.r.t. a[i] is a[k-i] (for valid indices)
        # To reduce b(k), decrease a[i] where a[k-i] is large
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi < i_lo:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = k - i_range
        # Gradient contribution at each i
        grad_contrib = a_curr[j_range]
        # Step: reduce a[i] proportionally to a[k-i]
        reduction = step_size * grad_contrib
        # Apply (only reduce, don't go negative)
        a_curr[i_range] -= reduction
        a_curr = np.maximum(a_curr, 0.0)
        # Rescale to maintain sum
        s_curr = a_curr.sum()
        if s_curr > 1e-15:
            a_curr = a_curr * (S / s_curr)
        else:
            a_curr = project_simplex(a, S)

    c_new = fft_convolve_self(a_curr)
    return a_curr, c_new.max()

def nesterov_peak_descent(a_init, n, N, S, max_time, rng):
    """
    Nesterov-accelerated gradient descent using multi-peak subgradient.
    The subgradient of max_k b(k) at the peak is H_k * a where H_k is the
    Hankel matrix for lag k.
    """
    a = a_init.copy()
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    momentum = np.zeros(n)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()

        # Top-k peaks
        top_k = min(32, 5 + it // 10)
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break

        # Weighted subgradient: sum of weighted correlation vectors
        grad = np.zeros(n)
        for i, k in enumerate(top_indices):
            w = top_vals[i] / total
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                grad[i_lo:i_hi + 1] += w * y[k - i_lo:k - i_hi + 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.01 * it))

        # Momentum update
        momentum = 0.9 * momentum - step * grad / gnorm
        x_new = y + momentum
        x_new = project_simplex(x_new, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t_val = t_new

                best_a = x_new.copy()

        if it % 500 == 0 and rng.random() < 0.2:

            t_val = 1.0

    t = np.arange(n, dtype=np.float64) / n
    if method == "cosine_mix":
        a = (0.5 + 0.2 * np.cos(2 * np.pi * 2 * t) + 0.15 * np.cos(2 * np.pi * 5 * t)
             + 0.1 * np.cos(2 * np.pi * 9 * t) + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        a = 0.4 + 0.3 * np.cos(2 * np.pi * (3 * t + 5 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "alternating":
        base = np.where(np.arange(n) % 2 == 0, 0.8, 0.2)
        a = base + 0.15 * np.cos(2 * np.pi * 7 * t) + 0.05 * rng.standard_normal(n)
    elif method == "bimodal":
        a = (0.3 + 0.4 * np.exp(-0.5 * ((t - 0.25) * 6) ** 2)
             + 0.3 * np.exp(-0.5 * ((t - 0.75) * 6) ** 2)
             + 0.05 * rng.standard_normal(n))
    elif method == "random_smooth":
        # Random frequencies with random phases
        n_freq = 16
        freqs = rng.integers(1, n // 2, size=n_freq)
        amps = rng.uniform(0.05, 0.3, size=n_freq)
        phases = rng.uniform(0, 2 * np.pi, size=n_freq)
        a = np.full(n, 0.5)
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * f * t + ph)
        a += 0.05 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)

    a = np.clip(a, 0.01, 2.0)
    return a


def multi_scale_init(n, rng):
    """
    Build a good initialization by optimizing at a smaller scale
    and interpolating up.
    """
    # Start with a smaller sequence
    n_small = min(1024, n)
    a_small = structured_initialization(n_small, rng, "cosine_mix")
    S_small = np.sqrt(2.0 * n_small)
    a_small = a_small * (S_small / a_small.sum())

    # Quick optimization at small scale
    N_small = 1 << (2 * n_small - 1).bit_length()
    a_opt, _ = iterative_spectral_optimize(a_small, n_small, N_small, S_small,
                                           5.0, rng, target=1.5)

    # Interpolate to full size
    x_small = np.linspace(0, 1, n_small)
    x_full = np.linspace(0, 1, n)
    a_full = np.interp(x_full, x_small, a_opt)

    # Add small high-frequency noise to avoid artifacts
    a_full += 0.02 * rng.standard_normal(n)
    a_full = np.clip(a_full, 0.01, None)

    return a_full


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses bidirectional spectral balancing, analytical coordinate descent,
    Nesterov peak chasing, and multi-scale initialization.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Determine main length
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        if a_candidate is None:
            return False
        a_candidate = safe_normalize(a_candidate, n)
        c = fft_convolve_self(a_candidate)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            return True
        return False

    # --- Build initializations ---
    inits = []

    # Try to use previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given[:n], 0, 1000).astype(np.float64)))
        elif given.size > 0 and np.sum(np.clip(given, 0, 1000)) > 0:
            # Resize to n
            x_old = np.linspace(0, 1, len(given))
            x_new = np.linspace(0, 1, n)
            resampled = np.interp(x_new, x_old, given)
            inits.append(("given_interp", np.clip(resampled, 0, 1000)))
    except Exception:
        pass

    # Structured initializations
    for method in ["cosine_mix", "chirp", "alternating", "bimodal", "random_smooth"]:
        inits.append((method, structured_initialization(n, rng, method)))

    # Multi-scale
    inits.append(("multiscale", multi_scale_init(n, rng)))

    # Random
    inits.append(("rand", 0.3 + 0.7 * rng.random(n)))

    n_inits = len(inits)
    print("  %d initializations" % n_inits)

    # === Phase 1: Spectral optimization from multiple starts (35% time) ===
    print("=== Phase 1: Spectral optimization ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.35
    per_init = phase1_budget / n_inits

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        a_norm = safe_normalize(arr, n)

        # Quick score
        c0 = fft_convolve_self(a_norm)
        sc0 = c0.max()
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: %.6f" % (name, global_best))

        # Spectral optimization
        rem_init = min(per_init, 45.0)
        time_budget = max(1.0, min(rem_init, (deadline - 60 - time.monotonic())))
        if time_budget < 2:
            continue
        a_opt, sc_opt = iterative_spectral_optimize(
            a_norm, n, N, S, time_budget, rng, target=1.50)
        if update_best(a_opt):
            print("  [%s] spectral: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    if global_a is None:
        global_a = safe_normalize(structured_initialization(n, rng, "cosine_mix"), n)
        global_best = fft_convolve_self(global_a).max()

    # === Phase 2: Focused refinement (40% time) ===
    print("=== Phase 2: Focused refinement ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.40

    a_curr = global_a.copy()
    stall_count = 0
    iteration = 0

    while time.monotonic() < t_phase2 + phase2_budget:
        iteration += 1

        # Spectral optimization
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem < 5:
            break
        time_budget = min(rem * 0.35, 25.0)
        a_sw, sc_sw = iterative_spectral_optimize(a_curr, n, N, S,
                                                  time_budget, rng, target=1.49)
        if update_best(a_sw):
            a_curr = a_sw.copy()
            stall_count = 0
            if iteration % 5 == 0:
                print("  [P2 iter %d] spectral: %.6f" % (iteration, global_best))
        else:
            stall_count += 1

        # Nesterov peak descent
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 5:
            time_budget = min(rem * 0.35, 20.0)
            a_ng, sc_ng = nesterov_peak_descent(a_curr, n, N, S, time_budget, rng)
            if update_best(a_ng):
                a_curr = a_ng.copy()
                stall_count = 0
                if iteration % 5 == 0:
                    print("  [P2 iter %d] nesterov: %.6f" % (iteration, global_best))
            else:
                stall_count += 1

        # Peak chasing
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 3:
            step_size = 0.0005 * S / (1.0 + 0.1 * iteration)
            a_pc, sc_pc = peak_chasing_step(a_curr, n, step_size, n_peaks=5)
            if update_best(a_pc):
                a_curr = a_pc.copy()
                stall_count = 0
                if iteration % 10 == 0:
                    print("  [P2 iter %d] peak_chase: %.6f" % (iteration, global_best))

        # Diversification when stalled
        if stall_count >= 4:
            rem = (t_phase2 + phase2_budget) - time.monotonic()
            if rem > 10:
                for sigma in [0.05, 0.1, 0.2, 0.3]:
                    a_pert, _ = spectral_step_random(a_curr, N, n, sigma, rng)
                    if update_best(a_pert):
                        a_curr = a_pert.copy()
                        stall_count = 0
                        print("  [P2] perturb(%.2f): %.6f" % (sigma, global_best))
                        break
                    # Short spectral optimization from perturbed
                    a_sw2, _ = iterative_spectral_optimize(a_pert, n, N, S,
                                                           min(5.0, rem * 0.3),
                                                           rng, target=1.49)
                    if update_best(a_sw2):
                        a_curr = a_sw2.copy()
                        stall_count = 0
                        print("  [P2] perturb+opt(%.2f): %.6f" % (sigma, global_best))
                        break
                stall_count = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # === Phase 3: Final polish (25% time) ===
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if global_a is not None and remaining > 5:
        # Nesterov with larger top-k
        seg1 = min(remaining * 0.25, 100.0)
        a_p1, _ = nesterov_peak_descent(global_a, n, N, S, seg1, rng)
        if update_best(a_p1):
            print("  [P3] nesterov: %.6f" % global_best)

        # Coordinate descent
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            seg2 = min(rem2 * 0.30, 80.0)
            a_p2, _ = analytical_coordinate_descent(global_a, n, S, seg2,
                                                     n_top=30)
            if update_best(a_p2):
                print("  [P3] coord_desc: %.6f" % global_best)

        # Fine spectral tuning
        rem3 = deadline - time.monotonic()
        if rem3 > 3:
            a_p3, _ = iterative_spectral_optimize(global_a, n, N, S,
                                                  min(rem3 * 0.25, 40.0),
                                                  rng, target=1.49)
            if update_best(a_p3):
                print("  [P3] fine_spectral: %.6f" % global_best)

        # One more coordinate descent
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_p4, _ = analytical_coordinate_descent(global_a, n, S,
                                                    min(rem4 * 0.4, 60.0),
                                                    n_top=50)
            if update_best(a_p4):
                print("  [P3] final_coord: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 20.0)))
    return [float(x) for x in global_a]
