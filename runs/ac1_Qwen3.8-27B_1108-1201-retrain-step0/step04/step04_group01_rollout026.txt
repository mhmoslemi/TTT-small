def fft_convolve_full(a):
    n = len(a)
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    b = fft_convolve_full(a)
    b_max = b.max()
    scaled = beta * (b - b_max)

        scale_factor = 500.0 / max(1e-15, (b - b_max).max())

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using adaptive LSE gradient descent
with Nesterov momentum, backtracking line search, and periodic noise restarts.
Uses FFT-based convolution for O(n log n) per iteration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_gradient(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    b = fft_convolve_full(a)
    b_max = b.max()
    diff = b - b_max
    max_diff = diff.max()
    if max_diff > 500.0 / beta:
        beta_eff = 500.0 / max_diff
    else:
        beta_eff = beta
    scaled = beta_eff * diff
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    lse_val = b_max + np.log(tot) / beta_eff
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, b


def compute_peak_gradient(a, b, top_k=32):
    """Weighted subgradient from top-k peaks of convolution."""
    n = len(a)
    top_k = min(top_k, len(b))
    idx = np.argsort(b)[::-1][:top_k]
    mean_b = b.mean()
    weights = np.maximum(b[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), b
    weights = weights / wsum
    w_full = np.zeros(2 * n - 1)
    w_full[idx] = weights
    fsize = 1 << (2 * n).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, b


def objective_val(a, b=None):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    if b is None:
        b = fft_convolve_full(a)
    return 2.0 * n * b.max() / (s * s)


def optimize_sequence(a_init, max_time, rng, mode="hybrid"):
    """
    Optimize a sequence using LSE gradient with Nesterov momentum.
    mode: 'lse', 'peak', or 'hybrid' (switches between them)
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0

    b0 = fft_convolve_full(y)
    best_obj = objective_val(y, b0)
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    last_improve = 0
    step = n * 0.5 / (S * S)
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive beta: start moderate, increase over time
        if mode == "lse":
            beta = 20.0 + 480.0 * (progress ** 0.6)
            lse_val, grad, b = compute_lse_gradient(y, beta)
        elif mode == "peak":
            k = min(64, 4 + int(30 * progress))
            grad, b = compute_peak_gradient(y, b=None, top_k=k) if b is None else compute_peak_gradient(y, fft_convolve_full(y), top_k=k)
        else:
            # Hybrid: use LSE for most, switch to peak in last 30%
            if progress < 0.7:
                beta = 20.0 + 200.0 * progress
                lse_val, grad, b = compute_lse_gradient(y, beta)
            else:
                k = 8 + int(48 * (progress - 0.7) / 0.3)
                b = fft_convolve_full(y)
                grad, b = compute_peak_gradient(y, b, top_k=k)

        cur_obj = objective_val(y, b)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
            stuck_count = 0

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        # Try the current step, halve if objective worsens
        f_current = cur_obj
        x_trial = project_simplex(y - step * grad, S)
        b_trial = fft_convolve_full(x_trial)
        f_trial = objective_val(x_trial, b_trial)

        backtrack_count = 0
        while f_trial > f_current and backtrack_count < 8:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            b_trial = fft_convolve_full(x_trial)
            f_trial = objective_val(x_trial, b_trial)
            backtrack_count += 1

        if f_trial <= f_current:
            step = min(step * 1.1, 2.0 * n / (S * S))

        # Nesterov update
        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Stuck detection with periodic restart
        if it - last_improve > 800:
            stuck_count += 1
            noise_scale = (0.003 + 0.004 * stuck_count) * S / np.sqrt(n)
            noise_scale = min(noise_scale, 0.05 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it
            step = n * 0.5 / (S * S)

        # Periodic deterministic perturbation
        if it % 2000 == 0 and progress > 0.2:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    # Final check
    b_final = fft_convolve_full(best_a)
    final_obj = objective_val(best_a, b_final)
    return best_a, final_obj
