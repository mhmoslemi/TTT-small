import numpy as np

from scipy.fft import fft, ifft


def compute_score_fast(a):
    """Compute objective 2n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))[:2 * n - 1]
    s = a.sum()
    if s < 0.01:
        return np.inf, c, c.max()
    score = float(2.0 * n * c.max() / (s * s))
    return score, c, c.max()


def compute_pressure(a, excess, L):
    """
    Compute pressure E[i] = sum_j excess[i+j] * a[j] for i=0..n-1 using FFT cross-correlation.
    """
    A = fft(a, L)
    E = np.real(ifft(fft(excess, L) * np.conj(A)))[:len(a)]
    return E


def multiplicative_update(a, excess, eta, L, regularization=0.0, n_uniform=None):
    """
    One step of multiplicative convolution flattening.
    a_new[i] = a[i] * exp(-eta * pressure[i] / max_pressure)
    With optional regularization toward uniform.
    """
    pressure = compute_pressure(a, excess, L)
    max_p = np.max(np.abs(pressure))
    if max_p < 1e-15:
        return a.copy()

    log_scale = -eta * pressure / max_p
    # Clip to avoid numerical issues
    log_scale = np.clip(log_scale, -5.0, 5.0)
    a_new = a * np.exp(log_scale)

    if regularization > 0.0 and n_uniform is not None:
        a_new = (1.0 - regularization) * a_new + regularization * n_uniform

    # Renormalize
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)

            if verbose_prefix:
                print(f"  {verbose_prefix} Reached target {target} at iter {it}")
            break

        if verbose_prefix and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  {verbose_prefix} iter={it} score={best_score:.6f} eta={eta:.3f} elapsed={elapsed:.1f}s")

    return best_a, best_score


def subgradient_polish(a_init, S, budget_s, rng, top_k=50, verbose_prefix=""):
    """
    Subgradient method with Nesterov acceleration for polishing.
    """
    n = len(a_init)
    L = 1 << (3 * n - 2).bit_length()
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_param = 1.0
    best_a = a.copy()
    best_score, _, _ = compute_score_fast(a)
    t_start = time.monotonic()
    it = 0

    # Project onto simplex {x >= 0, sum(x) = S}
    def project_simplex(v, S_val):
        n_v = len(v)
        v = np.asarray(v, dtype=np.float64)
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, n_v + 1, dtype=np.float64)
        theta_candidates = (cumsum - S_val) / idx
        valid = theta_candidates <= w
        if not np.any(valid):
            rho = n_v
        else:
            rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S_val) / rho
        return np.maximum(v - theta, 0.0)

    while time.monotonic() - t_start < budget_s:
        it += 1
        _, c, _ = compute_score_fast(y)
        y_score, _, _ = compute_score_fast(y)
        if y_score < best_score:
            best_score = y_score
            best_a = y.copy()

        # Compute subgradient from top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Gradient: grad[i] = 2 * sum_k w_k * c_k's derivative w.r.t a[i]
        # d c[k] / d a[i] = a[k-i] (if valid) + a[i-k] (if valid, for k-i != i-k)
        # For self-convolution: d c[k] / d a[i] = a[k-i] + a[i-k] (when k-i != i-k)
        # Or 2*a[0] if k-i == i-k == 0
        # Use FFT for efficiency
        w_full = np.zeros(2 * n - 1)
        for i_idx, k in enumerate(top_idx):
            w_full[int(k)] = weights[i_idx]

        # grad[i] = 2 * sum_k w_full[k] * a[k-i] (for valid k-i)
        # This is 2 * cross-correlation of w_full and a
        W = fft(w_full, L)
        A = fft(y, L)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_param = t_new

        if it % 200 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_param = 1.0

    return best_a, best_score


def make_diverse_inits(n, rng, S):
    """Create diverse initializations."""
    inits = []
    t = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) * S / n)

    # Sine
    inits.append(np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * t), 0.01) * S / n)

    # Multi-sine
    sig = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.15 * np.sin(2 * np.pi * 7 * t) + 0.1 * np.sin(2 * np.pi * 13 * t)
    inits.append(np.maximum(sig, 0.01) * S / np.maximum(sig, 0.01).mean() / n)
    inits[-1] = inits[-1] * S / inits[-1].sum() * n

    # Chirp
    chirp = 0.5 + 0.3 * np.sin(2 * np.pi * (5 * t + 3 * t * t))
    inits.append(np.maximum(chirp, 0.01))
    inits[-1] = inits[-1] * S / inits[-1].sum()

    # Random
    r = rng.random(n) + 0.1
    inits.append(r * S / r.sum())

    # Smooth random (low-pass)
    noise = rng.standard_normal(n)
    size = 1 << (2 * n - 1).bit_length()
    N = fft(noise, size)
    # Zero out high frequencies
    N[size // 4:] = 0
    N[:size // 4] = 0
    smooth = np.real(ifft(N))[:n]
    smooth = np.maximum(smooth - smooth.min() + 0.1, 0.01)
    inits.append(smooth * S / smooth.sum())

    # Ramp
    ramp = np.linspace(0.2, 1.0, n)
    inits.append(ramp * S / ramp.sum())

    # Two-scale: alternating high/low blocks
    block_sz = max(1, n // 16)
    alternating = np.tile(np.array([1.0, 0.4] * 8), n // 16 + 1)[:n]
    inits.append(alternating * S / alternating.sum())

    return inits


def basin_hop(a_best, S, rng, n_hops, hop_budget_each, L, target=1.45, verbose_prefix=""):
    """
    Perform basin hopping: perturb the best solution and re-optimize.
    Returns the best found.
    """
    n = len(a_best)
    best_a = a_best.copy()
    best_score, _, _ = compute_score_fast(a_best)

    for h in range(n_hops):
        # Generate perturbation
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Gaussian noise
            scale = 0.1 + 0.1 * rng.random()
            a_pert = a_best + scale * S / np.sqrt(n) * rng.standard_normal(n)
        elif strategy == 1:
            # Multiply by random smooth function
            freq = rng.integers(1, n // 4 + 1)
            phase = rng.random() * 2 * np.pi
            amp = 0.3 + 0.3 * rng.random()
            mod = 1.0 + amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
            a_pert = a_best * mod
        elif strategy == 2:
            # Block permutation
            n_blocks = rng.integers(4, 17)
            block_sz = n // n_blocks
            indices = list(range(n_blocks))
            rng.shuffle(indices)
            a_pert = np.zeros(n)
            for new_i, old_i in enumerate(indices):
                start = new_i * block_sz
                end = (new_i + 1) * block_sz if new_i < n_blocks - 1 else n
                ostart = old_i * block_sz
                oend = (old_i + 1) * block_sz if old_i < n_blocks - 1 else n
                a_pert[start:end] = a_best[ostart:oend]
        else:
            # Random restart (partially)
            a_random = rng.random(n) + 0.1
            mix = 0.3 + 0.4 * rng.random()
            a_pert = (1 - mix) * a_best + mix * a_random * S / a_random.sum()

        a_pert = np.maximum(a_pert, 0.0)
        ps = a_pert.sum()
        if ps < 1e-15:
            continue
        a_pert = a_pert * (S / ps)

        # Optimize from perturbed point
        a_opt, sc_opt = run_multiplicative_flatten(
            a_pert, S, hop_budget_each, rng,
            target=target, eta_init=0.5,
            verbose_prefix=f"{verbose_prefix}hop{h}"
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()

    return best_a, best_score


def optimize_length(n, budget_s, rng, target=1.45, verbose_prefix=""):
    """
    Full optimization pipeline for a single sequence length.
    """
    S = np.sqrt(2.0 * float(n))
    L = 1 << (3 * n - 2).bit_length()

    if verbose_prefix:
        print(f"=== n={n}, S={S:.4f}, L={L}, budget={budget_s:.0f}s ===")

    # Try provided sequence if available
    inits = []
    init_names = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            g = np.clip(given, 0, 1000)
            gs = g.sum()
            if gs > 0.01:
                inits.append(g * (S / gs))
                init_names.append("provided")
    except Exception:
        pass

    # Add diverse inits
    diverse = make_diverse_inits(n, rng, S)
    for i, d in enumerate(diverse):
        inits.append(d)
        init_names.append(f"diverse_{i}")

    global_best = np.inf
    global_a = None

    t_start = time.monotonic()
    per_init_budget = budget_s * 0.6 / len(inits)  # 60% for initial optimization

    for idx, (a_init, name) in enumerate(zip(inits, init_names)):
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s * 0.7:
            break

        alloc = min(per_init_budget, budget_s * 0.7 - elapsed)
        if alloc < 5:
            break

        a_opt, sc_opt = run_multiplicative_flatten(
            a_init, S, alloc, rng,
            target=target, eta_init=0.8,
            verbose_prefix=f"{verbose_prefix}{name}"
        )
        if sc_opt < global_best:
            global_best = sc_opt
            global_a = a_opt.copy()
            if verbose_prefix or idx < 3:
                print(f"  [{name}] score={sc_opt:.6f} (elapsed {time.monotonic()-t_start:.1f}s)")

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best, _, _ = compute_score_fast(global_a)

    # Basin hopping phase
    elapsed = time.monotonic() - t_start
    bh_budget = max(0, budget_s * 0.2 - (elapsed - t_start))
    if bh_budget > 20:
        n_hops = max(2, int(bh_budget / 30))
        hop_budget = bh_budget / n_hops
        if verbose_prefix:
            print(f"  Basin hopping: {n_hops} hops, {hop_budget:.1f}s each")
        a_bh, sc_bh = basin_hop(global_a, S, rng, n_hops, hop_budget, L, target, verbose_prefix)
        if sc_bh < global_best:
            global_best = sc_bh
            global_a = a_bh

    # Final subgradient polish
    elapsed = time.monotonic() - t_start
    polish_budget = max(0, budget_s - elapsed)
    if polish_budget > 5:
        a_pol, sc_pol = subgradient_polish(global_a, S, polish_budget, rng,
                                           top_k=min(100, n // 64), verbose_prefix=verbose_prefix)
        if sc_pol < global_best:
            global_best = sc_pol
            global_a = a_pol
            if verbose_prefix:
                print(f"  After polish: {sc_pol:.6f}")

    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multiplicative convolution flattening + basin hopping + subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    deadline = t_global_start + budget_s - 30  # 30s reserve

    print(f"Starting optimization, budget={budget_s:.0f}s, seed={seed}")

    # Determine lengths to try
    lengths_to_try = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            lengths_to_try.append(len(given))
    except Exception:
        pass

    for L in [4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Remove duplicates, keep order
    seen = set()
    lengths_unique = []
    for L in lengths_to_try:
        if L not in seen:
            seen.add(L)
            lengths_unique.append(L)
    lengths_to_try = lengths_unique

    global_best = np.inf
    global_a = None

    # Allocate budget: favor the provided length and mid-range
    total_budget = budget_s - 60  # 60s reserve
    n_lengths = len(lengths_to_try)

    # Weight allocation: more time for provided length and 8192
    weights = []
    for L in lengths_to_try:
        if L == 8192:
            weights.append(1.5)
        elif L == 4096:
            weights.append(1.0)
        elif L == 16384:
            weights.append(0.8)
        else:
            weights.append(1.2)  # provided length gets priority
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_global_start
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(budgets[i], remaining - 30)
        if alloc < 30:
            break

        print(f"\n--- Length n={n}, allocated {alloc:.0f}s ---")
        a_result, score = optimize_length(n, alloc, rng, target=1.45,
                                          verbose_prefix=f"n{n}: ")

        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print(f"  *** New global best: {global_best:.6f} at n={n} ***")
        else:
            print(f"  Best for n={n}: {score:.6f} (global: {global_best:.6f})")

    # If we have time, do additional basin hopping on the global best
    elapsed = time.monotonic() - t_global_start
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        L_best = 1 << (3 * n_best - 2).bit_length()
        print(f"\n--- Final basin hopping: {remaining:.0f}s remaining ---")
        n_hops = max(3, int(remaining / 40))
        hop_budget = remaining * 0.7 / n_hops
        a_final, sc_final = basin_hop(global_a, S_best, rng, n_hops, hop_budget,
                                      L_best, target=1.45, verbose_prefix="final:")
        if sc_final < global_best:
            global_best = sc_final
            global_a = a_final
            print(f"  *** Improved to {global_best:.6f} ***")

        # Final quick polish
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_pol, sc_pol = subgradient_polish(global_a, S_best, rem2 - 5, rng,
                                               top_k=min(200, n_best // 32))
            if sc_pol < global_best:
                global_best = sc_pol
                global_a = a_pol

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best, _, _ = compute_score_fast(global_a)
        print(f"Fallback uniform n={n_fb}, score={global_best:.4f}")

    # Verify with evaluate_sequence
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"Verification: evaluate_sequence = {ver:.6f}")
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned inf")
            global_a = np.ones(8192, dtype=np.float64) * np.sqrt(2 * 8192) / 8192
            final_list = [float(x) for x in global_a]
    except Exception as e:
        print(f"Verification error: {e}")

    print(f"\nFinal: length={len(global_a)}, score={global_best:.6f}, time={time.monotonic()-t_global_start:.1f}s")
    return final_list
