def fft_convolve_self(a):

    A = fft(a, fsize)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

def normalize_sum(a, n):

def fista_exact(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=10000.0,

    """FISTA with LSE smoothing transitioning to exact subgradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_curr = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

        it += 1

        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur

            cm = c.max()
            for i, kk in enumerate(top_k):
                weights[i] = np.exp(beta * 0.01 * (c[kk] - cm))
            weights /= weights.sum()
            for i, kk in enumerate(top_k):
                lo = max(0, kk - n + 1)

        else:

            if tot < 1e-300:

            else:

                A = fft(a, fsize)

                            sk2 = src - kk
                            if 0 <= dk1 < n:
                                contrib += a[dk1]
                            if 0 <= dk2 < n:
                                contrib += a[dk2]
                            if 0 <= sk1 < n:
                                contrib -= a[sk1]
                            if 0 <= sk2 < n:
                                contrib -= a[sk2]
                            c_new = c[kk] + mid_d * contrib
                            if kk == 0:
                                c_new += 2.0 * mid_d * (a[dest] - a[src]) + 2.0 * mid_d * mid_d
                            c_trial_max = max(c_trial_max, c_new)
                        if c_trial_max < best * 1.0001:
                            lo_d = mid_d
                        else:
                            hi_d = mid_d
                    delta = lo_d * 0.9
                    if delta > best_delta:
                        best_delta = delta
                        best_dest = dest

            if best_dest >= 0 and best_delta > 1e-10:
                a[src] -= best_delta
                a[best_dest] += best_delta
                # Update c approximately
                for kk in range(2 * n - 1):
                    contrib = 0.0
                    dk1 = best_dest + kk
                    dk2 = best_dest - kk
                    sk1 = src + kk
                    sk2 = src - kk
                    if 0 <= dk1 < n:
                        contrib += a[dk1] - (a[dk1] - best_delta)  # a[dk1] already updated
                    # This is getting complicated; just recompute periodically
                    pass
                # Quick recompute of c for verification (expensive but accurate)
                if it % 5 == 0:
                    c = fft_convolve_self(a)
                    if c.max() < best:
                        best = c.max()
                        best_a = a.copy()
                        improved = True
                    top_lags = np.argsort(c)[::-1][:10]
                else:
                    # Approximate update
                    for kk in top_lags:
                        contrib = 0.0
                        dk1 = best_dest + kk
                        dk2 = best_dest - kk
                        sk1 = src + kk
                        sk2 = src - kk
                        # Use pre-update values (approximately)
                        if 0 <= dk1 < n:
                            contrib += a[dk1]
                        if 0 <= dk2 < n:
                            contrib += a[dk2]
                        if 0 <= sk1 < n:
                            contrib -= a[sk1]
                        if 0 <= sk2 < n:
                            contrib -= a[sk2]
                        c[kk] += best_delta * contrib
                    improved = True

        # Full recompute every 10 iterations
        if it % 10 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()
                improved = True
            top_lags = np.argsort(c)[::-1][:10]

        if not improved and it > 20:
            # Inject noise
            decay = np.exp(-it / 80.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            top_lags = np.argsort(c)[::-1][:10]
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def dct_subspace_opt(a_init, n, S, max_time, rng, d=128):
    """Optimize in a DCT subspace to make global shape changes."""
    a = normalize_sum(a_init, n)

    # DCT basis (type II)
    k = np.arange(n)
    m = np.arange(n)
    DCT = np.zeros((n, n))
    for j in range(n):
        DCT[j, :] = np.cos(np.pi * (2 * m + 1) * j / (2 * n))
    DCT[0, :] *= np.sqrt(1.0 / n)
    DCT[1:, :] *= np.sqrt(2.0 / n)

    # Work in the subspace spanned by the first d basis vectors
    D = DCT[:d, :].T  # n x d
    # Current coordinates in this subspace
    x = D.T @ a  # d x 1
    # The sum constraint: 1^T D x = sum(a), so we need D[0] * x[0] to account for sum
    # Actually, let's just optimize and project

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    best_x = x.copy()
    x_curr = x.copy()
    x_mom = x.copy()
    lr = 0.01
    it = 0
    patience = 500
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Reconstruct
        a_trial = a - D @ x_curr + D @ x_curr  # identity, just for clarity
        a_trial = normalize_sum(a, n)  # base
        a_trial = np.maximum(a_trial + D @ (x_curr - D.T @ a_trial), 0)
        # Simpler: just use D @ x + residual
        residual = a - D @ (D.T @ a)  # component orthogonal to subspace
        a_curr = np.maximum(residual + D @ x_curr, 0)
        a_curr = normalize_sum(a_curr, n)

        c = fft_convolve_self(a_curr)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = a_curr.copy()
            best_x = x_curr.copy()
            last_improve = it

        if it - last_improve > patience:
            break

        # Gradient in x-space
        # c_k = sum_i (residual_i + sum_j D[i,j] x_j) * (residual_{i+k} + sum_j D[i+k,j] x_j)
        # dc_k/dx_j = sum_i D[i,j] * a_curr[i+k] + sum_i D[i+k,j] * a_curr[i]
        # For max: use subgradient at argmax
        k_max = int(np.argmax(c))
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        grad_k = np.zeros(d)
        for j in range(d):
            grad_k[j] = np.sum(D[idx, j] * a_curr[k_max - idx]) + np.sum(D[k_max - idx + k_max - k_max:][:0], )
            # Simplified: grad_j = sum_i D[i,j]*a[i+k] + D[i+k,j]*a[i]
            # = sum_i D[i,j]*a[i+k] + sum_i D[i+k,j]*a[i]
            # Let p = i, q = i+k: sum_p D[p,j]*a[p+k] + sum_q D[q,j]*a[q-k]
            # = sum_i D[i,j]*a_curr[i+k_max] + sum_i D[(i+k_max) mod ?, j]*a_curr[i]
            # This is getting complex. Use finite differences instead for reliability.

        # Use numerical gradient (simpler and more robust for small d)
        eps = 0.01
        grad_num = np.zeros(d)
        for j in range(min(d, 64)):  # Limit for speed
            x_plus = x_curr.copy()
            x_plus[j] += eps
            a_plus = np.maximum(residual + D @ x_plus, 0)
            a_plus = normalize_sum(a_plus, n)
            c_plus = fft_convolve_self(a_plus)
            grad_num[j] = (c_plus.max() - cur) / eps

        # Gradient descent with momentum
        x_mom = 0.9 * x_mom - lr * grad_num
        x_curr = x_curr + x_mom

        # Project x to keep a non-negative (simple: if any a_i < 0, reduce x)
        a_check = residual + D @ x_curr
        if (a_check < -1e-6).any():
            x_curr = 0.9 * x_curr  # Scale back

        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.95

    return best_a, best


def spectral_refine(a_init, n, S, max_time, rng):
    """Iterative spectral flattening with adaptive parameters."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    alpha = 0.3

    while time.monotonic() - start < max_time:
        it += 1
        if it - getattr(spectral_refine, '_last_improve', 0) > 2000:
            break

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Adaptive flattening
        progress = min(1.0, it / 1000.0)
        alpha_cur = alpha + 0.4 * progress

        # Flat target with peak suppression
        P_ratio = P / P_mean
        P_suppress = np.where(P_ratio > 2.0, 1.0 / np.sqrt(P_ratio), 1.0)
        P_new = (1.0 - alpha_cur) * P / P_mean + alpha_cur * P_suppress
        P_new = P_new / P_new.mean()

        mag_new = np.sqrt(P_new)
        mag_new *= np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_new = np.real(ifft(F_new))[:n]

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
            spectral_refine._last_improve = it
        else:
            if not hasattr(spectral_refine, '_last_improve'):
                spectral_refine._last_improve = 0
        a = a_new

    return best_a, best


def generate_structural_variants(a, n, rng):
    """Generate diverse structural variants."""
    variants = [a.copy()]
    tt = np.linspace(0, 1, n, endpoint=False)
    S = np.sqrt(2.0 * n)

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, n))

    # Gaussian windowed
    gauss = a * np.exp(-4.0 * (tt - 0.5) ** 2)
    variants.append(normalize_sum(gauss, n))

    # Cosine windowed
    cosw = a * 0.5 * (1 + np.cos(2 * np.pi * tt))
    s = cosw.sum()
    if s > 1e-10:
        variants.append(cosw * (S / s))

    # Perturbed
    for scale in [0.02, 0.05, 0.1]:
        noise = rng.normal(0, scale, n)
        v = project_simplex(a + scale * S / np.sqrt(n) * noise, S)
        variants.append(v)

    # Smoothed version
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    for bw in [0.2, 0.5, 1.0]:
        F_sm = F * np.exp(-5.0 * (freqs / bw) ** 2)
        a_sm = np.real(ifft(F_sm))[:n]
        a_sm = np.maximum(a_sm, 0)
        s = a_sm.sum()
        if s > 1e-10:
            variants.append(a_sm * (S / s))

    return variants


def propose_candidate(seed=42, budget_s=2100):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 15.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192
        parent = np.ones(n) * (np.sqrt(2.0 * n) / n)

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, budget=%.0fs, parent_len=%s" % (
        n, budget_s, len(parent)))

    # Evaluate parent
    parent = normalize_sum(parent, n)
    parent_score = compute_objective(parent)
    print("Parent score: %.6f" % parent_score)

    global_best = parent_score
    global_a = parent.copy()

    def try_update(a_cand, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_cand, len(a_cand))
        sc = compute_objective(a_norm)
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [IMPROVE %s] score=%.6f (t=%.0fs)" % (label, sc, elapsed))
        return sc

    # Phase 1: FISTA from parent (40% of time)
    print("\n=== Phase 1: FISTA from parent ===")
    p1_time = min(700, (deadline - time.monotonic()) * 0.40)
    a_fista, sc_fista = fista_exact(parent, n, S, p1_time, rng,
                                     beta_init=50.0, beta_max=5000.0, beta_ramp=300)
    try_update(a_fista, "fista_parent")

    # Phase 2: Directed coordinate descent (25% of time)
    print("\n=== Phase 2: Directed CD ===")
    p2_time = min(500, (deadline - time.monotonic()) * 0.25)
    a_cd, sc_cd = directed_coord_descent(global_a, n, S, p2_time, rng, n_pairs=80)
    try_update(a_cd, "cd")

    # Phase 3: Spectral refinement (15% of time)
    print("\n=== Phase 3: Spectral refinement ===")
    p3_time = min(300, (deadline - time.monotonic()) * 0.15)
    spectral_refine._last_improve = 0
    a_spec, sc_spec = spectral_refine(global_a, n, S, p3_time, rng)
    try_update(a_spec, "spectral")

    # Phase 4: Second FISTA pass from best (10% of time)
    print("\n=== Phase 4: Second FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_time = min(200, remaining * 0.20)
        a_f2, sc_f2 = fista_exact(global_a, n, S, p4_time, rng,
                                   beta_init=200.0, beta_max=10000.0, beta_ramp=100)
        try_update(a_f2, "fista2")

    # Phase 5: Multi-restart with structural variants
    print("\n=== Phase 5: Multi-restart ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        variants = generate_structural_variants(global_a, n, rng)
        # Evaluate all variants quickly
        best_var_score = np.inf
        best_var = global_a.copy()
        for i, v in enumerate(variants):
            sc_v = compute_objective(v)
            if sc_v < best_var_score:
                best_var_score = sc_v
                best_var = v.copy()
            print("  variant_%d: %.6f" % (i, sc_v))

        if best_var_score < global_best:
            global_best = best_var_score
            global_a = best_var.copy()
            print("  [IMPROVE] best_variant: %.6f" % global_best)

        # Run FISTA on best variant
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_alloc = min(remaining2 * 0.5, 180)
            a_v, sc_v = fista_exact(best_var, n, S, t_alloc, rng,
                                     beta_init=100.0, beta_max=8000.0, beta_ramp=200)
            try_update(a_v, "variant_fista")

    # Phase 6: Final coordinate descent polish
    print("\n=== Phase 6: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_polish = min(remaining * 0.8, 120)
        a_polish, sc_polish = directed_coord_descent(global_a, n, S, t_polish, rng, n_pairs=100)
        try_update(a_polish, "polish")

    # Final verification
    print("\n=== Final Verification ===")
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Score: %.6f" % ver)
    else:
        print("Score: inf - using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returned: n=%d, score=%.6f, time=%.0fs" % (
        len(global_a), evaluate_sequence(final_list), time.monotonic() - t_start))
    return final_list
