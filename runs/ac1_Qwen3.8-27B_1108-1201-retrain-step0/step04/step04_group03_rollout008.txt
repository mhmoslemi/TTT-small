Minimize 2*n*max(f*f)/sum(f)^2 using:
1. Nesterov-accelerated subgradient with multi-peak averaging
2. LP-based best response with active set
3. Spectral domain shaping
4. Element-level local search

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient(f, c, top_k):
    """
    Compute a weighted subgradient from the top-k peaks of the convolution.
    Weight each peak by its softmax probability.
    """
    n = len(f)
    # Get top k values
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    
    # Softmax weights
    top_vals = c[top_idx]
    max_val = top_vals.max()
    weights = np.exp(50.0 * (top_vals - max_val))
    weights /= weights.sum()
    
    # Gradient from each lag
    grad = np.zeros(n)
    for i, lag in enumerate(top_idx):
        if weights[i] < 1e-10:
            continue
        # (f*f)(lag) = sum_j f(j) f(lag-j)
        # d/d f(m) = f(lag-m) + f(m-lag) (with boundary handling)

        # More precisely: c[lag] = sum_j f[j] f[lag-j] where 0<=j<n and 0<=lag-j<n
        # d c[lag] / d f[m] = f[lag-m] (if 0<=lag-m<n) + f[m-lag] (if 0<=m-lag<n)
        # But since we're computing the gradient of the quadratic form, it's 2 * (P_lag f)[m]
        # where (P_lag f)[m] = f[lag-m] if 0<=lag-m<n else 0
        
        # Vectorized: (P_lag f)[m] = f[lag-m] for m in range(max(0, lag-n+1), min(lag, n-1)+1)
        lo = max(0, lag - n + 1)
        hi = min(lag, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += weights[i] * f[lag - np.arange(lo, hi+1)]
    
    return 2.0 * grad

    """
    Nesterov-accelerated subgradient method with multi-peak averaging.
    """
    f = normalize_sum(f_init, n)
    y = f.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0
    
    # Polyak step size parameters
    f_opt_est = S * S / (2 * n - 1)  # Lower bound: sum(c) = S^2, so max >= S^2/(2n-1)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        
        if cur < best:
            best = cur
            best_f = y.copy()
            last_improve = it
        
        # Compute subgradient
        grad = compute_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Polyak step size: step = (cur - f_opt_est) / ||grad||^2
        # But cap it to avoid overshooting
        step_polyak = (cur - f_opt_est) / max(gnorm**2, 1e-15)
        step = min(step_polyak, S / (n * gnorm))  # Cap step
        
        # Nesterov acceleration
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new
        
        # Stagnation handling

            # Reset acceleration
            t = 1.0
            # Add noise

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            f = y.copy()
            last_improve = it
            # Increase top_k to get a broader gradient
            top_k = min(top_k + 5, 50)
    
    return best_f, best

"""

1. Nesterov-accelerated subgradient with adaptive multi-peak weighting and Polyak steps

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 1:
        return np.array([max(S, 0.0)])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_multi_peak_gradient(f, c, top_k, sharpness):
    """
    Compute gradient as weighted sum of shifts from top-k peaks.
    Uses softmax weighting with given sharpness.
    """
    n = len(f)
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    max_val = top_vals.max()
    weights = np.exp(sharpness * (top_vals - max_val))
    tot = weights.sum()
    if tot < 1e-300:
        weights = np.ones(k) / k
    else:
        weights /= tot

    grad = np.zeros(n)
    for i in range(k):
        if weights[i] < 1e-12:
            continue
        lag = top_idx[i]
        lo = max(0, lag - n + 1)
        hi = min(lag, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += weights[i] * f[lag - js]
    return 2.0 * grad


def run_nesterov_multi_peak(f_init, n, S, max_time, rng, top_k_init=20, top_k_max=80,
                            sharpness_init=20.0, sharpness_max=200.0):
    """
    Nesterov-accelerated subgradient with adaptive multi-peak averaging.
    Uses Polyak step size with adaptive lower bound estimation.
    """
    f = normalize_sum(f_init, n)
    y = f.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0
    top_k = top_k_init
    sharpness = sharpness_init

    # Lower bound on max(c): sum(c) = S^2, so max >= S^2/(2n-1)
    lb = S * S / (2.0 * n - 1)

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptively increase sharpness and top_k
        sharpness = min(sharpness_max, sharpness_init * (1.0 + 0.001 * it))
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()

        if cur < best:
            best = cur
            best_f = y.copy()
            last_improve = it
            # Tighten lower bound estimate
            lb = max(lb, cur * (1.0 - 0.3 * (1.0 - (2.0 * n - 1) / (2.0 * n))))

        grad = compute_multi_peak_gradient(y, c, top_k, sharpness)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step size
        gap = cur - lb
        if gap < 1e-12:
            step = 1e-6
        else:
            step = gap / (gnorm * gnorm + 1e-30)
            # Cap step to prevent instability
            max_step = S / (n * gnorm + 1e-15)
            step = min(step, max_step)
            # Also cap to prevent going too fast
            step = min(step, 0.5 * S / max(cur, 1e-10))

        # Nesterov update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new

        # Stagnation: reset and perturb
        if it - last_improve > 300:
            t = 1.0
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                f = y.copy()
            # Occasionally increase top_k for broader gradient
            top_k = min(top_k_max, top_k + 3)
            last_improve = it

    return best_f, best


def solve_lp_best_response(f, n, S, top_k=200, solver='SCS', time_limit=15.0):
    """
    LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max() * 1.0 + 1e-10  # Slight relaxation for numerical safety

    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start_t = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start_t > time_limit * 4:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def run_lp_iterations(f_init, n, S, max_time, rng, initial_top_k=100, max_top_k=400):
    """
    Iterative LP best response with bisection on step size.
    Gradually increase the number of constraints.
    """
    f = normalize_sum(f_init, n)
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0

    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(max_top_k, initial_top_k + 20 * it)
        if top_k > 400 and it > 5:
            top_k = 400  # Cap for computational tractability

        g = solve_lp_best_response(f, n, S, top_k=top_k, solver='SCS', time_limit=10.0)
        if g is None:
            continue

        # Bisection on step size
        c_old = fft_convolve_self(f)
        M_old = c_old.max()

        def eval_step(t_step):
            f_test = (1 - t_step) * f + t_step * g
            f_test = np.maximum(f_test, 0)
            f_test = project_simplex(f_test, S)
            c_test = fft_convolve_self(f_test)
            return c_test.max(), f_test

        # Try several step sizes
        best_t = 0.0
        best_val = M_old
        best_f_new = f.copy()

        # Try t=1 first (full replacement)
        val1, f1 = eval_step(1.0)
        if val1 < best_val:
            best_val = val1
            best_t = 1.0
            best_f_new = f1

        # Bisection
        lo, hi = 0.0, max(best_t, 0.5) if best_t > 0 else 1.0
        for _ in range(10):
            mid = (lo + hi) / 2.0
            val_mid, f_mid = eval_step(mid)
            if val_mid < best_val:
                best_val = val_mid
                best_t = mid
                best_f_new = f_mid
                lo = mid
            else:
                hi = mid

        if best_val < best * 0.9999:
            f = best_f_new
            if best_val < best:
                best = best_val
                best_f = f.copy()
                print("    LP iter %d (top_k=%d): %.6f" % (it, top_k, best))

        if time.monotonic() - start > max_time * 0.9:
            break

    return best_f, best


def run_spectral_shaping(f_init, n, S, max_time, rng, n_inner=50):
    """
    Frequency-domain iterative peak reduction.
    Work with |F(omega)| to flatten the autocorrelation.
    """
    f = normalize_sum(f_init, n)
    L = 2 * n - 1
    fsize = 1 << (L - 1).bit_length()

    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0

    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()

    while time.monotonic() - start < max_time:
        it += 1

        # Compute spectrum
        f_padded = np.zeros(fsize)
        f_padded[:n] = f
        F = fft(f_padded)
        G = np.abs(F) ** 2

        # IDFT of G gives circular autocorrelation
        r = np.real(ifft(G))[:L]

        # Find peaks and reduce them
        r_sorted = np.sort(r)[::-1]
        r_max = r_sorted[0]
        r_second = r_sorted[1] if len(r_sorted) > 1 else r_max * 0.9
        r_mean = np.sum(r) / L

        # Target: reduce peaks toward the mean
        target = max(r_mean * 1.5, r_second)
        clip_frac = 0.1 + 0.05 * min(it, 10)  # Gradually increase clipping

        # Soft clip: reduce values above target
        r_new = r.copy()
        excess = np.maximum(r - target, 0)
        # Redistribute the excess to below-target bins
        below_mask = r < target
        if np.any(below_mask) and np.any(excess > 1e-15):
            redist = excess.sum() / below_mask.sum()
            r_new[below_mask] += redist * clip_frac
            r_new[excess > 1e-15] -= excess[excess > 1e-15] * clip_frac
            r_new = np.maximum(r_new, 0)
        else:
            r_new = r

        # DFT of new autocorrelation
        G_new = np.abs(fft(r_new, fsize)) ** 2
        # Hmm, this isn't quite right. Let me use a different approach.
        # Actually: we want |F_new|^2 to give autocorrelation close to r_new
        # So: F_new = sqrt(DFT(r_new)) * exp(i * phase(F))
        # But DFT(r_new) should be non-negative for this to work...
        # It won't be in general. Let's use a different approach.

        # Alternative: directly modify |F| to reduce peaks
        # The peak of r is at some index p. To reduce r[p], we need to
        # reduce the "coherence" of F at frequencies that contribute to r[p].
        # r[k] = IDFT(|F|^2)[k] = (1/fsize) * sum_m |F[m]|^2 * exp(2*pi*i*m*k/fsize)
        # To reduce r[p], we want to reduce |F[m]|^2 for m where exp(2*pi*i*m*p/fsize) is positive.

        # Simpler approach: iteratively reduce |F| at "problematic" frequencies
        F_phase = np.angle(F)
        F_mag = np.abs(F)

        # Compute which frequency bins contribute most to the peak
        p = np.argmax(r)
        phase_factor = np.cos(2.0 * np.pi * np.arange(fsize) * p / fsize)

        positive_freqs = phase_factor > 0.5

        # Reduce |F| at those frequencies
        reduction = 0.02 * (1.0 + 0.1 * min(it, 20))
        F_mag[positive_freqs] *= (1.0 - reduction)
        F_mag[np~positive_freqs] *= (1.0 + reduction * 0.5)  # Slight increase elsewhere

        # Reconstruct
        F_new = F_mag * np.exp(1j * F_phase)
        f_new_padded = np.real(ifft(F_new))[:n]

        # Project to feasible set
        f_new = np.maximum(f_new_padded, 0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new * (S / s_new)
        f_new = project_simplex(f_new, S)

        # Check improvement
        c_new = fft_convolve_self(f_new)
        val_new = c_new.max()

        if val_new < best:
            best = val_new
            best_f = f_new.copy()
            f = f_new.copy()
        else:
            # Rejected, add small random perturbation
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            f = project_simplex(f + noise, S)

    return best_f, best


def run_local_search(f_init, n, S, max_time, rng):
    """
    Coordinate descent style local search.
    For each element, try small changes and accept if max(f*f) decreases.
    Uses incremental convolution update for efficiency.
    """
    f = normalize_sum(f_init, n)
    c = fft_convolve_self(f)
    S_total = f.sum()
    start = time.monotonic()
    best = c.max()
    best_f = f.copy()

    # Precompute: for each index i, the "shift" vectors
    # When we change f[i] by delta, c changes by:
    # c[k] += delta * f[k-i] + delta * f[i-k] + delta^2 (if k=2i)
    # (with f[j]=0 for j outside [0,n-1])

    # We'll use a "warm start" approach: try random subsets of elements
    improvement_streak = 0
    delta_base = S / n * 0.01

    while time.monotonic() - start < max_time:
        # Pick a random element to perturb
        i = rng.integers(0, n)
        f_val = f[i]

        # Try several deltas
        best_delta = 0.0
        for sign in [-1, 1]:
            for mult in [0.5, 1.0, 2.0, 5.0]:
                delta = sign * delta_base * mult
                # Skip if it would make f[i] negative
                new_val = f[i] + delta
                if new_val < -1e-10:
                    continue

                # Compute new c incrementally
                c_new = c.copy()
                # c[k] changes by: delta * f[k-i] + delta * f[i-k] + delta^2 * [k==2i]
                for k in range(2 * n - 1):
                    contrib = 0.0
                    if 0 <= k - i < n:
                        contrib += delta * f[k - i]
                    if 0 <= i - k < n:
                        contrib += delta * f[i - k]
                    if k == 2 * i:
                        contrib += delta * delta
                    if contrib != 0:
                        c_new[k] += contrib

                # Check if max decreased
                new_max = c_new.max()
                if new_max < best * (1.0 - 1e-7):
                    best_delta = delta
                    c = c_new
                    f[i] = max(f[i] + delta, 0.0)
                    break
            if best_delta != 0:
                break

        if best_delta != 0:
            # Renormalize occasionally
            s = f.sum()
            if abs(s - S_total) / S_total > 0.01:
                f = f * (S_total / s)
                c = fft_convolve_self(f)
            improvement_streak += 1
            c_check = fft_convolve_self(f)
            if c_check.max() < best:
                best = c_check.max()
                best_f = f.copy()
        else:
            improvement_streak += 1
            if improvement_streak > 200:
                # Perturb randomly to escape
                idx_perturb = rng.integers(0, n, 5)
                for ip in idx_perturb:
                    f[ip] = max(f[ip] + rng.normal(0, delta_base), 0)
                f = project_simplex(f, S_total)
                c = fft_convolve_self(f)
                improvement_streak = 0

        # Recompute c periodically for accuracy
        if improvement_streak % 50 == 0:
            c = fft_convolve_self(f)
            cur_max = c.max()
            if cur_max < best:
                best = cur_max
                best_f = f.copy()

    return best_f, best

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    result = np.interp(t_new, t_old, a)
    return np.maximum(result, 0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: Nesterov subgradient -> LP refinement -> spectral shaping -> local search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Get initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)
    print("Main length: %d, S=%.4f, budget=%.0fs" % (n, S, budget_s))

    # Initialize
    if given is not None:
        f_main = normalize_sum(given, n)
    else:
        f_main = np.ones(n) / n * S

    c_init = fft_convolve_self(f_main)
    global_best = c_init.max()
    global_a = f_main.copy()
    obj_init = 2.0 * n * global_best / (f_main.sum() ** 2)
    print("Initial: max_conv=%.6f, objective=%.6f" % (global_best, obj_init))

    # Phase 1: Combined Nesterov + LP optimization
    remaining = deadline - time.monotonic()
    if remaining > 100:
        phase1_time = remaining * 0.45
        print("=== Phase 1: Combined Nesterov+LP (%.0fs) ===" % phase1_time)
        t_start = time.monotonic()
        f1, s1 = run_combined_optimization(f_main, n, S, phase1_time, rng)
        f1 = normalize_sum(f1, n)
        c1 = fft_convolve_self(f1)
        obj1 = 2.0 * n * c1.max() / (f1.sum() ** 2)
        print("  Phase 1 done in %.0fs: max_conv=%.6f, obj=%.6f" %
              (time.monotonic() - t_start, c1.max(), obj1))
        if obj1 < global_best:
            global_best = obj1
            global_a = f1.copy()

    # Phase 2: Dedicated LP refinement
    remaining = deadline - time.monotonic()
    if remaining > 100:
        phase2_time = remaining * 0.25
        print("=== Phase 2: LP refinement (%.0fs) ===" % phase2_time)
        t_start = time.monotonic()
        f2, s2 = run_lp_iterations(global_a, n, S, phase2_time, rng,
                                   initial_top_k=150, max_top_k=350)
        f2 = normalize_sum(f2, n)
        c2 = fft_convolve_self(f2)
        obj2 = 2.0 * n * c2.max() / (f2.sum() ** 2)
        print("  Phase 2 done in %.0fs: max_conv=%.6f, obj=%.6f" %
              (time.monotonic() - t_start, c2.max(), obj2))
        if obj2 < global_best:
            global_best = obj2
            global_a = f2.copy()

    # Phase 3: Spectral shaping
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase3_time = remaining * 0.2
        print("=== Phase 3: Spectral shaping (%.0fs) ===" % phase3_time)
        t_start = time.monotonic()
        f3, s3 = run_spectral_shaping(global_a, n, S, phase3_time, rng)
        f3 = normalize_sum(f3, n)
        c3 = fft_convolve_self(f3)
        obj3 = 2.0 * n * c3.max() / (f3.sum() ** 2)
        print("  Phase 3 done in %.0fs: max_conv=%.6f, obj=%.6f" %
              (time.monotonic() - t_start, c3.max(), obj3))
        if obj3 < global_best:
            global_best = obj3
            global_a = f3.copy()

    # Phase 4: Local search
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase4_time = remaining * 0.3
        print("=== Phase 4: Local search (%.0fs) ===" % phase4_time)
        t_start = time.monotonic()
        f4, s4 = run_local_search(global_a, n, S, phase4_time, rng)
        f4 = normalize_sum(f4, n)
        c4 = fft_convolve_self(f4)
        obj4 = 2.0 * n * c4.max() / (f4.sum() ** 2)
        print("  Phase 4 done in %.0fs: max_conv=%.6f, obj=%.6f" %
              (time.monotonic() - t_start, c4.max(), obj4))
        if obj4 < global_best:
            global_best = obj4
            global_a = f4.copy()

    # Phase 5: Final high-precision Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase5_time = remaining * 0.6
        print("=== Phase 5: Final polish (%.0fs) ===" % phase5_time)
        t_start = time.monotonic()
        f5, s5 = run_nesterov_multi_peak(global_a, n, S, phase5_time, rng,
                                         top_k_init=50, top_k_max=100,
                                         sharpness_init=50.0, sharpness_max=500.0)
        f5 = normalize_sum(f5, n)
        c5 = fft_convolve_self(f5)
        obj5 = 2.0 * n * c5.max() / (f5.sum() ** 2)
        print("  Phase 5 done in %.0fs: max_conv=%.6f, obj=%.6f" %
              (time.monotonic() - t_start, c5.max(), obj5))
        if obj5 < global_best:
            global_best = obj5
            global_a = f5.copy()

    # Phase 6: Try a slightly different n if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 10000 else n // 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            phase6_time = remaining * 0.5
            print("=== Phase 6: Try n=%d (%.0fs) ===" % (alt_n, phase6_time))
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            t_start = time.monotonic()
            f6, s6 = run_nesterov_multi_peak(alt_init, alt_n, S_alt, phase6_time, rng,
                                             top_k_init=30, top_k_max=60,
                                             sharpness_init=20.0, sharpness_max=300.0)
            f6 = normalize_sum(f6, alt_n)
            c6 = fft_convolve_self(f6)
            obj6 = 2.0 * alt_n * c6.max() / (f6.sum() ** 2)
            print("  Phase 6 (n=%d): max_conv=%.6f, obj=%.6f" % (alt_n, c6.max(), obj6))
            if obj6 < global_best:
                global_best = obj6
                global_a = f6.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
            if ver > global_best * 1.001:
                print("  Warning: verification differs from internal score")
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, best score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
