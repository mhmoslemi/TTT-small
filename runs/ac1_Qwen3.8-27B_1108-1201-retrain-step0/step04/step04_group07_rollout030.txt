import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

- X[j, k-j] <= M
- diag(X) is a perfect square (dropped in relaxation)
- X is rank 1 (dropped in relaxation)

3. Repeat until convergence

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_score(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return float('inf'), c
    return float(2 * n * c.max() / (S * S)), c

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute objective value and convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return float('inf'), c
    return float(2 * n * c.max() / (S * S)), c


def lp_linearization_step(a_t, S, top_k=100, solver=None):
    """One step of LP successive linearization."""
    n = len(a_t)
    c_t = fft_convolve_self(a_t)
    top_idx = np.argsort(c_t)[::-1][:top_k]

    a_var = cp.Variable(n)
    M_var = cp.Variable()

    constraints = [a_var >= 0, cp.sum(a_var) == S]

    a_t_flat = a_t.ravel()
    for k in top_idx:
        ki = int(k)
        coeffs = np.zeros(n)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            coeffs[j] = a_t_flat[ki - j]
        constraints.append(2.0 * coeffs @ a_var - M_var <= c_t[ki])

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver] if solver else [cp.ECOS, cp.SCS, cp.CLARABEL]
    for slv in solvers_to_try:
        if slv is None:
            continue
        try:
            prob.solve(solver=slv)
            break
        except Exception:
            continue

    if a_var.value is None:
        return a_t, None

    a_new = np.maximum(a_var.value, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a_t, M_var.value
    a_new = a_new * (S / s)
    return a_new, M_var.value


def lp_polish(a_init, S, max_time, top_k_init=50, top_k_max=200, rng=None):
    """Iterative LP linearization polishing."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score, _ = compute_score(a)
    start = time.monotonic()
    it = 0
    patience = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adapt top_k: start small, grow over time
        progress = it / max(it + 100, 1)
        top_k = min(top_k_max, top_k_init + int(progress * (top_k_max - top_k_init)))
        top_k = max(20, top_k)

        a_new, M_lp = lp_linearization_step(a, S, top_k=top_k)

        if a_new is not None and not np.allclose(a_new, a):
            sc_new, c_new = compute_score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
            a = a_new
        else:
            patience += 1

        if patience > 15:
            # Perturb to escape
            if rng is not None:
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            patience = 0

        if best_score <= 1.502:
            break

    return best_a, best_score


def subgradient_nesterov(a_init, n, S, max_time, rng, top_k_init=30, top_k_max=100,
                         target_obj=1.502):
    """Nesterov-accelerated subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)[0]
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        S_y = y.sum()
        sc = 2.0 * n * cur_max / (S_y ** 2) if S_y > 0.01 else float('inf')
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
            if best_score <= target_obj:
                break

        # Adaptive top_k
        top_k = min(top_k_max, top_k_init + it // 200)
        top_k = max(10, top_k)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute gradient via FFT
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0
        fsize = 1 << (3 * n - 1).bit_length()
        W = fft(w, fsize)
        A_rev = fft(y[::-1], fsize)
        ca = np.real(ifft(W * A_rev))[:3 * n - 2]
        grad = 2.0 * ca[n - 1:2 * n - 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.05 * t))
        x_new = project_simplex(y - step * grad / gnorm, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y * (S / sy)
        a_prev = a.copy()
        a = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % 300 == 0 and rng is not None and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a_prev = y.copy()
            t = 1.0

    return best_local, best_score


def spectral_flatten(a, N, alpha):
    """One spectral flattening step."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    p_median = np.median(P)
    if p_median < 1e-30:
        return a
    ratio = P / p_median
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * np.maximum(0, 1.0 - alpha * (1.0 - ratio[above] ** (-alpha)))
    P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - ratio[~above] ** alpha))
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new


def make_initializations(n, rng):
    """Diverse initializations with emphasis on asymmetry."""
    inits = []

    # Exponential ramp (asymmetric)
    inits.append(("exp_ramp", np.exp(1.5 * tt) / np.exp(1.5 * tt).sum() * n * 0.5))

    # Power ramp
    inits.append(("pow_ramp", (tt ** 2 + 0.1) / ((tt ** 2 + 0.1).sum() / n) * 0.5))

    # Sine-modulated ramp (asymmetric with oscillation)
    inits.append(("mod_ramp", (0.3 + 0.7 * tt) * (1 + 0.2 * np.sin(2 * np.pi * 7 * tt)) + 0.05))

    # Sine
    inits.append(("sin1", 0.5 + 0.3 * np.sin(2 * np.pi * tt)))
    inits.append(("sin5", 0.5 + 0.3 * np.sin(2 * np.pi * 5 * tt)))

    inits.append(("chirp", 0.5 + 0.25 * np.sin(2 * np.pi * (3 * tt + 5 * tt ** 2))))

    # Multi-frequency
    inits.append(("multi", 0.4 + 0.2 * np.sin(2 * np.pi * 3 * tt) +
                  0.15 * np.sin(2 * np.pi * 11 * tt) + 0.1 * np.sin(2 * np.pi * 23 * tt)))

    inits.append(("random", rng.random(n) + 0.05))

    kern = np.ones(ks) / ks
    smooth = np.convolve(noise, kern, mode='same')

    inits.append(("cos_win", 0.3 + 0.4 * (1 + np.cos(2 * np.pi * (tt - 0.5))) / 2))
