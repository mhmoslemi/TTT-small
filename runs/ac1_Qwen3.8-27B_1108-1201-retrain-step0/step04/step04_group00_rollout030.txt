"""
Multi-resolution FISTA with adaptive LSE smoothing and active-set polishing.
Optimizes non-negative sequences to minimize 2*n*max_conv/sum^2.
Strategy: progressive resolution increase (2048→8192→16384→32768) with
adaptive beta scheduling, dual-weighted gradients, and structured restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.empty(n)
    np.sort(v, out=w)
    w = w[::-1]  # descending
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    """LSE-smoothed gradient of max(c) w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_gradient(y, c, k, power=1.0):
    """Gradient using top-k entries of c."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.maximum(vals, 1e-30) ** (power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def entropic_gradient(y, c, temperature=1.0):
    """Gradient using entropic weighting of active lags."""
    n = len(y)
    cm = c.max()
    # Temperature controls how "sharp" the weighting is
    shifted = (c - cm) / max(temperature, 1e-10)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w_full = np.zeros(len(c))
        w_full[np.argmax(c)] = 1.0
    else:
        w_full = e / tot
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def upsample(a, factor=2):
    """Upsample by interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def downsample(a, factor=2):
    """Downsample by block averaging."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

def run_fista(a_init, max_time, rng, beta_init=10.0, beta_final=5000.0,

    FISTA with adaptive smoothing.

    Returns (best_a, best_conv_max).
    """
    n = len(a_init)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()

    best_a = a.copy()

    last_improve = 0

    total_iters_est = max_time / 0.05  # estimate total iterations (50ms each)

            break

            step *= 0.5

            last_improve = it

            t_mom = 1.0

    return best_a, max(c_final.max(), best_conv)

"""
Multi-resolution FISTA with adaptive LSE smoothing and multi-mode polishing.
Minimizes 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: progressive resolution (8192→16384→32768), adaptive beta scheduling,
dual-weighted gradients, structured perturbations for escaping local minima,
and exact subgradient polishing at the end.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.empty(n)
    np.sort(v, out=w)
    w = w[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE-smoothed gradient of max(c) w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Gradient using top-k entries of c with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.maximum(vals, 1e-30) ** (power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def dual_weighted_gradient(y, c, alpha=1.0):
    """
    Gradient using weights proportional to (c_k)^alpha normalized.
    Spreads weight across all lags weighted by their value.
    """
    n = len(y)
    cm = c.max()
    if cm < 1e-30:
        return np.zeros(n)
    w_full = np.power(np.maximum(c, 0.0), alpha)
    tot = w_full.sum()
    if tot < 1e-300:
        w_full = np.zeros(len(c))
        w_full[np.argmax(c)] = 1.0
    else:
        w_full = w_full / tot
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def upsample_interp(a, n_new):
    """Upsample by linear interpolation to n_new points."""
    n = len(a)
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample_avg(a, n_new):
    """Downsample by block averaging to n_new points."""
    n = len(a)
    if n == n_new:
        return a.copy()
    a_trunc = a[:n_new * (n // n_new)]
    blocks = a_trunc.reshape(n_new, n // n_new).mean(axis=1)
    return blocks


def run_fista_adaptive(a_init, max_time, rng, beta_init=10.0, beta_max=8000.0,
                       step_scale=1.0, noise_period=4000, use_dual=False):
    """
    FISTA with adaptive LSE smoothing and optional dual-weighted gradient mixing.
    Cosine beta schedule. Returns (best_a, best_conv_max).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Estimate total iterations for cosine schedule
    # Measure first iteration time
    t0 = time.monotonic()
    c0 = fft_convolve_self(y)
    _ = lse_gradient(y, c0, beta_init)
    iter_time = max(time.monotonic() - t0, 0.005)
    est_total_iters = max(int(max_time / iter_time), 100)

    while time.monotonic() - start < max_time:
        it += 1

        # Cosine annealing for beta
        progress = min(1.0, it / est_total_iters)
        beta = beta_init + (beta_max - beta_init) * 0.5 * (1.0 - np.cos(np.pi * progress))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Gradient computation with adaptive mode switching
        if beta < 50:
            grad = lse_gradient(y, c, beta)
        elif beta < 500:
            g1 = lse_gradient(y, c, beta)
            if use_dual:
                g2 = dual_weighted_gradient(y, c, alpha=beta / 100.0)
                mix = min(1.0, (beta - 50) / 450.0)
                grad = (1.0 - mix) * g1 + mix * g2
            else:
                k = max(4, min(32, int(4 + (beta - 50) / 25)))
                g2 = topk_gradient(y, c, k, power=1.0)
                mix = min(1.0, (beta - 50) / 450.0)
                grad = (1.0 - mix) * g1 + mix * g2
        else:
            if stuck > 2000:
                k = min(48, 2 + stuck // 250)
                grad = topk_gradient(y, c, k, power=1.5)
            else:
                grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with adaptive reduction
        step = step_scale / max(beta, 1.0)
        if stuck > 2000:
            step *= 0.6
        if stuck > 5000:
            step *= 0.5

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if stuck > noise_period and it % 400 == 0:
            scale = max(1e-5, 0.01 * S / np.sqrt(n) * np.exp(-stuck / (1.5 * noise_period)))
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t_mom = 1.0
            last_improve = it

        # Periodic momentum reset
        if it % 8000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    actual_best = max(c_final.max(), best_conv)
    return best_a, actual_best

    No momentum, very careful step sizes.
    """

    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Exact subgradient: single argmax lag
        k_max = int(np.argmax(c))
        w_full = np.zeros(len(c))
        w_full[k_max] = 1.0
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Conservative step: O(1/sqrt(it))
        step = S / (gnorm * np.sqrt(1.0 + 0.005 * it))
        step = min(step, 5.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        y = x_new
        a = x_new

            y = project_simplex(y + rng.normal(0, scale, n), S)
            a = y.copy()
            last_improve = it

def try_perturbations(a_base, n, rng, budget_per=10.0):
    """
    Try several structured perturbations of a_base and return the best.
    Perturbations: add low-freq cosine, add high-freq noise, symmetric flip, etc.
    """
    S = np.sqrt(2.0 * n)
    t = np.linspace(0, 1, n, endpoint=False)
    candidates = [a_base.copy()]

    # Low-frequency cosine perturbation
    for freq in [2, 3, 5]:
        for amp in [0.05, 0.1, 0.2]:
            pert = a_base + amp * S / n * np.cos(2 * np.pi * freq * t)
            pert = np.maximum(pert, 0)
            if pert.sum() > 0.01:
                pert = pert * (S / pert.sum())
                candidates.append(pert)

    # Sine bump mixing
    bump = np.sin(np.pi * t)
    for mix in [0.1, 0.3, 0.5]:
        pert = (1 - mix) * a_base + mix * bump * S / bump.sum()
        pert = np.maximum(pert, 0)
        if pert.sum() > 0.01:
            pert = pert * (S / pert.sum())
            candidates.append(pert)

    # Small random perturbation
    for scale in [0.05, 0.1, 0.2]:
        pert = a_base + rng.normal(0, scale * S / np.sqrt(n), n)
        pert = np.maximum(pert, 0)
        if pert.sum() > 0.01:
            pert = pert * (S / pert.sum())
            candidates.append(pert)

    best_obj = compute_obj(a_base)
    best_a = a_base.copy()
    for cand in candidates[1:]:
        obj = compute_obj(cand)
        if obj < best_obj:
            best_obj = obj
            best_a = cand.copy()

    return best_a, best_obj

    # === Phase 1: FISTA at n=8192 (or resize initial to 8192) ===
    n1 = 8192
    if n_start != n1:
        t_old = np.linspace(0, 1, n_start, endpoint=False)
        t1 = np.linspace(0, 1, n1, endpoint=False)
        a1_init = np.interp(t1, t_old, a_start)
    else:
        a1_init = a_start.copy()
    S1 = np.sqrt(2.0 * n1)
    a1_init = a1_init * (S1 / a1_init.sum())

    print("\n=== Phase 1: FISTA at n=%d ===" % n1)
    t_phase = time.monotonic()

    a1, _ = run_fista_adaptive(a1_init, budget1, rng, beta_init=10.0,
                                beta_max=10000.0, step_scale=1.0,
                                noise_period=3000, use_dual=False)
    update_best(a1, "FISTA-8192-A")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        budget1b = min(remaining * 0.08, 180)

        a_pert, obj_pert = try_perturbations(global_best_a, n_pert, rng)

        # If perturbation helped, do a short FISTA from there
        if obj_pert < global_best_obj + 0.001 and remaining > 60:

    remaining = deadline - time.monotonic()
    if remaining > 120:

        if len(global_best_a) < n2:

        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())

        print("\n=== Phase 2: FISTA at n=%d ===" % n2)

                                    beta_max=8000.0, step_scale=1.0,
                                    noise_period=4000, use_dual=False)
        update_best(a2, "FISTA-16384")

        # Continue with top-k polish
        remaining = deadline - time.monotonic()
        if remaining > 40:

    # === Phase 3: Try n=32768 ===
    remaining = deadline - time.monotonic()
    if remaining > 300:

    # === Phase 4: Final exact polish on the best ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_best_a is not None:
        print("\n=== Phase 4: Exact polish (n=%d) ===" % len(global_best_a))
        t_phase = time.monotonic()
        budget4 = remaining * 0.50

    if remaining > 30 and global_best_a is not None:
        print("\n=== Phase 5: High-beta FISTA (n=%d) ===" % len(global_best_a))
        t_phase = time.monotonic()
        a5, _ = run_fista_adaptive(global_best_a, remaining * 0.8, rng,

        update_best(a5, "HighBeta-FISTA")

    # === Final verification ===
    print("\n=== Final ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected result, using fallback")
            n_fb = 8192
            final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb
    else:
        n_fb = 8192
        final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb

    return final_list
