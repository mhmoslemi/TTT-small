Minimize 2*n*max(convolve(f,f))/sum(f)^2 over non-negative sequences.
Uses a combination of multiplicative mirror descent, LP best-response,
and FISTA polishing with adaptive LSE smoothing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import coo_matrix


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def get_objective(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    c = fft_convolve_self(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def compute_repulsion(f, w):
    """
    Compute the repulsion gradient: grad[j] = sum_k w[k] * (f[k-j] + f[j-k])
    Using the symmetry of w (w[k] = w[2n-2-k]), this is 2 * cross-correlation.
    """
    n = len(f)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * cross


def multiplicative_update(f, S, beta, lr):
    """One multiplicative mirror descent step."""
    n = len(f)
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy(), c
    w /= tot
    grad = compute_repulsion(f, w)
    # Scale gradient for numerical stability
    gscale = max(np.abs(grad).max(), 1e-10)
    f_new = f * np.exp(-lr * grad / gscale)
    f_new = np.maximum(f_new, 1e-15)
    s = f_new.sum()
    if s < 1e-10:
        f_new = np.full(n, S / n)
    else:
        f_new = f_new * (S / s)
        f_new = project_simplex(f_new, S)
    return f_new, c


def solve_lp_best_response(f, S, top_k=500, solver='SCS', time_limit=15.0):
    """Solve the LP best-response problem."""
    n = len(f)
    c = fft_convolve_self(f)
    M = c.max()
    
    if M < 1e-15:
        return None
    
    # Select top-k lags (those closest to the maximum)
    num_lags = len(c)
    if top_k < num_lags:
        idx = np.argsort(c)[::-1][:top_k]
    else:
        idx = np.arange(num_lags)
    
    n_constraints = len(idx)
    
    # Build constraint matrix: A[i, j] = f[k_i - j] if 0 <= k_i - j < n
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    
    if not rows:
        return None
    
    A = coo_matrix((vals, (rows, cols)), shape=(n_constraints, n)).tocsr()
    
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    solvers_to_try = [solver, 'MOSEK', 'SCS', 'ECOS', 'CLARABEL']
    g = None
    for s in solvers_to_try:
        try:
            t0 = time.monotonic()
            prob.solve(solver=s, verbose=False)
            if time.monotonic() - t0 > time_limit:
                continue
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                if g.sum() > 1e-10:
                    g = g * (S / g.sum())
                    break
        except Exception:
            continue
    
    return g


def fista_step(f, S, beta, lr=None):
    """One FISTA step with LSE objective."""
    n = len(f)
    c = fft_convolve_self(f)
    cm = c.max()
    # LSE gradient
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return f.copy(), c
    w = e / tot
    grad = compute_repulsion(f, w)
    if lr is None:
        lr = 1.0 / (2.0 * beta)
    f_new = project_simplex(f - lr * grad, S)
    return f_new, c


def run_mirror_descent(f_init, S, max_time, rng, beta_start=1.0, beta_end=2000.0,
                       lr_start=0.5, lr_end=0.05, top_k_lp=300, lp_every=100,
                       noise_every=500):
    """
    Main optimization loop: multiplicative mirror descent with periodic LP.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    f = project_simplex(f, S)
    
    start = time.monotonic()
    best_obj = get_objective(f)
    best_f = f.copy()
    it = 0
    last_improve = 0
    lp_improvements = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Annealing schedule
        frac = min(1.0, it / (max_time / 0.01))  # This is wrong, let me fix
        frac = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Multiplicative update
        f, c = multiplicative_update(f, S, beta, lr)
        
        # Check improvement
        cur_obj = 2.0 * n * c.max() / (f.sum() ** 2)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_f = f.copy()
            last_improve = it
        
        # Periodic LP
        if it % lp_every == 0 and it > 50:
            t0 = time.monotonic()
            g = solve_lp_best_response(f, S, top_k=top_k_lp, time_limit=10.0)
            lp_time = time.monotonic() - t0
            if g is not None:
                # Try different step sizes
                for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    f_test = (1 - t) * f + t * g
                    f_test = project_simplex(np.maximum(f_test, 0), S)
                    obj_test = get_objective(f_test)
                    if obj_test < best_obj:
                        best_obj = obj_test
                        best_f = f_test.copy()
                        f = f_test.copy()
                        last_improve = it
                        lp_improvements += 1
                        print("  LP it=%d t=%.1f obj=%.6f (lp_time=%.1fs)" % (it, t, best_obj, lp_time))
                        break
                else:
                    # No improvement from LP, but keep f as is
                    pass
        
        # Periodic noise to escape local minima
        if it % noise_every == 0:
            decay = np.exp(-3.0 * frac)
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)
        
        # If stuck, try a bigger perturbation
        if it - last_improve > 2000:
            f = project_simplex(f + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
            last_improve = it
    
    return best_f, best_obj, lp_improvements


def run_fista_polish(f_init, S, max_time, rng, beta_start=50.0, beta_end=5000.0):
    """FISTA polishing with increasing beta."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    f = project_simplex(f, S)
    y = f.copy()
    t_k = 1.0
    
    start = time.monotonic()
    best_obj = get_objective(f)
    best_f = f.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        grad = compute_repulsion(y, w)
        
        lr = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - lr * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0)
        f = x_new
        t_k = t_new
        
        if it % 10 == 0:
            cur_obj = get_objective(f)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_f = f.copy()
                last_improve = it
        
        if it - last_improve > 5000:
            # Stuck, add noise
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            f = y.copy()
            last_improve = it
    
    cur_obj = get_objective(f)
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_f = f.copy()
    
    return best_f, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine-based
    for k in [1, 2, 3, 5]:
        for phase in [0, 0.25, 0.5]:
            base = 1.0 + 0.3 * np.cos(2 * np.pi * k * tt + 2 * np.pi * phase)
            inits.append(np.maximum(base, 0.01))
    
    # Gaussian-like
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            base = np.exp(-((tt - center) ** 2) / (2 * width ** 2))
            inits.append(base)
    
    # Random smooth
    for _ in range(5):
        coeffs = rng.normal(0, 0.3, min(n, 20))
        t_eval = tt
        signal = np.zeros(n)
        for i, c in enumerate(coeffs):
            signal += c * np.cos(2 * np.pi * (i + 1) * t_eval)
        inits.append(np.maximum(1.0 + signal, 0.01))
    
    # Parent sequence (if available)
    if parent is not None:
        inits.append(parent.copy())
    
    return [np.maximum(x, 0.01) for x in inits]
