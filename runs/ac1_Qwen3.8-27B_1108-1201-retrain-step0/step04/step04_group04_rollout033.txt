```python
from scipy.optimize import minimize

def lse_objective(a, beta):
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    return (cm + np.log(e.sum()) / beta)

def lse_gradient(a, beta):
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    w = e / e.sum()
    # grad of LSE wrt a
    fsize = 1 << (len(w) + len(a) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
```

```python
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def topk_subgradient(a, c, k=3):
    """Computes a weighted subgradient focusing on top-k correlation peaks.
    
    Returns gradient vector of length n.
    The subgradient of max_k c_k at a point where the max is attained at lag k*

    """

    max_c = c.max()
    
    # Find top-k lags
    top_idx = np.argsort(c)[::-1][:k]

    # but others get small weights
    beta_w = 50.0

    for w, k_lag in zip(weights, top_idx):

        # d c[k]/d a[j] = a[k-j] + a[j-k] (with bounds checking)

def lse_value_and_grad(a, beta):
    """Compute LSE value and gradient."""
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c


def bb_projected_gradient(a_init, n, S, max_time, rng, topk=3, use_lse=False, beta=200.0):
    """Projected gradient with Barzilai-Borwein step sizes.
    
    Alternates between:
    - Exact max subgradient (top-k weighted)
    - LSE smoothed gradient (when use_lse)
    """
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    a_prev = a.copy()
    g_prev = topk_subgradient(a, c, topk)
    it = 0
    stalled = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute gradient
        if use_lse and it % 10 == 0:
            _, g, c = lse_value_and_grad(a, beta)
        else:
            c = fft_convolve_self(a)
            g = topk_subgradient(a, c, topk)
        
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1
        
        # BB step size
        s_vec = a - a_prev
        y_vec = g - g_prev
        ss = np.dot(s_vec, s_vec)
        sy = np.dot(s_vec, y_vec)
        yy = np.dot(y_vec, y_vec)
        
        if yy < 1e-30 or ss < 1e-30:
            gamma = 1.0 / np.linalg.norm(g) if np.linalg.norm(g) > 1e-30 else 1.0
        elif sy > 1e-10:
            # BB+ and BB-
            gamma_plus = ss / sy
            gamma_minus = sy / yy
            gamma = 0.5 * (gamma_plus + gamma_minus)
            # Clip to reasonable range
            gnorm = np.linalg.norm(g)
            gamma = np.clip(gamma, 0.01 / gnorm, 10.0 / gnorm)
        else:
            gnorm = np.linalg.norm(g)
            gamma = 1.0 / gnorm if gnorm > 1e-30 else 1.0
        
        # Line search (backtracking)
        step = gamma
        a_trial = a.copy()
        for ls_iter in range(20):
            a_trial = project_simplex(a - step * g, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur - 1e-10:
                break
            step *= 0.5
        
        a_prev = a
        g_prev = g
        a = a_trial
        
        # Periodic noise
        if stalled > 100 and it % 50 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-stalled / 200.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                a_prev = a.copy()
                g_prev = topk_subgradient(a, fft_convolve_self(a), topk)
                stalled = 0
        
        if stalled > 500:
            # Major perturbation
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            a_prev = a.copy()
            g_prev = topk_subgradient(a, fft_convolve_self(a), topk)
            stalled = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def spectral_error_reduction(a_init, n, S, max_time, rng, target=0.0):
    """Spectral domain error reduction: push autocorrelation towards flat.
    
    This is a Gerchberg-Saxton variant where we modify the spectrum
    to reduce peaks in the autocorrelation.
    """
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        a_pad = np.zeros(fsize, dtype=np.complex128)
        a_pad[:n] = a
        F = fft(a_pad)
        
        # Current autocorrelation (periodic, but close to aperiodic for padded size)
        P = np.abs(F) ** 2
        c_periodic = np.real(ifft(P))[:2*n-1]
        c_periodic = c_periodic / c_periodic.max()  # Normalize
        
        # Target: as flat as possible
        # We want to reduce peaks: where c > mean, push down
        c_mean = c_periodic.mean()
        c_max = c_periodic.max()
        
        # Desired spectrum: reduce power at frequencies that contribute to peaks
        # This is approximate: modify P to flatten c
        # Simple approach: P_new = P / (c convolved with something)
        # Better: iterate - in time domain, set c to target, then go back to freq
        
        # Error reduction: e = c - c_target, where c_target = c_mean * ones
        c_target = np.ones(2 * n - 1) * c_mean
        e = c_periodic - c_target
        
        # In frequency domain, the correction is: delta_P = -FFT(e) * conj(F) / |F|^2 * F
        # Simplified: multiply F by a correction factor
        # |F_new|^2 = |F|^2 - alpha * IFFT(conj(F) * delta_c) ... this gets complex
        
        # Simpler approach: just flatten the power spectrum
        # P_new = P^alpha normalized
        progress = min(1.0, it / 500.0)
        alpha = 0.3 + 0.5 * progress
        P_flat = np.power(np.maximum(P, 1e-30), alpha)
        P_flat = P_flat / P_flat.mean() * P.mean()
        
        F_new = F * (P_flat / np.maximum(P, 1e-30)) ** 0.5
        F_new = F_new * (P.sum() / np.sum(np.abs(F_new)**2)) ** 0.5
        
        a_new = np.real(ifft(F_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        else:
            # Blend with current
            t = 0.5
            a_new = t * a_old + (1 - t) * a_new if 'a_old' in dir() else a_new
        
        a = a_new
    
    return best_a, best
```

```python
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_lse_grad(a, c, beta):
    """Compute LSE gradient given precomputed convolution c."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    size = 2 * n - 1
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient_at_max(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if hi >= lo:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_subgradient(a, c, k=5):
    """Weighted combination of subgradients at top-k lags."""
    n = len(a)
    max_c = c.max()
    # Get top-k indices
    if k >= len(c):
        top_idx = np.argsort(c)[::-1]
    else:
        # Use argpartition for efficiency
        part_idx = np.argpartition(c, -k)[-k:]
        top_idx = part_idx[np.argsort(c[part_idx])[::-1]]
    
    # Weights: exponential of (c[top] - max) * scale
    scale = 20.0
    vals = c[top_idx] - max_c
    weights = np.exp(scale * vals)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, kl in zip(weights, top_idx):
        lo = max(0, kl - n + 1)
        hi = min(kl, n - 1)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[kl - idx]
    return grad


def run_continuation_fista(a_init, n, S, max_time, rng):

    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Beta schedule: geometric from 5 to 3000
    beta_start = 5.0
    beta_end = 3000.0
    total_iters_estimate = 3000  # rough
    
    while time.monotonic() - start < max_time:
        it += 1
        # Geometric beta ramp
        progress = min(1.0, it / total_iters_estimate)
        beta = (beta_end / beta_start) ** progress * beta_start
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # LSE gradient
        g = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic noise for exploration
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best
```

Wait, I realize the penalty approach might not work well because L-BFGS-B might struggle with the penalty. Let me instead use the softmax parameterization.

Actually, for n=8192, the softmax parameterization means 8192 free variables, and L-BFGS-B should handle that fine. The issue is just numerical stability of the softmax.

Let me use a different approach: since the problem is scale-invariant, I can fix `sum(a) = S` by expressing one variable as the sum constraint:
`a[n-1] = S - sum(a[0:n-1])`

Then I optimize over `a[0:n-1]` with bounds `[0, S]` (to keep `a[n-1] >= 0`).

This reduces the dimension by 1 and eliminates the equality constraint. The bounds handle the non-negativity.

```python
def optimize_lse_lbfgs(a_init, n, S, max_time, beta=1000.0):
    """L-BFGS-B optimization on LSE-smoothed objective."""
    a = normalize_sum(a_init, n)
    
    def objective(x):
        a_full = np.empty(n)
        a_full[:n-1] = x
        a_full[n-1] = S - x.sum()
        if a_full[n-1] < 0:
            return 1e10, np.zeros(n-1)
        c = fft_convolve_self(a_full)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        val = cm + np.log(max(tot, 1e-300)) / beta
        
        w = e / max(tot, 1e-300)
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a_full, fsize)
        grad_full = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        # Chain rule: d/dx_i = d/d a_i for i < n-1, and d/d a_{n-1} * (-1)
        grad = grad_full[:n-1] - grad_full[n-1]
        return val, grad
    
    x0 = a[:n-1].copy()
    bounds = [(0.0, S)] * (n-1)
    
    start = time.monotonic()
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 10000, 'maxfun': 5000,
                               'max_time': max_time, 'gtol': 1e-8})
    
    a_opt = np.empty(n)
    a_opt[:n-1] = result.x
    a_opt[n-1] = S - result.x.sum()
    a_opt = np.maximum(a_opt, 0.0)
    a_opt = normalize_sum(a_opt, n)
    
    c = fft_convolve_self(a_opt)
    return a_opt, c.max()
```

This should work well for the smooth optimization phase. The key advantage of L-BFGS is that it uses second-order information (approximate Hessian) to take better steps than first-order methods.
