"""
Minimize 2*n*max_conv/sum^2 using multi-scale spectral shaping with adaptive
peak reduction, time-domain shaving, and simulated annealing polish.

Key improvements over previous approach:
1. Multi-n strategy: try n=8192, 16384 to exploit the 1/n decay of the excess.
2. Time-domain peak shaving: directly reduce elements contributing to peak lags.
3. Adaptive threshold shaping: use a decreasing threshold for which lags to target.
4. Simulated annealing on individual elements for final polish.
5. Better initialization using sum of sinusoids with optimized frequencies.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute the score 2*n*max_b/S^2 for a non-negative sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def init_sinusoidal(n, rng, n_harmonics=8):
    """Initialize with sum of random sinusoids (guaranteed non-negative base)."""
    t = np.arange(n)
    a = np.ones(n) * 0.5
    for _ in range(n_harmonics):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        a += amp * np.sin(2 * np.pi * freq * t / n + phase)
    a = np.clip(a, 0.01, None)
    return a


def init_smooth_random(n, rng, bandwidth=0.1):
    """Generate a smooth random non-negative sequence."""
    # Generate in frequency domain with low-pass filter
    N = 1 << (2 * n - 1).bit_length()
    spec = rng.standard_normal(N) + 0.5j * rng.standard_normal(N)
    # Low-pass filter
    m = np.arange(N)
    # Keep low frequencies
    cutoff = int(bandwidth * N / 2)
    spec[m > cutoff] *= 0.01
    spec[m < N - cutoff] *= 0.01  # Both sides for real signal
    a = np.real(ifft(spec))[:n]
    # Make non-negative
    a = np.abs(a) + 0.1
    return a


def init_spectral_flat(n, rng):
    """Initialize with flat spectrum (random phases) and project to non-negative."""
    N = 1 << (2 * n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N)
    # Flat amplitude

    # Ensure Hermitian symmetry for real output
    amps[N//2] = 0.05
    A = amps * np.exp(1j * phases)

    # Enforce Hermitian
    A = (A + np.conj(A[::-1])) / 2  # This isn't quite right for DFT
    # Actually, let's just use random phases and take real part
    A = amps * np.exp(1j * phases)
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.1
    return a


def spectral_shapeshart(a, n, N, S, target_ratio, max_iters, rng, 
                         top_k=10, alpha=0.3):
    """
    Spectral shaping to reduce peak autocorrelation.
    target_ratio: target value for max_b / (S^2/(2n-1)), i.e., target score / (2n/(2n-1))
    Returns (best_a, best_cmax).
    """
    a = a.copy()
    a = project_simplex(a, S)
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    
    m = np.arange(N)
    
    for it in range(max_iters):
        if time.monotonic() - start > max_iters:  # Use max_iters as time budget in seconds
            break
        
        c = fft_convolve_self(a)
        cmax = c.max()
        mean_c = c.sum() / (2 * n - 1)  # = S^2 / (2n-1)
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Check if we've hit the target
        target_cmax = target_ratio * mean_c
        if cmax <= target_cmax * 1.01:
            break
        
        # Find lags to target: those above target
        above = c > target_cmax
        n_above = above.sum()
        if n_above == 0:
            above = np.zeros(len(c), dtype=bool)
            above[np.argmax(c)] = True
        
        # Weight by excess
        excess = np.maximum(c - target_cmax, 0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            excess = np.maximum(c - cmax * 0.9, 0)
            total_excess = excess.sum()
        if total_excess < 1e-30:
            break
        weights = excess / total_excess
        
        # Compute spectral modification
        # The autocorrelation c[k] = sum_i a[i]*a[i+k]
        # In frequency: c[k] = (1/N) * sum_f |A(f)|^2 * exp(2*pi*i*f*k/N)
        # To reduce c at specific lags, we want to reduce |A(f)|^2 where
        # exp(2*pi*i*f*k/N) aligns with the peak direction.
        
        # Simplified: compute cos(2*pi*f*k/N) for each peak lag and
        # reduce where the weighted sum is positive.
        
        # Use top-k lags for efficiency
        top_indices = np.argsort(c)[::-1][:top_k]
        top_weights = c[top_indices]
        tw = top_weights / top_weights.sum()
        
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_indices):
            kk = int(k)
            cos_eff += tw[i] * np.cos(2 * np.pi * m * kk / N)
        
        # Exponential reduction where cos_eff > 0
        adaptive_alpha = alpha * min(2.0, (cmax - target_cmax) / (cmax + 1e-10) * 3 + 0.3)
        A = fft(a, N)
        P = np.abs(A) ** 2
        P_new = P * np.exp(-adaptive_alpha * np.maximum(cos_eff, 0.0))
        
        # Generate new sequence
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Accept if better (or with small probability for exploration)
        c_check = fft_convolve_self(a_new)
        if c_check.max() < cmax or (it > 50 and rng.random() < 0.05):
            a = a_new
    
    return best_a


def time_domain_shave(a, S, n_shave_steps=100, rng=None):

    Directly reduce elements that contribute most to peak autocorrelation.
    """

    best_a = a.copy()
    best_cmax = np.inf
    
    for step in range(n_shave_steps):
        c = fft_convolve_self(a)
        cmax = c.max()
        peak_lag = int(np.argmax(c))
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Find elements contributing most to the peak lag
        n = len(a)
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)

        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Reduce the top contributors
        n_top = min(10, n)
        top_idx = np.argsort(contribs)[::-1][:n_top]
        
        reduced = False
        for idx in top_idx:
            i = int(idx)
            orig = a[i]
            if orig < 1e-10:

"""
import numpy as np

import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_internal_score(a):

    S = a.sum()

    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)
