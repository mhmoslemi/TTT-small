    return a_new

    """
    Compute C_unaff[i] for all i: max c[k] among top M lags with k outside [i, i+n-1].
    Vectorized using prefix/suffix max.
    """
    # top_val: array of size 2n-1, 0 except at top lag positions
    total_lags = 2 * n - 1
    top_val = np.zeros(total_lags)
    for j in range(min(M, total_lags)):
        top_val[int(lag_order[j])] = c_sorted[j]

    running = 0.0
    for i in range(n):
        if i < total_lags:
            running = max(running, top_val[i])
        prefix_max[i] = running  # max of lags 0..i-1 = lags < i
    prefix_max[n] = running  # max of all lags 0..n-1

    suffix_max = np.zeros(total_lags + 1)
    running = 0.0
    for i in range(total_lags - 1, -1, -1):
        running = max(running, top_val[i])
        suffix_max[i] = running
    suffix_max[total_lags] = 0.0

    # prefix_max[i] = max of top lags with index < i
    # suffix_max[i+n] = max of top lags with index >= i+n
    C_unaff = np.maximum(prefix_max[:n], suffix_max[n:n+n])  # suffix_max[i+n] for i=0..n-1

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using vectorized coordinate descent with
exact optimal step computation, periodic spectral reshaping, and multi-start strategy.

Key innovations:
1. Vectorized coordinate descent: O(M*n) per pass with exact 1D optimal step
2. Prefix/suffix max for O(n) computation of unaffected-lag bounds
3. np.maximum.at for vectorized delta computation
4. Periodic spectral reshaping with random phase noise for global exploration
5. Multi-n search and structured initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def spectral_reshape(a, N, n, alpha, top_k, rng=None):
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for k in top_idx:
        cos_eff += np.cos(2.0 * np.pi * m_arr * int(k) / N)
    cos_eff /= top_k
    P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))
    phase = np.angle(A)
    if rng is not None and rng.random() < 0.4:
        phase += rng.normal(0, 0.03 * alpha, N)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new *= S / s
    return a_new


def compute_C_unaff(top_vals, top_order, n, M):
    total_lags = 2 * n - 1
    top_val = np.zeros(total_lags)
    for j in range(min(M, total_lags)):
        top_val[int(top_order[j])] = top_vals[j]
    prefix_max = np.zeros(n + 1)
    running = 0.0
    for i in range(n):
        if i < total_lags:
            if top_val[i] > running:
                running = top_val[i]
        prefix_max[i] = running
    prefix_max[n] = running
    suffix_max = np.zeros(total_lags + 1)
    running = 0.0
    for i in range(total_lags - 1, -1, -1):
        if top_val[i] > running:
            running = top_val[i]
        suffix_max[i] = running
    C_unaff = np.maximum(prefix_max[:n], suffix_max[n:n + n])
    return C_unaff


def coord_descent_loop(a, n, deadline, seed=42, M=80, reshape_every=40, alpha=0.35, delta_cap=0.8):
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1).bit_length()
    total_lags = 2 * n - 1
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve_self(a)
        P = c.max()
        S = a.sum()
        if S < 1e-15:
            break
        score = 2.0 * n * P / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 500 == 1:
            elapsed = time.monotonic() - start
            nz = np.sum(a > 1e-10)
            print("    CD%4d: score=%.6f P=%.3e nz=%d t=%.0fs" % (it, best_score, P, nz, elapsed))

        top_M = min(M, total_lags)
        top_order = np.argsort(c)[::-1][:top_M].astype(np.int64)
        top_vals = c[top_order]

        C_unaff = compute_C_unaff(top_vals, top_order, n, top_M)
        active_mask = (C_unaff < P) & (a > 1e-15)

        if not np.any(active_mask):
            a = spectral_reshape(a, N, n, alpha, 20, rng)
            continue

        delta_needed = np.zeros(n)

        for j in range(top_M):
            k = int(top_order[j])
            v = float(top_vals[j])
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo > i_hi:
                continue
            i_range = np.arange(i_lo, i_hi + 1)
            cu = C_unaff[i_range]
            mask = v > cu
            if not np.any(mask):
                continue
            i_sel = i_range[mask]
            cu_sel = cu[mask]
            j_vals = k - i_sel
            rates = 2.0 * a[j_vals]
            valid = rates > 1e-30
            if not np.any(valid):
                continue
            i_valid = i_sel[valid]
            cu_valid = cu_sel[valid]
            rates_valid = rates[valid]
            candidates = (v - cu_valid) / rates_valid
            np.maximum.at(delta_needed, i_valid, candidates)

        for j in range(top_M):
            k = int(top_order[j])
            if k % 2 == 0:
                i_center = k // 2
                if 0 <= i_center < n and a[i_center] > 1e-15:
                    v = float(top_vals[j])
                    C_u = C_unaff[i_center]
                    if v > C_u:
                        disc = a[i_center] ** 2 - v + C_u
                        if disc >= 0:
                            qd = a[i_center] - np.sqrt(disc)
                            if qd > delta_needed[i_center]:
                                delta_needed[i_center] = qd

        deltas = np.minimum(delta_needed, a * delta_cap)
        updated = deltas > 1e-15
        if np.any(updated):
            a[updated] -= deltas[updated]

        if it % reshape_every == 0:
            a = spectral_reshape(a, N, n, alpha, 20, rng)
            if it % (reshape_every * 5) == 0:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0.0)
                s2 = a.sum()
                if s2 > 1e-30:
                    a *= S / s2

    return best_a, best_score


def structured_init(n, rng, method):
    t = np.arange(n) / n
    if method == "cos1":
        a = 1.0 + 0.4 * np.cos(2 * np.pi * t)
    elif method == "cos2":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * t) + 0.15 * np.cos(2 * np.pi * 5 * t)
    elif method == "sin1":
        a = 1.0 + 0.4 * np.sin(2 * np.pi * t)
    elif method == "multi":
        a = 1.0 + 0.25 * np.cos(2 * np.pi * t) + 0.2 * np.cos(2 * np.pi * 3 * t) + 0.1 * np.cos(2 * np.pi * 7 * t)
    elif method == "bimodal":
        a = 0.5 + 0.8 * np.exp(-20 * (t - 0.25) ** 2) + 0.8 * np.exp(-20 * (t - 0.75) ** 2)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def safe_finalize(a, n):

    return [float(x) for x in a]
