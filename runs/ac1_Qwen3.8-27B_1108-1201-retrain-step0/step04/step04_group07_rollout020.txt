def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)

    c = np.real(ifft(A * A))

"""

multi-start search. Key innovations: large adaptive top-k active set, water-filling

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def get_topk_weights(c, top_k):
    """Get indices and normalized weights of top-k convolution values."""
    idx = np.argsort(c)[::-1][:top_k]
    vals = c[idx]
    total = vals.sum()
    if total < 1e-30:
        return None, None
    return idx, vals / total


def compute_grad_topk(y, top_idx, weights, n):
    """Gradient of weighted sum of top c[k] values, via FFT."""
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_project(a, S, alpha=0.4, n_steps=30):
    """
    Water-filling spectral projection: iteratively compress |FFT(a)|^2
    toward uniform using phase-preserving amplitude modification.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = a.copy()

    for _ in range(n_steps):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        ratio = P / P_mean
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P_mean * np.power(ratio[above], alpha)
        P_new[~above] = P_mean * np.power(ratio[~above], 1.0 / max(alpha, 0.01))
        P_new = np.maximum(P_new, 0.0)

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
        a = project_simplex(a, S)

    return a

            inits.insert(0, np.clip(given, 0, 1000).astype(np.float64))
    except Exception:
        pass

    n_inits = len(inits)
    # Allocate time: more for first (provided) init, less for later
    base_time = budget_s * 0.75 / n_inits

    t_start = time.monotonic()

    for i, init in enumerate(inits):
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s * 0.8:
            break

        # First init (provided) gets more time
        if i == 0:
            alloc = min(base_time * 2.5, budget_s * 0.8 - elapsed)
        else:
            alloc = min(base_time, budget_s * 0.8 - elapsed)

        if alloc < 5:
            break

        a = np.clip(init, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        # Main subgradient optimization
        a_res, sc = subgradient_optimize(a, n, S, alloc, rng,
                                         top_k_base=min(200, n // 40))

        if sc < global_best:
            global_best = sc
            global_a = a_res.copy()

        print("  [init %d/%d n=%d] score=%.6f elapsed=%.0fs" %
              (i + 1, n_inits, n, global_best, elapsed))

    # Coordinate descent polish on best
    if global_a is not None:
        rem = max(0.0, budget_s - (time.monotonic() - t_start))
        if rem > 10:
            a_cd, sc_cd = coordinate_descent_polish(global_a, n, S,
                                                     min(rem, 120.0), rng)
            if sc_cd < global_best:
                global_best = sc_cd
                global_a = a_cd.copy()
            print("  [CD polish n=%d] score=%.6f" % (n, global_best))

            # One more subgradient pass after CD
            rem2 = max(0.0, budget_s - (time.monotonic() - t_start))
            if rem2 > 10:
                a_sg2, sc_sg2 = subgradient_optimize(global_a, n, S,
                                                      min(rem2, 60.0), rng,
                                                      top_k_base=min(300, n // 32))
                if sc_sg2 < global_best:
                    global_best = sc_sg2
                    global_a = a_sg2.copy()
                print("  [Final SG n=%d] score=%.6f" % (n, global_best))

    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    Uses adaptive subgradient + spectral projection + coordinate descent,
    with multi-length and multi-start strategy.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + budget_s - 25

    print("Starting search, budget=%.0fs" % budget_s)
    t_global = time.monotonic()

    # Lengths to try
    lengths = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            lengths.append(given.size)
    except Exception:
        pass

    for L in [4096, 8192, 16384, 32768]:
        if L not in lengths:
            lengths.append(L)

    # Budget allocation: favor 8192-16384 (good balance of DOF and speed)
    allocs = {}
    for L in lengths:
        if L <= 4096:
            allocs[L] = 0.15
        elif L <= 8192:
            allocs[L] = 0.30
        elif L <= 16384:
            allocs[L] = 0.30
        else:
            allocs[L] = 0.25

    total_frac = sum(allocs.values())
    total_budget = budget_s - 40

    global_best = np.inf
    global_a = None

    # Sort lengths by expected quality (mid-range first)
    lengths_sorted = sorted(lengths, key=lambda x: abs(x - 8192))

    for n in lengths_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = total_budget * allocs[n] / total_frac
        alloc = min(alloc, remaining - 25)
        if alloc < 15:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        a_res, score = optimize_single_length(n, alloc, rng)

        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  *** New global best: %.6f at n=%d ***" % (global_best, n))
        elif a_res is not None:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

    # Fallback
    if global_a is None:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.ones(n_fb) * S_fb / n_fb
        global_best = compute_score(global_a)
        print("Fallback uniform: score=%.4f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: evaluate_sequence = %.6f" % ver)
    else:
        print("WARNING: verification failed, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %.4f" % ver)

    print("FINAL: length=%d, score=%.6f" % (len(final_list), ver if np.isfinite(ver) else global_best))
    return final_list
