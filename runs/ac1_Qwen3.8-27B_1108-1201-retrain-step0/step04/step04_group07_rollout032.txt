```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    rho = int(np.max(np.where(valid)[0])) + 1 if np.any(valid) else n
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_score(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))
```

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using soft-max gradient optimization
with Adam, spectral flattening, and subgradient polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def compute_convolution(a):
    """Self-convolution of a."""
    return fft_convolve(a, a)


def compute_score(a):
    """Compute the objective."""
    n = len(a)
    c = compute_convolution(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * np.max(c) / (S * S))


def soft_max_gradient(c, tau):
    """
    Compute softmax weights p[k] = exp(tau*c[k]) / sum(exp(tau*c)))
    and the gradient of soft_max_tau(c) with respect to a.
    Returns (p, grad) where grad has the same length as a.
    """
    # Stable softmax
    c_shifted = tau * (c - c.max())
    exp_c = np.exp(c_shifted)
    p = exp_c / exp_c.sum()
    return p


def compute_grad_soft_max(a, tau):
    """
    Compute gradient of soft_max_tau(convolve(a,a)) with respect to a.
    """
    n = len(a)
    c = compute_convolution(a)  # length 2n-1
    p = soft_max_gradient(c, tau)  # length 2n-1
    
    # grad = 2 * (p * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv = fft_convolve(p, a_rev)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv[n-1 : 2*n-1]
    return grad, c, p


def adam_soft_max_optimize(a_init, n, S, max_time, rng, tau_init=1.0, tau_max=2000.0,
                           lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                           tau_anneal_schedule='linear', noise_interval=1000, noise_std=0.001):
    """
    Optimize using Adam on the soft-max smoothed objective with tau annealing.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    t = 0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    best_tau = tau_init
    
    total_iters = max_time * 200  # rough estimate
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal tau
        progress = min(1.0, it / (total_iters * 0.8))
        if tau_anneal_schedule == 'linear':
            tau = tau_init + (tau_max - tau_init) * progress
        elif tau_anneal_schedule == 'exponential':
            tau = tau_init * (tau_max / tau_init) ** progress
        else:
            tau = tau_init + (tau_max - tau_init) * progress
        tau = min(tau, tau_max)
        
        # Compute gradient
        grad, c, p = compute_grad_soft_max(a, tau)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_simplex(a_new, S)
        
        # Noise for exploration
        if it % noise_interval == 0 and rng.random() < 0.3:
            a_new = project_simplex(a_new + rng.normal(0, noise_std * S / np.sqrt(n), n), S)
        
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_tau = tau
        
        if it % 5000 == 0:
            print("    soft_max it=%d tau=%.1f score=%.6f (time %.0fs)" % (it, tau, best_score, time.monotonic() - start))
    
    return best_a, best_score


def spectral_flatten(a, n_iters, alpha=0.3, S=None):
    """
    Iterative spectral flattening: push |FFT(a)|^2 toward uniform.
    """
    n = len(a)
    if S is None:
        S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        
        # Target: uniform power
        P_target = np.full(N, P.mean())
        # Blend
        P_new = (1 - alpha) * P + alpha * P_target
        P_new = np.maximum(P_new, 0)
        
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        a_new = project_simplex(a_new, S)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k=50):
    """
    Subgradient method on the actual max (not smoothed).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    N = 1 << (2 * n - 1).bit_length()
    
    # Momentum
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    while time.monotonic() - start < max_time:
        it += 1
        c = compute_convolution(a)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (a.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient: grad[i] = 2 * sum_k w_k * a[k-i] (valid k)
        # = 2 * (w * a_rev)[n-1 : 2n-1]
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        a_rev = a[::-1]
        conv = fft_convolve(w_full, a_rev)
        grad = 2.0 * conv[n-1 : 2*n-1]
        
        # Step size (decreasing)
        step_size = 2.0 * S / (n * np.sqrt(it + 1))
        
        # Update with momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        a_new = a - step_size * momentum
        a_new = project_simplex(a_new, S)
        a = a_new
        
        # Occasional noise
        if it % 2000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            momentum = np.zeros(n)
        
        if it % 5000 == 0:
            print("    subgrad it=%d score=%.6f (time %.0fs)" % (it, best_score, time.monotonic() - start))
    
    return best_a, best_score


def multi_start_spectral(n, n_starts, rng, S):
    """Generate multiple spectral-flattened starting points."""
    results = []
    N = 1 << (2 * n - 1).bit_length()
    
    for s in range(n_starts):
        if s == 0:
            a = np.ones(n) * S / n
        elif s == 1:
            a = np.abs(rng.standard_normal(n)) 
            a = a / a.sum() * S
        elif s == 2:
            # Asymmetric: weighted toward one end
            weights = np.linspace(0.1, 1.0, n) ** 2
            a = weights / weights.sum() * S
        elif s == 3:
            # Two peaks
            a = np.exp(-0.5 * ((np.arange(n) - n*0.2) / (n*0.1))**2) + \
                np.exp(-0.5 * ((np.arange(n) - n*0.8) / (n*0.1))**2) + 0.01
            a = a / a.sum() * S
        else:
            # Random with structure
            freq = rng.uniform(0.5, 5.0, size=3)
            phase = rng.uniform(0, 2*np.pi, size=3)
            t = np.arange(n) / n
            a = 1.0
            for f, p in zip(freq, phase):
                a += 0.3 * np.cos(2 * np.pi * f * t + p)
            a = np.maximum(a, 0.01)
            a = a / a.sum() * S
        
        a = project_simplex(a, S)
        results.append(a)
    
    return results
```

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using soft-max gradient optimization
with Adam, spectral flattening, peak-directed coordinate updates, and
subgradient polishing. Multi-length search with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    if fsize == 0:
        fsize = 1
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def compute_convolution(a):
    """Self-convolution of a (length 2n-1)."""
    return fft_convolve(a, a)


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = compute_convolution(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def compute_soft_max_grad(a, tau):
    """
    Compute gradient of soft_max_tau(convolve(a,a)) w.r.t. a.
    Returns (grad, c, p) where c is convolution and p is softmax weights.
    """
    n = len(a)
    c = compute_convolution(a)
    # Stable softmax
    c_shifted = tau * (c - c.max())
    exp_c = np.exp(c_shifted)
    p = exp_c / exp_c.sum()
    # grad = 2 * (p * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv = fft_convolve(p, a_rev)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c, p


def spectral_flatten(a, n_iters, alpha, S):
    """Iterative spectral flattening toward uniform |FFT|^2."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        P_new = (1.0 - alpha) * P + alpha * P_mean
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        a_new = project_simplex(a_new, S)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def adam_optimize_soft_max(a_init, n, S, max_time, rng,
                           tau_init=1.0, tau_max=2000.0,
                           lr_base=0.01, noise_interval=500, noise_scale=0.002):
    """Adam optimizer on soft-max smoothed objective with tau annealing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    while time.monotonic() - start < max_time:
        it += 1
        # Anneal tau: start smooth, end sharp
        progress = min(1.0, it / max(100.0, max_time * 80.0))
        tau = tau_init * (tau_max / tau_init) ** progress
        tau = min(tau, tau_max)

        # Adaptive learning rate: smaller as tau grows (gradient becomes peaked)
        lr = lr_base / (1.0 + 0.05 * tau)

        grad, c, p = compute_soft_max_grad(a, tau)

        # Normalize gradient for stability
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-30:
            grad = grad / gnorm * S / n

        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_simplex(a_new, S)

        # Noise for exploration
        if it % noise_interval == 0 and rng.random() < 0.3:
            a_new = project_simplex(
                a_new + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0

        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if it % 10000 == 0:
            print("      adam_soft it=%d tau=%.1f best=%.6f (%.0fs)" %
                  (it, tau, best_score, time.monotonic() - start))

    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k=50):
    """Subgradient method on exact max with momentum."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    momentum = np.zeros(n)
    beta_mom = 0.85

    while time.monotonic() - start < max_time:
        it += 1
        c = compute_convolution(a)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (a.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Top-k subgradient
        effective_k = min(top_k + it // 10000, len(c))
        top_idx = np.argpartition(c, -effective_k)[-effective_k:]
        top_vals = c[top_idx]
        # Weight by amount above median
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0.0)
        tot = weights.sum()
        if tot < 1e-30:
            weights = top_vals / top_vals.sum()
            tot = 1.0
        else:
            weights = weights / tot

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]

        a_rev = a[::-1]
        conv = fft_convolve(w_full, a_rev)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Decreasing step size
        step_size = 1.0 / (1.0 + 0.001 * it) * S / np.sqrt(n)

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        a_new = a - step_size * momentum
        a_new = project_simplex(a_new, S)
        a = a_new

        # Occasional restart noise
        if it % 5000 == 0 and rng.random() < 0.15:
            a = project_simplex(
                a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            momentum = np.zeros(n)

        if it % 10000 == 0:
            print("      subgrad it=%d best=%.6f (%.0fs)" %
                  (it, best_score, time.monotonic() - start))

    return best_a, best_score


def peak_directed_update(a, n, S, n_steps=100):
    """
    Direct peak reduction: identify peak lag, reduce contributing pairs.
    """
    best_a = a.copy()
    best_score = compute_score(a)
    a = a.copy()

    for step in range(n_steps):
        c = compute_convolution(a)
        peak_lag = int(np.argmax(c))
        peak_val = c[peak_lag]
        sc = 2.0 * n * peak_val / (a.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Find pairs contributing to peak
        # (a*a)[peak_lag] = sum_i a[i]*a[peak_lag - i] for valid i
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] = a[j]

        # Reduce high-contributing positions
        scale = 0.1 * peak_val / max(contrib.max(), 1e-15)
        a_new = a * (1.0 - scale * contrib)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        a_new = project_simplex(a_new, S)

        new_sc = compute_score(a_new)
        if new_sc <= best_score * 1.001:
            a = a_new
        else:
            break

    return best_a, best_score


def make_starts(n, S, rng, provided_seq=None):
    """Generate diverse starting points."""
    starts = []
    tt = np.arange(n) / n

    # Uniform
    starts.append(np.ones(n) * S / n)

    # Asymmetric ramp (favors one side)
    w = np.linspace(0.3, 1.0, n)
    starts.append(w / w.sum() * S)

    # Asymmetric ramp other direction
    w = np.linspace(1.0, 0.3, n)
    starts.append(w / w.sum() * S)

    # Sine-modulated
    a = 1.0 + 0.4 * np.cos(2 * np.pi * 1.5 * tt)
    a = np.maximum(a, 0.01)
    starts.append(a / a.sum() * S)

    # Random
    a = np.abs(rng.standard_normal(n)) + 0.1
    starts.append(a / a.sum() * S)

    # Smooth random (low-pass)
    a = rng.standard_normal(n)
    # Simple box filter
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0) + 0.01
    starts.append(a / a.sum() * S)

    # Cosine window
    a = 0.5 * (1.0 + np.cos(2 * np.pi * tt)) + 0.1
    starts.append(a / a.sum() * S)

    # Two-scale structure
    a = 1.0 + 0.3 * np.cos(2 * np.pi * 7.0 * tt) + 0.2 * np.cos(2 * np.pi * 13.0 * tt + 0.5)
    a = np.maximum(a, 0.01)
    starts.append(a / a.sum() * S)

    # Provided sequence if available
    if provided_seq is not None and len(provided_seq) == n:
        ps = np.clip(provided_seq, 0, 1000)
        if ps.sum() > 0.01:
            starts.insert(0, ps / ps.sum() * S)

    return starts


def optimize_single_length(n, budget, rng, provided_seq=None):
    """Full pipeline for one sequence length."""
    S = np.sqrt(2.0 * float(n))
    t_start = time.monotonic()
    best_score = np.inf
    best_a = None

    def elapsed():
        return time.monotonic() - t_start

    def try_update(a):
        nonlocal best_score, best_a
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if sc < 1.505:
                print("    >> New best: %.6f (n=%d, t=%.0fs)" % (sc, n, elapsed()))

    starts = make_starts(n, S, rng, provided_seq)
    n_starts = len(starts)
    remaining = budget - 15
    per_start = remaining / max(n_starts, 1)

    for si in range(n_starts):
        rem = budget - 10 - elapsed()
        if rem < 10:
            break

        a_work = starts[si].copy()
        alloc = min(per_start, rem)
        if alloc < 5:
            break

        init_sc = compute_score(a_work)
        print("  [n=%d start %d/%d] init=%.6f" % (n, si + 1, n_starts, init_sc))

        # Phase 1: Spectral flattening (20% of alloc)
        p1_budget = alloc * 0.2
        n_sf = max(30, int(p1_budget / 0.03))
        n_sf = min(n_sf, 300)
        a1, sc1 = spectral_flatten(a_work, n_sf, alpha=0.15, S=S)
        try_update(a1)

        rem = budget - 10 - elapsed()
        if rem < 5:
            continue

        # Phase 2: Soft-max Adam (50% of alloc)
        p2_budget = min(alloc * 0.5, rem * 0.6)
        a2, sc2 = adam_optimize_soft_max(
            a1, n, S, p2_budget, rng,
            tau_init=1.0, tau_max=500.0,
            lr_base=S / (n * 5.0),
            noise_interval=300, noise_scale=0.003)
        try_update(a2)

        rem = budget - 10 - elapsed()
        if rem < 5:
            continue

        # Phase 3: Subgradient polish (20% of alloc)
        p3_budget = min(alloc * 0.2, rem * 0.5)
        a3, sc3 = subgradient_polish(
            a2, n, S, p3_budget, rng,
            top_k=min(80, n // 200))
        try_update(a3)

        rem = budget - 10 - elapsed()
        if rem < 5:
            continue

        # Phase 4: Peak-directed + final soft-max
        p4_budget = min(alloc * 0.1, rem * 0.5)
        a4, sc4 = peak_directed_update(a3, n, S, n_steps=50)
        try_update(a4)

        rem = budget - 10 - elapsed()
        if rem > 10:
            a5, sc5 = adam_optimize_soft_max(
                a4, n, S, min(p4_budget, rem), rng,
                tau_init=50.0, tau_max=1500.0,
                lr_base=S / (n * 20.0),
                noise_interval=500, noise_scale=0.001)
            try_update(a5)

    # Final extended refinement on best
    if best_a is not None:
        rem = budget - 5 - elapsed()
        if rem > 20:
            print("  [n=%d] Final refinement, %.0fs remaining" % (n, rem))
            # High-tau soft-max for sharp peak targeting

                best_a, n, S, min(rem, 120.0), rng,
                tau_init=200.0, tau_max=3000.0,
                lr_base=S / (n * 50.0),
                noise_interval=1000, noise_scale=0.0005)
            try_update(a_fin)

            rem2 = budget - 5 - elapsed()
            if rem2 > 10:
                a_fin2, sc_fin2 = subgradient_polish(
                    best_a, n, S, min(rem2, 60.0), rng,

                try_update(a_fin2)

    return best_a, best_score

                    print("  Final subgrad improved to: %.6f" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("FALLBACK: uniform sequence, score=2.0")

    except Exception as e:
        print("Verification skipped: %s" % str(e))

    print("\n=== FINAL: length=%d, score=%.6f ===" % (len(global_a), global_best))
    return final_list
```
