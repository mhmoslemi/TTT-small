"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Hybrid of FISTA with LSE smoothing and exact LP best-response
iterations using the dual formulation for efficiency. Uses adaptive step
sizes, column generation for the LP, and multi-phase optimization.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def solve_dual_lp(f, M, n, max_time=30.0, top_k=None):
    """
    Solve the dual LP for the best-response problem.
    Dual: min M * sum(lambda) s.t. H @ lambda >= 1, lambda >= 0
    where H[i,k] = f[k-i] if 0 <= k-i < n.
    
    Returns the primal solution g (best-response) or None.
    """
    if top_k is not None:
        # Use only top-k most binding constraints (primal side)
        c = fft_convolve_self(f)
        idx = np.argsort(c)[::-1][:top_k]
        # Build reduced constraint matrix (top_k x n) for primal
        rows = np.zeros((top_k, n))
        for ii, k in enumerate(idx):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            rows[ii, js] = f[k - js]
        g_var = cp.Variable(n)
        constraints = [rows @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    else:
        # Use full dual formulation
        # Dual: min M * sum(lambda) s.t. H @ lambda >= 1, lambda >= 0
        # H is n x (2n-1), H[i,k] = f[k-i] if 0 <= k-i < n
        # Build sparse matrix H
        n_lambda = 2 * n - 1
        row_idx = []
        col_idx = []
        data = []
        for i in range(n):
            # For row i, non-zeros at columns k where 0 <= k-i < n
            # i.e., k in [i, i+n-1] ∩ [0, 2n-2]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            for k in range(k_start, k_end + 1):
                row_idx.append(i)
                col_idx.append(k)
                data.append(f[k - i])
        H = csr_matrix((data, (row_idx, col_idx)), shape=(n, n_lambda))
        lambda_var = cp.Variable(n_lambda)
        constraints = [H @ lambda_var >= 1, lambda_var >= 0]
        prob = cp.Problem(cp.Minimize(M * cp.sum(lambda_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver='MOSEK', verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                return None

    elapsed = time.monotonic() - start
    if elapsed > max_time:
        return None

    if top_k is not None:
        if g_var.value is not None:
            g = np.maximum(g_var.value, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                return None
            return g, None
    else:
        if lambda_var.value is not None:
            lam = np.maximum(lambda_var.value, 0)
            # Recover primal g from dual via complementary slackness
            # At optimum: g_i > 0 iff (H^T lambda)_i = 1
            # And: lambda_k > 0 iff (f*g)_k = M
            # We need to recover g. Use the fact that at KKT:
            # g_i * (1 - (H^T lambda)_i) = 0
            # So g_i > 0 only where (H^T lambda)_i = 1 (approximately)
            # For those i, we need to find g_i such that (f*g)_k = M for active k
            # This is a linear system. Let's use a simpler approach:
            # Solve the primal LP directly using the active set from dual
            active_k = np.where(lam > 1e-8)[0]
            if len(active_k) == 0:
                return None
            # Build subset of primal constraints for active k
            rows = np.zeros((len(active_k), n))
            for ii, k in enumerate(active_k):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                js = np.arange(lo, hi + 1)
                rows[ii, js] = f[k - js]
            g_var2 = cp.Variable(n)
            constraints2 = [rows @ g_var2 <= M, g_var2 >= 0]
            prob2 = cp.Problem(cp.Maximize(cp.sum(g_var2)), constraints2)
            try:
                prob2.solve(solver='MOSEK', verbose=False)
            except Exception:
                try:
                    prob2.solve(solver='SCS', verbose=False)
                except Exception:
                    return None
            if g_var2.value is not None:
                g = np.maximum(g_var2.value, 0)
                g_sum = g.sum()
                if g_sum < 1e-10:
                    return None
                return g, lam
            return None
    return None


def solve_best_response(f, M, n, max_time=20.0):
    """
    Solve the best-response LP. Tries exact dual first, falls back to top-k.
    Returns (g, active_lags) or (None, None).
    """
    # First try top-k=500 which is fast
    result = solve_dual_lp(f, M, n, max_time=max_time, top_k=500)
    if result is not None:
        return result
    # If that fails, try smaller k
    result = solve_dual_lp(f, M, n, max_time=max_time, top_k=200)
    if result is not None:
        return result
    return None, None


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002,
              beta_max=800.0, use_repulsion=True, repulsion_freq=300, repulsion_lr=0.04):
    """FISTA with LSE smoothing and optional repulsion updates."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 2000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it
            it = max(0, int(np.log(max(beta_init, 1) / beta_init) / beta_rate) if beta_rate > 0 else it // 3)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if use_repulsion and it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(beta * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 0)
                y = project_simplex(y, S)
                t = 1.0

        if it % 8000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def lp_iteration(f, n, S, max_time=15.0):
    """
    One LP best-response iteration with line search.
    Returns (f_new, M_new).
    """
    c_old = fft_convolve_self(f)
    M_old = c_old.max()

    g, active = solve_best_response(f, M_old, n, max_time=max_time)
    if g is None:
        return f.copy(), M_old

    # Rescale g to match sum
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f.copy(), M_old
    g = g * (S / g_sum)

    # Line search for best step size
    def eval_step(t_val):
        f_test = (1 - t_val) * f + t_val * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max()

    # Try a few step sizes and pick the best
    best_val = M_old
    best_t = 0.0
    for t_val in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1]:
        val = eval_step(t_val)
        if val < best_val:
            best_val = val
            best_t = t_val

    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val

    return f.copy(), M_old


def multiplicative_repulsion(f, n, S, beta=50.0, lr=0.08):
    """Fast multiplicative repulsion update targeting high autocorrelation."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def variance_repulsion(f, n, S, lr=0.05):
    """Repulsion based on variance of autocorrelation (targets non-peak lags too)."""
    c = fft_convolve_self(f)
    cm = c.max()
    # Target: push down all lags that are above mean
    c_mean = c.mean()
    # Weight: emphasize lags above mean
    w = np.maximum(c - c_mean, 0)
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w = w / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new

                    a = a_test.copy()
                    last_improve = it
                    print("  LP iter %d (frac=%.2f): %.6f" % (lp_count, t_frac, best))

        # Periodic FISTA polishing (a few iterations)
        fista_budget = min(30.0, remaining * 0.1)
        if fista_budget > 5:
            a_p, s_p = run_fista(a, n, S, fista_budget, rng,
                                 beta_init=50.0, beta_rate=0.003, beta_max=500.0,
                                 use_repulsion=False)
            a_p = normalize_sum(a_p, n)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < best:
                best = c_p.max()
                best_a = a_p.copy()
                a = a_p.copy()
                last_improve = it
                print("  Fista polish iter %d: %.6f" % (it, best))

        # Periodic repulsion
        if it % 3 == 0:
            a = multiplicative_repulsion(a, n, S, beta=30.0, lr=0.03)
            c_r = fft_convolve_self(a)
            if c_r.max() < best:
                best = c_r.max()
                best_a = a.copy()

    return best_a, best, lp_count


def run_enhanced_fista(a_init, n, S, max_time, rng):
    """
    Enhanced FISTA with multiple strategies:
    - Adaptive beta
    - Periodic variance-based repulsion
    - Periodic multiplicative repulsion
    - Periodic noise with cooling
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 30.0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase slowly
        beta = min(2000.0, 10.0 * np.exp(0.001 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        # Stuck detection
        if it - last_improve > 1500:
            # Try variance repulsion
            y = variance_repulsion(y, n, S, lr=0.05)
            c_v = fft_convolve_self(y)
            if c_v.max() < best:
                best = c_v.max()
                best_a = y.copy()
            # Add noise
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 30000.0)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            beta = max(10.0, beta * 0.5)
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic multiplicative repulsion
        if it % 200 == 0 and it > 100:
            y = multiplicative_repulsion(y, n, S, beta=40.0, lr=0.04)
            t = 1.0

        # Periodic variance repulsion
        if it % 500 == 0:
            y = variance_repulsion(y, n, S, lr=0.03)
            t = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-phase approach: FISTA warmup, LP best-response, enhanced
    FISTA polishing, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)
    print("Main length: %d, budget: %.0fs" % (main_len, budget_s))

    # Phase 1: Quick FISTA warmup (15% of time)
    phase1_time = (total_deadline - time.monotonic()) * 0.15
    if given is not None:
        init = normalize_sum(given, n)
        print("=== Phase 1: FISTA warmup (n=%d, %.0fs) ===" % (n, phase1_time))
        a1, s1 = run_enhanced_fista(init, n, S, phase1_time, rng)
        a1 = normalize_sum(a1, n)
        c1 = fft_convolve_self(a1)
        obj1 = 2.0 * n * c1.max() / (a1.sum() ** 2)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
        print("  Phase 1 best: %.6f" % global_best)
    else:
        global_a = np.ones(n) * S / n

    # Phase 2: LP best-response iteration (45% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 60:
        phase2_time = remaining * 0.45
        print("=== Phase 2: LP best-response (remaining %.0fs) ===" % phase2_time)
        a2, s2, nlp = run_lp_phase(global_a, n, S, phase2_time, rng, lp_time=15.0)
        a2 = normalize_sum(a2, n)
        c2 = fft_convolve_self(a2)
        obj2 = 2.0 * n * c2.max() / (a2.sum() ** 2)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            print("  Phase 2 best: %.6f (LP calls: %d)" % (global_best, nlp))
        else:
            print("  Phase 2: no improvement (best=%.6f, phase2=%.6f, LPs=%d)" %
                  (global_best, obj2, nlp))

    # Phase 3: Enhanced FISTA polishing (25% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        phase3_time = remaining * 0.25
        print("=== Phase 3: Enhanced FISTA polish (remaining %.0fs) ===" % phase3_time)
        a3, s3 = run_enhanced_fista(global_a, n, S, phase3_time, rng)
        a3 = normalize_sum(a3, n)
        c3 = fft_convolve_self(a3)
        obj3 = 2.0 * n * c3.max() / (a3.sum() ** 2)
        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()
            print("  Phase 3 best: %.6f" % global_best)
        else:
            print("  Phase 3: no improvement (best=%.6f, phase3=%.6f)" % (global_best, obj3))

    # Phase 4: Try a slightly different n if time permits
    remaining = total_deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            phase4_time = remaining * 0.35
            print("=== Phase 4: Trying n=%d (remaining %.0fs) ===" % (alt_n, phase4_time))
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            a4, s4 = run_enhanced_fista(alt_init, alt_n, S_alt, phase4_time, rng)
            a4 = normalize_sum(a4, alt_n)
            c4 = fft_convolve_self(a4)
            obj4 = 2.0 * alt_n * c4.max() / (a4.sum() ** 2)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                print("  Phase 4 (n=%d) best: %.6f" % (alt_n, global_best))
            else:
                print("  Phase 4 (n=%d): %.6f (no improvement vs %.6f)" %
                      (alt_n, obj4, global_best))

    # Phase 5: Final intensive LP + FISTA combo
    remaining = total_deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        phase5_time = remaining * 0.5
        print("=== Phase 5: Final LP+FISTA (remaining %.0fs, n=%d) ===" % (phase5_time, n_p))
        a5, s5, nlp5 = run_lp_phase(global_a, n_p, S_p, phase5_time, rng, lp_time=10.0)
        a5 = normalize_sum(a5, n_p)
        c5 = fft_convolve_self(a5)
        obj5 = 2.0 * n_p * c5.max() / (a5.sum() ** 2)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("  Phase 5 best: %.6f (LP calls: %d)" % (global_best, nlp5))

    # Phase 6: Very final micro-polish
    remaining = total_deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        a_f, s_f = run_enhanced_fista(global_a, n_f, S_f, max(5.0, remaining * 0.5), rng)
        a_f = normalize_sum(a_f, n_f)
        c_f = fft_convolve_self(a_f)
        obj_f = 2.0 * n_f * c_f.max() / (a_f.sum() ** 2)
        if obj_f < global_best:
            global_best = obj_f
            global_a = a_f.copy()
            print("  Final micro-polish: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
        else:
            print("Final verification: INFINITE, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return final_list
