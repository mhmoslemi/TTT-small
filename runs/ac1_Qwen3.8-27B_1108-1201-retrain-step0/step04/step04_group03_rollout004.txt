"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses adaptive FISTA with LSE smoothing, multiplicative repulsion updates,
and periodic LP best-response corrections. Multi-length exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Normalize array to have sum S, clip to [0, 1000]."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, c, beta):
    """Gradient of beta-LSE of c, back-propagated through convolution."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Backprop: grad_a[i] = 2 * sum_k w[k] * c_k_grad(i)
    # c_k = sum_j a[j]*a[k-j], dc_k/da[i] = 2*a[k-i] (if valid)
    # So grad_a[i] = 2 * sum_k w[k] * a[k-i] (if k-i in [0,n))
    # = 2 * (w * a_reversed)[i] (circularly, truncated)
    # More precisely: grad_a[i] = 2 * sum_{k: k-i in [0,n)} w[k] * a[k-i]
    # Let j = k-i, so k = i+j: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    # This is (w padded) convolved with a, evaluated at position... 
    # Actually it's the cross-correlation of w with a.
    # Let's use FFT: if W = fft(w, 2n), A = fft(a, 2n), then
    # crosscorr[i] = sum_j w[i+j] * a[j] = ifft(W * conj(A))[i+n]... 
    # Let me just do it directly with FFT.
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # sum_j w[i+j] * a[j] for i in [0, n)
    # = cross-correlation of w and a at lag i
    # In FFT: ifft(conj(W) * A) gives cross-correlation
    # Actually: crosscorr[i] = sum_j w[i+j] * a[j]
    # Let's define: (w ⊛ a)[i] = sum_j w[i+j] * a[j]
    # This is the correlation, which in FFT is: ifft(conj(FFT(w)) * FFT(a))
    # But we need to be careful with the convention.
    # Standard: (f * g)[n] = sum_k f[k] * g[n-k] (convolution)
    # (f ⊛ g)[n] = sum_k f[k] * g[n+k] (correlation) = ifft(conj(F) * G)
    # We want: sum_j w[i+j] * a[j]
    # Let's substitute m = i+j, so j = m-i: sum_m w[m] * a[m-i]
    # = (w * a_reversed)[i + something]... 
    # Actually: sum_j w[i+j] * a[j] = sum_j w[m] * a[m-i] where m = i+j
    # = correlation of w and a at lag i
    # = ifft(conj(FFT(w, fsize)) * FFT(a, fsize))[i]
    # Let me verify: ifft(conj(W) * A)[i] = (1/fsize) * sum_k conj(W[k]) * A[k] * exp(2*pi*i*k/fsize)
    # This equals the circular correlation... let me just use the direct approach.
    
    # Alternative: use the fact that grad_a = 2 * ifft(conj(W) * A)[:n]
    # where W = fft(w, fsize) and A = fft(a, fsize)
    result = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * result


def run_fista_adaptive(a_init, n, S, max_time, rng, 
                       beta_init=5.0, beta_max=5000.0, beta_growth=1.005,
                       restart_patience=500):
    """FISTA with adaptive beta and restarts."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    beta = beta_init
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    lse_vals = []

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_lse = c.max() + np.log(np.exp(beta * (c - c.max())).sum()) / beta
        
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: 1/L where L is Lipschitz constant of gradient
        # For LSE with parameter beta, L ≈ 2*beta*max(a) (rough bound)
        step = 1.0 / (2.0 * beta * max(y.max(), 1e-10) + 1e-10)
        # Actually, let's use a simpler step: 1/(2*beta)
        step = 1.0 / (2.0 * beta)
        
        # Adaptive restart
        if it - last_improve > restart_patience:
            # Add noise and reset momentum
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
            # Don't reset beta, keep increasing
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Increase beta
        beta = min(beta_max, beta * beta_growth)

    return best_a, best


def multiplicative_update(f, n, S, beta=30.0, lr=0.3):
    """Multiplicative repulsion update (EM-like)."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    
    # Compute "pressure" at each position: sum_k w[k] * f[k-i] (if valid)
    # = correlation of w and f
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    F = fft(f, fsize)
    pressure = np.real(ifft(np.conj(W) * F))[:n]
    
    # Multiplicative update: f_new[i] = f[i] * (S/n) / pressure[i]
    # But this might be too aggressive. Use lr to control.
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = (S / n) / np.maximum(pressure, 1e-15)
    ratio = np.clip(ratio, 0.5, 2.0)  # Limit the change
    f_new = f * (1 - lr + lr * ratio)
    f_new = np.maximum(f_new, 1e-10)
    f_new = normalize_to_sum(f_new, S)
    return f_new


def solve_lp_small(f, n, S, top_k=50, solver='ECOS', time_limit=5.0):
    """Solve LP best response with reduced constraints."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Top k lags
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return None
    
    if time.monotonic() - start > time_limit * 2:
        return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def lp_step(f, n, S, top_k=50):
    """One LP-based step with bisection for step size."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    
    g = solve_lp_small(f, n, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old
    
    # Bisection for step size
    def eval_t(t):
        f_t = (1 - t) * f + t * g
        f_t = np.maximum(f_t, 0)
        f_t = normalize_to_sum(f_t, S)
        return fft_convolve_self(f_t).max()
    
    best_t = 0.0
    best_val = M_old
    lo, hi = 0.0, 1.0
    for _ in range(6):
        mid = (lo + hi) / 2
        val = eval_t(mid)
        if val < best_val:
            best_val = val
            best_t = mid
            lo = mid
        else:
            hi = mid
    
    if best_t > 0.05:
        f_new = (1 - best_t) * f + best_t * g
        f_new = normalize_to_sum(f_new, S)
        return f_new, best_val
    return f.copy(), M_old


def topk_lse_grad(a, c, beta, top_k=20):
    """Gradient focusing on top-k peaks only (sharper than full LSE)."""
    n = len(a)
    # Take top-k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    cm = c_top.max()
    w_top = np.exp(beta * (c_top - cm))
    w_top /= w_top.sum()
    
    # Zero out non-top peaks in the weight vector
    w = np.zeros(len(c))
    w[top_idx] = w_top
    
    # Same backprop as before
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * result


def run_peak_focused(a_init, n, S, max_time, rng, 
                     beta_init=20.0, beta_max=10000.0, beta_growth=1.01,
                     top_k=10, restart_patience=300):
    """FISTA focusing on top-k peaks only."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    beta = beta_init
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = topk_lse_grad(y, c, beta, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        
        if it - last_improve > restart_patience:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        beta = min(beta_max, beta * beta_growth)

    return best_a, best


def run_multiplicative(f_init, n, S, max_time, beta=30.0, lr=0.3, 
                       refit_every=100):
    """Pure multiplicative update iteration."""
    f = normalize_to_sum(f_init, S)
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        f = multiplicative_update(f, n, S, beta=beta, lr=lr)
        
        if it % refit_every == 0 or it == 1:
            c = fft_convolve_self(f)
            if c.max() < best:
                best = c.max()
                best_f = f.copy()
        
        # Adaptive beta increase
        if it % 500 == 0:
            beta *= 1.2
            lr *= 0.9

    c = fft_convolve_self(f)
    if c.max() < best:
        best = c.max()
        best_f = f.copy()

    return best_f, best


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine variants
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * tt))
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * 3 * tt) + 0.1 * np.cos(2 * np.pi * 7 * tt))
    
    # Gaussian
    inits.append(np.exp(-5 * (tt - 0.5) ** 2 / 0.1))
    
    # Power of sine
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    
    # Random
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    
    # Block structure
    for nb in [7, 11, 19]:
        bl = n // nb
        heights = rng.uniform(0.5, 2.0, nb)
        seq = np.repeat(heights, bl)
        if len(seq) < n:
            seq = np.concatenate([seq, np.full(n - len(seq), heights[-1])])
        inits.append(seq[:n])
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.8
    inits.append(alt)
    
    return [np.maximum(x, 0.01) for x in inits]


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def evaluate_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S ** 2)


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)
    
    global_best = np.inf
    global_a = None

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Determine main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192
    
    print("Main length: %d, budget: %.0fs" % (main_len, budget_s))
    
    # Phase 1: Quick FISTA on given sequence
    n = main_len
    S = np.sqrt(2.0 * n)
    remaining = deadline - time.monotonic()
    
    if given is not None and remaining > 120:
        print("=== Phase 1: Peak-focused FISTA (n=%d, %.0fs) ===" % (n, remaining * 0.25))
        t1 = time.monotonic()
        a1, s1 = run_peak_focused(given, n, S, remaining * 0.25, rng,
                                   beta_init=30.0, beta_max=20000.0, beta_growth=1.02,
                                   top_k=10, restart_patience=400)
        a1 = normalize_to_sum(a1, S)
        obj1 = evaluate_obj(a1)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
        print("  Phase 1: obj=%.6f (time=%.1fs)" % (obj1, time.monotonic() - t1))
    
    # Phase 2: Multiplicative updates
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print("=== Phase 2: Multiplicative updates (remaining %.0fs) ===" % remaining)
        t2 = time.monotonic()
        a2, s2 = run_multiplicative(global_a, n, S, remaining * 0.15, beta=20.0, lr=0.2)

        if obj2 < global_best:
            global_best = obj2

        print("  Phase 2: obj=%.6f (time=%.1fs)" % (obj2, time.monotonic() - t2))
    
    # Phase 3: LP-based refinement
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print("=== Phase 3: LP refinement (remaining %.0fs) ===" % remaining)
        t3 = time.monotonic()
        a3 = global_a.copy()
        S3 = np.sqrt(2.0 * n)
        while time.monotonic() < deadline - 60:
            a3, m3 = lp_step(a3, n, S3, top_k=40)
            obj3 = evaluate_obj(a3)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()
                print("  LP: obj=%.6f" % obj3)
            else:
                # No improvement, try a different top_k
                a3, m3 = lp_step(a3, n, S3, top_k=20)
                obj3b = evaluate_obj(a3)
                if obj3b < global_best:
                    global_best = obj3b
                    global_a = a3.copy()
                    print("  LP (top20): obj=%.6f" % obj3b)
                else:
                    break
        print("  Phase 3 done (time=%.1fs)" % (time.monotonic() - t3))
    
    # Phase 4: Try longer sequence
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2
        if alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 4: Longer sequence n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            t4 = time.monotonic()
            if global_a is not None:
                alt_init = interpolate_seq(global_a, alt_n)
            else:
                alt_init = np.ones(alt_n) * 0.01
            a4, s4 = run_peak_focused(alt_init, alt_n, S_alt, remaining * 0.4, rng,
                                        beta_init=20.0, beta_max=10000.0, beta_growth=1.015,
                                        top_k=10, restart_patience=300)

    if remaining > 60 and global_a is not None:

                global_a = ai.copy()
                print("  Divers
