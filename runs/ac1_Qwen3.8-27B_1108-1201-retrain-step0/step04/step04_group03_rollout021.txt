"""
Minimize 2*n*max(f*f)/sum(f)^2 using iterative LP best-response with full constraints,
exact line search, and multi-scale exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def line_search(f, g, S, n):
    """Find best t in [0,1] minimizing max(((1-t)f + t*g) * ((1-t)f + t*g))"""
    def obj(t):
        x = (1 - t) * f + t * g
        x = np.maximum(x, 0)
        x = project_simplex(x, S)
        c = fft_convolve_self(x)
        return c.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2

```python
def solve_lp_cutting_plane(f, n, S, solver='SCS', max_iters=50, time_limit=60.0):
    """Solve LP best-response using cutting plane method."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Start with top-K constraints
    K = min(n, 512)
    active = np.argsort(c)[::-1][:K]
    
    for iter in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Build constraint matrix for active set
        rows = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]
        
        g_var = cp.Variable(n)
        constraints = [rows @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if g_var.value is None:
            return None
        
        g = np.maximum(g_var.value, 0)
        
        # Check all constraints
        fg = fft_convolve(f, g)  # cross-convolution
        violations = fg - M
        if np.max(violations) < 1e-6 * M:
            break  # All constraints satisfied
        
        # Add violated constraints
        violated = np.where(violations > 1e-8 * M)[0]
        active = np.concatenate([active, violated])
        active = np.unique(active)
        # Keep only top constraints by violation
        if len(active) > 2 * K:
            # Prioritize most violated
            viol_scores = np.zeros(len(active))
            for i, k in enumerate(active):
                viol_scores[i] = max(0, fg[k] - M)
            order = np.argsort(viol_scores)[::-1][:K]
            active = active[order]
        
        if len(active) == 2 * n - 1:
            break  # All constraints active
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    
    return None
```

"""
Minimize 2*n*max(f*f)/sum(f)^2 using iterative LP best-response with cutting-plane
constraint generation, golden-section line search, and multi-phase optimization.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(f, n):
    """Compute the objective value."""
    c = fft_convolve_self(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def golden_section_search(f, g, S, n, max_evals=25):
    """
    Golden section search for best t in [0,1] minimizing max(((1-t)f + t*g) * ((1-t)f + t*g)).
    Returns (best_f, best_M, best_t).
    """
    def eval_t(t):
        x = (1.0 - t) * f + t * g
        x = np.maximum(x, 0.0)
        x = project_simplex(x, S)
        c = fft_convolve_self(x)
        return c.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = eval_t(c_pt)
    fd = eval_t(d_pt)
    n_evals = 2

    for _ in range(max_evals - 2):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = eval_t(c_pt)
            n_evals += 1
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = eval_t(d_pt)
            n_evals += 1

    t_best = (a + b) / 2.0
    M_best = eval_t(t_best)
    x_best = (1.0 - t_best) * f + t_best * g
    x_best = np.maximum(x_best, 0.0)
    x_best = project_simplex(x_best, S)
    return x_best, M_best, t_best


def solve_lp_best_response(f, n, S, solver='SCS', time_limit=30.0, max_cp_iters=30):
    """
    Solve LP best-response using cutting-plane method.
    max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()
    if M < 1e-15:
        return f.copy()

    # Initial active set: top-K lags
    K0 = min(n, 256)
    active = np.argsort(c)[::-1][:K0]

    g = None
    start_time = time.monotonic()

    for cp_iter in range(max_cp_iters):
        if time.monotonic() - start_time > time_limit:
            break

        # Build constraint matrix for current active set
        n_active = len(active)
        rows = np.zeros((n_active, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]

        g_var = cp.Variable(n)
        constraints = [rows @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            try:
                prob.solve(solver='CLARABEL', verbose=False)
            except Exception:
                try:
                    prob.solve(solver='ECOS', verbose=False)
                except Exception:
                    break

        if g_var.value is None:
            break

        g = np.maximum(g_var.value, 0)

        # Check all constraints
        fg = fft_convolve(f, g)
        violations = fg - M
        max_viol = np.max(violations) if len(violations) > 0 else 0

        if max_viol < 1e-4 * M:
            break  # Constraints satisfied to tolerance

        # Add violated constraints
        tol = max(1e-6 * M, 1e-10)
        violated = np.where(violations > tol)[0]
        if len(violated) == 0:
            break

        # Merge into active set, keeping at most 2*K0 constraints
        active = np.unique(np.concatenate([active, violated]))
        if len(active) > 2 * K0:
            # Keep the most violated ones
            viol_scores = np.array([max(0, fg[k] - M) for k in active])
            order = np.argsort(viol_scores)[::-1][:2 * K0]
            active = active[order]

        if len(active) >= 2 * n - 1:
            break

    if g is None:
        return None

    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g = g * (S / g_sum)
    return g


def multiplicative_update(f, n, S, beta=30.0, lr=0.1):
    """Multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 0)
    f_new = project_simplex(f_new, S)
    return f_new

    Main optimization loop using LP best-response + line search.
    """

    best_f = f.copy()

    last_improve = 0
    lp_success = 0

            lp_success += 1
            # Line search

            if M_new < M * (1.0 - 1e-6):

            elif M_new <= M:
                f = f_new
            else:

                    f_try = (1 - frac) * f + frac * g
                    f_try = np.maximum(f_try, 0)

        else:
            lp_fail += 1
            # Fallback: multiplicative update

            last_improve = it  # Reset counter

"""
Minimize 2*n*max(f*f)/sum(f)^2 using iterative LP best-response with cutting-plane
constraint generation and golden-section line search.
Multi-phase approach: exploration with diverse inits, exploitation with LP, polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_obj(f, n):
    c = fft_convolve_self(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def line_search(f, g, S, n, coarse_points=10, fine_evals=15):
    """Two-stage line search: coarse grid then golden section."""
    def eval_t(t):
        x = (1.0 - t) * f + t * g
        x = np.maximum(x, 0.0)
        x = project_simplex(x, S)
        return fft_convolve_self(x).max()

    # Coarse grid
    ts = np.linspace(0, 1, coarse_points)
    vals = np.array([eval_t(t) for t in ts])
    best_idx = np.argmin(vals)
    
    # Fine search around best
    if best_idx == 0:
        a, b = 0.0, ts[1]
    elif best_idx == len(ts) - 1:
        a, b = ts[-2], 1.0
    else:
        a, b = ts[best_idx - 1], ts[best_idx + 1]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = eval_t(c_pt)
    fd = eval_t(d_pt)
    
    for _ in range(fine_evals):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = eval_t(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = eval_t(d_pt)
    
    t_best = (a + b) / 2.0
    M_best = eval_t(t_best)
    x_best = (1.0 - t_best) * f + t_best * g
    x_best = np.maximum(x_best, 0.0)
    x_best = project_simplex(x_best, S)
    return x_best, M_best, t_best


def solve_lp(f, n, S, time_limit=25.0, init_active=None, solver='SCS'):
    """Solve LP best-response with cutting plane."""
    c = fft_convolve_self(f)
    M = c.max()
    if M < 1e-15:
        return f.copy()

    if init_active is None:
        K0 = min(n, 512)
        active = np.argsort(c)[::-1][:K0]
    else:
        active = init_active.copy()

    g = None
    start = time.monotonic()

    for _ in range(40):
        if time.monotonic() - start > time_limit:
            break

        n_act = len(active)
        rows = np.zeros((n_act, n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]

        g_var = cp.Variable(n)
        cons = [rows @ g_var <= M, g_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), cons)

        solved = False
        for s in [solver, 'CLARABEL', 'ECOS', 'SCS']:
            try:
                prob.solve(solver=s, verbose=False)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            break

        g = np.maximum(g_var.value, 0)

        # Check violations
        fg = fft_cross_convolve(f, g)
        viol = fg - M
        if np.max(viol) < 1e-5 * M + 1e-10:
            break

        # Add violated constraints
        tol = 1e-5 * M + 1e-10
        violated = np.where(viol > tol)[0]
        active = np.unique(np.concatenate([active, violated]))
        
        # Cap active set size
        Kmax = min(2 * n, 4096)
        if len(active) > Kmax:
            scores = np.array([max(0.0, fg[k] - M) for k in active])
            # Also score by original c[k] (importance)
            c_scores = np.array([c[k] for k in active])
            combined = 0.5 * scores / max(scores.max(), 1e-10) + 0.5 * c_scores / max(c_scores.max(), 1e-10)
            order = np.argsort(combined)[::-1][:Kmax]
            active = active[order]

        if len(active) >= 2 * n - 1:
            break

    if g is None:
        return None

    gs = g.sum()
    if gs < 1e-10:
        return None
    return g * (S / gs)


def subgradient_step(f, n, S, lr=1.0):
    """One subgradient step on the objective."""
    c = fft_convolve_self(f)
    k_star = np.argmax(c)
    M = c[k_star]
    
    # Subgradient of max(c) at f
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    subgrad = np.zeros(n)
    js = np.arange(lo, hi + 1)
    subgrad[js] = f[k_star - js]
    subgrad *= 2.0  # Because of the self-convolution
    
    # Full gradient of F = 2n*M/S^2
    S = f.sum()
    grad = (2.0 * n / S**2) * subgrad - (4.0 * n * M / S**3) * np.ones(n)
    
    # Projected gradient step
    f_new = f - lr * grad
    f_new = project_simplex(f_new, S)
    return f_new
