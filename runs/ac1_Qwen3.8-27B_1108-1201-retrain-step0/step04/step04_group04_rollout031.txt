"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: conv(a,a)[k] = sum_j a_j * a_{k-j} (linear convolution, NOT autocorrelation).
With normalization sum(a)=sqrt(2n), objective = max(conv(a,a)).
Strategy: Combined spectral flattening + LSE proximal gradient + peak-aware coordinate descent.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient_conv(a, beta):
    """
    Compute gradient of LSE approximation to max(conv(a,a)).
    Returns (gradient, max_value).
    grad f = (2/beta) * cross_correlation(a, w) where w = softmax(beta * conv(a,a)).
    """
    n = len(a)
    c = fft_conv_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot

    # Compute c_m = sum_l a_l * w_{m+l} for m=0..n-1
    # This is the cross-correlation of a and w.
    # cross_correlation(a, w)[m] = sum_l a_l * w_{m+l}
    # Using FFT: IFFT(FFT(a) * conj(FFT(w)))
    fsize = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    corr = np.real(ifft(A * np.conj(W)))
    grad = (2.0 / beta) * corr[:n]
    return grad, cm


def run_lse_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=3000.0,
                  beta_ramp=2000, restart_every=5000):
    """FISTA with LSE smoothing for minimizing max(conv(a,a))."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = fft_conv_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp))
        else:
            beta = beta_max

        c = fft_conv_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_gradient_conv(y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 1.0 / (2.0 * beta * max(1.0, np.max(np.abs(y))))
        if it - last_improve > 2000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain feasibility
        s = y.sum()
        if s < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * (S / s)
        a = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % 2000 == 0 and it < restart_every * 3:
            decay = np.exp(-it / (restart_every * 2.0))
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng, alpha_start=0.3, alpha_end=0.9):
    """Gerchberg-Saxton style: flatten power spectrum to reduce convolution peak."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = fft_conv_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 3000.0)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Push spectrum toward flat
        P_flat = np.ones(fsize)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_flat
        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_conv_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        a = a_new

    return best_a, best


def coordinate_descent_peak(a_init, n, S, max_time, rng):
    """Coordinate descent that specifically targets the peak lag."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = fft_conv_self(a).max()
    best_a = a.copy()
    it = 0

    c = fft_conv_self(a)
    peak_lag = int(np.argmax(c))

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Focus on top few lags
        top_lags = np.argsort(c)[::-1][:5]

        for k in top_lags:
            if time.monotonic() - start > max_time - 1:
                    break
            # Try to reduce c[k] by moving mass
            # c[k] = sum_j a_j * a_{k-j}
            # Transfer from position i to position j reduces c[k] by:
            # a_{k-i}*delta + a_{i}*a_{k-j} - a_{k-j}*delta - a_{j}*a_{k-i}
            # = delta*(a_{k-i} - a_{k-j}) + a_i*(a_{k-j} - a_{k-i})
            
            n_try = 30
            for _ in range(n_try):
                # Choose source: position where a_{k-src} is large
                valid_src = np.arange(max(0, k - n + 1), min(n, k + 1))
                if len(valid_src) < 2:
                    continue
                # Weight by a_{k-src} to find "expensive" positions
                weights_src = np.array([a[k - s] if 0 <= k - s < n else 0 for s in valid_src])
                weights_src = np.maximum(weights_src * a[valid_src], 1e-15)
                weights_src /= weights_src.sum()
                src = valid_src[rng.choice(len(valid_src), p=weights_src)]

                # Choose dest: position where a_{k-dest} is small
                valid_dst = np.arange(max(0, k - n + 1), min(n, k + 1))
                valid_dst = valid_dst[valid_dst != src]
                if len(valid_dst) < 1:
                    continue
                weights_dst = np.array([1.0 / (a[k - d] + 1e-15) if 0 <= k - d < n else 1.0 for d in valid_dst])
                weights_dst = np.maximum(a[valid_dst], 1e-15)  # can only move from positions with mass
                # Actually, transfer FROM src TO dest reduces c[k] if a_{k-dest} < a_{k-src}
                weights_dst = (a[src] - np.array([a[k - d] if 0 <= k - d < n else 0 for d in valid_dst]))
                weights_dst = np.maximum(weights_dst, 0)
                total = weights_dst.sum()
                if total < 1e-15:
                    continue
                weights_dst /= total
                dst = valid_dst[rng.choice(len(valid_dst), p=weights_dst)]

                delta = min(a[src], 0.1 * S / n)
                if delta < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta

                if a_trial[src] < 0:
                    continue

                # Quick evaluation: just update the affected lags
                c_trial = c.copy()
                for idx in range(max(0, src - n + 1), min(2 * n - 1, src + n)):
                    if idx != k:
                        pass  # skip for speed, just check peak
                # Full recompute for correctness (only if close)
                c_trial = fft_conv_self(a_trial)
                cur_trial = c_trial.max()

                if cur_trial < best - 1e-12:
                    a = a_trial
                    c = c_trial
                    best = cur_trial
                    best_a = a.copy()
                    improved = True
                    peak_lag = int(np.argmax(c))
                    break

        if not improved and it > 3:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-it / 200.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_conv_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            peak_lag = int(np.argmax(c))

    return best_a, best

    tt = np.linspace(0, 1, n, endpoint=False)

    # Cosine window
    a = np.maximum(np.cos(np.pi * tt) ** 2 + 0.01, 0)
    inits.append(normalize_sum(a, n))

    # Gaussian centered
    a = np.exp(-20.0 * (tt - 0.5) ** 2)
    inits.append(normalize_sum(a, n))

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * tt))
    a = np.maximum(a, 0.01)
    inits.append(normalize_sum(a, n))

    # Flat-top window
    a = 1.0 - 0.42 * np.cos(2 * np.pi * tt) + 0.28 * np.cos(4 * np.pi * tt) - 0.08 * np.cos(6 * np.pi * tt)
    a = np.maximum(a, 0.01)
    inits.append(normalize_sum(a, n))

    # Sinc-like
    a = np.sinc(tt - 0.5) ** 2 + 0.05
    a = np.maximum(a, 0)
    inits.append(normalize_sum(a, n))

    # Random
    a = rng.random(n) + 0.1
    inits.append(normalize_sum(a, n))

    # Two-peak
    a = np.exp(-50 * (tt - 0.3) ** 2) + np.exp(-50 * (tt - 0.7) ** 2)
    inits.append(normalize_sum(a, n))

    # Uniform (baseline)
    inits.append(np.ones(n) * S / n)

    return inits


def propose_candidate(seed=42, budget_s=2100):
    """Search for a non-negative sequence minimizing 2*n*max(conv)/sum^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 512 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, budget=%.0fs, parent=%s" % (n, budget_s, "yes" if parent is not None else "no"))

    global_best = np.inf
    global_a = None

    def eval_score(a_candidate):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_conv_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  NEW BEST: score=%.6f (elapsed %.0fs)" % (score, elapsed))
        return score

    # Build starting points
    starts = []
    if parent is not None:
        starts.append(("parent", normalize_sum(parent, n)))
        # Perturbed parent
        for scale in [0.05, 0.15]:
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            v = project_simplex(parent + noise, S)
            starts.append(("parent_noise_%d" % int(scale * 100), v))
        # Spectral variant of parent
        fsize = 1 << (2 * n - 1).bit_length()
        F = fft(parent, fsize)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        spect = mag ** 2
        spect = np.power(np.maximum(spect, 1e-30), 0.5)
        spect = spect / spect.mean()
        mag_new = np.sqrt(spect)
        F_new = phase * mag_new
        a_spec = np.real(ifft(F_new))[:n]
        a_spec = np.maximum(a_spec, 0)
        s = a_spec.sum()
        if s > 1e-15:
            starts.append(("parent_spectral", a_spec * (S / s)))

    # Add structured inits
    for i, (name, arr) in enumerate(make_initializations(n, S, rng)):
        starts.append(("init_%d_%s" % (i, name.split()[0]), arr))

    print("Total starting points: %d" % len(starts))

    # Phase 1: Quick spectral flattening on all starts
    print("\n=== Phase 1: Spectral Flattening ===")
    p1_total = (deadline - time.monotonic()) * 0.25
    t_each = p1_total / len(starts)
    for name, arr in starts:
        if time.monotonic() > deadline - (deadline - t_start) * 0.7:
            break
        if t_each < 2:
            continue
        a_opt, score = spectral_flatten(arr, n, S, max_time=t_each, rng=rng)
        eval_score(a_opt)

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: LSE FISTA on top candidates
    print("\n=== Phase 2: LSE FISTA ===")
    p2_total = (deadline - time.monotonic()) * 0.35
    # Run on best from phase 1 and a couple others
    candidates = []
    if global_a is not None:
        candidates.append(("best_p1", global_a))
    candidates.append(("sinc", starts[4][1] if len(starts) > 4 else np.ones(n) * S / n))
    candidates.append(("cosine", starts[0][1]))

    t_each = p2_total / len(candidates)
    for name, arr in candidates:
        if time.monotonic() > deadline - (deadline - t_start) * 0.3:
            break
        a_opt, score = run_lse_fista(arr, n, S, max_time=t_each, rng=rng,
                                      beta_init=5.0, beta_max=2000.0, beta_ramp=1500)
        eval_score(a_opt)

    print("After Phase 2: best=%.6f" % global_best)

    # Phase 3: Coordinate descent polish
    print("\n=== Phase 3: Coordinate Descent ===")
    if global_a is not None:
        p3_total = (deadline - time.monotonic()) * 0.20
        a_opt, score = coordinate_descent_peak(global_a, n, S, max_time=p3_total, rng=rng)
        eval_score(a_opt)

    print("After Phase 3: best=%.6f" % global_best)

    # Phase 4: High-beta LSE polish
    print("\n=== Phase 4: High-Beta Polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_opt, score = run_lse_fista(global_a, n, S, max_time=remaining * 0.6, rng=rng,
                                          beta_init=100.0, beta_max=5000.0, beta_ramp=500)
            eval_score(a_opt)

            # Another spectral pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_opt2, score2 = spectral_flatten(global_a, n, S, max_time=remaining2 * 0.7, rng=rng,
                                                   alpha_start=0.5, alpha_end=0.95)
                eval_score(a_opt2)

    # Phase 5: Try slightly different lengths if time permits
    remaining = deadline - time.monotonic()
    if remaining > 180 and global_a is not None:
        print("\n=== Phase 5: Alternate Length ===")
        for alt_n in [n * 2, n // 2]:
            if remaining < 60:
                break
            if 1024 <= alt_n <= 65536:
                S_alt = np.sqrt(2.0 * alt_n)
                if alt_n > n:
                    t_old = np.linspace(0, 1, n)
                    t_new = np.linspace(0, 1, alt_n)
                    alt_start = np.interp(t_new, t_old, global_a)
                else:
                    nb = n // alt_n
                    alt_start = global_a[:nb * alt_n].reshape(alt_n, nb).mean(axis=1)
                alt_start = normalize_sum(alt_start, alt_n)
                t_alloc = remaining * 0.4
                a_alt, score = run_lse_fista(alt_start, alt_n, S_alt, max_time=t_alloc, rng=rng,
                                              beta_init=10.0, beta_max=1000.0)
                eval_score(a_alt)

    # Final verification
    print("\n=== Final ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, score=%.6f, time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
