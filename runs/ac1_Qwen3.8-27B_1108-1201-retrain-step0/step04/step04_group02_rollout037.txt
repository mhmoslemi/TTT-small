Minimize 2*n*max_conv/sum^2 using smooth-surrogate FISTA with adaptive temperature,
backtracking step size, diverse multi-start, and LP-inspired polishing.
Key insight: use smooth max (log-sum-exp) with gradually increasing temperature

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on Lagrange multiplier."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.max(idx * w - cumsum + S > 0)
    if rho == 0:
        return np.zeros(n)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_weights(c, beta):
    """Compute softmax(beta * c) safely."""
    bc = beta * (c - c.max())
    e = np.exp(bc)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / s


def smooth_max_gradient(a, w):
    """Compute gradient of smooth_max(conv(a)) given softmax weights w.
    grad[j] = 2 * sum_k w[k] * a[k-j] for valid k.
    This is 2 * cross_correlation(w, a)[j].
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # cross-correlation: ifft(fft(w) * conj(fft(a)))
    xc = np.real(ifft(W * np.conj(A)))[:size]
    # We need: for each j in 0..n-1, sum_{k=j}^{j+n-1} w[k]*a[k-j]

    # The full cross-correlation xc[j] = sum_k w[k]*a[k-j] already handles this

    if sum_a < 0.01:

    return float(2.0 * n * max_b / (sum_a ** 2))

        while time.monotonic() < phase_deadline:
            it += 1

            # FISTA momentum update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = a + ((t_mom - 1.0) / t_new) * (a - y + 0.0)  # Standard FISTA
            # Actually standard: y = x_new + ((t_old - 1)/t_new) * (x_new - x_old)
            # Let me redo properly

            # Compute gradient at y
            c_y = fft_convolve_self(y)
            w_y = softmax_weights(c_y, beta)
            grad_y = smooth_max_gradient(y, w_y)
            gnorm = np.linalg.norm(grad_y)
            if gnorm < 1e-15:
                break

            # Backtracking step size
            max_iter_bt = 30
            for _ in range(max_iter_bt):
                x_try = project_simplex(y - step * grad_y, S)
                c_try = fft_convolve_self(x_try)
                w_try = softmax_weights(c_try, beta)
                obj_try_smooth = np.log(np.sum(np.exp(beta * (c_try - c_try.max())))) / beta
                obj_y_smooth = np.log(np.sum(np.exp(beta * (c_y - c_y.max())))) / beta

                # Descent condition (strong Wolfe-like)
                directional = np.dot(grad_y, x_try - y)
                if obj_try_smooth <= obj_y_smooth + 0.01 * directional + 1e-12 * np.linalg.norm(x_try - y) ** 2:
                    break
                step *= 0.5

            x_new = project_simplex(y - step * grad_y, S)

            # Check true objective periodically
            if it % 100 == 0:
                true_obj = true_objective_value(x_new)
                if true_obj < best_obj - 1e-9:
                    best_obj = true_obj
                    best_a = x_new.copy()
                    stall_count = 0
                elif true_obj < best_phase_obj - 1e-9:
                    best_phase_obj = true_obj
                    best_phase_a = x_new.copy()
                else:
                    stall_count += 1
                    if stall_count > 5:
                        # Reset momentum
                        t_mom = 1.0
                        stall_count = 0

            # FISTA update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            # Occasionally increase step if making progress
            if it % 200 == 0 and best_phase_obj < obj_prev - 1e-6:
                step = min(step * 1.3, 1.0 / (beta * 10.0))

        # After phase, update a to best found in this phase
        if best_phase_obj < best_obj:
            a = best_phase_a.copy()
            best_obj = best_phase_obj
            best_a = a.copy()
        else:
            a = best_phase_a.copy()

    return best_a, best_obj


def run_gradient_descent(a_init, n, S, max_time, rng):
    """
    Plain gradient descent with momentum and backtracking on smooth surrogate.
    More robust than FISTA for non-convex problems.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start_time = time.monotonic()
    best_obj = true_objective_value(a)
    best_a = a.copy()

    beta = 30.0
    momentum = 0.9
    vel = np.zeros(n)
    step = 1.0 / (beta * S * 0.05)
    it = 0
    stall = 0
    last_best = best_obj

    while time.monotonic() - start_time < max_time:
        it += 1

        c = fft_convolve_self(a)
        w = softmax_weights(c, beta)
        grad = smooth_max_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Normalize gradient for stable step
        grad_normed = grad / gnorm
        step = min(step * 1.5, S / gnorm * 0.1)

        # Momentum update
        vel = momentum * vel - step * grad_normed * gnorm * 0.01
        a_try = project_simplex(a + vel, S)

        # Backtracking
        obj_cur = 2.0 * n * c.max() / S ** 2
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S ** 2

        retries = 0
        while obj_try >= obj_cur and retries < 20:
            step *= 0.5
            vel = momentum * vel - step * grad_normed * gnorm * 0.01
            a_try = project_simplex(a + vel, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2
            retries += 1

        a = a_try
        c = c_try

        # Check true objective
        if it % 200 == 1:
            true_obj = true_objective_value(a)
            if true_obj < best_obj - 1e-9:
                best_obj = true_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 3:
                    # Perturb to escape
                    noise_scale = 0.005 * S / np.sqrt(n) * (1.0 + 0.5 * (stall / 10.0))
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    vel = np.zeros(n)
                    step = 1.0 / (beta * S * 0.05)
                    stall = 0

        # Gradually increase beta
        if it % 3000 == 0 and beta < 300:
            beta *= 1.5

        # Decay momentum slightly
        momentum = max(0.85, momentum * 0.9999)

    return best_a, best_obj


def run_lp_polish(a, n, S, max_time, rng):
    """
    LP-inspired direction finding: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k, then blend.
    Uses multiplicative weights on the dual.
    """
    a = a.copy()
    c = fft_convolve_self(a)
    M = c.max()
    start_time = time.monotonic()
    best_obj = true_objective_value(a)
    best_a = a.copy()

    size = 2 * n - 1
    # Dual variables (weights on constraints)
    lam = np.ones(size) / size

    # Initialize primal
    g = np.ones(n) / n
    g = g * S  # Scale to have sum S

    for outer in range(200):
        if time.monotonic() - start_time > max_time:
            break

        # Compute (a * g) convolution
        fsize = 1 << (size - 1).bit_length()
        A = fft(a, fsize)
        G = fft(g, fsize)
        ag = np.real(ifft(A * G))[:size]

        # Check violation
        violated = ag > M * (1 + 1e-10)
        if not np.any(violated):
            break

        # Update dual: increase weights on violated constraints
        eta_dual = 0.05
        for k in np.where(violated)[0][:min(200, size)]:
            lam[k] *= np.exp(eta_dual * (ag[k] - M) / max(M, 1e-10))
        lam = lam / lam.sum()  # Normalize

        # Compute gradient for g: d(consr)/dg[j] at violated k
        # (a*g)[k] = sum_i a[i]*g[k-i], so d/dg[j] = a[k-j] for valid k
        # Weighted sum over violated k:
        red = np.zeros(n)
        viol_idx = np.where(ag > M * (1 + 1e-10))[0]
        for k in viol_idx[:min(50, len(viol_idx))]:
            weight = (ag[k] - M) / max(M, 1e-10)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            red[lo:hi + 1] += weight * a[k - (hi - lo - 1 + lo):k - lo + 1] if False else weight * a[k - np.arange(lo, hi + 1)]

        # Update g: reduce where it causes violation
        g = np.maximum(g - 0.1 * red, 0)

        # Normalize g to sum S
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) * S / n
        else:
            g = g * (S / gs)

        # Blend: try a = (1-t)*a + t*g and see if it helps
        for t in [0.1, 0.3, 0.5]:
            a_try = (1 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum() * S
            obj_try = true_objective_value(a_try)
            if obj_try < best_obj - 1e-9:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a, best_obj


def run_micro_refine(a, n, S, max_time, rng):
    """
    Fine-grained refinement: small steps with exact (non-smooth) subgradient.
    """
    a = a.copy()
    start_time = time.monotonic()
    best_obj = true_objective_value(a)
    best_a = a.copy()

    step = 0.05
    it = 0
    stall = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S ** 2

        # Top-1 subgradient: reduce the peak lag
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        grad[lo:hi + 1] = a[k_star - np.arange(lo, hi + 1)]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try multiple k values (top few peaks)
        best_try = None
        best_try_obj = cur_obj

        peak_vals, peak_idx = np.sort(c)[::-1], None
        top_k = np.argsort(c)[::-1][:min(5, len(c))]

        for k_test in top_k:
            lo_t = max(0, k_test - n + 1)
            hi_t = min(k_test, n - 1)
            g_t = np.zeros(n)
            g_t[lo_t:hi_t + 1] = a[k_test - np.arange(lo_t, hi_t + 1)]
            gn_t = np.linalg.norm(g_t)
            if gn_t < 1e-15:
                continue

            eff_step = step * S / (n * gn_t * 10)
            a_try = project_simplex(a - eff_step * g_t, S)
            obj_try = true_objective_value(a_try)
            if obj_try < best_try_obj - 1e-10:
                best_try_obj = obj_try
                best_try = a_try

        if best_try is not None:
            a = best_try
            step = min(step * 1.1, 1.0)
            stall = 0
            if best_try_obj < best_obj - 1e-9:
                best_obj = best_try_obj
                best_a = best_try.copy()
        else:
            stall += 1
            step *= 0.7
            if stall > 20:
                # Perturb
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                step = 0.05
                stall = 0

        if it % 500 == 0:
            print("    micro: it=%d, obj=%.8f" % (it, best_obj))

    return best_a, best_obj


def generate_diverse_inits(n, rng, prev_seq=None):
    """Generate diverse initialization sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if prev_seq is not None:
        prev = np.maximum(prev_seq, 0).astype(np.float64)
        if len(prev) == n:
            inits.append(("prev", prev.copy()))
        elif len(prev) >= n:
            inits.append(("prev_trunc", prev[:n].copy()))

    # Linear ramp (asymmetric)
    inits.append(("ramp", 1.0 + 0.8 * (2 * tt - 1)))

    # Quadratic (parabolic, shifted)
    inits.append(("quad", 1.0 + 0.6 * (2 * tt - 0.7) ** 2 - 0.2))

    # Cosine window
    inits.append(("cosine", np.sin(np.pi * tt) ** 2 + 0.1))

    # Raised cosine
    inits.append(("raised_cos", 0.5 + 0.5 * np.cos(2 * np.pi * tt)))

    # Tangent-like (steep in middle)
    inits.append(("tangent", np.tanh(3 * (2 * tt - 1)) + 1.5))

    # Power law
    inits.append(("pow", tt ** 0.3 + 0.1))

    # Inverse power
    inits.append(("inv_pow", (1 - tt) ** 0.3 + 0.1))

    # Multi-scale: large asymmetry + small oscillation
    inits.append(("multi", (1.0 + 0.5 * tt) * (1.0 + 0.1 * np.cos(20 * np.pi * tt))))

    # Sinusoidal modulation with asymmetry
    inits.append(("sin_asym", (1.0 + 0.3 * np.sin(4 * np.pi * tt)) * (0.7 + 0.3 * tt)))

    # Gaussian bump (off-center)
    inits.append(("gauss", np.exp(-4 * (tt - 0.3) ** 2) * 2 + 0.1))

    # Random positive
    inits.append(("rand", rng.random(n) * 0.5 + 0.5))

    # Two-hump
    inits.append(("two_hump", np.exp(-20*(tt-0.25)**2) + np.exp(-20*(tt-0.75)**2) + 0.1))

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses smooth-surrogate FISTA/gradient descent with adaptive temperature,
    diverse multi-start, LP polishing, and micro-refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 45.0)

    # Get previous best if available
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if prev_seq.size >= 64 and np.sum(np.clip(prev_seq, 0, 1000)) > 0.01:
            prev_seq = np.clip(prev_seq, 0, 1000)
    except Exception:
        pass

    # Choose length
    if prev_seq is not None and 2048 <= len(prev_seq) <= 16384:
        n = len(prev_seq)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, S=%.2f, budget=%.0fs" % (n, S, budget_s))
    t0 = time.monotonic()

    global_best = np.inf
    global_a = None

    # Generate initializations
    inits = generate_diverse_inits(n, rng, prev_seq)
    print("Generated %d initializations" % len(inits))

    # Strategy: allocate time to initializations based on expected quality
    # Give more time to "prev" and "ramp" (most promising)
    usable_time = max(60.0, deadline - time.monotonic())

    # Time allocation weights
    weights = {}
    for name, _ in inits:
        if name == "prev":
            weights[name] = 3.0
        elif name in ("ramp", "quad", "multi"):
            weights[name] = 2.0
        elif name in ("cosine", "sin_asym"):
            weights[name] = 1.5
        else:
            weights[name] = 1.0

    total_w = sum(weights.values())
    time_per_init = {}
    for name, _ in inits:
        time_per_init[name] = usable_time * weights[name] / (total_w * 1.5)
        # Cap at 600s per init
        time_per_init[name] = min(time_per_init[name], 600.0)

    # Phase 1: Run optimizations from each initialization
    print("=== Phase 1: Multi-start optimization ===")
    results = []

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break

        t_alloc = time_per_init.get(name, 60.0)
        t_alloc = min(t_alloc, deadline - time.monotonic() - 60.0)
        if t_alloc < 10:
            continue

        arr_n = len(arr)
        if arr_n != n:
            # Resample to correct length
            if arr_n > n:
                arr = arr[:n]
            else:
                # Interpolate
                old_t = np.linspace(0, 1, arr_n)
                new_t = np.linspace(0, 1, n)
                arr = np.interp(new_t, old_t, arr)
            arr_n = n

        S_i = np.sqrt(2.0 * arr_n)
        print("  [%s] starting, alloc=%.0fs" % (name, t_alloc))

        a_opt, score = run_smooth_fista(arr, arr_n, S_i, t_alloc, rng)
        a_opt = project_simplex(a_opt, S_i)
        real_score = true_objective_value(a_opt)

        results.append((name, real_score, a_opt))
        print("  [%s] obj=%.8f (elapsed=%.0fs)" % (name, real_score, time.monotonic() - t0))

        if real_score < global_best:
            global_best = real_score
            global_a = a_opt.copy()

    # Sort results by quality
    results.sort(key=lambda x: x[1])
    print("\n  Top 5 results:")
    for name, score, _ in results[:5]:
        print("    %s: %.8f" % (name, score))

    print("\nAfter Phase 1: best=%.8f" % global_best)

    # Phase 2: Intensive optimization on best candidate
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print("\n=== Phase 2: Intensive optimization (remaining %.0fs) ===" % remaining)

        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Run gradient descent (different algorithm for diversity)
        t_fd = remaining * 0.3
        a_fd, s_fd = run_gradient_descent(global_a, n_best, S_best, t_fd, rng)
        a_fd = project_simplex(a_fd, S_best)
        s_fd = true_objective_value(a_fd)
        print("  After gradient descent: %.8f" % s_fd)
        if s_fd < global_best:
            global_best = s_fd
            global_a = a_fd.copy()

        # Run smooth FISTA again on the best (with different schedule)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_f2 = remaining2 * 0.4
            beta_sched = [20, 50, 100, 200, 400, 800]
            a_f2, s_f2 = run_smooth_fista(global_a, n_best, S_best, t_f2, rng,
                                          beta_schedule=beta_sched)
            a_f2 = project_simplex(a_f2, S_best)
            s_f2 = true_objective_value(a_f2)
            print("  After 2nd FISTA: %.8f" % s_f2)
            if s_f2 < global_best:
                global_best = s_f2
                global_a = a_f2.copy()

    # Phase 3: LP polishing
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 3: LP polishing (remaining %.0fs) ===" % remaining)

        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_lp = min(remaining * 0.3, 180.0)
        a_lp, s_lp = run_lp_polish(global_a, n_best, S_best, t_lp, rng)
        a_lp = project_simplex(a_lp, S_best)
        s_lp = true_objective_value(a_lp)
        print("  After LP: %.8f" % s_lp)
        if s_lp < global_best:
            global_best = s_lp
            global_a = a_lp.copy()

    # Phase 4: Micro refinement
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 4: Micro refinement (remaining %.0fs) ===" % remaining)

        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_micro = min(remaining * 0.5, 300.0)
        a_micro, s_micro = run_micro_refine(global_a, n_best, S_best, t_micro, rng)
        a_micro = project_simplex(a_micro, S_best)
        s_micro = true_objective_value(a_micro)
        print("  After micro: %.8f" % s_micro)
        if s_micro < global_best:
            global_best = s_micro
            global_a = a_micro.copy()

    # Phase 5: Final attempt with a different n if time allows
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        # Try a different length to see if it's better
        n_alt = len(global_a) * 2 if len(global_a) <= 8192 else len(global_a) // 2
        if n_alt >= 1024 and n_alt <= 32768 and n_alt != len(global_a):
            print("\n=== Phase 5: Try n=%d (remaining %.0fs) ===" % (n_alt, remaining))
            S_alt = np.sqrt(2.0 * n_alt)
            # Resample the best sequence
            old_t = np.linspace(0, 1, len(global_a))
            new_t = np.linspace(0, 1, n_alt)
            init_alt = np.interp(new_t, old_t, global_a)

            t_alt = remaining * 0.6
            a_alt, s_alt = run_smooth_fista(init_alt, n_alt, S_alt, t_alt, rng)
            a_alt = project_simplex(a_alt, S_alt)
            s_alt = true_objective_value(a_alt)
            print("  n=%d: %.8f" % (n_alt, s_alt))
            if s_alt < global_best:
                global_best = s_alt
                global_a = a_alt.copy()

    # Final: one more micro-refinement pass
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_final, s_final = run_micro_refine(global_a, n_best, S_best, min(remaining - 10, 120.0), rng)
        a_final = project_simplex(a_final, S_best)
        s_final = true_objective_value(a_final)
        if s_final < global_best:
            global_best = s_final
            global_a = a_final.copy()

    # Verify with evaluate_sequence
    print("\n=== Final verification ===")
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("evaluate_sequence: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned inf, using safe fallback")
            global_a = np.ones(256) * (np.sqrt(512) / 256)
            global_best = true_objective_value(global_a)

    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512) / 256)
        global_best = true_objective_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512) / 256)

    print("\nReturning: length=%d, obj=%.8f, total_time=%.0fs" % (
        len(global_a), global_best, time.monotonic() - t0))

    return [float(x) for x in global_a]
