"""
Minimize 2*n*max_conv/sum^2 using:
1. L-BFGS on top-k LSE-smoothed objective (primary optimizer)
2. Nesterov accelerated subgradient on top-k average (secondary)

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_info(a, N, beta, top_k=None):
    """Compute LSE value and gradient info. Returns (lse_val, c, c_max, top_indices)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    
    if top_k is not None and top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c
    
    exp_bc = np.exp(beta * (c_top - c_top.max()))
    total = exp_bc.sum()
    if total < 1e-300:
        lse_val = c_top.max()
        w = np.zeros_like(c_top)
        w[0] = 1.0
    else:
        lse_val = c_top.max() + np.log(total) / beta
        w = exp_bc / total
    
    return lse_val, c, c_max, top_idx, w

def lse_gradient(a, N, beta, top_k=None):
    """Gradient of top-k LSE-smoothed peak. Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    lse_val, c, c_max, top_idx, w = lse_info(a, N, beta, top_k)
    
    # Build weight vector in full convolution space
    fsize = N  # Use same FFT size
    w_full = np.zeros(fsize)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = w[i]
    
    W = fft(w_full)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    Af = fft(a_pad)
    
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max

def lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy = np.dot(y_last, y_last)
    if yy > 1e-30:
        gamma = max(1e-10, np.dot(s_last, y_last) / yy)
    else:
        gamma = 1.0
    r = gamma * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def nesterov_block(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=200):
    """Nesterov accelerated top-k subgradient."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        fsize = N

def freq_perturb(a, N, rng, strength=0.1):

"""
Minimize 2*n*max_conv/sum^2 using:
1. L-BFGS on top-k LSE-smoothed objective (primary optimizer)
2. Nesterov accelerated top-k subgradient (secondary)
3. Frequency-domain perturbation for global moves
4. Multi-length search with adaptive time allocation

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_gradient(a, N, beta, top_k=None):
    """Gradient of top-k LSE-smoothed peak. Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()

    if top_k is not None and top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        c_top = c

    c_top_max = c_top.max()
    exp_bc = np.exp(beta * (c_top - c_top_max))
    total = exp_bc.sum()
    if total < 1e-300:
        lse_val = c_top_max
        w = np.zeros_like(c_top)
        w[np.argmax(c_top)] = 1.0
    else:

    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

def lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy = np.dot(y_last, y_last)
    if yy > 1e-30:
        gamma = max(1e-10, np.dot(s_last, y_last) / yy)
    else:
        gamma = 1.0
    r = gamma * q

    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)

        a = a_new
        g = g_new

        if it % 600 == 0 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            g, _, _, c_max = lse_gradient(a, N, beta, top_k)

    return best_a, best_cmax


def nesterov_block(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=300):
    """Nesterov accelerated top-k subgradient."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = compute_conv(y, N).max()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 40)

        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()

        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(N)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full)
        y_pad = np.zeros(N)
        y_pad[:n] = y
        Af = fft(y_pad)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 1500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_a, best_cmax


def freq_perturb(a, N, rng, strength=0.05):
    """Perturb in frequency domain and project back."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)
    perturbation = rng.normal(0, strength, N)
    mag_new = np.maximum(mag * (1.0 + perturbation), 1e-10)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def safe_normalize(a, n, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + 0.5 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_aperiodic(n, rng, N):
    """Design in frequency domain: nearly flat magnitude spectrum."""
    mag = np.ones(N)
    mag[0] = float(n)
    # Add small random variation to avoid exact flatness
    mag[1:] *= (1.0 + 0.1 * rng.standard_normal(N - 1))
    mag = np.maximum(mag, 0.1)
    phase = rng.uniform(0, 2 * np.pi, N)
    phase[0] = 0.0
    A = mag * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-strategy: L-BFGS on top-k LSE with adaptive beta,
    Nesterov subgradient, frequency perturbation.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_best(a_candidate, n_candidate, label=""):
        nonlocal global_best, global_a, global_n
        S = np.sqrt(2.0 * n_candidate)
        a_cand = safe_normalize(a_candidate, n_candidate, S)
        N = 1 << (2 * n_candidate - 1).bit_length()
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            global_n = n_candidate
            if label:
                print("  [%s] n=%d obj=%.8f" % (label, n_candidate, sc))
            return True
        return False

    # Determine which lengths to try
    primary_n = 8192
    try:
        given_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given_seq.size >= 1024 and given_seq.size <= 32768:
            primary_n = given_seq.size
    except Exception:
        pass

    S = np.sqrt(2.0 * float(primary_n))
    N = 1 << (2 * primary_n - 1).bit_length()
    n = primary_n

    print("Primary: n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    p1_budget = min(200.0, (deadline - t1) * 0.12)

    inits = []
    try:
        given_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given_seq.size == n:
            inits.append(("given", np.clip(given_seq, 0, 1000)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("aperiodic", make_init_aperiodic(n, rng, N)))

    # Smooth structured init
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("smooth", np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                                     0.2 * np.cos(2 * np.pi * 11 * tt) +
                                     0.05 * rng.standard_normal(n), 0.01, None)))

    n_inits = len(inits)
    per_init = p1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        a_norm = safe_normalize(arr, n, S)
        rem = max(3.0, per_init * 0.9)
        a_l, sc_l = lbfgs_block(a_norm, n, S, N, rem, rng,
                                 beta_init=5.0, beta_max=80.0, m_lbfgs=8, top_k=30)
        update_best(a_l, n, name + "/L1")

    print("Phase 1 done. Best: %.8f" % global_best)

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = compute_conv(global_a, N).max()
        global_n = n

    # --- Phase 2: Progressive L-BFGS with increasing beta ---
    print("=== Phase 2: Progressive L-BFGS ===")
    t2 = time.monotonic()
    p2_budget = min(1400.0, (deadline - t2) * 0.55)

    a_curr = global_a.copy()
    stall = 0
    beta_stages = [8, 15, 30, 60, 120, 250, 500, 1000]
    stage_idx = 0

    while time.monotonic() < t2 + p2_budget:
        block_time = min(200.0, (t2 + p2_budget) - time.monotonic())
        if block_time < 15:
            break

        beta_start = beta_stages[min(stage_idx, len(beta_stages) - 1)]
        beta_end = min(3000.0, beta_start * 4)

        # Reduce top_k as beta increases (more focused)
        tk = max(15, 50 - stage_idx * 5)

        a_new, sc_new = lbfgs_block(a_curr, n, S, N, block_time, rng,
                                     beta_init=beta_start, beta_max=beta_end,
                                     m_lbfgs=12, top_k=tk)
        if sc_new < global_best - 1e-12:
            global_best = sc_new
            global_a = a_new.copy()
            global_n = n
            a_curr = a_new.copy()
            stall = 0
            print("  [P2 s%d] beta=%d-%d tk=%d obj=%.8f" % (stage_idx, beta_start, beta_end, tk, sc_new))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                a_curr = freq_perturb(a_curr, N, rng, strength=0.04)
                a_curr = safe_normalize(a_curr, n, S)
                stall = 0
                stage_idx = min(stage_idx + 1, len(beta_stages) - 1)
                print("  [P2 s%d] perturbed, advancing to beta=%d" % (stage_idx, beta_stages[stage_idx]))

    # --- Phase 3: Nesterov polish ---
    print("=== Phase 3: Nesterov polish ===")
    t3 = time.monotonic()
    remaining_p3 = min(250.0, (deadline - t3) * 0.3)
    if remaining_p3 > 30:
        a_ng, sc_ng = nesterov_block(a_curr, n, S, N, remaining_p3, rng,
                                      top_k_init=20, top_k_max=300)
        update_best(a_ng, n, "nesterov")

    # --- Phase 4: Final high-beta L-BFGS ---
    print("=== Phase 4: Final high-beta L-BFGS ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        a_fin, sc_fin = lbfgs_block(global_a.copy(), n, S, N, min(remaining, 400.0), rng,
                                     beta_init=200.0, beta_max=5000.0, m_lbfgs=10, top_k=15)
        update_best(a_fin, n, "final_hi_beta")

    # --- Phase 5: One more Nesterov pass on the best found ---
    remaining = deadline - time.monotonic() - 8
    if remaining > 20:
        a_ng2, sc_ng2 = nesterov_block(global_a.copy(), n, S, N, min(remaining, 150.0), rng,
                                        top_k_init=15, top_k_max=200)
        update_best(a_ng2, n, "nesterov2")

    # --- Verification ---
    print("=== Verification ===")
    print("Best internal objective: %.8f (n=%d)" % (global_best, global_n))

    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
        else:
            print("Verification: inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d" % len(global_a))
    return [float(x) for x in global_a]
