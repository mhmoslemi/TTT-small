```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def cross_corr_lag_j(f, g, N):
    """Compute (f ⋆ g)[j] for j = 0, ..., len(g)-1"""
    F = fft(f, N)
    G = fft(g, N)
    return np.real(ifft(F * np.conj(G)))[:len(g)]

def lse_smooth_max(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    return cm + np.log(np.sum(np.exp(scaled))) / beta

def lse_gradient(c, beta, a, N):
    """Gradient of LSE-smoothed max of c = convolve(a,a) w.r.t. a"""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    w = e / e.sum()
    # gradient_j = 2 * sum_k w[k] * a[k-j] = 2 * (w ⋆ a)[j]
    W = fft(w, N)
    A = fft(a, N)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]

def lp_gradient(c, a, p, N):
    """Gradient direction of L_p norm of convolution w.r.t. a"""
    if p == 2:
        weights = c
    else:
        weights = c ** (p - 1)
    # direction_j = 2 * (weights ⋆ a)[j]
    W = fft(weights, N)
    A = fft(a, N)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
```

```python
            break
        weights /= wsum

        # Gradient: 2 * (weights ⋆ a)
        W = fft(weights, N_xcorr)
        A = fft(a, N_xcorr)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Decreasing step
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        a_new = project_simplex(a - step * g, S)

        # Check if improvement
        c_new = fft_convolve_self(a_new, N_conv)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            step *= 0.1
            a_new = project_simplex(a - step * g, S)
            c_new = fft_convolve_self(a_new, N_conv)
            if c_new.max() < cur_max:
                a = a_new
            else:
                # Perturb
                noise = 0.0005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best


def make_initializations(n, rng, base_seq=None):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None:
        bs = np.asarray(base_seq, dtype=np.float64).flatten()
        if len(bs) == n:
            inits.append(np.clip(bs, 0, 1000))
        elif len(bs) > n:
            # Downsample by averaging
            ratio = len(bs) // n
            if ratio * n <= len(bs):
                small = bs[:ratio * n].reshape(n, ratio).mean(axis=1)
                inits.append(np.clip(small, 0, 1000))
        else:
            # Upscale by interpolation
            t_old = np.linspace(0, 1, len(bs))
            t_new = np.linspace(0, 1, n)
            inits.append(np.clip(np.interp(t_new, t_old, bs), 0, 1000))

    # Standard windows
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 5 * tt + 0.3) - 0.2 * np.cos(2 * np.pi * 11 * tt))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Block structure
    for nb in [8, 16, 32]:
        bl = n // nb
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, bl)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n])

    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the actual objective."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses L_p annealing (p=2 to 256) with FISTA, then LSE polishing and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size >= 64 and given.sum() > 0:
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
        else:
            given = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Focus on n=8192
    n = 8192
    S = np.sqrt(2.0 * n)
    N_conv = 1 << (2 * n - 1).bit_length()  # 16384
    N_xcorr = 1 << (3 * n - 2).bit_length()  # 32768

    print("n=%d, S=%.4f, N_conv=%d, N_xcorr=%d" % (n, S, N_conv, N_xcorr))

    # --- Phase 1: Quick exploration (300s) ---
    print("\n=== Phase 1: Exploration (%.0fs) ===" % 300)
    t_phase1 = min(300.0, deadline - time.monotonic() - 1500.0)
    if t_phase1 < 30:
        t_phase1 = max(30.0, deadline - time.monotonic() - 100.0)

    inits = make_initializations(n, rng, base_seq=given)
    best_a = None
    best_val = np.inf

    # Try top inits with quick L_p=2 optimization
    t_per_init = t_phase1 / min(5, len(inits))
    for i in range(min(5, len(inits))):
        if time.monotonic() > deadline - 1500:
            break
        t_alloc = min(t_per_init, deadline - time.monotonic() - 1500)
        if t_alloc < 5:
            break
        a_opt, val = run_fista_lp(inits[i], S, N_conv, N_xcorr, p=2,
                                   max_time=t_alloc, rng=rng, step_init=2.0)
        if val < best_val:
            best_val = val
            best_a = a_opt.copy()
            obj = 2.0 * n * val / (a_opt.sum() ** 2)
            print("  Init %d: max_b=%.6f, obj=%.6f" % (i, val, obj))

    if best_a is None:
        best_a = inits[0].copy()
        best_a = best_a * S / best_a.sum()
        best_val = fft_convolve_self(best_a, N_conv).max()

    if 2.0 * n * best_val / S ** 2 < global_best:
        global_best = 2.0 * n * best_val / S ** 2
        global_a = best_a.copy()
    print("Phase 1 best obj: %.6f" % global_best)

    # --- Phase 2: L_p annealing (800s) ---
    remaining = deadline - time.monotonic()
    t_phase2 = min(800.0, remaining * 0.35)
    print("\n=== Phase 2: L_p annealing (%.0fs) ===" % t_phase2)

    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256]
    # Allocate more time to lower p values
    p_weights = [0.15, 0.15, 0.15, 0.12, 0.12, 0.1, 0.1, 0.11]
    p_weights = np.array(p_weights) / np.array(p_weights).sum()

    for pi, p in enumerate(p_schedule):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = p_weights[pi] * t_phase2
        t_alloc = min(t_alloc, deadline - time.monotonic() - 600)
        if t_alloc < 5:
            break

        step_init = 3.0 / (p ** 0.5)
        a_opt, val = run_fista_lp(best_a, S, N_conv, N_xcorr, p=p,
                                   max_time=t_alloc, rng=rng, step_init=step_init)
        if val < best_val:
            best_val = val
            best_a = a_opt.copy()
            obj = 2.0 * n * val / (best_a.sum() ** 2)
            print("  p=%d: max_b=%.6f, obj=%.6f" % (p, val, obj))
        else:
            print("  p=%d: max_b=%.6f (no improvement)" % (p, val))

        if 2.0 * n * best_val / S ** 2 < global_best:
            global_best = 2.0 * n * best_val / S ** 2
            global_a = best_a.copy()

    print("Phase 2 best obj: %.6f" % global_best)

    # --- Phase 3: LSE polishing (700s) ---
    remaining = deadline - time.monotonic()
    t_phase3 = min(700.0, remaining * 0.35)
    print("\n=== Phase 3: LSE polishing (%.0fs) ===" % t_phase3)

    a_opt, val = run_fista_lse(best_a, S, N_conv, N_xcorr,
                                beta_start=50.0, beta_end=5000.0,
                                max_time=t_phase3, rng=rng)
    if val < best_val:
        best_val = val
        best_a = a_opt.copy()
        obj = 2.0 * n * val / (best_a.sum() ** 2)
        print("  LSE: max_b=%.6f, obj=%.6f" % (val, obj))
    else:
        print("  LSE: max_b=%.6f (no improvement)" % val)

    if 2.0 * n * best_val / S ** 2 < global_best:
        global_best = 2.0 * n * best_val / S ** 2
        global_a = best_a.copy()
    print("Phase 3 best obj: %.6f" % global_best)

    # --- Phase 4: Peak shaving (300s) ---
    remaining = deadline - time.monotonic()
    t_phase4 = min(300.0, remaining * 0.3)
    print("\n=== Phase 4: Peak shaving (%.0fs) ===" % t_phase4)

    a_opt, val = run_peak_shave_fine(best_a, S, N_conv, N_xcorr,
                                       max_time=t_phase4, rng=rng)
    if val < best_val:
        best_val = val
        best_a = a_opt.copy()
        obj = 2.0 * n * val / (best_a.sum() ** 2)
        print("  Peak shave: max_b=%.6f, obj=%.6f" % (val, obj))

    if 2.0 * n * best_val / S ** 2 < global_best:
        global_best = 2.0 * n * best_val / S ** 2
        global_a = best_a.copy()
    print("Phase 4 best obj: %.6f" % global_best)

    # --- Phase 5: Diversification + re-optimization (remaining time) ---
    remaining = deadline - time.monotonic()
    print("\n=== Phase 5: Diversification (%.0fs) ===" % remaining)

    if remaining > 60:
        # Try a few perturbed restarts
        n_restarts = min(4, int(remaining / 80))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the current best
            noise_scale = 0.01 * S / np.sqrt(n) * (1.0 + 0.5 * r)
            perturbed = best_a + rng.normal(0, noise_scale, n)
            perturbed = project_simplex(perturbed, S)

            # Quick LSE optimization
            t_alloc = min(60.0, remaining - 30 - r * 70)
            if t_alloc < 10:
                break
            a_opt, val = run_fista_lse(perturbed, S, N_conv, N_xcorr,
                                        beta_start=100.0, beta_end=5000.0,
                                        max_time=t_alloc, rng=rng)
            if val < best_val:
                best_val = val
                best_a = a_opt.copy()
                obj = 2.0 * n * val / (best_a.sum() ** 2)
                print("  Restart %d: max_b=%.6f, obj=%.6f" % (r, val, obj))
                if 2.0 * n * best_val / S ** 2 < global_best:
                    global_best = 2.0 * n * best_val / S ** 2
                    global_a = best_a.copy()
            else:
                print("  Restart %d: max_b=%.6f (no improvement)" % (r, val))

    # --- Final intensive polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("\n=== Final polishing (%.0fs) ===" % remaining)

        # Very high beta LSE
        t_final = remaining * 0.6
        a_opt, val = run_fista_lse(global_a, S, N_conv, N_xcorr,
                                    beta_start=500.0, beta_end=20000.0,
                                    max_time=t_final, rng=rng)
        if val < fft_convolve_self(global_a, N_conv).max():
            global_a = a_opt.copy()
            obj = 2.0 * n * val / (global_a.sum() ** 2)
            global_best = min(global_best, obj)
            print("  Final LSE: obj=%.6f" % global_best)

        # Quick peak shave
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_opt, val = run_peak_shave_fine(global_a, S, N_conv, N_xcorr,
                                              max_time=remaining2 * 0.7, rng=rng)
            if val < fft_convolve_self(global_a, N_conv).max():
                global_a = a_opt.copy()
                obj = 2.0 * n * val / (global_a.sum() ** 2)
                global_best = min(global_best, obj)
                print("  Final peak shave: obj=%.6f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    s = global_a.sum()
    if s < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_obj = objective_value(global_a)
    print("\nFinal internal objective: %.6f" % final_obj)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification via evaluate_sequence: %.6f" % ver)

    if not np.isfinite(ver):
        print("evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d" % len(global_a))
    return final_list
```
