    """

        c = fft_convolve_self(y)

            c_top = c[top_idx]
            # Softmax weights
            c_max = c_top.max()
            w = np.exp(beta * (c_top - c_max)) / np.sum(np.exp(beta * (c_top - c_max)))
            # Gradient: sum_i w_i * dc[i]/dy
            w_full = np.zeros_like(c)

            w = np.exp(beta * (c - c_max)) / np.sum(np.exp(beta * (c - c_max)))

        if sc < best_score:

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using FISTA with log-sum-exp smoothing
and spectral pre-conditioning. 

Key ideas:
1. The problem is convex (minimize max of convex quadratics s.t. simplex constraint)
2. Use log-sum-exp to smooth the non-differentiable max, enabling accelerated gradient
3. Beta-annealing: start with low beta for fast convergence, increase for precision
4. Spectral pre-shaping to get close to optimum before FISTA
5. Top-k focus: only track the most important lags in the log-sum-exp
6. Proper line search with backtracking for robust step selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_gradient(weights_full, y, N_grad):
    """
    Compute gradient of sum_k weights[k]*c[k] w.r.t. y.
    grad[m] = 2 * sum_k weights[k] * y[k-m] (valid indices only)
    Uses FFT with sufficient zero-padding for linear cross-correlation.
    """
    n = len(y)
    W = fft(weights_full, N_grad)
    Y = fft(y, N_grad)
    cross = np.real(ifft(np.conj(W) * Y))[:n]
    return 2.0 * cross


def spectral_precondition(a, N, S, n_steps, rng):
    """
    Pre-condition by iteratively flattening the power spectrum.
    Returns the best sequence found.
    """
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    a_curr = a.copy()

    for it in range(n_steps):
        progress = it / max(n_steps - 1, 1)
        alpha = 0.6 * (1.0 - 0.7 * progress) + 0.05

        A = fft(a_curr, N)
        P = np.abs(A) ** 2
        mean_P = P.mean()

        if mean_P < 1e-30:
            break

        # Interpolate toward flat spectrum
        P_new = P * (1.0 - alpha) + mean_P * alpha
        P_new = np.maximum(P_new, 0.0)

        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]

        # Project to simplex
        a_new = project_simplex(a_new, S)

        # Line search: try a few mixing ratios
        best_local = best_score
        best_mix = a_curr
        for t in [1.0, 0.8, 0.5, 0.3]:
            candidate = (1.0 - t) * a_curr + t * a_new
            candidate = project_simplex(candidate, S)
            sc = compute_score(candidate)
            if sc < best_local:
                best_local = sc
                best_mix = candidate

        a_curr = best_mix
        if best_local < best_score:
            best_score = best_local
            best_a = a_curr.copy()

    return best_a, best_score


def fista_smooth(a_init, n, S, N, N_grad, max_time, beta_start, beta_end,
                 top_k, rng, use_annealing=True):
    """
    FISTA (accelerated proximal gradient) with log-sum-exp smoothing.
    
    Parameters:
    - a_init: initial point on simplex {x>=0, sum(x)=S}
    - n: sequence length
    - S: target sum
    - N: FFT size for convolution (>= 2n-1)
    - N_grad: FFT size for gradient (>= 3n-2)
    - max_time: time budget in seconds
    - beta_start: initial temperature
    - beta_end: final temperature
    - top_k: number of top lags to focus on
    - rng: random number generator
    - use_annealing: whether to anneal beta
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0

    # Step size initialization
    step = 2.0 * S / (n * beta_start * 10.0)
    step_min = step * 0.01
    step_max = 2.0 * S / (n * beta_start) * 5.0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Compute convolution of y
        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()

        # Anneal beta
        if use_annealing and it > 10:
            progress = min(1.0, (it - 10) / max(100, it))
            beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        else:
            beta = beta_start

        # Compute top-k weights (softmax over top lags)
        if top_k is not None and top_k < len(c):
            top_idx = np.argsort(c)[::-1][:top_k]
            c_top = c[top_idx]
            c_max_top = c_top.max()
            exps = np.exp(beta * (c_top - c_max_top))
            total = exps.sum()
            if total < 1e-30:
                break
            w_top = exps / total
            w_full = np.zeros(len(c))
            w_full[top_idx] = w_top
        else:
            c_max = c.max()
            exps = np.exp(beta * (c - c_max))
            total = exps.sum()
            if total < 1e-30:
                break
            w_full = exps / total

        # Compute gradient via FFT
        grad = compute_gradient(w_full, y, N_grad)

        # Gradient norm check
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        f_y = (1.0 / beta) * np.log(total + 1e-30)
        found_step = False
        step_try = min(step, step_max)

        for ls in range(10):
            x_new = y - step_try * grad
            x_new = project_simplex(x_new, S)

            # Evaluate objective at x_new
            c_new = fft_convolve_self(x_new)
            if top_k is not None and top_k < len(c_new):
                top_idx_new = np.argsort(c_new)[::-1][:top_k]
                c_new_max = c_new[top_idx_new].max()
                c_new_shifted = c_new[top_idx_new] - c_new[top_idx_new].max()
                total_new = np.exp(beta * c_new_shifted).sum()
            else:
                c_new_max = c_new.max()
                c_new_shifted = c_new - c_new.max()
                total_new = np.exp(beta * c_new_shifted).sum()
            f_new = (1.0 / beta) * np.log(total_new + 1e-30)

            # Linear decrease condition
            if f_new <= f_y + 0.001 * np.dot(grad, x_new - y) + 0.0001 * step_try * gnorm ** 2:
                found_step = True
                break
            step_try *= 0.5
            if step_try < step_min:
                found_step = True
                break

        if not found_step:
            x_new = y - step_try * grad
            x_new = project_simplex(x_new, S)

        # Accept or reject (for robustness)
        c_check = fft_convolve_self(x_new)
        sc_check = 2.0 * n * c_check.max() / (x_new.sum() ** 2)
        if sc_check > best_score * 1.5:
            # Reject this step
            step = step_try * 0.5
            x_new = y.copy()

        # Update step for next iteration
        step = np.clip(step_try * 1.05, step_min, step_max)

        # Nesterov acceleration
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y_new = x_new + momentum * (x_new - y)
        y_new = np.maximum(y_new, 0.0)

        # Renormalize to maintain sum constraint
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new * (S / s_y)

        a = y  # old y becomes new a for momentum
        y = y_new
        t = t_new

        # Periodic perturbation to escape potential local issues
        if it % 2000 == 1000 and rng.random() < 0.15:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            a = y.copy()

    return best_local, best_score


def make_initializations(n, rng, provided_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if provided_seq is not None and len(provided_seq) == n:
        inits.append(("provided", provided_seq.copy()))

    # Uniform
    inits.append(("uniform", np.ones(n) * 0.5))

    # Sine-based
    inits.append(("sin1", 0.5 + 0.3 * np.sin(2 * np.pi * tt)))
    inits.append(("sin_multi", 0.5 + 0.15 * np.sin(2 * np.pi * 3 * tt) +
                                 0.1 * np.sin(2 * np.pi * 7 * tt) +
                                 0.08 * np.sin(2 * np.pi * 13 * tt)))

    # Chirp
    inits.append(("chirp", 0.5 + 0.25 * np.sin(2 * np.pi * (3 * tt + 2 * tt * tt))))

    # Random smooth
    noise = rng.standard_normal(n)
    kern = max(3, n // 128)
    kernel = np.ones(kern) / kern
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_rand", 0.2 + 0.6 * smooth))

    # Pure random
    inits.append(("random", rng.random(n) * 0.8 + 0.1))

    # Cosine window
    inits.append(("cos_win", 0.3 + 0.4 * (1 + np.cos(2 * np.pi * tt)) / 2))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(tt - 0.5)
    inits.append(("triangle", 0.3 + 0.4 * tri))

    # Inverse cosine (peaked at edges)
    inits.append(("inv_cos", 0.3 + 0.4 * (1 - (1 + np.cos(2 * np.pi * tt)) / 2)))

    return inits


def optimize_single_length(n, budget, rng, provided_seq=None, target=1.50):

        if sc < global_best:
            global_best = sc
            global_a = a.copy()
        return sc

    inits = make_initializations(n, rng, provided_seq)

    t_start = time.monotonic()
    n_inits = len(inits)

    else:
        budgets = [budget / n_inits] * n_inits

    for i, (name, arr) in enumerate(inits):
        elapsed = time.monotonic() - t_start
        remaining = budget - elapsed
        if remaining < 10:
            break

        alloc = min(budgets[i] if i < len(budgets) else budget / n_inits, remaining - 5)
        if alloc < 5:
            break

        print("  [%s] n=%d, budget=%.0fs" % (name, n, alloc))

        a = np.clip(arr.astype(np.float64), 0.0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        sc = try_update(a)
        print("    Initial score: %.6f" % sc)

        # Phase 1: Spectral preconditioning
        t1 = time.monotonic()
        n_sp_steps = max(50, int(alloc * 0.25 / max(0.001, N * 14 * 1e-9)))
        n_sp_steps = min(n_sp_steps, 2000)
        a_sp, sc_sp = spectral_precondition(a, N, S, n_sp_steps, rng)
        sc = try_update(a_sp)
        t1_elapsed = time.monotonic() - t1
        print("    After spectral: %.6f (%.0fs)" % (sc, t1_elapsed))

        # Phase 2: FISTA with beta annealing
        t2 = time.monotonic()
        fista_budget = max(5, alloc - (time.monotonic() - t_start) - 5)
        if fista_budget > 5:
            a_f, sc_f = fista_smooth(a_sp, n, S, N, N_grad, fista_budget,
                                     beta_start=5.0, beta_end=50.0,
                                     top_k=min(200, n // 32), rng=rng,
                                     use_annealing=True)
            sc = try_update(a_f)
            print("    After FISTA: %.6f" % sc)

        # Phase 3: Second FISTA pass with higher beta (if time remains)
        remaining = budget - (time.monotonic() - t_start)
        if remaining > 10 and global_best > target:
            a_f2, sc_f2 = fista_smooth(a_f if 'a_f' in dir() else a_sp, n, S,
                                       N, N_grad, min(remaining - 5, alloc * 0.3),
                                       beta_start=20.0, beta_end=200.0,
                                       top_k=min(50, n // 128), rng=rng,
                                       use_annealing=False)
            sc = try_update(a_f2)
            print("    After FISTA2: %.6f" % sc)

        if global_best <= target:
            print("    Target reached!")
            break

    # Final polish on global best
    if global_a is not None:
        remaining = budget - (time.monotonic() - t_start)
        if remaining > 15:
            a_final, _ = fista_smooth(global_a, n, S, N, N_grad,
                                      min(remaining - 5, 60.0),
