"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA-LSE (aggressive beta) + spectral de-symmetrization + vectorized CD.
Key insight: optimal sequences are near-palindromic with small anti-palindromic perturbation
to reduce the center convolution lag below the palindromic lower bound.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad_and_val(a, c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    max_sh = shifted.max()
    e = np.exp(shifted - max_sh)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(len(a))
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    val = cm + max_sh / beta + np.log(tot) / beta
    return val, grad


def run_fista_aggressive(a_init, n, S, max_time, rng,
                         beta_init=50.0, beta_max=10000.0,
                         beta_ramp=2000, restart_every=2000,
                         noise_scale_init=0.005):
    a = normalize_to_sum(a_init, S)
    best = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_init + (beta_max - beta_init) * min(1.0, it / beta_ramp)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        val, grad = lse_grad_and_val(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            grad = np.zeros(n)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_star - idx]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

        step = 1.0 / (4.0 * beta)
        step = min(step, 0.05)

        a_new = project_simplex(a - step * grad, S)

        # Periodic restart from best
        if it % restart_every == 0:
            a = best_a.copy()
            # Add small noise to escape
            elapsed_frac = (time.monotonic() - start) / max_time
            ns = noise_scale_init * S / np.sqrt(n) * (1.0 - elapsed_frac)
            if ns > 1e-9:
                a = project_simplex(a + rng.normal(0, ns, n), S)
            it = 0
            continue

        # Intermediate noise (decaying)
        if it % 500 == 0 and it > 0:
            elapsed_frac = (time.monotonic() - start) / max_time
            ns = noise_scale_init * S / np.sqrt(n) * (1.0 - elapsed_frac) * 0.5
            if ns > 1e-10:
                a_test = project_simplex(a_new + rng.normal(0, ns, n), S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < best:
                    a = a_test
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_depeaking(a_init, n, S, max_time, rng):
    """Target spectral peaking: reduce specific high lags via frequency-domain notch."""
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Identify top lags and their frequency contributions
        top_k = 5
        max_val = c.max()
        # Weight: peaks get more suppression
        progress = min(1.0, it / 500.0)
        alpha = 0.3 + 0.5 * progress

        # Flatten power spectrum with peak weighting
        P_ratio = P / P_mean
        P_new = np.ones_like(P)
        P_new = (1.0 - alpha) * P_ratio + alpha * np.ones_like(P)
        # Extra suppression where power is concentrated
        P_new = P_new / (1.0 + 0.5 * np.maximum(P_ratio - 1.0, 0.0) * progress)

        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        # Blend with current for stability
        a = 0.7 * a_new + 0.3 * a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

    return best_a, best


def desymmetrize(a, n, S, alpha=0.1):
    """Add anti-palindromic component to reduce center lag."""
    a_sym = (a + a[::-1]) / 2.0
    a_anti = (a - a[::-1]) / 2.0
    a_new = a_sym + alpha * a_anti
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return normalize_to_sum(a, S)
    return a_new * (S / s)


def fast_coordinate_descent(a_init, n, S, max_time, rng):
    """Vectorized coordinate descent with golden section search."""
    a = normalize_to_sum(a_init, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved_this_round = False
        n_pairs = min(100 + it * 2, 500)

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose pair: biased toward positions that affect peak lags
            k_peak = int(np.argmax(c))
            if rng.random() < 0.4:
                # Choose j near the peak lag region
                j = int(np.clip(rng.integers(0, n), 0, n - 1))
                # m chosen to affect the peak
                m = int(np.clip(k_peak - j + rng.integers(-n // 4, n // 4), 0, n - 1))
            else:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))

            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # Compute d[k] = change in c[k] per unit delta (transfer m -> j)
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val -= a[mk]
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val

            lo = -a[j]
            hi = a[m]
            if hi - lo < 1e-12:
                continue

            # Golden section search on max(c + delta*d + 2*delta^2 at k=0 correction)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            ax, bx = lo, hi
            cx_ = bx - gr * (bx - ax)
            dx_ = ax + gr * (bx - ax)

            def obj(dlt):
                vals = c + dlt * d
                vals[0] += 2.0 * (a[j] - a[m]) * dlt + 2.0 * dlt * dlt
                return vals.max()

            fc, fd = obj(cx_), obj(dx_)
            for _ in range(30):
                if fc < fd:
                    bx = dx_
                    dx_ = cx_
                    fd = fc
                    cx_ = bx - gr * (bx - ax)
                    fc = obj(cx_)
                else:
                    ax = cx_
                    cx_ = dx_
                    fc = fd
                    dx_ = ax + gr * (bx - ax)
                    fd = obj(dx_)

            delta = (ax + bx) / 2.0
            delta = max(lo, min(hi, delta))

            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue

            c_trial = c + delta * d
            c_trial[0] += 2.0 * (a[j] - a[m]) * delta + 2.0 * delta * delta
            cur_trial = c_trial.max()

            if cur_trial < best - 1e-12:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved_this_round = True

        if not improved_this_round:
            stagnant += 1
            if stagnant > 20:
                # Restart with noise
                ns = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 20.0)
                if ns < 1e-9:
                    break
                a = project_simplex(a + rng.normal(0, ns, n), S)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()
                stagnant = 0
        else:
            stagnant = 0

        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def generate_starting_points(parent, n, rng):
    """Generate diverse starting points."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    starts = []

    if parent is not None and len(parent) == n:
        starts.append(("parent", normalize_to_sum(parent, S)))
        # De-symmetrized parent
        p = normalize_to_sum(parent, S)
        starts.append(("p_desym", desymmetrize(p, n, S, 0.3)))
        # Reversed
        starts.append(("p_rev", p[::-1].copy()))
        # Small perturbations
        for sc in [0.01, 0.03]:
            v = project_simplex(p + rng.normal(0, sc * S / np.sqrt(n), n), S)
            starts.append(("p_pert_%d" % int(sc * 1000), v))

    # Structured starts
    starts.append(("uniform", np.ones(n) * S / n))
    starts.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.001)))
    starts.append(("gauss", np.exp(-4.0 * (tt - 0.5) ** 2)))
    starts.append(("cos2", np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * tt), 0.001)))
    starts.append(("ramp_left", np.maximum(np.linspace(0, 1, n) ** 0.5 + 0.01, 0.001)))
    starts.append(("ramp_right", np.maximum(np.linspace(1, 0, n) ** 0.5 + 0.01, 0.001)))
    starts.append(("bimodal", np.exp(-20 * (tt - 0.25) ** 2) + np.exp(-20 * (tt - 0.75) ** 2)))
    starts.append(("random", rng.random(n) + 0.05))

    # Normalize all
    result = []
    for name, arr in starts:
        result.append((name, normalize_to_sum(arr, S)))
    return result
