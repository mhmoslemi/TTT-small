import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def obj_value(c, n):
    """Compute objective from convolution array."""
    return 2.0 * n * c.max() / (n * c.sum() / n) ** 2 if False else None


def compute_objective(a, n):
    """Compute 2*n*max_c/(sum_a)^2 directly."""
    c = compute_conv(a, 1 << (2 * n - 1).bit_length())
    cmax = c.max()
    s = a.sum()
    if s < 1e-10:
        return 1e10
    return 2.0 * n * cmax / (s * s)


def lse_value_and_grad(a, N, beta):
    """Compute LSE-smoothed max of conv and its gradient. Returns (lse_val, grad, c, cmax)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g, m_max=10):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        return -g / gn if gn > 1e-18 else np.zeros(n)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS optimization on LSE-smoothed objective. Returns (best_a, best_cmax)."""
    a = project_simplex(a_init.copy(), S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    _, g, c, c_max = lse_value_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    last_improve = best_cmax

    while time.monotonic() - start_time < max_time:
        it += 1
        if time.monotonic() - start_time > max_time - 2:
            break

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.1)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            _, g, c, c_max = lse_value_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                last_improve = best_cmax
                stall_count = 0

        if it % 80 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Backtracking line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        c_new = c
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            a_new = candidate
            c_new = c_ls
            cmax_new = cm_ls
            if cm_ls <= c_max * 1.001 or ls == 11:

            alpha_ls *= 0.5

            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        _, g_new, _, _ = lse_value_and_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)

        g = g_new

        c_max = cmax_new

            s_hist = []
            y_hist = []
            _, g, c, c_max = lse_value_and_grad(a, N, beta)
            stall_count = 0

def toppk_block(a_init, n, S, N, max_time, rng, top_k_init=10, top_k_max=256, momentum=0.9):
    """Top-k subgradient with momentum for polishing. Returns (best_a, best_cmax)."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    velocity = np.zeros(n)

    while time.monotonic() - start_time < max_time:
        it += 1
        if time.monotonic() - start_time > max_time - 1:
            break

        top_k = min(top_k_max, top_k_init + it // 50)
        c = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        fsize = 1 << (3 * n).bit_length()
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        velocity = momentum * velocity + (1 - momentum) * (-grad / gnorm)
        a = project_simplex(a + step * velocity, S)

    return best_a, best_cmax


def spectral_step(a, N, top_k, alpha):
    """One spectral shaping step."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def make_init_powerlaw(n, alpha, rng):
    """Power law initialization: a_j = ((j+1)/(n+1))^alpha."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = (j / (n + 1)) ** alpha
    noise = 0.01 * rng.standard_normal(n)
    a = a + noise * a.mean()
    return np.clip(a, 0.001, None)


def make_init_exp(n, rate, rng):
    """Exponential initialization: a_j = exp(rate * j/n)."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(rate * j / n)
    noise = 0.01 * rng.standard_normal(n)
    a = a + noise * a.mean()
    return np.clip(a, 0.001, None)


def make_init_gabor(n, center, width, rng):
    """Gaussian-bump initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-((j - center) ** 2) / (2 * width ** 2))
    noise = 0.02 * rng.standard_normal(n)
    a = a + noise * a.mean()
    return np.clip(a, 0.001, None)


def make_init_linear_ramp(n, rng):
    """Linear ramp: a_j = j+1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    noise = 0.005 * rng.standard_normal(n)
    a = a + noise * a.mean()
    return np.clip(a, 0.001, None)


def make_init_flat(n, rng):
    """Flat with small noise."""
    a = np.ones(n) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-n exploration with L-BFGS, top-k subgradient, and spectral shaping.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_best(a_candidate, n_candidate, label=""):
        nonlocal global_best, global_a, global_n
        a_cand = safe_normalize(a_candidate, n_candidate, np.sqrt(2.0 * n_candidate))
        N_temp = 1 << (2 * n_candidate - 1).bit_length()
        c = compute_conv(a_cand, N_temp)
        cmax = c.max()
        s = a_cand.sum()
        obj = 2.0 * n_candidate * cmax / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a_cand.copy()
            global_n = n_candidate
            if label:
                print("  [%s] n=%d obj=%.6f" % (label, n_candidate, global_best))
        return obj

    # Determine which n values to try based on budget
    n_candidates = [2048, 4096, 8192]
    if budget_s > 1800:
        n_candidates.append(12288)

    # Try to use the previous solution as a starting point
    previous_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512:
            previous_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Phase 1: Quick exploration across n values and initializations
    print("=== Phase 1: Multi-n exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.2)

    for n in n_candidates:
        if time.monotonic() > deadline - 180:
            break
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        per_init = max(5.0, phase1_budget / (n_candidates.index(n) + 1) / 5)

        inits = []
        if previous_seq is not None and len(previous_seq) == n:
            inits.append(("prev", previous_seq.copy()))
        inits.append(("ramp", make_init_linear_ramp(n, rng)))
        inits.append(("pow1", make_init_powerlaw(n, 1.0, rng)))
        inits.append(("pow2", make_init_powerlaw(n, 2.0, rng)))
        inits.append(("exp", make_init_exp(n, 2.0, rng)))

        for name, arr in inits:
            if time.monotonic() > deadline - 180:
                break
            a_norm = safe_normalize(arr, n, S)
            rem = max(3.0, per_init - (time.monotonic() - t1) * 0.1)
            a_l, sc_l = lbfgs_block(a_norm, n, S, N, min(rem, 40.0), rng,
                                     beta_init=5.0, beta_max=50.0, m_lbfgs=8)
            update_best(a_l, n, "P1/" + name)

    print("Phase 1 done. Best: %.6f (n=%d)" % (global_best, global_n))

    if global_a is None:
        # Fallback
        n = 4096
        global_a = make_init_linear_ramp(n, rng)
        global_a = safe_normalize(global_a, n, np.sqrt(2.0 * n))
        global_n = n
        global_best = 2.0

    # Phase 2: Focused optimization on the best n
    print("=== Phase 2: Focused L-BFGS optimization ===")
    t2 = time.monotonic()
    n_best = global_n
    S = np.sqrt(2.0 * float(n_best))
    N = 1 << (2 * n_best - 1).bit_length()
    phase2_budget = min(1500.0, (deadline - t2) * 0.6)

    a_curr = global_a.copy()
    stall = 0
    n_blocks = 0

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(100.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        n_blocks += 1
        beta_start = 8.0 + 3.0 * min(stall, 10)
        beta_end = min(400.0, 40.0 * (1 + 0.3 * stall))
        a_new, sc_new = lbfgs_block(a_curr, n_best, S, N, block_time, rng,
                                     beta_init=beta_start, beta_max=beta_end, m_lbfgs=12)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2 block %d] %.6f (beta %.0f)" % (n_blocks, global_best, beta_start))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 3:
                # Bigger perturbation
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n_best), n_best), S)
                stall = 0

    # Phase 3: Top-k subgradient polishing
    print("=== Phase 3: Top-k subgradient polishing ===")
    t3 = time.monotonic()
    remaining3 = deadline - t3 - 60
    if remaining3 > 20:
        a_ng, sc_ng = toppk_block(global_a, n_best, S, N, min(remaining3 * 0.5, 400.0),
                                   rng, top_k_init=10, top_k_max=256)
        update_best(a_ng, n_best, "topk")

    # Phase 4: Spectral shaping
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 30
    if remaining4 > 10:
        a = global_a.copy()
        S4 = a.sum()
        for _ in range(int(min(200, remaining4 * 2))):
            if time.monotonic() > deadline - 30:
                break
            c = compute_conv(a, N)
            excess = max(0.0, c.max() / (S4 / n_best) - 1.0)
            alpha = 0.1 * (0.2 + 0.8 * min(excess, 1.0))
            a = spectral_step(a, N, top_k=8, alpha=alpha)
        update_best(a, n_best, "spectral")

    # Phase 5: Final high-beta L-BFGS
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 15:
        a_fin, sc_fin = lbfgs_block(global_a, n_best, S, N, min(remaining5, 200.0), rng,
                                     beta_init=50.0, beta_max=500.0, m_lbfgs=10)
        update_best(a_fin, n_best, "final_LBFGS")

    # Phase 6: Try a different n with the best shape as template
    remaining6 = deadline - time.monotonic() - 10
    if remaining6 > 120:
        # Interpolate best shape to a different n
        for n_try in [4096, 8192, 16384]:
            if n_try == n_best or time.monotonic() > deadline - 40:
                continue
            S_try = np.sqrt(2.0 * float(n_try))
            N_try = 1 << (2 * n_try - 1).bit_length()
            # Resample the best sequence to new length
            old_idx = np.linspace(0, 1, n_best)
            new_idx = np.linspace(0, 1, n_try)
            a_resampled = np.interp(new_idx, old_idx, global_a)
            a_resampled = safe_normalize(a_resampled, n_try, S_try)
            a_try, sc_try = lbfgs_block(a_resampled, n_try, S_try, N_try,
                                         min(remaining6 * 0.3, 80.0), rng,
                                         beta_init=10.0, beta_max=100.0, m_lbfgs=10)
            update_best(a_try, n_try, "re-n")

    # Verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f (internal: %.6f)" % (ver, global_best))
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, best=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
