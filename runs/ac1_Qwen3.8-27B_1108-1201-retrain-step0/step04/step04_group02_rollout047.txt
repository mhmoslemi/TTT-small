def project_simplex(v, S):

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:

def weighted_topk_subgradient(a, c, K, temperature=1.0):
    """Weighted subgradient: peaks weighted by softmax of (c - max_c)/temperature."""
    n = len(a)
    K = min(K, len(c))
    idx = np.argsort(c)[::-1][:K]
    c_vals = c[idx]
    c_max = c_vals[0]
    weights = np.exp(-(c_max - c_vals) / max(temperature, 1e-10))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, j in zip(weights, idx):
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += w * a[j - i_vals]
    return grad


def topk_subgradient(a, c, k):
    """Uniform top-k subgradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]

    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += a[j - i_vals]
    return grad / k


def mirror_update(a, grad, eta, S):
    """Multiplicative (mirror) descent step."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)

        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    n = 8192
    S = np.sqrt(2.0 * n)

    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))
    t0 = time.monotonic()

    # Build initialization list
    inits = []
    if given is not None and len(given) == n:
        inits.append(("given", given.copy()))
    # Always include given even if different length (we'll resize)
    if given is not None and len(given) != n:
        # Interpolate to target length
        from numpy import interp
        old_tt = np.linspace(0, 1, len(given))
        new_tt = np.linspace(0, 1, n)
        resized = interp(new_tt, old_tt, given)
        inits.append(("given_resized", np.maximum(resized, 0.01)))

    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits):
        inits.append(("init_%d" % i, g))

    # Add structured perturbations of the given
    if given is not None and len(given) == n:
        for i in range(4):
            tt = np.arange(n) / n
            if i == 0:
                mod = 1.0 + 0.2 * np.cos(4 * np.pi * tt)
            elif i == 1:
                mod = 1.0 + 0.15 * (2 * tt - 1) ** 2
            elif i == 2:
                mod = 1.0 + 0.1 * np.sin(6 * np.pi * tt + rng.uniform(0, 6.28))
            else:
                mod = 1.0 + 0.3 * np.exp(-20.0 * (tt - rng.uniform(0.2, 0.8)) ** 2)
            pert = np.maximum(given * mod, 0.01)
            inits.append(("pert_%d" % i, pert))

    n_inits = len(inits)
    print("Total initializations: %d" % n_inits)

    # Phase 1: Quick exploration (40% of time)
    phase1_time = max(30.0, min(600.0, (deadline - time.monotonic()) * 0.40))
    per_init = phase1_time / n_inits
    print("=== Phase 1: Exploration (%d inits, total %.0fs, per-init %.1fs) ===" % (
        n_inits, phase1_time, per_init))

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - t0) * 0.50:
            break
        if per_init < 3:
            continue
        arr = np.maximum(arr, 0.01)
        if len(arr) != n:
            continue

        a_opt, score = optimize_adaptive(arr, n, S, per_init, rng, verbose=False)
        a_norm = normalize_sum(a_opt, n)
        real_score = compute_objective(a_norm)

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] obj=%.7f (elapsed %.0fs)" % (name, global_best, time.monotonic() - t0))

    print("=== After Phase 1: best=%.7f, elapsed=%.0fs ===" % (
        global_best, time.monotonic() - t0))

    # If no good result from exploration, use the given directly
    if global_a is None:
        if given is not None:
            global_a = normalize_sum(given, n) if len(given) == n else np.ones(n) * S / n
        else:
            global_a = np.ones(n) * S / n
        global_best = compute_objective(global_a)
        print("  Fallback to uniform: obj=%.7f" % global_best)

    # Phase 2: Main optimization on best (35% of time)
    remaining = deadline - time.monotonic()
    phase2_time = max(60.0, remaining * 0.50)
    print("=== Phase 2: Main optimization (remaining %.0fs) ===" % phase2_time)

    n_best = len(global_a)
    S_best = np.sqrt(2.0 * n_best)

    a_main, s_main = optimize_adaptive(global_a, n_best, S_best, phase2_time, rng, verbose=True)
    s_main = compute_objective(normalize_sum(a_main, n_best))
    if s_main < global_best:
        global_best = s_main
        global_a = normalize_sum(a_main, n_best)
        print("  After main optimization: %.7f" % global_best)

    # Phase 3: Polishing (20% of time)
    remaining = deadline - time.monotonic()
    phase3_time = max(30.0, remaining * 0.60)
    print("=== Phase 3: Polishing (remaining %.0fs) ===" % phase3_time)

    n_best = len(global_a)
    S_best = np.sqrt(2.0 * n_best)

    a_pol, s_pol = polish_peaks(global_a, n_best, S_best, phase3_time, rng, verbose=True)
    s_pol = compute_objective(normalize_sum(a_pol, n_best))
    if s_pol < global_best:
        global_best = s_pol
        global_a = normalize_sum(a_pol, n_best)
        print("  After polishing: %.7f" % global_best)

    # Phase 4: Final micro-optimization with different K schedule
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("=== Phase 4: Final micro-opt (remaining %.0fs) ===" % remaining)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Try a few more optimization runs with slightly different parameters
        for trial in range(3):
            if remaining < 20:
                break
            alloc = min(remaining * 0.4, 120.0)

            # Slight perturbation then re-optimize
            noise = 0.002 * S_best / np.sqrt(n_best)
            a_pert = project_simplex(global_a + rng.normal(0, noise, n_best), S_best)

            a_try, s_try = optimize_adaptive(a_pert, n_best, S_best, alloc, rng, verbose=False)
            a_try = normalize_sum(a_try, n_best)
            s_try = compute_objective(a_try)
            if s_try < global_best:
                global_best = s_try
                global_a = a_try.copy()
                print("  Trial %d: %.7f" % (trial, global_best))
            remaining = deadline - time.monotonic()

    # Final: one more polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_fin, s_fin = polish_peaks(global_a, n_best, S_best, min(remaining * 0.8, 300.0), rng, verbose=False)
        a_fin = normalize_sum(a_fin, n_best)
        s_fin = compute_objective(a_fin)
        if s_fin < global_best:
            global_best = s_fin
            global_a = a_fin.copy()
            print("  Final polish: %.7f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.7f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = compute_objective(global_a)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = compute_objective(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = compute_objective(global_a)

    print("Returning sequence of length %d, obj=%.7f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
