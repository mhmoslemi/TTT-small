import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(x, y):
    """Full linear cross-convolution via FFT."""
    n = len(x) + len(y) - 1
    fsize = 1 << (n - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    return np.real(ifft(X * Y))[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lp_gradient(a, p=4):
    """Gradient of Lp norm of self-convolution."""
    n = len(a)
    b = fft_convolve_self(a)
    b = np.maximum(b, 0.0)
    b_p = b ** p
    sum_bp = b_p.sum()
    if sum_bp < 1e-30:
        return np.zeros(n), 0.0
    w = b ** (p - 1)
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    factor = (sum_bp) ** (1.0 / p - 1.0)
    grad = factor * cross
    obj = sum_bp ** (1.0 / p)
    return grad, obj


def peak_shave_gradient(a, c, k=16):
    """Exact subgradient from the top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.maximum(c[idx] - c.mean(), 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):

    if s < 0.01:

    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

sum_j a_j * g_{k-j} <= M

For a fixed a, the matrix A where A[k, j] = a_j if 0 <= k-j < n else 0 is a Toeplitz-like matrix. We can represent it compactly.

In CVXPY, we can use sparse matrices. The matrix A has non-zero entries only where 0 <= k-j < n, i.e., j is in [max(0, k-n+1), min(k, n-1)]. For each row k, there are at most n non-zero entries. The total number of non-zeros is sum_k min(n, k+1, 2n-1-k) ≈ n^2.

So the matrix A is n^2 * (2n-1) in size... no wait, A is (2n-1) x n with about n^2 non-zeros. For n=8192, that's about 67 million non-zeros, which is too many for most LP solvers.

A[k, j] = a_j if 0 <= k-j < n... no, let me redefine.

The coefficient of g_i in this constraint is a_{k-i} (if 0 <= k-i < n), else 0.

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Multi-phase optimization:
1. Lp-norm (p=4) FISTA for smooth bulk optimization
2. LSE-FISTA with adaptive beta for peak targeting  
3. Exact peak shaving with top-k lags
4. Final polishing with very high beta
Includes backtracking line search and periodic restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_obj_and_grad(a, c, beta):
    """Compute LSE objective and gradient."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    obj = cm + np.log(tot) / beta
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad


def lp_obj_and_grad(a, p=4):
    """Compute Lp objective and gradient."""
    n = len(a)
    b = fft_convolve_self(a)
    b = np.maximum(b, 0.0)
    bp = b ** p
    sum_bp = bp.sum()
    if sum_bp < 1e-30:
        return 0.0, np.zeros(n)
    obj = sum_bp ** (1.0 / p)
    w = b ** (p - 1)
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    factor = (sum_bp) ** (1.0 / p - 1.0)
    grad = factor * cross
    return obj, grad


def peak_grad(a, c, k=32):

    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]
