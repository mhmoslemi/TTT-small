```python
import time
from scipy.fft import fft, ifft

def compute_score(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

    F(a) = (1/beta) * log(sum_k exp(beta * c[k]))
    dF/da[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
    where w[k] = exp(beta*c[k]) / sum exp(beta*c[l])

    size = 2 * n - 1
    A = fft(a, N)
    c = np.real(ifft(A * A))[:size]
    
    # Compute softmax weights (numerically stable)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
```

```python
def peak_reduction(a, n, S, N, rng, n_iter, target=1.50):
    """
    Iteratively reduce the peak of the convolution by:
    1. Reducing a at positions that contribute to the peak
    2. Adding mass back at "safe" positions
    """
    a = a.copy()
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        # Compute convolution
        A = fft(a, N)
        c = np.real(ifft(A * A))[:2*n-1]
        c_max = c.max()
        k_peak = int(np.argmax(c))
        
        score = 2.0 * n * c_max / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if score <= target:
            break
        
        # Influence vector: how much each a[j] contributes to c[k_peak]
        # c[k_peak] = sum_j a[j] * a[k_peak - j]
        # d c[k_peak] / d a[j] = 2 * a[k_peak - j]
        influence = np.zeros(n)
        for j in range(n):
            k = k_peak - j
            if 0 <= k < n:
                influence[j] = a[k]
        
        # "Safety" vector: how much each position contributes to the max
        # (approximate: max over all lags)
        safety = np.zeros(n)
        # For position j, the contribution to c[k] is a[k-j]
        # The "cost" of increasing a[j] is max_k 2*a[k-j]
        # Approximate by: max over a few top lags
        top_k = np.argsort(c)[::-1][:5]
        for k in top_k:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    safety[j] = max(safety[j], a[idx])
        
        # Reduce high-influence, low-safety positions
        # and increase low-influence, high-safety... wait, that's backward
        # We want to:
        # - Decrease a[j] where influence is high (reduces peak)
        # - Increase a[j] where safety is low (less cost)
        
        # Compute a "score" for each position: high = should decrease, low = should increase
        # score = influence / (safety + epsilon)
        # Actually, let's think of it as: we want to move mass from high-influence to low-safety
        
        # Simple approach: a_new = a - alpha * influence + beta * (1/safety_normalized)
        # Then project
        
        alpha = 0.1  # reduction strength
        a_new = a - alpha * influence / max(influence.max(), 1e-15)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        a = a_new
    
    return best_a, best_score
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.max() < 0:
        return np.ones(n) * S / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, N, beta, top_k=None):
    """
    Compute gradient of (1/beta) * log(sum_k exp(beta * c[k]))
    or if top_k is given, (1/beta) * log(sum_{k in top_k} exp(beta * c[k]))
    """
    n = len(a)
    size = 2 * n - 1
    A = fft(a, N)
    c = np.real(ifft(A * A))[:size]
    
    if top_k is not None and top_k < size:
        # Use only top-k lags
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_subset = c[idx]
        c_max = c_subset.max()
        exp_bc = np.exp(beta * (c_subset - c_max))
        w_full = np.zeros(size)
        w_full[idx] = exp_bc / exp_bc.sum()
    else:
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        w_full = exp_bc / exp_bc.sum()
    
    # Gradient: dF/da[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is 2 * cross_correlation(w_full, a)
    # Using FFT: 2 * IFFT(FFT(w_full_padded) * conj(FFT(a_padded)))
    w_pad = np.zeros(N)
    w_pad[:size] = w_full
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    grad = 2.0 * np.real(ifft(fft(w_pad) * np.conj(fft(a_pad))))[:n]
    
    return grad, c, c_max

def optimize_nesterov(a_init, n, S, N, max_time, rng, 
                       beta_start=0.5, beta_end=100.0,
                       top_k=None, target=1.50):
    """
    Nesterov-accelerated gradient descent on the LSE-smoothed objective.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Anneal beta
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.3)
        
        # Ensure y is valid
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y * (S / sy)
        
        # Compute gradient
        grad, c, c_max = lse_gradient(y, N, beta, top_k=top_k)
        
        # Check score
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
        
        if best_score <= target:
            break
        
        # Gradient step
        # Step size: use a schedule that starts large and decreases
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        
        # Step size heuristic: we want to move a meaningful fraction of the "spread"
        # of a, but not too much
        spread = a.max() - a.min() if n > 1 else 1.0
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.005 * it))
        
        x = y - step * grad
        x = project_simplex(x, S)
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            break
        x = x * (S / sx)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y * (S / sy)
        t = t_new
        
        # Periodic random restart (mild)
        if it % 1000 == 500 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0  # Reset momentum
    
    return best_a, best_score
```
