from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 1 << (2*n - 1 - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}"""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    mask = w > (cumsum - S) / idx
    rho = np.sum(mask)
    if rho == 0:
        return np.full(n, S/n)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    diff = c - cm
    # Clip for numerical stability
    diff = np.clip(beta * diff, -500, 500)
    e = np.exp(diff)
    w = e / e.sum()
    # Compute 2 * correlate(w, a)
    size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def power_sum_gradient(a, c, p):
    """Gradient of sum(c_k^p)"""
    n = len(a)
    if p == 1:
        w = np.ones_like(c)
    else:
        w = p * np.power(np.maximum(c, 0), p-1)

    w = w / w.max()  # Normalize for numerical stability

    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    mean_c = c.mean()

    A = fft(a, size)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        if gnorm < 1e-14:
            break
        
        # Step
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_mom * t_mom))
        y = x_new + ((t_mom - 1) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        a = x_new

            y = project_simplex(y + rng.normal(0, noise, len(a)), S)
            t_mom = 1.0
            last_improve = it

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines power-sum FISTA (new smooth objective), LSE FISTA, peak shaving,
and coordinate perturbation for final polishing.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = w > (cumsum - S) / idx
    rho = int(np.sum(mask))
    if rho == 0:
        return np.full(n, S / n)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_sum_gradient(a, c, p):
    """Gradient of sum(c_k^p) with respect to a."""
    n = len(a)
    c_pos = np.maximum(c, 1e-30)
    if p == 1:

    else:

    size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    return 2.0 * wmax * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=32):
    """Exact subgradient from top-k lags weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]

    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    size = 1 << (len(w_full) + n - 1 - 1).bit_length()
    W = fft(w_full, size)
    A = fft(a, size)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((b_k - T)^p_+)."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    wsum = excess.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    if p == 1.0:

    else:

    size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    return 2.0 * nrm * np.real(ifft(W * np.conj(A)))[:n]

        if s_try < 1e-15:
            continue
        a_try = project_simplex(a_try, S)

        c_try = fft_convolve_self(a_try)
        if c_try.max() < cur_max - 1e-14:
            a = a_try
            c = c_try
            cur_max = c.max()

    return a, cur_max


def generate_inits(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None:
        inits.append(given.copy())

    inits.append(project_simplex(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01), np.sqrt(2 * n)))
    inits.append(project_simplex(np.sin(np.pi * tt) ** 2 + 0.01, np.sqrt(2 * n)))
    inits.append(project_simplex(np.exp(-4 * (tt - 0.5) ** 2 / 0.09) + 0.01, np.sqrt(2 * n)))
    inits.append(project_simplex(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5), np.sqrt(2 * n)))
    inits.append(project_simplex(rng.random(n) + 0.1, np.sqrt(2 * n)))
    inits.append(project_simplex(np.exp(rng.normal(0, 0.3, n)), np.sqrt(2 * n)))

    return inits


def upscale_sequence(a_small, n_large):
    """Upscale by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: exploration, power-sum FISTA, excess annealing, LSE FISTA,
    peak shaving, and coordinate perturbation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 25.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        given = np.clip(given, 0.0, 1000.0)
        if given.size >= 64 and given.sum() > 0.01:
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
        else:
            given = None
    except Exception:
        pass

    # Determine working length
    if given is not None:
        n = len(given)
        S = given.sum()
    else:
        n = 8192
        S = np.sqrt(2.0 * n)
        given = np.ones(n) * S / n

    a_best = given.copy()
    n_best = n
    S_best = S
    c_best = fft_convolve_self(a_best)
    best_obj = 2.0 * n_best * c_best.max() / (S_best * S_best)
    print("Initial: n=%d, obj=%.8f, max_b=%.6f" % (n_best, best_obj, c_best.max()))

    t_total = deadline - time.monotonic()
    if t_total < 60:
        t_total = 60

    # === Phase 1: Quick exploration with diverse inits (12%) ===
    t1 = t_total * 0.12
    if t1 > 10:
        print("\n--- Phase 1: Exploration (%.0fs) ---" % t1)
        inits = generate_inits(n, rng, given=given)
        t_per = t1 / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - t_total * 0.2:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - t_total * 0.15)
            if t_alloc < 5:
                break
            a_test = project_simplex(init, S)
            a_opt, m = fista_run(a_test, S, t_alloc, rng, mode='lse', beta=50.0)
            obj_test = 2.0 * n * m / (S * S)
            if obj_test < best_obj:
                best_obj = obj_test
                a_best = a_opt.copy()
                n_best = n
                S_best = S
                print("  Init %d: obj=%.8f" % (i, best_obj))

    # === Phase 2: Power-sum FISTA with increasing p (28%) ===
    t2 = t_total * 0.28
    if t2 > 15:
        print("\n--- Phase 2: Power-sum FISTA (%.0fs) ---" % t2)
        a_work = a_best.copy()
        S_work = S_best
        n_work = n_best
        p_values = [2, 4, 8, 16, 32]
        t_per_p = t2 / len(p_values)
        for p in p_values:
            if time.monotonic() > deadline - t_total * 0.15:
                break
            t_alloc = min(t_per_p, deadline - time.monotonic() - t_total * 0.12)
            if t_alloc < 5:
                break
            a_opt, m = fista_run(a_work, S_work, t_alloc, rng, mode='power', p=p)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  p=%d: obj=%.8f, max_b=%.6f" % (p, best_obj, m))
            a_work = a_opt.copy()

    # === Phase 3: Excess annealing (20%) ===
    t3 = t_total * 0.20
    if t3 > 10:
        print("\n--- Phase 3: Excess annealing (%.0fs) ---" % t3)
        a_work = a_best.copy()
        S_work = S_best
        n_work = n_best
        c_work = fft_convolve_self(a_work)
        T_start = c_work.max()
        T_end = c_work.mean() + 0.02 * (c_work.max() - c_work.mean())
        
        y = a_work.copy()
        t_mom = 1.0
        start = time.monotonic()
        it = 0
        best_m = c_work.max()
        best_a3 = a_work.copy()
        last_improve = 0

        while time.monotonic() - start < t3:
            it += 1
            progress = min(1.0, (time.monotonic() - start) / t3)
            T = T_start + (T_end - T_start) * (progress ** 0.6)
            T = max(T, T_end)

            c = fft_convolve_self(y)
            if c.max() < best_m - 1e-12:
                best_m = c.max()
                best_a3 = y.copy()
                last_improve = it

            grad = excess_gradient(y, c, T, p=2.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 0.5 / gnorm

            x_new = project_simplex(y - step * grad, S_work)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a_work)
            y = np.maximum(y, 0.0)
            a_work = x_new
            t_mom = t_new

            if it % 1500 == 0:
                noise = 0.001 * S_work / np.sqrt(n_work) * (1.0 - progress)
                if noise > 1e-10:
                    y = project_simplex(y + rng.normal(0, noise, n_work), S_work)
                    t_mom = 1.0

            if it - last_improve > 2000:
                noise = 0.002 * S_work / np.sqrt(n_work)
                y = project_simplex(y + rng.normal(0, noise, n_work), S_work)
                t_mom = 1.0
                last_improve = it

        obj3 = 2.0 * n_work * best_m / (S_work * S_work)
        if obj3 < best_obj:
            best_obj = obj3
            a_best = best_a3.copy()
            n_best = n_work
            S_best = S_work
            print("  Excess: obj=%.8f, max_b=%.6f" % (best_obj, best_m))

    # === Phase 4: LSE FISTA with increasing beta (20%) ===
    t4 = t_total * 0.20
    if t4 > 10:
        print("\n--- Phase 4: LSE FISTA (%.0fs) ---" % t4)
        a_work = a_best.copy()
        S_work = S_best
        n_work = n_best
        betas = [100, 300, 1000, 3000]
        t_per_beta = t4 / len(betas)
        for beta in betas:
            if time.monotonic() > deadline - t_total * 0.12:
                break
            t_alloc = min(t_per_beta, deadline - time.monotonic() - t_total * 0.10)
            if t_alloc < 5:
                break
            a_opt, m = fista_run(a_work, S_work, t_alloc, rng, mode='lse', beta=beta)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  beta=%d: obj=%.8f, max_b=%.6f" % (beta, best_obj, m))
            a_work = a_opt.copy()

    # === Phase 5: Peak shaving (10%) ===
    t5 = t_total * 0.10
    if t5 > 5:
        print("\n--- Phase 5: Peak shaving (%.0fs) ---" % t5)
        a_work = a_best.copy()
        S_work = S_best
        n_work = n_best
        for k in [8, 16, 32]:
            t_alloc = t5 / 3
            if time.monotonic() > deadline - t_total * 0.08:
                break
            a_opt, m = fista_run(a_work, S_work, t_alloc, rng, mode='peak', k=k)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  k=%d: obj=%.8f" % (k, best_obj))
            a_work = a_opt.copy()

    # === Phase 6: Final polishing with remaining time ===
    t6 = deadline - time.monotonic()
    if t6 > 10:
        print("\n--- Phase 6: Final polishing (%.0fs) ---" % t6)
        a_work = a_best.copy()
        S_work = S_best
        n_work = n_best

        # Very high beta LSE
        t_alloc = t6 * 0.40
        if t_alloc > 5:
            a_opt, m = fista_run(a_work, S_work, t_alloc, rng, mode='lse', beta=5000.0)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  Final LSE: obj=%.8f" % best_obj)
            a_work = a_opt.copy()

        # Peak shave with large k
        t_remain = deadline - time.monotonic()
        if t_remain > 5:
            a_opt, m = fista_run(a_work, S_work, t_remain * 0.4, rng, mode='peak', k=64)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  Final peak: obj=%.8f" % best_obj)
            a_work = a_opt.copy()

        # Coordinate perturbation
        t_remain = deadline - time.monotonic()
        if t_remain > 5:
            print("  Coordinate perturbation: %.0fs" % t_remain)
            a_opt, m = coordinate_perturb(a_work, S_work, t_remain, rng, n_trials=100000)
            obj_opt = 2.0 * n_work * m / (S_work * S_work)
            if obj_opt < best_obj:
                best_obj = obj_opt
                a_best = a_opt.copy()
                n_best = n_work
                S_best = S_work
                print("  Coord pert: obj=%.8f" % best_obj)

    # Final safety check and return
    a_best = np.clip(a_best, 0.0, 1000.0)
    final_S = a_best.sum()
    if final_S < 0.01:
        a_best = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in a_best]
    ver = evaluate_sequence(final_list)
    print("\nFinal verification: obj=%.8f (n=%d)" % (ver, len(a_best)))

    if not np.isfinite(ver):
        print("  Rejected! Using fallback.")
        a_best = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in a_best]

    print("Returning sequence of length %d, obj=%.8f" % (len(final_list), ver))
    return final_list
