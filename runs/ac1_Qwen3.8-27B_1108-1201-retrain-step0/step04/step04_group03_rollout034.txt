def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:

    return a * (S / s)

"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-scale FISTA with adaptive LSE,
periodic LP refinement, and sinusoidal perturbation post-processing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.003, beta_max=2000.0,
              repulsion_freq=300, repulsion_lr=0.05, noise_every=4000, noise_scale_base=0.003):
    """
    FISTA with LSE smoothing and periodic noise/repulsion.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        # Adaptive: if stuck, reduce step and add noise
        stuck = it - last_improve_it
        if stuck > 3000:
            step *= 0.3
            ns = noise_scale_base * S / np.sqrt(n) * np.exp(-stuck / 10000)
            y = project_simplex(y + rng.normal(0, max(ns, 1e-6), n), S)
            t_mom = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic repulsion (multiplicative)
        if it % repulsion_freq == 0 and it > 200:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(40.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 0)
                y = project_simplex(y, S)
                t_mom = 1.0

        # Periodic noise
        if it % noise_every == 0:
            decay = np.exp(-it / 30000.0)
            ns = noise_scale_base * S / np.sqrt(n) * decay
            if ns > 1e-7:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()

    return best_a, best


def solve_lp_best_response(f, n, S, top_k=100, solver='SCS', time_limit=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 4:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g

    return None


def upsample(a, new_n):
    """Upsample sequence by interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def sinusoidal_perturbation(f, n, S, rng, num_trials=200, max_eps=0.1):
    """
    Try small sinusoidal perturbations: f * (1 + eps*sin(2*pi*k*t + phase)).
    """
    best = f.copy()
    best_val = fft_convolve_self(f).max()
    t = np.linspace(0, 1, n, endpoint=False)

    for _ in range(num_trials):
        k = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        eps = rng.uniform(-max_eps, max_eps)
        
        mod = 1.0 + eps * np.sin(2 * np.pi * k * t + phase)
        f_trial = f * mod
        f_trial = np.maximum(f_trial, 0)
        f_trial = project_simplex(f_trial, S)
        
        c_trial = fft_convolve_self(f_trial)
        if c_trial.max() < best_val:
            best_val = c_trial.max()
            best = f_trial.copy()

    return best, best_val


def run_lp_polish(f, n, S, max_time, rng, lp_every_iterations=1, top_k=100):
    """
    Repeated LP-based best response with convex combination.
    """
    a = f.copy()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        g = solve_lp_best_response(a, n, S, top_k=top_k)
        if g is None:
            break

        # Try different fractions
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - frac) * a + frac * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = project_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best:
                best = c_trial.max()
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                break

        if not improved:
            # Try adding noise
            ns = 0.002 * S / np.sqrt(n)
            a_trial = project_simplex(a + rng.normal(0, ns, n), S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best:
                best = c_trial.max()
                best_a = a_trial.copy()
                a = a_trial.copy()
            else:
                break

    return best_a, best

"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-scale FISTA with high-beta LSE,
LP-based best response, sinusoidal perturbation, and adaptive noise scheduling.
Key improvements: much higher max beta (10000), aggressive LP usage,
structured perturbation search, and careful time management.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_val(a, n):
    """Compute the objective 2n*max(R)/sum(a)^2."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002,
              beta_max=5000.0, repulsion_freq=400, repulsion_lr=0.04,
              noise_every=3000, noise_base=0.003, min_improve_every=2000):
    """
    FISTA with LSE smoothing, adaptive beta, periodic repulsion and noise.
    Returns (best_a, best_conv_Max).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        stuck = it - last_improve_it
        if stuck > min_improve_every:
            step *= max(0.2, 1.0 - stuck / (5.0 * min_improve_every))
            ns = noise_base * S / np.sqrt(n) * np.exp(-stuck / (2.0 * min_improve_every))
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
            t_mom = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic multiplicative repulsion
        if it % repulsion_freq == 0 and it > 100:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(50.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                scale = rep / max(y.max(), 1e-10)
                y = y * np.exp(-repulsion_lr * scale)
                y = np.maximum(y, 0)
                y = project_simplex(y, S)
                t_mom = 1.0

        # Periodic noise for exploration
        if it % noise_every == 0:
            decay = np.exp(-it / (4.0 * noise_every))
            ns = noise_base * S / np.sqrt(n) * decay
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def solve_lp_best_response(f, n, S, top_k=120, solver='SCS', time_limit=12.0):
    """
    LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 5:
        return None

    if g_var.value is not None and np.isfinite(g_var.value).all():
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def run_hybrid(a_init, n, S, max_time, rng, lp_every=150, lp_top_k=100,
               beta_init=20.0, beta_rate=0.002, beta_max=5000.0):
    """
    FISTA with periodic LP corrections.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    lp_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # LP correction
        if it % lp_every == 0 and (time.monotonic() - start) < max_time * 0.85:
            t0 = time.monotonic()
            g_lp = solve_lp_best_response(y, n, S, top_k=lp_top_k)
            lp_dt = time.monotonic() - t0
            lp_count += 1

            if g_lp is not None:
                c_lp = fft_convolve_self(g_lp)
                improved = False
                for frac in [1.0, 0.6, 0.4, 0.2, 0.1]:
                    y_t = (1.0 - frac) * y + frac * g_lp
                    y_t = np.maximum(y_t, 0)
                    y_t = project_simplex(y_t, S)
                    c_t = fft_convolve_self(y_t)
                    if c_t.max() < best:
                        best = c_t.max()
                        best_a = y_t.copy()
                        y = y_t.copy()
                        a = y_t.copy()
                        t_mom = 1.0
                        last_improve_it = it
                        improved = True
                        if lp_dt > 1:
                            print("    LP %d (frac=%.1f, %.1fs): %.6f" % (lp_count, frac, lp_dt, best))
                        break
                if not improved and lp_dt > 5:
                    print("    LP %d: no improvement (%.1fs)" % (lp_count, lp_dt))

        # FISTA step
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        stuck = it - last_improve_it
        if stuck > 2500:
            step *= 0.3
            ns = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
            t_mom = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Repulsion
        if it % 500 == 0 and it > 300:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(40.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-0.03 * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 0)
                y = project_simplex(y, S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best, lp_count


def sinusoidal_perturbation(f, n, S, rng, num_trials=300, max_eps=0.08):
    """
    Try f * (1 + eps*sin(2*pi*k*t + phase)) for random k, phase, eps.
    Also try mixed-frequency perturbations.
    """
    best = f.copy()
    best_val = fft_convolve_self(f).max()
    t = np.linspace(0, 1, n, endpoint=False)
    improved_count = 0

    for trial in range(num_trials):
        r = rng.random()
        if r < 0.6:
            # Single sinusoid
            k = int(rng.integers(1, max(2, n // 6)))
            phase = rng.uniform(0, 2 * np.pi)
            eps = rng.uniform(-max_eps, max_eps)
            mod = 1.0 + eps * np.sin(2.0 * np.pi * k * t + phase)
        elif r < 0.8:
            # Two sinusoids
            k1 = int(rng.integers(1, max(2, n // 10)))
            k2 = int(rng.integers(k1 + 1, max(k1 + 2, n // 6)))
            p1 = rng.uniform(0, 2 * np.pi)
            p2 = rng.uniform(0, 2 * np.pi)
            e1 = rng.uniform(-max_eps, max_eps)
            e2 = rng.uniform(-max_eps, max_eps)
            mod = 1.0 + e1 * np.sin(2.0 * np.pi * k1 * t + p1) + e2 * np.sin(2.0 * np.pi * k2 * t + p2)
        else:
            # Smooth random perturbation (low-pass filtered noise)
            noise = rng.normal(0, 1, n)
            # Simple low-pass: moving average
            w = max(3, n // 200)
            kernel = np.ones(w) / w
            if w > 1:
                noise = np.convolve(noise, kernel, mode='same')
            noise = noise - noise.mean()
            eps = rng.uniform(-max_eps * 0.5, max_eps * 0.5)
            mod = 1.0 + eps * noise / max(np.std(noise), 1e-10)

        f_trial = f * mod
        f_trial = np.maximum(f_trial, 0)
        f_trial = project_simplex(f_trial, S)

        c_trial = fft_convolve_self(f_trial)
        if c_trial.max() < best_val:
            best_val = c_trial.max()
            best = f_trial.copy()
            improved_count += 1

    if improved_count > 0:
        print("    Sinusoidal: %d improvements out of %d trials" % (improved_count, num_trials))
    return best, best_val


def upsample(a, new_n):
    """Upsample by linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

    # === Phase 6: Final micro-perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a6, val6 = sinusoidal_perturbation(global_a, n, S, rng,
                                            num_trials=min(300, int(remaining * 3)),
                                            max_eps=0.02)
        a6 = normalize_sum(a6, n)
        obj6 = objective_val(a6, n)
        if obj6 < global_best:
            global_best = obj6
            global_a = a6.copy()
        print("  Phase 6 result: %.6f" % global_best)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("=== FINAL: length=%d, verified_score=%.6f ===" % (len(global_a), ver))
        else:
            print("WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        print("FALLBACK used.")

    return [float(x) for x in global_a]
