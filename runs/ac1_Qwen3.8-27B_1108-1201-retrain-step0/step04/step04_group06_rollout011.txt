"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Lp-smoothed Nesterov accelerated
gradient descent with adaptive p, spectral perturbations, and coordinate polish.

Key innovations:
1. Lp-smoothed objective: use ||conv(a,a)||_p as a smooth surrogate for max,
   with p increased adaptively.
2. Efficient O(n log n) gradient computation via FFT cross-correlation.
3. Nesterov acceleration for O(1/t^2) convergence rate.
4. Stable large-p computation using normalized ratios (b_k/b_max)^(p-1).
5. Spectral perturbations for global exploration.
6. Coordinate descent final polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / rho_candidates
    valid = theta <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta_val, 0.0)
    # Fix numerical issues
    result[result < 0] = 0
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference to non-zero elements
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result[result < 0] = 0
    return result


def fft_self_conv(a, N):
    """Compute self-convolution of a using FFT of size N. Returns length 2n-1."""
    n = len(a)
    A = fft(a, N)
    b = np.real(ifft(A * A))
    return b[:2 * n - 1]


def compute_lp_gradient(a, b, p, N):
    """
    Compute gradient of ||b||_p where b = conv(a, a).
    Uses stable formulation for large p.
    """
    n = len(a)
    b_max = b.max()
    if b_max <= 1e-30:
        return np.zeros(n)
    
    # Normalized ratios: r_k = b_k / b_max (stable, in [0, 1])
    r = b / b_max
    
    if p <= 1.001:
        # p ≈ 1: use L1 subgradient
        w = (r > 1e-10).astype(np.float64)
        norm_factor = 1.0
    else:
        # w_k = r_k^(p-1) (stable since r_k <= 1)
        if p > 50:
            # For very large p, use log to avoid underflow of small r_k
            log_r = np.where(r > 1e-300, np.log(np.maximum(r, 1e-300)), -700)
            log_w = (p - 1) * log_r
            max_log_w = log_w.max()
            w = np.exp(log_w - max_log_w)
        else:
            w = np.where(r > 0, r ** (p - 1), 0.0)
        
        # Normalization: (sum r_k^p)^(-(p-1)/p)
        if p > 50:
            log_r2 = np.where(r > 1e-300, np.log(np.maximum(r, 1e-300)), -700)
            log_sum = np.logaddexp.reduce(p * log_r2 - p * 0)  # p*log(r_k)
            # Actually: sum r_k^p = sum exp(p * log(r_k))
            # log(sum exp(x)) = max(x) + log(sum exp(x - max(x)))
            x = p * log_r2
            x_max = x.max()
            log_sum_p = x_max + np.log(np.sum(np.exp(x - x_max)))
            norm_factor = np.exp(-(p - 1) / p * log_sum_p)
        else:
            sum_p = np.sum(r ** p)
            norm_factor = sum_p ** (-(p - 1) / p)
    
    # Gradient: 2 * b_max * norm_factor * sum_k w_k * a_{k-i}
    # sum_k w_k * a_{k-i} = cross-correlation computed via FFT
    A = fft(a, N)
    W = fft(w, N)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    
    grad = 2.0 * b_max * norm_factor * corr
    return grad


def nesterov_lp_descent(a_init, n, N, S, max_time, p_start=4, p_end=4096,
                        step_scale=0.5, rng=None, target_score=1.50):
    """
    Nesterov-accelerated gradient descent on ||conv(a,a)||_p with adaptive p.
    """
    a = a_init.copy()
    a = project_simplex(a, S)
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    start = time.monotonic()
    best_a = a.copy()
    best_bmax = np.inf
    it = 0
    p = p_start
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive p: increase gradually
        frac = min(1.0, (time.monotonic() - start) / max_time)
        p = p_start * (p_end / p_start) ** frac
        # p = p_start * exp(frac * log(p_end/p_start))
        
        # Compute objective at y
        b_y = fft_self_conv(y, N)
        bmax_y = b_y.max()
        
        if bmax_y < best_bmax:
            best_bmax = bmax_y
            best_a = y.copy()
            if best_bmax <= target_score:
                break
        
        # Compute gradient at y
        grad = compute_lp_gradient(y, b_y, p, N)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Step size: adaptive
        step = step_scale * S / (gnorm + 1e-15)
        step = min(step, 0.1 * S / n)  # Cap the step
        
        # Nesterov step
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        x = project_simplex(y - step * grad, S)
        y = x + ((t - 1.0) / t_new) * (x - a_prev)
        y = np.maximum(y, 0.0)
        a_prev = x
        t = t_new
        
        # Periodic check and perturbation
        if it % 500 == 0 and rng is not None:
            b_check = fft_self_conv(y, N)
            if b_check.max() < best_bmax:
                best_bmax = b_check.max()
                best_a = y.copy()
            
            # Small perturbation to avoid stagnation
            if rng.random() < 0.1:
                y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
                t = 1.0
                a_prev = y.copy()
    
    return best_a, best_bmax


def spectral_perturb(a, N, n, S, sigma, rng):
    """Perturb the spectrum of a and project back."""
    A = fft(a, N)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def coordinate_polish(a_init, n, S, N, max_time, rng, n_top=20):
    """
    Coordinate descent on top-contributing coordinates at the peak lag.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_bmax = np.inf
    
    multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 0.1, 0.8, 1.2]
    
    while time.monotonic() - start < max_time:
        b = fft_self_conv(a, N)
        bmax = b.max()
        peak_lag = int(np.argmax(b))
        
        if bmax < best_bmax:
            best_bmax = bmax
            best_a = a.copy()
        
        # Find coordinates that contribute to the peak
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_coords:
            if time.monotonic() - start > max_time:
                break
            i = int(idx)
            j = peak_lag - i
            if j < 0 or j >= n:
                continue
            
            orig_val = a[i]
            if orig_val < 1e-15:
                continue
            
            best_frac = 1.0
            best_local_bmax = bmax
            
            for frac in multipliers:
                a[i] = orig_val * frac
                # Quick recompute of affected lags
                new_b = b.copy()
                # Lags affected: k = i to i+n-1 (and k = ... let me think)
                # Changing a[i] affects b_k for k where a[i] participates:
                # b_k includes a[i]*a[k-i] and a[k-i]*a[i]
                # So k-i must be in [0, n-1], i.e., k in [i, i+n-1]
                # Also k-i+i = k, and the other term is a[k-i]*a[i] where k-i in [0,n-1]
                # Actually both are the same set: k in [i, i+n-1]
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j2 = k - i
                    if 0 <= j2 < n:
                        # Original contribution: a[i]*a[j2] (and a[j2]*a[i] if j2 != i)
                        # New contribution: a[i]*frac * a[j2]
                        if j2 == i:
                            new_b[k] = b[k] - a[i]**2 + (a[i]*frac)**2
                        else:
                            new_b[k] = b[k] - 2*a[i]*a[j2] + 2*(a[i]*frac)*a[j2]
                
                new_bmax = new_b.max()
                if new_bmax < best_local_bmax:
                    best_local_bmax = new_bmax
                    best_frac = frac
            
            a[i] = orig_val * best_frac
            if best_frac != 1.0:
                improved = True
        
        a = project_simplex(a, S)
        
        if not improved and rng is not None:
            # Perturb some coordinates
            if contribs.sum() > 0:
                probs = contribs / contribs.sum()
                idxs = rng.choice(n, size=min(5, n), replace=False, p=probs)
                for idx in idxs:
                    a[int(idx)] *= rng.uniform(0.5, 1.5)
                a = project_simplex(a, S)
    
    return best_a, best_bmax


def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def structured_init(n, rng, method="mixed"):
    """Create diverse initializations."""
    t = np.arange(n, dtype=np.float64) / n
    if method == "sin":
        a = 0.5 + 0.3*np.sin(2*np.pi*3*t) + 0.2*np.sin(2*np.pi*7*t) \
            + 0.1*np.sin(2*np.pi*13*t) + 0.05*rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3*np.sin(2*np.pi*(2*t + 3*t*t)) + 0.1*rng.standard_normal(n)
    elif method == "bump":
        a = 0.3 + 0.5*np.exp(-0.5*((t-0.3)*4)**2) + 0.2*np.exp(-0.5*((t-0.7)*4))
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.1*rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = 64
        piece_len = n // n_pieces
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, 0.3 + 0.7*rng.random(n - len(a))])
        a = a[:n]
    elif method == "gaussian":
        a = np.exp(-0.5 * ((t - 0.5) * 4) ** 2) + 0.1 * rng.standard_normal(n)
    elif method == "two_peak":
        a = 0.2 + 0.6*np.exp(-0.5*((t-0.25)*6)**2) + 0.6*np.exp(-0.5*((t-0.75)*6)**2) \
            + 0.05*rng.standard_normal(n)
    else:
        a = 0.5 + 0.3*rng.standard_normal(n)
    
    a = np.clip(a, 0.01, 3.0)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Lp-smoothed Nesterov accelerated gradient descent with adaptive p,
    spectral perturbations for global exploration, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    # Use n = 16384 for more degrees of freedom
    n = 16384
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 32768
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def try_update(a_candidate):
