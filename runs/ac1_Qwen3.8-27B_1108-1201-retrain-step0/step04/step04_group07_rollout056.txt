"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using softmax gradient descent
with adaptive temperature annealing, multiple initializations, and periodic
restarting. The key innovation is using the analytically computed gradient of
the softmax approximation to the max-convolution objective, computed efficiently
via FFT-based cross-correlation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_softmax_gradient(a, T, N_fft):
    """
    Compute gradient of T*log(sum exp(c[k]/T)) w.r.t. a.
    Returns (gradient, c, score).
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # Softmax weights (numerically stable)
    c_scaled = c / T
    c_shifted = c_scaled - c_scaled.max()
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    
    # Cross-correlation via FFT
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    w_pad = np.zeros(N_fft)
    w_pad[:2*n-1] = w
    
    A = fft(a_pad)
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    S = a.sum()
    score = float(2.0 * n * c.max() / (S * S))
    return grad, c, score

    Main optimization using softmax gradient descent with adaptive temperature.
    """
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N_fft = 1 << (3 * n - 2).bit_length()
    
    # Initial temperature
    c0 = fft_convolve_self(a)
    T_init = max((c0.max() - c0.min()) / 4.0, 0.05)
    T = T_init
    T_min = 0.01
    
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    stagnant = 0
    last_best = best_score
    
    # Learning rate base
    lr_base = 2.0 / n
    
    while time.monotonic() - start < budget:
        it += 1
        
        # Compute gradient
        grad, c, score = compute_softmax_gradient(a, T, N_fft)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Check convergence

            break
        
        # Learning rate with decay

def spectral_repair(a, n, N_fft, n_steps, alpha_init, rng, S):
    """
    Spectral repair: directly reduce the largest spectral components
    that contribute to convolution peaks.
    """
    a = a.copy()
    c = fft_convolve_self(a)
    S = a.sum()
    
    for step in range(n_steps):
        c = fft_convolve_self(a)
        c_max = c.max()
        
        # Identify top lags
        top_k = min(20, 2*n-1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # For each top lag, reduce contributing frequency bins
        a_pad = np.zeros(N_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A)**2
        phase = np.angle(A)
        
        # Compute which bins contribute most to the peak
        # c[k] = (1/N) * sum_m A[m]^2 * exp(2*pi*i*m*k/N)
        # For the peak lag k*, reduce bins where Re(A[m]^2 * exp(2*pi*i*m*k*/N)) > 0
        k_star = top_idx[0]
        contributions = np.zeros(N_fft)
        for m in range(N_fft):
            contributions[m] = P[m] * np.cos(2*np.pi*m*k_star/N_fft - 2*phase[m])
        
        # Reduce bins that positively contribute to the peak
        above = contributions > np.median(contributions)
        alpha = alpha_init * (1.0 - step / max(n_steps-1, 1) * 0.7)
        P[above] *= (1.0 - alpha)
        
        # Slightly boost bins that negatively contribute (to maintain energy)
        below = contributions < np.median(contributions)
        P[below] *= (1.0 + alpha * 0.3)
        
        P = np.maximum(P, 0)
        A_new = np.sqrt(P) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        sc_new = compute_score(a_new)
        sc_old = compute_score(a)
        if sc_new < sc_old:
            a = a_new
        # else keep old a
    
    return a

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using softmax gradient descent
with adaptive temperature annealing, FFT-based analytic gradients,
spectral repair, and multi-start strategy. Key innovation: the gradient of
the softmax approximation to max(convolution) is computed in O(N log N)
via cross-correlation in the frequency domain, enabling many more iterations
than subgradient or spectral-flattening approaches.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_softmax_gradient(a, T, N_fft):
    """
    Compute gradient of T*log(sum exp(c[k]/T)) w.r.t. a.
    The gradient is 2 * cross_correlation(w, a) where w = softmax(c/T).
    Computed efficiently via FFT.
    Returns (gradient, c, score).
    """
    n = len(a)
    c = fft_convolve_self(a)

    # Stable softmax weights
    c_scaled = c / T
    c_shifted = c_scaled - c_scaled.max()
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum

    # Cross-correlation via FFT: grad[i] = 2 * sum_k w[k]*a[k-i]
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    w_pad = np.zeros(N_fft)
    w_pad[:2 * n - 1] = w

    A = fft(a_pad)
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    S = a.sum()
    score = float(2.0 * n * c.max() / (S * S))
    return grad, c, score


def softmax_gd_optimize(a_init, n, S, budget, rng, target=1.50):
    """
    Softmax gradient descent with adaptive temperature annealing,
    stagnation detection, and periodic noise restarts.
    """
    a = np.maximum(a_init, 0.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N_fft = 1 << (3 * n - 2).bit_length()

    # Initial temperature from data
    c0 = fft_convolve_self(a)
    T = max((c0.max() - c0.min()) / 4.0, 0.05)
    T_min = 0.005

    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    stagnant = 0

    while time.monotonic() - start < budget:
        it += 1

        grad, c, score = compute_softmax_gradient(a, T, N_fft)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if best_score <= target:
            break

        # Learning rate: starts moderate, decays slowly
        lr = 2.0 / (n * (1.0 + 0.003 * it))

        # Gradient step with projection
        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)

        # If poorly stuck, take a bolder step
        if stagnant > 500:
            lr_bold = 5.0 / (n * (1.0 + 0.01 * it))
            a_bold = project_simplex(a - lr_bold * grad, S)
            sc_bold = compute_score(a_bold)
            sc_new = compute_score(a_new)
            if sc_bold < sc_new:
                a_new = a_bold
                sc_new = sc_bold
            # Occasionally negate the step (push away from local min structure)
            if stagnant > 2000 and rng.random() < 0.1:
                a_push = project_simplex(a + lr_bold * grad, S)
                sc_push = compute_score(a_push)
                if sc_push < sc_new:
                    a_new = a_push
                    sc_new = sc_push

        a = a_new

        # Anneal temperature
        if it % 300 == 0:
            T = max(T_min, T * 0.88)

        # Noise restarts at different scales
        if stagnant > 400 and it % 100 < 50 and rng.random() < 0.4:
            noise_level = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_level, n), S)
            stagnant = 0
            c_tmp = fft_convolve_self(a)
            T = max((c_tmp.max() - c_tmp.min()) / 4.0, 0.05)

        if stagnant > 1500 and rng.random() < 0.3:
            noise_level = 0.08 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_level, n), S)
            stagnant = 0
            c_tmp = fft_convolve_self(a)
            T = max((c_tmp.max() - c_tmp.min()) / 3.0, 0.1)

        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print("    it=%d, best=%.6f, T=%.5f, stag=%d, time=%.0fs" % (
                it, best_score, T, stagnant, elapsed))

    return best_a, best_score


def spectral_repair(a, n, N_fft, budget, rng, S):
    """
    Directly reduce spectral components that contribute positively to
    the convolution peak. Iterates until time budget is exhausted.
    """
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    start = time.monotonic()
    step = 0

    while time.monotonic() - start < budget:
        step += 1
        c = fft_convolve_self(a)
        k_star = int(np.argmax(c))

        a_pad = np.zeros(N_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute per-bin contribution to the peak lag
        contrib = P * np.cos(2.0 * np.pi * np.arange(N_fft) * k_star / N_fft - 2.0 * phase)

        # Reduce positive contributors, slightly boost negative ones
        alpha = 0.05 * max(0.1, 1.0 - step / 100.0)
        med = np.median(contrib)
        P[contrib > med] *= (1.0 - alpha)
        P[contrib < med] *= (1.0 + alpha * 0.2)
        P = np.maximum(P, 0.0)

        A_new = np.sqrt(P) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

        if step % 50 == 0:
            print("    spectral_repair step=%d, score=%.6f" % (step, best_sc))

    return best_a, best_sc


def coordinate_descent_polish(a, n, S, budget, rng):
    """
    Coordinate-descent style polish: identify small entries (near zero)
    and large entries, and try to optimize locally by adjusting groups.
    """
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    start = time.monotonic()

    while time.monotonic() - start < budget:
        # Identify entries below and above median
        med = np.median(a)
        below_mask = a < med * 0.3
        above_mask = a > med * 2.0

        n_below = np.sum(below_mask)
        n_above = np.sum(above_mask)

        # Try reducing the largest entries and increasing the smallest
        if n_above > 0 and n_below > 0:
            a_trial = a.copy()
            # Reduce top entries
            top_idx = np.where(above_mask)[0]
            a_trial[top_idx] *= 0.95
            # Increase bottom entries (redistribute)
            bot_idx = np.where(below_mask)[0]
            a_trial[bot_idx] += 0.05 * a_trial[top_idx].mean()
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial * S / s
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial
            else:
                # Try the opposite: reduce bottom, increase top
                a_trial2 = a.copy()
                a_trial2[top_idx] *= 1.02
                a_trial2[bot_idx] *= 0.98
                a_trial2 = np.maximum(a_trial2, 0)
                s2 = a_trial2.sum()
                if s2 > 1e-15:
                    a_trial2 = a_trial2 * S / s2
                sc2 = compute_score(a_trial2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a_trial2.copy()
                    a = a_trial2

        # Random coordinate perturbation
        if rng.random() < 0.5:
            n_perturb = max(10, n // 100)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, 0.01 * S / n, n_perturb)
            a_trial = project_simplex(a_trial, S)
            sc = compute_score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial

    return best_a, best_sc


def make_initializations(n, rng, S):
    """Create diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * S / n))

    # Sine wave
    inits.append(("sin1", (0.5 + 0.4 * np.sin(2 * np.pi * tt)) * S / n))

    # Multi-sine
    inits.append(("multi_sin", (0.5 + 0.2 * np.sin(2 * np.pi * 3 * tt) +
                                  0.15 * np.sin(2 * np.pi * 7 * tt) +
                                  0.1 * np.sin(2 * np.pi * 11 * tt)) * S / n))

    # Chirp
    inits.append(("chirp", (0.5 + 0.3 * np.sin(2 * np.pi * (4 * tt + 2 * tt**2))) * S / n))

    # Random
    inits.append(("random", (rng.random(n) + 0.01) * S / n))

    # Binary-like
    b = (rng.random(n) > 0.5).astype(float)
    if b.sum() > 0:
        b = b * S / b.sum()
    inits.append(("binary", b))

    # Smoothed noise
    noise = rng.standard_normal(n)
    k = max(3, n // 64)
    smooth = np.convolve(noise, np.ones(k) / k, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth", (0.1 + 0.9 * smooth) * S / n))

    # Bump
    inits.append(("bump", (np.exp(-2.0 * (tt - 0.5)**2) + 0.1) * S / n))

    return inits
