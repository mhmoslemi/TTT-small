    # (w * flip(a)) has length (2n-1) + n - 1 = 3n-2
    # We need indices n-1, n, ..., 2n-2 (i.e., m = 0, 1, ..., n-1)
    
    grad = a.copy() + conv[n-1:n-1+n]
    return grad

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. FFT-based mirror descent with LSE-smoothed subgradient (entropic updates)
2. Adaptive beta (smoothing) annealing with restarts
3. Vectorized coordinate descent for polishing
4. Multi-start with diverse initializations
5. Periodic noise injection to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_grad_fft(a, c, beta):
    """Compute LSE gradient: grad[j] = a[j] + (w * flip_a)[n-1+j] where w = softmax(beta*c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot

    # Cross-correlation via FFT: conv = IFFT(FFT(w) * FFT(flip_a))
    # grad[j] = a[j] + conv[n-1+j]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    flip_a = a[::-1]
    FA = fft(flip_a, fsize)
    conv = np.real(ifft(W * FA))
    grad = a + conv[n - 1: n - 1 + n]
    return grad


def topk_grad(a, c, k):
    """Compute subgradient using top-k lags (unweighted average)."""
    n = len(a)
    if k >= len(c):
        topk_idx = np.arange(len(c))
    else:
        topk_idx = np.argpartition(c, -k)[-k:]
    grad = np.zeros(n)
    for lag in topk_idx:
        lo = max(0, lag - n + 1)
        hi = min(lag, n - 1)
        j_idx = np.arange(lo, hi + 1)
        grad[j_idx] += a[j_idx] + a[lag - j_idx]
    grad /= len(topk_idx)
    return grad


def mirror_descent(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=3000.0,
                   eta_mult=1.0, use_lse=True, topk_size=5, restart_patience=500):
    """Mirror descent with adaptive LSE smoothing and restarts."""
    a = normalize_sum(a_init, S)
    c = fft_convolve_self(a)
    best = c.max()
    a_best = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    beta = beta_start
    total_iters = max(10000, int(max_time * 100))  # rough estimate
    beta_ramp = min(3000, total_iters // 3)

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta schedule
        if it < beta_ramp:
            beta = beta_start + (beta_end - beta_start) * (it / beta_ramp) ** 0.7
        else:
            beta = beta_end

        # Compute gradient
        if use_lse:
            grad = lse_grad_fft(a, c, beta)
            step = eta_mult / max(beta, 1.0)
        else:
            grad = topk_grad(a, c, topk_size)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = eta_mult * S / (gnorm + 1e-10)

        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-step * grad)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Evaluate
        c_new = fft_convolve_self(a_new)
        cur = c_new.max()

        if cur < best - 1e-12:
            best = cur
            a_best = a_new.copy()
            last_improve = it

        a = a_new
        c = c_new

        # Restart: if no improvement for a while, add noise
        if it - last_improve > restart_patience:
            decay = np.exp(-(it - last_improve) / restart_patience)
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                it_restart = it
                # Reset patience
                last_improve = it
                if it - last_improve > restart_patience * 3:
                    # Larger noise
                    noise_scale = 0.05 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    c = fft_convolve_self(a)
                    last_improve = it

    return a_best, best


def coord_descent_fast(a_init, n, S, max_time, rng):
    """Vectorized coordinate descent: try random pairs, apply best improvement."""
    a = normalize_sum(a_init, S)
    c = fft_convolve_self(a)
    best = c.max()
    a_best = a.copy()
    start = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0

    while time.monotonic() - start < max_time:
        # Try a batch of random pairs
        n_pairs = min(2000, n)
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 0.5:
                break
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m:
                continue
            aj = a[j]
            am = a[m]
            if aj < 1e-12 and am < 1e-12:
                continue

            # Compute d-vector: change in c per unit delta (transfer from m to j)
            # d[k] = a[j+k] + a[j-k] - a[m+k] - a[m-k] (with bounds)
            d = np.zeros(2 * n - 1)
            # Contribution from j
            jk_hi = min(n - 1, j + (2 * n - 2))
            jk_lo = max(0, j)
            for k in range(2 * n - 1):
                vj = 0.0
                if 0 <= j + k < n:
                    vj += a[j + k]
                if 0 <= j - k < n:
                    vj += a[j - k]
                vm = 0.0
                if 0 <= m + k < n:
                    vm += a[m + k]
                if 0 <= m - k < n:
                    vm += a[m - k]
                d[k] = vj - vm

            # Golden section search for delta in [-aj, am]
            lo = -aj
            hi = am

            # Quick check: if delta=0 is already optimal, skip
            # Evaluate at endpoints and midpoint
            def f_delta(dlt):
                return (c + dlt * d).max()

            fc = f_delta(lo * 0.5 + hi * 0.5)  # midpoint
            f0 = c.max()
            if fc >= f0 - 1e-14:
                # Try small steps
                step = (hi - lo) * 0.1
                f_plus = f_delta(step)
                f_minus = f_delta(-step)
                if min(f_plus, f_minus) >= f0 - 1e-14:
                    continue

            # Full golden section
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc2 = f_delta(c_g)
            fd2 = f_delta(d_g)

            for _ in range(30):
                if fc2 < fd2:
                    b_g = d_g
                    d_g = c_g
                    fd2 = fc2
                    c_g = b_g - gr * (b_g - a_g)
                    fc2 = f_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc2 = fd2
                    d_g = a_g + gr * (b_g - a_g)
                    fd2 = f_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = max(lo, min(hi, delta))
            if abs(delta) < 1e-14:
                continue

            # Apply
            a[j] += delta
            a[m] -= delta
            c = c + delta * d
            # Correct c[0] for the quadratic term
            c[0] = c[0] + 2.0 * (aj - am) * delta + 2.0 * delta * delta

            cur = c.max()
            if cur < best - 1e-12:
                best = cur
                a_best = a.copy()
                improved = True

            # Clamp tiny negatives
            if a[j] < 0:
                a[j] = 0
            if a[m] < 0:
                a[m] = 0

        # Periodic full recompute
        if not improved:
            stall += 1
            if stall > 3:
                # Add noise
                noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stall / 5.0)
                if noise_scale < 1e-10:
                    break
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                stall = 0
        else:
            stall = 0

        # Full recompute every 1000 iterations to avoid drift
        if it if hasattr(locals(), 'it') else 0:
            pass

    return a_best, best

    beta_ramp_end = total_est * 40 // 100
    momentum_start = total_est * 30 // 100

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta
        if it < beta_ramp_end:
            progress = it / beta_ramp_end
            beta = beta_target * (1.0 / (1.0 + 200.0 * np.exp(-15.0 * progress)))
            beta = max(beta, 5.0)
        else:
            beta = beta_target

        # Update momentum
        if it > momentum_start:
            momentum = min(0.3, (it - momentum_start) / total_est)

        # Compute gradient
        grad = lse_grad_fft(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = eta_mult / (beta * 0.1 + 1.0)
        # Scale by sum for stability
        step *= S / (gnorm + 1e-10)
        step = min(step, 0.1)

        # Mirror descent update
        update = np.exp(-step * grad)
        a_new = a * update
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Momentum (in the "log" space would be better, but use simple blend)
        if momentum > 0:
            a_new = (1 - momentum) * a_new + momentum * a_mom
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * (S / s)

        # Evaluate (every other iteration for speed)
        if it % 2 == 0 or it < 100:
            c_new = fft_convolve_self(a_new)
            cur = c_new.max()
            if cur < best - 1e-12:
                best = cur
                a_best = a_new.copy()
                last_improve = it
            a = a_new
            c = c_new
        else:
            a = a_new
            # Skip c update for odd iterations (will recompute)
            c = None

        a_mom = a.copy()

        # Restart logic
        if c is not None and it - last_improve > 300:
            # Try different strategies
            strategy = rng.integers(0, 3)
            if strategy == 0:
                # Noise
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                last_improve = it
            elif strategy == 1:
                # Increase beta temporarily
                beta = min(beta_target * 2, beta * 2)
            else:
                # Rescale eta
                eta_mult = max(0.5, eta_mult * 0.5)

        # Update beta back to target after restart
        if beta > beta_target:
            beta = max(beta_target, beta * 0.8)

    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        a_best = a.copy()

    return a_best, best


def make_variants(a, n, S, rng):
    """Generate diverse variants of a sequence."""
    variants = [a.copy()]
    tt = np.linspace(0, 1, n, endpoint=False)

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, S))

    # Smoothed
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    for cutoff in [0.1, 0.3, 0.5]:
        F_filt = F * np.exp(-20.0 * (freqs / cutoff) ** 2)
        a_filt = np.real(ifft(F_filt))[:n]
        a_filt = np.maximum(a_filt, 0)
        s = a_filt.sum()
        if s > 1e-10:
            variants.append(a_filt * (S / s))

    # Perturbed
    for scale in [0.02, 0.05, 0.15]:
        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        v = project_simplex(a + noise, S)
        variants.append(v)

    # Truncated + padded
    for frac in [0.7, 0.85]:
        m = int(n * frac)
        a_trunc = np.zeros(n)
        a_trunc[:m] = a[:m]
        variants.append(normalize_sum(a_trunc, S))

    # Power-law weighting
    for p in [0.5, 2.0]:
        weights = np.power(tt + 0.1, p)
        a_pw = a * weights
        s = a_pw.sum()
        if s > 1e-10:
            variants.append(a_pw * (S / s))

    return variants
