import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    N = 1 << (2*n - 2).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2*n-1]
    return 2.0 * n * c.max() / (S * S)

    """
    One step of soft subgradient descent.
    Uses exponential weighting of peaks for smoother gradient.
    """

Uses soft subgradient descent with Nesterov acceleration,
combined with spectral magnitude shaping.

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return np.array([])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho_candidates) == 0:
        return np.zeros(n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N=None):
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 2).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, N

def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c, _ = compute_conv(a)
    return 2.0 * n * c.max() / (S * S)

    n = len(a)
    A = fft(a, N)

    phase = np.angle(A)
    
    # Geometric mean of magnitudes (excluding zeros)

    # Geometric blending: mag_new = mag^(1-alpha) * target^alpha
    log_new = (1 - alpha) * log_mag + alpha * np.log(target + 1e-30)
    mag_new = np.exp(log_new)
    
    # Reconstruct

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:

    return a_new * (a.sum() / s)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses soft subgradient descent (FFT-accelerated cross-correlation gradient)
with Nesterov momentum, adaptive temperature, spectral magnitude shaping,
and perturbation-based restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho_candidates) == 0:
        return np.zeros(n, dtype=np.float64)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_fft(a, N):
    """Compute linear self-convolution via FFT. Returns c of length 2n-1."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def compute_obj_fast(c, n, S):
    """Compute objective from precomputed convolution."""
    if S < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_gradient_fft(a, w, N):
    """
    Compute g[m] = 2 * sum_k w[k] * a[k-m] (with bounds).
    This is the subgradient direction using FFT cross-correlation.
    """
    n = len(a)
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    # Cross-correlation: corr[m] = sum_j w[j+m] * a[j]
    # In FFT: ifft(conj(FFT(w)) * FFT(a))
    Wc = np.conj(fft(w_pad))
    A = fft(a_pad)
    corr = np.real(ifft(Wc * A))
    return 2.0 * corr[:n]


def spectral_shape_step(a, N, alpha):
    """Push |FFT(a)| toward geometric mean (flatten spectrum)."""
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    log_mag = np.log(mag + 1e-30)
    target_log = np.mean(log_mag)
    log_new = (1.0 - alpha) * log_mag + alpha * target_log
    mag_new = np.exp(log_new)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)


def phase_alignment_step(a, N, alpha):
    """
    Nudge F[k] toward being real (phase 0 or pi) to help flatten F[k]*F[k].
    For real a, F has Hermitian symmetry. We blend phase toward 0/pi.
    """
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Target phase: 0 if cos(phase) >= 0, pi otherwise (i.e., make F real)
    # Blend: new_phase = (1-alpha)*phase + alpha*target_phase (circular blend)
    # Simpler: multiply by a factor that reduces the imaginary part
    # F_new = (1-alpha)*F + alpha*real(F) (for the "make real" part)
    # But this breaks Hermitian symmetry... Let's just partially reduce imaginary part
    A_new = A * (1.0 - alpha) + np.real(A) * alpha
    # Enforce Hermitian symmetry for real output
    A_new[0] = np.real(A_new[0])
    for k in range(1, N // 2 + 1):
        A_new[N - k] = np.conj(A_new[k])
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)

    if best_obj < target:
        return best_a, best_obj

    start_time = time.monotonic()
    t_nest = 1.0
    y = a.copy()
    it = 0
    stagnation_count = 0
    last_best_obj = best_obj

    # Adaptive temperature: start moderate, decrease over time
    T0 = S * S / n  # Initial temperature scale
    step_base = S / (2.0 * np.sqrt(n))

        it += 1
        progress = (time.monotonic() - start_time) / max(budget, 1e-6)

        # Temperature annealing
        T = T0 * (0.01 + 0.99 * (1.0 - progress) ** 2)

        if wsum < 1e-30:

        weights /= wsum

            t_nest = 1.0

            t_nest = 1.0

                0.08 * np.sin(2 * np.pi * 23 * t))

    # Random
    inits.append(rng.random(n) + 0.05)

    # Two bumps
    inits.append(0.1 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 5) ** 2) +
                0.5 * np.exp(-0.5 * ((t - 0.7) * 5) ** 2))

    # Cosine window
    inits.append(0.3 + 0.4 * (1.0 + np.cos(2 * np.pi * t)) / 2.0)

    # Smooth noise
    noise = rng.standard_normal(n)
    ks = max(3, n // 128)
    kernel = np.ones(ks) / ks
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(0.15 + 0.7 * smooth)

    # Exponential ramp
    inits.append(0.1 + 0.8 * np.exp(-2 * t) * (1 - np.exp(-2 * t)) + 0.1)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses FFT-accelerated soft subgradient descent with Nesterov momentum,
    spectral magnitude sharpening, phase alignment, and multi-start restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    print("Starting optimization, budget = %.0fs" % budget_s)

    global_best = np.inf
    global_a = None

    # Try to load provided sequence
    provided = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        provided = np.clip(provided, 0.0, 1000.0)
        if provided.sum() < 0.01:
            provided = None
    except (NameError, Exception):
        pass

    # Determine lengths to search
    lengths = [8192]
    if provided is not None:
        if len(provided) != 8192:
            lengths.insert(0, len(provided))
    lengths.append(4096)
    lengths.append(16384)

    # Deduplicate and sort
    lengths = sorted(set(lengths))

    # Budget allocation
    total_budget = budget_s - 25.0
    # Weight toward 8192 (sweet spot of DOF vs. convergence speed)
    w_map = {}
    for L in lengths:
        if L == 8192:
            w_map[L] = 0.45
        elif L == 4096:
            w_map[L] = 0.20
        elif L == 16384:
            w_map[L] = 0.20
        else:
            w_map[L] = 0.15
    total_w = sum(w_map.values())

    for n in lengths:
        if time.monotonic() > deadline:
            break
        remaining_total = deadline - time.monotonic()
        if remaining_total < 20:
            break

        alloc = total_budget * w_map.get(n, 0.1) / total_w
        alloc = min(alloc, remaining_total - 10)
        if alloc < 10:
            continue

        print("\n=== n=%d, time budget=%.0fs ===" % (n, alloc))
        N = 1 << (2 * n - 2).bit_length()
        S = np.sqrt(2.0 * n)

        # Build list of starting points
        start_list = []

        # Primary: provided sequence (if matches length)
        if provided is not None and len(provided) == n:
            start_list.append(("provided", provided.copy()))

        # Secondary: diversity initializations
        inits = make_initializations(n, rng)
        # Limit number of starts based on budget
        n_starts = max(2, min(6, int(alloc / 30)))
        for i in range(min(n_starts, len(inits))):
            start_list.append(("init_%d" % i, inits[i]))

        # Shuffle (but keep provided first)
        if len(start_list) > 1:
            rest = start_list[1:]
            rng.shuffle(rest)
            start_list = [start_list[0]] + rest
            if provided is not None and len(provided) == n:
                start_list = [start_list[0]]  # provided already first
            else:
                start_list = rest[:n_starts] if len(start_list) > n_starts else start_list

        per_start_budget = alloc / max(len(start_list), 1)

        local_best_a = None
        local_best_obj = np.inf

        for name, a0 in start_list:
            if time.monotonic() > deadline - 5:
                break
            rem = min(per_start_budget, deadline - time.monotonic() - 5)
            if rem < 3:
                break

            a0 = np.clip(a0, 0.0, 1000.0).astype(np.float64)
            s0 = a0.sum()
            if s0 < 1e-15:
                a0 = np.ones(n) * S / n
            else:
                a0 = a0 * (S / s0)

            a_res, obj = optimize_sequence(a0, S, N, rem, rng, target=1.49)

            if obj < local_best_obj:
                local_best_obj = obj
                local_best_a = a_res.copy()

            if obj < global_best:
                global_best = obj
                global_a = a_res.copy()
                print("  [%s n=%d] NEW GLOBAL BEST: %.6f" % (name, n, obj))
            else:
                print("  [%s n=%d] obj=%.6f" % (name, n, obj))

        # Final polish on best for this length
        if local_best_a is not None:
            rem = max(0.0, deadline - time.monotonic() - 5)
            if rem > 5:
                print("  Polishing n=%d with %.0fs..." % (n, rem))
                a_pol, obj_pol = optimize_sequence(local_best_a, S, N, rem, rng, target=1.49)
                if obj_pol < local_best_obj:
                    local_best_obj = obj_pol
                    local_best_a = a_pol.copy()
                if obj_pol < global_best:
                    global_best = obj_pol
                    global_a = a_pol.copy()
                    print("  [polish n=%d] NEW GLOBAL BEST: %.6f" % (n, obj_pol))

    # Cross-length refinement: take best and try to improve with different N
    if global_a is not None:
        n_best = len(global_a)
        N_best = 1 << (2 * n_best - 2).bit_length()
        S_best = global_a.sum()
        rem = max(0.0, deadline - time.monotonic() - 5)
        if rem > 10:
            print("\nFinal refinement (%.0fs)..." % rem)
            # Try a few perturbation + re-optimize cycles
            for trial in range(3):
                if time.monotonic() > deadline - 5:
                    break
                pert = rng.normal(0, 0.005 * S_best / np.sqrt(n_best), n_best)
                a_pert = project_simplex(global_a + pert, S_best)
                sub_budget = rem / (3 - trial)
                a_tri, obj_tri = optimize_sequence(a_pert, S_best, N_best,
                                                   sub_budget, rng, target=1.49)
                if obj_tri < global_best:
                    global_best = obj_tri
                    global_a = a_tri.copy()
                    print("  Trial %d: %.6f (improved!)" % (trial, obj_tri))
                else:
                    print("  Trial %d: %.6f" % (trial, obj_tri))

    # Fallback
    if global_a is None or global_best >= np.inf:
        print("WARNING: Using fallback uniform sequence")
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0

    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(4096, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== FINAL ===")
    print("Length: %d" % len(global_a))
    print("Computed objective: %.6f" % global_best)
    print("Verified evaluate_sequence: %.6f" % ver)

    if np.isfinite(ver) and ver < global_best:
        global_best = ver

    return final_list
