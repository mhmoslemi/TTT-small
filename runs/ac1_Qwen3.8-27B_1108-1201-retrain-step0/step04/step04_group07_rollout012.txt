import numpy as np
import time

def compute_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = A * np.conj(A)
    c = np.fft.irfft(P, N)[:2*n-1]
    return c

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for theta
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    theta = (cumsum - S) / idx
    # Find largest rho such that theta[rho-1] <= w[rho-1]
    valid = (w - theta) >= 0
    rho = np.max(np.where(valid)[0]) + 1
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def subgradient_autocorr(a, k_max):
    """
    Compute subgradient of max(c) at point a, where k_max is the argmax lag.
    Returns gradient vector of length n.
    """
    n = len(a)
    # grad[m] = 2 * a[k_max - m] if 0 <= k_max - m < n, else 0
    grad = np.zeros(n)
    m_range = np.arange(n)
    k_minus_m = k_max - m_range
    valid = (k_minus_m >= 0) & (k_minus_m < n)
    grad[valid] = 2.0 * a[k_minus_m[valid]]
    return grad

def multi_lag_subgradient(a, c, top_k):
    """
    Compute weighted subgradient using top-k lags.
    Weights are proportional to c values (or a smoothed version).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    weights = top_vals / total
    
    grad = np.zeros(n)
    m_range = np.arange(n)
    for i, k in enumerate(top_idx):
        k_minus_m = k - m_range
        valid = (k_minus_m >= 0) & (k_minus_m < n)
        grad[valid] += weights[i] * 2.0 * a[k_minus_m[valid]]
    return grad

def nesterov_subgradient(a_init, n, max_time, rng, top_k_init=1, top_k_max=100,
                          step_scale=1.0, shake_interval=200, shake_prob=0.3):
    """
    Nesterov accelerated projected subgradient method.
    """
    S = a_init.sum()
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = a / S  # normalize to sum = 1
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = compute_obj(a)
    
    it = 0
    N = 1 << (2*n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute autocorrelation at y
        c = compute_autocorr(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max  # since sum(y) = 1
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
        
        # Adaptive top-k: start small, grow
        progress = min(1.0, it / 5000)
        top_k = max(top_k_init, int(top_k_max * progress))
        
        # Compute subgradient
        grad = multi_lag_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak step size or fixed
        # Using diminishing step size
        step = step_scale / (np.sqrt(it) + 1)
        
        # Nesterov update
        x_new = project_simplex(y - step * grad, 1.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        # Re-project to ensure sum = 1
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = x_new.copy()
        
        x_prev = x_new
        t = t_new
        
        # Periodic shaking
        if it % shake_interval == 0 and rng.random() < shake_prob:
            noise = rng.standard_normal(n) * 0.001
            y = project_simplex(y + noise, 1.0)
            x_prev = y.copy()
            t = 1.0
    
    best_a = best_a * S  # denormalize
    return best_a, 2.0 * n * best_a.max() if False else compute_obj(best_a)

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = compute_autocorr(a)

where f* is a lower bound on the optimal value. We know f* >= 2n/(2n-1) (the theoretical minimum when all c values are equal).

2. Compute a_new = proj(y - step * g)
3. If f(a_new) > f(y) * (1 - c * step * ||g||), halve step and repeat.
4. Accept a_new.

This is more robust but requires multiple function evaluations per iteration.

I'll use the Polyak step size with a cap on the maximum step.

import numpy as np
import time

def compute_autocorr(a):
    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.abs(A)**2
    c = np.fft.irfft(P, N)[:2*n-1]
    return c

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    # Find largest rho where w[rho-1] >= theta[rho-1]
    valid = (w >= theta)
    rho = np.max(np.where(valid)[0]) + 1
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = compute_autocorr(a)
    return 2.0 * n * c.max() / (S * S)

def subgradient_single_lag(a, k):
    n = len(a)
    grad = np.zeros(n)
    m_start = max(0, k - (n-1))
    m_end = min(k, n-1)
    a_idx_start = k - m_end
    a_idx_end = k - m_start + 1
    grad[m_start:m_end+1] = 2.0 * a[a_idx_start:a_idx_end][::-1]
    return grad

def subgradient_multi_lag(a, c, top_k):
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Use softmax-like weights for smoother gradient
    # Or just proportional
    max_val = top_vals[0]
    weights = top_vals / top_vals.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        m_start = max(0, k - (n-1))
        m_end = min(k, n-1)
        a_idx_start = k - m_end
        a_idx_end = k - m_start + 1
        grad[m_start:m_end+1] += weights[i] * 2.0 * a[a_idx_start:a_idx_end][::-1]
    return grad

def optimize_nesterov(a_init, max_time, rng, target=1.50):
    n = len(a_init)
    S = a_init.sum()
    a = a_init / S  # normalize to sum = 1
    
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = 2.0 * n * compute_autocorr(a).max()  # since sum=1
    f_lower = 2.0 * n / (2*n - 1)  # theoretical lower bound
    
    it = 0
    N = 1 << (2*n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = compute_autocorr(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
        
        if best_score <= target:
            break
        
        # Adaptive top-k
        progress = min(1.0, it / 10000)
        top_k = max(1, int(50 * progress + 1))
        top_k = min(top_k, 200)
        
        grad = subgradient_multi_lag(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak step size with cap
        polyak_step = (cur_score - f_lower) / (gnorm * gnorm + 1e-30)
        step = min(polyak_step, 0.1)  # cap to avoid divergence
        
        # Nesterov update
        x_new = project_simplex(y - step * grad, 1.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = x_new.copy()
        
        x_prev = x_new
        t = t_new
        
        # Periodic shaking to escape local minima
        if it % 500 == 250 and rng.random() < 0.3:
            noise_scale = 0.001 * (1.0 + 0.5 * np.sin(it / 100))
            y = project_simplex(y + rng.standard_normal(n) * noise_scale, 1.0)

            t = 1.0
    
    best_a = best_a * S
    return best_a, compute_obj(best_a)

def multiplicative_weights_step(a, c, top_k, step):
    """Apply multiplicative weights to reduce peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    weights = top_vals / top_vals.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        m_start = max(0, k - (n-1))
        m_end = min(k, n-1)
        a_idx_start = k - m_end
        a_idx_end = k - m_start + 1
        grad[m_start:m_end+1] += weights[i] * 2.0 * a[a_idx_start:a_idx_end][::-1]
    
    # Multiplicative update: a_new[j] = a[j] * exp(-step * grad[j])
    a_new = a * np.exp(-step * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using Nesterov-accelerated projected

diversification. Focuses on refining the provided warm start with careful

import numpy as np
import time


def _compute_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(np.abs(A) ** 2, N)[:2 * n - 1]
    return c


def _project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = w >= theta
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def _compute_obj(a):
    """Compute objective: 2*n*max(c)/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = _compute_autocorr(a)
    return float(2.0 * n * c.max() / (S * S))


def _subgradient(a, c, top_k):
    """Weighted subgradient using top-k autocorrelation lags."""
    n = len(a)

def _nesterov_optimize(a_init, max_time, rng, target=1.50, verbose=True):

    S = a_init.sum()
    a = a_init / S  # normalize to sum = 1

    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    start = time.monotonic()

    best_score = 2.0 * n * _compute_autocorr(a).max()

            x_prev = y.copy()

        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Small noise occasionally
        if it % 50 == 25 and rng.random() < 0.3:
            a_new = _project_simplex(a_new + rng.standard_normal(n) * 0.0005, 1.0)

        a = a_new

    best_a = best_a * S
    return best_a, _compute_obj(best_a)


def _try_perturbations(a_best, max_time, rng, n_trials=50):
    """Try random perturbations around the best solution."""
    n = len(a_best)
    S = a_best.sum()
    a_norm = a_best / S
    best = a_best.copy()
    best_score = _compute_obj(a_best)
    start = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - start > max_time:
            break

        # Decreasing perturbation scale
        scale = 0.003 * (1.0 - trial / n_trials) + 0.0003
        a_trial = _project_simplex(a_norm + rng.standard_normal(n) * scale, 1.0)
        sc = 2.0 * n * _compute_autocorr(a_trial).max()

        if sc < best_score:
            best_score = sc
            best = a_trial * S
            a_norm = a_trial  # center future perturbations on new best
            print(f"    [perturb {trial}] new best: {best_score:.6f}")

    return best, best_score


def _make_initialization(n, rng):
    """Generate a structured initialization of length n."""
    tt = np.linspace(0, 1, n, endpoint=False)
    # Multi-frequency sinusoidal with phases
    phases = rng.uniform(0, 2 * np.pi, size=5)
    a = np.ones(n) * 0.5
    for i, f in enumerate([3, 7, 11, 17, 23]):
        a += 0.15 * np.sin(2 * np.pi * f * tt + phases[i])
    a = np.maximum(a, 0.01)
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov projected subgradient + multiplicative refinement + perturbation.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + budget_s - 20

    print(f"Starting optimization, budget={budget_s:.0f}s")

    # Get starting point
    base = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.clip(base, 0, 1000)
        if base.sum() < 0.01:
            base = None
    except Exception:
        pass

    global_best_a = None
    global_best_score = np.inf

    def _update_global(a, score):
        nonlocal global_best_a, global_best_score
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return
        sc = _compute_obj(a)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = a.copy()
            print(f"  ** Global best updated: {sc:.6f} (n={len(a)})")

    # Determine lengths to try
    lengths = []
    if base is not None:
        lengths.append(len(base))
    for L in [8192, 4096, 16384]:
        if L not in lengths:
            lengths.append(L)

    # Budget allocation: primary gets 55%, secondary 25%, tertiary 20%
    # Primary is the provided length if available, else 8192
    primary_idx = 0 if (base is not None) else lengths.index(8192) if 8192 in lengths else 0
    lengths_sorted = [lengths[primary_idx]] + [l for i, l in enumerate(lengths) if i != primary_idx]

    total_alloc = budget_s - 40  # reserve for verification
    budgets = []
    for i, L in enumerate(lengths_sorted):
        if i == 0:
            budgets.append(total_alloc * 0.55)
        elif i == 1:
            budgets.append(total_alloc * 0.25)
        else:
            budgets.append(total_alloc * 0.20)

    for idx, n in enumerate(lengths_sorted):
        if time.monotonic() > deadline:
            break
        alloc = min(budgets[idx], deadline - time.monotonic())
        if alloc < 15:
            break

        print(f"\n=== Length n={n}, allocated {alloc:.0f}s ===")

        # Initialize
        if base is not None and n == len(base):
            a_init = base.copy()
        elif base is not None:
            # Resize: use interpolation or truncation
            a_init = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, len(base)),
                base
            )
            a_init = np.maximum(a_init, 0.0)
        else:
            a_init = _make_initialization(n, rng)

        a_init = np.clip(a_init, 0, 1000)
        if a_init.sum() < 0.01:
            a_init = np.ones(n) * 0.5

        init_score = _compute_obj(a_init)
        print(f"  Initial score: {init_score:.6f}")
        _update_global(a_init, init_score)

        # Phase 1: Nesterov subgradient (60% of budget)
        phase1_time = alloc * 0.60
        if phase1_time > 5:
            best_a, best_sc = _nesterov_optimize(a_init, phase1_time, rng, target=1.49, verbose=True)
            print(f"  After Nesterov: {best_sc:.6f}")
            _update_global(best_a, best_sc)
        else:
            best_a = a_init.copy()
            best_sc = init_score

        # Phase 2: Multiplicative refinement (25%)
        phase2_time = alloc * 0.25
        if phase2_time > 5 and time.monotonic() < deadline - 10:
            best_a2, best_sc2 = _refine_multiplicative(best_a, phase2_time, rng, top_k=150, base_step=0.02)
            print(f"  After multiplicative: {best_sc2:.6f}")
            _update_global(best_a2, best_sc2)
            if best_sc2 < best_sc:
                best_a = best_a2
                best_sc = best_sc2

        # Phase 3: Another Nesterov pass with different params (10%)
        phase3_time = alloc * 0.10
        if phase3_time > 3 and time.monotonic() < deadline - 10:
            best_a3, best_sc3 = _nesterov_optimize(best_a, phase3_time, rng, target=1.49, verbose=False)
            print(f"  After 2nd Nesterov: {best_sc3:.6f}")
            _update_global(best_a3, best_sc3)

        # Phase 4: Perturbations (5%)
        phase4_time = alloc * 0.05
        if phase4_time > 2 and time.monotonic() < deadline - 5:
            best_a4, best_sc4 = _try_perturbations(best_a, phase4_time, rng)
            _update_global(best_a4, best_sc4)

    # Final long refinement on the overall best
    if global_best_a is not None and time.monotonic() < deadline - 30:
        final_time = min(deadline - time.monotonic() - 10, 120)
        if final_time > 10:
            n = len(global_best_a)
            print(f"\n=== Final refinement (n={n}, {final_time:.0f}s) ===")
            # Very careful Nesterov with small steps
            a_best = global_best_a.copy()
            S = a_best.sum()
            a_norm = a_best / S

            # Use a longer, more patient optimization
            y = a_norm.copy()
            x_prev = a_norm.copy()
            t = 1.0
            start = time.monotonic()
            f_lower = 2.0 * n / (2 * n - 1)

            while time.monotonic() - start < final_time:
                c = _compute_autocorr(y)
                cur_score = 2.0 * n * c.max()

                if cur_score * (S * S) / (S * S) < global_best_score:
                    global_best_score = cur_score
                    global_best_a = y * S

                # Very small top-k for fine-grained control
                top_k = 10
                grad = _subgradient(y, c, top_k)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-14:
                    break

                polyak_step = (cur_score - f_lower) / (gnorm * gnorm + 1e-30)
                step = min(polyak_step, 0.01)
                step = max(step, 1e-7)

                x_new = _project_simplex(y - step * grad, 1.0)
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
                y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y / s_y
                x_prev = x_new
                t = t_new

            print(f"  After final refinement: {global_best_score:.6f}")

    # Fallback
    if global_best_a is None:
        global_best_a = np.ones(8192) * 0.5
        global_best_score = 2.0
        print("Using fallback uniform sequence")

    # Final verification
    global_best_a = np.clip(global_best_a, 0, 1000)
    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print(f"\nFinal verified score: {ver:.6f}")
        if ver < global_best_score:
            global_best_score = ver
    else:
        print("WARNING: verification failed, using fallback")
        global_best_a = np.ones(4096, dtype=np.float64) * 0.5
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        global_best_score = ver if np.isfinite(ver) else 2.0
        print(f"Fallback score: {global_best_score:.6f}")

    print(f"Done. length={len(global_best_a)}, score={global_best_score:.6f}")
    return final_list
