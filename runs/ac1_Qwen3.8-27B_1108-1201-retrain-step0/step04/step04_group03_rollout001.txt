"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-smoothed gradient descent,
LP best-response iterations, and spectral initialization.
Key insight: The objective is scale-invariant (depends only on shape),
so we optimize at a moderate n and rescale.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    n2 = v.shape[0]
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n2 + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n2
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of LSE wrt a: 2 * (w * a) (circular corr, truncated)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the actual objective."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_lse_gradient_descent(a_init, n, S, max_time, rng, beta_init=50.0, 
                              beta_growth=1.002, beta_max=5000.0,
                              use_momentum=True, momentum=0.9):

    Gradient descent on LSE approximation with adaptive step size.
    """
    a = np.array(a_init, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    else:
        a = np.ones(n) * S / n
    
    y = a.copy()  # Momentum variable
    v = np.zeros(n)  # Velocity
    t = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * beta_init)
    
    while time.monotonic() - start < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta_init * beta_growth ** it)
        else:
            beta = min(beta_max, beta_init * beta_growth ** it)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (y.sum() ** 2)
        
        if cur_obj < best_obj - 1e-8:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        
        # Adaptive step size (Barzilai-Borwein style)
        step = 1.0 / (2.0 * beta)
        
        # Backtracking line search
        alpha = 1.0
        y_max = c.max()
        for _ in range(20):
            candidate = y - alpha * step * grad
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs > 0:
                candidate = candidate * (S / cs)
            else:
                candidate = np.ones(n) * S / n
            c_new = fft_convolve_self(candidate)
            if c_new.max() < y_max * (1 + 1e-4) or alpha < 1e-6:
                break
            alpha *= 0.5
        
        new_a = y - alpha * step * grad
        new_a = np.maximum(new_a, 0)
        cs = new_a.sum()
        if cs > 0:
            new_a = new_a * (S / cs)
        else:
            new_a = np.ones(n) * S / n

            y = new_a + v
            y = np.maximum(y, 0)

            if ys > 0:

            y = new_a

def solve_lp_best_response(f, n, S, n_constraints=None, time_limit=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for selected lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max()
    
    # Select constraints: all lags where c_k > 0.9 * M
    threshold = 0.9 * M
    active_lags = np.where(c > threshold)[0]
    
    if n_constraints is not None and len(active_lags) > n_constraints:
        # Keep the most binding ones
        active_lags = active_lags[np.argsort(c[active_lags])[::-1][:n_constraints]]
    
    if len(active_lags) == 0:
        return None
    
    # Build constraint matrix
    n_lags = len(active_lags)
    # A[i, j] = f_{lag_i - j} if 0 <= lag_i - j < n, else 0
    # This is sparse; use dense for small problems, or build sparsely
    if n_lags * n < 50_000_000:
        A = np.zeros((n_lags, n))
        for i, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A[i, lo:hi+1] = f[k - (lo:hi+1)]
    else:
        # Too large, use only the top constraints
        top_k = min(n_constraints, 200) if n_constraints else 200
        active_lags = active_lags[np.argsort(c[active_lags])[::-1][:top_k]]
        n_lags = len(active_lags)
        A = np.zeros((n_lags, n))
        for i, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A[i, lo:hi+1] = f[k - (lo:hi+1)]
    
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    result = None
    for solver in ['SCS', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - start > time_limit:
                continue
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                result = g_var.value
                break
        except Exception:
            continue
    
    if result is None:
        return None
    
    g = np.maximum(result, 0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g = g * (S / g_sum)
    return g


def lp_step(f, n, S, n_constraints=500):
    """One LP-based step with optimal line search."""
    g = solve_lp_best_response(f, n, S, n_constraints=n_constraints)
    if g is None:
        return f.copy(), None
    
    # Find best step size
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    
    best_t = 0.0
    best_val = M_old
    best_f = f.copy()
    
    # Try a few step sizes
    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        cs = f_test.sum()
        if cs < 1e-10:
            continue
        f_test = f_test * (S / cs)
        c_test = fft_convolve_self(f_test)
        if c_test.max() < best_val:
            best_val = c_test.max()
            best_t = t
            best_f = f_test.copy()
    
    return best_f, best_val


def spectral_init(n, rng):
    """
    Create initial sequence with approximately flat autocorrelation
    using iterative spectral shaping.
    """
    # Start with random phase, flat magnitude
    fsize = 1 << (2 * n - 1).bit_length()
    
    # Target: flat |FFT|^2 (gives delta autocorrelation)
    # But we need non-negative f, so we'll iterate
    a = np.zeros(fsize)
    a[:n] = 1.0 + 0.1 * rng.standard_normal(n)
    
    for _ in range(50):
        # Compute FFT
        A = fft(a)
        # Modify to have flat magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        target_mag = np.ones(fsize)  # Flat magnitude
        A_new = target_mag * np.exp(phase)
        a = np.real(ifft(A_new))
        # Enforce non-negativity and support
        a = np.maximum(a, 0)
        a[n:] = 0  # Truncate to support [0, n)
    
    a = a[:n]
    a = np.maximum(a, 1e-6)
    return a

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Aggressive LP best-response iterations with LSE-graded

import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n2 = v.shape[0]
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n2 + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n2
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_grad(a, c, beta):
    n = len(a)
    cm = c.max()
    shifted = np.clip(beta * (c - cm), -400, 400)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def lp_best_response(f, n, S, n_cons=None, timeout=20.0):
    """
    LP: max 1'g s.t. A*f*g <= M (for selected lags), g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()
    
    # Select active constraints
    if n_cons is None or n_cons >= 2 * n - 1:
        active = np.arange(2 * n - 1)
    else:
        # Top n_cons lags by c value
        active = np.argsort(c)[::-1][:n_cons]
    
    n_lag = len(active)
    
    # Build constraint matrix (sparse-friendly)
    # A[i, j] = f[active[i] - j] if 0 <= active[i] - j < n
    A = np.zeros((n_lag, n))
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[i, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    
    g_var = cp.Variable(n)
    cons = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), cons)
    
    t0 = time.monotonic()
    g_val = None
    for solver in ['SCS', 'ECOS', 'CLARABEL', 'OSQP']:
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - t0 > timeout:
                break
            if g_var.value is not None:
                gv = np.array(g_var.value, dtype=np.float64)
                if not np.any(np.isnan(gv)) and np.any(gv > 0):
                    g_val = gv
                    break
        except Exception:
            continue
    
    if g_val is None:
        return None
    
    g = np.maximum(g_val, 0)
    gs = g.sum()
    if gs < 1e-10:
        return None
    return g * (S / gs)

    c_f = fft_convolve_self(f)
    M_f = c_f.max()
