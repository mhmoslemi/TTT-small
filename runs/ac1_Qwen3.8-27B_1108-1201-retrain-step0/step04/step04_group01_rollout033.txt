But we need to project onto `sum(a) = S` and `a >= 0`.

The non-negativity is automatically satisfied (exponential is positive).
The sum constraint requires a rescaling step.

So the algorithm is:
1. Compute c = conv(a, a)
2. Compute p = softmax(beta * c)
3. Compute grad_u = 2 * a * (p ⋆ a)  [correlation of p and a, element-wise multiplied by a]
4. Update: a = a * exp(-eta * grad_u)
5. Rescale: a = a * S / sum(a)

This is a multiplicative update! It's similar to the "iterative proportional fitting" used in statistics.

The advantage of this approach:
- Non-negativity is preserved automatically
- The update is "natural" in the information geometry sense (Fisher metric)
- It's related to mirror descent with the KL divergence

The step size eta needs to be chosen carefully. Too large and it oscillates; too small and it's slow.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative updates,
LSE-smoothed FISTA with adaptive step sizes, and spectral preconditioning.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_gradient(a, c, beta=None, top_k=None):
    """Compute gradient of max(conv) or LSE-smoothed max."""
    n = len(a)
    if beta is not None:
        # LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        if scaled.max() > 500:
            beta_eff = 500.0 / max(1e-15, (c - cm).max())
            scaled = beta_eff * (c - cm)
        else:
            beta_eff = beta
        e = np.exp(scaled)
        w = e / e.sum()
    else:
        # Use top-k peaks with excess weighting
        if top_k is None:
            top_k = min(32, len(c) // 4)
        top_k = min(top_k, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        # Weight by excess above median
        med = np.median(c)
        weights = np.maximum(c[idx] - med, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(top_k)
            wsum = 1.0
        weights = weights / wsum
        w = np.zeros(len(c))
        w[idx] = weights
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista(a_init, S, max_time, rng, beta_schedule=None, mode='lse'):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    if beta_schedule is None:
        beta_schedule = (10.0, 0.005, 10000.0)  # init, rate, max
    
    beta_init, beta_rate, beta_max = beta_schedule
    
    stagnation = 0
    last_best = best
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it
            stagnation = 0
        else:
            stagnation += 1
        
        # Compute gradient
        grad = compute_gradient(y, c, beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use Lipschitz estimate
        # L ≈ 2 * beta * ||a||_inf * n (conservative)
        # Better: L ≈ 2 + 2 * beta * ||a||^2
        a_norm_sq = np.sum(a**2)
        L = 2.0 + 2.0 * beta * a_norm_sq
        step = 1.0 / L
        
        # Try backtracking
        # (skip for speed, use fixed step)
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart with perturbation
        if stagnation > 2000:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stagnation / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stagnation = 0
    
    return best_a, best


def run_multiplicative(a_init, S, max_time, rng, beta=50.0):
    """Multiplicative (exponential) update method."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        # Compute gradient in log-space
        grad = compute_gradient(a, c, beta=beta)
        grad_u = a * grad  # gradient in log-space
        
        # Step size
        gmax = np.max(np.abs(grad_u))
        if gmax < 1e-15:
            break
        eta = 1.0 / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad_u)
        
        # Ensure positive and rescale
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * S / a_new.sum()
        
        # Safety: if the objective got much worse, take a smaller step
        c_new = fft_convolve_self(a_new)
        if c_new.max() > c.max() * 1.5:
            # Halve the step
            a_new = a * np.exp(-0.5 * eta * grad_u)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new * S / a_new.sum()
        
        a = a_new
    
    return best_a, best


def run_hybrid(a_init, S, max_time, rng):
    """Combine FISTA and multiplicative updates."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    # Alternate between additive and multiplicative updates
    use_additive = True
    
    while time.monotonic() - start < max_time:
        it += 1
        # Switch method every 100 iterations
        if it % 200 == 0:
            use_additive = not use_additive
        
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        beta = 20.0 + it * 0.5
        beta = min(beta, 5000.0)
        
        if use_additive:
            grad = compute_gradient(a, c, beta=beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            L = 2.0 + 2.0 * beta * np.sum(a**2)
            step = 1.0 / L
            a = project_simplex(a - step * grad, S)
        else:
            grad = compute_gradient(a, c, beta=beta)
            grad_u = a * grad
            gmax = np.max(np.abs(grad_u))
            if gmax < 1e-15:
                break
            eta = 1.0 / gmax
            a_new = a * np.exp(-eta * grad_u)
            a_new = np.maximum(a_new, 1e-15)
            a = a_new * S / a_new.sum()
    
    return best_a, best

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid FISTA with proper
Lipschitz-based step sizes, multiplicative (exponential) updates,
and aggressive LSE beta annealing. Key improvements over previous:
- Correct FISTA gradient evaluation at momentum point y
- Proper Lipschitz constant for step size
- Multiplicative updates that preserve non-negativity naturally
- Aggressive beta schedule up to 50000
- Multi-length exploration (8192, 16384)
- Random restarts with geometrically decaying perturbations

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_gradient(a, c, beta):
    """Compute LSE-smoothed gradient of max(conv(a,a))."""
    n = len(a)
    cm = c.max()
    spread = (c - cm).max()
    if spread > 1e-15:
        beta_eff = min(beta, 400.0 / spread)
    else:
        beta_eff = beta
    scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    p = e / e.sum()
    fsize = 1 << (len(p) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return grad, beta_eff


def compute_peak_gradient(a, c, top_k=16, temperature=1.0):
    """Weighted subgradient from top-k peaks with exponential temperature weighting."""
    n = len(a)
    top_k = min(top_k, len(c))
    order = np.argsort(c)[::-1][:top_k]
    c_top = c[order]
    c_min_top = c_top.min()
    c_max_top = c_top.max()
    spread = c_max_top - c_min_top
    if spread > 1e-15:
        weights = np.exp(-temperature * (c_max_top - c_top) / spread)
    else:
        weights = np.ones(top_k)
    weights = weights / weights.sum()
    w = np.zeros(len(c))
    w[order] = weights
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_refined(a_init, S, max_time, rng, beta_start=15.0, beta_end=30000.0):
    """
    FISTA with proper gradient at y, Lipschitz-based step, and
    geometrically-decaying beta annealing. Periodic restarts.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    stagnation = 0
    restart_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Beta schedule: smooth exponential ramp with soft cap
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        beta = min(beta, beta_end)

        # Evaluate at y (momentum point)
        c_y = fft_convolve_self(y)
        if c_y.max() < best:
            best = c_y.max()
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Gradient at y
        grad, beta_eff = compute_lse_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Lipschitz constant: L = 2 + 2*beta_eff*||y||^2
        y_norm_sq = np.sum(y * y)
        L = 2.0 + 2.0 * beta_eff * y_norm_sq
        step = 1.0 / L

        # Extra safety: cap step so we don't overshoot
        # If gradient is very large relative to y, reduce step
        grad_inf = np.max(np.abs(grad))
        if grad_inf > 0:
            step_cap = 0.5 * np.max(y) / grad_inf
            step = min(step, step_cap)

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if c_a.max() < best:

            best_a = a.copy()

    return best_a, best


def run_multiplicative_refined(a_init, S, max_time, rng, beta_start=30.0, beta_end=5000.0):
    """
    Multiplicative (exponential) update: a <- a * exp(-eta * grad_log)
    where grad_log = a * grad_a (gradient in log-parameterization).
    Preserves non-negativity automatically.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.8)
        beta = min(beta, beta_end)

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        grad, beta_eff = compute_lse_gradient(a, c, beta)

        # Gradient in log-space: grad_log[m] = a[m] * grad[m]
        grad_log = a * grad
        gmax = np.max(np.abs(grad_log))
        if gmax < 1e-15:
            break

        # Step size: eta = 1/gmax gives a "normalized" step
        # But we can be more aggressive early on
        eta = 1.0 / (gmax * (1.0 + 0.5 * progress))

        a_new = a * np.exp(-eta * grad_log)

        a_new = a_new * (S / a_new.sum())

        # Safety check: revert if objective worsened too much

            # Halve the step

            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new * (S / a_new.sum())
            c_new2 = fft_convolve_self(a_new)
            if c_new2.max() > c.max() * 1.05 + 1e-10:
                # Quarter the step

                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new * (S / a_new.sum())

    return best_a, best

        grad, beta_eff = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        L = 2.0 + 2.0 * beta_eff * np.sum(a * a)
        step = 1.0 / L
        a = project_simplex(a - step * grad, S)

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

    return best_a, best


def generate_inits(n, rng):
    """Generate diverse initializations for optimization."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Various window functions
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(np.pi * tt) ** 0.5)

    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.cos(10 * np.pi * tt + 0.5))
    inits.append(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt))

    # Random
    inits.append(rng.random(n) + 0.2)
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    # Log-spaced decay
    inits.append(np.exp(-3.0 * tt))
    inits.append(1.0 / (1.0 + 10.0 * tt))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Hybrid approach: FISTA + multiplicative updates + peak shaving,
    with adaptive LSE smoothing and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    print("Starting optimization, budget=%.0fs" % budget_s)
    t0 = time.monotonic()

    # Load previous best if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            given_s = given.sum()
            if given_s < 0.01:
                given = None
            else:
                print("Using given sequence of length %d, sum=%.2f" % (len(given), given_s))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Determine lengths to try
    # Primary: 8192 (proven). Secondary: 16384 (more degrees of freedom).
    lengths_to_try = []
    if given is not None:
        if len(given) == 8192:
            lengths_to_try = [8192, 16384]
        elif len(given) == 16384:
            lengths_to_try = [16384, 8192]
        elif 2048 <= len(given) <= 32768:
            lengths_to_try = [len(given), 8192, 16384]
        else:
            lengths_to_try = [8192, 16384]
    else:
        lengths_to_try = [8192, 16384]

    total_time = deadline - time.monotonic()

    # Time allocation: 60% to primary (first in list), 30% to second, 10% buffer
    if len(lengths_to_try) >= 2:
        time_alloc = {lengths_to_try[0]: total_time * 0.55,
                      lengths_to_try[1]: total_time * 0.30}
        for n in lengths_to_try[2:]:
            time_alloc[n] = total_time * 0.10 / max(1, len(lengths_to_try) - 2)
    elif len(lengths_to_try) == 1:
        time_alloc = {lengths_to_try[0]: total_time * 0.9}
    else:
        time_alloc = {8192: total_time * 0.8}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        t_budget = time_alloc.get(n, 100)
        S = np.sqrt(2.0 * n)

        print("\n=== n=%d, time budget=%.0fs ===" % (n, t_budget))
        tn_start = time.monotonic()

        # Prepare initializations
        inits = []
        if given is not None:
            if len(given) == n:
                inits.append(given.copy())
            elif len(given) > n:
                # Downsample by averaging
                ratio = len(given) // n
                if ratio >= 1:
                    usable = given[:n * ratio]
                    small = usable.reshape(n, ratio).mean(axis=1)
                    inits.append(small)
            else:
                # Upsample by interpolation
                t_old = np.linspace(0, 1, len(given))
                t_new = np.linspace(0, 1, n)
                inits.append(np.interp(t_new, t_old, given))

        # Add generated inits
        gen_inits = generate_inits(n, rng)
        inits.extend(gen_inits[:4])

        best_a_n = None
        best_val_n = np.inf

        # Phase A: Run different methods on different inits
        # Allocate time for 4 sub-phases
        n_methods = min(len(inits), 5)
        t_per_method = t_budget * 0.50 / n_methods

        for i in range(n_methods):
            if time.monotonic() > deadline - 60:
                break
            t_avail = min(t_per_method, deadline - time.monotonic() - 60)
            if t_avail < 15:
                continue

            init = inits[i]
            n_i = len(init)
            S_i = np.sqrt(2.0 * n_i)

            # Select method based on index
            if i == 0:
                # Best init: FISTA with moderate beta
                method_name = "FISTA-moderate"
                a_opt, val = run_fista_refined(init, S_i, t_avail, rng,
                                                beta_start=15.0, beta_end=10000.0)
            elif i == 1:
                # FISTA with high beta
                method_name = "FISTA-high"
                a_opt, val = run_fista_refined(init, S_i, t_avail, rng,
                                                beta_start=100.0, beta_end=50000.0)
            elif i == 2:
                # Multiplicative
                method_name = "Multiplicative"
                a_opt, val = run_multiplicative_refined(init, S_i, t_avail, rng,
                                                         beta_start=30.0, beta_end=5000.0)
            elif i == 3:
                # Peak shave
                method_name = "PeakShave"
                a_opt, val = run_peak_shave_refined(init, S_i, t_avail, rng)
            else:
                # Another FISTA variant
                method_name = "FISTA-slow"
                a_opt, val = run_fista_refined(init, S_i, t_avail, rng,
                                                beta_start=5.0, beta_end=20000.0)

            val_check = fft_convolve_self(a_opt).max()
            obj_i = 2.0 * n_i * val_check / (a_opt.sum() ** 2)

            if val_check < best_val_n:
                best_val_n = val_check
                best_a_n = a_opt.copy()
            print("  [%d] %s (n=%d): max_b=%.8f, obj=%.8f, time=%.0fs" %
                  (i, method_name, n_i, val_check, obj_i, time.monotonic() - tn_start))

        if best_a_n is None:
            best_a_n = np.ones(n) * S / n
            best_val_n = fft_convolve_self(best_a_n).max()

        # Phase B: Polish best with high-beta FISTA
        t_polish1 = min(t_budget * 0.20, deadline - time.monotonic() - 60)
        if t_polish1 > 20:
            a_p1, val_p1 = run_fista_refined(best_a_n, len(best_a_n), t_polish1, rng,
                                              beta_start=200.0, beta_end=50000.0)
            if val_p1 < best_val_n:
                best_val_n = val_p1
                best_a_n = a_p1.copy()
                print("  After high-beta polish: max_b=%.8f" % best_val_n)

        # Phase C: Multiplicative refinement
        t_mul = min(t_budget * 0.12, deadline - time.monotonic() - 60)
        if t_mul > 15:
            a_m, val_m = run_multiplicative_refined(best_a_n, len(best_a_n), t_mul, rng,
                                                     beta_start=100.0, beta_end=10000.0)
            if val_m < best_val_n:
                best_val_n = val_m
                best_a_n = a_m.copy()
                print("  After multiplicative refine: max_b=%.8f" % best_val_n)

        # Phase D: Peak shave with high k
        t_ps = min(t_budget * 0.08, deadline - time.monotonic() - 60)
        if t_ps > 10:
            a_ps, val_ps = run_peak_shave_refined(best_a_n, len(best_a_n), t_ps, rng)
            if val_ps < best_val_n:
                best_val_n = val_ps
                best_a_n = a_ps.copy()
                print("  After peak shave: max_b=%.8f" % best_val_n)

        # Update global best
        n_best = len(best_a_n)
        obj_n = 2.0 * n_best * best_val_n / (best_a_n.sum() ** 2)
        if obj_n < global_best:
            global_best = obj_n
            global_a = best_a_n.copy()
            print("  >>> New global best: obj=%.8f (n=%d)" % (global_best, n_best))

    # Final intensive polishing with all remaining time
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        if remaining > 50:
            print("\n=== Final polishing (%.0fs remaining, n=%d) ===" % (remaining, n_best))

            # Split remaining time into segments
            seg1 = remaining * 0.45
            seg2 = remaining * 0.30
            seg3 = remaining * 0.25

            # Segment 1: Very high beta FISTA
            a_f1, val_f1 = run_fista_refined(global_a, n_best, seg1, rng,
                                              beta_start=1000.0, beta_end=100000.0)
            if val_f1 < fft_convolve_self(global_a).max():
                global_a = a_f1.copy()
                global_best = 2.0 * n_best * val_f1 / (global_a.sum() ** 2)
                print("  After final FISTA: obj=%.8f" % global_best)

            # Segment 2: Multiplicative with very high beta
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_f2, val_f2 = run_multiplicative_refined(global_a, n_best,
                                                           min(seg2, remaining2), rng,
                                                           beta_start=1000.0, beta_end=20000.0)
                if val_f2 < fft_convolve_self(global_a).max():
                    global_a = a_f2.copy()
                    global_best = 2.0 * n_best * val_f2 / (global_a.sum() ** 2)
                    print("  After final multiplicative: obj=%.8f" % global_best)

            # Segment 3: Peak shaving
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15:
                a_f3, val_f3 = run_peak_shave_refined(global_a, n_best,
                                                       min(seg3, remaining3), rng)
                if val_f3 < fft_convolve_self(global_a).max():
                    global_a = a_f3.copy()
                    global_best = 2.0 * n_best * val_f3 / (global_a.sum() ** 2)
                    print("  After final peak shave: obj=%.8f" % global_best)

    # Verify with evaluate_sequence
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            print("WARNING: sum too small, using fallback")
            global_a = np.ones(256) * np.sqrt(512.0) / 256
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * np.sqrt(512.0) / 256
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback verification: %.8f" % ver)
    else:
        global_a = np.ones(256) * np.sqrt(512.0) / 256
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - t0
    print("\nDone. Total time: %.0fs. Sequence length: %d. Objective: %.8f" %
          (elapsed, len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
