with direct convolution flattening in frequency domain.

Key improvements:
1. Multi-scale upstarting: optimize at increasing lengths for better initialization.
2. Direct convolution flattening: shape FFT(P)^2 (not |FFT(P)|^2) to reduce peak.

4. Nesterov-accelerated projected gradient with adaptive step size.

import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, L=None):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    if L is None:
        L = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, L)
    c = np.real(np.fft.ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def weighted_subgradient_step(a, step, top_k=50, temperature=1.0):
    """
    One step of weighted subgradient descent.
    Weights are based on how much each lag exceeds the mean.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    c = fft_convolve_self(a, L)
    S = a.sum()
    mean_c = S * S / (2 * n - 1)
    
    # Excess over mean
    excess = np.maximum(c - mean_c, 0.0)
    total_excess = excess.sum()
    
    if total_excess < 1e-30:
        return a
    
    # Weighted subgradient
    # Use top-k lags with softmax weighting
    # Actually, let's use all lags with exponential weighting
    weights = np.exp(excess / max(temperature * mean_c, 1e-30))
    weights = weights / weights.sum()
    
    # Only use the top-k for efficiency
    top_idx = np.argsort(weights)[::-1][:top_k]
    top_weights = weights[top_idx] / weights[top_idx].sum()
    
    # Compute gradient: grad[j] = sum_k top_weights[k] * a[k-j] (if valid)
    # This is a weighted cross-correlation
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = top_weights[i]
    
    # Gradient via FFT
    fsize = L
    W = np.fft.fft(w_full, fsize)
    A = np.fft.fft(a, fsize)
    grad = 2.0 * np.real(np.fft.ifft(W * A))[:n]
    
    # Take step
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new * (S / s)
    return a_new


def nesterov_subgradient(a_init, n, S, max_time, rng, top_k=100, 
                          step_init=None, decay=0.01, target_obj=1.50):

    Nesterov-accelerated projected subgradient descent.
    """

        step_init = 2.0 * S / (n * 0.1)
    
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    L = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y, L)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
        
        if best_score <= target_obj:
            break
        
        # Weighted subgradient
        mean_c = (y.sum() ** 2) / (2 * n - 1)
        excess = np.maximum(c - mean_c, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            break
        
        # Top-k selection
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx] - mean_c
        top_vals = np.maximum(top_vals, 1e-30)
        # Exponential weighting
        w = np.exp(top_vals / max(mean_c * 0.1, 1e-30))
        w = w / w.sum()
        
        # Compute gradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = w[i]
        
        W = np.fft.fft(w_full, L)
        A_y = np.fft.fft(y, L)
        grad = 2.0 * np.real(np.fft.ifft(W * A_y))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = step_init / (1.0 + decay * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15) * gnorm, S)
        
        # Wait, let me reconsider the step size
        # The gradient of max(c) w.r.t. a is: for the peak lag k*, grad[j] = a[k*-j] (if valid)
        # The norm of this gradient is approximately sqrt(n) * mean(a) = sqrt(n) * S/n = S/sqrt(n)
        # So a reasonable step is O(S/n) for the normalized gradient
        
        # Let me use: step * grad / gnorm * step_scale
        # where step_scale controls the actual movement
        # Actually, let me just use step * grad directly with proper scaling

        y = project_simplex(y, S)
        
        a = x_new
        t = t_new

            a = project_simplex(a + noise * 0.5, S)

    return best_local, best_score


def spectral_correction_step(a, strength, rng=None):
    """
    Directly shape FFT(P)^2 to reduce the peak of the convolution.
    This is different from flattening |FFT(P)|^2.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    S = a.sum()
    
    A = np.fft.fft(a, L)
    Q = A * A  # This determines the convolution
    c = np.real(np.fft.ifft(Q))[:2 * n - 1]
    
    mean_c = S * S / (2 * n - 1)
    excess = np.maximum(c - mean_c, 0.0)
    
    if excess.max() < 1e-30:
        return a.copy()
    
    # We want to reduce excess. In frequency domain:
    # c = ifft(Q), so ifft(Q - eta * FFT(excess_padded)) = c - eta * excess
    # We need Q_new = Q - eta * FFT(excess_padded) to be a valid square.
    # First order: A_new ≈ A - eta * FFT(excess_padded) / (2A)
    # But this has division issues. Let's use a safer approach.
    
    # Alternative: modify |A| and phase separately
    # Q = |A|^2 * exp(2i*phase(A))
    # We want Q_new = Q * (1 - eta * FFT(excess_padded)/Q)
    # So |A_new| = |A| * sqrt(1 - eta * real(FFT(excess_padded)/Q))
    # And phase(A_new) = phase(A) - eta/2 * imag(FFT(excess_padded)/Q) / |Q|
    
    # Simpler approach: just add a small negative perturbation to Q and take sqrt
    E_padded = np.zeros(L, dtype=np.complex128)
    E_padded[:2 * n - 1] = excess
    E_fft = np.fft.fft(E_padded)
    
    # Dampen high-frequency content of excess
    # (This effectively does a low-pass on the correction)
    mask = np.ones(L)
    # Reduce the correction at high frequencies to avoid instability
    center = L // 2
    for k in range(L):
        dist = min(abs(k - center), k) if k > center else k
        # Actually, let's just use a simple attenuation
        freq_dist = min(k, L - k)  # distance from DC or Nyquist
        mask[k] = np.exp(-freq_dist / (L * 0.1))
    
    E_fft_smoothed = E_fft * mask
    
    # First-order correction to A
    # A_new^2 = Q - strength * E_fft_smoothed
    # A_new ≈ A - strength * E_fft_smoothed / (2A)
    # Safe division:
    A_abs = np.abs(A)
    safe_A = np.where(A_abs > 1e-10, A, 1e-10)
    correction = strength * E_fft_smoothed / (2.0 * safe_A)
    
    A_new = A - correction
    a_new_full = np.real(np.fft.ifft(A_new))[:n]
    
    # Project to non-negative with same sum
    a_new = np.maximum(a_new_full, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    return a_new

def upsample(a, new_n):
    """Upsample array a to length new_n via linear interpolation."""
    old_n = len(a)

    best_local = a.copy()
    it = 0
    L = 1 << (2 * n - 1).bit_length()
    patience = 0
    patience_limit = max(2000, n // 2)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y, L)
        cur_max = c.max()
        Scur = y.sum()
        sc = 2.0 * n * cur_max / (Scur * Scur)
        if sc < best_score - 1e-8:
            best_score = sc
            best_local = y.copy()
            patience = 0
        else:
            patience += 1

        if sc <= target_obj:
            break
        if patience > patience_limit:
            break

        mean_c = (Scur * Scur) / (2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = np.maximum(c[top_idx] - mean_c, 1e-30)
        w = np.exp(top_vals / max(mean_c * 0.05, 1e-30))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = w[i]

        W = np.fft.fft(w_full, L)
        A_y = np.fft.fft(y, L)
        grad = 2.0 * np.real(np.fft.ifft(W * A_y))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_init / (1.0 + 0.005 * it)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * (S / s_y)

        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.4:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = project_simplex(a + 0.5 * noise, S)
            t = 1.0
            patience = 0

    return best_local, best_score


def spectral_correction_step(a, strength, L=None):
    """
    Shape FFT(P)^2 to reduce convolution peak by correcting the spectrum.
    """
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    S = a.sum()

    A = np.fft.fft(a, L)
    Q = A * A
    c = np.real(np.fft.ifft(Q))[:2 * n - 1]

    mean_c = (S * S) / (2 * n - 1)
    excess = np.maximum(c - mean_c, 0.0)
    if excess.max() < 1e-30:
        return a.copy()

    E_padded = np.zeros(L, dtype=np.complex128)
    E_padded[:2 * n - 1] = excess
    E_fft = np.fft.fft(E_padded)

    # Low-pass the correction for stability
    k = np.arange(L)
    freq_dist = np.minimum(k, L - k)
    mask = np.exp(-freq_dist / max(L * 0.05, 1))
    E_fft = E_fft * mask

    A_abs = np.abs(A)
    safe_A = np.where(A_abs > 1e-10, A, 1e-10 + 1j * 1e-10)
    correction = strength * E_fft / (2.0 * safe_A)

    A_new = A - correction
    a_new_full = np.real(np.fft.ifft(A_new))[:n]

    a_new = np.maximum(a_new_full, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    return a_new


def variance_reduction_step(a, strength, L=None):
    """Reduce variance of convolution via spectral reweighting (phase-aware)."""
    n = len(a)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    S = a.sum()

    A = np.fft.fft(a, L)
    Q = A * A
    c = np.real(np.fft.ifft(Q))[:2 * n - 1]

    mean_c = (S * S) / (2 * n - 1)
    if mean_c < 1e-30:
        return a.copy()

    # Reweight Q to reduce variance of ifft(Q)
    Q_reweight = Q * (1.0 - strength * (Q - Q.mean()) / (Q.var() + 1e-30))
    Q_reweight = np.maximum(np.real(Q_reweight), 0.0) + 1j * np.imag(Q_reweight)

    A_new = np.sqrt(Q_reweight)
    a_new_full = np.real(np.fft.ifft(A_new))[:n]

    a_new = np.maximum(a_new_full, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    return a_new


def make_structured_init(n, rng, style="mixed"):
    """Generate a structured initial sequence."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if style == "uniform":
        a = np.ones(n)
    elif style == "sine":
        a = 0.5 + 0.4 * np.sin(2 * np.pi * tt)
    elif style == "multi_sine":
        a = 0.4 + 0.3 * np.sin(2 * np.pi * tt) + 0.2 * np.sin(2 * np.pi * 3 * tt + 0.7)
    elif style == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt * tt))
    elif style == "smooth_noise":
        noise = rng.standard_normal(n)
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(noise, kernel, mode='same')
        smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
        a = 0.3 + 0.5 * smooth
    elif style == "two_bump":
        a = np.exp(-0.5 * ((tt - 0.25) * 5.0) ** 2) + np.exp(-0.5 * ((tt - 0.75) * 5.0) ** 2) + 0.1
    else:
        # Mixed: sum of sines with random phases
        a = np.ones(n) * 0.3
        for j in range(5):
            freq = rng.integers(1, n // 4 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.15 / (j + 1)
            a += amp * np.sin(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    return a


def optimize_at_scale(a_init, n, S, budget_s, rng, target_obj=1.50):
    """Full optimization pipeline at a single scale."""
    t_start = time.monotonic()
    best_score = np.inf
    best_a = None
    L = 1 << (2 * n - 1).bit_length()

    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    sc = compute_score(a)
    best_score = sc
    best_a = a.copy()

    def try_update(a_cand):
        nonlocal best_score, best_a
        sc_c = compute_score(a_cand)
        if sc_c < best_score:
            best_score = sc_c
            best_a = a_cand.copy()
        return sc_c

    # Phase 1: Alternating spectral correction + subgradient
    remaining = budget_s - (time.monotonic() - t_start)
    n_blocks = max(3, int(remaining / 8.0))
    block_time = remaining / n_blocks

    strength = 0.3
    for block in range(n_blocks):
        if time.monotonic() - t_start > budget_s - 10:
            break

        # Spectral correction iterations
        n_corr = max(10, int(block_time * 0.4 / max(0.001, L * 14 * 1e-9)))
        n_corr = min(n_corr, 3000)
        for it in range(n_corr):
            if time.monotonic() - t_start > budget_s - 10:
                break
            a = spectral_correction_step(a, strength, L)
            a = project_simplex(a, S)
            if it % 50 == 49:
                sc = try_update(a)
                if sc <= target_obj:
                    return best_a, best_score

        # Subgradient phase
        sg_time = min(4.0, budget_s - 10 - (time.monotonic() - t_start))
        if sg_time > 1:
            a, sc = nesterov_subgradient(a, n, S, sg_time, rng,
                                          top_k=min(200, n // 10), target_obj=target_obj)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Anneal
        strength = max(0.02, strength * 0.85)

        # Occasional perturbation
        if block % 3 == 2 and rng.random() < 0.5:
            a = project_simplex(a + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)

    # Phase 2: Extended subgradient with higher top_k
    remaining = budget_s - (time.monotonic() - t_start)
    if remaining > 20:
        a, sc = nesterov_subgradient(best_a, n, S, remaining * 0.5,
                                      rng, top_k=min(500, n // 5), target_obj=target_obj)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Phase 3: Final fine-tuning
    remaining = budget_s - (time.monotonic() - t_start)
    if remaining > 10:
        for _ in range(3):
            if time.monotonic() - t_start > budget_s - 5:
                break
            a_tmp = spectral_correction_step(best_a, 0.05, L)
            a_tmp = project_simplex(a_tmp, S)
            a_tmp, sc_tmp = nesterov_subgradient(a_tmp, n, S, remaining * 0.1,
                                                  rng, top_k=min(300, n // 8),
                                                  target_obj=target_obj)
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_a = a_tmp.copy()
            remaining = budget_s - (time.monotonic() - t_start)
            if remaining < 5:
                break

    return best_a, best_score


def multi_scale_optimize(target_n, budget_s, rng, start_n=64, target_obj=1.50):
    """Multi-scale optimization with upstarting."""
    t_start = time.monotonic()
    best_score = np.inf
    best_a = None

    scales = []
    n = start_n
    while n < target_n:
        scales.append(n)
        n = min(n * 2, target_n)
    if scales[-1] != target_n:
        scales.append(target_n)

    # Initial solution at smallest scale
    n0 = scales[0]
    S0 = np.sqrt(2.0 * n0)
    a = make_structured_init(n0, rng, style="mixed")
    a = a / a.sum() * S0

    for i, n in enumerate(scales):
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed - 10
        if remaining < 10:
            break

        S = np.sqrt(2.0 * n)
        if i > 0:
            a = upsample(a, n)
            a = a / a.sum() * S

        # Budget allocation: more time for larger scales
        if i < len(scales) - 1:
            sc_budget = min(remaining * 0.4, 120.0)
        else:
            sc_budget = remaining

        a, sc = optimize_at_scale(a, n, S, sc_budget, rng, target_obj)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print("    [scale n=%d] score=%.6f (elapsed %.0fs)" % (n, best_score, time.monotonic() - t_start))

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-scale upstarting with direct convolution flattening.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    deadline = t_global_start + budget_s - 25.0

    print("Starting optimization with budget %.0fs" % budget_s)

    global_best = np.inf
    global_a = None

    # Load previous best if available
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size >= 64 and given.sum() > 0.01:
            n_given = len(given)
            sc_given = compute_score(given)
            print("Given sequence: n=%d, score=%.6f" % (n_given, sc_given))
            # Normalize
            S_g = np.sqrt(2.0 * n_given)
            given = given / given.sum() * S_g
            global_best = sc_given
            global_a = given.copy()
    except Exception:
        pass

    elapsed = time.monotonic() - t_global_start
    remaining = deadline - time.monotonic()

    # Strategy 1: Improve the given sequence directly
    if global_a is not None and remaining > 60:
        n_g = len(global_a)
        S_g = np.sqrt(2.0 * n_g)
        alloc = min(remaining * 0.35, 500.0)
        print("=== Improving given n=%d, budget=%.0fs ===" % (n_g, alloc))
        a1, sc1 = optimize_at_scale(global_a, n_g, S_g, alloc, rng, target_obj=1.50)
        if sc1 < global_best:
            global_best = sc1
            global_a = a1.copy()
            print("  Improved to %.6f" % global_best)

    elapsed = time.monotonic() - t_global_start
    remaining = deadline - time.monotonic()

    # Strategy 2: Multi-scale from scratch at various lengths
    lengths_to_try = [8192, 16384, 32768, 4096]
    # Remove already tried length
    if global_a is not None:
        lengths_to_try = [L for L in lengths_to_try if L != len(global_a)]

    if remaining > 60:
        per_len = remaining / max(1, len(lengths_to_try))
        for n in lengths_to_try:
            if time.monotonic() > deadline - 30:
                break
            alloc = min(per_len, deadline - time.monotonic() - 30)
            if alloc < 30:
                break
            print("=== Multi-scale target n=%d, budget=%.0fs ===" % (n, alloc))
            a_ms, sc_ms = multi_scale_optimize(n, alloc, rng, start_n=64, target_obj=1.50)
            if a_ms is not None and sc_ms < global_best:
                global_best = sc_ms
                global_a = a_m
                print("  New global best: %.6f at n=%d" % (global_best, n))
            else:
                print("  Best for n=%d: %.6f" % (n, sc_ms))

    # Strategy 3: Random restarts from best
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n_r = len(global_a)
        S_r = np.sqrt(2.0 * n_r)
        n_restarts = max(1, int(remaining / 40))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Perturb and re-optimize
            a_r = global_a + rng.normal(0, 0.03 * S_r / np.sqrt(n_r), n_r)
            a_r = project_simplex(a_r, S_r)
            a_r, sc_r = optimize_at_scale(a_r, n_r, S_r, min(30.0, deadline - time.monotonic() - 20),
                                           rng, target_obj=global_best)
            if sc_r < global_best:
                global_best = sc_r
                global_a = a_r.copy()
                print("  [restart %d] new best=%.6f" % (r, global_best))

    # Final verification
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = compute_score(global_a)
        print("Using fallback uniform, score=%.4f" % global_best)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
        global_best = compute_score(global_a)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
    else:
        print("WARNING: verification failed, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])

    print("Final: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
