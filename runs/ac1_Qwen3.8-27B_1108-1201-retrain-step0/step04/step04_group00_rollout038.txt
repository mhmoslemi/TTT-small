"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj_normalized(a):
    """Compute 2*n*max_conv/sum^2 (scale invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE gradient."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def effective_topk(c, rel_tol=0.01):
    """Estimate effective number of top lags within rel_tol of max."""
    max_val = c.max()
    if max_val < 1e-30:
        return 1
    count = np.sum(c >= max_val * (1.0 - rel_tol))
    return int(count)


def run_adaptive_fista(a_init, max_time, rng, beta_init=20.0, beta_max=50000.0,

    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    periodic_it = 0

        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        eff_k = effective_topk(c, rel_tol=0.01)

        gnorm = np.linalg.norm(grad)

        a = x_new
        t_mom = t_new

        # Annealing: periodic restarts with noise

def run_exact_polish(a_init, max_time, rng):

    Intensive polishing with exact subgradient and adaptive tolerance.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance
        if stuck < 100:
            tol_frac = 0.0
        elif stuck < 500:
            tol_frac = 0.002
        elif stuck < 2000:
            tol_frac = 0.005
        elif stuck < 10000:
            tol_frac = 0.01
        else:
            tol_frac = 0.02

        grad = exact_subgradient(y, c, tol_frac=tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step
        step = 1.0 / (0.5 * gnorm + 1e-10)
        step = min(step, S / (0.5 * gnorm + 1e-10))
        
        # Add small LSE component for smoothing
        if stuck > 500:
            lse_g = lse_gradient(y, c, 50.0)
            grad = 0.8 * grad + 0.2 * lse_g
            gnorm = np.linalg.norm(grad)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when very stuck

            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))

            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:

    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase adaptive optimization combining:
1. LP-based active set identification (dual covering LP solved greedily)
2. FISTA with flatness-adaptive LSE smoothing
3. Multi-mode gradient (LSE + subgradient blend based on stuckness)
4. Multi-size exploration (n, 2n) with upsampled initialization
5. Simulated annealing restarts to escape local minima
6. Final intensive polish with exact subgradient
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2 (scale invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_from_active(y, c, active_set):
    """Subgradient using a given active set of lags."""
    n = len(y)
    w_full = np.zeros(len(c))
    w_full[active_set] = 1.0 / len(active_set)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    return subgradient_from_active(y, c, active)


def solve_dual_covering(a, M):
    """
    Solve the dual covering LP greedily:
    min sum y_k s.t. sum_{k=j}^{j+n-1} y_k >= 1/a[j] for all j, y_k >= 0
    Returns (y, active_lags) where active_lags are positions where y > 0.
    """
    n = len(a)
    size = 2 * n - 1
    y = np.zeros(size)
    d = 1.0 / np.maximum(a, 1e-30)

    window_sum = np.zeros(n)
    for j in range(n):
        deficit = d[j] - window_sum[j]
        if deficit > 0:
            pos = j + n - 1
            y[pos] += deficit
            window_sum[j:] += deficit

    active_lags = np.where(y > 1e-15)[0]
    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(a * a)])
    return y, active_lags


def effective_topk(c, rel_tol=0.01):
    """Estimate effective number of top lags."""
    max_val = c.max()
    if max_val < 1e-30:
        return 1
    count = np.sum(c >= max_val * (1.0 - rel_tol))
    return max(1, int(count))


def run_adaptive_fista(a_init, max_time, rng, beta_init=20.0, beta_max=50000.0,
                       base_step=1.0, use_lp=False):
    """
    FISTA with adaptive beta, multi-mode gradient, and optional LP active set.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    beta = beta_init
    lp_interval = 200

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive beta based on flatness of top
        eff_k = effective_topk(c, rel_tol=0.01)
        target_beta = max(5.0, min(beta_max, 200.0 / max(1, eff_k)))
        beta = 0.9 * beta + 0.1 * target_beta
        beta = max(beta, beta_init * np.exp(0.002 * it))
        beta = min(beta, beta_max)

        # Compute gradient based on stuckness
        lse_grad = lse_gradient(y, c, beta)

        if use_lp and it % lp_interval == 1 and stuck_count > 50:
            _, active_lags = solve_dual_covering(y, cur_max)
            lp_grad = subgradient_from_active(y, c, active_lags)
            if stuck_count < 500:
                grad = 0.5 * lse_grad + 0.5 * lp_grad
            else:
                grad = 0.3 * lse_grad + 0.7 * lp_grad
        elif stuck_count < 100:
            grad = lse_grad
        elif stuck_count < 1000:
            sub_grad = exact_subgradient(y, c, tol_frac=0.005)
            grad = 0.7 * lse_grad + 0.3 * sub_grad
        elif stuck_count < 5000:
            sub_grad = exact_subgradient(y, c, tol_frac=0.01)
            grad = 0.4 * lse_grad + 0.6 * sub_grad
        else:
            sub_grad = exact_subgradient(y, c, tol_frac=0.03)
            grad = 0.2 * lse_grad + 0.8 * sub_grad

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = base_step / max(beta, 1.0)
        if stuck_count > 3000:
            step *= 0.5

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Annealing restarts
        if stuck_count > 3000 and it % 2000 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
            if stuck_count > 15000:
                # Hard restart from best
                pert_scale = 0.02 * S / np.sqrt(n)
                y = project_simplex(best_a + rng.normal(0, pert_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_exact_polish(a_init, max_time, rng):
    """

    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    if a.sum() < 1e-15:

    else:

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance for active set
        if stuck < 50:
            tol_frac = 0.0
        elif stuck < 300:
            tol_frac = 0.001
        elif stuck < 1000:
            tol_frac = 0.003
        elif stuck < 5000:
            tol_frac = 0.008
        else:
            tol_frac = 0.015

        grad = exact_subgradient(y, c, tol_frac=tol_frac)

        # Add small LSE component for stability when stuck
        if stuck > 200:
            lse_g = lse_gradient(y, c, 100.0)
            blend = min(0.3, 0.05 * np.log1p(stuck / 100.0))
            grad = (1.0 - blend) * grad + blend * lse_g

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: 1/L approximation
        step = min(S / (0.5 * gnorm + 1e-10), 2.0 / (gnorm + 1e-10))
        step = min(step, S * 0.1)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when very stuck

            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
