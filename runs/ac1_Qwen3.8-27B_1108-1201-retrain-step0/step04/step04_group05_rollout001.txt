    best_cmax = c_max
    
    for bt in range(max_backtracks):
        candidate = project_simplex(a + step * direction, S)
        c_ls = compute_conv(candidate, N)
        cm = c_ls.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = candidate.copy()
            if cm < c_max * 0.999:
                break
        step *= 0.5
    
    return best_a, best_cmax


def single_peak_targeted(a, N, c, S, max_steps):
    """Aggressively target the single maximum peak for reduction."""
    n = len(a)
    best_a = a.copy()
    best_cmax = c.max()
    
    for _ in range(max_steps):
        peak_lag = int(np.argmax(c))
        # Gradient of c[peak_lag] with respect to a
        # c[k] = sum_i a[i]*a[i+k], grad_i = a[i+k] + a[i-k] (where defined)
        grad = np.zeros(n)
        k = peak_lag
        if k == 0:
            grad = 2.0 * a
        else:
            grad[:n - k] += a[k:]
            grad[k:] += a[:n - k]
        
        dir_norm = np.linalg.norm(grad)
        if dir_norm < 1e-18:
            break
        
        # Try multiple step sizes
        step = 0.5 * S / (n * dir_norm)
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            candidate = project_simplex(a - scale * step * grad, S)
            c_test = compute_conv(candidate, N)
            cm = c_test.max()
            if cm < best_cmax - 1e-12:
                best_cmax = cm
                best_a = candidate.copy()
                a = candidate
                c = c_test
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses phased top-k subgradient with spectral mutations and peak targeting.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine length
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # Initialize
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a = project_simplex(np.clip(given, 0, 1000), S)
            print("  Initialized from height_sequence_1")
        else:
            a = project_simplex(rng.random(n), S)
    except Exception:
        a = project_simplex(rng.random(n), S)
        print("  Initialized randomly")

    global_best = np.inf
    global_a = None

    def try_update(a_cand, label=""):
        nonlocal global_best, global_a
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            if label:
                print("  [%s] max_conv=%.6f (score=%.6f)" % (label, sc, sc))
        return sc, c

    # Initial evaluation
    cur_c = compute_conv(a, N)
    cur_max = cur_c.max()
    global_best = cur_max
    global_a = a.copy()
    print("  Initial: max_conv=%.6f" % global_best)

    # === Main optimization loop ===
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    last_improve_time = start_time

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = elapsed / (budget_s - 30.0)
        
        # Determine phase
        if progress < 0.25:
            top_k = max(50, int(50 * (1 - progress * 4)))
            step_scale = 3.0
            label_prefix = "P1"
        elif progress < 0.5:
            top_k = max(10, int(50 - 40 * (progress - 0.25) * 4))
            step_scale = 2.0
            label_prefix = "P2"
        elif progress < 0.75:
            top_k = max(2, int(10 - 8 * (progress - 0.5) * 4))
            step_scale = 1.5
            label_prefix = "P3"
        else:
            top_k = 1
            step_scale = 1.0
            label_prefix = "P4"

        # Compute subgradient
        grad = topk_subgrad_conv(a, cur_c, N, top_k)
        dir_norm = np.linalg.norm(grad)
        
        if dir_norm > 1e-18:
            direction = -grad / dir_norm
            a_new, cmax_new = adaptive_line_search(a, direction, S, N, cur_max, max_backtracks=10)
        else:
            a_new, cmax_new = a, cur_max
        
        if cmax_new < cur_max - 1e-10:
            a = a_new
            cur_max = cmax_new
            cur_c = compute_conv(a, N)
            stall_count = 0
        else:
            stall_count += 1
        
        # Try to update global best every few iterations
        if it % 10 == 0:
            try_update(a, label_prefix + "/" + str(it))

        # Periodic spectral mutations when stalled
        if stall_count > 20 and it % 5 == 0:
            mutation_strength = 0.01 + 0.03 * (stall_count / 100.0)
            a_mut = spectral_mutate(a, N, rng, mutation_strength)
            c_mut = compute_conv(a_mut, N)
            if c_mut.max() < cur_max:
                a = a_mut
                cur_max = c_mut.max()
                cur_c = c_mut
                stall_count = 0
            elif rng.random() < 0.1:  # Accept some worse solutions (annealing)
                a = a_mut
                cur_max = c_mut.max()
                cur_c = c_mut
        
        # Periodic spectral peak shrinking
        if stall_count > 50 and it % 10 == 0:
            shrink_frac = 0.05 + 0.1 * min(1.0, stall_count / 200.0)
            a_shrunk = spectral_shrink_peaks(a, N, cur_c, rng, min(20, top_k + 5), shrink_frac)
            c_shr = compute_conv(a_shrunk, N)
            if c_shr.max() < cur_max:
                a = a_shrunk
                cur_max = c_shr.max()
                cur_c = c_shr
                stall_count = 0

        # Periodic single-peak targeting in later phases
        if progress > 0.6 and it % 20 == 0:
            a_peak, cmax_peak = single_peak_targeted(a, N, cur_c, S, max_steps=20)
            if cmax_peak < cur_max - 1e-10:
                a = a_peak
                cur_max = cmax_peak
                cur_c = compute_conv(a, N)
                stall_count = 0

        # Random restart (small perturbation) if very stuck
        if stall_count > 500 and it % 100 == 0:
            perturbation = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, perturbation, n), S)
            cur_c = compute_conv(a, N)
            cur_max = cur_c.max()
            stall_count = 0
            print("  [restart] max_conv=%.6f" % cur_max)

    # === Final polishing ===
    print("=== Final polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 5:
        # Single peak targeting
        a_final, cmax_final = single_peak_targeted(global_a, N, compute_conv(global_a, N), S,
                                                   max_steps=max(50, int(remaining // 0.05)))
        try_update(a_final, "final_peak")
        
        # More top-k with k=1
        time_budget = max(1.0, remaining * 0.3)
        t_polish = time.monotonic()
        a_curr = global_a.copy()
        c_curr = compute_conv(a_curr, N)
        m_curr = c_curr.max()
        
        while time.monotonic() - t_polish < time_budget:
            grad = topk_subgrad_conv(a_curr, c_curr, N, 1)
            dir_norm = np.linalg.norm(grad)
            if dir_norm < 1e-18:
                break
            direction = -grad / dir_norm
            a_new, cmax_new = adaptive_line_search(a_curr, direction, S, N, m_curr, max_backtracks=8)
            if cmax_new < m_curr - 1e-12:
                a_curr = a_new
                m_curr = cmax_new
                c_curr = compute_conv(a_curr, N)
            else:
                break
        
        try_update(a_curr, "polish_k1")

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, best max_conv=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
