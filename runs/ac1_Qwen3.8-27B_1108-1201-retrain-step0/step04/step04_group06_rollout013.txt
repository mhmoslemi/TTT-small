```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return v * (S / v.sum()) if v.sum() > 0 else np.ones(n) * S / n
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def compute_conv(a, N, n):
    """Compute linear self-convolution using FFT."""
    A = fft(a, N)
    b = np.real(ifft(A * A))[:2*n-1]
    return b

def phase_optimize(a, N, n, S, max_time, rng, top_k=10, lr_init=0.1):
    """
    Iterative phase optimization to minimize max(convolution).
    """
    a = a.copy()
    best_a = a.copy()
    best_score = np.inf
    lr = lr_init
    start = time.monotonic()
    m = np.arange(N)
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        A = fft(a, N)
        b = np.real(ifft(A * A))[:2*n-1]
        score = b.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top-k peak lags
        peak_indices = np.argsort(b)[::-1][:top_k]
        peak_vals = b[peak_indices]
        
        # Weight by excess above the average
        avg = S * S / (2*n - 1)
        excess = np.maximum(peak_vals - avg, 0)
        if excess.sum() < 1e-10:
            excess = peak_vals
        weights = excess / excess.sum()
        
        # Compute phase gradient (weighted sum)
        phi = np.angle(A)
        A_mag = np.abs(A)
        
        grad = np.zeros(N)
        for i, k in enumerate(peak_indices):
            k_int = int(k)
            if k_int >= N:
                k_int = k_int - N  # wrap around for circular
            grad += weights[i] * np.sin(2 * phi + 2 * np.pi * k_int * m / N)
        
        # Gradient descent on phase
        dphi = lr * grad
        A_new = A * np.exp(1j * dphi)
        
        # Enforce Hermitian symmetry
        # A_new[0] should be real, A_new[N/2] should be real (if N even)
        # A_new[f] = conj(A_new[N-f])
        A_new = 0.5 * (A_new + np.conj(A_new[np.mod(-np.arange(N), N)]))
        
        # Compute new a
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Check if the update actually helps
        b_check = compute_conv(a_new, N, n)
        if b_check.max() < score + 0.01 * score:  # allow small increase
            a = a_new
        else:
            # Revert and try smaller step
            a = a  # keep original
            lr *= 0.9
        
        # Cooling
        if it % 100 == 0:
            lr *= 0.995
        
        if best_score <= 1.45:
            break
    
    return best_a, best_score
```

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if np.sum(v) <= 0:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S)
