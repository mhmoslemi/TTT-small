import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_fft(a, N):
    """Compute FFT of a padded to size N."""
    a_pad = np.zeros(N)
    n = len(a)
    a_pad[:n] = a
    return fft(a_pad)

def spectral_shape_v2(a, N, n, top_k, alpha, S, beta):
    """
    Improved spectral shaping:
    - Uses soft-peak (softmax) weighting of the top autocorrelation values
    - Adaptive alpha based on how much the peak exceeds the mean
    - Preserves phase from current FFT
    """
    A = compute_fft(a, N)
    P = np.abs(A) ** 2
    
    # Compute autocorrelation from P
    c = np.real(ifft(P))[:2*n-1]
    c_max = c.max()
    c_mean = c.sum() / len(c)  # should be S^2/(2n-1)
    
    if c_max <= c_mean * 1.01:
        return a, c_max
    
    # Soft peak weighting
    # Use top-K with softmax weights
    sorted_idx = np.argsort(c)[::-1]
    top_indices = sorted_idx[:top_k]
    top_vals = c[top_indices]
    
    # Softmax weights ( temperature-controlled)
    if top_vals.max() > 1e-30:
        safe_beta = beta / max(top_vals.max(), 1e-30)
        exps = np.exp(safe_beta * (top_vals - top_vals.max()))
        weights = exps / exps.sum()
    else:
        weights = np.ones(top_k) / top_k
    
    # Compute weighted cosine
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    # Adaptive alpha
    excess_ratio = (c_max - c_mean) / max(c_max, 1e-10)
    alpha_eff = alpha * (0.5 + excess_ratio)
    
    # Shape the spectrum
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))
    
    # Reconstruct with preserved phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max

def nesterov_soft_peak(a_init, n, N, S, max_time, rng, top_k_init=10, top_k_max=200, beta=20.0):
    """
    Nesterov-accelerated projected gradient with soft-peak objective.
    Uses softmax-weighted top-K lags for a smoother subgradient.
    """
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # BB step size parameters
    prev_grad = None
    prev_x = None
    gamma = 2.0 * S / (n * (1.0 + 0.001))
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        
        c = fft_convolve_self(y)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()
        
        # Soft peak: top-K with softmax weights
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        
        # Softmax weights
        if top_vals.max() > 1e-30:
            safe_beta = beta / top_vals.max()
            exps = np.exp(safe_beta * (top_vals - top_vals.max()))
            weights = exps / exps.sum()
        else:
            weights = np.ones(top_k) / top_k
        
        # Compute gradient: 2 * sum_j weights[j] * f[lag_j - i] (flipped)
        # grad[i] = 2 * sum_j weights[j] * f[lag_j - i] (with bounds)
        # This is equivalent to 2 * (w_reversed * f) where w is the weight sequence
        grad = np.zeros(n)
        for j in range(top_k):
            lag = int(top_indices[j])
            w = weights[j]
            # (f*f)(lag) = sum_i f[i] f[lag-i]
            # d/d f[idx] = 2 * f[lag - idx] (when 0 <= lag-idx < n)
            lo = max(0, lag - n + 1)
            hi = min(lag, n - 1)
            # grad[lo:hi+1] += 2 * w * f[lag-hi : lag-lo][::-1]
            f_slice = f_ = y[lag - hi : lag - lo + 1][::-1]
            grad[lo:hi+1] += 2.0 * w * f_slice
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        normed_grad = grad / gnorm
        
        # BB step size
        if prev_grad is not None and prev_x is not None:
            sy = normed_grad - prev_grad
            xs = y - prev_x
            sy_norm = np.dot(sy, sy)
            if sy_norm > 1e-20:
                gamma_new = np.dot(xs, sy) / sy_norm
                if gamma_new > 1e-10:
                    gamma = 0.5 * gamma + 0.5 * gamma_new
        prev_grad = normed_grad.copy()
        prev_x = y.copy()
        
        # Decay
        x_new = project_simplex(y - gamma * normed_grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize to keep sum = S
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.15:
            sigma = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, sigma, n), S)
            t = 1.0
    
    return best_local, best_score

def coordinate_descent_fast(a_init, n, S, max_time, rng, n_top=50):

**Improved spectral shaping:**

The key insight is that the autocorrelation peak is caused by "constructive interference" in the spectrum. To reduce the peak at lag $m_0$, we want to "de-construct" the interference at that lag.

The autocorrelation at lag $m$ is:
$r[m] = \frac{1}{N}\sum_{k=0}^{N-1} P[k] e^{2\pi i mk/N}$

For $r[m_0]$ to be small, we want the complex numbers $P[k] e^{2\pi i m_0 k/N}$ to cancel each other. Since $P[k] \geq 0$, the "direction" of each term is determined by $2\pi m_0 k/N$. Terms in the "upper half" (where $\cos(2\pi m_0 k/N) > 0$) push $r[m_0]$ up, and terms in the "lower half" push it down.

To reduce $r[m_0]$, we should:
- Reduce $P[k]$ where $\cos(2\pi m_0 k/N) > 0$ (reduce the "up" push)
- Or increase $P[k]$ where $\cos(2\pi m_0 k/N) < 0$ (increase the "down" push)

Both are valid strategies. The previous code only does the first. Let me add the second as well.

Actually, increasing $P[k]$ where $\cos < 0$ is tricky because it might increase $r[m]$ at other lags. Let me stick with just reducing where $\cos > 0$, but use a more adaptive strategy.

**Adaptive strategy for spectral shaping:**

1. Compute $r = f*f$ (autocorrelation).
2. Find the top-K lags by value.
3. For each bin $k$, compute the "net contribution" to the top lags: $\text{net}[k] = \sum_j w_j \cos(2\pi m_j k/N)$.
4. Reduce $P[k]$ where $\text{net}[k] > 0$, by an amount proportional to $\text{net}[k]$.
5. Additionally, apply a "regularization" to keep the spectrum from becoming too peaked: $P_{\text{new}}[k] = \max(P_{\text{new}}[k], \epsilon \cdot \bar{P})$ where $\bar{P}$ is the mean power.

This regularization prevents the spectrum from concentrating all its power in a few bins, which would make the time-domain signal too "spiky" (and hence harder to make non-negative).

**Another key improvement: Use the "water-filling" analogy.**

Think of the autocorrelation values $r[m]$ as "water levels" in a container. We want to level them out. The "water" is the total energy $(\sum f)^2$, which is fixed. The "container shape" is determined by the spectrum.

Water-filling: pour water (reduce $P[k]$) from the "high" frequency bins (those that contribute to the peaks) into the "low" frequency bins (those that contribute to the valleys). This levels out the autocorrelation.

The constraint is that $P[k] \geq 0$ (can't have negative power) and that the resulting time-domain signal must be non-negative.

"""
Minimize 2*n*max_conv/sum^2 using:
1. Sequence length 16384 for lower theoretical bound (~1.00003)
2. Soft-peak (softmax-weighted top-K) objective for smoother optimization
3. Adaptive spectral shaping with water-filling and regularization
4. Nesterov-accelerated projected gradient with BB step size
5. Incremental coordinate descent with O(n) convolution updates
6. Simulated annealing for global exploration
7. Multiple structured initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_fft_padded(a, N):
    """Compute FFT of a padded to size N."""
    a_pad = np.zeros(N)
    n = len(a)
    a_pad[:n] = a
    return fft(a_pad)


def spectral_shape_adaptive(a, N, n, S, top_k, alpha, beta, min_power_frac):
    """
    Adaptive spectral shaping with regularization.
    Reduces power at frequencies that constructively interfere at peak lags.
    Applies a floor to prevent spectral collapse.
    """
    A = compute_fft_padded(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Compute autocorrelation from spectrum
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    c_mean = c.sum() / max(len(c), 1)

    if c_max <= c_mean * 1.005:
        return a, c_max

    # Find top-K peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]

    # Softmax weights
    if top_vals.max() > 1e-30:
        safe_beta = beta / top_vals.max()
        exps = np.exp(safe_beta * (top_vals - top_vals.max()))
        weights = exps / exps.sum()
    else:
        weights = np.ones(top_k) / top_k

    # Compute effective cosine (weighted sum over peak lags)
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i in range(top_k):
        lag = int(top_indices[i])
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * lag / N)

    # Adaptive alpha based on excess
    excess_ratio = (c_max - c_mean) / max(c_max, 1e-10)
    alpha_eff = alpha * (0.3 + excess_ratio)

    # Shape spectrum: reduce where cos_eff > 0
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    # Regularization: floor the power to prevent collapse
    mean_P = P.sum() / N
    floor = min_power_frac * mean_P
    P_new = np.maximum(P_new, floor)

    # Reconstruct with preserved phase
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def nesterov_soft_gradient(a_init, n, N, S, time_budget, rng,
                            top_k_init=10, top_k_max=300, beta=15.0):
    """
    Nesterov-accelerated projected gradient with softmax-weighted subgradient.
    Uses BB step size adaptation.
    """
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    momentum_t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Initial step size
    gamma = 4.0 * S / n
    prev_diff = None
    prev_y = None

    while time.monotonic() - start < time_budget:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = fft_convolve_self(y)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()

        # Compute softmax-weighted subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]

        if top_vals.max() > 1e-30:
            safe_beta = beta / top_vals.max()
            exps = np.exp(safe_beta * (top_vals - top_vals.max()))
            weights = exps / exps.sum()
        else:
            weights = np.ones(top_k) / top_k

        # Gradient: grad[i] = 2 * sum_j weights[j] * y[lag_j - i]
        grad = np.zeros(n)
        for j in range(top_k):
            lag = int(top_indices[j])
            w = weights[j]
            lo = max(0, lag - n + 1)
            hi = min(lag, n - 1)
            if lo > hi:
                continue
            # grad[lo:hi+1] += 2*w * y[lag-hi : lag-lo+1][::-1]
            f_slice = y[lag - hi: lag - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * w * f_slice

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        normed_grad = grad / gnorm

        # BB step size update
        if prev_diff is not None and prev_y is not None:
            d_grad = normed_grad - prev_diff
            d_y = y - prev_y
            denom = np.dot(d_grad, d_grad)
            if denom > 1e-20:
                gamma_new = np.dot(d_y, d_grad) / denom
                if gamma_new > 1e-10 and gamma_new < 10.0 * S / n:
                    gamma = 0.6 * gamma + 0.4 * gamma_new
        prev_diff = normed_grad.copy()
        prev_y = y.copy()

        # Projected gradient step
        x_new = project_simplex(y - gamma * normed_grad, S)

        # Nesterov momentum update
        momentum_t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t))
        y = x_new + ((momentum_t - 1.0) / momentum_t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        momentum_t = momentum_t_new

        # Periodic perturbation for escape
        if it % 200 == 0 and rng.random() < 0.2:
            sigma = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, sigma, n), S)
            momentum_t = 1.0

        # Decay step size slowly
        gamma = gamma / (1.0 + 0.0001 * it)

    return best_local, best_score


def spectral_waterfill_iterate(a_init, n, N, S, time_budget, rng,
                                top_k_init=5, top_k_max=100,
                                alpha_init=0.6, beta_init=10.0):

    """

    s = a.sum()

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 20)
        alpha = alpha_init / (1.0 + 0.02 * it)
        beta = beta_init + 0.1 * it

        # Spectral shaping step
        a_new, c_max = spectral_shape_adaptive(a, N, n, S, top_k, alpha, beta, 0.01)

        # Verify

        actual_cmax = c_check.max()

        if actual_cmax < best_score:

            a = a_new.copy()
            stall = 0
            if it % 200 == 0:
                print("    SW iter %d: %.6f" % (it, best_score))
        else:
            a = a_new.copy()
            stall += 1
            if stall > 50:
                # Perturb to escape
                sigma = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, sigma, n), S)
                stall = 0

        # Occasional gradient step between spectral steps

def coordinate_descent_incremental(a_init, n, S, time_budget, rng, n_top=40):
    """
    Coordinate descent with O(n) incremental convolution updates.
    Tries multiple multipliers per coordinate.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    multipliers = np.array([0.2, 0.4, 0.6, 0.8, 1.3, 1.6, 2.0, 0.05])

    c = fft_convolve_self(a)
    c_max = c.max()
    if c_max < best_score:
        best_score = c_max
        best_a = a.copy()

    round_count = 0
    while time.monotonic() - start < time_budget:
        round_count += 1
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find peak lag
        peak_lag = int(np.argmax(c))

        # Find top-contributing coordinates at peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)

        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for idx in top_coords:
            i = int(idx)
            if time.monotonic() - start > time_budget:
                break
            j = peak_lag - i
            if j < 0 or j >= n:
                continue
            orig_val = a[i]
            if orig_val < 1e-15:
                continue

            # Determine affected lag range
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            affected_len = k_end - k_start + 1

            # Precompute unaffected max
            if k_start > 0 and k_end < 2 * n - 2:
                unaffected_max = max(c[:k_start].max(), c[k_end + 1:].max())
            elif k_start > 0:
                unaffected_max = c[:k_start].max()
            elif k_end < 2 * n - 2:
                unaffected_max = c[k_end + 1:].max()
            else:
                unaffected_max = 0.0

            # Partner values: a[k-i] for k in [k_start, k_end]
            partner = a[k_start - i: k_end - i + 1]

            best_frac = 1.0
            best_cmax = c_max + 1e-30

            for frac in multipliers:
                new_val = orig_val * frac
                delta = new_val - orig_val
                if abs(delta) < 1e-15:
                    continue

                # Compute new values at affected lags
                c_affected = c[k_start:k_end + 1].copy()
                c_affected += 2.0 * delta * partner

                # Correction for k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    c_affected[pos] -= 2.0 * delta * partner[pos]
                    c_affected[pos] += delta * (2.0 * orig_val + delta)

                trial_max = max(c_affected.max(), unaffected_max)
                if trial_max < best_cmax:
                    best_cmax = trial_max
                    best_frac = frac

            # Apply best
            if best_frac != 1.0:
                new_val = orig_val * best_frac
                delta = new_val - orig_val
                c[k_start:k_end + 1] += 2.0 * delta * partner
                if 2 * i >= k_start and 2 * i <= k_end:
                    pos = 2 * i - k_start
                    c[2 * i] -= 2.0 * delta * partner[pos]
                    c[2 * i] += delta * (2.0 * orig_val + delta)
                a[i] = new_val
                improved = True

        # Rescale to maintain sum
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a * (S / a_sum)
            c = fft_convolve_self(a)  # Recompute after rescaling

        # Perturb if no improvement
        if not improved and rng.random() < 0.4:
            n_pert = min(20, n)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            for ip in idx_pert:
                a[int(ip)] *= rng.uniform(0.5, 1.8)
            a = project_simplex(a, S)
            c = fft_convolve_self(a)

    return best_a, best_score


def annealed_spectral_search(a_init, n, N, S, time_budget, rng):
    """
    Simulated annealing with spectral modifications.
    Accepts worse solutions with decreasing probability.
    """
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    current_score = fft_convolve_self(a).max()
    T = 0.05  # Initial temperature
    T_min = 0.001
    cool_rate = 0.999
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        top_k = min(50, 5 + it // 10)
        alpha = 0.3 / (1.0 + 0.01 * it)

        # Generate candidate via spectral shaping
        a_new, _ = spectral_shape_adaptive(a, N, n, S, top_k, alpha, 20.0, 0.01)

        # Add random spectral noise with amplitude proportional to T
        A = compute_fft_padded(a_new, N)
        noise_scale = T * np.sqrt(np.mean(np.abs(A) ** 2))
        noise = noise_scale * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
        A_new = A + noise
        a_cand = np.real(ifft(A_new))[:n]
        a_cand = project_simplex(a_cand, S)

        # Evaluate
        c_cand = fft_convolve_self(a_cand)
        cand_score = c_cand.max()

        # Metropolis acceptance
        if cand_score < current_score:
            accept = True
        else:
            delta = cand_score - current_score
            if delta < 1e-15:
                accept = True
            else:
                accept_prob = np.exp(-delta / max(T, 1e-10))
                accept = rng.random() < accept_prob

        if accept:
            a = a_cand
            current_score = cand_score

        if current_score < best_score:
            best_score = current_score
            best_local = a.copy()
            if it % 100 == 0:
                print("    Anneal iter %d: %.6f (T=%.4f)" % (it, best_score, T))

        T = max(T_min, T * cool_rate)

    return best_local, best_score


def structured_init(n, rng, method="windowed_chirp"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "windowed_chirp":
        # Hann window times chirp
        window = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        window = 0.1 + 0.9 * window  # Avoid exact zeros
        chirp = np.sin(2 * np.pi * (2.0 * t + 1.5 * t * t))
        a = window * (0.8 + 0.4 * chirp) + 0.05 * rng.standard_normal(n)
    elif method == "gaussian":
        center = 0.5
        width = 0.25
        a = np.exp(-0.5 * ((t - center) / width) ** 2) + 0.05 * rng.standard_normal(n)
    elif method == "multi_gaussian":
        a = 0.6 * np.exp(-0.5 * ((t - 0.3) / 0.15) ** 2) \
            + 0.8 * np.exp(-0.5 * ((t - 0.5) / 0.1) ** 2) \
            + 0.6 * np.exp(-0.5 * ((t - 0.7) / 0.15) ** 2) \
            + 0.05 * rng.standard_normal(n)
    elif method == "spiral":
        # Spiral frequency modulation
        freq = 3 + 5 * t
        a = 0.5 + 0.3 * np.sin(2 * np.pi * np.cumsum(freq) / n) + 0.15 * rng.standard_normal(n)
    elif method == "alternating_window":
        # Alternating pattern with window
        window = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        window = 0.1 + 0.9 * window
        base = np.where(np.arange(n) % 3 == 0, 1.0, 0.4)
        a = window * base + 0.1 * rng.standard_normal(n)
    elif method == "barker_repeated":
        # Repeated Barker-like pattern, windowed
        bark = np.array([1, 1, 1, 1, 1, -1, -1])
        pattern = np.tile(bark, n // len(bark) + 1)[:n]
        window = 0.2 + 0.8 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
        a = window * (0.5 + 0.5 * pattern) + 0.05 * rng.standard_normal(n)
    elif method == "random_piecewise":
        n_pieces = 128
        piece_len = n // n_pieces
        heights = 0.2 + 0.8 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, 0.2 + 0.8 * rng.random(n - len(a))])
        a = a[:n]
        # Apply window
        window = 0.2 + 0.8 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
        a = a * window
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)

    a = np.clip(a, 0.001, 3.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def resize_sequence(a, new_n):
    """Resize a sequence to new_n using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n < new_n:
        # Upsample
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    else:
        # Downsample
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses larger sequence length (16384), soft-peak optimization,
    adaptive spectral shaping, Nesterov gradient, coordinate descent,
    and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Main parameters
    n = 16384
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 32768

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = safe_normalize(a_candidate, n)
        c = fft_convolve_self(a_candidate)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            return True
        return False

    # --- Build initializations ---
    inits = []

    # Try to use previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size > 0 and np.sum(np.clip(given, 0, 1000)) > 0:
            resized = resize_sequence(np.clip(given, 0, 1000), n)
            inits.append(("prev_best", resized))
            print("  Loaded previous best (len=%d), resized to %d" % (given.size, n))
    except Exception:
        pass

    # Structured initializations
    for method in ["windowed_chirp", "gaussian", "multi_gaussian",
                   "spiral", "alternating_window", "barker_repeated", "random_piecewise"]:
        inits.append((method, structured_init(n, rng, method)))

    # Pure random
    inits.append(("rand", 0.2 + 0.8 * rng.random(n)))

    # Windowed constant
    t = np.arange(n) / n
    window = 0.1 + 0.9 * (0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(("flat_windowed", window + 0.02 * rng.standard_normal(n)))

    n_inits = len(inits)
    print("  %d initializations" % n_inits)

    # === Phase 1: Quick evaluation + spectral water-filling from multiple starts ===
    print("=== Phase 1: Multi-start spectral water-filling ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.35
    per_init_time = phase1_budget / n_inits

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        a_norm = safe_normalize(arr, n)

        # Quick score
        c0 = fft_convolve_self(a_norm)
        sc0 = c0.max()
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: %.6f" % (name, global_best))

        # Spectral water-filling
        time_budget = min(per_init_time, 40.0)
        if time_budget > 2:
            a_sw, sc_sw = spectral_waterfill_iterate(a_norm, n, N, S,
                                                      time.monotonic() + time_budget,
                                                      rng, top_k_init=5, top_k_max=50,
                                                      alpha_init=0.5, beta_init=10.0)
            if update_best(a_sw):
                print("  [%s] waterfill: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    if global_a is None:
        global_a = safe_normalize(structured_init(n, rng, "windowed_chirp"), n)
        global_best = fft_convolve_self(global_a).max()

    # === Phase 2: Focused optimization on best found ===
    print("=== Phase 2: Focused optimization ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.40

    a_curr = global_a.copy()
    stall_count = 0

    while time.monotonic() < t_phase2 + phase2_budget:
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem < 5:
            break

        # Spectral water-filling burst
        if rem > 10:
            tb = min(rem * 0.25, 25.0)
            a_sw, sc_sw = spectral_waterfill_iterate(a_curr, n, N, S,
                                                      time.monotonic() + tb,
                                                      rng, top_k_init=10, top_k_max=80,
                                                      alpha_init=0.3, beta_init=15.0)
            if update_best(a_sw):
                a_curr = a_sw.copy()
                stall_count = 0
                print("  [P2] waterfill: %.6f" % global_best)
            else:
                stall_count += 1

        # Nesterov gradient
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 5:
            tb = min(rem * 0.25, 20.0)
            a_ng, sc_ng = nesterov_soft_gradient(a_curr, n, N, S, tb, rng,
                                                  top_k_init=20, top_k_max=200, beta=15.0)
            if update_best(a_ng):
                a_curr = a_ng.copy()
                stall_count = 0
                print("  [P2] nesterov: %.6f" % global_best)
            else:
                stall_count += 1

        # Coordinate descent
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 5:
            tb = min(rem * 0.2, 20.0)
            a_cd, sc_cd = coordinate_descent_incremental(a_curr, n, S, tb, rng, n_top=30)
            if update_best(a_cd):
                a_curr = a_cd.copy()
                stall_count = 0
                print("  [P2] coord: %.6f" % global_best)
            else:
                stall_count += 1

        # Annealed spectral search when stuck
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 10 and stall_count >= 3:
            tb = min(rem * 0.2, 20.0)
            a_anneal, sc_anneal = annealed_spectral_search(a_curr, n, N, S, tb, rng)
            if update_best(a_anneal):
                a_curr = a_anneal.copy()
                stall_count = 0
                print("  [P2] anneal: %.6f" % global_best)
            else:
                stall_count += 1

        # Random perturbation + quick optimization
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 8 and stall_count >= 4:
            sigmas = [0.02, 0.05, 0.1, 0.2]
            for sigma in sigmas:
                if rem < 3:
                    break
                a_pert = project_simplex(
                    a_curr + rng.normal(0, sigma * S / np.sqrt(n), n), S)
                # Quick spectral shaping from perturbed state
                a_sw2, _ = spectral_waterfill_iterate(a_pert, n, N, S,
                                                       time.monotonic() + min(5.0, rem / 2),
                                                       rng, top_k_init=5, top_k_max=30,
                                                       alpha_init=0.4, beta_init=10.0)
                if update_best(a_sw2):
                    a_curr = a_sw2.copy()
                    stall_count = 0
                    print("  [P2] perturb(%.2f)+wf: %.6f" % (sigma, global_best))
                    break
            stall_count = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # === Phase 3: Final intensive polish ===
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if global_a is not None and remaining > 5:
        # Long Nesterov with high top-k
        seg1 = min(remaining * 0.25, 120.0)
        if seg1 > 3:
            a_p1, _ = nesterov_soft_gradient(global_a, n, N, S, seg1, rng,
                                              top_k_init=50, top_k_max=500, beta=25.0)
            if update_best(a_p1):
                print("  [P3] nesterov: %.6f" % global_best)

        # Coordinate descent (long)
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            seg2 = min(rem2 * 0.30, 100.0)
            a_p2, _ = coordinate_descent_incremental(global_a, n, S, seg2, rng, n_top=60)
            if update_best(a_p2):
                print("  [P3] coord: %.6f" % global_best)

        # Fine spectral water-filling
        rem3 = deadline - time.monotonic()
        if rem3 > 3:
            seg3 = min(rem3 * 0.25, 60.0)
            a_p3, _ = spectral_waterfill_iterate(global_a, n, N, S,
                                                  time.monotonic() + seg3,
                                                  rng, top_k_init=30, top_k_max=200,
                                                  alpha_init=0.1, beta_init=30.0)
            if update_best(a_p3):
                print("  [P3] fine wf: %.6f" % global_best)

        # Final coordinate descent
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_p4, _ = coordinate_descent_incremental(global_a, n, S, rem4 - 1, rng, n_top=80)
            if update_best(a_p4):
                print("  [P3] final coord: %.6f" % global_best)

        # One more Nesterov pass on the best
        rem5 = deadline - time.monotonic()
        if rem5 > 5:
            a_p5, _ = nesterov_soft_gradient(global_a, n, N, S, rem5 - 2, rng,
                                              top_k_init=100, top_k_max=800, beta=40.0)
            if update_best(a_p5):
                print("  [P3] final nesterov: %.6f" % global_best)

    # --- Verification and safety ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
