def fft_convolve_self(a):

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def l2_proj_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v

    else:
        rho = int(np.where(valid)[0].max() + 1)
    theta = (cumsum[rho - 1] - S) / rho

def grad_max_conv(a, c, top_k=8):
    """
    Gradient of max of convolution using top-k peaks.
    Returns gradient vector (length n).
    grad b[k] wrt a[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    So gradient = 2 * sum_{k in active} w_k * shift_k(a)
    where shift_k(a)[j] = a[k-j] if valid.
    """
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above median
    median_c = np.median(c)
    weights = np.maximum(c[idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    
    # Compute gradient: for each active lag k, grad_i = 2 * a[k-i] (if valid)
    # This is equivalent to: 2 * sum_k w_k * flip_shift(a, k)
    # We can compute this via FFT:
    # The "correlation" of a with delta at position k is just a shifted/reversed a.
    # grad = 2 * sum_k w_k * (a reversed and shifted by k)
    # 
    # Using FFT: if W = fft(weights_padded), A = fft(a), 
    # then the "cross-correlation" is ifft(W * conj(A)) ... 
    # Actually, let me think about this more carefully.
    
    # b[k] = sum_i a[i] * a[k-i]
    # db[k]/da[j] = 2 * a[k-j] (if 0 <= k-j < n)
    # 
    # So the full gradient of sum_k w_k * b[k] is:
    # g[j] = 2 * sum_k w_k * a[k-j] (if 0 <= k-j < n)
    # 
    # This is 2 * (w * a_reversed)[j] where * is convolution... 
    # Let me define r[i] = a[n-1-i] (reversed a)
    # Then a[k-j] (for 0 <= k-j < n) = r[n-1-(k-j)] = r[n-1-k+j]
    # So g[j] = 2 * sum_k w_k * r[n-1-k+j]
    # Let m = n-1-k+j, so k = n-1+j-m
    # g[j] = 2 * sum_m r[m] * w[n-1+j-m]
    # = 2 * (r * w_full)[j + (n-1)]  ... hmm, this is getting complicated.
    #
    # Simpler approach: use the fact that it's a convolution.
    # g[j] = 2 * sum_k w_k * a[k-j]
    # Let u[d] = sum_k w_k * delta(k - j - d) ... no.
    #
    # Actually: g[j] = 2 * sum_k w_k * a[k-j]
    # Define v[d] = a[d] (for 0 <= d < n, else 0)
    # Then g[j] = 2 * sum_k w_k * v[k-j]
    # This is 2 * (w * v_reversed)[j] where v_reversed[d] = v[-d] = v[-d]...
    #
    # Let me just use the direct FFT approach:
    # The operator "map a -> g where g[j] = sum_k w_k * a[k-j]" is a convolution.
    # Define h[m] = w[m] (for 0 <= m < 2n-1, else 0). Then:
    # g[j] = 2 * sum_m h[m] * a[m-j]
    # = 2 * (h * flip(a))[j] where flip(a)[d] = a[-d]... 
    #
    # OK simplest: g[j] = 2 * sum_k w_k * a[k-j]
    # If I define the "full" weight vector W of length 2n-1 with W[k] = w_k for active k,
    # then g = 2 * ifft(fft(W_extended) * conj(fft(a_extended))) ... 
    #
    # Actually the cleanest: g is the "cross-correlation" of W and a (with specific convention).
    # In numpy terms: g[j] = 2 * sum_k W[k] * a[k-j]
    # This is np.correlate(W, a, mode='full')[n-1-j] or something...
    
    # Let me just do it directly with FFT:
    # g[j] = 2 * sum_{k: k-j in [0,n-1]} w_k * a[k-j]
    # = 2 * sum_{d: d in [0,n-1]} w_{j+d} * a[d]  (substituting d = k-j)
    # = 2 * (w_shifted * a)[j] where w_shifted is w indexed starting from j
    
    # This is the convolution of w (as a full vector) with a, evaluated at specific positions.
    # Let's define: W_full[k] = w_k for k in active set, 0 otherwise (length 2n-1)
    # Then g[j] = 2 * sum_k W_full[k] * a[k-j] (with a[m]=0 for m outside [0,n-1])
    # = 2 * (W_full * a_flipped)[n-1-j] where a_flipped[d] = a[n-1-d]
    #
    # Hmm, let me just use the correlation formula:
    # np.correlate(W_full, a, 'full') gives sum_k W_full[k] * a[k + j - offset]
    # 
    # OK I'll just use the FFT approach properly:
    fsize = 1 << (2 * n).bit_length()
    W = np.zeros(fsize)
    W[idx] = weights  # idx are the active lag positions
    A = fft(a, fsize)
    # We want: g[j] = 2 * sum_k W[k] * a[k-j]
    # This is the cross-correlation corr(W, a)[j] = sum_k W[k] * a[k-j]
    # In FFT: corr = ifft(fft(W) * conj(fft(a)))
    # But we need to be careful about the convention.
    # 
    # Actually: ifft(fft(W) * conj(fft(a)))[m] = sum_k W[k] * conj(a[k-m]) 
    # For real a: = sum_k W[k] * a[k-m]
    # So ifft(fft(W) * conj(fft(a)))[j] = sum_k W[k] * a[k-j]
    # That's exactly what we want!
    
    g = 2.0 * np.real(ifft(fft(W) * np.conj(A)))[:n]
    return g

def grad_lse(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    fsize = 1 << (2 * n).bit_length()
    W = np.zeros(fsize)
    W[:len(w)] = w
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(fft(W) * np.conj(A)))[:n]
    return g

def fista_exact_subgrad(a_init, max_time, rng, top_k=16, use_acceleration=True):
    """FISTA with exact subgradient (top-k peaks) and backtracking."""
    a = normalize_sum(a_init, 1.0)
    n = len(a)

    best_obj = obj(a)
    best_a = a.copy()
    
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * np.linalg.norm(a) ** 2 * max(1, top_k))
    
    c_prev = None
    step_prev = None
    x_prev = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        # Compute gradient
        g = grad_max_conv(y, c, top_k=min(top_k, 2 * it // 10 + 4))
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Backtracking step size
        x = y.copy()
        step_bt = step
        for _ in range(20):
            x_candidate = l2_proj_simplex(y - step_bt * g, 1.0)
            c_check = fft_convolve_self(x_candidate)
            obj_check = 2.0 * n * c_check.max()
            # Check sufficient decrease (for non-smooth, use a weaker condition)
            if obj_check <= cur_obj + step_bt * gnorm * gnorm * 0.5:
                break
            step_bt *= 0.5
        
        step = step_bt  # Use for next iteration (BB-like)
        x = x_candidate
        
        # Nesterov update
        if use_acceleration:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x + ((t_mom - 1.0) / t_new) * (x - a)
            y = np.maximum(y, 0.0)
            # Renormalize
            y = normalize_sum(y, 1.0)
        else:
            y = x.copy()
        
        a = x
        if use_acceleration:
            t_mom = t_new
        
        # Periodic restart
        if it - last_improve > 2000:
            noise = rng.normal(0, 0.002 / np.sqrt(n), n)
            y = l2_proj_simplex(y + noise, 1.0)
            t_mom = 1.0
            last_improve = it
            step = max(step * 0.5, 1e-10)
    
    # Final check
    final_obj = obj(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    
    return best_a, best_obj

def peak_shave_refine(a_init, max_time, rng):
    """
    Fine-grained peak shaving: repeatedly identify the single worst peak
    and take a tiny step to reduce it.
    """
    a = normalize_sum(a_init, 1.0)
    n = len(a)
    start = time.monotonic()
    best = 2.0 * n * fft_convolve_self(a).max()
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = 2.0 * n * c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find the single worst peak
        k_peak = np.argmax(c)
        
        # Gradient of b[k_peak] only
        # g[j] = 2*a[k_peak - j] if 0 <= k_peak - j < n
        g = np.zeros(n)
        # a[k_peak - j] for j = 0..n-1, valid when 0 <= k_peak - j < n
        # i.e., k_peak - n + 1 <= j <= k_peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        if lo <= hi:
            # g[j] = 2 * a[k_peak - j] for j in [lo, hi]
            # This is 2 * a[k_peak - lo : k_peak - lo + (hi-lo+1)] reversed
            segment = a[k_peak - hi : k_peak - lo + 1]  # a[k_peak-hi], ..., a[k_peak-lo]
            g[lo:hi+1] = 2.0 * segment[::-1]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Tiny step
        step = 0.01 / gnorm
        a_new = l2_proj_simplex(a - step * g, 1.0)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if 2.0 * n * c_new.max() < cur:
            a = a_new
        else:
            # Try smaller step
            step *= 0.1
            a_new = l2_proj_simplex(a - step * g, 1.0)
            c_new = fft_convolve_self(a_new)
            if 2.0 * n * c_new.max() < cur:
                a = a_new
            else:
                # Add small noise to escape
                a = l2_proj_simplex(a + rng.normal(0, 0.0005/np.sqrt(n), n), 1.0)
    
    return best_a, best
