"""
Minimize 2*n*max_conv/sum^2 using KKT-aware multiplicative updates with
adaptive LSE smoothing, Adam-like per-coordinate step sizes, and periodic
structured perturbations. Key insight: at the optimum, the correlation of
a with each active peak should be constant; we exploit this for better
gradient directions. Combined with mirror descent (multiplicative updates)
in the log domain and aggressive peak-chasing phases.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(w, a):
    """
    Compute cross-correlation: result[i] = sum_k w[k] * a[k-i]
    where w has length 2n-1 and a has length n.
    Uses FFT: ifft(fft(w) * conj(fft(a)))
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 1 - 1 + 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(W * np.conj(A)))
    return result[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta):
    """
    Compute gradient of LSE-smoothed max: (1/beta) * LSE(beta * c)
    Gradient w.r.t. a is: 2 * sum_k p_k * (K_k a)
    where p_k = softmax(beta * c)
    This equals 2 * cross_corr(p, a)
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    p = e / tot
    # Gradient = 2 * cross_corr(p, a)
    # where cross_corr(p, a)[i] = sum_k p[k] * a[k-i]
    g = 2.0 * fft_cross_corr(p, a)
    return g


def kkt_gradient(a, c, topk=10):
    """
    KKT-aware gradient: at the optimum, sum_{k in active} a[k-i] should be
    constant for all i with a[i] > 0. Compute the "excess" correlation
    with active peaks as the gradient direction.
    """
    n = len(a)
    c_max = c.max()
    # Active set: lags where c[k] is close to max
    threshold = c_max - 0.05 * (c_max - c.min())
    active = c > threshold
    if not active.any():
        active[np.argmax(c)] = True
    p = active.astype(np.float64) / active.sum()
    # Gradient = 2 * cross_corr(p, a) - constant (centered)
    v = 2.0 * fft_cross_corr(p, a)
    # Center: subtract mean for valid support
    v_mean = v.mean()
    return v - v_mean


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    p = np.zeros(len(c))
    p[idx] = 1.0 / k
    return 2.0 * fft_cross_corr(p, a)


def mirror_update(a, grad, eta, S):
    """Multiplicative (mirror) descent step maintaining sum = S and a >= 0."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)


def adam_step(a, grad, m, v, t, lr, beta1, beta2, eps, S):
    """Adam update with simplex projection."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a_new, S)
    return a_new, m, v


def run_optimization(a_init, n, S, max_time, rng, method="kkt_adam"):
    """
    Main optimization loop.
    method: 'kkt_adam', 'lse_adam', 'mirror_kkt', 'hybrid'
    """
    a = np.clip(a_init.copy(), 1e-10, None)
    a = a / a.sum() * S

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    lr_base = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step_size = 0.1
    consecutive_fail = 0
    last_improve_it = 0

    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1

        # Choose gradient based on method and phase
        if method == "kkt_adam":
            if it % 500 < 100:
                grad = topk_subgradient(a, c, 1)
            else:
                grad = kkt_gradient(a, c, topk=20)
        elif method == "lse_adam":
            beta = min(200.0, 2.0 + 0.01 * it)
            grad = lse_gradient(a, c, beta)
        elif method == "mirror_kkt":
            if it % 300 < 50:
                grad = topk_subgradient(a, c, 5)
            else:
                grad = kkt_gradient(a, c, topk=16)
        elif method == "hybrid":
            phase = it % 1000
            if phase < 100:
                k = min(32, 2 + it // 500)
                grad = topk_subgradient(a, c, k)
            elif phase < 400:
                grad = kkt_gradient(a, c, topk=16)
            elif phase < 700:
                beta = min(150.0, 5.0 + 0.02 * it)
                grad = lse_gradient(a, c, beta)
            else:
                grad = kkt_gradient(a, c, topk=4)
        else:
            grad = kkt_gradient(a, c, topk=10)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Normalize gradient for stable updates
        grad_norm = grad / gnorm

        if method == "mirror_kkt":
            # Mirror descent with adaptive step
            eta = lr_base / (1.0 + 0.1 * np.log1p(it / 100))

            best_a = a_new.copy()
            last_improve_it = it
            consecutive_fail = 0
            # Increase learning rate slightly
            lr_base = min(lr_base * 1.03, 5.0)
        else:
            a = a_new  # Still accept (Adam can have non-monotonic progress)
            c = c_new
            consecutive_fail += 1
            lr_base = max(lr_base * 0.95, 0.05)

            if consecutive_fail > 100:
                # Perturbation to escape
                noise_scale = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                m = np.zeros(n)
                v = np.zeros(n)
                lr_base = 0.5
                consecutive_fail = 0

            if consecutive_fail > 300:
                # Larger perturbation
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                m = np.zeros(n)
                v = np.zeros(n)
                lr_base = 1.0
                consecutive_fail = 0

            if consecutive_fail > 800:
                # Structured perturbation

                a = project_simplex(a * np.maximum(mod, 0.1), S)
                c = fft_convolve_self(a)
                m = np.zeros(n)

                consecutive_fail = 0

        if it % 10000 == 0:

    return best_a, best_obj


def run_peak_chase(a_init, n, S, max_time, rng):
    """
    Aggressive peak-chasing: always focus on the single worst peak,
    use line search to find the best step.
    """
    a = np.clip(a_init.copy(), 1e-10, None)
    a = a / a.sum() * S

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S ** 2

        # Find the peak
        k_star = int(np.argmax(c))
        # Subgradient at the peak
        grad = topk_subgradient(a, c, 1)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try several step sizes (line search)
        best_step_obj = cur_obj
        best_a_try = a
        best_c_try = c

        for trial in range(8):
            step = S / (n * gnorm) * (0.01 * (1.5 ** trial))
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_a_try = a_try
                best_c_try = c_try

        a = best_a_try
        c = best_c_try

        if best_obj - 1e-12 > cur_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Periodic perturbation
        if it % 500 == 0:
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            c = fft_convolve_self(a)

        if it % 5000 == 0:
            print("    peak_chase it=%d, obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def run_mirror_descent(a_init, n, S, max_time, rng):
    """
    Pure mirror descent (multiplicative updates) with KKT-aware gradient
    and adaptive step size via backtracking.
    """
    a = np.clip(a_init.copy(), 1e-10, None)
    a = a / a.sum() * S

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    eta = 1.0
    consecutive_fail = 0

    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1

        # Use KKT gradient most of the time, top-k for diversity
        if it % 100 < 20:
            k = min(8, 1 + it // 200)
            grad = topk_subgradient(a, c, k)
        else:
            grad = kkt_gradient(a, c, topk=12)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent step
        eta_try = eta
        a_new = mirror_update(a, grad, eta_try, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S ** 2
        cur_obj = 2.0 * n * c.max() / S ** 2

        if obj_new < cur_obj:
            a = a_new
            c = c_new
            eta = min(eta * 1.1, S / (n * gnorm + 1e-15))
            consecutive_fail = 0
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
        else:
            eta *= 0.8
            consecutive_fail += 1
            if consecutive_fail > 50:
                # Reset with perturbation
                noise = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                eta = 1.0
                consecutive_fail = 0

        if it % 10000 == 0:
            print("    mirror it=%d, obj=%.6f, eta=%.4f" % (it, best_obj, eta))

    return best_a, best_obj


def generate_initializations(n, rng):

    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * tt) + 0.1 * np.sin(6 * np.pi * tt))

            it_f += 1
            c_f = fft_convolve_self(a_fine)
            cur_f = 2.0 * n_best * c_f.max() / S_best ** 2

            # Use KKT gradient
            grad_f = kkt_gradient(a_fine, c_f, topk=8)
            gnorm_f = np.linalg.norm(grad_f)
            if gnorm_f < 1e-15:
                break
            grad_f = grad_f / gnorm_f

            a_try, m_f, v_f = adam_step(a_fine, grad_f, m_f, v_f, it_f, lr_f, 0.9, 0.999, 1e-8, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best ** 2

            if obj_try < cur_f:
                a_fine = a_try
                lr_f = min(lr_f * 1.01, 0.1)
            else:
                lr_f = max(lr_f * 0.9, 0.001)
                if lr_f <= 0.001 and it_f % 100 == 0:
                    # Small random perturbation
                    noise = 0.001 * S_best / np.sqrt(n_best)
                    a_fine = project_simplex(a_fine + rng.normal(0, noise, n_best), S_best)
                    m_f = np.zeros(n_best)
                    v_f = np.zeros(n_best)
                    lr_f = 0.01

            if obj_try < best_fine - 1e-12:
                best_fine = obj_try

            if it_f % 20000 == 0:
                print("    fine it=%d, obj=%.6f" % (it_f, best_fine))

        a_fine = np.clip(a_fine, 0, 1000)
        s_fine = compute_objective(a_fine)
        if s_fine < global_best:
            global_best = s_fine
            global_a = a_fine.copy()
            print("  After fine-tuning: %.6f" % global_best)

    # Phase 5: Try a different length if time permits
    remaining = deadline - time.monotonic()
    print("=== Phase 5: Alternate length (remaining=%.0fs) ===" % remaining)

    if remaining > 120:
        # Try length 4096 or 16384
        alt_len = 4096 if n > 4096 else 16384
        S_alt = np.sqrt(2.0 * alt_len)
        alt_time = min(remaining * 0.7, 300)

        # Initialize by interpolating from best
        if global_a is not None:
            x_old = np.linspace(0, 1, len(global_a))
            x_new = np.linspace(0, 1, alt_len)
            alt_init = np.interp(x_new, x_old, global_a)
            alt_init = np.maximum(alt_init, 0.001)
        else:
            alt_init = generate_initializations(alt_len, rng)[0]

        try:
            a_alt, s_alt = run_optimization(alt_init, alt_len, S_alt, alt_time, rng, method="kkt_adam")
            a_alt = np.clip(a_alt, 0, 1000)
            s_alt = compute_objective(a_alt)
            if s_alt < global_best:
                global_best = s_alt
                global_a = a_alt.copy()
                print("  Alternate length %d: %.6f (NEW BEST!)" % (alt_len, global_best))
            else:
                print("  Alternate length %d: %.6f (not better)" % (alt_len, s_alt))
        except Exception as e:
            print("  Alternate length error: %s" % str(e))

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
    else:
        print("evaluate_sequence returned non-finite, using fallback")
        final_list = [float(x) for x in np.ones(256) * (np.sqrt(512.0) / 256)]

    print("Returning sequence of length %d, obj=%.6f" % (len(final_list), best_obj))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return final_list
