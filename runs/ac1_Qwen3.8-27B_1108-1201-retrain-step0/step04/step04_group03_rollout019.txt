Minimize 2*n*max(f*f)/sum(f)^2 using FISTA with LSE smoothing,
peak-shaving multiplicative updates, and LP polishing.
Explores multiple sequence lengths for better optimization landscape.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:

    return 2.0 * n * c.max() / (s * s)

def solve_lp_best_response(f, n, S, top_k, time_limit=5.0):
    c = fft_convolve_self(f)
    M = c.max()
    
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    g = None
    for solver in ['SCS', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - start > time_limit:
                break
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                if g.sum() > 1e-10:
                    g = g * (S / g.sum())
                    break
        except Exception:
            continue
    
    return g

def lp_polish(f, n, S, max_time, top_k=100):

    start = time.monotonic()
    best = np.inf
    best_f = f_current.copy()

    best = c.max()

        if g is None:
            break
        
        # Try different step sizes
        improved = False
        for t in [1.0, 0.5, 0.3, 0.2, 0.1]:
            f_test = (1 - t) * f_current + t * g
            f_test = np.maximum(f_test, 0)
            f_test = project_simplex(f_test, S)
            c_test = fft_convolve_self(f_test)
            if c_test.max() < best:
                best = c_test.max()
                best_f = f_test.copy()
                f_current = f_test.copy()
                improved = True
                break

            # Try pure FISTA step
            f_current = f_current + rng.normal(0, 0.001 * S / np.sqrt(n), n)
            f_current = project_simplex(f_current, S)
            if time.monotonic() - start < max_time * 0.9:

            else:
                break

"""
Minimize 2*n*max(f*f)/sum(f)^2 using aggressive FISTA with LSE smoothing,
adaptive peak-focused gradients, multiplicative repulsion, and LP polishing.
Explores multiple sequence lengths (8192, 16384, 32768) to exploit

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, k_indices, fsize):
    """Compute gradient contribution from specific top lags using FFT."""
    n = len(a)
    w = np.zeros(len(c))
    for k in k_indices:
        w[k] = 1.0
    w = w / len(k_indices)
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def multiplicative_repulsion_update(y, n, S, beta_rep, lr_rep, fsize_grad):
    """Apply multiplicative repulsion to reduce peaks."""
    c_rep = fft_convolve_self(y)
    cm_rep = c_rep.max()
    if cm_rep < 1e-300:
        return y
    w_rep = np.exp(beta_rep * (c_rep - cm_rep))
    tot_rep = w_rep.sum()
    if tot_rep < 1e-300:
        return y
    w_rep /= tot_rep
    W_r = fft(w_rep, fsize_grad)
    A_r = fft(y, fsize_grad)
    rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
    y_new = y * np.exp(-lr_rep * rep / (y.max() + 1e-10))
    y_new = np.maximum(y_new, 1e-15)
    return project_simplex(y_new, S)


def run_fista_aggressive(a_init, n, S, max_time, rng, beta_start=500, beta_max=5000,
                         repulsion_freq=250, repulsion_beta=40.0, repulsion_lr=0.03,
                         reset_freq=400, noise_freq=3000):
    """
    FISTA with LSE smoothing, aggressive beta, multiplicative repulsion,
    and adaptive stagnation handling.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fsize_grad = 1 << (size + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: ramp up over time
        progress = min(1.0, it / (max_time * 300.0))
        beta = beta_start * (beta_max / beta_start) ** progress
        beta = min(beta, beta_max)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Compute gradient
        if beta > 200 and (it % 50 == 0):
            # Use top-5 gradient for speed at high beta
            top5 = np.argsort(c)[::-1][:5]
            grad = topk_gradient(y, c, top5, fsize_grad)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (4.0 * beta)

        # Stagnation handling
        if it - last_improve > reset_freq:
            t_mom = 1.0
            # Try repulsion update
            y = multiplicative_repulsion_update(y, n, S, repulsion_beta,
                                                repulsion_lr * 2.0, fsize_grad)
            a = y.copy()
            last_improve = it
            # If still stuck, add noise
            if it - last_improve > reset_freq:
                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                last_improve = it
                t_mom = 1.0

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic multiplicative repulsion
        if it % repulsion_freq == 0 and it > 500:
            y = multiplicative_repulsion_update(y, n, S, repulsion_beta,
                                                repulsion_lr, fsize_grad)
            a = y.copy()
            t_mom = 1.0

        # Periodic noise (decreasing)
        if it % noise_freq == 0:
            decay = np.exp(-it / (noise_freq * 20.0))
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()

    return best_a, best


def solve_lp_best_response(f, n, S, top_k, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    g = None
    for solver in ['SCS', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - start > time_limit:
                break
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                if g.sum() > 1e-10:
                    g = g * (S / g.sum())
                    break
        except Exception:
            continue

    return g


def lp_polish(f, n, S, max_time, top_k=80):
    """Apply LP-based polishing iterations."""
    f_current = f.copy()
    start = time.monotonic()
    c = fft_convolve_self(f_current)
    best = c.max()
    best_f = f_current.copy()
    rng = np.random.default_rng(12345)
    stalled = 0

    while time.monotonic() - start < max_time:
        g = solve_lp_best_response(f_current, n, S, top_k=top_k, time_limit=3.0)
        if g is None:
            break

        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            f_test = (1 - t) * f_current + t * g
            f_test = np.maximum(f_test, 0)
            f_test = project_simplex(f_test, S)
            c_test = fft_convolve_self(f_test)
            if c_test.max() < best - 1e-10:
                best = c_test.max()
                best_f = f_test.copy()
                f_current = f_test.copy()
                improved = True
                stalled = 0
                break

        if not improved:
            stalled += 1
            if stalled > 5:
                break
            f_current = f_current + rng.normal(0, 0.002 * S / np.sqrt(n), n)
            f_current = project_simplex(f_current, S)

    return best_f, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def compute_objective_internal(a):
    """Compute the objective value internally."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_fista_with_peak_focus(a_init, n, S, max_time, rng):
    """
    FISTA variant that alternates between LSE gradient and direct peak reduction.
    Uses a 'peak focus' mode where we directly target the top lag.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fsize_grad = 1 << (size + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        progress = min(1.0, it / (max_time * 200.0))
        beta = 500.0 * (4000.0 / 500.0) ** progress
        beta = min(beta, 4000.0)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Alternate gradient strategies
        if it % 3 == 0:
            # Peak-focused: use top-3 lags
            top3 = np.argsort(c)[::-1][:3]
            grad = topk_gradient(y, c, top3, fsize_grad)
            step = 1.0 / (2.0 * len(top3))
        else:
            # Standard LSE
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (4.0 * beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Stagnation: aggressive restart
        if it - last_improve > 300:
            t_mom = 1.0
            # Strong repulsion
            y = multiplicative_repulsion_update(y, n, S, 60.0, 0.05, fsize_grad)
            a = y.copy()
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic repulsion
        if it % 200 == 0 and it > 400:
            y = multiplicative_repulsion_update(y, n, S, 30.0, 0.02, fsize_grad)
            a = y.copy()
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses FISTA with LSE, peak-focused gradients, multiplicative repulsion,
    and LP polishing across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Get starting sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    def try_update(a, label):
        nonlocal global_best, global_a
        if a is None or len(a) < 32:
            return
        n = len(a)
        S = np.sqrt(2.0 * n)
        a_norm = normalize_sum(a, n)
        c = fft_convolve_self(a_norm)
        obj = 2.0 * n * c.max() / (a_norm.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  NEW BEST [%s]: %.8f (n=%d)" % (label, global_best, n))

    print("=== Starting optimization ===")
    print("Given sequence length: %s" % ("None" if given is None else str(len(given))))
    print("Total budget: %.0f seconds" % budget_s)

    # === Phase 1: Optimize at given length ===
    if given is not None and len(given) >= 1024:
        n1 = len(given)
        S1 = np.sqrt(2.0 * n1)
        t1_budget = min(600.0, (deadline - time.monotonic()) * 0.4)
        print("Phase 1: FISTA at n=%d, budget=%.0fs" % (n1, t1_budget))

        a1, s1 = run_fista_aggressive(given, n1, S1, t1_budget, rng,
                                       beta_start=500, beta_max=4000,
                                       repulsion_freq=200, repulsion_beta=50.0,
                                       repulsion_lr=0.04, reset_freq=350)
        try_update(a1, "P1-FISTA n=%d" % n1)

        # Peak-focus variant on the result
        remaining = deadline - time.monotonic()
        if remaining > 100 and global_a is not None:
            n1b = len(global_a)
            S1b = np.sqrt(2.0 * n1b)
            t1b = min(200.0, remaining * 0.3)
            print("Phase 1b: Peak-focus at n=%d, budget=%.0fs" % (n1b, t1b))
            a1b, s1b = run_fista_with_peak_focus(global_a, n1b, S1b, t1b, rng)
            try_update(a1b, "P1b-Peak n=%d" % n1b)

    # If no starting point, create one
    if global_a is None:
        n_init = 4096
        t = np.linspace(0, 1, n_init, endpoint=False)
        given = np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * t), 0.01)
        S_init = np.sqrt(2.0 * n_init)
        t_init_budget = min(200.0, (deadline - time.monotonic()) * 0.2)
        a_init, _ = run_fista_aggressive(given, n_init, S_init, t_init_budget, rng)
        try_update(a_init, "Init n=%d" % n_init)

    # === Phase 2: Try n=16384 ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        init2 = interpolate_sequence(global_a, n2)
        init2 = normalize_sum(init2, n2)
        t2_budget = min(500.0, remaining * 0.35)
        print("Phase 2: FISTA at n=%d, budget=%.0fs (remaining %.0fs)" % (n2, t2_budget, remaining))

        a2, s2 = run_fista_aggressive(init2, n2, S2, t2_budget, rng,
                                       beta_start=400, beta_max=3000,
                                       repulsion_freq=300, repulsion_beta=40.0,
                                       repulsion_lr=0.03, reset_freq=500)
        try_update(a2, "P2-FISTA n=%d" % n2)

    # === Phase 3: Try n=32768 if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 250 and global_a is not None:
        n3 = 32768
        S3 = np.sqrt(2.0 * n3)
        init3 = interpolate_sequence(global_a, n3)
        init3 = normalize_sum(init3, n3)
        t3_budget = min(400.0, remaining * 0.4)
        print("Phase 3: FISTA at n=%d, budget=%.0fs (remaining %.0fs)" % (n3, t3_budget, remaining))

        a3, s3 = run_fista_aggressive(init3, n3, S3, t3_budget, rng,
                                       beta_start=300, beta_max=2500,
                                       repulsion_freq=400, repulsion_beta=35.0,
                                       repulsion_lr=0.025, reset_freq=600)
        try_update(a3, "P3-FISTA n=%d" % n3)

    # === Phase 4: LP polish on the best ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        t4_budget = min(120.0, remaining * 0.35)
        print("Phase 4: LP polish at n=%d, budget=%.0fs" % (n_p, t4_budget))
        try:
            a4, s4 = lp_polish(global_a, n_p, S_p, t4_budget, top_k=60)
            try_update(a4, "P4-LP n=%d" % n_p)
        except Exception as e:
            print("  LP failed: %s" % str(e))

    # === Phase 5: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        t5_budget = min(80.0, remaining * 0.5)
        print("Phase 5: Final polish at n=%d, budget=%.0fs" % (n5, t5_budget))

        # High-beta FISTA for final refinement
        a5, s5 = run_fista_aggressive(global_a, n5, S5, t5_budget, rng,
                                       beta_start=1000, beta_max=8000,
                                       repulsion_freq=150, repulsion_beta=80.0,
                                       repulsion_lr=0.02, reset_freq=250)
        try_update(a5, "P5-final n=%d" % n5)

    # === Phase 6: Try a completely different initialization ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and global_a is not None:
        n6 = len(global_a)
        S6 = np.sqrt(2.0 * n6)
        t6_budget = min(80.0, remaining * 0.4)

        # Generate a structured random initialization
        t = np.linspace(0, 1, n6, endpoint=False)
        # Multi-frequency structure
        init6 = (1.0
                 + 0.3 * np.sin(2 * np.pi * 3 * t + rng.uniform(0, 2 * np.pi))
                 + 0.2 * np.cos(2 * np.pi * 7 * t + rng.uniform(0, 2 * np.pi))
                 + 0.1 * np.sin(2 * np.pi * 13 * t + rng.uniform(0, 2 * np.pi))
                 + 0.05 * rng.normal(0, 1, n6))
        init6 = np.maximum(init6, 0.01)
        init6 = normalize_sum(init6, n6)

        print("Phase 6: Different init at n=%d, budget=%.0fs" % (n6, t6_budget))
        a6, s6 = run_fista_aggressive(init6, n6, S6, t6_budget, rng,
                                       beta_start=300, beta_max=3000)
        try_update(a6, "P6-alt-init n=%d" % n6)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final evaluate_sequence: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned non-finite!")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid sequence found, using fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("=== Done ===")
    print("Returning sequence of length %d, best=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
