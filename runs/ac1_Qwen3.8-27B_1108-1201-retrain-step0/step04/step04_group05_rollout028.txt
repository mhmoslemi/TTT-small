"""

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    # Clip tiny negative values from FFT roundoff
    c = np.maximum(c, 0.0)
    return c


def compute_grad_weighted(a, w, N):
    """
    Compute 2 * ifft(fft(w_pad) * conj(fft(a_pad)))[:n]
    where w_pad = [w, 0, ..., 0] (length N).
    This gives the gradient of sum(w[i] * c[i]) where c = a*a.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    Af = fft(a_pad)
    
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    Wf = fft(w_pad)
    
    grad_complex = ifft(Wf * np.conj(Af))
    return 2.0 * np.real(grad_complex)[:n]


def lse_objective_and_grad(a, N, beta):
    """Compute LSE-smoothed max of c = a*a, and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
        return c_max, compute_grad_weighted(a, w, N), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    grad = compute_grad_weighted(a, w, N)
    return lse_val, grad, c, c_max


def toppk_objective_and_grad(a, N, k):
    """Compute top-k average of c = a*a, and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    top_k = min(k, len(c))
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    obj = top_vals.mean() if top_k > 0 else 0.0
    w = np.zeros(2*n-1)
    w[top_idx] = 1.0 / top_k
    grad = compute_grad_weighted(a, w, N)
    return obj, grad, c, c.max()


def lbfgs_two_loop(s_hist, y_hist, g, m=20):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = min(m, len(s_hist))
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    # Ensure descent
    gd = np.dot(g, direction)
    if gd > 0:
        direction = -g
    return direction


def wolfe_line_search(a, direction, f_val, grad, N, S, c1=1e-4, c2=0.9, max_iter=30):
    """
    Wolfe line search for step size.
    Returns (alpha, a_new, f_new, c_new, cmax_new).
    """
    n = len(a)
    gtd = np.dot(grad, direction)
    if gtd >= 0:
        # Not a descent direction, use steepest descent
        direction = -grad
        gtd = -np.dot(grad, grad)
    
    # Initial step size
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-18:
        return 0.0, a, f_val, None, None
    alpha = np.linalg.norm(a - project_simplex(np.zeros(n), S)) / (gnorm * np.linalg.norm(direction) + 1e-30)
    alpha = min(alpha, 1.0)
    alpha = max(alpha, 1e-10)
    
    for ls in range(max_iter):
        a_new = project_simplex(a + alpha * direction, S)
        c_new = compute_conv(a_new, N)
        cmax_new = c_new.max()
        
        # Compute objective at new point (use cmax as proxy for during line search)
        f_new = cmax_new  # We're really minimizing cmax
        
        # Check Armijo condition (sufficient decrease)
        # f_new <= f_val + c1 * alpha * gtd
        # Since we use cmax as proxy, this is approximate
        if f_new <= f_val + c1 * alpha * gtd / (gnorm + 1e-30):
            break
        alpha *= 0.5
    
    return alpha, a_new, f_new, c_new, cmax_new


def lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0, 
                m_lbfgs=15, use_toppk=False, top_k=10):
    """
    L-BFGS on LSE-smoothed (or top-k) objective.
    Returns (best_a, best_cmax).
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    if use_toppk:
        f_val, g, c, c_max = toppk_objective_and_grad(a, N, top_k)
    else:
        f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
    
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive beta ramping
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.1)
            if not use_toppk:
                f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        # Trim history
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        # Get direction
        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            # Reset with steepest descent
            direction = -g
            dir_norm = np.linalg.norm(direction)
            if dir_norm < 1e-18:
                break
            s_hist = []
            y_hist = []
        
        # Line search (simple backtracking for speed)
        alpha = S / (n * dir_norm + 1e-30) * 2.0
        alpha = min(alpha, 10.0)
        
        best_alpha = 0.0
        a_new = a
        c_new = c
        cmax_new = c_max
        
        for ls in range(20):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            
            if cm_ls <= c_max * (1.0 - 1e-10) or ls == 19:
                a_new = candidate
                c_new = c_ls
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    best_alpha = alpha
                    break
            alpha *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        # Compute new gradient
        if use_toppk:
            f_new, g_new, c_nw, cmax_nw = toppk_objective_and_grad(a_new, N, top_k)
        else:
            f_new, g_new, c_nw, cmax_nw = lse_objective_and_grad(a_new, N, beta)
        
        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c = c_nw
        c_max = cmax_nw
        f_val = f_new
        
        # Periodic perturbation to escape local minima
        if it % 300 == 0 and rng.random() < 0.3:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 + best_cmax / 2.0)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            if use_toppk:
                f_val, g, c, c_max = toppk_objective_and_grad(a, N, top_k)
            else:
                f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
    
    return best_a, best_cmax


def active_set_equalize(a, n, S, N, k=10, n_steps=50):
    """
    Active set equalization: find top-k peaks and try to equalize them.
    Returns improved (a, cmax).
    """
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        s = S
    a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = np.inf
    
    for step in range(n_steps):
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Find top-k peaks
        top_k = min(k, 2*n-1)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Compute gradient at each peak
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0 / top_k
        grad = compute_grad_weighted(a, w, N)
        
        # Descent direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step size
        alpha = 2.0 * S / (n * gnorm)
        alpha = min(alpha, 1.0)
        
        a_new = project_simplex(a - alpha * grad, S)
        
        # Check improvement
        c_new = compute_conv(a_new, N)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Try smaller step
            alpha *= 0.5
            a_new = project_simplex(a - alpha * grad, S)
            c_new = compute_conv(a_new, N)
            if c_new.max() < c_max:
                a = a_new
            else:
                break
    
    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multisine(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    n_freqs = 5
    for i in range(n_freqs):
        freq = rng.uniform(2, 20)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Random phase in frequency domain."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-8)
    phase = rng.uniform(0, 2 * np.pi, N)
    # Ensure Hermitian symmetry for real signal
    phase[0] = 0
    if N % 2 == 0:
        phase[N//2] = 0
    A = np.sqrt(P) * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    A[:N//2+1] = A[:N//2+1]
    A[N//2+1:] = np.conj(A[N//2:0:-1])
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_gaussian(n, rng):
    """Gaussian-shaped initialization."""
    center = n // 2
    sigma = rng.uniform(n//8, n//3)
    tt = np.arange(n)
    a = np.exp(-0.5 * ((tt - center) / sigma) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_binary_smoothed(n, rng, window=8):
    """Binary sequence smoothed."""
    a = (rng.random(n) > 0.5).astype(np.float64)
    # Smooth with a window
    kernel = np.ones(window) / window
    a = np.convolve(a, kernel, mode='same')
    return np.clip(a, 0.01, None)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    S_curr = np.sqrt(2.0 * float(n_curr))
    a_curr = safe_normalize(a_curr, n_curr, S_curr)
    N_curr = 1 << (2 * n_curr - 1).bit_length()
    
    stall = 0
    last_best = global_best
    block_count = 0
    
    while time.monotonic() < t2 + phase2_budget:
        block_count += 1
        block_time = min(150.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        
        # Increase beta with stall
        beta_start = 10.0 + 5.0 * min(stall, 20)
        beta_end = min(1000.0, 100.0 * (1 + 0.3 * min(stall, 20)))
        
        # Alternate between LSE and top-k
        use_toppk = (block_count % 3 == 1)
        top_k = 5 + block_count * 2
        
        a_new, sc_new = lbfgs_block(a_curr, n_curr, S_curr, N_curr, block_time, rng,
                                      beta_init=beta_start, beta_max=beta_end,
                                      m_lbfgs=12, use_toppk=use_toppk, top_k=top_k)
        
        S_new = a_new.sum()
        obj_new = 2.0 * n_curr * sc_new / (S_new * S_new)
        
        if obj_new < global_best - 1e-8:
            global_best = obj_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2-%d] obj=%.6f (beta %.0f-%.0f, %s)" % (
                block_count, global_best, beta_start, beta_end, 
                "topk" if use_toppk else "LSE"))
        else:
            a_curr = safe_normalize(a_new, n_curr, S_curr)
            stall += 1
            if stall >= 2:
                # Perturb and reset
                noise = 0.01 * S_curr / np.sqrt(n_curr)
                a_curr = project_simplex(a_curr + rng.normal(0, noise, n_curr), S_curr)
                stall = 0
                
                # Also try active set equalization on perturbed
                a_eq, sc_eq = active_set_equalize(a_curr, n_curr, S_curr, N_curr, k=15, n_steps=200)
                S_eq = a_eq.sum()
                obj_eq = 2.0 * n_curr * sc_eq / (S_eq * S_eq)
                if obj_eq < global_best:
                    global_best = obj_eq
                    global_a = a_eq.copy()
                    a_curr = a_eq.copy()
                    print("  [P2-%d/eq] obj=%.6f" % (block_count, global_best))
    
    # === Phase 3: Active set + high-beta polish (400s) ===
    print("=== Phase 3: Active set + high-beta polish ===")

    remaining = deadline - t3
    
    if remaining > 30:
        # Active set equalization
