"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-method optimization combining:
1. FISTA with LSE smoothing (main workhorse)
2. Exact peak-chasing with backtracking (for final polishing)
3. Multiplicative weights update (for maintaining non-negativity naturally)
4. Multi-length exploration with warm starts
5. Adaptive restart with noise injection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure sum is exactly S (fix floating point)
    diff = S - result.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference to non-zero elements
        mask = result > 0
        result[mask] += diff / mask.sum()
        result = np.maximum(result, 0.0)
    return result


def lse_smooth_max(c, beta):
    """Log-sum-exp smooth approximation of max(c)."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    return cm + np.log(np.exp(scaled).sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_topk_gradient(a, c, k=8):
    """Exact subgradient from top-k lags, weighted uniformly."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # For each lag, compute the gradient contribution
    # (a*a)[lag] gradient wrt a[m] = a[m-lag] + a[m+lag] (with boundary)
    grad = np.zeros(n)
    for lag in idx:
        lag = int(lag)
        if lag == 0:
            grad += 2.0 * a
        else:
            # For lag > 0: (a*a)[lag] = sum_{j=lag}^{n-1} a[j]*a[j-lag]
            # grad[m] = a[m-lag] (if m >= lag) + a[m+lag] (if m+lag < n)
            if lag < n:
                grad[lag:] += a[:n-lag]
                grad[:n-lag] += a[lag:]
    grad /= k
    return grad


def multiplicative_update(a, grad, eta):
    """Multiplicative (mirror) descent step."""
    n = len(a)
    # a_j <- a_j * exp(-eta * grad_j)
    # Clip to avoid overflow
    exponent = -eta * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new * (a.sum() / s)


def run_fista_lse(a_init, S, max_time, rng, beta_init=50.0, beta_max=50000.0, 
                   beta_growth=1.002, name=""):
    """FISTA with LSE smoothing and careful step size."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    # Initial evaluation
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        beta = min(beta_max, beta_init * (beta_growth ** it))
        
        # Compute current LSE objective
        c = fft_convolve_self(y)
        obj = c.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        
        # Compute gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use 1/L where L is Lipschitz constant
        # For LSE with parameter beta, the Hessian of the LSE is bounded by
        # beta * (Jacobian of conv)^T (Jacobian of conv)
        # The Jacobian of a -> a*a (as a linear operator at point a) has norm <= 2*||a||_1
        # So L <= 2 * beta * (2*||a||_1)^2 = 8 * beta * ||a||_1^2
        # But this is a very loose bound. Let's use empirical backtracking instead.
        
        step = 1.0 / (8.0 * beta * S * S)  # Conservative initial step
        # Actually, let's be less conservative:
        # The gradient norm is at most 2 * ||a||_2 * (sum w_k)^0.5 = 2 * ||a||_2
        # And the "curvature" is roughly beta * 2 * ||a||_1
        # So step ~ 1 / (2 * beta * 2 * ||a||_1) = 1 / (4 * beta * S)
        step = 1.0 / (4.0 * beta * S)
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        # Restart if stuck
        if it - last_improve > 500:
            t_k = 1.0
            # Add small noise
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it
        
        # Periodic check
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_obj:
                best_obj = c_check.max()
                best_a = a.copy()
                last_improve = it
    
    return best_a, best_obj


def run_peak_chase(a_init, S, max_time, rng, k=4, name=""):
    """Peak chasing with exact subgradients and backtracking."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Get exact subgradient from top-k lags
        grad = exact_topk_gradient(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking step size
        step = 0.1 * S / (n + 1)
        max_backtracks = 10
        improved = False
        for _ in range(max_backtracks):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_obj * (1 - 1e-10):
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try a different approach: use LSE gradient
            beta = 100.0
            grad_lse = lse_gradient(a, c, beta)
            step = 1.0 / (4.0 * beta * S)
            a_new = project_simplex(a - step * grad_lse, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_obj:
                a = a_new
            else:
                # Stuck, add noise
                if it - last_improve > 200:
                    noise_scale = 0.002 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    last_improve = it
                else:
                    break
        else:
            a = a_new
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best_obj:
        best_obj = c_final.max()
        best_a = a.copy()
    
    return best_a, best_obj


def run_multiplicative(a_init, S, max_time, rng, name=""):
    """Multiplicative weights / mirror descent with entropy regularization."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    n = len(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    a = a * (S / s)
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0
    eta = 1.0 / (4.0 * S)  # Initial step size
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Decay eta
        eta = eta * (1.0 - 0.0001 * min(it, 10000))
        eta = max(eta, 1e-6 / S)
        
        # Use LSE gradient for smoothness
        beta = 50.0 + 5.0 * min(it, 1000)
        grad = lse_gradient(a, c, beta)
        
        # Multiplicative update
        a_new = multiplicative_update(a, grad, eta)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_obj * 1.01:
            # Reduce step
            eta *= 0.5
            a_new = multiplicative_update(a, grad, eta)
        
        a = a_new
        
        # Restart
        if it - last_improve > 500:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = np.maximum(a + rng.normal(0, noise_scale, n), 1e-10)
            a = a * (S / a.sum())
            last_improve = it
    
    return best_a, best_obj


def run_hybrid(a_init, S, max_time, rng, name=""):
    """Hybrid: alternate between LSE-FISTA and peak chasing."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    phase = 0  # 0: FISTA, 1: peak chase
    phase_time = max_time / 20.0  # Alternate every ~20 phases
    
    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break
        
        if phase == 0:
            # LSE FISTA phase
            alloc = min(phase_time, remaining)
            a_opt, obj = run_fista_lse(a, S, alloc, rng, 
                                         beta_init=50.0 * (1 + phase * 0.1),
                                         beta_max=50000.0,
                                         beta_growth=1.003)
        else:
            # Peak chase phase
            alloc = min(phase_time, remaining)
            k = max(2, min(16, 2 + phase))
            a_opt, obj = run_peak_chase(a, S, alloc, rng, k=k)
        
        c_new = fft_convolve_self(a_opt)
        if c_new.max() < best_obj:
            best_obj = c_new.max()
            best_a = a_opt.copy()
        
        a = best_a.copy()
        phase = 1 - phase
        phase_time *= 0.95  # Slightly decrease phase duration
    
    return best_a, best_obj


def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())
    
    # Various deterministic inits
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt))
    inits.append(rng.random(n) + 0.1)
    
    # Block structure
    for nb in [4, 8]:
        blen = n // nb
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, blen)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n])
    
    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def upscale(a_small, n_large):
    """Upscale by interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_single_lag(a, k):
    """Compute (a*a)[k] for a single lag in O(n)."""
    n = len(a)
    if k < 0:
        k = -k
    if k >= n:
        return 0.0
    return np.dot(a[k:], a[:n - k])


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix floating point: ensure sum is exactly S
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0.0)
    return result


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_topk_gradient_vec(a, c, k=8):
    """Exact subgradient from top-k lags using vectorized computation."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for lag in idx:
        lag = int(lag)
        if lag == 0:
            grad += 2.0 * a
        elif lag < n:
            grad[lag:] += a[:n - lag]
            grad[:n - lag] += a[lag:]
    grad /= k
    return grad


def normalize_seq(a, S):
    """Clip and normalize to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_fista_backtrack(a_init, S, max_time, rng, beta_init=30.0,
                        beta_growth=1.003, beta_max=30000.0):
    """FISTA with LSE smoothing and backtracking line search."""
    a = normalize_seq(a_init, S)
    n = len(a)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0
    step = 1.0 / (6.0 * beta_init * S)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (beta_growth ** it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_obj - 1e-10:
            best_obj = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking: try step, halve if no improvement
        step_try = step
        improved = False
        for _ in range(8):
            x_new = project_simplex(y - step_try * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max * (1.0 + 1e-12):
                improved = True
                break
            step_try *= 0.5
        
        if not improved:
            # Use smaller fixed step
            x_new = project_simplex(y - 0.01 * grad / max(gnorm, 1e-15), S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max:
                improved = True
            else:
                # Stuck
                if it - last_improve > 300:
                    noise = 0.002 * S / np.sqrt(n)
                    y = project_simplex(a + rng.normal(0, noise, n), S)
                    t_k = 1.0
                    last_improve = it
                step = step_try
                # Small perturbation
                x_new = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
        
        if improved:
            step = step_try  # Remember successful step
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_k = t_new
        else:
            a = x_new
        
        # Adaptive restart
        if it - last_improve > 800:
            t_k = 1.0
            y = a.copy()
            
        # Grow step slowly (since beta grows, step should shrink)
        step = 1.0 / (6.0 * beta * S)
    
    # Final
    c_f = fft_convolve_self(a)
    if c_f.max() < best_obj:
        best_obj = c_f.max()
        best_a = a.copy()
    return best_a, best_obj


def run_peak_chase(a_init, S, max_time, rng, k=6):
    """Peak chasing with exact top-k subgradient and backtracking."""
    a = normalize_seq(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0
    step_base = 0.05 * S / n
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj - 1e-10:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it
        
        # Get exact subgradient from top-k lags
        grad = exact_topk_gradient_vec(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = step_base * (1.0 + 0.5 / (1.0 + 0.01 * it))
        improved = False
        for _ in range(12):
            a_new = project_simplex(a - step * grad, S)
            # Check only the top lags (cheaper than full FFT)
            top_lags = np.argsort(c)[::-1][:min(20, 2 * k)]
            max_new = 0.0
            for lag in top_lags:
                val = compute_single_lag(a_new, int(lag))
                if val > max_new:
                    max_new = val
            # Also check lag 0
            v0 = compute_single_lag(a_new, 0)
            max_new = max(max_new, v0)
            
            if max_new < cur_max * (1.0 - 1e-9):
                improved = True
                break
            step *= 0.5
        
        if improved:
            a = a_new
            step_base = max(step * 2.0, 0.001 * S / n)
        else:
            # Try LSE with small beta as fallback
            beta = 50.0
            grad_lse = lse_gradient(a, c, beta)
            step_lse = 1.0 / (8.0 * beta * S)
            a_new = project_simplex(a - step_lse * grad_lse, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
            elif it - last_improve > 150:
                # Add noise to escape
                noise_scale = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                last_improve = it
                step_base = 0.05 * S / n
            else:
                break
    
    c_f = fft_convolve_self(a)
    if c_f.max() < best_obj:
        best_obj = c_f.max()
        best_a = a.copy()
    return best_a, best_obj


def run_mirror_descent(a_init, S, max_time, rng):
    """Mirror descent with entropy (multiplicative weights)."""
    a = normalize_seq(a_init, S)
    n = len(a)
    a = np.maximum(a, 1e-15)

    best_a = a.copy()

        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj - 1e-10:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it
        
        # Decrease eta over time
        eta = max(0.1 / (S * S * (1 + 0.001 * it)), 1e-8 / S)
        
        # Use LSE gradient
        beta = 30.0 + 1.0 * min(it, 2000)
        grad = lse_gradient(a, c, beta)

        exponent = -eta * grad * S  # Scale for numerical stability
        exponent = np.clip(exponent, -30, 30)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 1e-30)

        c_new = fft_convolve_self(a_new)

        best_obj = c_f.max()
        best_a = a.copy()
    return best_a, best_obj


def run_kkt_polish(a_init, S, max_time, rng):
    """KKT-based polishing: iteratively adjust to satisfy KKT conditions."""
    a = normalize_seq(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    it = 0
    
    # Determine support (elements that are "active" / positive)
    threshold = 1e-6 * S / n
    support = a > threshold
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj - 1e-10:
            best_obj = cur_max
            best_a = a.copy()
        
        # Find active lags (within 1% of max)
        active_tol = 0.01 * cur_max
        active_lags = np.where(c >= cur_max - active_tol)[0]
        active_lags = active_lags[active_lags < n]  # Only positive lags matter (by symmetry)
        k_active = len(active_lags)
        
        if k_active == 0:
            active_lags = np.array([0])
            k_active = 1
        
        # Compute weighted gradient from active lags
        # Weights: uniform over active lags
        weights = np.ones(k_active) / k_active
        
        grad = np.zeros(n)
        for i, lag in enumerate(active_lags):
            lag = int(lag)
            if lag == 0:
                grad += weights[i] * 2.0 * a
            elif lag < n:
                grad[lag:] += weights[i] * a[:n - lag]
                grad[:n - lag] += weights[i] * a[lag:]
        
        # KKT: on the support, grad should be constant (= mu)
        # Compute mu as the mean of grad on support
        if support.sum() > 10:
            mu = grad[support].mean()
            # Correction: push grad[support] toward mu
            # The direction to reduce KKT violation is:
            # delta_j = grad[j] - mu for j in support
            # We want to move a in the direction that makes grad more uniform
            # Since grad depends linearly on a (approximately), we use:
            # a_new = project(a - eta * (grad[support] - mu) projected on support)
            correction = np.zeros(n)
            correction[support] = grad[support] - mu
            c_norm = np.linalg.norm(correction)
            if c_norm > 1e-15:
                eta_kkt = 0.1 * S / (c_norm * np.sqrt(n))
                a_new = project_simplex(a - eta_kkt * correction, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max * 1.001:
                    a = a_new
                else:
                    a = project_simplex(a - 0.1 * eta_kkt * correction, S)
        
        # Update support periodically
        if it % 50 == 0:
            threshold = 1e-7 * S / n
            support = a > threshold
        
        # Add tiny noise to avoid stagnation
        if it % 100 == 0:
            noise = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
    
    c_f = fft_convolve_self(a)
    if c_f.max() < best_obj:
        best_obj = c_f.max()
        best_a = a.copy()
    return best_a, best_obj

def objective_value(a):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def upscale_seq(a_small, n_large):
    """Upscale by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)

        total_phase1_time = max(60.0, (deadline - time.monotonic()) * 0.4)
        time_per_init = total_phase1_time / n_inits
        
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            
            t_alloc = min(time_per_init, deadline - time.monotonic() - 60)
            if t_alloc < 5:
                continue
            
            n_i = len(init)
            S_i = np.sqrt(2.0 * n_i)
            
            # Choose method
            if i == 0 and given is not None:
                # Best init gets the best method (alternating)
                a_opt, obj = run_alternating(init, S_i, t_alloc, rng)
                method_name = "alternating"
            elif i % 3 == 1:
                a_opt, obj = run_fista_backtrack(init, S_i, t_alloc, rng,
                                                 beta_init=20.0, beta_growth=1.005,
                                                 beta_max=15000.0)
                method_name = "fista"
            elif i % 3 == 2:
                a_opt, obj = run_peak_chase(init, S_i, t_alloc, rng, k=6)
                method_name = "peak"
            else:
                a_opt, obj = run_mirror_descent(init, S_i, t_alloc, rng)
                method_name = "mirror"
            
            obj_check = objective_value(a_opt)
            max_b = fft_convolve_self(a_opt).max()
            print("  [%d/%d] %s: obj=%.6f, max_b=%.6f (t=%.0fs)" %
                  (i + 1, n_inits, method_name, obj_check, max_b, t_alloc))
            
            if obj_check < best_obj_n:
                best_obj_n = obj_check
                best_a_n = a_opt.copy()
        
        if best_a_n is None:
            best_a_n = np.ones(n) * S / n
            best_obj_n = objective_value(best_a_n)
        
        # Phase 2: Intensive polishing on the best
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_a_n is not None:
            n_best = len(best_a_n)
            S_best = np.sqrt(2.0 * n_best)
            
            seg1 = remaining * 0.4
            print("  Polishing phase 1 (alternating, %.0fs)..." % seg1)
            a_p1, obj_p1 = run_alternating(best_a_n, S_best, seg1, rng)
            obj_p1 = objective_value(a_p1)
            if obj_p1 < best_obj_n:
                best_obj_n = obj_p1
                best_a_n = a_p1.copy()
                print("    Improved to: %.6f" % best_obj_n)
            
            seg2 = remaining * 0.3
            if time.monotonic() < deadline - 20:
                print("  Polishing phase 2 (peak chase, %.0fs)..." % seg2)
                a_p2, obj_p2 = run_peak_chase(best_a_n, S_best, seg2, rng, k=8)
                obj_p2 = objective_value(a_p2)
                if obj_p2 < best_obj_n:
                    best_obj_n = obj_p2
                    best_a_n = a_p2.copy()
                    print("    Improved to: %.6f" % best_obj_n)
            
            seg3 = remaining * 0.2
            if time.monotonic() < deadline - 10:
                print("  Polishing phase 3 (KKT, %.0fs)..." % seg3)
                a_p3, obj_p3 = run_kkt_polish(best_a_n, S_best, seg3, rng)
                obj_p3 = objective_value(a_p3)
                if obj_p3 < best_obj_n:
                    best_obj_n = obj_p3
                    best_a_n = a_p3.copy()
                    print("    Improved to: %.6f" % best_obj_n)
        
        # Update global
        if best_obj_n < global_best_obj:
            global_best_obj = best_obj_n
            global_a = best_a_n.copy()
            print("  >>> Global best: obj=%.6f (n=%d)" % (global_best_obj, len(global_a)))
    
    # Final verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: evaluate_sequence = %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
        else:
            global_best_obj = min(global_best_obj, ver)
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
    
    print("FINAL: n=%d, obj=%.6f" % (len(global_a), global_best_obj))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
