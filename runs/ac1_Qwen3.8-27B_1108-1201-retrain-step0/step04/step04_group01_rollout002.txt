import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S, non-negative."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = scaled * (500.0 / scaled.max())
    terms = np.exp(scaled)
    effective_beta = beta * min(1.0, 500.0 / max(scaled.max(), 1e-10))
    return cm + np.log(terms.sum()) / effective_beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution with respect to a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scale_factor = 500.0 / max(scaled.max(), 1e-10)
        scaled = scaled * scale_factor
        beta_eff = beta * scale_factor
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def fista_optimize(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.003,
                   beta_max=8000.0, use_backtrack=False):
    """FISTA with LSE smoothing and optional backtracking line search."""
    a = normalize_to_sum(a_init, S)
    n = len(a)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if use_backtrack:
            # Backtracking line search
            step = 1.0 / (4.0 * beta * 0.3)
            for _ in range(30):
                x_new = project_simplex(y - step * grad, S)
                c_new = fft_convolve_self(x_new)
                new_val = lse_smooth_max(c_new, beta)
                old_val = lse_smooth_max(c, beta)
                # Check sufficient decrease
                if new_val <= old_val - 0.1 * step * gnorm * gnorm / 2:
                    break
                step *= 0.5
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            step = 1.0 / (4.0 * beta * 0.3)
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new

        # Periodic restart with noise
        if stagnation > 2000:
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 + 0.5 * np.exp(-stagnation / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stagnation = 0
            beta = beta_init

        if it % 5000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.0008 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def barrett_stopford(a_init, S, max_time, rng, n_iter_max=50000):
    """
    Barrett-Stopford style spectral shaping.
    Iteratively filters the spectrum to reduce autocorrelation peaks.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time and it < n_iter_max:
        it += 1
        progress = it / n_iter_max

        A = fft(a, N)
        P = np.abs(A) ** 2
        ac = np.real(ifft(P))
        ac_lin = ac[:n]

        # Adaptive threshold: start high, decrease
        threshold = ac_lin.mean() + (1.0 - 0.3 * progress) * (ac_lin.max() - ac_lin.mean())

        # Identify peaks above threshold
        peak_mask = ac_lin > threshold
        if not np.any(peak_mask):
            break

        # Build filter in lag domain
        F = np.zeros(N)
        F[:n] = peak_mask.astype(np.float64)
        F[N - n + 1:] = peak_mask[1:][::-1].astype(np.float64)

        # Frequency domain filter
        F_spec = np.abs(fft(F)) ** 2

        # Update spectrum: reduce components that contribute to peaks
        eps = 1e-20
        alpha = 0.3 + 0.3 * (1.0 - progress)
        ratio = 1.0 - alpha * F_spec / (F_spec.max() + eps)
        ratio = np.maximum(ratio, 0.01)
        A = A * np.sqrt(ratio)

        # Time domain: enforce constraints
        a_new = np.real(ifft(A))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)
        a = a_new

        cur = fft_convolve_self(a).max()
        if cur < best:
            best = cur
            best_a = a.copy()

    return best_a, best


def directed_local_search(a_init, S, max_time, rng, n_trials_per_iter=500):
    """
    Directed local search: for each trial, pick a peak-contributing pair
    and try to reduce it via a swap.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 100000.0)
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Find top lags
        top_k = min(8, n // 4)
        top_lags = np.argsort(c[:n])[::-1][:top_k]

        improved = False
        for lag in top_lags:
            if time.monotonic() - start > max_time:
                    break

            # For this lag, the contribution is sum_j a[j]*a[j+lag]
            # Find the positions (j, j+lag) with largest product
            j_max = min(n - lag, n)
            products = a[:j_max] * a[lag:lag + j_max]
            n_try = min(n_trials_per_iter, j_max)
            top_idx = np.argpartition(products, -n_try)[-n_try:]

            for j in top_idx:
                # Try: reduce a[j], increase a[j'] where j' is "far" from peak positions
                # Choose j' that minimizes its contribution to all top lags
                # Approximate: pick j' randomly from the "quiet" region
                delta = a[j] * rng.uniform(0.05, 0.2)

                # Find a good target position
                # Score each position by how much it would hurt other peaks
                candidates = rng.choice(n, size=min(50, n), replace=False)
                best_score = np.inf
                best_j2 = candidates[0]
                for j2 in candidates:
                    if j2 == j or j2 == j + lag or j2 == j - lag:
                        continue
                    # Estimate new peak values if we move delta from j to j2
                    score = 0.0
                    for l in top_lags:
                        if 0 <= j2 + l < n:
                            score += a[j2 + l] * delta
                        if 0 <= j2 - l < n:
                            score += a[j2 - l] * delta
                    if score < best_score:
                        best_score = score
                        best_j2 = j2

                if best_score >= c[lag] * 0.5:
                    continue  # Would hurt too much

                # Apply swap
                a_test = a.copy()
                a_test[j] -= delta
                a_test[best_j2] += delta
                a_test = np.maximum(a_test, 0.0)
                a_test = a_test * (S / a_test.sum())

                c_test = fft_convolve_self(a_test)
                if c_test.max() < c.max() * (1.0 - 1e-8):
                    a = a_test
                    improved = True
                    break

            if improved:
                break

        # If no improvement after many iterations, add small noise
        if not improved and it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                it = max(0, it - 100)  # Reset stagnation counter

        if it > 200000:
            break

    return best_a, best


def simulated_annealing(a_init, S, max_time, rng, T_init=0.01, T_min=1e-6, cool_rate=0.9995):
    """
    Simulated annealing with structured perturbations.
    Perturbation: pick a random lag, adjust a few elements to reduce it.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    T = T_init
    it = 0

    while time.monotonic() - start < max_time and T > T_min:
        it += 1
        progress = (time.monotonic() - start) / max_time

        # Structured perturbation: pick a random "block" and adjust
        block_size = rng.integers(max(2, n // 256), max(3, n // 64))
        start_pos = rng.integers(0, n - block_size)
        
        # Random perturbation within the block
        perturbation = rng.normal(0, 1, block_size)
        a_test = a.copy()
        a_test[start_pos:start_pos + block_size] += perturbation * (T * S / n)
        a_test = np.maximum(a_test, 0.0)
        a_test = a_test * (S / a_test.sum())

        c_test = fft_convolve_self(a_test)
        new_val = c_test.max()
        old_val = fft_convolve_self(a).max()

        if new_val < old_val or rng.random() < np.exp(-(new_val - old_val) / (T * old_val + 1e-15)):
            a = a_test
            if new_val < best:
                best = new_val
                best_a = a.copy()

        T *= cool_rate
        if T < T_min:
            T = T_min

        # Occasional large perturbation (restart-like)
        if it % 5000 == 0:
            large_scale = 0.01 * S / np.sqrt(n) * (1.0 + progress)
            a = project_simplex(a + rng.normal(0, large_scale, n), S)
            c_check = fft_convolve_self(a)
            if c_check.max() < best:
                best = c_check.max()
                best_a = a.copy()

    return best_a, best


def spectral_equalize(a_init, S, max_time, rng, n_outer=100):
    """
    Spectral equalization: iteratively try to flatten the power spectrum.
    Each outer iteration: shape spectrum, then re-project to feasible set.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    for outer in range(n_outer):
        if time.monotonic() - start > max_time:
            break

        A = fft(a, N)
        P = np.abs(A) ** 2

        # Target: make P[1:] as flat as possible (DC is fixed)
        P_non_dc = P[1:]
        mean_p = P_non_dc.mean()

        if mean_p < 1e-20:
            break

        # Create target spectrum: geometric mean between current and flat
        alpha = 0.3
        P_target = P.copy()
        P_target[1:] = P[1:] * ((mean_p / P[1:]) ** alpha)
        P_target = np.maximum(P_target, 0)

        # Amplitude of target spectrum
        target_amplitude = np.sqrt(P_target)
        current_amplitude = np.abs(A)

        # Phase is preserved, amplitude is modified
        phase = A / (current_amplitude + 1e-30)
        A_new = target_amplitude * phase

        # Time domain
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)
        a = a_new

        # Refine with a few FISTA steps
        a, val = fista_optimize(a, S, min(2.0, (max_time - (time.monotonic() - start)) * 0.1),
                                 rng, beta_init=50.0, beta_rate=0.01, beta_max=2000.0)
        if val < best:
            best = val
            best_a = a.copy()

    return best_a, best


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def downsample_sequence(a_large, n_small):
    """Downsample by block averaging."""
    n_large = len(a_large)
    if n_large < n_small:
        return upscale_sequence(a_large, n_small)
    ratio = n_large // n_small
    truncated = a_large[:n_small * ratio]
    return truncated.reshape(n_small, ratio).mean(axis=1)


def generate_inits(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None:
        if len(given) == n:
            inits.append(given.copy())
        elif len(given) > n:
            inits.append(downsample_sequence(given, n))
        else:
            inits.append(upscale_sequence(given, n))

    # Various window shapes
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / (0.25 ** 2)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.2) + 0.15 * np.cos(2 * np.pi * 7 * tt))

    # Random with structure
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Block structure
    for nb in [8, 16, 32]:
        bl = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block_seq = np.repeat(heights, bl)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), 0.5)])
        inits.append(block_seq[:n])

    # Binary-like
    inits.append((rng.random(n) > 0.5).astype(float) * 0.5 + 0.5)

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA, spectral shaping, directed search, SA, polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, label=""):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        n = len(a)
        s = a.sum()
        if s < 0.01:
            return
        obj = objective_value(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  [%s] New global best: obj=%.8f (n=%d)" % (label, obj, n))

    # Primary length
    n_primary = 8192
    if given is not None and 2048 <= len(given) <= 16384:
        n_primary = len(given)

    S = np.sqrt(2.0 * n_primary)
    print("Primary length: n=%d, S=%.4f" % (n_primary, S))

    # Phase 1: Multi-start FISTA (explore landscape)
    print("\n=== Phase 1: Multi-start FISTA ===")
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.2)
    inits = generate_inits(n_primary, rng, given)
    best_a_p1 = None
    best_val_p1 = np.inf

    n_starts = min(8, len(inits))
    t_per_start = phase1_time / n_starts

    for i in range(n_starts):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per_start, deadline - time.monotonic() - 30)
        if t_alloc < 5:
            break
        a_opt, val = fista_optimize(inits[i], S, t_alloc, rng,
                                     beta_init=10.0, beta_rate=0.004, beta_max=5000.0)
        if val < best_val_p1:
            best_val_p1 = val
            best_a_p1 = a_opt.copy()
            obj = 2.0 * n_primary * val / (a_opt.sum() ** 2)
            print("  Init %d: max_b=%.6f, obj=%.6f" % (i, val, obj))

    if best_a_p1 is not None:
        update_global(best_a_p1, "FISTA")

    # Phase 2: Barrett-Stopford spectral shaping
    print("\n=== Phase 2: Barrett-Stopford spectral shaping ===")
    if time.monotonic() < deadline - 60:
        phase2_time = min(300.0, (deadline - time.monotonic()) * 0.15)
        if best_a_p1 is not None:
            a_bs, val_bs = barrett_stopford(best_a_p1, S, phase2_time, rng)
            if val_bs < best_val_p1:
                best_val_p1 = val_bs
                best_a_p1 = a_bs.copy()
                print("  BS improved: max_b=%.6f" % val_bs)
            update_global(a_bs, "BS")

    # Phase 3: Spectral equalization
    print("\n=== Phase 3: Spectral equalization ===")
    if time.monotonic() < deadline - 60:
        phase3_time = min(200.0, (deadline - time.monotonic()) * 0.1)
        if best_a_p1 is not None:
            a_se, val_se = spectral_equalize(best_a_p1, S, phase3_time, rng)
            if val_se < best_val_p1:
                best_val_p1 = val_se
                best_a_p1 = a_se.copy()
                print("  SE improved: max_b=%.6f" % val_se)
            update_global(a_se, "SE")

    # Phase 4: FISTA refinement with high beta
    print("\n=== Phase 4: High-beta FISTA refinement ===")
    if time.monotonic() < deadline - 60:
        phase4_time = min(300.0, (deadline - time.monotonic()) * 0.15)
        if best_a_p1 is not None:
            a_r, val_r = fista_optimize(best_a_p1, S, phase4_time, rng,
                                         beta_init=100.0, beta_rate=0.002, beta_max=15000.0,
                                         use_backtrack=True)
            if val_r < best_val_p1:
                best_val_p1 = val_r
                best_a_p1 = a_r.copy()
                print("  Refined: max_b=%.6f" % val_r)
            update_global(a_r, "Refine")

    # Phase 5: Directed local search
    print("\n=== Phase 5: Directed local search ===")
    if time.monotonic() < deadline - 30:
        phase5_time = min(400.0, (deadline - time.monotonic()) * 0.2)
        if best_a_p1 is not None:
            a_dl, val_dl = directed_local_search(best_a_p1, S, phase5_time, rng,
                                                  n_trials_per_iter=300)
            if val_dl < best_val_p1:
                best_val_p1 = val_dl
                best_a_p1 = a_dl.copy()
                print("  DL improved: max_b=%.6f" % val_dl)
            update_global(a_dl, "DL")

    # Phase 6: Simulated annealing
    print("\n=== Phase 6: Simulated annealing ===")
    if time.monotonic() < deadline - 30:
        phase6_time = min(200.0, (deadline - time.monotonic()) * 0.12)
        if best_a_p1 is not None:
            a_sa, val_sa = simulated_annealing(best_a_p1, S, phase6_time, rng,
                                                T_init=0.005, T_min=1e-7, cool_rate=0.9998)
            if val_sa < best_val_p1:
                best_val_p1 = val_sa
                best_a_p1 = a_sa.copy()
                print("  SA improved: max_b=%.6f" % val_sa)
            update_global(a_sa, "SA")

    # Phase 7: Iterated local search (repeat: perturb + directed search)
    print("\n=== Phase 7: Iterated local search ===")
    if time.monotonic() < deadline - 30:
        phase7_time = (deadline - time.monotonic()) * 0.25
        if best_a_p1 is not None:
            n_ils = max(1, int(phase7_time / 30.0))
            for ils_i in range(n_ils):
                if time.monotonic() > deadline - 20:
                    break
                # Perturb the current best
                noise_scale = 0.003 * S / np.sqrt(n_primary)
                a_pert = project_simplex(best_a_p1 + rng.normal(0, noise_scale, n_primary), S)
                # Quick directed search from perturbed point
                t_sub = min(25.0, deadline - time.monotonic() - 20)
                if t_sub < 5:
                    break
                a_ils, val_ils = directed_local_search(a_pert, S, t_sub, rng,
                                                        n_trials_per_iter=200)
                if val_ils < best_val_p1:
                    best_val_p1 = val_ils
                    best_a_p1 = a_ils.copy()
                    print("  ILS %d: max_b=%.6f" % (ils_i, val_ils))
                update_global(a_ils, "ILS")

    # Final polishing
    print("\n=== Final polishing ===")
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        seg = remaining * 0.6
        a_fin, val_fin = fista_optimize(global_a, S_best, seg, rng,
                                         beta_init=500.0, beta_rate=0.001, beta_max=30000.0,
                                         use_backtrack=True)
        if val_fin < fft_convolve_self(global_a).max():
            global_a = a_fin.copy()
            obj_fin = objective_value(global_a)
            global_best = obj_fin
            print("  Final FISTA: obj=%.8f" % global_best)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_ps, val_ps = directed_local_search(global_a, S_best, remaining2 * 0.7, rng,
                                                  n_trials_per_iter=200)
            if val_ps < fft_convolve_self(global_a).max():
                global_a = a_ps.copy()
                obj_ps = objective_value(global_a)
                global_best = obj_ps
                print("  Final DL: obj=%.8f" % global_best)

    # Verify and return
    if global_a is None:
        # Fallback
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        global_best = objective_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: evaluate_sequence = %.8f" % ver)
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Final: length=%d, objective=%.8f" % (len(global_a), global_best))
    return final_list
