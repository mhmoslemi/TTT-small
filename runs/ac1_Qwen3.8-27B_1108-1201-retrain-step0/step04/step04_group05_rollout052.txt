def equalize_step(a, n, N, top_k=20, mu=1.0):
    """Equalize the top peaks using a linearized system."""
    S = a.sum()
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    
    # Get top peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    # Target: equalize to the maximum
    target = top_vals[0]
    
    # Build the linear system: conv(d, a)[k] = (target - c[k])/2 for k in top_idx
    # conv(d, a)[k] = sum_j d[j] * a[k-j]
    # This is a linear operator applied to d
    
    # We want to minimize ||d||^2 subject to Conv_a * d = rhs
    # where Conv_a is the matrix with (Conv_a * d)[k] = sum_j d[j] * a[k-j]
    
    # This can be solved via FFT!
    # Conv_a * d = ifft(fft(d) * fft(a))
    # So we want: ifft(fft(d) * fft(a))[top_idx] = rhs[top_idx]
    
    # In frequency domain: fft(d)[k] * fft(a)[k] = fft(rhs_zero_padded)[k]
    # So: fft(d)[k] = fft(rhs)[k] / fft(a)[k] (where fft(a)[k] != 0)
    
    # This is NOT exact because we only have constraints at top_idx, not all positions.
    # But as an approximation, we can use the "psuedo-inverse" approach.
    
    # Alternative: use the fact that Conv_a is self-adjoint (for real a)
    # Conv_a * d = conv(d, a) = conv(a, d)
    # So we're solving: Conv_a * d = rhs (on the subspace of top_idx)
    
    # For the full system (all 2n-1 positions), the solution would be:
    # d = ifft(fft(rhs_full) / fft(a))
    # But we only have rhs at top_idx positions.
    
    # Approximation: set rhs_full = 0 except at top_idx
    rhs_full = np.zeros(2*n-1)
    for i, k in enumerate(top_idx):
        rhs_full[k] = (target - top_vals[i]) / 2.0
    
    # d = ifft(fft(rhs_full, N) / fft(a, N)) (element-wise division)
    af = fft(a, N)
    rf = fft(rhs_full, N)
    
    # Where af is small, clip the division
    denom = np.where(np.abs(af) > 1e-10, af, 1e-10 + 0j)
    df = rf / denom
    d = np.real(ifft(df))[:n]
    
    # Project to maintain sum and non-negativity
    d -= d.mean()  # preserve sum
    # Step size control
    step = mu * S / (n * np.linalg.norm(d))
    d = step * d
    
    a_new = a + d
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new * (S / a_new.sum())  # renormalize
    
    return a_new

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * S / n
    # Simple: if mostly non-negative, just scale
    frac_pos = np.sum(v > 0)
    if frac_pos == n:
        return v * (S / s)
    # Full projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_and_fft(a, N):
    """Returns (c, A) where c = conv(a,a) and A = fft(a padded to N)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def peak_equalization_step(a, c, A, N, top_k, target_frac=1.0, lam=0.01, step_scale=1.0):
    """
    Find a perturbation that equalizes top_k peaks using regularized frequency-domain solve.
    Returns (a_new, c_new, improvement).
    """
    n = len(a)
    S = a.sum()
    c_max = c.max()
    
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    target = c_max * target_frac
    
    # rhs: we want conv(delta, a)[k_i] = (target - c[k_i]) / 2
    rhs = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        rhs[int(k)] = (target - top_vals[i]) / 2.0
    
    # Solve in frequency domain with Tikhonov regularization
    rhs_pad = np.zeros(N)
    rhs_pad[:2 * n - 1] = rhs
    rhs_f = fft(rhs_pad)
    A_abs_sq = np.abs(A) ** 2
    delta_f = rhs_f * np.conj(A) / (A_abs_sq + lam * np.max(A_abs_sq))
    delta = np.real(ifft(delta_f))[:n]
    
    # Remove mean to preserve sum
    delta -= delta.mean()
    
    # Scale step
    delta_norm = np.linalg.norm(delta)
    if delta_norm < 1e-30:
        return a.copy(), c, 0.0
    step = step_scale * S / (n * delta_norm) * np.linalg.norm(c) / c_max
    delta = step * delta
    
    a_new = a + delta
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-30:
        return a.copy(), c, 0.0
    a_new = a_new * (S / s_new)
    
    c_new = compute_conv_full(a_new, N)
    cmax_new = c_new.max()
    improvement = c_max - cmax_new
    return a_new, c_new, improvement


def subgradient_descent_step(a, c, N, top_k, step_size):
    """One subgradient step on the max-convolution (on the simplex)."""
    n = len(a)
    S = a.sum()
    c_max = c.max()
    
    # Find active set: lags where c is within 1% of max
    thresh = c_max * 0.99
    active = np.where(c >= thresh)[0][:top_k]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    m = len(active)
    # Subgradient of max(conv(a,a)) w.r.t. a[j]:
    # g[j] = (2/m) * sum_{k in active} a[k-j] (for valid indices)
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/d a[j'] = 2*a[k-j'] (symmetric)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                g[j] += 2.0 * a[kj]
    g /= m
    
    # Normalize gradient
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-30:
        return a.copy()
    
    # Mirror descent: a_new = a * exp(-eta * g) / Z
    eta = step_size * S / (n * gnorm)
    log_ratio = -eta * g / max(g.max(), 1e-30)
    # Clip to avoid overflow
    log_ratio = np.clip(log_ratio, -30, 30)
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-30:
        return a.copy()
    a_new = a_new * (S / s_new)
    return a_new


def spectral_perturb(a, N, alpha, rng):
    """Perturb in frequency domain: multiply |A| by (1 + alpha * noise) and IFFT."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    
    # Random phase perturbation (preserves symmetry for real result)
    noise = rng.normal(0, alpha, N)
    noise[N//2] = 0.0  # keep Nyquist real if N even
    # Ensure conjugate symmetry
    noise_sym = np.zeros(N)
    for k in range(N//2 + 1):
        if k == 0 or k == N//2:
            noise_sym[k] = noise[k]
        else:
            noise_sym[k] = noise[k]
            noise_sym[N-k] = noise[k]  # same real noise for conjugate pair
    
    mag = np.abs(A)
    phase = np.angle(A)
    new_mag = np.maximum(mag * (1.0 + noise_sym * 0.1), 0.0)
    A_new = new_mag * np.exp(1j * phase)
    # Enforce conjugate symmetry
    for k in range(1, N//2 + 1):
        A_new[N-k] = np.conj(A_new[k])
    A_new[0] = np.abs(A_new[0])  # DC is real
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-30:
        return a.copy()
    return a_new * (S / s_new)


def lse_smoothed_conv(c, beta):
    """Compute LSE-smoothed max and its gradient contribution."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, c_max * np.ones_like(c)
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return lse_val, w


def lse_gradient(a, N, beta):
    """Gradient of LSE-smoothed max of conv(a,a)."""
    n = len(a)
    c, A = compute_conv_and_fft(a, N)
    lse_val, w = lse_smoothed_conv(c, beta)
    
    # grad[lse_max] = 2 * ifft(fft(w) * conj(fft(a)))[:n]
    W = fft(w, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c.max()


def objective_value(a, n, N):
    """Compute the actual objective: 2*n*max_b/sum^2."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = compute_conv_full(a, N)
    return 2.0 * n * c.max() / (S * S)


def make_init_from_given(n, N):
    """Try to use the given height_sequence_1."""
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            return np.clip(given, 0, 1000).astype(np.float64)
        # Resize by interpolation
        if len(given) >= 10:
            from numpy import interp
            new_x = np.linspace(0, 1, n)
            old_x = np.linspace(0, 1, len(given))
            return np.clip(interp(new_x, old_x, given), 0, 1000)
    except Exception:
        pass
    return None


def make_init_rand(n, rng):
    return rng.random(n) + 0.01


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.7 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt) + rng.random() * 2 * np.pi)
    return np.clip(a, 0.001, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for freq in [2, 5, 11, 17, 31]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.001, None)


def make_init_ramp(n, rng):
    """Ramp-like initialization (asymmetric)."""
    tt = np.linspace(0, 1, n)
    a = tt**0.5 + 0.1 * np.sin(2 * np.pi * 7 * tt) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_fourier_box(n, N, rng):
    """IFFT of a box in frequency domain (with non-negativity enforcement)."""
    bw = rng.integers(n//10, n//3)
    center = rng.integers(0, N//2 - bw)
    A = np.zeros(N, dtype=complex)
    A[center:center+bw] = 1.0
    A[N-center:N-center+bw] = 1.0  # conjugate
    A[0] = float(n) * 0.5
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.01
    return a


def search_single_start(a_init, n, S, N, max_time, rng, label=""):
    """
    Main optimization loop for a single starting point.
    Uses peak equalization + subgradient + spectral perturbation.
    Returns best (a, objective) found.
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a * (S / s)
    
    best_a = a.copy()
    best_obj = np.inf
    
    c = compute_conv_full(a, N)
    cur_obj = 2.0 * n * c.max() / (S * S)
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_a = a.copy()
    
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    last_best_time = start_time
    
    # Adaptive parameters
    top_k_start = 5
    top_k_max = 256
    lam = 0.1  # regularization for peak equalization
    step_size_sg = 0.5  # subgradient step
    spect_interval = 200
    spect_alpha = 0.3
    
    while time.monotonic() - start_time < max_time:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = elapsed / max_time
        
        # Adapt top_k: increase over time
        top_k = min(top_k_max, top_k_start + int(top_k_max * progress * 0.7))
        
        # Re-compute conv
        c = compute_conv_full(a, N)
        c_max = c.max()
        cur_obj = 2.0 * n * c_max / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            last_best_time = time.monotonic()
            stall_count = 0
            if label and it % 100 == 0:
                print("    [%s] iter %d: obj=%.6f (t=%.0fs)" % (label, it, best_obj, elapsed))
        else:
            stall_count += 1
        
        # If stuck for too long, perturb
        if stall_count > 500 or (time.monotonic() - last_best_time > 120 and rng.random() < 0.1):
            alpha_p = 0.5 + 0.5 * rng.random()
            a = spectral_perturb(a, N, alpha_p, rng)
            # Also try random kick
            if rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            stall_count = 0
            continue
        
        # Choose step type
        rand = rng.random()
        
        if rand < 0.5:
            # Peak equalization
            A = fft(np.concatenate([a, np.zeros(N - n)]))
            a_new, c_new, improvement = peak_equalization_step(
                a, c, A, N, top_k, target_frac=0.995, lam=lam, step_scale=1.0
            )
            if improvement > 0:
                c_check = compute_conv_full(a_new, N)
                obj_check = 2.0 * n * c_check.max() / (S * S)
                if obj_check < cur_obj:
                    a = a_new
                    c = c_new
                    lam = max(0.001, lam * 0.95)  # reduce regularization (be bolder)
                else:
                    lam = min(1.0, lam * 1.2)  # increase regularization
            else:
                lam = min(1.0, lam * 1.1)
        
        elif rand < 0.8:
            # Subgradient step
            a_new = subgradient_descent_step(a, c, N, top_k, step_size_sg)
            c_new = compute_conv_full(a_new, N)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new < cur_obj:
                a = a_new
                c = c_new
                step_size_sg = min(2.0, step_size_sg * 1.05)
            else:
                step_size_sg = max(0.05, step_size_sg * 0.8)
        
        else:
            # Small spectral perturbation
            a = spectral_perturb(a, N, 0.1 + 0.1 * rng.random(), rng)
    
    return best_a, best_obj


def fine_polish(a_init, n, S, N, max_time, rng):
    """Fine polishing: small equalization steps with large active set."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    a = a * (S / s) if s > 1e-10 else np.ones(n) * S / n
    
    best_a = a.copy()
    best_obj = 2.0 * n * compute_conv_full(a, N).max() / (S * S)
    
    start_time = time.monotonic()
    it = 0
    top_k = 100
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv_full(a, N)
        c_max = c.max()
        cur_obj = 2.0 * n * c_max / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Gradually increase top_k
        progress = (time.monotonic() - start_time) / max_time
        top_k = min(512, 50 + int(500 * progress))
        
        A = fft(np.concatenate([a, np.zeros(N - n)]))
        a_new, c_new, improvement = peak_equalization_step(
            a, c, A, N, top_k, target_frac=0.999, lam=0.005, step_scale=0.5
        )
        
        if improvement > 0:
            a = a_new
        else:
            # Try subgradient
            a_new = subgradient_descent_step(a, c, N, top_k, 0.2)
            c_new = compute_conv_full(a_new, N)
            if c_new.max() < c_max:
                a = a_new
            else:
                # Tiny random perturbation to escape
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-method approach: peak equalization, subgradient descent,
    spectral perturbation, with progressive refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    # Determine length (try the given one and a few alternatives)
    n_candidates = [8192]
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= len(given) <= 32768:
            n_candidates.insert(0, len(given))
    except Exception:
        pass
    n_candidates.extend([4096, 16384])
    # Remove duplicates, keep order
    seen = set()
    unique_n = []
    for x in n_candidates:
        if x not in seen:
            unique_n.append(x)
            seen.add(x)
    n_candidates = unique_n
    
    print("Candidate lengths: %s, budget=%.0fs" % (n_candidates, budget_s))
    
    global_best = np.inf
    global_a = None
    global_n = None
    
    def update_best(a_cand, n_val):
        nonlocal global_best, global_a, global_n
        N = 1 << (2 * n_val - 1).bit_length()
        S = a_cand.sum()
        if S < 0.01:
            return
        c = compute_conv_full(a_cand, N)
        obj = 2.0 * n_val * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_cand.copy()
            global_n = n_val
            print("  NEW BEST: obj=%.6f (n=%d)" % (obj, n_val))
        return obj
    
    # ---- Phase 1: Initial exploration with multiple starts and lengths ----
    print("=== Phase 1: Multi-start exploration ===")
    t0 = time.monotonic()
    phase1_budget = min(budget_s * 0.3, 700.0)
    
    # Prioritize the given sequence if available
    all_inits = []
    
    for n_val in n_candidates:
        N = 1 << (2 * n_val - 1).bit_length()
        S = np.sqrt(2.0 * n_val)
        
        inits_for_n = []
        g = make_init_from_given(n_val, N)
        if g is not None and g.sum() > 0.01:
            inits_for_n.append(("given", g))
        inits_for_n.append(("chirp", make_init_chirp(n_val, rng)))
        inits_for_n.append(("multifreq", make_init_multi_freq(n_val, rng)))
        inits_for_n.append(("ramp", make_init_ramp(n_val, rng)))
        
        for name, arr in inits_for_n:
            all_inits.append((n_val, name, arr))
    
    # Sort: given sequences first, then by n (prefer 8192)
    all_inits.sort(key=lambda x: (0 if x[1] == "given" else 1, abs(x[0] - 8192)))
    
    n_inits = len(all_inits)
    per_init_time = phase1_budget / max(n_inits, 1)
    print("  %d initializations, ~%.1f sec each" % (n_inits, per_init_time))
    
    for n_val, name, arr in all_inits:
        if time.monotonic() > deadline - 300:
            break
        N = 1 << (2 * n_val - 1).bit_length()
        S = np.sqrt(2.0 * n_val)
        a_norm = project_simplex(arr, S)
        
        rem = max(2.0, min(per_init_time, deadline - time.monotonic() - 300))
        if rem < 5:
            break
        
        a_best, obj_best = search_single_start(a_norm, n_val, S, N, rem, rng, label="%s/n%d" % (name, n_val))
        update_best(a_best, n_val)
    
    print("Phase 1 done. Best: %.6f at n=%d" % (global_best, global_n))
    
    # ---- Phase 2: Focused optimization on best length ----
    if global_a is not None:
        n_best = global_n
    else:
        n_best = 8192
        global_a = np.ones(n_best) * 1.0
        global_best = 2.0
    
    N = 1 << (2 * n_best - 1).bit_length()
    S = np.sqrt(2.0 * n_best)
    
    print("=== Phase 2: Focused optimization (n=%d) ===" % n_best)
    t2 = time.monotonic()
    phase2_budget = min(budget_s * 0.35, 900.0)
    
    a_curr = global_a.copy()
    a_curr = project_simplex(a_curr, S)
    stall = 0
    
    while time.monotonic() < t2 + phase2_budget:
        block_time = min(180.0, t2 + phase2_budget - time.monotonic())
        if block_time < 10:
            break
        
        a_new, obj_new = search_single_start(a_curr, n_best, S, N, block_time, rng, label="P2")
        update_best(a_new, n_best)
        
        if obj_new < global_best + 0.001:
            a_curr = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                # Try a different perturbation to escape
                a_curr = spectral_perturb(a_curr, N, 1.0, rng)
                if rng.random() < 0.5:
                    a_curr = project_simplex(a_curr + rng.normal(0, 0.02 * S / np.sqrt(n_best), n_best), S)
                stall = 0
    
    print("Phase 2 done. Best: %.6f" % global_best)
    
    # ---- Phase 3: Try other lengths with the best "shape" ----
    print("=== Phase 3: Cross-length transfer ===")
    t3 = time.monotonic()
    phase3_budget = min(budget_s * 0.15, 300.0)
    
    if time.monotonic() < deadline - 180:
        # Try transferring to nearby lengths
        for n_try in [n_best // 2, n_best * 2]:
            if n_try < 512 or n_try > 32768 or n_try == n_best:
                continue
            N_try = 1 << (2 * n_try - 1).bit_length()
            S_try = np.sqrt(2.0 * n_try)
            
            # Resize by interpolation
            from numpy import interp
            new_x = np.linspace(0, 1, n_try)
            old_x = np.linspace(0, 1, n_best)
            a_transfered = interp(new_x, old_x, global_a)
            a_transfered = project_simplex(a_transfered, S_try)
            
            rem = max(10.0, phase3_budget / 2 - (time.monotonic() - t3))
            if rem < 10:
                break
            
            a_t, obj_t = search_single_start(a_transfered, n_try, S_try, N_try, rem, rng, label="transfer_n%d" % n_try)
            update_best(a_t, n_try)
    
    # ---- Phase 4: Fine polishing ----
    print("=== Phase 4: Fine polishing ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        n_polish = global_n
        N_polish = 1 << (2 * n_polish - 1).bit_length()
        S_polish = np.sqrt(2.0 * n_polish)
        
        a_pol = global_a.copy()
        a_pol = project_simplex(a_pol, S_polish)
        
        a_fin, obj_fin = fine_polish(a_pol, n_polish, S_polish, N_polish, min(remaining - 10, 400.0), rng)
        update_best(a_fin, n_polish)
    
    # ---- Phase 5: Final verification and return ----
    print("=== Final ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f (n=%d)" % (ver, len(final_list)) if np.isfinite(ver) else "Verification: inf")
        
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    
    print("Returning length %d, best internal obj=%.6f" % (len(final_list), global_best))
    return final_list
