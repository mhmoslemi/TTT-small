"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Multi-phase optimization using adaptive LSE proximal gradient with
Nesterov momentum, spectral domain refinement, and vectorized coordinate descent.
Employs multiple sequence lengths and diverse initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def gradient_lse_conv(a, c, beta):
    """Gradient of LSE(conv(a,a), beta) w.r.t. a using FFT."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(A * np.conj(W)))[:n]
    return grad


def run_proximal_gradient(a_init, n, S, max_time, rng, beta_init=25.0, beta_end=1200.0):
    """Nesterov-accelerated proximal gradient with adaptive LSE smoothing."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t_nest = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    no_improve = 0
    step = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_init + (beta_end - beta_init) * (progress ** 0.6)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = gradient_lse_conv(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        cm = c.max()
        e = np.exp(beta * (c - cm))
        max_w = e.max() / e.sum()
        L = 4.0 * beta * max_w * n * (S / n)
        step = min(1.0 / (L + 1e-10), 0.8 * S / (gnorm + 1e-15))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_nest = t_new

        # Noise injection to escape local minima
        if no_improve > 200 and it % 50 == 0:
            decay = np.exp(-no_improve / 800.0)
            noise_scale = 0.008 * S / np.sqrt(n) * decay
            if noise_scale > S / n * 0.005:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_nest = 1.0
            elif no_improve > 1000:
                # Large restart
                noise_scale2 = 0.05 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale2, n), S)
                a = y.copy()
                t_nest = 1.0

        if beta > 800 and no_improve > 600:
            break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_refine(a_init, n, S, max_time, rng):
    """Fienup-style spectral domain refinement with adaptive flattening."""
    a = project_simplex(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / 3000.0)
        alpha = 0.15 + 0.75 * progress

        # Flatten power spectrum with peak suppression
        P_flat = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_flat
        P_suppress = 1.0 / np.maximum(P / P_mean, 1e-10)
        P_suppress /= P_suppress.mean()
        P_new = P_new * (0.75 + 0.25 * P_suppress)

        mag_new = np.sqrt(np.maximum(P_new, 0.0))
        mag_new *= np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < S * 0.05:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        # Stable blending
        blend = 0.6 + 0.3 * progress
        a = blend * a_new + (1.0 - blend) * a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a *= S / s

    return best_a, best


def compute_d_vec(a, j, m, n):
    """Vectorized computation of convolution change vector for transfer m->j."""
    k_range = np.arange(2 * n - 1)
    jk = j + k_range
    jmk = j - k_range
    mk = m + k_range
    mmk = m - k_range

    mask_j = (jk >= 0) & (jk < n)
    mask_jm = (jmk >= 0) & (jmk < n)
    mask_m = (mk >= 0) & (mk < n)
    mask_mm = (mmk >= 0) & (mmk < n)

    vec = np.zeros(2 * n - 1)
    if np.any(mask_j):
        vec[mask_j] += a[jk[mask_j]]
    if np.any(mask_jm):
        vec[mask_jm] += a[jmk[mask_jm]]
    if np.any(mask_m):
        vec[mask_m] -= a[mk[mask_m]]
    if np.any(mask_mm):
        vec[mask_mm] -= a[mmk[mask_mm]]
    return vec


def run_coord_descent(a_init, n, S, max_time, rng):
    """Vectorized coordinate descent with golden-section search."""
    a = project_simplex(a_init, S)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best = c.max()
    best_a = a.copy()
    it = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        n_try = min(150 + it * 2, 600)

        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1.0:
                break
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            d_vec = compute_d_vec(a, j, m, n)
            lo = -a[j]
            hi = a[m]
            if hi - lo < 1e-12:
                continue

            # Golden section search minimizing max(c + d*d_vec + 2d^2 at k=0)
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = (c + c_g * d_vec).max() + 2.0 * c_g * c_g
            fd = (c + d_g * d_vec).max() + 2.0 * d_g * d_g

            for _ in range(35):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = (c + c_g * d_vec).max() + 2.0 * c_g * c_g
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = (c + d_g * d_vec).max() + 2.0 * d_g * d_g

            d_opt = (a_g + b_g) / 2.0
            if abs(d_opt) < 1e-12:
                continue

            a_trial = a.copy()
            a_trial[j] += d_opt
            a_trial[m] -= d_opt
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue

            c_trial = c + d_opt * d_vec
            c_trial[0] = c[0] + 2.0 * (a[j] - a[m]) * d_opt + 2.0 * d_opt * d_opt

            cur = c_trial.max()
            if cur < best - 1e-11:
                a = a_trial
                c = c_trial
                best = cur
                best_a = a.copy()
                improved = True

        if not improved:
            noise = 0.003 * S / np.sqrt(n) * np.exp(-it / 40.0)
            if noise < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        # Periodic full recompute to prevent drift
        if it % 30 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def multi_beta_optimize(a_init, n, S, max_time, rng):
    """Run proximal gradient at multiple beta levels and combine."""
    # Split time among different beta ranges
    alphas = [
        (15.0, 100.0, 0.35),   # low-mid beta
        (80.0, 500.0, 0.40),   # mid-high beta
        (300.0, 2000.0, 0.25), # high beta
    ]
    best_a = a_init.copy()
    best_score = np.inf

    for beta_lo, beta_hi, frac in alphas:
        t_alloc = max_time * frac
        if t_alloc < 5:
            continue
        a_opt, score = run_proximal_gradient(
            best_a, n, S, t_alloc, rng,
            beta_init=beta_lo, beta_end=beta_hi
        )
        if score < best_score:
            best_score = score
            best_a = a_opt

    return best_a, best_score


def generate_starts(parent, n, S, rng):
    """Generate diverse starting sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if len(p) == n:
            starts.append(project_simplex(p, S))
        else:
            # Resample
            tt_p = np.linspace(0, 1, len(p))
            resampled = np.interp(tt, tt_p, p)
            starts.append(project_simplex(resampled, S))

    # Cosine taper
    starts.append(project_simplex(0.5 * (1 + np.cos(2 * np.pi * tt)), S))
    # Gaussian
    starts.append(project_simplex(np.exp(-6.0 * (tt - 0.5) ** 2), S))
    # Raised cosine squared
    starts.append(project_simplex((0.5 * (1 + np.cos(2 * np.pi * tt))) ** 2, S))
    # Sinc-like
    starts.append(project_simplex(np.maximum(np.sinc(tt - 0.5), 0.01), S))
    # Random
    starts.append(project_simplex(rng.random(n) + 0.1, S))
    # Two-peak
    starts.append(project_simplex(
        np.exp(-20 * (tt - 0.3) ** 2) + np.exp(-20 * (tt - 0.7) ** 2), S))
    # Triangular
    starts.append(project_simplex(np.maximum(1 - 2 * np.abs(tt - 0.5), 0.01), S))

    return starts
