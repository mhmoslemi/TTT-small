"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy:
1. FISTA with LSE smoothing for initial optimization
2. Active-set QP chasing (linearized constraints) for precise max reduction
3. Local coordinate descent for final polish
4. Multi-length exploration (8192, 16384)

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002, beta_max=5000.0):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 3000:
            step *= 0.3
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def solve_qp_chase(f, n, S, M_target, top_k=40):
    """
    Solve QP: min ||f_new - f||^2 s.t. sum(f_new)=S, f_new>=0,
    and linearized (f*f)_k <= M_target for top_k active lags.
    Returns f_new or None if infeasible.
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Find top_k active lags (those closest to max)
    idx = np.argsort(c)[::-1][:top_k]

    # Build constraint matrix: row_k[i] = f[k-i] if 0<=k-i<n else 0
    rows = np.zeros((top_k, n))
    rhs = np.zeros(top_k)
    for ii, k in enumerate(idx):
        # (f*f)_k = sum_j f[j]*f[k-j], linearized around f:
        # approx c_k + 2*sum_j f[j]*(f_new[k-j] - f[k-j])
        # Constraint: c_k + 2*sum_j f[j]*f_new[k-j] - 2*c_k <= M_target
        # i.e., 2*sum_j f[j]*f_new[k-j] <= M_target + c_k
        # i.e., sum_j f[j]*f_new[k-j] <= (M_target + c_k) / 2
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                rows[ii, j] = f[ki]
        rhs[ii] = (M_target + c[k]) / 2.0

    # Solve QP
    f_var = cp.Variable(n)
    obj = cp.sum_squares(f_var - f)
    constraints = [rows @ f_var <= rhs, cp.sum(f_var) == S, f_var >= 0]

    try:
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    except Exception:
        try:
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None

    if f_var.value is None:
        return None

    f_new = np.maximum(np.asarray(f_var.value, dtype=np.float64), 0.0)
    f_new = project_simplex(f_new, S)

    # Verify the solution actually satisfies constraints reasonably
    c_new = fft_convolve_self(f_new)
    if c_new.max() > M_target * 1.05 + 0.01:
        # Solution doesn't satisfy constraints well, reject
        return None

    return f_new


def qp_chase(f, n, S, max_time, rng, initial_rate=0.003, min_rate=1e-6, top_k=40):
    """
    Iteratively chase the max down using QP with decreasing M_target.
    """
    start = time.monotonic()
    c = fft_convolve_self(f)
    M = c.max()
    best = M
    best_f = f.copy()
    rate = initial_rate
    stuck_count = 0
    it = 0

    print("  QP chase: starting M = %.8f, top_k = %d" % (M, top_k))

    while time.monotonic() - start < max_time:
        it += 1
        M_target = M * (1.0 - rate)

        if M_target < best * 0.999:
            break

        f_new = solve_qp_chase(f, n, S, M_target, top_k=top_k)

        if f_new is not None:
            c_new = fft_convolve_self(f_new)
            M_new = c_new.max()
            if M_new < best:
                best = M_new
                best_f = f_new.copy()
                f = f_new
                M = M_new
                stuck_count = 0
                it %= 1000000  # prevent overflow
                if it % 10 == 0:
                    print("    QP iter %d: M = %.8f (rate=%.6f)" % (it, M, rate))
            else:
                stuck_count += 1
                if stuck_count > 5:
                    rate *= 0.5
                    stuck_count = 0
                    if rate < min_rate:
                        break
        else:
            rate *= 0.5
            if rate < min_rate:
                break
            # Try a smaller step
            M_target = M * (1.0 - rate)

    print("  QP chase: final M = %.8f (from %.8f), iters=%d" % (best, fft_convolve_self(f).max(), it))
    return best_f, best


def coordinate_descent(f, n, S, max_time, rng):
    """
    Coordinate descent: try small perturbations of individual elements.
    """
    start = time.monotonic()
    c = fft_convolve_self(f)
    M = c.max()
    best = M
    best_f = f.copy()
    base = f.mean()
    no_improve_passes = 0

    while time.monotonic() - start < max_time:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > max_time:
                break

            # Try a few deltas
            for delta in [-0.5 * base * 0.01, -0.1 * base * 0.01, 0.1 * base * 0.01, 0.5 * base * 0.01,
                          -0.5 * base * 0.005, 0.5 * base * 0.005]:
                f_temp = f.copy()
                f_temp[i] = max(0.0, f[i] + delta)
                if abs(f_temp[i] - f[i]) < 1e-15:
                    continue
                f_temp = project_simplex(f_temp, S)
                c_temp = fft_convolve_self(f_temp)
                M_temp = c_temp.max()
                if M_temp < M - 1e-10:
                    f = f_temp
                    c = c_temp
                    M = M_temp
                    if M < best:
                        best = M
                        best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best


def random_perturb_search(f, n, S, max_time, rng, perturb_scale=0.01):
    """
    Try random perturbations and keep improvements.
    """
    start = time.monotonic()
    c = fft_convolve_self(f)
    M = c.max()
    best = M
    best_f = f.copy()
    base = f.mean()

    while time.monotonic() - start < max_time:
        # Perturb a random subset
        n_perturb = rng.integers(n // 10, n // 3)
        idx = rng.choice(n, size=n_perturb, replace=False)
        f_temp = f.copy()
        f_temp[idx] += rng.normal(0, perturb_scale * base, n_perturb)
        f_temp = np.maximum(f_temp, 0.0)
        f_temp = project_simplex(f_temp, S)
        c_temp = fft_convolve_self(f_temp)
        M_temp = c_temp.max()

        if M_temp < best:
            best = M_temp
            best_f = f_temp.copy()
            f = f_temp
            # Keep the current perturb_scale or reduce slightly
            if rng.random() < 0.1:
                perturb_scale *= 0.8

        # Occasionally reduce perturb scale
        if rng.random() < 0.05:
            perturb_scale *= 0.9

    return best_f, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def multiplicative_repulsion(f, n, S, beta=50.0, lr=0.05):
    """Fast multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best = np.inf
    global_a = None

    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - 60))

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    print("Main length: %d" % main_len)

    # Phase 1: FISTA warmup
    n = main_len
    S = np.sqrt(2.0 * n)
    remaining = deadline - time.monotonic()
    print("=== Phase 1: FISTA warmup (n=%d, %.0fs) ===" % (n, remaining * 0.2))

    if given is not None:
        init = given[:n] if len(given) >= n else np.pad(given, (0, n - len(given)))
    else:
        init = np.ones(n) * (S / n)
        tt = np.linspace(0, 1, n, endpoint=False)
        init = 0.5 + 0.3 * np.cos(2 * np.pi * tt) + 0.2 * np.sin(4 * np.pi * tt)
        init = np.maximum(init, 0.01)

    a1, m1 = run_fista(init, n, S, remaining * 0.2, rng,
                       beta_init=20.0, beta_rate=0.003, beta_max=3000.0)
    a1 = normalize_sum(a1, n)
    c1 = fft_convolve_self(a1)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
        print("  Phase 1 best: %.8f" % global_best)

    # Phase 2: QP chase
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print("=== Phase 2: QP chase (remaining %.0fs) ===" % remaining)
        a2, m2 = qp_chase(global_a, n, S, remaining * 0.35, rng,
                          initial_rate=0.002, min_rate=1e-6, top_k=30)
        a2 = normalize_sum(a2, n)
        c2 = fft_convolve_self(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            print("  Phase 2 best: %.8f" % global_best)
        else:
            print("  Phase 2: no improvement (best=%.8f, phase2=%.8f)" % (global_best, c2.max()))

    # Phase 3: More FISTA at higher beta
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Phase 3: High-beta FISTA (remaining %.0fs) ===" % remaining)
        a3, m3 = run_fista(global_a, n, S, remaining * 0.2, rng,
                           beta_init=200.0, beta_rate=0.001, beta_max=10000.0)
        a3 = normalize_sum(a3, n)
        c3 = fft_convolve_self(a3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a3.copy()
            print("  Phase 3 best: %.8f" % global_best)
        else:
            print("  Phase 3: no improvement (best=%.8f, phase3=%.8f)" % (global_best, c3.max()))

    # Phase 4: Another QP chase round
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Phase 4: QP chase round 2 (remaining %.0fs) ===" % remaining)
        a4, m4 = qp_chase(global_a, n, S, remaining * 0.25, rng,
                          initial_rate=0.001, min_rate=5e-7, top_k=50)
        a4 = normalize_sum(a4, n)
        c4 = fft_convolve_self(a4)
        if c4.max() < global_best:
            global_best = c4.max()
            global_a = a4.copy()
            print("  Phase 4 best: %.8f" % global_best)
        else:
            print("  Phase 4: no improvement (best=%.8f, phase4=%.8f)" % (global_best, c4.max()))

    # Phase 5: Local coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("=== Phase 5: Coordinate descent (remaining %.0fs) ===" % remaining)
        a5, m5 = coordinate_descent(global_a, n, S, remaining * 0.3, rng)
        a5 = normalize_sum(a5, n)
        c5 = fft_convolve_self(a5)
        if c5.max() < global_best:
            global_best = c5.max()
            global_a = a5.copy()
            print("  Phase 5 best: %.8f" % global_best)
        else:
            print("  Phase 5: no improvement (best=%.8f, phase5=%.8f)" % (global_best, c5.max()))

    # Phase 6: Random perturbation search
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("=== Phase 6: Random perturb (remaining %.0fs) ===" % remaining)
        a6, m6 = random_perturb_search(global_a, n, S, remaining * 0.25, rng,
                                        perturb_scale=0.005)
        a6 = normalize_sum(a6, n)
        c6 = fft_convolve_self(a6)
        if c6.max() < global_best:
            global_best = c6.max()
            global_a = a6.copy()
            print("  Phase 6 best: %.8f" % global_best)
        else:
            print("  Phase 6: no improvement (best=%.8f, phase6=%.8f)" % (global_best, c6.max()))

    # Phase 7: Try n=16384
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        print("=== Phase 7: Try n=%d (remaining %.0fs) ===" % (n2, remaining))
        if global_a is not None:
            init2 = interpolate_sequence(global_a, n2)
            init2 = normalize_sum(init2, n2)
        else:
            init2 = np.ones(n2) * (S2 / n2)

        # Quick FISTA
        a7, m7 = run_fista(init2, n2, S2, remaining * 0.3, rng,
                           beta_init=30.0, beta_rate=0.003, beta_max=5000.0)
        a7 = normalize_sum(a7, n2)
        c7 = fft_convolve_self(a7)
        obj7 = 2.0 * n2 * c7.max() / (a7.sum() ** 2)
        if obj7 < global_best:
            global_best = obj7
            global_a = a7.copy()
            print("  Phase 7 (n=%d) best: %.8f" % (n2, global_best))
        else:
            print("  Phase 7 (n=%d): %.8f (no improvement)" % (n2, obj7))

        # QP chase on n=16384 if improved
        remaining = deadline - time.monotonic()
        if remaining > 30 and len(global_a) == n2:
            a7b, m7b = qp_chase(global_a, n2, S2, remaining * 0.4, rng,
                                initial_rate=0.002, min_rate=1e-6, top_k=30)
            a7b = normalize_sum(a7b, n2)
            c7b = fft_convolve_self(a7b)
            obj7b = 2.0 * n2 * c7b.max() / (a7b.sum() ** 2)
            if obj7b < global_best:
                global_best = obj7b
                global_a = a7b.copy()
                print("  Phase 7b (n=%d) best: %.8f" % (n2, global_best))

    # Phase 8: Final polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        print("=== Phase 8: Final polish (n=%d, remaining %.0fs) ===" % (n_p, remaining))

        # One more QP chase with very small rate
        a8, m8 = qp_chase(global_a, n_p, S_p, remaining * 0.4, rng,
                          initial_rate=0.0005, min_rate=1e-7, top_k=60)
        a8 = normalize_sum(a8, n_p)
        c8 = fft_convolve_self(a8)
        obj8 = 2.0 * n_p * c8.max() / (a8.sum() ** 2)
        if obj8 < global_best:
            global_best = obj8
            global_a = a8.copy()
            print("  Phase 8 QP best: %.8f" % global_best)

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a8b, m8b = coordinate_descent(global_a, n_p, S_p, remaining * 0.5, rng)
            a8b = normalize_sum(a8b, n_p)
            c8b = fft_convolve_self(a8b)
            obj8b = 2.0 * n_p * c8b.max() / (a8b.sum() ** 2)
            if obj8b < global_best:
                global_best = obj8b
                global_a = a8b.copy()
                print("  Phase 8 CD best: %.8f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.8f" % ver)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, best internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
