"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Key insight: For non-negative sequences, the max of the self-convolution
is always at lag 0 (b_0 = sum a_j^2), by AM-GM inequality. Thus the
objective simplifies to 2n * sum(a_j^2) / (sum a_j)^2, which is a smooth
function on the simplex. The global minimum is 2 (uniform sequence).

Strategy:
1. Evaluate a uniform sequence (guaranteed 2.0)
2. Evaluate the provided height_sequence_1
3. If h1 is better (which would indicate numerical effects), optimize further
4. Use exact gradient descent on the simplex for fast convergence
5. Try multiple initialization strategies and return the best
"""
import numpy as np
import time


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n * max_conv / sum^2 using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    # For non-negative sequences, max is at lag 0: sum(a_j^2)
    # But use actual convolution to be safe
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)[:size]
    max_b = c.max()
    return 2.0 * n * max_b / (s * s)


def compute_objective_simple(a):
    """Faster version: use the fact that max is at lag 0 for non-neg sequences."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    q = np.sum(a * a)
    return 2.0 * n * q / (s * s)


def exact_gradient(a):
    """Exact gradient of f(a) = 2n * sum(a^2) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    q = np.sum(a * a)
    # df/da_i = 4n(a_i * s - q) / s^3
    return 4.0 * n * (a * s - q) / (s * s * s)


def gradient_descent_simplex(a_init, max_iters, step_init, momentum_decay=0.0):
    """
    Projected gradient descent on the simplex with fixed sum.
    Uses the exact gradient of 2n*sum(a^2)/(sum a)^2.
    """
    n = len(a_init)
    S = a_init.sum()
    a = np.clip(a_init.copy(), 0, None)
    a = a * (S / a.sum())

    v = np.zeros(n)  # momentum
    t = 1.0
    best_obj = compute_objective_simple(a)
    best_a = a.copy()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        grad = exact_gradient(a)
        # FISTA-style momentum
        v = momentum_decay * v + (1 - momentum_decay) * grad
        a_new = project_simplex(a - step * v, S)

        obj = compute_objective_simple(a_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step size when stuck
            if no_improve % 100 == 0:
                step *= 0.8

        a = a_new

        # Adaptive step: use Lipschitz-like estimate
        if it % 500 == 0 and it > 0:
            grad2 = exact_gradient(a)
            gn = np.linalg.norm(grad2)
            if gn > 1e-15:
                step = min(step, S / (2 * gn))

    return best_a, best_obj


def fista_optimize(a_init, budget_s, seed=42):
    """
    FISTA with exact gradient for the smooth objective on the simplex.
    The objective f(a) = 2n*sum(a^2)/(sum a)^2 is convex on the simplex,
    so FISTA converges to the global minimum (uniform point, f=2).
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S = a_init.sum()
    a = np.clip(a_init.astype(np.float64), 0, None)
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_objective_simple(a)
    best_a = a.copy()
    step = S / (2.0 * n)  # Initial step based on Lipschitz constant estimate
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        grad = exact_gradient(y)
        gn = np.linalg.norm(grad)
        if gn < 1e-20:
            break

        # Adaptive step size
        step = S / (1.0 + 0.1 * it)
        step = min(step, S / (1.0 * gn + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        a = x_new
        t_mom = t_new

        obj = compute_objective_simple(a)
        if obj < best_obj - 1e-15:

            best_a = a.copy()
