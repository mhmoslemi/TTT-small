def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_gradient(y, c, beta, mode='lse', k=1):

            return np.zeros(n)
        w = e / tot

        idx = np.argsort(c)[::-1][:k]

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimizer: FISTA with adaptive LSE smoothing, backtracking,
multi-start initialization, active-set polishing, and coordinate descent refinement.
Key improvements: smarter step size via LSE-based backtracking, piecewise beta

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_and_weights(c, beta):
    """Compute LSE value and softmax weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.full(len(c), 1.0 / len(c))
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return lse_val, w


def gradient_from_weights(y, w):
    """Compute gradient given weights w over convolution lags."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_weights(c, k):
    """Get uniform weights on top-k lags."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    return w


def run_fista_v2(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                 beta_max=20000.0, mode='lse', step_scale=1.0,
                 use_backtracking=True, jump_beta=True):
    """FISTA with adaptive LSE, optional backtracking, and beta jumps."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    step = step_scale / beta_init
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Update beta
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_lse, w_lse = compute_lse_and_weights(c, beta)

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Compute gradient based on mode
        if mode == 'adaptive':
            if stuck_count < 300:
                w = w_lse
            elif stuck_count < 1500:
                k_blend = min(16, 4 + stuck_count // 200)
                w_topk = topk_weights(c, k_blend)
                blend = 0.3 + 0.4 * min(1.0, stuck_count / 1500.0)
                w = (1 - blend) * w_lse + blend * w_topk
            else:
                k_use = min(32, 8 + stuck_count // 300)
                w = topk_weights(c, k_use)
        elif mode == 'topk':
            k_use = max(2, min(64, n // 64))
            w = topk_weights(c, k_use)
        else:
            w = w_lse

        grad = gradient_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        if use_backtracking:
            step = step_scale / max(beta, 1.0)
            # Cap step to avoid overshooting
            step = min(step, 2.0 * S / gnorm)
            # Simple backtracking: ensure LSE doesn't increase too much
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            lse_trial, _ = compute_lse_and_weights(c_trial, beta)
            for _ in range(10):
                if lse_trial <= cur_lse + 1e-12:
                    break
                step *= 0.5
                x_trial = project_simplex(y - step * grad, S)
                c_trial = fft_convolve_self(x_trial)
                lse_trial, _ = compute_lse_and_weights(c_trial, beta)
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Beta jump when stuck
        if jump_beta and stuck_count > 2000 and stuck_count % 1000 == 0:
            beta = min(beta_max, beta * 2.0)

        # Noise injection when very stuck
        if stuck_count > 4000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it

        # Periodic momentum reset
        if it % 8000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_polish_v2(a_init, max_time, rng):
    """Intensive polishing with adaptive active set and exact line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive: increase number of active lags when stuck
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = min(8, 2 + stuck // 100)
        elif stuck < 5000:
            k = min(32, 8 + stuck // 500)
        else:
            k = min(128, 32 + stuck // 500)

        w = topk_weights(c, k)
        grad = gradient_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search: find step that minimizes max along the direction
        step_init = S / (gnorm * (1.0 + 0.005 * it))
        step_init = min(step_init, 2.0 * S / gnorm)
        step = step_init

        # Backtracking
        best_x = y.copy()
        best_val = cur_max
        for ls in range(15):
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            val = c_trial.max()
            if val < best_val:
                best_val = val
                best_x = x_trial
            if val <= cur_max - 0.01 * step * gnorm:
                break
            step *= 0.5

        y = best_x

        # Noise when very stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_coord_descent(a_init, max_time, rng, max_iter_per_coord=50):
    """Coordinate descent polish - update one coordinate at a time."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best_conv = c.max()
    best_a = a.copy()

    # Precompute: for each coordinate j, which convolution lags are affected
    # Changing a[j] affects c[k] for k in [j, j+n-1] (since (a*a)[k] involves a[j]*a[k-j])
    # specifically c[k] += 2*delta*a[k-j] for k-j in [0,n-1]

    order = np.random.permutation(n)
    pass_count = 0

    while time.monotonic() - start < max_time:
        for j in order:
            if time.monotonic() - start > max_time:
                break

            # Current values
            # c[k] for k in [j, j+n-1] are affected by a[j]
            # c[k] = sum_i a[i]*a[k-i], the terms involving a[j] are:
            #   a[j]*a[k-j] when k-j in [0,n-1], i.e., k in [j, j+n-1]
            #   a[k-j]*a[j] when k-j in [0,n-1] (same thing, it's symmetric)
            # Actually c[k] includes a[j]*a[k-j] once (from i=j) and a[k-j]*a[j] once (from i=k-j)
            # If j == k-j (i.e., k=2j), it's a[j]^2, derivative is 2a[j]
            # Otherwise derivative of c[k] w.r.t. a[j] is 2*a[k-j] (if k-j in [0,n-1])

            # We want to find delta that minimizes max_{k} (c[k] + 2*delta*a[k-j])
            # subject to a[j]+delta >= 0 and maintaining sum = S (approximately)

            # For simplicity, just do a small perturbation
            # Find the lags most affected
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)

            # Gradient at this coordinate (partial derivative of max c[k])
            # d(c[k])/d(a[j]) = 2*a[k-j] for k-j in [0,n-1]
            # We want to decrease the max, so if a[j] > 0, we want to decrease a[j]
            # if it's contributing to the max, or increase it if it's not.

            # Simple approach: try small perturbations
            a_old = a[j]
            if a_old < 1e-15:
                continue

            # The current contribution to max: find max of c[k] + 2*delta*a[k-j]
            # For delta < 0 (decreasing a[j]): reduces c[k] for k where a[k-j] > 0
            # For delta > 0: increases those c[k]

            # Heuristic: if a[j] is correlated with high c values, decrease it
            affected = c[k_min:k_max + 1]
            coeff = a[k_max - n + 1:min(k_min, n - 1) + 1]  # a[k-j] for k in [k_min, k_max]
            # Actually: for k in [j, min(j+n-1, 2n-2)], coefficient is a[k-j]
            indices_k = np.arange(k_min, k_max + 1)
            coeff = a[indices_k - j]  # a[k-j]

            # If the weighted max of (c + 2*delta*coeff) can be reduced by decreasing a[j]:
            # The lags where c[k] is near max AND coeff is large should be decreased
            # This means if a[j] * max(coeff) is significant, we should reduce a[j]

            max_c_near = c.max()
            # Try: reduce a[j] slightly and reproject
            delta = -0.01 * a_old
            a_new = a.copy()
            a_new[j] = max(0, a[j] + delta)
            a_new = a_new * (S / a_new.sum())

            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_conv * (1 - 1e-12):
                a = a_new
                c = c_new
                # Update
            else:
                # Try increasing (if a[j] is small)
                if a_old < S / n * 2:
                    delta = 0.01 * S / n
                    a_new = a.copy()
                    a_new[j] = a[j] + delta
                    a_new = a_new * (S / a_new.sum())
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < best_conv:
                        a = a_new
                        c = c_new

        pass_count += 1
        if pass_count % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best_conv:
                best_conv = c.max()
                best_a = a.copy()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
        best_a = a.copy()  # Actually best_a already tracks this
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def try_length_variants(a_best, max_time, rng):
    """Try different sequence lengths derived from best found."""
    results = []
    n = len(a_best)
    S = np.sqrt(2.0 * n)
    a_norm = a_best * (S / a_best.sum())

    # Try half length
    if n >= 512 and max_time > 60:
        a_half = downsample(a_norm, factor=2)
        n_half = len(a_half)
        S_half = np.sqrt(2.0 * n_half)
        a_half = a_half * (S_half / a_half.sum())
        obj = compute_obj(a_half)
        results.append((obj, a_half, "half"))

    # Try double length
    if n <= 8192 and max_time > 120:
        a_dbl = upsample(a_norm, factor=2)
        n_dbl = len(a_dbl)
        S_dbl = np.sqrt(2.0 * n_dbl)
        a_dbl = a_dbl * (S_dbl / a_dbl.sum())
        obj = compute_obj(a_dbl)
        results.append((obj, a_dbl, "double"))

    return results


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Phases: multi-start init, FISTA exploration, upsampling, active-set polish,
    coordinate descent refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    best_obj = np.inf
    best_a = None

    def update_best(a):
        nonlocal best_obj, best_a
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            return True
        return False

    # --- Gather initial candidates ---
    candidates = []

    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_parent = np.clip(a_parent, 0, 1000)
        if a_parent.sum() >= 0.01:
            candidates.append(("parent", a_parent))
    except Exception:
        pass

    n_base = len(candidates[0][1]) if candidates else 8192

    if not candidates:
        tt = np.linspace(0, 1, n_base, endpoint=False)
        candidates.append(("cos1", 0.5 + 0.5 * np.cos(2 * np.pi * tt)))

    # Add diverse initializations
    tt = np.linspace(0, 1, n_base, endpoint=False)
    candidates.append(("cos2", 0.3 + 0.7 * np.cos(np.pi * tt)**2))
    candidates.append(("gauss", np.exp(-6 * (tt - 0.5)**2)))
    candidates.append(("ramp", np.maximum(0, 1 - np.abs(2 * tt - 1))))
    candidates.append(("cos4", 0.5 + 0.5 * np.cos(4 * np.pi * tt)))
    candidates.append(("mixed", 0.4 + 0.3 * np.cos(2 * np.pi * tt) + 0.3 * np.cos(4 * np.pi * tt + 0.5)))

    # Random starts
    for i in range(4):
        candidates.append((f"rand{i}", rng.random(n_base) + 0.05))

    # Quick evaluation
    print("=== Initial candidates ===")
    best_init = None
    best_init_obj = np.inf
    for name, a in candidates:
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            continue
        obj = compute_obj(a)
        print(f"  {name}: obj={obj:.6f}" + (f"  *" if obj < 1.55 else ""))
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        update_best(a)

    print(f"  Best initial: {best_init_obj:.6f}")

    # === Phase 1: FISTA on best init (30% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 500.0)
    n_cur = len(best_a)
    print(f"\n=== Phase 1: FISTA-LSE at n={n_cur} (budget {p1_budget:.0f}s) ===")
    t_phase = time.monotonic()

    a_r, _ = run_fista_v2(best_a, p1_budget, rng,
                           beta_init=10.0, beta_rate=0.006,
                           beta_max=10000.0, mode='lse', step_scale=1.0,
                           use_backtracking=True)
    update_best(a_r)
    elapsed = time.monotonic() - t_phase
    print(f"  Result: obj={best_obj:.6f} (t={elapsed:.0f}s)")

    # === Phase 2: Adaptive mode FISTA (25% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.28, 550.0)
    print(f"\n=== Phase 2: FISTA-adaptive at n={n_cur} (budget {p2_budget:.0f}s) ===")
    t_phase = time.monotonic()

    a_r, _ = run_fista_v2(best_a, p2_budget, rng,
                           beta_init=50.0, beta_rate=0.004,
                           beta_max=50000.0, mode='adaptive', step_scale=0.6,
                           use_backtracking=True)
    update_best(a_r)
    elapsed = time.monotonic() - t_phase
    print(f"  Result: obj={best_obj:.6f} (t={elapsed:.0f}s)")

    # === Phase 3: Upsample and optimize (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and len(best_a) <= 16384:
        p3_budget = min(remaining * 0.22, 400.0)
        n2 = len(best_a) * 2
        S2 = np.sqrt(2.0 * n2)
        a_up = upsample(best_a, factor=2)
        a_up = a_up * (S2 / a_up.sum())
        update_best(a_up)

        obj_up = compute_obj(a_up)
        if obj_up < best_obj * 1.01:  # Only worthwhile if close
            print(f"\n=== Phase 3: Upsampled to n={n2} (budget {p3_budget:.0f}s) ===")
            t_phase = time.monotonic()

            a_r, _ = run_fista_v2(a_up, p3_budget * 0.7, rng,
                                   beta_init=5.0, beta_rate=0.005,
                                   beta_max=5000.0, mode='lse', step_scale=1.0,
                                   use_backtracking=True)
            update_best(a_r)

            remaining3 = p3_budget - (time.monotonic() - t_phase)
            if remaining3 > 30:
                a_r, _ = run_polish_v2(best_a, remaining3, rng)
                update_best(a_r)

            elapsed = time.monotonic() - t_phase
            print(f"  Result: obj={best_obj:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"\n  Phase 3 skipped: upsampled obj={obj_up:.6f} not close enough")

    # === Phase 4: Active-set polishing (15% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    print(f"\n=== Phase 4: Active-set polish at n={len(best_a)} (budget {p4_budget:.0f}s) ===")
    t_phase = time.monotonic()

    a_r, _ = run_polish_v2(best_a, p4_budget, rng)
    update_best(a_r)
    elapsed = time.monotonic() - t_phase
    print(f"  Result: obj={best_obj:.6f} (t={elapsed:.0f}s)")

    # === Phase 5: Final high-beta FISTA + polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 5: Final refinement (budget {remaining:.0f}s) ===")
        t_phase = time.monotonic()

        # High-beta FISTA
        a_r, _ = run_fista_v2(best_a, remaining * 0.5, rng,
                               beta_init=200.0, beta_rate=0.001,
                               beta_max=100000.0, mode='adaptive', step_scale=0.3,
                               use_backtracking=False)
        update_best(a_r)

        remaining5 = remaining - (time.monotonic() - t_phase)
        if remaining5 > 10:
            a_r, _ = run_polish_v2(best_a, remaining5, rng)
            update_best(a_r)

        elapsed = time.monotonic() - t_phase
        print(f"  Result: obj={best_obj:.6f} (t={elapsed:.0f}s)")

    # === Try length variants if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Trying length variants (budget {remaining:.0f}s) ===")
        variants = try_length_variants(best_a, remaining, rng)
        for obj_v, a_v, desc in variants:
            print(f"  {desc}: obj={obj_v:.6f}")
            if obj_v < best_obj:
                update_best(a_v)
                # Quick polish
                rem = deadline - time.monotonic()
                if rem > 20:
                    a_r, _ = run_polish_v2(best_a, min(rem, 120), rng)
                    update_best(a_r)

    # Final verification
    if best_a is not None:
        final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"\n=== FINAL: verified obj={ver:.6f} (n={len(final_list)}) ===")
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")

    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    print(f"Using fallback (n={n_fb}, obj={compute_obj(a_fb):.6f})")
    return [float(x) for x in a_fb]
