- c[1] = 2*a[0]*a[1] (so a[1] = c[1]/(2*a[0]))
- c[2] = 2*a[0]*a[2] + a[1]^2 (so a[2] = (c[2] - a[1]^2)/(2*a[0]))
- etc.

So given c, we can recursively determine a (if a[0] > 0). The constraint is that a[i] >= 0 for all i.

This means: a[0] = sqrt(c[0]) > 0, and a[i] = (c[i] - sum_{j=1}^{i-1} a[j]*a[i-j] - (a[i/2]^2 if i even)) / (2*a[0]) for i >= 1.

This gives a necessary condition: a[i] >= 0 for all i, which translates to:
- c[0] > 0
- c[1] >= 0
- c[2] >= a[1]^2 = c[1]^2 / (4*c[0])
- c[3] >= 2*a[0]*a[2] = ... (gets complex)

This recursive structure suggests an alternative optimization approach: optimize c directly (which is easier since we just want c to be flat) subject to the constraints that c = a*a for some non-negative a.

But the constraints are complex. Let me stick with optimizing a directly.

**Revised Strategy:**

I'll implement a combination of:
1. **Adaptive LSE gradient descent** (with multiple active peaks) - this is the main workhorse
2. **Penalty-based peak shaving** - explicitly penalize the top M peaks
3. **Stochastic perturbations** to escape local minima
4. **Multi-start** with diverse initializations

Key improvements over the previous program:
- Use the top M peaks (not just the single max) in the LSE approximation
- Use a penalty method that explicitly targets the peaks
- Better adaptive beta scheduling
- More diverse initializations
- Try to exploit the recursive structure of the convolution

Let me also consider: what if I use a different length n? The previous program used n=8192. Let me try n=4096, n=16384, etc.

For the evaluation function, larger n makes the lower bound 2n/(2n-1) closer to 1, so in principle larger n is better. But the optimization is harder for larger n.

Given the time budget, I think n=8192 or n=16384 are reasonable choices.

"""
Minimize 2*n*max_self_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Adaptive LSE with multi-peak awareness
2. Penalty-based peak shaving (top-M peaks)
3. Stochastic perturbations for escape from local minima
4. Coordinate descent polish with golden section search
5. Multi-start with diverse initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, target_sum):
    """Clamp and rescale to sum = target_sum."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / s)


def lse_gradient_multi_peak(a, c, beta, top_m=5):
    """
    Gradient of LSE approximation to max(c), with emphasis on top peaks.
    Uses a weighted scheme: peaks above a threshold get extra weight.
    """
    n = len(a)
    cm = c.max()
    # Standard LSE weights
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    
    w = e / tot
    
    # Compute gradient via FFT
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def penalty_gradient(a, c, T, top_m=10):
    """
    Gradient of sum_{k in top_M} (c[k] - T)_+^2 
    This explicitly penalizes the peaks above threshold T.
    """
    n = len(a)
    # Get top M lags
    top_indices = np.argsort(c)[::-1][:top_m]
    
    grad = np.zeros(n)
    count = 0
    for k in top_indices:
        if c[k] > T:
            excess = c[k] - T
            # grad of c[k] = 2 * (a a_{k-:}) reversed appropriately
            # c[k] = sum_i a[i]*a[k-i], so d(c[k])/d(a[j]) = a[k-j] + a[j-k] (if valid)
            # = sum of a[k-j] for valid j + sum of a[j-k] for valid j
            # which is the convolution of a with a delta at position k
            # = a flipped and shifted
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            dck_da = a[k - idx]  # this is d(c[k])/d(a[idx])
            # Wait: c[k] = sum_{i} a[i] * a[k-i]
            # d(c[k])/d(a[j]) = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
            # But a[k-j] and a[j-k] are the same when we sum over all valid indices
            # Actually: c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # d/d(a[j]): term where i=j gives a[k-j] (if valid), and term where k-i=j gives a[i]*a[k-i] = a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]
            # So d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n and j != k-j
            # = a[j] if 2j = k (i.e., j = k/2)
            # Hmm, let me just use the FFT approach for accuracy
            
            count += 1
    
    # Use FFT for the gradient
    # grad = 2 * sum_{k in top, c[k]>T} (c[k]-T) * (a shifted by k)
    # This is 2 * sum_{k in top} (c[k]-T) * ifft(fft(a) * exp(-i*2*pi*k*j/N))
    # = 2 * ifft(fft(a) * fft(sum_{k in top} (c[k]-T) * delta_{k}))
    weights = np.zeros(2 * n - 1)
    for k in top_indices:
        if c[k] > T:
            weights[k] = c[k] - T
    
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(weights, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    
    return grad


def run_adaptive_lse(a_init, n, S, max_time, rng, 
                     beta_init=20.0, beta_max=3000.0,
                     beta_ramp=2000, top_m=1, use_penalty=False):
    """
    FISTA with adaptive LSE smoothing and optional penalty on top peaks.
    """
    a = normalize_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    patience = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Budget check
        if it % 10 == 0 and time.monotonic() - start > max_time:
            break
        
        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # Choose gradient
        if use_penalty and it > 500:
            # Use penalty gradient on top peaks
            T = cur - 0.05 * cur  # threshold slightly below max
            grad = penalty_gradient(y, c, T, top_m=max(5, top_m * 3))
            # Blend with LSE gradient
            lse_grad, _ = lse_gradient_multi_peak(y, c, beta, top_m)
            gnorm_p = np.linalg.norm(grad)
            gnorm_l = np.linalg.norm(lse_grad)
            if gnorm_p > 1e-14 and gnorm_l > 1e-14:
                grad = (grad / gnorm_p + lse_grad / gnorm_l) / 2.0
            elif gnorm_p > 1e-14:
                pass
            else:
                grad = lse_grad
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 0.5 / gnorm
        else:
            grad, _ = lse_gradient_multi_peak(y, c, beta, top_m)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
        
        # Adaptive step reduction
        if it - last_improve > 2000:
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart with noise
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
        
        # Patience-based restart
        if it - last_improve > 10000:
            # Big restart
            noise_scale = 0.05 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def penalty_peak_shave(a_init, n, S, max_time, rng, 
                       n_top=20, decay_rate=0.99):
    """
    Iteratively reduce the top peaks of the self-convolution.
    At each step, find the top M peaks, compute the gradient of their sum,
    and take a gradient step to reduce them.
    """
    a = normalize_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    current_top = n_top
    
    while time.monotonic() - start < max_time:
        if it % 5 == 0 and time.monotonic() - start > max_time:
            break
        it += 1
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Get top M peaks
        top_m = min(current_top, len(c))
        top_indices = np.argsort(c)[::-1][:top_m]
        
        # Gradient of sum of top peaks
        # d(sum_k c[k])/d(a[j]) = sum_k d(c[k])/d(a[j])
        # d(c[k])/d(a[j]) = a[k-j] (if valid) + a[j-k] (if valid)
        # = the j-th element of (a * delta_k) + (a * delta_{-k})
        # In FFT: grad = 2 * real(ifft(fft(a) * fft(weights)))
        
        weights = np.zeros(2 * n - 1)
        for k in top_indices:
            weights[k] = 1.0
        
        fsize = 1 << (2 * n - 1).bit_length()
        A = fft(a, fsize)
        W = fft(weights, fsize)
        grad = 2.0 * np.real(ifft(W * A))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Line search
        step = 0.1 * S / gnorm
        a_new = project_simplex(a - step * grad, S)
        
        # Verify improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur:
            # Reduce step
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
        
        a = a_new
        
        # Adaptively reduce top_m as peaks flatten
        if it % 100 == 0:
            current_top = max(5, int(current_top * decay_rate))
    
    return best_a, best


def coordinate_polish(a_init, n, S, max_time, rng):
    """
    Fast coordinate descent: try random pairs and use the exact 
    1D minimization to find the best transfer.
    """
    a = normalize_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < max_time:
        # Batch of random pairs
        n_pairs = min(100, max(30, n // 50))
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 1:
                break
            
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j == m:
                continue
            if a[j] < 1e-10 or a[m] < 1e-10:
                continue
            
            # d(k) = change in c[k] per unit transfer from m to j
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                if 0 <= jk < n:
                    val += a[jk]
                jmk = j - k
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                if 0 <= mk < n:
                    val -= a[mk]
                mmk = m - k
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val
            
            lo = -a[j]
            hi = a[m]
            
            # Golden section
            a_g, b_g = lo, hi
            if b_g - a_g < 1e-15:
                continue
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            
            def eval_delta(dlt):
                return (c + dlt * d).max()
            
            fc = eval_delta(c_g)
            fd = eval_delta(d_g)
            
            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = eval_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = eval_delta(d_g)
            
            delta = (a_g + b_g) / 2.0
            delta = max(lo, min(hi, delta))
            
            if abs(delta) < 1e-12:
                continue
            
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            
            if a_trial[j] < 0 or a_trial[m] < 0:
                continue
            
            c_trial = c + delta * d
            # Correct c[0]: the quadratic term
            # c[0] = a[0]^2, change = (a[0]+delta*1[j==0])^2 - a[0]^2 + ...
            # Actually for self-convolution, c[k] = sum_i a[i]*a[k-i]
            # When we change a[j] by +delta and a[m] by -delta:
            # c[k] changes by delta*(a[k-j] + a[j-k] - a[k-m] - a[m-k]) 
            # but we also need the quadratic correction when j==j or m==m
            # For k where both j and k-j are in {j, m}, there's a cross term
            # This is complex; let's just verify with recompute if close
            
            cur_trial = c_trial.max()
            if cur_trial < best - 1e-10:
                a = a_trial
                c = fft_convolve_self(a)  # recompute to be safe
                best = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Add small noise
            noise_scale = 0.002 * S / np.sqrt(n)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best


def spectral_init(n, rng):
    """
    Generate an initial sequence by designing its spectrum to give
    a flat self-convolution. Start from random phase and iterate.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    
    # Start with a random sequence
    a = rng.random(n) + 0.1
    a = a / a.sum() * np.sqrt(2 * n)
    
    # Iterate: in frequency domain, flatten |A|^4 (since C = ifft(A^2), 
    # we want C to be flat, so A^2 should be a delta, which is impossible.
    # Instead, we want ifft(A^2) to be flat, so A^2 should be approximately 
    # a delta in the spatial domain... 
    # Actually, if C is flat, then FFT(C) has only the DC term.
    # FFT(C) = A^2 (circular, in the padded domain)
    # So we want A^2 to be approximately a DC-only signal.
    # This means |A| should be approximately constant (since A^2 has DC = sum|A|^2/N^2... 
    # no, this is circular convolution in the padded domain).
    
    # Let me think differently. We want C = ifft(FFT(a)^2) to be flat.
    # C flat means FFT(C) = [S^2, 0, 0, ...] (DC only).
    # FFT(C) = FFT(FFT(a)^2) = ... this is getting circular.
    
    # Alternative: use the fact that if A = FFT(a) (padded to 2n-1),
    # then C = ifft(A^2). We want C to be as flat as possible.
    # C[k] = (1/N_fsize) * sum_j A[j]^2 * exp(i*2*pi*j*k/N_fsize)
    # For C to be flat, A[j]^2 should be concentrated at j=0.
    # But A[0] = sum(a) = S, and A[j] for j>0 should be small.
    # This means a should be approximately uniform!
    # Which gives us back the triangular convolution.
    
    # So the "flatten C" approach in frequency domain doesn't directly work.
    # Let's just use diverse random inits.
    
    inits = []
    
    # Uniform
    inits.append(np.ones(n) * np.sqrt(2 * n) / n)
    
    # Gaussian centered
    t = np.linspace(0, 1, n)
    inits.append(np.exp(-10 * (t - 0.5) ** 2))
    
    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.01))
    
    # Sinc-like
    inits.append(np.maximum(np.sinc(3 * (t - 0.5)) ** 2, 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    
    # Two-peak (non-palindromic)
    a_two = np.exp(-10 * (t - 0.3) ** 2) + 0.5 * np.exp(-10 * (t - 0.7) ** 2)
    inits.append(a_two)
    
    # Linear ramp (very non-palindromic)
    inits.append(t ** 2)
    
    # Power law
    inits.append((t + 0.01) ** 3)
    
    for init in inits:
        init = np.clip(init, 0.01, None)
        s = init.sum()
        inits[-1] = init * (np.sqrt(2 * n) / s)  # This overwrites, bug!
    
    # Fix: normalize each properly
    result = []
    for init in [np.ones(n), np.exp(-10*(t-0.5)**2), 
                 np.maximum(0.5*(1+np.cos(2*np.pi*t)), 0.01),
                 np.maximum(np.sinc(3*(t-0.5))**2, 0.01),
                 rng.random(n)+0.1,
                 np.exp(-10*(t-0.3)**2) + 0.5*np.exp(-10*(t-0.7)**2),
                 t**2, (t+0.01)**3]:
        init = np.clip(init, 0.0, None)
        s = init.sum()
        if s > 1e-10:
            result.append(init * (np.sqrt(2 * n) / s))
    
    return result


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Multi-phase approach:
    1. Multi-start LSE optimization with diverse initializations
    2. Penalty-based peak shaving on best candidate
    3. Coordinate descent polish
    4. Final LSE refinement
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)
    
    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None
    
    # Determine length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192
    
    S = np.sqrt(2.0 * n)
    
    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes" if parent is not None else "no"))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, S)
        c_chk = fft_convolve_self(a_norm)
        score = 2.0 * n * c_chk.max() / S**2
        if score < global_best:

            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start

    # Build starting points
    starts = []
    if parent is not None:
        starts.append(("parent", parent.copy()))

        starts.append(("shifted", shifted))

        update_best(a_opt, "lse_" + name)
