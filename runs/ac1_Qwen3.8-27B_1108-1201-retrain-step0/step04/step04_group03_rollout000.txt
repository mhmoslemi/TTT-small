Minimize 2*n*max(f*f)/sum(f)^2 using LP power iteration with adaptive
constraints, multi-scale exploration, and FISTA polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT, matching np.convolve(a, a)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def objective(f, n):
    """Compute the objective 2*n*max(f*f)/sum(f)^2."""
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(f)
    return 2.0 * n * c.max() / (s * s)


def solve_lp(f, M, n, top_k, solver='SCS'):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for selected lags, g >= 0.
    Returns g or None.
    """
    c = fft_convolve_self(f)
    total_lags = 2 * n - 1
    
    # Select constraints: all near-active + top remaining
    if top_k is not None and top_k < total_lags:
        threshold = M * 0.9
        near_active = np.where(c >= threshold)[0]
        if len(near_active) >= top_k:
            idx = near_active[np.argsort(c[near_active])[::-1][:top_k]]
        else:
            remaining = top_k - len(near_active)
            other_idx = np.argsort(c)[::-1]
            # Exclude already selected
            other_mask = np.ones(total_lags, dtype=bool)
            other_mask[near_active] = False
            other_candidates = other_idx[other_mask][:remaining]
            idx = np.concatenate([near_active, other_candidates])
    else:
        idx = np.arange(total_lags)
    
    idx = np.array(idx, dtype=np.int64)
    n_con = len(idx)
    
    # Build constraint matrix: n_con x n
    A_mat = np.zeros((n_con, n), dtype=np.float64)
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        A_mat[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [A_mat @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0.0)
        if g.sum() < 1e-10:
            return None
        return g
    
    return None


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta):
    cm = c.max()
    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def run_fista_polish(a, n, S, max_time, rng, beta_init=50.0, beta_max=2000.0):
    """FISTA polishing with LSE smoothing."""
    a = normalize_sum(a, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.002 ** it))
        c = fft_convolve_self(y)
        cur = objective(y, n)
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best


def multiplicative_repulsion(f, n, S, beta=40.0, lr=0.05):
    """Multiplicative repulsion update to flatten the convolution."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 0)
    f_new = normalize_sum(f_new, S)
    return f_new


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)
