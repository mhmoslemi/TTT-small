    cols = []
    vals = []

        r_end = min(2*n-2, j+n-1)  # k-j < n => k < j+n

    # Non-zero columns j satisfy: 0 <= k - j < n, i.e., k-n+1 <= j <= k

    row_idx = np.concatenate(row_idx)

"""
Minimize 2*n*max_conv/sum^2 using cutting-plane LP polishing + FISTA.

Key innovation: Instead of only using LSE-smoothed gradients, we alternate
with a cutting-plane LP that directly solves the subproblem from the KKT
conditions of the non-smooth max. This helps escape flat regions where
gradient methods stall, and provides a theoretical monotonicity guarantee.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy import sparse
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.01):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,

def build_conv_matrix(f, constraint_lags):
    """Build sparse matrix A where A[i,j] = f[j] if 0 <= constraint_lags[i]-j < n."""
    n = len(f)
    m = len(constraint_lags)
    row_idx = []
    col_idx = []
    values = []
    for i, k in enumerate(constraint_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row_idx.extend([i] * (j_end - j_start + 1))
            col_idx.extend(range(j_start, j_end + 1))
            values.extend(f[j_start:j_end + 1].tolist())
    if not row_idx:
        return sparse.csr_matrix((m, n))
    return sparse.coo_matrix((values, (row_idx, col_idx)), shape=(m, n)).tocsr()


def lp_direction(f, M, max_cp=15, max_cons=80, time_limit=15):
    """
    Cutting-plane LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    c_ff = fft_convolve_self(f)
    top_idx = np.argsort(c_ff)[::-1][:min(5, len(c_ff))]
    active = list(set(int(k) for k in top_idx))

    g = None
    for cp_it in range(max_cp):
        A = build_conv_matrix(f, active)
        b = np.full(len(active), M)
        c_obj = -np.ones(n)
        g_ub = 10.0 * np.sqrt(2.0 * n)

        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, g_ub)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

        if not res.success:
            break

        g = res.x.copy()

        # Check full feasibility
        fg = fft_convolve(f, g)
        tol = 1e-9 * max(M, 1.0)
        viol_mask = fg > M + tol

        if not np.any(viol_mask):
            break

        # Add most violated constraints
        viol_idx = np.where(viol_mask)[0]
        n_add = min(15, len(viol_idx))
        worst = viol_idx[np.argsort(fg[viol_idx])[::-1][:n_add]]
        for k in worst:
            if int(k) not in active:
                active.append(int(k))

        if len(active) > max_cons:
            fg_active = fg[active]
            order = np.argsort(fg_active)[::-1][:max_cons]
            active = [int(active[i]) for i in order]
            break

    if g is not None and g.sum() > 1e-10:
        return g
    return None
