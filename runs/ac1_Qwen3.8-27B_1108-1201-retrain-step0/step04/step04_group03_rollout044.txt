6. **New algorithm idea - Subgradient/Bundle method**:
   Since the objective `max_k (f*f)(k)` is the maximum of `2n-1` linear functions (in `f`), it's a convex piecewise-linear function. We can use:
   - **Cutting plane method**: At each iteration, linearize the max at the current point, add the constraint, and solve the resulting LP (minimize the linearized objective subject to the constraint and `sum(f) = S, f >= 0`). Over time, we accumulate more and more linearized constraints, and the solution converges to the optimum.
   - **Proximal subgradient**: Take a subgradient step projected onto the simplex.

   The cutting plane approach is particularly appealing because:
   - Each iteration solves an LP with `n` variables and a growing number of constraints (but each constraint is just one row)

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: FISTA with LSE + fast LP best-response via scipy/HiGHS + coordinate descent polish + multi-n exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_max(a, c):
    """Exact subgradient of max(c) at the argmax lag."""
    n = len(a)
    kstar = int(np.argmax(c))
    # Build one-hot at kstar in length 2n-1
    w = np.zeros(2 * n - 1)
    w[kstar] = 1.0
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002, beta_max=1000.0,
              use_subgrad=False, subgrad_freq=100, use_repulsion=True, repulsion_freq=300, repulsion_lr=0.04):
    """FISTA with LSE smoothing and optional exact subgradient / repulsion."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Choose gradient: LSE or exact subgradient
        if use_subgrad and it % subgrad_freq == 0:
            grad = subgradient_at_max(y, c)
            # Scale subgradient to be comparable to LSE gradient magnitude
            step = 1.0 / (2.0 * max(beta * 0.1, 5.0))
        else:
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (2.0 * beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: reduce if stuck
        if it - last_improve_it > 2500:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic repulsion
        if use_repulsion and it % repulsion_freq == 0 and it > 200:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(25.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-100:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 0)
                y = project_simplex(y, S)
                t = 1.0

        # Periodic noise injection
        if it % 8000 == 0:
            decay = np.exp(-it / 60000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def solve_best_response_lp(f, n, S, top_k=200, time_limit=15.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k_i) <= M for top-k lags, g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Get top-k lags by correlation value
    idx = np.argsort(c)[::-1][:top_k]

    # Build constraint matrix A (top_k x n)
    # A[i, j] = f[k_i - j] if valid, else 0
    A = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[i, lo:hi + 1] = f[lo:hi + 1][::-1]  # f[k - j] for j in [lo, hi]

    # Solve: max 1^T g s.t. A g <= M, g >= 0
    # In linprog: min -1^T g s.t. A g <= M, g >= 0
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(top_k, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g * (S / g_sum)
    except Exception:
        pass
    return None


def lp_iteration(f, n, S, top_k=200):
    """One LP-based iteration with step-size search."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()

    g = solve_best_response_lp(f, n, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old

    # Check if full replacement works
    c_g = fft_convolve_self(g)
    val_full = c_g.max()
    if val_full < M_old:
        return g, val_full

    # Bisection for best step
    best_t = 0.0
    best_val = M_old
    lo, hi = 0.0, 1.0
    for _ in range(7):
        mid = (lo + hi) / 2.0
        f_test = (1.0 - mid) * f + mid * g
        f_test = np.maximum(f_test, 0.0)
        f_test = project_simplex(f_test, S)
        val_mid = fft_convolve_self(f_test).max()
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0.05:
        f_new = (1.0 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val

    return f.copy(), M_old


def coordinate_descent_polish(f, n, S, max_time, rng, sweeps=3):

    f = normalize_sum(f, n)

            break
        order = rng.permutation(n)
        improved = 0

            # Current value
            fj = f[j]

            # c[k] involves f[j] for k in [j, j+n-1] (with bounds)
            # For efficiency, compute the "rest" without f[j]
            # c[k] = f[j]*f[k-j] + f[k-j]*f[j] + (other terms)
            # More precisely: c[k] has terms f[j]*f[k-j] if 0<=k-j<n, and f[k-j']*f[j] if k-j'=j i.e. j'=k-j (same)
            # So the contribution of f[j] to c[k] is:
            #   if k-j is in [0, n-1]: 2*f[j]*f[k-j] (or f[j]^2 if k=2j)
            #   plus the "background" (terms not involving f[j])

            # The dependence of c[k] on f[j]:
            # valid k range: max(0, j) to min(2n-2, j+n-1) = [j, j+n-1]
            # For k in [j, min(j+n-1, 2n-2)]:
            #   if k - j < n (always true since k <= j+n-1): c[k] includes f[j]*f[k-j]
            #   if k - j == j (k=2j): the term is f[j]^2, so quadratic
            #   else: linear in f[j], coefficient 2*f[k-j]
            # Plus from the other side: f[k' - j]*f[j] where k' - j = j means k' = 2j... 
            # Wait, I'm overcomplicating. Let me just use ternary search.

            # Ternary search for optimal f[j] in [0, 1000]
            # The objective max_k c[k] is convex in f[j] (max of convex functions)
            lo_v, hi_v = 0.0, max(2.0 * fj + 1.0, 2.0 * S / np.sqrt(n) * 5.0)

            for _ in range(12):
                m1 = lo_v + (hi_v - lo_v) / 3.0
                m2 = hi_v - (hi_v - lo_v) / 3.0

                f[j] = m1
                c1 = fft_convolve_self(f).max()
                f[j] = m2
                c2 = fft_convolve_self(f).max()

                    hi_v = m2
                else:

            f_opt = (lo_v + hi_v) / 2.0
            f[j] = f_opt
            # Re-normalize (sum constraint)
            s = f.sum()
            if s > 1e-10:
                f = f * (S / s)
            
            cur = fft_convolve_self(f).max()
            if cur < best - 1e-10:
                best = cur
                best_f = f.copy()
                improved += 1
            f[j] = best_f[j]  # revert if not improved (actually keep since we're tracking best)

        if improved == 0:
            break

    return best_f, best


def multiplicative_update(f, n, S, beta=30.0, lr=0.05):
    """Multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    return project_simplex(f_new, S)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

            S_alt = np.sqrt(2.0 * alt_n)
            t4 = remaining * 0.25
            print("Phase 4: Try n=%d (%.0fs)" % (alt_n, t4))
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            a4, s4 = run_fista(alt_init, alt_n, S_alt, t4, rng,
                              beta_init=15.0, beta_rate=0.002, beta_max=500.0,
                              use_subgrad=True, subgrad_freq=100,
                              use_repulsion=True, repulsion_freq=300, repulsion_lr=0.04)
            a4 = normalize_sum(a4, alt_n)
            c4 = fft_convolve_self(a4)
            obj4 = 2.0 * alt_n * c4.max() / (a4.sum() ** 2)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                print("  Phase 4 (n=%d) improved: %.6f" % (alt_n, global_best))
            else:
                print("  Phase 4 (n=%d): %.6f (no improvement)" % (alt_n, obj4))

    # === Phase 5: More LP iterations on best ===
    if global_a is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        n_5 = len(global_a)
        S_5 = np.sqrt(2.0 * n_5)
        t5 = remaining * 0.5
        print("Phase 5: Final hybrid polish (n=%d, %.0fs)" % (n_5, t5))
        a5, s5, nlp5 = run_hybrid(global_a, n_5, t5, rng, lp_every=100, top_k=250)
        a5 = normalize_sum(a5, n_5)
        obj5 = evaluate_internal(a5, n_5)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("  Phase 5 improved: %.6f (LP: %d)" % (global_best, nlp5))
        else:
            print("  Phase 5: %.6f (no improvement, LP: %d)" % (obj5, nlp5))

    # === Phase 6: Final coordinate descent ===
    if global_a is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_6 = len(global_a)
        S_6 = np.sqrt(2.0 * n_6)
        t6 = remaining * 0.6
        if t6 > 20:
            print("Phase 6: Final CD polish (%.0fs)" % t6)
            a6, s6 = coordinate_descent_polish(global_a, n_6, S_6, t6, rng, sweeps=3)
            a6 = normalize_sum(a6, n_6)
            obj6 = evaluate_internal(a6, n_6)
            if obj6 < global_best:
                global_best = obj6
                global_a = a6.copy()
                print("  Phase 6 improved: %.6f" % global_best)

    # === Final verification ===
    if global_a is None:
        # Fallback
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: len=%d, internal=%.6f, verified=%.6f" % (len(global_a), global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        n_fb = 256
        final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb

    return final_list
