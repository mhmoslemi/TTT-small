import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def objective(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def subgradient_peak(a, c, k_weights):
    """Compute 2*conv(w, a) where w is a sparse weight vector on the convolution."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(k_weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def optimize_fista(a_init, S, max_time, rng, mode="lse", 
                   beta_init=50.0, beta_rate=0.003, beta_max=5000.0):
    """FISTA optimization with LSE or subgradient."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Backtracking parameters
    c1 = 0.01  # Armijo parameter
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = y.copy()
        
        if mode == "lse":
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))
            cm = c.max()
            scaled = beta * (c - cm)
            if scaled.max() > 500:
                beta = 500.0 / max(1e-15, scaled.max())
                scaled = beta * (c - cm)
            e = np.exp(scaled)
            w = e / e.sum()
            grad = subgradient_peak(y, c, w)
        elif mode == "subgrad":
            # Weighted top-k peaks
            k = max(4, min(64, n // 64))
            idx = np.argsort(c)[::-1][:k]
            mean_c = c.mean()
            weights = np.maximum(c[idx] - mean_c, 0.0)
            wsum = weights.sum()
            if wsum < 1e-30:
                break
            w_full = np.zeros(len(c))
            w_full[idx] = weights / wsum
            grad = subgradient_peak(y, c, w_full)
        elif mode == "excess":
            T = c.mean() + 0.3 * (c.max() - c.mean())
            excess = np.maximum(c - T, 0.0)
            if excess.max() < 1e-15:
                T = c.max() - 0.001
            excess = np.maximum(c - T, 0.0)
            w_full = excess / max(excess.sum(), 1e-30)
            grad = subgradient_peak(y, c, w_full)
        else:
            break
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0
        obj_y = c.max()
        for _ in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            obj_new = c_new.max()
            if obj_new <= obj_y - c1 * step * gnorm * gnorm + 1e-10:
                break
            step *= 0.5
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize to maintain sum constraint approximately
        a = x_new
        t_k = t_new
        
        # Periodic restart to escape local minima
        if it % 5000 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_k = 1.0

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip to [0,1000] and scale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    scaled = beta * (c - cm)
    max_s = scaled.max()
    if max_s > 500:
        scale_factor = 500.0 / max_s
        scaled = scaled * scale_factor
        beta_eff = beta * scale_factor
    else:
        beta_eff = beta
    e = np.exp(scaled)
    return e / e.sum(), beta_eff


def topk_weights(c, k):
    """Compute weights on top-k peaks, proportional to excess over mean."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 1e-10)
    w_full = np.zeros(n)
    w_full[idx] = weights / weights.sum()
    return w_full


def conv_with_weights(a, w):
    """Compute 2*conv(w, a) up to length len(a)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0


def fista_lse(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.004,
              beta_max=50000.0, step_factor=1.0, stuck_threshold=2000):
    """FISTA with LSE smoothing and adaptive beta."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best - 1e-12:
            best = cm
            best_a = y.copy()
            last_improve = it
        
        # Update beta
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        # Compute LSE gradient
        w, beta_eff = softmax_weights(c, beta)
        grad = conv_with_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: based on Lipschitz constant estimate
        # L ≈ 2 * beta_eff * ||y||_1^2 / ||y||_n-like
        step = step_factor / (2.0 * beta_eff * S / n)
        step = min(step, 1.0 / (beta_eff * 0.1))
        
        x_new = project_simplex(y - step * grad, S)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        # Stuck detection
        if it - last_improve > stuck_threshold:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it
            beta = max(beta_init, beta * 0.5)  # Reset beta
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def subgrad_optimize(a_init, S, max_time, rng, k=8, step_mode="adaptive"):
    """Optimize using exact subgradient on top-k peaks."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best - 1e-12:
            best = cm
            best_a = y.copy()
            last_improve = it
        
        # Top-k weighted subgradient
        w = topk_weights(c, k)
        grad = conv_with_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        if step_mode == "adaptive":
            # Step based on current gap
            gap = cm - c.mean()
            if gap < 1e-10:
                break
            step = 0.5 * gap / (gnorm * gnorm / n)
            step = min(step, S / n * 0.1)
        else:
            step = 0.1 * S / (n * (1.0 + 0.005 * it))
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        if it - last_improve > 5000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def local_search(a_init, S, max_time, rng):
    """Local search via random perturbations of small groups."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()
    
    group_sizes = [1, 2, 3, 4, 6, 8, 12, 16]
    
    while time.monotonic() - start < max_time:
        # One second batch
        batch_end = min(start + 1.0, start + max_time)
        while time.monotonic() < batch_end:
            gs = group_sizes[rng.integers(len(group_sizes))]
            idx = rng.choice(n, size=gs, replace=False)
            
            a_new = a.copy()
            # Multiplicative perturbation
            factors = rng.lognormal(0, 0.3, gs)
            a_new[idx] *= factors
            a_new = normalize_to_sum(a_new, S)
            
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best - 1e-12:
                best = c_new.max()
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best


def excess_optimize(a_init, S, max_time, rng):
    """Excess-based optimization: minimize sum((b_k - T)^2_+) with annealed T."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    c0 = fft_convolve_self(a)
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.5)
        T = max(T, T_end)
        
        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best - 1e-12:
            best = cm
            best_a = y.copy()
        
        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            # All below threshold, try to reduce the max directly
            T = 0.999 * cm
            excess = np.maximum(c - T, 0.0)
        
        wsum = excess.sum()
        if wsum < 1e-30:
            break
        w = excess / wsum
        grad = conv_with_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = S / (n * (1.0 + 0.01 * progress) * 2.0)
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()
