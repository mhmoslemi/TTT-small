import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_gradient(a, c, beta):

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Uses adaptive LSE-smoothed FISTA with cyclic beta, backtracking line
search, spectral post-refinement, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """LSE-smoothed max."""
    cm = c.max()
    d = (c - cm) * beta
    d_max = d.max()
    if d_max > 500:
        return cm + np.log(np.exp(d - 500).sum()) / beta + 500 / beta
    # Actually, let me just be safe
    d_shifted = d - d_max
    return cm + (d_max + np.log(np.exp(d_shifted).sum())) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Shift for numerical stability
    s_max = scaled.max()
    if s_max > 500:
        e = np.exp(scaled - 500)
    else:
        e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: 2 * convolve(w, a) (truncated to n)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, S, max_time, rng, beta_init=20.0, beta_max=5000.0,
                  beta_cycle=2000):
    """Adaptive LSE FISTA with cyclic beta and backtracking."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall = 0
    
    # For backtracking
    prev_lse_val = None
    prev_y = None
    prev_grad = None
    step_size = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Cyclic beta: go from beta_init to beta_max and back
        phase = (it % beta_cycle) / beta_cycle
        beta = beta_init * (beta_max / beta_init) ** (0.5 * (1 - np.cos(2 * np.pi * phase)))
        beta = max(beta, beta_init)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = step_size
        cur_lse = lse_value(c, beta)
        
        x_new = project_simplex(y - step * grad, S)
        c_new = fft_convolve_self(x_new)
        lse_new = lse_value(c_new, beta)
        
        tries = 0
        while lse_new > cur_lse - 1e-15 * step * gnorm and tries < 15:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, beta)
            tries += 1
            if step < 1e-15:
                break
        
        if step < 1e-15:
            # Can't make progress, add noise
            noise = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
            step_size = 1.0
            continue
        
        step_size = min(step * 1.5, 1.0 / (beta * max(a.max(), 1e-10)))
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t = t_new
        
        # Reset if stuck
        if stall > 300:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stall / 500)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            else:
                y = a.copy()
            t = 1.0
            step_size = max(step_size * 2, 1.0 / (beta * max(a.max(), 1e-10)))
            stall = 0
            last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best
