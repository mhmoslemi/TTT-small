Uses adaptive LSE-smoothed FISTA with multiple initializations and length exploration.

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w  # strict inequality for proper handling
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*b[k]))) where b = conv(a,a)."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    # Gradient: 2 * conv(w, a) truncated to length n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lse_value(c, beta):
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    return cm + np.log(np.exp(scaled).sum()) / beta

    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf

        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)

def peak_shave_step(a, S, k=32):
    """One step of peak shaving: reduce the top-k peaks."""
    n = len(a)
    c = fft_convolve_self(a)
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above median
    med = np.median(c)
    weights = np.maximum(c[idx] - med, 0.0)

    if wsum < 1e-30:

Actually, one more important consideration: the `project_simplex` function. The standard algorithm for projecting onto the simplex `{x >= 0, sum(x) = S}` is:

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Uses adaptive LSE-smoothed FISTA with backtracking, peak shaving,
multi-length exploration, and simulated annealing restarts.
Key insight: optimal sequences are asymmetric; reduce center convolution peak
by breaking left-right symmetry while keeping other lags controlled.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, beta):
    """Gradient of LSE-smoothed max of full self-convolution."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    zfsize = 1 << (size + n - 1).bit_length()
    W = fft(w, zfsize)
    AF = fft(a, zfsize)
    grad = 2.0 * np.real(ifft(W * np.conj(AF)))[:n]
    return grad, c


def lse_value(c, beta):
    """Compute LSE-smoothed max value."""
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    return cm + np.log(np.exp(scaled).sum()) / beta


def compute_objective(a):
    """Compute 2n*max_b/sum_a^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def run_fista(a_init, S, max_time, rng, beta_init=30.0, beta_end=5000.0,
              use_backtrack=True):
    """
    FISTA on LSE-smoothed max(conv(a,a)).
    Adaptive beta, optional backtracking step size, periodic restarts.
    """
    n = len(a_init)
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_maxb = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stall = 0
    step = 1.0 / (2.0 * beta_init * S * max(a.max(), 1e-10) + 1.0)
    lse_prev = None

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive beta: start low (explore), end high (exploit)
        if progress < 0.3:
            beta = beta_init * (1.0 + progress / 0.3)
        elif progress < 0.7:
            beta = beta_init * (beta_end / beta_init) ** ((progress - 0.3) / 0.4 * 0.5)
        else:
            beta = (beta_init * (beta_end / beta_init) ** 0.5) * \
                   (beta_end / (beta_init * (beta_end / beta_init) ** 0.5)) ** \
                   ((progress - 0.7) / 0.3)
        beta = min(beta, 50000.0)

        # Compute gradient at y
        grad, c_y = lse_gradient(y, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Tak
        if use_backtrack:
            # Backtracking line search on LSE value
            if lse_prev is not None:
                lse_cur = lse_value(c_y, beta)
                if lse_cur > lse_prev + 0.01 * step * gn * gn:
                    step *= 0.5
                elif step < 10.0 / (2.0 * beta * S * max(y[y > 0].max() if (y > 0).any() else 1.0, 1e-10) + 1.0):
                    step *= 1.1
        else:
            step = 1.0 / (2.0 * beta * S * max(np.max(np.abs(y)), 1e-10) + 1.0)

        lse_prev = lse_value(c_y, beta) if use_backtrack else None

        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        a = x_new
        t_k = t_new

        # Track best using actual max (not LSE)
        if it % 10 == 0 or it <= 20:
            c_a = fft_convolve_self(a)
            if c_a.max() < best_maxb:
                best_maxb = c_a.max()
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Restart with noise if stalled
        if stall > 300 and it % 50 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * max(0.1, 1.0 - progress)
            y = a + rng.normal(0, noise_scale, n)
            y = project_simplex(y, S)
            t_k = 1.0
            stall = 0
            step = 1.0 / (2.0 * beta * S * max(a.max(), 1e-10) + 1.0)

        # Hard restart every 5000 iters
        if it % 5000 == 0:
            noise_scale = 0.01 * S / np.sqrt(n)
            best_a = best_a + rng.normal(0, noise_scale, n)
            best_a = project_simplex(best_a, S)

        if time.monotonic() - start > max_time:
            break

    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best_maxb:
        best_maxb = c_final.max()
        best_a = a.copy()

    return best_a, best_maxb


def peak_shave_iterate(a, S, max_time, rng, k_schedule=None):
    """
    Iterative exact subgradient peak shaving on top-k lags.
    """
    n = len(a)
    a = project_simplex(np.asarray(a, dtype=np.float64), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        # Increase k over time (start with few peaks, end with more)
        if k_schedule is None:
            k = int(4 + 60 * progress)
        else:
            k = int(k_schedule[0] + (k_schedule[1] - k_schedule[0]) * progress)
        k = min(k, 2 * n - 1)

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Get top-k lags
        idx = np.argsort(c)[::-1][:k]
        median_c = np.median(c)
        weights = np.maximum(c[idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum

        w_full = np.zeros(len(c))
        w_full[idx] = weights
        zfsize = 1 << (len(c) + n - 1).bit_length()
        W = fft(w_full, zfsize)
        A = fft(a, zfsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Decreasing step
        step = S / (0.3 * n) * (1.0 - 0.5 * progress)
        a = project_simplex(a - step * grad, S)

        # Occasional noise
        if it % 1000 == 0 and rng.random() < 0.3:
            noise = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best


def excess_gradient_and_step(a, S, max_time, rng):

    best = fft_convolve_self(a).max()
    best_a = a.copy()

    c0 = fft_convolve_self(a)
    T_start = c0.max() * 0.95
    T_end = c0.mean() + 0.05 * (c0.max() - c0.mean())

    y = a.copy()
    t_k = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.6)
        T = max(T, T_end)

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()

        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            T = c.max() - 0.005 * max(c.max() - c.mean(), 1e-10)
            excess = np.maximum(c - T, 0.0)

        if excess.max() < 1e-12:
            break

        # Quadratic excess gradient
        w = 2.0 * excess
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w = w / wsum

        w_full = w  # already full length
        zfsize = 1 << (len(c) + n - 1).bit_length()
        W = fft(w_full, zfsize)
        A = fft(y, zfsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (0.5 * S * max(np.max(np.abs(y)), 1e-10) * 2.0 + 1.0)
        step = max(step, 1.0 / (4.0 * S * n))
        step = min(step, 2.0 / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        a = x_new
        t_k = t_new

        # Noise injection
        if it % 2000 == 0 and rng.random() < 0.2:
            noise = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
                t_k = 1.0

    return best_a, best


def make_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None:
        if len(base_seq) == n:
            inits.append(base_seq.copy())
        elif len(base_seq) > n:
            # Block average downsample
            ratio = len(base_seq) // n
            if ratio > 0:
                small = base_seq[:n * ratio].reshape(n, ratio).mean(axis=1)
                inits.append(small)
        else:
            t_old = np.linspace(0, 1, len(base_seq))
            t_new = np.linspace(0, 1, n)
            inits.append(np.interp(t_new, t_old, base_seq))

    # Anti-symmetric sine perturbations (key insight: breaks center peak)
    for freq in [1, 2, 3]:
        for amp in [0.4, 0.6, 0.8]:
            a_s = 1.0 + amp * np.sin(2 * np.pi * freq * tt + 0.3)
            inits.append(np.maximum(a_s, 0.01))

    # Cosine modulation
    for freq in [1, 2]:
        a_c = 1.0 + 0.5 * np.cos(2 * np.pi * freq * tt + 0.5)
        inits.append(np.maximum(a_c, 0.01))

    # Gaussian-bump asymmetry
    center = 0.35
    a_g = np.exp(-4.0 * (tt - center) ** 2) + 0.3 * np.exp(-4.0 * (tt - 0.7) ** 2)
    inits.append=a_g if hasattr(inits, 'append') else None
    inits.append(np.maximum(a_g, 0.01))

    # Random
    inits.append(0.3 + 0.7 * rng.random(n))
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    # Power-law decay (asymmetric)
    inits.append((tt + 0.1) ** 0.5)
    inits.append((1.0 - tt + 0.1) ** 0.5)

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2380):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA exploration -> excess annealing -> peak shaving -> polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    print(f"Starting search, budget={budget_s}s, seed={seed}")

    # Get initial sequence if available
    base_seq = None
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        base_seq = np.clip(base_seq, 0, 1000)
        if len(base_seq) >= 64 and base_seq.sum() > 0:
            print(f"Using given sequence: len={len(base_seq)}, obj={compute_objective(base_seq):.6f}")
        else:
            base_seq = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Determine lengths to try
    lengths = [8192]
    if base_seq is not None and len(base_seq) not in lengths:
        lengths.insert(0, len(base_seq))
    # Add a larger length for exploration
    if 16384 not in lengths:
        lengths.append(16384)

    total_time = budget_s - 60  # reserve 60s for final polish
    # Allocate time: 50% to primary, 30% to secondary, 20% to explore
    n_lengths = len(lengths)
    time_allocs = []
    remaining_t = total_time
    for i, n in enumerate(lengths):
        frac = 0.5 / (i + 1) if i < 2 else 0.15
        ta = min(frac * budget_s, remaining_t - 60)
        ta = max(ta, 30)
        time_allocs.append(ta)
        remaining_t -= ta

    for li, n in enumerate(lengths):
        t_left = deadline - time.monotonic()
        if t_left < 60:
            break
        t_alloc = min(time_allocs[li], t_left - 60)
        if t_alloc < 30:
            continue

        S = np.sqrt(2.0 * n)
        print(f"\n{'='*50}")
        print(f"Length n={n}, S={S:.4f}, time_alloc={t_alloc:.0f}s")
        print(f"{'='*50}")

        # Prepare inits
        inits = make_inits(n, rng, base_seq)
        # Limit number of inits based on time
        max_inits = max(2, int(t_alloc / 40))
        inits = inits[:max_inits]
        t_per_init = t_alloc * 0.6 / len(inits)
        t_per_init = max(t_per_init, 10)

        best_n_a = None
        best_n_maxb = np.inf

        # Phase 1: FISTA exploration with diverse inits
        for i, init in enumerate(inits):
            t_rem = deadline - time.monotonic()
            if t_rem < 90:
                break
            t_i = min(t_per_init, t_rem - 60)
            if t_i < 5:
                continue
            n_i = len(init)
            S_i = np.sqrt(2.0 * n_i)

            a_opt, maxb = run_fista(init, S_i, t_i, rng,
                                     beta_init=20.0, beta_end=5000.0)
            a_opt = project_simplex(a_opt, S_i)
            obj = compute_objective(a_opt)
            actual_maxb = fft_convolve_self(a_opt).max()

            if actual_maxb < best_n_maxb:
                best_n_maxb = actual_maxb
                best_n_a = a_opt.copy()
                print(f"  FISTA init[{i}] (n={n_i}): obj={obj:.6f}, maxb={actual_maxb:.6f}")

        if best_n_a is None:
            best_n_a = np.ones(n) * S / n
            best_n_maxb = fft_convolve_self(best_n_a).max()

        # Phase 2: Excess annealing on best
        t_rem = deadline - time.monotonic()
        if t_rem > 90:
            t_excess = min(t_alloc * 0.2, t_rem - 60)
            if t_excess > 10:
                n_best = len(best_n_a)
                S_best = np.sqrt(2.0 * n_best)
                a_e, maxb_e = excess_gradient_and_step(best_n_a, S_best, t_excess, rng)
                a_e = project_simplex(a_e, S_best)
                obj_e = compute_objective(a_e)
                actual_e = fft_convolve_self(a_e).max()
                if actual_e < best_n_maxb:
                    best_n_maxb = actual_e
                    best_n_a = a_e.copy()
                    print(f"  After excess annealing: obj={obj_e:.6f}, maxb={actual_e:.6f}")

        # Phase 3: Peak shaving
        t_rem = deadline - time.monotonic()
        if t_rem > 60:
            t_ps = min(t_alloc * 0.1, t_rem - 60)
            if t_ps > 10:
                n_best = len(best_n_a)
                S_best = np.sqrt(2.0 * n_best)
                a_p, maxb_p = peak_shave_iterate(best_n_a, S_best, t_ps, rng)
                a_p = project_simplex(a_p, S_best)
                obj_p = compute_objective(a_p)
                actual_p = fft_convolve_self(a_p).max()
                if actual_p < best_n_maxb:
                    best_n_maxb = actual_p
                    best_n_a = a_p.copy()
                    print(f"  After peak shaving: obj={obj_p:.6f}, maxb={actual_p:.6f}")

        # Update global best
        n_best = len(best_n_a)
        obj_n = compute_objective(best_n_a)
        if obj_n < global_best:
            global_best = obj_n
            global_a = best_n_a.copy()
            print(f"  *** New global best: obj={global_best:.6f} (n={n_best}) ***")

    # Final intensive polishing
    if global_a is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            print(f"\n{'='*50}")
            print(f"Final polishing: n={n_f}, time={t_rem:.0f}s")
            print(f"{'='*50}")

            # High-beta FISTA polish
            t_pol1 = t_rem * 0.5
            if t_pol1 > 10:
                a_f1, _ = run_fista(global_a, S_f, t_pol1, rng,
                                     beta_init=500.0, beta_end=30000.0)
                a_f1 = project_simplex(a_f1, S_f)
                obj_f1 = compute_objective(a_f1)
                if obj_f1 < global_best:
                    global_best = obj_f1
                    global_a = a_f1.copy()
                    print(f"  After high-beta FISTA: obj={global_best:.6f}")

            # Peak shave polish
            t_rem2 = deadline - time.monotonic()
            if t_rem2 > 15:
                t_pol2 = t_rem2 * 0.6
                a_f2, _ = peak_shave_iterate(global_a, S_f, t_pol2, rng)
                a_f2 = project_simplex(a_f2, S_f)
                obj_f2 = compute_objective(a_f2)
                if obj_f2 < global_best:
                    global_best = obj_f2
                    global_a = a_f2.copy()
                    print(f"  After final peak shave: obj={global_best:.6f}")

    # Final verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
    else:
        global_a = np.ones(8192)
        global_a[:] = 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\nFinal sequence length: {len(final_list)}")
    print(f"Verified objective: {ver:.6f}")

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(8192)
        final_list = [1.0] * 8192
        ver = evaluate_sequence(final_list)
        print(f"Fallback objective: {ver:.6f}")

    return final_list
