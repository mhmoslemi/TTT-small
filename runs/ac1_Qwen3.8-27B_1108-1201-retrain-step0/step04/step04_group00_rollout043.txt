def project_simplex(v, S):
    n=len(v)
    if n==0: return v
    v=np.asarray(v,float)
    w=np.sort(v)[::-1]
    css=np.cumsum(w)
    rho=np.nonzero(w * np.arange(1,n+1) > (css - S))[0]
    if len(rho)==0: return np.zeros(n)

    # adjust for floating error

    if s>0: u *= S/s
    return u

def resize_sequence(a, m):
    n=len(a)
    if n==m: return a.copy()
    t_old=np.linspace(0,1,n,endpoint=False)
    t_new=np.linspace(0,1,m,endpoint=False)
    return np.interp(t_new,t_old,a)

"""
Minimize evaluate_sequence for nonnegative step-function heights.

The objective is scale-invariant.  Internally all candidates are normalized to
sum sqrt(2n), for which the objective equals the maximum of the full linear
self-convolution.  The search uses:
  * diverse smooth/heuristic seeds,
  * accelerated projected p-norm/LSE-smoothed descent,
  * adaptive restarts,
  * upsampling to a larger length,
  * final high-smoothness polishing.

All heavy local objectives are computed with FFTs; the authoritative
evaluate_sequence is called only on promising candidates and with a time budget.
"""

import time
import math

import numpy as np
from scipy.fft import fft, ifft


def next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def get_conv_size(n):
    return next_power_of_two(max(1, 2 * int(n) - 1))


def get_fft_size(n):
    # Enough for the correlation/gradient involving weights of length 2n-1.
    return next_power_of_two(max(1, 3 * int(n) - 2))


def normalize_to_sum(a, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return None
    with np.errstate(invalid="ignore", over="ignore"):
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = float(np.sum(a))
    if not math.isfinite(s) or s <= 0.0:
        return None
    return a * (S / s)


def resize_sequence(a, m):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    m = int(m)
    if n == m:
        return a.copy()
    if n == 0:
        return np.zeros(m, dtype=np.float64)
    t_old = np.linspace(0.0, 1.0, n, endpoint=False)
    t_new = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(t_new, t_old, a)


def convolve_self(a, fsize=None):
    n = len(a)
    size = 2 * n - 1
    if fsize is None:
        fsize = get_conv_size(n)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return c, A


def local_max_norm(a):
    """Assume a.sum() is already sqrt(2n). Returns max full convolution."""
    try:
        c, _ = convolve_self(a, get_conv_size(len(a)))
        return float(np.max(np.maximum(c, 0.0)))
    except Exception:
        return np.inf


def weighted_gradient(a, w, A, fsize):
    """
    Gradient of sum_k w_k c_k for symmetrically symmetrized weights.
    The returned vector is 2 * correlation(w_sym, a) truncated to len(a).
    """
    n = len(a)
    ws = 0.5 * (w + w[::-1])
    W = fft(ws, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g


def project_simplex(v, S):
    n = v.size
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([S if S > 0 else 0.0], dtype=np.float64)

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    css = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)

    valid = w * idx > (css - S)
    nz = np.nonzero(valid)[0]
    if nz.size == 0:
        u = np.full(n, S / float(n), dtype=np.float64)
    else:
        rho = int(nz[0])
        theta = (css[rho] - S) / float(rho + 1)
        u = np.maximum(v - theta, 0.0)

    s = float(np.sum(u))
    if s > 0.0 and math.isfinite(s):
        u *= (S / s)
    else:
        u = np.full(n, S / float(n), dtype=np.float64)
    return u


def pnorm_weights(c, p, uniform_mix=0.0):
    u = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    m = u.size
    if m == 0:
        return np.empty(0, dtype=np.float64)

    cm = float(np.max(u))
    if not math.isfinite(cm) or cm <= 0.0:
        w = np.ones(m, dtype=np.float64) / m
    else:
        scaled = u / cm
        if p <= 1.0:
            w = np.ones(m, dtype=np.float64) / m
        else:
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                w = np.power(scaled, p - 1.0)
            wsum = float(np.sum(w))
            if not math.isfinite(wsum) or wsum <= 0.0:
                w = np.zeros(m, dtype=np.float64)
                w[int(np.argmax(u))] = 1.0
            else:
                w /= wsum

    if uniform_mix > 0.0:
        mix = min(1.0, float(uniform_mix))
        w = (1.0 - mix) * w + (mix / float(m))

    s = float(np.sum(w))
    if not math.isfinite(s) or s <= 0.0:
        w = np.ones(m, dtype=np.float64) / m
    else:
        w /= s
    return w


def lse_weights(c, beta, uniform_mix=0.0):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return np.empty(0, dtype=np.float64)

    cmax = float(np.max(c))
    if not math.isfinite(cmax):
        w = np.ones(m, dtype=np.float64) / m
    else:
        shifted = beta * (c - cmax)
        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            e = np.exp(np.clip(shifted, -700.0, 0.0))
        tot = float(np.sum(e))
        if not math.isfinite(tot) or tot <= 0.0:
            w = np.zeros(m, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / tot

    if uniform_mix > 0.0:
        mix = min(1.0, float(uniform_mix))
        w = (1.0 - mix) * w + (mix / float(m))

    s = float(np.sum(w))
    if not math.isfinite(s) or s <= 0.0:
        w = np.ones(m, dtype=np.float64) / m
    else:
        w /= s
    return w


def anneal_value(init, rate, it, mx):
    init = float(init)
    mx = float(mx)
    if init >= mx:
        return mx
    if rate <= 0.0:
        return init
    arg = float(rate) * int(it)
    if arg > 50.0:
        return mx
    val = init * math.exp(arg)
    return min(mx, val)


def run_opt(a_init,
            budget,
            deadline,
            rng,
            mode="pnorm",
            p_init=4.0,
            p_rate=0.002,
            p_max=128.0,
            beta_init=10.0,
            beta_rate=0.003,
            beta_max=500.0,
            step_scale=0.4,
            uniform_mix=0.0,
            noise_base=0.001):
    """
    Accelerated projected descent for a smoothed sum/power objective over the
    nonnegativity simplex.  Returns (best sequence, best local max, final y).
    """
    n = len(a_init)
    S = math.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    if a is None:
        return a_init, np.inf, a_init

    fsize = get_fft_size(n)

    y = a.copy()
    x = a.copy()
    t = 1.0

    c, A = convolve_self(y, fsize)
    best_val = float(np.max(np.maximum(c, 0.0)))
    best_a = y.copy()

    it = 0
    stuck = 0
    last = 0
    eta_mult = 1.0

    start = time.monotonic()
    end = min(start + float(budget), float(deadline))

    while time.monotonic() < end:
        it += 1

        c, A = convolve_self(y, fsize)
        val = float(np.max(np.maximum(c, 0.0)))

        if val < best_val - 1e-14:
            best_val = val
            best_a = y.copy()
            stuck = 0
            last = it
            if eta_mult < 1.5:
                eta_mult = min(1.5, eta_mult * 1.05)
        else:
            stuck = it - last
            if stuck > 500:
                eta_mult = max(0.3, eta_mult * 0.99)

        if mode == "pnorm":
            param = anneal_value(p_init, p_rate, it, p_max)
            w = pnorm_weights(c, param, uniform_mix)
            denom = max(param, 1.0)
        else:
            param = anneal_value(beta_init, beta_rate, it, beta_max)
            w = lse_weights(c, param, uniform_mix)
            denom = max(param, 1.0)

        grad = weighted_gradient(y, w, A, fsize)
        gmax = float(np.max(np.abs(grad)))
        if not math.isfinite(gmax) or gmax < 1e-16:
            break

        step = (float(step_scale) * eta_mult) / denom
        if step * gmax > 5.0:
            step = 5.0 / max(gmax, 1e-30)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))
        if t_new > 100.0:
            t_new = 100.0

        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        np.maximum(y, 0.0, out=y)
        sy = float(np.sum(y))
        if math.isfinite(sy) and sy > 0.0:
            y *= (S / sy)
        else:
            y = x_new.copy()

        x = x_new
        t = t_new

        if stuck > 2000 and (it % 100 == 0):
            ns = float(noise_base) * S / math.sqrt(float(n)) * math.exp(-stuck / 5000.0)
            if ns < 1e-9:
                ns = 1e-9
            y = project_simplex(y + rng.normal(0.0, ns, n), S)
            x = y.copy()
            t = 1.0
            stuck = 0
            last = it
            eta_mult = 1.0

    if time.monotonic() < float(deadline) - 1.0:
        try:
            c, _ = convolve_self(y, fsize)
            val = float(np.max(np.maximum(c, 0.0)))
            if val < best_val:
                best_val = val
                best_a = y.copy()
        except Exception:
            pass

    return best_a, best_val, y.copy()

def generate_candidates(n, rng, parent):
    cands = []
    n = int(n)

    tt = np.linspace(0.0, 1.0, n, endpoint=False)

    # Uniform.
    cands.append(np.ones(n, dtype=np.float64))

    # Smooth periodic profiles.
    for r in (0.15, 0.35, 0.55, 0.75, 0.92):
        for phi in (0.0, math.pi / 6.0, math.pi / 4.0, math.pi / 3.0,
                    math.pi / 2.0, 2.0 * math.pi / 3.0, 5.0 * math.pi / 6.0):
            for p in (0.5, 1.0, 1.5, 2.0, 3.0):
                base = 1.0 + r * np.cos(2.0 * np.pi * tt + phi)
                base = np.maximum(base, 1e-6)
                with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                    a = np.power(base, p)
                if np.all(np.isfinite(a)):
                    cands.append(a)

    # Centered/border Gaussian-like profiles.
    for c in (0.2, 0.35, 0.5, 0.65, 0.8):
        for w in (0.04, 0.08, 0.15):
            a = 0.04 + np.exp(-((tt - c) ** 2) / (2.0 * w * w))
            cands.append(a)

    # Random smooth Fourier profiles.
    for K in (2, 4, 8):
        for _ in range(2):
            x = np.zeros(n, dtype=np.float64)
            amp = 0.35 * rng.random(K)
            phase = 2.0 * math.pi * rng.random(K)
            for k in range(1, K + 1):
                x += amp[k - 1] * np.cos(2.0 * math.pi * k * tt + phase[k - 1])
            a = np.exp(np.clip(x, -30.0, 30.0))
            cands.append(a)

    # Simple random positive profiles.
    cands.append(0.05 + rng.random(n))
    cands.append(0.05 + rng.exponential(1.0, n))
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        par = 0.05 + rng.pareto(1.1, n)
    par = np.minimum(np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0), 10.0)
    cands.append(par)

    # Tapers.
    tri = np.maximum(0.0, 1.0 - np.abs(2.0 * tt - 1.0))
    for base_mix in (0.0, 0.2, 0.5, 0.8):
        cands.append(base_mix + (1.0 - base_mix) * tri)

    sinpow = np.sin(math.pi * tt)
    for p in (0.5, 1.0, 2.0, 4.0):
        cands.append(np.power(np.maximum(sinpow, 1e-8), p))

    # Parent-transformed seeds, if available.
    if parent is not None:
        par = normalize_to_sum(parent, 1.0)
        if par is not None:
            resized = resize_sequence(par, n)
            cands.append(resized)
            cands.append(resized[::-1])
            for shift in (1, max(1, n // 4), max(1, n // 3)):
                cands.append(np.roll(resized, shift))
            cands.append(resized + 0.01 * rng.normal(0.0, 1.0, n))
            cands.append(resized + 0.03 * rng.normal(0.0, 1.0, n))

    return cands

def choose_best(cands, parent, n, deadline, max_actual=12, top_count=8):
    best_a = None
    best_score = np.inf
    eval_seen = False

    S = math.sqrt(2.0 * float(n))
    scored = []

    for idx, raw in enumerate(cands):
        a = normalize_to_sum(raw, S)
        if a is None:
            continue
        val = local_max_norm(a)
        if math.isfinite(val):
            scored.append((val, idx, a))

    scored.sort(key=item0)
    top = scored[:max(1, int(top_count))]

    # Parent at its original length, if valid.
    if parent is not None:
        a_par = normalize_to_sum(parent, math.sqrt(2.0 * float(len(parent))))
        if a_par is not None:
            sc = eval_actual_budgeted(a_par, deadline)
            if np.isfinite(sc):
                eval_seen = True
                if sc < best_score:
                    best_score = sc
                    best_a = a_par.copy()
                    eval_seen = True

    for val, idx, a in top:
        if time.monotonic() > float(deadline) - 30.0:
            break
        sc = eval_actual_budgeted(a, deadline)
        if np.isfinite(sc):
            eval_seen = True
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        if top:
            best_a = top[0][2].copy()
            best_score = top[0][0]
        else:
            best_a = np.ones(n, dtype=np.float64)
            best_a *= S / float(n)
            best_score = local_max_norm(best_a)

    return best_a, best_score, top, eval_seen


def improve_with_actual(cand, val, best_a, best_score, deadline, eval_seen):
    """
    Update best from a candidate.  If the authoritative evaluator is available,
    trust it; otherwise fall back to the local objective.
    """
    sc = eval_actual_budgeted(cand, deadline)
    if np.isfinite(sc):
        if sc < best_score:
            return cand.copy(), sc, True
        return best_a, best_score, True

    if not eval_seen and np.isfinite(val) and val < best_score:
        return cand.copy(), val, eval_seen

    return best_a, best_score, eval_seen

            if len(best_a) == n0:
                seeds[0] = best_a.copy()

            print("Main config %d mode=%s score=%s" % (ci, mode, best_score), flush=True)

    # Upsample to 16384 when beneficial and time remains.
    if len(best_a) < 16384 and time.monotonic() < deadline - 200.0:
        n1 = 16384
        up_budget = (deadline - time.monotonic()) * 0.30
        S1 = math.sqrt(2.0 * float(n1))

        seed1 = normalize_to_sum(resize_sequence(best_a, n1), S1)
        if seed1 is None:
            seed1 = np.ones(n1, dtype=np.float64)
            seed1 *= S1 / float(n1)

        up_configs = [
            ("pnorm", {
                "frac": 0.30,
                "p_init": 2.0, "p_rate": 0.0012, "p_max": 32.0,
                "step_scale": 0.40, "uniform_mix": 0.05, "noise_base": 0.0015,
            }),
            ("pnorm", {
                "frac": 0.35,
                "p_init": 6.0, "p_rate": 0.0018, "p_max": 96.0,
                "step_scale": 0.35, "uniform_mix": 0.02, "noise_base": 0.0010,
            }),
            ("lse", {
                "frac": 0.35,
                "beta_init": 8.0, "beta_rate": 0.003, "beta_max": 400.0,
                "step_scale": 0.30, "uniform_mix": 0.02, "noise_base": 0.0010,
            }),
        ]

        print("Upsampling to n=%d" % n1, flush=True)

        for mode, cfg in up_configs:
            if time.monotonic() > deadline - 60.0:
                break
            frac = float(cfg.get("frac", 1.0 / len(up_configs)))
            budget = up_budget * frac
            budget = min(budget, max(0.0, (deadline - time.monotonic()) - 35.0))
            if budget <= 12.0:
                continue

            params = dict(cfg)
            params.pop("frac", None)

            a_opt, val_opt, _ = run_opt(seed1, budget, deadline, rng, mode=mode, **params)
            best_a, best_score, eval_seen = improve_with_actual(
                a_opt, val_opt, best_a, best_score, deadline, eval_seen
            )

            if len(a_opt) == n1:
                # Continue the upsized run from its local best if it looks good.
                seed1 = a_opt.copy()

            print("Upsized %s score=%s" % (mode, best_score), flush=True)

    # Final intensive polishing on the best sequence currently known.
    if time.monotonic() < deadline - 70.0:
        final_budget = (deadline - time.monotonic()) * 0.80

        a1, v1, _ = run_opt(
            best_a, final_budget * 0.55, deadline, rng,
            mode="pnorm", p_init=16.0, p_rate=0.0018, p_max=256.0,
            step_scale=0.30, uniform_mix=0.0, noise_base=0.0005
        )
        best_a, best_score, eval_seen = improve_with_actual(a1, v1, best_a, best_score, deadline, eval_seen)
        print("Final pnorm score=%s" % best_score, flush=True)

        if time.monotonic() < deadline - 40.0:
            a2, v2, _ = run_opt(
                best_a, (deadline - time.monotonic()) * 0.70, deadline, rng,
                mode="lse", beta_init=40.0, beta_rate=0.0018, beta_max=1500.0,
                step_scale=0.25, uniform_mix=0.0, noise_base=0.0004
            )
            best_a, best_score, eval_seen = improve_with_actual(a2, v2, best_a, best_score, deadline, eval_seen)
            print("Final LSE score=%s" % best_score, flush=True)

    # Fallback to a valid sequence if something went wrong.
    if best_a is None:
        best_a = np.ones(n0, dtype=np.float64)
        best_a *= math.sqrt(2.0 * float(n0)) / float(n0)

    best_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    best_a = np.clip(best_a, 0.0, 1000.0)

    # If all authoritative evaluations failed, normalize to guarantee sum > 0.01.
    if best_a.size == 0:
        best_a = np.ones(8192, dtype=np.float64)

    s = float(np.sum(best_a))
    if not math.isfinite(s) or s < 0.01:
        n_fb = 8192
        S_fb = math.sqrt(2.0 * n_fb)
        best_a = np.full(n_fb, S_fb / float(n_fb), dtype=np.float64)

    return best_a.tolist()
