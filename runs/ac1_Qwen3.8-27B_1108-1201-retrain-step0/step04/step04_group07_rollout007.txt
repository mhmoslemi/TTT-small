Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive p-norm smoothing with
FFT-based gradient computation, multi-length search, and subgradient polishing.

Key ideas:
1. Use sum(c^p) as smooth approximation to max(c), with annealed p.
2. Compute gradients via FFT (convolution structure).
3. Projected gradient with adaptive step size and backtracking.
4. Multi-length search (n=4096 to 32768).
5. Nesterov-accelerated subgradient for final polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == S:
        return np.maximum(v, 0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_truncated(a, b):
    """Compute convolve(a, b) truncated to length len(a) (first n elements)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:n]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def pnorm_smooth_step(a, p, lr, n_backtrack=10):
    """
    One step of projected gradient descent on sum(c^p) approximation.
    Uses backtracking line search.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Clip c to avoid numerical issues
    c_safe = np.maximum(c, 0.0)
    
    # Weights for gradient
    if p > 1:
        weights = c_safe ** (p - 1)
    else:
        weights = np.ones_like(c_safe)
    
    # Gradient: 2p * convolve(weights, a) truncated to [0, n-1]
    grad = 2.0 * p * fft_convolve_truncated(a, weights)
    
    # Backtracking line search
    a_new = a
    f_new = None
    for bt in range(n_backtrack):
        step = lr * (0.5 ** bt)
        a_trial = project_simplex(a - step * grad, S)
        c_trial = fft_convolve_self(a_trial)
        f_trial = np.sum(np.maximum(c_trial, 0.0) ** p)
        f_cur = np.sum(c_safe ** p)
        if f_trial < f_cur * (1 - 1e-4 * step):
            a_new = a_trial
            f_new = f_trial
            break
        a_new = a_trial
    else:
        # If all backtracks fail, just use the smallest step
        a_new = project_simplex(a - lr * (0.5 ** (n_backtrack-1)) * grad, S)
    
    # Renormalize
    s = a_new.sum()
    if s < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * (S / s)
    
    return a_new


def pnorm_optimize(a_init, p_start, p_end, n_iter, lr_start, lr_end, rng, target_score=1.50):
    """
    Optimize using p-norm smoothing with annealed p and learning rate.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        # Anneal p: start moderate, end high (sharper approximation to max)
        p = p_start + (p_end - p_start) * progress
        # Anneal learning rate
        lr = lr_start + (lr_end - lr_start) * progress
        
        a_new = pnorm_smooth_step(a, p, lr)
        sc = compute_score(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if best_score <= target_score:
            break
        
        # Periodic perturbation to escape local minima
        if it % 200 == 100 and rng.random() < 0.3:
            perturbation = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + perturbation, S)
    
    return best_a, best_score

def spectral_refine(a, N, n_iter, alpha, rng, target_score=1.50):

    Spectral flattening: push |FFT(a)|^2 toward uniform.

        P = np.abs(A) ** 2
        
        p_median = np.median(P)
        if p_median < 1e-30:
            break
        
        ratio = P / p_median
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * (1.0 - alpha * (1.0 - np.power(ratio[above], -alpha)))
        P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - np.power(ratio[~above], alpha)))
        P_new = np.maximum(P_new, 0.0)
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new_full, S)
        a_new = np.maximum(a_new, 0.0)

            a_new = a.copy()

            a_new = a_new * (S / s)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if best_score <= target_score:
            break
    
    return best_a, best_score

def fft_convolve_full(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_trunc(a, w):
    """Compute first len(a) elements of convolve(a, w) via FFT."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    c = np.real(ifft(A * W))
    return c[:n]

    """Compute the objective 2*n*max(c)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_full(a)
    return float(2.0 * n * c.max() / (S * S))


def pnorm_gradient(a, p, fsize):
    """
    Compute gradient of sum(c^p) where c = convolve(a,a).
    grad[i] = 2*p * sum_k c[k]^(p-1) * a[k-i]  (truncated to [0,n-1])

    Computed via FFT.

    c = fft_convolve_full(a)
    c_safe = np.maximum(c, 1e-30)
    weights = c_safe ** (p - 1)
    grad = 2.0 * p * fft_convolve_trunc(a, weights)
    return grad, c


def spectral_refine(a, n_iter, alpha, target_score=1.50):
    """
    Spectral flattening: iteratively push |FFT(a)|^2 toward uniform.
    Returns (best_a, best_score).
    """
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2

        p_med = np.median(P)
        if p_med < 1e-30:
            break

        ratio = P / p_med
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * (1.0 - alpha * (1.0 - ratio[above] ** (-alpha)))
        P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - ratio[~above] ** alpha))
        P_new = np.maximum(P_new, 0.0)

            best_a = a_new.copy()
        a = a_new

        if best_score <= target_score:

            c_trial = fft_convolve_full(x_trial)
            f_trial = np.sum(np.maximum(c_trial, 1e-30) ** p)
            if f_trial < f_y * (1.0 - 1e-4):
                x_new = x_trial
                break
            step *= 0.3
        else:
            x_new = project_simplex(y - step * grad, S)
            s = x_new.sum()
            if s < 1e-15:
                x_new = x.copy()
            else:
                x_new = x_new * (S / s)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new
        t = t_new

        # Track best
        sc = compute_score(x)
        if sc < best_score:
            best_score = sc
            best_a = x.copy()

        if best_score <= target_score:
            break

        # Periodic perturbation
        if it % 300 == 150 and rng.random() < 0.25:
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            y = project_simplex(x + noise, S)
            t = 1.0

    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k=100):
    """
    Nesterov-accelerated subgradient for max(c) with exponential weighting.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_full(y)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()

        # Exponentially weighted subgradient
        k_actual = min(top_k, len(c))
        idx = np.argsort(c)[::-1][:k_actual]
        vals = c[idx]
        v_max = vals[0]
        if v_max > 1e-30:
            weights = np.exp(15.0 * (vals - v_max) / v_max)
            weights /= weights.sum()
        else:
            weights = np.ones(k_actual) / k_actual

        w_full = np.zeros(len(c))
        for i in range(k_actual):
            w_full[int(idx[i])] = weights[i]

        W = fft(w_full, fsize)
        Ay = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Ay)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.5 * S / (np.sqrt(n) * (1.0 + 0.003 * it) * gnorm)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 400 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0

        if best_score <= 1.50:
            break

    return best_local, best_score


def make_inits(n, rng):
    """Generate diverse initializations for length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.25 * np.sin(2 * np.pi * tt))
    inits.append(0.5 + 0.15 * np.sin(2 * np.pi * 3 * tt) +
                0.1 * np.sin(2 * np.pi * 7 * tt) +
                0.08 * np.sin(2 * np.pi * 13 * tt))
    inits.append(0.5 + 0.15 * np.sin(2 * np.pi * (4 * tt + 2 * tt * tt)))
    inits.append(0.5 + 0.1 * np.sin(2 * np.pi * 5 * tt) +
                0.1 * np.sin(2 * np.pi * 11 * tt) +
                0.08 * np.sin(2 * np.pi * 17 * tt) +
                0.05 * np.sin(2 * np.pi * 23 * tt))

    # Smooth random
    noise = rng.standard_normal(n)
    ks = max(3, n // 32)
    kern = np.ones(ks) / ks
    sm = np.convolve(noise, kern, mode='same')
    sm = (sm - sm.min()) / max(sm.max() - sm.min(), 1e-10)
    inits.append(0.2 + 0.5 * sm)

    # Bump
    inits.append(np.exp(-0.5 * ((tt - 0.5) * 3.0) ** 2) + 0.1)

    # Cosine window
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * tt)) + 0.1)

    # Random binary-like
    b = (rng.random(n) > 0.5).astype(float)
    inits.append(0.3 + 0.4 * b)

    # Multi-scale noise
    n1 = rng.standard_normal(n)
    ks1 = max(3, n // 64)
    k1 = np.ones(ks1) / ks1
    s1 = np.convolve(n1, k1, mode='same')
    n2 = rng.standard_normal(n)
    ks2 = max(3, n // 16)
    k2 = np.ones(ks2) / ks2
    s2 = np.convolve(n2, k2, mode='same')
    combined = s1 + 0.5 * s2
    combined = (combined - combined.min()) / max(combined.max() - combined.min(), 1e-10)
    inits.append(0.15 + 0.6 * combined)

    return inits


def optimize_one_length(n, budget_s, rng, target_score=1.50):
    """Full pipeline for one sequence length. Returns (best_a, best_score)."""
    S = np.sqrt(2.0 * float(n))
    global_best = np.inf
    global_a = None

    def try_update(a):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
        return sc

    inits = make_inits(n, rng)

    # Try provided sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            gc = np.clip(given, 0, 1000)
            if gc.sum() > 0.01:
                inits.insert(0, gc)
                print("  Using provided sequence as init")
    except Exception:
        pass

    t_start = time.monotonic()
    n_init = len(inits)
    per_init = budget_s * 0.8 / n_init

    for ii, arr in enumerate(inits):
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s * 0.8:
            break

        a = np.clip(arr.astype(np.float64), 0.0, 1000.0)
        s0 = a.sum()
        if s0 < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s0)

        rem_slot = min(per_init * 1.5, budget_s * 0.8 - elapsed)
        if rem_slot < 5:
            break

        # Stage 1: Spectral refinement (coarse, fast)
        n_spec = max(100, int(rem_slot * 0.2 / max(0.01, n * 20 * 1e-9)))
        n_spec = min(n_spec, 500)
        a1, sc1 = spectral_refine(a, n_spec, alpha=0.6, target_score=target_score)
        sc = try_update(a1)

        # Stage 2: p-norm Nesterov (main workhorse)
        elapsed = time.monotonic() - t_start
        rem2 = min(per_init * 1.0, budget_s * 0.8 - elapsed)
        if rem2 > 10:
            n_pn = max(500, int(rem2 / max(0.01, n * 30 * 1e-9)))
            n_pn = min(n_pn, 5000)
            a2, sc2 = pnorm_nesterov(None, a1, p_start=4, p_end=60,
                                     n_iter=n_pn, lr_start=0.5, lr_end=0.05,
                                     rng=rng, target_score=target_score)
            sc = try_update(a2)

        # Stage 3: Subgradient polish
        elapsed = time.monotonic() - t_start
        rem3 = min(per_init * 0.5, budget_s * 0.8 - elapsed)
        if rem3 > 3:
            a3, sc3 = subgradient_polish(a2, n, S, rem3, rng,
                                         top_k=min(200, n // 32))
            sc = try_update(a3)

        if (ii + 1) % 3 == 0:
            print("  [n=%d, init %d/%d] best=%.6f (%.0fs elapsed)" %
                  (n, ii + 1, n_init, global_best, time.monotonic() - t_start))

    # Final extended polish on global best
    if global_a is not None:
        elapsed = time.monotonic() - t_start
        rem_final = budget_s - elapsed
        if rem_final > 15:
            # Long subgradient
            t0 = time.monotonic()
            af1, _ = subgradient_polish(global_a, n, S, rem_final * 0.4, rng,
                                        top_k=min(300, n // 20))
            try_update(af1)
            # Long p-norm with high p
            rem_extra = budget_s - (time.monotonic() - t_start)
            if rem_extra > 10:
                n_extra = max(200, int(rem_extra * 0.5 / max(0.01, n * 30 * 1e-9)))
                af2, _ = pnorm_nesterov(None, global_a, p_start=30, p_end=100,
                                        n_iter=min(n_extra, 3000),
                                        lr_start=0.2, lr_end=0.02,
                                        rng=rng, target_score=target_score)
                try_update(af2)

    return global_a, global_best


def sort_by_priority(x):
    """Sort key: prioritize lengths near 16384."""
    return abs(x - 16384)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-length p-norm Nesterov + spectral + subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)
    print("Optimization budget: %.0fs" % budget_s)

    lengths = [4096, 8192, 16384, 32768]

    # Add provided length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 2048 and given.size not in lengths:
            lengths.append(given.size)
    except Exception:
        pass

    lengths = sorted(set(lengths), key=sort_by_priority)

    global_best = np.inf
    global_a = None
    t_global = time.monotonic()

    # Budget allocation: weight favoring 8192-16384
    total_budget = budget_s - 50
    weights = []
    for L in lengths:
        if L <= 4096:
            weights.append(0.6)
        elif L <= 8192:
            weights.append(1.0)
        elif L <= 16384:
            weights.append(1.2)
        else:
            weights.append(0.8)
    wsum = sum(weights)
    budgets = [total_budget * w / wsum for w in weights]

    for i, n in enumerate(lengths):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(budgets[i], remaining - 40)
        if alloc < 30:
            break

        print("=== n=%d, alloc=%.0fs ===" % (n, alloc))
        a_res, score = optimize_one_length(n, alloc, rng, target_score=1.50)

        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  NEW GLOBAL BEST: %.6f (n=%d)" % (global_best, n))
        else:
            print("  n=%d best=%.6f (global=%.6f)" % (n, score, global_best))

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        global_best = 2.0
        print("Fallback uniform, score=%.4f" % global_best)

    # Verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified score: %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate returned inf, fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0

    print("FINAL: n=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
