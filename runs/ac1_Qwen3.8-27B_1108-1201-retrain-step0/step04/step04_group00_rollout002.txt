"""

Improved multi-phase optimization with:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power2_gradient(y, c):
    """Gradient of sum(c^2) w.r.t. a. Since d(sum c^2)/da = 4*(c*a)[0:n]."""
    n = len(y)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(y, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_power2_phase(a_init, max_time, rng):
    """Warm-up phase: minimize sum(c^2) using FISTA."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj2 = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        obj2 = np.sum(c * c)
        if obj2 < best_obj2:
            best_obj2 = obj2
            best_a = y.copy()

        grad = power2_gradient(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: 1/L where L ~ 8*||a||_2^2 / n + 4*max(c)
        step = 0.5 / (8.0 * np.max(a) + 4.0 * c.max() / n + 1.0)
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a


def run_lse_phase(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                  beta_max=10000.0, use_hybrid=False):
    """Main LSE FISTA phase with adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta
        if stuck < 500:
            beta = beta_init * (1.0 + beta_rate * it)
        elif stuck < 3000:
            beta = min(beta_max, beta * 1.01)
        else:
            beta = min(beta_max, beta * 1.03)
        beta = min(beta, beta_max)

        # Gradient
        if use_hybrid and stuck > 2000 and (it % 3 == 0):
            grad = exact_subgradient(y, c, tol_frac=0.01)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = 1.0 / max(beta, 1.0)
        # Conservative bound
        L_bound = 2.0 * beta * np.max(a) + 2.0 * np.max(a)
        step = min(step, 0.8 / max(L_bound, 1.0))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with noise when stuck
        if stuck > 5000:
            t_mom = 1.0
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        # Periodic momentum restart
        if it % 8000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_mirror_descent(a_init, max_time, rng, beta_init=20.0):

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 1000:
            beta = min(10000.0, beta * 1.02)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent step: a_new[j] = a[j] * exp(-step * grad[j])
        # Step size for mirror descent
        step = 1.0 / (beta * gnorm + 1.0)
        step = min(step, 0.1)

        y_new = y * np.exp(-step * grad)
        y_new = normalize_seq(y_new, S)

        # Check improvement (simple step, no momentum)
        c_check = fft_convolve_self(y_new)
        if c_check.max() < cur_max or it % 100 == 0:
            y = y_new
        else:
            # Try smaller step
            y_new = y * np.exp(-step * 0.5 * grad)
            y_new = normalize_seq(y_new, S)
            c_check = fft_convolve_self(y_new)
            if c_check.max() < cur_max:
                y = y_new
            else:
                # Revert, try opposite direction partially
                y = normalize_seq(y + 0.01 * y * (grad / gnorm), S)

        if stuck > 8000 and it % 500 == 0:
            noise = rng.normal(0, 0.01, n)
            y = normalize_seq(y * np.exp(noise * 0.01), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)

    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / n)


def make_random_init(n, rng):

    a = a ** 2  # Create some structure
    return a


def make_square_wave_init(n):

def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

            global_best_obj = obj_md
            global_best_a = a_md.copy()

    # === Phase 4: Active-set polishing ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.30
    if p4_budget > 10:
        print("\n=== Phase 4: Active-set polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a_ap1, _ = run_active_polish(global_best_a, p4_budget * 0.6, rng)
        obj_ap1 = compute_obj(a_ap1)
        print("  Active1: obj=%.6f (t=%.1fs)" % (obj_ap1, time.monotonic() - t4))
        if obj_ap1 < global_best_obj:
            global_best_obj = obj_ap1
            global_best_a = a_ap1.copy()

        rem4 = p4_budget - (time.monotonic() - t4)
        if rem4 > 5:
            # Continue with LSE at very high beta
            a_ap2, _ = run_lse_phase(global_best_a, rem4, rng,
                                      beta_init=200.0, beta_rate=0.002,
                                      beta_max=50000.0, use_hybrid=True)
            obj_ap2 = compute_obj(a_ap2)
            print("  Active2: obj=%.6f (t=%.1fs)" % (obj_ap2, time.monotonic() - t4))
            if obj_ap2 < global_best_obj:
                global_best_obj = obj_ap2
                global_best_a = a_ap2.copy()

    # === Phase 5: Multi-scale (upsample to 2n) ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p5_budget = remaining * 0.70
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_budget))
        t5 = time.monotonic()

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)

        a_up1, _ = run_lse_phase(a_up, p5_budget * 0.5, rng,
                                  beta_init=5.0, beta_rate=0.006, beta_max=5000.0)
        obj_up1 = compute_obj(a_up1)
        print("  Upsampled LSE: obj=%.6f (t=%.1fs)" % (obj_up1, time.monotonic() - t5))
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 10:
            a_up2, _ = run_active_polish(global_best_a, rem5, rng)
            obj_up2 = compute_obj(a_up2)
            print("  Upsampled polish: obj=%.6f (t=%.1fs)" % (obj_up2, time.monotonic() - t5))
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p6_budget = remaining * 0.80
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_budget)
        t6 = time.monotonic()

        a_fin, _ = run_active_polish(global_best_a, p6_budget * 0.5, rng)
        obj_fin = compute_obj(a_fin)
        print("  Final1: obj=%.6f (t=%.1fs)" % (obj_fin, time.monotonic() - t6))
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 5:
            a_fin2, _ = run_lse_phase(global_best_a, rem6, rng,
                                       beta_init=500.0, beta_rate=0.001,
                                       beta_max=100000.0, use_hybrid=True)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final2: obj=%.6f (t=%.1fs)" % (obj_fin2, time.monotonic() - t6))
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - (deadline - budget_s + 30.0)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
