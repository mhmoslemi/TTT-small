"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using Nesterov-accelerated
projected subgradient with logsumexp smoothing, adaptive temperature,
spectral nudging, and multi-length search.

Key improvements:
1. Logsumexp smoothing for smooth optimization in early phases
2. Nesterov acceleration for O(1/T^2) convergence
3. Adaptive temperature annealing for the peak weighting
4. Periodic spectral nudging to escape local minima
5. Multi-length search (4096, 8192, 16384)
6. Proper FFT-based gradient computation
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_autocorr(a):
    """Full autocorrelation (self-convolution) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_autocorr(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_gradient(a, weights_full, n):
    """
    Compute grad[j] = sum_{i=0}^{n-1} a[i] * weights_full[i+j]
    using FFT. weights_full has length 2n-1.
    """
    N = 1 << (3 * n - 2).bit_length()
    A_flip = fft(a[::-1], N)
    W = fft(weights_full, N)
    conv = np.real(ifft(A_flip * W))
    return conv[n - 1:2 * n - 1]


def logsumexp_step(a, beta, top_k, lr, S):
    """
    One Nesterov step using logsumexp-smoothed objective.
    Uses top-k weights for efficiency.
    """
    n = len(a)
    c = fft_autocorr(a)
    
    # Get top-k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    
    # Compute softmax weights with temperature 1/beta
    max_val = top_vals[0]
    if max_val < 1e-30:
        return a, c, 0.0
    exp_vals = np.exp(beta * (top_vals - max_val))
    # Also account for the "tail" (other peaks not in top-k)
    # For simplicity, ignore the tail (top-k should capture most of the mass for large beta)
    weights = exp_vals / exp_vals.sum()
    
    # Build weight vector
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    
    # Compute gradient
    grad = compute_gradient(a, w_full, n)
    
    # The actual gradient of logsumexp is beta * (above), but we absorb beta into lr
    step = lr * grad
    a_new = project_simplex(a - step, S)
    return a_new, c, c.max()


def nesterov_optimize(a0, n, S, budget, rng, top_k=50, target=1.50, beta_start=0.5, beta_end=50.0):
    """
    Nesterov-accelerated projected gradient on smoothed objective.
    """
    a = np.maximum(a0, 0.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    
    # Learning rate schedule
    base_lr = S / (n ** 0.7)
    
    it = 0
    while time.monotonic() - start < budget:
        it += 1
        # Anneal beta
        progress = min(it / (budget * 100), 1.0)  # rough progress estimate
        beta = beta_start + (beta_end - beta_start) * progress ** 0.5
        
        # Ensure y is on the simplex
        y_pos = np.maximum(y, 0.0)
        y_sum = y_pos.sum()
        if y_sum < 1e-15:
            y = a.copy()
            continue
        y_norm = y_pos * (S / y_sum)
        
        # Compute gradient at y_norm
        c = fft_autocorr(y_norm)
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = y_norm.copy()
            if best_score <= target:
                break
        
        # Top-k subgradient with softmax weighting
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        max_val = top_vals[0]
        if max_val < 1e-30:
            break
        
        # Softmax with adaptive temperature
        temp = 1.0 / beta
        exp_vals = np.exp((top_vals - max_val) / temp)
        weights = exp_vals / exp_vals.sum()
        
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        grad = compute_gradient(y_norm, w_full, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        
        # Polyak-type step size
        # f* lower bound ≈ 2n/(2n-1) ≈ 1
        f_star = 1.0
        gap = sc - f_star
        if gap > 1e-10:
            lr = gap / (gnorm * gnorm)
        else:
            lr = base_lr / (1.0 + 0.001 * it)
        lr = min(lr, base_lr)  # cap the step
        
        x_new = project_simplex(y_norm - lr * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - y_norm)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic noise for exploration
        if it % 2000 == 0 and rng.random() < 0.3:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_a, best_score


def spectral_nudge(a, n, S, rng, strength=0.1):
    """
    Apply a spectral nudge: modify the power spectrum to reduce
    autocorrelation peaks, then project back.
    """
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Compute current autocorrelation
    c = np.real(ifft(P))[:2 * n - 1]
    c_mean = c.mean()
    c_error = c - c_mean
    
    # The gradient of max(c) w.r.t. P is approximately the "dual" variable
    # We want to reduce P in directions that reduce c peaks
    # dc/dP[m] = (1/N) * exp(-2πi*m*peak_lag/N) for the peak lag
    # Approximate: reduce P at frequencies that align with the peak
    
    # Simpler approach: flatten P toward uniform
    P_mean = P.mean()
    P_new = P + strength * (P_mean - P) * 0.1  # gentle flattening
    
    # Also add small random phase perturbation to break coherence
    if rng.random() < 0.5:
        phase_pert = rng.normal(0, 0.01, N)
        phase = phase + phase_pert
    
    A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    return a_new


def make_initializations(n, rng):

    inits.append(("random", rng.random(n) + 0.05))

    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth", 0.2 + 0.6 * smooth))
    
    return inits

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_autocorr(a):
    """Full self-convolution via FFT. Returns length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute objective value."""
    n = len(a)
    c = fft_autocorr(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def compute_grad(a, w_full, n):
    """Compute grad[j] = sum_i a[i]*w_full[i+j] via FFT."""
    N = 1 << (3 * n - 2).bit_length()
    A_flip = fft(a[::-1], N)
    W = fft(w_full, N)
    conv = np.real(ifft(A_flip * W))
    return conv[n - 1:2 * n - 1]


def softmax_weights(top_vals, temperature):
    """Compute softmax weights for top-k peak values."""
    max_val = top_vals[0]
    if max_val < 1e-30 or temperature < 1e-15:
        w = np.zeros(len(top_vals))
        w[0] = 1.0
        return w
    exp_vals = np.exp((top_vals - max_val) / temperature)
    s = exp_vals.sum()
    if s < 1e-30:
        w = np.zeros(len(top_vals))
        w[0] = 1.0
        return w
    return exp_vals / s

    temp_start = 0.5

        max_val = top_vals[0]

        w_full = np.zeros(2 * n - 1)

        lr = base_lr

def peak_peel(a, n, S, num_steps=200, budget=60.0):
    """
    Peak peeling: iteratively identify the worst peak and reduce
    the contributing pairs, then renormalize.
    """
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    for step in range(num_steps):
        if time.monotonic() - start > budget:
            break

        c = fft_autocorr(a)
        k_star = int(np.argmax(c))
        c_max = c[k_star]

        # Find contributing pairs for this peak
        # c[k_star] = sum_j a[j] * a[k_star - j] for valid j
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)

        best_prod = 0.0
        best_j = -1
        for j in range(j_start, j_end):
            k2 = k_star - j
            if 0 <= k2 < n:
                prod = a[j] * a[k2]
                if prod > best_prod:
                    best_prod = prod
                    best_j = j

        if best_j < 0:
            break

        k2 = k_star - best_j
        # Reduce the larger of the pair
        if a[best_j] >= a[k2]:
            reduction = 0.02 * a[best_j]
            a[best_j] -= reduction
        else:
            reduction = 0.02 * a[k2]
            a[k2] -= reduction

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        else:
            break

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def spectral_nudge(a, n, S, rng, strength=0.05):
    """Apply spectral flattening nudge then project."""
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Gentle flattening
    P_mean = P.mean()
    P_new = P + strength * (P_mean - P)

    # Small random phase perturbation
    phase_pert = rng.normal(0, 0.005, N)
    phase = phase + phase_pert

    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    return a_new

    for L in [8192, 16384, 4096]:
        if L not in lengths:
            lengths.append(L)

    # Budget allocation: favor the primary length heavily
    total_budget = budget_s - 50  # reserve for verification
    if len(lengths) == 1:
        budgets = [total_budget]
    else:
        # 80% to first, split rest
        budgets = [total_budget * 0.80]
        rest = total_budget * 0.20
        for i in range(1, len(lengths)):
            budgets.append(rest / (len(lengths) - 1))

    for i, n in enumerate(lengths):
        elapsed = time.monotonic() - t_global
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc = min(budgets[i], remaining - 40)
        if alloc < 30:
            break

        print("\n=== n=%d, budget=%.0fs ===" % (n, alloc))
        start_a = prev_best if (n == prev_len and prev_best is not None) else None
        a_res, score = optimize_one_length(n, alloc, rng, target=1.50, start_a=start_a)

        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  *** New global best: %.6f at n=%d ***" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

        if global_best <= 1.5027:
            print("Target reached!")
            break

    # Final fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using uniform fallback")

    # Ensure valid output
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned non-finite, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback score: %.6f" % ver)

    print("\n=== Final: length=%d, score=%.6f ===" % (len(global_a), min(global_best, ver)))
    return final_list
