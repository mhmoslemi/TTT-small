"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Smooth LSE optimization with Adam, coordinate descent polishing,
and multi-scale annealing. Exploits convexity of the problem.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to prevent overflow
    scaled = np.clip(scaled, -50, 50)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return (2.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta):
    """Value of (1/beta)*log(sum exp(beta*c))."""
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -50, 50)
    return cm + np.log(np.exp(scaled).sum()) / beta


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def run_lse_adam(a_init, n, S, max_time, rng):
    """
    LSE smooth approximation + Adam optimizer with simplex projection.
    Anneals beta from small to large.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Annealing schedule for beta
    beta_start = 2.0
    beta_end = 50.0
    n_phases = 8
    phase_len = max(50, 2000 // n_phases)

    c = fft_convolve_self(a)

    for phase in range(n_phases):
        t = phase / max(n_phases - 1, 1)
        # Exponential annealing
        beta = beta_start * (beta_end / beta_start) ** t
        lr = alpha * (1.0 / (1.0 + 0.1 * phase))

        for inner in range(phase_len):
            if time.monotonic() - start > max_time:
                break

            it += 1
            grad = lse_gradient(a, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            update = m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - lr * update
            a_new = project_simplex(a_new, S)

            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / S ** 2

            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new
            c = c_new

        # End of phase: check true objective
        a = project_simplex(a, S)
        c = fft_convolve_self(a)
        true_obj = 2.0 * n * c.max() / S ** 2
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()

        if time.monotonic() - start > max_time:
            break

    return best_a, best_obj


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate j, the objective max_k b[k] as a function of a[j]
    is convex (piecewise quadratic). Use ternary search.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)

    def obj_at_j(j, val):
        """Objective with a[j] = val, others fixed (adjusting sum)."""
        # We need to maintain sum = S. If we change a[j] from a[j] to val,
        # the sum changes by (val - a[j]). We redistribute the difference
        # among all other coordinates proportionally.
        # Actually, for coordinate descent on the simplex, we fix sum = S
        # and vary a[j]. The constraint is that all a[i] >= 0.
        # Let's just compute max_b with the modified a.
        a_mod = a.copy()
        a_mod[j] = val
        # Don't renormalize - we want to see the effect on max_b
        # The sum changes, so the objective changes via both max_b and sum^2
        c_mod = fft_convolve_self(a_mod)
        sum_mod = a_mod.sum()
        if sum_mod < 0.01:
            return np.inf
        return 2.0 * n * c_mod.max() / (sum_mod ** 2)

    # Instead of full coordinate descent (too slow for large n),
    # use a block approach: randomly pick blocks of coordinates
    # and optimize them jointly

    # Better approach: subgradient + occasional coordinate refinement
    # Actually, for large n, let's use a different strategy:
    # Pick the top few lag positions and do a local quadratic optimization

    # Let's use a simpler but effective approach:
    # At each step, find the top-k lag positions, compute the aggregate
    # gradient, and take a step with backtracking line search.

    a = best_a.copy()
    c = fft_convolve_self(a)
    top_k_list = [1, 3, 7, 15, 31, 63]
    step = 0.1
    last_improve = 0
    total_it = 0

    while time.monotonic() - start < max_time:
        total_it += 1
        k = top_k_list[min(total_it // 200, len(top_k_list) - 1)]
        k = min(k, 2 * n - 1)

        # Top-k subgradient
        idx = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for j in idx:
            lo = max(0, j - n + 1)
            hi = min(j, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[j - i_vals]
        grad /= k

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        cur_obj = 2.0 * n * c.max() / S ** 2
        found = False
        for _ in range(20):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2
            if obj_try < cur_obj * (1 - 1e-7):
                a = a_try
                c = c_try
                step *= 1.1
                found = True
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a.copy()
                    last_improve = total_it
                break
            step *= 0.7
            if step < 1e-6:
                break

        if not found:
            # Perturbation to escape
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            step = 0.1

        if total_it - last_improve > 500:
            # Larger perturbation
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            step = 0.5

    return best_a, best_obj


def run_mirror_descent(a_init, n, S, max_time, rng):
    """
    Mirror descent (multiplicative weights) with LSE gradient.
    Maintains non-negativity naturally.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 1e-300, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)

    eta = 0.1
    beta = 5.0
    it = 0
    last_improve = 0
    consec_fail = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Anneal beta
        beta = min(100.0, 5.0 * (1.0 + 0.01 * it))

        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Multiplicative update
        log_a = np.log(a)
        log_a_new = log_a - eta * grad
        log_a_new = np.clip(log_a_new, -700, 700)
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S ** 2

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            consec_fail = 0

        if obj_new < 2.0 * n * c.max() / S ** 2:
            a = a_new
            c = c_new
            eta = min(eta * 1.01, 1.0)
            consec_fail = 0
        else:
            consec_fail += 1
            eta *= 0.9
            if consec_fail > 30:
                # Perturbation
                noise = rng.normal(0, 0.01, n)
                a = np.maximum(a * (1 + noise), 1e-300)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                eta = 0.1
                consec_fail = 0

    return best_a, best_obj


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Previous best-like shapes
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * t), 0.01))
    inits.append(1.0 / (1.0 + 10.0 * t * (1 - t)))
    inits.append(t ** 2 + (1 - t) ** 2 + 0.1)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t))
    inits.append(np.exp(-8.0 * (t - 0.5) ** 2) + 0.2)
    inits.append(rng.random(n) * 0.5 + 0.1)
    inits.append(1.0 / (1.0 + np.exp(-5 * (t - 0.3))))
    inits.append(0.5 + 0.5 * np.sin(np.pi * t) ** 4)
    inits.append(1.0 + 0.4 * np.cos(4 * np.pi * t) - 0.2 * np.cos(8 * np.pi * t))

    return [np.maximum(x, 0.01) for x in inits]


def two_point_refinement(a, n, S, max_time, rng):
    """
    Two-point (pairwise) refinement: pick pairs of coordinates and
    optimize their values jointly while keeping sum constant.
    This is more effective than single-coordinate descent.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    a = a * (S / s)

    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S ** 2
    best_a = a.copy()
    c = fft_convolve_self(a)

    # Precompute: for each pair (i, j), the change in b[k] when we move
    # mass from j to i (keeping sum constant) is:
    # delta_b[k] = delta * (a[k-i] - a[k-j]) + quadratic terms
    # For small delta, this is approximately linear.

    # Strategy: repeatedly find the pair (i, j) and transfer amount
    # that most reduces the max convolution.

    # For efficiency, use a greedy approach:
    # 1. Find the peak lag k*
    # 2. Find the pair (i, j) that most reduces b[k*] by transferring mass
    # 3. Transfer a small amount

    it = 0
    step_base = S / n * 0.1

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S ** 2

        # Find top few problematic lags
        top_lags = np.argsort(c)[::-1][:5]

        # For each top lag, find the best "transfer" direction
        best_reduction = 0
        best_i, best_j, best_delta = -1, -1, 0

        # Sample random pairs (for efficiency with large n)
        n_samples = min(200, n)
        pairs = rng.integers(0, n, (n_samples, 2))

        for lag in top_lags[:3]:
            # Gradient of b[lag] w.r.t. a[j] is proportional to a[lag-j]
            # (with boundary effects)
            lo = max(0, lag - n + 1)
            hi = min(lag, n - 1)
            grad_lag = np.zeros(n)
            i_vals = np.arange(lo, hi + 1)
            grad_lag[i_vals] = 2.0 * a[lag - i_vals]

            # We want to decrease a[j] and increase a[i] where
            # grad_lag[j] > grad_lag[i] (to reduce b[lag])
            # But we also need to not increase other lags too much.

            # Simple heuristic: find j with highest grad_lag and i with lowest
            j_candidate = int(np.argmax(grad_lag))
            i_candidate = int(np.argmin(grad_lag))

            if j_candidate != i_candidate:
                delta = step_base
                # Try transferring delta from j to i
                a_try = a.copy()
                a_try[i_candidate] += delta
                a_try[j_candidate] -= delta
                if a_try[j_candidate] >= 0:
                    c_try = fft_convolve_self(a_try)
                    obj_try = 2.0 * n * c_try.max() / S ** 2
                    if obj_try < cur_obj:
                        reduction = cur_obj - obj_try
                        if reduction > best_reduction:
                            best_reduction = reduction
                            best_i, best_j, best_delta = i_candidate, j_candidate, delta

        if best_i >= 0:
            a[best_i] += best_delta
            a[best_j] -= best_delta
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            new_obj = 2.0 * n * c.max() / S ** 2
            if new_obj < best_obj - 1e-10:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # No improvement found; perturb
            noise_scale = step_base * 0.5
            mask = rng.random(n) < 0.1
            pert = np.zeros(n)
            pert[mask] = rng.normal(0, noise_scale, mask.sum())
            a = a + pert
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        if it % 10000 == 0:
            print("    Two-point it=%d, obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LSE-smoothed Adam optimization, mirror descent, coordinate refinement,
    and two-point exchange methods.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)
    t0 = time.monotonic()

    # Get previous best if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 2048 <= len(given) <= 16384:
        n = len(given)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, budget=%.0fs" % (n, budget_s))

    # Prepare initializations
    inits = []
    if given is not None:
        inits.append(given.copy())
        # Perturbations of given
        for _ in range(2):
            p = given.copy()
            noise = rng.normal(0, 0.02, n)
            p = np.maximum(p + noise, 0)
            p = p * (S / p.sum())
            inits.append(p)

    # Add diverse initializations
    for init in generate_diverse_inits(n, rng):
        inits.append(init.copy())

    global_best = np.inf
    global_a = None

    # Quick evaluation of all inits
    for i, init in enumerate(inits):
        a = np.clip(init, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)
            obj = compute_objective(a)
            if obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Init %d: obj=%.6f" % (i, obj))
            if time.monotonic() > deadline - budget_s * 0.8:
                break

    print("Initial best: %.6f" % global_best)

    # Phase 1: LSE + Adam (40% of time)
    p1_time = budget_s * 0.40
    p1_deadline = t0 + p1_time
    print("=== Phase 1: LSE+Adam (%.0fs) ===" % p1_time)

    if global_a is not None:
        a, obj = run_lse_adam(global_a, n, S, p1_time, rng)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)
            obj = compute_objective(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  LSE+Adam: %.6f" % global_best)

    # Phase 2: Mirror descent (20% of time)
    elapsed = time.monotonic() - t0
    p2_time = min(budget_s * 0.20, deadline - time.monotonic())
    print("=== Phase 2: Mirror Descent (%.0fs) ===" % p2_time)

    if global_a is not None and p2_time > 30:
        a, obj = run_mirror_descent(global_a, n, S, p2_time, rng)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)
            obj = compute_objective(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Mirror: %.6f" % global_best)

    # Phase 3: Coordinate/top-k subgradient refinement (25% of time)
    p3_time = min(budget_s * 0.25, deadline - time.monotonic())
    print("=== Phase 3: Top-k subgradient (%.0fs) ===" % p3_time)

    if global_a is not None and p3_time > 30:
        a, obj = run_coordinate_descent(global_a, n, S, p3_time, rng)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)
            obj = compute_objective(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Top-k: %.6f" % global_best)

    # Phase 4: Two-point exchange (10% of time)
    p4_time = min(budget_s * 0.10, deadline - time.monotonic() - 20)
    print("=== Phase 4: Two-point exchange (%.0fs) ===" % p4_time)

    if global_a is not None and p4_time > 30:
        a, obj = two_point_refinement(global_a, n, S, p4_time, rng)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)
            obj = compute_objective(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Two-point: %.6f" % global_best)

    # Phase 5: Final micro-optimization with very small steps
    remaining = deadline - time.monotonic()
    print("=== Phase 5: Final polish (%.0fs) ===" % remaining)

    if global_a is not None and remaining > 20:
        a = global_a.copy()
        c = fft_convolve_self(a)
        best_local = global_best
        step = 0.005
        it = 0
        start_local = time.monotonic()

        while time.monotonic() - start_local < remaining - 10:
            it += 1
            cur = 2.0 * n * c.max() / S ** 2

            # Top-1 subgradient
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            grad = np.zeros(n)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] = 2.0 * a[k_star - i_vals]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S ** 2

            if obj_try < cur * (1 - 1e-8):
                a = a_try
                c = c_try
                step = min(step * 1.02, 0.1)
                if obj_try < best_local - 1e-10:
                    best_local = obj_try
            else:
                step *= 0.95
                if step < 1e-5:
                    # Tiny perturbation
                    a = project_simplex(a + rng.normal(0, 1e-4 * S / n, n), S)
                    c = fft_convolve_self(a)
                    step = 0.005

        a = project_simplex(a, S)
        final_obj = compute_objective(a)
        if final_obj < global_best:
            global_best = final_obj
            global_a = a.copy()
            print("  Polish: %.6f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver > global_best:
                print("Warning: verified value differs, using verified")
        else:
            print("evaluate_sequence rejected input, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
    except Exception as e:
        print("Verification error: %s" % str(e))

    print("Final: len=%d, obj=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return final_list
