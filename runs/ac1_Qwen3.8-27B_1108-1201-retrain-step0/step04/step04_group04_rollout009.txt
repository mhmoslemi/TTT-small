```python
from scipy.sparse import csr_matrix

def build_conv_matrix(f, n, active_lags):
    """Build sparse matrix A where A[k_idx, j] = f[lag - j] for active lags."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    return A
```
