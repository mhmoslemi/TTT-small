def hankel_matvec(w, a, N):
    """Compute A @ a where A[i,j] = w[i+j]. Uses FFT."""
    n = len(a)
    # Reverse a
    a_rev = a[::-1]
    # Convolve w and a_rev
    full_conv = np.real(ifft(fft(w, N) * fft(a_rev, N)))
    # Extract the right slice
    return full_conv[n-1 : 2*n-1]

def waterfill_solve(w, S, n, N, max_iter=1000, tol=1e-10):
    """
    Solve: A @ a = lambda/2 * 1, sum(a) = S, a >= 0
    where A[i,j] = w[i+j].
    Uses augmented system with CG.
    """
    # Augmented system:
    # [A   1] [a]   [lambda/2 * 1]
    # [1^T 0] [lam] = [S]
    #
    # This is (n+1) x (n+1) symmetric system.
    # [A a + (lam/2)*1 = 0]  -- wait, let me set it up properly.
    #
    # We want: A @ a = (lambda/2) * 1  and  1^T @ a = S
    # Equivalently: A @ a - (lambda/2) * 1 = 0
    #
    # Let u = [a; lambda/2]. Then:
    # [A  -1] [a]       [0]
    # [1^T  0] [lam/2] = [S]
    #
    # This is symmetric. Use CG.
    
    # For CG, we need the matrix-vector product of the augmented system.
    # M @ [a; s] = [A@a - s*1; 1^T@a]
    # where s = lambda/2
    
    def aug_matvec(u):
        a_part = u[:n]
        s_part = u[n]
        result = np.empty(n + 1)
        result[:n] = hankel_matvec(w, a_part, N) - s_part
        result[n] = a_part.sum()
        return result
    
    # Initial guess
    u = np.zeros(n + 1)
    u[:n] = S / n
    u[n] = 0
    
    r = np.zeros(n + 1) - aug_matvec(u)  # r = b - M@u, where b = [0; S]
    p = r.copy()
    rs = r @ r
    
    for it in range(max_iter):
        Ap = aug_matvec(p)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        u += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if rs_new < tol * tol:
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    
    # Enforce non-negativity and sum constraint
    a_sol = np.maximum(u[:n], 0)
    a_sol_sum = a_sol.sum()
    if a_sol_sum > 0:
        a_sol = a_sol * S / a_sol_sum
    else:
        a_sol = np.full(n, S / n)
    
    return a_sol

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Water-filling active-set method with FFT-accelerated CG
2. Nesterov-accelerated gradient on softmax-smoothed objective
3. Multi-start with structured initializations
4. Random spectral perturbations for global exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def hankel_matvec(w, x, N):
    """Compute (A@x)[i] = sum_j w[i+j]*x[j] using FFT. w has length 2n-1, x has length n."""
    n = len(x)
    x_rev = x[::-1]
    result = np.real(ifft(fft(w, N) * fft(x_rev, N)))
    return result[n-1:2*n-1]

def waterfill_solve(w, S, n, N, max_iter=2000, tol=1e-12):
    """
    Solve A x = 1 (CG), then return a = S*x/sum(x).
    A is the Hankel matrix with A[i,j] = w[i+j].
    """
    b = np.ones(n)
    
    # Initial guess: uniform
    x = np.full(n, 1.0 / n)
    r = b - hankel_matvec(w, x, N)
    p = r.copy()
    rs = r @ r
    b_norm_sq = n  # ||b||^2 = n
    
    for it in range(max_iter):
        Ap = hankel_matvec(w, p, N)
        pAp = p @ Ap
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = r_new @ r_new
        if np.sqrt(rs_new) < tol * np.sqrt(b_norm_sq):
            x = x_new
            break
        x = x_new
        r = r_new
        p = r + (rs_new / max(rs, 1e-30)) * p
        rs = rs_new
    
    x_sum = x.sum()
    if x_sum <= 1e-15:
        return np.full(n, S / n)
    a = S * x / x_sum
    return a

def soft_max_objective(c, T):
    """T * log(sum exp(c/T)) using log-sum-exp trick."""
    c_max = c.max()
    return c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))

def soft_max_gradient(a, T, N):
    """Gradient of T*log(sum exp(c[k]/T)) w.r.t. a."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2*n-1]
    c_max = c.max()
    p = np.exp((c - c_max) / T)
    p /= p.sum()
    p_pad = np.zeros(N)
    p_pad[:2*n-1] = p
    P = fft(p_pad)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return grad

def nesterov_smooth(a_init, n, N, S, T_start=0.5, T_end=0.05, max_time=60.0, 
                     max_iter=5000, step_lr=1.0):
    """Nesterov-accelerated projected gradient on softmax-smoothed objective."""
    a = a_init.astype(np.float64).copy()
    y = a.copy()
    t = 1.0
    T = T_start
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() - start > max_time:
            break
        
        # Adaptive temperature
        frac = min(1.0, it / max_iter)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute gradient at y
        c = np.real(ifft(fft(y, N) ** 2))[:2*n-1]
        cur_val = soft_max_objective(c, T)
        if cur_val < best_val:
            best_val = cur_val
            best_a = y.copy()
        
        grad = soft_max_gradient(y, T, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 1/L ~ T (heuristic)
        step = step_lr * T / max(gnorm * n, 1e-10)
        step = min(step, 1.0)  # cap step
        
        x_new = y - step * grad
        x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y * S / y_sum
        a = x_new
        t = t_new

            c_best = fft_convolve_self(best_a)
            print(f"      NS it={it}, T={T:.4f}, soft={cur_val:.6f}, hard={c_best.max():.6f}")
    
    return best_a

def get_top_k_weights(c, top_k, power=1.0):
    """Get weights for top-k peaks."""
    n_c = len(c)
    top_idx = np.argsort(c)[::-1][:top_k]
    c_max = c.max()
    # Weight by relative excess
    w = np.zeros(n_c)
    top_vals = c[top_idx]
    # Use power-law weighting
    w[top_idx] = top_vals ** power
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w[top_idx] = 1.0 / top_k
    return w

def waterfill_iteration(a, n, N, S, top_k=50, power=1.0, max_cg_iter=1000):
    """One water-filling iteration."""
    c = fft_convolve_self(a)
    c_max = c.max()
    w = get_top_k_weights(c, top_k, power)
    a_new = waterfill_solve(w, S, n, N, max_iter=max_cg_iter)
    a_new = project_simplex(a_new, S)
    return a_new, c_max

        A = np.exp(1j * phases)
        a = np.real(ifft(A))[:n]
        a = np.clip(a, 0.01, None)
    elif method == "raised_cosine":

        a = 0.5 * (1 + np.cos(2*np.pi*t))
        a += 0.01 * rng.standard_normal(n)
        a = np.clip(a, 0.001, None)
    elif method == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-3*t**2)
        a += 0.01 * rng.standard_normal(n)
        a = np.clip(a, 0.001, None)
    elif method == "sin":
        t = np.arange(n) / n
        a = 0.5 + 0.3*np.sin(2*np.pi*3*t) + 0.2*np.sin(2*np.pi*7*t)
        a += 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == "uniform":
        a = np.ones(n)
    else:
        a = 0.5 + 0.5 * rng.random(n)
    return a
