def solve_lp_br(x, S, top_k, time_limit):
    n = len(x)
    c = conv_full(x)  # full convolution, via FFT
    M = np.max(c)
    # top-k lags
    idx = np.argsort(c)[::-1][:top_k]
    # Build A (top_k x n): A[i, j] = x_{idx_i - j} (0 out of range)
    A = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A[i, j] = x[k-j]
    y_var = cp.Variable(n)
    cons = [A @ y_var <= M, y_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), cons)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=100000, eps=1e-5)
    except Exception:
        try:
            prob.solve(solver=cp.PDLP, verbose=False)
        except Exception:
            return None
    y = y_var.value
    if y is None: return None
    y = np.maximum(y, 0)
    return y, M

def lp_best_response(x, S, top_k, time_budget=30.0):
    n = len(x)
    c = conv_full(x)
    M = float(np.max(c))
    idx = np.argsort(c)[::-1][:top_k]
    A = np.zeros((top_k, n), float)
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[i, lo:hi+1] = np.flip(x[k - hi : k - lo + 1])
    yv = cp.Variable(n)
    cons = [A @ yv <= M, yv >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(yv)), cons)
    solvers = [cp.PDLP, cp.SCS, cp.CLARABEL]  ... 
    for name in solvers:
        try:
            t0 = time.time()
            prob.solve(solver=name, verbose=False)
            if time.time() - t0 > time_budget:
                # could be interrupted? CVXPY not easily; skip
                pass
            y = yv.value
            if y is not None and np.isfinite(y).all():
                y = np.maximum(y, 0.0)
                return y, M, name
        except Exception:
            continue
    return None, M, None

def repulse_deflate(x, S, top_lags=16, eta=0.5, beta=40.0):
    n = len(x)
    c = conv_full(x)
    M = c.max()
    if M <= 0: return x.copy(), float(M)
    idx = np.argsort(c)[::-1][:top_lags]
    # weights
    w = np.exp(beta * (c[idx] - M))
    w = w / w.sum()
    # repulse vector r[j] (in OUR conv convention, j index) = 2 * sum_{k in top} w_k x[k - j] (0 out of range)
    r = np.zeros(n)
    for i in range(len(idx)):
        k = idx[i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # for j in [lo, hi]: contribution 2 * w_i * x[k - j]
        r[lo:hi+1] += 2.0 * w[i] * x[k - hi : k - lo + 1][::-1]  # x[k - j] for j=lo->hi is x[k-lo]->x[k-hi] reversed
    # scale
    r_norm = np.max(r) if r.size else 0.0
    if r_norm <= 0: return x.copy(), float(M)
    r = r / r_norm
    # multiplicative
    xm = x * np.exp(-eta * r)
    xm = normalize_sum(xm, S)
    return xm, 0.0

def permute_hillclimb(x, S, n_steps, rng, T0=0.05, Tend=1e-4):

    cur_peak = peak_of(cur)

    for it in range(n_steps):

        j = rng.integers(0, n)
        if i == j: continue

        new_peak = peak_of(cur)

            if new_peak < best_peak:

            # revert
            cur[j], cur[i] = cur[i], cur[j]
    return best, best_peak

"""
Minimize 2*n*max(a*a)/sum(a)^2 over non-negative sequences.
Strategy: inherit SOTA start; drive with (1) LP best-response (HiGHS/LinProg)
  maximizing cross-mass under the top peak lags, and (2) sharp exponential
  deflation (moving mass off peak-contributing entries), plus optional
  permutation simulated-annealing hill-climb; multi-start + annealed schedule.
Objective (under sum=sqrt(2n)) equals max_full_conv(a), so we minimize that.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def conv_full(x, y=None):
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    c = np.real(ifft(X * Y))[:size]
    return c


def peak_self(x):
    return float(np.max(conv_full(x)))


def normalize_sum(x, S):
    x = np.asarray(x, dtype=np.float64).copy()
    x[~np.isfinite(x)] = 0.0
    np.clip(x, 0.0, 1000.0, out=x)
    s = float(x.sum())
    if s < 1e-12:
        x[:] = S / float(len(x))
        return x
    x *= (S / s)
    return x


def build_rows(x, n, k_list):
    k_list = np.array([int(v) for v in k_list])
    R = np.zeros((len(k_list), n), dtype=np.float64)
    for i in range(len(k_list)):
        k = int(k_list[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            R[i, lo:hi+1] = np.flip(x[k - hi : k - lo + 1])
    return R


def lp_best_response(x, S, top_k):
    n = len(x)
    c = conv_full(x)
    M = float(np.max(c))
    idx = np.argsort(c)[::-1][:top_k]
    A = build_rows(x, n, idx)
    # avoid infeasibility guard
    b = np.ones(len(idx)) * (M + 1e-9)
    try:
        res = linprog(c=np.zeros(n) - 1.0, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n, method='highs')
        if res.status in (0, 3) and res.x is not None:
            y = np.asarray(res.x, dtype=np.float64)
            y[~np.isfinite(y)] = 0.0
            y = np.maximum(y, 0.0)
            if float(y.sum()) > 1e-12:
                return y
    except Exception:
        pass
    return None


def deflation_step(x, S, n, c, beta, eta, top_lags):
    M = float(np.max(c))
    if M <= 1e-12:
        return x.copy(), M
    idx_top = np.argsort(c)[::-1][:top_lags]
    vals = c[idx_top]
    w = np.exp(beta * (vals - vals.max()))
    wsum = float(w.sum())
    if wsum <= 1e-12:
        w = np.ones(len(idx_top)) / len(idx_top)
    else:
        w = w / wsum
    R = build_rows(x, n, idx_top)
    r = R.T @ (2.0 * w)  # shape (n,)
    rmax = float(np.max(r))
    if rmax <= 1e-12:
        return x.copy(), M
    r = r / rmax
    xm = x * np.exp(-eta * r)
    xm = normalize_sum(xm, S)
    return xm, M


def permute_sa(x, S, n, n_steps, rng, T0=0.03, Tend=0.0005):
    n = len(x)
    cur = x.copy()
    cur_prev = peak_self(cur)
    best = cur.copy(); best_prev = cur_prev
    logT0 = math.log(max(T0, 1e-6)); logT1 = math.log(max(Tend, 1e-6))
    for it in range(n_steps):
        i = int(rng.integers(0, n)); j = int(rng.integers(0, n))
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_prev = peak_self(cur)
        Tv = math.exp(logT0 + (logT1 - logT0) * (it / max(1, n_steps - 1)))
        accept = (new_prev < cur_prev) or (np.random.random() < math.exp(-max(0.0, new_prev - cur_prev) / max(Tv, 1e-6)))
        if accept:
            cur_prev = new_prev
            if new_prev < best_prev:
                best_prev = new_prev
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_prev


def deflation_drill(init, S, n, budget, rng, tag, allow_lp, allow_restart, lp_k, lp_gap):
    x = normalize_sum(init, S)
    val = peak_self(x)
    best_x = x.copy(); best_val = val
    t0 = time.time(); dl = t0 + budget
    it = 0
    eta = 0.5; beta = 20.0; top_lags = 16
    last_lp = 0.0; last_print = 0.0
    while time.time() < dl:
        c = conv_full(x)
        xm, _ = deflation_step(x, S, n, c, beta, eta, top_lags)
        vm = peak_self(xm)
        if vm < val - 1e-12:
            x = xm; val = vm
        if vm < best_val:
            best_val = vm; best_x = xm.copy()
        if allow_lp:
            now = time.time()
            if now - last_lp > lp_gap and now < dl - 20.0:
                nlp_top = min(lp_k, n + n - 1)
                y = lp_best_response(x, S, nlp_top)
                if y is not None:
                    yn = normalize_sum(y, S)
                    best_t_val = best_val
                    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
                        mix = (1.0 - t) * x + t * yn
                        m = normalize_sum(mix, S)
                        mv = peak_self(m)
                        if mv < best_t_val:
                            best_t_val = mv
                    # adopt best mix if it also improves the live run
                    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
                        m = normalize_sum((1.0 - t) * x + t * yn, S)
                        mv = peak_self(m)
                        if mv < val - 1e-12 and mv <= best_t_val*(1+1e-9):
                            x = m; val = mv
                last_lp = time.time()
            it += 1
            # periodic restart (early)

                x_try = normalize_sum(x + rng.normal(0.0, scale, n), S)
                vt = peak_self(x_try)
                if vt < best_val:

            # anneal

        now = time.time()

            print("[%s] it=%d val=%.6f best=%.6f remain=%.0fs" % (tag, it, val, best_val, dl - now), flush=True)
            last_print = now
    return best_x, best_val

"""
Minimize 2*n*max(a*a)/sum(a)^2 over non-negative sequences.

Strategy (deliberately different from the previous FISTA+LP hybrid):
  * Note: under the normalization sum(a)=sqrt(2n), the objective equals
    max_full_conv(a), so all internal optimization minimizes the peak of
    the full self-convolution while keeping sum(a)=sqrt(2n).
  * Inherit the previous SOTA (height_sequence_1) as the primary start.
  * Core driver: LP "best-response" (scipy HiGHS): maximize sum(y) subject
    to (x*y)[k] <= M on the currently top lags (M = peak of x*x), then a
    line-searched damp mix x <- (1-t)x + t*y.  This is a different, safer
    core than LSE-FISTA.
  * Sharp "exponential deflation" step: move probability mass off the
    entries that contribute to the top lags (r = top_lag_row^T w), using
    a softmax lag-weighting (annealed) and multiplicative re-normalization.
  * Occasional stochastic local restarts (non-negative simplex noise) for
    basin-hopping within a start.
  * Final phase: permutation simulated-annealing hill-climb (swap adjacent
    re-orderings) on the global best, plus a fast deflation-only polish.
  * Multi-start: SOTA + smooth + uniform/structured inits, with adaptive
    time sharing; keeps global best throughout.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def conv_full(x, y=None):
    """Full linear convolution via padded real FFT."""
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    c = np.real(ifft(X * Y))[:size]
    return c


def peak_self(x):
    """Peak of the full self-convolution."""
    return float(np.max(conv_full(x)))


def normalize_sum(x, S):
    """Clip to [0,1000], d/negin, and rescale so sum == S."""
    x = np.asarray(x, dtype=np.float64).copy()
    x[~np.isfinite(x)] = 0.0
    np.clip(x, 0.0, 1000.0, out=x)
    s = float(x.sum())
    if s < 1e-12:
        return np.full(len(x), S / float(len(x)), dtype=np.float64)
    x *= (S / s)
    return x

    """Rows R[i, :] with R[i, j] = x[k_i - j] (0 out of range).
       Then (R[i] @ y) == full_conv(x, y)[k_i]."""
    k_list = np.array([int(v) for v in k_list])
    R = np.zeros((len(k_list), n), dtype=np.float64)
    for i in range(len(k_list)):
        k = int(k_list[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            R[i, lo:hi + 1] = np.flip(x[k - hi : k - lo + 1])
    return R


def lp_best_response(x, S, top_k):
    """Solve max sum(y) s.t. (x*y)[k] <= M for top-k lags, y >= 0.
       Returns raw y or None on failure."""
    n = len(x)
    c = conv_full(x)
    M = float(np.max(c))
    idx = np.argsort(c)[::-1][:top_k]
    A = build_rows(x, n, idx)
    b = np.ones(len(idx)) * (M + 1e-9)
    try:
        res = linprog(c=np.zeros(n) - 1.0, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n, method='highs')
        if res.status in (0, 3) and res.x is not None:
            y = np.asarray(res.x, dtype=np.float64)
            y[~np.isfinite(y)] = 0.0
            y = np.maximum(y, 0.0)
            if float(y.sum()) > 1e-12:
                return y
    except Exception:
        pass
    return None


def deflation_step(x, S, n, c, beta, eta, top_lags):
    """One sharp exponential deflation step: shrink entries that feed the
       most-binding lags, then re-normalize to sum S.  Returns (xm, M)."""
    M = float(np.max(c))
    if M <= 1e-12:
        return x.copy(), M
    idx_top = np.argsort(c)[::-1][:top_lags]
    vals = c[idx_top]
    if beta < 1e-6:
        w = np.ones(len(idx_top)) / len(idx_top)
    else:
        w = np.exp(beta * (vals - vals.max()))
        wsum = float(w.sum())
        w = (w / wsum) if wsum > 1e-12 else (np.ones(len(idx_top)) / len(idx_top))
    R = build_rows(x, n, idx_top)
    r = R.T @ (2.0 * w)
    rmax = float(np.max(r))
    if rmax <= 1e-12:
        return x.copy(), M
    r = r / rmax
    xm = x * np.exp(-eta * r)
    xm = normalize_sum(xm, S)
    return xm, M


def permute_sa(x, S, n, n_steps, rng, T0=0.02, Tend=0.0005):
    """Simulated annealing over random swaps (preserves multiset and sum).
       Returns (best_permuted, best_peak)."""
    cur = x.copy()
    cur_peak = peak_self(cur)
    best = cur.copy()
    best_peak = cur_peak
    logT0 = math.log(max(T0, 1e-6))
    logT1 = math.log(max(Tend, 1e-6))
    for it in range(n_steps):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_peak = peak_self(cur)
        frac = (it / max(1, n_steps - 1))
        Tv = math.exp(logT0 + (logT1 - logT0) * frac)
        accept = (new_peak < cur_peak) or (
            float(rng.random()) < math.exp(-max(0.0, new_peak - cur_peak) / max(Tv, 1e-9)))
        if accept:
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_peak


def deflation_drill(init, S, n, budget, rng, tag, allow_lp, allow_restart,
                    lp_k, lp_gap):
    """Main driver: deflation + optional LP best-response + stochastic
       non-negative simplex restarts.  Returns (best_x, best_val)."""
    x = normalize_sum(init, S)
    val = peak_self(x)
    best_x = x.copy()
    best_val = val
    t0 = time.time()
    dl = t0 + max(0.0, budget)
    it = 0
    eta = 0.5
    beta = 20.0
    top_lags = 16
    last_lp = 0.0
    last_print = 0.0
    worst_lp_time = 0.0
    lp_on = (allow_lp is True)
    while time.time() < dl:
        c = conv_full(x)
        xm, _ = deflation_step(x, S, n, c, beta, eta, top_lags)
        vm = peak_self(xm)
        if vm < val - 1e-12:
            x = xm
            val = vm
        if vm < best_val:
            best_val = vm
            best_x = xm.copy()

        if lp_on:
            now = time.time()
            if now - last_lp > lp_gap and now < dl - 30.0:
                nlp_top = min(int(lp_k), max(8, 2 * n - 1))
                tp = time.time()
                y = lp_best_response(x, S, nlp_top)
                el = time.time() - tp
                worst_lp_time = max(worst_lp_time, el)
                if worst_lp_time > 140.0:
                    lp_on = False
                    print("[%s] LP too slow (worst %.1fs), switching to deflation-only" % (tag, worst_lp_time), flush=True)
                if y is not None:
                    yn = normalize_sum(y, S)
                    cand = []
                    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0):
                        m = normalize_sum((1.0 - t) * x + t * yn, S)
                        mv = peak_self(m)
                        cand.append((mv, t))
                    cand.sort(key=lambda z: z[0])
                    mv, _t = cand[0]
                    if mv < best_val:
                        best_val = mv
                        best_x = normalize_sum((1.0 - _t) * x + _t * yn, S).copy()
                    if mv < val - 1e-12:
                        x = best_x.copy()
                        val = mv
                last_lp = time.time()
        it += 1
        if allow_restart and (it % 600 == 0) and (it < 4000):
            scale = (S / math.sqrt(n)) * 0.008
            x_try = normalize_sum(x + rng.normal(0.0, scale, n), S)
            vt = peak_self(x_try)
            if vt < best_val:
                best_val = vt
                best_x = x_try.copy()
                x = x_try.copy()
                val = vt
        if it % 250 == 0:
            eta = max(0.18, eta - 0.005)
            beta = min(170.0, beta + 1.1)
            top_lags = min(36, 16 + it // 900)
        now = time.time()
        if now - last_print > 20.0:
            print("[%s] it=%d val=%.6f best=%.6f remain=%.0fs" % (tag, it, val, best_val, dl - now), flush=True)
            last_print = now
    return best_x, best_val


def make_smooth_inits(n, rng, amplitudes):
    """Produce a list of non-negative, structurally-diverse initializations."""
    inits = []
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    # smooth single / multi bell
    for amp in amplitudes:
        base = 1.0 - amp * np.sin(2 * np.pi * t) ** 4
        base = np.maximum(base, 1e-3)
        base += rng.normal(0.0, 0.02 * base.mean(), n)
        inits.append(np.maximum(base, 1e-3))
    # flat / structured / partial random
    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 2 * t)
                            + 0.2 * np.sin(2 * np.pi * 5 * t), 1e-3))
    # plateau + ramp
    pr = np.concatenate([
        np.full(n // 3, 1.2),
        np.linspace(1.2, 0.6, n - 2 * (n // 3)),
    ])[:n]
    inits.append(np.maximum(pr + rng.normal(0.0, 0.02, n), 1e-3))
    # mild gaussian centered
    gc = np.exp(-2.0 * (t - 0.5) ** 2 / 0.10)
    inits.append(np.maximum(0.4 + 0.6 * gc, 1e-3))
    return [np.asarray(a, dtype=np.float64) for a in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    t0 = time.time()
    budget_s = float(budget_s)
    dl = t0 + max(0.0, budget_s - 45.0)
    rng = np.random.default_rng(seed)

    # ---- primary length + SOTA inheritance ----
    n_default = 8192
    n = n_default
    sota = None
    hsq = None
    try:
        hsq = globals().get('height_sequence_1', None)
    except Exception:
        hsq = None
    if isinstance(hsq, (list, tuple, np.ndarray)) and len(hsq) > 0:
        try:
            arr = np.asarray(hsq, dtype=np.float64).reshape(-1)
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if 512 <= len(arr) <= 32768 and float(arr.sum()) > 0.5:
                sota = arr
                n = len(arr)
        except Exception:
            pass
    S = math.sqrt(2.0 * n)
    print("Starting n=%d S=%.3f budget=%.0fs sota=%s" % (n, S, budget_s, (sota is not None)), flush=True)

    best_x = None
    best_val = math.inf

    # ---- helper-ish inline update of global best via local vars (no closures) ----
    # (implemented explicitly by comparing each phase result below)

    # Phase 1: SOTA-driven heavy drill
    if sota is not None:
        b1 = min(1150.0, max(40.0, dl - time.time()))
        x1, v1 = deflation_drill(sota, S, n, b1, rng, tag="sota",
                                 allow_lp=True, allow_restart=True,
                                 lp_k=150, lp_gap=22.0)
        best_x, best_val = x1, v1
        print("Phase1 sota drill best=%.6f remain=%.0fs" % (best_val, dl - time.time()), flush=True)

        # Phase 2: deflation-only polish on sota line
        if dl - time.time() > 60:
            b2 = min(320.0, dl - time.time() - 250.0)
            if b2 > 40:
                x2, v2 = deflation_drill(x1 if x1 is not None else best_x, S, n, b2, rng,
                                         tag="sota2", allow_lp=False, allow_restart=False,
                                         lp_k=150, lp_gap=22.0)
                if v2 < best_val:
                    best_val = v2
                    best_x = x2
                print("Phase2 sota2 best=%.6f remain=%.0fs" % (best_val, dl - time.time()), flush=True)

    # If we have no SOTA, run a core drill from a smooth init to seed the global best
    if best_x is None:
        sm = make_smooth_inits(n, rng, amplitudes=[0.8, 0.5, 0.3])
        init0 = normalize_sum(sm[0], S)
        b0 = min(700.0, max(60.0, dl - time.time()))
        x0, v0 = deflation_drill(init0, S, n, b0, rng, tag="noSOTA",
                                 allow_lp=True, allow_restart=True,
                                 lp_k=150, lp_gap=25.0)
        best_x, best_val = x0, v0
        print("Seeded from smooth (no sota) best=%.6f remain=%.0fs" % (best_val, dl - time.time()), flush=True)

    # Phase 3: secondary starts (smooth + structured) on the chosen n
    remaining3 = dl - time.time()
    if remaining3 > 200:
        sm = make_smooth_inits(n, rng, amplitudes=[0.85, 0.45, 0.25])
        amench = sm[:3]
        per3 = (remaining3 * 0.5) / max(1, len(amench))
        for k in range(len(amench)):
            if dl - time.time() < 90:
                break
            initk = normalize_sum(amench[k], S)
            xk, vk = deflation_drill(initk, S, n, max(30.0, per3), rng,
                                     tag=("s%d" % k), allow_lp=True, allow_restart=True,
                                     lp_k=120, lp_gap=30.0)
            if vk < best_val:
                best_val = vk
                best_x = xk
                print("Secondary start %d improved to %.6f remain=%.0fs" % (k, best_val, dl - time.time()), flush=True)
            else:
                print("Secondary start %d: %.6f (no improvement) remain=%.0fs" % (k, vk, dl - time.time()), flush=True)

    # Phase 4: permutation SA hill-climb on the global best (cheap, orthogonal)
    remaining4 = dl - time.time()
    if best_x is not None and remaining4 > 120:
        # a couple of SA bursts, keep improvement
        for burst in range(2):
            if dl - time.time() < 70:
                break
            n_steps = 9000
            xb, vb = permute_sa(best_x, S, n, n_steps=n_steps, rng=rng,
                                T0=0.02, Tend=0.0008)
            if vb < best_val:
                best_val = vb
                best_x = xb
                print("SA burst %d improved to %.6f remain=%.0fs" % (burst, best_val, dl - time.time()), flush=True)
            else:
                print("SA burst %d: %.6f (no improvement) remain=%.0fs" % (burst, vb, dl - time.time()), flush=True)

    # Phase 5: final fast deflation-only polish on global best
    if best_x is not None and dl - time.time() > 40:
        b5 = min(250.0, dl - time.time() - 35.0)
        if b5 > 20:
            x5, v5 = deflation_drill(best_x, S, n, b5, rng, tag="polish",
                                     allow_lp=False, allow_restart=False,
                                     lp_k=150, lp_gap=30.0)
            if v5 < best_val:
                best_val = v5
                best_x = x5
            print("Final polish best=%.6f remain=%.0fs" % (best_val, dl - time.time()), flush=True)

    # ---- validation & safe return ----
    if best_x is None:
        best_x = np.ones(n, dtype=np.float64)
        best_x = normalize_sum(best_x, S)
        best_val = peak_self(best_x)
    best_x = np.asarray(best_x, dtype=np.float64).reshape(-1)
    best_x = np.nan_to_num(best_x, nan=0.0, posinf=0.0, neginf=0.0)
    best_x = np.clip(best_x, 0.0, 1000.0)
    best_x = normalize_sum(best_x, S)
    ret = [float(v) for v in best_x]
    try:
        gv = evaluate_sequence(ret)
        print("evaluate_sequence(final)=%.6f internal=%.6f" % (float(gv), best_val), flush=True)
    except NameError:
        print("evaluate_sequence not available; internal=%.6f" % best_val, flush=True)
    except Exception as e:
        print("evaluate_sequence error: %r" % e, flush=True)
    return ret
