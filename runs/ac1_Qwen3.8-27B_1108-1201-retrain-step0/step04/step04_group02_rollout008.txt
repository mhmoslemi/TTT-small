This is a neat formula! The gradient in log-space is proportional to `a[i] * (corr(w,a)[i] - LSE)`.

Intuitively: if `corr(w,a)[i] > LSE` (i.e., coordinate i is "contributing more than average" to the peaked parts of the convolution), we should decrease x[i] (decrease a[i]). If less, we increase.

This is very similar to multiplicative weight updates!

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using log-space Adam
optimization with LSE-smoothed max, temperature annealing, and multiple restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(a, c, beta):
    """
    Compute LSE-smoothed max and its gradient in log-space.
    Returns: (lse_value, grad_x) where grad_x[i] = dLSE/dx[i]
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    
    # corr(w, a)[j] = sum_k w[k] * a[k-j] (for valid k)
    fsize = 1 << (n + 2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A), fsize))[:n]
    
    # dLSE/dx[i] = 2 * a[i] * corr[i]
    grad_x = 2.0 * a * corr
    
    return lse_val, grad_x


def optimize_logspace(a_init, budget, seed=42):
    """
    Optimize a sequence in log-space using Adam with LSE smoothing.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    
    # Initialize in log-space
    a_init = np.maximum(a_init, 1e-10)
    x = np.log(a_init)
    x = x - x.mean()  # normalize
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_obj = compute_obj(np.exp(x))
    best_x = x.copy()
    no_improve = 0
    
    start = time.monotonic()
    
    while time.monotonic() - start < budget:
        elapsed = time.monotonic() - start
        progress = elapsed / budget
        
        # Temperature schedule: starts low, increases with annealing restarts
        # Use a "cyclical" temperature that occasionally resets
        if t < 5000:
            beta = 5.0 + 50.0 * (t / 5000.0)
        else:
            beta = min(300.0, 50.0 + 0.01 * (t - 5000))
        
        a = np.exp(x)
        c = fft_convolve_self(a)
        
        lse_val, grad_lse = lse_and_grad(a, c, beta)
        
        # Full objective gradient in log-space:
        # d/dx [2n * LSE / S^2] where S = sum(a)
        # = (2n/S^2) * grad_lse - (4n*LSE/S^2) * a
        S = a.sum()
        grad = (2.0 * n / (S * S)) * (grad_lse - 2.0 * lse_val * a)
        
        # Adam update
        t += 1
        # Learning rate with warmup and decay
        if t < 100:
            lr = 0.01 * (t / 100.0)
        else:
            lr = 0.01 / (1.0 + 0.0001 * (t - 100))
        
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        
        x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Clip for numerical stability
        x = np.clip(x, x.mean() - 20, x.mean() + 20)
        
        # Periodic check
        if t % 500 == 0:
            a_check = np.exp(x)
            obj_check = compute_obj(a_check)
            if obj_check < best_obj - 1e-10:
                best_obj = obj_check
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 20:
                # Restart with perturbation
                noise = rng.normal(0, 0.1, n)
                x = best_x + noise
                x = x - x.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

            x = x - x.mean()
        
        # Print progress
        if t % 5000 == 0:
            print("    t=%d, beta=%.1f, best_obj=%.6f" % (t, beta, best_obj))

def topk_refine(a, budget, k=1):
    """Fine-tune using exact top-k subgradient with small steps."""
    n = len(a)
    S = a.sum()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0.5
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < budget:
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Top-k subgradient
        idx = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for j in idx:
            lo = max(0, j - n + 1)
            hi = min(j, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[j - i_vals]
        grad /= k
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try a step
        a_try = np.maximum(a - step * grad / (gnorm / S), 0)
        s_try = a_try.sum()
        a_try = a_try * (S / s_try)
        
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2
        
        if obj_try < cur_obj:
            a = a_try
            step = min(step * 1.1, S)
        else:
            step *= 0.8
            if step < 0.01 * S / n:
                # Perturb and restart
                a = np.maximum(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), 0)
                a = a * (S / a.sum())
                step = 1.0
    
    return best_a, best_obj
