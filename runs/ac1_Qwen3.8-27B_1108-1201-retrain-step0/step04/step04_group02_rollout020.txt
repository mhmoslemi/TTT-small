"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using multi-phase
optimization: FISTA with LSE-smoothed gradients, epsilon-active-set
subgradients, adaptive step sizes, and multi-length search.
Starts from previous best and refines with careful parameter scheduling.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    mask = tc < w
    if not np.any(mask):
        rho = n
    else:
        rho = int(np.max(np.where(mask)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_lse_grad(a, c, beta, n):
    """LSE-smoothed gradient of max(c)."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * result


def compute_active_grad(a, c, n, rel_eps):
    """Average subgradient over active set (c_k within rel_eps*max of max)."""
    max_c = c.max()
    if max_c < 1e-300:
        return np.zeros(n)
    threshold = max_c * (1.0 - rel_eps)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    grad = np.zeros(n)
    for j in active:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / len(active)


def compute_topk_grad(a, c, n, k):
    """Subgradient using top-k peaks."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def get_obj(a, c, n, S):
    """Compute objective from precomputed convolution."""
    return 2.0 * n * c.max() / (S * S)


def run_fista_lse(a_init, n, max_time, rng, beta_start=30.0, beta_end=800.0):
    """
    FISTA with LSE-smoothed gradient and restarting.
    Adaptive beta increasing over time for increasing accuracy.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()

    c = fft_convolve_self(a)
    cur_obj = get_obj(a, c, n, S)
    best_obj = cur_obj
    best_a = a.copy()
    it = 0
    fail_streak = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)

        # Beta schedule: increasing for more accurate gradient
        if frac < 0.3:
            beta = beta_start + (beta_start * 2 - beta_start) * (frac / 0.3)
        elif frac < 0.6:
            beta = beta_start * 2 + (beta_start * 8 - beta_start * 2) * ((frac - 0.3) / 0.3)
        elif frac < 0.85:
            beta = beta_start * 8 + (beta_end - beta_start * 8) * ((frac - 0.6) / 0.25)
        else:
            beta = beta_end

        # Compute gradient at y
        c_y = fft_convolve_self(y)
        grad = compute_lse_grad(y, c_y, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size: balance between accuracy (needs small step for high beta)
        # and progress (wants large step)
        step = 0.8 / (beta * 2.0 * S / n + 1e-15)
        step = min(step, 2.0 * S / (n * gnorm + 1e-15))
        step = min(step, 1.5)

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum with restarting
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum_coeff = (t - 1.0) / t_new
        y_candidate = x_new + momentum_coeff * (x_new - a)

        # Check if momentum helps (restart if not)
        c_x = fft_convolve_self(x_new)
        obj_x = get_obj(x_new, c_x, n, S)

        y_candidate = np.maximum(y_candidate, 0.0)
        s_yc = y_candidate.sum()
        if s_yc > 1e-15:
            y_candidate = y_candidate * (S / s_yc)
            c_yc = fft_convolve_self(y_candidate)
            obj_yc = get_obj(y_candidate, c_yc, n, S)
        else:
            obj_yc = cur_obj

        if obj_yc > cur_obj + 1e-12 * cur_obj:
            # Restart momentum
            y = x_new
            t = 1.0
        else:
            y = y_candidate
            t = min(t_new, 500.0)  # Cap momentum to prevent divergence

        a = x_new
        c = c_x
        cur_obj = obj_x

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            fail_streak = 0
        else:
            fail_streak += 1

        # Periodic perturbation to escape local minima
        if fail_streak > 2000 and rng.random() < 0.05:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.05, 1.0 - frac)
            y = project_simplex(a + rng.normal(0, noise_scale, n), S)
            t = 1.0
            fail_streak = 0

        if it % 10000 == 0:
            print("    FISTA-LSE it=%d, obj=%.8f, beta=%.0f" % (it, best_obj, beta))

    return best_a, best_obj


def run_active_set_refine(a_init, n, max_time, rng):
    """
    Fine-tuning using epsilon-active-set subgradient with adaptive step.
    Designed for the very end of optimization when we're close to a local min.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    cur_obj = get_obj(a, c, n, S)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()
    step = 0.01
    it = 0
    fail = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)

        # Decreasing epsilon for tighter active set
        rel_eps = 0.005 * (1.0 - 0.5 * frac)

        grad = compute_active_grad(a, c, n, rel_eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Small adaptive step
        eff_step = step * (S / (gnorm * n + 1e-15))
        eff_step = min(eff_step, 0.5)

        a_new = project_simplex(a - eff_step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = get_obj(a_new, c_new, n, S)

        if obj_new < cur_obj - 1e-14:
            a = a_new
            c = c_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            step = min(step * 1.01, 0.1)
            fail = 0
        else:
            fail += 1
            step *= 0.9
            if fail > 100:
                # Small perturbation
                noise_scale = 0.0005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                cur_obj = get_obj(a, c, n, S)
                step = 0.01
                fail = 0

        if it % 5000 == 0:
            print("    Active-set it=%d, obj=%.8f" % (it, best_obj))

    return best_a, best_obj

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Cosine bump
    inits.append(np.abs(np.cos(np.pi * tt)))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    # Sine squared
    inits.append(np.sin(np.pi * tt) ** 2)
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.3))
    # Edge-heavy
    inits.append(1.0 + 1.5 * (tt ** 3 + (1 - tt) ** 3))
    # Random
    inits.append(rng.random(n) + 0.05)
    # Gaussian
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2))
    # Two bumps
    inits.append(np.exp(-40.0 * (tt - 0.3) ** 2) + np.exp(-40.0 * (tt - 0.7) ** 2))

    if parent is not None:
        p = np.clip(parent, 0, 1000)
        inits.append(p / p.sum() if p.sum() > 0 else np.ones(n) / n)
        # Perturbed parent
        noise = 0.02 * np.ones(n)
        inits.append(np.maximum(p + rng.normal(0, noise, n), 0.0))

    return inits
