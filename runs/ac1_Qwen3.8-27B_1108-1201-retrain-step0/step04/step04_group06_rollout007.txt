import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def spectral_shape_step(a, N, n, top_k, alpha, rng, target_peak):
    """
    Single spectral shaping step.
    Targets top-k peak lags simultaneously using exponential shaping.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    if c_max <= target_peak:
        return a, c_max
    
    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Weight by excess
    excess = np.maximum(top_vals - target_peak, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = top_vals.sum()
        if excess_sum < 1e-30:
            return a, c_max
    weights = excess / excess_sum
    
    # Compute effective cosine
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) % N / N)
    
    # Adaptive alpha
    alpha_eff = alpha * min(1.5, (c_max - target_peak) / max(c_max * 0.1, 1e-10) + 0.5)
    
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))
    
    # Reconstruct with magnitude modification
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    
    # Small random phase perturbation

        phase_noise = rng.normal(0, 0.02 * alpha_eff, N)
        A_new = A_new * np.exp(1j * phase_noise)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max

        top_i = np.argsort(contrib)[::-1][:n_coords]

        for idx in top_i:
            i = int(idx)
            j = lag - i
            if j < 0 or j >= n or i == j:
                continue
            if a[i] < 1e-10:
                continue

            orig = a[i]
            best_frac = 1.0
            best_cm = max(c[lag], c.max() * 0.99)

            fracs = np.array([0.4, 0.55, 0.7, 0.82, 0.92])
            for frac in fracs:
                a[i] = orig * frac
                delta = a[i] - orig
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    jk = k - i
                    if 0 <= jk < n:
                        if jk != i:
                            c[k] += 2.0 * delta * a[jk]
                        else:
                            c[k] += 2.0 * delta * orig + delta * delta
                new_cm = c.max()
                if new_cm < best_cm:
                    best_cm = new_cm
                    best_frac = frac
            # Apply best
            if best_frac != 1.0:
                a[i] = orig * best_frac
                delta = a[i] - orig
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    jk = k - i
                    if 0 <= jk < n:
                        if jk != i:
                            c[k] += 2.0 * delta * a[jk]
                        else:
                            c[k] += 2.0 * delta * orig + delta * delta
            else:
                a[i] = orig
                # Revert
                delta = orig - a[i]  # 0, no-op
    a = project_simplex(a, S)
    return a


def nesterov_topk(a, n, S, budget, rng, top_k_init=10, top_k_max=100):
    """Nesterov-accelerated projected gradient with top-k subgradient."""
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        s = S
    else:
        a = a * (S / s)
        s = a.sum()

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_conv(y).max()
    best_local = y.copy()
    it = 0

    while time.monotonic() - start < budget:
        it += 1
        tk = min(top_k_max, top_k_init + it // 15)
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_idx = np.argsort(c)[::-1][:tk]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        wts = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = wts[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_conv(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def spectral_optimize(a_init, n, N, budget, rng):
    """Main spectral optimization loop."""
    a = a_init.copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) * np.sqrt(2.0 * n)
        S = np.sqrt(2.0 * n)
    else:
        S = np.sqrt(2.0 * n)
        a = a * (S / a.sum())

    best_a = a.copy()
    best_cmax = fft_conv(a).max()
    best_score = 2.0 * n * best_cmax / (S ** 2)

    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < budget:
        it += 1
        top_k = min(60, 5 + it // 15)
        alpha = 0.7 / (1.0 + 0.01 * it)

        a_new, _ = spectral_shave(a, N, n, top_k, alpha, rng)
        c_new = fft_conv(a_new)
        cm_new = c_new.max()

        if cm_new < best_cmax:
            best_cmax = cm_new
            best_a = a_new.copy()
            best_score = 2.0 * n * best_cmax / (S ** 2)
            a = a_new.copy()
            stall = 0
            if it % 200 == 0:
                print(f"      iter {it}: score={best_score:.6f}")
        else:
            if cm_new < best_cmax * 1.002:
                a = a_new.copy()
            stall += 1

        if it % 8 == 0:
            c_curr = fft_conv(a)
            a = coord_shave(a, c_curr, n, S, top_peaks=4, n_coords=10, n_tries=4)
            c_chk = fft_conv(a)
            if c_chk.max() < best_cmax:
                best_cmax = c_chk.max()
                best_a = a.copy()
                best_score = 2.0 * n * best_cmax / (S ** 2)
                stall = 0
                if it % 160 == 0:
                    print(f"      iter {it} coord: score={best_score:.6f}")

        if stall > 40:
            # Annealing perturbation
            a_pad = np.zeros(N)
            a_pad[:n] = a
            A = fft(a_pad)
            sig = 0.03 * S / np.sqrt(n)
            noise = sig * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
            A_pert = A + noise
            a_pert = np.real(ifft(A_pert))[:n]
            a_pert = project_simplex(a_pert, S)
            c_pert = fft_conv(a_pert)
            if c_pert.max() < best_cmax:
                best_cmax = c_pert.max()
                best_a = a_pert.copy()
                best_score = 2.0 * n * best_cmax / (S ** 2)
                a = a_pert.copy()
                stall = 0
                print(f"      iter {it} perturb: score={best_score:.6f}")
            elif c_pert.max() < best_cmax * 1.03:
                T = max(0.005, 0.05 * np.exp(-0.005 * it))
                if rng.random() < np.exp(-(c_pert.max() - best_cmax) / (T * best_cmax)):
                    a = a_pert.copy()
            stall = 0

    return best_a, best_score


def upsample(a, n_from, n_to, rng):
    """Upsample a sequence by zero-insertion with small random fill."""
    a = a.astype(np.float64)
    a_new = np.zeros(n_to)
    step = n_to / n_from
    for i in range(n_from):
        pos = int(round(i * step))
        if pos < n_to:
            a_new[pos] = a[i]
    # Add small random values in gaps
    mask = a_new < 1e-10
    a_new[mask] = rng.random(mask.sum()) * 0.05
    return a_new


def structured_init(n, rng, method="sin"):
    t = np.arange(n) / n
    if method == "sin":
        a = (0.5 + 0.3 * np.sin(2 * np.pi * 3 * t)
             + 0.2 * np.sin(2 * np.pi * 7 * t)
             + 0.1 * np.sin(2 * np.pi * 13 * t)
             + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.4) + 0.08 * rng.standard_normal(n)
    elif method == "bump":
        a = (0.3 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 4) ** 2)
             + 0.2 * np.exp(-0.5 * ((t - 0.7) * 4) ** 2))
    elif method == "piecewise":
        n_pieces = min(32, n)
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, n // n_pieces)
        if len(a) < n:
            a = np.concatenate([a, rng.random(n - len(a)) * 0.7 + 0.3])
        a = a[:n]
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, 2.0)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def try_n(n, budget, rng, init_a=None):
    """Optimize for a given n. Returns (best_a, best_score)."""
    N = 1 << (2 * n - 1).bit_length()
    S = np.sqrt(2.0 * n)

    if init_a is not None and len(init_a) == n:
        a_init = safe_normalize(init_a, n)
        inits = [("given", a_init)]
    else:
        a_init = safe_normalize(structured_init(n, rng, "sin"), n)
        inits = [("sin", a_init)]

    best_a = a_init.copy()
    best_score = 2.0 * n * fft_conv(a_init).max() / (S ** 2)
    print(f"    n={n}: init score={best_score:.6f}")

    # Spectral optimization
    a_opt, sc_opt = spectral_optimize(best_a, n, N, budget * 0.55, rng)
    if 2.0 * n * fft_conv(a_opt).max() / (S ** 2) < best_score:
        best_score = 2.0 * n * fft_conv(a_opt).max() / (S ** 2)
        best_a = a_opt.copy()
        print(f"    n={n}: spec opt score={best_score:.6f}")

    # Nesterov polish
    rem = budget * 0.30
    a_ng, _ = nesterov_topk(best_a, n, S, rem, rng, top_k_init=15, top_k_max=120)
    sc_ng = 2.0 * n * fft_conv(a_ng).max() / (S ** 2)
    if sc_ng < best_score:
        best_score = sc_ng
        best_a = a_ng.copy()
        print(f"    n={n}: nesterov score={best_score:.6f}")

    # Final coord pass
    rem2 = budget * 0.15
    c_final = fft_conv(best_a)
    a_cd = coord_shave(best_a, c_final, n, S, top_peaks=6, n_coords=15, n_tries=5)
    sc_cd = 2.0 * n * fft_conv(a_cd).max() / (S ** 2)
    if sc_cd < best_score:
        best_score = sc_cd
        best_a = a_cd.copy()
        print(f"    n={n}: coord score={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-n strategy with spectral water-filling, coordinate shaving,
    and Nesterov-accelerated gradient descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Try to use previous best
    prev_a = None
    prev_n = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size > 100 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            prev_a = np.clip(given, 0, 1000)
            prev_n = len(prev_a)
            print(f"Previous sequence: n={prev_n}, sum={prev_a.sum():.2f}")
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Determine candidate n values
    n_candidates = []
    if prev_n is not None and 1024 <= prev_n <= 32768:
        n_candidates.append(prev_n)
    n_candidates.extend([4096, 8192, 16384])
    # Remove duplicates, keep order
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)
    n_candidates = n_unique
    print(f"Trying n values: {n_candidates}")

    # Phase 1: Initial optimization for each n (60% time)
    t0 = time.monotonic()
    phase1_budget = (deadline - t0) * 0.55
    n_trials = len(n_candidates)
    per_n = phase1_budget / n_trials

    for i_n, n in enumerate(n_candidates):
        if time.monotonic() > deadline - 30:
            break
        print(f"=== Phase 1: n={n} (budget={per_n:.0f}s) ===")
        a_init = None
        if prev_a is not None and n == prev_n:
            a_init = prev_a
        a_opt, sc = try_n(n, per_n, rng, a_init)
        if sc < global_best:
            global_best = sc
            global_a = a_opt.copy()
            print(f"  *** New best: n={n}, score={global_best:.6f} ***")

    # Phase 2: Focus on best n, more iterations (30% time)
    t1 = time.monotonic()
    phase2_budget = (deadline - t1) * 0.30
    if global_a is not None and phase2_budget > 10:
        best_n = len(global_a)
        N = 1 << (2 * best_n - 1).bit_length()
        S = np.sqrt(2.0 * best_n)
        print(f"=== Phase 2: focus n={best_n} (budget={phase2_budget:.0f}s) ===")

        # Multiple rounds of spectral + nesterov + coord
        a_curr = global_a.copy()
        for rnd in range(3):
            rem = (deadline - 30) - time.monotonic()
            if rem < 5:
                break
            seg = rem / max(1, 3 - rnd) * 0.9
            a_opt, _ = spectral_optimize(a_curr, best_n, N, seg * 0.6, rng)
            sc = 2.0 * best_n * fft_conv(a_opt).max() / (S ** 2)
            if sc < global_best:
                global_best = sc
                global_a = a_opt.copy()
                a_curr = a_opt.copy()
                print(f"  [P2 r{rnd}] spec: score={global_best:.6f}")

            rem2 = (deadline - 30) - time.monotonic()
            if rem2 > 3:
                a_ng, _ = nesterov_topk(a_curr, best_n, S, rem2 * 0.5, rng,
                                         top_k_init=20, top_k_max=150)
                sc_ng = 2.0 * best_n * fft_conv(a_ng).max() / (S ** 2)
                if sc_ng < global_best:
                    global_best = sc_ng
                    global_a = a_ng.copy()
                    a_curr = a_ng.copy()
                    print(f"  [P2 r{rnd}] nesterov: score={global_best:.6f}")

            rem3 = (deadline - 30) - time.monotonic()
            if rem3 > 2:
                c_f = fft_conv(a_curr)
                a_cd = coord_shave(a_curr, c_f, best_n, S, top_peaks=6, n_coords=15, n_tries=5)
                sc_cd = 2.0 * best_n * fft_conv(a_cd).max() / (S ** 2)
                if sc_cd < global_best:
                    global_best = sc_cd
                    global_a = a_cd.copy()
                    a_curr = a_cd.copy()
                    print(f"  [P2 r{rnd}] coord: score={global_best:.6f}")

    # Phase 3: Final polish (15% time)
    t2 = time.monotonic()
    rem_final = deadline - t2
    if global_a is not None and rem_final > 3:
        fn = len(global_a)
        S = np.sqrt(2.0 * fn)
        print(f"=== Phase 3: final polish (budget={rem_final:.0f}s) ===")

        # Long Nesterov
        seg = rem_final * 0.4
        a_p, _ = nesterov_topk(global_a, fn, S, seg, rng, top_k_init=30, top_k_max=200)
        sc_p = 2.0 * fn * fft_conv(a_p).max() / (S ** 2)
        if sc_p < global_best:
            global_best = sc_p
            global_a = a_p.copy()
            print(f"  [P3] nesterov: score={global_best:.6f}")

        # Final coord
        rem_f = deadline - time.monotonic()
        if rem_f > 2:
            c_f = fft_conv(global_a)
            a_cd = coord_shave(global_a, c_f, fn, S, top_peaks=8, n_coords=20, n_tries=5)
            sc_cd = 2.0 * fn * fft_conv(a_cd).max() / (S ** 2)
            if sc_cd < global_best:
                global_best = sc_cd
                global_a = a_cd.copy()
                print(f"  [P3] coord: score={global_best:.6f}")

        # One more spectral pass
        rem_g = deadline - time.monotonic()
        if rem_g > 3:
            N = 1 << (2 * fn - 1).bit_length()
            a_sp, _ = spectral_optimize(global_a, fn, N, rem_g, rng)
            sc_sp = 2.0 * fn * fft_conv(a_sp).max() / (S ** 2)
            if sc_sp < global_best:
                global_best = sc_sp
                global_a = a_sp.copy()
                print(f"  [P3] spectral: score={global_best:.6f}")

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"Verification: {ver:.6f}" if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print(f"Returning n={len(global_a)}, score={global_best:.6f}")
    return [float(x) for x in global_a]
