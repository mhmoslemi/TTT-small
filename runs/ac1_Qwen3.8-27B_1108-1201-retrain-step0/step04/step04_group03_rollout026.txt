"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses subgradient descent, LP best-response (PDP1), and coordinate descent.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def subgradient_direction(f, c, beta=50.0):
    """
    Compute the softmax-weighted subgradient direction.
    Returns a vector of length n: grad[i] = sum_k w[k]*f[k-i]
    """
    n = len(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w /= tot
    
    # Compute g[i] = sum_k w[k] * f[k-i] for i = 0, ..., n-1
    # Using convolution: g[i] = conv(w_rev, f)[len(w)-1-i]
    fsize = 1 << (n + len(w) - 1).bit_length()
    w_rev = w[::-1]
    conv = np.real(ifft(fft(w_rev, fsize) * fft(f, fsize)))
    idx = len(w) - 1 - np.arange(n)
    grad = conv[idx]
    return grad


def subgradient_step(f, c, lr, beta=50.0):
    """One subgradient descent step."""
    S = f.sum()
    grad = subgradient_direction(f, c, beta)
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    f_new = project_simplex(f_new, S)
    return f_new


def solve_lp_best_response(f, n, S, top_k=500, solver='PDSLP', time_limit=15.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Select top-k most binding lags
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix: rows[i, j] = f[idx[i] - j] if valid
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def lp_iteration(f, n, S, top_k=500, solver='PDP1'):
    """One LP iteration with best step size."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    
    g = solve_lp_best_response(f, n, S, top_k=top_k, solver=solver)
    if g is None:
        return f.copy(), M_old
    
    # Find best step size
    best_val = M_old
    best_t = 0.0
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        val = c_test.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), M_old


def multiplicative_update(f, c, lr=0.1, beta=50.0):
    """Multiplicative repulsion update."""
    n = len(f)
    S = f.sum()
    grad = subgradient_direction(f, c, beta)
    # Multiplicative update: f[i] *= exp(-lr * grad[i] / max(f))
    scale = max(f.max(), 1e-10)
    f_new = f * np.exp(-lr * grad / scale)
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new

Actually, I want to think about one more thing: the step size for the subgradient method.

The subgradient of `max_k c[k]` has norm at most `2 * max(f)` (since each component is at most `2 * max(f)`). A good step size is `lr = c / (iter + 1)` or `lr = c / sqrt(iter)` for standard subgradient methods, where c is a constant.
