"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: FISTA with LSE smoothing + QP best-response corrections + local search polish.
Uses adaptive beta scheduling, stochastic subgradients, and multi-scale exploration.

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_lse_grad(a, c, beta, stochastic_frac=1.0, rng=None):
    """Gradient of LSE approximation, optionally using only top fraction of lags."""
    n = len(a)
    m = len(c)
    cm = c.max()
    if stochastic_frac < 1.0 and rng is not None:
        k_use = max(1, int(m * stochastic_frac))
        perm = rng.permutation(m)
        idx = perm[:k_use]
        e = np.zeros(m)
        c_sub = c[idx]
        e[idx] = np.exp(beta * (c_sub - cm))
        tot = e.sum()
        if tot < 1e-300:
            return np.zeros(n)
        w = e / tot
    else:
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            return np.zeros(n)
        w = e / tot
    fsize = 1 << (m + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def solve_qp_best_response(f, n, S, top_k=250, time_limit=30.0):
    """
    Solve QP: min ||g - f||^2 s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    This gives a stable "best response" that stays close to f.
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Find top-k most binding lags
    idx = np.argsort(c)[::-1][:top_k]

    # Build constraint matrix (top_k x n)
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    # Objective: minimize ||g - f||^2 (equivalently, maximize sum(g) - (1/2)||g||^2 for a fixed offset)
    # Actually let's use: minimize (g - f)^T (g - f)
    obj = cp.Minimize(cp.sum_squares(g_var - f))
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(obj, constraints)

    start = time.monotonic()
    solvers_to_try = ['ECOS', 'SCS', 'CLARABEL']
    result = None
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, verbose=False)
            result = prob.status
            break
        except Exception:
            continue
        if time.monotonic() - start > time_limit:
            break

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def solve_lp_best_response(f, n, S, top_k=250, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    """
    c = fft_convolve_self(f)
    M = c.max()

    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    for solver in ['SCS', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue
        if time.monotonic() - start > time_limit:
            break

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def multiplicative_best_response(f, n, S, iters=50, eta=0.5):
    """
    Fast multiplicative approximation to the LP best response.
    Uses exponential weights on the most violated constraints.
    """
    g = np.ones(n) * S / n
    c_f = fft_convolve_self(f)
    M = c_f.max()

    for _ in range(iters):
        # Compute f * g
        fsize = 1 << (2 * n - 1).bit_length()
        F = fft(f, fsize)
        G = fft(g, fsize)
        fg = np.real(ifft(F * G))[:2 * n - 1]

        # Find most violated constraint
        k_star = np.argmax(fg)
        violation = fg[k_star] - M

        if violation < 1e-10 * M:
            break

        # Multiplicative update: reduce g where f[k_star - j] is large
        # The gradient of (f*g)(k_star) w.r.t. g is f[k_star - j]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        grad = f[k_star - js]

        # Update
        step = eta * grad / (M + 1e-10)
        g[js] *= np.exp(-step)

        # Renormalize
        g = np.maximum(g, 1e-15)
        g = g * (S / g.sum())

    return g


def local_search(f, n, S, max_iter=5000, rng=None, step_scale=0.01):
    """
    Local coordinate descent: try perturbing individual elements.
    """
    f = f.copy()
    c = fft_convolve_self(f)
    best_val = c.max()

    if rng is None:
        rng = np.random.default_rng(0)

    for it in range(max_iter):
        # Try random coordinate
        j = rng.integers(0, n)
        delta = step_scale * S / np.sqrt(n) * rng.normal(0, 1)
        f_new = f.copy()
        f_new[j] = max(0.0, f_new[j] + delta)
        f_new = project_simplex(f_new, S)

        # Only recompute the affected part of the convolution
        # For efficiency, just do full recompute (FFT is fast enough)
        c_new = fft_convolve_self(f_new)
        val_new = c_new.max()

        if val_new < best_val:
            f = f_new
            c = c_new
            best_val = val_new

    return f, best_val


def run_fista_advanced(a_init, n, S, max_time, rng,
                       beta_init=15.0, beta_rate=0.002, beta_max=1500.0,
                        use_stochastic=False, stochastic_frac=0.3,
                        use_repulsion=True, repulsion_freq=300, repulsion_lr=0.04,
                        momentum_reset_freq=2000):
    """Advanced FISTA with LSE smoothing, stochastic gradients, and repulsion."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: increase when improving, reset when stuck
        if it - last_improve_it > 500:
            # Stuck: either increase beta to sharpen, or add noise
            if beta < beta_max * 0.5:
                beta *= 1.5
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve_it = it
        else:
            beta = min(beta_max, beta * (1 + beta_rate))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Compute gradient (stochastic or full LSE)
        if use_stochastic and it % 3 == 0:
            grad = compute_lse_grad(y, c, beta, stochastic_frac=stochastic_frac, rng=rng)
        else:
            grad = compute_lse_grad(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        # Adaptive step reduction
        if it - last_improve_it > 1000:
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum with warm restart
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new

        # Warm restart: if y is "behind" a, reset momentum
        if it % momentum_reset_freq == 0:
            t_mom = 1.0

        t_mom = t_new

        # Periodic multiplicative repulsion
        if use_repulsion and it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(30.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 1e-15)
                y = project_simplex(y, S)
                t_mom = 1.0

        # Periodic noise injection (decreasing)
        if it % 4000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

            print("  Phase 2 best: %.6f (corrections: %d)" % (global_best, ncorr))

    # Phase 3: Try a slightly different size
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        alt_n = n  # Keep same size for consistency
        # Try a chirp initialization at same size
        tt = np.linspace(0, 1, n, endpoint=False)
        chirp = np.exp(2 * np.pi * tt * 3)  # Frequency chirp
        chirp = np.maximum(chirp, 0.01)
        chirp = normalize_sum(chirp, n)

        phase3_time = remaining * 0.25
        if phase3_time > 30:
            print("=== Phase 3: Chirp init FISTA (n=%d, %.0fs) ===" % (n, phase3_time))
            a3, s3 = run_fista_advanced(chirp, n, S, phase3_time, rng,
                                         beta_init=10.0, beta_rate=0.003, beta_max=800.0,
                                         use_repulsion=True, repulsion_freq=200, repulsion_lr=0.04,
                                         use_stochastic=True, stochastic_frac=0.5)
            a3 = normalize_sum(a3, n)
            c3 = fft_convolve_self(a3)
            obj3 = 2.0 * n * c3.max() / (a3.sum() ** 2)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()
                print("  Phase 3 (chirp) improved: %.6f" % global_best)
            else:
                print("  Phase 3 (chirp): %.6f (no improvement vs %.6f)" % (obj3, global_best))

    # Phase 4: Intensive polish on the best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_p = len(global_a)
            S_p = np.sqrt(2.0 * n_p)
            phase4_time = remaining * 0.4
            print("=== Phase 4: Intensive polish (n=%d, %.0fs) ===" % (n_p, phase4_time))

            # High-beta FISTA
            a4, s4 = run_fista_advanced(global_a, n_p, S_p, phase4_time, rng,
                                         beta_init=80.0, beta_rate=0.001, beta_max=3000.0,
                                         use_repulsion=False,
                                         use_stochastic=False)
            a4 = normalize_sum(a4, n_p)
            c4 = fft_convolve_self(a4)
            obj4 = 2.0 * n_p * c4.max() / (a4.sum() ** 2)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                print("  Phase 4a best: %.6f" % global_best)

            # LP correction
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                print("  Phase 4b: LP correction...")
                g_lp = solve_lp_best_response(global_a, n_p, S_p, top_k=300, time_limit=min(40.0, remaining2 * 0.5))
                if g_lp is not None:
                    for frac in [1.0, 0.8, 0.6, 0.4, 0.2]:
                        y_test = (1 - frac) * global_a + frac * g_lp
                        y_test = np.maximum(y_test, 0)
                        y_test = project_simplex(y_test, S_p)
                        c_test = fft_convolve_self(y_test)
                        obj_test = 2.0 * n_p * c_test.max() / (y_test.sum() ** 2)
                        if obj_test < global_best:
                            global_best = obj_test
                            global_a = y_test.copy()
                            print("    LP frac=%.1f: %.6f" % (frac, global_best))
                            break

    # Phase 5: Local search polish
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_l = len(global_a)
        S_l = np.sqrt(2.0 * n_l)
        if remaining > 30:
            ls_time = min(60.0, remaining * 0.3)
            print("=== Phase 5: Local search (%.0fs) ===" % ls_time)
            # Convert time budget to iterations
            max_ls_iter = int(ls_time * 50)  # ~50 iterations per second estimate
            a5, s5 = local_search(global_a, n_l, S_l, max_iter=min(max_ls_iter, 20000),
                                   rng=rng, step_scale=0.008)
            c5 = fft_convolve_self(a5)
            obj5 = 2.0 * n_l * c5.max() / (a5.sum() ** 2)
            if obj5 < global_best:
                global_best = obj5
                global_a = a5.copy()
                print("  Local search improved: %.6f" % global_best)
            else:
                print("  Local search: %.6f (no improvement)" % obj5)

    # Phase 6: Final verification and cleanup
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0

        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final score: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]

    else:
        # Fallback: uniform sequence
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback score: %.6f" % ver)

    print("Returning sequence of length %d, best score=%.6f" % (len(final_list), global_best))
    return final_list
