def gerchberg_saxton(a_init, n_iter, rng):
    n = len(a_init)
    N = 1 << (2*n-1).bit_length()
    S = a_init.sum()
    a = a_init.copy()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        # Step 1: FFT
        A = fft(a, N)
        
        # Step 2: Enforce flat magnitude in frequency domain
        # Keep phase, set magnitude to 1 (or to target)
        phase = np.angle(A)
        A_flat = np.exp(1j * phase)  # |A| = 1 everywhere
        
        # Step 3: IFFT
        a_new = np.real(ifft(A_flat))[:n]
        
        # Step 4: Enforce non-negativity and sum constraint
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def ps2(n):
    """Next power of 2 >= n."""
    return 1 << max(0, (n-1).bit_length())

def compute_conv(a):
    n = len(a)
    N = ps2(2*n - 1)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def compute_score(a):
    n = len(a)
    c = compute_conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Simple approach: clamp and rescale (not exact projection but fast)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S/n)
    return v * (S / s)

def project_simplex_exact(v, S):
    """Exact Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max() + 1)
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def smoothed_grad(a, w, N):
    """
    Compute gradient of (1/beta)*logsumexp(beta*c) w.r.t. a.
    w = softmax(beta * c), c = conv(a,a)
    Returns grad of shape (n,).
    """
    n = len(a)
    size = ps2(2*n - 1 + n - 1)  # for conv of w (len 2n-1) with a_rev (len n)
    W = fft(w, size)
    Ar = fft(a[::-1], size)
    conv = np.real(ifft(W * Ar))
    return 2.0 * conv[n-1 : 2*n-1]

def gs_step(a, N, omega=1.0, target_mag=None):
    """One Gerchberg-Saxton iteration."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    if target_mag is None:
        target_mag = np.ones(N) * (np.sum(mag) / N)
    A_new = (1 - omega) * A + omega * target_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * (S / s)
    return a_new

    return float(2.0 * n * c.max() / (S * S))


def project_simplex_fast(v, S):
    """Fast (non-exact) projection to {x>=0, sum(x)=S}: clamp and rescale."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(len(v), S / len(v))
    return v * (S / s)


def project_simplex_exact(v, S):
    """Exact Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max() + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, w, N):
    """
    Gradient of sum_k w[k]*c[k] where c = conv(a,a).
    grad[i] = 2 * sum_k w[k] * a[k-i] (with a[m]=0 out of range).
    Computed via FFT: 2 * convolve(w, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    size = ps2(2 * n - 1 + n - 1)  # full conv of len 2n-1 with len n gives len 3n-2
    W = fft(w, size)
    Ar = fft(a[::-1], size)
    conv_result = np.real(ifft(W * Ar))
    return 2.0 * conv_result[n - 1:2 * n - 1]


def verify_gradient(a, beta, rng):
    """Numerical gradient check (for debugging)."""
    n = len(a)
    N = ps2(2 * n - 1)
    c = compute_conv(a)
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w /= w.sum()
    grad_analytic = compute_smoothed_gradient(a, w, N)

    eps = 1e-6
    grad_numeric = np.zeros(n)
    f0 = np.sum(w * c)
    for i in range(min(5, n)):
        a_pert = a.copy()
        a_pert[i] += eps
        c_pert = compute_conv(a_pert)
        grad_numeric[i] = (np.sum(w * c_pert) - f0) / eps
        a_pert[i] -= 2 * eps
        c_pert2 = compute_conv(a_pert)
        grad_numeric[i] = (np.sum(w * c) - np.sum(w * c_pert2)) / (2 * eps)

    diff = np.abs(grad_analytic[:min(5, n)] - grad_numeric[:min(5, n)])
    print("  Gradient check: max diff = %.2e" % diff.max())
    return diff.max() < 1e-3


def gs_step(a, N, omega=1.0):
    """One Gerchberg-Saxton iteration: flatten spectrum, enforce non-negativity."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    target_mag = np.ones(N) * mag.mean()
    A_new = (1.0 - omega) * A + omega * target_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def optimize_gs(a0, N, n_iter, S, rng):
    """Run GS iterations with relaxation and periodic restarts."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        omega = 0.7 if it < n_iter // 3 else (0.5 if it < 2 * n_iter // 3 else 0.3)
        a = gs_step(a, N, omega)

        if it % 200 == 199:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 1000 == 500 and rng.random() < 0.25:
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex_fast(a + rng.normal(0, noise_scale, n), S)

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
