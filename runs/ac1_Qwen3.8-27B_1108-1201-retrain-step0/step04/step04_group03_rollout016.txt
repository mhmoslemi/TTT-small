import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        # Distribute remainder
        nonzero = result > 0
        if nonzero.sum() > 0:
            result[nonzero] += diff / nonzero.sum()
            result = np.maximum(result, 0)
    return result


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_smooth_max(c, beta):

    cm = c.max()
    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta

def solve_lp_topk(f, n, S, top_k, solver='SCS', time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Get top-k lags
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix rows
    # (f*g)(k) = sum_j f(k-j) * g(j) for j in [max(0,k-n+1), min(k,n-1)]
    A_rows = []
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        row = np.zeros(n)
        row[lo:hi+1] = f[k - hi : k - lo : -1]  # f(k-j) for j=lo..hi
        A_rows.append(row)
    A = np.array(A_rows)
    
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False, max_iter=100000)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    elapsed = time.monotonic() - start
    if elapsed > time_limit * 3:
        return None
    
    if g_var.value is None:
        return None
    g = np.maximum(g_var.value, 0)
    if g.sum() < 1e-10:
        return None
    return g * (S / g.sum())

def fista_optimize(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=3000.0, beta_rate=0.003):
    """FISTA with LSE smoothing."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(c, y, beta)
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best

        g = solve_lp_topk(f, n, S, top_k=top_k, solver=solver, time_limit=8.0)
    
    if g is None:
        return f.copy(), M_old
    
    # Bisection for best step size
    def eval_step(t):
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()
    
    val1 = eval_step(1.0)
    if val1 <= M_old:
        f_new = project_simplex(g, S)
        return f_new, eval_step(1.0)
    
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = M_old
    for _ in range(10):
        mid = (lo + hi) / 2
        val_mid = eval_step(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid
    
    if best_t > 0.005:
        f_new = (1 - best_t) * f + best_t * g
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), M_old


def multiplicative_update(f, n, S, beta=50.0, lr=0.05, top_k=None):
    """Multiplicative repulsion update focusing on peaks."""
    c = fft_convolve_self(f)
    cm = c.max()
    
    if top_k is not None:
        # Only focus on top-k peaks
        w = np.zeros_like(c)
        top_idx = np.argsort(c)[::-1][:top_k]
        w[top_idx] = np.exp(beta * (c[top_idx] - cm))
    else:
        w = np.exp(beta * (c - cm))
    
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    
    # Gradient contribution
    w_flip = w[::-1]
    fsize = 1 << (len(w_flip) + n - 1).bit_length()
    W = fft(w_flip, fsize)
    A = fft(f, fsize)
    rep = 2.0 * np.real(ifft(W * A))[:n]
    
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Iterative LP best-response with cutting planes + FISTA polishing
+ multiplicative peak-shaving. Multiple lengths tried with interpolation.
Key innovations over previous:
1. Cutting-plane LP (adaptively adds violated constraints)
2. Targeted multiplicative updates on actual peaks (not smooth approx)
3. Aggressive diversity restarts
4. Multi-length exploration with proper interpolation

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        nonzero = result > 0
        if nonzero.sum() > 0:
            result[nonzero] += diff / nonzero.sum()
            result = np.maximum(result, 0)
    return result


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(c, a, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    w_flip = w[::-1]
    fsize = 1 << (len(w_flip) + n - 1).bit_length()
    W = fft(w_flip, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad


def fista_optimize(a_init, n, S, max_time, rng, beta_init=30.0, beta_max=3000.0, beta_rate=0.003):
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        grad = lse_gradient(c, y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom_new if (t_mom_new := t_new) else t_mom

    return best_a, best


def build_lp_matrix(f, n, lags):
    """Build constraint matrix rows for given lags."""
    rows = []
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        row = np.zeros(n)
        # (f*g)(k) = sum_j f(k-j) g(j) for j in [lo, hi]
        # f(k-j) for j=lo..hi is f(k-hi), f(k-hi+1), ..., f(k-lo)
        row[lo:hi + 1] = f[k - hi:k - lo + 1]
        rows.append(row)
    return np.array(rows)


def solve_lp_with_constraints(f, n, S, A, M, solver='SCS', max_iter=500000):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g normalized to sum S."""
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False, max_iter=max_iter)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False, max_iter=max_iter)
        except Exception:
            return None

    if g_var.value is None:
        return None
    g = np.maximum(g_var.value, 0)
    if g.sum() < 1e-10:
        return None
    return g * (S / g.sum())


def lp_best_response(f, n, S, top_k=80, solver='SCS', time_limit=8.0, use_cutting=True):
    """
    LP best response with optional cutting plane.
    Returns (g, M) or (None, M).
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Start with top constraints
    idx = np.argsort(c)[::-1][:top_k]
    A = build_lp_matrix(f, n, idx)

    g = None
    start = time.monotonic()

    if use_cutting:
        # Cutting plane: add violated constraints
        max_rounds = min(5, max(1, int(time_limit / 2)))
        for rnd in range(max_rounds):
            if time.monotonic() - start > time_limit:
                break
            g = solve_lp_with_constraints(f, n, S, A, M, solver=solver)
            if g is None:
                break

            # Check violations
            fsize = 1 << (2 * n - 1).bit_length()
            F = fft(f, fsize)
            G = fft(g, fsize)
            conv = np.real(ifft(F * G))[:2 * n - 1]

            violations = conv > M * 1.001 + 1e-10
            if not np.any(violations):
                break

            # Add worst violated
            worst = np.argmax(conv)
            if worst not in idx:
                idx = np.append(idx, worst)
                A = build_lp_matrix(f, n, idx)
            else:
                break
    else:
        g = solve_lp_with_constraints(f, n, S, A, M, solver=solver)

    if g is None:
        return None, M
    return g, M


def lp_iteration(f, n, S, top_k=80, solver='SCS', use_cutting=True, time_limit=8.0):
    """Full LP iteration: solve LP + bisection step size."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()

    g, M = lp_best_response(f, n, S, top_k=top_k, solver=solver,
                            use_cutting=use_cutting, time_limit=time_limit)
    if g is None:
        return f.copy(), M_old

    # Bisection for step size
    def eval_step(t):
        f_test = (1.0 - t) * f + t * g
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()

    val1 = eval_step(1.0)
    if val1 <= M_old * 1.0001:
        return project_simplex(g, S), val1

    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = M_old
    for _ in range(12):
        mid = (lo + hi) / 2.0
        val_mid = eval_step(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0.003:
        f_new = project_simplex((1.0 - best_t) * f + best_t * g, S)
        return f_new, best_val

    return f.copy(), M_old


def multiplicative_peak_shave(f, n, S, beta=80.0, lr=0.04, top_k=30):
    """Aggressive multiplicative update targeting only the top peaks."""
    c = fft_convolve_self(f)
    cm = c.max()

    # Focus on top-k peaks only
    w = np.zeros_like(c)
    if top_k is not None and top_k > 0:
        top_idx = np.argsort(c)[::-1][:top_k]
        w[top_idx] = np.exp(beta * (c[top_idx] - cm))
    else:
        w = np.exp(beta * (c - cm))

    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot

    w_flip = w[::-1]
    fsize = 1 << (len(w_flip) + n - 1).bit_length()
    W = fft(w_flip, fsize)
    A = fft(f, fsize)
    rep = 2.0 * np.real(ifft(W * A))[:n]

    # Multiplicative update with clamping
    ratio = rep / max(f.max(), 1e-10)
    f_new = f * np.exp(-lr * ratio)
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def interpolate_seq(a, new_n):
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def diverse_init(n, rng):
    """Generate a diverse initialization."""
    choice = rng.integers(0, 5)
    t = np.linspace(0, 1, n, endpoint=False)
    if choice == 0:
        return np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t + rng.uniform(0, 2 * np.pi)), 0.01)
    elif choice == 1:
        return np.exp(-3.0 * (t - 0.5) ** 2 / 0.2)
    elif choice == 2:
        return np.sin(np.pi * t) ** 2 + 0.05
    elif choice == 3:
        return 0.5 + 0.3 * np.sin(4 * np.pi * t) + 0.2 * np.cos(7 * np.pi * t + 0.5)
    else:
        return np.exp(rng.normal(0, 0.4, n))
    return np.ones(n) * 0.1


def propose_candidate(seed=42, budget_s=2400):
    """
    Minimize 2n*max(f*f)/sum(f)^2 using hybrid LP + FISTA + peak shaving.

    Phases:
    1. FISTA warmup on given sequence
    2. LP iterations with cutting planes (main optimization)
    3. Peak shaving + FISTA polish
    4. Multi-length exploration
    5. Final LP refinement
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 90.0)

    global_best = np.inf
    global_a = None

    # Load initial sequence
    init_seq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if h.size >= 128 and h.sum() > 0:
            init_seq = np.clip(h, 0, 1000)
    except Exception:
        pass

    # Determine main length
    if init_seq is not None:
        main_n = len(init_seq)
    else:
        main_n = 8192

    S = np.sqrt(2.0 * main_n)
    print("Phase 0: n=%d, S=%.4f, budget=%.0fs" % (main_n, S, budget_s))

    # === Phase 1: FISTA Warmup ===
    remaining = deadline - time.monotonic()
    t1 = min(90.0, remaining * 0.08)
    print("=== Phase 1: FISTA warmup (%.0fs) ===" % t1)

    if init_seq is not None:
        a1, s1 = fista_optimize(init_seq, main_n, S, t1, rng,
                                beta_init=50.0, beta_max=1500.0, beta_rate=0.004)
    else:
        a1 = np.ones(main_n) * (S / main_n)
        a1, s1 = fista_optimize(a1, main_n, S, t1, rng,
                                beta_init=20.0, beta_max=800.0, beta_rate=0.003)

    a1 = normalize_to_sum(a1, S)
    obj1 = 2.0 * main_n * s1 / S ** 2
    global_best = obj1
    global_a = a1.copy()
    print("  Phase 1: obj=%.6f" % global_best)

    # === Phase 2: LP Iterations (Main) ===
    remaining = deadline - time.monotonic()
    t2_budget = remaining * 0.45
    print("=== Phase 2: LP iterations (%.0fs budget) ===" % t2_budget)

    f = global_a.copy()
    it = 0
    lp_count = 0
    stall_count = 0
    prev_best = global_best

    while time.monotonic() < deadline - t2_budget * 0.55:
        it += 1

        # Choose LP parameters adaptively
        if it <= 10:
            top_k = 60
            use_cut = True
        elif it <= 30:
            top_k = 80
            use_cut = (it % 3 == 0)
        else:
            top_k = 100
            use_cut = (it % 2 == 0)

        f_new, M_new = lp_iteration(f, main_n, S, top_k=top_k,
                                     solver='SCS', use_cutting=use_cut, time_limit=7.0)
        lp_count += 1

        obj_new = 2.0 * main_n * M_new / S ** 2
        if obj_new < global_best - 1e-9:
            global_best = obj_new
            global_a = f_new.copy()
            stall_count = 0
            if it % 5 == 0:
                print("  LP iter %d: obj=%.6f" % (it, global_best))
        else:
            stall_count += 1

        f = f_new

        # Periodic FISTA micro-polish
        if it % 8 == 0:
            c_cur = fft_convolve_self(f)
            M_cur = c_cur.max()
            f_temp, M_temp = fista_optimize(f, main_n, S, 8.0, rng,
                                            beta_init=200.0, beta_max=3000.0, beta_rate=0.004)
            if M_temp < M_cur * 0.9999:
                f = f_temp
                obj_temp = 2.0 * main_n * M_temp / S ** 2
                if obj_temp < global_best:
                    global_best = obj_temp
                    global_a = f_temp.copy()
                    stall_count = 0

        # Periodic peak shaving
        if it % 12 == 0:
            f_temp = multiplicative_peak_shave(f, main_n, S, beta=100.0, lr=0.03, top_k=20)
            c_temp = fft_convolve_self(f_temp)
            if c_temp.max() < M_new * 0.9999:
                f = f_temp
                obj_temp = 2.0 * main_n * c_temp.max() / S ** 2
                if obj_temp < global_best:
                    global_best = obj_temp
                    global_a = f_temp.copy()
                    stall_count = 0

        # Diversity restart if stuck
        if stall_count > 20:
            strength = min(0.1, 0.02 * (stall_count - 20) / 10.0)
            noise = rng.normal(0, strength * S / np.sqrt(main_n), main_n)
            f = project_simplex(f + noise, S)
            # Also try a more aggressive restart every few stalls
            if stall_count > 40:
                f_alt = diverse_init(main_n, rng)
                f_alt = normalize_to_sum(f_alt, S)
                f_alt, M_alt = fista_optimize(f_alt, main_n, S, 15.0, rng,
                                              beta_init=30.0, beta_max=500.0, beta_rate=0.003)
                obj_alt = 2.0 * main_n * M_alt / S ** 2
                if obj_alt < global_best:
                    global_best = obj_alt
                    global_a = f_alt.copy()
                    f = f_alt.copy()
                    print("  Diversity restart found better: %.6f" % global_best)
                stall_count = 0

            if it % 30 == 0:
                print("  Iter %d: stall=%d, best=%.6f" % (it, stall_count, global_best))

    print("  Phase 2 done: %d LP iters, best=%.6f" % (lp_count, global_best))

    # === Phase 3: Peak Shaving + FISTA Polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        t3 = min(remaining * 0.2, 150.0)
        print("=== Phase 3: Polish (%.0fs, n=%d) ===" % (t3, n3))

        # High-beta FISTA
        a3, s3 = fista_optimize(global_a, n3, S3, t3 * 0.5, rng,
                                beta_init=500.0, beta_max=8000.0, beta_rate=0.001)
        obj3 = 2.0 * n3 * s3 / S3 ** 2
        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()
            print("  FISTA polish: %.6f" % global_best)

        # Peak shaving iterations
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            f = global_a.copy()
            for ps_it in range(20):
                if time.monotonic() > deadline - 30:
                    break
                f_new = multiplicative_peak_shave(f, n3, S3, beta=120.0, lr=0.02, top_k=15)
                c_new = fft_convolve_self(f_new)
                obj_new = 2.0 * n3 * c_new.max() / S3 ** 2
                if obj_new < global_best:
                    global_best = obj_new
                    global_a = f_new.copy()
                    f = f_new
                    print("  Peak shave %d: %.6f" % (ps_it, global_best))
                else:
                    f = f_new  # still continue, might help next iteration

    # === Phase 4: Multi-length Exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and global_a is not None:
        base_n = len(global_a)
        candidates = []
        if base_n <= 8192:
            candidates.append(base_n * 2)
        if base_n >= 4096:
            candidates.append(base_n // 2)
        if base_n == 8192:
            candidates.append(16384)

        for alt_n in candidates:
            if not (2048 <= alt_n <= 32768):
                continue
            if remaining < 60:
                break

            S_alt = np.sqrt(2.0 * alt_n)
            alt_init = interpolate_seq(global_a, alt_n)
            alt_init = normalize_to_sum(alt_init, S_alt)

            t4 = min(remaining * 0.35, 120.0)
            print("=== Phase 4: n=%d (%.0fs) ===" % (alt_n, t4))

            a4, s4 = fista_optimize(alt_init, alt_n, S_alt, t4, rng,
                                     beta_init=50.0, beta_max=2000.0, beta_rate=0.003)
            obj4 = 2.0 * alt_n * s4 / S_alt ** 2

            # Quick LP on this
            if time.monotonic() < deadline - 30:
                f4 = a4.copy()
                for lp4_it in range(5):
                    if time.monotonic() > deadline - 30:
                        break
                    f4_new, M4 = lp_iteration(f4, alt_n, S_alt, top_k=60,
                                              solver='SCS', use_cutting=True, time_limit=6.0)
                    obj4_new = 2.0 * alt_n * M4 / S_alt ** 2
                    if obj4_new < obj4:
                        f4 = f4_new
                        obj4 = obj4_new

            if obj4 < global_best:
                global_best = obj4
                global_a = f4.copy()
                print("  Phase 4 (n=%d): %.6f (NEW BEST)" % (alt_n, global_best))
            else:
                print("  Phase 4 (n=%d): %.6f" % (alt_n, obj4))

            remaining = deadline - time.monotonic()

    # === Phase 5: Final LP Refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        print("=== Phase 5: Final LP (%.0fs, n=%d) ===" % (remaining - 30, n5))

        f5 = global_a.copy()
        for it5 in range(30):
            if time.monotonic() > deadline - 30:
                break
            f5_new, M5 = lp_iteration(f5, n5, S5, top_k=100,
                                       solver='SCS', use_cutting=True, time_limit=6.0)
            obj5 = 2.0 * n5 * M5 / S5 ** 2
            if obj5 < global_best:
                global_best = obj5
                global_a = f5_new.copy()
                f5 = f5_new
                print("  Final LP %d: %.6f" % (it5, global_best))
            else:
                f5 = f5_new

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        if global_a.sum() < 0.01:
            print("  Sum too small, using fallback")
            final_list = [1.0] * 128
    else:
        final_list = [1.0] * 128

    ver = evaluate_sequence(final_list)
    if not np.isfinite(ver):
        print("  Evaluation failed, using safe fallback")
        safe = np.ones(256, dtype=np.float64) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in safe]
        ver = evaluate_sequence(final_list)

    print("FINAL: len=%d, internal=%.6f, evaluate_sequence=%.6f" % (len(final_list), global_best, ver))
    return final_list
