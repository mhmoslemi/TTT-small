"""
Minimize R = 2n*max(self_conv)/sum^2 using adaptive L-BFGS with peak-focused gradients,
spectral shaping, and multi-phase optimization. Key improvements over previous:
1. Peak-focused gradient (top-k with excess weighting) instead of full LSE for later phases.
2. Adaptive step sizing with proper backtracking.

4. Aggressive spectral reshaping between optimization blocks.
5. Coordinate-descent-style local search for final polish.
6. Brief multi-length exploration (n=4096, 8192, 16384).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N, n):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_gradient(a, N, n, beta):
    """Gradient of LSE-smoothed max of conv(a,a). Returns (grad, c, c_max)."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c, c_max
    w = exp_bc / total
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c, c_max


def topk_gradient(a, N, n, top_k, margin_frac=0.05):
    """Peak-focused gradient: top-k lags weighted by excess over (c_max - margin)."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    margin = margin_frac * c_max
    target = c_max - margin
    weights = np.maximum(c - target, 0.0)
    # Focus on top-k
    if top_k < 2 * n - 1:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[top_idx] = True
        weights = np.where(mask, weights, 0.0)
    total = weights.sum()
    if total < 1e-30:
        return np.zeros(n), c, c_max
    weights = weights / total
    W = fft(weights, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18) if gn >= 1e-18 else np.zeros(n)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-10, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def spectral_step(a, N, n, top_k, alpha):
    """One spectral shaping step: reduce power at frequencies that constructively interfere at peaks."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    S = a.sum()
    return project_simplex(a_new_full, S)


def spectral_block(a_init, n, S, N, max_time, rng, top_k=10, alpha=0.12):
    """Spectral shaping block."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        a = spectral_step(a, N, n, top_k, alpha)
        c = compute_conv(a, N, n)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
        if it % 200 == 100 and rng.random() < 0.25:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def coord_descent_block(a_init, n, S, N, max_time, rng):
    """Coordinate descent: perturb groups of elements and keep improvements."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N, n).max()
    start_time = time.monotonic()
    it = 0
    group_size = max(1, n // 64)

    while time.monotonic() - start_time < max_time:
        it += 1
        # Pick a random group
        start = rng.integers(0, n - group_size)
        end = start + group_size
        # Perturb the group
        delta = rng.normal(0, 0.01 * S / n, group_size)
        a_new = a.copy()
        a_new[start:end] = a_new[start:end] + delta
        a_new = project_simplex(a_new, S)
        c_new = compute_conv(a_new, N, n)
        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best_a = a_new.copy()
            a = a_new
        # Occasionally try a larger perturbation
        if it % 10 == 0:
            group_size_large = max(4, n // 16)
            start2 = rng.integers(0, n - group_size_large)
            end2 = start2 + group_size_large
            a_try = a.copy()
            a_try[start2:end2] += rng.normal(0, 0.02 * S / n, group_size_large)
            a_try = project_simplex(a_try, S)
            c_try = compute_conv(a_try, N, n)
            if c_try.max() < best_cmax:
                best_cmax = c_try.max()
                best_a = a_try.copy()
                a = a_try

    return best_a, best_cmax


def make_init_ramp(n, rng, anti_sym=True):
    """Linear ramp initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    if anti_sym:
        a = 0.5 + 0.5 * (2 * t - 1) + 0.1 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
