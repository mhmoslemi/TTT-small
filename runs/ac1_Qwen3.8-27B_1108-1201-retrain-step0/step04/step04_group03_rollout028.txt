```python
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def build_constraint_rows(f, n, k_indices):
    """Build sparse constraint rows for given lag indices."""
    n_constraints = len(k_indices)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        values = f[k - js]
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(values.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))

def solve_lp_best_response(f, n, S, active_lags, M, solver='ECOS'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    g_var = cp.Variable(n)
    constraints = [g_var >= 0]
    
    # Build constraints
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        expr = f[k - js] @ g_var[js]
        constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
            # If SCS, maybe it's slow, try ECOS
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                return None
    
    if g_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        return g / g_sum  # Normalize to sum = 1, caller will rescale
    
    return None

def convolve_with(f, g, n):
    """Compute f * g (full convolution) using FFT."""
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    return np.real(ifft(F * G))[:2*n-1]

def lp_iteration(f, n, S, max_active=200, solver='ECOS'):
    """One LP-based iteration with constraint generation."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Start with top-k most binding lags
    k = max_active
    idx = np.argsort(c)[::-1][:k]
    active_lags = list(idx)
    
    g = None
    for _ in range(10):  # Max 10 constraint generation rounds
        g = solve_lp_best_response(f, n, S, active_lags, M, solver=solver)
        if g is None:
            return f.copy(), M
        
        # Check for violated constraints
        fg = convolve_with(f, g * S, n)  # g is normalized to sum=1, scale to S
        violations = np.where(fg > M * (1 + 1e-6))[0]
        if len(violations) == 0:
            break
        # Add violated constraints
        for v in violations[:20]:  # Add at most 20 new constraints
            if v not in active_lags:
                active_lags.append(v)
    
    if g is None:
        return f.copy(), M
    
    # Rescale g to sum = S
    g = g * S
    
    # Find best step size via bisection
    def eval_max(t):
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max(), f_test
    
    # Try t = 1 first
    val1, f1 = eval_max(1.0)
    if val1 < M:
        return f1, val1
    
    # Bisection
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = M
    best_f = f.copy()
    for _ in range(10):
        mid = (lo + hi) / 2
        val_mid, f_mid = eval_max(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            best_f = f_mid
            lo = mid
        else:
            hi = mid
    
    if best_t > 0.01:
        return best_f, best_val
    
    return f.copy(), M
```

```python
def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
```

```python
def solve_lp_with_active_set(f, n, M, active_lags, solver_list):
    """Solve LP and return (g, active_constraints)."""
    g_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        expr = cp.sum(cp.multiply(f[k - js], g_var[js]))
        constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    return g / g_sum
        except Exception:
            continue
    return None
```
