        mom = (t_nest - 1.0) / t_new
        y_new = x_new + mom * (x_new - y)
        y = np.maximum(y_new, 0.0)
        sy2 = y.sum()
        if sy2 > 1e-10:
            y = y * (S / sy2)
        else:
            y = best_a.copy()
            t_nest = 1.0
            continue
        t_nest = t_new

        # Periodic tau increase (every ~3000 iterations)
        if it % 3000 == 0 and tau_idx < len(tau_schedule) - 1:
            tau_idx += 1
            tau = tau_schedule[tau_idx]
            t_nest = 1.0
            y = best_a.copy()
            if verbose:
                print("    it=%d: tau -> %.1f" % (it, tau))

        # Periodic perturbation for exploration
        if it % 4000 == 2000 and rng.random() < 0.6:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_nest = 1.0

        # Stagnation handling: restart with larger perturbation
        if stagnant > 8000:
            noise_scale = 0.03 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            t_nest = 1.0
            step_scale = max(0.1, step_scale * 0.7)
            stagnant = 0
            if verbose and stagnant == 0 and it % 5000 < 3000:
                print("    it=%d: STAGNANT restart, score=%.6f" % (it, best_score))

        # Adapt step_scale based on recent performance
        if it % 2000 == 0:
            recent_gain = it - last_improve
            if recent_gain < 500:
                step_scale = min(step_scale * 1.2, 3.0)
            elif recent_gain > 3000:
                step_scale = max(step_scale * 0.7, 0.05)

        # Progress logging
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print("    it=%d, score=%.6f, tau=%.1f, step=%.5f, stagnant=%d, elapsed=%.0fs" % (
                it, best_score, tau, step, stagnant, elapsed))

    return best_a, best_score, it


def spectral_reflection_step(a, N, strength=0.3):
    """
    Apply a spectral reflection: in frequency domain, try to make A[m]^2 more
    uniform by adjusting the phase/magnitude of A while keeping a >= 0.
    This is a heuristic that helps escape local minima.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)

    # Compute current power spectrum
    P = np.abs(A) ** 2
    P_median = np.median(P[P > 0]) if np.any(P > 0) else 1.0

    # Compress high-frequency components, boost low
    ratio = P / max(P_median, 1e-30)
    P_new = P * np.exp(-strength * (ratio - 1.0) ** 2)
    P_new = np.maximum(P_new, 0.0)

    # Preserve phase from a
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def make_diverse_starts(n, S, rng):
    """Generate diverse starting points for exploration."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Chirp
    a = 0.5 + 0.35 * np.sin(2 * np.pi * (2 * tt + 1.5 * tt * tt))
    a = np.clip(a, 0.01, None)
    starts.append(a * S / a.sum())

    # Multi-frequency
    a = 0.5 + 0.15 * np.sin(2 * np.pi * 3 * tt) + 0.12 * np.sin(2 * np.pi * 7 * tt) + \
        0.1 * np.sin(2 * np.pi * 13 * tt) + 0.08 * np.sin(2 * np.pi * 21 * tt)
    a = np.clip(a, 0.01, None)
    starts.append(a * S / a.sum())

    # Asymmetric ramp with oscillation
    a = 0.3 + 0.4 * tt + 0.2 * np.sin(2 * np.pi * 5 * tt)
    a = np.clip(a, 0.01, None)
    starts.append(a * S / a.sum())

    # Two-lobe (high at edges, low in middle)
    a = 0.3 + 0.5 * np.cos(np.pi * tt) ** 2
    a = np.clip(a, 0.01, None)
    starts.append(a * S / a.sum())

    # Random smooth
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    a = 0.5 + 0.3 * (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    a = np.clip(a, 0.01, None)
    starts.append(a * S / a.sum())

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive log-sum-exp projected gradient with Nesterov acceleration,
    periodic diversification, and spectral reflection.
    """
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)
    rng = np.random.default_rng(seed)

    print("=== Starting optimization (budget %.0fs) ===" % budget_s)

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    # Load provided sequence
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and given.sum() > 0.01:
            a_start = np.clip(given, 0.0, 1000.0).copy()
            print("Loaded provided sequence (n=%d)" % n)
        else:
            print("Provided sequence length mismatch (%d), using default" % given.size)
    except Exception as e:
        print("Could not load provided: %s" % str(e))

    if a_start is None:
        a_start = np.ones(n) * S / n
        print("Using uniform start")

    # Normalize
    s = a_start.sum()
    if s > 0.01:
        a_start = a_start * (S / s)

    global_best = np.inf
    global_a = a_start.copy()

    def update(a, score):
        nonlocal global_best, global_a
        if score < global_best:
            global_best = score
            global_a = a.copy()
            print("  ** New global best: %.6f **" % score)

    # === Phase 1: Main optimization from provided start (60% budget) ===
    budget1 = (budget_s - 40) * 0.6
    print("\n[Phase 1] Main LSE gradient optimization (budget %.0fs)" % budget1)
    a1, score1, it1 = optimize_lse_gradient(
        a_start, budget1, seed=seed, verbose=True, n=n)
    update(a1, score1)
    print("  Phase 1 done: score=%.6f after %d iters" % (score1, it1))

    elapsed = time.monotonic() - (deadline - (budget_s - 40))

    # === Phase 2: Spectral reflection + re-optimize (15% budget) ===
    remaining = deadline - time.monotonic()
    budget2 = min((budget_s - 40) * 0.15, remaining - 60)
    if budget2 > 30:
        print("\n[Phase 2] Spectral reflection variants (budget %.0fs)" % budget2)
        for reflect_strength in [0.2, 0.5]:
            a_ref = spectral_reflection_step(global_a, N, strength=reflect_strength)
            # Apply a few reflection steps
            for _ in range(3):
                a_ref = spectral_reflection_step(a_ref, N, strength=reflect_strength * 0.5)
            a_ref = project_simplex(a_ref, S)

            sub_budget = budget2 / 2 - 5
            if sub_budget > 10:
                a2, score2, it2 = optimize_lse_gradient(
                    a_ref, sub_budget, seed=seed + 10, verbose=False, n=n)
                update(a2, score2)
                print("  Reflection strength %.1f: score=%.6f" % (reflect_strength, score2))

    # === Phase 3: Diverse starts (15% budget) ===
    remaining = deadline - time.monotonic()
    budget3 = min((budget_s - 40) * 0.15, remaining - 60)
    if budget3 > 30:
        print("\n[Phase 3] Diverse initialization search (budget %.0fs)" % budget3)
        starts = make_diverse_starts(n, S, rng)
        per_budget = budget3 / len(starts)
        for i, st in enumerate(starts):
            if time.monotonic() > deadline - 60:
                break
            if per_budget < 10:
                per_budget = 10
            print("  Starting %d/%d (%.1fs each)" % (i + 1, len(starts), per_budget))
            a3, score3, it3 = optimize_lse_gradient(
                st, per_budget, seed=seed + 100 + i, verbose=False, n=n)
            update(a3, score3)
            print("    -> score=%.6f" % score3)

    # === Phase 4: Final fine-tuning (10% budget) ===
    remaining = deadline - time.monotonic()
    budget4 = min((budget_s - 40) * 0.10, remaining - 30)
    if budget4 > 20:
        print("\n[Phase 4] Final fine-tuning (budget %.0fs)" % budget4)

        # Use higher tau for fine-tuning
        # Manually run with high tau
        a_fin = global_a.copy()
        # Perturb slightly
        a_fin = project_simplex(
            a_fin + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

        a4, score4, it4 = optimize_lse_gradient(
            a_fin, budget4, seed=seed + 500, verbose=False, n=n)
        update(a4, score4)
        print("  Phase 4 done: score=%.6f" % score4)

    # === Phase 5: One more try from best with perturbation ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n[Phase 5] Final perturbation attempt (budget %.0fs)" % (remaining - 30))
        a5_start = project_simplex(
            global_a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
        a5, score5, it5 = optimize_lse_gradient(
            a5_start, remaining - 30, seed=seed + 999, verbose=False, n=n)
        update(a5, score5)
        print("  Phase 5 done: score=%.6f" % score5)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64)
        global_best = 2.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        if ver < global_best:
            global_best = ver
        print("\n=== FINAL: length=%d, score=%.6f ===" % (len(global_a), global_best))
    else:
        print("\nWARNING: evaluate returned inf, using fallback")
        global_a = np.ones(n, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback score: %.6f" % global_best)

    return final_list
