To find the best `da`, we need to minimize this max. This is a 1D problem! For a fixed i, we can find the optimal `da` by considering the "tipping point" where the max transitions from one k to another.

This is related to the "minimax" or "Chebyshev" approximation problem in 1D. The optimal `da` is where the maximum of `{b[k] + da * v_i[k]}` is minimized.

For efficiency, we can:
1. Sort the pairs `(b[k], v_i[k])` by the value of `b[k]`.
2. The current max is at k* where `b[k*]` is largest.
3. We want to find `da` such that `b[k*] + da * v_i[k*]` is reduced without making any other `b[k] + da * v_i[k]` exceed it.
4. The constraint is: for all k, `b[k] + da * v_i[k] <= b[k*] + da * v_i[k*]`.
5. This gives: `da <= (b[k*] - b[k]) / (v_i[k] - v_i[k*])` when `v_i[k] > v_i[k*]` (and the reverse inequality when `v_i[k] < v_i[k*]`).

So the optimal `da` is the minimum of the upper bounds (if we're decreasing the peak) or the maximum of the lower bounds.

This is O(n) per coordinate (to compute all the bounds), which is much better than O(n^2).

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def nesterov_subgradient(a_init, n, S, max_time, rng, top_k=32):
    """Nesterov-accelerated projected subgradient method."""
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        b_y = fft_convolve_self(y)
        cur = b_y.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Subgradient at y
        top_indices = np.argsort(b_y)[::-1][:top_k]
        top_vals = b_y[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient
        grad = np.zeros(n)
        for i_idx in range(len(top_indices)):
            l = int(top_indices[i_idx])
            w = weights[i_idx]
            if l < n:
                grad[:l+1] += w * a[l::-1]  # a[l], a[l-1], ..., a[0]
            grad[l:] += w * a[:n-l]  # a[0], a[1], ..., a[n-1-l]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size (diminishing)
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Project y to simplex
        y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Occasional perturbation
        if it % 500 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

                # b[k] + da*v[k] <= b[k*] + da*v[k*] for all k
                # da*(v[k] - v[k*]) <= b[k*] - b[k]
                # For v[k] > v[k*]: da <= (b[k*]-b[k])/(v[k]-v[k*])
                # For v[k] < v[k*]: da >= (b[k*]-b[k])/(v[k]-v[k*])
                
                mask_hi = v > v_kstar + 1e-15
                if np.any(mask_hi):
                    ub = (b[k_star] - b[mask_hi]) / (v[mask_hi] - v_kstar)
                    da_max = np.min(ub)
                else:
                    da_max = 0.0
                
                mask_lo = v < v_kstar - 1e-15
                if np.any(mask_lo):
                    lb = (b[k_star] - b[mask_lo]) / (v[mask_lo] - v_kstar)
                    da_min_c = np.max(lb)
                else:
                    da_min_c = -np.inf
                
                da_min = max(da_min_c, -a[i])
                
                if da_min < da_max - 1e-15:
                    da = da_min  # most negative (maximum reduction)
                    new_val = a[i] + da
                    if new_val < 0:
                        new_val = 0.0
                    a[i] = new_val
                    improved = True
                    # Update b incrementally
                    b += da * v
                    # Update b_max
                    b_max = b.max()
                    k_star = int(np.argmax(b))
            else:
                # v_kstar < 0: Want to increase a[i]
                # da > 0
                # b[k] + da*v[k] <= b[k*] + da*v[k*] for all k
                # da*(v[k] - v[k*]) <= b[k*] - b[k]
                # For v[k] < v[k*] (more negative): da >= (b[k*]-b[k])/(v[k]-v[k*]) (dividing by negative flips)
                # For v[k] > v[k*]: da <= (b[k*]-b[k])/(v[k]-v[k*])
                
                mask_lo = v < v_kstar - 1e-15  # v[k] - v[k*] < 0
                if np.any(mask_lo):
                    lb = (b[k_star] - b[mask_lo]) / (v[mask_lo] - v_kstar)
                    da_min = np.max(lb)
                else:
                    da_min = 0.0
                
                mask_hi = v > v_kstar + 1e-15  # v[k] - v[k*] > 0
                if np.any(mask_hi):
                    ub = (b[k_star] - b[mask_hi]) / (v[mask_hi] - v_kstar)
                    da_max = np.min(ub)
                else:
                    da_max = np.inf
                
                if da_max > da_min + 1e-15:
                    da = da_max  # most positive (maximum increase)
                    a[i] = a[i] + da
                    improved = True
                    b += da * v
                    b_max = b.max()
                    k_star = int(np.argmax(b))
        
        # Rescale to maintain sum
        a = project_simplex(a, S)
        
        if not improved and sweep > 10:
            # Perturb to escape
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return a, b_max


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_sequence(a):
    """Compute the score (before 2n factor) = max(conv)/sum^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return c.max() / (s * s)

            return False
