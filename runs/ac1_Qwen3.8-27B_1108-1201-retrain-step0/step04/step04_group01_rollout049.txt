"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using iterative peak shaving
with momentum, adaptive step sizes, and multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for theta
    lo, hi = -1e15, max(v)
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0.0)


def project_simplex_fast(v, S):
    """Faster version using sorting."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_subgradient(a, c, k=8):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above mean
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:

import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex_fast(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sorting."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_weighted(a, lags, weights):
    """Compute weighted sum of subgradients at given lags of self-convolution."""
    n = len(a)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):

        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)

            indices = np.arange(lo, hi + 1)

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Iterative peak shaving with heavy-ball momentum
2. LSE-smoothed gradient for continuous optimization
3. Multi-length exploration (n=8192, 16384)
4. Adaptive step sizes and periodic restarts
5. Final high-precision polishing with exact subgradient
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex_fast(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_weighted(a, lags, weights):
    n = len(a)
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w < 1e-15:
            continue
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            g[lo:hi+1] += w * 2.0 * a[lag - hi:lag - lo + 1][::-1]
    return g


def lse_gradient_fft(a, c, beta):
    """Gradient of LSE(beta, c) where c = convolve(a,a)."""
    n = len(a)
    M = c.max()
    scaled = beta * (c - M)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - M).max())
        scaled = beta * (c - M)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    h = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * beta * h


def lse_value(c, beta):
    M = c.max()
    scaled = beta * (c - M)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - M).max())
        scaled = beta * (c - M)
    return M + np.log(np.exp(scaled).sum()) / beta


def peak_shave_momentum(a, S, time_budget, rng, k_peaks=16, momentum=0.9, 
                        lr_init=None, verbose=True):
    """Peak shaving with heavy-ball momentum."""
    n = len(a)
    a = normalize_to_sum(a, S).copy()
    if lr_init is None:
        lr_init = 0.5 * S / n
    
    b = fft_convolve_self(a)
    best_obj = b.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        b = fft_convolve_self(a)
        cur_max = b.max()
        
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k = min(k_peaks, len(b))
        peak_idx = np.argsort(b)[::-1][:k]
        mean_b = b.mean()
        weights = np.maximum(b[peak_idx] - mean_b, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        g = subgradient_weighted(a, peak_idx, weights)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        v = momentum * v - lr * g
        a_new = project_simplex_fast(a + v, S)
        
        b_new = fft_convolve_self(a_new)
        if b_new.max() <= cur_max + 1e-12:
            a = a_new
            if no_improve == 0:
                lr = min(lr * 1.02, 10 * S / n)
        else:
            # Halve the step
            v = np.zeros(n)
            lr *= 0.5
            a_new = project_simplex_fast(a - lr * g, S)
            b_new = fft_convolve_self(a_new)
            if b_new.max() < cur_max:
                a = a_new
        
        it += 1
        if it % 5000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    [PS] it={it}, max_b={cur_max:.8f}, best={best_obj:.8f}, "
                  f"lr={lr:.2e}, elapsed={elapsed:.0f}s")
        
        if no_improve > 2000:
            noise = 0.01 * S / np.sqrt(n)
            a = project_simplex_fast(a + rng.normal(0, noise, n), S)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def lse_fista(a, S, time_budget, rng, beta_init=5.0, beta_growth=0.001, 
              beta_max=500.0, verbose=True):
    """FISTA with LSE-smoothed max."""
    n = len(a)
    a = normalize_to_sum(a, S).copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < time_budget:
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient_fft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Lipschitz constant estimate: ||grad|| is bounded by 2*beta*||a||_2
        # Use step = 1/L where L ≈ 2*beta * S / sqrt(n)
        L = 2.5 * beta * S / np.sqrt(n)
        step = 1.0 / L
        
        x_new = project_simplex_fast(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        it += 1
        
        if it % 5000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    [FISTA] it={it}, max_b={cur:.8f}, best={best:.8f}, "
                  f"beta={beta:.1f}, elapsed={elapsed:.0f}s")
        
        # Periodic restart
        if it % 10000 == 0:
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex_fast(y + rng.normal(0, noise, n), S)
            t = 1.0
    
    # Final step
    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()
    
    return best_a, best
