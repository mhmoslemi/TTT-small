def project_simplex(v, S):

def project_simplex(v, S):
    """Euclidean projection onto {x: x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)

    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np(where_w)(0)) + 1) if (where_w := np.where(valid)) is not None else n
    # Redo properly without walrus in case of issues

    if where_valid[0].size > 0:
        rho = int(where_valid[0].max()) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta, N_corr):
    """
    Compute LSE-smoothed max of self-conv and its gradient.
    Returns: (gradient [n], lse_value, c [2n-1], c_max)
    Gradient formula: grad_i = 2 * sum_j a[j] * w[i+j]
    Computed via: conv(a, w_rev)[2n-2-i]
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max

    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta

    # Gradient via cross-correlation trick
    w_rev = w[::-1].copy()
    a_full = np.zeros(N_corr)
    a_full[:n] = a
    w_full = np.zeros(N_corr)
    w_full[:2 * n - 1] = w_rev
    conv_res = np.real(ifft(fft(a_full) * fft(w_full)))
    idx = np.arange(n)
    grad = 2.0 * conv_res[2 * n - 2 - idx]

    return grad, lse_val, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion. Returns descent direction."""
    n = len(g)
    m = min(len(s_hist), 12)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-20:
            return np.zeros(n)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(si, q) / yy
            q = q - alphas[i] * yi

    sl = s_hist[-1]
    yl = y_hist[-1]
    slsl = np.dot(sl, sl)
    ylsl = np.dot(yl, yl)
    if ylsl > 1e-30:
        sy = np.dot(sl, yl)
        gamma = max(1e-10, sy / ylsl)
    else:
        gamma = max(1e-8, 1.0 / max(slsl, 1e-10))

    r = gamma * q
    for i in range(m):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy < 1e-30:
            continue
        beta_i = np.dot(si, yi) / yy
        r = r + (alphas[i] - beta_i) * yi

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

- Non-negativity enforced via simplex projection throughout
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x: x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    where_valid = np.where(valid)
    if where_valid[0].size > 0:
        rho = int(where_valid[0].max()) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta, N_corr):
    """
    Compute LSE-smoothed max of self-conv and its gradient.
    Returns: (gradient [n], lse_value, c [2n-1], c_max)
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max

    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta

    # Gradient: grad_i = 2 * sum_j a[j] * w[i+j]
    # Via: conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1].copy()
    a_full = np.zeros(N_corr)
    a_full[:n] = a
    w_full = np.zeros(N_corr)
    w_full[:2 * n - 1] = w_rev
    conv_res = np.real(ifft(fft(a_full) * fft(w_full)))
    idx = np.arange(n)
    grad = 2.0 * conv_res[2 * n - 2 - idx]

    return grad, lse_val, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion. Returns descent direction."""
    n = len(g)
    m = min(len(s_hist), 12)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-20:
            return np.zeros(n)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(si, q) / yy
            q = q - alphas[i] * yi

    sl = s_hist[-1]
    yl = y_hist[-1]
    yy_l = np.dot(yl, yl)
    if yy_l > 1e-30:
        sy_l = np.dot(sl, yl)
        gamma = max(1e-10, sy_l / yy_l)
    else:
        gamma = 1.0

    r = gamma * q
    for i in range(m):
        si = s_hist[i]
        yi = y_hist[i]
        yy = np.dot(yi, yi)
        if yy < 1e-30:
            continue
        beta_i = np.dot(si, yi) / yy
        r = r + (alphas[i] - beta_i) * yi

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_block(a_init, n, S, N, N_corr, max_time, rng,
                beta_init=5.0, beta_max=300.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf

    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_grad(a, N, beta, N_corr)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.08)
            g, f_val, c, c_max = lse_grad(a, N, beta, N_corr)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-22:
            break

        step_scale = 2.0 * S / (n * max(beta, 2.0))
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        accepted = False

        for _ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max:
                a_new = candidate
                cmax_new = cm_ls
                accepted = True
                break
            alpha_ls *= 0.5

        if not accepted:
            a_new = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(a_new, N)
            cmax_new = c_ls.max()

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw, cmax_nw = lse_grad(a_new, N, beta, N_corr)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)

        a = a_new
        g = g_new
        c = c_nw
        c_max = cmax_nw

        if it % 600 == 0 and rng.random() < 0.25:
            noise_scale = 0.004 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_grad(a, N, beta, N_corr)

    return best_a, best_cmax


def spectral_waterfill_step(a, N, S, top_k, alpha):
    """One spectral water-filling step."""
    n = len(a)
    c, A = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i in range(top_k):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(top_idx[i]) / N)

    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)
