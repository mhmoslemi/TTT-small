```python
        if D_2i <= T_opt:
            w_opt = min(w_opt, np.sqrt(T_opt - D_2i)) if w_opt > 0 else np.sqrt(max(0, T_opt - D_2i))
    
    ks = np.arange(k_min, k_max + 1)
    ks = ks[ks != 2*i]
    if len(ks) > 0:
        js = ks - i
        a_js = a[js]
        D_ks = B[ks] - 2*v*a_js
        mask_pos = a_js > 1e-15
        if np.any(mask_pos):
            w_limits = (T_opt - D_ks[mask_pos]) / (2*a_js[mask_pos])
            w_opt = min(w_opt, np.min(w_limits)) if w_opt > 0 else np.min(w_limits)
    
    w_opt = max(0, w_opt)
    
    # Update
    a[i] = w_opt
    # Update B
    for k in range(k_min, k_max + 1):
        j = k - i
        if k == 2*i:
            B[k] = D_2i + w_opt*w_opt  # need to recompute
        else:
            B[k] = B[k] - 2*v*a[j] + 2*w_opt*a[j]
    
    return a, B
```

Hmm, this is getting complicated and the inner loop over k is O(n) per coordinate, making a full cycle O(n^2). For n=8192, that's ~67M operations per cycle, which is manageable.

But actually, I can vectorize the bisection check. Let me rewrite:

For a given T, the feasible w range is [0, w_max(T)] where:
w_max(T) = min(
    sqrt(T - D_{2i}) if 2i in [k_min, k_max] and T >= D_{2i}, else inf,
    min_{k≠2i, a[k-i]>0} (T - D_k) / (2*a[k-i]) if D_k <= T for all k with a[k-i]=0
)

This can be computed in O(n) per bisection step, and we need ~40 bisection steps, so O(40n) per coordinate. A full cycle is O(40n^2).

For n=8192: 40 * 8192^2 ≈ 2.7 billion operations. That's too slow for a single cycle in Python (even with numpy, this would take minutes).

Let me reduce the bisection to 20 steps (giving precision of ~1e-6 relative), making it ~1.3 billion per cycle. Still too slow for many cycles.

Alternative: Instead of full bisection, use a "greedy" approach:
- Compute the current max T_cur
- For coordinate i, find the w that minimizes the max of the affected B values
- This can be done in O(n) without bisection using a "water-filling" approach

Actually, the problem for coordinate i is:
min_w max(f_0(w), f_1(w), ..., f_m(w))

where f_0(w) = w^2 + D_{2i} (parabola) and f_k(w) = 2*w*a[k-i] + D_k (lines).

The minimum of the max of a parabola and several lines can be found by:
1. Sort the lines by their slope (2*a[k-i])
2. The optimum is where the parabola intersects the "upper envelope" of the lines

This is a 1D convex optimization that can be solved in O(m log m) where m is the number of affected k values.

In practice, for a quick implementation, I'll use a reduced bisection (10-15 steps) which should be sufficient.

Let me reconsider the overall strategy. Given the time budget of 2400 seconds and the need to beat 1.507:

2. **Phase 2 (30% of time)**: Run coordinate descent polishing. Use a vectorized approach:
   - For each coordinate, compute the optimal update in O(n) with ~20 bisection steps
   - Do multiple passes
   - This should give a small but consistent improvement

3. **Phase 3 (20% of time)**: Try a larger n (16384) by upsample

```python
"""
Minimize 2*n*max(a*a)/sum(a)^2 over non-negative sequences.
Strategy: Hybrid of (1) FISTA with LSE smoothing, (2) fast coordinate descent
with bisection on per-coordinate optimal values, (3) LP best-response, and
(4) multi-scale exploration with diverse initializations (exponential decay,
spectral flat, etc.). Key innovation: vectorized coordinate descent that
solves the 1D subproblem for each coordinate via bisection on the threshold.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(a*a)/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def coordinate_descent_pass(a, B, n, num_bisect=25, subset_indices=None):
    """
    One pass of coordinate descent. For each coordinate i, find optimal value
    minimizing max of affected B values. B is the self-convolution (kept updated).
    Returns updated a, B, and the max of B after this pass.
    """
    if subset_indices is None:
        indices = np.arange(n)
    else:
        indices = subset_indices

    for i in indices:
        v = a[i]
        if v < 1e-15:
            # If a[i] is ~0, try a small positive value
            new_v = 1e-5
            a[i] = new_v
            # Update B
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            js = np.arange(k_lo, k_hi + 1) - i
            if i * 2 <= 2 * n - 2 and i * 2 >= 0:
                B[i * 2] += new_v * new_v
                B[k_lo:k_hi + 1] += 2 * new_v * a[js]
                B[i * 2] -= new_v * new_v  # fix double count at k=2i
                B[i * 2] = B[i * 2]  # already has v^2 contribution
            else:
                B[k_lo:k_hi + 1] += 2 * new_v * a[js]
            continue

        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)

        # Gather affected B values and corresponding a[k-i]
        ks = np.arange(k_lo, k_hi + 1)
        js = ks - i  # indices into a, range [0, n-1]
        a_js = a[js]
        B_ks = B[ks].copy()

        # Identify k=2i position
        two_i = 2 * i
        is_two_i = (ks == two_i)

        # D_k = B[k] - contribution of a[i]
        # For k=2i: contribution = v^2
        # For k!=2i: contribution = 2*v*a[k-i]
        contrib = np.where(is_two_i, v * v, 2.0 * v * a_js)
        D = B_ks - contrib

        # Bisection on T to find min max of new B values
        cur_max_region = B_ks.max()
        lo = max(0.0, D.max())  # T must be at least max(D) (with w=0)
        hi = cur_max_region

        for _ in range(num_bisect):
            mid = (lo + hi) / 2.0

            # Check feasibility: exists w >= 0 with all new B[k] <= mid
            feasible = True
            w_ub = np.inf

            # Constraint from k=2i: w^2 + D[2i] <= mid
            if np.any(is_two_i):
                D_2i = D[is_two_i][0]
                if D_2i > mid + 1e-15:
                    feasible = False
                else:
                    w_ub = min(w_ub, np.sqrt(max(0.0, mid - D_2i)))

            # Constraints from k != 2i: 2*w*a[k-i] + D[k] <= mid
            mask_not_2i = ~is_two_i
            if np.any(mask_not_2i):
                a_sub = a_js[mask_not_2i]
                D_sub = D[mask_not_2i]
                # For a_sub > 0: w <= (mid - D_sub) / (2*a_sub)
                # For a_sub == 0: need D_sub <= mid
                pos_mask = a_sub > 1e-14
                if np.any(~pos_mask):
                    if np.any(D_sub[~pos_mask] > mid + 1e-10):
                        feasible = False
                if feasible and np.any(pos_mask):
                    limits = (mid - D_sub[pos_mask]) / (2.0 * a_sub[pos_mask])
                    w_ub = min(w_ub, limits.min())

            if feasible and w_ub >= 0:
                hi = mid
            else:
                lo = mid

        T_opt = hi

        # Find optimal w at T_opt
        w_opt = 0.0
        if np.any(is_two_i):
            D_2i = D[is_two_i][0]
            if D_2i <= T_opt:
                w_opt = min(w_opt if w_opt > 0 else np.inf, np.sqrt(max(0.0, T_opt - D_2i)))

        mask_not_2i = ~is_two_i
        if np.any(mask_not_2i):
            a_sub = a_js[mask_not_2i]
            D_sub = D[mask_not_2i]
            pos_mask = a_sub > 1e-14
            if np.any(pos_mask):
                limits = (T_opt - D_sub[pos_mask]) / (2.0 * a_sub[pos_mask])
                w_opt = min(w_opt if w_opt > 0 else np.inf, limits.min())

        w_opt = max(0.0, min(w_opt, 1000.0))

        if abs(w_opt - v) > 1e-12:
            # Update B
            new_contrib = np.where(is_two_i, w_opt * w_opt, 2.0 * w_opt * a_js)
            B[k_lo:k_hi + 1] = D + new_contrib
            a[i] = w_opt

    return a, B, B.max()


def run_coordinate_descent(a_init, n, S, max_time, num_passes_max=200, bisect_steps=25):
    """Run coordinate descent with periodic projection."""
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    a = project_simplex(a, S)
    B = fft_convolve_self(a)
    start = time.monotonic()
    best_max = B.max()
    best_a = a.copy()
    no_improve = 0

    for pass_num in range(num_passes_max):
        if time.monotonic() - start > max_time:
            break

        # Randomized order for diversity
        np.random.shuffle_indices = None
        indices = np.arange(n)
        np.random.default_rng(pass_num * 7 + 1).shuffle(indices)

        a, B, cur_max = coordinate_descent_pass(a, B, n, num_bisect=bisect_steps, subset_indices=indices)

        # Periodic projection to maintain sum constraint
        if pass_num % 10 == 0:
            a = project_simplex(a, S)
            B = fft_convolve_self(a)
            cur_max = B.max()

        if cur_max < best_max * (1 - 1e-9):
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if pass_num % 20 == 0:
            obj = 2.0 * n * cur_max / (S * S)
            print("  CD pass %d: max_B=%.8f, obj=%.6f" % (pass_num, cur_max, obj))

        if no_improve > 30:
            # Add small perturbation to escape
            rng_local = np.random.default_rng(pass_num)
            a = a + rng_local.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a, S)
            B = fft_convolve_self(a)
            cur_max = B.max()
            no_improve = 0

    return best_a, best_max


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.002, beta_max=1000.0,
              use_repulsion=True, repulsion_freq=300):
    """FISTA with LSE smoothing."""
    a = np.clip(a_init, 0.0, 1000.0)
    a = project_simplex(a, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 1500:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if use_repulsion and it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(30.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-0.05 * rep / max(y.max(), 1e-10))
                y = project_simplex(y, S)
                t = 1.0

    return best_a, best


def solve_lp_best_response(f, n, S, top_k=100, solver='SCS', time_limit=15.0):
    """LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 2:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        return g * (S / g_sum)
    return None


def generate_exponential_init(n, decay, rng):
    """Generate exponential decay initialization."""
    a = decay ** np.arange(n, dtype=np.float64)
    noise = rng.normal(0, 0.05, n)
    a = a + noise
    return np.maximum(a, 0.01)


def generate_spectral_flat_init(n, rng):
    """Generate sequence with approximately flat |A(w)|."""
    # Random phases, constant magnitude
    mags = np.ones(n) * np.sqrt(n)
    phases = rng.uniform(-np.pi, np.pi, n)
    A = mags * (np.cos(phases) + 1j * np.sin(phases))
    a = np.real(ifft(A))
    return np.maximum(a, 0.01)


def generate_bathtub_init(n, depth, rng):
    """U-shaped: high at edges, low in middle."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + depth * np.cos(2 * np.pi * t)
    a = a + rng.normal(0, 0.02, n)
    return np.maximum(a, 0.01)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def multiplicative_repulsion(f, n, S, beta=40.0, lr=0.05):
    """Fast multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new

    global_best = np.inf

            print("  NEW GLOBAL BEST: obj=%.6f (n=%d)" % (global_obj, n))
        else:
            print("  No improvement (best=%.6f, this=%.6f)" % (global_obj, final_obj))

    # Final intensive polishing on the global best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_p = len(global_a)
            S_p = np.sqrt(2.0 * n_p)
            print("\n=== Final Polish (remaining %.0fs, n=%d) ===" % (remaining, n_p))

            # Coordinate descent with more passes
            cd_time2 = remaining * 0.5
            a_pol, max_pol = run_coordinate_descent(global_a, n_p, S_p, cd_time2,
                                                    num_passes_max=1000, bisect_steps=25)
            obj_pol = 2.0 * n_p * max_pol / (S_p * S_p)
            print("  Final CD: obj=%.6f" % obj_pol)
            if obj_pol < global_obj:
                global_obj = obj_pol
                global_a = a_pol.copy()

            # Another FISTA round with high beta
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_fin, max_fin = run_fista(global_a, n_p, S_p, remaining2 * 0.6, rng,
                                           beta_init=50.0, beta_rate=0.001, beta_max=2000.0,
                                           use_repulsion=False)
                obj_fin = 2.0 * n_p * max_fin / (S_p * S_p)
                print("  Final FISTA: obj=%.6f" % obj_fin)
                if obj_fin < global_obj:
                    global_obj = obj_fin
                    global_a = a_fin.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f" % ver)
        else:
            print("\nWARNING: evaluate_sequence rejected! Using fallback.")
            n_fb = 256
            global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_a]
    else:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)

    print("Returning sequence of length %d, internal best=%.6f" % (len(global_a), global_obj))
    return [float(x) for x in global_a]
```
