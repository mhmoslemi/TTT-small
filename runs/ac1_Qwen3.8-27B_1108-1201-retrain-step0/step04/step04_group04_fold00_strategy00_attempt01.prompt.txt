def evaluate_sequence(sequence: list[float]) -> float:
    """

    # Check each element in the list for validity
    for x in sequence:

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def spectral_flatten(a_init, n, S, max_time, rng, alpha=0.5, n_inner=50):
    """Gerchberg-Saxton style spectral flattening."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2

        # Flatten the power spectrum
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        # Soft flattening: P_new = P^alpha / mean(P^alpha)
        P_flat = np.power(np.maximum(P, 1e-30), alpha)
        P_flat = P_flat / P_flat.mean()
        # Blend with original for stability
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_flat

        # Reconstruct with new magnitude, same phase
        mag_new = np.sqrt(P_new)
        # Ensure total power is conserved (approximately)
        mag_new = mag_new * np.sqrt(P.sum() / P_new.sum())
        F_new = phase * mag_new
        a_new = np.real(ifft(F_new))[:n]

        # Project to non-negative with fixed sum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

        # Adaptive alpha: increase flattening over time
        alpha = min(0.95, alpha + 0.002)

    return best_a, best


def coordinate_descent(a_init, n, S, max_time, rng):
    """Paired coordinate descent on the simplex."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()

    # Precompute indices for efficiency
    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Try a set of (j, m) pairs: local neighbors and random
        n_try = min(50 + it // 10, 200)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break

            # Choose pair: mix of local and random
            if rng.random() < 0.5:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, n - 2)
                m = j + rng.integers(1, min(5, n - 1))
                m = m % n

            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # Compute d_k = change in c_k per unit delta (transfer from m to j)
            # c_k = sum_i a_i * a_{i+k}
            # dc_k/ddelta = a_{j+k} + a_{j-k} - a_{m+k} - a_{m-k} (with bounds)

            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                # Contribution from j
                jk = j + k
                jmk = j - k
                val_j = 0.0
                if 0 <= jk < n:
                    val_j += a[jk]
                if 0 <= jmk < n:
                    val_j += a[jmk]
                # Contribution from m
                mk = m + k
                mmk = m - k
                val_m = 0.0
                if 0 <= mk < n:
                    val_m += a[mk]
                if 0 <= mmk < n:
                    val_m += a[mmk]
                d[k] = val_j - val_m

            # c_k(delta) = c[k] + delta * d[k] (ignoring k=0 quadratic for now)
            # For k=0: c_0(delta) = c_0 + delta*(a_j + a_m - a_j - a_m) + ... 
            # Actually c_0 = sum a_i^2, change = (a_j+d)^2 - a_j^2 + (a_m-d)^2 - a_m^2
            # = 2a_j*d + d^2 - 2a_m*d + d^2 = 2(a_j-a_m)*d + 2d^2
            # So d[0] should be 2*(a_j - a_m) and we have a quadratic term 2*d^2

            # Find delta* minimizing max_k (c[k] + delta * d[k])
            # with delta in [-a[j], a[m]]
            lo = -a[j]
            hi = a[m]

            # Ternary search for 1D convex minimization
            def f_of_delta(dlt):
                vals = c + dlt * d
                vals[0] += 2.0 * dlt ** 2  # quadratic correction for k=0
                return vals.max()

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = f_of_delta(c_g)
            fd = f_of_delta(d_g)

            for _ in range(40):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_of_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_of_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = np.maximum(lo, np.minimum(hi, delta))

            # Apply if improvement
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            # Ensure non-negative
            if a_trial[j] < 0 or a_trial[m] < 0:
                continue

            c_trial = c + delta * d
            c_trial[0] += 2.0 * a[j] * delta + 2.0 * delta ** 2
            # More precisely recompute c_0
            c_trial[0] = c[0] + 2 * (a[j] - a[m]) * delta + 2 * delta ** 2

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-12:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Full recompute periodically to avoid drift
        if it % 20 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        if not improved and it > 5:
            # No improvement in this pass, add small noise
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 100.0)
            if noise_scale < 1e-8:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def spectral_flatten_v2(a_init, n, S, max_time, rng):
    """Improved spectral flattening with adaptive parameters and momentum."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    a_prev = a.copy()
    momentum = 0.8

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Adaptive alpha based on progress
        progress = min(1.0, it / 2000.0)
        alpha = 0.3 + 0.6 * progress

        # Flattening: push P towards constant
        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target
        # Also try "inverse" weighting for peaks
        P_peak_weight = 1.0 / np.sqrt(np.maximum(P / P_mean, 1e-10))
        P_new = P_new * (0.7 + 0.3 * P_peak_weight / P_peak_weight.mean())

        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        # Momentum
        a_new = momentum * a_new + (1 - momentum) * a_prev
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a_prev = a
        a = a_new

    return best_a, best


def make_palindromic(a, n):
    """Make a sequence palindromic."""
    half = a[:n // 2]
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])


def generate_variants(a, n, rng):
    """Generate diverse variants of a sequence."""
    variants = [a.copy()]

    # Palindromic
    pal = make_palindromic(a, n)
    variants.append(normalize_sum(pal, n))

    # Reversed
    variants.append(a[::-1].copy())

    # Small random perturbations
    for scale in [0.01, 0.05, 0.1]:
        noise = rng.normal(0, scale * S_global / np.sqrt(n), n)
        v = project_simplex(a + noise, S_global)
        variants.append(v)

    # Low-pass filtered version
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    cutoff = 0.3
    F_filtered = F * np.exp(-10.0 * (freqs / cutoff) ** 2)
    a_filtered = np.real(ifft(F_filtered))[:n]
    a_filtered = np.maximum(a_filtered, 0)
    s = a_filtered.sum()
    if s > 1e-15:
        variants.append(a_filtered * (S_global / s))

    # High-pass (smoothed complement)
    F_hp = F - F * np.exp(-10.0 * (freqs / cutoff) ** 2)
    a_hp = np.real(ifft(F_hp))[:n]
    a_hp = np.maximum(a_hp, 0)
    s = a_hp.sum()
    if s > 1e-15:
        variants.append(a_hp * (S_global / s))

    return variants
