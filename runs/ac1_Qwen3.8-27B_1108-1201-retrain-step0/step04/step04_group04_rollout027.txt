    A = fft(a, fsize)
    c = np.real(ifft(A * A))

    if s < 1e-15:

    return a * (S / s)

def lse_value(c, beta):

    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Multi-phase optimization:
1. FISTA with LSE smoothing (adaptive beta ramp)
2. Mirror descent with negative entropy (multiplicative updates)
3. High-precision FISTA with very high beta
4. Block coordinate descent with exact 1D line search
5. Final polish with exact subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_grad(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_grad(a, c, beta):
    """Gradient using exponential weighting near the max (top-k style)."""
    n = len(a)
    cm = c.max()
    w = np.exp(-beta * (cm - c))
    tot = w.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = w / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def _compute_d(a, j, m, n):
    """Compute derivative vector for transfer from m to j."""
    size = 2 * n - 1
    d = np.zeros(size)
    for k in range(size):
        val = 0.0
        jk = j + k
        jmk = j - k
        mk = m + k
        mmk = m - k
        if 0 <= jk < n:
            val += a[jk]
        if 0 <= jmk < n:
            val += a[jmk]
        if 0 <= mk < n:
            val -= a[mk]
        if 0 <= mmk < n:
            val -= a[mmk]
        d[k] = val
    return d


def _golden_search(c, d, a, j, m, lo, hi):
    """Golden section search for optimal delta."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = lo
    b_g = hi
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)

    def obj(dlt):
        mx = np.max(c + dlt * d)
        q = 2.0 * dlt * (a[j] - a[m]) + 2.0 * dlt * dlt
        idx0 = 0
        val0 = c[0] + dlt * d[0] + q
        return max(mx, val0)

    fc = obj(c_g)
    fd = obj(d_g)
    for _ in range(35):
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr * (b_g - a_g)
            fc = obj(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr * (b_g - a_g)
            fd = obj(d_g)
    return (a_g + b_g) / 2.0

def spectral_refine(a_init, S, max_time, rng):

    n = len(a_init)

        a_pad = np.zeros(fsize)

        if P_mean < 1e-30:

        P_new = (1.0 - alpha) * P / P_mean + alpha * P_flat

        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        phase = F / np.maximum(np.abs(F), 1e-30)
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        a = a_new

    return best_a, best


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Main search function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)

    parent = None
    try:
        parr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parr.size >= 64 and np.sum(np.clip(parr, 0, 1000)) > 0.01:
            parent = np.clip(parr, 0, 1000)
    except Exception:
        parent = None

    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Config: n=%d, S=%.4f, budget=%.0fs" % (n, S, budget_s))

    a_best = parent.copy() if parent is not None else np.ones(n) * (S / n)
    a_best = normalize_sum(a_best, S)
    c_chk = fft_convolve_self(a_best)
    best_score = c_chk.max()
    print("Initial score: %.6f" % best_score)

    def _update(a_cand, label):
        nonlocal a_best, best_score
        a_cand = normalize_sum(a_cand, n)
        c_v = fft_convolve_self(a_cand)
        sc = c_v.max()
        if sc < best_score:
            a_best = a_cand.copy()
            best_score = sc
            el = time.monotonic() - t_start
            print("  [%s] score=%.6f (t=%.0fs)" % (label, sc, el))
        return sc

    # Phase 1: FISTA with moderate beta
    rem = deadline - time.monotonic()
    t1 = rem * 0.20
    print("\n=== Phase 1: FISTA LSE beta->5000 (%.0fs) ===" % t1)
    if t1 > 10:
        a_r, _ = run_fista_lse(a_best, S, t1, rng, beta_init=10.0, beta_max=5000.0,
                               ramp_iters=1500, restart_every=3000)
        _update(a_r, "fista1")

    # Phase 2: Mirror descent
    rem = deadline - time.monotonic()
    t2 = rem * 0.15
    print("\n=== Phase 2: Mirror Descent (%.0fs) ===" % t2)
    if t2 > 10:
        a_r, _ = run_mirror_descent(a_best, S, t2, rng)
        _update(a_r, "mirror")

    # Phase 3: Spectral refinement
    rem = deadline - time.monotonic()
    t3 = rem * 0.15
    print("\n=== Phase 3: Spectral (%.0fs) ===" % t3)
    if t3 > 10:
        a_r, _ = spectral_refine(a_best, S, t3, rng)
        _update(a_r, "spectral")

    # Phase 4: High-precision FISTA
    rem = deadline - time.monotonic()
    t4 = rem * 0.25
    print("\n=== Phase 4: High-beta FISTA (%.0fs) ===" % t4)
    if t4 > 10:
        a_r, _ = run_fista_highbeta(a_best, S, t4, rng, beta_start=5000.0, beta_end=50000.0)
        _update(a_r, "fista_hb")

    # Phase 5: Coordinate descent
    rem = deadline - time.monotonic()
    t5 = rem * 0.20
    print("\n=== Phase 5: Coord Descent (%.0fs) ===" % t5)
    if t5 > 10:
        a_r, _ = run_coord_descent(a_best, S, t5, rng, n_pairs=100)
        _update(a_r, "cd")

    # Phase 6: Final exact subgradient FISTA
    rem = deadline - time.monotonic()
    t6 = rem * 0.50
    print("\n=== Phase 6: Exact Subgrad FISTA (%.0fs) ===" % t6)
    if t6 > 10:
        a_r, _ = run_fista_exact(a_best, S, t6, rng)
        _update(a_r, "exact")

    # Final CD polish
    rem = deadline - time.monotonic()
    if rem > 10:
        a_r, _ = run_coord_descent(a_best, S, rem * 0.8, rng, n_pairs=150)
        _update(a_r, "final_cd")

    # Verify and return
    a_final = np.clip(a_best, 0.0, 1000.0)
    if a_final.sum() < 0.01:
        a_final = np.ones(256) * (np.sqrt(2 * 256) / 256)
    final_list = [float(x) for x in a_final]
    ver = evaluate_sequence(final_list)
    print("\nFinal: length=%d, score=%.6f, time=%.0fs" % (len(final_list), ver, time.monotonic() - t_start))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        a_final = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in a_final]

    return final_list
