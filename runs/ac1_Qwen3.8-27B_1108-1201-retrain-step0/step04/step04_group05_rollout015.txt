"""
Minimize 2*n*max_conv/sum^2 via Nesterov-accelerated exact subgradient
with adaptive peak focusing, spectral moves, and diverse restarts.

Key innovations over previous approach:
1. Exact subgradient (no LSE bias) with Nesterov acceleration - avoids the smooth
   approximation bias that limited the previous method.
2. Adaptive top-k focusing: start with few peaks, gradually include more.
3. Two-scale optimization: coarse phase for global progress, fine phase for precision.
4. Spectral domain moves targeted at specific peak structures.
5. Intelligent restart strategy that preserves progress while exploring.
6. Pairwise interpolation moves for escaping saddle points.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_subgrad(a, top_idx, weights):
    """Compute weighted subgradient of top-k peaks of self-convolution."""
    n = len(a)
    g = np.zeros(n)
    for k, w in zip(top_idx, weights):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * w * a[ki - idx]
    return g


def get_top_peaks(c, k):
    """Get top-k peak indices and normalized weights (exponential decay)."""
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    max_val = top_vals[0]
    min_val = top_vals[-1]
    span = max_val - min_val
    if span < 1e-12:
        weights = np.ones(k) / k
    else:
        # Heavier weight on the actual maximum
        t = (max_val - top_vals) / span
        weights = np.exp(-2.0 * t)
        weights /= weights.sum()
    return top_idx, weights


def spectral_step(a, N, top_k, alpha, c):
    """Spectral shaping: reduce |A|^2 at frequencies contributing to peaks."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    if c is None:
        c = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    # Reduce power where it contributes constructively to peaks
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def nesterov_subgrad_block(a_init, n, S, N, max_time, rng,
                           top_k_start=5, top_k_end=50,
                           step_init=None, step_decay=0.999,
                           momentum_reset_iter=800, label=""):
    """
    Nesterov-accelerated exact subgradient on max(self-conv).
    Returns (best_a, best_maxval).
    """
    if step_init is None:
        step_init = 2.0 * S / np.sqrt(n)

    a = np.array(a_init, dtype=np.float64)
    a = project_simplex(a, S)
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = np.inf
    start_time = time.monotonic()
    it = 0
    step = step_init
    n_evals = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive top-k: grow from top_k_start to top_k_end
        frac = min(1.0, it / 2000.0)
        k_curr = int(top_k_start + (top_k_end - top_k_start) * frac)

        c = compute_conv(y, N)
        cur_max = c.max()
        n_evals += 1

        if cur_max < best_val:
            best_val = cur_max
            best_a = y.copy()
            if it % 200 == 0:
                elapsed = time.monotonic() - start_time
                print("    [%s] iter=%d, val=%.6f, time=%.0fs" % (label, it, best_val, elapsed))

        # Compute subgradient
        top_idx, weights = get_top_peaks(c, k_curr)
        g = compute_subgrad(y, top_idx, weights)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Nesterov step
        x_new = project_simplex(y - step * g / gnorm, S)

        # Update momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint approximately
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        x_prev = x_new
        t = t_new

        # Decay step size
        step *= step_decay

        # Periodic momentum reset to avoid oscillation
        if it % momentum_reset_iter == 0:
            t = 1.0
            x_prev = x_new.copy()
            y = x_new.copy()

        # Small random perturbation to escape local structure
        if it % 300 == 0 and rng.random() < 0.3:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            x_prev = y.copy()

    return best_a, best_val


def spectral_block(a_init, n, S, N, max_time, rng, top_k=10, alpha=0.15, label=""):
    """Iterative spectral shaping."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_val = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            if it % 100 == 0:
                print("    [%s/ spectral] iter=%d, val=%.6f" % (label, it, best_val))
        # Adaptive alpha based on peak excess
        mean_c = c.sum() / len(c)
        excess = max(0.0, cur - mean_c) / max(cur, 1e-10)

        a = spectral_step(a, N, top_k, alpha_curr, c)

            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best_val


def pairwise_interpolation_block(a_init, n, S, N, max_time, rng, label=""):
    """
    Try pairwise coordinate interpolation to escape local minima.
    For pairs (i,j), try interpolating a[i],a[j] -> (1-t)a[i]+t*a[j], (1-t)a[j]+t*a[i]
    and keep if it reduces the peak.
    """
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    c = compute_conv(a, N)
    best_val = c.max()
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Select random pairs
        n_pairs = min(200, n)
        for _ in range(n_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            t = rng.uniform(0.1, 0.9)
            a_new = a.copy()
            a_new[i] = (1 - t) * a[i] + t * a[j]
            a_new[j] = (1 - t) * a[j] + t * a[i]
            if a_new[i] < 0 or a_new[j] < 0:
                continue
            c_new = compute_conv(a_new, N)
            if c_new.max() < best_val - 1e-12:
                best_val = c_new.max()
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                # Revert
                a_new[i] = a[i]
                a_new[j] = a[j]

        if it % 10 == 0 and rng.random() < 0.1:
            # Occasional full random perturbation
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = compute_conv(a, N)
            if c.max() < best_val:
                best_val = c.max()
                best_a = a.copy()
            elif c.max() > best_val * 1.05:
                a = best_a.copy()  # Revert if too bad

    return best_a, best_val


def fine_polish(a_init, n, S, N, max_time, rng, label=""):
    """
    Very fine-grained polishing with tiny steps and high top-k.
    Uses a decreasing step schedule for precision.
    """

    best_val = c.max()
    start_time = time.monotonic()

        it += 1
        c = compute_conv(a, N)
        cur = c.max()

            best_a = a.copy()
