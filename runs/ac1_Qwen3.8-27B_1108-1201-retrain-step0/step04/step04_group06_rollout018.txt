        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
            stall = 0
            if it % 20000 == 0:
                print(f"    [SG {phase}] iter {it}: {best_score:.6f}")
        else:
            stall += 1

        # Temperature schedule
        conv_range = conv.max() - conv.min()
        if conv_range < 1e-10:
            break
        if phase == "main":
            temp = conv_range / (50.0 + 0.05 * it)
            tk = min(15, 3 + it // 3000)
        else:
            temp = conv_range / (200.0 + 0.02 * it)
            tk = min(8, 1 + it // 2000)

        temp = max(temp, 0.0005 * conv_range)

        # Compute smooth gradient
        grad = compute_smooth_gradient(conv, y, tk, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Adaptive step size
        step = step_scale * S / (n * (1.0 + 0.0005 * it))

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        beta = (t_mom - 1.0) / t_new
        x = project_simplex(y - step * grad / gnorm, S)
        y = x + beta * (x - a)
        y = np.maximum(y, 0.0)
        # Re-project y to maintain sum constraint
        y = project_simplex(y, S)
        a = x
        t_mom = t_new

        # Stall detection: try spectral kick
        if stall > 3000 and it % 3000 == 0:
            alpha = 0.2 + 0.3 * min(1.0, stall / 10000.0)
            a_kick = spectral_kick(best_a, N, n, alpha, rng,
                                   num_lags=min(10, 3 + it // 5000))
            conv_kick = fft_convolve_self(a_kick)
            if conv_kick.max() < best_score:
                best_score = conv_kick.max()
                best_a = a_kick.copy()
                a = a_kick.copy()
                y = a_kick.copy()
                t_mom = 1.0
                stall = 0
                step_scale = 1.0
            else:
                # Random perturbation
                a = project_simplex(best_a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                y = a.copy()
                t_mom = 1.0
                step_scale = max(0.1, step_scale * 0.5)
                stall = 0

        # Occasionally reduce step if not improving
        if it % 5000 == 0 and stall > 1000:
            step_scale *= 0.7

    return best_a, best_score


def coordinate_polish(a, n, S, max_time, rng, n_top=30):
    """
    Coordinate descent on top-contributing coordinates at the peak lag.
    """
    a = a.copy()
    best_a = a.copy()
    best_score = np.inf
    start = time.monotonic()

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    while time.monotonic() - start < max_time:
        conv = fft_convolve_self(a)
        c_max = conv.max()
        peak_k = int(np.argmax(conv))

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find top contributing coordinates
        j_lo = max(0, peak_k - n + 1)
        j_hi = min(n - 1, peak_k)
        j_range = np.arange(j_lo, j_hi + 1)
        contribs = a[j_range] * a[peak_k - j_range]

        top_coords = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for idx in top_coords:
            i = int(j_range[idx])
            j = peak_k - i
            if j < 0 or j >= n or i == j:
                continue

            orig_val = a[i]
            best_frac = 1.0
            best_cmax = c_max

            for frac in multipliers:
                a[i] = orig_val * frac
                # Recompute affected conv values
                # Changing a[i] affects conv[k] for k in [i, i+n-1]
                # Delta at conv[k]: (a[i]*frac - orig_val) * a[k-i] + (a[i]*frac - orig_val) * a[i-k] (if i-k in range and i-k != k-i)
                delta = a[i] - orig_val
                # Only check the peak and nearby
                k_check_lo = max(0, peak_k - 5)
                k_check_hi = min(2 * n - 2, peak_k + 5)
                # Recompute conv at peak
                new_peak = c_max
                # conv[peak_k] = sum_t a[t]*a[peak_k - t]
                # = orig conv[peak_k] - orig_val*a[j] + new_val*a[j] (if j != i)
                # = orig conv[peak_k] + delta * a[j] + (delta * a[j] if i != j else delta*orig_val + delta^2)
                if i != j:
                    new_peak_val = c_max + delta * a[j]
                else:
                    new_peak_val = c_max + 2 * delta * orig_val + delta * delta
                # Also need to check if other lags become the new max
                # For speed, just check if peak decreased
                if new_peak_val < best_cmax:
                    best_cmax = new_peak_val
                    best_frac = frac

            a[i] = orig_val * best_frac
            if best_frac != 1.0:
                improved = True

        # Rescale
        a = project_simplex(a, S)

        if not improved:
            # Perturb a few random top coordinates
            for _ in range(min(5, len(top_coords))):
                i = int(top_coords[np.random.randint(len(top_coords))])
                a[i] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)

    return best_a, best_score


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Returns the best sequence found.
    """
    deadline = time.monotonic() + max(0, budget_s - 25)
    rng = np.random.default_rng(777)

    global_best = np.inf
    global_a = None

    def try_update(a_candidate, n):
        nonlocal global_best, global_a
        a_candidate = np.clip(np.asarray(a_candidate, dtype=np.float64), 0, 1000)
        if a_candidate.sum() < 0.01:
            return False
        c = fft_convolve_self(a_candidate)
        sc = c.max()
        # Actual score: 2*n*sc/sum^2, but since we normalize sum=sqrt(2n), score=sc
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            print(f"  NEW BEST: {sc:.8f} (n={n})")
            return True
        return False

    # === MAIN: n = 8192 ===
    n = 8192
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()
    print(f"Main: n={n}, S={S:.4f}, FFT size={N}")

    # Try to load previous best
    inits = []
    a_prev = make_init(n, rng, "previous")
    inits.append(("prev", a_prev))

    # Additional initializations
    inits.append(("two_level", make_init(n, rng, "two_level")))
    inits.append(("sin", make_init(n, rng, "sin")))
    inits.append(("four_level", make_init(n, rng, "four_level")))

    # Score initializations
    for name, a_init in inits:
        c0 = fft_convolve_self(a_init)
        sc0 = c0.max()
        print(f"  Init [{name}]: {sc0:.6f}")
        try_update(a_init, n)

    # === Phase 1: Subgradient from best init (50% time) ===
    print("=== Phase 1: Subgradient (main) ===")
    t1_start = time.monotonic()
    t1_budget = (deadline - t1_start) * 0.50

    # Start from the best initialization
    best_init = min(inits, key=lambda x: fft_convolve_self(x[1]).max())[1]
    a1, s1 = subgradient_phase(best_init, n, N, S, min(t1_budget, 120), rng, phase="main")
    try_update(a1, n)

    # Continue with subgradient from current best
    a1b, s1b = subgradient_phase(global_a, n, N, S,
                                 max(0, deadline - time.monotonic()) * 0.45,
                                 rng, phase="main")
    try_update(a1b, n)

    print(f"  After Phase 1: {global_best:.6f}")

    # === Phase 2: Refined subgradient (25% time) ===
    print("=== Phase 2: Refined subgradient ===")
    t2_start = time.monotonic()
    t2_budget = (deadline - t2_start) * 0.50

    a2, s2 = subgradient_phase(global_a, n, N, S, min(t2_budget, 180), rng, phase="fine")
    try_update(a2, n)

    a2b, s2b = subgradient_phase(global_a, n, N, S,
                                 max(0, deadline - time.monotonic()) * 0.40,
                                 rng, phase="fine")
    try_update(a2b, n)

    print(f"  After Phase 2: {global_best:.6f}")

    # === Phase 3: Coordinate descent polish (remaining) ===
    print("=== Phase 3: Coordinate descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        a3, s3 = coordinate_polish(global_a, n, S, min(remaining, 300), rng, n_top=40)
        try_update(a3, n)

        # One more round of subgradient from CD result
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a4, s4 = subgradient_phase(global_a, n, N, S,
                                       min(remaining2, 120), rng, phase="fine")
            try_update(a4, n)

            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                a5, s5 = coordinate_polish(global_a, n, S,
                                           max(0, remaining3 - 5), rng, n_top=50)
                try_update(a5, n)

    print(f"  Final: {global_best:.6f}")

    # === Optional: Try n=4096 if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"=== Bonus: n=4096 ({remaining:.0f}s remaining) ===")
        n2 = 4096
        S2 = np.sqrt(2.0 * n2)
        N2 = 1 << (2 * n2 - 1).bit_length()

        a2_try = make_init(n2, rng, "two_level")
        a2_try2 = make_init(n2, rng, "sin")
        try_update(a2_try, n2)
        try_update(a2_try2, n2)

        best_n2 = min([a2_try, a2_try2], key=lambda x: fft_convolve_self(x).max())
        a_r, s_r = subgradient_phase(best_n2, n2, N2, S2,
                                     min(remaining - 30, 120), rng, phase="main")
        try_update(a_r, n2)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        result = [float(x) for x in global_a]
        ver = evaluate_sequence(result)
        print(f"Verification score: {ver:.8f}")
        if not np.isfinite(ver) or ver > global_best + 0.01:
            print("  Warning: verification mismatch, using fallback")
            result = [1.0] * 256
            ver = evaluate_sequence(result)
            print(f"  Fallback verification: {ver:.6f}")
        return result
    else:
        result = [1.0] * 256
        print(f"Fallback verification: {evaluate_sequence(result):.6f}")
        return result
