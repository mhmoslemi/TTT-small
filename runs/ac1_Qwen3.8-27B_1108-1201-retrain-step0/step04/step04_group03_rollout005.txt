"""
Minimize 2*n*max(convolve(f,f))/sum(f)^2 over non-negative sequences.
Hybrid approach: FISTA with adaptive LSE smoothing, spectral compression,
multiplicative repulsion, and periodic LP best-response.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta * c).max() approximation wrt a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_compress(f, gamma=2.0):
    """Compress the dynamic range of the power spectrum to flatten autocorrelation."""
    n = len(f)
    F = fft(f)
    power = np.abs(F) ** 2
    total_power = power.sum()
    dc_power = power[0]
    # Compress non-DC power
    non_dc = power[1:].copy()
    non_dc = non_dc ** (1.0 / gamma)
    # Renormalize non-DC to preserve total power
    non_dc_sum = non_dc.sum()
    if non_dc_sum > 1e-300:
        scale = (total_power - dc_power) / non_dc_sum
        non_dc *= scale
    power_new = np.empty(n)
    power_new[0] = dc_power
    power_new[1:] = non_dc
    F_new = np.sqrt(power_new) * np.exp(1j * np.angle(F))
    f_new = np.real(ifft(F_new))
    return np.maximum(f_new, 0.0)


def multiplicative_repulsion(f, S, beta=30.0, lr=0.05):
    """Multiplicative update based on convolution peaks."""
    n = len(f)
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    f_new = f_new * (S / s)
    return project_simplex(f_new, S)


def solve_lp_best_response(f, n, S, top_k=100, time_limit=8.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    result = None
    for solver in ['PDLP', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=solver, verbose=False)
            if time.monotonic() - start > time_limit:
                break
            result = g_var.value
            break
        except Exception:
            continue
    
    if result is not None and np.all(np.isfinite(result)):
        g = np.maximum(result, 0)
        s = g.sum()
        if s > 1e-10:
            g = g * (S / s)
            return g
    return None
