The issue is that reducing one peak might increase others. But if the step is small enough, the reduction in the target peak should dominate.

- After step: c[k*] ≈ M - alpha * ||proj(grad)||^2 (to first order)
- Other peaks c[k] change by: c[k] ≈ c[k] + alpha * grad_k · proj(grad_{k*})

If the peaks are well-separated (M >> c[k] for k != k*), then reducing the highest peak is safe.

    n = len(a)
    for it in range(max_iterations):
        c = compute_conv(a, N)
        k_star = np.argmax(c)
        M = c[k_star]

        # If the top peak is within a small fraction of the second peak, stop
        sorted_c = np.sort(c)[::-1]
        if sorted_c[0] - sorted_c[1] < 1e-6 * M:
            break

        k_int = int(k_star)

"""
Minimize 2*n*max_conv/sum^2 using exact active-set subgradient with momentum,
frequency-domain equalization, and adaptive multi-phase optimization.

Key differences from previous L-BFGS+LSE approach:
1. Exact active-set subgradient (top-K peaks) with momentum instead of LSE smoothing
2. Backtracking line search on the TRUE max (not smoothed surrogate)
3. Frequency-domain spectral equalization as complementary global move
4. Adaptive K (active set size) that grows as optimization progresses
5. Multi-scale: coarse-to-fine refinement
6. Longer sequences (n=16384) for better attack surface

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def next_pow2(x):
    """Next power of 2 >= x."""
    return 1 << (int(x) - 1).bit_length()


def compute_conv_fft(a, N):
    """Compute linear self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_grad_toppk(a, N, top_idx, weights, fsize):
    """Compute weighted subgradient for given active peaks using FFT."""
    n = len(a)
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def exact_toppk_subgrad(a, N, K, top_idx=None, top_vals=None):
    """Compute exact subgradient from top-K active peaks.
    Returns (grad, c, c_max)."""
    n = len(a)
    if top_idx is None:
        c = compute_conv_fft(a, N)
        c_max = c.max()
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]
    else:
        c = compute_conv_fft(a, N)
        c_max = c.max()
        top_vals = c[top_idx]
    
    # Use softmax-like weights to focus on the worst peaks
    margin = top_vals - top_vals.min()
    if margin.max() < 1e-15:
        weights = np.ones(K) / K
    else:
        # Exponential weighting: emphasize the worst peaks
        beta_w = 10.0 / max(margin.max(), 1e-15)
        exp_w = np.exp(beta_w * margin)
        weights = exp_w / exp_w.sum()
    
    fsize = next_pow2(3 * n - 2)
    grad = compute_grad_toppk(a, N, top_idx, weights, fsize)
    return grad, c, c_max, top_idx, top_vals


def nesterov_momentum_step(a, grad, S, n, N, momentum, step_scale, fsize):
    """One Nesterov-style momentum step with backtracking line search.
    Returns (a_new, c_max_new, momentum_new, K_new)."""
    c = compute_conv_fft(a, N)
    c_max_old = c.max()
    
    alpha = step_scale
    best_a = a
    best_cmax = c_max_old
    best_alpha = 0.0
    
    for ls in range(20):

        c_try = compute_conv_fft(a_try, N)

        if cmax_try <= c_max_old * (1.0 + 1e-12):
            best_a = a_try
            best_cmax = cmax_try
            best_alpha = alpha
            break
        alpha *= 0.5
    
    return best_a, best_cmax, best_alpha


def spectral_equalize(a, N, n, S, strength, n_freqs):
    """Reduce spectral peaks to flatten autocorrelation.
    In frequency domain, reduce |A[k]|^2 where it contributes most to time-domain peaks."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    
    # Find the time-domain autocorrelation
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    # Identify worst time-domain lag
    k_worst = int(np.argmax(c))
    
    # In freq domain, the contribution to c[k_worst] from frequency m is:
    # P[m] * cos(2*pi*m*k_worst/N)
    # We want to reduce P where this is large (positive contribution to peak)
    m_arr = np.arange(N)
    cos_contrib = np.cos(2.0 * np.pi * m_arr * k_worst / N)
    
    # Only reduce where contribution is positive
    mask = cos_contrib > 0
    reduction = strength * cos_contrib * mask
    P_new = np.maximum(P * (1.0 - reduction), 0.0)
    
    # Also equalize the spectrum: reduce top n_freqs spectral peaks
    top_spec = np.argsort(P_new)[::-1][:n_freqs]
    for k in top_spec:
        if k != 0:  # Don't touch DC (fixed by sum constraint)
            P_new[k] *= (1.0 - strength * 0.5)
    
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

        else:
            no_improve += 1
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step with momentum
        lr = 2.0 * S / (n * beta * max(gnorm / (S/n), 1e-10))
        lr = min(lr, 50.0 / gnorm)
        
        momentum_arr = mu * momentum_arr - lr * grad
        a_try = project_simplex(a + momentum_arr, S)
        
        c_try = compute_conv_fft(a_try, N)
        if c_try.max() <= c_max:
            a = a_try
        else:
            a = project_simplex(a - 0.1 * lr * grad / max(gnorm, 1e-15), S)
            momentum_arr *= 0.5
        
        # Perturbation
        if no_improve > 300 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            momentum_arr = np.zeros(n)
            no_improve = 0
    
    return best_a, best_cmax


def fine_polish(a_init, n, S, N, max_time, rng, fsize):
    """Very fine polishing with tiny exact subgradient steps."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    best_a = a.copy()
    best_cmax = np.inf
    K = 3  # Very small active set for precision
    
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        K = min(20, 3 + it // 100)
        
        grad, c, c_max, top_idx, top_vals = exact_toppk_subgrad(a, N, K)
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Very small step
        step = 0.5 * S / (n * gnorm)
        a_try = project_simplex(a - step * grad, S)
        c_try = compute_conv_fft(a_try, N)
        
        if c_try.max() < c_max:
            a = a_try
        else:
            # Try even smaller
            for s in [0.1, 0.01]:
                a_try2 = project_simplex(a - step * s * grad, S)
                c_try2 = compute_conv_fft(a_try2, N)
                if c_try2.max() < c_max:
                    a = a_try2
                    break
    
    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses exact active-set subgradient with momentum, frequency-domain equalization,
    and LSE-smoothed gradient as complementary methods.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    # Determine length
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= len(given) <= 65536:
            n = len(given)
    except Exception:
        pass
    
    # Try longer sequence if we have time
    if budget_s > 2000 and n < 16384:
        n_try = 16384
        # Check if FFT is feasible
        N_try = next_pow2(2 * n_try - 1)
        if N_try <= 65536:
            n = n_try
    
    S = np.sqrt(2.0 * float(n))
    N = next_pow2(2 * n - 1)
    fsize = next_pow2(3 * n - 2)
    
    print("n=%d, S=%.4f, N(FFT)=%d, fsize=%d, budget=%.0fs" % (n, S, N, fsize, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_cand, label=""):
        nonlocal global_best, global_a
        a_norm = project_simplex(np.asarray(a_cand, dtype=np.float64), S)
        c = compute_conv_fft(a_norm, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()
            if label:
                print("  [%s] %.8f" % (label, global_best))
        return sc
    
    # --- Phase 0: Load previous best ---
    prev_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            prev_a = np.clip(given, 0, 1000)
            S_prev = prev_a.sum()
            prev_a = prev_a * (S / S_prev)
            update_best(prev_a, "prev_best")
    except Exception:
        pass
    
    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(400.0, (deadline - t1) * 0.15)
    
    inits = []
    if prev_a is not None:
        inits.append(("prev", make_init_from_prev(prev_a, n, rng)))
    inits.append(("uniform", make_init_uniform(n, rng)))
    inits.append(("ramp", make_init_ramp(n, rng)))
    inits.append(("twobump", make_init_two_bumps(n, rng)))
    inits.append(("osc", make_init_oscillatory(n, rng)))
    
    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))
    
    for name, arr in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        a_norm = project_simplex(arr, S)
        # Quick main optimizer
        a_res, sc = main_optimizer(a_norm, n, S, N, min(per_init, rem - 10), rng, fsize,
                                    K_init=5, K_max=50, momentum_init=0.5)
        update_best(a_res, name + "/main")
    
    print("Phase 1 done. Best: %.8f" % global_best)
    
    # --- Phase 2: Focused optimization on best ---
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0
    
    print("=== Phase 2: Focused optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.5)
    
    a_curr = global_a.copy()
    stall = 0
    last_best = global_best
    
    while time.monotonic() < t2 + phase2_budget:
        rem = t2 + phase2_budget - time.monotonic()
        if rem < 30:
            break
        block_time = min(180.0, rem)
        
        # Alternate between main optimizer and LSE
        if stall % 2 == 0:
            a_new, sc_new = main_optimizer(a_curr, n, S, N, block_time, rng, fsize,
                                            K_init=10, K_max=300 + stall * 20,
                                            momentum_init=0.6 + 0.05 * min(stall, 5))
        else:
            a_new, sc_new = lise_optimizer(a_curr, n, S, N, block_time, rng, fsize,
                                            beta_init=20 + 10 * stall,
                                            beta_max=500 + 100 * stall)
        
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2] %.8f (stall was %d)" % (global_best, stall))
        else:
            a_curr = project_simplex((a_curr + a_new) / 2.0, S)
            stall += 1
            if stall >= 4:
                # Reset with perturbation
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                stall = 0
    
    # --- Phase 3: Spectral equalization refinement ---
    print("=== Phase 3: Spectral equalization ===")
    t3 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t3) * 0.15)
    
    a_sp = global_a.copy()
    for i in range(3):
        rem = phase3_budget / 3
        a_sp = spectral_equalize(a_sp, N, n, S, strength=0.05 * (i + 1), n_freqs=n // 100)
        a_sp = project_simplex(a_sp, S)
        # Follow up with main optimizer
        a_sp, sc_sp = main_optimizer(a_sp, n, S, N, rem, rng, fsize,
                                      K_init=20, K_max=200, momentum_init=0.5)
        update_best(a_sp, "spectral_%d" % i)
    
    # --- Phase 4: Fine polishing ---
    print("=== Phase 4: Fine polishing ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 30:
        a_fin, sc_fin = fine_polish(global_a, n, S, N, min(remaining - 10, 400.0), rng, fsize)
        update_best(a_fin, "polish")
    
    # --- Phase 5: Final LSE with very high beta ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final, sc_final = lise_optimizer(global_a, n, S, N, min(remaining - 5, 200.0),
                                            rng, fsize, beta_init=100, beta_max=2000)
        update_best(a_final, "final_LSE")
    
    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.8f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    print("Returning length %d, internal_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
