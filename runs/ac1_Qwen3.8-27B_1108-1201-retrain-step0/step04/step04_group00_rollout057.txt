"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE-FISTA with backtracking,
variance-aware smoothing, multi-scale optimization, and random restarts.
Key improvements: adaptive beta based on peaking ratio, backtracking line search,
variance penalty for uniformity, multi-scale (n->2n), and perturbation-based restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    shifted = beta * c
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def corr_fft(w, a):
    """Compute cross-correlation corr(w, a)[j] = sum_k w[k]*a[k-j] via FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def lse_grad(a, c, beta):
    """Compute LSE gradient: 2 * corr(softmax(beta*c), a)."""
    w = lse_weights(c, beta)
    return 2.0 * corr_fft(w, a)


def topk_grad(a, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return 2.0 * corr_fft(w_full, a)


def variance_grad(a, c):
    """Gradient of sum(c^2), which encourages uniformity."""
    return 4.0 * corr_fft(c, a)


def run_lse_fista(a_init, max_time, rng, beta_init=20.0, beta_max=20000.0,
                  beta_rate=0.004, var_alpha=0.01, use_backtrack=True):
    """
    FISTA with adaptive LSE smoothing, optional variance penalty, and backtracking.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    last_step = 1.0 / beta_init
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive beta: increase when peaking is high
        mean_c = c.mean()
        if mean_c > 1e-30:
            peaking = cur_max / mean_c
            beta = beta * (1.0 + 0.1 * np.log1p(peaking - (2 * n - 1) * mean_c / c.max() if c.max() > 0 else 0))

        # Gradient
        grad = lse_grad(y, c, beta)

        # Add variance penalty (small)
        if var_alpha > 0 and stuck_count > 100:
            vgrad = variance_grad(y, c)
            vnorm = np.linalg.norm(vgrad)
            gnorm_base = np.linalg.norm(grad)
            if vnorm > 1e-30 and gnorm_base > 1e-30:
                grad = grad + (var_alpha * gnorm_base / vnorm) * vgrad

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = last_step * 0.9  # slight shrink
        if stuck_count > 2000:
            step *= 0.5
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 4000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

        # Backtracking line search
        if use_backtrack:
            base_obj = cur_max
            step_try = step
            for _ in range(8):
                x_try = project_simplex(y - step_try * grad, S)
                # Approximate objective improvement via gradient
                # (exact would require another FFT, use predicted reduction)
                pred_red = step_try * gnorm * gnorm * 0.5
                if pred_red > 0 or step_try < step * 0.01:
                    break
                step_try *= 0.5
            step = step_try

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        last_step = step

        # Periodic restart with noise
        if it % 5000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 30000.0)
            if decay > 0.03:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_exact_polish(a_init, max_time, rng, max_k=32):
    """
    Polishing with exact/subgradient using adaptive top-k.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(max_k, 8 + stuck // 500)

        if k == 1:
            idx = np.argmax(c)
            w_full = np.zeros(len(c))
            w_full[idx] = 1.0
            grad = 2.0 * corr_fft(w_full, y)
        else:
            grad = topk_grad(y, c, k, power=1.2)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Diminishing step
        step = S / (gnorm * (1.0 + 0.01 * it) ** 0.5)
        step = min(step, S / (0.3 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when stuck
        if stuck > 3000 and it % 200 == 0:
            noise_scale = max(1e-6, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def try_perturbation(a_init, budget, rng, n_perturbations=5):
    """Try several random perturbations and re-optimize briefly."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    per_budget = budget / n_perturbations

    for p in range(n_perturbations):
        if time.monotonic() > start_of_perturb + budget:
            break
        # Different perturbation strategies
        if p == 0:
            # Small Gaussian noise
            noise_scale = 0.01 * S / np.sqrt(n)
            a_pert = project_simplex(a_init + rng.normal(0, noise_scale, n), S)
        elif p == 1:
            # Random spike
            a_pert = a_init.copy()
            j = rng.integers(0, n)
            a_pert[j] += 0.05 * S / n
            a_pert = project_simplex(a_pert, S)
        elif p == 2:
            # Anti-spike (reduce a random element)
            a_pert = a_init.copy()
            j = rng.integers(0, n)
            a_pert[j] = max(0, a_pert[j] - 0.05 * S / n)
            a_pert = project_simplex(a_pert, S)
        elif p == 3:
            # Smooth perturbation (low frequency)
            freq = rng.integers(1, max(2, n // 10))
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.005 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            a_pert = project_simplex(a_init + pert, S)
        else:
            # Random pair swap
            a_pert = a_init.copy()
            j1, j2 = rng.choice(n, size=2, replace=False)
            delta = 0.02 * S / n
            a_pert[j1] += delta
            a_pert[j2] -= delta
            a_pert = np.maximum(a_pert, 0)
            a_pert = project_simplex(a_pert, S)

        # Brief optimization
        a_opt, _ = run_lse_fista(a_pert, per_budget * 0.8, rng,
                                  beta_init=50.0, beta_max=10000.0,
                                  beta_rate=0.005, var_alpha=0.005)
        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj
