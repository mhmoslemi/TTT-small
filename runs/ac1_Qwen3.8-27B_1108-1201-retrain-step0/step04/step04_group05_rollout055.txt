"""
Minimize 2*n*max_conv/sum^2 using adaptive peak-shaving with spectral flattening.

Key innovations:
1. Exponential-weighted peak shaving: concentrate gradient effort on tallest peaks
2. Alternating time/frequency domain operations for complementary progress
3. Simulated annealing style acceptance to escape local minima
4. Adaptive K-schedule: progressively target more peaks as convergence improves
5. Tight line search on the exact (non-smoothed) max for precision
"""
import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. N must be >= 2*len(a)-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_gradient_weighted(a, w_peaks, N_grad):
    """
    Compute gradient of sum_k w_k * (a*a)_k with respect to a.
    w_peaks: array of length 2n-1 with weights at selected peak positions.
    Returns gradient of length n.
    """
    W = fft(w_peaks, N_grad)
    A = fft(a, N_grad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return grad


def spectral_flatten(a, N, alpha):
    """
    Apply spectral flattening: compress the dynamic range of |FFT(a)|^2.
    alpha in (0,1): smaller means more aggressive flattening.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_safe = np.maximum(P, 1e-30)
    P_new = P_safe ** alpha
    # Rescale to preserve sum (approximately)
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def spectral_flatten_v2(a, N, alpha, n_top):
    """
    Targeted spectral flattening: only flatten the top frequency bins.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    # Find top frequency bins
    top_idx = np.argsort(P)[::-1][:n_top]
    P_mean = P.mean()
    # Reduce top bins toward the mean
    for idx in top_idx:
        if P[idx] > P_mean:
            P[idx] = P[idx] * (1.0 - alpha) + P_mean * alpha
    A_new = np.sqrt(np.maximum(P, 1e-30)) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def peak_shaving_step(a, conv, n, S, N, N_grad, K, beta, alpha_init, rng):
    """
    One peak-shaving step with exponential weights and line search.
    Returns (new_a, new_max, accepted).
    """
    cur_max = conv.max()
    c_max_idx = np.argmax(conv)
    
    # Select top-K peaks
    if K >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -K)[-K:]
    
    top_vals = conv[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, cur_max, False
    
    # Exponential weights
    if beta > 0.01:
        w_vals = np.exp(beta * (top_vals - top_vals.max()))
        w_vals /= w_vals.sum()
    else:
        w_vals = top_vals / total
    
    # Build weight sequence
    w_seq = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_seq[int(k)] = w_vals[i]
    
    # Compute gradient
    grad = compute_gradient_weighted(a, w_seq, N_grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-18:
        return a, cur_max, False
    
    # Line search: try decreasing step sizes
    best_a = a
    best_max = cur_max
    alpha = alpha_init
    for _ in range(12):
        a_new = project_simplex(a - alpha * grad, S)
        c_new = compute_conv(a_new, N)
        max_new = c_new.max()
        if max_new < best_max:
            best_max = max_new
            best_a = a_new
            if max_new < cur_max * (1.0 - 0.001):
                break
        alpha *= 0.6
        if alpha < 1e-12:
            break
    
    accepted = best_max < cur_max
    return best_a, best_max, accepted
