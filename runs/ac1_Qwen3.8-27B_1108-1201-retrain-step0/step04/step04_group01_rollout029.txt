import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    """
    Iterative peak shaving:
    1. Compute b = convolve(a, a)
    2. Find top_k peaks
    3. Create symmetric weight vector w
    4. Compute gradient = 2 * convolve(w, a)[:n]
    5. Take a step in negative gradient direction, project to simplex
    """
    a = normalize_to_sum(a, S)
    fsize = 1 << (2*n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        k = min(top_k, len(c))
        idx = np.argsort(c)[::-1][:k]

        w = np.zeros(len(c))
        for i in idx:

            w[2*n-2-i] = 1.0  # Symmetrize

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_peak_gradient(a, c, top_k=32):
    """
    Compute gradient of max(b) using symmetric peak weights.
    
    Uses the fact that for symmetric w, gradient = 2 * (w * a)[:n].
    """
    n = len(a)
    c_len = len(c)  # 2n-1
    k = min(top_k, c_len)
    
    # Find top k peaks
    idx = np.argpartition(c, -k)[-k:]
    idx = idx[np.argsort(c[idx])[::-1]]
    
    # Create symmetric weight vector
    w = np.zeros(c_len)
    for i in idx:
        w[i] = 1.0
        symmetric_idx = c_len - 1 - i
        w[symmetric_idx] = 1.0
    
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    
    # Compute gradient: 2 * convolve(w, a)[:n]
    fsize = 1 << (c_len + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    
    return grad


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_peak_shave(a_init, S, max_time, rng, top_k=32, verbose=True):
    """
    Iterative peak shaving with adaptive step size.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    it = 0
    step = 1.0
    stall_count = 0
    last_best = best
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    iter {it}, elapsed {elapsed:.0f}s, max_b={cur_max:.6f}, best={best:.6f}")
        
        # Compute gradient
        grad = compute_peak_gradient(a, c, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Normalize gradient
        grad_norm = grad / gnorm
        
        # Adaptive step size
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Decrease step size as we progress
        base_step = 0.01 * S / np.sqrt(n)
        step = base_step * (1.0 + 2.0 * (1.0 - progress))
        
        # Try the step
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
            # Can try to increase step
            if c_new.max() < cur_max * 0.999:
                step *= 1.1
        else:
            # Backtrack
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            if fft_convolve_self(a_new).max() < cur_max:
                a = a_new
            else:
                # Give up this direction, add small noise
                step = base_step
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
        
        # Periodic restart with noise
        if stall_count > 3000 and progress < 0.85:
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stall_count = 0
            step = base_step
    
    return best_a, best


def run_lse_fista(a_init, S, max_time, rng, beta_init=50.0, beta_max=5000.0,
                  beta_rate=0.003, verbose=True):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: inverse of Lipschitz constant estimate
        # L ≈ 2 * max(|convolve(w, a)|) * ... 
        # Use a simpler estimate
        step = 1.0 / (beta * 0.3 + 1.0)
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart if stuck
        if it - last_improve > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
    
    return best_a, best


def run_combined(a_init, S, max_time, rng, verbose=True):
    """
    Combined optimization: LSE FISTA for exploration, then peak shaving for polishing.
    """
    n = len(a_init)
    start = time.monotonic()

        return a1, best1
