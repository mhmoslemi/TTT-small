def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)

"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Multi-phase approach:
1. Adaptive FISTA with LSE smoothing and backtracking
2. Spectral shaping guided by autocorrelation peak distribution  
3. Dual-guided coordinate descent with efficient O(n) objective updates
4. Simulated annealing for final local escape
5. Optional LP inner iterations for strong convexification

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max_b/sum_a^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def fista_gradient(a, beta):
    """Compute gradient of LSE-smoothed max autocorrelation.
    Returns (grad, c, fval) where fval = LSE approximation of max(c)."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        fval = cm
    else:
        w = e / tot
        fval = cm + np.log(tot) / beta

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c, fval


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=3000.0,
              beta_ramp=3000, use_backtrack=True, noise_period=5000):
    """FISTA with adaptive LSE smoothing, backtracking, and noise injection."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max

        grad, c, fval = fista_gradient(y, beta)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Step size with backtracking
        step = 1.0 / (2.0 * beta) * step_scale

        # Compute trial step
        x_new = project_simplex(y - step * grad, S)

        # Backtracking check (every 100 iters)
        if use_backtrack and it % 100 == 0:
            c_trial = fft_convolve_self(x_new)
            cur_trial = c_trial.max()
            cur_y = c.max()
            if cur_trial > cur_y * 1.05:
                step_scale = max(0.1, step_scale * 0.5)
            elif cur_trial < cur_y * 0.95:
                step_scale = min(2.0, step_scale * 1.1)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Track best
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Noise injection to escape local minima
        if it % noise_period == 0 and it - last_improve > noise_period // 2:
            decay = np.exp(-0.1 * (it - last_improve) / noise_period)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_momentum = 1.0
                step_scale = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_shaping(a_init, n, S, max_time, rng, mode='guided'):
    """Spectral shaping: modify |A(f)| to flatten autocorrelation.
    Mode 'guided': weight the flattening by which lags are currently problematic.
    """
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    alpha = 0.3

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        if mode == 'guided':
            # Identify which autocorrelation lags are problematic
            c = fft_convolve_self(a)
            c_mean = c.mean()
            # Weights: lags that are above mean get more "flattening pressure"
            excess = np.maximum(c - c_mean, 0)
            # The excess at lag k is controlled by |F(k)|^2 in frequency domain
            # Use the excess to modulate the flattening strength per frequency bin
            # Simple heuristic: stronger flattening when peaks are more concentrated
            peak_ratio = c.max() / max(c_mean, 1e-15)
            alpha = 0.3 + 0.5 * min(1.0, (peak_ratio - 1.0) / 2.0)
        else:
            alpha = 0.3 + 0.4 * min(1.0, it / 3000.0)

        # Soft flattening of power spectrum
        P_clipped = np.maximum(P, 1e-30)
        P_flat = np.power(P_clipped, alpha) / np.power(P_clipped, alpha).mean()
        P_new = (1.0 - alpha * 0.5) * P / P_mean + alpha * 0.5 * P_flat

        # Reconstruct
        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project to non-negative simplex
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def coord_descent(a_init, n, S, max_time, rng, n_pairs=100):
    """Coordinate descent with efficient O(2n) objective updates.
    Transfers mass between pairs (j, m) to reduce peak autocorrelation."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Select pairs: focus on indices that contribute to peak lags
        peak_lags = np.argsort(c)[::-1][:20]

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose j and m
            if rng.random() < 0.4:
                # Target indices involved in peak lags
                k_star = peak_lags[rng.integers(0, len(peak_lags))]
                j = rng.integers(0, n)
                m = (k_star - j + n) % n
                if m == j:
                    m = (j + 1) % n
            elif rng.random() < 0.7:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, n - 1)
                m = j + 1

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute change in c[k] per unit delta (transfer delta from m to j)
            # c[k] changes by: delta*(a[k-j]+a[j-k]) - delta*(a[k-m]+a[m-k])
            # (for the linear part; k=0 has an extra 2*delta^2 term for self-products)
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                if 0 <= k - j < n:
                    val += a[k - j]
                if 0 <= j - k < n:
                    val += a[j - k]
                if 0 <= k - m < n:
                    val -= a[k - m]
                if 0 <= m - k < n:
                    val -= a[m - k]
                d[k] = val

            # For k=0: additional quadratic term 2*delta^2 (from a[j]^2 - a[m]^2 change)
            # c[0] = a[0]^2 ... actually c[0] = a[0]*a[0] only if n>1
            # Let's handle this: c[k] = sum_i a[i]*a[k-i]
            # When we change a[j] -> a[j]+d and a[m] -> a[m]-d:
            # The change in c[k] includes cross terms involving both j and m
            # For simplicity, use the linear approximation + check

            lo = -a[j] + 1e-15
            hi = a[m] - 1e-15
            if lo >= hi:
                continue

            # Golden section search for min max(c + delta*d)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = (c + c_g * d).max()
            fd = (c + d_g * d).max()

            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = (c + c_g * d).max()
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = (c + d_g * d).max()

            delta = (a_g + b_g) / 2.0
            delta = max(lo, min(hi, delta))

            if abs(delta) < 1e-14:
                continue

            # Apply and check
            a[j] += delta
            a[m] -= delta
            if a[j] < 0 or a[m] < 0:
                a[j] -= delta
                a[m] += delta
                continue

            c = c + delta * d
            # Fix c[0] and c[2n-2] which have quadratic corrections
            # c[0] = a[0]^2: if j==0, c[0] += 2*a[0]*delta + delta^2
            # c[2n-2] = a[n-1]^2: if j==n-1, similar
            if j == 0:
                c[0] = a[0] ** 2
            elif m == 0:
                c[0] = a[0] ** 2
            if j == n - 1:
                c[2 * n - 2] = a[n - 1] ** 2
            elif m == n - 1:
                c[2 * n - 2] = a[n - 1] ** 2

            cur = c.max()
            if cur < best - 1e-13:
                best = cur
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                # Revert
                a[j] -= delta
                a[m] += delta
                c = c - delta * d
                if j == 0 or m == 0:
                    c[0] = a[0] ** 2
                if j == n - 1 or m == n - 1:
                    c[2 * n - 2] = a[n - 1] ** 2
                stall += 1

        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()
                stall = 0

        if stall > 200:
            # Add noise to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stall / 100.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            stall = 0

    return best_a, best


def simulated_annealing(a_init, n, S, max_time, rng, T_init=0.1, T_min=0.001):
    """Simulated annealing with efficient O(n) objective updates."""
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    cur_val = c.max()
    best = cur_val
    best_a = a.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Temperature schedule
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_init * (T_min / T_init) ** progress

        # Generate move: transfer delta from j to m
        j = rng.integers(0, n)
        m = rng.integers(0, n)
        if j == m or a[j] < 1e-12:
            continue

        # Delta: random small transfer
        max_delta = a[j] * 0.1
        delta = rng.uniform(-max_delta, max_delta)
        if abs(delta) < 1e-14:
            continue

        # Compute objective change approximately
        # d[k] = a[k-j] + a[j-k] - a[k-m] - a[m-k]
        d_k = 0.0
        # We only need the peak change, so compute d at the current peak
        # and at a few nearby lags
        peak_k = int(np.argmax(c))
        # For efficiency, just compute at peak and a few neighbors
        max_new = cur_val
        for k_check in [peak_k, peak_k - 1, peak_k + 1, 0, n - 1, 2 * n - 2]:
            k_check = max(0, min(2 * n - 2, k_check))
            val = 0.0
            if 0 <= k_check - j < n:
                val += a[k_check - j]
            if 0 <= j - k_check < n:
                val += a[j - k_check]
            if 0 <= k_check - m < n:
                val -= a[k_check - m]
            if 0 <= m - k_check < n:
                val -= a[m - k_check]
            new_c = c[k_check] + delta * val
            if new_c > max_new:
                max_new = new_c

        delta_obj = max_new - cur_val

        # Acceptance
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
            a[j] += delta
            a[m] -= delta
            if a[j] < 0 or a[m] < 0:
                a[j] -= delta
                a[m] += delta
                continue
            # Update c (approximate: only update affected entries)
            # For correctness, periodically do full recompute
            if it % 100 == 0:
                c = fft_convolve_self(a)
                cur_val = c.max()
            else:
                # Approximate update at peak
                c[peak_k] = max_new
                cur_val = max_new

            if cur_val < best:
                best = cur_val
                best_a = a.copy()

    return best_a, best


def lp_inner_iteration(f, n, S, n_constraints=200, max_time_lp=30.0):
    """Solve LP: min t s.t. (f*g)[k] <= t for top-k lags, g>=0, sum g=S.
    Uses scipy.optimize.linprog for speed."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    c_ff = fft_convolve_self(f)
    # Select active constraints (top lags)
    active_lags = np.argsort(c_ff)[::-1][:n_constraints]

    # Build constraint matrix: for each lag k, (f*g)[k] = sum_j f[k-j]*g[j]
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
            fi = k - j
            if 0 <= fi < n:
                rows.append(idx)
                cols.append(j)
                vals.append(f[fi])

    if len(rows) == 0:
        return f.copy()

    C = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))

    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: C@g - t <= 0  =>  C@g + [0,...,0,-1]*t <= 0
    # A_ub @ [g, t] <= b_ub where b_ub = 0
    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = C.toarray()
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(len(active_lags))

    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Bounds: g >= 0, t unbounded
    bounds = [(0, None)] * n + [(None, None)]

    try:
        start_lp = time.monotonic()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time_lp})
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                g = g * (S / s)
            return g
    except Exception:
        pass

    return f.copy()


def generate_structured_starts(n, rng):
    """Generate diverse structured initial sequences."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    starts.append(np.ones(n))

    # Cosine (half-wave rectified)
    starts.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))

    # Gaussian bump
    starts.append(np.exp(-4.0 * (tt - 0.5) ** 2))

    # Sinc-like
    starts.append(np.maximum(np.sinc(tt - 0.5) ** 2, 0.01))

    # Raised cosine with roll-off
    rc = 0.5 * (1 + np.cos(2 * np.pi * tt / 2))
    starts.append(rc)

    # Two-hump
    starts.append(0.5 * (np.exp(-20 * (tt - 0.25) ** 2) + np.exp(-20 * (tt - 0.75) ** 2)))

    # Random
    starts.append(rng.random(n) + 0.1)

    # Log-uniform random
    starts.append(10 ** rng.uniform(-2, 0, n))

    return starts


def upsample(a, n_new):
    """Upsample sequence to new length using FFT interpolation."""
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    fsize_old = 1 << (2 * n_old - 1).bit_length()
    fsize_new = 1 << (2 * n_new - 1).bit_length()
    A = fft(a, fsize_old)
    # Zero-pad in frequency (bandlimit)
    A_new = np.zeros(fsize_new, dtype=complex)
    half_old = fsize_old // 2
    half_new = fsize_new // 2
    A_new[:half_new] = A[:half_old]
    A_new[-half_new:] = A[-half_old:]
    a_new = np.real(ifft(A_new))[:n_new]
    return np.maximum(a_new, 0.0)

        S_i = np.sqrt(2.0 * arr_n)
        a_opt, _ = run_fista(arr, arr_n, S_i, t_alloc, rng,
                             beta_init=5.0, beta_max=2000.0,
                             beta_ramp=1000, use_backtrack=True)
        update_best(a_opt, "fista_" + name)

    print("Phase 1 complete: best=%.6f" % global_best)

    # Phase 2: LP inner iterations (20% budget)
    print("\n=== Phase 2: LP Inner Iterations ===")
    p2_end = t_start + (deadline - t_start) * 0.60
    if global_a is not None and time.monotonic() < p2_end - 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        f = global_a.copy()
        n_lp_iters = 0
        while time.monotonic() < p2_end - 10:
            n_lp_iters += 1
            # Solve LP for g
            g = lp_inner_iteration(f, n_best, S_best,
                                   n_constraints=300, max_time_lp=20.0)
            if g is None or np.allclose(g, f):
                break
            # Mix
            for mix in [0.3, 0.5, 0.7]:
                f_trial = (1 - mix) * f + mix * g
                f_trial = normalize_sum(f_trial, n_best)
                update_best(f_trial, "lp_mix_%d_%d" % (n_lp_iters, int(mix * 10)))
            f = normalize_sum(0.5 * f + 0.5 * g, n_best)

        print("Phase 2 complete: %d LP iters, best=%.6f" % (n_lp_iters, global_best))

    # Phase 3: Spectral shaping (15% budget)
    print("\n=== Phase 3: Spectral Shaping ===")
    p3_end = t_start + (deadline - t_start) * 0.75
    if global_a is not None and time.monotonic() < p3_end - 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_alloc = p3_end - time.monotonic() - 5
        a_spec, _ = spectral_shaping(global_a, n_best, S_best, t_alloc, rng, mode='guided')
        update_best(a_spec, "spectral")

        # Also try from a slightly perturbed version
        pert = project_simplex(global_a + rng.normal(0, 0.01 * S_best / np.sqrt(n_best), n_best), S_best)
        a_spec2, _ = spectral_shaping(pert, n_best, S_best, min(t_alloc, 60), rng, mode='guided')
        update_best(a_spec2, "spectral_pert")

    print("Phase 3 complete: best=%.6f" % global_best)

    # Phase 4: Coordinate descent (15% budget)
    print("\n=== Phase 4: Coordinate Descent ===")
    p4_end = t_start + (deadline - t_start) * 0.90
    if global_a is not None and time.monotonic() < p4_end - 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_alloc = p4_end - time.monotonic() - 5
        a_cd, _ = coord_descent(global_a, n_best, S_best, t_alloc, rng, n_pairs=80)
        update_best(a_cd, "coord_descent")

    print("Phase 4 complete: best=%.6f" % global_best)

    # Phase 5: Final FISTA + Simulated Annealing (10% budget)
    print("\n=== Phase 5: Final Polish ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # FISTA polish with high beta
            t_fista = remaining * 0.5
            a_fin, _ = run_fista(global_a, n_best, S_best, t_fista, rng,
                                 beta_init=200.0, beta_max=8000.0,
                                 beta_ramp=500, use_backtrack=True, noise_period=3000)
            update_best(a_fin, "final_fista")

            # Simulated annealing on best
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_sa, _ = simulated_annealing(global_a, n_best, S_best,
                                              remaining2 * 0.6, rng,
                                              T_init=0.05, T_min=0.001)
                update_best(a_sa, "sim_anneal")

                # One more CD pass
                remaining3 = deadline - time.monotonic()
                if remaining3 > 10:
                    a_cd2, _ = coord_descent(global_a, n_best, S_best,
                                             remaining3 * 0.8, rng, n_pairs=50)
                    update_best(a_cd2, "final_cd")

    # Phase 6: Try different length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Phase 6: Alternate Length ===")
        alt_n = n * 2 if n <= 8192 else n // 2
        if 1024 <= alt_n <= 65536:
            S_alt = np.sqrt(2.0 * alt_n)
            alt_start = upsample(global_a, alt_n)
            alt_start = normalize_sum(alt_start, alt_n)
            t_alt = remaining * 0.4
            a_alt, _ = run_fista(alt_start, alt_n, S_alt, t_alt, rng,
                                 beta_init=5.0, beta_max=1500.0,
                                 beta_ramp=500)
            update_best(a_alt, "alt_n_%d" % alt_n)

            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_alt2, _ = coord_descent(a_alt, alt_n, S_alt, remaining2 * 0.7, rng, n_pairs=50)
                update_best(a_alt2, "alt_n_cd")

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification: inf, using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback score: %.6f" % ver)

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
