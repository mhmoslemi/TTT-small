**Technique: Subgradient with exact line search**

The function $g(S) = \max_k c_k$ is convex in $S$ (as a function on $\mathbb{R}^n$). Wait, is it? Each $c_k$ is a quadratic form $S^T Q_k S$ where $Q_k$ has 1s at positions $(i, k-i)$. This is not necessarily convex.

Actually, $c_k = \sum_i S_i S_{k-i}$. The Hessian of $c_k$ with respect to $S$ is a matrix with 1s at positions $(i, k-i)$ and $(k-i, i)$. This matrix is NOT positive semi-definite in general. For example, for $k=0$, $c_0 = S_0^2$, and the Hessian has a 1 at position (0,0) and 0 elsewhere, which IS PSD. For $k=1$, $c_1 = 2S_0 S_1$, and the Hessian has 1s at (0,1) and (1,0), which is NOT PSD (its eigenvalues are 1 and -1).

So $c_k$ is not convex for $k \geq 1$. And $\max_k c_k$ is not convex either.

This means we can't use standard convex optimization. We need non-convex optimization techniques.

**Technique: Alternating minimization**

Split $S$ into two halves: $S = (S_L, S_R)$ where $S_L = (S_0, \ldots, S_{n/2-1})$ and $S_R = (S_{n/2}, \ldots, S_{n-1})$.

The peak $c_{n-1} = \sum_i S_i S_{n-1-i}$. This pairs $S_L$ with $S_R$ (in reversed order) and within-halves terms.

If we fix $S_L$ and optimize $S_R$ (and vice versa), the problem becomes:
$$\min_{S_R \geq 0} \max_k c_k \quad \text{s.t.} \quad \sum S_R = S_{\text{total}} - \sum S_L$$

This is still non-convex but has fewer variables.

with spectral domain optimization, phase perturbation, and adaptive

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a, N):
    """Full self-convolution via FFT. N is the FFT size."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2*len(a)-1]

def compute_score(a):
    """Compute the objective."""
    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    c = fft_convolve_self(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf, c
    return float(2 * n * c.max() / (S * S)), c

def spectral_flatten_iter(a, N, alpha, beta, rng):
    """
    One iteration of spectral flattening.
    alpha: compression exponent (0 < alpha < 1 means compress)
    beta: noise level for phase perturbation
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Flatten the power spectrum
    if alpha < 1.0:
        # Compress: bring high values down
        P_max = P.max()
        if P_max > 1e-30:
            P_normalized = P / P_max
            P_compressed = np.power(P_normalized, 1.0/alpha)  # > 1 exponent to compress
            P_new = P_compressed * P_max
        else:
            P_new = P.copy()
    elif alpha > 1.0:
        # Expand: bring low values up (less common)
        P_max = P.max()
        if P_max > 1e-30:
            P_normalized = P / P_max
            P_compressed = np.power(P_normalized, 1.0/alpha)
            P_new = P_compressed * P_max
        else:
            P_new = P.copy()
    else:
        P_new = P.copy()
    
    # Phase perturbation for diversity
    if beta > 0 and rng.random() < 0.3:
        noise_phase = rng.standard_normal(N) * beta
        phase = phase + noise_phase
    
    A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

def local_search_spectral(a, N, max_iters, rng, target_obj=1.50, verbose=False):
    """
    Intensive local search using spectral flattening with adaptive parameters.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score, _ = compute_score(a)
    no_improve = 0
    max_no_improve = 500
    
    # Start aggressive, gradually become gentle
    for it in range(max_iters):
        progress = it / max(max_iters - 1, 1)
        # Alpha annealing: start near 1 (mild compression), go toward smaller (aggressive)
        # Actually, let's start aggressive and become gentle
        alpha = 0.5 + 0.5 * progress  # 0.5 -> 1.0
        beta = 0.01 * (1 - progress)  # Phase noise decreases
        
        a_new = spectral_flatten_iter(a, N, alpha, beta, rng)
        sc, _ = compute_score(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, try a smaller step
            a_new = 0.5 * a + 0.5 * a_new
            a_new = project_simplex(a_new, S)
            sc, _ = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
        
        a = a_new
        
        if best_score <= target_obj:
            break
        if no_improve > max_no_improve and progress > 0.5:
            break
        if verbose and it % 1000 == 0:
            print("    spectral iter %d: score=%.6f" % (it, best_score))
    
    return best_a, best_score

def subgradient_local_search(a, N, max_time, rng, top_k=50):
    """
    Subgradient-based local search with Nesterov acceleration.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score, _ = compute_score(a)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y, N)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
        
        # Compute subgradient: use top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient of c_k w.r.t. a[i] is a[k-i] + a[i-(k-n+1)] (for valid indices)
        # More precisely: d(c_k)/d(a_i) = a[k-i] if 0 <= k-i < n
        # For the weighted sum: grad[i] = sum_j weights[j] * a[top_idx[j] - i] (if valid)
        # = sum_j weights[j] * (a * a_rev_padded)[i] ... 
        # Actually, the gradient of the circular convolution c_k = sum_i a_i a_{k-i} (mod N)
        # w.r.t. a_i is: a_{k-i} (for the specific k).
        # For the linear convolution, it's more complex but similar.
        
        # Simpler: compute the gradient via FFT
        # c = conv(a, a), so in frequency domain C = A * A
        # dc/dA = 2*A (for the magnitude)
        # But we want dc/da, which is 2 * conv(a, delta) where delta corresponds to the peak lag
        # Actually: c_k = sum_i a_i * a_{k-i}
        # dc_k/da_i = a_{k-i} + a_{i-k} (for the self-convolution, these might be the same or different)
        # Wait: c_k = sum_j a_j * a_{k-j}
        # dc_k/da_i: term where j=i gives a_{k-i}; term where k-j=i gives a_{k-i}... 
        # Actually: dc_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if i != k-i, 
        # or a_i (just once) if i == k-i.
        # More carefully: c_k = sum_j a_j a_{k-j}
        # d/d(a_i): sum_j [da_j/da_i * a_{k-j} + a_j * d(a_{k-j})/da_i]
        # = a_{k-i} + a_{k-i} = 2 a_{k-i} (when i != k-i, i.e., k is even and i = k/2, it's just a_{k/2})
        
        # For the weighted sum: grad_i = sum_lights weights_l * 2 * a[top_l - i]
        # This is like a weighted convolution of a with a "delta train" at the peak positions.
        
        # Efficient computation via FFT:
        # Create a "response" function r where r[k] = sum_lights weights_l for k in top_idx
        # Then grad = 2 * conv(r, a) (truncated to length n)
        r = np.zeros(N)
        for l in range(top_k):
            r[int(top_idx[l])] = weights[l]
        
        R = fft(r, N)
        A = fft(y, N)
        grad = 2.0 * np.real(ifft(R * A))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use diminishing step
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.01 * it))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        y = y_new
        # y is the "extrapolated" point, x_new is the actual iterate
        # For Nesterov, we evaluate at y and update from x_new
        
        t = t_new
        
        # Periodic restart
        if it % 1000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_a, best_score

def phase_annealing(a, N, max_iters, rng, T_start=0.05, T_end=0.001):
    """
    Simulated annealing in the phase domain.
    */
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score, _ = compute_score(a)
    
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    
    for it in range(max_iters):
        progress = it / max(max_iters - 1, 1)
        T = T_start * (T_end / T_start) ** progress
        
        # Perturb phase
        noise_scale = T * 0.1
        phase_pert = phase + rng.standard_normal(N) * noise_scale
        
        A_pert = mag * np.exp(1j * phase_pert)
        a_pert = np.real(ifft(A_pert))[:n]
        a_pert = project_simplex(a_pert, S)
        
        sc, _ = compute_score(a_pert)
        
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
            # Accept: update phase
            A = fft(a_pert, N)
            mag = np.abs(A)
            phase = np.angle(A)
        elif rng.random() < np.exp(-(sc - best_score) / T):
            # Accept worse solution (annealing)
            A = fft(a_pert, N)
            mag = np.abs(A)
            phase = np.angle(A)
    
    return best_a, best_score

def coordinate_descent_polish(a, N, max_time, rng):
    """
    Coordinate descent: randomly perturb individual elements.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score, _ = compute_score(a)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        # Pick a random element and perturb it
        i = rng.integers(n)
        delta = rng.normal(0, 0.01 * S / n)
        a[i] += delta
        if a[i] < 0:
            a[i] = 0
        # Renormalize to maintain sum
        s = a.sum()
        if s > 0:
            a = a * (S / s)
        
        sc, _ = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            a[i] = best_a[i]  # Revert
    
    return best_a, best_score

def make_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # U-shaped (high at edges, low in middle)

    # Random
    inits.append(("random", rng.random(n) + 0.05))
    
    # Smooth random
    noise = rng.standard_normal(n)
    from scipy.ndimage import gaussian_filter1d
    smooth = gaussian_filter1d(noise, sigma=n//32)
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth_rand", 0.2 + 0.6 * smooth))
    
    # Binary-like (balanced)

    inits.append(("binary", b))
    
    # Chirp
    inits.append(("chirp", 0.3 + 0.4 * np.abs(np.sin(2*np.pi*(2*tt + 3*tt**2)))))
    
    return inits

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using intensive local search combining:
1. Adaptive subgradient descent with Nesterov acceleration on top-k convolution lags
2. Spectral domain flattening with adaptive compression
3. Phase-domain simulated annealing for escaping local minima
4. Coordinated multi-scale search across sequence lengths

Key improvements over previous:
- Focused intensive local search rather than spreading thin
- Adaptive top-k window that narrows as we converge
- Better annealing schedule for phase perturbation
- Direct peak-equalization strategy in time domain

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full linear self-convolution via zero-padded FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def compute_score(a, N=None):
    """Compute objective value. Returns (score, convolution)."""
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    c = fft_convolve_self(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf, c
    return float(2.0 * n * c.max() / (S * S)), c


def spectral_flatten_step(a, N, compress, phase_noise, rng):
    """
    One spectral flattening iteration.
    compress: in (0,1), lower means more aggressive compression of dynamic range
    phase_noise: std of uniform phase perturbation (radians)
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    if compress < 1.0:
        P_max = P.max()
        if P_max > 1e-30:
            P_norm = P / P_max
            P_new = np.power(P_norm, 1.0 / compress) * P_max
        else:
            P_new = P.copy()
    else:
        P_new = P.copy()

    if phase_noise > 1e-8:
        phase = phase + rng.standard_normal(N) * phase_noise

    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def peak_equalization_step(a, c, N, top_k, strength):
    """
    Directly try to equalize the top-k convolution peaks.
    Direction: reduce a[i] proportionally to its contribution to the highest peaks,
    increase a[i] proportionally to its low contribution.
    """
    n = len(a)
    S = a.sum()

    # Get top-k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    c_max = c.max()

    if c_max < 1e-30:
        return a

    # For each top lag k, the "gradient" w.r.t. a[i] is a[k-i] (for valid i)
    # Build a composite gradient
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # d(c_k)/d(a_i) = a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k - i] for i in [lo, hi] -> indices k-hi to k-lo in a
            ai_slice = a[k - hi:k - lo + 1]
            grad[lo:hi + 1] += ai_slice[::-1]

    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a

    # Step: move against the gradient (to reduce peaks)
    a_new = a - strength * grad / gnorm
    a_new = project_simplex(a_new, S)
    return a_new


def intensive_subgradient(a, N, max_time, rng, base_top_k=100, verbose=False):
    """
    Adaptive subgradient descent with Nesterov acceleration.
    The top-k window shrinks as we converge (focusing on the critical lags).
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score, _ = compute_score(a, N)

    # State for Nesterov
    x = a.copy()  # current iterate
    y = a.copy()  # extrapolated point
    t = 1.0
    start = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y, N)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)

        if sc < best_score:
            best_score = sc
            best_a = y.copy()
            stale = 0
        else:
            stale += 1

        if verbose and it % 5000 == 0:
            print("      subgrad it=%d score=%.6f stale=%d" % (it, best_score, stale))

        if stale > 20000:
            break

        # Adaptive top-k: start wide, narrow as we converge
        progress = min(1.0, it / 50000.0)
        top_k = max(10, int(base_top_k * (1.0 - 0.7 * progress)))

        # Compute composite subgradient from top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Build response vector for FFT-based gradient
        r = np.zeros(N)
        for idx in top_idx:
            r[int(idx)] = weights[int(idx) - top_idx[0]] if len(top_idx) > 1 else weights[0]
        # Actually simpler: just accumulate

        for j in range(len(top_idx)):
            r[int(top_idx[j])] = weights[j]

        R = fft(r, N)
        A = fft(y, N)
        grad = 2.0 * np.real(ifft(R * A))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step_base = S / np.sqrt(n)
        step = step_base / (1.0 + 0.001 * it)

        x_new = project_simplex(y - step * grad / gnorm, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        y_sum = y_new.sum()
        if y_sum > 0:

        y = y_new
        x = x_new
        t = t_new

        if it % 5000 == 0:

            y = project_simplex(y + rng.standard_normal(n) * noise_scale, S)
            t = 1.0

    return best_a, best_score

        pn = 0.02 * (1.0 - progress)

            a_blend = project_simplex(a_blend, S)

                a = a_blend

        if verbose and it % 5000 == 0:

def phase_annealing(a, N, max_time, rng, target_obj=1.49):
    """
    Simulated annealing in the phase domain.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()

    start = time.monotonic()

    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    T_start = 0.1
    T_end = 0.001
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start * (T_end / T_start) ** progress

        # Perturb phase
        noise_scale = T * 0.2
        phase_pert = phase + rng.standard_normal(N) * noise_scale
        A_pert = mag * np.exp(1j * phase_pert)
        a_pert = np.real(ifft(A_pert))[:n]
        a_pert = project_simplex(a_pert, S)

        sc, _ = compute_score(a_pert, N)

        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
            A = fft(a_pert, N)
            mag = np.abs(A)
            phase = np.angle(A)
        elif rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
            A = fft(a_pert, N)
            mag = np.abs(A)

def make_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("u1", 0.5 + 0.4 * np.cos(np.pi * tt)))
    inits.append(("u2", 0.3 + 0.5 * (2 * tt - 1) ** 2))
    inits.append(("v", 0.3 + 0.4 * np.abs(2 * tt - 1)))
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("multi", 0.3 + 0.25 * np.abs(np.sin(3 * np.pi * tt)) + 0.15 * np.abs(np.sin(7 * np.pi * tt))))
    inits.append(("chirp", 0.3 + 0.35 * np.abs(np.sin(2 * np.pi * (2 * tt + 2 * tt**2)))))

    # Smoothed random
    noise = rng.standard_normal(n)
    # Simple boxcar smoothing
    k = max(3, n // 32)
    kernel = np.ones(k) / k
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth", 0.2 + 0.6 * smooth))

    return inits

            a_new, sc = intensive_subgradient(best_a, N, t_alloc, rng, base_top_k=tk)
        elif mname == "spectral":
            a_new, sc = spectral_local_search(best_a, N, t_alloc, rng, target_obj=best_score)
        elif mname == "peak_dir":
            # Peak direction descent
            S_val = best_a.sum()
            a_cur = best_a.copy()
            sc_cur, c_cur = compute_score(a_cur, N)
            it = 0
            while time.monotonic() - t0 < t0 + t_alloc:
                c_cur = fft_convolve_self(a_cur, N)
                strength = 0.1 * S_val / n * (0.5 + 0.5 * np.random.random())
                a_cur = peak_direction_step(a_cur, c_cur, N, strength)
                sc, _ = compute_score(a_cur, N)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
                it += 1
            a_new, sc = best_a, best_score
        elif mname == "phase":
            a_new, sc = phase_annealing(best_a, N, t_alloc, rng, target_obj=best_score)
        else:
            a_new, sc = best_a, best_score

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            break  # If we hit target, stop

    return best_a, best_score
