"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:

2. Entropy-guided exploration to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to have the desired sum."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the objective value (before the 2n/sum^2 normalization)."""
    c = fft_convolve_self(a)
    return c.max()


def multi_lag_subgradient(a, c, K=5, alpha=2.0):
    """Compute weighted subgradient over top-K lags of c.
    
    For each top-k lag k_m with weight w_m, the subgradient at coordinate j is:
        a[k_m - j] (if valid) + a[j - k_m] (if valid)
    """
    n = len(a)
    if K > len(c):
        K = len(c)
    # Get top K indices
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    # Geometric decay weights
    max_val = top_vals[0]
    weights = np.exp(-alpha * (max_val - top_vals) / max(max_val, 1e-30))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        k = int(k)
        # Contribution from a[k-j]: valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            # grad[lo1..hi1] += w * a[k - lo1 .. k - hi1] = w * a[k-hi1 .. k-lo1][::-1]
            src_lo = k - hi1
            src_hi = k - lo1
            grad[lo1:hi1+1] += w * a[src_lo:src_hi+1][::-1]
        # Contribution from a[j-k]: valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            # grad[lo2..hi2] += w * a[lo2-k .. hi2-k]
            src_lo = lo2 - k
            src_hi = hi2 - k
            grad[lo2:hi2+1] += w * a[src_lo:src_hi+1]
    return grad


def lse_gradient(a, c, beta):
    """Gradient of log-sum-exp approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def entropy_gradient(a, c):
    """Gradient of negative entropy of c (encourages flat c)."""
    n = len(a)
    c_safe = np.maximum(c, 1e-30)
    # d/da[j] of -sum_k c[k]*log(c[k]) = -sum_k (1+log(c[k])) * dc[k]/da[j]
    # = -sum_k (1+log(c[k])) * (a[k-j] + a[j-k])
    # This is a convolution: grad = -convolve((1+log(c_safe)), a_flip) - convolve((1+log(c_safe)), a)
    # More precisely: grad[j] = -sum_k (1+log(c[k]))*(a[k-j]+a[j-k])
    # = -conv(a, 1+log(c))[j] - conv(1+log(c), a)[j]... 
    # Actually it's: grad[j] = -sum_k w[k]*(a[k-j]+a[j-k]) where w[k]=1+log(c[k])
    # First term: sum_k w[k]*a[k-j] = (w * a_flipped)[j] = conv(w, a[::-1])[j + n - 1]... 
    # Let's just use FFT for efficiency
    w = 1.0 + np.log(c_safe)
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # conv(w, a) at position n-1+j gives sum_k w[k]*a[n-1+j-k]... 
    # Actually let me just compute directly using the formula:
    # grad[j] = -sum_k w[k]*(a[k-j] + a[j-k])
    # First sum: sum_k w[k]*a[k-j]. Let m = k-j, k = m+j. Sum over valid m: w[m+j]*a[m]
    # This is (w shifted) convolved with a. 
    # Using FFT: IFFT(FFT(w) * FFT(a_rev)) gives cross-correlation
    A_rev = fft(a[::-1].copy(), fsize)
    # conv(w, a)[k] = sum_i w[i]*a[k-i], what we want is sum_k w[k]*a[k-j]
    # = sum_i w[i+j]*a[i] = (w[j:] * a[:len])[sum] ... this is cross-correlation
    # cross_corr(w, a)[j] = sum_i w[i+j]*a[i]
    # = IFFT(conj(FFT(w)) * FFT(a))[j] (circular, with proper padding)
    corr1 = np.real(ifft(np.conj(W) * A))[:n]
    # Second sum: sum_k w[k]*a[j-k] = sum_i w[j-i]*a[i]... 
    # = sum_i a[i]*w[j-i] = conv(a, w)[j] (if we define conv properly)
    # conv(a,w)[j] = sum_i a[i]*w[j-i]
    # = IFFT(FFT(a) * FFT(w))[j]
    corr2 = np.real(ifft(A * W))[:n]
    grad = -(corr1 + corr2)
    return grad


def fista_multilag(a_init, n, S, max_time, rng, 
                   beta_init=5.0, beta_max=3000.0, beta_ramp=2000,
                   K_lags=5, alpha_decay=2.0, use_backtrack=True,
                   noise_have=0.001):
    """FISTA with multi-lag subgradient and optional backtracking."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step = 0.1 / n  # initial step size
    last_improve = 0
    c = fft_convolve_self(y)
    best = c.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp))
        else:
            beta = beta_max
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
            step = min(step * 1.05, 1.0 / (2.0 * beta + 1e-10))
        
        # Choose gradient: blend of LSE and multi-lag subgradient
        if it < 500:
            grad = lse_gradient(y, c, beta)
        else:
            # Blend: LSE for smoothness + multi-lag for sharpness
            w_smooth = max(0.0, 1.0 - it / 5000.0)
            w_sharp = 1.0 - w_smooth
            g_smooth = lse_gradient(y, c, min(beta, 500.0))
            g_sharp = multi_lag_subgradient(y, c, K=K_lags, alpha=alpha_decay)
            grad = w_smooth * g_smooth + w_sharp * g_sharp
            # Normalize to have similar scale
            gnorm_s = np.linalg.norm(g_smooth)
            gnorm_h = np.linalg.norm(g_sharp)
            if gnorm_s > 1e-30 and gnorm_h > 1e-30:
                grad = w_smooth * g_smooth * (gnorm_h / gnorm_s) + w_sharp * g_sharp
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search / step update
        if use_backtrack:
            # Backtracking: try to decrease the hard max
            trial_step = step
            for _ in range(10):
                x_new = project_simplex(y - trial_step * grad, S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= cur + 1e-10:
                    break
                trial_step *= 0.5
            step = trial_step
        else:
            step = 1.0 / (2.0 * beta + gnorm)
            if it - last_improve > 2000:
                step *= 0.7
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Noise injection
        if it % 2000 == 0:
            decay = np.exp(-it / 50000.0)
            ns = noise_have * S / np.sqrt(n) * decay
            if ns > 1e-10:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t = 1.0
        
        # Periodic momentum reset
        if it % 10000 == 0:
            t = 1.0
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

def peak_shaving(a_init, n, S, max_time, rng):
    """Targeted peak shaving: find top-contributing pairs and optimize transfers."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top peaks
        peak_lags = np.argsort(c)[::-1][:5]
        improved = False
        
        for k in peak_lags:
            k = int(k)
            # Find the coordinate pair (i, k-i) that most contributes to c[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            contributions = a[idx] * a[k - idx]
            # Find top contributor
            top_local = np.argmax(contributions)
            i_j = idx[top_local]
            i_k = k - i_j
            
            if i_j == i_k:
                # Can't transfer from a coordinate to itself
                continue
            
            # Optimal transfer: move delta from i_k to i_j (or vice versa)
            # c[k] changes by: delta * (a[i_j - other] stuff) - delta * (a[i_k - other] stuff)
            # More precisely, for a small transfer from i_k to i_j:
            # The change in c[m] for each lag m is:
            # dc[m] = delta * (a[m - i_j] - a[m - i_k]) + delta * (a[i_j - m + k... 
            # This is complex. Let's use the full gradient approach:
            # dc[m]/ddelta at delta=0 for transfer from i_k to i_j:
            # = a[m - i_j] + a[i_j - m] - a[m - i_k] - a[i_k - m] (with bounds)
            
            d = np.zeros(2 * n - 1)
            # Contribution from adding to i_j
            for m in range(max(0, i_j), min(2 * n - 1, i_j + n)):
                # a[m - i_j] term
                j2 = m - i_j
                if 0 <= j2 < n and j2 != i_j:
                    d[m] += a[j2]
                # a[i_j - m + ... ] term: a[i_j] appears as second factor when m - j = i_j, j = m - i_j
                # Wait, let me use the formula: dc[m]/da[j] = a[m-j] + a[j-m] (with bounds)
                pass
            
            # Simpler: just compute d directly
            # For transfer delta from i_k to i_j:
            # a[i_j] += delta, a[i_k] -= delta
            # dc[m] = delta * (da_c[m]/da[i_j] - da_c[m]/da[i_k])
            # da_c[m]/da[j] = a[m-j] (if 0<=m-j<n) + a[j-m] (if 0<=j-m<n)
            
            for m in range(2 * n - 1):
                # da_c[m]/da[i_j]
                v1 = 0.0
                t1 = m - i_j
                if 0 <= t1 < n:
                    v1 += a[t1]
                t2 = i_j - m
                if 0 <= t2 < n:
                    v1 += a[t2]
                # da_c[m]/da[i_k]
                v2 = 0.0
                t1 = m - i_k
                if 0 <= t1 < n:
                    v2 += a[t1]
                t2 = i_k - m
                if 0 <= t2 < n:
                    v2 += a[t2]
                d[m] = v1 - v2
            
            # Minimize max_m (c[m] + delta * d[m]) over delta in [-a[i_j], a[i_k]]
            # Use golden section search
            lo_d = -a[i_j]
            hi_d = a[i_k]
            
            if hi_d - lo_d < 1e-15:
                continue
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = np.max(c + c_g * d)
            fd = np.max(c + d_g * d)
            
            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = np.max(c + c_g * d)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = np.max(c + d_g * d)
            
            delta = (a_g + b_g) / 2.0
            
            # Check improvement
            c_trial = c + delta * d
            # Correction for the diagonal term (when i_j == m - i_j, i.e., m = 2*i_j)
            # c[m] contains a[i_j]^2 when m = 2*i_j, which changes to (a[i_j]+delta)^2
            # The linear approximation gives 2*a[i_j]*delta, but we also have delta^2
            if 0 <= 2 * i_j < 2 * n - 1:
                c_trial[2 * i_j] += delta * delta
            if 0 <= 2 * i_k < 2 * n - 1:
                c_trial[2 * i_k] -= delta * delta
            # Also cross terms at m = i_j + i_k
            if 0 <= i_j + i_k < 2 * n - 1:
                # The term a[i_j]*a[i_k] appears in c[i_j+i_k]
                # After transfer: (a[i_j]+delta)*(a[i_k]-delta) = a[i_j]*a[i_k] + delta*(a[i_j]-a[i_k]) - delta^2
                # Linear part is already in d, quadratic is -delta^2
                c_trial[i_j + i_k] -= delta * delta
                # And at m = i_k + i_j (same), already handled
                # Also m = |i_j - i_k| ... no, these are the same lag
            
            cur_trial = c_trial.max()
            if cur_trial < cur - 1e-12:
                a[i_j] += delta
                a[i_k] -= delta
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = np.ones(n) * S / n
                else:
                    a = a * (S / s)
                c = c_trial
                improved = True
        
        if not improved:
            # Add noise and recompute
            ns = 0.002 * S / np.sqrt(n) * np.exp(-it / 500.0)
            if ns < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best


def spectral_equalize(a_init, n, S, max_time, rng, 
                      alpha_init=0.3, alpha_max=0.9, momentum=0.7):
    """Spectral amplitude equalization with adaptive parameters."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    a_prev = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Adaptive alpha
        progress = min(1.0, it / float(max(1, it + 100)))
        alpha = alpha_init + (alpha_max - alpha_init) * min(1.0, it / 2000.0)
        
        # Flatten: P_new = P^alpha normalized
        P_alpha = np.power(np.maximum(P, 1e-30), alpha)
        P_alpha = P_alpha / P_alpha.mean()
        
        # Blend
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_alpha
        
        # Power conservation
        mag_new = np.sqrt(P_new)
        total_p = P.sum()
        total_new = P_new.sum()
        if total_new > 1e-30:
            mag_new *= np.sqrt(total_p / total_new)
        
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]
        
        # Project to non-negative with fixed sum
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)
        
        # Momentum
        a_new = momentum * a_new + (1.0 - momentum) * a_prev
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        
        # Adaptive momentum: reduce when converging
        if it % 1000 == 0:
            momentum = max(0.3, momentum * 0.95)
    
    return best_a, best


def make_antisymmetric_start(n, rng):
    """Create an asymmetric starting point with one-sided concentration."""
    a = np.zeros(n)
    # Exponential decay from one side
    t = np.arange(n)
    a = np.exp(-t / (n * 0.3))
    a += 0.1 * rng.random(n)
    return a


def make_bimodal_start(n, rng):
    """Create a bimodal starting point (concentrated at two locations)."""
    a = np.zeros(n)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.1
    a += np.exp(-((np.arange(n) - center1) / width) ** 2)
    a += 0.5 * np.exp(-((np.arange(n) - center2) / width) ** 2)
    a += 0.05 * rng.random(n)
    return a


def make_cosine_asymmetric_start(n, rng):
    """Cosine window with asymmetric slope."""
    t = np.arange(n) / n
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.3)))
    a = np.maximum(a, 0.01)
    # Apply asymmetric weighting
    a *= np.exp(-0.5 * t)
    a += 0.05 * rng.random(n)
    return a


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)
    
    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None
    
    # Determine length
    if parent is not None and 4096 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192
    
    S = np.sqrt(2.0 * n)
    
    print("Starting: n=%d, budget=%.0fs, parent='%s'" % (
        n, budget_s, 'available' if parent is not None else 'none'))
    
    global_best = np.inf
    global_a = None
    
    def _update_best(a_cand, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_cand, len(a_cand))
        score = compute_score(a_norm)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: conv_max=%.8f, obj=%.6f (t=%.0fs)" % (
                label, len(a_norm), score, 2.0 * len(a_norm) * score / (np.sum(a_norm)**2), elapsed))
        return score
    
    # Build diverse starting points
    starts = []
    if parent is not None:
        starts.append(("parent", normalize_sum(parent, n)))
        # Perturbed parent
        for scale in [0.02, 0.05, 0.1]:
            p = normalize_sum(parent, n)
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            starts.append(("parent_pct_%.2f" % scale, project_simplex(p + noise, S)))
        # Anti-symmetric perturbation of parent
        p = normalize_sum(parent, n)
        asym = np.linspace(1.0, 0.5, n)  # decay from left to right
        starts.append(("parent_asym", project_simplex(p * asym, S)))
    
    # Structured starts
    t = np.linspace(0, 1, n, endpoint=False)
    starts.append(("uniform", np.ones(n)))
    starts.append(("gaussian_c", np.exp(-20.0 * (t - 0.5)**2)))
    starts.append(("cosine", np.maximum(0.5*(1+np.cos(2*np.pi*t)), 0.01)))
    starts.append(("sinc_sq", np.maximum(np.sinc(t - 0.5)**2, 0.01)))
    starts.append(("anti_sym", make_antisymmetric_start(n, rng)))
    starts.append(("bimodal", make_bimodal_start(n, rng)))
    starts.append(("cos_asym", make_cosine_asymmetric_start(n, rng)))
    
    # Sinc with phase (asymmetric)
    a_sinc = np.sinc(t - 0.5)
    a_sinc = np.maximum(a_sinc, 0) * 0.5 + np.maximum(-a_sinc, 0) * 0.8 + 0.01
    starts.append(("sinc_asym", a_sinc))
    
    # Power-law decay
    a_pl = np.power(1.0 - t + 0.001, 0.5)
    starts.append(("power_decay", a_pl))
    
    print("  %d starting points generated" % len(starts))
    
    # Phase 1: Quick evaluation of all starts
    print("\n=== Phase 1: Evaluate starts ===")
    for name, arr in starts:
        arr_n = len(arr)
        S_i = np.sqrt(2.0 * arr_n)
        arr_norm = normalize_sum(arr, arr_n)
        _update_best(arr_norm, "start_" + name)
    
    # Phase 2: FISTA multi-lag on best starts
    print("\n=== Phase 2: FISTA Multi-Lag ===")
    p2_end = deadline - (deadline - t_start) * 0.45
    # Sort starts by initial score
    start_scores = []
    for name, arr in starts:
        s = compute_score(normalize_sum(arr, n))
        start_scores.append((s, name, arr))
    start_scores.sort()
    
    # Allocate time proportional to quality (best starts get more time)
    n_runs = min(6, len(start_scores))
    top_starts = start_scores[:n_runs]
    
    for i, (score, name, arr) in enumerate(top_starts):
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        # Allocate more time to earlier (better) starts
        t_alloc = remaining * (1.0 / (i + 1) - 1.0 / (n_runs + 1))
        if t_alloc < 10:
            t_alloc = remaining / (n_runs - i)
        if t_alloc < 5:
            break
        print("  Running FISTA on '%s' for %.0fs" % (name, t_alloc))
        a_opt, sc = fista_multilag(arr, n, S, t_alloc, rng,
                                    beta_init=5.0, beta_max=2000.0,
                                    beta_ramp=1500, K_lags=5, alpha_decay=2.0,
                                    use_backtrack=True, noise_have=0.001)
        _update_best(a_opt, "fista_" + name)
    
    print("  After Phase 2: best_obj=%.6f" % (2.0 * n * global_best / S**2))
    
    # Phase 3: Entropy-based exploration
    print("\n=== Phase 3: Entropy Exploration ===")
    if global_a is not None:
        p3_end = deadline - (deadline - t_start) * 0.25
        t_alloc = min(p3_end - time.monotonic(), 300)
        if t_alloc > 20:
            a_ent, sc = fista_entropy(global_a, n, S, t_alloc, rng)
            _update_best(a_ent, "entropy")
    
    print("  After Phase 3: best_obj=%.6f" % (2.0 * n * global_best / S**2))
    
    # Phase 4: Peak shaving
    print("\n=== Phase 4: Peak Shaving ===")
    if global_a is not None:
        p4_end = deadline - (deadline - t_start) * 0.15
        t_alloc = min(p4_end - time.monotonic(), 400)
        if t_alloc > 20:
            a_ps, sc = peak_shaving(global_a, n, S, t_alloc, rng)
            _update_best(a_ps, "peak_shave")
    
    print("  After Phase 4: best_obj=%.6f" % (2.0 * n * global_best / S**2))
    
    # Phase 5: Spectral equalization
    print("\n=== Phase 5: Spectral Equalization ===")
    if global_a is not None:
        t_alloc = min(deadline - time.monotonic() - (deadline - t_start) * 0.1, 300)
        if t_alloc > 20:
            a_sp, sc = spectral_equalize(global_a, n, S, t_alloc, rng,
                                          alpha_init=0.4, alpha_max=0.85, momentum=0.6)
            _update_best(a_sp, "spectral")
    
    print("  After Phase 5: best_obj=%.6f" % (2.0 * n * global_best / S**2))
    
    # Phase 6: Final FISTA polish with high beta
    print("\n=== Phase 6: Final Polish ===")
    if global_a is not None:
        t_alloc = min(deadline - time.monotonic(), 600)
        if t_alloc > 30:
            a_fin, sc = fista_multilag(global_a, n, S, t_alloc, rng,
                                        beta_init=200.0, beta_max=5000.0,
                                        beta_ramp=500, K_lags=3, alpha_decay=3.0,
                                        use_backtrack=True, noise_have=0.0005)
            _update_best(a_fin, "final_fista")
        
        # Another peak shaving pass
        t_alloc2 = deadline - time.monotonic()
        if t_alloc2 > 30:
            a_ps2, sc = peak_shaving(global_a, n, S, t_alloc2 * 0.8, rng)
            _update_best(a_ps2, "final_peak")
    
    # Phase 7: Try alternate length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        print("\n=== Phase 7: Alternate Length ===")
        # Try 2x or 0.5x length
        for alt_factor in [2, 0.5]:
            alt_n = int(n * alt_factor)
            if alt_n < 2048 or alt_n > 65536:
                continue
            alt_S = np.sqrt(2.0 * alt_n)
            if alt_factor > 1:
                # Upsample
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                # Downsample
                nblocks = n // alt_n
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)
            
            t_alt = (deadline - time.monotonic()) * 0.4
            if t_alt > 30:
                a_alt, sc = fista_multilag(alt_start, alt_n, alt_S, t_alt, rng,
                                            beta_init=5.0, beta_max=1000.0,
                                            beta_ramp=1000, K_lags=5, alpha_decay=2.0,
                                            use_backtrack=True)
                _update_best(a_alt, "alt_n_%d" % alt_n)
    
    # Phase 8: Multi-start from best with different seeds
    remaining = deadline - time.monotonic()
    if remaining > 100 and global_a is not None:
        print("\n=== Phase 8: Multi-Start Polish ===")
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            # Create perturbed version
            scale = 0.03 * (trial + 1)
            pert = global_a + rng.normal(0, scale * S / np.sqrt(n), n)
            pert = project_simplex(pert, S)
            t_trial = (deadline - time.monotonic() - 10) / (3 - trial)
            a_tr, sc = fista_multilag(pert, n, S, t_trial, rng,
                                       beta_init=50.0, beta_max=3000.0,
                                       beta_ramp=500, K_lags=5, alpha_decay=2.5,
                                       use_backtrack=True)
            _update_best(a_tr, "multistart_%d" % trial)
    
    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification: inf - using fallback")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
    
    print("Final: len=%d, score=%.6f, time=%.0fs" % (
        len(global_a), ver if np.isfinite(ver) else float('inf'), time.monotonic() - t_start))
    
    return final_list
