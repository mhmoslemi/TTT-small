"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smooth-max gradient descent
with mirror descent updates, adaptive temperature, and multi-phase optimization.

Key improvements over previous approach:
1. Log-sum-exp smooth approximation of the max for smooth gradient signals
2. Mirror descent (multiplicative) updates natural for non-negative variables
3. Adaptive temperature (beta) schedule: starts smooth, ends sharp
4. Adaptive step size based on gradient infinity norm
5. Multi-phase: quick flatten → smooth max optimization → subgradient polish
6. Periodic noise injection with annealing schedule

import numpy as np
import time
from scipy.fft import fft, ifft


def compute_conv(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = compute_conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def smooth_max_and_gradient(a, beta):
    """
    Compute smooth max of convolution and its gradient w.r.t. a.
    
    Returns: (smooth_max_value, gradient_array)
    """
    n = len(a)
    c = compute_conv(a)
    c_max = c.max()
    # Shift to avoid overflow
    c_shifted = c - c_max
    beta_c = beta * c_shifted
    # Clip to avoid overflow
    beta_c = np.clip(beta_c, -500, 500)
    exp_bc = np.exp(beta_c)
    log_sum_exp = c_max + np.log(exp_bc.sum()) / beta
    w = exp_bc / exp_bc.sum()
    
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m] where 0 <= k-m < n
    # This is 2 * (w cross-correlated with a)
    size = 1 << (2 * n - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    g = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    
    return log_sum_exp, g, c


def mirror_descent_update(a, g, eta):
    """Multiplicative mirror descent update with normalization."""
    # Center the gradient to preserve the sum more naturally
    g_centered = g - g.mean()
    # Scale to avoid overflow
    g_max = np.abs(g_centered).max()
    if g_max < 1e-30:
        return a.copy()
    # Use a scaled version
    exponent = -eta * g_centered / g_max
    exponent = np.clip(exponent, -5, 5)  # safety clip
    a_new = a * np.exp(exponent)
    a_new = np.maximum(a_new, 1e-30)
    S = a.sum()
    a_new = a_new * (S / a_new.sum())
    return a_new


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_polish(a_init, n, S, max_time, rng, top_k=50):
    """Nesterov-accelerated subgradient method on the exact max."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    
    N = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = compute_conv(y)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.02 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def spectral_flatten_step(a, N, alpha):
    """One step of spectral flattening."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    p_median = np.median(P)
    if p_median < 1e-30:
        return a.copy()
    
    ratio = P / p_median
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * (1.0 - alpha * (1.0 - np.power(ratio[above], -alpha)))
    P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - np.power(ratio[~above], alpha)))
    P_new = np.maximum(P_new, 0.0)
    
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    S = a.sum()
    a_new = project_simplex(a_new_full, S)
    return a_new

    N = 1 << (n - 1).bit_length()
    # Start with random phases
    phi = rng.uniform(0, 2*np.pi, N)

    if a_sum > 1e-10:
