"""
Minimize 2*n*max_conv/sum^2 using aggressive L-BFGS on LSE-smoothed peak,
peak-targeted coordinate shaving, and spectral flatness maximization.

Key innovations:
1. Peak-shaving: directly identify and reduce coordinates contributing to top peaks.
2. Spectral flattening: push power spectrum towards uniformity in frequency domain.
3. Aggressive L-BFGS with adaptive beta ramping and large memory.
4. Multi-phase schedule: coarse -> fine -> polish with different strategies per phase.
5. Restart mechanism with diverse perturbations when stalled.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, N, beta):
    """
    Gradient of LSE-smoothed max of conv(a,a).
    Returns (grad, lse_val, c, c_max).
    """
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def toppk_grad(a, N, top_k, beta):
    """
    Gradient using top-k weighted peaks (sharper than full LSE).
    Returns (grad, peak_val, c, c_max).
    """
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    # Get top-k indices
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    exp_bc = np.exp(beta * (top_vals - top_vals.max()))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c, c_max
    weights = exp_bc / total
    # Build weight vector
    w_full = np.zeros(len(c))
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    peak_val = top_vals.max() + np.log(total) / beta
    return grad, peak_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def peak_shave(a, N, top_k=5, alpha=0.05):
    """
    Direct peak shaving: identify top peaks, reduce contributing coordinates,
    redistribute to low-contribution coordinates.
    """
    n = len(a)
    S = a.sum()
    c, A = compute_conv(a, N)
    if top_k >= len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx = sorted(top_idx.astype(int))
    top_vals = c[top_idx]
    max_val = top_vals.max()
    if max_val < 1e-15:
        return a

    # Compute blame: which coordinates contribute most to top peaks
    blame = np.zeros(n)
    for k in top_idx:
        k = int(k)
        weight = (c[k] / max_val) ** 2  # emphasize larger peaks
        # conv[k] = sum_i a[i]*a[i+k] (with bounds)
        if k < n:
            # Forward: a[i] * a[i+k] for i = 0..n-1-k
            for_pair_a = a[:n - k]
            for_pair_b = a[k:n]
            blame[:n - k] += weight * for_pair_b
            blame[k:n] += weight * for_pair_a
        else:
            # k >= n: conv[k] = sum_{i=k-n+1}^{n-1} a[i]*a[i-(k-n+1)]
            # Actually for k in [n, 2n-2]: conv[k] = sum_{i=k-n+1}^{n-1} a[i] * a[i-(k-n+1)]
            # Let d = k - (n-1), then conv[k] = sum_{i=d}^{n-1-d} ... 
            # Simpler: use the symmetry conv[k] = conv[2n-2-k]
            k_sym = 2 * n - 2 - k
            if k_sym < n:
                for_pair_a = a[:n - k_sym]
                for_pair_b = a[k_sym:n]
                blame[:n - k_sym] += weight * for_pair_b
                blame[k_sym:n] += weight * for_pair_a

    # Normalize blame
    blame_max = blame.max()
    if blame_max < 1e-15:
        return a

    # Reduce high-blame coordinates, add to low-blame ones
    # Find threshold: reduce top 20% of blamed coordinates
    threshold = np.percentile(blame, 80)
    reduce_mask = blame > threshold
    add_mask = blame < np.percentile(blame, 20)

    total_reduce = a[reduce_mask].sum() * alpha
    if total_reduce < 1e-15:
        return a

    a_new = a.copy()
    a_new[reduce_mask] -= a[reduce_mask] * alpha
    # Distribute to add_mask proportionally to (S/n - a) to flatten
    add_weights = np.maximum(S / n - a[add_mask], 0.0)
    if add_weights.sum() < 1e-15:
        add_weights = np.ones(add_mask.sum())
    add_weights = add_weights / add_weights.sum()
    a_new[add_mask] += total_reduce * add_weights

    a_new = np.maximum(a_new, 0.0)
    return safe_normalize(a_new, n, S)


def spectral_flatten(a, N, alpha=0.3):
    """
    Push power spectrum towards flatness.
    In frequency domain: P_new = P^(1-alpha) * P_median^alpha
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Keep DC fixed, flatten the rest
    P_non_dc = P[1:]
    if P_non_dc.max() < 1e-15:
        return a
    P_median = np.median(P_non_dc[P_non_dc > 1e-15]) if np.any(P_non_dc > 1e-15) else 1.0
    P_median = max(P_median, 1e-10)

    P_new = P.copy()
    P_new[1:] = P_new[1:] ** (1 - alpha) * P_median ** alpha
    P_new[1:] = np.maximum(P_new[1:], 0.0)

    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return safe_normalize(np.abs(a_new), n, S)


def chirp_init(n, rng, rate=1.0):
    """Chirp-like initialization with anti-palindromic structure."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Anti-palindromic: large on one side, small on other
    a = 0.5 + 0.3 * t + 0.15 * np.sin(2 * np.pi * rate * t)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def binary_init(n, rng, density=0.5):
    """Random binary-like initialization."""
    a = (rng.random(n) < density).astype(np.float64)
    a *= 0.5 + 0.5 * rng.random(n)
    return np.clip(a, 0.0, None)


def anti_pal_init(n, rng):
    """Anti-palindromic initialization: a[i] * a[n-1-i] should be small."""
    left = rng.random(n // 2) + 0.1
    right = 1.0 / (left + 0.5)  # inversely related
    if n % 2 == 0:
        a = np.concatenate([left, right[::-1]])
    else:
        a = np.concatenate([left, [0.5], right[::-1]])
    return np.clip(a, 0.01, 10.0)

def spectral_block(a_init, n, S, N, budget, rng, alpha=0.2, flatten_interval=50):
    """Spectral flattening block."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < budget:
        it += 1
        c, A = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Spectral flattening
        if it % flatten_interval == 0 or it == 1:
            a = spectral_flatten(a, N, alpha=alpha)
        else:
            # Small gradient step
            beta = 30.0
            g, _, c, c_max = lse_grad(a, N, beta)
            gnorm = np.linalg.norm(g)
            if gnorm > 1e-15:
                step = 2.0 * S / (n * beta)
                a = project_simplex(a - step * g / gnorm, S)

        # Occasional noise
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_cmax

            a_ps, sc_ps = optimize_block(global_a, n, S, N, min(remaining2 - 5, 120.0), rng,
                                          beta_init=80.0, beta_max=400.0, m_lbfgs=15,
                                          use_toppk=True, top_k=3)
            update_best(a_ps, "polish_shaved")

    # Verification
    print("=== Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f (internal: %.6f)" % (ver, 2 * n * global_best / S ** 2))
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d" % len(global_a))
    return [float(x) for x in global_a]
