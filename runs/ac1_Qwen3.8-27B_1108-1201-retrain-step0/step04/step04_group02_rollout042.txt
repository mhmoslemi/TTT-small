def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):

    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def compute_objective(a):

    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()

def normalize(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def fourier_flat_design(n, rng):
    """
    Design a sequence whose spectrum is as flat as possible (subject to DC constraint).
    Uses iterative amplitude/phase adjustment.
    """
    S = np.sqrt(2.0 * n)

    fsize = 1 << (2 * n - 1).bit_length()

    # and freq-domain (enforce flat magnitude)

    A[0] = S
    A[1:] = np.sqrt(n / (fsize - 1)) * np.exp(1j * rng.uniform(0, 2 * np.pi, fsize - 1))
    
    a = np.zeros(n)
    for it in range(500):

        a_full = np.real(ifft(A))
        a = a_full[:n]
        a = np.maximum(a, 0)
        a = a * (S / max(a.sum(), 1e-15))
        
        # Freq domain: project to flat magnitude
        A = fft(a, fsize)
        mag = np.abs(A)
        target_mag = np.ones(fsize)
        target_mag[0] = S
        # For k > 0, target magnitude
        # We want |A[k]|^2 to be proportional to 1 (flat)
        # But A[0] = S, so if flat: |A[k]| = S/fsize for k>0 (by Parseval)
        # Actually, sum |A[k]|^2 = fsize * ||a||^2
        # We want |A[k]| = c for k>0, and A[0]=S
        # S^2 + (fsize-1)*c^2 = fsize * ||a||^2
        # Since ||a||^2 >= S^2/n (by C-S), we have fsize*S^2/n <= S^2 + (fsize-1)*c^2
        # So c^2 >= (fsize*S^2/n - S^2) / (fsize-1) = S^2*(fsize/n - 1)/(fsize-1)
        # For fsize >> n: c^2 ≈ S^2/n = 2
        # Let's just use c = sqrt(2) (since S^2 = 2n, so S^2/n = 2)

        A = target_mag * np.exp(1j * phase)
    
    a = np.maximum(a, 0)
    a = a * (S / a.sum())

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Mirror descent with LSE-smoothed objective, temperature continuation,
backtracking line search, targeted peak deflation, and multi-start exploration.
Focus on making the linear self-convolution as flat (low peak-to-mean ratio) as possible.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Compute gradient of LSE(beta, c) w.r.t. a, given c = a*a (full conv)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation of a and w)
    N = 1 << (n + len(w) - 1).bit_length()
    A_full = fft(a, N)
    W_full = fft(w, N)
    corr = np.real(ifft(np.conj(A_full) * W_full))[:n]
    return 2.0 * corr


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def mirror_step(a, grad, eta, S):
    """Multiplicative (mirror) descent step maintaining sum = S, a >= 0."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    log_a_new = log_a_new - log_a_new.max()
    a_new = np.exp(np.clip(log_a_new, -700, 700))
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new * (S / s)


def main_optimize(a_init, n, max_time, rng, verbose=True):
    """
    Main optimization: mirror descent with LSE, adaptive beta, backtracking.
    """
    a = normalize(a_init, n).copy()
    S = np.sqrt(2.0 * n)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)

    if verbose:
        print("  Main opt start: obj=%.6f" % best_obj)

    it = 0
    last_improve = 0
    eta = 1.0 / (30.0 * S)  # initial step for beta=30
    stuck_counter = 0

    total_time = max_time

    while time.monotonic() - start < total_time:
        it += 1
        frac = (time.monotonic() - start) / total_time

        # Beta schedule: 30 -> 800 with ease-in
        beta = 30.0 + 770.0 * (frac ** 0.6)

        # Compute gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: scale inversely with beta
        eta_target = 1.0 / (beta * S * 0.5)

        # Backtracking line search (max 8 halvings)
        step = min(eta, eta_target * 2)
        cur_obj = 2.0 * n * c.max() / (S * S)
        a_accepted = False

        for bt in range(8):
            a_try = mirror_step(a, grad, step, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)

            if obj_try < cur_obj - 1e-13:
                a = a_try
                c = c_try
                a_accepted = True
                break
            else:
                step *= 0.5

        if not a_accepted:
            # Add perturbation to escape
            stuck_counter += 1
            if stuck_counter > 20:
                # Structured perturbation
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    # Gaussian noise
                    noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                    a = np.maximum(a + noise, 1e-300)
                elif strategy == 1:
                    # Anti-peak: reduce coefficients contributing to the max
                    k_star = int(np.argmax(c))
                    lo = max(0, k_star - n + 1)
                    hi = min(k_star, n - 1)
                    i_vals = np.arange(lo, hi + 1)
                    contrib = a[k_star - i_vals]
                    scale = 0.01 * c[k_star] / max(contrib.max(), 1e-300)
                    a[i_vals] = np.maximum(a[i_vals] - scale * contrib, 1e-300)
                elif strategy == 2:
                    # Frequency domain perturbation
                    A = fft(a)
                    A *= (1 + rng.normal(0, 0.03, n) + 1j * rng.normal(0, 0.03, n))
                    a = np.real(ifft(A))
                    a = np.maximum(a, 1e-300)
                else:
                    # Edge compression (reduce a[0], a[n-1] which cause c[0], c[2n-2])
                    edge_decay = 0.5 ** (np.minimum(np.arange(n), n - 1 - np.arange(n)) / max(1, n // 16))
                    a = a * (0.5 + 0.5 * edge_decay)

                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                cur_obj = 2.0 * n * c.max() / (S * S)
                eta = 1.0 / (beta * S * 0.5)
                step = eta
                stuck_counter = 0
                last_improve = it
                continue

        # Update tracking
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            # Increase step slightly on success
            eta = min(eta * 1.03, eta_target)
        else:
            eta *= 0.85

        if verbose and it % 5000 == 0:
            print("    it=%d, beta=%.1f, obj=%.6f, best=%.6f" % (it, beta, cur_obj, best_obj))

    a = normalize(a, n)
    final_obj = compute_objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    if verbose:
        print("  Main opt done: best=%.6f (it=%d)" % (best_obj, it))
    return best_a, best_obj


def exact_polish(a_init, n, max_time, rng, top_k=1, verbose=True):
    """
    Fine polishing with exact top-k subgradient and RMSProp step size.
    """
    a = normalize(a_init, n).copy()
    S = np.sqrt(2.0 * n)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)

    if verbose:
        print("  Polish start: obj=%.6f" % best_obj)

    G_sq = np.zeros(n)  # RMSProp accumulator
    lr = 0.05
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Top-k subgradient
        k = min(top_k, 2 * n - 1)
        idx = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for j in idx:
            lo = max(0, j - n + 1)
            hi = min(j, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[j - i_vals]
        grad /= k

        # RMSProp step
        G_sq = 0.95 * G_sq + 0.05 * (grad ** 2)
        step = lr / (np.sqrt(G_sq) + 1e-10)

        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            a_try = a
        else:
            a_try = a_try * (S / s_try)

        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / (S * S)

        if obj_try < cur_obj - 1e-14:
            a = a_try
            c = c_try
            lr = min(lr * 1.005, 0.5)
        else:
            lr *= 0.7
            if it - last_improve > 5000:
                # Perturbation
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                G_sq = np.zeros(n)
                lr = 0.05
                last_improve = it

        if verbose and it % 10000 == 0:
            print("    polish it=%d, obj=%.6f" % (it, best_obj))

    a = normalize(a, n)
    final_obj = compute_objective(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    if verbose:
        print("  Polish done: best=%.6f (it=%d)" % (best_obj, it))
    return best_a, best_obj


def quick_exploration(a_init, n, max_time, rng):
    """Quick LSE mirror descent to evaluate a starting point."""
    a = normalize(a_init, n).copy()
    S = np.sqrt(2.0 * n)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        frac = (time.monotonic() - start) / max_time
        beta = 30.0 + 200.0 * frac

        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eta = 1.0 / (beta * S)
        cur_obj = 2.0 * n * c.max() / (S * S)

        for bt in range(6):
            a_try = mirror_step(a, grad, eta, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            if obj_try < cur_obj - 1e-13:
                a = a_try
                c = c_try
                break
            eta *= 0.5
        else:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = np.maximum(a + noise, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            continue

        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses mirror descent with LSE-smoothed objective, temperature continuation,
    backtracking, and multi-phase optimization with exact subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 50.0)
    t0 = time.monotonic()

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 1.0:
            given = np.clip(given, 0, 1000)
    except (NameError, Exception):
        pass

    global_best = np.inf
    global_a = None

    if given is not None:
        n = len(given)
        print("Using previous best with n=%d" % n)
    else:
        n = 8192
        print("No previous best found, using n=8192")

    S = np.sqrt(2.0 * n)

    # Generate initializations
    inits = []
    if given is not None:
        inits.append(("given", given.copy()))
        # Perturbations of given
        for i in range(4):
            p = given.copy()
            if i == 0:
                t = np.arange(n) / n
                p *= (1 + 0.2 * np.sin(2 * np.pi * (rng.integers(2, 8)) * t + rng.uniform(0, 6)))
            elif i == 1:
                A = fft(p)
                A *= (1 + rng.normal(0, 0.04, n) + 1j * rng.normal(0, 0.04, n))
                p = np.real(ifft(A))
            elif i == 2:
                nb = rng.integers(16, 64)
                scales = rng.uniform(0.8, 1.2, nb)
                for b in range(nb):
                    lo = b * n // nb
                    hi = (b + 1) * n // nb
                    p[lo:hi] *= scales[b]
            else:
                p += rng.normal(0, 0.02 * S / np.sqrt(n), n)
            p = np.maximum(p, 0)
            p = p * (S / p.sum())
            inits.append(("pert%d" % i, p))

    # Additional diverse inits
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(("cos1", np.maximum(1 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t + 0.7), 0.01)))
    inits.append(("cos2", np.maximum(1 + 0.25 * np.cos(2 * np.pi * 3 * t + 0.3) + 0.15 * np.cos(2 * np.pi * 7 * t), 0.01)))
    inits.append(("gauss", np.exp(-3.0 * (t - 0.5) ** 2 / 0.15)))
    inits.append(("rand", rng.random(n) + 0.05))

    n_inits = len(inits)
    print("Total initializations: %d" % n_inits)

    # Phase 1: Quick exploration (15% of available time)
    available = deadline - time.monotonic()
    phase1_time = available * 0.15
    per_init = phase1_time / n_inits

    print("=== Phase 1: Quick exploration (%.0fs total) ===" % phase1_time)
    for name, arr in inits:
        if time.monotonic() > deadline - available * 0.7:
            break
        t_alloc = max(3.0, per_init)
        a_opt, score = quick_exploration(arr, n, t_alloc, rng)
        score = compute_objective(a_opt)
        if score < global_best:
            global_best = score
            global_a = a_opt.copy()
            print("  [%s] obj=%.6f (t=%.0fs)" % (name, score, time.monotonic() - t0))

    print("Phase 1 best: %.6f" % global_best)

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    # Phase 2: Main optimization (60% of available time)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.62
    print("=== Phase 2: Main optimization (%.0fs) ===" % phase2_time)

    a_opt, score = main_optimize(global_a, n, phase2_time, rng, verbose=True)
    score = compute_objective(a_opt)
    if score < global_best:
        global_best = score
        global_a = a_opt.copy()

    # Phase 3: Second main optimization with different beta range (18% of remaining)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.40
    if phase3_time > 30:
        print("=== Phase 3: High-beta refinement (%.0fs) ===" % phase3_time)
        # Start from current best with higher initial beta
        a_opt2, score2 = main_optimize(global_a, n, phase3_time, rng, verbose=True)
        score2 = compute_objective(a_opt2)
        if score2 < global_best:
            global_best = score2
            global_a = a_opt2.copy()

    # Phase 4: Exact subgradient polishing (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Phase 4: Exact polish (%.0fs) ===" % remaining)
        a_opt3, score3 = exact_polish(global_a, n, remaining, rng, top_k=2, verbose=True)
        score3 = compute_objective(a_opt3)
        if score3 < global_best:
            global_best = score3
            global_a = a_opt3.copy()

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Verification: %.6f" % ver)
                if ver < global_best:
                    global_best = ver
            else:
                print("WARNING: evaluate_sequence returned inf, using fallback")
                global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        except Exception as e:
            print("Verification error: %s" % str(e))

    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    print("FINAL: length=%d, obj=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
