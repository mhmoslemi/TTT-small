"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP best-response iterations (with many/all constraints) at increasing n,
followed by FISTA LSE polishing. Uses Gurobi/SCS for LP solves with adaptive top_k.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v.sort()[::-1]
    # Use in-place sort for efficiency
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_lp_matrix(f, n, lags):
    """Build sparse constraint matrix for LP best response."""
    A = lil_matrix((len(lags), n))
    for row, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[row, i] = f[k - i]
    return csr_matrix(A)


def solve_best_response_lp(f, n, S, top_k, solver='GUROBICL', time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for selected lags, g >= 0.
    Returns normalized g or None.
    """
    c = fft_convolve_self(f)
    M = c.max()

    if top_k is None or top_k >= len(c):
        lags = np.arange(len(c))
    else:
        lags = np.argsort(c)[::-1][:top_k]

    # Build constraint matrix
    A = build_lp_matrix(f, n, lags)

    # Solve LP
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    solved = False
    for sol in [solver, 'SCS', 'ECOS']:
        try:
            prob.solve(solver=sol, verbose=False)
            if time.monotonic() - start > time_limit * 2:
                break
            solved = True
            break
        except Exception:
            continue

    if g_var.value is not None:
        g = np.maximum(np.asarray(g_var.value, dtype=np.float64).flatten(), 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g

    return None


def lp_iteration(f, n, S, top_k, solver='GUROBICL', time_limit=30.0):
    """One LP best-response iteration with step size selection."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()

    g = solve_best_response_lp(f, n, S, top_k, solver=solver, time_limit=time_limit)
    if g is None:
        return f.copy(), M_old, 0

    def eval_at_t(t):
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max()

    # Try t=1 first
    val1 = eval_at_t(1.0)
    if val1 < M_old:
        f_new = project_simplex(np.maximum(f + (g - f), 0), S)
        return f_new, val1, 1

    # Bisection for best t
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = M_old
    for _ in range(10):
        mid = (lo + hi) / 2.0
        val_mid = eval_at_t(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0.001:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val, best_t

    return f.copy(), M_old, 0


def run_lp_optimization(f_init, n, S, max_time, top_k, solver='GUROBICL',
                        time_per_lp=30.0, fista_every=20):
    """Run LP best-response iterations with periodic FISTA polishing."""
    f = normalize_sum(f_init, S)
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    lp_improvements = 0

    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        it += 1

        # LP iteration
        t_start = time.monotonic()
        f_new, M_new, t_used = lp_iteration(f, n, S, top_k, solver=solver,
                                             time_limit=min(time_per_lp, remaining * 0.3))
        lp_time = time.monotonic() - t_start

        if M_new < best:
            best = M_new
            best_f = f_new.copy()
            lp_improvements += 1
            f = f_new.copy()
            if lp_improvements % 5 == 0:
                print("    LP it %d: max_c=%.8f (t=%.3f, lp_time=%.1fs)" %
                      (it, best, t_used, lp_time))
        else:
            # No improvement from LP; try FISTA step
            f = f_new.copy()

        # Periodic FISTA polishing
        if it % fista_every == 0 and remaining > 30:
            f = fista_step(f, n, S, steps=200, beta=50.0)
            c_check = fft_convolve_self(f)
            if c_check.max() < best:
                best = c_check.max()
                best_f = f.copy()

    return best_f, best, lp_improvements


def fista_step(f, n, S, steps=200, beta=50.0):
    """Quick FISTA polishing step."""
    a = normalize_sum(f, S)
    y = a.copy()
    t = 1.0

    for _ in range(steps):
        c = fft_convolve_self(y)
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step_size = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step_size * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    return a


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.002,
              beta_max=800.0, use_repulsion=True, repulsion_freq=300):
    """Full FISTA with LSE and optional repulsion."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if time.monotonic() - start > max_time * 0.95:
            break

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve > 2000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if use_repulsion and it % repulsion_freq == 0:
            y = multiplicative_repulsion(y, n, S, beta=40.0, lr=0.04)
            t = 1.0

        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def multiplicative_repulsion(f, n, S, beta=50.0, lr=0.05):
    """Fast multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return f.copy()
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length using cubic-like interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def compute_objective(a, n):
    """Compute the objective 2*n*max(c)/(sum(a))^2."""
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def smart_top_k(c, n, max_k):
    """Choose adaptive top_k based on the distribution of c values."""
    cm = c.max()
    # Include all lags within 1% of the max, plus some margin
    threshold = cm * 0.999
    n_above = np.sum(c >= threshold)
    # Also include at least 20% of lags for robustness
    min_k = max(100, n // 5)
    return min(max_k, max(n_above + 50, min_k))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses LP best-response iterations with increasing dimensionality,
    followed by FISTA polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 45.0)

    global_best = np.inf
    global_a = None

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Budget: %.0fs, given seq len: %s" % (budget_s, len(given) if given is not None else "None"))

    # === Phase 1: LP optimization at n=2048 (all constraints, fast) ===
    n1 = 2048
    S1 = np.sqrt(2.0 * n1)
    t0 = time.monotonic()

    if given is not None:
        f1 = interpolate_sequence(given, n1)
    else:
        tt = np.linspace(0, 1, n1, endpoint=False)
        f1 = 0.54 + 0.46 * np.cos(2 * np.pi * tt)
    f1 = normalize_sum(f1, S1)

    print("=== Phase 1: LP at n=%d ===" % n1)
    obj1 = compute_objective(f1, n1)
    print("  Initial obj: %.6f" % obj1)

    # Use all constraints for n=2048
    f1, best1, n_imp1 = run_lp_optimization(f1, n1, S1, 250.0, top_k=None,
                                             solver='GUROBICL', time_per_lp=20.0,
                                             fista_every=10)
    obj1 = compute_objective(f1, n1)
    print("  Phase 1 best: %.6f (improvements: %d, time: %.0fs)" %
          (obj1, n_imp1, time.monotonic() - t0))

    if obj1 < global_best:
        global_best = obj1
        global_a = f1.copy()

    # === Phase 2: LP optimization at n=4096 (all or near-all constraints) ===
    remaining = deadline - time.monotonic()
    if remaining < 60:
        print("Time running out, skipping phase 2+")
    else:
        n2 = 4096
        S2 = np.sqrt(2.0 * n2)
        f2 = interpolate_sequence(global_a, n2)
        f2 = normalize_sum(f2, S2)
        t0 = time.monotonic()

        print("=== Phase 2: LP at n=%d ===" % n2)
        obj2 = compute_objective(f2, n2)
        print("  Initial obj: %.6f" % obj2)

        # Use top 3000 constraints for n=4096
        f2, best2, n_imp2 = run_lp_optimization(f2, n2, S2, min(remaining * 0.3, 500.0),
                                                 top_k=3000,
                                                 solver='GUROBICL', time_per_lp=30.0,
                                                 fista_every=15)
        obj2 = compute_objective(f2, n2)
        print("  Phase 2 best: %.6f (improvements: %d, time: %.0fs)" %
              (obj2, n_imp2, time.monotonic() - t0))

        if obj2 < global_best:
            global_best = obj2
            global_a = f2.copy()

    # === Phase 3: LP optimization at n=8192 ===
    remaining = deadline - time.monotonic()
    if remaining < 60:
        print("Time running out, skipping phase 3+")
    else:
        n3 = 8192
        S3 = np.sqrt(2.0 * n3)
        f3 = interpolate_sequence(global_a, n3)
        f3 = normalize_sum(f3, S3)
        t0 = time.monotonic()

        print("=== Phase 3: LP at n=%d ===" % n3)
        obj3 = compute_objective(f3, n3)
        print("  Initial obj: %.6f" % obj3)

        # Adaptive top_k: start with 1000, increase if useful
        time_for_p3 = min(remaining * 0.35, 800.0)
        f3, best3, n_imp3 = run_lp_optimization(f3, n3, S3, time_for_p3,
                                                 top_k=1000,
                                                 solver='GUROBICL', time_per_lp=30.0,
                                                 fista_every=10)
        obj3 = compute_objective(f3, n3)
        print("  Phase 3 best: %.6f (improvements: %d, time: %.0fs)" %
              (obj3, n_imp3, time.monotonic() - t0))

        if obj3 < global_best:
            global_best = obj3
            global_a = f3.copy()

    # === Phase 4: FISTA polishing at the best n ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n4 = len(global_a)
        S4 = np.sqrt(2.0 * n4)
        t0 = time.monotonic()
        print("=== Phase 4: FISTA polish at n=%d (time=%.0fs) ===" % (n4, remaining * 0.3))

        f4, best4 = run_fista(global_a, n4, S4, remaining * 0.3, rng,
                              beta_init=30.0, beta_rate=0.003, beta_max=1000.0,
                              use_repulsion=True, repulsion_freq=200)
        obj4 = compute_objective(f4, n4)
        print("  Phase 4 best: %.6f (time: %.0fs)" % (obj4, time.monotonic() - t0))

        if obj4 < global_best:
            global_best = obj4
            global_a = f4.copy()

    # === Phase 5: Try a different n (larger) ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n5_base = len(global_a)
        n5 = n5_base * 2 if n5_base <= 8192 else n5_base // 2
        if 2048 <= n5 <= 32768:
            S5 = np.sqrt(2.0 * n5)
            f5 = interpolate_sequence(global_a, n5)
            f5 = normalize_sum(f5, S5)
            t0 = time.monotonic()
            print("=== Phase 5: Trying n=%d (time=%.0fs) ===" % (n5, remaining * 0.3))

            # Brief LP + FISTA
            f5, best5, _ = run_lp_optimization(f5, n5, S5, remaining * 0.15,
                                                top_k=500, solver='GUROBICL',
                                                time_per_lp=20.0, fista_every=20)
            f5, best5b = run_fista(f5, n5, S5, remaining * 0.15, rng,
                                   beta_init=50.0, beta_rate=0.002, beta_max=800.0)
            f5 = f5 if best5 < best5b else f5
            # Use whichever is better
            c5a = fft_convolve_self(f5)
            obj5 = compute_objective(f5, n5)
            print("  Phase 5 (n=%d) best: %.6f (time: %.0fs)" % (n5, obj5, time.monotonic() - t0))

            if obj5 < global_best:
                global_best = obj5
                global_a = f5.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n6 = len(global_a)
        S6 = np.sqrt(2.0 * n6)
        print("=== Phase 6: Final polish (remaining=%.0fs, n=%d) ===" % (remaining, n6))

        # High-beta FISTA
        f6, best6 = run_fista(global_a, n6, S6, remaining * 0.5, rng,
                              beta_init=100.0, beta_rate=0.001, beta_max=2000.0,
                              use_repulsion=False)
        obj6 = compute_objective(f6, n6)
        if obj6 < global_best:
            global_best = obj6
            global_a = f6.copy()
            print("  Phase 6a: %.6f" % obj6)

        # Quick LP round
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            f6b, best6b, _ = run_lp_optimization(global_a, n6, S6, remaining2 * 0.6,
                                                  top_k=500, solver='GUROBICL',
                                                  time_per_lp=15.0, fista_every=30)
            obj6b = compute_objective(f6b, n6)
            if obj6b < global_best:
                global_best = obj6b
                global_a = f6b.copy()
                print("  Phase 6b: %.6f" % obj6b)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final verification: %.6f" % ver if np.isfinite(ver) else "Final verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
