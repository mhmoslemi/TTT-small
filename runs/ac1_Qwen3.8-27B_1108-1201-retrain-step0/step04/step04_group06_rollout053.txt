"""
Minimize 2*n*max_conv/sum^2 using LP-based spectral water-filling with
reduced active-set constraints, logsumexp smooth-max refinement via
multiplicative weights, and coordinate descent polish.

Key innovations over parent:
1. LP water-filling: solves a reduced linear program to find the optimal

2. Logsumexp smooth-max with multiplicative weight updates: uses a smooth

3. Adaptive active-set management: grows the constraint set as peaks stabilize.
4. Combined multi-phase strategy balancing global (LP) and local (CD) search.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = (S - cumsum[-1]) / n if cumsum[-1] < S else 0.0
        rho = 0
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    B = fft(b, L)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]

def lp_waterfill_step(f, M, top_k, S, timeout=30.0):

    rows_list = []
    cols_list = []
    vals_list = []
    for row_i, k in enumerate(top_idx):

    for t in t_candidates:
        c_new = (1 - t)**2 * c_f + 2 * t * (1 - t) * c_fg + t**2 * c_g
        M_new = c_new.max()
        if M_new < best_M:
            best_M = M_new
            best_t = t
            best_f = (1 - t) * f + t * g

    if best_t == 0.0:
        return f, M, b_opt, False

    best_f = np.maximum(best_f, 0.0)
    best_f = project_simplex(best_f, S)
    M_final = fft_self_convolve(best_f).max()
    return best_f, M_final, b_opt, True


def logsumexp_phase(a, S, max_seconds, beta_start=5.0, beta_end=50.0,
                    eta_start=0.05, eta_end=0.01, n_steps=None):
    """
    Multiplicative weight descent with logsumexp smooth-max objective.
    """
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()
    best_score = fft_self_convolve(a_curr).max()
    start = time.monotonic()

    if n_steps is None:
        n_steps = 500

    for step in range(n_steps):
        if time.monotonic() - start > max_seconds:
            break

        frac = step / max(n_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start + (eta_end - eta_start) * frac

        c = fft_self_convolve(a_curr)
        cur_max = c.max()
        score = cur_max
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        # Compute softmax weights
        c_shifted = beta * (c - c.max())
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / w_sum

        # Compute gradient via FFT
        L = 1 << (3 * n - 1).bit_length()
        W = fft(w, L)
        A = fft(a_curr, L)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Normalize gradient for stability
        g_max = g.max()
        if g_max < 1e-15:
            break
        g = g / g_max

        # Multiplicative update
        a_new = a_curr * np.exp(-eta * g)
        a_new = project_simplex(a_new, S)

        # Line search: try a few eta values
        best_eta_val = eta
        best_eta_score = score
        for eta_try in [eta * 0.5, eta, eta * 1.5, eta * 2.0]:
            a_try = a_curr * np.exp(-eta_try * g / g_max)
            a_try = project_simplex(a_try, S)
            s_try = fft_self_convolve(a_try).max()
            if s_try < best_eta_score:
                best_eta_score = s_try
                best_eta_val = eta_try
                a_new = a_try

        a_curr = a_new

        # Periodic random perturbation to escape local minima
        if step % 100 == 50:
            rng_local = np.random.default_rng(step)
            a_curr = a_curr + rng_local.normal(0, 0.002 * S / np.sqrt(n), n)
            a_curr = project_simplex(a_curr, S)

    return best_a, best_score


def coordinate_descent_polish(a_init, S, max_seconds, n_top=25, n_rounds=200):
    """
    Coordinate descent on top-contributing coordinates at peak lags.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = fft_self_convolve(a).max()
    start = time.monotonic()
    multipliers = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.1]

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_seconds:
            break

        c = fft_self_convolve(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Focus on top few lags
        top_lags = np.argsort(c)[::-1][:5]

        improved = False
        for lag in top_lags:
            lag = int(lag)
            if time.monotonic() - start > max_seconds:
                break

            j_lo = max(0, lag - n + 1)
            j_hi = min(lag, n - 1)
            contribs = np.array([a[j] * a[lag - j] for j in range(j_lo, j_hi + 1)])
            top_coords = np.argsort(contribs)[::-1][:n_top]

            for tc in top_coords:
                j = j_lo + int(tc)
                if j >= n:
                    continue
                k = lag - j
                if k < 0 or k >= n or j == k:
                    continue

                orig = a[j]
                best_frac = 1.0
                best_local = c_max

                for frac in multipliers:
                    a[j] = orig * frac
                    # Quickly recompute affected lags
                    k_lo = max(0, j)
                    k_hi = min(2 * n - 2, j + n - 1)
                    c_check = c.copy()
                    for kk in range(k_lo, k_hi + 1):
                        delta = (a[j] - orig)
                        m1 = kk - j
                        if 0 <= m1 < n:
                            c_check[kk] += delta * a[m1]
                        m2 = j
                        kk2 = kk - m2
                        if 0 <= kk2 < n and kk2 != m1:
                            c_check[kk] += delta * a[kk2]

                    new_max = c_check[k_lo:k_hi + 1].max()
                    other_part = 0.0
                    if k_lo > 0:
                        other_part = max(other_part, c_check[:k_lo].max())
                    if k_hi + 1 < len(c_check):
                        other_part = max(other_part, c_check[k_hi + 1:].max())
                    total_max = max(new_max, other_part)

                    if total_max < best_local:
                        best_local = total_max
                        best_frac = frac

                a[j] = orig * best_frac
                if best_frac != 1.0:
                    improved = True

        a = project_simplex(a, S)

        if not improved:
            rng_local = np.random.default_rng(rnd + 1000)
            n_perturb = min(10, n // 100)
            idxs = rng_local.choice(n, size=n_perturb, replace=False)
            for idx in idxs:
                a[int(idx)] *= rng_local.uniform(0.5, 1.5)
            a = project_simplex(a, S)

    return best_a, best_score


def structured_init(n, rng, method="sin"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "sin":
        a = (0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t)
             + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n))
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = (0.3 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 4) ** 2)
             + 0.2 * np.exp(-0.5 * ((t - 0.7) * 4) ** 2))
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.1 * rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = 64
        piece_len = max(1, n // n_pieces)
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, 0.3 + 0.7 * rng.random(n - len(a))])
        a = a[:n]
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.01, 3.0)


def safe_normalize(a, n):
    """Clip to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def multi_start_lp(a_init, S, n, max_seconds, initial_top_k=20, max_top_k=500):
    """
    Iterative LP water-filling with growing active set.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_M = fft_self_convolve(a).max()
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < max_seconds:
        it += 1
        top_k = min(max_top_k, initial_top_k + it * 5)
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

        f_new, M_new, b_opt, success = lp_iteration(a, best_M, S, top_k, t_candidates)

        if success and M_new < best_M:
            best_M = M_new
            best_a = f_new.copy()
            a = f_new.copy()
            stall = 0
            if it % 10 == 0:
                print("      LP iter %d: M=%.8f, top_k=%d" % (it, best_M, top_k))
        else:
            # Try with current max as the bound (tighter)
            c_curr = fft_self_convolve(a)
            M_curr = c_curr.max()
            if M_curr < best_M:
                best_M = M_curr
                best_a = a.copy()

            f_new, M_new, b_opt, success = lp_iteration(a, M_curr * 1.001, S, top_k, t_candidates)
            if success and M_new < best_M:
                best_M = M_new
                best_a = f_new.copy()
                a = f_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    # Perturb and retry
                    rng_local = np.random.default_rng(it + 7777)
                    a = a + rng_local.normal(0, 0.003 * S / np.sqrt(n), n)
                    a = project_simplex(a, a.sum(), )
                    # Renormalize
                    a = project_simplex(a, S)
                    stall = 0

    return best_a, best_M


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses LP-based water-filling, logsumexp refinement, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))

    print("n=%d, S=%.6f, budget=%.0fs" % (n, S, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = safe_normalize(a_candidate, n)
        c = fft_self_convolve(a_candidate)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            return True
        return False

    # --- Build initializations ---
    inits = []

    # Try previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= n and np.sum(np.clip(given[:n], 0, 1000)) > 0:
            inits.append(("given", np.clip(given[:n], 0, 1000).astype(np.float64)))
        elif given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    for method in ["sin", "chirp", "bump", "alternating", "piecewise"]:
        inits.append((method, structured_init(n, rng, method)))
    inits.append(("rand", 0.3 + 0.7 * rng.random(n)))

    print("  %d initializations" % len(inits))

    # === Phase 1: LP Water-Filling from best init (45% time) ===
    print("=== Phase 1: LP Water-Filling ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.45

    # Find best init quickly
    best_init = None
    best_init_score = np.inf
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c0 = fft_self_convolve(a_norm)
        sc0 = c0.max()
        if sc0 < best_init_score:
            best_init_score = sc0
            best_init = a_norm
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()

    if global_a is None:
        global_a = safe_normalize(structured_init(n, rng, "sin"), n)
        global_best = fft_self_convolve(global_a).max()

    print("  Best init score: %.8f" % global_best)

    # Run LP water-filling from best init
    a_curr = global_a.copy()
    rem = deadline - time.monotonic()
    lp_time = min(rem * 0.6, phase1_budget)
    a_lp, M_lp = multi_start_lp(a_curr, S, n, lp_time,
                                initial_top_k=30, max_top_k=400)
    if update_best(a_lp):
        print("  LP phase 1: %.8f" % global_best)

    # Run LP from a few other inits if time permits
    rem = deadline - time.monotonic()
    if rem > 30:
        other_inits = [arr for name, arr in inits if name not in ("given",)]
        per_init_time = rem * 0.4 / max(len(other_inits), 1)
        for name, arr in inits:
            if time.monotonic() > deadline - 60:
                break
            if per_init_time < 10:
                break
            a_norm = safe_normalize(arr, n)
            a_lp2, M_lp2 = multi_start_lp(a_norm, S, n, min(per_init_time, 60.0),
                                          initial_top_k=20, max_top_k=200)
            if update_best(a_lp2):
                print("  LP from [%s]: %.8f" % (name, global_best))

    print("=== Phase 1 done. Best: %.8f ===" % global_best)

    # === Phase 2: Logsumexp refinement (30% time) ===
    print("=== Phase 2: Logsumexp Refinement ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.30

    a_curr = global_a.copy()
    # Run logsumexp from current best
    a_ls, score_ls = logsumexp_phase(a_curr, S, min(phase2_budget * 0.5, 300.0),
                                     beta_start=10.0, beta_end=80.0,
                                     eta_start=0.03, eta_end=0.005,
                                     n_steps=300)
    if update_best(a_ls):
        print("  Logsumexp 1: %.8f" % global_best)

    # Another round with different params
    rem = deadline - time.monotonic()
    if rem > 20:
        a_ls2, score_ls2 = logsumexp_phase(global_a, S, min(rem * 0.5, 200.0),
                                           beta_start=30.0, beta_end=200.0,
                                           eta_start=0.02, eta_end=0.003,
                                           n_steps=200)
        if update_best(a_ls2):
            print("  Logsumexp 2: %.8f" % global_best)

    # Interleave with short LP bursts
    rem = deadline - time.monotonic()
    if rem > 30 and global_a is not None:
        a_burst, M_burst = multi_start_lp(global_a, S, n, min(rem * 0.4, 120.0),
                                          initial_top_k=50, max_top_k=600)
        if update_best(a_burst):
            print("  LP burst: %.8f" % global_best)

    print("=== Phase 2 done. Best: %.8f ===" % global_best)

    # === Phase 3: Coordinate descent polish (15% time) ===
    print("=== Phase 3: Coordinate Descent ===")
    t_phase3 = time.monotonic()
    phase3_budget = (deadline - t_phase3) * 0.15

    if global_a is not None and phase3_budget > 10:
        a_cd, score_cd = coordinate_descent_polish(global_a, S, min(phase3_budget, 300.0),
                                                   n_top=30, n_rounds=300)
        if update_best(a_cd):
            print("  CD polish: %.8f" % global_best)

        # One more LP round after CD
        rem = deadline - time.monotonic()
        if rem > 20:
            a_lp3, M_lp3 = multi_start_lp(global_a, S, n, min(rem * 0.5, 120.0),
                                          initial_top_k=100, max_top_k=800)
            if update_best(a_lp3):
                print("  LP after CD: %.8f" % global_best)

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 10:
            a_cd2, _ = coordinate_descent_polish(global_a, S, min(rem, 120.0),
                                                 n_top=40, n_rounds=200)
            if update_best(a_cd2):
                print("  Final CD: %.8f" % global_best)

    print("=== Phase 3 done. Best: %.8f ===" % global_best)

    # === Phase 4: Final exploration (10% time) ===
    print("=== Phase 4: Final Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 30 and global_a is not None:
        # Try different beta ranges
        for beta_s, beta_e in [(5, 40), (50, 300)]:
            if time.monotonic() > deadline - 30:
                break
            a_try, _ = logsumexp_phase(global_a, S, min((deadline - time.monotonic()) * 0.5, 60.0),
                                       beta_start=beta_s, beta_end=beta_e,
                                       eta_start=0.01, eta_end=0.002,
                                       n_steps=100)
            if update_best(a_try):
                print("  Exploration (beta %d-%d): %.8f" % (beta_s, beta_e, global_best))

        # Short LP with larger active set
        if time.monotonic() < deadline - 30:
            a_try, _ = multi_start_lp(global_a, S, n, min(deadline - time.monotonic() - 30, 90.0),
                                      initial_top_k=200, max_top_k=1500)
            if update_best(a_try):
                print("  Exploration LP: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.8f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
