"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization with:
1. Multi-scale initialization (coarse-to-fine via interpolation upsampling)
2. FISTA with adaptive LSE smoothing and simulated annealing restarts
3. Direct spectral reshaping (Gerchberg-Saxton with adaptive parameters)
4. Peak-focused coordinate descent
5. Anti-palindromic structured initializations

Key novelty: Exploits the insight that anti-palindromic structures
(low a[i]*a[n-1-i]) reduce the central convolution peak, combined
with boundary tapers to suppress edge contributions.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the evaluation objective (without the evaluate_sequence wrapper)."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_smooth_max(c, beta):
    """Log-sum-exp approximation to max(c)."""
    cm = c.max()
    if beta < 1.0:
        return cm
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm
    return cm + np.log(tot) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot  # weights on each lag
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def run_fista_sa(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=3000.0,
                 beta_ramp_iters=3000, sa_prob=0.05, sa_noise=0.02):
    """FISTA with adaptive LSE and simulated annealing restarts."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    best_fista_it = 0

    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: ramp up then hold
        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()
            last_improve = it

        # If no improvement for a while, do an SA restart
        stalled = it - last_improve > 500

        if stalled and sa_prob > 0.001:
            # Simulated annealing: accept a worse solution with probability
            noise_scale = sa_noise * S / np.sqrt(n)
            y_perturbed = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            c_pert = fft_convolve_self(y_perturbed)
            # Accept with probability exp(-delta/beta_sa)
            delta = c_pert.max() - best
            beta_sa = max(10.0, beta * 0.01)
            if delta < 0 or rng.random() < np.exp(-delta / beta_sa):
                y = y_perturbed.copy()
                t = 1.0
                last_improve = it
                # Decay noise
                sa_noise *= 0.95

        # Compute gradient
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(L) where L ≈ 2*beta (Lipschitz constant of LSE grad)
        step = 1.0 / (2.0 * beta + 1.0)
        # Reduce step if stalled
        if stalled:
            step *= 0.7

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum constraint (projection may drift slightly)
        s = y.sum()
        if s > 1e-15:
            y = y * (S / s)
        a = x_new
        t = t_new

        # Periodic small noise to escape very shallow local minima
        if it % 5000 == 0 and not stalled:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_reshape(a_init, n, S, max_time, rng, alpha_start=0.3, alpha_end=0.9):
    """Gerchberg-Saxton spectral reshaping with adaptive alpha."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    a_prev = a.copy()

    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Adaptive alpha
        progress = min(1.0, it / max(1, int(max_time * 10)))
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # Flatten: push P towards constant
        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target

        # Additional peak suppression: reduce the largest spectral components
        P_sorted = np.sort(P)[::-1]
        P_median = np.median(P)
        # Damp peaks above 3x median
        threshold = 3.0 * P_median
        P_damp = np.where(P > threshold, P * (threshold / P) ** 0.5, P)
        P_new = 0.5 * P_new + 0.5 * P_damp / P_damp.mean()

        mag_new = np.sqrt(P_new)
        # Preserve total power (approximately preserves sum)
        total_power = P.sum()
        P_new_sum = P_new.sum()
        if P_new_sum > 1e-30:
            mag_new = mag_new * np.sqrt(total_power / P_new_sum)
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project to non-negative with fixed sum
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        # Momentum blend
        momentum = 0.7 + 0.2 * progress
        a_new = momentum * a_new + (1 - momentum) * a_prev
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a_prev = a
        a = a_new

    return best_a, best


def coordinate_descent_peak(a_init, n, S, max_time, rng):
    """Coordinate descent focused on reducing the peak convolution value."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Identify the peak lag(s)
        peak_val = c.max()
        peak_lags = np.where(c >= peak_val - 1e-10 * peak_val)[0]

        n_try = min(100 + it * 5, 500)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose a pair: bias towards elements that contribute to peak lags
            if rng.random() < 0.4:
                # Target a specific peak lag
                k = peak_lags[rng.integers(0, len(peak_lags))]
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if hi <= lo:
                    continue
                j = lo + rng.integers(0, hi - lo + 1)
                m = k - j
                if m < 0 or m >= n or j == m:
                    continue
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
                if j == m:
                    continue

            if a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute the derivative of each c_k w.r.t. transferring delta from m to j
            # c_k(delta) = c_k + delta * (a[j+k] + a[j-k if valid] - a[m+k] - a[m-k if valid])
            # For efficiency, compute the change vector d
            d = np.zeros(2 * n - 1)
            # Contribution of +delta at position j: adds delta * a[j+k] for each k
            for k in range(2 * n - 1):
                jk = j + k
                jmk = j - k
                val = 0.0
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                val_m = 0.0
                if 0 <= mk < n:
                    val_m += a[mk]
                if 0 <= mmk < n:
                    val_m += a[mmk]
                d[k] = val - val_m

            # c_0 has a quadratic correction: (a[j]+d)^2 - a[j]^2 + (a[m]-d)^2 - a[m]^2
            # = 2(a[j]-a[m])*d + 2d^2
            # So for k=0, the linear part is already in d[0], plus quadratic 2d^2
            # Actually d[0] = a[j] + a[j] - a[m] - a[m] = 2(a[j]-a[m])
            # And the quadratic correction is 2*delta^2

            lo_d = -a[j]
            hi_d = a[m]

            # Binary/golden search for best delta
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            if b_g - a_g < 1e-14:
                continue

            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            def f_delta(dlt):
                vals = c + dlt * d
                vals[0] += 2.0 * dlt * dlt  # quadratic for k=0
                return vals.max()

            fc = f_delta(c_g)
            fd = f_delta(d_g)

            for _ in range(50):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = max(lo_d, min(hi_d, delta))

            if abs(delta) < 1e-14:
                continue

            # Apply
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < -1e-12 or a_trial[m] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)

            # Compute new c
            c_trial = c + delta * d
            c_trial[0] += 2.0 * a[j] * delta + 2.0 * delta * delta
            # More precise: c_trial[0] = c[0] + delta*(2*a[j]-2*a[m]) + 2*delta^2
            c_trial[0] = c[0] + delta * d[0] + 2.0 * delta * delta

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-13:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Periodic full recompute
        if it % 30 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        if not improved and it > 10:
            # Add noise to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def generate_step_init(n, rng, S):
    """Generate an anti-palindromic step-function initialization."""
    # "Cliff" in the middle: first half larger, second half smaller
    # With smooth boundary tapers
    t = np.linspace(0, 1, n)
    # Main body: sigmoid cliff
    cliff_width = rng.uniform(0.1, 0.3)
    cliff_center = 0.5 + rng.uniform(-0.1, 0.1)
    body = 1.0 / (1.0 + np.exp((t - cliff_center) / cliff_width))
    # Boundary taper: reduce at edges
    taper_width = rng.uniform(0.02, 0.1)
    left_taper = np.minimum(1.0, t / taper_width)
    right_taper = np.minimum(1.0, (1.0 - t) / taper_width)
    a = body * left_taper * right_taper
    a = np.maximum(a, 0.0)
    return normalize_sum(a, n)


def generate_spectral_init(n, rng, S):
    """Generate initialization from a flat-ish spectrum with random phases."""
    fsize = 1 << (2 * n - 1).bit_length()
    # Random phases with flat magnitude
    phases = rng.uniform(0, 2 * np.pi, fsize)
    F = np.exp(1j * phases)
    # Ensure Hermitian symmetry for real output
    F = np.fft.fftshift(F)
    # Make it properly Hermitian
    F_herm = np.zeros(fsize, dtype=np.complex128)
    for k in range(fsize // 2 + 1):
        F_herm[k] = F[k]
        F_herm[fsize - k] = np.conj(F[k])
    F = np.fft.ifftshift(F_herm)
    a = np.real(ifft(F))[:n]
    a = np.maximum(a, 0.0)
    return normalize_sum(a, n)


def upsample_interpolate(a, n_old, n_new):
    """Interpolate a sequence from length n_old to n_new."""
    t_old = np.linspace(0, 1, n_old, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    return a_new


def propose_candidate(seed=42, budget_s=2100):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 512 and np.sum(np.clip(parent, 0, 1000)) > 0.1:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine target length
    if parent is not None and 2048 <= len(parent) <= 65536:
        n_target = len(parent)
    else:
        n_target = 8192

    S_target = np.sqrt(2.0 * n_target)

    print("Starting: n_target=%d, budget=%.0fs, parent=%s" % (
        n_target, budget_s, "yes" if parent is not None else "no"))

    best_score = np.inf
    best_a = None

    def update_best(a_cand, label):
        nonlocal best_score, best_a
        a_n = len(a_cand)
        S_n = np.sqrt(2.0 * a_n)
        a_norm = normalize_sum(a_cand, a_n)
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()  # Since S = sqrt(2n), obj = max(c)
        if score < best_score:
            best_score = score
            best_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: score=%.6f (%.0fs elapsed)" % (label, a_n, score, elapsed))
        return score

    # === Phase 0: Multi-scale initialization ===
    print("\n=== Phase 0: Multi-scale Initialization ===")
    p0_budget = (deadline - time.monotonic()) * 0.15
    p0_start = time.monotonic()

    # Start at a smaller scale
    n_coarse = min(1024, n_target)
    S_coarse = np.sqrt(2.0 * n_coarse)

    # Generate diverse coarse initializations
    coarse_starts = []
    if parent is not None:
        # Downsample parent to coarse size
        if n_target > n_coarse:
            t_old = np.linspace(0, 1, n_target, endpoint=False)
            t_new = np.linspace(0, 1, n_coarse, endpoint=False)
            parent_coarse = np.interp(t_new, t_old, parent)
        else:
            parent_coarse = parent[:n_coarse]
        coarse_starts.append(("parent", normalize_sum(parent_coarse, n_coarse)))

    coarse_starts.append(("step1", generate_step_init(n_coarse, rng, S_coarse)))
    coarse_starts.append(("step2", generate_step_init(n_coarse, rng, S_coarse)))
    coarse_starts.append(("spectral", generate_spectral_init(n_coarse, rng, S_coarse)))
    coarse_starts.append(("gauss", normalize_sum(np.exp(-4.0 * (np.linspace(0, 1, n_coarse) - 0.5) ** 2), n_coarse)))
    coarse_starts.append(("random", normalize_sum(rng.random(n_coarse) + 0.01, n_coarse)))

    # Quick FISTA on each coarse start
    coarse_best_a = None
    coarse_best_score = np.inf
    per_start_time = max(1.0, p0_budget / len(coarse_starts) * 0.8)

    for name, arr in coarse_starts:
        if time.monotonic() > t_start + p0_budget * 0.7:
            break
        a_opt, score = run_fista_sa(arr, n_coarse, S_coarse, per_start_time, rng,
                                     beta_init=10.0, beta_max=500.0,
                                     beta_ramp_iters=500, sa_prob=0.1, sa_noise=0.05)
        if score < coarse_best_score:
            coarse_best_score = score
            coarse_best_a = a_opt
            print("  Coarse [%s]: score=%.4f" % (name, score))

    # Upsample to target size
    if coarse_best_a is not None:
        fine_start = upsample_interpolate(coarse_best_a, n_coarse, n_target)
        fine_start = normalize_sum(fine_start, n_target)
        update_best(fine_start, "coarse_upsampled")
    else:
        fine_start = normalize_sum(parent, n_target) if parent is not None else \
            normalize_sum(rng.random(n_target) + 0.01, n_target)

    # If parent is available and longer, also try it directly
    if parent is not None:
        parent_norm = normalize_sum(parent, n_target)
        update_best(parent_norm, "parent_direct")
        # Use the better of parent and upsampled
        c_parent = fft_convolve_self(parent_norm).max()
        c_up = fft_convolve_self(fine_start).max() if fine_start is not best_a else best_score
        if c_parent < c_up:
            fine_start = parent_norm

    if best_a is None:
        best_a = fine_start
        best_score = fft_convolve_self(best_a).max()

    print("After Phase 0: best=%.6f" % best_score)

    # === Phase 1: Main FISTA optimization ===
    print("\n=== Phase 1: FISTA Optimization ===")
    p1_budget = (deadline - time.monotonic()) * 0.35
    if p1_budget > 20:
        a_fista, _ = run_fista_sa(best_a, n_target, S_target, p1_budget, rng,
                                   beta_init=50.0, beta_max=5000.0,
                                   beta_ramp_iters=5000, sa_prob=0.08, sa_noise=0.03)
        update_best(a_fista, "fista_main")

    print("After Phase 1: best=%.6f" % best_score)

    # === Phase 2: Spectral reshaping ===
    print("\n=== Phase 2: Spectral Reshaping ===")
    p2_budget = (deadline - time.monotonic()) * 0.20
    if p2_budget > 15 and best_a is not None:
        a_spec, _ = spectral_reshape(best_a, n_target, S_target, p2_budget, rng,
                                      alpha_start=0.4, alpha_end=0.85)
        update_best(a_spec, "spectral")

    print("After Phase 2: best=%.6f" % best_score)

    # === Phase 3: Second FISTA pass (from spectral result) ===
    print("\n=== Phase 3: FISTA Polish ===")
    p3_budget = (deadline - time.monotonic()) * 0.20
    if p3_budget > 15 and best_a is not None:
        a_fista2, _ = run_fista_sa(best_a, n_target, S_target, p3_budget, rng,
                                    beta_init=200.0, beta_max=10000.0,
                                    beta_ramp_iters=2000, sa_prob=0.03, sa_noise=0.01)
        update_best(a_fista2, "fista_polish")

    print("After Phase 3: best=%.6f" % best_score)

    # === Phase 4: Coordinate descent ===
    print("\n=== Phase 4: Coordinate Descent ===")
    p4_budget = (deadline - time.monotonic()) * 0.18
    if p4_budget > 10 and best_a is not None:
        a_cd, _ = coordinate_descent_peak(best_a, n_target, S_target, p4_budget, rng)
        update_best(a_cd, "coord_descent")

    print("After Phase 4: best=%.6f" % best_score)

    # === Phase 5: Final quick passes if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        print("\n=== Phase 5: Final Refinement ===")
        # Alternate short FISTA and spectral
        for phase in range(3):
            if time.monotonic() > deadline - 10:
                break
            t_sub = min(remaining * 0.25, 60.0)
            if phase % 2 == 0:
                a_sub, _ = run_fista_sa(best_a, n_target, S_target, t_sub, rng,
                                         beta_init=500.0, beta_max=15000.0,
                                         beta_ramp_iters=1000, sa_prob=0.02, sa_noise=0.005)
                update_best(a_sub, "final_fista_%d" % phase)
            else:
                a_sub, _ = coordinate_descent_peak(best_a, n_target, S_target, t_sub, rng)
                update_best(a_sub, "final_cd_%d" % phase)

    # === Final verification ===
    print("\n=== Final Verification ===")
    if best_a is None:
        best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in best_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification FAILED, using fallback")
        best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in best_a]

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(best_a), ver, time.monotonic() - t_start))
    return final_list
