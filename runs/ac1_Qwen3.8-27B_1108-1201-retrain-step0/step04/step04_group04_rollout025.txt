"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Spectral gradient descent with adaptive low-pass smoothing (novel)
2. Peak-targeted coordinate descent with exact 1D line search
3. Multi-length exploration and diverse initializations
4. Iterative reweighting to flatten the auto-convolution

Key novelty: Uses frequency-domain smoothing of the subgradient to create
descent directions that reduce the peak while avoiding creating new peaks.
Combines with a targeted coordinate descent that focuses on peak-contributing pairs.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-30:
        return np.ones(n) * S / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def spectral_gradient_step(a, S, step_size, sigma, rng):
    """
    Spectral gradient descent step:
    1. Compute subgradient at peak lags
    2. Smooth in frequency domain to avoid creating new peaks
    3. Step and project
    """
    n = len(a)
    c = fft_convolve_self(a)
    cur_max = c.max()

    # Find significant lags (those within 5% of max)
    threshold = cur_max * 0.95
    significant = np.where(c >= threshold)[0]

    if len(significant) == 0:
        significant = np.array([int(np.argmax(c))])

    # Compute weighted subgradient
    # grad[j] = sum_k w_k * 2*a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)
    weights = c[significant] / c[significant].sum()

    for idx, k in enumerate(significant):
        w = 2.0 * weights[idx]
        # a[k-j] for valid j: j ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

    # Low-pass filter in frequency domain
    fsize = 1 << (n - 1).bit_length()
    G = fft(grad, fsize)
    freqs = np.fft.fftfreq(fsize)
    # Gaussian low-pass
    filt = np.exp(-sigma ** 2 * (freqs / 0.5) ** 2)
    G_filtered = G * filt
    grad_smooth = np.real(ifft(G_filtered))[:n]

    # Ensure gradient points in descent direction
    if grad_smooth @ (a - a.mean()) < 0:
        grad_smooth = -grad_smooth

    # Step and project
    a_new = a - step_size * grad_smooth
    a_new = project_simplex(a_new, S)

    c_new = fft_convolve_self(a_new)
    return a_new, c_new.max(), cur_max


def smart_coordinate_descent(a, S, max_time, rng, n_pairs_per_iter=30):
    """
    Coordinate descent that focuses on peak-reducing moves.
    For each iteration, identifies the peak lag and tries transfers
    that specifically reduce that lag.
    """
    n = len(a)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()

    # Precompute FFT size
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        improved = False

        # Find current peak
        peak_lag = int(np.argmax(c))
        cur_max = c[peak_lag]
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Identify top contributors to the peak lag
        # c[peak_lag] = sum_i a[i] * a[peak_lag - i]
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)
        if lo > hi:
            break

        contributors = np.arange(lo, hi + 1)
        contrib_vals = a[contributors] * a[peak_lag - contributors]
        total_contrib = contrib_vals.sum()

        if total_contrib < 1e-30:
            break

        # Sort by contribution (largest first)
        sorted_idx = np.argsort(contrib_vals)[::-1]
        top_contributors = contributors[sorted_idx[:min(20, len(sorted_idx))]]

        # Generate candidate moves
        moves = []

        # Type 1: Transfer from a large contributor to a small-position
        for j in top_contributors:
            # Partner in the peak: partner = peak_lag - j
            partner = peak_lag - j
            if partner < 0 or partner >= n or partner == j:
                continue
            # Target: transfer to a position that doesn't contribute to peak
            # Good targets: positions near 0 or n-1 that are currently small
            # Or positions far from the peak interaction
            for _ in range(3):
                # Random target
                m = rng.integers(0, n)
                if m != j and m != partner:
                    moves.append((j, m))

        # Type 2: Symmetric transfers (reduce both a[j] and a[partner])
        for j in top_contributors[:10]:
            partner = peak_lag - j
            if partner < 0 or partner >= n or partner == j:
                continue
            # Transfer from j to a "safe" position
            m = rng.integers(0, n)
            if m != j and m != partner:
                moves.append((j, m))
            m = rng.integers(0, n)
            if m != j and m != partner:
                moves.append((partner, m))

        # Type 3: Random moves
        for _ in range(n_pairs_per_iter // 3):
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if j != m:
                moves.append((j, m))

        # Evaluate moves
        for j, m in moves:
            if time.monotonic() - start > max_time - 1:
                break
            if j == m or a[j] < 1e-12:
                continue

            # Compute direction of change in c for transfer from j to m
            # delta > 0 means move mass from j to m
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                # Contribution from position j (being reduced)
                jk = k - j
                jmk = j + k
                if 0 <= jk < n:
                    val -= a[jk]
                if 0 <= jmk < n and jmk != jk:
                    val -= a[jmk]
                # Contribution from position m (being increased)
                mk = k - m
                mmk = m + k
                if 0 <= mk < n:
                    val += a[mk]
                if 0 <= mmk < n and mmk != mk:
                    val += a[mmk]
                d[k] = val

            # Quadratic correction for k where j or m is self-interacting
            # c[0] change: (a[m]+d)^2 - a[m]^2 - (a[j]-d)^2 + a[j]^2 = 2*a[m]*d + d^2 - 2*a[j]*d + d^2
            # Actually for k=0: only a[0]^2 contributes if j or m is 0
            # Let's handle this with a quadratic term

            # Find optimal delta using bisection on the maximum
            lo_d = 0.0  # move from j to m (j decreases, m increases)
            hi_d = a[j] * 0.5  # don't move more than half

            if hi_d < 1e-15:
                continue

            # Also try moving from m to j
            # f(delta) = max(c + delta * d) for delta in [lo_d, hi_d]

            # Quick check: does this direction help?
            peak_change = d[peak_lag]
            if peak_change >= -1e-15:
                # This move doesn't reduce the peak; try the other direction
                lo_d = -a[m] * 0.5
                hi_d = 0.0
                if hi_d - lo_d < 1e-15:
                    continue
                if d[peak_lag] >= -1e-15:
                    continue

            # Bisection for optimal delta (convex function)
            # Actually, max of linear functions is convex, so we can use ternary search
            lo_b = lo_d
            hi_b = hi_d

            # Evaluate at a few points first to get a sense
            def eval_delta(dlt):
                new_c_max = (c + dlt * d).max()
                # Quadratic correction for k=0 (self-convolution)
                if 0 <= 0 < 2 * n - 1:
                    # c[0] = a[0]^2, change if j==0 or m==0
                    pass
                return new_c_max

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_b
            b_g = hi_b
            if b_g - a_g < 1e-15:
                continue
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = eval_delta(c_g)
            fd = eval_delta(d_g)

            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = eval_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = eval_delta(d_g)

            delta = (a_g + b_g) / 2.0

            # Apply the move
            a_trial = a.copy()
            a_trial[j] -= delta
            a_trial[m] += delta

            if a_trial[j] < -1e-12 or a_trial[m] < -1e-12:
                continue

            a_trial = np.maximum(a_trial, 0.0)

            # Compute actual improvement
            c_trial = c + delta * d
            # Fix up c[0] and c[2n-2] for quadratic terms
            # c[0] = sum a[i]^2 ... actually c[0] = a[0]*a[0] for linear conv
            # Wait, np.convolve(a,a)[0] = a[0]*a[0]
            # Let me just recompute c[0] and c[-1]
            c_trial[0] = a_trial[0] ** 2
            c_trial[-1] = a_trial[-1] ** 2

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-12:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Periodic full recompute
        if improved and time.monotonic() - start < max_time - 5:
            # Recompute c periodically to avoid drift
            if rng.random() < 0.1:
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()

        if not improved:
            # Add noise to escape
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best

        it += 1

        sigma = min(sigma_max, sigma * (1.0 + 0.001))

        if new_max < best - 1e-14:

            no_improve_count = 0
        else:
            no_improve_count += 1
            # If no improvement, try a smaller step
            if no_improve_count % 20 == 0:

                # Occasionally add noise
                if no_improve_count % 100 == 0:
                    noise_scale = 0.002 * S / np.sqrt(n)

                best_a = a_cd.copy()
                a = a_cd.copy()

def weighted_lse_descent(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=5000.0):
    """
    Log-sum-exp smoothed descent with adaptive beta.
    More stable than pure spectral method.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    t_fista = 1.0
    y = a.copy()
    prev_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta
        beta = min(beta_max, beta_init * (1.05 ** it))

        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        # LSE gradient
        c_shift = c_y - cur_max
        e = np.exp(beta * c_shift)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        # grad = 2 * ifft(fft(w) * conj(fft(a)))[:n]
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        # Cap step size
        step = min(step, 0.1 / max(gnorm / S, 1e-10))

        x_new = project_simplex(y - step * grad, S)

        # Fista momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - prev_a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 0:
            y = y * (S / s)
        prev_a = x_new
        t_fista = t_new

        # Periodic noise
        if it % 2000 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_fista = 1.0

    return best_a, best


def tapered_init(n, rng, style=0):
    """Generate tapered initializations (small at edges, large in middle)."""
    t = np.linspace(0, 1, n)

    if style == 0:
        # Gaussian taper
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif style == 1:
        # Cosine taper
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 0.01)
    elif style == 2:
        # Power taper
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 0.01)
    elif style == 3:
        # Asymmetric taper (skewed)
        a = np.sin(np.pi * t ** 0.7) ** 2
    elif style == 4:
        # Double bump
        a = np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2)
    elif style == 5:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.random()
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(0.5, 2.0)
            a += amp * np.exp(-((t - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    else:
        a = np.exp(-3.0 * (t - 0.5) ** 2)

    return a


def block_coordinate_optimize(a_init, n, S, max_time, rng, n_blocks=16):
    """
    Block-coordinate optimization: divide sequence into blocks,
    optimize each block's profile (using parametric shapes) while
    keeping other blocks fixed.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    block_size = n // n_blocks
    n_actual_blocks = n // block_size
    # Use only the portion that divides evenly
    a_work = a[:n_actual_blocks * block_size]

    # Parametric shapes for blocks
    t_block = np.linspace(0, 1, block_size)
    shapes = [
        np.ones(block_size),
        t_block,
        1.0 - t_block,
        np.sin(np.pi * t_block),
        np.exp(-2.0 * (t_block - 0.5) ** 2),
        np.maximum(1.0 - 2.0 * np.abs(t_block - 0.5), 0.1),
    ]

    while time.monotonic() - start < max_time:
        improved = False

        for b in range(n_actual_blocks):
            if time.monotonic() - start > max_time - 5:
                break

            offset = b * block_size
            current_block = a_work[offset:offset + block_size].copy()
            block_sum = current_block.sum()

            if block_sum < 1e-10:
                continue

            # Try each shape
            for shape in shapes:
                # Scale shape to match block sum
                shape_scaled = shape * (block_sum / shape.sum())

                a_trial = a_work.copy()
                a_trial[offset:offset + block_size] = shape_scaled

                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best - 1e-12:
                    a_work = a_trial.copy()
                    best = c_trial.max()
                    best_a = np.concatenate([a_work, a[n_actual_blocks * block_size:]])
                    improved = True

        if not improved:
            # Perturb
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex([a[:n_actual_blocks * block_size],])
            a_work = np.maximum(a_work + rng.normal(0, noise_scale, n_actual_blocks * block_size), 0)
            a_work = a_work * (block_sum / a_work.sum()) if a_work.sum() > 0 else a_work

    return best_a, best

        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * n)
        print("\n=== Optimizing n=%d (sum=%.4f) ===" % (n, S))

        # Build initializations
        init_list = []
        if parent is not None and len(parent) == n:
            init_list.append(("parent", parent))
        elif parent is not None:
            # Interpolate parent to this length
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            interp = np.interp(t_new, t_old, parent)
            init_list.append(("parent_interp", interp))

        for s in range(6):
            init_list.append(("taper_%d" % s, tapered_init(n, rng, s)))

        init_list.append(("random", rng.random(n) + 0.1))

        # Phase 1: Try spectral gradient on top initializations
        p1_time = min(300.0, (deadline - time.monotonic()) * 0.35)
        n_inits = len(init_list)
        time_per_init = p1_time / n_inits

        for name, init_arr in init_list:
            if time.monotonic() > t_start + budget_s * 0.4:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                continue

            a_opt, _ = spectral_gradient_optimization(
                init_arr, n, S, t_alloc, rng,
                sigma_init=0.03, sigma_max=0.2, step_init=0.05)
            update_best(a_opt, "specgrad_" + name)

        # Phase 2: Weighted LSE on best
        if global_a is not None and len(global_a) == n:
            p2_time = min(200.0, (deadline - time.monotonic()) * 0.25)
            if p2_time > 20:
                a_opt, _ = weighted_lse_descent(
                    global_a, n, S, p2_time, rng,
                    beta_init=50.0, beta_max=3000.0)
                update_best(a_opt, "lse")

        # Phase 3: Coordinate descent
        if global_a is not None and len(global_a) == n:
            p3_time = min(300.0, (deadline - time.monotonic()) * 0.30)
            if p3_time > 20:
                a_opt, _ = smart_coordinate_descent(
                    global_a, S, p3_time, rng, n_pairs_per_iter=40)
                update_best(a_opt, "coord")

        # Phase 4: Final LSE polish
        if global_a is not None and len(global_a) == n:
            p4_time = (deadline - time.monotonic()) * 0.5
            if p4_time > 15:
                a_opt, _ = weighted_lse_descent(
                    global_a, n, S, p4_time * 0.6, rng,
                    beta_init=200.0, beta_max=10000.0)
                update_best(a_opt, "lse_polish")

                # Another CD pass
                remaining = deadline - time.monotonic() - 15
                if remaining > 10:
                    a_opt, _ = smart_coordinate_descent(
                        global_a, S, remaining, rng, n_pairs_per_iter=50)
                    update_best(a_opt, "cd_final")

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        n_fb = 1024
        global_a = tapered_init(n_fb, rng, 0)
        global_a = normalize_sum(global_a, n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 1024
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        n_fb = 1024
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Returning length=%d, eval_score=%.6f, internal_best=%.6f, time=%.0fs" % (
        len(global_a), ver, global_best, time.monotonic() - t_start))
    return final_list
