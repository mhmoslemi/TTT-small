   So the formula is confirmed: ∂b_k/∂a_j = 2 * a_{k-j} if 0 <= k-j < n, else 0.
   
   Now, for transferring δ from m to j:
   Δb_k = δ * (∂b_k/∂a_j - ∂b_k/∂a_m)
   = δ * (2*a_{k-j} [0<=k-j<n] - 2*a_{k-m} [0<=k-m<n])
   
   This is very simple to compute! For a given pair (j, m) and lag k, the change is:
   2δ * (a_{(k-j) mod n} [valid] - a_{(k-m) mod n} [valid])
   
   where [valid] means the index is in [0, n-1].

   Actually, I realize there's a subtlety: this is only the linear (first-order) change. The actual change in b_k when we modify a_j and a_m is:
   
   Δb_k = (new_b_k) - (old_b_k)
   
   For the term involving both a_j and a_m (if k-j = m or k-m = j), there's a cross term. But for the first-order approximation, the formula above is correct.

   For an exact computation, we'd need to account for:
   - The change in a_j^2 terms (if k = 2j)
   - The change in a_m^2 terms (if k = 2m)
   - The cross term a_j * a_m (if k-j = m, i.e., k = j+m)

   But for a first-order optimization, the linear approximation is sufficient, and we can verify with exact computation after.

   OK so now I have an efficient way to do coordinate descent:
   1. Find the top M lags (e.g., M=10)
   2. For a candidate pair (j, m), compute Δb_k for k in top M lags using the formula above
   3. The worst-case change is max_k Δb_k (if positive, it might increase the peak)
   4. Optimize δ to minimize the worst-case

   This is O(M) per pair evaluation, which is much faster than O(n).

   Let me now write the full algorithm:

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization with efficient peak-directed coordinate descent,
adaptive FISTA, and spectral flattening. Key innovation: O(1) per-pair evaluation
in coordinate descent by precomputing sensitivity at top lags only.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_sensitivities(a, peak_lags):
    """
    For each peak lag k, precompute sensitivity of b_k w.r.t. each a_j.
    Returns array of shape (len(peak_lags), n) where entry [l, j] = ∂b_{k_l}/∂a_j.
    """
    n = len(a)
    M = len(peak_lags)
    sens = np.zeros((M, n))
    for l, k in enumerate(peak_lags):
        # ∂b_k/∂a_j = 2*a_{k-j} if 0 <= k-j < n, else 0
        # So for j in range(max(0, k-n+1), min(k+1, n)):
        #   sens[l, j] = 2 * a[k-j]
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            js = np.arange(lo, hi)
            sens[l, js] = 2.0 * a[k - js]
    return sens


def lse_objective_and_grad(a, beta):
    """Compute LSE(max(conv)) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    # LSE = cm + (1/beta) * log(sum(exp(beta*(c-cm))))
    # For very large beta, just use cm + (1/beta)*log(M_active)
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    obj = cm + np.log(tot) / beta
    # Gradient
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad


def run_fista_v2(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=3000.0
                 , beta_ramp=3000, use_restarts=True, restart_interval=5000):
    """
    Improved FISTA with:
    - Backtracking line search
    - Periodic restarts to escape local minima
    - Hybrid gradient (LSE + exact subgradient)
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    no_improve = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp) ** 2
        else:
            beta = beta_max

        # Compute gradient
        c = fft_convolve_self(y)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Choose gradient: blend LSE and exact
        if beta < 100:
            # Use LSE gradient
            e = np.exp(beta * (c - cm))
            tot = e.sum()
            w = e / tot
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        else:
            # Use top-K lags for sharper gradient
            k_order = np.argsort(c)[::-1][:20]
            k_max = c.max()
            e = np.exp(beta * (c - k_max))
            tot = e.sum()
            if tot < 1e-300:
                # Essentially exact subgradient
                k_star = int(np.argmax(c))
                grad = np.zeros(n)
                lo = max(0, k_star - n + 1)
                hi = min(k_star, n - 1)
                js = np.arange(lo, hi + 1)
                grad[js] = 2.0 * a[k_star - js]
            else:
                w = e / tot
                # Only use top lags for efficiency
                w_trunc = np.zeros_like(w)
                top_k = np.argsort(c)[::-1][:50]
                w_trunc[top_k] = w[top_k]
                w_trunc = w_trunc / w_trunc.sum()
                fsize = 1 << (len(w_trunc) + n - 1).bit_length()
                W = fft(w_trunc, fsize)
                A = fft(a, fsize)
                grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(L) where L is Lipschitz constant (approximate)
        # For the quadratic part, L ≈ 2*max(a) * 2 * n (rough)
        # Use backtracking
        step = 1.0 / (beta + 1.0)
        for _ in range(5):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cm + 0.5 * step * np.dot(grad, x_new - y):
                break
            step *= 0.5

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y_new, 0.0)
        a = x_new
        t = t_new

        # Periodic restart
        if use_restarts and it % restart_interval == 0 and no_improve > restart_interval // 2:
            # Perturb and reset momentum
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-no_improve / (2 * restart_interval))
            if noise_scale > 1e-8:
                y = project_simplex(a + rng.normal(0, noise_scale, n), S)
                t = 1.0
                no_improve = 0

    return best_a, best


def efficient_coord_descent(a_init, n, S, max_time, rng, n_top_lags=10, n_pairs_per_iter=100):
    """
    Efficient coordinate descent using precomputed sensitivities at top lags.
    Key: O(1) evaluation per (j,m) pair using precomputed sensitivity matrix.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    # Precompute sensitivities
    peak_lags = np.argsort(c)[::-1][:n_top_lags]
    sens = compute_sensitivities(a, peak_lags)  # shape (M, n)

    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Update peak lags periodically
        if it % 5 == 0:
            peak_lags = np.argsort(c)[::-1][:n_top_lags]
            sens = compute_sensitivities(a, peak_lags)

        # Try pairs
        for _ in range(n_pairs_per_iter):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose pair: mix of random and structured
            r = rng.random()
            if r < 0.3:
                # Random pair
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            elif r < 0.6:
                # Local pair (close together)
                j = rng.integers(0, n)
                m = (j + rng.integers(1, max(2, n // 100))) % n
            else:
                # Pair targeting peak reduction: j where sens is low, m where sens is high
                # At the peak lag, we want to decrease b_k
                # Δb_k = δ * sens[k,j] - δ * sens[k,m]
                # To decrease: pick j with small sens and m with large sens
                k_idx = rng.integers(0, n_top_lags)
                row = sens[k_idx]
                # j: low sensitivity (receiving), m: high sensitivity (donating)
                j_candidates = np.argsort(row)[:max(1, n // 50)]
                m_candidates = np.argsort(row)[::-1][:max(1, n // 50)]
                j = j_candidates[rng.integers(0, len(j_candidates))]
                m = m_candidates[rng.integers(0, len(m_candidates))]

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute change in each top lag
            # Δb_k = δ * (sens[k,j] - sens[k,m]) for each k
            ds = sens[:, j] - sens[:, m]  # shape (M,)

            # We want to find δ that minimizes max_k (c[k_peak] + δ * ds[k])
            # subject to -a[j] <= δ <= a[m]
            # (approximately, ignoring second-order effects)

            base = c[peak_lags]  # current values at peak lags
            lo = -a[j]
            hi = a[m]
            
            if lo >= hi:
                continue

            # The function f(δ) = max_k (base[k] + δ * ds[k]) is piecewise linear and convex
            # Minimum is where the lines cross
            # Use a simple binary/ternary search
            # Actually, for piecewise linear convex, the minimum is at a breakpoint
            
            # Simple approach: try δ = 0, δ = lo/2, δ = hi/2, and the breakpoint
            # The breakpoint occurs where base[i] + δ*ds[i] = base[j] + δ*ds[j]
            # δ = (base[j] - base[i]) / (ds[i] - ds[j])
            
            # For efficiency, just use golden section on the convex piecewise linear
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = lo, hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = np.max(base + c_g * ds)
            fd = np.max(base + d_g * ds)

            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = np.max(base + c_g * ds)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = np.max(base + d_g * ds)

            delta_opt = (a_g + b_g) / 2.0
            delta_opt = max(lo, min(hi, delta_opt))

            if abs(delta_opt) < 1e-14:
                continue

            # Apply and verify
            a[j] += delta_opt
            a[m] -= delta_opt

            # Ensure non-negativity
            if a[j] < 0:
                a[j] = 0
                a[m] -= a[j]  # adjust
            if a[m] < 0:
                a[m] = 0
                a[j] -= a[m]
                continue

            # Update c approximately
            for l in range(n_top_lags):
                c[peak_lags[l]] += delta_opt * ds[l]
            # Also update c[0] for the quadratic term (if peak_lags includes 0)
            # c[0] = sum a_i^2, change = (a_j+δ)^2 - a_j^2 + (a_m-δ)^2 - a_m^2 = 2δ(a_j-a_m) + 2δ^2
            # But this is already accounted for in the sensitivity formula if k=0 is in peak_lags

            cur_max = np.max(c[peak_lags])
            if cur_max < best - 1e-10:
                best = cur_max
                improved = True
            elif cur_max < best + 1e-8:
                # Accept if approximately same (to allow exploration)
                if rng.random() < 0.3:
                    pass  # keep the change
            else:
                # Revert
                a[j] -= delta_opt
                a[m] += delta_opt
                for l in range(n_top_lags):
                    c[peak_lags[l]] -= delta_opt * ds[l]

        # Periodic full recompute
        if it % 3 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()
            else:
                stagnation += 1

            # Re-normalize to prevent drift
            s = a.sum()
            if abs(s - S) > 0.01 * S:
                a = a * (S / s)
                c = fft_convolve_self(a)

            # If stuck, add noise
            if stagnation > 20:
                noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnation / 30.0)
                if noise_scale > 1e-8:
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    c = fft_convolve_self(a)
                    best = min(best, c.max())
                    best_a = a.copy() if c.max() < best else best_a
                    stagnation = 0

    # Final
    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()

    return best_a, best


def spectral_nudge(a_init, n, S, max_time, rng, n_iter=200):
    """
    Spectral flattening with better non-negativity handling.
    Uses soft projection and adaptive step size.
    """
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    # DC component
    A0 = a.sum()  # = S

    for it in range(n_iter):
        if time.monotonic() - start > max_time - 2:
            break

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)

        # Power spectrum
        P = np.abs(F) ** 2

        # Target: flat spectrum except DC
        # The "excess" power at DC is A0^2, the rest should be (total_power - A0^2) / (fsize-1)
        total_power = P.sum()  # = fsize * sum(a^2) / fsize = sum(a^2) * fsize... 
        # Actually P.sum() = fsize * sum(a^2) by Parseval
        # We want P[0] = A0^2 (fixed) and P[k] = (total - A0^2)/(fsize-1) for k>0

        # Adaptive flattening strength
        alpha = min(0.9, 0.1 + 0.8 * it / n_iter)

        # Create target spectrum
        P_target = np.ones(fsize) * (P.sum() - P[0]) / max(fsize - 1, 1)
        P_target[0] = P[0]  # Keep DC

        # Blend
        P_new = (1 - alpha) * P + alpha * P_target

        # Preserve total power
        P_new = P_new * (P.sum() / P_new.sum())

        # Inverse FFT with new magnitude, same phase
        phase = F / np.maximum(np.abs(F), 1e-30)
        mag_new = np.sqrt(np.maximum(P_new, 0))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        # Project to non-negative with sum S
        a_raw = np.maximum(a_raw, 0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        cur = c_new.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't accept, but try a smaller step
            alpha2 = alpha * 0.3
            P_new2 = (1 - alpha2) * P + alpha2 * P_target
            P_new2 = P_new2 * (P.sum() / P_new2.sum())
            mag_new2 = np.sqrt(np.maximum(P_new2, 0))
            F_new2 = phase * mag_new2
            a_raw2 = np.real(ifft(F_new2))[:n]
            a_raw2 = np.maximum(a_raw2, 0)
            s2 = a_raw2.sum()
            if s2 > 1e-15:
                a_new2 = a_raw2 * (S / s2)
                c_new2 = fft_convolve_self(a_new2)
                if c_new2.max() < best:
                    best = c_new2.max()
                    best_a = a_new2.copy()
                    a = a_new2

    return best_a, best


def simulated_annealing_coord(a_init, n, S, max_time, rng):
    """
    Simulated annealing with coordinate moves.
    Accepts worse solutions with probability exp(-Δ/T) to escape local minima.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    cur = c.max()
    best = cur
    best_a = a.copy()

    T = 0.01 * S  # Initial temperature
    T_min = 1e-6 * S
    cool_rate = 0.999

    peak_lags = np.argsort(c)[::-1][:10]
    sens = compute_sensitivities(a, peak_lags)

    while time.monotonic() - start < max_time:
        # Update temperature
        T = max(T_min, T * cool_rate)

        if T <= T_min * 1.01:
            break

        # Choose move
        j = rng.integers(0, n)
        m = rng.integers(0, n)
        if j == m or a[j] < 1e-12 or a[m] < 1e-12:
            continue

        # Small random delta
        max_delta = 0.01 * S / np.sqrt(n)
        delta = rng.uniform(-max_delta, max_delta)
        delta = max(-a[j], min(a[m], delta))
        if abs(delta) < 1e-15:
            continue

        # Update
        a[j] += delta
        a[m] -= delta

        # Approximate objective change at top lags
        ds = sens[:, j] - sens[:, m]
        base = c[peak_lags]
        new_peak_approx = np.max(base + delta * ds)
        old_peak = np.max(base)
        delta_obj = new_peak_approx - old_peak

        # Accept or reject
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-15)):
            # Accept
            for l in range(len(peak_lags)):
                c[peak_lags[l]] += delta * ds[l]
            cur = c.max()  # approximate
            if cur < best:
                best = cur
                best_a = a.copy()
                # Update sensitivities
                peak_lags = np.argsort(c)[::-1][:10]
                sens = compute_sensitivities(a, peak_lags)
        else:
            # Reject
            a[j] -= delta
            a[m] += delta

        # Periodic full recompute
        if int((time.monotonic() - start) * 10) % 50 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()
            peak_lags = np.argsort(c)[::-1][:10]
            sens = compute_sensitivities(a, peak_lags)
            # Renormalize
            s = a.sum()
            if abs(s - S) > 0.01 * S:
                a = a * (S / s)

    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()

    return best_a, best


def multi_start_search(n, S, max_time, rng, parent=None):
    """
    Try multiple diverse starting points and optimize each.
    """
    tt = np.linspace(0, 1, n, endpoint=False)

    candidates = []
    names = []

    if parent is not None:
        candidates.append(parent.copy())
        names.append("parent")
        # Perturbed parent
        for scale in [0.02, 0.05, 0.1, 0.2]:
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            candidates.append(project_simplex(parent + noise, S))
            names.append("parent_n%d" % int(scale * 100))

    # Structured candidates
    candidates.append(np.ones(n) * S / n)
    names.append("uniform")

    candidates.append(np.maximum(np.sinc(tt - 0.5) ** 2, 0.01))
    candidates.append(normalize_sum(candidates[-1], n))
    names.append("sinc")

    candidates.append(np.exp(-5.0 * (tt - 0.5) ** 2))
    candidates.append(normalize_sum(candidates[-1], n))
    names.append("gaussian")

    candidates.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    candidates.append(normalize_sum(candidates[-1], n))
    names.append("cosine")

    candidates.append(rng.random(n) + 0.1)
    candidates.append(normalize_sum(candidates[-1], n))
    names.append("random")

    # Log-normal distribution
    log_norm = rng.lognormal(0, 0.5, n)
    candidates.append(normalize_sum(log_norm, n))
    names.append("lognormal")

    # Bimodal
    bimodal = np.exp(-30 * (tt - 0.3) ** 2) + np.exp(-30 * (tt - 0.7) ** 2)
    candidates.append(normalize_sum(bimodal, n))
    names.append("bimodal")

    # Triangle
    candidates.append(normalize_sum(1 - 2 * np.abs(tt - 0.5), n))
    names.append("triangle")

    # Sawtooth
    candidates.append(normalize_sum(np.abs(2 * tt - 1), n))
    names.append("sawtooth")

    # Evaluate all
    best_score = np.inf
    best_a = None
    best_name = ""

    for name, cand in zip(names, candidates):
        cand = normalize_sum(cand, n)
        c = fft_convolve_self(cand)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            best_name = name

    return best_a, best_score, best_name, candidates, names

    4. Spectral nudging

        else:
            parent = None

        parent = None

        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes" if parent is not None else "no"))

    global_a = None

        a_norm = normalize_sum(a_cand, len(a_cand))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()

        final_list = [float(x) for x in global_a]

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
```
