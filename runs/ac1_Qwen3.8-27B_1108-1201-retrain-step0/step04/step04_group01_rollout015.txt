Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using LSE-smoothed gradient
descent with backtracking line search, adaptive beta, and spectral perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_and_grad(a, beta, fsize):
    """Compute LSE value and its gradient for the self-convolution of a."""
    n = len(a)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:2*n-1]
    cm = c.max()
    scaled = beta * (c - cm)
    # Guard against overflow
    if scaled.max() > 500:
        # This shouldn't happen if beta is reasonable
        max_scaled = scaled.max()
        scaled = scaled - max_scaled
        cm = cm + max_scaled / beta
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    
    # Gradient: 2 * ifft(W * conj(A))[:n]
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return lse_val, grad, c, cm

def backtrack_line_search(a, grad, beta, fsize, c_current, lse_current, 
                         max_iter=20, c_armijo=1e-4):
    """Backtracking line search for LSE objective."""
    n = len(a)
    S = a.sum()
    alpha = 1.0
    gnorm_sq = np.dot(grad, grad)
    if gnorm_sq < 1e-30:
        return a, alpha
    
    for _ in range(max_iter):
        a_new = project_simplex(a - alpha * grad, S)
        if np.all(a_new == 0):
            alpha *= 0.5
            continue
        lse_new, _, _, _ = lse_and_grad(a_new, beta, fsize)
        if lse_new <= lse_current - c_armijo * alpha * gnorm_sq:
            return a_new, alpha
        alpha *= 0.5
    
    # If all attempts fail, try a very small step
    a_new = project_simplex(a - alpha * grad, S)
    return a_new, alpha

def spectral_perturb(a, fsize, rng, strength=0.1):
    """Apply a random phase perturbation in the frequency domain."""
    n = len(a)
    A = fft(a, fsize)

            A_perturbed[fsize - k] = np.conj(A_perturbed[k])
    a_new = np.real(ifft(A_perturbed))[:n]
    return a_new

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using LSE-smoothed gradient
descent with backtracking line search, adaptive beta annealing, and
spectral-domain perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on Lagrange multiplier."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_grad(a, beta, fsize):
    """Compute LSE-smoothed max of self-convolution and its gradient."""
    n = len(a)
    A = fft(a, fsize)
    c = np.real(ifft(A * A)[:2 * n - 1 + 1])[:2 * n - 1]
    cm = c.max()
    scaled = beta * (c - cm)
    # Guard against overflow in exp
    max_scaled = scaled.max()
    if max_scaled > 500.0:
        scaled = scaled - max_scaled
        cm = cm + max_scaled / beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n), c, cm
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    # Gradient: dLSE/da = (2/beta) * sum_k w_k * dc_k/da
    # In FFT domain: 2 * ifft(W * conj(A))[:n]
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, cm


def lse_value_only(a, beta, fsize):
    """Compute only the LSE value (faster, no gradient)."""
    n = len(a)
    A = fft(a, fsize)
    c = np.real(ifft(A * A)[:2 * n])[:2 * n - 1]
    cm = c.max()
    scaled = beta * (c - cm)
    max_scaled = scaled.max()
    if max_scaled > 500.0:
        scaled = scaled - max_scaled
        cm = cm + max_scaled / beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm
    return cm + np.log(tot) / beta


def backtrack_search(a, grad, beta, fsize, S, lse_cur, max_ls=30, c_armijo=1e-4):
    """Backtracking line search with Armijo condition."""
    gnorm_sq = np.dot(grad, grad)
    if gnorm_sq < 1e-30:
        return a, 0.0
    alpha = 1.0
    for _ in range(max_ls):
        a_new = project_simplex(a - alpha * grad, S)
        lse_new = lse_value_only(a_new, beta, fsize)
        if lse_new <= lse_cur - c_armijo * alpha * gnorm_sq:
            return a_new, alpha
        alpha *= 0.4
    # Fallback: very small step
    a_new = project_simplex(a - alpha * grad, S)
    return a_new, alpha


def spectral_perturb(a, fsize, rng, strength=0.1):
    """Apply random phase perturbation in frequency domain, preserving DC."""
    n = len(a)
    A = fft(a, fsize)
    mask = rng.random(fsize) < 0.25
    mask[0] = False
    phases = rng.uniform(-strength, strength, fsize)
    A_new = A.copy()
    idx = np.where(mask)[0]
    A_new[idx] = A[idx] * np.exp(1j * phases[idx])
    # Enforce Hermitian symmetry
    for k in range(1, (fsize // 2) + 1):
        mirror = fsize - k
        if mirror < fsize:
            avg = (A_new[k] + np.conj(A_new[mirror])) / 2.0
            A_new[k] = avg
            A_new[mirror] = np.conj(avg)
    a_new = np.real(ifft(A_new))[:n]
    return a_new

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

            best_a = a.copy()

        idx = np.argsort(c)[::-1][:k]
        # Use a soft weighting: weights proportional to (c[i] - median)^+
        c_sorted = c[idx]
        ref = c_sorted[-1]  # smallest of top-k
        weights = np.maximum(c_sorted - ref, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum

        w_full = np.zeros(2 * n - 1)
        w_full[idx] = weights

        A = fft(a, fsize)
        W = fft(w_full, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Adaptive step: decrease over time
        progress = min(1.0, (time.monotonic() - start) / max_time)
        step = 1.0 / (1.0 + 0.01 * it + 2.0 * progress)
        step *= S / (np.sqrt(n) * 5.0)

        a = project_simplex(a - step * grad, S)

        # Occasional noise to escape
        if it % 1000 == 0:
            decay = np.exp(-progress * 3.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_cm:
        best_cm = c_final.max()
    return best_a, best_cm


def coarse_to_fine(base_seq, n_target, S_target, max_time, rng):
    """Optimize at smaller n, then upscale and continue at n_target."""
    # Optimize at n/8 first
    n_coarse = max(256, n_target // 8)
    S_coarse = np.sqrt(2.0 * n_coarse)

    if base_seq is not None:
        if len(base_seq) == n_coarse:
            a_c = base_seq.copy()
        elif len(base_seq) > n_coarse:
            ratio = len(base_seq) // n_coarse
            a_c = base_seq[:n_coarse * ratio].reshape(n_coarse, ratio).mean(axis=1)
        else:
            t_old = np.linspace(0, 1, len(base_seq))
            t_new = np.linspace(0, 1, n_coarse)
            a_c = np.interp(t_new, t_old, base_seq)
    else:
        a_c = np.ones(n_coarse) * 0.5

    a_c = normalize_to_sum(a_c, S_coarse)

    # Quick optimization at coarse scale
    t_coarse = max(10.0, max_time * 0.1)
    a_c, cm_c = run_lse_optimize(a_c, S_coarse, t_coarse, rng,
                                   beta_init=50.0, beta_max=10000.0, beta_rate=0.005)
    print("  Coarse (n=%d): max_c=%.6f" % (n_coarse, cm_c))

    # Upscale to target
    t_old = np.linspace(0, 1, len(a_c))
    t_new = np.linspace(0, 1, n_target)
    a_fine = np.interp(t_new, t_old, a_c)
    a_fine = normalize_to_sum(a_fine, S_target)

    # Also try direct upscale of base if available
    if base_seq is not None:
        if len(base_seq) == n_target:
            a_direct = normalize_to_sum(base_seq, S_target)
        elif len(base_seq) > n_target:
            ratio = len(base_seq) // n_target
            a_direct = base_seq[:n_target * ratio].reshape(n_target, ratio).mean(axis=1)
            a_direct = normalize_to_sum(a_direct, S_target)
        else:
            t_old2 = np.linspace(0, 1, len(base_seq))
            a_direct = np.interp(t_new, t_old2, base_seq)
            a_direct = normalize_to_sum(a_direct, S_target)

        c_up = fft_convolve_self(a_fine).max()
        c_dir = fft_convolve_self(a_direct).max()
        if c_dir < c_up:
            a_fine = a_direct
            print("  Using direct base (better than coarse upscale)")

    return a_fine


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LSE-smoothed gradient descent with backtracking, adaptive beta,
    spectral perturbations, and multi-phase peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, budget_s - 40.0)

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)

    # Get base sequence
    base = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        base = np.clip(base, 0.0, 1000.0)
        print("Base sequence: length=%d, sum=%.4f" % (len(base), base.sum()))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)

    # Prepare initial sequence using coarse-to-fine
    print("\n--- Phase 0: Coarse-to-fine initialization ---")
    t_coarse = 60.0
    a_init = coarse_to_fine(base, n, S, t_coarse, rng)

    c0 = fft_convolve_self(a_init)
    obj0 = 2.0 * n * c0.max() / (a_init.sum() ** 2)
    print("Initial: obj=%.6f, max_c=%.6f" % (obj0, c0.max()))

    global_best = obj0
    global_a = a_init.copy()

    # --- Phase 1: Main LSE optimization ---
    t_remain = deadline - time.monotonic()
    t_p1 = t_remain * 0.55
    print("\n--- Phase 1: LSE optimization (%.0fs) ---" % t_p1)

    a1, cm1 = run_lse_optimize(a_init, S, t_p1, rng,
                                beta_init=80.0, beta_max=80000.0, beta_rate=0.002)
    obj1 = 2.0 * n * cm1 / (a1.sum() ** 2)
    print("After Phase 1: obj=%.6f, max_c=%.6f" % (obj1, cm1))

    if obj1 < global_best:
        global_best = obj1
        global_a = a1.copy()

    # --- Phase 2: Peak shaving ---
    t_remain = deadline - time.monotonic()
    t_p2 = t_remain * 0.30
    if t_p2 > 20:
        print("\n--- Phase 2: Peak shaving (%.0fs) ---" % t_p2)
        a2, cm2 = peak_shave_optimize(global_a, S, t_p2, rng)
        obj2 = 2.0 * n * cm2 / (a2.sum() ** 2)
        print("After Phase 2: obj=%.6f, max_c=%.6f" % (obj2, cm2))
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()

    # --- Phase 3: High-beta LSE polish ---
    t_remain = deadline - time.monotonic()
    t_p3 = t_remain * 0.40
    if t_p3 > 15:
        print("\n--- Phase 3: High-beta polish (%.0fs) ---" % t_p3)
        a3, cm3 = run_lse_optimize(global_a, S, t_p3, rng,
                                    beta_init=5000.0, beta_max=200000.0, beta_rate=0.001)
        obj3 = 2.0 * n * cm3 / (a3.sum() ** 2)
        print("After Phase 3: obj=%.6f, max_c=%.6f" % (obj3, cm3))
        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()

    # --- Phase 4: Final peak shave ---
    t_remain = deadline - time.monotonic()
    if t_remain > 10:
        print("\n--- Phase 4: Final peak shave (%.0fs) ---" % t_remain)
        a4, cm4 = peak_shave_optimize(global_a, S, t_remain, rng)
        obj4 = 2.0 * n * cm4 / (a4.sum() ** 2)
        print("After Phase 4: obj=%.6f, max_c=%.6f" % (obj4, cm4))
        if obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification: %.6f ===" % ver)
        if not np.isfinite(ver):
            print("evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return final_list
