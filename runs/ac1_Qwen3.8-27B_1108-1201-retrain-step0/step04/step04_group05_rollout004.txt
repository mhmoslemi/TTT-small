def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_and_fft(a, N):
    """Self-convolution via FFT, also returns A for reuse."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_weights(c, beta):
    """Compute softmax weights for LSE. Returns (weights, lse_value, c_max)."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w, c_max, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return w, lse_val, c_max


def grad_from_weights(w, a, N, fsize):
    """Compute gradient of sum_k w_k * c_k w.r.t. a using FFT."""
    n = len(a)
    w_pad = np.zeros(fsize)
    w_pad[:len(w)] = w
    W = fft(w_pad)
    a_pad = np.zeros(fsize)
    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def lbfgs_two_loop(s_hist, y_hist, g, m_max=12):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-30:
        return -g / max(gn, 1e-30)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    sy_last = np.dot(s_last, y_last)
    if yy_last > 1e-30 and sy_last > 1e-30:
        gamma_use = sy_last / yy_last
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def spectral_deflate(a, N, top_k, alpha, rng):
    """Spectral deflation: reduce frequency components contributing to temporal peaks."""
    n = len(a)
    S = a.sum()
    c, A = compute_conv_and_fft(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    P = np.abs(A) ** 2
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    # Maintain total power approximately
    power_ratio = P.sum() / max(P_new.sum(), 1e-30)
    P_new *= min(power_ratio, 2.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

    per_init = phase1_budget / n_inits
    print("  %d initializations, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        a_norm = safe_normalize(arr, n, S)
        rem = max(2.0, per_init - (time.monotonic() - t_start) - 0.5)
        rem = min(rem, per_init)
        a_l, sc_l = lbfgs_optimize(a_norm, n, S, N, fsize, rem, rng,
                                    beta_init=5.0, beta_rate=1.07, beta_max=100.0,
                                    m_lbfgs=8, verbose_label=name)
        update_best(a_l, name)

    print("Phase 1 best: %.6f" % global_best)

    # === Phase 2: Focused L-BFGS with increasing beta ===
    print("=== Phase 2: Focused L-BFGS ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.5)

    a_curr = global_a.copy() if global_a is not None else np.ones(n) * S / n
    stall = 0
    last_best = global_best
    block_count = 0

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(180.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        block_count += 1
        beta_start = 15.0 + 8.0 * min(block_count, 10)
        beta_end = min(500.0, 80.0 * (1 + 0.3 * block_count))
        a_new, sc_new = lbfgs_optimize(a_curr, n, S, N, fsize, block_time, rng,
                                        beta_init=beta_start, beta_rate=1.04,
                                        beta_max=beta_end, m_lbfgs=12,
                                        verbose_label="P2_%d" % block_count)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2_%d] %.6f" % (block_count, global_best))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                # Targeted perturbation based on peak structure
                c_curr = compute_conv(a_curr, N)
                top_idx = np.argsort(c_curr)[::-1][:10]
                w_top = np.zeros(2 * n - 1)
                for k in top_idx:
                    w_top[k] = c_curr[k]
                w_top /= w_top.sum()
                grad_top = grad_from_weights(w_top, a_curr, N, fsize)
                a_curr = project_simplex(a_curr - 0.02 * S / np.sqrt(n) * grad_top, S)
                stall = 0

    # === Phase 3: Spectral deflation refinement ===
    print("=== Phase 3: Spectral deflation ===")
    t3 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t3) * 0.2)

    a_sp = global_a.copy()
    best_sp = global_best
    for i in range(5):
        rem = max(5.0, phase3_budget / 5 - (time.monotonic() - t3))
        top_k = max(3, 20 - 4 * i)
        alpha = 0.12 - 0.015 * i
        a_sp = spectral_deflate(a_sp, N, top_k, alpha, rng)
        c_sp = compute_conv(a_sp, N)
        sc_sp = c_sp.max()
        if sc_sp < best_sp:
            best_sp = sc_sp
            update_best(a_sp, "spectral_%d" % i)
        # Follow up with short L-BFGS
        if rem > 20:
            a_l, sc_l = lbfgs_optimize(a_sp, n, S, N, fsize, min(rem - 5, 60.0), rng,
                                        beta_init=30.0, beta_rate=1.05, beta_max=200.0,
                                        m_lbfgs=10, verbose_label="spec_lbg_%d" % i)
            update_best(a_l, "spec_lbg_%d" % i)
            a_sp = a_l

    # === Phase 4: High-beta precision L-BFGS ===
    print("=== Phase 4: High-beta precision ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        a_fin, sc_fin = lbfgs_optimize(global_a, n, S, N, fsize, min(remaining - 10, 400.0), rng,
                                        beta_init=50.0, beta_rate=1.03, beta_max=800.0,
                                        m_lbfgs=12, verbose_label="final")
        update_best(a_fin, "final")

    # === Phase 5: Final perturbation search ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("=== Phase 5: Final search ===")
        for i in range(min(10, int(remaining // 30))):
            if time.monotonic() > deadline - 15:
                break
            # Small random perturbation
            a_pert = project_simplex(global_a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            a_l, sc_l = lbfgs_optimize(a_pert, n, S, N, fsize, 25.0, rng,
                                        beta_init=40.0, beta_rate=1.04, beta_max=400.0,
                                        m_lbfgs=10, verbose_label="pert_%d" % i)
            update_best(a_l, "pert_%d" % i)

    # === Verification and return ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        obj_val = 2.0 * n * global_best / (S * S)
        print("Final: len=%d, internal_obj=%.6f, verify=%.6f" % (len(global_a), obj_val, ver if np.isfinite(ver) else float('inf')))
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
