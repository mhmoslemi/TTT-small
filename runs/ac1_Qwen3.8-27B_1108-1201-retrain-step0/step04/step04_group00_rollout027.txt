"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: Start from asymmetric initializations to avoid the symmetry trap
that causes gradient methods to get stuck at obj≈2. Linear ramp gives obj≈4/3
before optimization, and further optimization can improve this.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, k, power=1.0):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = c = fft_convolve_self(a)
    best_conv = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if stuck_count > 2000:
                k = max(1, k // 8)
            grad = topk_gradient(y, c, k, power=1.5)
        else:  # hybrid
            if stuck_count > 1000:
                grad = topk_gradient(y, c, max(1, min(32, n//64)), power=1.0)
            else:
                grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if stuck_count > 3000:
            step *= 0.5
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_multiplicative(a_init, max_time, rng, beta_init=5.0, beta_rate=0.004,
                       beta_max=2000.0, eta_init=0.1, eta_decay=0.9999):
    """Multiplicative (mirror descent) update for the LSE-smoothed objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    eta = eta_init
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Multiplicative update with clip to prevent overflow
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad / gnorm * np.sqrt(n)
        a = np.exp(log_a_new - log_a_new.max())
        a = a * (S / a.sum())

        eta *= eta_decay

        # Noise when stuck
        if it - last_improve > 5000 and it % 500 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = np.maximum(a + noise, 0)
            a = a * (S / a.sum())

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 4 + stuck // 200)
        else:
            k = min(64, 16 + stuck // 500)

        grad = topk_gradient(y, c, k, power=1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def make_asymmetric_init(n, kind="ramp"):
    t = np.linspace(0, 1, n, endpoint=False)
    if kind == "ramp":
        a = t + 0.1
    elif kind == "ramp2":
        a = (t + 0.1) ** 2
    elif kind == "ramp3":
        a = (t + 0.1) ** 3
    elif kind == "exp":
        a = np.exp(2 * t)
    elif kind == "sin":
        a = np.sin(np.pi * t / 2) ** 2 + 0.01
    elif kind == "logistic":
        a = 1.0 / (1.0 + np.exp(-4 * (t - 0.3)))
    elif kind == "bump_left":
        a = np.exp(-((t - 0.25) ** 2) / 0.05) + 0.01
    elif kind == "bump_right":
        a = np.exp(-((t - 0.75) ** 2) / 0.05) + 0.01
    elif kind == "cos_asym":
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.3 * t
    else:
        a = t + 0.1
    a = np.maximum(a, 0)
    return a
