def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

    """
    Peak-focusing subgradient method.
    At each step, compute the convolution, find top_k peaks,
    compute weighted gradient, and update.
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    a = a * (S / a.sum())
    
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    best_obj = 2.0 * n * best_cmax / (S * S)
    
    start_time = time.monotonic()
    it = 0
    step = step_init
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        cmax = c.max()
        cur_obj = 2.0 * n * cmax / (S * S)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            best_cmax = cmax
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Add noise to escape local minimum
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            no_improve = 0
        
        # Find top_k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            grad += weights[i] * peak_gradient(a, int(k))
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Try the step
        a_candidate = project_simplex(a - step * grad / gnorm, S)
        c_candidate = compute_conv(a_candidate, N)
        cmax_candidate = c_candidate.max()
        
        if cmax_candidate <= cmax:
            a = a_candidate
            # Optionally increase step
            if it % 50 == 0:
                step = min(step * 1.1, 1.0)
        else:
            # Backtrack
            step *= 0.7
            a = a_candidate if cmax_candidate < cmax * 1.5 else a
        
        # Decay step slowly
        if it % 100 == 0:
            step = max(step * step_decay, 0.001)
    
    return best_a, best_obj


def spectral_step(a, N, top_k, alpha):
    """One spectral shaping step: dampen the top peaks in the power spectrum."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        kk = int(k)
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * kk / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def spectral_block(a_init, n, S, N, max_time, rng, top_k=5, alpha=0.15):
    """Spectral shaping block."""
    a = a_init.copy()
    best_a = a.copy()
    best_obj = objective_value(a, N)
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        a = spectral_step(a, N, top_k, alpha)
        if it % 50 == 0:
            cur_obj = objective_value(a, N)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
    
    final_obj = objective_value(a, N)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    return best_a, best_obj


def make_power_init(n, p, rng):
    """Create a power-law initialization: a_j = (j+1)^p."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** p
    return a


def make_asymmetric_init(n, p, shift, rng):
    """Asymmetric power-law: a_j = (1 + p*j/n + shift*exp(-j/n))^p or similar."""
    t = np.linspace(0, 1, n)
    a = 0.1 + 0.9 * (t ** p)
    a += shift * np.exp(-5 * t) * 0.1
    return np.maximum(a, 0.01)
