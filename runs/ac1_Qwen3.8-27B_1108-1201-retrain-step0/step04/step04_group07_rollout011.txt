"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smoothed projected gradient
descent with Nesterov acceleration and adaptive smoothing parameter.

Key ideas:
1. The problem is convex (minimize max of quadratic forms over the simplex)
2. Use softmax smoothing of the max to get a differentiable objective
3. Use Nesterov-accelerated projected gradient descent
4. Adaptively increase the smoothing parameter (anneal)
5. Use FFT for fast convolution and gradient computation
"""
import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / rho
    valid = theta <= w
    if not np.any(valid):
        theta = (v.sum() - S) / n
    else:
        idx = np.max(np.where(valid)[0])
        theta = (cumsum[idx] - S) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(b, fsize)
    return np.fft.irfft(A * B, fsize)[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2 * n * c.max() / (S * S))


def compute_gradient_smoothed(a, c, beta):

    Compute the gradient of the smoothed objective:

    where c = a * a.
    
    Returns the gradient w.r.t. a.

    # corr = ifft(fft(w_padded) * conj(fft(a_padded))) 
    # Actually: corr[i] = sum_j a[j] * w[j+i]
    # Let's define: b = reverse(w), then corr[i] = sum_j a[j] * w[j+i] = (a * b)[len(b)-1-i]... 
    # This is getting confusing. Let me use a different approach.

    # Note: w has length 2n-1, a has length n

        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    
    start = time.monotonic()
    it = 0
    step_base = S / (n * 2.0)  # initial step size estimate
    
    print("  Starting accelerated PG: initial score = %.6f" % best_score)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute convolution of y
        c = fft_convolve(y, y)
        
        # Compute gradient
        beta = beta_init * (beta_final / beta_init) ** (it / max(1, int(max_time / 0.01)))
        # Actually, let's just linearly interpolate

            y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Check score periodically
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:

def spectral_rebalance(a, N, n_iter, rng):
    """
    Spectral rebalancing: iteratively modify |FFT(a)| to flatten the 
    power spectrum, which helps reduce convolution peaks.
    """
    n = len(a)
    S = a.sum()
    a_curr = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        # Pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a_curr
        A = np.fft.fft(a_pad)
        magnitude = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: bring magnitudes toward the mean
        mean_mag = magnitude.mean()
        if mean_mag < 1e-30:
            break
        
        # Compress: new_mag = mag^alpha for alpha < 1
        alpha = 0.7 + 0.3 * (it / max(1, n_iter))  # start aggressive, end gentle
        new_mag = np.power(magnitude + 1e-30, alpha)
        # Normalize to preserve sum
        a_new = np.real(np.fft.ifft(new_mag * np.exp(1j * phase)))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a_curr
        else:
            a_new = a_new * (S / s)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_curr = a_new
        # Regardless of improvement, continue from a_curr (or a_new?)
        # Use a_curr = a_new only if it improved, otherwise try a blend
        if sc >= best_score:
            # Blend
            a_curr = 0.7 * best_a + 0.3 * a_new
            a_curr = np.maximum(a_curr, 0.0)
            a_curr = a_curr * (S / a_curr.sum())
        
        if best_score <= 1.50:
            break
    
    return best_a, best_score


def warm_start_extend(a_short, n_target, S_target):

    Extend a shorter optimized sequence to a target length by

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smoothed projected gradient descent
with Nesterov acceleration, adaptive smoothing, and multi-phase optimization.

Key improvements over previous approach:
1. Correct gradient computation for the smoothed max-convolution objective

4. Better initialization strategies exploiting anti-palindromic structure
5. Efficient FFT-based gradient evaluation
6. Adaptive beta (smoothing parameter) annealing

import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = (v.sum() - S) / n
    else:
        idx = int(np.max(np.where(valid)[0]))
        theta = (cumsum[idx] - S) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, b):
    """Linear convolution using FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(b, fsize)
    return np.fft.irfft(A * B, fsize)[:size]


def self_convolve(a):
    """Self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = self_convolve(a)
    return float(2.0 * n * c.max() / (S * S))


def compute_smoothed_gradient(a, c, S, beta):

    f(a) = (2n / (beta * S^2)) * log(sum_k exp(beta * c_k / S^2))
    
    Returns (gradient, max_c, active_lags)
    """
    n = len(a)
    c_scale = beta / (S * S)

    c_shifted = c * c_scale - c.max() * c_scale
    exp_c = np.exp(c_shifted)
    w = exp_c / exp_c.sum()

def spectral_rebalance(a_init, n, S, max_time, rng=None):
    """
    Iterative spectral magnitude flattening to reduce convolution peaks.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten magnitudes toward geometric mean
        log_mag = np.log(mag + 1e-30)
        target_log_mag = np.median(log_mag)
        blend = 0.3 + 0.5 * (it / max(1, int(max_time * 100)))
        new_log_mag = (1.0 - blend) * log_mag + blend * target_log_mag
        new_mag = np.exp(new_log_mag)
        
        # Preserve DC component (sum)
        new_a = np.real(np.fft.ifft(new_mag * np.exp(1j * phase)))[:n]
        new_a = np.maximum(new_a, 0.0)
        ns = new_a.sum()
        if ns < 1e-15:
            new_a = a.copy()
        else:
            new_a = new_a * (S / ns)
        
        sc = compute_score(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
        else:
            # Blend
            a = 0.8 * best_a + 0.2 * new_a
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
        
        if it % 200 == 0:
            sc_now = compute_score(a)
            if sc_now < best_score:
                best_score = sc_now
                best_a = a.copy()
            if best_score <= 1.50:
                break
    
    return best_a, best_score
