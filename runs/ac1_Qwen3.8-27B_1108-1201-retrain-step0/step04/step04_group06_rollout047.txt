"""
Minimize 2*n*max_conv/sum^2 using adaptive-temperature soft-peak gradient descent
with simplex projection, multi-start exploration, and spectral diversification.

Key innovations:
1. Soft-peak (log-sum-exp) gradient: smooth approximation to max-subgradient
   that provides direction from ALL peaks, weighted by their prominence.
2. Adaptive temperature schedule: starts high (global exploration) and
   decreases (local exploitation).
3. Efficient FFT-based gradient computation: O(N log N) per iteration.
4. Multi-scale approach: try multiple sequence lengths.
5. Spectral diversification: periodic frequency-domain perturbations to
   escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N1):
    """Compute self-convolution of a using FFT of size N1."""
    A = fft(a, N1)
    c = np.real(ifft(A * A))
    n = len(a)
    return c[:2*n-1]


def compute_soft_gradient(a, w, N2):
    """
    Compute gradient of soft-peak objective.
    g[i] = 2 * sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n-1])
    Using FFT cross-correlation.
    """
    A = fft(a, N2)
    W = fft(w, N2)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return g


def soft_max_weights(c, T):
    """Compute soft-max weights from convolution values with temperature T."""
    # Numerically stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    Z = exp_c.sum()
    if Z < 1e-300:
        # All weights equal (uniform)
        return np.ones(len(c)) / len(c)
    return exp_c / Z

    """

    a = a_init.copy()
    a = project_simplex(a, S)
    
    best_a = a.copy()
    best_score = np.inf
    
    if T_start is None:
        c0 = compute_conv(a, N1)
        T_start = c0.max() * 5.0
    
    T = T_start
    start_time = time.monotonic()
    it = 0
    stall = 0
    last_best_score = np.inf
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Compute convolution
        c = compute_conv(a, N1)
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
            stall = 0
        
        # Compute soft-max weights
        w = soft_max_weights(c, T)
        
        # Compute gradient
        g = compute_soft_gradient(a, w, N2)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Step size: adaptive based on gradient magnitude and current scale
        # The convolution values are O(S^2/n), gradient is O(S^2/n) per component
        # We want the step to change a by O(S/n) per component
        step = step_scale * S / (n * (1.0 + 0.1 * T / T_min))
        
        a_new = a - step * g / g_norm * S  # Scale step by S for scale invariance
        
        # Project onto simplex
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c_new = compute_conv(a_new, N1)
        c_max_new = c_new.max()
        
        if c_max_new < best_score:
            best_score = c_max_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        elif c_max_new < c_max:
            a = a_new.copy()
            stall += 1
        else:
            # Reject this step, try smaller step
            a_reject = project_simplex(a - step * 0.5 * g / g_norm * S, S)
            c_reject = compute_conv(a_reject, N1)
            if c_reject.max() < c_max:
                a = a_reject.copy()
            stall += 1
        
        # Decrease temperature
        T = max(T_min, T * T_decay)
        
        # Periodic noise for exploration
        if it % noise_period == 0 and stall > 50:
            noise = rng.standard_normal(n) * noise_scale * S / np.sqrt(n)
            a = project_simplex(a + noise, S)
            T = max(T_min * 10, T * 2)  # Increase temperature to escape
            stall = 0
        
        # Occasional larger perturbation
        if it % 1000 == 500 and stall > 100:
            noise = rng.standard_normal(n) * 0.01 * S / np.sqrt(n)
            a = project_simplex(a + noise, S)
            T = T_start * 0.5  # Reset to moderate temperature
            stall = 0
        
        if it % 10000 == 0:
            print("    iter %d: score=%.6f, T=%.4f, stall=%d" % (it, best_score, T, stall))
        
        # Early stop if we're doing very well
        if best_score < 1.01:
            break
    
    return best_a, best_score


def spectral_perturb(a, N1, n, S, sigma, rng):
    """Add random complex noise to the FFT, project back to simplex."""
    A = fft(a, N1)
    noise = sigma * (rng.standard_normal(N1) + 1j * rng.standard_normal(N1))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def spectral_flatten(a, N1, n, S, alpha, rng):
    """
    Flatten the power spectrum to reduce peak autocorrelation.
    P_new = P^(1-alpha) * P_mean^alpha (geometric interpolation with flat)
    """
    A = fft(a, N1)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    
    # Interpolate towards flat spectrum
    P_new = P ** (1.0 - alpha) * P_mean ** alpha
    
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def structured_init(n, rng, method="v2"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2*np.pi*3*t) + 0.2 * np.sin(2*np.pi*7*t) \
            + 0.1 * np.sin(2*np.pi*13*t) + 0.05 * rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2*np.pi*(2*t + 3*t*t)) + 0.1 * rng.standard_normal(n)
    elif method == "alt":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.1 * rng.standard_normal(n)
    elif method == "multi_sin":
        # Many sinusoidal components for a "random-looking" but structured signal
        a = 0.5
        for k in range(1, 8):
            a += 0.1 * np.sin(2*np.pi*(2*k+1)*t + rng.random()*2*np.pi)
        a += 0.05 * rng.standard_normal(n)
    elif method == "random_smooth":
        # Random but smooth (low-pass filtered)
        a = 0.3 + 0.7 * rng.random(n)
        # Smooth it
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=n//32)
    elif method == "binary_smooth":
        # Start binary, then smooth
        a = np.where(rng.random(n) > 0.5, 1.0, 0.2)
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(a, size=n//16)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    
    a = np.clip(a, 0.01, 2.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
