"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using:
1. FISTA with adaptive LSE smoothing and backtracking
2. Direct subgradient peak shaving with Polyak step
3. Spectral domain mixing for diversity
4. Multi-length exploration (n=8192, 16384)

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    """Smooth max via log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Prevent overflow
    max_scaled = scaled.max()
    if max_scaled > 500:
        scaled = scaled - max_scaled + 500 / beta  # Hmm, this doesn't work
    # Actually, let's just clip
    scaled = np.clip(scaled, -500, 500)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta

def lse_gradient(c, beta):
    """Weights for LSE gradient computation."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def compute_subgradient_from_weights(a, w):
    """Compute 2 * cross_correlate(a, w) via FFT, truncated to length n."""
    n = len(a)
    fsize = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def objective_value(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def spectral_mix(a1, a2, alpha=0.5):
    """Mix two sequences in frequency domain."""
    n1, n2 = len(a1), len(a2)
    n = max(n1, n2)
    # Resample to common length
    if n1 != n:
        a1_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n1), a1)
    else:
        a1_r = a1
    if n2 != n:
        a2_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n2), a2)
    else:
        a2_r = a2
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1_r, fsize)
    A2 = fft(a2_r, fsize)
    A_mix = alpha * A1 + (1 - alpha) * A2
    a_mix = np.real(ifft(A_mix))[:n]
    return np.maximum(a_mix, 0.0)

    a = normalize_to_sum(a_init, S)

                    g[:n-k] += a[k:]
                # a_{j-k} contributes to g_j for j=k..n-1

                    g[k:] += a[:n-k]

                continue

            # c_new[k_star] ≈ c[k_star] - step * (g · a_shifted)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_weights(c, beta):
    """Compute normalized weights for LSE-smoothed max gradient."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def lse_value(c, beta):
    """Compute LSE-smoothed max value."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    return cm + np.log(np.exp(scaled).sum()) / beta


def compute_grad_from_weights(a, w):
    """Compute 2 * cross_correlate(a, w) truncated to len(a)."""
    n = len(a)
    fsize = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def spectral_mix(a1, a2, alpha=0.5):
    """Mix two sequences in frequency domain (linear combination of FFTs)."""
    n = max(len(a1), len(a2))
    if len(a1) != n:
        a1 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a1)), a1)
    if len(a2) != n:
        a2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a2)), a2)
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    A_mix = alpha * A1 + (1.0 - alpha) * A2
    a_mix = np.real(ifft(A_mix))[:n]
    return np.maximum(a_mix, 0.0)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.003, beta_max=10000.0):
    """FISTA with LSE smoothing and backtracking line search."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Gradient via LSE weights
        w = lse_gradient_weights(c, beta)
        grad = compute_grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Backtracking line search
        f_curr = lse_value(c, beta)
        step = min(1.0 / (4.0 * beta), 2.0 * S / (n * gnorm + 1e-10))
        step = max(step, 1e-12)

        x_new = y
        accepted = False
        for _ in range(8):
            x_cand = project_simplex(y - step * grad, S)
            c_cand = fft_convolve_self(x_cand)
            f_cand = lse_value(c_cand, beta)
            if f_cand <= f_curr - 1e-12 * step * gnorm * gnorm:
                x_new = x_cand
                accepted = True
                break
            step *= 0.5

        if not accepted:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Restart if stuck
        if consecutive_no_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0
            consecutive_no_improve = 0

        # Gentle periodic perturbation for exploration
        if it % 4000 == 0 and consecutive_no_improve > 500:
            decay = np.exp(-it / 80000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12 and rng.random() < 0.5:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

    return best_a, best


def run_peak_shave(a_init, S, max_time, rng, top_k=5):
    """Direct peak shaving: target top peaks with carefully chosen steps."""
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    stall = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Perturb and reset
            noise_scale = 0.001 * S / np.sqrt(n) * max(0.01, 1.0 - stall / 2000.0)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stall = 0
            continue

        # Find top peaks
        k = min(top_k, n)
        peak_indices = np.argsort(c)[::-1][:k]
        c_mean = c.mean()

        improved = False
        for idx in peak_indices:
            k_star = int(idx)
            if c[k_star] <= c_mean * 1.01:
                break

            # Compute subgradient: g_j = a_{j+k*} + a_{j-k*} (boundary = 0)
            g = np.zeros(n)
            ks = k_star
            if 0 <= ks < n:
                g[:n - ks] += a[ks:]
            if ks > 0:
                g[ks:] += a[:n - ks]

            gnorm2 = np.dot(g, g)
            if gnorm2 < 1e-20:
                continue

            # Polyak-like step: try to reduce excess
            excess = c[k_star] - c_mean
            # The change in c[k_star] from step a -> a - step*g:
            # Δc[k_star] ≈ -2*step*(g_{k_star component}) ...
            # Use heuristic step
            step = 0.5 * excess / (gnorm2 + 1e-20)
            step = min(step, 0.05 * S / (np.sqrt(gnorm2) + 1e-10))

            a_new = project_simplex(a - step * g, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
                improved = True
                break

        if not improved:
            # Try all top peaks with smaller steps
            found = False
            for idx in peak_indices[:3]:
                k_star = int(idx)
                if c[k_star] <= c_mean:
                    continue
                g = np.zeros(n)
                ks = k_star
                if 0 <= ks < n:
                    g[:n - ks] += a[ks:]
                if ks > 0:
                    g[ks:] += a[:n - ks]
                gnorm2 = np.dot(g, g)
                if gnorm2 < 1e-20:
                    continue
                for frac in [0.1, 0.05, 0.01]:
                    step = frac * S / (np.sqrt(gnorm2) + 1e-10)
                    a_new = project_simplex(a - step * g, S)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < c.max():
                        a = a_new
                        found = True
                        break
                if found:
                    break
            if not found:
                # Small random perturbation
                noise_scale = 0.0003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_iterative_linearization(a_init, S, max_time, rng):
    """
    Alternating linearization: fix one factor of convolution,
    solve linear least squares in the other factor.
    minimize ||convolve(x, a_fixed) - c_target||^2 + mu*||x - a||^2
    subject to x >= 0, sum(x) = S
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    # Use iterative gradient method on the subproblem
    mu = 0.1  # regularization parameter

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Target: flatten the convolution
        c_mean = c.sum() / len(c)
        # Target is the current convolution but with peaks reduced
        c_target = np.minimum(c, c_mean * 1.5)  # cap peaks at 1.5x mean
        # Also ensure non-negativity
        c_target = np.maximum(c_target, 0.0)

        # Solve: minimize ||convolve(x, a) - c_target||^2 + mu*||x - a||^2
        # Gradient: 2 * a_correlate(convolve(x,a) - c_target, a) + 2*mu*(x - a)
        # Use a few gradient steps
        x = a.copy()
        for inner in range(3):
            c_x = fft_convolve(x, a)
            # Truncate to length of c_target
            L = min(len(c_x), len(c_target))
            resid = c_x[:L] - c_target[:L]
            # Gradient of ||resid||^2 w.r.t. x = 2 * cross_correlate(resid, a)
            # cross_correlate(resid, a)[j] = sum_k resid[k] * a[k-j]
            g = compute_grad_from_weights(x, c_x[:L])  # This is 2*convolve_stuff
            # Actually, let me compute it properly
            # d/dx_j ||convolve(x,a) - target||^2 = 2 * (convolve(target - convolve(x,a), a_rev))[j]
            # where a_rev is the reversed a
            # Simpler: use FFT
            resid_full = np.zeros(2 * n - 1)
            resid_full[:L] = resid
            a_rev = a[::-1]
            grad_conv = -2.0 * fft_convolve(resid_full, a_rev)[:n]
            grad_reg = 2.0 * mu * (x - a)
            grad_total = grad_conv + grad_reg

            gnorm = np.linalg.norm(grad_total)
            if gnorm < 1e-15:
                break
            step = 1.0 / (4.0 * mu + 2.0 * np.max(np.abs(a)) * len(a))
            step = min(step, 0.5 / (mu + 1e-10))

            x_new = project_simplex(x - step * grad_total, S)
            x = x_new

        a = x

        # Occasionally add perturbation
        if rng.random() < 0.05:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 5 * tt + 0.5))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))
    inits.append(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using multi-phase optimization:
    FISTA with LSE, direct peak shaving, iterative linearization, and spectral mixing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)

    # Load starting sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000)
            given_obj = objective_value(given)
            print("Loaded given sequence: length=%d, obj=%.6f" % (len(given), given_obj))
    except Exception:
        print("No given sequence available.")

    global_best = np.inf
    global_a = None

    # === PRIMARY: n=8192 ===
    n1 = 8192
    S1 = np.sqrt(2.0 * n1)

    # Prepare initializations
    inits1 = []
    if given is not None:
        if len(given) == n1:
            inits1.append(given.copy())
        elif len(given) > n1:
            ratio = len(given) // n1
            if ratio > 0:
                small = given[:n1 * ratio].reshape(n1, ratio).mean(axis=1)
                inits1.append(small)
        else:
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, n1)
            inits1.append(np.interp(t_new, t_old, given))

    gen_inits = generate_inits(n1, rng)
    inits1.extend(gen_inits[:5])

    # Phase 1: Quick FISTA on best inits (20% of time)
    t_phase1 = (deadline - time.monotonic()) * 0.20
    print("\n--- Phase 1: Quick FISTA exploration (%.0fs) ---" % t_phase1)

    best_a_n1 = None
    best_val_n1 = np.inf
    time_per_init = t_phase1 / min(4, len(inits1))

    for i, init in enumerate(inits1[:4]):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.6:
            break
        t_alloc = min(time_per_init, deadline - time.monotonic() - 60.0)
        if t_alloc < 5:
            break

        a_opt, val = run_fista(init, S1, t_alloc, rng,
                                beta_init=5.0, beta_rate=0.004, beta_max=3000.0)
        obj = objective_value(a_opt)
        print("  Init %d: obj=%.6f" % (i, obj))
        if val < best_val_n1:
            best_val_n1 = val
            best_a_n1 = a_opt.copy()
            global_best = obj
            global_a = a_opt.copy()

    if best_a_n1 is None:
        best_a_n1 = np.ones(n1) * S1 / n1
        best_val_n1 = fft_convolve_self(best_a_n1).max()

    # Phase 2: Intensive FISTA on best (30% of time)
    remaining_frac = 0.30
    t_phase2 = (deadline - time.monotonic()) * remaining_frac
    print("\n--- Phase 2: Intensive FISTA (%.0fs) ---" % t_phase2)

    a2, val2 = run_fista(best_a_n1, S1, t_phase2, rng,
                          beta_init=20.0, beta_rate=0.002, beta_max=8000.0)
    obj2 = objective_value(a2)
    print("  After intensive FISTA: obj=%.6f" % obj2)
    if val2 < best_val_n1:
        best_val_n1 = val2
        best_a_n1 = a2.copy()
        global_best = obj2
        global_a = a2.copy()

    # Phase 3: Direct peak shaving (20% of time)
    t_phase3 = (deadline - time.monotonic()) * 0.20
    print("\n--- Phase 3: Peak shaving (%.0fs) ---" % t_phase3)

    a3, val3 = run_peak_shave(best_a_n1, S1, t_phase3, rng, top_k=6)
    obj3 = objective_value(a3)
    print("  After peak shaving: obj=%.6f" % obj3)
    if val3 < best_val_n1:
        best_val_n1 = val3
        best_a_n1 = a3.copy()
        global_best = obj3
        global_a = a3.copy()

    # Phase 4: Iterative linearization (10% of time)
    t_phase4 = (deadline - time.monotonic()) * 0.10
    print("\n--- Phase 4: Iterative linearization (%.0fs) ---" % t_phase4)

    a4, val4 = run_iterative_linearization(best_a_n1, S1, t_phase4, rng)
    obj4 = objective_value(a4)
    print("  After linearization: obj=%.6f" % obj4)
    if val4 < best_val_n1:
        best_val_n1 = val4
        best_a_n1 = a4.copy()
        global_best = obj4
        global_a = a4.copy()

    # Phase 5: Final polishing with high beta (remaining time)
    t_phase5 = deadline - time.monotonic()
    if t_phase5 > 15:
        print("\n--- Phase 5: Final polishing (%.0fs) ---" % t_phase5)
        half = t_phase5 * 0.5

        a5, val5 = run_fista(best_a_n1, S1, half, rng,
                              beta_init=200.0, beta_rate=0.001, beta_max=30000.0)
        obj5 = objective_value(a5)
        print("  High-beta FISTA: obj=%.6f" % obj5)
        if val5 < best_val_n1:
            best_val_n1 = val5
            best_a_n1 = a5.copy()
            global_best = obj5
            global_a = a5.copy()

        # Final peak shave
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a6, val6 = run_peak_shave(best_a_n1, S1, remaining * 0.7, rng, top_k=4)
            obj6 = objective_value(a6)
            print("  Final peak shave: obj=%.6f" % obj6)
            if val6 < best_val_n1:
                best_val_n1 = val6
                best_a_n1 = a6.copy()
                global_best = obj6
                global_a = a6.copy()

    # === SECONDARY: Try n=16384 if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        print("\n--- Secondary: n=%d (%.0fs) ---" % (n2, remaining))

        # Upscale best solution
        t_old = np.linspace(0, 1, len(best_a_n1))
        t_new = np.linspace(0, 1, n2)
        init_n2 = np.interp(t_new, t_old, best_a_n1)

        a_n2, val_n2 = run_fista(init_n2, S2, remaining * 0.6, rng,
                                   beta_init=5.0, beta_rate=0.002, beta_max=3000.0)
        obj_n2 = objective_value(a_n2)
        print("  n=%d: obj=%.6f" % (n2, obj_n2))

        if obj_n2 < global_best:
            global_best = obj_n2
            global_a = a_n2.copy()
            print("  New global best from n=%d!" % n2)

        # Quick peak shave on n2
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_n2b, val_n2b = run_peak_shave(a_n2, S2, rem2 * 0.5, rng, top_k=4)
            obj_n2b = objective_value(a_n2b)
            print("  n=%d after peak shave: obj=%.6f" % (n2, obj_n2b))
            if obj_n2b < global_best:
                global
