"""
Minimize 2*n*max(self-convolution)/sum^2 using FISTA with LSE smoothing,
spectral post-processing, and adaptive LP corrections.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_grad(a, c, beta):
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(a))
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def run_var_descent(a_init, n, S, max_time, rng, lr_init=0.01, lr_decay=0.9999):
    """Gradient descent on the circular variance objective."""
    a = normalize(a_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    lr = lr_init
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        grad = grad_var_circ(a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = lr / max(gnorm / S, 1e-10)
        step = min(step, 1.0)  # Cap step size
        
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
            last_improve = it
            lr = min(lr * 1.05, 1.0)  # Increase step slightly on improvement
        else:
            lr *= 0.9  # Decrease step on non-improvement
        
        a = a_new
        
        # Periodic noise injection
        if it % 500 == 0 and it - last_improve > 200:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 5000)
            if noise_scale > 1e-8:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
    return best_a, best

def run_peak_descent(a_init, n, S, max_time, rng, top_k=5):
    """Gradient descent targeting the top-k peak lags."""
    a = normalize(a_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    lr = 0.01
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it
        
        # Get top-k lags
        idx = np.argsort(c)[::-1][:top_k]
        cm = c.max()
        
        # Weighted gradient
        weights = np.zeros(len(c))
        for k in idx:
            weights[k] = (c[k] / cm) ** 2
        
        # Compute weighted gradient
        g = np.zeros(n)
        for k in idx:
            g += weights[k] * grad_peak(a, k)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        step = lr / max(gnorm / S, 1e-10)
        step = min(step, 0.5)
        
        a_new = a - step * g
        a_new = project_simplex(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur * 1.1:
            # Step too large, reduce
            lr *= 0.5
            continue
        else:
            a = a_new
            lr = min(lr * 1.02, 0.5)
        
        if it % 300 == 0 and it - last_improve > 150:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def run_fista_lse(a_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.001, beta_max=5000.0):
    """FISTA with LSE smoothing."""
    a = normalize(a_init, n).copy()
    y = a.copy()
    t_fista = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        
        if it - last_improve > 3000:
            noise = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_fista = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista**2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_fista = t_new
    
    return best_a, best

def spectral_flatness_step(a, n, S, rng, strength=0.1):
    """Try to flatten the power spectrum."""
    A = fft(a, n)
    P = np.abs(A)**2
    P_mean = P.mean()
    
    # Dampen peaks in the spectrum
    damp = np.ones(n)
    threshold = P_mean * (1.0 + strength)
    damp[P > threshold] = 1.0 / np.sqrt(P[P > threshold] / P_mean)
    damp[0] = 1.0  # Don't touch DC
    
    new_mag = np.sqrt(P) * damp
    phase = np.angle(A)
    new_a = np.real(ifft(new_mag * np.exp(1j * phase), n))
    new_a = np.maximum(new_a, 0)
    return project_simplex(new_a, S)
