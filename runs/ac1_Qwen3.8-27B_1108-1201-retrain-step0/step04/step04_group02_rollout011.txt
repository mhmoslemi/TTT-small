Actually, L-BFGS-B handles bounds by only moving into the feasible region. So if a coordinate hits 0, L-BFGS-B won't push it below 0. The gradient computation should still be valid at the boundary (we just need to make sure we don't get NaN).

Let me also reconsider the gradient computation. The objective is:
f(a) = 2n * max_k c[k] / S^2

The LSE approximation:
f_smooth(a) = 2n * LSE_beta(c) / S^2

where LSE_beta(c) = cm + (1/beta) * log(sum exp(beta*(c[k]-cm)))

And c[k] = sum_i a[i] * a[k-i] (valid i).

"""
Minimize 2*n*max_conv/sum^2 using LSE-annealed optimization with
L-BFGS-B and projected gradient, multi-start from diverse initializations,
and multiple sequence lengths.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))

def lse_gradient(a, beta):
    """Compute gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    exponents = np.clip(beta * (c - cm), -700, 700)
    e = np.exp(exponents)
    tot = e.sum()
    w = e / tot
    # dmax/da[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    dmax = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return dmax, c, cm + np.log(tot) / beta

def lse_obj_grad(x, beta, n):
    """For scipy.optimize.minimize."""
    a = np.maximum(x, 0.0)
    if a.sum() < 1e-15:
        return 1e10, np.zeros(n)
    dmax, c, LSE = lse_gradient(a, beta)
    S = a.sum()
    val = 2.0 * n * LSE / (S * S)
    grad = 2.0 * n * dmax / (S * S) - 4.0 * n * LSE / (S * S * S)
    return val, grad

def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def run_lbfgsb(a_init, n, S, max_time, rng):
    """LSE-annealed L-BFGS-B."""
    a = project_nonneg_sum(a_init.copy(), S)
    best_obj = compute_objective(a)
    best_a = a.copy()
    start = time.monotonic()
    
    # Beta schedule: start smooth, end sharp
    betas = [5, 10, 20, 50, 100, 200, 400, 800, 1500]
    beta_idx = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        if beta_idx >= len(betas):
            beta_idx = len(betas) - 1
        
        beta = betas[beta_idx]
        
        # Determine max iterations based on time
        maxiter = int(min(200, max(20, remaining / 1.0)))
        
        try:
            res = minimize(lse_obj_grad, a, args=(beta, n),
                         method='L-BFGS-B',
                         bounds=[(0.0, 1000.0)] * n,
                         options={'maxiter': maxiter, 'ftol': 1e-16,
                                  'gtol': 1e-14, 'maxfun': maxiter * 10,
                                  'maxls': 50})
            a_new = project_nonneg_sum(np.maximum(res.x, 0.0), S)
            obj_new = compute_objective(a_new)
        except Exception:
            a_new = a.copy()
            obj_new = best_obj
        
        if obj_new
