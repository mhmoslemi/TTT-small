import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)

    if s < 1e-30:

    return a * (S / s)

    # Weight by softmax of top-k values
    c_top = c[top_k]
    weights = np.exp(100 * (c_top - c_top.max()))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, kstar in zip(weights, top_k):
        kstar = int(kstar)
        lo = max(0, kstar - n + 1)
        hi = min(kstar, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        g[idx] += w * 2.0 * a[kstar - idx]
    return g


def run_mirror_descent(a_init, n, S, max_time, rng, 
                       eta_init=0.05, momentum_decay=0.999,
                       noise_period=5000, noise_decay=20000):
    """Multiplicative (mirror) descent with Nesterov momentum."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    eta = eta_init
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # Compute subgradient
        g = compute_topk_subgradient(y, c, n, k=3)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        # Scale eta so that the relative change is reasonable
        eta_eff = eta * S / (gnorm + 1e-10)
        eta_eff = min(eta_eff, 0.5)  # cap
        
        # Multiplicative step
        log_scale = -eta_eff * g
        log_scale = np.clip(log_scale, -20, 20)
        a_new = y * np.exp(log_scale)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = np.ones(n) * S / n
        else:
            y_new = y_new * (S / s)
        
        a = project_simplex(a_new, S)
        y = y_new
        t = t_new
        
        # Decay eta
        eta *= momentum_decay
        
        # Noise injection
        if it % noise_period == 0:
            decay = np.exp(-it / noise_decay)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
        
        # Break if no improvement for a long time
        if it - last_improve > 50000 and eta < 1e-6:
            break
    
    return best_a, best


def run_spectral_shaping(a_init, n, S, max_time, rng, 
                         alpha_init=0.3, alpha_end=0.8,
                         blend=0.5):
    """Spectral shaping: iteratively flatten the power spectrum."""
    a = normalize_to_sum(a_init, S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    a_prev = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 5000.0)
        alpha = alpha_init + (alpha_end - alpha_init) * progress
        
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        
        if P_mean < 1e-30:
            break
        
        # Shape the spectrum
        # Method: P_new = P^alpha / mean(P^alpha), blended with flat
        P_shaped = np.power(np.maximum(P, 1e-30), alpha)
        P_shaped = P_shaped / P_shaped.mean()
        P_new = (1.0 - blend) * P / P_mean + blend * P_shaped
        
        # Also apply a "notch" at the peak autocorrelation frequency
        c = fft_convolve_self(a)
        k_peak = int(np.argmax(c))
        if k_peak > 0:
            # The peak at lag k_peak corresponds to frequency 2*pi*k_peak/fsize
            # Reduce power at that frequency and its harmonics
            freqs = np.arange(fsize)
            target_freq = k_peak  # in DFT bins
            notch = 1.0 - 0.3 * np.exp(-0.5 * ((freqs - target_freq) / max(1, fsize/50))**2)
            notch = np.minimum(notch, 1.0)
            P_new = P_new * notch
            # Also notch the negative frequency
            neg_target = fsize - target_freq
            notch2 = 1.0 - 0.3 * np.exp(-0.5 * ((freqs - neg_target) / max(1, fsize/50))**2)
            P_new = P_new * notch2
        
        mag_new = np.sqrt(P_new)
        # Preserve total power
        total_power = P.sum()
        P_new_sum = (mag_new ** 2).sum()
        if P_new_sum > 1e-30:
            mag_new *= np.sqrt(total_power / P_new_sum)
        
        A_new = phase * mag_new
        a_new = np.real(ifft(A_new))[:n]
        
        # Project
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, S)
        
        # Blend with previous for stability
        a_new = 0.7 * a_new + 0.3 * a_prev
        a_new = normalize_to_sum(a_new, S)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
    
    return best_a, best


def run_pair_cd(a_init, n, S, max_time, rng):
    """Targeted pair coordinate descent."""
    a = normalize_to_sum(a_init, S)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best = c.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        # Determine number of pairs to try this round
        n_pairs = min(100 + it * 10, 1000)
        
        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 0.5:
                break
            
            # Choose pair: targeted or random
            if rng.random() < 0.6:
                # Targeted: find elements contributing to peak
                k_peak = int(np.argmax(c))
                # Find top contributors to c[k_peak]
                lo = max(0, k_peak - n + 1)
                hi = min(k_peak, n - 1)
                idx = np.arange(lo, hi + 1)
                contrib = a[idx] * a[k_peak - idx]
                top_idx = idx[np.argsort(contrib)[::-1][:5]]
                m = int(top_idx[rng.integers(0, min(5, len(top_idx)))])
                
                # Choose j: a small element far from the peak contribution
                small_mask = a < np.median(a)
                if small_mask.sum() > 0:
                    j_candidates = np.where(small_mask)[0]
                    j = int(j_candidates[rng.integers(0, len(j_candidates))])
                else:
                    j = int(rng.integers(0, n))
            else:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))
            
            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue
            
            # Compute direction d_k = change in c_k per unit delta (transfer from m to j)
            # d_k = a[j+k] + a[j-k] - a[m+k] - a[m-k] (with zero-padding)
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val -= a[mk]
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val
            
            # Find optimal delta in [-a[j], a[m]] minimizing max(c + delta*d)
            # (ignoring the O(delta^2) correction at k=0 for speed)
            lo_d = -a[j]
            hi_d = a[m]
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = (c + c_g * d).max()
            fd = (c + d_g * d).max()
            
            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = (c + c_g * d).max()
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = (c + d_g * d).max()
            
            delta = (a_g + b_g) / 2.0
            delta = max(lo_d, min(hi_d, delta))
            
            if abs(delta) < 1e-12:
                continue
            
            # Apply
            a[j] += delta
            a[m] -= delta
            if a[j] < 0: a[j] = 0
            if a[m] < 0: a[m] = 0
            
            # Update c (approximate - recompute)
            c = c + delta * d
            # Fix c[0] (quadratic term)
            c[0] = c[0] + 2 * (a[j] - a[m]) * delta + 2 * delta ** 2
            # Actually, let me just recompute
            # c = fft_convolve_self(a)  # too expensive per pair
            
            cur = c.max()
            if cur < best - 1e-12:
                best = cur
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[j] -= delta
                a[m] += delta
                c = c - delta * d
                c[0] = c[0] - 2 * (a[j] - a[m]) * delta - 2 * delta ** 2
        
        # Periodic full recompute
        if it % 10 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        # Add noise if stuck
        if not improved:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 200.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best
