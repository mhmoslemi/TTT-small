def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v

    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]

    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):

    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a

    c = np.real(ifft(A * A))[:2 * n - 1]

def topk_gradient(a, c, k, N):
    """Compute gradient targeting top-k peaks. Returns (grad, c_max, weights)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    c_max = c[top_idx[0]]
    top_vals = c[top_idx]
    # Softmax weights emphasizing the highest peaks
    exp_v = np.exp(-(top_vals - c_max) * 2.0)
    weights = exp_v / exp_v.sum()
    
    fsize = 1 << (2 * n).bit_length()
    w_full = np.zeros(fsize)
    for i, k_idx in enumerate(top_idx):
        w_full[int(k_idx) % fsize] = weights[i]
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c_max, top_idx, weights


def backtracking_ls(a, direction, c_max, S, N, step, max_iter=20):
    """Backtracking line search with Armijo condition."""
    n = len(a)
    grad_dir = np.dot(direction, direction)
    if grad_dir < 1e-30:
        return a, c_max
    alpha = step
    for _ in range(max_iter):
        a_new = project_simplex(a + alpha * direction, S)
        c_new, _ = compute_conv(a_new, N)
        cm_new = c_new.max()
        if cm_new <= c_max + 1e-12:
            return a_new, cm_new
        alpha *= 0.5
    return a_new, cm_new

                continue
            beta_i = np.dot(s_i, y_i) / yy
            r = r + (alphas[i] - beta_i) * y_i
        direction = -r
        if np.dot(g, direction) > 0:
            direction = -g
        return direction
    
    g, c_max, c = lse_and_grad(a, beta)
    best_cmax = c_max
    step_scale = 3.0 * S / (n * max(beta, 1.0))
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.05)
            step_scale = 3.0 * S / (n * max(beta, 1.0))
            g, c_max, c = lse_and_grad(a, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
            if it % 500 == 0:
                print("    [lbfgs] it=%d beta=%.0f cmax=%.6f" % (it, beta, best_cmax))
        
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, _, _ = lse_and_grad(a_new, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_new
        c = None
        
        if it % 800 == 400 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, c_max, c = lse_and_grad(a, beta)
    
    return best_a, best_cmax


def spectral_block(a_init, n, S, N, max_time, rng, top_k=10, alpha=0.15):
    """Spectral shaping: reduce peak power spectrum contributions."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
        
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
        
        excess = max(0.0, cur - S * S / (2 * n)) / max(cur, 1e-10)
        alpha_curr = alpha * (0.3 + 0.7 * excess)
        
        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        # Maintain DC
        P_new[0] = P[0]
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        # Enforce conjugate symmetry
        A_new[N - np.arange(1, N // 2 + 1)] = np.conj(A_new[np.arange(1, N // 2 + 1)])
        if N % 2 == 0:
            A_new[N // 2] = np.real(A_new[N // 2])
        
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(np.abs(a_new_full) + 1e-6, S)
        
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax


def fourier_flat_init(n, N, S, rng):
    """Initialize with approximately flat power spectrum."""
    M = np.ones(N) * (S / np.sqrt(N))
    M[0] = S
    # Conjugate symmetric
    phases = rng.uniform(0, 2 * np.pi, N)
    phases[0] = 0.0
    if N % 2 == 0:
        phases[N // 2] = 0.0
    for i in range(1, N // 2 + (1 if N % 2 == 0 else 0)):
        phases[N - i] = -phases[i]
    A = M * np.exp(1j * phases)
    a_full = np.real(ifft(A))
    a = np.abs(a_full[:n]) + 0.05
    return project_simplex(a, S)


def chirp_init(n, S, rng):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * (3.0 * t + 0.5 * t * t / n)
    a = np.abs(np.cos(phase)) + 0.1 * rng.random(n) + 0.05
    return project_simplex(a, S)


def multi_freq_init(n, S, rng):
    """Multi-frequency initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 11, 17, 29]:
        a += 0.3 * np.sin(2 * np.pi * freq * t + rng.random() * 2 * np.pi)
    a = np.abs(a) + 0.05
    return project_simplex(a, S)


def combined_sequence(a1, a2, S, n, rng, mix=0.3):
    """Combine two sequences element-wise (geometric mean style)."""
    # Geometric mean preserves non-negativity and tends to smooth
    a1 = np.maximum(a1, 1e-10)
    a2 = np.maximum(a2, 1e-10)
    a_comb = (a1 * (1 - mix) + a2 * mix)
    return project_simplex(a_comb, S)


def safe_normalize(a, n, S):
    """Clamp and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses combined momentum top-k gradient, L-BFGS, and spectral shaping.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    # Primary length
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    print("Primary: n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    global_n = n
    
    def update_best(a_candidate, n_cand, label=""):
        nonlocal global_best, global_a, global_n
        a_cand = safe_normalize(a_candidate, n_cand, np.sqrt(2.0 * float(n_cand)))
        N_cand = 1 << (2 * n_cand - 1).bit_length()
        c, _ = compute_conv(a_cand, N_cand)
        sc = c.max()
        obj = 2.0 * n_cand * sc / (a_cand.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a_cand.copy()
            global_n = n_cand
            if label:
                print("  [%s] obj=%.6f (n=%d)" % (label, global_best, n_cand))
        return obj
    
    # --- Load previous solution as starting point ---
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.all(given >= 0):
            a_start = safe_normalize(given, n, S)
            c_check, _ = compute_conv(a_start, N)
            obj_check = 2.0 * n * c_check.max() / (a_start.sum() ** 2)
            print("  Previous solution loaded: obj=%.6f" % obj_check)
            update_best(a_start, n, "previous")
    except Exception as e:
        print("  Could not load previous: %s" % str(e))
    
    if a_start is None:
        a_start = fourier_flat_init(n, N, S, rng)
    
    # === Phase 1: Momentum top-k gradient (primary) ===
    print("=== Phase 1: Momentum top-k gradient ===")
    t1 = time.monotonic()
    phase1_budget = min(500.0, (deadline - t1) * 0.22)
    a_p1, sc_p1 = momentum_toppk_block(a_start, n, S, N, phase1_budget, rng, top_k=30)
    update_best(a_p1, n, "P1_mom")
    
    # === Phase 2: L-BFGS with high beta ===
    print("=== Phase 2: L-BFGS polishing ===")
    t2 = time.monotonic()
    phase2_budget = min(700.0, (deadline - t2) * 0.30)
    
    a_curr = global_a.copy() if global_a is not None else a_p1
    stall = 0
    last_best = global_best
    
    while time.monotonic() < t2 + phase2_budget:
        block_time = min(150.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 5:
            break
        beta_start = 30.0 + 10.0 * stall
        beta_end = min(800.0, 100.0 * (1 + stall))
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end, m_lbfgs=15)
        obj_new = 2.0 * n * sc_new / S ** 2
        if obj_new < global_best - 1e-8:
            update_best(a_new, n, "P2_lbfgs")
            a_curr = a_new.copy()
            stall = 0
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                a_curr = project_simplex(a_curr + rng.normal(0, 0.006 * S / np.sqrt(n), n), S)
                stall = 0
    
    # === Phase 3: Spectral shaping ===
    print("=== Phase 3: Spectral shaping ===")
    t3 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t3) * 0.15)
    
    a_sp, sc_sp = spectral_block(global_a, n, S, N, phase3_budget, rng, top_k=8, alpha=0.12)
    update_best(a_sp, n, "P3_spec")
    
    a_sp2, sc_sp2 = spectral_block(global_a, n, S, N, min(150.0, max(0, (deadline - time.monotonic()) * 0.1)),
                                    rng, top_k=5, alpha=0.06)
    update_best(a_sp2, n, "P3_spec_fine")
    
    # === Phase 4: Try n=16384 ===
    remaining = deadline - time.monotonic()
    if remaining > 300:
        print("=== Phase 4: n=16384 exploration ===")
        n2 = 16384
        S2 = np.sqrt(2.0 * float(n2))
        N2 = 1 << (2 * n2 - 1).bit_length()
        print("  n2=%d, N2=%d, remaining=%.0fs" % (n2, N2, remaining))
        
        # Initialize from upsampling the best n=8192 solution
        a_best_up = np.interp(np.linspace(0, 1, n2, endpoint=False),
                              np.linspace(0, 1, n, endpoint=False),
                              global_a)
        a_init2 = project_simplex(a_best_up, S2)
        
        t4 = time.monotonic()
        phase4_budget = min(350.0, remaining - 100)
        
        a_p4, sc_p4 = momentum_toppk_block(a_init2, n2, S2, N2, min(phase4_budget * 0.5, 150), rng, top_k=40)
        update_best(a_p4, n2, "P4_mom_16k")
        
        rem2 = deadline - time.monotonic() - 50
        if rem2 > 50:
            a_p4b, sc_p4b = lbfgs_lse_block(a_p4, n2, S2, N2, min(rem2, 200), rng,
                                             beta_init=30.0, beta_max=300.0, m_lbfgs=12)
            update_best(a_p4b, n2, "P4_lbfgs_16k")
    
    # === Phase 5: Final intensive polish on best sequence ===
    print("=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        # Run momentum again on the best
        rem_polish = min(remaining - 10, 200.0)
        a_fin, sc_fin = momentum_toppk_block(global_a, global_n, np.sqrt(2.0 * global_n),
                                              1 << (2 * global_n - 1).bit_length(),
                                              rem_polish, rng, top_k=20)
        update_best(a_fin, global_n, "P5_mom")
        
        # L-BFGS final
        rem2 = deadline - time.monotonic() - 5
        if rem2 > 20:
            a_fin2, sc_fin2 = lbfgs_lse_block(global_a, global_n, np.sqrt(2.0 * global_n),
                                               1 << (2 * global_n - 1).bit_length(),
                                               min(rem2, 150), rng,
                                               beta_init=100.0, beta_max=1000.0, m_lbfgs=15)
            update_best(a_fin2, global_n, "P5_lbfgs")
    
    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("  Verification failed (inf), using fallback")
            global_a = np.ones(256, dtype=np.float64)
            global_n = 256
    else:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256
    
    print("Returning length %d, best_obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
