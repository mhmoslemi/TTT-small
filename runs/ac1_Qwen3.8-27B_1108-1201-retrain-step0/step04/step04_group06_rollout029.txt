import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    else:
        a = np.full_like(a, S / len(a))
    return a

def compute_smooth_max_grad(b, T, M):
    """Compute soft-max weights and gradient contribution."""
    # b has length 2n-1
    # w_k = exp(T*b[k]/M) / sum exp(T*b[l]/M)
    scaled = T * b / M
    scaled -= scaled.max()  # for numerical stability
    w = np.exp(scaled)
    w /= w.sum()
    return w

def gradient_soft_max(a, b, w, n):
    """
    Compute gradient of soft-max objective w.r.t. a.
    grad_j = sum_k w[k] * (2*a[k-j] - a[j]*[k==2j]) * [0<=k-j<n]
    
    We can write this as:
    grad_j = 2 * sum_k w[k] * a[k-j] (for valid k-j) - a[j] * sum_k w[k] * [k==2j]
    
    The second term: sum_k w[k] * [k==2j] = w[2j] (if 2j < 2n-1)
    The first term: 2 * sum_m w[j+m] * a[m] (for m = 0..n-1, j+m < 2n-1)
    """
    # First term: cross-correlation

    # This is the "cross-correlation" of a and w

    # where w_padded has w placed at positions 0..2n-2

    A = fft(a, N)
    W = fft(w, N)  # w is already length 2n-1, padded by fft to N

    # Let me verify: ifft(conj(A)*W)[j] = (1/N) sum_k conj(A[k]) W[k] exp(2pi i jk/N)
    # = (1/N) sum_k (sum_m a[m] exp(-2pi i mk/N)) (sum_l w[l] exp(2pi i lk/N)) exp(2pi i jk/N)
    # = sum_m a[m] (1/N) sum_l w[l] exp(2pi i (l + j - m)k/N)
    # = sum_m a[m] w[(j-m) mod N]... 
    
    # Hmm, this is getting confusing. Let me use a different approach.
    # 
    # C[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # This is the cross-correlation of a and w, which equals:
    # C = ifft(conj(fft(a, N)) * fft(w, N)) ... need to be careful
    #
    # Actually, the standard result is:
    # (x * y)[n] = sum_k x[k] y[n-k] (convolution) <-> X(f) Y(f)
    # <x, y>[n] = sum_k x[k] y[k+n] (cross-correlation) <-> conj(X(f)) Y(f)

"""
Minimize 2*n*max_conv/sum^2 using soft-max gradient descent with adaptive temperature
annealing, FFT-based gradient computation, Nesterov momentum, diverse U-shaped
initializations, and coordinate descent polish.

Key differences from previous approach:
1. Direct time-domain soft-max gradient (not spectral shaping)
2. Adaptive temperature annealing from smooth to sharp
3. FFT-accelerated gradient via cross-correlation
4. Diverse structured initializations (U-shapes, two-peaked, anti-triangular)
5. Simulated annealing acceptance for escaping local minima
6. Efficient coordinate descent with exact delta computation
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-30:
        a = a * (S / s)
    else:
        a = np.full_like(a, S / len(a))
    return a


def compute_soft_max_weights(b, T, M):
    """Compute softmax weights w_k = exp(T*b_k/M) / sum(exp(T*b_l/M))."""
    scaled = T * b / M
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


def gradient_soft_max(y, w, n, N_corr):
    """
    Compute gradient of soft-max(b, T, M) w.r.t. y.
    grad_j = sum_k w[k] * (2*y[k-j] - y[j]*[k==2j]) * [0<=k-j<n]
           = 2 * C[j] - y[j] * w[2j]
    where C[j] = sum_m y[m] * w[j+m] (cross-correlation)
    """
    A = fft(y, N_corr)
    W = fft(w, N_corr)
    C = np.real(ifft(np.conj(A) * W))[:n]
    two_j = 2 * np.arange(n)
    diag = np.zeros(n)
    valid = two_j < 2 * n - 1
    diag[valid] = w[two_j[valid]]
    grad = 2.0 * C - y * diag
    return grad


def subgradient_top_k(b, y, n, top_k):
    """Compute subgradient of max(b) using top-k lags."""
    top_indices = np.argsort(b)[::-1][:top_k]
    top_vals = b[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    N = 1 << (3 * n - 2).bit_length()
    A = fft(y, N)
    W = fft(w_full, N)
    C = np.real(ifft(np.conj(A) * W))[:n]
    two_j = 2 * np.arange(n)
    diag = np.zeros(n)
    valid = two_j < 2 * n - 1
    diag[valid] = w_full[two_j[valid]]
    grad = 2.0 * C - y * diag
    return grad


def coordinate_descent(a, S, max_time, rng, n_top=30, n_rounds=200, multipliers=None):
    """
    Coordinate descent: adjust top-contributing coordinates at the peak lag.
    Uses exact delta computation for efficiency.
    """
    n = len(a)
    a = a.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    if multipliers is None:
        multipliers = [0.3, 0.5, 0.65, 0.8, 1.2, 1.4, 1.6]

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break

        b = fft_convolve_self(a)
        b_max = b.max()
        peak_lag = int(np.argmax(b))

        if b_max < best_score:
            best_score = b_max
            best_a = a.copy()

        # Find contributing range at peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)

        if high < low:
            break

        # Compute contributions
        contribs = np.zeros(n)
        idx_range = range(low, high + 1)
        for i in idx_range:
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]
        improved = False

        for idx in top_coords:
            i0 = int(idx)
            if time.monotonic() - start > max_time:
                break
            j0 = peak_lag - i0
            if j0 < 0 or j0 >= n:
                continue

            orig = a[i0]
            if orig < 1e-15:
                continue

            best_frac = 1.0
            best_bmax = b_max

            # Precompute affected range
            k_lo = i0
            k_hi = min(2 * n - 2, i0 + n - 1)

            for frac in multipliers:
                new_val = orig * frac
                delta = new_val - orig
                if abs(delta) < 1e-15:
                    continue

                # Compute new max efficiently
                # Only lags in [k_lo, k_hi] are affected
                new_bmax = b_max
                affected_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    m = k - i0
                    if 0 <= m < n:
                        # delta_b[k] = (frac-1)*orig*a[m] for the first-factor term
                        # plus (frac-1)*orig*a[k-i0] for second-factor term (same if k!=2*i0)
                        # Actually: delta_b[k] = 2*(frac-1)*orig*a[k-i0] if k != 2*i0
                        #            delta_b[k] = (frac^2-1)*orig^2 if k == 2*i0
                        if k == 2 * i0:
                            db = (frac * frac - 1.0) * orig * orig
                        else:
                            db = 2.0 * (frac - 1.0) * orig * a[m]
                        new_val_k = b[k] + db
                        if new_val_k > affected_max:
                            affected_max = new_val_k

                # Check if unaffected lags have a larger max
                if k_lo > 0:
                    unaffected_max = b[:k_lo].max()
                else:
                    unaffected_max = 0.0
                if k_hi + 1 < len(b):
                    rest_max = b[k_hi + 1:].max()
                    unaffected_max = max(unaffected_max, rest_max)

                candidate_max = max(affected_max, unaffected_max)
                if candidate_max < best_bmax:
                    best_bmax = candidate_max
                    best_frac = frac

            a[i0] = orig * best_frac
            if best_frac != 1.0:
                improved = True

        # Renormalize
        a = project_nonneg_sum(a, S)

        # If no improvement, try random perturbation on top coords
        if not improved:
            top5 = top_coords[:min(5, len(top_coords))]
            for idx in top5:
                a[int(idx)] *= rng.uniform(0.5, 1.5)
            a = project_nonneg_sum(a, S)

    return best_a, best_score


def u_shape_init(n, delta, rng):
    """Create U-shaped initialization: high at ends, low in middle."""
    t = np.linspace(0, 1, n)
    a = 1.0 + delta * np.cos(np.pi * t)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a


def two_peak_init(n, rng, pos1=0.25, pos2=0.75, width=0.15):
    """Two-peaked initialization."""
    t = np.linspace(0, 1, n)
    a = 0.3 + 1.0 * np.exp(-0.5 * ((t - pos1) / width) ** 2) \
        + 1.0 * np.exp(-0.5 * ((t - pos2) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a


def alternating_init(n, ratio, rng):
    """Alternating high/low pattern."""
    a = np.where(np.arange(n) % 2 == 0, 1.0, ratio)
    a += 0.03 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a

def spectral_rebalance(a, n, S, rng, alpha=0.3):
    """
    Rebalance the spectrum: reduce peaks in A(f)^2 by redistributing energy.
    This is a spectral-domain step that helps escape local minima.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    B = A * A  # This is the spectrum of the convolution

    # Find the "peaked" frequencies in B (where |B| is large)
    B_mag = np.abs(B)
    B_mean = B_mag.mean()

    # Reduce high-magnitude components, boost low-magnitude ones
    B_mag_new = B_mag * (1.0 - alpha * 0.5) + alpha * 0.5 * B_mean
    # Preserve the phase
    B_phase = B / np.where(np.abs(B) > 1e-30, np.abs(B), 1.0)
    B_new = B_mag_new * B_phase

    # Recover A from B: A = sqrt(B) (need to handle phases carefully)
    # A_new^2 = B_new => A_new = sqrt(|B_new|) * exp(i*angle(B_new)/2)
    A_mag_new = np.sqrt(np.abs(B_new))
    A_phase_new = np.angle(B_new) / 2.0
    A_new = A_mag_new * np.exp(1j * A_phase_new)

    # Ensure A[0] is real and positive (DC component = sum)
    A_new[0] = np.abs(A_new[0])

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_nonneg_sum(a_new, S)
    return a_new

    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    internal_score = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / (global_a.sum() ** 2)
    print("Final: length=%d, internal_score=%.6f" % (len(global_a), internal_score))
    return [float(x) for x in global_a]
