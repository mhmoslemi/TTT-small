from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_step(y, grad, eta, S):
    """Mirror descent step with entropic regularization."""
    # a_i = S * y_i * exp(-eta*grad_i) / sum_j y_j * exp(-eta*grad_j)
    log_weights = np.log(np.maximum(y, 1e-300)) - eta * grad
    # Shift for numerical stability
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    total = weights.sum()
    if total < 1e-300:
        return np.ones(len(y)) * S / len(y)
    return S * weights / total


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_weights(c, beta):
    """Compute LSE soft-max weights."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.full(len(c), 1.0 / len(c)), cm
    return e / tot, cm


def lse_gradient(a, w, fsize):
    """Gradient of LSE: grad_i = 2 * sum_k w_k * a[k-i] (convolution)."""
    n = len(a)
    # grad = 2 * convolve(w, a) truncated to n
    # Using FFT: IFFT(FFT(w) * FFT(a)) gives circular convolution
    W = fft(w, fsize)
    A = fft(a, fsize)
    # The result is the circular convolution of w and a
    # We need the linear convolution truncated to first n elements
    conv = np.real(ifft(W * A))
    # The linear convolution of w (length 2n-1) and a (length n) has length 3n-2
    # But we computed circular conv of length fsize
    # The first n elements of the linear conv are what we want
    grad = 2.0 * conv[:n]
    return grad


def run_fista_v2(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0,
                 beta_ramp=2000, use_mirror=False):
    """Improved FISTA with adaptive LSE and optional mirror descent."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    patience = 5000

    while time.monotonic() - start < max_time:
        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print("    FISTA iter %d, best=%.6f, elapsed=%.0f/%.0fs" % (it, best, elapsed, max_time))

        # Adaptive beta
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if it - last_improve > patience:
            break

        # Compute LSE weights and gradient
        w, _ = lse_weights(c, beta)
        grad = lse_gradient(y, w, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (2.0 * beta)
        # Adaptive: if no recent improvement, reduce step
        if it - last_improve > patience // 2:
            step *= 0.7

        if use_mirror:
            x_new = mirror_step(y, grad, step, S)
        else:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize to maintain sum constraint approximately
        s = y.sum()
        if s > 1e-15:
            y = y * (S / s)
        a = x_new
        t = t_new

        # Periodic noise injection
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def targeted_coord_descent(a_init, n, S, max_time, rng, top_k=20):
    """Coordinate descent focused on coordinates affecting the peak lags."""
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best = c.max()
    best_a = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    # Precompute top-k lags
    top_lags = np.argsort(c)[::-1][:top_k]

    while time.monotonic() - start < max_time:
        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - start
            print("    CD iter %d, best=%.6f, elapsed=%.0f/%.0fs" % (it, best, elapsed, max_time))

        # Recompute top lags periodically
        if it % 10 == 0:
            c = fft_convolve_self(a)
            top_lags = np.argsort(c)[::-1][:top_k]
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        improved = False
        n_pairs = min(200 + it * 2, 1000)

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose a pair of coordinates
            if rng.random() < 0.4:
                # Choose based on peak lag influence
                k = top_lags[rng.integers(0, len(top_lags))]
                # Coordinates that affect b_k: a_i and a[k-i] for valid i
                valid_i = np.arange(max(0, k - n + 1), min(k + 1, n))
                if len(valid_i) == 0:
                    continue
                # Pick two from the influential coordinates
                i1 = valid_i[rng.integers(0, len(valid_i))]
                i2 = valid_i[rng.integers(0, len(valid_i))]
                j, m = int(i1), int(i2)
            else:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute the effect of transferring delta from m to j
            # b_k changes by delta * (a[k-j] + a[j-k] - a[k-m] - a[m-k]) for each k
            # For efficiency, only check the top lags
            d = np.zeros(len(top_lags))
            for idx, k in enumerate(top_lags):
                # Effect on b_k from changing a[j] by +delta and a[m] by -delta
                # b_k = sum_i a_i * a[k-i]
                # db_k = delta * (a[k-j] if valid + a[j-k] if valid) 
                #        - delta * (a[k-m] if valid + a[m-k] if valid)
                val_j = 0.0
                kj = k - j
                jk = j - k
                if 0 <= kj < n:
                    val_j += a[kj]
                if 0 <= jk < n:
                    val_j += a[jk]
                val_m = 0.0
                km = k - m
                mk = m - k
                if 0 <= km < n:
                    val_m += a[km]
                if 0 <= mk < n:
                    val_m += a[mk]
                d[idx] = val_j - val_m

            # Also estimate effect on non-top lags (approximate)
            # For now, just check top lags

            # Find delta that minimizes max(b_k + delta * d_k) for k in top_lags
            # subject to -a[j] <= delta <= a[m]
            lo = -a[j]
            hi = a[m]

            if hi - lo < 1e-14:
                continue

            # Binary search for the optimal delta
            # f(delta) = max_k (c[top_lags[k]] + delta * d[k])
            # This is convex in delta (max of linear functions)
            # Use ternary search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = lo, hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = np.max(c[top_lags] + c_g * d)
            fd = np.max(c[top_lags] + d_g * d)

            for _ in range(50):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = np.max(c[top_lags] + c_g * d)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = np.max(c[top_lags] + d_g * d)

            delta_star = (a_g + b_g) / 2.0
            delta_star = max(lo, min(hi, delta_star))

            if abs(delta_star) < 1e-14:
                continue

            # Apply the update
            a[j] += delta_star
            a[m] -= delta_star

            # Verify: check if this actually improves (using top lags as proxy)
            new_top_vals = c[top_lags] + delta_star * d
            if np.max(new_top_vals) < best - 1e-12:
                best = np.max(new_top_vals)
                best_a = a.copy()
                improved = True
                # Update c for top lags
                c[top_lags] = new_top_vals
            else:
                # Revert
                a[j] -= delta_star
                a[m] += delta_star

        # Full recompute periodically
        if it % 5 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            top_lags = np.argsort(c)[::-1][:top_k]

        if not improved and it > 10:
            # Add small noise and restart
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def spectral_project(a_init, n, S, max_time, rng, n_iter_max=5000):
    """Spectral domain optimization: flatten |FFT|^2 subject to a>=0, sum(a)=S."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time and it < n_iter_max:
        it += 1
        if it % 500 == 0:
            elapsed = time.monotonic() - start
            print("    Spectral iter %d, best=%.6f, elapsed=%.0f/%.0fs" % (it, best, elapsed, max_time))

        # Zero-pad and FFT
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = np.angle(F)
        P = mag ** 2

        # Target: flatten P (make it as uniform as possible)
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Gradual flattening
        alpha = min(0.9, 0.1 + 0.8 * it / n_iter_max)
        P_new = (1 - alpha) * P + alpha * P_mean

        # Reconstruct
        mag_new = np.sqrt(P_new)
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(F_new))[:n]

        # Enforce non-negativity and sum constraint
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def sobol_like_init(n, rng):
    """Generate a diverse set of initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) * 0.1)

    # Gaussian
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2))

    # Sinc-like
    inits.append(np.maximum(np.sinc(2 * t - 1) ** 2, 0.01))

    # Cosine window
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))

    # Raised cosine
    inits.append(0.5 * (1 + np.cos(np.pi * (2 * t - 1))))

    # Power law
    inits.append((1.0 - t) ** 0.5 + 0.01)

    # Random
    inits.append(rng.random(n) + 0.05)

    # Bimodal
    inits.append(np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2))

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))

    # Chirp-like (real part, made non-negative)
    inits.append(np.maximum(np.abs(np.sin(2 * np.pi * (1 + 4 * t) * t)), 0.01))

    return inits


def upsample(a, new_n):
    """Upsample a sequence to a new length using interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, new_n):
    """Downsample a sequence to a new length using block averaging."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    nblocks = old_n // new_n
    if nblocks < 1:
        return a[:new_n].copy()
    result = a[:nblocks * new_n].reshape(new_n, nblocks).mean(axis=1)
    return result

    else:

        return 2.0 * len(a) * c.max() / (s * s)
