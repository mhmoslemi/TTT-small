```python
import numpy as np
from scipy import sparse
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def build_toeplitz_matrix(a):
    """Build the (2n-1) x n Toeplitz matrix T where T[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j: T[k, j] = a[k-j] for k = j, j+1, ..., j+n-1 (capped at 2n-2)
        k_start = j
        k_end = min(j + n, 2*n - 1)
        k_range = np.arange(k_start, k_end)
        a_idx = k_range - j  # = 0, 1, ..., k_end-1-j
        rows.extend(k_range.tolist())
        cols.extend([j] * len(k_range))
        vals.extend(a[a_idx].tolist())
    T = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T
```

```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using:
1. FISTA with adaptive LSE smoothing for fast bulk optimization
2. LP-based direction finding for high-quality steps
3. Coordinate descent and subgradient polishing
4. Multi-length exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy import sparse
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - S) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / n
    else:
        theta = (cssv[rho[-1]] - S) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def lse_smooth_max(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = scaled * 500.0 / scaled.max()
    # Use log-sum-exp trick
    m = scaled.max()
    return cm + (np.log(np.exp(scaled - m).sum()) + m) / beta


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = scaled * 500.0 / scaled.max()
    m = scaled.max()
    e = np.exp(scaled - m)
    w = e / e.sum()
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_lse(a_init, S, max_time, rng, beta_start=20.0, beta_end=5000.0):
    """FISTA with increasing LSE beta."""
    a = np.clip(a_init, 0, 1000)
    a = a * (S / a.sum())
    n = len(a)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal beta
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        if c.max() < best_max:
            best_max = c.max()
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: for LSE with parameter beta, the gradient is 2*beta-Lipschitz
        # (approximately), so step = 1 / (2*beta * lip_const)
        # The convolution operator has norm <= ||a||_1 = S (approximately)
        # Actually the Lipschitz constant of LSE_max(conv(a,a)) is about 2*beta*S
        step = 1.0 / (4.0 * beta * S * 0.1)  # conservative
        step = min(step, 1.0 / (beta * 0.5))
        
        x = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
        
        # Periodic restart with noise
        if it % 2000 == 0 and progress < 0.9:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_a, best_max


def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. T_a @ g <= M*1, g >= 0."""
    n = len(a)
    
    # Build sparse Toeplitz matrix T_a: (2n-1) x n
    # T_a[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # More precisely: (convolve(a, g))[k] = sum_j a[k-j] * g[j]
    # = sum_j T_a[k,j] * g[j]
    
    # For efficiency, build using offsets
    # Column j has nonzeros at rows j, j+1, ..., j+n-1 (capped at 2n-2)
    # with values a[0], a[1], ..., a[n-1]
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        length = k_end - k_start
        if length > 0:
            rows.append(np.arange(k_start, k_end))
            cols.append(np.full(length, j))
            vals.append(a[:length])
    
    if rows:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
    
    T = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: maximize sum(g) subject to T@g <= M*1, g >= 0
    # In linprog form: minimize -sum(g) subject to T@g <= M*1, g >= 0
    c_obj = -np.ones(n)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None


def lp_based_update(a, S, max_lp_time=60.0):
    """Use LP to find a good direction, then take a step."""
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    start = time.monotonic()
    g = solve_lp_direction(a, M)
    
    if g is None:
        return a, M
    
    # Scale g to have sum S
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a, M
    
    g = g * (S / g_sum)
    
    # Line search: find best t in [0, 1]
    best_obj = np.inf
    best_a = a.copy()
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1 - t) * a + t * g
        a_try = np.clip(a_try, 0, 1000)
        a_try = a_try * (S / a_try.sum())
        c_try = fft_convolve_self(a_try)
        obj = c_try.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a, best_obj


def coordinate_descent(a, S, max_time, rng):
    """Coordinate descent: update one coordinate at a time."""
    n = len(a)
    a = np.clip(a, 0, 1000)
    a = a * (S / a.sum())
    start = time.monotonic()
    best_max = fft_convolve_self(a).max()
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Random coordinate
        j = rng.integers(n)
        
        # Try a few values for a[j]
        current = a[j]
        # The contribution of a[j] to convolution:
        # (a*a)(k) gets a[j] * a[k-j] added for each valid k-j
        # To minimize max, we want to reduce a[j] if it contributes to peaks
        
        # Compute the "sensitivity": how much does each lag change when a[j] changes?
        # d(a*a)(k)/da[j] = a[k-j] + a[j-k] (symmetric)
        # (with a[i] = 0 for i out of range)
        
        # For a small increase da in a[j]:
        # (a+a_ej*da)*(a+a_ej*da)(k) ≈ (a*a)(k) + da*(a[k-j] + a[j-k]) + da^2 * delta(k)
        # where delta(k) = 1 if k = 2j - 2j = 0... wait
        
        # Actually (a + da*e_j) * (a + da*e_j)(k)
        # = sum_i (a_i + da*δ(i,j)) * (a_{k-i} + da*δ(k-i,j))
        # = (a*a)(k) + da * (a_{k-j} + a_{j-k}) + da^2 * δ(k-2j, 0)... 
        # Hmm, let me be more careful.
        # = sum_i a_i * a_{k-i} + da * sum_i δ(i,j) * a_{k-i} + da * sum_i a_i * δ(k-i,j) + da^2 * δ(j,j)δ(k-j,j)
        # = (a*a)(k) + da * a_{k-j} + da * a_{k-j} + da^2 * δ(k-2j+... 
        # Wait: sum_i δ(i,j) * a_{k-i} = a_{k-j}
        # sum_i a_i * δ(k-i,j) = a_{k-j}
        # da^2 * δ(i,j)*δ(k-i,j) = da^2 * δ(j,j)*δ(k-j,j) = da^2 * δ(k, 2j)
        
        # So the change is: da * (a_{k-j} + a_{k-j}) + da^2 * δ(k,2j) = 2*da*a_{k-j} + da^2*δ(k,2j)
        # (where a[i] = 0 for i out of [0, n-1])
        
        # To find the best da, we want to minimize max_k [(a*a)(k) + 2*da*a[k-j] + da^2*δ(k,2j)]
        # subject to a[j] + da >= 0 and sum stays S (so we need to adjust other coords)
        
        # This is getting complicated for exact coordinate descent with sum constraint.
        # Let's use a simpler approach: just try a few values and keep the best.
        
        candidates = [max(0, current * 0.8), max(0, current * 0.9), current, 
                      current * 1.1, current * 1.2]
        best_val = np.inf
        best_candidate = current
        
        for da_val in candidates:
            a_try = a.copy()
            diff = da_val - current
            a_try[j] = da_val
            # Adjust one other coordinate to maintain sum (random choice)
            if abs(diff) > 1e-15:
                other = rng.integers(n)
                if other == j:
                    other = (j + 1) % n
                a_try[other] = max(0, a_try[other] - diff)
                if a_try[other] < 0:
                    continue
                # Renormalize
                a_try = a_try * (S / a_try.sum())
            
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_candidate = da_val
                best_a_try = a_try.copy()
        
        if best_val < best_max:
            best_max = best_val
            best_a = best_a_try.copy()
            a = best_a_try.copy()
            c = fft_convolve_self(a)
    
    return best_a, best_max
```

```python
import numpy as np
```

```python
def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    valid = u - (cssv - S) / arange > 0
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
        theta = (cssv[rho - 1] - S) / rho
    else:
        theta = cssv[-1] / n
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution with parameter beta."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = scaled * (500.0 / max(scaled.max(), 1e-30))
    m = scaled.max()
    e = np.exp(scaled - m)
    w = e / e.sum()
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k=32):
    """Exact subgradient from top-k lag peaks, weighted by excess above median."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    med = np.median(c[idx])
    weights = np.maximum(c[idx] - med, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def build_convolution_matrix(a):
    """Build sparse (2n-1) x n matrix T where (T@g)(k) = (a*g)(k)."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []

    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        length = k_end - k_start
        if length > 0:
            rows_list.append(np.arange(k_start, k_end, dtype=np.int32))
            cols_list.append(np.full(length, j, dtype=np.int32))
            vals_list.append(a[:length])

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)

    return sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(a, S, top_k_constraints=None, time_limit=30.0):
    """
    LP-based best response: find g maximizing sum(g) subject to
    (a*g)(k) <= M for top-k constraints (or all), g >= 0.
    Returns (g_normalized, success).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()

    # Optionally restrict to top-k constraints for speed
    if top_k_constraints is not None and top_k_constraints < 2 * n - 1:
        k = min(top_k_constraints, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        # Build reduced constraint matrix
        T_full = build_convolution_matrix(a)
        T_reduced = T_full[top_idx, :]
        b_reduced = np.full(k, M)
    else:
        T_reduced = build_convolution_matrix(a)
        b_reduced = np.full(2 * n - 1, M)

    # LP: max sum(g) s.t. T_reduced @ g <= b_reduced, g >= 0
    # linprog minimizes, so we negate: min -sum(g)
    c_obj = -np.ones(n)

    start = time.monotonic()
    try:
        res = linprog(c_obj,
                      A_ub=T_reduced,
                      b_ub=b_reduced,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception as e:
        print("  LP failed: %s" % str(e))
        return a, False

    if res.success and res.x is not None:
        g = np.clip(res.x, 0, 1000)
        g_sum = g.sum()
        if g_sum > 1e-10:
            g = g * (S / g_sum)
            return g, True
    return a, False


def lp_based_iteration(a, S, top_k=None, time_limit=30.0):
    """
    One LP-based iteration: solve LP, then line search for best step.
    Returns (new_a, new_max_b, improved).

            best_val = c_try.max()
            best_a = a_try.copy()

    improved = best_val < current_max - 1e-10
    return best_a, best_val, improved

def objective_value(a):
    """Compute objective: 2*n*max_b / sum^2."""

    n = len(a)
    s = a.sum()
    if s < 0.01:

    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def upscale_sequence(a_small, n_large):
    """Upscale by linear interpolation."""
    t_old = np.linspace(0, 1, len(a_small))
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)

    ver = evaluate_sequence(final_list)
    print("\nFinal verification: evaluate_sequence = %.6f" % ver)
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_best_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_best_a]

    print("Returning sequence of length %d, objective=%.6f" %
          (len(final_list), global_best_obj))
    return final_list
```
