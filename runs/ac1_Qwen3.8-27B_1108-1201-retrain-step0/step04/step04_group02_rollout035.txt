    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def run_optimization(a_init, n, S, max_time, rng, verbose=True):

    a = project_simplex(np.clip(a_init, 0, 1000.0), S)
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()
    
    # LSE parameters
    beta = 5.0
    beta_growth = 1.003
    beta_max = 500.0
    
    # Step size (BB)
    step = 1.0 / (2.0 * np.max(a) + 1e-10)
    
    prev_a = None
    prev_grad = None
    
    it = 0
    last_improve_it = 0
    stuck_count = 0
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute gradient
        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # BB step size
        if prev_a is not None and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            dot_sy = np.dot(s_vec, y_vec)
            dot_yy = np.dot(y_vec, y_vec)
            if dot_yy > 1e-20 and 0 < dot_sy < 1e10:
                bb_step = dot_sy / dot_yy
                # Clip BB step
                bb_step = np.clip(bb_step, 1e-8, 1.0 / (gnorm * 10))
                step = 0.5 * step + 0.5 * bb_step
        
        # Adaptive step based on gnorm
        max_allowed_step = S / (n * gnorm + 1e-10)
        step = min(step, max_allowed_step)
        step = max(step, 1e-10)

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

            # Accept
            prev_a = a.copy()
            prev_grad = grad.copy()
            a = a_new
            c = c_new
            cur_obj = obj_new
            step = min(step * 1.1, max_allowed_step * 2)

                best_obj = cur_obj
                best_a = a.copy()
                last_improve_it = it
                stuck_count = 0
        else:

            step *= 0.7
            stuck_count += 1

                A = A * (1 + noise)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE(c; beta) w.r.t. a. c is precomputed self-convolution."""
    n = len(a)
    c_max = c.max()
    exp_vals = np.exp(beta * (c - c_max))
    total = exp_vals.sum()
    if total < 1e-300:
        return np.zeros(n)
    p = exp_vals / total
    # Cross-correlation of p (len 2n-1) and a (len n): need pad to 3n-2
    L = 1 << (3 * n - 2).bit_length()
    P = fft(p, L)
    A = fft(a, L)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return grad


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def structured_perturb(a, n, S, rng):
    """Apply a structured perturbation to escape local minima."""
    strategy = rng.integers(0, 6)
    a_new = a.copy()
    t = np.arange(n) / n

    if strategy == 0:
        # Gaussian noise
        scale = 0.01 * S / np.sqrt(n)
        a_new = a + rng.normal(0, scale, n)
    elif strategy == 1:
        # Sinusoidal modulation
        freq = rng.uniform(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t + phase)
        a_new = a * mod
    elif strategy == 2:
        # Edge damping / boosting
        edge_mod = 1.0 + rng.uniform(-0.5, 0.5) * (1 - np.cos(np.pi * t))
        a_new = a * edge_mod
    elif strategy == 3:
        # Frequency domain perturbation
        L = len(a)
        A = fft(a, L)
        noise = 0.03 * (rng.normal(0, 1, L) + 1j * rng.normal(0, 1, L))
        A = A * (1.0 + noise)
        a_new = np.real(ifft(A))
    elif strategy == 4:
        # Block modulation
        nblocks = rng.integers(3, 12)
        block_mod = rng.uniform(0.5, 1.5, nblocks)
        mod = np.interp(t, np.linspace(0, 1, nblocks + 1), np.concatenate([[block_mod[0]], block_mod]))
        a_new = a * mod
    else:
        # Super-Gaussian reshaping
        p = rng.uniform(1.5, 5.0)
        base = np.sin(np.pi * t) ** 2
        a_new = base ** (1.0 / p) * (S / (base ** (1.0 / p)).sum() * n)

    a_new = np.maximum(a_new, 0.0)
    return project_simplex(a_new, S)


def run_lse_optimization(a_init, n, S, max_time, rng, verbose=True):
    """
    LSE-smoothed gradient descent with BB step size and adaptive beta.
    Minimizes max(a*a) subject to sum(a) = S, a >= 0.
    """
    a = project_simplex(np.clip(a_init, 0, 1000.0), S)
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()

    # LSE parameters
    beta = 8.0
    beta_min = 3.0
    beta_max = 600.0

    # Step size
    step = 0.5

    # BB state
    prev_a = None
    prev_grad = None

    # Momentum
    momentum = 0.0
    vel = np.zeros(n)

    it = 0
    stuck_count = 0
    last_improve_it = 0
    n_fails = 0
    fail_window = 0

    start = time.monotonic()

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start

        # Compute LSE gradient
        grad = compute_lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size update
        if prev_a is not None and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            dot_sy = np.dot(s_vec, y_vec)
            dot_yy = np.dot(y_vec, y_vec)
            if dot_yy > 1e-20 and 1e-6 < dot_sy < 1e6:
                bb_step = dot_sy / dot_yy
                bb_step = np.clip(bb_step, 1e-8, 2.0 / (gnorm + 1e-10))
                # Smooth BB step
                step = 0.3 * step + 0.7 * bb_step

        # Safety cap on step
        max_step = S / (n * gnorm * 0.5 + 1e-10)
        step = min(step, max_step)
        step = max(step, 1e-12)

        # Momentum update
        momentum = min(0.9, momentum + 1e-6)
        vel = momentum * vel + (1 - momentum) * step * grad

        # Take step
        a_new = project_simplex(a - vel, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < cur_obj - 1e-13:
            # Accept
            prev_a = a.copy()
            prev_grad = grad.copy()
            a = a_new
            c = c_new
            cur_obj = obj_new
            step = min(step * 1.05, max_step)
            fail_window = 0

            if cur_obj < best_obj - 1e-11:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve_it = it
                stuck_count = 0
        else:
            fail_window += 1
            step *= 0.75
            vel *= 0.5
            if fail_window > 30:
                stuck_count += 1
            else:
                stuck_count = 0

        # Adaptive beta based on peak structure
        c_sorted = np.sort(c)[::-1]
        top_gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
        rel_gap = top_gap / (c_sorted[0] + 1e-15)
        if rel_gap < 0.05:
            # Many similar peaks: increase beta for sharper focus
            beta = min(beta_max, beta * 1.01)
        elif rel_gap > 0.3:
            # Clear dominant peak: can use smoother beta
            beta = max(beta_min, beta * 0.998)
        else:
            beta = min(beta_max, beta * 1.00

3)

        # Periodic restart if stuck
        if stuck_count > 150:
            a = structured_perturb(a, n, S, rng)
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / (S * S)
            step = 0.5
            vel = np.zeros(n)
            momentum = 0.0
            stuck_count = 0
            fail_window = 0
            prev_a = None
            prev_grad = None
            beta = max(beta_min, beta * 0.5)

        # Log
        if verbose and it % 2000 == 0:
            print("    it=%d, obj=%.8f, best=%.8f, beta=%.1f, step=%.5f, e=%.0fs" % (
                it, cur_obj, best_obj, beta, step, elapsed))

    return best_a, best_obj


def run_hybrid_optimization(a_init, n, S, max_time, rng, verbose=True):
    """
    Hybrid optimizer: alternates between LSE gradient and top-k subgradient.
    Uses adaptive switching based on convergence behavior.
    """
    a = project_simplex(np.clip(a_init, 0, 1000.0), S)
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()

    beta = 10.0
    beta_max = 500.0
    step = 1.0
    prev_a = None
    prev_grad = None
    it = 0
    stuck = 0
    mode = "lse"  # or "topk"
    mode_count = 0

    start = time.monotonic()

    while time.monotonic() - start < max_time:
        it += 1

        # Choose gradient mode based on progress
        mode_count += 1
        if mode_count > 500:
            # Switch modes periodically
            if mode == "lse":
                mode = "topk"
            else:
                mode = "lse"
            mode_count = 0

        if mode == "lse":
            grad = compute_lse_grad(a, c, beta)
            beta = min(beta_max, beta * 1.005)
        else:
            k = min(16, 2 + it // 500)
            grad = topk_subgradient(a, c, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step
        if prev_a is not None and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            dot_sy = np.dot(s_vec, y_vec)
            dot_yy = np.dot(y_vec, y_vec)
            if dot_yy > 1e-20 and 1e-8 < dot_sy < 1e6:
                bb = dot_sy / dot_yy
                bb = np.clip(bb, 1e-8, 1.0 / (gnorm + 1e-10))
                step = 0.5 * step + 0.5 * bb

        max_s = S / (n * gnorm * 0.5 + 1e-10)
        step = min(step, max_s)
        step = max(step, 1e-12)

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < cur_obj - 1e-13:
            prev_a = a.copy()
            prev_grad = grad.copy()
            a = a_new
            c = c_new
            cur_obj = obj_new
            step = min(step * 1.08, max_s * 1.5)
            stuck = 0
            if cur_obj < best_obj - 1e-11:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            step *= 0.7
            stuck += 1

        if stuck > 200:
            a = structured_perturb(a, n, S, rng)
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / (S * S)
            step = 1.0
            stuck = 0
            prev_a = None
            prev_grad = None

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            print("    hybrid it=%d, obj=%.8f, best=%.8f, mode=%s, e=%.0fs" % (
                it, cur_obj, best_obj, mode, elapsed))

    return best_a, best_obj


def generate_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t))
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2 / 0.04))
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * t) + 0.15 * np.cos(10 * np.pi * t + 0.5))
    inits.append(rng.random(n) + 0.05)
    inits.append(1.0 + 2.0 * (t ** 4 + (1 - t) ** 4))
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2))
    inits.append(1.0 + 0.25 * np.cos(6 * np.pi * t) + 0.15 * np.sin(14 * np.pi * t))
    inits.append(np.maximum(1.0 - 0.6 * np.abs(t - 0.5) ** 0.5, 0.05))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2 / 0.02))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses multi-phase optimization: exploration, LSE gradient refinement,
    hybrid top-k/LSE polishing, and micro-optimization.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)
    t0 = time.monotonic()

    # Get initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
        else:
            given = None
    except Exception:
        given = None

    # Main sequence length
    if given is not None and 2048 <= len(given) <= 16384:
        n = len(given)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Main length: n=%d, S=%.4f" % (n, S))

    global_best = np.inf
    global_a = None

    # === Phase 1: Quick exploration (15% of budget) ===
    phase1_budget = (deadline - time.monotonic()) * 0.15
    print("=== Phase 1: Exploration (%.0fs) ===" % phase1_budget)

    inits = []
    if given is not None:
        inits.append(("given", given.copy()))
    gen = generate_inits(n, rng)
    for i, g in enumerate(gen):
        inits.append(("init_%d" % i, g))

    # Perturbations of given
    if given is not None:
        for i in range(2):
            g_norm = project_simplex(given, S)
            pert = structured_perturb(g_norm, n, S, rng)
            inits.append(("pert_given_%d" % i, pert))

    n_inits = len(inits)
    per_init = max(3.0, phase1_budget / n_inits)

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - t0) * 0.75:
            break
        arr_n = len(arr)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        t_alloc = min(per_init, deadline - time.monotonic() - 120)
        if t_alloc < 3:
            continue

        # Use LSE optimization for exploration
        a_opt, score = run_lse_optimization(arr, arr_n, S_i, t_alloc, rng, verbose=False)
        a_norm = project_simplex(np.clip(a_opt, 0, 1000.0), S_i)
        real_score = compute_objective(a_norm)

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: obj=%.6f" % (name, arr_n, global_best))

    print("Phase 1 done: best=%.6f" % global_best)

    # === Phase 2: Intensive LSE refinement (40% of budget) ===
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = compute_objective(global_a)

    remaining = deadline - time.monotonic()
    phase2_budget = remaining * 0.40
    print("=== Phase 2: LSE refinement (%.0fs) ===" % phase2_budget)

    n_best = len(global_a)
    S_best = np.sqrt(2.0 * n_best)

    a2, s2 = run_lse_optimization(global_a, n_best, S_best, phase2_budget, rng, verbose=True)
    a2 = project_simplex(np.clip(a2, 0, 1000.0), S_best)
    s2 = compute_objective(a2)
    if s2 < global_best:
        global_best = s2
        global_a = a2.copy()
        print("After Phase 2: %.6f" % global_best)

    # === Phase 3: Hybrid optimization (25% of budget) ===
    remaining = deadline - time.monotonic()
    phase3_budget = remaining * 0.42
    print("=== Phase 3: Hybrid optimization (%.0fs) ===" % phase3_budget)

    a3, s3 = run_hybrid_optimization(global_a, n_best, S_best, phase3_budget, rng, verbose=True)
    a3 = project_simplex(np.clip(a3, 0, 1000.0), S_best)
    s3 = compute_objective(a3)
    if s3 < global_best:
        global_best = s3
        global_a = a3.copy()
        print("After Phase 3: %.6f" % global_best)

    # === Phase 4: Micro-optimization (15% of budget) ===
    remaining = deadline - time.monotonic()
    phase4_budget = remaining * 0.50
    print("=== Phase 4: Micro-optimization (%.0fs) ===" % phase4_budget)

    if phase4_budget > 10 and global_a is not None:
        a4 = global_a.copy()
        n4 = len(a4)
        S4 = np.sqrt(2.0 * n4)
        c4 = fft_convolve_self(a4)
        cur4 = 2.0 * n4 * c4.max() / S4**2
        best4 = cur4
        step4 = 0.05
        it4 = 0
        micro_start = time.monotonic()

        while time.monotonic() - micro_start < phase4_budget:
            it4 += 1
            c4 = fft_convolve_self(a4)
            cur4 = 2.0 * n4 * c4.max() / S4**2
            if cur4 < best4 - 1e-12:
                best4 = cur4

            # Use LSE with high beta (essentially top-k)
            k_use = 1
            grad4 = topk_subgradient(a4, c4, k_use)
            gn4 = np.linalg.norm(grad4)
            if gn4 < 1e-15:
                break

            eff_step4 = step4 / (1.0 + 1e-5 * it4)
            a_try = project_simplex(a4 - eff_step4 * grad4, S4)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n4 * c_try.max() / S4**2

            if obj_try < cur4 - 1e-14:
                a4 = a_try
                step4 = min(step4 * 1.01, 0.3)
            else:
                step4 *= 0.88
                if step4 < 1e-6:
                    # Try with top-k
                    for kk in [3, 7, 15]:
                        grad_kk = topk_subgradient(a4, c4, kk)
                        a_try2 = project_simplex(a4 - eff_step4 * grad_kk, S4)
                        c_try2 = fft_convolve_self(a_try2)
                        obj_try2 = 2.0 * n4 * c_try2.max() / S4**2
                        if obj_try2 < cur4:
                            a4 = a_try2
                            step4 = min(step4 * 1.5, 0.1)
                            break
                    else:
                        # Perturb
                        noise = 0.002 * S4 / np.sqrt(n4)
                        a4 = project_simplex(a4 + rng.normal(0, noise, n4), S4)
                        step4 = 0.05

            if it4 % 20000 == 0 and verbose:
                print("    micro it=%d, best=%.8f" % (it4, best4))

        a4 = project_simplex(np.clip(a4, 0, 1000.0), S4)
        s4 = compute_objective(a4)
        if s4 < global_best:
            global_best = s4
            global_a = a4.copy()
            print("After Phase 4: %.6f" % global_best)

    # === Phase 5: Final with different seed ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        print("=== Phase 5: Final refinement (%.0fs) ===" % remaining)
        rng5 = np.random.default_rng(seed + 7777)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a5, s5 = run_lse_optimization(global_a, n_best, S_best, remaining * 0.8, rng5, verbose=False)
        a5 = project_simplex(np.clip(a5, 0, 1000.0), S_best)
        s5 = compute_objective(a5)
        if s5 < global_best:
            global_best = s5
            global_a = a5.copy()
            print("After Phase 5: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    print("=== FINAL: n=%d, obj=%.8f, time=%.0fs ===" % (
        len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
