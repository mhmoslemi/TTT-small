def project_simplex(v, S):
    v=np.asarray(v,dtype=np.float64)
    n=len(v)
    if S <= 0 or n==0: return np.zeros(n)
    if not np.all(np.isfinite(v)): v=np.nan_to_num(v, nan=0.0, posinf=S, neginf=0.0)

        rho=n
    else:

    theta=(cumsum[rho-1]-S)/rho

  2) Active-set LP direction finding (approximation of the full LP

  3) Top-peak projected subgradient polishing.
  4) Optional longer length exploration.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def next_pow2(m):
    m = int(max(1, m))
    return 1 << max(0, (m - 1).bit_length())


def fft_convolve_self(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = next_pow2(size)
    A = fft(a, L)
    c = np.real(ifft(A * A))[:size]
    return np.maximum(c, 0.0)


def fft_convolve_binary(a, b):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    size = a.size + b.size - 1
    L = next_pow2(size)
    c = np.real(ifft(fft(a, L) * fft(b, L)))[:size]
    return np.maximum(c, 0.0)


def score_from_conv(a, c, Ssum=None):
    n = len(a)
    total = float(Ssum) if Ssum is not None else float(np.sum(a))
    if total <= 1e-15 or not np.isfinite(total):
        return np.inf
    if c.size == 0:
        return 0.0
    return float(2.0 * n * np.max(c) / (total * total))


def normalize_sum(a, Ssum):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.full(a.size, float(Ssum) / a.size, dtype=np.float64)
    return a * (float(Ssum) / s)


def project_simplex(v, Ssum):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=Ssum, neginf=0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - Ssum) / idx
    valid = tc <= w
    valid_idx = np.where(valid)[0]
    if valid_idx.size == 0:
        rho = n
    else:
        rho = int(valid_idx.max()) + 1
    theta = (cumsum[rho - 1] - Ssum) / float(rho)
    return np.maximum(v - theta, 0.0)


def select_lags(c, n, max_K, grid_step, top_count):
    L = len(c)
    if L <= 0:
        return np.array([], dtype=np.int64)
    if grid_step < 1:
        grid_step = 1
    grid = np.arange(0, L, int(grid_step), dtype=np.int64)

    if top_count <= 0:
        top = np.array([], dtype=np.int64)
    elif top_count >= L:
        top = np.arange(L, dtype=np.int64)
    else:
        top = np.argpartition(c, -int(top_count))[-int(top_count):].astype(np.int64)

    mc = float(np.max(c)) if c.size else 0.0
    if mc > 0.0:
        near = np.where(c >= 0.999 * mc)[0].astype(np.int64)
    else:
        near = np.array([], dtype=np.int64)

    core_parts = []
    if top.size:
        core_parts.append(top)
    if grid.size:
        core_parts.append(grid)
    core = np.unique(np.concatenate(core_parts)) if core_parts else np.array([], dtype=np.int64)

    if core.size > max_K:
        max_grid = max(0, int(max_K) - int(top.size))
        if max_grid < grid.size:
            core = np.unique(np.concatenate([top, grid[:max_grid]]))
        else:
            core = core[:max_K]

    if core.size < max_K:
        extra = np.setdiff1d(near, core, assume_unique=False)
        room = int(max_K) - core.size
        if extra.size > room:
            extra = extra[:room]
        core = np.unique(np.concatenate([core, extra]))

    if core.size < max_K:
        all_parts = []
        if top.size:
            all_parts.append(top)
        if grid.size:
            all_parts.append(grid)
        if near.size:
            all_parts.append(near)
        combined = np.unique(np.concatenate(all_parts)) if all_parts else np.array([], dtype=np.int64)
        extra = np.setdiff1d(combined, core, assume_unique=False)
        room = int(max_K) - core.size
        if extra.size > room:
            order = np.argsort(-c[extra])[:room]
            extra = extra[order]
        core = np.unique(np.concatenate([core, extra]))

    return core.astype(np.int64, copy=False)


def build_A(p, K):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    K = np.asarray(K, dtype=np.int64).reshape(-1)
    A = np.zeros((K.size, n), dtype=np.float64)
    for r, k in enumerate(K):
        k = int(k)
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            A[r, lo:hi + 1] = p[k - hi:k + 1][::-1]
    return A


def solve_lp_active(p, K, M, U, time_left):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    K = np.asarray(K, dtype=np.int64).reshape(-1)
    n = p.size
    if n == 0 or K.size == 0:
        return None

    A = build_A(p, K)
    b = np.full(K.size, float(M), dtype=np.float64)
    cvec = -np.ones(n, dtype=np.float64)
    bounds = np.zeros((n, 2), dtype=np.float64)
    bounds[:, 1] = float(U)

    opts_basic = {"presolve": True}
    opts_timed = dict(opts_basic)
    if time_left is not None and time_left > 15.0:
        opts_timed["time_limit"] = float(max(5.0, min(60.0, 0.5 * float(time_left))))

    try:
        res = linprog(cvec, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=opts_timed)
        if res.status == 0 and res.x is not None:
            q = np.asarray(res.x, dtype=np.float64)
            q = np.clip(q, 0.0, None)
            if np.all(np.isfinite(q)) and q.sum() > 0.0:
                return q
    except Exception:
        pass

    try:
        res = linprog(cvec, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=opts_basic)
        if res.status == 0 and res.x is not None:
            q = np.asarray(res.x, dtype=np.float64)
            q = np.clip(q, 0.0, None)
            if np.all(np.isfinite(q)) and q.sum() > 0.0:
                return q
    except Exception:
        pass

    return None


def blend_max_conv(c_p, c_q, c_pq, t):
    u = 1.0 - float(t)
    val = u * u * c_p + 2.0 * float(t) * u * c_pq + float(t) * float(t) * c_q
    return float(np.max(val)) if val.size else 0.0


def line_search_blend(p, q, c_p, c_q, c_pq, n, Ssum, best_score, max_t=0.65):
    base_ts = np.array([0.01, 0.02, 0.05, 0.10, 0.18, 0.28, 0.40, 0.55, 0.70], dtype=np.float64)
    base_ts = base_ts[base_ts <= max_t]
    if base_ts.size == 0:
        return None, None

    best_t = 0.0
    best_val = np.inf
    for t in base_ts:
        mc = blend_max_conv(c_p, c_q, c_pq, float(t))
        sc = 2.0 * n * mc / (Ssum * Ssum)
        if sc < best_val:
            best_val = sc
            best_t = float(t)

    if best_t > 0.0:
        lo = max(0.0, best_t - 0.12)
        hi = min(max_t, best_t + 0.12)
        ts = np.linspace(lo, hi, 41, dtype=np.float64)
        for t in ts:
            mc = blend_max_conv(c_p, c_q, c_pq, float(t))
            sc = 2.0 * n * mc / (Ssum * Ssum)
            if sc < best_val:
                best_val = sc
                best_t = float(t)

    if best_val < best_score * (1.0 - 1e-12) and best_t > 1e-5:
        p_new = project_simplex((1.0 - best_t) * p + best_t * q, Ssum)
        return p_new, float(best_val)
    return None, None


def sqrt_square_close(C, P):
    R = np.sqrt(C)
    L = R.size
    if L == 0:
        return R

    if float(np.real(P[0] * np.conj(R[0]))) < 0.0:
        R[0] = -R[0]

    half = L // 2
    if L % 2 == 0:
        ny = half
        if ny < L and float(np.real(P[ny] * np.conj(R[ny]))) < 0.0:
            R[ny] = -R[ny]

    if half > 1:
        idx = np.arange(1, half, dtype=np.int64)
        nidx = L - idx
        pair_score = (
            np.real(P[idx] * np.conj(R[idx])) +
            np.real(P[nidx] * np.conj(R[nidx]))
        )
        flip = pair_score < 0.0
        if np.any(flip):
            ii = idx[flip]
            jj = nidx[flip]
            R[ii] = -R[ii]
            R[jj] = -R[jj]

    return R


def run_spectral_sqrt(p_init, n, Ssum, budget, rng):
    p = normalize_sum(p_init, Ssum)
    c = fft_convolve_self(p)
    total = Ssum * Ssum
    cur_score = score_from_conv(p, c, Ssum)
    best = p.copy()
    best_score = cur_score

    if not np.isfinite(cur_score):
        return best, best_score

    Mlen = 2 * n - 1
    L = next_pow2(Mlen)
    flat = np.full(Mlen, total / float(Mlen), dtype=np.float64)
    Mlower = total / float(Mlen)

    start = time.monotonic()
    deadline = start + max(1.0, float(budget))
    it = 0
    no_imp = 0

    while time.monotonic() < deadline and n > 0:
        it += 1
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / max(1.0, float(budget)))

        lam = 0.25 + 0.55 * prog
        if no_imp > 18:
            lam = max(0.10, lam * 0.85)

        Mcur = float(np.max(c))
        if not np.isfinite(Mcur) or Mcur <= 0.0:
            break

        Mclip = Mlower + (Mcur - Mlower) * (1.0 - prog)
        if Mclip < Mlower * 1.02:
            Mclip = Mlower * 1.02
        if Mclip > Mcur:
            Mclip = max(Mlower * 1.02, Mcur * 0.985)

        P = fft(p, L)
        Ccur = P * P
        ccur = np.real(ifft(Ccur))[:Mlen]
        ccur = np.maximum(ccur, 0.0)
        scc = float(ccur.sum())
        if scc <= 1e-15:
            break
        ccur = ccur * (total / scc)

        clipped = np.minimum(ccur, Mclip)
        scl = float(clipped.sum())
        deficit = total - scl
        if deficit > 0.0:
            headroom = np.maximum(Mclip - clipped, 0.0)
            hsum = float(headroom.sum())
            if hsum > 1e-15:
                cleared = clipped + (deficit / hsum) * headroom
                cs = float(cleared.sum())
                if cs > 1e-15:
                    clipped = cleared * (total / cs)
                else:
                    clipped = flat.copy()
            else:
                clipped = flat.copy()
        elif abs(scl - total) > 1e-9:
            clipped = clipped * (total / scl)

        ctar = (1.0 - lam) * ccur + lam * clipped
        Ct = fft(ctar, L)
        R = sqrt_square_close(Ct, P)
        p_raw = np.real(ifft(R))[:n]

        best_mu = 0.0
        best_cand = None
        best_cand_score = cur_score
        for mu in (0.10, 0.25, 0.45, 0.70, 1.00):
            cand = project_simplex((1.0 - mu) * p + mu * p_raw, Ssum)
            cc = fft_convolve_self(cand)
            sc = score_from_conv(cand, cc, Ssum)
            if np.isfinite(sc) and sc < best_cand_score - 1e-12:
                best_cand_score = sc
                best_cand = cand
                best_mu = mu

        if best_cand is not None:
            p = best_cand
            c = fft_convolve_self(p)
            cur_score = float(np.max(c)) * (2.0 * n / total)
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            no_imp = 0
        else:
            no_imp += 1

        if it % 200 == 0:
            print("spectral_sqrt it=%d score=%.6f best=%.6f" % (it, cur_score, best_score))

    return best, best_score


def run_top_gradient(p_init, n, Ssum, budget, rng, top_count=32):
    p = normalize_sum(p_init, Ssum)
    c = fft_convolve_self(p)
    total = Ssum * Ssum
    cur_score = score_from_conv(p, c, Ssum)
    best = p.copy()
    best_score = cur_score

    if not np.isfinite(cur_score) or n == 0:
        return best, best_score

    Lc = 2 * n - 1
    start = time.monotonic()
    deadline = start + max(1.0, float(budget))
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1
        if top_count >= Lc:
            top_idx = np.arange(Lc, dtype=np.int64)
        else:
            top_idx = np.argpartition(c, -top_count)[-top_count:].astype(np.int64)

        w = np.zeros(Lc, dtype=np.float64)
        w[top_idx] = 1.0
        w[Lc - 1 - top_idx] = 1.0
        peak_idx = np.nonzero(w)[0]

        g = np.zeros(n, dtype=np.float64)
        for s in peak_idx:
            s = int(s)
            lo = s - (n - 1)
            if lo < 0:
                lo = 0
            hi = s
            if hi > n - 1:
                hi = n - 1
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * p[s - hi:s + 1][::-1]

        gmax = float(np.max(np.abs(g))) if g.size else 0.0
        if gmax <= 1e-14:
            break

        base_scale = (Ssum / float(n)) / gmax
        best_cand = None
        best_cand_score = cur_score

        for mult in (0.05, 0.10, 0.20, 0.35, 0.60, 1.0, 2.0):
            step = base_scale * mult
            cand = project_simplex(p - step * g, Ssum)
            cc = fft_convolve_self(cand)
            sc = score_from_conv(cand, cc, Ssum)
            if np.isfinite(sc) and sc < best_cand_score - 1e-12:
                best_cand_score = sc
                best_cand = cand

        if best_cand is not None:
            p = best_cand
            c = fft_convolve_self(p)
            cur_score = float(np.max(c)) * (2.0 * n / total)
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 10:
                noise = 0.004 * Ssum / np.sqrt(float(n))
                cand = project_simplex(p + rng.normal(0.0, noise, n), Ssum)
                cc = fft_convolve_self(cand)
                sc = score_from_conv(cand, cc, Ssum)
                if np.isfinite(sc) and sc < cur_score * 1.005:
                    p = cand
                    c = cc
                    cur_score = sc
                    if cur_score < best_score:
                        best = p.copy()
                        best_score = cur_score
                no_imp = 0

        if it % 200 == 0:
            print("top_grad it=%d score=%.6f best=%.6f" % (it, cur_score, best_score))

    return best, best_score


def lp_params_for_n(n):
    if n <= 4096:
        return 420, 160, max(16, n // 64)
    if n <= 8192:
        return 600, 250, max(32, n // 64)
    if n <= 16384:
        return 450, 220, max(64, n // 64)
    return 320, 170, max(128, n // 64)


def run_active_lp(p_init, n, Ssum, budget, rng):
    p = normalize_sum(p_init, Ssum)
    c = fft_convolve_self(p)
    total = Ssum * Ssum
    cur_score = score_from_conv(p, c, Ssum)
    best = p.copy()
    best_score = cur_score

    if not np.isfinite(cur_score) or n == 0:
        return best, best_score

    max_K, top_count, grid_step = lp_params_for_n(n)
    Mlower = total / float(2 * n - 1)

    start = time.monotonic()
    deadline = start + max(1.0, float(budget))
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        it += 1
        Mcur = float(np.max(c))
        if not np.isfinite(Mcur) or Mcur <= 0.0:
            break

        if no_imp == 0:
            M_lp = Mcur
        elif no_imp <= 2:
            M_lp = max(Mlower * 1.02, Mcur * 0.975)
        elif no_imp <= 5:
            M_lp = max(Mlower * 1.02, Mcur * 0.94)
        else:
            M_lp = max(Mlower * 1.02, Mcur * 0.90)

        pmax = float(np.max(p)) if p.size else 0.0
        U = max(10.0 * pmax + 1e-9, 1.2 * np.sqrt(max(M_lp, 0.0)), 5.0 * Ssum / float(n))
        U = min(float(U), 0.5 * Ssum)

        top_cur = min(int(top_count) + it // 6, int(top_count * 1.6))
        K = select_lags(c, n, max_K, grid_step, top_cur)
        remaining = max(0.0, deadline - time.monotonic())

        q = solve_lp_active(p, K, M_lp, U, remaining)
        improved = False

        if q is not None and q.size == n and q.sum() > 0.0:
            q = normalize_sum(q, Ssum)
            c_p = c
            c_q = fft_convolve_self(q)
            c_pq = fft_convolve_binary(p, q)

            p_new, _ = line_search_blend(p, q, c_p, c_q, c_pq, n, Ssum, cur_score, max_t=0.60)
            if p_new is not None:
                cc = fft_convolve_self(p_new)
                sc = score_from_conv(p_new, cc, Ssum)
                if np.isfinite(sc) and sc < cur_score - 1e-12:
                    p = p_new
                    c = cc
                    cur_score = sc
                    if cur_score < best_score:
                        best = p.copy()
                        best_score = cur_score
                    improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                noise = 0.002 * Ssum / np.sqrt(float(n))
                cand = project_simplex(p + rng.normal(0.0, noise, n), Ssum)
                cc = fft_convolve_self(cand)
                sc = score_from_conv(cand, cc, Ssum)
                if np.isfinite(sc) and sc < cur_score * 1.003:
                    p = cand
                    c = cc
                    cur_score = sc
                    if cur_score < best_score:
                        best = p.copy()
                        best_score = cur_score
                no_imp = 0

        if it % 10 == 0:
            print("active_lp it=%d cur=%.6f best=%.6f Mdiff=%.6f" %
                  (it, cur_score, best_score, cur_score * (2.0 * n / total) - 2.0 * n * Mlower / total))

    return best, best_score


def resample_sequence(a, new_n, Ssum):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.full(new_n, Ssum / float(max(1, new_n)), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    b = np.interp(x_new, x_old, a)
    return normalize_sum(b, Ssum)


def make_initial_sequence(n, Ssum, rng):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    base = 1.0 + 0.45 * np.cos(2.0 * np.pi * t) + 0.18 * np.cos(4.0 * np.pi * t + 0.35)
    base = np.maximum(base, 0.02)
    base = base + rng.normal(0.0, 0.04, n)
    base = np.maximum(base, 0.0)
    return normalize_sum(base, Ssum)


def true_score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    lst = [float(x) for x in a]
    try:
        val = evaluate_sequence(lst)
    except Exception:
        return np.inf
    return float(val) if np.isfinite(val) else np.inf

    if not candidates:
        internal_best = np.full(256, Ssum / 256.0, dtype=np.float64)
        internal_score = 2.0
        candidates = [internal_best.copy()]
        cand_scores = [2.0]

    scores_arr = np.array([float(x) for x in cand_scores], dtype=np.float64)
    order = np.argsort(scores_arr)
    best_final = None
    best_true = np.inf
    evaluated = 0
    limit = 8

    for idx in order:
        if evaluated >= limit:
            break
        cand = candidates[int(idx)]
        clen = cand.size
        if clen > 32768 and evaluated >= 3:
            break
        if clen == 0:
            continue
        val = true_score(cand)
        if np.isfinite(val) and val < best_true:
            best_true = val
            best_final = cand.copy()
        evaluated += 1

    if best_final is None:
        fallback = np.full(256, Ssum / 256.0, dtype=np.float64)
        val = true_score(fallback)
        if np.isfinite(val):
            best_final = fallback
        else:
            best_final = internal_best

    out = np.asarray(best_final, dtype=np.float64).reshape(-1)
    out = np.clip(out, 0.0, 1000.0)
    final_score = true_score(out)
    print("final length=%d internal=%.6f true=%.6f time=%.1fs" %
          (out.size, internal_score, final_score, time.monotonic() - t_start))

    if not np.isfinite(final_score):
        fallback = np.full(256, Ssum / 256.0, dtype=np.float64)
        out = fallback

    return [float(x) for x in out]
