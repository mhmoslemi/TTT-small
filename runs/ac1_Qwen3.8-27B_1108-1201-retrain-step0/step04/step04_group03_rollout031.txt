import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_fista_v2(a_init, n, S, max_time, rng, 
                  beta_init=50.0, beta_growth=0.003, beta_max=3000.0,
                  repulsion_freq=200, repulsion_beta=30.0, repulsion_lr=0.05,
                  noise_freq=3000, shake_freq=8000):
    """Improved FISTA with better schedules and frequency-domain shaking."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_counter = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best * 0.9999:
            best = cur
            best_a = y.copy()
            last_improve = it
            stall_counter = 0
        else:
            stall_counter += 1
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        
        # Adaptive: if stalled, reduce step and add noise
        if stall_counter > 5000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stall_counter = 0
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic repulsion
        if it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(repulsion_beta * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 1e-300:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                # Multiplicative update
                y = y * np.exp(-repulsion_lr * rep / (y.max() + 1e-10))
                y = np.maximum(y, 1e-15)
                y = project_simplex(y, S)
                a = y.copy()
                t_mom = 1.0
        
        # Frequency-domain shake: modify the spectrum to reduce peak
        if it % shake_freq == 0 and stall_counter > 2000:
            fsize_s = 1 << (2 * n - 1 - 1).bit_length()
            A_s = fft(y, fsize_s)
            # Find the peak of |A|^2 and reduce it
            power = np.abs(A_s)**2
            pmax_idx = power.argmax()
            # Reduce the peak frequency components
            A_s[pmax_idx] *= 0.9
            # Also tweak neighbors
            for d in [-2, -1, 1, 2]:
                idx2 = (pmax_idx + d) % fsize_s
                A_s[idx2] *= 0.95
            y_new = np.real(ifft(A_s))[:n]
            if np.all(y_new >= -0.1 * y_new.max()):
                y_new = np.maximum(y_new, 0)
                y_new = project_simplex(y_new, S)
                c_new = fft_convolve_self(y_new)
                if c_new.max() < best * 1.01:
                    y = y_new
                    a = y_new.copy()
                    t_mom = 1.0
                    stall_counter = 0
        
        # Periodic noise
        if it % noise_freq == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    # Final evaluation
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    
    return best_a, best


def direct_subgradient_polish(a, n, S, max_time, rng, lr_init=0.01, lr_decay=0.9999):
    """Polish using direct subgradient of the max function."""
    a = normalize_sum(a, n).copy()
    lr = lr_init
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top few max positions (for stability)
        top_k = min(5, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        # Weight by how close to max
        weights = np.exp((c[idx] - c.max()) * 10.0)
        weights /= weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for wi, ki in zip(weights, idx):
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            js = np.arange(lo, hi + 1)
            grad[js] += wi * 2.0 * a[ki - js]
        
        # Multiplicative update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr * grad / (a.max() + 1e-10)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = project_simplex(a_new, S)
        
        # Line search: accept if better
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(-step * 0.5)
            a_half = np.maximum(a_half, 1e-15)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur:
                a = a_half
            else:
                lr *= 0.95
        
        lr = max(lr * lr_decay, 1e-6)
    
    return best_a, best


def solve_lp_best_response(f, n, S, top_k=200, time_limit=15.0):
    """Solve LP for best response."""
    c = fft_convolve_self(f)
    M = c.max()
    
    idx = np.argsort(c)[::-1][:top_k]
    
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        prob.solve(solver='SCS', verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    
    return None


def lp_iteration(f, n, S, top_k=150):
    """One LP iteration with line search."""
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    
    g = solve_lp_best_response(f, n, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old
    
    def eval_step(t):
        f_test = np.maximum((1-t)*f + t*g, 0)
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()
    
    # Try full step
    val1 = eval_step(1.0)
    if val1 < M_old:
        f_new = project_simplex(np.maximum(g, 0), S)
        return f_new, val1
    
    # Bisection
    lo, hi = 0.0, 1.0
    best_t, best_val = 0.0, M_old
    for _ in range(10):
        mid = (lo + hi) / 2
        val_mid = eval_step(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid
    
    if best_t > 0.05:
        f_new = project_simplex(np.maximum((1-best_t)*f + best_t*g, 0), S)
        return f_new, best_val
    
    return f.copy(), M_old


def interpolate_sequence(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def generate_structured_init(n, rng, seed_type=0):
    """Generate structured initializations based on known good patterns."""
    t = np.linspace(0, 1, n, endpoint=False)
    
    if seed_type == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif seed_type == 1:
        # Sinusoidal modulation
        a = 0.7 + 0.3 * np.cos(2 * np.pi * 3 * t)
    elif seed_type == 2:
        # Two-harmonic
        a = 0.6 + 0.25 * np.cos(2 * np.pi * 5 * t) + 0.15 * np.cos(2 * np.pi * 13 * t)
    elif seed_type == 3:
        # Chaotic (deterministic)
        x = 0.42
        a = np.zeros(n)
        for i in range(n):
            x = 4.0 * x * (1 - x)
            a[i] = x
    elif seed_type == 4:
        # Smooth random
        freqs = rng.integers(1, 20, 5)
        phases = rng.uniform(0, 2*np.pi, 5)
        amps = rng.uniform(0.1, 0.4, 5)
        a = np.ones(n) * 0.5
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * t + p)
    elif seed_type == 5:
        # Block structure with smooth transitions
        nblocks = rng.integers(5, 15)
        block_heights = rng.uniform(0.3, 1.0, nblocks)
        block_len = n // nblocks
        a = np.repeat(block_heights, block_len)
        if len(a) < n:
            a = np.concatenate([a, np.full(n - len(a), block_heights[-1])])
        a = a[:n]
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=3)
    elif seed_type == 6:
        # Smooth noise
        a = rng.random(n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=10)
        a = a - a.min() + 0.1
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    
    return np.maximum(a, 0.01)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Multi-phase FISTA with adaptive LSE smoothing, frequency-domain shaking,
LP refinement with cutting planes, and direct subgradient polishing.
Key improvements: better beta schedule, peak-aware repulsion, cutting-plane LP,
and aggressive final polishing with adaptive step sizes.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.ndimage import gaussian_filter1d


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_fista_v2(a_init, n, S, max_time, rng,
                 beta_init=50.0, beta_growth=0.002, beta_max=3000.0,
                 repulsion_freq=200, repulsion_beta=40.0, repulsion_lr=0.04,
                 noise_freq=4000, shake_freq=12000):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best * 0.9999:
            best = cur
            best_a = y.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        if stall > 5000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            stall = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        t_mom = t_new

            c_rep = fft_convolve_self(y)

                y = np.maximum(y, 1e-15)

                a = y.copy()
                t_mom = 1.0

        if it % shake_freq == 0 and stall > 3000:
            fsize_s = 1 << (2 * n - 1 - 1).bit_length()

                    a = y_new.copy()
                    t_mom = 1.0

                t_mom = 1.0

    c_final = fft_convolve_self(best_a)

    return best_a, best


def direct_subgradient_polish(a, n, S, max_time, rng, lr_init=0.008, lr_decay=0.99995):
    a = normalize_sum(a, n).copy()
    lr = lr_init
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        top_k = min(8, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        weights = np.exp((c[idx] - c.max()) * 15.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, ki in zip(weights, idx):
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            js = np.arange(lo, hi + 1)
            grad[js] += wi * 2.0 * a[ki - js]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr * grad / (a.max() + 1e-10)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = project_simplex(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            a_half = a * np.exp(-step * 0.5)
            a_half = np.maximum(a_half, 1e-15)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur:
                a = a_half
            else:
                lr *= 0.9

        lr = max(lr * lr_decay, 1e-7)

    return best_a, best


def solve_lp_response(f, n, S, top_k=200, time_limit=20.0):
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver='SCS', verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - t0 > time_limit * 3:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def lp_iteration(f, n, S, top_k=200):
    c_old = fft_convolve_self(f)
    M_old = c_old.max()
    g = solve_lp_response(f, n, S, top_k=top_k)
    if g is None:
        return f.copy(), M_old

    def eval_step(t):
        f_test = np.maximum((1 - t) * f + t * g, 0)
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()

    val1 = eval_step(1.0)
    if val1 < M_old:
        f_new = project_simplex(np.maximum(g, 0), S)
        return f_new, val1

    lo, hi = 0.0, 1.0
    best_t, best_val = 0.0, M_old
    for _ in range(12):
        mid = (lo + hi) / 2.0
        val_mid = eval_step(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0.05:
        f_new = project_simplex(np.maximum((1 - best_t) * f + best_t * g, 0), S)
        return f_new, best_val
    return f.copy(), M_old

def interpolate_sequence(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def generate_structured_init(n, rng, seed_type=0):

    elif seed_type == 3:
        x = 0.42
        a = np.zeros(n)
        for i in range(n):

            a[i] = x

        for _ in range(4):
            f = rng.integers(2, 30)
            p = rng.uniform(0, 2 * np.pi)

            a += amp * np.cos(2.0 * np.pi * f * t + p)
    elif seed_type == 5:
        a = rng.random(n)
        a = gaussian_filter1d(a, sigma=8)
