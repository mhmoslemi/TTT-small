"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization with:
1. FISTA with adaptive LSE smoothing (coarse to fine)
2. Subgradient descent with Armijo backtracking line search
3. Active-set peak equalization via LP-free projection
4. Multi-length exploration
5. Final high-precision polishing

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute objective: 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def compute_grad_topk(a, b, k=1, soft_width=0.0):
    """Compute gradient of max(b) approximated by top-k peaks.
    
    If soft_width > 0, uses smooth weighting: w_k = exp(-(max_b - b_k)/soft_width)
    for b_k above (max_b - soft_width * 5), else 0.
    """
    n = len(a)
    max_b = b.max()
    
    if soft_width > 0:
        weights = np.exp(-(max_b - b) / soft_width)
        # Zero out far-from-max
        weights[b < max_b - 5 * soft_width] = 0.0
        wsum = weights.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        weights /= wsum
    else:
        k = min(k, len(b))
        idx = np.argsort(b)[::-1][:k]
        weights = np.zeros(len(b))
        weights[idx] = 1.0 / k
    
    # Gradient = 2 * (weights circ-conv-like with a), but linear
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def line_search(a, grad, S, max_iter=30):
    """Armijo backtracking line search.
    
    Finds step size t such that obj(a - t*grad_projected) < obj(a) - c*t*||grad||^2
    where we project after each step.
    """
    n = len(a)
    cur_obj = obj_value(a)
    if not np.isfinite(cur_obj):
        return 0.0
    
    # Project gradient to sum-zero subspace
    grad = grad - np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return 0.0
    
    # Initial step
    t = 1.0 / (gnorm * 0.1)
    # Cap step to avoid going too negative
    min_a = a.min()
    neg_grad = np.minimum(grad, 0)
    if np.max(-neg_grad) > 0:
        t = min(t, 0.5 * min_a / np.max(-neg_grad))
    t = min(t, 1.0)
    
    c = 1e-4  # Armijo parameter
    for _ in range(max_iter):
        a_trial = project_simplex(a - t * grad, S)
        trial_obj = obj_value(a_trial)
        if trial_obj < cur_obj - c * t * gnorm * gnorm:
            return t
        t *= 0.5
        if t < 1e-15:
            return t
    return t

def run_peak_equalize(a_init, S, max_time, rng, tol=1e-6):
    """Peak equalization: try to make top peaks equal.
    
    At each step:
    1. Find the set of "active" peaks (within tolerance of max)
    2. Compute the average gradient from these peaks
    3. Project to sum-zero subspace
    4. Take a step that reduces the max peak
    5. If the max peak is now below some other peak, update active set
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s
    a = project_simplex(a, S)
    
    start = time.monotonic()
    best_b = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        b = fft_convolve_self(a)
        max_b = b.max()
        if max_b < best_b - 1e-10:
            best_b = max_b
            best_a = a.copy()
        
        # Active set: peaks within tol of max
        active = (b >= max_b - max(tol, max_b * 1e-6))
        n_active = active.sum()
        
        if n_active == 0:
            active[np.argmax(b)] = True
            n_active = 1
        
        # Weighted gradient from active peaks
        # Weights proportional to (max_b - b_k) for non-max + 1 for max
        weights = np.zeros(len(b))
        max_indices = np.where(b >= max_b - max(tol * 0.1, max_b * 1e-8))[0]
        near_indices = np.where((b >= max_b - max_b * 0.01) & (b < max_b - max(tol * 0.1, max_b * 1e-8)))[0]
        
        # Simple: uniform weight on top-k where k adapts
        k = min(max(1, n_active // 2), 64)
        idx = np.argsort(b)[::-1][:k]
        weights[idx] = 1.0 / k
        
        fsize = 1 << (len(weights) + n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Project to sum-zero
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with line search
        t = line_search(a, grad, S)
        
        a_new = project_simplex(a - t * grad, S)
        new_max = fft_convolve_self(a_new).max()
        
        if new_max < max_b:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - t * 0.1 * grad, S)
            new_max = fft_convolve_self(a_new).max()
            if new_max < max_b:
                a = a_new
            else:
                break  # Stuck
    
    c_final = fft_convolve_self(a).max()
    if c_final < best_b:
        best_b = c_final
        best_a = a.copy()
    
    return best_a, best_b


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def downscale_sequence(a_large, n_small):
    """Downscale by averaging blocks."""
    n_large = len(a_large)
    ratio = n_large // n_small
    if ratio < 1:
        return a_large.copy()
    small = a_large[:n_small * ratio].reshape(n_small, ratio).mean(axis=1)
    return small

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_max_b(a):
    """Compute max of self-convolution."""
    return fft_convolve_self(a).max()


def weighted_subgradient(a, b, k=1, lam=0.0):
    """Compute weighted subgradient of max(b).
    
    Arguments:
        a: the sequence
        b: the convolution (length 2n-1)
        k: number of top peaks to include (if lam=0)
        lam: inverse temperature for soft weighting (if lam>0)
    
    Returns:
        Gradient vector of length n
    """
    n = len(a)
    b_max = b.max()
    
    if lam > 0:
        # Soft weighting: w = exp(-(b_max - b)/lam) for b near max
        weights = np.exp(-(b_max - b) / lam)
        # Zero out distant peaks

        wsum = weights.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        weights /= wsum
    else:
        k = min(k, len(b))
        idx = np.argsort(b)[::-1][:k]
        weights = np.zeros(len(b))
        weights[idx] = 1.0 / k
    
    # Gradient = 2 * IFFT(FFT(weights) * conj(FFT(a)))[:n]
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def backtracking_linesearch(a, grad, S, b_current_max, max_backtracks=20):
    """Armijo backtracking line search.
    
    Finds t such that max(conv(project_simplex(a - t*grad, S))) < b_current_max.
    Returns the step size t and the new point.
    """
    n = len(a)
    
    # Initial step size
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return 0.0, a.copy()
    
    # Heuristic initial step
    t = 1.0 / (gnorm * 0.5)
    # Cap by non-negativity (rough estimate)
    pos_grad = np.maximum(grad, 0)
    if pos_grad.max() > 1e-15:
        t = min(t, 0.9 * a.min() / pos_grad.max())
    t = min(t, 2.0)

        a_trial = project_simplex(a - t * grad, S)
        new_max = compute_max_b(a_trial)
        if new_max < b_current_max - 1e-12:

        t *= 0.5

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm:
1. FISTA with adaptive LSE smoothing (coarse-to-fine bulk optimization)
2. Subgradient descent with Armijo backtracking line search (monotonic refinement)
3. Targeted coordinate descent on peak-contributing elements (fine polish)
4. Multi-length exploration with the given sequence as warm start

Key improvements over previous approach:
- Proper backtracking line search guarantees monotonic decrease
- Adaptive gradient direction (smooth k=1 to broad soft weighting)
- Targeted coordinate descent exploits peak structure
- Multiple independent runs with different seeds for robustness
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_max_b(a):
    """Compute max of self-convolution."""
    return fft_convolve_self(a).max()


def obj_value(a):
    """Compute objective: 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def weighted_subgradient(a, b, k=1, lam=0.0):
    """Compute weighted subgradient of max(b) at point a.

    If lam > 0: soft weighting w_k = exp(-(b_max - b_k)/lam) for near-peak
    If lam == 0: uniform weight on top-k peaks
    """
    n = len(a)
    b_max = b.max()

    if lam > 0:
        weights = np.exp(-(b_max - b) / lam)
        weights[b < b_max - 10.0 * lam] = 0.0
        wsum = weights.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        weights /= wsum
    else:
        k = min(k, len(b))
        idx = np.argsort(b)[::-1][:k]
        weights = np.zeros(len(b))
        weights[idx] = 1.0 / k

    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def backtrack_step(a, grad, S, current_max, max_bt=25):
    """Backtracking line search: find t s.t. max(conv(proj(a-t*grad))) < current_max."""
    n = len(a)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return 0.0, a.copy()

    # Initial step heuristic
    t = 0.5 / gnorm
    # Cap by non-negativity
    pos_grad = np.maximum(grad, 0.0)
    if pos_grad.max() > 1e-15:
        t = min(t, 0.8 * a.min() / pos_grad.max())
    t = min(t, 1.0)
    if t < 1e-12:
        return 0.0, a.copy()

    for _ in range(max_bt):
        a_trial = project_simplex(a - t * grad, S)
        new_max = compute_max_b(a_trial)
        if new_max < current_max - 1e-12:
            return t, a_trial
        t *= 0.5
        if t < 1e-14:
            break

    # Last resort: try the smallest step
    a_trial = project_simplex(a - t * grad, S)
    new_max = compute_max_b(a_trial)
    if new_max < current_max:
        return t, a_trial
    return 0.0, a.copy()


def run_fista_phase(a_init, S, max_time, rng, beta_init=5.0, beta_rate=0.004, beta_max=5000.0):
    """FISTA with LSE smoothing of max(conv)."""
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s
    a = project_simplex(a, S)

    y = a.copy()
    t_k = 1.0
    start_time = time.monotonic()
    best_b = compute_max_b(a)
    best_a = a.copy()

    it = 0
    last_improve = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        b = fft_convolve_self(y)
        cur_max = b.max()
        if cur_max < best_b - 1e-12:
            best_b = cur_max
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = b.max()
        diff = b - cm
        scaled = beta * diff
        if scaled.max() > 400:
            beta = 400.0 / max(1e-15, diff.max())
            scaled = beta * diff
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Lipschitz estimate for LSE: L ~ 2 * beta * max|b|
        L = 2.0 * beta * max(b.max(), 1e-10)
        step = 1.0 / max(L, 1e-15)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0
                last_improve = it

    b_final = fft_convolve_self(a)
    if b_final.max() < best_b:
        best_b = b_final.max()
        best_a = a.copy()

    return best_a, best_b


def run_subgrad_linesearch(a_init, S, max_time, rng,
                           k_start=1, k_end=16, lam_end=0.003):
    """Subgradient descent with backtracking line search.

    Adaptive: starts exact (k=1, no softening), broadens over time.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s
    a = project_simplex(a, S)

    start_time = time.monotonic()
    b = fft_convolve_self(a)
    best_b = b.max()
    best_a = a.copy()

    it = 0
    no_improve = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max_time)

        # Adaptive parameters
        k = max(1, int(k_start + (k_end - k_start) * progress))
        lam = lam_end * progress ** 2

        cur_max = b.max()

        # Compute gradient
        if lam > 0.001:
            grad = weighted_subgradient(a, b, k=32, lam=lam)
        else:
            grad = weighted_subgradient(a, b, k=k)

        # Line search
        t_step, a_new = backtrack_step(a, grad, S, cur_max)

        if t_step > 0:
            a = a_new
            b = fft_convolve_self(a)
            if b.max() < best_b - 1e-10:
                best_b = b.max()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Escape from local minima
        if no_improve > 30:
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            a_noisy = project_simplex(a + rng.normal(0, noise_scale, n), S)
            b_noisy = fft_convolve_self(a_noisy)
            if b_noisy.max() < b.max():
                a = a_noisy
                b = b_noisy
                if b.max() < best_b:
                    best_b = b.max()
                    best_a = a.copy()
            no_improve = 0

        if no_improve > 300:
            break

        # Periodic larger perturbation
        if it % 500 == 0 and progress < 0.6:
            ns = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            a_trial = project_simplex(a + rng.normal(0, ns, n), S)
            b_trial = fft_convolve_self(a_trial)
            if b_trial.max() < b.max():
                a = a_trial
                b = b_trial

    b_final = fft_convolve_self(a)
    if b_final.max() < best_b:
        best_b = b_final.max()
        best_a = a.copy()

    return best_a, best_b


def run_targeted_cd(a_init, S, max_time, rng):

        j_min = max(0, peak_lag - n + 1)

            # Try increasing (sometimes helps by reducing other peaks)

def run_multi_subgrad(a_init, S, max_time, rng, n_runs=3):
    """Run subgradient from multiple perturbed starts, keep best."""
    n = len(a_init)
    results = []

    t_per_run = max_time / n_runs

    for r in range(n_runs):
        if time.monotonic() > time.monotonic() + max_time - 5:
            break

        if r == 0:
            a_start = a_init.copy()
        else:
            # Perturbed start
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 + 0.5 * r)
            a_start = project_simplex(a_init + rng.normal(0, noise_scale, n), S)

        # Vary hyperparameters per run

        a_opt, b_opt = run_subgrad_linesearch(
            a_start, S, t_per_run, rng,

        )

    # Return best

            a_e1, b_e1 = run_fista_phase(a_exp_init, S_exp, t_e1, rng,
                                          beta_init=10.0, beta_rate=0.003, beta_max=3000.0)

            # Subgrad
            t_e2 = (remaining - 30) * 0.3
            t_e2 = min(t_e2, deadline - time.monotonic() - 20)
            if t_e2 > 5:
                a_e2, b_e2 = run_subgrad_linesearch(a_e1, S_exp, t_e2, rng,
                                                     k_start=1, k_end=12, lam_end=0.002)
            else:
                a_e2, b_e2 = a_e1, b_e1

            obj_exp = 2.0 * n_explore * b_e2 / (a_e2.sum() ** 2)
            print("  Exploration obj=%.6f" % obj_exp)

            if obj_exp < global_best:
                global_best = obj_exp
                global_a = a_e2.copy()
                print("  New global best from exploration!")

    # === FINAL POLISH ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_fin = len(global_a)
        S_fin = global_a.sum()

        if remaining > 20:
            print("\n=== Final polish (%.0fs) ===" % remaining)

            # Very high beta FISTA
            t_p1 = remaining * 0.4
            a_p1, b_p1 = run_fista_phase(global_a, S_fin, t_p1, rng,
                                          beta_init=200.0, beta_rate=0.002, beta_max=20000.0)
            if b_p1 < compute_max_b(global_a):
                global_a = a_p1
                global_best = 2.0 * n_fin * b_p1 / (S_fin ** 2)
                print("  After final FISTA: obj=%.6f" % global_best)

            # Subgrad with very tight parameters
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_p2, b_p2 = run_subgrad_linesearch(global_a, S_fin, remaining2 * 0.5, rng,
                                                     k_start=1, k_end=4, lam_end=0.001)
                if b_p2 < compute_max_b(global_a):
                    global_a = a_p2
                    global_best = 2.0 * n_fin * b_p2 / (S_fin ** 2)
                    print("  After final Subgrad: obj=%.6f" % global_best)

            # Quick CD
            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                a_p3, b_p3 = run_targeted_cd(global_a, S_fin, remaining3 * 0.7, rng)
                if b_p3 < compute_max_b(global_a):
                    global_a = a_p3
                    global_best = 2.0 * n_fin * b_p3 / (S_fin ** 2)
                    print("  After final CD: obj=%.6f" % global_best)

    # Verify and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, internal obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
