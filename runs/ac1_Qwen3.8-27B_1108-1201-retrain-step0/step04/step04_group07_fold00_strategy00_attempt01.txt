def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Order from largest to smallest
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx_candidates = (cumsum - S) / np.arange(1, n+1, dtype=np.float64)
    idx_valid = idx_candidates <= w
    if not np.any(idx_valid):
        rho = n
    else:
        rho = np.min(np.arange(1, n+1)[idx_candidates <= w])
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)
