import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

Strategy: Multi-phase FISTA with backtracking line search and adaptive LSE smoothing.
Phase 1: Coarse optimization with moderate beta (explore landscape).
Phase 2: Medium beta with momentum (refine peaks).
Phase 3: High-beta gradient descent polishing (fine-tune near optimum).
Phase 4: Local coordinate search for final micro-improvements.
Uses the given sequence as primary warm start, with diverse inits for multi-start.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    return cm + np.log(np.exp(scaled).sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

    t = 1.0

    it = 0

    best_x = x.copy()

        elapsed_frac = (time.monotonic() - start) / max_time

        cur_max = c_y.max()
        if cur_max < best_val - 1e-12:
            best_val = cur_max

            last_improve = it

        grad = lse_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        f_y = lse_smooth_max(c_y, beta)
        step = step_cache
        x_new = x
        converged = False

                x_new = x_try
                converged = True

            mom_scale = 1.0 - 0.8 * decay

            noise_scale = 0.002 * S / np.sqrt(n) * max(0.01, 1.0 - elapsed_frac)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
            last_improve = it

        # Also restart periodically
        if it % (restart_every * 2) == 0:
            t = 1.0
            y = x.copy()

    return best_x, best_val


def gd_polish(a_init, S, max_time, rng, beta=5000.0):
    """Plain gradient descent polishing with very high beta and small steps."""
    n = len(a_init)
    x = project_simplex(np.clip(a_init, 0.0, 1000.0), S)
    start = time.monotonic()
    best_val = fft_convolve_self(x).max()
    best_x = x.copy()
    step = 1.0 / (2.0 * beta * S * 0.1)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(x)
        if c.max() < best_val - 1e-12:
            best_val = c.max()
            best_x = x.copy()

        grad = lse_gradient(x, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: decrease slowly
        step = 1.0 / (2.0 * beta * S * (0.1 + 0.05 * it / 1000))

        x_new = project_simplex(x - step * grad, S)
        # Simple acceptance: only if it helps
        c_new = fft_convolve_self(x_new)
        if c_new.max() <= c.max() + 1e-10:
            x = x_new
        else:
            # Try half step
            x_half = project_simplex(x - 0.5 * step * grad, S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < c.max():
                x = x_half
            else:
                step *= 0.5

    return best_x, best_val


def coordinate_refine(a_init, S, max_time, rng):
    """Coordinate-wise local search: try small perturbations at each coordinate."""
    n = len(a_init)
    a = project_simplex(np.clip(a_init, 0.0, 1000.0), S)
    start = time.monotonic()
    best_val = fft_convolve_self(a).max()
    best_a = a.copy()
    c = fft_convolve_self(a)
    passes = 0

    while time.monotonic() - start < max_time:
        passes += 1
        # Sample a subset of coordinates to try (for speed)
        num_try = min(n, max(100, n // 4))
        indices = rng.choice(n, size=num_try, replace=False)

        for i in indices:
            if time.monotonic() - start > max_time:
                break
            if a[i] < 1e-10:
                continue

            # Try several perturbations
            base = a[i]
            for frac in [0.9, 1.1, 0.8, 1.2, 0.95, 1.05]:
                trial = a.copy()
                old_val = trial[i]
                trial[i] = base * frac
                trial = project_simplex(trial, S)
                c_trial = fft_convolve_self(trial)
                if c_trial.max() < best_val - 1e-10:
                    best_val = c_trial.max()
                    best_a = trial.copy()
                    a = trial
                    c = c_trial
                    break
                elif c_trial.max() < base * 1.01:
                    # Keep the best trial even if not global best
                    pass

        # Periodic global normalization check
        if passes % 5 == 0:
            a = project_simplex(best_a, S)

    # Final: use best_a
    return best_a, best_val


def generate_inits(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None and len(given) == n:
        inits.append(np.clip(given, 0.0, 1000.0).copy())

    # Raised cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Hann window
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    # Gaussian
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    # Sin^2
    inits.append(np.sin(np.pi * tt) ** 2)
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.2) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    # Random
    inits.append(rng.random(n) + 0.1)
    # Triangle
    inits.append(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01))

    return [np.maximum(x, 0.001) for x in inits]


def upscale_sequence(a_small, n_large):
    """Upscale via linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase FISTA with backtracking, GD polishing, and coordinate refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Primary length: 8192 (proven to work well)
    n = 8192
    S = np.sqrt(2.0 * n)

    # Also try 16384 if we have the given at that length
    n_alt = 16384
    S_alt = np.sqrt(2.0 * n_alt)

    # Prepare initializations for n=8192
    if given is not None:
        if len(given) == n:
            init_given = given.copy()
        elif len(given) > n:
            # Downsample by averaging
            ratio = len(given) // n
            if ratio > 1:
                init_given = given[:n * ratio].reshape(n, ratio).mean(axis=1)
            else:
                init_given = upscale_sequence(given, n)
        else:
            init_given = upscale_sequence(given, n)
    else:
        init_given = np.ones(n) * S / n

    inits = generate_inits(n, rng, given=None)
    inits.insert(0, init_given)

    # Phase 1: Try a few initializations with moderate FISTA (40% of time)
    time_phase1 = (deadline - time.monotonic()) * 0.35
    num_inits = min(5, len(inits))
    time_per_init = time_phase1 / num_inits

    best_a_n = None
    best_val_n = np.inf

    for i in range(num_inits):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.55:
            break
        t_alloc = min(time_per_init, deadline - time.monotonic() - 30)
        if t_alloc < 10:
            continue

        print("  Init %d/%d (time=%.0fs)" % (i + 1, num_inits, t_alloc))
        a_opt, val = fista_backtrack(inits[i], S, t_alloc, rng,
                                      beta_init=5.0, beta_max=500.0,
                                      beta_rate=0.005, momentum_decay_start=0.5)
        val_check = objective_value(a_opt)
        print("    obj=%.6f, max_b=%.6f" % (val_check, val))
        if val < best_val_n:
            best_val_n = val
            best_a_n = a_opt.copy()

    if best_a_n is None:
        best_a_n = project_simplex(init_given, S)
        best_val_n = fft_convolve_self(best_a_n).max()

    obj_n = 2.0 * n * best_val_n / S**2
    if obj_n < global_best:
        global_best = obj_n
        global_a = best_a_n.copy()
    print("After Phase 1: obj=%.6f" % global_best)

    # Phase 2: Higher beta FISTA on the best (30% of time)
    time_phase2 = (deadline - time.monotonic()) * 0.35
    if time_phase2 > 20:
        print("Phase 2: High-beta FISTA (%.0fs)" % time_phase2)
        a_opt, val = fista_backtrack(best_a_n, S, time_phase2, rng,
                                      beta_init=50.0, beta_max=3000.0,
                                      beta_rate=0.003, momentum_decay_start=0.4)
        val_check = objective_value(a_opt)
        print("  obj=%.6f" % val_check)
        if val < best_val_n:
            best_val_n = val
            best_a_n = a_opt.copy()
        obj_n = 2.0 * n * best_val_n / S**2
        if obj_n < global_best:
            global_best = obj_n
            global_a = best_a_n.copy()
    print("After Phase 2: obj=%.6f" % global_best)

    # Phase 3: GD polishing (20% of time)
    time_phase3 = (deadline - time.monotonic()) * 0.4
    if time_phase3 > 15:
        print("Phase 3: GD polishing (%.0fs)" % time_phase3)
        a_opt, val = gd_polish(best_a_n, S, time_phase3 * 0.6, rng, beta=5000.0)
        val_check = objective_value(a_opt)
        print("  After GD: obj=%.6f" % val_check)
        if val < best_val_n:
            best_val_n = val
            best_a_n = a_opt.copy()
        obj_n = 2.0 * n * best_val_n / S**2
        if obj_n < global_best:
            global_best = obj_n
            global_a = best_a_n.copy()

        # Coordinate refinement with remaining time
        time_coord = deadline - time.monotonic() - 10
        if time_coord > 10:
            print("  Coordinate refinement (%.0fs)" % time_coord)
            a_opt, val = coordinate_refine(best_a_n, S, time_coord, rng)
            val_check = objective_value(a_opt)
            print("  After coord: obj=%.6f" % val_check)
            if val < best_val_n:
                best_val_n = val
                best_a_n = a_opt.copy()
            obj_n = 2.0 * n * best_val_n / S**2
            if obj_n < global_best:
                global_best = obj_n
                global_a = best_a_n.copy()

    print("After Phase 3: obj=%.6f" % global_best)

    # Try n=16384 if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120 and given is not None and len(given) >= n_alt:
        print("Trying n=%d with %.0fs" % (n_alt, remaining))
        init_alt = given[:n_alt].copy() if len(given) >= n_alt else upscale_sequence(given, n_alt)
        a_alt, val_alt = fista_backtrack(init_alt, S_alt, remaining * 0.5, rng,
                                          beta_init=5.0, beta_max=500.0,
                                          beta_rate=0.004, momentum_decay_start=0.4)
        obj_alt = 2.0 * n_alt * val_alt / S_alt**2
        print("  n=%d obj=%.6f" % (n_alt, obj_alt))
        if obj_alt < global_best:
            global_best = obj_alt
            global_a = a_alt.copy()

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
