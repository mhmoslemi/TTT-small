Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smooth peak approximation
with projected gradient descent, multi-restart, and adaptive parameter scheduling.

Approach:
1. Use softmax-based smooth approximation of the max function
2. Efficient gradient computation via FFT
3. Projected gradient descent with adaptive learning rate
4. Multi-restart with diverse initializations
5. Progressive sharpening of the softmax (annealing)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full self-convolution of a via FFT using precomputed FFT size N."""
    n = len(a)
    size = 2 * n - 1
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a, 1 << (2 * n - 1).bit_length())
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_smooth_gradient(a, w, A_cached, N):
    """
    Compute gradient of smooth peak approximation g_t(a).
    grad[i] = 2 * sum_j w[i+j] * a[j]
    Computed via cross-correlation using FFT.
    Reuses A_cached = fft(a, N).
    """
    W = fft(w, N)
    h = np.real(ifft(W * np.conj(A_cached)))
    return 2.0 * h[:len(a)]


def spectral_refine(a, N, strength):
    """
    One step of spectral domain refinement: compress |FFT(a)|^2 toward uniform.
    This helps with global shaping (reducing overall spectral concentration).
    """
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    p_med = np.median(P)
    if p_med < 1e-30:
        return a
    ratio = P / p_med
    # Compress: high bins down, low bins up
    P_new = P.copy()
    above = ratio > 1.0
    P_new[above] = P[above] - strength * P[above] * (ratio[above] - 1.0) / ratio[above].max()
    below = ratio < 0.5
    P_new[below] = P[below] + strength * P[below] * (1.0 - ratio[below])
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    S = a.sum()
    a_new = project_simplex(a_new, S)
    return a_new

    for phase in range(n_phases):
        phase_start = time.monotonic()
        t_lo = t_min * (t_max / t_min) ** (phase / n_phases)
        t_hi = t_min * (t_max / t_min) ** ((phase + 1) / n_phases)

        it = 0
        while time.monotonic() - phase_start < phase_duration:
            # Check global budget
            if time.monotonic() - start_time > budget:
                break

            it += 1
            progress_in_phase = (time.monotonic() - phase_start) / phase_duration
            t = t_lo * (t_hi / t_lo) ** progress_in_phase

            # Compute autocorrelation
            A_cached = fft(a, N)
            c = np.real(ifft(A_cached * A_cached))[:2 * n - 1]

            # Compute softmax weights (numerically stable)
            c_max = c.max()
            exp_c = np.exp(t * (c - c_max))
            w = exp_c / exp_c.sum()

            # Compute gradient
            grad = compute_smooth_gradient(a, w, A_cached, N)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Backtracking line search
            lr = 2.0 * S / (n * gnorm)
            improved = False
            for bt in range(8):
                a_trial = project_simplex(a - lr * grad, S)
                if np.any(a_trial < -1e-15):
                    lr *= 0.5
                    continue
                sc = compute_score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    no_improve_count = 0
                    improved = True
                    break
                lr *= 0.5

            if not improved:
                no_improve_count += 1
                # Try spectral refinement periodically
                if no_improve_count > 1000 and no_improve_count % 1000 == 0:
                    a_spec = spectral_refine(a, N, 0.1)
                    sc_spec = compute_score(a_spec)
                    if sc_spec < best_score:
                        best_score = sc_spec
                        best_a = a_spec.copy()
                        a = a_spec
                        no_improve_count = 0
                    elif rng.random() < 0.1:
                        # Random restart from best
                        a = project_simplex(best_a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)

                if no_improve_count > max_no_improve:
                    # Restart from best with larger perturbation
                    a = project_simplex(best_a + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
                    no_improve_count = 0

            # Periodic print
            if it % 5000 == 0:
                print("    phase %d, it %d, t=%.1f, score=%.6f" % (phase, it, t, best_score))

    return best_a, best_score


def make_inits(n, rng, provided_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Provided sequence (if matches length)
    if provided_seq is not None and len(provided_seq) == n:
        ps = np.clip(provided_seq, 0, 1000)
        if ps.sum() > 0.01:
            inits.append(ps.copy())

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Random
    inits.append(rng.random(n) + 0.05)

    # Smooth random (low-pass)
    noise = rng.standard_normal(n)
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(0.2 + 0.6 * smooth)

    # Multi-sine
    inits.append(0.5 + 0.2 * np.sin(2 * np.pi * 3 * tt) +
                0.15 * np.sin(2 * np.pi * 7 * tt) +
                0.1 * np.sin(2 * np.pi * 11 * tt))

    # Chirp
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt * tt)))

    # Banded: high values in middle
    center = np.exp(-0.5 * ((tt - 0.5) * 3.0) ** 2)
    inits.append(0.2 + 0.6 * center)

    # Two bumps
    inits.append(0.1 + 0.4 * np.exp(-0.5 * ((tt - 0.3) * 5.0) ** 2) +
                0.4 * np.exp(-0.5 * ((tt - 0.7) * 5.0) ** 2))

    # Random phase (simulates flat spectrum)
    A = fft(np.ones(n) + rng.normal(0, 0.3, n))
    P = np.abs(A) ** 2
    P = P + rng.exponential(0.1 * P.mean(), n)  # perturb
    phase = rng.uniform(0, 2 * np.pi, n)
    # Make even/odd symmetry for real signal
    phase[n // 2] = 0
    if n % 2 == 0:
        phase[n // 2] = 0
    A_new = np.sqrt(P) * np.exp(1j * phase)
    # Enforce Hermitian symmetry for real output
    if n % 2 == 0:
        A_new[n // 2:] = np.conj(A_new[:n // 2][::-1])
    else:
        A_new[(n // 2 + 1):] = np.conj(A_new[:n // 2][::-1])
    a_rand_phase = np.real(ifft(A_new))
    a_rand_phase = np.maximum(a_rand_phase, 0)
    if a_rand_phase.sum() > 0.01:
        inits.append(a_rand_phase / a_rand_phase.sum())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses smooth gradient descent with adaptive softmax, multi-restart,
    and spectral refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    final_deadline = t_global_start + budget_s - 30  # leave 30s for verification

    print("Starting optimization with budget %.0fs" % budget_s)

    # Get provided sequence
    provided = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        print("Provided sequence length: %d" % len(provided))
    except Exception:
        pass

    # Determine lengths to try
    lengths_to_try = [8192]
    if provided is not None and len(provided) in [4096, 16384, 32768]:
        lengths_to_try.insert(0, len(provided))
    # Add 16384 if not already there
    if 16384 not in lengths_to_try:
        lengths_to_try.append(16384)

    global_best = np.inf
    global_a = None

    for n in lengths_to_try:
        if time.monotonic() > final_deadline:
            break

        remaining = final_deadline - time.monotonic()
        if remaining < 60:
            break

        # Allocate budget: more time for 8192, less for others
        if n == 8192:
            alloc = min(remaining * 0.6, remaining - 30)
        elif n == 16384:
            alloc = min(remaining * 0.3, remaining - 30)
        else:
            alloc = min(remaining * 0.5, remaining - 30)

        if alloc < 30:
            alloc = remaining * 0.4
            if alloc < 20:
                break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        S = np.sqrt(2.0 * float(n))
        N = 1 << (3 * n - 1).bit_length()

        inits = make_inits(n, rng, provided)
        n_inits = len(inits)

        # More time for the provided sequence init, less for others
        per_init_budget = alloc / (n_inits * 1.5)  # leave extra

        for init_idx, init in enumerate(inits):
            if time.monotonic() > final_deadline:
                break
            rem = final_deadline - time.monotonic()
            if rem < 30:
                break

            # First init (provided) gets more time
            if init_idx == 0 and init is not inits[-1]:
                init_budget = min(per_init_budget * 2.5, rem - 30)
            else:
                init_budget = min(per_init_budget, rem - 30)

            if init_budget < 10:
                break

            print("  Init %d/%d, budget=%.0fs" % (init_idx + 1, n_inits, init_budget))
            a_result, score = optimize_one(init, n, S, init_budget, rng, N)

            if a_result is not None and score < global_best:
                global_best = score
                global_a = a_result.copy()
                print("  New global best: %.6f (n=%d, init=%d)" % (global_best, n, init_idx))
            else:
                print("  Best for this init: %.6f (global: %.6f)" % (score, global_best))

    # Final polish: run a short optimization from the best found
    if global_a is not None and time.monotonic() < final_deadline:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        # Renormalize
        s = global_a.sum()
        if s > 0.01:
            global_a = global_a * (S_best / s)
        N_best = 1 << (3 * n_best - 1).bit_length()
        polish_budget = min(60.0, final_deadline - time.monotonic())
        if polish_budget > 10:
            print("=== Final polish, budget=%.0fs ===" % polish_budget)
            a_pol, sc_pol = optimize_one(global_a, n_best, S_best, polish_budget, rng, N_best)
            if sc_pol < global_best:
                global_best = sc_pol
                global_a = a_pol
                print("  Polished score: %.6f" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64)
        global_best = 2.0
        print("Using fallback uniform sequence")

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        global_best = 2.0

    print("Final: length=%d, score=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t_global_start))
    return [float(x) for x in global_a]
