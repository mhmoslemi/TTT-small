"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization with aggressive spectral shaping,
FISTA with adaptive LSE + exact subgradient hybrid, fast coordinate
descent with peak-focused pair selection, and multi-length exploration.

intelligent coordinate descent focusing on top-peak lags,
and adaptive restarts with diverse perturbations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient for max(c) approximation."""
    cm = c.max()
    shifted = beta * (c - cm)
    max_shifted = shifted.max()
    e = np.exp(shifted - max_shifted)
    tot = e.sum()
    lse = cm + max_shifted / beta + np.log(tot) / beta
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse, grad, w


def exact_subgradient_at_peak(a, c):
    """Exact subgradient at the argmax lag of c."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    for i in range(lo, hi + 1):
        grad[i] += a[k - i]
    return grad, k


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=5000.0,
                       ramp_iters=2000, spectral_interval=500, restart_threshold=2000):
    """FISTA with adaptive LSE, hybrid subgradient, spectral re-tracing, and restarts."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        if it < ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / float(ramp_iters))
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-13:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient source
        use_exact = (it > 500 and rng.random() < min(0.4, 0.001 * it))
        if use_exact:
            grad, k_peak = exact_subgradient_at_peak(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (gnorm + 1e-10)
            step = min(step, 0.1 / max(1.0, beta * 0.01))
        else:
            _, grad, _ = lse_value_and_grad(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)

        # Backtracking: reduce step if no recent improvement
        if it - last_improve > 1000:
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * (S / s)
        a = x_new
        t_mom = t_new

        # Spectral re-tracing
        if it % spectral_interval == 0:
            c = fft_convolve_self(y)
            c_mean = c.mean()
            # Identify top peaks
            n_peaks = min(10, len(c) // 4)
            peak_indices = np.argsort(c)[-n_peaks:]
            peak_mask = np.zeros_like(c)
            peak_mask[peak_indices] = 1.0
            # FFT of peak mask tells us which frequencies to suppress
            peak_fft = fft(peak_mask, fsize)
            # Get current spectrum
            y_pad = np.zeros(fsize)
            y_pad[:n] = y
            Y = fft(y_pad)
            P = np.abs(Y) ** 2
            # Suppress frequency components that align with peaks
            suppression = 1.0 - 0.3 * np.abs(peak_fft) ** 2 / max(np.abs(peak_fft) ** 2, 1e-30).sum() * n_peaks
            suppression = np.clip(suppression, 0.3, 1.0)
            P_new = P * suppression
            mag_new = np.sqrt(P_new)
            phase = Y / np.maximum(np.abs(Y), 1e-30)
            Y_new = phase * mag_new
            y_new = np.real(ifft(Y_new))[:n]
            y_new = np.maximum(y_new, 0.0)
            s = y_new.sum()
            if s > 1e-10:
                y_new = y_new * (S / s)
                # Only accept if improvement
                c_new = fft_convolve_self(y_new)
                if c_new.max() <= cur + 0.01 * cur:
                    y = y_new
                    t_mom = 1.0

        # Periodic noise injection
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

        # Stagnation restart
        if it - last_improve > restart_threshold and it > 5000:
            # Restart from best with perturbation
            noise_scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it
            restart_threshold = int(restart_threshold * 1.3)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_waterfilling(a_init, n, S, max_time, rng):
    """Water-filling approach on the power spectrum to flatten autocorrelation."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    gamma = 0.1

    while time.monotonic() - start < max_time:
        it += 1

        y_pad = np.zeros(fsize)
        y_pad[:n] = a
        Y = fft(y_pad)
        P = np.abs(Y) ** 2
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best - 1e-13:
            best = cur
            best_a = a.copy()

        c_mean = c.mean()
        # Compute excess above threshold
        threshold = c_mean * 1.1
        excess = np.maximum(c - threshold, 0.0)
        if excess.max() < 1e-10:
            excess = c - c_mean
            excess = np.maximum(excess, 0.0)
        if excess.max() < 1e-10:
            break

        # FFT of excess gives frequency content to suppress
        E = fft(excess, fsize)
        E_mag = np.abs(E) ** 2
        E_norm = E_mag / max(E_mag.sum(), 1e-30)

        # Water-filling: reduce P where excess has high frequency content
        P_new = P * (1.0 - gamma * E_norm * fsize / n)
        P_new = np.maximum(P_new, 1e-10)

        # Also add a small amount of "noise" to fill in the spectrum
        P_noise = P.mean() * rng.random(fsize) * 0.05
        P_new = P_new + P_noise

        # Normalize total power
        P_new = P_new * (P.sum() / P_new.sum())

        mag_new = np.sqrt(P_new)
        phase = Y / np.maximum(np.abs(Y), 1e-30)
        Y_new = phase * mag_new
        a_new = np.real(ifft(Y_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Adaptive gamma
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            gamma = min(0.5, gamma * 1.05)
        else:
            gamma = max(0.01, gamma * 0.8)

        a = a_new

    return best_a, best


def fast_coordinate_descent(a_init, n, S, max_time, rng, n_peaks=20):
    """Fast coordinate descent focusing on peak lags."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stagnant = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Get top peak lags
        peak_lags = np.argsort(c)[-n_peaks:]

        n_trials = min(100 + it // 5, 500)
        for _ in range(n_trials):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose a pair
            if rng.random() < 0.4:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))
            else:
                # Focus on regions that contribute to peaks
                k_peak = int(peak_lags[int(rng.integers(0, n_peaks))])
                lo = max(0, k_peak - n + 1)
                hi = min(k_peak, n - 1)
                j = int(rng.integers(lo, hi + 1)) if hi > lo else int(rng.integers(0, n))
                m = int(rng.integers(0, n))

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute effect on top peaks only
            # delta transfer from m to j
            # c_new[k] ≈ c[k] + delta * (a[j+k]+a[j-k] - a[m+k]-a[m-k])
            d_peaks = np.zeros(n_peaks)
            for pi, k in enumerate(peak_lags):
                val_j = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val_j += a[jk]
                if 0 <= jmk < n:
                    val_j += a[jmk]
                val_m = 0.0
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val_m += a[mk]
                if 0 <= mmk < n:
                    val_m += a[mmk]
                d_peaks[pi] = val_j - val_m

            # Also check c[0] (energy)
            # c[0] changes by: (a[j]+d)^2 - a[j]^2 + (a[m]-d)^2 - a[m]^2
            # = 2(a[j]-a[m])*d + 2d^2
            # Find optimal delta via bisection on the max of peak values
            lo_d = -a[j] * 0.99
            hi_d = a[m] * 0.99

            if lo_d >= hi_d:
                continue

            # Golden section search on peak values
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            def peak_val_at_delta(dlt):
                # Evaluate max at peak lags + c[0]
                vals = c[peak_lags] + dlt * d_peaks
                c0_new = c[0] + 2.0 * (a[j] - a[m]) * dlt + 2.0 * dlt * dlt
                return max(vals.max(), c0_new)

            fc = peak_val_at_delta(c_g)
            fd = peak_val_at_delta(d_g)

            for _ in range(25):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = peak_val_at_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = peak_val_at_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = max(lo_d, min(hi_d, delta))

            if abs(delta) < 1e-14:
                continue

            # Apply and verify
            a[j] += delta
            a[m] -= delta
            if a[j] < 0 or a[m] < 0:
                a[j] -= delta
                a[m] += delta
                continue

            # Update c incrementally for peak lags
            for pi, k in enumerate(peak_lags):

                jk = int(j) + k
                jmk = int(j) - k
                if 0 <= jk < n:
                    val_j += a[jk]

                val_m = 0.0

            # Just recompute: c[0] = sum(a^2)

            cur_max = max(c.max(), c[0])
            if c[0] > c.max():
                cur_max = c[0]
            else:
                cur_max = c.max()

            if cur_max < best - 1e-12:
                best = cur_max
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[j] -= delta
                a[m] += delta
                for pi, k in enumerate(peak_lags):
                    val_j = 0.0
                    jk = int(j) + k
                    jmk = int(j) - k
                    if 0 <= jk < n:
                        val_j += a[jk]
                    if 0 <= jmk < n:
                        val_j += a[jmk]
                    val_m = 0.0
                    mk = int(m) + k
                    mmk = int(m) - k
                    if 0 <= mk < n:
                        val_m += a[mk]
                    if 0 <= mmk < n:
                        val_m += a[mmk]
                    c[k] -= delta * (val_j - val_m)
                c[0] = np.sum(a * a)

        # Periodic full recompute

            c = fft_convolve_self(a)
            actual = c.max()

        it += 1

        # every 500 iters, increase alpha
        if it % 500 == 0 and current_alpha_idx < len(alpha_schedule) - 1:
            current_alpha_idx += 1

        alpha = alpha_schedule[current_alpha_idx]

        y_pad = np.zeros(fsize)
        y_pad[:n] = a
        Y = fft(y_pad)
        P = np.abs(Y) ** 2
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best - 1e-13:
            best = cur
            best_a = a.copy()

        c_mean = c.mean()
        # Target: make c as flat as possible
        # In frequency domain: P should be as flat as possible
        # But we need to account for the triangular window effect

        # Method: iterative power reduction at peaks
        n_top = min(20, len(c) // 4)
        top_indices = np.argsort(c)[-n_top:]
        top_vals = c[top_indices]
        target = c_mean * 1.05

        # Compute which frequency bins to suppress
        excess = np.zeros(fsize)
        for idx in top_indices:
            excess[idx] = max(0.0, c[idx] - target)

        E = fft(excess, fsize)
        E_mag = np.abs(E) ** 2
        E_total = E_mag.sum()
        if E_total < 1e-30:
            E_norm = np.zeros(fsize)
        else:
            E_norm = E_mag / E_total

        # Suppress
        P_new = P * (1.0 - alpha * E_norm * n_top)
        P_new = np.maximum(P_new, P.mean() * 0.1)

        # Add small noise to avoid degeneracy
        P_new += P.mean() * 0.01 * rng.random(fsize)

        # Normalize power
        P_new = P_new * (P.sum() / P_new.sum())

        mag_new = np.sqrt(P_new)
        phase = Y / np.maximum(np.abs(Y), 1e-30)
        Y_new = phase * mag_new
        a_new = np.real(ifft(Y_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Verify
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur * 1.01:
            a = a_new
        else:
            # Revert, reduce alpha
            a_current_alpha = alpha
            # Just keep current a

    return best_a, best


def diverse_initialization(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) == n:
        inits.append(("parent", parent.copy()))

    # Sinc-like
    inits.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.02)))

    # Gaussian
    inits.append(("gauss", np.exp(-6.0 * (tt - 0.5) ** 2)))

    # Cosine window
    inits.append(("cos", np.maximum(0.5 * (1 + np.cos(2.0 * np.pi * tt)), 0.02)))

    # Raised cosine
    inits.append(("raised_cos", 0.5 + 0.5 * np.cos(2.0 * np.pi * tt)))

    # Random
    inits.append(("rand", rng.random(n) + 0.05))

    # Bimodal (peaks at edges)
    inits.append(("bimodal", np.exp(-20.0 * tt ** 2) + np.exp(-20.0 * (tt - 1.0) ** 2)))

    # Multi-frequency
    inits.append(("multi_freq", 1.0 + 0.3 * np.sin(4.0 * np.pi * tt) + 0.2 * np.sin(8.0 * np.pi * tt)))
    inits[-1] = (inits[-1][0], np.maximum(inits[-1][1], 0.01))

    # White noise (high frequency dominant)
    noise = rng.standard_t(3, n)
    inits.append(("heavy_tail", np.maximum(noise, -0.5)))

    return inits


def upsample_sequence(a, old_n, new_n):
    """Upsample a sequence to a new length using interpolation."""
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def downsample_sequence(a, old_n, new_n):
    """Downsample by block averaging."""
    n_blocks = old_n // new_n
    if n_blocks < 1:
        return a[:new_n].copy()
    trimmed = a[:n_blocks * new_n]
    return trimmed.reshape(new_n, n_blocks).mean(axis=1)


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine primary length
    if parent is not None and 2048 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes (%d)" % len(parent) if parent is not None else "no"))

    S = np.sqrt(2.0 * n)

    global_best = np.inf
    global_a = None

    def check_and_update(a_cand, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_cand, len(a_cand))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        eval_score = 2.0 * len(a_norm) * score / (a_norm.sum() ** 2)
        if score < global_best - 1e-13:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: max_c=%.6f, score=%.6f (t=%.0fs)" % (
                label, len(a_norm), score, eval_score, elapsed))
        return score

    # Phase 1: Multi-start with quick FISTA
    print("\n=== Phase 1: Multi-start FISTA ===")
    p1_budget = min(400.0, (deadline - t_start) * 0.2)
    inits = diverse_initialization(n, rng, parent)
    n_starts = len(inits)
    time_per_start = p1_budget / n_starts

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - t_start) * 0.6:
            break
        t_alloc = max(10.0, time_per_start)
        S_i = np.sqrt(2.0 * n)
        a_opt, _ = run_fista_adaptive(arr, n, S_i, t_alloc, rng,
                                       beta_init=30.0, beta_max=2000.0,
                                       ramp_iters=int(t_alloc * 20),
                                       spectral_interval=200, restart_threshold=1000)
        check_and_update(a_opt, "p1_" + name)

    print("After Phase 1: best_score=%.6f" % (2.0 * n * global_best / (S ** 2)))

    # Phase 2: Deep optimization on best candidates
    print("\n=== Phase 3: Deep spectral + FISTA ===")
    if global_a is not None:
        S_best = np.sqrt(2.0 * n)
        p2_budget = (deadline - time.monotonic()) * 0.35
        if p2_budget > 50:
            # Split between FISTA and spectral
            t_fista = p2_budget * 0.55
            t_spec = p2_budget * 0.45

            # Deep FISTA
            a_fin, _ = run_fista_adaptive(global_a, n, S_best, t_fista, rng,
                                           beta_init=100.0, beta_max=8000.0,
                                           ramp_iters=int(t_fista * 15),
                                           spectral_interval=300, restart_threshold=3000)
            check_and_update(a_fin, "p2_fista")

            # Spectral water-filling
            a_sf, _ = spectral_waterfilling(global_a, n, S_best, t_spec * 0.5, rng)
            check_and_update(a_sf, "p2_waterfill")

            # Spectral retrace deep
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_sr, _ = spectral_retrace_deep(global_a, n, S_best, min(t_spec * 0.5, remaining * 0.5), rng)
                check_and_update(a_sr, "p2_retrace")

    print("After Phase 2: best_score=%.6f" % (2.0 * n * global_best / (S ** 2)))

    # Phase 3: Coordinate descent polish
    print("\n=== Phase 3: Coordinate Descent ===")
    if global_a is not None:
        S_best = np.sqrt(2.0 * n)
        p3_budget = (deadline - time.monotonic()) * 0.3
        if p3_budget > 30:
            a_cd, _ = fast_coordinate_descent(global_a, n, S_best, p3_budget, rng, n_peaks=30)
            check_and_update(a_cd, "p3_cd")

            # Second pass with more peaks
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_cd2, _ = fast_coordinate_descent(global_a, n, S_best, min(remaining * 0.5, 200), rng, n_peaks=50)
                check_and_update(a_cd2, "p3_cd2")

    print("After Phase 3: best_score=%.6f" % (2.0 * n * global_best / (S ** 2)))

    # Phase 4: Final FISTA polish
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        S_best = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_fin, _ = run_fista_adaptive(global_a, n, S_best, remaining * 0.6, rng,
                                           beta_init=500.0, beta_max=15000.0,
                                           ramp_iters=int(remaining),
                                           spectral_interval=200, restart_threshold=5000)
            check_and_update(a_fin, "p4_final")

            # Quick CD
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_cd3, _ = fast_coordinate_descent(global_a, n, S_best, remaining2 * 0.7, rng, n_peaks=40)
                check_and_update(a_cd3, "p4_cd")

    # Phase 5: Try alternate length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 180 and global_a is not None:
        print("\n=== Phase 5: Alternate Length ===")
        # Try 2x and 0.5x
        alt_lengths = [n * 2, n // 2]
        for alt_n in alt_lengths:
            if time.monotonic() > deadline - 60:
                break
            if not (2048 <= alt_n <= 65536):
                continue
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > n:
                alt_start = upsample_sequence(global_a, n, alt_n)
            else:
                alt_start = downsample_sequence(global_a, n, alt_n)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alloc = min(120.0, (deadline - time.monotonic()) * 0.5)
            a_alt, _ = run_fista_adaptive(alt_start, alt_n, S_alt, t_alloc, rng,
                                           beta_init=30.0, beta_max=3000.0,
                                           ramp_iters=int(t_alloc * 10),
                                           spectral_interval=300, restart_threshold=2000)
            check_and_update(a_alt, "p5_n%d" % alt_n)

            # Quick CD on alternate
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_alt_cd, _ = fast_coordinate_descent(a_alt, alt_n, S_alt, min(60, remaining2 * 0.3), rng, n_peaks=30)
                check_and_update(a_alt_cd, "p5_cd_n%d" % alt_n)

    # Final verification and return
    print("\n=== Final ===")
    if global_a is None:
        n_fb = 1024
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 1024
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verification: score=%.6f, length=%d, time=%.0fs" % (
        ver, len(global_a), time.monotonic() - t_start))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(1024) * (np.sqrt(2.0 * 1024) / 1024)
        final_list = [float(x) for x in global_a]

    return final_list
