def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)

    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho

    a_pad[:n] = a
    A = fft(a_pad)

"""
Minimize 2*n*max_conv/sum^2 using hybrid L-BFGS + Top-K gradient + Mirror Descent.
Key improvements: top-k peak targeting (avoids LSE ill-conditioning), adaptive k scheduling,
multi-length exploration (n=8192, 16384), Adam complement, and intensive local optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns (c of length 2n-1, A of length N)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def compute_toppk_grad(a, A, c, top_k, weight_decay=1.0):
    """Compute gradient of weighted top-k peaks. weight_decay<1 gives decaying weights."""
    n = len(a)
    N = len(A)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Decay weights: w_i = top_vals[i] / sum(top_vals)
    w = top_vals / top_vals.sum()
    # Apply decay for emphasis on top peaks
    if weight_decay < 1.0:
        order = np.arange(top_k)
        w = w * (weight_decay ** order)
        w = w / w.sum()
    w_full = np.zeros(N)
    for i in range(top_k):
        pos = int(top_idx[i]) % N
        w_full[pos] += w[i]
    W = fft(w_full)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, top_vals[0], top_idx


def compute_lse_grad(a, A, c, beta):
    """Compute LSE-smoothed gradient."""
    n = len(a)
    N = len(A)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        ysy = np.dot(y_i, s_i)
        if abs(ysy) < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / ysy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    ysy_last = np.dot(y_last, s_last)
    if abs(ysy_last) > 1e-30:
        gamma = ysy_last / max(np.dot(y_last, y_last), 1e-30)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        ysy = np.dot(y_i, s_i)
        if abs(ysy) < 1e-30:
            continue
        beta_i = np.dot(y_i, r) / ysy
        r = r + (alphas[i] - beta_i) * s_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g / gn
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_toppk_block(a_init, n, S, N, max_time, rng,
                       top_k_init=5, top_k_max=200, top_k_growth=200,
                       m_lbfgs=15, weight_decay=0.95):
    """L-BFGS with top-k peak gradient and adaptive k."""
    a = safe_normalize(a_init, n, S)
    top_k = top_k_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    c, A = compute_conv_full(a, N)
    g, cur_max, _ = compute_toppk_grad(a, A, c, top_k, weight_decay)
    if cur_max < best_cmax:
        best_cmax = cur_max
        best_a = a.copy()

    step_scale = S / n
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adapt top_k
        top_k = min(top_k_init + it // top_k_growth, top_k_max)
        if top_k > top_k_max:
            top_k = top_k_max

        # Compute gradient with current top_k
        g, cur_peak, _ = compute_toppk_grad(a, A, c, top_k, weight_decay)

        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        direction = direction / dir_norm

        # Line search
        alpha_ls = step_scale
        a_new = a
        c_new = c
        cmax_new = cur_max
        accepted = False

        for ls in range(20):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, A_ls = compute_conv_full(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= cur_peak + 1e-12 or ls == 19:
                a_new = candidate
                c_new = c_ls
                A_new = A_ls
                cmax_new = cm_ls
                if cm_ls <= cur_peak + 1e-12:
                    accepted = True
                    break
            alpha_ls *= 0.6

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        # Update L-BFGS history
        g_new, _, _ = compute_toppk_grad(a_new, A_new, c_new, top_k, weight_decay)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        else:
            if len(s_hist) > 5 and rng.random() < 0.3:
                s_hist = s_hist[-m_lbfgs // 2:]
                y_hist = y_hist[-m_lbfgs // 2:]

        a = a_new
        c = c_new
        A = A_new
        cur_peak = cmax_new

        # Periodic perturbation to escape local minima
        if it % 300 == 0 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            c, A = compute_conv_full(a, N)

        # Reduce step scale slowly
        step_scale *= 0.9997

    return best_a, best_cmax


def adam_toppk_block(a_init, n, S, N, max_time, rng,
                      top_k_init=10, top_k_max=300,
                      lr_init=1.0, lr_decay=0.9995,
                      beta1=0.9, beta2=0.999):
    """Adam optimizer with top-k peak gradient."""
    a = safe_normalize(a_init, n, S)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    top_k = top_k_init
    lr = lr_init
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_init + it // 50, top_k_max)
        c, A = compute_conv_full(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()

        g, _, _ = compute_toppk_grad(a, A, c, top_k, 1.0)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break
        g = g / gnorm  # normalize gradient

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * (g * g)
        m_hat_hat = m_hat / (1 - beta1 ** (it + 1))
        v_hat_hat = v_hat / (1 - beta2 ** (it + 1))
        step = lr * m_hat_hat / (np.sqrt(v_hat_hat) + 1e-8)

        a_new = project_simplex(a - step, S)
        a = a_new
        lr *= lr_decay

        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)

    return best_a, best_cmax


def mirror_descent_block(a_init, n, S, N, max_time, rng,
                          top_k=20, eta_init=0.5, eta_decay=0.999):
    """Mirror descent (multiplicative) with top-k peak gradient."""
    a = np.maximum(a_init.copy(), 1e-15)
    a = a * (S / a.sum())
    eta = eta_init
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = compute_conv_full(a, N)
        cur_max = c.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()

        g, _, _ = compute_toppk_grad(a, A, c, top_k, 1.0)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        # Mirror descent: a_j <- a_j * exp(-eta * g_j / gnorm)
        log_a = np.log(a) - (eta / gnorm) * g
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            break
        a_new = a_new * (S / a_new_sum)
        a = np.maximum(a_new, 1e-15)
        a = a * (S / a.sum())
        eta *= eta_decay

        if it % 1000 == 0 and rng.random() < 0.1:
            a = np.maximum(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), 1e-15)
            a = a * (S / a.sum())

    return best_a, best_cmax


def spectral_step(a, N, top_k, alpha):
    """One spectral shaping step."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_asymmetric(n, rng):
    """Asymmetric initialization: left-heavy or right-heavy."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if rng.random() < 0.5:
        a = 0.3 + 0.7 * np.exp(-4 * tt) + 0.1 * rng.standard_normal(n)
    else:
        a = 0.3 + 0.7 * np.exp(-4 * (1 - tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Fourier-based initialization with random phase."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_window(n, rng):
    """Raised cosine window with random perturbation."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.2 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
