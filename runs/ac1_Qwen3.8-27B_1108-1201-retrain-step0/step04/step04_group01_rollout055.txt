"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 for non-negative sequences.
Uses FISTA with adaptive LSE/hinge smoothing, away-step perturbations,
coordinate descent polishing, and multi-length exploration.
Key improvements: cleaner gradient computation, adaptive threshold annealing,

import numpy as np
import time


def fft_convolve(a):
    """Compute full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    return np.real(np.fft.ifft(A * A))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - S) / idx
    if np.any(cond):
        rho = int(np.max(np.where(cond)[0])) + 1
    else:
        rho = n
    theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_gradient(a, w):
    """
    Compute gradient of sum_k w[k]*conv[k] w.r.t. a.
    w has length 2n-1, a has length n.
    Returns vector of length n.
    """
    n = len(a)
    a_flip = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    Wf = np.fft.fft(w, fsize)
    Af = np.fft.fft(a_flip, fsize)
    conv_result = np.real(np.fft.ifft(Wf * Af))
    h = conv_result[n - 1: 2 * n - 2]
    return 2.0 * h


def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 450:
        beta = 450.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    w = e / e.sum()
    obj = cm + np.log(e.sum()) / beta
    grad = conv_gradient(a, w)
    return c, obj, grad, beta


def hinge_objective_and_grad(a, T, p=2.0):
    """Compute sum((c_k - T)^p_+) and its gradient."""
    n = len(a)
    c = fft_convolve(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return c, 0.0, np.zeros(n)
    if p == 1.0:
        w = (excess > 0).astype(np.float64)
    else:
        w = p * excess ** (p - 1)
    wsum = w.sum()
    if wsum < 1e-30:
        return c, float(np.sum(excess ** p)), np.zeros(n)
    w = w / wsum
    w_full = np.maximum(c - T, 0.0)
    if w_full.sum() < 1e-30:
        return c, 0.0, np.zeros(n)
    w_full = w_full / w_full.sum()
    grad = conv_gradient(a, w_full)
    obj_val = float(np.sum(excess ** p))
    return c, obj_val, grad


def fista_optimize(a_init, max_time, rng, beta_start=20.0, beta_end=3000.0,
                   mode_schedule="auto", use_hinge_after=0.6):
    """
    FISTA optimization with adaptive smoothing.
    mode_schedule: 'auto' switches from LSE to hinge at use_hinge_after fraction of time.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    a = project_simplex(a, S)

    y = a.copy()
    t_k = 1.0
    start_time = time.monotonic()
    best_c_max = fft_convolve(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_counter = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        elapsed_frac = (time.monotonic() - start_time) / max_time

        # Adaptive beta schedule
        if mode_schedule == "auto":
            if elapsed_frac < use_hinge_after:
                frac = elapsed_frac / use_hinge_after
                beta = beta_start * (beta_end / beta_start) ** frac
                use_hinge = False
            else:
                frac = (elapsed_frac - use_hinge_after) / (1.0 - use_hinge_after)
                beta = beta_end
                use_hinge = True
        else:
            beta = beta_start * (1.0 + 0.01 * it)
            use_hinge = False

        c = fft_convolve(y)
        if c.max() < best_c_max - 1e-12:
            best_c_max = c.max()
            best_a = y.copy()
            last_improve = it
        else:
            stall_counter += 1

        # Compute gradient
        if use_hinge:
            T = c.mean() + 0.3 * (c.max() - c.mean())
            c_h, _, grad = hinge_objective_and_grad(y, T, p=2.0)
            # Lipschitz estimate for hinge
            L = 2.0 * T * 2.0 + 4.0 * y.max()
            step = 1.0 / max(L, 1e-6)
        else:
            _, _, grad, beta_eff = lse_objective_and_grad(y, beta)
            L = 2.0 * beta_eff * 2.0 * y.max() + 0.01
            step = 1.0 / max(L, 1e-6)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum
        y = project_simplex(y, S)
        a = x_new
        t_k = t_new

        # Periodic restart with noise
        if it % 500 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * max(0.1, 1.0 - elapsed_frac)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

        # Away-step perturbation when stalled
        if stall_counter > 2000 and it % 100 == 0:
            c_peak = fft_convolve(a)
            peak_idx = np.argmax(c_peak)
            # Find which element contributes most to the peak
            shift = peak_idx
            valid = (shift < n)
            if valid:
                # a[peak_idx - j] is the contribution pattern
                # The element that most increases the peak when increased is at j where a[peak_idx-j] is large
                # We want to DECREASE the element at the position that contributes most
                contrib = np.zeros(n)
                lo = max(0, peak_idx - n + 1)
                hi = min(peak_idx, n - 1)
                for j in range(lo, hi + 1):
                    if 0 <= peak_idx - j < n:
                        contrib[j] = a[peak_idx - j]
                if contrib.max() > 0:
                    j_max = int(np.argmax(contrib))
                    # Decrease a[j_max] slightly, redistribute to the min element
                    j_min = int(np.argmin(a))
                    delta = 0.01 * a[j_max]
                    if a[j_max] - delta > 0:
                        a[j_max] -= delta
                        a[j_min] += delta
                        a = np.maximum(a, 0)
                        a = project_simplex(a, S)
                        y = a.copy()
                        t_k = 1.0
                        stall_counter = 0

    # Final check
    c_final = fft_convolve(a)
    if c_final.max() < best_c_max:
        best_c_max = c_final.max()
        best_a = a.copy()

    return best_a, best_c_max


def coordinate_descent_refine(a_init, max_time, rng, num_rounds=5):
    """
    Coordinate descent: try adjusting pairs of elements to reduce the peak.
    Uses the insight that reducing elements that contribute to the peak
    and increasing elements that don't can help.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    start_time = time.monotonic()
    c = fft_convolve(a)
    best_max = c.max()
    best_a = a.copy()

    for rnd in range(num_rounds):
        if time.monotonic() - start_time > max_time:
            break
        for _ in range(n // 4):
            if time.monotonic() - start_time > max_time:
                break
            # Find the current peak
            peak_idx = int(np.argmax(c))
            # Compute contribution pattern for this peak
            # contrib[j] = a[peak_idx - j] (how much a[j] contributes to c[peak_idx])
            contrib = np.zeros(n)
            for j in range(n):
                k = peak_idx - j
                if 0 <= k < n:
                    contrib[j] = a[k]
                else:
                    # Check the other side: a[j] * a[peak_idx - j] 
                    # Actually for conv: c[k] = sum_j a[j]*a[k-j]
                    # Contribution of a[j] to c[peak_idx] = a[peak_idx-j] if valid
                    pass
            # Also check: c[peak_idx] = sum_j a[j]*a[peak_idx-j]
            # The contribution of a[j] is a[peak_idx-j] (if 0<=peak_idx-j<n)
            # plus a[j-peak_idx] (if 0<=j-peak_idx<n) -- wait, that's for correlation
            # For convolution: c[k] = sum_j a[j]*a[k-j]
            # d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # = a[peak_idx-j] (if valid) + a[j-peak_idx] (if valid)
            contrib2 = np.zeros(n)
            for j in range(n):
                if 0 <= peak_idx - j < n:
                    contrib2[j] += a[peak_idx - j]
                if 0 <= j - peak_idx < n:
                    contrib2[j] += a[j - peak_idx]

            # Find the element that most contributes to the peak (to decrease)
            j_decrease = int(np.argmax(contrib2))
            # Find an element that contributes least (to increase)
            j_increase = int(np.argmin(contrib2))
            if j_decrease == j_increase:
                j_increase = (j_decrease + 1) % n
            if a[j_decrease] <= 1e-10:
                continue

            # Try several ratios of decrease/increase
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = frac * a[j_decrease]
                if a[j_decrease] - delta < 0:
                    continue
                a_trial = a.copy()
                a_trial[j_decrease] -= delta
                a_trial[j_increase] += delta
                if a_trial[j_increase] > 10:  # sanity check
                    continue
                c_trial = fft_convolve(a_trial)
                if c_trial.max() < best_max - 1e-12:
                    a = a_trial.copy()
                    c = c_trial
                    best_max = c.max()
                    best_a = a.copy()
                    break

    return best_a, best_max


def spectral_equalization(a_init, max_time, rng):
    """
    Try to equalize the power spectrum using iterative projections.
    Since max(conv) relates to peak of |A|^2 (for the cyclic case),
    we try to flatten |A|^2 while maintaining non-negativity.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    start_time = time.monotonic()
    best_max = fft_convolve(a).max()
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()
    mu = 0.01  # mixing parameter

    for it in range(100000):
        if time.monotonic() - start_time > max_time:
            break
        # Compute current spectrum
        A = np.fft.fft(a, fsize)
        power = np.abs(A) ** 2
        power[0] = 0  # ignore DC
        p_mean = power.mean()

        # Target: reduce peaks in power spectrum
        # Gradient of max(power) w.r.t. a in time domain
        p_max_idx = int(np.argmax(power))
        if power[p_max_idx] < p_mean * 1.01:
            break  # spectrum is flat enough

        # The phase of A[p_max] determines the time-domain gradient
        phase = np.angle(A[p_max])
        # Gradient of |A[k]|^2 w.r.t. time is proportional to cos(2*pi*k*t/n + 2*phase)
        t = np.arange(n)
        grad = np.cos(2.0 * np.pi * p_max * t / fsize + 2.0 * phase)

        # Also add gradient from the linear convolution peak
        c = fft_convolve(a)
        peak_idx = int(np.argmax(c))
        contrib = np.zeros(n)
        for j in range(n):
            if 0 <= peak_idx - j < n:
                contrib[j] += a[peak_idx - j]
            if 0 <= j - peak_idx < n:
                contrib[j] += a[j - peak_idx]

        # Combined gradient
        combined_grad = grad + 0.5 * contrib / max(contrib.max(), 1e-10)

        # Step
        step = 0.001 * S / n
        a_new = a - step * combined_grad
        a_new = project_simplex(a_new, S)

        c_new = fft_convolve(a_new)
        if c_new.max() < best_max:
            best_max = c_new.max()
            best_a = a_new.copy()
        a = a_new

        # Periodic noise
        if it % 1000 == 0:
            noise = 0.001 * S / np.sqrt(n) * rng.normal(0, 1, n)
            a = project_simplex(a + noise, S)

    return best_a, best_max

def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        if len(given_seq) == n:
            inits.append(given_seq.copy())
        elif len(given_seq) > n:
            # Downsample by averaging
            ratio = len(given_seq) // n
            small = given_seq[:n * ratio].reshape(n, ratio).mean(axis=1)
            inits.append(small)
        else:
            # Upsample
            t_old = np.linspace(0, 1, len(given_seq))
            small = np.interp(tt, t_old, given_seq)
            inits.append(small)

    # Various structured inits
    inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * np.sin(4 * np.pi * tt) + 0.3 * np.cos(6 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.1)
    inits.append(np.exp(-2.0 * (tt - 0.5) ** 2 / 0.1) * 0.5 + 0.5)
    inits.append(rng.random(n) + 0.05)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 7 * tt) + 0.2 * np.sin(2 * np.pi * 13 * tt), 0.01))

    # Block init
    nblocks = 8
    blen = n // nblocks
    heights = rng.uniform(0.5, 1.5, nblocks)
    block = np.repeat(heights, blen)
    if len(block) < n:
        block = np.concatenate([block, np.full(n - len(block), heights[-1])])
    inits.append(block[:n])

    return [np.maximum(x, 1e-4) for x in inits]


def objective_value(a):
    """Compute the actual objective."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve(a)
    return 2.0 * n * b.max() / (s * s)

                                           use_hinge_after=0.5)
            obj = 2.0 * n * c_max  # since sum=1
            if c_max < best_max_n:
                best_max_n = c_max
                best_a_n = a_opt.copy()
                print("  Init %d: max_b=%.8f, obj=%.6f" % (i, c_max, obj))

        if best_a_n is None:
            best_a_n = np.ones(n) / n
            best_max_n = fft_convolve(best_a_n).max()

        # Phase 2: Deep optimization
        deep_time = min(600.0, max(60.0, (deadline - time.monotonic()) * 0.4))
        print("  Deep FISTA: %.0fs" % deep_time)
        a_deep, c_deep = fista_optimize(best_a_n, deep_time, rng,
                                         beta_start=50.0, beta_end=10000.0,
                                         use_hinge_after=0.4)
        if c_deep < best_max_n:
            best_max_n = c_deep
            best_a_n = a_deep.copy()
            print("  After deep FISTA: max_b=%.8f" % best_max_n)

        # Phase 3: Coordinate descent refinement
        cd_time = min(120.0, max(20.0, (deadline - time.monotonic()) * 0.15))
        if cd_time > 20:
            print("  Coordinate descent: %.0fs" % cd_time)
            a_cd, c_cd = coordinate_descent_refine(best_a_n, cd_time, rng, num_rounds=3)
            if c_cd < best_max_n:
                best_max_n = c_cd
                best_a_n = a_cd.copy()
                print("  After coord descent: max_b=%.8f" % best_max_n)

        # Phase 4: Multi-start polishing
        ms_time = min(180.0, max(20.0, (deadline - time.monotonic()) * 0.2))
        if ms_time > 30:
            print("  Multi-start: %.0fs" % ms_time)
            a_ms, c_ms = multi_start_refine(best_a_n, ms_time, rng, num_starts=4)
            if c_ms < best_max_n:
                best_max_n = c_ms
                best_a_n = a_ms.copy()
                print("  After multi-start: max_b=%.8f" % best_max_n)

        # Update global best
        obj_n = 2.0 * len(best_a_n) * best_max_n  # sum=1 so obj = 2n*max_b
        if obj_n < global_best_obj:
            global_best_obj = obj_n
            global_best_a = best_a_n.copy()
            print("  New global best: obj=%.6f (n=%d)" % (global_best_obj, len(best_a_n)))

    # Final intensive polishing
    if global_best_a is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(global_best_a)
        print("\n=== Final polishing: %.0fs ===" % remaining)

        # High-beta FISTA
        t1 = remaining * 0.4
        a_fin, c_fin = fista_optimize(global_best_a, t1, rng,
                                       beta_start=500.0, beta_end=20000.0,
                                       use_hinge_after=0.3)
        if c_fin < fft_convolve(global_best_a).max():
            global_best_a = a_fin.copy()
            global_best_obj = 2.0 * n_best * c_fin
            print("  After final FISTA: obj=%.6f" % global_best_obj)

        # Coordinate descent
        t2 = remaining * 0.3
        if t2 > 15:
            a_cd2, c_cd2 = coordinate_descent_refine(global_best_a, t2, rng, num_rounds=5)
            if c_cd2 < fft_convolve(global_best_a).max():
                global_best_a = a_cd2.copy()
                global_best_obj = 2.0 * n_best * c_cd2
                print("  After final CD: obj=%.6f" % global_best_obj)

        # One more multi-start with very small perturbations
        t3 = deadline - time.monotonic()
        if t3 > 30:
            a_ms2, c_ms2 = multi_start_refine(global_best_a, t3, rng, num_starts=3)
            if c_ms2 < fft_convolve(global_best_a).max():
                global_best_a = a_ms2.copy()
                global_best_obj = 2.0 * n_best * c_ms2
                print("  After final multi-start: obj=%.6f" % global_best_obj)

    # Verify and return
    if global_best_a is None:
        global_best_a = np.ones(256) / 256.0

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256) / 256.0

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: obj=%.6f" % ver)

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Fallback.")
        global_best_a = np.ones(256) / 256.0
        final_list = [float(x) for x in global_best_a]

    print("Returning sequence of length %d, obj=%.6f" % (len(global_best_a), ver))
    return final_list
