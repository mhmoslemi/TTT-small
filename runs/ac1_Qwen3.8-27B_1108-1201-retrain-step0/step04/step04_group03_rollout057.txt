"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Targeted peak reduction with LSE polishing, diversification restarts,
and adaptive cycle length. Uses exact max tracking and surgical gradient steps
on the most constraining lags.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a, fsize=None):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if fsize is None:
        fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta, fsize=None):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    if fsize is None:
        fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta):
    """Value of LSE approximation."""
    cm = c.max()
    # Numerically stable
    shifted = beta * (c - cm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    return cm + np.log(np.exp(shifted).sum()) / beta


def targeted_peak_reduction(f, n, S, top_k=20, lr=0.01, fsize=None):
    """
    Surgically reduce the top-k peaks of the self-convolution.
    For each top peak at lag k*, compute the gradient of c[k*] w.r.t. f
    and take a step to reduce it.
    """
    c = fft_convolve_self(f, fsize)
    
    # Find top-k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Compute combined gradient: weighted sum of gradients of top lags
    grad = np.zeros(n)
    weights = np.zeros(2*n-1)
    for k in top_idx:
        weights[k] = 1.0
    
    # For each top lag k, the gradient of c[k] w.r.t. f_j is:
    # d(c[k])/d(f_j) = f[k-j] (if valid) + f[j-k] (if valid and j != k-j)
    # More precisely: c[k] = sum_j f[j]*f[k-j]
    # dc[k]/df[j] = f[k-j] (from the term where j'=j)
    #             + f[j-k] (from the term where j'=k-j, if k-j != j)
    # But since we're summing over all valid j', the gradient is:
    # dc[k]/df[j] = f[k-j] if 0 <= k-j < n
    # Plus, if j != k-j (i.e., j != k/2), there's also a contribution
    # from the term j'=k-j: dc[k]/df[j] += f[k-j] (same value)
    # Wait, let me think again.
    # c[k] = sum_{j': valid} f[j'] * f[k-j']
    # d(c[k])/d(f[j]) = sum over terms where f[j] appears:
    #   - Term j'=j: gives f[k-j] (if k-j is valid)
    #   - Term j'=k-j: gives f[j] ... no wait.
    # Actually: d/d(f[j]) [f[j'] * f[k-j']] 
    #   = delta(j', j) * f[k-j'] + f[j'] * (-delta(k-j', j))
    #   = delta(j', j) * f[k-j'] - f[j'] * delta(j', k-j)
    #   = delta(j', j) * f[k-j] - f[k-j] * delta(j', k-j)
    # Summing over j':
    #   = f[k-j] (from j'=j, if k-j valid) - f[k-(k-j)] (from j'=k-j, if k-j valid and k-j != j)
    #   = f[k-j] - f[j] (if both valid and j != k-j)
    #   = f[k-j] (if j = k-j, i.e., j = k/2)
    
    # Hmm, this is getting complicated. Let me just use the FFT approach.
    # The gradient of c[k] w.r.t. f is the cross-correlation of the "delta at k" with f.
    # In the frequency domain: FFT(delta_k) * conj(FFT(f)) -> IFFT
    # But delta_k has FFT = exp(-2*pi*i*k/n) (for circular) which is just a phase.
    
    # Actually, for linear convolution, the gradient of c[k] w.r.t. f is:
    # g_j = f[k-j] (if 0 <= k-j < n)  [from the j'=j term]
    #     + f[k-j] (if 0 <= k-j < n and k-j != j)  [from the j'=k-j term]
    # Wait no. Let me redo.
    
    # c[k] = sum_{j'=max(0,k-n+1)}^{min(k, n-1)} f[j'] * f[k-j']
    # 
    # d(c[k])/d(f[j]) for a specific j:
    # f[j] appears in the sum wherever j' = j or k-j' = j (i.e., j' = k-j).
    # If j' = j is in range: contributes f[k-j]
    # If j' = k-j is in range: contributes f[j] ... wait no.
    # 
    # The term is f[j'] * f[k-j']. If j' = j, the term is f[j] * f[k-j], and the derivative w.r.t. f[j] is f[k-j].
    # If j' = k-j, the term is f[k-j] * f[j], and the derivative w.r.t. f[j] is f[k-j].
    # So if both j' = j and j' = k-j are in range (and j != k-j), the contribution is 2*f[k-j].
    # If j = k-j (i.e., j = k/2), the term is f[j]*f[j] = f[j]^2, and the derivative is 2*f[j].
    
    # So: d(c[k])/d(f[j]) = 
    #   0 if j < max(0, k-n+1) or j > min(k, n-1) and k-j < 0 or k-j >= n
    #   2*f[k-j] if j != k-j and both j and k-j are in [0, n-1]
    #   2*f[j] if j = k-j (i.e., k = 2j) and j is in range
    #   f[k-j] if only one of j or k-j is in range (this happens at the boundaries)
    
    # This is getting too complex for a direct computation. Let me use the FFT trick.
    # The gradient of (f*f)[k] w.r.t. f is the cross-correlation of "delta at position k" with f,
    # which in the linear convolution sense is just f shifted.
    
    # Actually, the simplest way: the gradient of the LSE of c w.r.t. f is:
    # sum_k w_k * d(c[k])/d(f)
    # where w_k are the LSE weights.
    # And d(c[k])/d(f) is the "unrolled" cross-correlation, which can be computed via FFT.
    
    # The standard result: if c = f*f (linear convolution), then
    # d(c)/d(f) in the sense that sum_k w_k * c[k] = <w, f*f>,
    # the gradient w.r.t. f is 2 * (w * f) (linear cross-correlation), with proper boundary handling.
    # More precisely, the gradient is the linear convolution of w with f, times 2,
    # but with w zero-padded to the right.
    
    # Actually, the standard result for linear convolution:
    # If c = conv(f, g), then d(c[k])/d(f[j]) = g[k-j] (if valid).
    # For self-convolution c = conv(f, f): d(c[k])/d(f[j]) = f[k-j] + f[j-k] (with boundary conditions)
    # = 2*f[k-j] (interior) or f[k-j] (boundary)
    
    # Using FFT: the gradient of sum_k w[k]*c[k] w.r.t. f is:
    # 2 * IFFT(FFT(w_padded) * conj(FFT(f_padded)))[:n]
    # where w_padded is w zero-padded to the full convolution length.

    A = fft(f, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    f_new = project_simplex(f - lr * grad, S)
    return f_new, c


def run_lse_optimize(f_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.005, beta_max=5000.0,
                     use_targeted=False, targeted_freq=100, top_k=10):
    """
    LSE-based optimization with optional targeted peak reduction.
    """
    f = np.clip(f_init, 0.0, 1000.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    y = f.copy()
    t_fista = 1.0
    fsize = 1 << (2 * n - 1).bit_length()
    
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y, fsize)
        cur = c.max()
        if cur < best:
            best = cur
            best_f = y.copy()
            last_improve = it
        
        # LSE gradient step
        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t_fista = t_new
        
        # Targeted peak reduction periodically
        if use_targeted and it % targeted_freq == 0:
            # Find the top lags and do a surgical update
            c_top = fft_convolve_self(y, fsize)
            top_idx = np.argsort(c_top)[::-1][:top_k]
            weights_top = np.zeros(2*n-1)
            for k in top_idx:
                weights_top[k] = 1.0
            
            Wt = fft(weights_top, fsize)
            At = fft(y, fsize)
            grad_top = 2.0 * np.real(ifft(Wt * np.conj(At)))[:n]
            
            # Take a small step in the direction that reduces the top peaks
            lr_top = 0.005 * S / max(np.linalg.norm(grad_top), 1e-15)
            y_new = project_simplex(y - lr_top * grad_top, S)
            
            # Check if this helps
            c_new = fft_convolve_self(y_new, fsize)
            if c_new.max() < cur:
                y = y_new
                f = y.copy()
                t_fista = 1.0
        
        # Stuck detection and restart
        if it - last_improve > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            f = y.copy()
            t_fista = 1.0
            last_improve = it
    
    return best_f, best


def run_diversified_search(f_init, n, S, max_time, rng):
    """
    Try multiple different perturbation strategies.
    """
    best_f = f_init.copy()
    best_val = fft_convolve_self(f_init).max()
    
    strategies = [
        ("gaussian", lambda f: f + rng.normal(0, 0.01 * f.mean(), n)),
        ("multiplicative", lambda f: f * np.exp(rng.normal(0, 0.05, n))),
        ("mirror_break", None),  # special
        ("block_swap", None),  # special
    ]
    
    per_strategy = max_time / len(strategies)
    
    for name, fn in strategies:
        if time.monotonic() + per_strategy > time.monotonic() + max_time:
            break
        
        if name == "mirror_break":
            # Break the symmetry pattern: swap random pairs
            f_pert = best_f.copy()
            for _ in range(n // 100):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                f_pert[i], f_pert[j] = f_pert[j], f_pert[i]
        elif name == "block_swap":
            # Swap random blocks
            f_pert = best_f.copy()
            block_size = max(1, n // 20)
            for _ in range(5):
                i = rng.integers(0, n - block_size)
                j = rng.integers(0, n - block_size)
                f_pert[i:i+block_size], f_pert[j:j+block_size] = f_pert[j:j+block_size].copy(), f_pert[i:i+block_size].copy()
        else:
            f_pert = fn(best_f)
        
        f_pert = np.clip(f_pert, 0, None)
        f_pert = project_simplex(f_pert, S)
        
        # Optimize the perturbed sequence
        f_opt, val = run_lse_optimize(f_pert, n, S, per_strategy * 0.8, rng,
                                        beta_init=50.0, beta_rate=0.003, beta_max=3000.0,
                                        use_targeted=True, targeted_freq=200, top_k=10)
        
        if val < best_val:
            best_val = val
            best_f = f_opt.copy()
            print("  Strategy '%s' improved: %.6f" % (name, best_val))
    
    return best_f, best_val
