"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using LSE-smoothed
gradient descent with adaptive parameters, BB step size, and LP-based
direction finding for escape from local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Compute gradient of LSE approximation of max of c."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(a))
    w = e / tot
    # gradient of LSE wrt a[i] = sum_k w_k * a[k-i] (valid indices)
    # = cross-correlation of w and a
    n = len(a)
    size_c = len(c)  # 2n-1
    fsize = 1 << (size_c + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # We need grad[i] = sum_k w[k] * a[k-i] for valid k, i
    # This is the cross-correlation of w and a
    # corr[i] = sum_k w[k] * conj(a[k-i]) = ifft(fft(w) * conj(fft(a)))
    # But since a is real, conj(a) = a
    cross = np.real(ifft(W * np.conj(A)))
    return cross[:n]

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def run_lse_optimization(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0):
    """LSE-smoothed gradient descent with BB step size."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = project_simplex(a, S)
    
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    beta = beta_init
    
    # BB step size
    step = S / (n * 0.1)  # initial step
    prev_grad = None
    prev_a = None
    
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Increase beta over time
        beta = min(beta_max, beta_init + 0.5 * np.sqrt(it))
        
        # Compute LSE gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # BB step size
        if prev_grad is not None and prev_a is not None:
            s = a - prev_a
            y = grad - prev_grad
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-15 and sy > 1e-15:
                step_bb = np.dot(s, s) / sy
                step = 0.5 * (step + step_bb)
        
        # Clamp step
        step = min(step, 2.0 * S / (n * gnorm + 1e-15))
        step = max(step, 1e-10)
        
        prev_grad = grad.copy()
        prev_a = a.copy()
        
        a_new = project_simplex(a - step * grad, S)
        
        # Check for improvement
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new <= cur_obj:
            a = a_new
            c = c_new
            step = min(step * 1.1, 5.0 * S / n)
        else:
            # Backtrack
            for _ in range(10):
                step *= 0.5
                a_try = project_simplex(a - step * grad, S)
                c_try = fft_convolve_self(a_try)
                if 2.0 * n * c_try.max() / S**2 <= cur_obj:
                    a = a_try
                    c = c_try
                    break
            else:
                # Stuck
                stagnant += 5
                if stagnant > 100:
                    # Perturb
                    noise = 0.005 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise, n), S)
                    c = fft_convolve_self(a)
                    step = 0.1 * S / n
                    stagnant = 0
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    LSE: it={it}, obj={best_obj:.6f}, beta={beta:.1f}, step={step:.4f}, time={elapsed:.0f}s")
    
    return best_a, best_obj

"""
Minimize 2*n*max(autoconv)/sum^2 using LSE-smoothed gradient descent with
adaptive step sizes, frequency-domain initialization, multi-scale upscaling,
and LP-inspired direction finding. Key insight: optimal sequences must break
symmetry to achieve objective < 2.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):
    """Full linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Compute gradient of LSE(beta, c) with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    size_c = len(c)
    fsize = 1 << (size_c + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))
    return cross[:n]


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def upsample_sequence(a, factor):
    """Upsample a sequence by inserting zeros and smoothing."""
    n = len(a)
    new_n = n * factor
    new_a = np.zeros(new_n)
    # Place original values at regular intervals
    for i in range(n):
        new_a[i * factor] = a[i]
    # Smooth with a small kernel to avoid isolated spikes
    if factor > 1:
        kernel = np.ones(factor) / factor
        new_a = np.convolve(new_a, kernel, mode='same')
    return new_a


def freq_domain_init(n, rng, mode=0):
    """Initialize in frequency domain for flat-spectrum sequences."""
    if mode == 0:
        # Slightly perturbed flat magnitude
        mag = 1.0 + 0.1 * rng.standard_normal(n)
        phase = rng.standard_normal(n)
        # Enforce conjugate symmetry for real result
        mag = np.abs(mag)
        X = mag * np.exp(1j * phase)
        X[0] = np.abs(X[0])
        for k in range(1, n // 2 + 1):
            X[n - k] = np.conj(X[k])
            mag[k] = mag[n - k]
            phase[k] = -phase[n - k]
        X = mag * np.exp(1j * phase)
        a = np.real(ifft(X))
    elif mode == 1:
        # Cosine-tapered flat spectrum
        mag = np.ones(n)
        taper = 0.8 + 0.2 * np.cos(np.pi * np.arange(n) / n)
        mag *= taper
        phase = 0.0
        X = mag * np.exp(1j * phase)
        a = np.real(ifft(X))
    else:
        # Random phase, flat magnitude
        mag = 1.0
        phase = rng.standard_normal(n)
        X = mag * np.exp(1j * phase)
        X[0] = np.real(X[0])
        for k in range(1, n // 2 + 1):
            X[n - k] = np.conj(X[k])
        a = np.real(ifft(X))
    return np.maximum(a, 0.0)


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    t = np.arange(n) / n
    if mode == 0:
        # Asymmetric ramp with modulation
        a = (1.0 + t) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * t + rng.uniform(0, 6)))
    elif mode == 1:
        # Expanding window (breaks left-right symmetry)
        a = np.exp(-2.0 * np.abs(t - 0.3) ** 2 / 0.1) + 0.3 * np.exp(-2.0 * np.abs(t - 0.7) ** 2 / 0.05)
    elif mode == 2:
        # Two-scale structure
        a = 1.0 + 0.4 * np.sin(2 * np.pi * t) + 0.2 * np.sin(2 * np.pi * 7 * t + 0.5)
    elif mode == 3:
        # Skewed Gaussian mixture
        a = np.exp(-3.0 * (t - 0.25) ** 2 / 0.1) + 0.5 * np.exp(-3.0 * (t - 0.65) ** 2 / 0.15) + 0.1
    elif mode == 4:
        # Polynomial with asymmetry
        a = 1.0 + 0.5 * t + 0.3 * t ** 2 - 0.4 * t ** 3
    elif mode == 5:
        # Step-like with smooth transitions
        a = 1.0 / (1.0 + np.exp(-8.0 * (t - 0.4))) + 0.5 / (1.0 + np.exp(-8.0 * (t - 0.75)))
    else:
        a = np.maximum(1.0 + rng.standard_normal(n) * 0.3, 0.01)
    return np.maximum(a, 0.01)

def run_lp_direction(a, n, S, M, rng, max_iter=50):
    """
    Approximate LP direction: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k.
    Uses multiplicative weights / projected gradient.
    """
    g = np.ones(n)
    size = 2 * n - 1

    for it in range(max_iter):
        ag = fft_convolve_cross(a, g)
        # Scale to check constraint
        ag_scaled = ag / max(ag.max(), 1e-30)
        violated = ag > M + 1e-10

        if not np.any(violated):
            break

        # Subgradient descent on the constraints
        viol_idx = np.where(violated)[0]
        # Sort by violation magnitude
        viol_vals = ag[viol_idx] - M
        order = np.argsort(-viol_vals)
        viol_idx = viol_idx[order[:min(200, len(order))]]
        viol_amt = viol_vals[order[:min(200, len(order))]]

        red = np.zeros(n)
        for k, amt in zip(viol_idx, viol_amt):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1)
            red[i_vals] += a[k - i_vals] * amt

        rnorm = np.linalg.norm(red)
        if rnorm < 1e-15:
            break

        lr = 0.1 * g.sum() / (n * rnorm)
        g = np.maximum(g - lr * red, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None
        g = g / gs * S

    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs * S

                a_try = project_simplex(a_micro - step_micro * grad, S_best)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n_best * c_try.max() / S_best ** 2
                if obj_try < cur:
                    a_micro = a_try
                    c_micro = c_try
                    step_micro = min(step_micro * 1.02, 0.5)
                else:
                    step_micro *= 0.7
                    if step_micro < 1e-6:
                        break

            a_micro = normalize(a_micro, n_best)
            obj_final = compute_objective(a_micro)
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_micro.copy()

    # Final verification
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %.8f" % ver)
    print("Final: n=%d, obj=%.8f, total_time=%.0fs" %
          (len(global_a), ver, time.monotonic() - t0))

    if not np.isfinite(ver):
        final_list = [float(x) for x in np.ones(256) * (np.sqrt(2 * 256) / 256)]

    return final_list
