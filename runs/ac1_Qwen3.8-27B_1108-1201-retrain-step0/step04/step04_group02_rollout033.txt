"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LSE-annealed FISTA with adaptive beta, multi-length search,
structured perturbations, and peak-targeted refinement.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Grad of LSE(beta * c) / beta, normalized."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(a))
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def structured_perturb(a, n, S, rng):
    """Apply structured perturbation."""
    strategy = rng.integers(0, 6)
    a_new = a.copy()
    
    if strategy == 0:
        freq = rng.uniform(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * mod
    elif strategy == 1:
        t = np.arange(n) / n
        mod = 1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.cos(8 * np.pi * t + rng.uniform(0, 1))
        a_new = a * mod
    elif strategy == 2:
        nblocks = rng.integers(4, 32)
        block_mod = rng.uniform(0.5, 1.5, nblocks)
        mod = np.repeat(block_mod, n // nblocks + 1)[:n]
        a_new = a * mod
    elif strategy == 3:
        # Frequency domain perturbation
        A = fft(a, 1 << (n - 1).bit_length())
        noise = rng.normal(0, 0.1, len(A)) + 1j * rng.normal(0, 0.1, len(A))
        A = A * (1 + noise)
        a_new = np.real(ifft(A))[:n]
    elif strategy == 4:
        # Shift mass from center to edges
        t = np.arange(n) / n
        mod = 1.0 + 0.5 * (t**2 + (1-t)**2) / 0.5
        a_new = a * mod
    else:
        # Gaussian noise
        a_new = a + rng.normal(0, 0.05 * S / np.sqrt(n), n)
    
    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses variance-minimizing gradient for global flattening, LSE-annealed FISTA
for progressive peak focus, direct peak shaving, KKT-inspired polishing,
multi-length search, and structured perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(a))
    w = e / tot
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def variance_gradient(a, c, S):
    """Gradient of sum_k (c[k] - mean)^2 w.r.t. a."""
    n = len(a)
    mu = S * S / (2 * n - 1)
    d = c - mu
    fsize = 1 << (len(d) + n - 1).bit_length()
    D = fft(d, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(D * np.conj(A)))[:n]


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def run_lse_fista(a_init, n, max_time, rng, beta_start=1.0, beta_end=2000.0, n_beta_steps=20, use_variance_first=False):
    """
    FISTA with LSE smoothing, annealing beta. Optionally starts with variance minimization.
    """
    S = np.sqrt(2.0 * n)
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    # Optional variance phase
    if use_variance_first:
        var_time = min(30.0, max_time * 0.15)
        var_start = time.monotonic()
        step = 1.0 / (4.0 * S)
        it = 0
        while time.monotonic() - var_start < var_time:
            it += 1
            c = fft_convolve_self(a)
            grad = variance_gradient(a, c, S)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
            if obj_try < 2.0 * n * c.max() / S**2 + 1e-15:
                a = a_try
                step = min(step * 1.05, 2.0 / max(gnorm, 1e-15))
            else:
                step *= 0.85
            if it % 500 == 0:
                cur = compute_objective(a)
                if cur < best_obj:
                    best_obj = cur
                    best_a = a.copy()

    # LSE annealing phase
    if n_beta_steps > 1:
        betas = np.geomspace(beta_start, beta_end, n_beta_steps)
    else:
        betas = [beta_start]

    for beta in betas:
        if time.monotonic() - start > max_time - 15:
            break

        y = a.copy()
        t_mom = 1.0
        step = 1.0 / max(4.0 * beta * 2.0 * S, 1.0)
        step = min(step, 1.0)

        max_iter = 20000
        it = 0
        stagnation = 0

        while it < max_iter and time.monotonic() - start < max_time - 15:
            it += 1
            c_y = fft_convolve_self(y)
            grad = lse_gradient(y, c_y, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            x_new = project_simplex(y - step * grad, S)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            if it % 500 == 0:
                cur_obj = compute_objective(a)
                if cur_obj < best_obj - 1e-10:
                    best_obj = cur_obj
                    best_a = a.copy()
                    stagnation = 0
                else:
                    stagnation += 1
                if stagnation > 3:
                    break

        cur_obj = compute_objective(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def run_peak_shaving(a_init, n, max_time, rng):
    """
    Direct peak shaving with adaptive top-k weighting.
    """
    S = np.sqrt(2.0 * n)
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    step = 0.5
    fail_count = 0

    while time.monotonic() - start < max_time - 10:
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            fail_count = 0

        # Adaptive k: start broad, narrow down
        elapsed = time.monotonic() - start
        frac = elapsed / max_time
        k = max(1, int(16 * (1.0 - frac) + 1))
        k = min(k, 32)

        idx = np.argsort(c)[::-1][:k]
        weights = np.zeros(len(c))
        # Weight by excess over mean
        mean_b = (S * S) / (2 * n - 1)
        for i in idx:
            weights[i] = max(0.0, c[i] - mean_b) + 0.1
        wsum = weights.sum()
        if wsum < 1e-300:
            weights[idx] = 1.0 / len(idx)
        else:
            weights = weights / wsum

        fsize = 1 << (len(weights) + n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2

        if obj_new < cur_obj - 1e-12:
            a = a_new
            step = min(step * 1.08, 3.0)
            fail_count = 0
        else:
            step *= 0.75
            fail_count += 1
            if fail_count > 30:
                noise_scale = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 0.5
                fail_count = 0
            if fail_count > 80:
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 1.0
                fail_count = 0

    return best_a, best_obj


def run_fine_tune(a_init, n, max_time, rng):
    """
    Very fine-grained optimization: alternate between top-1 and top-4 subgradients
    with tiny steps, using backtracking line search.
    """
    S = np.sqrt(2.0 * n)
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    step = 0.05
    it = 0
    fail_count = 0

    while time.monotonic() - start < max_time - 5:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            fail_count = 0

        # Alternate between top-1 and top-4
        k = 1 if (it % 3 == 0) else 4
        idx = np.argsort(c)[::-1][:k]

        fsize = 1 << (len(c) + n - 1).bit_length()
        W = np.zeros(len(c))
        W[idx] = 1.0 / k
        Wf = fft(W, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(Wf * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        s = step
        a_new = None
        for _ in range(10):
            a_try = project_simplex(a - s * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < cur_obj - 1e-14 * cur_obj:
                a_new = a_try
                break
            s *= 0.5

        if a_new is not None:
            a = a_new
            step = min(s * 1.5, 1.0)
            fail_count = 0
        else:
            fail_count += 1
            step *= 0.7
            if fail_count > 50:
                # Small perturbation
                noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                step = 0.05
                fail_count = 0

    return best_a, best_obj


def kkt_polish(a_init, n, max_time, rng):
    """
    KKT-inspired polishing: identify active peaks, solve approximate KKT system
    via iterative linearization.
    """
    S = np.sqrt(2.0 * n)
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    c = fft_convolve_self(a)
    M = c.max()

    for iteration in range(200):
        if time.monotonic() - start > max_time - 5:
            break

        # Identify active peaks (within 2% of max)
        threshold = M * 0.98
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Cap at 20 active peaks for tractability
        active = np.argsort(c)[::-1][:min(len(active), 20)]

        # Build the linear system iteratively
        # (T*a)[i] = sum_{k in active} a[k-i] * 1(0<=k-i<n) = lambda
        # We solve this as a fixed-point: given a, compute the "target" a_new

        # For each i: target[i] = lambda (same for all i)
        # Current: cur[i] = sum_{k in active} a[k-i] (valid)
        # We want cur = lambda * 1

        # Compute current T*a
        Ta = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            j_vals = np.arange(lo, hi + 1)
            Ta[j_vals] += a[k - j_vals]

        # lambda estimate
        lambda_est = Ta.mean()

        # Update: a_new should satisfy (T*a_new) = lambda_est
        # This is a linear system. Use a few Jacobi iterations.
        # Actually, let's use a simpler approach: gradient descent on the KKT residual
        # residual = Ta - lambda_est * 1
        # grad of sum(residual^2) w.r.t. a[i] = 2 * sum_{k: k-i valid} residual[k]... 
        # This gets complicated. Let's just use a damped update.

        # Simple approach: move a in the direction that equalizes Ta
        # We want Ta to be constant. Gradient of 0.5*||Ta||^2 w.r.t. a:
        # d(0.5*Ta^2)/da[i] = sum_{k in active} Ta_contribution... 
        # Actually, let's just use the variance gradient restricted to active peaks.

        # Compute gradient to equalize active peaks
        d = np.zeros(2 * n - 1)
        d[active] = c[active] - c[active].mean()
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        D = fft(d, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(D * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Damped step
        step = 0.1 * S / (gnorm * np.sqrt(n))
        step = min(step, 0.5)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new / S**2
        obj_cur = 2.0 * n * M / S**2

        if obj_new < obj_cur:
            a = a_new
            c = c_new
            M = M_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
        else:
            # Try smaller step
            a_new = project_simplex(a - 0.1 * step * grad, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / S**2
            if obj_new < obj_cur:
                a = a_new
                c = c_new
                M = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()

    return best_a, best_obj


def structured_perturb(a, n, S, rng):
    """Apply structured perturbation to escape local minima."""
    strategy = rng.integers(0, 8)
    a_new = a.copy()

    if strategy == 0:
        freq = rng.uniform(1, 30)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * mod
    elif strategy == 1:
        t = np.arange(n) / n
        mod = 1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.15 * np.cos(8 * np.pi * t + rng.uniform(0, 2))
        a_new = a * mod
    elif strategy == 2:
        nblocks = rng.integers(4, 64)
        block_mod = rng.uniform(0.5, 1.5, nblocks)
        mod = np.repeat(block_mod, n // nblocks + 1)[:n]
        a_new = a * mod
    elif strategy == 3:
        A = fft(a, 1 << (n - 1).bit_length())
        mask = rng.random(len(A)) > 0.3
        noise = (rng.normal(0, 0.08, len(A)) + 1j * rng.normal(0, 0.08, len(A))) * mask
        A = A * (1 + noise)
        a_new = np.real(ifft(A))[:n]
    elif strategy == 4:
        t = np.arange(n) / n
        mod = 1.0 + 0.3 * (2 * t - 1)**6
        a_new = a * mod
    elif strategy == 5:
        # Shift mass
        shift = rng.integers(1, max(2, n // 10))
        a_new = np.roll(a, shift)
    elif strategy == 6:
        # Multiply by exponential
        t = np.arange(n) / n
        rate = rng.uniform(-5, 5)
        mod = np.exp(rate * t)
        a_new = a * mod
    else:
        a_new = a + rng.normal(0, 0.03 * S / np.sqrt(n), n)

    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

        a_fine, s_fine = run_fine_tune(global_a, n, phase3_time, rng)
        a_fine = project_simplex(a_fine, S)
        s_fine = compute_objective(a_fine)
        if s_fine < global_best:
            global_best = s_fine
            global_a = a_fine.copy()
            print("  Fine tuning result: %.6f" % global_best)

    # Phase 4: KKT polish (10% of time)
    remaining = deadline - time.monotonic()
    phase4_time = min(remaining * 0.3, 150.0)
    if phase4_time > 10:
        print("=== Phase 4: KKT polish (%.0fs) ===" % phase4_time)
        a_kkt, s_kkt = kkt_polish(global_a, n, phase4_time, rng)
        a_kkt = project_simplex(a_kkt, S)
        s_kkt = compute_objective(a_kkt)
        if s_kkt < global_best:
            global_best = s_kkt
            global_a = a_kkt.copy()
            print("  KKT polish result: %.6f" % global_best)

    # Phase 5: Multi-restart with perturbations (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print("=== Phase 5: Multi-restart (%.0fs remaining) ===" % remaining)
        n_restarts = max(2, int(remaining / 40))
        per_restart = remaining / (n_restarts + 1)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            perturb = structured_perturb(global_a, n, S, rng)
            t_r = min(per_restart * 0.85, remaining - 20)
            if t_r < 10:
                break
            a_r, s_r = run_lse_fista(perturb, n, t_r, rng,
                                      beta_start=20.0, beta_end=3000.0,
                                      n_beta_steps=12, use_variance_first=False)
            a_r = project_simplex(a_r, S)
            s_r = compute_objective(a_r)
            if s_r < global_best:
                global_best = s_r
                global_a = a_r.copy()
                print("  Restart %d: %.6f" % (i, global_best))

    # Phase 6: Final micro-optimization
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Phase 6: Final micro (%.0fs) ===" % remaining)
        a_fin, s_fin = run_fine_tune(global_a, n, remaining - 5, rng)
        a_fin = project_simplex(a_fin, S)
        s_fin = compute_objective(a_fin)
        if s_fin < global_best:
            global_best = s_fin
            global_a = a_fin.copy()
            print("  Final: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            n_fb = 256
            global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)

    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
