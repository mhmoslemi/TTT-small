    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / (restart_every * 4))
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.8)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        if use_peak:
            grad = peak_grad(y, c, k=min(peak_k, len(c) // 4))
        else:
            grad = lse_grad(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_peak:
            step = 1.0 / (max(beta, 1.0) * 0.3)
        else:
            step = 1.0 / (beta * 0.4)

        x = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new

        if it % restart_every == 0:
            c_y = fft_convolve_self(y)
            if c_y.max() >= best * 1.0005:
                ns = noise_scale * S / np.sqrt(len(a)) * (1.0 - 0.5 * progress)
                if ns > 1e-10:
                    y = project_simplex(y + rng.normal(0, ns, len(a)), S)
                    t = 1.0

    return best_a, best


def basin_hop(a_best, S, max_time, rng, n_hops_target=None):
    """Basin hopping: perturb + local optimize. Returns (best_a, best_max)."""
    n = len(a_best)
    start = time.monotonic()
    best_a = a_best.copy()
    best_val = fft_convolve_self(a_best).max()
    hop_count = 0
    it = 0

    # Estimate time per hop: aim for ~200 FISTA iterations per hop
    # Each iteration is roughly 2 FFTs of size 2*n
    # For n=8192, fsize=16384, each FFT is ~0.5ms, so ~2ms per iteration
    # 200 iterations ≈ 0.4s per hop. We can do ~max_time/0.4 hops.
    time_per_hop = max(0.5, max_time / max(n_hops_target or 200, 50))

    while time.monotonic() - start < max_time:
        hop_count += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        # Noise scale: starts large, decays
        sigma = (0.03 * (1 - progress) ** 1.5 + 0.002) * S / np.sqrt(n)

        # Create perturbation
        perturbed = best_a + rng.normal(0, sigma, n)
        perturbed = project_simplex(perturbed, S)

        # Sometimes try a more aggressive perturbation (every 5th hop)
        if hop_count % 5 == 0:
            sigma_big = (0.08 * (1 - progress) ** 1.5 + 0.005) * S / np.sqrt(n)
            perturbed = best_a + rng.normal(0, sigma_big, n)
            perturbed = project_simplex(perturbed, S)

        # Short FISTA on perturbed
        t_for_hop = max(1.0, min(time_per_hop, max_time - (time.monotonic() - start)))
        beta_start = 50.0 + 50.0 * progress
        beta_end = 500.0 + 2000.0 * progress

        a_opt, val_opt = run_fista(perturbed, S, beta_start, beta_end,
                                    t_for_hop, rng, use_peak=False)

        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            it = 0

    return best_a, best_val


def final_polish(a0, S, max_time, rng):
    """Intensive final polishing with very high beta and peak shaving."""
    n = len(a0)
    start = time.monotonic()
    best_a = a0.copy()
    best_val = fft_convolve_self(a0).max()

    # Phase 1: Very high beta LSE
    t1 = max_time * 0.5
    a1, v1 = run_fista(best_a, S, 200.0, 5000.0, t1, rng,
                       use_peak=False, restart_every=5000, noise_scale=0.002)
    if v1 < best_val:
        best_val = v1
        best_a = a1

    # Phase 2: Peak shaving
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t2 = remaining * 0.7
        a2, v2 = run_fista(best_a, S, 100.0, 1000.0, t2, rng,
                           use_peak=True, peak_k=min(64, n // 8),
                           restart_every=3000, noise_scale=0.001)
        if v2 < best_val:
            best_val = v2
            best_a = a2

    # Phase 3: Ultra-fine peak shaving on top-8 peaks
    remaining2 = max_time - (time.monotonic() - start)
    if remaining2 > 3:
        t3 = remaining2 * 0.8
        a3, v3 = run_fista(best_a, S, 500.0, 10000.0, t3, rng,
                           use_peak=True, peak_k=8,
                           restart_every=5000, noise_scale=0.0005)
        if v3 < best_val:
            best_val = v3
            best_a = a3

    return best_a, best_val


def generate_structural_inits(n, rng, base=None):
    """Generate diverse structural initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base is not None and len(base) == n:
        inits.append(base.copy())

    # Symmetric modifications of base
    if base is not None and n > 10:
        # Reversed
        inits.append(base[::-1].copy())
        # Circular shift
        shift = n // 3
        inits.append(np.roll(base, shift).copy())
        # Window-modulated
        window = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
        inits.append(base * (0.5 + 0.5 * window))

    # Standard window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 5 * tt + 1.0))

    # Random
    inits.append(rng.random(n) * 0.5 + 0.25)

    return [np.maximum(x, 0.01) for x in inits]


def objective_val(a):
    """Compute objective value directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for optimal non-negative sequence minimizing the evaluation function.
    Strategy: warm-start from given sequence, aggressive basin hopping with
    adaptive noise, multi-length exploration, and intensive final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        print("No given sequence available, using defaults.")

    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        n = len(a)
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  [NEW BEST] n=%d, obj=%.6f, max_b=%.6f" % (n, obj, val))
            return True
        return False

    # Allocate time: 60% to main length, 20% to secondary, 20% to polishing
    time_main = budget_s * 0.55
    time_secondary = budget_s * 0.20
    time_polish = budget_s * 0.20

    # === MAIN PHASE: Optimize at the given length ===
    if given is not None and len(given) >= 256:
        n_main = len(given)
    else:
        n_main = 8192

    S_main = np.sqrt(2.0 * n_main)
    print("\n=== Main phase: n=%d, time=%.0fs ===" % (n_main, time_main))

    # Get initial sequence for main length
    if given is not None:
        if len(given) == n_main:
            a_init = given.copy()
        elif len(given) > n_main:
            ratio = len(given) // n_main
            a_init = given[:n_main * ratio].reshape(n_main, ratio).mean(axis=1)
        else:
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, n_main)
            a_init = np.interp(t_new, t_old, given)
    else:
        a_init = np.ones(n_main)

    a_init = project_simplex(a_init, S_main)
    init_val = fft_convolve_self(a_init).max()
    print("  Initial: max_b=%.6f, obj=%.6f" % (init_val, 2.0 * n_main * init_val / S_main**2))

    # Phase A: Quick FISTA warmup (short time to establish gradient direction)
    t_warmup = min(60.0, time_main * 0.1)
    a_warm, v_warm = run_fista(a_init, S_main, 20.0, 200.0, t_warmup, rng,
                               use_peak=False, restart_every=2000, noise_scale=0.01)
    print("  After warmup: max_b=%.6f" % v_warm)
    update_global(a_warm, v_warm)

    # Phase B: Basin hopping (main exploration)
    t_hop = time_main * 0.60
    print("  Basin hopping for %.0fs..." % t_hop)
    a_hopped, v_hopped = basin_hop(a_warm, S_main, t_hop, rng, n_hops_target=150)
    print("  After basin hopping: max_b=%.6f" % v_hopped)
    update_global(a_hopped, v_hopped)

    # Phase C: Intermediate polish
    t_mid = time_main * 0.15
    a_mid, v_mid = run_fista(a_hopped, S_main, 100.0, 1000.0, t_mid, rng,
                             use_peak=False, restart_every=5000, noise_scale=0.003)
    print("  After intermediate polish: max_b=%.6f" % v_mid)
    update_global(a_mid, v_mid)

    # Phase D: Second round of basin hopping from polished solution
    t_hop2 = time_main * 0.10
    if time.monotonic() < deadline - time_secondary - time_polish:
        print("  Second basin hopping round (%.0fs)..." % t_hop2)
        a_hop2, v_hop2 = basin_hop(a_mid, S_main, t_hop2, rng, n_hops_target=40)
        print("  After second hopping: max_b=%.6f" % v_hop2)
        update_global(a_hop2, v_hop2)
        if v_hop2 < v_mid:
            a_mid = a_hop2
            v_mid = v_hop2

    # === SECONDARY PHASE: Try a different length ===
    if time.monotonic() < deadline - time_polish * 0.5:
        # Try a smaller or larger length
        if n_main >= 4096:
            n_sec = n_main // 2
        else:
            n_sec = n_main * 2

        # Also try the other option
        alt_lengths = [4096, 16384]
        n_sec2 = None
        for nl in alt_lengths:
            if nl != n_main and nl != n_sec and 256 <= nl <= 32768:
                n_sec2 = nl
                break

        S_sec = np.sqrt(2.0 * n_sec)
        t_sec = min(time_secondary * 0.6, deadline - time.monotonic() - time_polish * 0.5)
        if t_sec > 30:
            print("\n=== Secondary phase: n=%d, time=%.0fs ===" % (n_sec, t_sec))

            # Initialize from interpolation of best
            t_old = np.linspace(0, 1, len(a_mid))
            t_new = np.linspace(0, 1, n_sec)
            a_sec_init = np.interp(t_new, t_old, a_mid)
            a_sec_init = project_simplex(a_sec_init, S_sec)

            # Quick FISTA
            t_sec_fista = t_sec * 0.4
            a_sec, v_sec = run_fista(a_sec_init, S_sec, 50.0, 500.0, t_sec_fista, rng,
                                     use_peak=False, restart_every=3000, noise_scale=0.01)
            print("  Secondary FISTA: max_b=%.6f, obj=%.6f" % (v_sec, 2.0*n_sec*v_sec/S_sec**2))
            update_global(a_sec, v_sec)

            # Basin hop on secondary
            t_sec_hop = t_sec * 0.5
            if time.monotonic() < deadline - time_polish * 0.3:
                a_sec2, v_sec2 = basin_hop(a_sec, S_sec, t_sec_hop, rng, n_hops_target=30)
                print("  Secondary hop: max_b=%.6f, obj=%.6f" % (v_sec2, 2.0*n_sec*v_sec2/S_sec**2))
                update_global(a_sec2, v_sec2)

        # Try another length if available and time permits
        if n_sec2 is not None and time.monotonic() < deadline - time_polish * 0.5:
            S_individual = np.sqrt(2.0 * n_sec2)
            t_ind = min(time_secondary * 0.4, deadline - time.monotonic() - time_polish * 0.5)
            if t_ind > 20:
                print("\n=== Tertiary phase: n=%d, time=%.0fs ===" % (n_sec2, t_ind))
                t_old = np.linspace(0, 1, len(a_mid))
                t_new = np.linspace(0, 1, n_sec2)
                a_ind_init = np.interp(t_new, t_old, a_mid)
                a_ind_init = project_simplex(a_ind_init, S_individual)

                a_ind, v_ind = run_fista(a_ind_init, S_individual, 30.0, 300.0, t_ind, rng,
                                         use_peak=False, restart_every=2000, noise_scale=0.01)
                print("  Tertiary: max_b=%.6f, obj=%.6f" % (v_ind, 2.0*n_sec2*v_ind/S_individual**2))
                update_global(a_ind, v_ind)

    # === POLISH PHASE ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_best = len(global_a)
        S_best = global_a.sum()
        # Re-normalize to standard
        S_norm = np.sqrt(2.0 * n_best)
        a_polish = project_simplex(global_a, S_norm)

        if remaining > 30:
            print("\n=== Final polishing: n=%d, time=%.0fs ===" % (n_best, remaining))
            a_final, v_final = final_polish(a_polish, S_norm, remaining * 0.85, rng)
            print("  After final polish: max_b=%.6f, obj=%.6f" % (v_final, 2.0*n_best*v_final/S_norm**2))
            update_global(a_final, v_final)

    # === Last resort: one more basin hop if we have time ===
    if global_a is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_best_norm = project_simplex(global_a, S_best)
        print("\n=== Last hop attempt (%.0fs) ===" % remaining)
        a_last, v_last = basin_hop(a_best_norm, S_best, remaining * 0.8, rng, n_hops_target=20)
        update_global(a_last, v_last)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver) or ver < 0:
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verification: %.6f" % ver)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
