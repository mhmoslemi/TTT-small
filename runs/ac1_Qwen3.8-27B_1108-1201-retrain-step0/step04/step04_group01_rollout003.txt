    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

One approach: parameterize a = softmax(z) * S for z ∈ R^n. Then we optimize over z ∈ R^n (unconstrained) and Adam works naturally.

The gradient with respect to z is:
∂f/∂z_i = Σ_j (∂f/∂a_j) * ∂a_j/∂z_i

where a_j = S * exp(z_j) / Σ_k exp(z_k)

∂a_j/∂z_i = a_j * (δ_ij - a_i/S) = a_j * δ_ij - a_i * a_j / S

So ∂f/∂z_i = Σ_j g_j * (a_j * δ_ij - a_i * a_j / S)
= g_i * a_i - a_i / S * Σ_j g_j * a_j
= a_i * (g_i - mean_weighted_g)

where g = ∇_a f and mean_weighted_g = (Σ g_j * a_j) / S

This is nice! The gradient in the softmax parameterization is simply:
∂f/∂z = a * (g - <g, a>/S)

where <g, a> = Σ g_i * a_i.

This is elegant and avoids the projection issue entirely!

The LSE gradient w.r.t. a is:

Then the gradient w.r.t. z (softmax parameters) is:
∂f/∂z_i = a_i * (g_i - (Σ g_j * a_j) / S)

And we optimize z with Adam!

Strategy: Softmax-parameterized Adam optimization with LSE smoothing,
combined with Fourier-domain crossover and periodic perturbation.
The softmax parameterization (a = S * softmax(z)) automatically satisfies
the non-negativity and sum constraints, enabling clean Adam optimization.

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

        return np.full(len(a), S / len(a))
    return a * (S / s)

    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return S * e / e.sum()

    """Given a = S * softmax(z), recover z (up to a constant)."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe / S)

    a = softmax_from_z(z, S)
    c = fft_convolve_self(a)
    cm = c.max()

        scaled = beta * (c - cm)

    n = len(z)

    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    A_new[:cross] = A1[:cross]
    A_new[cross:] = A2[cross:]

    A_new[0] = (A1[0] + A2[0]) / 2  # DC component

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Softmax-parameterized Adam optimization with LSE smoothing.
The softmax parameterization a = S*softmax(z) automatically satisfies
non-negativity and sum constraints. Combined with Fourier-domain crossover
between elite solutions and adaptive perturbation, this escapes local minima
effectively. Final high-beta polishing targets the exact maximum.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def softmax_from_z(z, S):
    """Compute a = S * softmax(z)."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    total = e.sum()
    if total < 1e-300:
        return np.full(len(z), S / len(z))
    return S * e / total


def z_from_a(a, S):
    """Recover z (up to constant) from a = S * softmax(z)."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe / S)


def lse_grad_and_max(z, S, beta):
    """Compute LSE gradient w.r.t. z and the exact max of convolution."""
    a = softmax_from_z(z, S)
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    cm = cmax
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cmax, np.zeros(n)
    w = e / tot
    fsize = 1 << (2 * len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    dot_ga = np.dot(g, a) / S
    grad_z = a * (g - dot_ga)
    return cmax, grad_z


def objective_value(a):
    """Compute 2n*max(conv)/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fourier_crossover(a1, a2, S, rng):
    """Crossover two sequences in the Fourier domain."""
    n = len(a1)
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    cross = rng.integers(2, fsize // 2 - 1)
    A_new = np.empty(fsize, dtype=np.complex128)
    A_new[:cross] = A1[:cross]
    A_new[cross:] = A2[cross:]
    A_new[0] = (A1[0] + A2[0]) / 2.0
    if fsize % 2 == 0:
        A_new[fsize // 2] = (A1[fsize // 2] + A2[fsize // 2]) / 2.0
    a_new = np.real(ifft(A_new))
    a_new = np.maximum(a_new, 0.0)
    return normalize_to_sum(a_new, S)


def downsample(a, n_new):
    """Downsample sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    ratio = n_old // n_new
    if ratio < 1:
        t_old = np.linspace(0, 1, n_old)
        t_new = np.linspace(0, 1, n_new)
        return np.interp(t_new, t_old, a)
    small = a[:n_new * ratio].reshape(n_new, ratio).mean(axis=1)
    return small


def upscale(a, n_new):
    """Upscale sequence to new length via interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.interp(t_new, t_old, a)

            S2 = np.sqrt(2.0 * n2)
            a2 = downsample(global_a, n2)
            a2 = normalize_to_sum(a2, S2)
            z2 = z_from_a(a2, S2)
            a2_opt, cmax2 = run_adam_optimize(
                z2, S2, n2, time_n2, rng,
                beta_init=30.0, beta_rate=0.004, beta_max=10000.0,
                lr_init=0.01
            )
            obj2 = 2.0 * n2 * cmax2 / (a2_opt.sum() ** 2)
            print("n=%d result: cmax=%.6f, obj=%.6f" % (n2, cmax2, obj2))
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_a = a2_opt.copy()

    # Phase 4: Final micro-polish with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 4: Final micro-polish (%.0fs) ===" % remaining)
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        global_a = normalize_to_sum(global_a, S_final)
        z_final = z_from_a(global_a, S_final)
        a_fin, cmax_fin = polish_high_beta(
            z_final, S_final, n_final, remaining, rng,
            beta_start=5000.0, beta_end=80000.0
        )
        obj_fin = 2.0 * n_final * cmax_fin / (a_fin.sum() ** 2)
        print("Final: cmax=%.6f, obj=%.6f" % (cmax_fin, obj_fin))
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_a = a_fin.copy()

    # Verify and return
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification via evaluate_sequence: %.6f" % ver)

    if not np.isfinite(ver):
        print("  Rejected! Using fallback.")
        final_list = [float(x) for x in np.clip(global_a / max(global_a.sum(), 1e-10) * np.sqrt(2 * len(global_a)), 0, 1000)]
        ver2 = evaluate_sequence(final_list)
        if not np.isfinite(ver2):
            final_list = [1.0] * 256

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), ver))
    return final_list
