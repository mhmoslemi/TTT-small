Uses exact 1D minimax sub-problems solved by binary search on M.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def compute_g_for_coord(a, i, n):
    """Compute g[k] = dc[k]/da[i] for all k in [0, 2n-2]."""
    size = 2 * n - 1

    lo1 = i
    hi1 = min(i + n, size)

    # Second term: a[i-k] for 0 <= i-k < n, i.e., i-n+1 <= k <= i

        # If i >= n-1: lo2 = i-n+1, length = n

"""
Optimized coordinate descent / mirror descent for minimizing 2*n*max_conv/sum^2.
Uses LSE-smoothed gradient computed via scipy.signal.correlate, adaptive beta,
basin hopping with LP-direction restarts, and multiplicative updates.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta):
    """Compute LSE-smoothed gradient of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # h[i] = sum_k w[k] * a[k-i] = correlate(w, a, 'valid')
    h = correlate(w, a, mode='valid')
    return 2.0 * h


def run_mirror_descent(a_init, n, S, max_time, rng, beta_start=10.0, beta_end=300.0):
    """Mirror descent with LSE gradient and adaptive step."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    step = 0.1
    total_iters = 100000  # for beta scheduling
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        frac = min(1.0, it / total_iters)
        beta = beta_start + (beta_end - beta_start) * frac
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Mirror descent step (multiplicative)
        # a_new[i] = a[i] * exp(-step * grad[i] / S) (scaled to maintain sum)
        eta = step / max(gnorm, 1e-15)
        log_a = np.log(np.maximum(a, 1e-300))
        new_log = log_a - eta * grad
        new_log = np.clip(new_log, -700, 700)
        a_new = np.exp(new_log - new_log.max())
        ss = a_new.sum()
        if ss < 1e-300:

            a_new = a_new * (S / ss)

        obj_new = 2.0 * n * c_new.max() / S**2

            step = min(step * 1.1, 10.0)

            if step < 1e-6:
                # Try perturbation
                noise = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                step = 0.1
    
    return best_a, best_obj

        if obj_new < cur_obj * (1 + 1e-12):
            a = a_new
            step = min(step * 1.05, 5.0)
        else:
            step *= 0.8
            fail_count += 1
            if fail_count > 30:
                noise = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                step = 0.5
                fail_count = 0
    
    return best_a, best_obj


def lp_direction(a, c, n, S, rng, max_iter=20):
    """Compute LP-inspired direction."""
    M = c.max()
    # LSE weights as proxy for dual variables
    beta = 50.0
    cm = c.max()
    e = np.exp(beta * (c - cm))
    w = e / e.sum()
    
    # Compute h (the "dual constraint" values)
    h = correlate(w, a, mode='valid')
    h_max = h.max()
    h_min = h.min()
    
    # Direction: increase where h is small, decrease where h is large
    # g[j] = max(0, h_max - h[j]) / (h_max - h_min + 1e-15)
    direction = np.maximum(0, h_max - h)
    ds = direction.sum()
    if ds < 1e-15:
        return None
    direction = direction / ds
    
    # Take a step in this direction
    a_new = a + 0.1 * S * (direction - 1.0/n)
    a_new = project_simplex(a_new, S)
    return a_new
