```python
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase with diverse initialization, adaptive LSE with
backtracking, and targeted active-set polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_gradient(y, beta):
    """Compute LSE value and its gradient."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: 2 * conv(w, y) truncated to n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

        # Actually, for c[k] = sum_i a[i] a[k-i], the gradient w.r.t. a[j] is:
        #   a[k-j] + a[j-k] (when both are in range), or just one term if only one is in range
        
        # Use FFT for efficiency
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = 1.0 / k
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polak-Ribiere step size
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, S / gnorm)  # Cap step
        
        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        
        if obj_new <= cur_obj + 1e-12:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.1 * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-12:
                a = a_new
            else:
                # Skip this step
                step *= 0.5
        
        # Noise when stuck
        if stuck > 5000 and it % 200 == 0:
            noise_scale = max(1e-5, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def make_initialization(n, method, rng):
    """Create various initializations."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if method == "uniform":
        a = np.ones(n)
    elif method == "cosine_bell":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif method == "chirp":
        # Linear chirp magnitude (constant) with phase modulation
        # For non-negative, use amplitude modulation
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt + np.pi/4)
    elif method == "multi_chirp":
        a = 0.3 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.sin(2 * np.pi * 5 * tt + 1.0) + 0.2 * np.sin(2 * np.pi * 9 * tt + 2.0)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n)
    elif method == "bump":
        # Triangular bump in the middle
        a = np.maximum(0, 1 - 2 * np.abs(tt - 0.5))
    elif method == "two_bumps":
        a = np.exp(-0.5 * ((np.arange(n) - n*0.3) / (n*0.1))**2) + np.exp(-0.5 * ((np.arange(n) - n*0.7) / (n*0.1))**2)
    elif method == "periodic_mod":
        a = (1 + np.sin(2 * np.pi * 7 * tt)) * (0.5 + 0.5 * np.cos(2 * np.pi * tt))
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Diverse initialization with quick FISTA screening
2. Adaptive LSE-FISTA with backtracking and restarts
3. Direct coordinate descent (mass transport) for fine refinement
4. Multi-scale upsample to increase degrees of freedom
5. Final high-precision polish with exact subgradients
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0, use_backtrack=True):
    """FISTA with adaptive LSE and optional backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.8
            else:
                step = min(step * 1.03, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 5000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def coordinate_descent(a_init, max_time, rng, max_pairs_per_iter=8):
    """
    Direct coordinate descent via mass transfer.
    Transfers small amounts of mass from high-contribution to low-contribution
    coordinates to reduce the peak of the autocorrelation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 20000:
            break

        k_star = int(np.argmax(c))
        # Contribution of coordinate j to c[k_star]:
        # a[j] appears in c[k_star] as a[j]*a[k_star-j] if 0<=k_star-j<n
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            contrib[lo:hi+1] = a[k_star - hi:k_star - lo + 1][::-1]

        # Also consider the top few lags (not just argmax)
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            lo2 = max(0, int(kk) - n + 1)
            hi2 = min(n - 1, int(kk))
            if hi2 >= lo2:
                multi_contrib[lo2:hi2+1] += a[int(kk) - hi2:int(kk) - lo2 + 1][::-1]

        # Select source (j): high contribution, relatively large value
        # Select target (l): low contribution, relatively small value
        # Score: we want to move from high multi_contrib to low multi_contrib
        src_scores = multi_contrib * np.maximum(a, 0.001)
        tgt_scores = -multi_contrib * (S / n - a + 0.001)

        # Try a few candidates
        n_try = min(max_pairs_per_iter, n)
        src_candidates = np.argsort(src_scores)[::-1][:n_try]
        tgt_candidates = np.argsort(-tgt_scores)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in src_candidates[:n_try // 2]:
            for l in tgt_candidates[:n_try // 2]:
                if j == l or a[j] < 1e-10:
                    continue
                # Try transferring a small fraction of a[j]
                delta = a[j] * 0.05
                if delta < 1e-12:
                    continue
                # Compute new max approximately
                # c changes by: -delta*a[k-j] + delta*a[k-l] for affected k
                delta_c = np.zeros(2 * n - 1)
                # Effect of reducing a[j] by delta
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                delta_c[js:je+1] -= delta * coeff_j
                # Effect of increasing a[l] by delta
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                delta_c[ls:le+1] += delta * coeff_l

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj:
                    best_delta_obj = new_obj
                    best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            # Update c directly
            js = max(0, j)
            je = min(2 * n - 2, j + n - 1)
            c[js:je+1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
            ls = max(0, l)
            le = min(2 * n - 2, l + n - 1)
            c[ls:le+1] += delta * a[max(0, ls - l):min(n, le - l + 1)]
        else:
            # No improvement found; try a random small perturbation
            if stuck > 1000:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    # Recompute c (expensive but needed after random change)
                    c = fft_convolve_self(a)

        # Periodic full recompute to avoid drift
        if it % 100 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def run_subgradient_polish(a_init, max_time, rng):

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 2
        elif stuck < 5000:
            k = min(8, 2 + stuck // 500)
        else:
            k = min(32, 8 + stuck // 1000)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.002 * it))
        step = min(step, 2.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 3000 and it % 100 == 0:
            ns = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init, FISTA, coordinate descent, upsample, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Diverse initialization screening (15% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.15
    if p1_time > 45:
        print("\n=== Phase 1: Diverse starts (budget %.0fs) ===" % p1_time)
        methods = ["cosine", "gaussian", "chirp", "multi_tone", "random", "bump"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = run_fista(a_div, per_time * 0.6, rng,
                                        beta_init=15, beta_rate=0.012,
                                        beta_max=5000, step_scale=0.8,
                                        use_backtrack=False)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 2: Intensive FISTA (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 60:
        print("\n=== Phase 2: Intensive FISTA (budget %.0fs) ===" % p2_time)

        # Run 3 FISTA variants
        variants = [
            (30, 0.01, 10000, 1.0, True),
            (100, 0.005, 30000, 0.6, True),
            (500, 0.002, 100000, 0.3, False),
        ]
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time * 0.4, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss,
                                        use_backtrack=bt)
            print("  FISTA%d: obj=%.6f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Coordinate descent (30% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.30
    if p3_time > 60:
        print("\n=== Phase 3: Coordinate descent (budget %.0fs) ===" % p3_time)
        a_cd, obj_cd = coordinate_descent(global_best_a, p3_time, rng,
                                           max_pairs_per_iter=6)
        print("  Coord descent: obj=%.6f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # Follow up with FISTA on the CD result
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            a_cd2, obj_cd2 = run_fista(global_best_a, remaining3 * 0.5, rng,
                                        beta_init=50, beta_rate=0.008,
                                        beta_max=20000, step_scale=0.7,
                                        use_backtrack=True)
            print("  FISTA after CD: obj=%.6f" % obj_cd2)
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()

    # === Phase 4: Multi-scale upsample (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 45 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_fista(a_up, p4_time * 0.6, rng,
                                          beta_init=5, beta_rate=0.008,
                                          beta_max=5000, step_scale=1.0,
                                          use_backtrack=False)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

        # Coordinate descent on upsampled
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_cd_up, obj_cd_up = coordinate_descent(global_best_a, remaining4 * 0.7,
                                                     rng, max_pairs_per_iter=4)
            print("  CD upsampled: obj=%.6f" % obj_cd_up)
            if obj_cd_up < global_best_obj:
                global_best_obj = obj_cd_up
                global_best_a = a_cd_up.copy()

    # === Phase 5: Final polish (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.85
    if p5_time > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)
        a_pol, obj_pol = run_subgradient_polish(global_best_a, p5_time * 0.5, rng)
        print("  Subgradient polish: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_pol2, obj_pol2 = run_fista(global_best_a, remaining5, rng,
                                          beta_init=1000, beta_rate=0.001,
                                          beta_max=200000, step_scale=0.2,
                                          use_backtrack=True)
            print("  High-beta final: obj=%.6f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
```
