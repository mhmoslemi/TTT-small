"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm:
1. Multi-start FISTA with adaptive LSE smoothing and backtracking
2. Active-set peak shaving for final polishing

4. Number-theoretic and random initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    # Binary search for threshold
    lo = 0.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        adjusted = np.maximum(v - mid, 0.0).sum()
        if adjusted > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_gradient(a, c, topk=5):
    """Gradient using only the top-k lags, weighted by their values."""
    n = len(a)
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    # Weight by value (higher peaks get more weight)
    weights = c[idx] ** 2  # Quadratic weighting emphasizes the max
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_raw(a):
    """Compute max of self-convolution (the part we minimize, since scale-invariant)."""
    c = fft_convolve_self(a)
    return c.max()


def full_objective(a):
    """Compute the full objective 2*n*max_b/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def generate_quad_residue_init(n, rng):
    """
    Generate initialization based on quadratic residues.
    For prime p, the sequence of QR indicators has near-flat autocorrelation.
    We stretch this to length n.
    """
    # Find a prime near n
    p = n
    while p > 2:
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1
    
    if p < 10:
        return rng.random(n) + 0.1
    
    # Create QR indicator sequence
    qr = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p-1)//2, p) == 1:
            qr[i] = 1.0
    qr[0] = 1.0  # Include 0
    
    # Interpolate to length n
    t_old = np.linspace(0, 1, p)
    t_new = np.linspace(0, 1, n)
    result = np.interp(t_new, t_old, qr)
    
    # Add some smoothness
    result = 0.5 + 0.5 * result
    return result


def generate_window_init(n, window_type, rng):
    """Generate a windowed initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    if window_type == "hann":
        return 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window_type == "hamming":
        return 0.54 + 0.46 * np.cos(2 * np.pi * t)
    elif window_type == "blackman":
        return 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif window_type == "gaussian":
        sigma = rng.uniform(0.15, 0.35)
        return np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
    elif window_type == "sin":
        power = rng.uniform(1.0, 4.0)
        return np.sin(np.pi * t) ** power
    elif window_type == "multisine":
        freqs = rng.choice([2, 3, 4, 5, 7], size=2, replace=False)
        amps = rng.uniform(0.1, 0.4, 2)
        phases = rng.uniform(0, 2 * np.pi, 2)
        result = np.ones(n)
        for f, a, ph in zip(freqs, amps, phases):
            result += a * np.sin(2 * np.pi * f * t + ph)
        return np.maximum(result, 0.05)
    else:
        return rng.random(n) + 0.1


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def downscale_sequence(a_large, n_small):
    """Downscale a sequence by averaging blocks."""
    n_large = len(a_large)
    ratio = n_large // n_small
    if ratio < 1:
        return upscale_sequence(a_large, n_small)
    truncated = a_large[:n_small * ratio]
    return truncated.reshape(n_small, ratio).mean(axis=1)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Backtracking FISTA with adaptive LSE smoothing (core optimizer)
2. Active-set peak shaving for final convergence
3. Targeted "surgery" moves that directly reduce peak convolution lags
4. Multi-start with number-theoretic and structured initializations
5. Length exploration around the proven-good 8192
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    if abs(s - S) < 1e-12:
        return v
    lo = 0.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        adj = np.maximum(v - mid, 0.0).sum()
        if adj > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_gradient(a, c, cmean, topk=5):
    """Gradient using top-k lags weighted by excess above mean."""
    n = len(a)
    topk = min(topk, len(c))
    idx = np.argsort(c)[::-1][:topk]
    weights = np.maximum(c[idx] - cmean, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def full_objective(a):
    """Compute the full objective 2*n*max_b/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def run_fista(a_init, S, max_time, rng, beta_init=5.0, beta_growth=0.002,
              beta_max=5000.0, mode="hybrid", topk=5, label="fista", verbose=True):
    """
    FISTA with backtracking line search and adaptive LSE smoothing.
    mode: 'lse', 'active', 'hybrid'
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best_max = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * beta_init * 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))

        c = fft_convolve_self(y)
        if c.max() < best_max:
            best_max = c.max()
            best_a = y.copy()
            last_improve = it

        # Compute gradient
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "active":
            grad = active_gradient(y, c, c.mean(), topk=topk)
        else:
            w_lse = 0.3 + 0.7 * min(1.0, it / 3000.0)
            grad = w_lse * lse_gradient(y, c, beta) + (1.0 - w_lse) * active_gradient(y, c, c.mean(), topk=topk)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))

        # Backtracking line search
        step_size = step
        accepted = False
        best_trial = None
        best_trial_max = c.max()
        for bt in range(8):
            x_new = project_simplex(y - step_size * grad, S)
            y_trial = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y_trial = np.maximum(y_trial, 0.0)

            c_trial = fft_convolve_self(y_trial)
            if c_trial.max() < best_trial_max:
                best_trial_max = c_trial.max()
                best_trial = y_trial
                accepted = True
            if c_trial.max() <= c.max() + 1e-14:
                break
            step_size *= 0.5

        if best_trial is not None:
            a = project_simplex(best_trial, S)
            y = best_trial.copy()
            step = min(step * 1.08, 2.0 / beta)
        else:
            # Take a very small step anyway
            x_new = project_simplex(y - step * 0.01 * grad, S)
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            step *= 0.5

        t_k = t_new

        # Periodic restart
        if it - last_improve > 1500:
            elapsed_frac = (time.monotonic() - start) / max_time
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - elapsed_frac)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_k = 1.0
                step = 1.0 / (2.0 * beta)
            last_improve = it
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [{label}] it={it} restart, max_b={best_max:.8f}, t={elapsed:.0f}s")

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [{label}] it={it}, max_b={best_max:.8f}, beta={beta:.0f}, step={step:.5f}, t={elapsed:.0f}s")

    c_final = fft_convolve_self(best_a)
    final_max = c_final.max()
    if final_max < best_max:
        best_max = final_max
    return best_a, best_max


def run_surgery(a_init, S, max_time, rng, label="surgery", verbose=True):
    """
    Targeted peak reduction: identify the max lag, find the dominant
    contributing pair, and transfer mass to reduce the peak.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        if it % 100 == 1:
            c = fft_convolve_self(a)
            if c.max() < best_max:
                best_max = c.max()
                best_a = a.copy()
                last_improve = it

        # Find the max lag
        k_max = c.argmax()
        # Find the dominant contributing index: a[i]*a[i+k_max] is largest
        # For lag k_max, c[k_max] = sum_i a[i] * a[k_max-i] (with bounds)
        # The "autocorrelation" interpretation: c[k] = sum_i a[i]*a[i+k] for k>=0
        # But np.convolve uses c[k] = sum_i a[i]*a[k-i]
        # So c[k_max] = sum_i a[i]*a[k_max-i] where 0<=i<n and 0<=k_max-i<n
        # i ranges from max(0, k_max-n+1) to min(n-1, k_max)

        lo_i = max(0, k_max - n + 1)
        hi_i = min(n - 1, k_max)
        if hi_i < lo_i:
            c = fft_convolve_self(a)
            continue

        # Products contributing to this lag
        i_range = np.arange(lo_i, hi_i + 1)
        products = a[i_range] * a[k_max - i_range]
        # Find the dominant contributor
        i_dom = i_range[np.argmax(products)]
        j_dom = k_max - i_dom

        # Try to reduce: move mass from the larger of a[i_dom], a[j_dom]
        # to a position that minimally affects the peak
        if a[i_dom] >= a[j_dom]:
            src = i_dom
            dec_amount = a[src] * 0.01
        else:
            src = j_dom
            dec_amount = a[src] * 0.01

        if dec_amount < 1e-15:
            break

        # Choose destination: random position far from the peak structure
        # or the position with minimum current value
        dest_candidates = rng.choice(n, size=min(20, n), replace=False)
        best_dest = -1
        best_dest_max = np.inf
        for d in dest_candidates:
            if d == src:
                continue
            a_trial = a.copy()
            a_trial[src] -= dec_amount
            a_trial[d] += dec_amount
            if a_trial[src] < 0 or a_trial[d] > 1000:
                continue
            a_trial = project_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_max and c_trial.max() < best_dest_max:
                best_dest_max = c_trial.max()
                best_dest = a_trial.copy()

        if best_dest is not None:
            a = best_dest
            c = fft_convolve_self(a)
        else:
            # Try a small random perturbation instead
            if rng.random() < 0.1:
                noise_scale = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)

        if it - last_improve > 5000:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [{label}] it={it}, max_b={best_max:.8f}, k_max={k_max}, t={elapsed:.0f}s")

    return best_a, best_max


def generate_quad_residue_init(n, rng):
    """Generate initialization based on quadratic residues mod prime."""
    # Find a prime near n/2 (to get enough structure)
    p = max(n // 2, 100)
    while p > 50:
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1

    if p < 50:
        return rng.random(n) + 0.1

    qr = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            qr[i] = 1.0
    qr[0] = 0.5

    # Interpolate to length n and smooth
    t_old = np.linspace(0, 1, p)
    t_new = np.linspace(0, 1, n)
    result = np.interp(t_new, t_old, qr)
    # Apply a light smoothing
    kernel = np.array([0.25, 0.5, 0.25])
    result = np.convolve(result, kernel, mode='same')
    return np.maximum(result, 0.05)


def generate_window_init(n, window_type, rng):
    """Generate a windowed initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    if window_type == "hann":
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif window_type == "hamming":
        return 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
    elif window_type == "blackman":
        return np.maximum(0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t), 0.01)
    elif window_type == "gaussian":
        sigma = rng.uniform(0.12, 0.3)
        return np.exp(-((t - 0.5) ** 2) / (2.0 * sigma ** 2))
    elif window_type == "sin_power":
        power = rng.uniform(1.5, 5.0)
        return np.sin(np.pi * t) ** power
    elif window_type == "multisine":
        freqs = rng.choice([2, 3, 4, 5, 7, 11], size=2, replace=False)
        amps = rng.uniform(0.1, 0.35, 2)
        phases = rng.uniform(0, 2 * np.pi, 2)
        result = np.ones(n) * 0.8
        for f, amp, ph in zip(freqs, amps, phases):
            result += amp * np.sin(2.0 * np.pi * f * t + ph)
        return np.maximum(result, 0.05)
    elif window_type == "asym":
        # Asymmetric window (breaks palindromic symmetry)
        alpha = rng.uniform(0.3, 0.7)
        return np.maximum(np.sin(np.pi * alpha * t / 0.5) ** 2, 0.01) * np.exp(-t * 0.5)
    else:
        return rng.random(n) + 0.1


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    if n_small == n_large:
        return a_small.copy()
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def downscale_sequence(a_large, n_small):
    """Downscale a sequence by averaging blocks."""
    n_large = len(a_large)
    if n_large <= n_small:
        return upscale_sequence(a_large, n_small)
    ratio = n_large // n_small
    truncated = a_large[:n_small * ratio]
    return truncated.reshape(n_small, ratio).mean(axis=1)


def run_combined_polish(a_init, S, max_time, rng, label="polish", verbose=True):
    """
    Combined polishing: alternates between FISTA and surgery.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best_max = fft_convolve_self(a).max()
    best_a = a.copy()

    # Alternate between FISTA bursts and surgery bursts
    remaining = max_time
    phase = 0
    while remaining > 5:
        elapsed = time.monotonic() - start
        progress = (max_time - remaining) / max_time if max_time > 0 else 0

        if phase % 3 == 0:
            # FISTA burst
            t_burst = min(remaining * 0.4, 120.0)
            beta_start = 50.0 * (1.0 + progress * 10.0)
            a, max_b = run_fista(a, S, t_burst, rng,
                                  beta_init=beta_start, beta_growth=0.002,
                                  beta_max=20000.0, mode="hybrid",
                                  topk=3 + int(progress * 3),
                                  label=f"{label}_f", verbose=verbose)
        elif phase % 3 == 1:
            # Surgery burst
            t_burst = min(remaining * 0.3, 90.0)
            a, max_b = run_surgery(a, S, t_burst, rng,
                                    label=f"{label}_s", verbose=verbose)
        else:
            # Active-set FISTA
            t_burst = min(remaining * 0.3, 90.0)
            beta_start = 200.0 * (1.0 + progress * 5.0)
            a, max_b = run_fista(a, S, t_burst, rng,
                                  beta_init=beta_start, beta_growth=0.001,
                                  beta_max=50000.0, mode="active",
                                  topk=3, label=f"{label}_a", verbose=verbose)

        c_check = fft_convolve_self(a)
        if c_check.max() < best_max:
            best_max = c_check.max()
            best_a = a.copy()
            if verbose:
                obj = full_objective(best_a)
                print(f"    [{label}] improvement: max_b={best_max:.8f}, obj={obj:.8f}, elapsed={time.monotonic()-start:.0f}s")

        remaining = max_time - (time.monotonic() - start)
        phase += 1

    return best_a, best_max

    labels.append("rand")

    num_starts = min(len(inits), 6)
    remaining = deadline - time.monotonic()
    time_per_start = min(100.0, remaining * 0.25 / max(1, num_starts))

    for i in range(num_starts):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        t_alloc = min(time_per_start, remaining - 100)
        if t_alloc < 15:
            break

        init = normalize_to_sum(inits[i], S)
        mode = ["hybrid", "lse", "active"][i % 3]

        a_opt, max_b = run_fista(init, S, t_alloc, rng,
                                  beta_init=5.0, beta_growth=0.002,
                                  beta_max=2000.0, mode=mode,
                                  topk=4, label=labels[i],
                                  verbose=(i < 2))
        obj_opt = full_objective(a_opt)
        print(f"  [{labels[i]}] obj={obj_opt:.8f}")

        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print(f"    -> New global best!")

    # ===== PHASE 3: Intensive combined polishing =====
    print(f"\n=== Phase 3: Intensive polishing ===")
    print(f"  Current best: obj={global_best_obj:.8f}")

    if global_best_a is not None:
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()
        t_p3 = remaining * 0.45

        a_pol, max_b = run_combined_polish(global_best_a, S_best, t_p3, rng,
                                            label="polish", verbose=True)
        obj_pol = full_objective(a_pol)
        print(f"  After combined polish: obj={obj_pol:.8f}")
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

    # ===== PHASE 4: Try alternate lengths =====
    remaining = deadline - time.monotonic()
    if remaining > 150 and global_best_obj > 1.503:
        print(f"\n=== Phase 4: Alternate lengths (remaining={remaining:.0f}s) ===")

        for n_alt in [7680, 8704, 9216]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            S_alt = np.sqrt(2.0 * n_alt)
            t_alt = min(80.0, remaining - 60)

            init_alt = upscale_sequence(global_best_a, n_alt)
            init_alt = normalize_to_sum(init_alt, S_alt)

            a_alt, max_b_alt = run_fista(init_alt, S_alt, t_alt, rng,
                                          beta_init=20.0, beta_growth=0.002,
                                          beta_max=5000.0, mode="hybrid",
                                          topk=5, label=f"n{n_alt}", verbose=True)
            obj_alt = full_objective(a_alt)
            print(f"  n={n_alt}: obj={obj_alt:.8f}")

            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_alt.copy()
                print(f"    -> New global best!")

    # ===== PHASE 5: Final polishing =====
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_best_a is not None:
        print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)

        a_fin, max_b_fin = run_combined_polish(global_best_a, S_best, remaining * 0.7,
                                                rng, label="final", verbose=True)
        obj_fin = full_objective(a_fin)
        print(f"  After final: obj={obj_fin:.8f}")
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

    # ===== Final verification =====
    print(f"\n=== Final Result ===")
    print(f"  Best objective: {global_best_obj:.10f}")

    if global_best_a is None:
        n_fallback = 8192
        global_best_a = np.ones(n_fallback) * (np.sqrt(2.0 * n_fallback) / n_fallback)
        global_best_obj = full_objective(global_best_a)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        n_fallback = 8192
        global_best_a = np.ones(n_fallback) * (np.sqrt(2.0 * n_fallback) / n_fallback)

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    print(f"  evaluate_sequence: {ver:.10f}")

    if not np.isfinite(ver):
        print("  WARNING: rejected, using fallback")
        n_fallback = 8192
        global_best_a = np.ones(n_fallback) * (np.sqrt(2.0 * n_fallback) / n_fallback)
        final_list = [float(x) for x in global_best_a]

    print(f"  Returning sequence of length {len(final_list)}")
    return final_list
