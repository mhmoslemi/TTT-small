Minimize 2*n*max_conv/sum^2 over non-negative sequences.

perturbation-based escape from local minima. Key differences from baseline:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient: grad of beta*log(sum(exp(c/beta)))."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_single(y, c, k):
    """Subgradient from a single lag k."""
    n = len(y)
    w_full = np.zeros(len(c))
    w_full[k] = 1.0
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_subgradient(y, c, weights):
    """Subgradient from weighted combination of lags."""
    n = len(y)
    w_full = np.zeros(len(c))
    w_full[:] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def l2_conv_gradient(y, c):

    A = fft(y, fsize)
    return 2.0 * np.real(ifft(C * A))[:n]


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()

def normalize_seq(a, S):

    if s < 1e-15:

    return a * (S / s)

    return 0.5 + 0.5 * np.cos(2 * np.pi * t)

def run_l2_minimize(a_init, max_time, rng):
    """Minimize L2 norm of convolution on simplex."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        l2_obj = np.sum(c ** 2)
        if l2_obj < best_obj:

            best_a = y.copy()

        grad = l2_conv_gradient(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        # Backtracking step size
        step = 1.0 / (gnorm + 1.0)
        step = min(step, 1.0 / (0.5 * it + 1.0))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                  beta_max=20000.0, step_scale=1.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)

    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        if cur_max < best_conv - 1e-12:

def run_active_polish(a_init, max_time, rng):
    """Active-set focused polishing with adaptive top-k and smart perturbations."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 100:
            tol_frac = 0.0
        elif stuck < 500:
            tol_frac = 0.002
        elif stuck < 2000:
            tol_frac = 0.005
        elif stuck < 5000:
            tol_frac = 0.01
        else:
            tol_frac = 0.02

        max_val = c.max()
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        k_active = len(active)

        if k_active == 1:
            grad = subgradient_single(y, c, int(active[0]))
        else:
            # Weighted: emphasize lags closer to max
            active_vals = c[active]
            spread = active_vals.max() - active_vals.min()
            if spread > 1e-15:
                w = (active_vals - active_vals.min()) / spread
                w = w + 0.1  # floor to avoid zeros
                w = w / w.sum()
            else:
                w = np.ones(k_active) / k_active
            w_full = np.zeros(len(c))
            w_full[active] = w
            grad = weighted_subgradient(y, c, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Smart perturbation when stuck
        if stuck > 3000 and it % 200 == 0:
            # Perturb in direction that reduces max lag
            max_k = int(np.argmax(c))
            grad_max = subgradient_single(y, c, max_k)
            pert_dir = -grad_max
            # Project perturbation to tangent space
            pert_dir = pert_dir - np.mean(pert_dir)
            scale = max(1e-4, 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            noise = rng.normal(0, scale, n)
            noise = noise - np.mean(noise)  # tangent
            y = project_simplex(y + 0.5 * scale * pert_dir / max(np.linalg.norm(pert_dir), 1e-15) + noise, S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

            a_opt2, conv_opt2 = run_lse_fista(a, sub_budget, rng,
                                               beta_init=100.0, beta_rate=0.003,
                                               beta_max=10000.0, step_scale=0.3)
            obj2 = 2.0 * n * conv_opt2 / S ** 2
            if obj2 < best_obj:
                best_obj = obj2
                best_conv = conv_opt2
                best_a = a_opt2.copy()
                a = a_opt2.copy()
                print("    Re-opt %d: improved to %.6f" % (attempt, obj2))

    return best_a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.arange(n, dtype=np.float64)
    t_new = np.linspace(0, n - 1, n_new)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse starts, L2 warm-start, adaptive LSE FISTA,
    active-set polish, perturbation-based escape.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a):
        nonlocal global_best_obj, global_best_a
        if a is not None and len(a) > 0:
            obj = compute_obj(a)
            if np.isfinite(obj) and obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a.copy()
                print("  NEW BEST: obj=%.6f (n=%d, t=%.0fs)" % (obj, len(a), time.monotonic() - (deadline - budget_s + 30.0)))

    # Get initial sequences
    a_starts = []

    # Try to use height_sequence_1
    try:
        a1 = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a1 = np.clip(a1, 0, 1000)
        if a1.sum() > 0.01:
            a_starts.append(a1)
            print("Loaded height_sequence_1: n=%d, obj=%.6f" % (len(a1), compute_obj(a1)))
    except Exception:
        pass

    n_base = 8192

    if not a_starts or len(a_starts[0]) != n_base:
        # Use cosine as primary
        a_starts.insert(0, make_cosine_seq(n_base))
        print("Using cosine start: n=%d" % n_base)

    # Add diverse starts
    a_starts.append(make_gaussian_seq(n_base))
    a_starts.append(make_smooth_random(n_base, rng))

    # Add a "ringing" sequence
    t = np.linspace(0, 1, n_base, endpoint=False)
    a_ring = (0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t))
    a_ring = np.maximum(a_ring, 0.0)
    a_starts.append(a_ring)

    n = len(a_starts[0])
    S = np.sqrt(2.0 * n)

    # Evaluate all starts
    for i, a_s in enumerate(a_starts):
        a_s = normalize_seq(np.asarray(a_s, dtype=np.float64), S)
        obj_s = compute_obj(a_s)
        print("  Start %d: obj=%.6f" % (i, obj_s))
        update_best(a_s)

    # === Phase 1: L2 warm-start from top 3 (20% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20
    print("\n=== Phase 1: L2 warm-start (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    # Sort starts by objective
    start_objs = [(compute_obj(normalize_seq(a, S)), a) for a in a_starts]
    start_objs.sort(key=lambda x: x[0])
    top_starts = [a for _, a in start_objs[:3]]

    l2_results = []
    per_start_budget = p1_budget / len(top_starts)
    for i, a_s in enumerate(top_starts):
        if time.monotonic() > deadline - 60:
            break
        a_l2 = run_l2_minimize(a_s, per_start_budget, rng)
        a_l2 = normalize_seq(a_l2, S)
        obj_l2 = compute_obj(a_l2)
        print("  L2 from start %d: max_conv=%.6f, obj=%.6f (t=%.0fs)" %
              (i, fft_convolve_self(a_l2).max(), obj_l2, time.monotonic() - t1))
        l2_results.append(a_l2)
        update_best(a_l2)

    # === Phase 2: LSE FISTA from best L2 results (35% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.35
    print("\n=== Phase 2: LSE FISTA (budget %.0fs) ===" % p2_budget)
    t2 = time.monotonic()

    # Take top 2 L2 results and optimize with different beta schedules
    candidates = l2_results[:2] if len(l2_results) >= 2 else l2_results
    if not candidates:
        candidates = [top_starts[0]]

    fista_results = []
    per_candidate_budget = p2_budget / (len(candidates) + 1)

    for i, a_c in enumerate(candidates):
        if time.monotonic() > deadline - 60:
            break
        # Run with moderate beta
        a_opt, conv_opt = run_lse_fista(a_c, per_candidate_budget, rng,
                                         beta_init=10.0, beta_rate=0.006,
                                         beta_max=5000.0, step_scale=1.0)
        a_opt = normalize_seq(a_opt, S)
        obj_opt = compute_obj(a_opt)
        print("  FISTA mod-beta from cand %d: obj=%.6f (t=%.0fs)" % (i, obj_opt, time.monotonic() - t2))
        fista_results.append(a_opt)
        update_best(a_opt)

    # Also run from the best overall with aggressive beta
    if global_best_a is not None and time.monotonic() < deadline - 60:
        remaining2 = deadline - time.monotonic()
        a_agg, conv_agg = run_lse_fista(global_best_a, remaining2 * 0.7, rng,
                                         beta_init=50.0, beta_rate=0.004,
                                         beta_max=10000.0, step_scale=0.8)
        a_agg = normalize_seq(a_agg, S)
        obj_agg = compute_obj(a_agg)
        print("  FISTA agg-beta: obj=%.6f (t=%.0fs)" % (obj_agg, time.monotonic() - t2))
        update_best(a_agg)

    # === Phase 3: Active-set polish (25% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    print("\n=== Phase 3: Active-set polish (budget %.0fs) ===" % p3_budget)
    t3 = time.monotonic()

    if global_best_a is not None:
        a_pol, conv_pol = run_active_polish(global_best_a, p3_budget * 0.6, rng)
        a_pol = normalize_seq(a_pol, S)
        obj_pol = compute_obj(a_pol)
        print("  Active polish: obj=%.6f (t=%.0fs)" % (obj_pol, time.monotonic() - t3))
        update_best(a_pol)

        # Continue with longer polish
        remaining3 = deadline - time.monotonic()
        if remaining3 > 30:
            a_pol2, conv_pol2 = run_active_polish(global_best_a, remaining3 * 0.8, rng)
            a_pol2 = normalize_seq(a_pol2, S)
            obj_pol2 = compute_obj(a_pol2)
            print("  Extended polish: obj=%.6f (t=%.0fs)" % (obj_pol2, time.monotonic() - t3))
            update_best(a_pol2)

    # === Phase 4: Perturbation-based escape (20% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    print("\n=== Phase 4: Perturbation escape (budget %.0fs) ===" % p4_budget)
    t4 = time.monotonic()

    if global_best_a is not None:
        a_pert = run_perturb_reoptimize(global_best_a, p4_budget, rng)
        a_pert = normalize_seq(a_pert, S)
        obj_pert = compute_obj(a_pert)
        print("  Perturb result: obj=%.6f (t=%.0fs)" % (obj_pert, time.monotonic() - t4))
        update_best(a_pert)

    # === Phase 5: Final intensive polish on best (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_best_a is not None:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()

        # Very high beta LSE
        a_fin, conv_fin = run_lse_fista(global_best_a, remaining * 0.4, rng,
                                         beta_init=200.0, beta_rate=0.002,
                                         beta_max=50000.0, step_scale=0.3)
        a_fin = normalize_seq(a_fin, S)
        obj_fin = compute_obj(a_fin)
        print("  High-beta final: obj=%.6f (t=%.0fs)" % (obj_fin, time.monotonic() - t5))
        update_best(a_fin)

        # Active polish on the result
        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_fin2, conv_fin2 = run_active_polish(global_best_a, remaining5 * 0.9, rng)
            a_fin2 = normalize_seq(a_fin2, S)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final active: obj=%.6f (t=%.0fs)" % (obj_fin2, time.monotonic() - t5))
            update_best(a_fin2)

    # === Phase 6: Try upsampled version if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_best_a is not None and len(global_best_a) <= 16384:
        n_up = len(global_best_a) * 2
        S_up = np.sqrt(2.0 * n_up)
        a_up = upsample(global_best_a, factor=2)
        a_up = normalize_seq(a_up, S_up)
        obj_up = compute_obj(a_up)
        print("\n=== Phase 6: Upsampled n=%d: initial obj=%.6f ===" % (n_up, obj_up))

        if obj_up < global_best_obj + 0.05:  # Only if close
            sub_budget = min(100.0, remaining * 0.5)
            a_up_opt, conv_up = run_lse_fista(a_up, sub_budget, rng,
                                               beta_init=20.0, beta_rate=0.004,
                                               beta_max=5000.0, step_scale=0.8)
            a_up_opt = normalize_seq(a_up_opt, S_up)
            obj_up_opt = compute_obj(a_up_opt)
            print("  Upsampled optimized: obj=%.6f" % obj_up_opt)

            # Update best (note: different n, so compare directly)
            if np.isfinite(obj_up_opt) and obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()
                print("  NEW BEST from upsample: obj=%.6f" % obj_up_opt)

    # === Final verification ===
    print("\n=== Final Result ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = make_cosine_seq(n_fb)
            S_fb = np.sqrt(2.0 * n_fb)
            global_best_a = normalize_seq(global_best_a, S_fb)
            final_list = [float(x) for x in global_best_a]
            ver = evaluate_sequence(final_list)
            print("Fallback objective: %.6f" % ver)
    else:
        n_fb = 8192
        global_best_a = make_cosine_seq(n_fb)
        S_fb = np.sqrt(2.0 * n_fb)
        global_best_a = normalize_seq(global_best_a, S_fb)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("Fallback objective: %.6f" % ver)

    return final_list
