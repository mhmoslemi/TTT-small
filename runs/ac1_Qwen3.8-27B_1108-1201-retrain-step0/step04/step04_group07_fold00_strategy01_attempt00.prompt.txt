def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def spectral_flatten_step(a, N, alpha, target_ratio=0.5, rng=None):
    """
    One step of global spectral flattening.
    Pushes |FFT(a)|^2 toward uniform by reducing high-magnitude bins
    and slightly boosting low-magnitude bins.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    # Target: reduce bins that are above the median
    p_median = np.median(P)
    if p_median < 1e-30:
        return a, c, c_max

    # Flatten: bring high bins down, low bins up (partially)
    ratio = P / p_median
    # Smooth compression: new_ratio = ratio^alpha for ratio > 1, ratio^(1/alpha) for ratio < 1
    # This pushes toward 1 (flat)
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * (1.0 - alpha * (1.0 - np.power(ratio[above], -alpha)))
    P_new[~above] = P[~above] * (1.0 + alpha * (1.0 - np.power(ratio[~above], alpha)))
    P_new = np.maximum(P_new, 0.0)

    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c, c_max

def peak_repulsion_step(a, N, strength, rng=None):
    """
    Identify positions in a that contribute most to the convolution peak
    and apply repulsive force.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    c_max = c.max()

    # Find the peak lag
    peak_lag = int(np.argmax(c))

    # The contribution of a[j] to c[peak_lag] is a[peak_lag - j] (if valid)
    # So positions j where peak_lag - j is in [0, n-1] and a[peak_lag-j] is large
    # contribute the most
    contrib = np.zeros(n)
    for j in range(n):
        k = peak_lag - j
        if 0 <= k < n:
            contrib[j] = a[k]
        else:
            # Also check the other direction for circular-like contribution
            k2 = j - peak_lag
            if 0 <= k2 < n:
                contrib[j] = a[k2]

    # Apply repulsion: reduce high-contributing positions, boost low ones
    a_new = a * (1.0 - strength * contrib / max(c_max, 1e-15))

def subgradient_polish(a_init, n, S, max_time, rng, top_k=20):
    """Subgradient method with Nesterov acceleration for the peak of convolution."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()

        a = a * (S / s)

    t = 1.0

    best_local = a.copy()
    it = 0

        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Gradient computation via FFT
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new

        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_local, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * 0.5))

    # Linear ramp
    inits.append(("ramp", tt + 0.1))

    # Sine
    inits.append(("sin1", 0.5 + 0.4 * np.sin(2 * np.pi * tt)))
    inits.append(("sin3", 0.5 + 0.4 * np.sin(2 * np.pi * 3 * tt)))

    # Chirp
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt * tt))))

    # Multi-sine
    inits.append(("multi_sin", 0.5 + 0.2 * np.sin(2 * np.pi * 3 * tt) +
                                0.15 * np.sin(2 * np.pi * 7 * tt) +
                                0.1 * np.sin(2 * np.pi * 13 * tt)))

    # Random
    inits.append(("random", rng.random(n) + 0.05))

    # Bump (Gaussian)
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.1))

    # Two bumps
    inits.append(("two_bump", np.exp(-0.5 * ((tt - 0.25) * 6.0) ** 2) +
                                 np.exp(-0.5 * ((tt - 0.75) * 6.0) ** 2) + 0.05))

    # M-sequence inspired (balanced binary scaled)
    b = (rng.random(n) > 0.5).astype(float)
    b = b - b.mean() + 0.5  # center around 0.5
    inits.append(("mseq_like", np.clip(b, 0.05, 0.95)))

    # Smooth noise (low-pass filtered random)
    noise = rng.standard_normal(n)
    # Simple low-pass: moving average
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_noise", 0.2 + 0.6 * smooth))

    # Cosine window
    inits.append(("cosine_win", 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1))

    return inits
