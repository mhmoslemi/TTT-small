   - This is like the "Gerchberg-Saxton" algorithm (used in phase retrieval) but with the non-negativity constraint instead of a fixed amplitude.

Let me develop the spectral equalization idea:

1. Start with `a` (non-negative, sum = S).
2. Compute `A = FFT(a)`.
3. The power spectrum is `P = |A|^2`.
4. We want to make P as flat as possible. The "target" power spectrum would be `P_target = S^2 / (2n-1)` for all frequencies (approximately).
5. To reduce peaks in P, we can:
   - Where P is above target: reduce |A| (keep phase)
   - Where P is below target: increase |A| (keep phase)
   - Or: set `A_new = A * (P_target / P)^α` for some α ∈ (0, 1) (partial flattening)
6. Compute `a_new = IFFT(A_new)` (real part).
7. Project `a_new` onto {a >= 0, sum = S}.
8. Repeat.

This is a form of "spectral shaping" or "iterative amplitude adjustment".

The key parameter is α (the flattening strength). Large α = aggressive flattening, small α = gentle.

This approach is fundamentally different from the gradient-based approaches and might find different local minima.

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Spectral equalization (novel: flatten the power spectrum iteratively)
2. FISTA with LSE smoothing and backtracking line search
3. Peak-directed subgradient with adaptive step size
4. Coordinate descent with 1D subproblem solving
5. Multi-length and multi-start exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_smooth_max(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def spectral_equalize(a, alpha=0.5, max_iters=1000, S=None):
    """
    Spectral equalization: iteratively flatten the power spectrum.
    In frequency domain, adjust |A| to be more uniform, then project back.
    """
    n = len(a)
    if S is None:
        S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a.copy(), S)
    fsize = 1 << (2 * n - 1).bit_length()
    
    for it in range(max_iters):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        
        # Target: uniform power (excluding DC which is fixed by sum constraint)
        # Actually, the DC component is A[0] = sum(a) = S, so power[0] = S^2
        # For non-DC, target power ≈ S^2 / (fsize - 1) but this isn't quite right
        # The actual constraint is on the convolution, not the DFT directly.
        # For the linear convolution of length 2n-1, the "flat" target is:
        # b_k = S^2 / (2n-1) for all k
        # In frequency domain, this corresponds to |A(ω)|^2 being constant on the unit circle
        # sampled at fsize points, with the appropriate window.
        
        # Simpler approach: make |A|^2 proportional to 1/sqrt of current max power
        # Or: geometric mean targeting
        
        mean_power = power[1:].mean()  # exclude DC
        target = mean_power
        
        # Adjust amplitude: reduce peaks, boost troughs
        ratio = target / np.maximum(power, 1e-30)
        amp_adjust = ratio ** alpha  # alpha < 1 for partial adjustment
        
        A_new = A * amp_adjust
        # Keep DC fixed
        A_new[0] = A[0]
        
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = project_simplex(a_new, S)
    
    c = fft_convolve_self(a)
    return a, c.max()

def run_fista_bt(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.008, beta_max=5000.0):
    """FISTA with backtracking line search on LSE objective."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / (2.0 * beta * gnorm)  # initial step
        c_obj = lse_smooth_max(c, beta)
        
        for bt_iter in range(20):
            x_candidate = project_simplex(y - step * grad, S)
            c_candidate = fft_convolve_self(x_candidate)
            obj_candidate = lse_smooth_max(c_candidate, beta)
            if obj_candidate <= c_obj - 1e-4 * step * gnorm * gnorm:
                break
            step *= 0.5
        
        x_new = x_candidate
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart to escape local minima
        if it % 2000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    return best_a, best

def run_peak_directed(a_init, n, S, max_time, rng):
    """
    Peak-directed optimization: at each step, focus on reducing the single
    worst peak using a projected gradient step with line search.
    """
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    y = a.copy()
    t = 1.0
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Find the top few peaks
        k = min(8, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        top_max = c[top_idx[0]]
        
        # Weighted subgradient: weight peaks by their excess above mean
        mean_c = c.mean()
        weights = np.zeros(len(c))
        for idx in top_idx:
            if c[idx] > mean_c:
                weights[idx] = c[idx] - mean_c
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        # Gradient
        W = fft(weights, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Line search with decreasing step
        progress = min(1.0, (time.monotonic() - start) / max_time)
        base_step = S / (gnorm * (1.0 + 2.0 * progress))
        step = base_step
        
        for _ in range(5):
            x_candidate = project_simplex(y - step * grad, S)
            c_candidate = fft_convolve_self(x_candidate)
            if c_candidate.max() < c.max():
                break
            step *= 0.5
        
        x_new = x_candidate
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best

def coordinate_descent_polish(a, n, S, max_time, rng, max_sweeps=100):
    """
    Coordinate descent: for each coordinate, find the best value
    that reduces the max convolution, respecting the sum constraint
    by adjusting one other coordinate.
    """
    a = normalize_to_sum(a.copy(), S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > max_time:
            break
        
        # Randomize the order for better convergence
        order = rng.permutation(n)
        
        improved = False
        for j in order:
            if time.monotonic() - start > max_time:
                    break
            
            aj = a[j]
            if aj < 1e-10:
                continue
            
            # Try reducing a[j] and adding the mass to a random other coordinate
            # Or: try the optimal a[j] given the sum constraint
            
            # Simple approach: try a few candidate values for a[j]
            # and pick the one that minimizes max(c)
            
            # The effect of changing a[j] to a[j] + delta (taking from coordinate j2):
            # c_k changes by: delta * (a[k-j] - a[k-j2]) + delta^2 * ([k==2j] - [k==2j2])
            # (approximately, for the k where both are in range)
            
            # Simplest: try a[j] = 0, a[j] * 0.5, a[j] * 0.1
            # and redistribute to a random j2
            
            # Actually, let's use a simpler "pair swap" approach:
            # Pick j2 = the coordinate with the largest a[k-j] for the peak k
            # Then try moving mass from j to j2 or vice versa.
            
            # For speed, just try a[j] -> 0 (move all mass to some j2)
            # and a[j] -> a[j] * 0.5
            pass
        
        # At the end of sweep, recompute
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            improved = True
    
    return best_a, best

Actually, let me try a fundamentally different approach: **Use the fact that the problem is a ratio of a convex function to a convex function (after normalization), and apply Dinkelbach's algorithm or a similar parametric method.**

Wait, the objective is `2n * max(b) / (sum(a))^2`. If we fix `sum(a) = S`, it's just `2n * max(b) / S^2 = max(b)` (since S^2 = 2n). So we're just minimizing `max(b)` subject to `a >= 0, sum(a) = S`.

This is a **min-max** problem: `min_a max_k a^T Q_k a` where `Q_k` is the matrix for convolution at lag k.

One approach: **Bisection on the target T.**
- For a given T, check if there exists `a >= 0, sum(a) = S` such that `a^T Q_k a <= T` for all k.
- This is a feasibility problem with many quadratic constraints (non-convex).
- We can use a "cutting plane" approach: start with a subset of constraints, solve the QP, add violated constraints, repeat.

For the QP subproblem (minimize `a^T Q_{k*} a` subject to `a >= 0, sum(a) = S, a^T Q_{k_j} a <= T for j in J`):
- This is a non-convex QP (the constraints are non-convex if Q_{k_j} is indefinite).
- But for `a >= 0` and `Q_k` being a "convolution matrix" (which is positive semi-definite!), the constraints ARE convex!

Wait, is `Q_k` positive semi-definite? The matrix `Q_k` represents the quadratic form `a^T Q_k a = sum_i sum_j a_i (Q_k)_{ij} a_j = sum_i a_i * a_{k-i}` (the convolution at lag k).

The eigenvalues of `Q_k` are `|P(ω_k)|^2` (the power spectrum of the "shifted" sequence), which are non-negative. So YES, `Q_k` is PSD!

This means the problem is:
`min_a max_k a^T Q_k a` subject to `a >= 0, sum(a) = S`

where all `Q_k` are PSD. This is a **convex** optimization problem (min of max of convex functions, with convex constraints)!

Wait, is the max of convex functions convex? Yes! The pointwise maximum of convex functions is convex. So `f(a) = max_k a^T Q_k a` is convex.

And the constraint set `{a >= 0, sum(a) = S}` is convex (a polytope).

So we're solving a **convex optimization problem**! This is great because it means:
1. No local minima (any local minimum is global).
2. Convex optimization algorithms (subgradient, proximal gradient, etc.) converge globally.
3. We can use any convex optimization solver.

The issue is that the max is over `2n-1` convex functions, making the problem large. But we can use:
- **Subgradient methods**: The subgradient of `f(a)` is in `{2 Q_k a : k in argmax_k a^T Q_k a}` (the convex hull of the active gradients).
- **Cutting plane / Ellipsoid method**: Approximate the max with a supporting hyperplane, solve the QP, add the violated constraint.
- **Interior point methods**: Formulate as an SOCP or SDP.

Actually, the problem can be written as:
```
min t
s.t. a^T Q_k a <= t, k = 0, ..., 2n-2
     a >= 0
     sum(a) = S
```

This is a **quatically constrained program (QCP)** with convex constraints. It can be solved by:
1. **Semidefinite programming (SDP)**: Lift to `X = aa^T`, but this is huge.
2. **Sequential QP (SQP)**: Linearize the quadratic constraints.
3. **Proximal gradient / accelerated proximal gradient**: Use the subgradient of the max.

Actually, for the proximal gradient approach, the key insight is:
- The function `f(a) = max_k a^T Q_k a` is convex (as we established).
- We can compute its subgradient at any point.
- The proximal operator of the indicator of `{a >= 0, sum(a) = S}` is the simplex projection.
- So **FISTA (accelerated proximal gradient)** should converge to the global optimum!

The convergence rate of FISTA is O(1/t^2) for the function value, which is optimal for first-order methods on convex problems.

So why didn't the previous approach (which used FISTA) achieve the target? Possible reasons:
1. **Finite β in LSE**: The LSE approximation is not exact, so FISTA converges to the optimum of the LSE approximation, not the true problem.
2. **Insufficient iterations**: The O(1/t^2) rate means we need a lot of iterations for high accuracy.
3. **Numerical issues**: The simplex projection might have numerical errors.
4. **The max is achieved at multiple points**: The subgradient direction might not be unique.

The key improvement: **Use the exact subgradient of the max (not LSE) with FISTA.**

For the exact subgradient:
- At point `a`, compute `c = conv(a, a)`.
- Find all `k` where `c_k = max(c)` (or within a tolerance).
- The subgradient is `2 * sum_k w_k * (Q_k * a)` where `w_k >= 0, sum w_k = 1` and `w_k > 0` only for active k.
- A natural choice is `w_k = 1 / |argmax|` for all active k (uniform weighting).

But there's a subtlety: the subgradient of `max_k a^T Q_k a` at `a` is the convex hull of `{2 Q_k a : k in argmax}`. Using the uniform weighting gives one valid subgradient.

The issue with the exact subgradient is that it's not smooth (the argmax can change discontinuously). But FISTA can handle this (it only requires a valid subgradient, not a smooth gradient).

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/(sum(a))^2` over non-negative sequences. The key insight is that this is a **convex** optimization problem (max of convex quadratic forms with convex constraints), so there are no local minima—only convergence speed matters.

My approach differs from the previous code in several ways:

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Exploit convexity of the problem (max of PSD quadratic forms) with:
1. Spectral equalization for fast coarse initialization
2. FISTA with LSE + backtracking line search (main phase)
3. Excess-p penalty with annealing (alternative smooth objective)
4. Exact subgradient SCA polish (shifted-reverse trick, no FFT for gradient)
5. Multi-length exploration with upsampling/downsampling
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c)."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta, fsize=None):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(c) + n - 1).bit_length()
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient_at_peak(a, k_star):
    """
    Exact subgradient of max(b) at the point where b_{k_star} is the unique max.
    Since b_k = sum_i a_i * a_{k-i}, the gradient wrt a is:
    grad[i] = 2 * a[k_star - i] (for valid indices)
    This is O(n) with no FFT needed!
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def weighted_subgradient(a, c, top_k=10):
    """
    Weighted subgradient using top-k peaks.
    Weights proportional to excess above mean.
    """
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum

    # Compute weighted gradient: sum_k w_k * 2 * (shifted-reversed a by k)
    grad = np.zeros(n)
    for i in range(k):
        ki = idx[i]
        wi = weights[i]
        if wi < 1e-15:
            continue
        for j in range(n):
            m = ki - j
            if 0 <= m < n:
                grad[j] += wi * 2.0 * a[m]
    return grad


def spectral_equalize(a, n_iters=2000, S=None, alpha=0.3):
    """
    Spectral equalization: iteratively flatten the power spectrum.
    Adjusts |A(omega)| to be more uniform, then projects back to feasible set.
    """
    n = len(a)
    if S is None:
        S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a.copy(), S)
    fsize = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power[1:].mean() if len(power) > 1 else power[0]

        if mean_p < 1e-30:
            break

        # Adjust amplitude to flatten power spectrum
        ratio = mean_p / np.maximum(power, 1e-30)
        amp_adj = ratio ** alpha

        A_new = A * amp_adj
        A_new[0] = A[0]  # Keep DC fixed

        a_new = np.real(ifft(A_new, fsize))[:n]
        a = project_simplex(a_new, S)

    c = fft_convolve_self(a)
    return a, c.max()


def run_fista_lse(a_init, S, max_time, rng,
                  beta_init=20.0, beta_rate=0.005, beta_max=10000.0):
    """
    FISTA with LSE smoothing and backtracking line search.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: increase over time
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Compute LSE gradient
        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / (2.0 * beta * 0.5)
        obj_cur = lse_smooth_max(c, beta)
        a_new = a.copy()

        for _ in range(15):
            a_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(a_new)
            obj_new = lse_smooth_max(c_new, beta)
            if obj_new <= obj_cur - 1e-4 * step * gnorm * gnorm or step < 1e-10:
                break
            step *= 0.4

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        momentum = (t_k - 1.0) / t_new
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_k = t_new

        # Periodic restart to handle non-smoothness
        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            a = y.copy()
            last_improve = it

    return best_a, best


def run_excess_annealing(a_init, S, max_time, rng, p=4.0):
    """
    Minimize sum((b_k - T)^p_+) with T annealed downward.
    Higher p focuses more on the top peaks.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()

    c0 = fft_convolve_self(a)
    T_start = c0.max()
    T_end = c0.mean() + 0.005 * (c0.max() - c0.mean())

    y = a.copy()
    t_k = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        # Anneal threshold
        T = T_start + (T_end - T_start) * (progress ** 0.6)
        T = max(T, T_end)

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()

        # Compute excess gradient
        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            # Use a slightly lower T
            T = c.mean()
            excess = np.maximum(c - T, 0.0)

        if excess.max() < 1e-15:
            break

        # Weight: proportional to excess^(p-1)
        w = p * excess ** (p - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w = w / wsum

        # Gradient via FFT
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (p * (p - 1) * excess.max() ** (p - 2) * w.max() * 2.0 + 1.0)
        step = min(step, S / gnorm)

        a_new = project_simplex(y - step * grad, S)
        c_new = fft_convolve_self(a_new)

        # Reject if it makes things worse
        if c_new.max() > c.max() * (1 + 0.01):
            step *= 0.5
            a_new = project_simplex(y - step * grad, S)

        t_new = 0. * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        momentum = (t_k - 1.0) / t_new
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_k = t_new

    return best_a, best
