   (square root of Dirichlet kernel shape, not flat spectrum).

Key insight: For a flat linear self-convolution c = a*a (length 2n-1),
the DFT of c is a Dirichlet kernel. So |A_N[k]| should be proportional
to sqrt(|DFT(rect_{2n-1})[k]|), NOT flat.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum_w = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum_w - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum_w[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2 * n * c.max() / (S * S))

    where D_k = DFT of rect_{2n-1} (the Dirichlet kernel).
    """
    W = 2 * n - 1
    # DFT of rectangle of width W: D[k] = exp(-pi*i*(W-1)*k/N) * sin(pi*W*k/N) / sin(pi*k/N)
    k = np.arange(N)
    # Handle k=0 separately (limit is W)
    numerator = np.sin(np.pi * W * k / N)
    denominator = np.sin(np.pi * k / N)
    
    D_mag = np.zeros(N)
    # For k=0: limit is W
    D_mag[0] = W
    # For other k where denominator != 0
    mask = np.abs(denominator) > 1e-15
    D_mag[mask] = np.abs(numerator[mask] / denominator[mask])
    # For k where denominator is ~0 but k != 0 (shouldn't happen for N power of 2 and W odd)
    # But handle numerically:
    near_zero = ~mask & (k != 0)
    if np.any(near_zero):
        # Use limit: for k = m*N/W, the value is W/W = 1... actually compute directly
        for ki in k[near_zero]:
            # D[k] = sum_{m=0}^{W-1} exp(-2*pi*i*m*ki/N)
            m = np.arange(W)
            D_mag[ki] = np.abs(np.sum(np.exp(-2j * np.pi * m * ki / N)))
    
    # Target: proportional to sqrt(D_mag), normalized so DC component gives sum = S
    target = np.sqrt(D_mag)
    target = target / target[0] * S
    return target


def ifta_optimize(a_init, n, N, S, n_iter, rng, beta_init=0.5, beta_end=1.0):
    """
    IFTA optimization with the correct spectral target.
    Iterates between:
    - Frequency domain: blend magnitude toward target
    - Time domain: project onto {x >= 0, sum(x) = S}
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    target = compute_target_spectrum(n, N, S)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        # Anneal beta (blending parameter toward target)
        progress = it / max(n_iter - 1, 1)
        beta = beta_init + (beta_end - beta_init) * progress
        
        # Forward FFT
        A = fft(a, N)
        
        # Blend magnitude toward target (geometric mean)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Handle zeros in magnitude
        mag_safe = np.maximum(mag, 1e-30)
        target_safe = np.maximum(target, 1e-30)
        
        # Geometric interpolation: |A_new| = |A|^(1-beta) * target^beta
        log_mag = (1 - beta) * np.log(mag_safe) + beta * np.log(target_safe)
        mag_new = np.exp(log_mag)
        
        # Construct modified spectrum
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_full = np.real(ifft(A_new))[:n]
        
        # Project: non-negative, sum = S
        a = project_simplex(np.maximum(a_full, 0.0), S)
        
        # Evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def smooth_max_gradient(a, N, n, tau, top_k=None):
    """
    Compute the gradient of the smooth max (log-sum-exp) of the convolution.
    f(a) = tau * log(sum_k exp(c[k]/tau))
    grad = 2 * Re(ifft(w * conj(A)))[:n]
    where w[k] = softmax(c[k]/tau) (weights on the convolution peaks)
    """
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    
    # Softmax weights
    c_scaled = c / tau
    c_scaled -= c_scaled.max()  # numerical stability
    exp_c = np.exp(c_scaled)
    w = exp_c / exp_c.sum()
    
    # Gradient: d/dc(max_smooth) = w (softmax)
    # dc/da via: c = ifft(A^2), A = fft(a)
    # dL/da = 2*Re(ifft(w_full * conj(A)))[:n]
    w_full = np.zeros(N, dtype=np.complex128)
    w_full[:2 * n - 1] = w
    w_full = w_full.astype(np.complex128)
    
    grad = 2.0 * np.real(ifft(w_full * np.conj(A)))[:n]
    return grad, c, c.max()


def smooth_max_descent(a_init, n, N, S, max_time, rng, tau_init=0.1, tau_min=0.001,
                       n_inner=5, restart_prob=0.05):
    """
    Smooth-max gradient descent with annealing tau.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        # Anneal tau
        elapsed_frac = (time.monotonic() - start) / max_time
        tau = max(tau_min, tau_init * (1.0 - 0.95 * elapsed_frac))
        
        # Inner loop: multiple gradient steps with same tau
        for _ in range(n_inner):
            grad, c, c_max = smooth_max_gradient(a, N, n, tau)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Step size: adaptive
            step = S / (n * (1.0 + 0.005 * it))
            a_new = a - step * grad / max(gnorm, 1e-15)
            a = project_simplex(a_new, S)
            it += 1
        
        # Evaluate
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Random restart / perturbation
        if rng.random() < restart_prob:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_score


def hard_subgradient_polish(a_init, n, N, S, max_time, rng, top_k=50):
    """
    Hard subgradient method focusing on the actual max of the convolution.
    Uses Nesterov acceleration.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        sc = 2.0 * n * cur / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(N, dtype=np.complex128)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        A = fft(y, N)
        grad = 2.0 * np.real(ifft(w_full * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 1000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(("uniform", np.ones(n) * 0.5))
    
    # Sine with various frequencies
    for freq in [1, 2, 3, 5]:
        inits.append(("sin%d" % freq, 0.5 + 0.3 * np.sin(2 * np.pi * freq * tt)))
    
    # Anti-symmetric structure (large on one half, small on other)
    half = n // 2
    anti = np.ones(n) * 0.3
    anti[:half] = 0.7
    inits.append(("anti_sym", anti))
    
    # Linear ramp
    inits.append(("ramp", 0.1 + 0.8 * tt))
    
    # Chirp
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt**2))))
    
    # Random
    inits.append(("random", rng.random(n) + 0.05))
    
    # Smooth noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_noise", 0.2 + 0.6 * smooth))
    
    # Two-tone (designed to spread convolution energy)
    f1 = max(1, n // 7)
    f2 = max(2, n // 3)
    two_tone = 0.5 + 0.2 * np.cos(2 * np.pi * f1 * tt) + 0.15 * np.cos(2 * np.pi * f2 * tt)
    inits.append(("two_tone", np.maximum(two_tone, 0.01)))
    
    # Cosine window (tapers at edges)
    inits.append(("cosine_win", 0.3 + 0.4 * (1 + np.cos(2 * np.pi * tt)) / 2))
    
    # Periodic multi-sine
    multi = 0.5
    for fq in [1, 3, 7, 13]:
        multi += 0.08 * np.sin(2 * np.pi * fq * tt + rng.random() * 2 * np.pi)
    inits.append(("multi_sin", np.maximum(multi, 0.01)))
    
    return inits


def optimize_single_length(n, budget_s, rng, has_provided, target_obj=1.50):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    global_best = np.inf
    global_a = None
    
    def update_best(a):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
        return sc
    
    # Get initializations
    inits = make_initializations(n, rng)
    
    # Add provided sequence if available and matches length
    if has_provided:
        try:
            given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            if len(given) == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
                inits.insert(0, ("provided", np.clip(given, 0, 1000).astype(np.float64)))
        except Exception:
            pass
    
    t_start = time.monotonic()
    n_inits = len(inits)
    per_init_budget = budget_s * 0.8 / n_inits
    
    for name, arr in inits:
        elapsed = time.monotonic() - t_start
        if elapsed > budget_s * 0.8:
            break
        
        remaining = budget_s * 0.8 - elapsed
        if remaining < 5:
            break
        
        a = np.clip(arr, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
        
        # Stage 1: IFTA with correct target
        t_ifta = min(per_init_budget * 0.4, 30.0)
        if t_ifta > 1:
            # Estimate iterations
            n_iter = max(200, int(t_ifta / 0.005))  # ~5ms per iteration
            n_iter = min(n_iter, 5000)
            a_ifta, sc_ifta = ifta_optimize(a, n, N, S, n_iter, rng,
                                            beta_init=0.3, beta_end=0.9)
            update_best(a_ifta)
            a = a_ifta
        
        elapsed = time.monotonic() - t_start
        remaining = budget_s * 0.8 - elapsed
        if remaining < 3:
            continue
        
        # Stage 2: Smooth-max gradient descent
        t_smooth = min(per_init_budget * 0.35, 40.0)
        if t_smooth > 2:
            a_smooth, sc_smooth = smooth_max_descent(a, n, N, S, t_smooth, rng,
                                                     tau_init=S * 0.05,
                                                     tau_min=S * 0.001,
                                                     n_inner=3,
                                                     restart_prob=0.03)
            update_best(a_smooth)
            a = a_smooth
        
        elapsed = time.monotonic() - t_start
        remaining = budget_s * 0.8 - elapsed
        if remaining < 3:
            continue
        
        # Stage 3: Hard subgradient polish
        t_sg = min(per_init_budget * 0.25, 30.0)
        if t_sg > 2:
            a_sg, sc_sg = hard_subgradient_polish(a, n, N, S, t_sg, rng,
                                                  top_k=min(100, n // 64))
            update_best(a_sg)
        
        print("  [%s n=%d] best=%.6f (elapsed %.0fs)" % (name, n, global_best, time.monotonic() - t_start))
    
    # Final polish on best found
    if global_a is not None:
        final_budget = max(0.0, budget_s - (time.monotonic() - t_start))
        if final_budget > 5:
            a_f1, _ = hard_subgradient_polish(global_a, n, N, S,
                                              min(final_budget * 0.5, 60.0),
                                              rng, top_k=min(200, n // 32))
            update_best(a_f1)
            
            rem = max(0.0, budget_s - (time.monotonic() - t_start))
            if rem > 5:
                n_iter = max(100, int(rem * 0.5 / 0.005))
                a_f2, _ = ifta_optimize(global_a, n, N, S, n_iter, rng,
                                        beta_init=0.1, beta_end=0.5)
                update_best(a_f2)
                
                rem2 = max(0.0, budget_s - (time.monotonic() - t_start))
                if rem2 > 3:
                    a_f3, _ = smooth_max_descent(global_a, n, N, S, rem2 * 0.7, rng,
                                                 tau_init=S * 0.02,
                                                 tau_min=S * 0.0005,
                                                 n_inner=5,
                                                 restart_prob=0.0)
                    update_best(a_f3)
    
    return global_a, global_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses IFTA with correct Dirichlet-kernel target + smooth-max + subgradient.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)
    
    print("Starting optimization with budget %.0fs" % budget_s)
    
    # Check for provided sequence
    has_provided = False
    provided_len = 0
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            has_provided = True
            provided_len = len(given)
    except Exception:
        pass
    
    # Determine which lengths to try
    lengths_to_try = []
    if has_provided and provided_len >= 1024:
        lengths_to_try.append(provided_len)
    
    for L in [4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)
    
    # Sort: provided length first, then by priority
    lengths_to_try = sorted(set(lengths_to_try),
                            key=lambda x: (0 if x == provided_len else 1, abs(x - 8192)))
    
    global_best = np.inf
    global_a = None
    
    t_global_start = time.monotonic()
    total_budget = budget_s - 30
    
    # Allocate budget: favor the provided length and mid-range lengths
    weights = []
    for L in lengths_to_try:
        if L == provided_len:
            w = 1.5
        elif L <= 4096:
            w = 0.6
        elif L <= 8192:
            w = 1.0
        elif L <= 16384:
            w = 1.1
        else:
            w = 0.7
        weights.append(w)
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]
    
    for i, n in enumerate(lengths_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        alloc = min(budgets[i], remaining - 30)
        if alloc < 15:
            break
        
        print("=== Optimizing n=%d with budget %.0fs ===" % (n, alloc))
        a_result, score = optimize_single_length(n, alloc, rng, has_provided, target_obj=1.50)
        
        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  New global best: %.6f at n=%d" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))
    
    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence, score=%.4f" % global_best)
    
    # Verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback verification: %.6f" % ver)
    
    print("Final answer: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
