def project_simplex(v, S):

    n = len(v)

    v = np.asarray(v, dtype=np.float64)

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    # Specifically, grad[j] = 2*a[k-j] if 0<=k-j<n, else 0.

def topk_gradient(a, N, n, top_k):
    """Compute weighted gradient of top-k correlation peaks."""
    c, A = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), c, top_idx, top_vals
    # Weight by peak value (higher peaks get more weight)
    weights = top_vals / total
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        grad += weights[i] * conv_grad_at_k(a, int(k), N, n)
    return grad, c, top_idx, top_vals


def topk_gradient_fft(a, N, n, top_k):
    """Faster top-k gradient using FFT."""
    c, A = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), c, top_idx, top_vals
    weights = top_vals / total
    
    # grad[j] = sum_k w_k * 2*a[k-j] (for valid k-j)
    # This is like a convolution of weights (placed at positions top_idx) with a, times 2.
    # Specifically, grad = 2 * conv(w_full, a) where w_full has weights at top_idx positions.
    # But we need to be careful about the convolution direction.
    # grad[j] = 2 * sum_k w_k * a[k-j] * I(0<=k-j<n)
    # Let's define w_full of length 2n-1 with w_full[k] = w_k for k in top_idx.
    # Then grad[j] = 2 * sum_k w_full[k] * a[k-j] = 2 * (w_full * a)[j] where * is cross-correlation.
    # In numpy terms, this is 2 * np.convolve(w_full, a) shifted appropriately.
    # Actually, it's 2 * np.correlate(w_full, a, mode='full') at the right offset.
    # Or equivalently, 2 * ifft(fft(w_full) * conj(fft(a))) at the right positions.
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    
    # grad[j] = 2 * sum_k w_full[k] * a[k-j] for valid k-j in [0, n-1]
    # = 2 * (w_full cross-correlated with a)
    # cross-correlation: (w_full ⋆ a)[j] = sum_k w_full[k] * conj(a[k-j]) = sum_k w_full[k] * a[k-j] (real)
    # In FFT: fft(w_full ⋆ a) = conj(fft(a)) * fft(w_full)
    # So w_full ⋆ a = ifft(conj(fft(a)) * fft(w_full))
    
    fsize = 1 << (3 * n).bit_length()
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(Af)))  # This gives (w_full ⋆ a)[k] for k=0..fsize-1
    # We need cross[j] for j=0..n-1, which corresponds to (w_full ⋆ a)[j] = sum_k w_full[k]*a[k-j]
    # But the FFT gives us circular cross-correlation. With fsize >= (2n-1) + n - 1 = 3n-2, there's no aliasing.
    grad = 2.0 * cross[:n]
    return grad, c, top_idx, top_vals


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def nesterov_peak_block(a_init, n, S, N, max_time, rng, top_k_init=5, top_k_max=100):
    """Nesterov-accelerated top-k peak reduction with adaptive step size."""
    a = safe_normalize(a_init, n, S)
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    step_size = S / (n * 0.1)  # initial step size
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        
        grad, c, _, _ = topk_gradient_fft(a, N, n, top_k)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size: target reduction of ~1% of current peak per step
        target_reduction = 0.01 * cmax
        # Expected reduction ≈ step_size * gnorm (rough estimate)
        # So step_size ≈ target_reduction / gnorm
        # But cap it to avoid instability
        step = min(step_size, max(1e-10, target_reduction / (gnorm * 10)))
        
        a_new = project_simplex(a - step * grad / gnorm, S)
        
        # Check if this improved things
        c_new, _ = compute_conv(a_new, N)
        cmax_new = c_new.max()
        
        if cmax_new <= cmax + 1e-10:
            a = a_new
            # Nesterov momentum update
            v = 0.9 * v + 0.1 * grad
            a = project_simplex(a - 0.1 * step * v / max(np.linalg.norm(v), 1e-18), S)
            # Decay step size slowly
            step_size *= 0.999
        else:
            # Rejected step - reduce step size
            step_size *= 0.5
        
        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            v = np.zeros(n)
            step_size = S / (n * 0.05)
    
    return best_a, best_cmax

Minimize 2*n*max_conv/sum^2 using ramp initialization + focused gradient optimization.
Key insight: a monotonically increasing sequence (ramp) achieves max_b ≈ 4/3,
far below the 1.507+ of uniform/random initializations. We refine from this
starting point using top-k peak gradient descent with momentum and adaptive steps.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def conv_grad_fft(a, N, n, weight_arr):
    """
    Compute gradient of sum_k weight_arr[k] * b[k] w.r.t. a.
    Uses FFT for efficiency.
    """
    # grad[j] = 2 * sum_k weight_arr[k] * a[k-j] (for valid k-j in [0,n-1])
    # = 2 * (weight_arr cross-correlated with a)[j]
    # cross-correlation via FFT: ifft(fft(weight_arr) * conj(fft(a)))
    fsize = 1 << (3 * n).bit_length()
    W = fft(weight_arr, fsize)
    Af = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(Af)))
    return 2.0 * cross[:n]


def topk_weighted_grad(a, N, n, top_k, power=1.0):
    """Compute gradient weighted by top-k peak values (raised to `power`)."""
    c, _ = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx].copy()
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), c
    # Weight by peak value^power (higher peaks get more weight)
    if power != 1.0:
        top_vals = np.maximum(top_vals, 0) ** power
        total = top_vals.sum()
        if total < 1e-30:
            return np.zeros(n), c
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    grad = conv_grad_fft(a, N, n, w_full)
    return grad, c


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def ramp_init(n, alpha=1.0):
    """Create a ramp initialization: a[i] proportional to (i+1)^alpha."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** alpha
    return a / a.sum()  # normalize to sum=1, will rescale later


def optimize_topk(a_init, n, S, N, max_time, rng, top_k=10, power=1.0,
                   step_init=0.5, step_min=1e-8, decay=0.9995):

    """
    a = safe_normalize(a_init, n, S)
    v = np.zeros(n)
    best_a = a.copy()
    best_cmax = np.inf
    step = step_init
    start_time = time.monotonic()
    it = 0
    momentum = 0.9
    
    while time.monotonic() - start_time < max_time:
        it += 1
        grad, c = topk_weighted_grad(a, N, n, top_k, power)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:

        target = 0.05 * cmax
        step = max(step * decay, step_min)
        step = min(step, target / max(gnorm, 1e-18))
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad / gnorm
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-18:
            break
        v = v / vnorm

        # Check improvement
        c_new, _ = compute_conv(a_new, N)
        cmax_new = c_new.max()

            a = a_new

            step *= 0.5
            v = momentum * v  # damp momentum

            a = project_simplex(a + noise, S)
            v = np.zeros(n)
            step = step_init * 0.1
    
    return best_a, best_cmax
