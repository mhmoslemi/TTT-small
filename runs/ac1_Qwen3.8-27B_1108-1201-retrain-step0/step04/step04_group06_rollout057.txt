```python
def unfolding_step(p, n, n_iters=100):
    """Iteratively reduce the peak of the self-convolution."""
    c = convolve(p, p)  # length 2n-1
    target = c.mean()  # = 1/(2n-1) since sum(c) = 1
    
    for _ in range(n_iters):
        c = convolve(p, p)
        excess = np.maximum(c - target, 0)
        
        # Compute load for each position
        load = np.zeros(n)
        for k in range(2*n-1):
            if excess[k] > 0:
                lo = max(0, k - n + 1)
                hi = min(k, n-1)
                for i in range(lo, hi+1):
                    j = k - i
                    if 0 <= j < n:
                        load[i] += excess[k] * p[j]
        
        # Move mass from high-load to low-load
        # Multiplicative update
        step = 0.1
        p_new = p * np.exp(-step * load / (load.max() + 1e-30))
        p_new = p_new / p_new.sum()
        p = p_new
    
    return p
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using intensive multi-method local search.
Combines analytical coordinate descent, spectral shaping, and simulated annealing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = max(0.0, (v.sum() - S) / n)
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute max_conv / sum^2 (the part we minimize, times 2n is the full objective)."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv_self(a)
    return c.max() / (s * s)


def coord_descent_analytic(a, n, max_time, rng, top_m=30, max_rounds=5000):
    """
    Coordinate descent with analytical line search on the peak lag.
    For each round, find the coordinate whose optimal scaling gives the most
    reduction in the peak, apply it, and verify.
    """
    a = a.copy()
    S = a.sum()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()
    stall = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_conv_self(a)
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            # Random perturbation to escape
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            c = fft_conv_self(a)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()
            stall = 0
            continue
        
        # Find peak lag and second peak
        peak_lag = int(np.argmax(c))
        c_sorted = np.sort(c)[::-1]
        second_peak = c_sorted[1] if len(c_sorted) > 1 else cmax
        
        # Find contributing coordinates at peak lag
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)
        
        # Contribution of a[i] to c[peak_lag]: a[i] * a[peak_lag - i]
        # (counted twice if i != peak_lag - i, once if i == peak_lag - i)
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = peak_lag - i
            if 0 <= j < n:
                if i == j:
                    contribs[i] = a[i] * a[i]  # contributes a[i]^2 to c[peak_lag]
                else:
                    contribs[i] = a[i] * a[j]  # contributes 2*a[i]*a[j] to c[peak_lag]
        
        # Get top contributing coordinates
        top_indices = np.argsort(contribs)[::-1][:top_m]
        
        best_improvement = 0
        best_coord = -1
        best_factor = 1.0
        
        for idx in top_indices:
            i = int(idx)
            j = peak_lag - i
            if j < 0 or j >= n:
                continue
            
            if i == j:
                # Diagonal case: c[peak] contains a[i]^2
                # Changing a[i] to a[i]*f: new contribution = (a[i]*f)^2
                # delta_c = (a[i]*f)^2 - a[i]^2 = a[i]^2*(f^2 - 1)
                # We want: cmax + a[i]^2*(f^2-1) = target
                # f^2 = 1 + (target - cmax) / a[i]^2
                ai = a[i]
                if ai < 1e-15:
                    continue
                target = second_peak
                f2 = 1.0 + (target - cmax) / (ai * ai)
                if f2 > 0:
                    f = np.sqrt(f2)
                else:
                    f = 0.0
            else:
                # Off-diagonal: c[peak] contains 2*a[i]*a[j]
                # Changing a[i] to a[i]*f: delta_c = 2*a[i]*(f-1)*a[j]
                # We want: cmax + 2*a[i]*(f-1)*a[j] = target
                # f = 1 + (target - cmax) / (2*a[i]*a[j])
                ai = a[i]
                aj = a[j]
                if ai < 1e-15 or aj < 1e-15:
                    continue
                target = second_peak
                f = 1.0 + (target - cmax) / (2.0 * ai * aj)
            
            # Clamp f to reasonable range
            f = max(0.0, min(f, 2.0))
            
            # Compute the resulting peak (approximate: just the modified lag)
            if i == j:
                new_peak_val = cmax + a[i]**2 * (f*f - 1)
            else:
                new_peak_val = cmax + 2.0 * a[i] * a[j] * (f - 1)
            
            improvement = cmax - new_peak_val
            if improvement > best_improvement:
                best_improvement = improvement
                best_coord = i
                best_factor = f
        
        if best_coord < 0 or best_factor == 1.0:
            # No improvement found with analytical step, try small random moves
            i = int(top_indices[0]) if len(top_indices) > 0 else int(rng.integers(n))
            best_factor = 1.0 + rng.uniform(-0.1, 0.1)
            best_factor = max(0.0, best_factor)
        
        # Apply the move
        a[best_coord] *= best_factor
        
        # Project back to sum = S
        a = project_simplex(a, S)
    
    return best_a, best_cmax


def spectral_shaping_v2(a, N, n, max_time, rng, target_cmax=None):
    """
    Spectral shaping: modify the power spectrum to reduce the peak convolution.
    Uses a smoother, more adaptive approach.
    """
    a = a.copy()
    S = a.sum()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()
    
    c = fft_conv_self(a)
    cur_cmax = c.max()
    best_cmax = cur_cmax
    best_a = a.copy()
    
    if target_cmax is None:
        target_cmax = cur_cmax * 0.999  # aim for 0.1% reduction per call
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        cur_cmax = c.max()
        
        if cur_cmax < best_cmax:
            best_cmax = cur_cmax
            best_a = a.copy()
        
        if cur_cmax <= target_cmax * 1.001:
            break
        
        # Find top lags
        top_k = min(20, it // 10 + 5)
        top_indices = np.argsort(c)[::-1][:top_k]
        
        # Compute effective cosine
        m = np.arange(N)
        excess = np.maximum(c[top_indices] - target_cmax, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            excess = c[top_indices]
            total_excess = excess.sum()
        weights = excess / total_excess
        
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_indices):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        # Adaptive alpha
        alpha = 0.3 / (1.0 + 0.01 * it)
        alpha *= 0.5 + 0.5 * (cur_cmax - target_cmax) / max(cur_cmax, 1e-10)
        
        # Shape the spectrum
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        
        # Only reduce where cos_eff > 0 (constructive interference with peak)
        P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))
        
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Verify
        c_new = fft_conv_self(a_new)
        if c_new.max() < cur_cmax:
            a = a_new
        else:
            # Reduce step size
            alpha *= 0.5
            A_new = np.sqrt(P * np.exp(-alpha * np.maximum(cos_eff, 0.0))) * np.exp(1j * np.angle(A))
            a_new = np.real(ifft(A_new))[:n]
            a_new = project_simplex(a_new, S)
            a = a_new
    
    return best_a, best_cmax
```

Actually, let me reconsider the coordinate descent approach. The analytical line search I described assumes that only the peak lag matters. But in practice, reducing one lag might increase another. A better approach:

1. Compute the full convolution c
2. For the peak lag k*, compute the gradient of c[k*] with respect to each a[i]
3. Take a step in the negative gradient direction, projected onto the simplex
4. Use a line search to find the best step size

The gradient of c[k] with respect to a[i] is:
- `dc[k]/da[i] = a[k-i] + a[i]` if i != k-i (both terms)
  Wait, more carefully:
  `c[k] = sum_{i'} a[i']*a[k-i']`
  `dc[k]/da[i] = a[k-i] (from i'=i) + a[i] (from i'=k-i, if k-i != i)`
  
  If k-i == i (i.e., k = 2i): `dc[k]/da[i] = 2*a[i]`
  If k-i != i: `dc[k]/da[i] = a[k-i] + a[i]` (from the two terms i'=i and i'=k-i)

So the gradient of `max_k c[k]` at the peak lags is a sum of the individual gradients.

For the top-k lags with weights w_k:
`grad[i] = sum_{k in top_k} w_k * (a[k-i] + a[i] or 2*a[i])`

This can be computed efficiently with FFT!

Let me derive: `grad[i] = sum_k w_k * (a[k-i] + a[i])` (ignoring the diagonal correction for now)
`= sum_k w_k * a[k-i] + sum_k w_k * a[i]`
`= (w * a)[i] + a[i] * sum_k w_k`

where `(w * a)[i]` is the cross-correlation of w and a.

In the frequency domain, this is: `ifft(fft(w) * conj(fft(a)))` for the cross-correlation.

This gives us an O(n log n) way to compute the gradient!

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using intensive multi-method local search.
Combines gradient descent with top-k subgradients, analytical coordinate descent,
spectral shaping, and basin hopping with simulated annealing perturbations.
Key improvements: vectorized gradient computation, adaptive line search,
multiple complementary local optimizers, and strategic diversification.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = max(0.0, (v.sum() - S) / n)
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cmax(a):
    """Compute max of self-convolution."""
    return fft_conv_self(a).max()


def gradient_topk(a, c, top_k):
    """
    Compute the subgradient of max(c) using top-k lags.
    grad[i] = sum_{k in top} w_k * (a[k-i] + a[i]) for valid (i, k-i).
    Returns (grad, top_indices, weights).
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        total = 1.0
    weights = top_vals / total

    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        k = int(top_indices[idx])
        w_k = weights[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        contrib = a[j_range] + a[lo:hi + 1]
        grad[lo:hi + 1] += w_k * contrib

    return grad, top_indices, weights


def gradient_descent_line_search(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=200):
    """
    Gradient descent on top-k lags with backtracking line search.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = fft_conv_self(a)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            # Perturb to escape
            sigma = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, sigma, n), S)
            c = fft_conv_self(a)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()
            stall = 0
            continue

        # Compute gradient
        grad, top_indices, weights = gradient_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step0 = 2.0 * S / (n * gnorm)
        best_t = 0.0
        found = False
        for t_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            t = step0 * t_factor
            a_trial = project_simplex(a - t * grad, S)
            c_trial = fft_conv_self(a_trial)
            if c_trial.max() < cmax:
                a = a_trial
                found = True
                break

        if not found:
            # Try a very small step
            t = step0 * 0.001
            a_trial = project_simplex(a - t * grad, S)
            c_trial = fft_conv_self(a_trial)
            if c_trial.max() < cmax:
                a = a_trial
            else:
                # No improvement; perturb
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def coord_descent_analytic(a_init, n, S, max_time, rng, top_m=40):

    Coordinate descent with analytical line search on the peak lag.
    For each round, find the best coordinate to scale and the optimal factor.
    """

    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()
    stall = 0

    while time.monotonic() - start < max_time:
        c = fft_conv_self(a)
        cmax = c.max()

        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            # Try random coordinate move
            for _ in range(5):
                i = int(rng.integers(n))
                old_val = a[i]
                if old_val > 1e-15:
                    a[i] *= rng.uniform(0.5, 1.5)
                else:
                    a[i] = S / (n * 5)
                a = project_simplex(a, S)
                c_new = fft_conv_self(a)
                if c_new.max() < cmax:
                    cmax = c_new.max()
                    c = c_new
                    if cmax < best_cmax:
                        best_cmax = cmax
                        best_a = a.copy()
                        stall = 0
                    break
            stall = 0
            continue

        # Find peak and second peak
        peak_lag = int(np.argmax(c))
        c_copy = c.copy()
        c_copy[peak_lag] = 0
        second_peak = c_copy.max()

        # Find contributing coordinates
        lo = max(0, peak_lag - n + 1)
        hi = min(peak_lag, n - 1)

        # Compute contributions
        i_range = np.arange(lo, hi + 1)
        j_range = peak_lag - i_range
        contribs = a[i_range] * a[j_range]

        # Get top contributors
        n_contrib = len(i_range)
        if n_contrib == 0:
            continue
        top_m_eff = min(top_m, n_contrib)
        top_local = np.argsort(contribs)[::-1][:top_m_eff]

        best_improvement = 0.0
        best_coord_global = -1
        best_factor = 1.0

        for tl in top_local:
            i = int(i_range[tl])
            j = int(j_range[tl])
            ai = a[i]
            aj = a[j]

            if ai < 1e-15:
                continue

            if i == j:
                # Diagonal: c[peak] has term ai^2
                # New: (ai*f)^2, change = ai^2*(f^2 - 1)
                # Want: cmax + ai^2*(f^2-1) = second_peak
                f2 = 1.0 + (second_peak - cmax) / (ai * ai)
                if f2 > 0:
                    f = np.sqrt(f2)
                else:
                    f = 0.0
            else:
                # Off-diagonal: c[peak] has term 2*ai*aj
                # New: 2*(ai*f)*aj, change = 2*ai*aj*(f-1)
                # Want: cmax + 2*ai*aj*(f-1) = second_peak
                if aj < 1e-15:
                    continue
                f = 1.0 + (second_peak - cmax) / (2.0 * ai * aj)

            f = max(0.0, min(f, 3.0))
            if abs(f - 1.0) < 1e-10:
                continue

            # Approximate new peak (only the modified lag)
            if i == j:
                new_peak = cmax + ai * ai * (f * f - 1.0)
            else:
                new_peak = cmax + 2.0 * ai * aj * (f - 1.0)

            improvement = cmax - new_peak
            if improvement > best_improvement:
                best_improvement = improvement
                best_coord_global = i
                best_factor = f

        if best_coord_global < 0:
            # No analytical improvement; try small random steps on top contributors
            for _ in range(3):
                tl = int(rng.integers(n_contrib))
                i = int(i_range[tl])
                old_val = a[i]
                if old_val > 1e-15:
                    a[i] *= (1.0 + rng.uniform(-0.05, 0.05))
                a = project_simplex(a, S)
                c_new = fft_conv_self(a)
                if c_new.max() < cmax:
                    cmax = c_new.max()
                    c = c_new
                    if cmax < best_cmax:
                        best_cmax = cmax
                        best_a = a.copy()
                        stall = 0
                    break
            continue

        # Apply the move with verification
        a[best_coord_global] *= best_factor
        a = project_simplex(a, S)
        c_new = fft_conv_self(a)
        if c_new.max() >= cmax:
            # Revert
            a[best_coord_global] /= best_factor if best_factor > 0 else 1.0
            a = project_simplex(a, S)
            # Try half step
            a[best_coord_global] *= (1.0 + (best_factor - 1.0) * 0.5)
            a = project_simplex(a, S)
            c_new2 = fft_conv_self(a)
            if c_new2.max() >= cmax:
                a[best_coord_global] /= (1.0 + (best_factor - 1.0) * 0.5)

def spectral_shape(a_init, N, n, S, max_time, rng):
    """
    Spectral shaping: reduce power at frequencies constructively interfering with peak lags.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()

    c = fft_conv_self(a)
    cur_cmax = c.max()
    best_cmax = cur_cmax
    best_a = a.copy()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        cur_cmax = c.max()

        if cur_cmax < best_cmax:
            best_cmax = cur_cmax
            best_a = a.copy()

        # Top lags
        top_k = min(30, 5 + it // 20)
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        excess = np.maximum(top_vals - top_vals[len(top_vals) // 2], 0.0)
        if excess.sum() < 1e-30:
            excess = top_vals
        weights = excess / excess.sum()

        # Effective cosine
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for idx in range(len(top_indices)):
            cos_eff += weights[idx] * np.cos(2.0 * np.pi * m * int(top_indices[idx]) / N)

        # Adaptive alpha
        alpha = 0.4 / (1.0 + 0.02 * it)

        # Shape spectrum
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))

        # Add small random phase for exploration
        if rng.random() < 0.3:
            phase_noise = rng.normal(0, 0.03 * alpha, N)
            A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
        else:
            A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        c_new = fft_conv_self(a_new)
        if c_new.max() < cur_cmax:
            a = a_new
        else:
            # Half step
            P_half = P * np.exp(-0.5 * alpha * np.maximum(cos_eff, 0.0))
            A_half = np.sqrt(P_half) * np.exp(1j * np.angle(A))
            a_new = np.real(ifft(A_half))[:n]
            a_new = project_simplex(a_new, S)
            a = a_new

    return best_a, best_cmax


def basin_hopping(a_init, n, S, N, max_time, rng):
    """
    Basin hopping: alternate between local optimization and perturbation.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()
    rejected = 0

    while time.monotonic() - start < max_time:
        rem = max_time - (time.monotonic() - start)
        if rem < 5:
            break

        # Local optimization phase (split among methods)
        t_local = min(rem * 0.6, 30.0)

        # Gradient descent
        t_gd = t_local * 0.4
        a_gd, _ = gradient_descent_line_search(a, n, S, t_gd, rng, top_k_init=20, top_k_max=100)
        c_gd = compute_cmax(a_gd)
        if c_gd < best_cmax:
            best_cmax = c_gd
            best_a = a_gd.copy()
            a = a_gd.copy()
            rejected = 0

        rem = max_time - (time.monotonic() - start)
        if rem < 5:
            break

        # Coordinate descent
        t_cd = min(rem * 0.2, 15.0)
        a_cd, _ = coord_descent_analytic(a, n, S, t_cd, rng, top_m=30)
        c_cd = compute_cmax(a_cd)
        if c_cd < best_cmax:
            best_cmax = c_cd
            best_a = a_cd.copy()
            a = a_cd.copy()
            rejected = 0
        else:
            rejected += 1

        rem = max_time - (time.monotonic() - start)
        if rem < 5:
            break

        # Spectral shaping
        t_sp = min(rem * 0.2, 10.0)
        a_sp, _ = spectral_shape(a, N, n, S, t_sp, rng)
        c_sp = compute_cmax(a_sp)
        if c_sp < best_cmax:
            best_cmax = c_sp
            best_a = a_sp.copy()
            a = a_sp.copy()
            rejected = 0

        # Perturbation for global exploration
        rem = max_time - (time.monotonic() - start)
        if rem < 10:
            break

        # Adaptive perturbation strength
        if rejected > 3:
            sigma = 0.02 * S / np.sqrt(n)
        else:
            sigma = 0.005 * S / np.sqrt(n)

        # Spectral perturbation
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
        A_new = A + noise
        a_pert = np.real(ifft(A_new))[:n]
        a_pert = project_simplex(a_pert, S)

        # Quick local optimization from perturbed state
        t_quick = min(rem * 0.2, 10.0)
        a_quick, _ = gradient_descent_line_search(a_pert, n, S, t_quick, rng, top_k_init=10, top_k_max=50)
        c_quick = compute_cmax(a_quick)

        if c_quick < best_cmax:
            best_cmax = c_quick
            best_a = a_quick.copy()
            a = a_quick.copy()
            rejected = 0
        # Always move to perturbed state for diversity (even if worse)
        else:
            a = a_pert.copy()
            rejected += 1

    return best_a, best_cmax


def multi_start_explore(n, S, N, max_time, rng):
    """Try several diverse random initializations and optimize each briefly."""
    best_a = None
    best_cmax = np.inf

    strategies = [
        "uniform_perturb", "sine_mix", "chirp", "two_bump", "random"
    ]

    per_strategy = max_time / len(strategies)

    for strat in strategies:
        if time.monotonic() > max_time:
            break

        if strat == "uniform_perturb":
            a = np.ones(n) * S / n + rng.normal(0, S / (n * 3), n)
        elif strat == "sine_mix":
            t = np.arange(n) / n
            a = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * np.sin(2 * np.pi * 11 * t) \
                + 0.1 * np.sin(2 * np.pi * 17 * t) + 0.1 * rng.standard_normal(n)
        elif strat == "chirp":
            t = np.arange(n) / n
            a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * t + 5 * t * t)) + 0.1 * rng.standard_normal(n)
        elif strat == "two_bump":
            t = np.arange(n) / n
            a = 0.3 + 0.6 * np.exp(-20 * (t - 0.3) ** 2) + 0.4 * np.exp(-20 * (t - 0.7) ** 2)
        else:
            a = 0.2 + 0.8 * rng.random(n)

        a = np.clip(a, 0.001, None)
        a = project_simplex(a, S)

        # Quick optimization
        t_opt = min(per_strategy * 0.7, 20.0)
        a_opt, cmax = gradient_descent_line_search(a, n, S, t_opt, rng, top_k_init=10, top_k_max=60)
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a_opt.copy()

    return best_a, best_cmax


def final_polish(a_init, n, S, N, max_time, rng):
    """Final intensive polish: alternate between all methods with decreasing step sizes."""
    a = a_init.copy()
    start = time.monotonic()
    best_cmax = np.inf
    best_a = a.copy()

    c = fft_conv_self(a)
    best_cmax = c.max()
    best_a = a.copy()

    phase = 0
    while time.monotonic() - start < max_time:
        rem = max_time - (time.monotonic() - start)
        if rem < 3:
            break

        phase += 1

        # Gradient descent with increasing top_k
        t1 = min(rem * 0.3, 20.0)
        tk = min(300, 20 + phase * 5)
        a1, _ = gradient_descent_line_search(a, n, S, t1, rng, top_k_init=tk, top_k_max=tk + 50)
        c1 = compute_cmax(a1)
        if c1 < best_cmax:
            best_cmax = c1
            best_a = a1.copy()
            a = a1.copy()

        rem = max_time - (time.monotonic() - start)
        if rem < 3:
            break

        # Coordinate descent
        t2 = min(rem * 0.3, 15.0)
        a2, _ = coord_descent_analytic(a, n, S, t2, rng, top_m=50)
        c2 = compute_cmax(a2)
        if c2 < best_cmax:
            best_cmax = c2
            best_a = a2.copy()
            a = a2.copy()

        rem = max_time - (time.monotonic() - start)
        if rem < 3:
            break

        # Fine spectral shaping
        t3 = min(rem * 0.3, 10.0)
        a3, _ = spectral_shape(a, N, n, S, t3, rng)
        c3 = compute_cmax(a3)
        if c3 < best_cmax:
            best_cmax = c3
            best_a = a3.copy()
            a = a3.copy()

    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses basin hopping with gradient descent, coordinate descent, and spectral shaping.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = np.clip(np.asarray(a_candidate, dtype=np.float64), 0.0, 1000.0)
        s = a_candidate.sum()
        if s < 0.01:
            return False
        cmax = compute_cmax(a_candidate)
        if cmax < global_best:
            global_best = cmax
            global_a = a_candidate.copy()
            return True
        return False

    # --- Build initializations ---
    inits = []

    # Try to use previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            given_clipped = np.clip(given, 0.0, 1000.0)
            if given_clipped.sum() > 0.01:
                given_norm = project_simplex(given_clipped, S)
                inits.append(("given", given_norm))
                print("Using given sequence of length %d" % n)
    except Exception:
        try:
            given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            print("Given sequence length %d != %d, ignoring" % (len(given), n))
        except Exception:
            pass

    # Additional initializations
    t_arr = np.arange(n) / n
    inits.append(("sine", project_simplex(
        0.5 + 0.3 * np.sin(2 * np.pi * 3 * t_arr) + 0.2 * np.sin(2 * np.pi * 7 * t_arr)
        + 0.1 * np.sin(2 * np.pi * 13 * t_arr) + 0.05 * rng.standard_normal(n), S)))
    inits.append(("chirp", project_simplex(
        0.5 + 0.3 * np.sin(2 * np.pi * (2 * t_arr + 3 * t_arr * t_arr))
        + 0.1 * rng.standard_normal(n), S)))
    inits.append(("rand", project_simplex(0.3 + 0.7 * rng.random(n), S)))

    print("  %d initializations" % len(inits))

    # === Phase 1: Quick evaluation of all inits, then optimize the best (20% time) ===
    print("=== Phase 1: Initial evaluation ===")
    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.20

    for name, arr in inits:
        c0 = compute_cmax(arr)
        if c0 < global_best:
            global_best = c0
            global_a = arr.copy()
            print("  [%s] init: %.8f" % (name, global_best))
            obj = 2.0 * n * global_best / (S * S)
            print("  [%s] objective: %.6f" % (name, obj))

    if global_a is None:
        global_a = inits[0][1].copy()
        global_best = compute_cmax(global_a)

    # Intensive optimization of the best init
    best_init = global_a.copy()
    t_opt = min(phase1_budget * 0.7, 60.0)
    a_opt, _ = gradient_descent_line_search(best_init, n, S, t_opt, rng, top_k_init=20, top_k_max=100)
    update_best(a_opt)
    print("  After phase 1 GD: objective=%.6f" % (2.0 * n * global_best / (S * S)))

    # === Phase 2: Basin hopping (50% time) ===
    print("=== Phase 2: Basin hopping ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.50

    a_bh, cmax_bh = basin_hopping(global_a, n, S, N, phase2_budget, rng)
    if update_best(a_bh):
        print("  Basin hopping: objective=%.6f" % (2.0 * n * global_best / (S * S)))

    # Quick multi-start exploration if time remains
    rem_ms = deadline - time.monotonic()
    if rem_ms > 30:
        t_ms = min(rem_ms * 0.3, 60.0)
        a_ms, cmax_ms = multi_start_explore(n, S, N, t_ms, rng)
        if a_ms is not None and update_best(a_ms):
            print("  Multi-start: objective=%.6f" % (2.0 * n * global_best / (S * S)))

    # === Phase 3: Final intensive polish (30% time) ===
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if remaining > 5:
        a_pol, cmax_pol = final_polish(global_a, n, S, N, remaining * 0.7, rng)
        if update_best(a_pol):
            print("  Polish 1: objective=%.6f" % (2.0 * n * global_best / (S * S)))

        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            a_pol2, _ = final_polish(global_a, n, S, N, rem2 * 0.8, rng)
            if update_best(a_pol2):
                print("  Polish 2: objective=%.6f" % (2.0 * n * global_best / (S * S)))

    # --- Final verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [1.0] * 256
    else:
        global_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256

    print("Returning sequence of length %d, internal cmax=%.8f, obj=%.6f" % (
        len(final_list), global_best, 2.0 * n * global_best / (S * S)))
    return final_list
```
