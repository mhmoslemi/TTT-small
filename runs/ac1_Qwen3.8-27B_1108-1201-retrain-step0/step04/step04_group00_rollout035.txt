import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k, power=1.0):
    """Subgradient from top-k lags with optional power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj_normalized(a):
    """Objective assuming sum(a)=1: return 2*n*max(c)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_lse_reheat(a_init, max_time, rng, beta_start=5.0, beta_end=2000.0,
                     reheat_interval=1500, reheat_fraction=0.4,
                     noise_base=0.005, step_mode="adaptive"):
    """
    FISTA with LSE smoothing and periodic reheating.
    Returns (best_a_normalized, best_objective).
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    it = 0
    last_improve = it

    while time.monotonic() - start < max_time:
        it += 1

        # Compute beta with reheating
        progress = min(1.0, it / max(1, reheat_interval * 4))
        base_beta = beta_start * (beta_end / beta_start) ** progress
        reheat_phase = it % reheat_interval
        if reheat_phase < int(reheat_interval * (1.0 - reheat_fraction)):
            beta = base_beta * reheat_fraction
        else:
            beta = base_beta

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (y.sum() ** 2)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Gradient: use LSE when exploring, exact subgradient when exploiting
        if stuck < 500 or reheat_phase < int(reheat_interval * (1.0 - reheat_fraction)):
            grad = lse_gradient(y, c, beta)
        elif stuck < 3000:
            k = max(1, min(8, stuck // 500))
            grad = topk_subgradient(y, c, k, power=1.0)
        else:
            k = max(1, min(32, 8 + stuck // 500))
            grad = topk_subgradient(y, c, k, power=1.5)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size
        if step_mode == "adaptive":
            step = 1.0 / max(beta, 1.0)
            step = min(step, 2.0 * n / (beta * c.max() + 1e-30))
        else:
            step = 1.0 / max(beta, 1.0)

        # Reheat: inject noise and reset momentum when stuck
        if stuck > 1000:
            noise_scale = noise_base / np.sqrt(n) * np.exp(-stuck / 4000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), 1.0)
                t_mom = 1.0
                last_improve = it

        # FISTA update
        x_new = project_simplex(y - step * grad, 1.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def active_set_polish(a_init, max_time, rng, k_max=16):

    """

            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)

        else:

        if gnorm < 1e-18:

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        t_mom = t_new

            noise_scale = max(1e-7, 0.002 / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), 1.0)

    return best_a, best_obj


def upsample_linear(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    result = np.interp(t_new, t_old, a)
    return result

        tt = np.linspace(0, 1, n_fb, endpoint=False)
        final_list = [float(x) for x in (0.5 + 0.5 * np.cos(2 * np.pi * tt)) / n_fb * np.sqrt(2 * n_fb)]

    print("Total computation time: %.0fs" % (time.monotonic() - (deadline - budget_s + 35.0)))
    return final_list
