import time
from scipy.fft import fft, ifft

def compute_convolution(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return c

def compute_objective(a):
    """Compute 2n*max(c)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = compute_convolution(a)
    return 2.0 * n * c.max() / (S * S)

def project_nonneg_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for lambda such that sum(max(v - lambda, 0)) = S
    lo = v.min() - S  # if lambda = lo, sum(max(v-lo, 0)) >= S
    hi = v.max() + 1  # if lambda = hi, sum(max(v-hi, 0)) = 0 < S
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.maximum(v - lam, 0)

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using Nesterov-accelerated
subgradient descent with adaptive softmax weighting in the frequency domain.

Strategy:
1. Multi-start with diverse initializations (including anti-symmetric patterns).
2. Nesterov subgradient with adaptive temperature (annealed softmax over lags).
3. Periodic restarts with perturbation to escape local minima.
4. Final polish with conservative parameters.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

# Global best tracker
_global_best_score = np.inf
_global_best_a = None

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sort-based method."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u - (css - S) / r > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = np.max(np.where(cond)[0]) + 1
    theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return c, fsize

def compute_score(a):
    """Compute objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c, _ = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)

def compute_gradient(a, w, N):
    """
    Compute subgradient of sum_k w[k]*c[k] w.r.t. a.
    w: weight vector of length 2n-1 (non-negative).
    N: FFT size (>= 3n-2 for linear cross-correlation without aliasing).
    Returns gradient of length n.
    """
    n = len(a)
    A = fft(a, N)
    W = fft(w, N)
    s1 = np.real(ifft(W * np.conj(A)))[:n]
    s2 = np.real(ifft(A * np.conj(W)))[:n]
    return s1 + s2

def nesterov_subgradient(a_init, n, S, time_budget, rng, 
                          tau_init=1.0, tau_final=0.01,
                          top_k=None, initial_step=None,
                          anneal_period=200):
    """
    Nesterov-accelerated subgradient descent.
    
    Parameters:
    - a_init: initial point (will be projected)
    - n: sequence length
    - S: target sum
    - time_budget: seconds to run
    - rng: random number generator
    - tau_init, tau_final: softmax temperature (annealed)
    - top_k: if set, only use top-k lags (instead of softmax)
    - initial_step: initial step size
    """
    global _global_best_score, _global_best_a
    
    a = project_simplex(a_init, S)
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    start = time.monotonic()
    best_local = a.copy()
    best_local_score = compute_score(a)
    it = 0
    
    # FFT sizes
    N_conv = 1 << (2 * n - 1).bit_length()
    N_grad = 1 << (3 * n - 2).bit_length()
    
    if initial_step is None:
        initial_step = S / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        it += 1
        # Compute convolution and score
        c, _ = fft_convolve(y)
        S_y = y.sum()
        if S_y < 1e-10:
            break
        score = 2.0 * n * c.max() / (S_y * S_y)
        if score < best_local_score:
            best_local_score = score
            best_local = y.copy()
            if score < _global_best_score:
                _global_best_score = score
                _global_best_a = y.copy()
        
        # Compute weights (top-k or softmax)
        if top_k is not None:
            top_idx = np.argsort(c)[::-1][:top_k]
            top_vals = c[top_idx]
            w = np.zeros(2 * n - 1)
            w[top_idx] = top_vals  # weight by value
            w = w / w.sum()
        else:
            # Annealed softmax
            progress = min(1.0, it / (anneal_period * 10))
            tau = tau_init * (tau_final / tau_init) ** progress
            c_centered = c - c.mean()
            weights = np.exp(c_centered / (tau * (c.max() - c.min() + 1e-10)))
            weights = weights / weights.sum()
            # Scale to match sum of top-k for fair comparison
            w = weights
        
        # Compute gradient
        grad = compute_gradient(y, w, N_grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size (diminishing)
        step = initial_step / (1.0 + 0.05 * it)
        
        # Projected gradient step
        x_new = project_simplex(y - step * grad / S_y, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Occasional perturbation to escape local minima
        if it % 500 == 250 and rng.random() < 0.3:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0  # Reset Nesterov momentum
    
    return best_local, best_local_score

                 0.4 * np.exp(-20.0 * (tt - 0.7) ** 2) + 0.1)
    
    # Sawtooth
    inits.append(tt + 0.1)
    
    # Triangle wave
    inits.append(0.5 - np.abs(tt * 2 - 1))
    
    # Bump on edges (U-shape)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * (tt - 0.5)) * -1)  # inverted cos
    
    return inits

def optimize_length(n, budget, rng, target=1.50):
    """Optimize for a specific sequence length."""
    global _global_best_score, _global_best_a
    
    S = np.sqrt(2.0 * n)
    inits = make_initializations(n, rng)
    
    # Also try the provided sequence if available
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n:
            inits.insert(0, np.clip(given, 0, 1000))
    except:
        pass
    
    n_inits = len(inits)
    per_init_time = budget / (n_inits + 2)  # +2 for final polish
    
    t_start = time.monotonic()
    
    for i, init in enumerate(inits):
        elapsed = time.monotonic() - t_start
        if elapsed > budget * 0.8:
            break
        
        remaining = budget * 0.8 - elapsed
        if remaining < 5:
            break
        
        a = project_simplex(init, S)
        
        # Run Nesterov subgradient
        a_best, sc_best = nesterov_subgradient(
            a, n, S, remaining * 0.9, rng,
            tau_init=2.0, tau_final=0.05,
            top_k=min(50, n // 100),
            initial_step=S / np.sqrt(n)
        )
        
        elapsed = time.monotonic() - t_start
        if i % 3 == 0:
            print("  [init %d/%d n=%d] local_score=%.6f global=%.6f (t=%.0fs)" % 
                  (i, n_inits, n, sc_best, _global_best_score, elapsed))
    
    # Final polish on best found
    if _global_best_a is not None and len(_global_best_a) == n:
        final_time = max(0, budget - (time.monotonic() - t_start))
        if final_time > 10:
            # Conservative subgradient
            a_polished, sc_polished = nesterov_subgradient(
                _global_best_a, n, S, final_time * 0.7, rng,
                tau_init=0.5, tau_final=0.01,
                top_k=min(100, n // 50),
                initial_step=S / (2 * np.sqrt(n))
            )
            if sc_polished < _global_best_score:
                _global_best_score = sc_polished
                _global_best_a = a_polished.copy()
    
    return _global_best_a, _global_best_score
