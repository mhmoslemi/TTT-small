"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-phase optimization: FISTA with LSE smoothing, spectral flattening,
and coordinate descent polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=3000.0,
              beta_ramp_iters=2000):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp_iters))
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        if it - last_improve > 2000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng, alpha=0.5):
    """Gerchberg-Saxton style spectral flattening."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / 5000.0)
        alpha_cur = 0.3 + 0.6 * progress

        P_target = np.ones_like(P)
        P_new = (1.0 - alpha_cur) * P / P_mean + alpha_cur * P_target

        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best

def generate_variants(a, n, S, rng):
    """Generate diverse variants of a sequence."""
    variants = [a.copy()]

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, n))

    # Reversed
    variants.append(a[::-1].copy())

    # Random perturbations
    for scale in [0.02, 0.08, 0.15]:
        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        v = project_simplex(a + noise, S)
        variants.append(v)

    # Low-pass filtered
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    cutoff = 0.2
    F_filtered = F * np.exp(-5.0 * (freqs / cutoff) ** 2)
    a_filtered = np.real(ifft(F_filtered))[:n]
    a_filtered = np.maximum(a_filtered, 0)
    s = a_filtered.sum()
    if s > 1e-15:
        variants.append(a_filtered * (S / s))

    return variants

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA+LSE, spectral flattening, coordinate descent polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c). Returns (grad, cmax)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def exact_subgradient(a, c):
    """Exact subgradient at argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=5000.0,
              beta_ramp_iters=2000):
    """FISTA with adaptive LSE smoothing and periodic noise."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    best_gap = 0.0

    while time.monotonic() - start < max_time:
        it += 1

        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / float(beta_ramp_iters))
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            gap = best - cur
            best = cur
            best_a = y.copy()
            last_improve = it
            best_gap = gap

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: smaller when close to convergence
        step = 1.0 / (2.0 * beta)
        if it - last_improve > 3000:
            step *= 0.3
        elif best_gap < 1e-5:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % 4000 == 0:
            decay = np.exp(-it / 40000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng, alpha_start=0.3, alpha_end=0.9):
    """Gerchberg-Saxton style spectral flattening."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / max(1, int(max_time / 0.05)))
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        P_target = np.ones_like(P)
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_target

        mag_new = np.sqrt(np.maximum(P_new, 0))
        P_new_sum = P_new.sum()
        if P_new_sum < 1e-30:
            break
        mag_new = mag_new * np.sqrt(P.sum() / P_new_sum)

        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]

        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def compute_d_vector(a, j, m, n):
    """Compute d[t] = 2*(a[t-j] - a[t-m]) with bounds for t in [0, 2n-2]."""
    d = np.zeros(2 * n - 1)
    # a[t-j] is nonzero for t in [j, j+n-1]
    end_j = min(j + n, 2 * n - 1)
    if end_j > j:
        d[j:end_j] += 2.0 * a[:end_j - j]
    # a[t-m] is nonzero for t in [m, m+n-1]
    end_m = min(m + n, 2 * n - 1)
    if end_m > m:
        d[m:end_m] -= 2.0 * a[:end_m - m]
    return d


def coord_desc_step(a, c, j, m, n, S, rng):
    """Try one coordinate descent step: transfer from m to j. Returns (a_new, c_new, improved)."""
    if j == m or a[j] < 1e-12 or a[m] < 1e-12:
        return a, c, False

    d = compute_d_vector(a, j, m, n)
    lo = -a[j]
    hi = a[m]

    # Try multiple candidate deltas
    n_cands = 20
    deltas = np.linspace(lo, hi, n_cands)
    # Also add some focused candidates around 0
    focus = np.linspace(lo * 0.1, hi * 0.1, n_cands // 2)
    deltas = np.concatenate([deltas, focus])
    deltas = np.unique(deltas)

    best_delta = 0.0
    best_val = c.max()
    aj = a[j]
    am = a[m]

    for delta in deltas:
        # Linear part
        trial_max = (c + delta * d).max()
        # Quadratic correction for c[0]
        c0_new = c[0] + 2.0 * (aj - am) * delta + 2.0 * delta * delta
        if c0_new > trial_max:
            trial_max = c0_new
        if trial_max < best_val:
            best_val = trial_max
            best_delta = delta

    if best_val >= c.max() - 1e-14:
        return a, c, False

    # Apply
    a_new = a.copy()
    a_new[j] += best_delta
    a_new[m] -= best_delta
    if a_new[j] < -1e-10 or a_new[m] < -1e-10:
        return a, c, False

    c_new = c + best_delta * d
    c_new[0] = c[0] + 2.0 * (aj - am) * best_delta + 2.0 * best_delta * best_delta

    return a_new, c_new, True


def coordinate_descent(a_init, n, S, max_time, rng):
    """Paired coordinate descent on the simplex."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        improved_this_iter = False

        n_try = min(80 + it, 200)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 0.5:
                break

            if rng.random() < 0.3:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))
            elif rng.random() < 0.6:
                j = int(rng.integers(0, n - 1))
                m = (j + int(rng.integers(1, min(20, n)))) % n
            else:
                # Far apart pairs
                j = int(rng.integers(0, n // 2))
                m = int(rng.integers(n // 2, n))

            a, c, improved = coord_desc_step(a, c, j, m, n, S, rng)
            if improved:
                cur = c.max()
                if cur < best:
                    best = cur
                    best_a = a.copy()
                improved_this_iter = True

        # Periodic recompute to avoid drift
        if it % 30 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()

        # Noise injection when stuck
        if not improved_this_iter:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-it / 150.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()

    return best_a, best


def generate_variants(a, n, S, rng):
    """Generate diverse variants of a sequence."""
    variants = [a.copy()]

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, n))

    # Reversed
    variants.append(a[::-1].copy())

    # Random perturbations
    for scale in [0.02, 0.08, 0.15]:
        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        v = project_simplex(a + noise, S)
        variants.append(v)

    # Low-pass
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    F_filtered = F * np.exp(-4.0 * (freqs / 0.15) ** 2)
    a_f = np.real(ifft(F_filtered))[:n]
    a_f = np.maximum(a_f, 0)
    s = a_f.sum()
    if s > 1e-15:
        variants.append(a_f * (S / s))

    return variants


def score_sequence(a):
    """Compute the convolution max for a normalized sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()


def propose_candidate(seed=42, budget_s=2400):
    """Search for non-negative sequence minimizing 2*n*max_conv/sum^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)

    # Get parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 128 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, budget=%.0fs, parent=%s" % (n, budget_s, "yes" if parent is not None else "no"))

    global_best = np.inf
    global_a = None

    # Build initial candidates
    starts = []
    if parent is not None:
        starts.append(("parent", parent.copy()))
        for i, v in enumerate(generate_variants(parent, n, S, rng)):
            starts.append(("var%d" % i, v))

    tt = np.linspace(0, 1, n, endpoint=False)
    starts.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.001)))
    starts.append(("gauss", np.exp(-5.0 * (tt - 0.5) ** 2)))
    starts.append(("cos", np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.001)))
    starts.append(("rand", rng.random(n) + 0.05))

    # Evaluate starts
    print("\n=== Phase 0: Initial ===")
    for name, arr in starts:
        s = score_sequence(normalize_sum(arr, len(arr)))
        if s < global_best:
            global_best = s
            global_a = normalize_sum(arr, len(arr)).copy()
            print("  [%s] score=%.6f" % (name, s))

    print("Init best: %.6f" % global_best)

    # Phase 1: FISTA on best starts
    print("\n=== Phase 1: FISTA ===")
    p1_budget = (deadline - time.monotonic()) * 0.35

    # Sort starts by score, pick top
    scored = []
    for name, arr in starts:
        s = score_sequence(normalize_sum(arr, len(arr)))
        scored.append((s, name, arr))
    scored.sort(key=lambda x: x[0])

    n_fista = min(3, len(scored))
    t_per = p1_budget / max(1, n_fista)
    for i in range(n_fista):
        if time.monotonic() > deadline - (deadline - t_start) * 0.50 - 60:
            break
        sc, nm, arr = scored[i]
        arr_n = len(arr)
        S_i = np.sqrt(2.0 * arr_n)
        a_opt, _ = run_fista(arr, arr_n, S_i, t_per, rng,
                             beta_init=50.0, beta_max=5000.0,
                             beta_ramp_iters=2000)
        s = score_sequence(a_opt)
        if s < global_best:
            global_best = s
            global_a = a_opt.copy()
            print("  [fista_%s] score=%.6f (t=%.0fs)" % (nm, s, time.monotonic() - t_start))

    print("After FISTA: %.6f" % global_best)

    # Phase 2: Spectral flattening
    print("\n=== Phase 2: Spectral ===")
    if global_a is not None:
        n_b = len(global_a)
        S_b = np.sqrt(2.0 * n_b)
        p2_bud = (deadline - time.monotonic()) * 0.18
        if p2_bud > 10:
            a_sf, _ = spectral_flatten(global_a, n_b, S_b, p2_bud * 0.6, rng, 0.3, 0.8)
            s = score_sequence(a_sf)
            if s < global_best:
                global_best = s
                global_a = a_sf.copy()
                print("  [spectral1] score=%.6f" % s)

            a_sf2, _ = spectral_flatten(global_a, n_b, S_b, p2_bud * 0.4, rng, 0.5, 0.95)
            s = score_sequence(a_sf2)
            if s < global_best:
                global_best = s
                global_a = a_sf2.copy()
                print("  [spectral2] score=%.6f" % s)

    print("After spectral: %.6f" % global_best)

    # Phase 3: Coordinate descent
    print("\n=== Phase 3: CoordDescent ===")
    if global_a is not None:
        n_b = len(global_a)
        S_b = np.sqrt(2.0 * n_b)
        p3_bud = (deadline - time.monotonic()) * 0.25
        if p3_bud > 10:
            a_cd, _ = coordinate_descent(global_a, n_b, S_b, p3_bud, rng)
            s = score_sequence(a_cd)
            if s < global_best:
                global_best = s
                global_a = a_cd.copy()
                print("  [coord] score=%.6f" % s)

    print("After CD: %.6f" % global_best)

    # Phase 4: Final polish - alternate FISTA and CD
    print("\n=== Phase 4: Polish ===")
    if global_a is not None:
        n_b = len(global_a)
        S_b = np.sqrt(2.0 * n_b)
        remaining = deadline - time.monotonic()
        ndirs = 3
        t_round = remaining / max(1, ndirs)
        for rnd in range(ndirs):
            if time.monotonic() > deadline - 15:
                break
            # FISTA
            t_f = t_round * 0.55
            if t_f > 5:
                a_f, _ = run_fista(global_a, n_b, S_b, t_f, rng,
                                   beta_init=200.0, beta_max=10000.0,
                                   beta_ramp_iters=500)
                s = score_sequence(a_f)
                if s < global_best:
                    global_best = s
                    global_a = a_f.copy()
                    print("  [polish_f%d] score=%.6f" % (rnd, s))
            # CD
            t_c = t_round * 0.40
            if time.monotonic() > deadline - 15:
                break
            if t_c > 5:
                a_c, _ = coordinate_descent(global_a, n_b, S_b, t_c, rng)
                s = score_sequence(a_c)
                if s < global_best:
                    global_best = s
                    global_a = a_c.copy()
                    print("  [polish_c%d] score=%.6f" % (rnd, s))

    # Final verification
    print("\n=== Final ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
    else:
        print("Verification: INF - falling back")
        global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Len=%d, score=%.6f, time=%.0fs" % (len(final_list), ver, time.monotonic() - t_start))
    return final_list
