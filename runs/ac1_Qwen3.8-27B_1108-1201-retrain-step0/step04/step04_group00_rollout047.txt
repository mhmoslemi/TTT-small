from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(v)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta, top_k=None):
    """Compute LSE gradient w.r.t. the sequence with optional top-k focus."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    if top_k is not None and top_k < len(w):
        # Focus on top-k entries
        idx = np.argsort(c)[::-1][:top_k]
        w_masked = np.zeros(len(w))
        w_masked[idx] = w[idx]

    W = fft(w, fsize)
    A = fft(y, fsize)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

            decay = np.exp(-it / 80000.0)
            if decay > 0.05:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_focused(y, c, beta, focus_frac=1.0):
    """

    focus_frac=1.0 means use all entries (standard LSE).
    focus_frac<1.0 means reweight towards the top entries.

    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot

    if focus_frac < 1.0:
        k_focus = max(1, int(focus_frac * len(w)))
        idx = np.argsort(c)[::-1][:k_focus]
        w_other = w.sum() - w[idx].sum()

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        if it % 12000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 100000.0)
            if decay > 0.03:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_exact_polish(a_init, max_time, rng):
    """
    Polishing using exact top-k subgradient with adaptive k and step size.
    This is for the final stages when we're very close to convergence.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: very focused when close, wider when stuck
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 800:
            k = max(4, stuck // 50)
        elif stuck < 3000:
            k = max(8, 4 + stuck // 100)
        else:
            k = min(64, 8 + stuck // 200)

        grad = exact_topk_subgradient(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Careful step size: not too aggressive
        step = 0.5 * S / gnorm
        step = min(step, S / (0.1 * gnorm + 0.01))
        # Decrease step over time
        step *= 1.0 / (1.0 + 0.002 * it)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Aggressive noise when very stuck
        if stuck > 2000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            # Try different noise patterns
            pattern = rng.integers(0, 3)
            if pattern == 0:
                noise = rng.normal(0, noise_scale, n)
            elif pattern == 1:
                # Low-frequency noise
                freq = np.fft.rfftfreq(n)
                amp = noise_scale / (1 + 10 * freq * n)
                phase = rng.uniform(0, 2 * np.pi, len(freq))
                noise = np.fft.irfft(amp * np.exp(1j * phase), n)
            else:
                # Block noise
                block = max(1, n // 16)
                noise = np.repeat(rng.normal(0, noise_scale, n // block + 1), block)[:n]
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using cubic-like interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    # Use quadratic interpolation for smoother result
    result = np.interp(t_new, t_old, a, kind='cubic') if hasattr(np, 'interp') else None
    # np.interp doesn't support kind, so use simple linear
    result = np.interp(t_new, t_old, a)
    return np.maximum(result, 0)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    if n_new == 0:
        return a
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def generate_diverse_init(n, rng, style=0):
    """Generate diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    if style == 0:
        # Cosine taper
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif style == 1:
        # Gaussian bump
        t = np.linspace(-1, 1, n, endpoint=False)
        a = np.exp(-2 * t * t)
    elif style == 2:
        # Random smooth
        freqs = rng.integers(1, 5, 4)
        phases = rng.uniform(0, 2 * np.pi, 4)
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * t + p)
        a = np.maximum(a, 0.01)
    elif style == 3:
        # Two humps
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2)
    else:
        # Truncated cosine window
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    1. Quick FISTA at original size to get a good starting point
    2. Upsample to 2n for more degrees of freedom
    3. Extended FISTA with progressive constraint focus at 2n
    4. Exact subgradient polishing
    5. Diverse random restarts to escape local minima
    6. Final intensive polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Quick FISTA at original size (15% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.15
    print("\n=== Phase 1: Quick FISTA at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_adaptive(global_best_a, p1_budget, rng,
                                beta_init=5.0, beta_rate=0.008,
                                beta_max=3000.0, step_scale=1.0,
                                focus_schedule=(1.0, 0.3, 3000))
    obj1 = compute_obj(a1)
    print("  FISTA: obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: Upsample to 2n and optimize (35% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 80 and n0 <= 16384:
        p2_budget = remaining * 0.35
        n2 = n0 * 2
        print("\n=== Phase 2: Upsample to n=%d (budget %.0fs) ===" % (n2, p2_budget))
        t2 = time.monotonic()

        a2_init = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())

        # First pass: broad exploration
        a2a, _ = run_fista_adaptive(a2_init, p2_budget * 0.4, rng,
                                     beta_init=3.0, beta_rate=0.005,
                                     beta_max=2000.0, step_scale=1.0,
                                     focus_schedule=(1.0, 0.5, 2000))
        obj2a = compute_obj(a2a)
        print("  FISTA broad at 2n: obj=%.6f (t=%.0fs)" % (obj2a, time.monotonic() - t2))
        if obj2a < global_best_obj:
            global_best_obj = obj2a
            global_best_a = a2a.copy()

        # Second pass: focused
        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 10:
            a2b, _ = run_fista_adaptive(global_best_a, remaining2, rng,
                                         beta_init=10.0, beta_rate=0.004,
                                         beta_max=5000.0, step_scale=0.8,
                                         focus_schedule=(0.5, 0.1, 5000))
            obj2b = compute_obj(a2b)
            print("  FISTA focused at 2n: obj=%.6f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Exact subgradient polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = remaining * 0.20
        n_best = len(global_best_a)
        print("\n=== Phase 3: Exact polish at n=%d (budget %.0fs) ===" % (n_best, p3_budget))
        t3 = time.monotonic()

        a3, _ = run_exact_polish(global_best_a, p3_budget, rng)
        obj3 = compute_obj(a3)
        print("  Exact polish: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Diverse restarts (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = remaining * 0.15
        n_best = len(global_best_a)
        print("\n=== Phase 4: Diverse restarts at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        n_restarts = max(2, int(p4_budget / 25))
        restart_budget = p4_budget / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Create diverse perturbation
            style = i % 5
            if style < 3:
                # Perturb existing best
                noise_scale = 0.05 + 0.15 * rng.random()
                S_best = np.sqrt(2.0 * n_best)
                a_pert = global_best_a + rng.normal(0, noise_scale * S_best / np.sqrt(n_best), n_best)
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 0.01:
                    a_pert = a_pert * (S_best / s_pert)
                else:
                    a_pert = generate_diverse_init(n_best, rng, style)
            else:
                # Completely new initialization
                a_pert = generate_diverse_init(n_best, rng, style)

            a_r, _ = run_fista_adaptive(a_pert, restart_budget, rng,
                                         beta_init=10.0, beta_rate=0.006,
                                         beta_max=4000.0, step_scale=1.0,
                                         focus_schedule=(1.0, 0.2, 2000))
            obj_r = compute_obj(a_r)
            print("  Restart %d (style %d): obj=%.6f" % (i + 1, style, obj_r))
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_r.copy()

    # === Phase 5: Try larger size if possible (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 50 and len(global_best_a) <= 8192:
        p5_budget = remaining * 0.10
        n5 = len(global_best_a) * 2
        print("\n=== Phase 5: Try n=%d (budget %.0fs) ===" % (n5, p5_budget))
        t5 = time.monotonic()

        a5_init = upsample(global_best_a, factor=2)
        S5 = np.sqrt(2.0 * n5)
        a5_init = a5_init * (S5 / a5_init.sum())

        a5, _ = run_fista_adaptive(a5_init, p5_budget, rng,
                                    beta_init=3.0, beta_rate=0.004,
                                    beta_max=2000.0, step_scale=1.0,
                                    focus_schedule=(1.0, 0.3, 3000))
        obj5 = compute_obj(a5)
        print("  At n=%d: obj=%.6f (t=%.0fs)" % (n5, obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final intensive polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p6_budget = remaining * 0.90
        n_best = len(global_best_a)
        print("\n=== Phase 6: Final polish at n=%d (budget %.0fs) ===" % (n_best, p6_budget))
        t6 = time.monotonic()

        # Alternate between smooth and exact
        split = p6_budget * 0.5
        a6a, _ = run_fista_adaptive(global_best_a, split, rng,
                                     beta_init=50.0, beta_rate=0.002,
                                     beta_max=20000.0, step_scale=0.5,
                                     focus_schedule=(0.3, 0.05, 2000))
        obj6a = compute_obj(a6a)
        print("  Smooth final: obj=%.6f (t=%.0fs)" % (obj6a, time.monotonic() - t6))
        if obj6a < global_best_obj:
            global_best_obj = obj6a
            global_best_a = a6a.copy()

        remaining6 = p6_budget - (time.monotonic() - t6)
        if remaining6 > 5:
            a6b, _ = run_exact_polish(global_best_a, remaining6, rng)
            obj6b = compute_obj(a6b)
            print("  Exact final: obj=%.6f (t=%.0fs)" % (obj6b, time.monotonic() - t6))
            if obj6b < global_best_obj:
                global_best_obj = obj6b
                global_best_a = a6b.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
