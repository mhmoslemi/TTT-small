"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining FISTA with LSE smoothing,
LP-based direction finding (active constraints), and exact subgradient
polishing. Includes multi-scale upsampling for more degrees of freedom

Key innovations over baseline:
- LP-based descent direction using only active constraints
- Adaptive constraint set sizing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def convolve_cross(f, g):
    """Compute f * g (linear convolution) via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    return np.real(ifft(F * G))[:size]


def solve_lp_direction(f, M, K=20):
    """
    Solve LP to find best direction g:
      maximize sum(g)
      subject to (f*g)[k] <= M for top-K active lags
                 g >= 0
    
    Returns the optimal g (or None if infeasible/time-out).
    """
    from scipy.optimize import linprog
    n = len(f)
    
    # Compute current self-convolution to identify active lags
    c = fft_convolve_self(f)
    
    # Select top-K lags by value
    sorted_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix A_eq: rows correspond to selected lags
    # (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    # For lag k: g indices j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
    
    A_ub = np.zeros((K, n))
    for i, k in enumerate(sorted_idx):
        # (f*g)[k] = sum_{j} f[j] * g[k-j]
        # g index = k - j, so we need 0 <= k-j <= n-1
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            g_idx = k - j
            A_ub[i, g_idx] = f[j]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.full(K, M),
                         bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success:
            return result.x
        else:
            return None
    except Exception:
        return None


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, mode="lse", step_scale=1.0, lp_interval=0):
    """
    FISTA with adaptive LSE smoothing, optionally augmented with LP directions.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / beta_init
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # LP-based direction (if enabled and at the right interval)
        used_lp = False
        if lp_interval > 0 and it % lp_interval == 0 and stuck_count > 100:
            g = solve_lp_direction(y, cur_max, K=min(20, n // 500 + 5))
            if g is not None and g.sum() > 1e-10:
                # Rescale g to match normalization
                g = g * (S / g.sum())
                # Line search: try a few step sizes
                best_t = 0.1
                best_val = cur_max
                for t_try in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    candidate = (1 - t_try) * y + t_try * g
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate * (S / candidate.sum())
                    c_test = fft_convolve_self(candidate)
                    if c_test.max() < best_val:
                        best_val = c_test.max()
                        best_t = t_try
                if best_t > 0:
                    y = (1 - best_t) * y + best_t * g
                    y = np.maximum(y, 0)
                    y = y * (S / y.sum())
                    t_mom = 1.0  # Reset momentum
                    used_lp = True

        if not used_lp:
            # Gradient computation
            if mode == "lse":
                grad = lse_gradient(y, c, beta)
            elif mode == "hybrid":
                if stuck_count > 500 and (it % 5 == 0 or rng.random() < 0.2):
                    grad = exact_subgradient(y, c, tol_frac=0.01)
                else:
                    grad = lse_gradient(y, c, beta)
            elif mode == "topk":
                k = max(4, min(256, n // 8))
                if stuck_count > 1000:
                    k = max(1, k // 4)
                grad = topk_gradient(y, c, k, power=1.5)
            else:
                grad = lse_gradient(y, c, beta)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Adaptive step: reduce when stuck
            if stuck_count > 3000:
                step *= 0.3
                noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
                if noise_scale > 1e-6:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_mom = 1.0
                    stuck_count = 0
                    last_improve = it

            # FISTA update
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            # Periodic restart
            if it % 8000 == 0:
                t_mom = 1.0
                decay = np.exp(-it / 50000.0)
                if decay > 0.05:
                    noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                    y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact subgradient with adaptive top-k.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: start with 1, increase when stuck
        if stuck < 200:
            k = 1
            tol_frac = 0.0
        elif stuck < 1000:
            k = 3
            tol_frac = 0.005
        elif stuck < 5000:
            k = min(16, 4 + stuck // 500)
            tol_frac = 0.01
        else:
            k = min(64, 16 + stuck // 1000)
            tol_frac = 0.02

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step size
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-5, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

        # LP-based escape when very stuck
        if stuck > 3000 and it % 200 == 0:
            g = solve_lp_direction(y, cur_max, K=min(10, n // 1000 + 3))
            if g is not None and g.sum() > 1e-10:
                g = g * (S / g.sum())
                candidate = 0.5 * y + 0.5 * g
                candidate = np.maximum(candidate, 0)
                if candidate.sum() > 1e-10:
                    candidate = candidate * (S / candidate.sum())
                    c_test = fft_convolve_self(candidate)
                    if c_test.max() < cur_max:
                        y = candidate
                        t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def run_lp_refine(a_init, max_time, rng, K=15):
    """
    Dedicated LP-based refinement loop.
    Iteratively solves the LP direction problem and takes steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = f.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = f.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 50:
            # Try noise
            noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 200.0))
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            continue
        
        # Solve LP
        g = solve_lp_direction(f, cur_max, K=K)
        if g is None or g.sum() < 1e-10:
            # LP failed, try noise
            noise_scale = 0.002 * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            continue
        
        g = g * (S / g.sum())
        
        # Line search
        best_t = 0.0
        best_val = cur_max
        for t_try in [0.01, 0.03, 0.1, 0.2, 0.3, 0.5]:
            candidate = (1 - t_try) * f + t_try * g
            candidate = np.maximum(candidate, 0)
            if candidate.sum() < 1e-10:
                continue
            candidate = candidate * (S / candidate.sum())
            c_test = fft_convolve_self(candidate)
            if c_test.max() < best_val:
                best_val = c_test.max()
                best_t = t_try
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f * (S / f.sum())
        else:
            # No improvement, inject noise
            noise_scale = max(1e-5, 0.003 * S / np.sqrt(n))
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. FISTA with adaptive LSE smoothing for fast convergence
2. LP-based direction finding (active constraints) for principled descent
3. Multi-scale upsampling for more degrees of freedom
4. Active-set and exact subgradient polishing
5. Adaptive noise injection to escape local minima

Key innovations:
- LP-based descent direction using only top-K active convolution constraints
- Backtracking line search for LP directions
- Dedicated LP refinement phase
- Dual LP formulation for efficiency with small K
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def build_lp_matrix(f, sorted_idx, n):
    """
    Build the LP constraint matrix for the K active lags.
    Each row i corresponds to lag k_i, with (f*g)[k_i] = sum_j A[i,j]*g[j].
    """
    K = len(sorted_idx)
    A_ub = np.zeros((K, n))
    for i, k in enumerate(sorted_idx):
        k = int(k)
        if k < n:
            # g indices 0..k, coefficients f[k], f[k-1], ..., f[0]
            A_ub[i, 0:k+1] = f[k::-1]
        else:
            start = k - n + 1
            # g indices start..n-1, coefficients f[n-1], f[n-2], ..., f[start]
            A_ub[i, start:n] = f[n-1:start-1:-1]
    return A_ub


def solve_lp_direction(f, c, M, K=15, time_limit=15):
    """
    Solve LP to find best direction g:
      maximize sum(g)
      subject to (f*g)[k] <= M for top-K active lags
                 g >= 0
    Uses dual formulation for efficiency when K << n.
    """
    from scipy.optimize import linprog
    n = len(f)

    sorted_idx = np.argsort(c)[::-1][:K]

    # Use dual formulation:
    # Primal: max 1'g s.t. A_ub @ g <= M*1, g >= 0
    # Dual:   min M * 1'y s.t. A_ub' @ y >= 1, y >= 0
    # Dual has K variables and n constraints - much smaller!

    A_ub = build_lp_matrix(f, sorted_idx, n)

    # Dual: minimize M * sum(y) subject to A_ub.T @ y >= 1, y >= 0
    # Which is: minimize M*sum(y) subject to -A_ub.T @ y <= -1, y >= 0
    c_dual = M * np.ones(K)
    A_dual = -A_ub.T  # shape (n, K)
    b_dual = -np.ones(n)
    bounds_dual = [(0, None)] * K

    try:
        result = linprog(c_dual, A_ub=A_dual, b_ub=b_dual,
                         bounds=b_dual, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            # Recover primal solution from dual (complementary slackness)
            # Or solve primal with the LP warm start
            # For now, just solve primal directly
            c_obj = -np.ones(n)
            bounds = [(0, None)] * n
            result_p = linprog(c_obj, A_ub=A_ub, b_ub=np.full(len(sorted_idx), M),
                               bounds=bounds, method='highs',
                               options={'time_limit': time_limit, 'presolve': True,
                                        'dual_feasibility_tolerance': 1e-9,
                                        'primal_feasibility_tolerance': 1e-9})
            if result_p.success:
                return result_p.x
            else:
                # Try dual variables to construct a direction
                y_opt = result.x
                # g_j > 0 where (A_ub.T @ y)_j < 1 (complementary slackness)
                # This gives us the support of g
                At_y = A_ub.T @ y_opt
                support = At_y < (1.0 - 1e-6)
                if not np.any(support):
                    return None
                # On the support, g is determined by the active constraints
                # For simplicity, use a heuristic: g_j proportional to slack
                g = np.zeros(n)
                g[support] = 1.0
                # Verify constraints
                cross = A_ub @ g
                if np.all(cross <= M * (1 + 1e-6)):
                    return g
                return None
        else:
            return None
    except Exception:
        return None


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    """
    FISTA with adaptive LSE smoothing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / beta_init
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient computation
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "hybrid":
            if stuck_count > 500 and (it % 5 == 0 or rng.random() < 0.2):
                grad = exact_subgradient(y, c, tol_frac=0.01)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 1000:
                k = max(1, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: reduce when stuck
        if stuck_count > 3000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 8000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 50000.0)
            if decay > 0.05:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lp_refine(a_init, max_time, rng, K=12):
    """
    Dedicated LP-based refinement loop.
    Iteratively solves the LP direction problem and takes steps with line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = f.copy()
    last_improve = 0
    it = 0
    lp_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = f.copy()
            last_improve = it

        stuck = it - last_improve

        # If stuck for a while, inject noise and continue
        if stuck > 80:
            noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 300.0))
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            continue

        # Solve LP for direction
        t_lp_start = time.monotonic()
        g = solve_lp_direction(f, c, cur_max, K=K, time_limit=10)
        lp_time = time.monotonic() - t_lp_start
        lp_count += 1

        if g is None or g.sum() < 1e-10:
            noise_scale = 0.002 * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            continue

        g = g * (S / g.sum())

        # Line search: try several step sizes
        best_t = 0.0
        best_val = cur_max
        for t_try in [0.01, 0.03, 0.08, 0.15, 0.25, 0.4]:
            candidate = (1 - t_try) * f + t_try * g
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate * (S / cs)
            c_test = fft_convolve_self(candidate)
            if c_test.max() < best_val:
                best_val = c_test.max()
                best_t = t_try

        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            fs = f.sum()
            if fs > 1e-10:
                f = f * (S / fs)
        else:
            # No improvement from LP direction, small noise
            noise_scale = max(1e-5, 0.002 * S / np.sqrt(n))
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)

        if it % 50 == 0:
            print("    LP iter %d: conv=%.8f, stuck=%d, lp_time=%.2fs" % (it, cur_max, stuck, lp_time))

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    print("  LP refine: %d LPs solved in %.0fs" % (lp_count, time.monotonic() - start))
    return best_a, best_conv
