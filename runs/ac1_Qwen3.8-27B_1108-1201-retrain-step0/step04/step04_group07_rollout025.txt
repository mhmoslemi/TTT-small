with dual softmax smoothing, spectral pre-conditioning, and adaptive restarts.

Key improvements over previous approach:
1. Dual softmax smoothing: instead of hard top-k, use a temperature-based soft
   selection of active lags for smoother gradients.
2. Nesterov acceleration with proper Polyak step sizes.
3. Spectral pre-conditioning: iteratively shape the power spectrum before
   entering the main optimization loop.
4. Adaptive restarts: detect stagnation and restart with larger perturbations.
5. Solution averaging: average the last T% of iterates for variance reduction.
6. Multi-length search with budget allocation based on progress.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def spectral_precondition(a, N, n_iter, rng, alpha=0.3):
    """
    Iteratively flatten the power spectrum to reduce convolution peak.
    This serves as a pre-conditioner before the main subgradient loop.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max_val = c.max()

        # Adaptive flattening: reduce bins above geometric mean
        p_mean = np.exp(np.mean(np.log(np.maximum(P, 1e-30))))
        if p_mean < 1e-30:
            break

        ratio = P / p_mean
        # Compress high bins, expand low bins
        P_new = np.zeros_like(P)
        above = ratio > 1.0
        P_new[above] = P[above] * np.power(ratio[above], -alpha)
        P_new[~above] = P[~above] * np.power(ratio[~above], alpha * 0.3)
        P_new = np.maximum(P_new, 0.0)

        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new_full, S)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        if best_score <= 1.50:
            break

    return best_a, best_score


def nesterov_subgradient(a_init, n, S, max_time, rng, temperature=10.0,
                         top_k_start=3, top_k_end=1, restart_interval=2000,
                         target_score=1.50):
    """
    Nesterov-accelerated subgradient method with dual softmax smoothing.
    
    The dual variable p is a softmax over the convolution values, giving a
    smooth approximation to the indicator of the peak lag.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.time()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0
    N = 1 << (2 * n - 1).bit_length()

    # For averaging
    avg_a = a.copy()
    avg_count = 0
    avg_start = 0

    # Stagnation tracking
    last_improvement_it = 0
    patience = 3000
    stagnation_count = 0

    base_step = 2.0 * S / (n ** 0.5)

    while time.time() - start < max_time:
        it += 1
        # Compute convolution
        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)

        if sc < best_score - 1e-8:
            best_score = sc
            best_local = y.copy()
            last_improvement_it = it
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Stagnation detection and restart
        if it - last_improvement_it > patience and it % restart_interval == 0:
            perturbation = S * 0.01 * (1 + stagnation_count * 0.5)
            y = project_simplex(y + rng.normal(0, perturbation / np.sqrt(n), n), S)
            t = 1.0
            avg_a = y.copy()
            avg_count = 0
            last_improvement_it = it
            stagnation_count = 0
            patience = max(2000, patience * 0.7)

        # Compute smooth dual weights (softmax over convolution values)
        # Temperature adapts: start hot (smooth), end cold (sharp)
        progress = min(it / max(1, int(max_time * 2000)), 1.0)
        temp = temperature * (1.0 + 3.0 * progress)  # increase temperature over time
        # Actually, we want it to decrease (sharper gradient later)
        temp = temperature * (1.0 / (1.0 + 2.0 * progress))

        c_shifted = c - c.max()
        p = np.exp(c_shifted / temp)
        p /= p.sum()

        # Adaptive top-k: blend between top-k sharp and full softmax
        k_eff = int(np.ceil(top_k_start + (top_k_end - top_k_start) * progress))
        top_idx = np.argsort(c)[::-1][:max(k_eff, 1)]
        top_mask = np.zeros_like(c)
        top_mask[top_idx] = 1.0
        blend = 0.5 + 0.5 * progress  # more weight on sharp subset as we progress
        p_blended = blend * (top_mask * p) / max((top_mask * p).sum(), 1e-30) + \
                    (1 - blend) * p
        p_blended /= p_blended.sum()

        # Compute gradient: d(sum_k p[k]*c[k])/da
        # c[k] = sum_i a[i]*a[k-i], so d(c[k])/da[j] = a[k-j] + a[j-k] (with bounds)
        # Using FFT: grad = IFFT(FFT(p_blended) * conj(FFT(y)))
        # Actually: grad[j] = sum_k p[k] * (a[k-j] + a[j-k]) with bounds
        # This is: grad = p_blended * a (circular conv) + reverse(p_blended) * a (circular conv)
        # For linear conv, we need to be careful. Let's use the FFT approach.
        
        fsize = N
        W = fft(p_blended, fsize)
        A_y = fft(y, fsize)
        # The gradient of b[k] w.r.t. a is: (b shifted) conj(A) in freq domain
        # More precisely: grad = IFFT( (1 + Z^{-k}) * P * conj(A) ) but this is complex
        # Let's use the direct formula: grad[j] = sum_k p[k] * (y[(k-j) mod n] + y[(j-k) mod n])
        # which is the circular convolution of p with y, plus the reverse
        gy = np.real(ifft(W * A_y))[:n]  # circular conv of p and y
        p_rev = p_blended[::-1]
        W_rev = fft(p_rev, fsize)
        gy_rev = np.real(ifft(W_rev * A_y))[:n]
        grad = 2.0 * (gy + gy_rev)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: adaptive
        # Use a decreasing step with some adaptation
        step = base_step / (1.0 + 0.1 * np.sqrt(it))
        # Cap the step to avoid large jumps
        step = min(step, 0.5 * S / max(gnorm, 1e-10))

        x_new = project_simplex(y - step * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y = x_new + beta * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project y to maintain feasibility
        sy = y.sum()
        if sy > 0:
            y = y * (S / sy)
        a = x_new

        # Averaging (keep running average of last 20% of iterates)
        if it > avg_start:
            avg_a = (avg_a * avg_count + a) / (avg_count + 1)
            avg_count += 1
        if avg_count == 0 and it > restart_interval:
            avg_start = it
            avg_a = a.copy()
            avg_count = 1

        t = t_new

    # Check averaged solution
    if avg_count > 100:
        avg_score = compute_score(avg_a)
        if avg_score < best_score:
            best_score = avg_score
            best_local = avg_a.copy()

    return best_local, best_score

    # Sinusoidal perturbation of uniform
    for freq in [1, 2, 3, 5]:
        inits.append(0.5 + 0.3 * np.sin(2 * np.pi * freq * tt + rng.uniform(0, 2 * np.pi)))

    # Cosine window (higher in middle)
    inits.append(0.3 + 0.7 * np.cos(np.pi * (tt - 0.5)) ** 2)

    # Asymmetric (higher at one end)
    inits.append(0.2 + 0.6 * (1 - tt))
    inits.append(0.2 + 0.6 * tt)

    # Bilinear
    inits.append(0.3 + 0.4 * np.sin(np.pi * tt))

    # Two-scale structure
    coarse = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * tt)
    fine = 0.1 * np.sin(2 * np.pi * 17 * tt)
    inits.append(np.clip(coarse + fine, 0.05, 0.95))

    return inits


def optimize_single_length(n, budget_s, rng, target_score=1.50):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    global_best = np.inf
    global_a = None

    t_start = time.time()

    # Try to load the provided sequence if it matches this length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
            pass
    except Exception:
        given = None

    inits = make_initializations(n, rng)

    # If we have a provided sequence of matching length, use it as first init
    if given is not None:
        inits.insert(0, np.clip(given, 0, 1000).astype(np.float64))

    n_inits = len(inits)
    # Allocate budget: more time for first few inits (they're more promising)
    # Use a decreasing allocation
    weights = [max(0.3, 1.0 - 0.3 * i / n_inits) for i in range(n_inits)]
    total_w = sum(weights)
    init_budgets = [budget_s * 0.8 * w / total_w for w in weights]

    for i, init_arr in enumerate(inits):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.85:
            break

        remaining = min(init_budgets[i], budget_s * 0.85 - elapsed)
        if remaining < 5:
            break

        a = np.clip(init_arr, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        # Stage 1: Spectral pre-conditioning (short)
        pre_time = min(remaining * 0.2, 30.0)
        if pre_time > 2:
            n_pre = max(50, int(pre_time * 2000))
            a, _ = spectral_precondition(a, N, n_pre, rng, alpha=0.4)

        # Stage 2: Main subgradient optimization
        sg_time = min(remaining * 0.7, max(5, remaining - 5))
        if sg_time > 3:
            a_sg, sc_sg = nesterov_subgradient(a, n, S, sg_time, rng,
                                               temperature=8.0,
                                               top_k_start=5, top_k_end=1,
                                               restart_interval=1500,
                                               target_score=target_score)
            a = a_sg
            sc = compute_score(a)
            if sc < global_best:
                global_best = sc
                global_a = a.copy()
                print("    n=%d init=%d: score=%.6f (elapsed %.0fs)" %
                      (n, i, global_best, time.time() - t_start))

        # Stage 3: Fine polishing (short, very small steps)
        polish_time = min(remaining * 0.1, 15.0)
        if polish_time > 2 and global_best < 1.51:
            # Fine-tune with very low temperature (sharp gradient)
            a_fine, sc_fine = nesterov_subgradient(a, n, S, polish_time, rng,
                                                   temperature=2.0,
                                                   top_k_start=1, top_k_end=1,
                                                   restart_interval=5000,
                                                   target_score=target_score)
            if sc_fine < global_best:
                global_best = sc_fine
                global_a = a_fine.copy()
                print("    n=%d init=%d polished: score=%.6f" % (n, i, global_best))

    # Final polish on the best found
    if global_a is not None:
        final_time = max(0, budget_s - (time.time() - t_start))
        if final_time > 10:
            # Run one more round of fine optimization
            a_final, sc_final = nesterov_subgradient(global_a, n, S,
                                                     min(final_time * 0.6, 60.0),
                                                     rng, temperature=3.0,
                                                     top_k_start=2, top_k_end=1,
                                                     restart_interval=3000,
                                                     target_score=target_score)
            if sc_final < global_best:
                global_best = sc_final
                global_a = a_final.copy()

            # Try a spectral polish on the result
            rem = max(0, budget_s - (time.time() - t_start))
            if rem > 5:
                a_sp, sc_sp = spectral_precondition(global_a, N,
                                                    int(rem * 1000), rng, alpha=0.1)
                if sc_sp < global_best:
                    global_best = sc_sp
                    global_a = a_sp.copy()

    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov-accelerated subgradient with dual softmax, spectral
    pre-conditioning, adaptive restarts, and multi-length search.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.time() + max(0.0, budget_s - 30.0)

    print("Starting optimization with budget %.0fs" % budget_s)

    # Determine lengths to try
    lengths_to_try = [8192, 4096, 16384]

    # If provided sequence has a different useful length, add it
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 2048 and given.size not in lengths_to_try:
            lengths_to_try.append(given.size)
            # Reorder: provided length first
            lengths_to_try.remove(given.size)
            lengths_to_try.insert(0, given.size)
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    t_global_start = time.time()

    # Allocate budget: prioritize n=8192 (balance of DOF and speed)
    # and the provided length if available
    total_budget = budget_s - 40  # reserve 40s for verification

    # Weights: favor 8192 and 16384
    weights = {}
    for L in lengths_to_try:
        if L == 8192:
            weights[L] = 1.5
        elif L >= 16384:
            weights[L] = 1.0
        elif L >= 4096:
            weights[L] = 0.8
        else:
            weights[L] = 0.5
    total_w = sum(weights.values())
    budgets = {L: total_budget * weights[L] / total_w for L in lengths_to_try}

    # Process lengths in priority order (8192 first if available)
    lengths_sorted = sorted(lengths_to_try, key=lambda x: (-weights[x], x))

    for n in lengths_sorted:
        elapsed = time.time() - t_global_start
        remaining_global = deadline - time.time()
        if remaining_global < 30:
            break

        alloc = min(budgets[n], remaining_global - 20)
        if alloc < 20:
            break

        print("=== Optimizing n=%d with budget %.0fs (global best: %s) ===" %
              (n, alloc, "%.6f" % global_best if np.isfinite(global_best) else "inf"))

        a_result, score = optimize_single_length(n, alloc, rng, target_score=1.50)

        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  New global best: %.6f at n=%d (elapsed %.0fs)" %
                  (global_best, n, time.time() - t_global_start))
        else:
            print("  Best for n=%d: %.6f (global: %s)" %
                  (n, score, "%.6f" % global_best if np.isfinite(global_best) else "inf"))

    # Cross-pollination: if we have results for multiple lengths,
    # try to use the best as initialization for other lengths (resize)
    if global_a is not None and time.time() < deadline - 60:
        pass  # Cross-pollination is complex, skip for now

    # Final verification and fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence, score=%.4f" % global_best)

    # Ensure valid output
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Final verification
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0

    print("Final answer: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
