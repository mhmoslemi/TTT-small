def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:

    return a * (S / s)

    M = c.max()

        return f.copy()

    idx = np.argsort(c)[::-1][:top_k]

    A = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        A[i, js] = f[k - js]
    
    # Solve: min -sum(g) s.t. A@g <= M, g >= 0
    cost = -np.ones(n)
    b = np.full(top_k, M)

        try:

        except Exception:
            return None

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Iterative LP best-response (using scipy linprog/HiGHS) with
adaptive active-set constraints, combined with FISTA polishing using
LSE-smoothed gradient and multiplicative repulsion updates.
Key improvements: faster LP via scipy (no CVXPY overhead), aggressive
time management, adaptive constraint selection, and multiple polish phases.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    M = c.max()

        return f.copy()

    deadline = time.monotonic() + max(60.0, budget_s - 60.0)

    # --- Get initial sequence ---
    n = 8192
    f = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if raw.size >= 512 and raw.size <= 65536:
            raw = np.clip(raw, 0, 1000)
            if raw.sum() >= 0.01:
                n = raw.size
                f = raw
    except Exception:
        pass

    if f is None:
        f = np.ones(n) * 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n, endpoint=False))
        f = np.maximum(f, 0.01)

    S = np.sqrt(2.0 * n)
    f = normalize_sum(f, n)

    # --- Initial evaluation ---
    c0 = fft_convolve_self(f)
    best_max = c0.max()
    best_f = f.copy()
    obj = 2.0 * n * best_max / S**2
    print("Start: n=%d, max_conv=%.8f, obj=%.6f" % (n, best_max, obj))

    # === PHASE 1: LP iterations with FISTA interspersed ===
    phase1_frac = 0.50
    total_time = budget_s - 60.0
    phase1_deadline = time.monotonic() + total_time * phase1_frac

    print("Phase 1: LP+FISTA (deadline in %.0fs)" % (phase1_deadline - time.monotonic()))
    lp_count = 0
    stagnant = 0
    adaptive_topk = 150

    while time.monotonic() < phase1_deadline:
        # LP iteration
        f_new, new_max = lp_iteration(f, S, top_k=adaptive_topk)
        lp_count += 1

        if new_max < best_max * 0.9995:
            # Improvement found
            if new_max < best_max:
                best_max = new_max
                best_f = f_new.copy()
                obj = 2.0 * n * best_max / S**2
            f = f_new
            stagnant = 0
            # Increase top_k slightly for more constraints as we converge
            adaptive_topk = min(400, 100 + lp_count // 5)
            if lp_count % 10 == 1:
                print("  LP %3d: max=%.8f, obj=%.6f (top_k=%d)" % (lp_count, best_max, obj, adaptive_topk))
        else:
            stagnant += 1
            f = f_new

        # If stagnant, try escape strategies
        if stagnant >= 8:
            # Strategy 1: Multiplicative repulsion
            f_rep = multiplicative_repulsion(f, S, beta=30.0, lr=0.04)
            c_rep = fft_convolve_self(f_rep)
            if c_rep.max() < best_max:
                best_max = c_rep.max()
                best_f = f_rep.copy()
                f = f_rep
                obj = 2.0 * n * best_max / S**2
                stagnant = 0
                print("  Repulsion escape %d: %.8f" % (lp_count, best_max))
            else:
                # Strategy 2: Short FISTA burst
                remaining_p1 = phase1_deadline - time.monotonic()
                if remaining_p1 > 15:
                    f_f, fista_max = fista_run(f, S, min(20.0, remaining_p1 * 0.3), rng,
                                                beta_init=50.0, beta_rate=0.01, beta_max=800.0,
                                                repulsion_every=200, repulsion_lr=0.04)
                    if fista_max < best_max:
                        best_max = fista_max
                        best_f = f_f.copy()
                        f = f_f
                        obj = 2.0 * n * best_max / S**2
                        stagnant = 0
                        print("  FISTA escape %d: %.8f" % (lp_count, best_max))
                    else:
                        # Strategy 3: Noise perturbation
                        noise_scale = 0.003 * S / np.sqrt(n)
                        f = project_simplex(f + rng.normal(0, noise_scale, n), S)
                        stagnant = 0

        # Periodic FISTA polish between LP iterations
        if lp_count % 5 == 0:
            remaining_p1 = phase1_deadline - time.monotonic()
            if remaining_p1 > 30:
                f_poly, poly_max = fista_run(f, S, min(30.0, remaining_p1 * 0.15), rng,
                                             beta_init=30.0, beta_rate=0.005, beta_max=500.0,
                                             repulsion_every=400, repulsion_lr=0.03)
                if poly_max < best_max:
                    best_max = poly_max
                    best_f = f_poly.copy()
                    f = f_poly
                    obj = 2.0 * n * best_max / S**2
                    print("  FISTA polish (LP %d): %.8f" % (lp_count, best_max))

    print("Phase 1 done: %d LP iters, best obj=%.6f" % (lp_count, obj))

    # === PHASE 2: Intensive FISTA with high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("Phase 2: High-beta FISTA (%.0fs)" % remaining)
        f2, max2 = fista_run(best_f, S, remaining * 0.6, rng,
                             beta_init=200.0, beta_rate=0.003, beta_max=5000.0,
                             repulsion_every=500, repulsion_beta=80.0, repulsion_lr=0.02)
        if max2 < best_max:
            best_max = max2
            best_f = f2.copy()
            obj = 2.0 * n * best_max / S**2
            print("  Phase 2 improved: obj=%.6f" % obj)
        else:
            print("  Phase 2: no improvement (best=%.6f)" % obj)

    # === PHASE 3: LP on best_f with more constraints ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 3: Final LP with more constraints (%.0fs)" % remaining)
        for topk in [300, 200, 150]:
            if time.monotonic() > deadline - 30:
                break
            f3, max3 = lp_iteration(best_f, S, top_k=topk)
            if max3 < best_max:
                best_max = max3
                best_f = f3.copy()
                obj = 2.0 * n * best_max / S**2
                print("  LP top_k=%d: obj=%.6f" % (topk, obj))

    # === PHASE 4: Multiplicative repulsion sweep ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("Phase 4: Repulsion sweep (%.0fs)" % remaining)
        for beta_r in [10, 30, 60, 100, 200]:
            for lr_r in [0.01, 0.03, 0.05, 0.1]:
                if time.monotonic() > deadline - 25:
                    break
                f_r = multiplicative_repulsion(best_f, S, beta=beta_r, lr=lr_r)
                c_r = fft_convolve_self(f_r)
                if c_r.max() < best_max:
                    best_max = c_r.max()
                    best_f = f_r.copy()
                    obj = 2.0 * n * best_max / S**2
                    print("  Rep(b=%d,l=%.2f): obj=%.6f" % (beta_r, lr_r, obj))

    # === FINAL: Verify and return ===
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_list = [float(x) for x in best_f]

    # Safety check
    if len(final_list) < 1 or sum(final_list) < 0.01:
        final_list = [float(x) for x in normalize_sum(np.ones(256), 256)]

    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("FINAL VERIFIED: n=%d, obj=%.6f" % (len(final_list), ver))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            final_list = [float(x) for x in normalize_sum(np.ones(256), 256)]
    except Exception as e:
        print("Verification error: %s" % str(e))
        final_list = [float(x) for x in normalize_sum(np.ones(256), 256)]

    elapsed = time.monotonic() - (deadline - total_time)
    print("Total time: %.1fs, returning length %d" % (elapsed, len(final_list)))
    return final_list
