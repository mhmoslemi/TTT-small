```python
"""
Natural-gradient search for non-negative sequences minimizing
    2*n*max(np.convolve(a,a)) / sum(a)**2.

The main idea is to optimize in log-density coordinates.  For probability
distribution a (sum(a)=1), full self-convolution c=a*a is computed by FFT.
For a soft active-set weight q over the lags, the stationarity direction for
the controlled objective is h_i = sum_k q_k a_{k-i} - C, where
C = sum_k q_k c_k.  Moving z_i <- z_i - alpha*(h_i - C) in log-space and
renormalizing is a natural-gradient step that keeps all entries non-negative.
Exact line search over several alpha-values is performed against the true
objective.  The method is run in multiple tau stages, multiple starts, and
optionally at one larger sequence length.
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def normalize(a):
    """Project to the non-negative simplex and return as numpy array."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        if a.size == 0:
            a = np.ones(1, dtype=np.float64)
        return a / float(a.size)
    return a / s


def normalize_to_length(seq, n, rng=None):
    """
    Convert an input sequence to a probability vector of length n.
    If the length differs, preserve the continuous density by interpolation.
    """
    n = int(n)
    arr = np.asarray(seq, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)

    if arr.size == n:
        a = arr
    else:
        # density = height * length, approximately integral-preserving
        density = arr * float(arr.size)
        if float(density.sum()) <= 1e-15:
            return np.ones(n, dtype=np.float64) / float(n)
        x_old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
        density_new = np.interp(x_new, x_old, density)
        a = density_new * float(n)

    a = np.maximum(a, 0.0)
    return normalize(a)


def self_conv_unit(a):
    """Full self-convolution of a probability vector using FFT."""
    n = int(a.size)
    length = 2 * n - 1
    nfft = 1 << max(1, (length - 1).bit_length())
    A = fft(a, nfft)
    c = ifft(A * A).real[:length]
    np.maximum(c, 0.0, out=c)
    return c


def score_unit(a):
    """Objective for a probability vector (sum(a)=1)."""
    n = int(a.size)
    c = self_conv_unit(a)
    return float(2.0 * n * float(np.max(c)))


def softmax_from_z(z):
    """Stable softmax returning a probability vector."""
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.zeros(z.shape, dtype=np.float64)
        e[int(np.argmax(z))] = 1.0
        s = 1.0
    return e / s


def softmax_weights(D, tau):
    """Stable softmax probability vector over lag energies D."""
    L = int(D.size)
    if tau <= 0.0 or not np.isfinite(tau):
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    x = tau * D
    x = np.where(np.isfinite(x), x, -np.inf)
    m = float(np.max(x))
    if not np.isfinite(m):
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    e = np.exp(x - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(np.isfinite(x) & (np.abs(x - m) <= 1e-12))[0]
        if idx.size == 0:
            return np.full(L, 1.0 / float(L), dtype=np.float64)
        q = np.zeros(L, dtype=np.float64)
        q[idx] = 1.0 / float(idx.size)
        return q
    return e / s


def cross_corr_q_a(q, a):
    """
    Return h_i = sum_k q_k a_{k-i}, for i=0,...,n-1.
    q has length 2n-1, a has length n.
    """
    n = int(a.size)
    L = 2 * n - 1
    p = q[::-1]
    conv_len = n + L - 1
    nfft = 1 << max(1, (conv_len - 1).bit_length())
    A = fft(a, nfft)
    P = fft(p, nfft)
    r = ifft(A * P).real
    return r[n - 1:L][::-1]


def jitter_sequence(a, rng, std):
    """Add log-normal noise and renormalize."""
    z = np.log(a + 1e-300)
    z = z - float(np.mean(z))
    z = z + rng.normal(0.0, std, size=a.size)
    z = z - float(np.mean(z))
    return softmax_from_z(z)


def make_inits(n, provided, rng):
    """Make a small collection of diverse tapered initializations."""
    inits = []

    if provided is not None:
        a0 = normalize_to_length(provided, n, rng)
        inits.append(("provided", a0))
        inits.append(("provided_jit", jitter_sequence(a0, rng, 0.25)))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    taper = x * (1.0 - x)

    # Asymmetric smooth bumps, forced low at both boundaries.
    y = (
        0.01 * taper
        + 1.00 * np.exp(-8.0 * (x - 0.32) ** 2)
        + 0.55 * np.exp(-6.0 * (x - 0.71) ** 2)
        + 0.05
    )
    inits.append(("asym_bumps", normalize(y)))

    # Tapered cosine with random phase.
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    y = 0.02 * taper * (1.0 + 0.9 * np.cos(2.0 * np.pi * x + phase)) + 0.04
    y = np.maximum(y, 0.0)
    inits.append(("cos_taper", normalize(y)))

    # Smoothed random density with taper.
    noise = rng.standard_normal(n)
    k = max(3, n // 24)
    kernel = np.ones(k, dtype=np.float64) / float(k)
    sm = np.convolve(noise, kernel, mode="same")
    sm = (sm - float(np.min(sm))) / max(float(np.max(sm) - np.min(sm)), 1e-12)
    y = 0.05 * taper * (0.1 + 0.9 * sm) + 0.03
    inits.append(("rand_taper", normalize(y)))

    # Mild one-sided ramp with taper.
    y = 0.05 * taper * (1.0 + 1.2 * float(x) ** 1.5) + 0.03
    inits.append(("ramp_taper", normalize(y)))

    return inits


def standard_phases(min_tau=0.0):
    """
    Return tau stages with normalized time fractions and initial step sizes.
    Smaller tau is more global, larger tau is more peak-directed.
    """
    phases = [
        (2.0, 0.07, 0.50),
        (5.0, 0.11, 0.35),
        (10.0, 0.15, 0.25),
        (20.0, 0.16, 0.18),
        (40.0, 0.16, 0.12),
        (80.0, 0.14, 0.08),
        (200.0, 0.12, 0.05),
        (500.0, 0.10, 0.03),
    ]
    if min_tau > 0.0:
        selected = [(t, f, a) for (t, f, a) in phases if t >= min_tau]
        if not selected:
            selected = [phases[-1]]
    else:
        selected = phases
    total = sum(f for _, f, _ in selected)
    if total <= 0.0:
        total = 1.0
    return [(t, f / total, a) for (t, f, a) in selected]


def gradient_phase(cur_a, cur_z, cur_score, best_a, best_score,
                   n, tau, time_budget, alpha0, rng, verbose=False):
    """
    Run natural-gradient line search for a fixed soft-max tau.
    Mutates/updates local best only; returns updated state.
    """
    if time_budget <= 0.5:
        return cur_a, cur_z, cur_score, best_a, best_score

    start = time.monotonic()
    deadline = start + float(time_budget)

    alpha = float(alpha0)
    no_improve = 0
    reset_count = 0
    it = 0
    last_print = float(best_score)

    mults = (1.0, 0.5, 2.0, 0.25, 4.0, 0.125)

    while True:
        now = time.monotonic()
        if now >= deadline - 0.05:
            break

        it += 1

        # Current score from exact convolution.
        c = self_conv_unit(cur_a)
        cur_score = float(2.0 * n * float(np.max(c)))
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        D = c * float(n)
        q = softmax_weights(D, tau)

        # Natural-gradient direction for current soft active set.
        h = cross_corr_q_a(q, cur_a)
        C = float(np.dot(q, c))
        d = float(n) * (h - C)

        if not np.all(np.isfinite(d)):
            break

        # additive constant has no effect after softmax normalization
        d = d - float(np.mean(d))
        span = float(np.max(d) - np.min(d))
        if not np.isfinite(span) or span < 1e-14:
            break
        d = d / span

        # Line search over multiple step sizes.
        best_try_score = np.inf
        best_try_a = None
        best_try_z = None
        best_al = alpha

        for mult in mults:
            al = alpha * float(mult)
            if al <= 1e-8 or not np.isfinite(al):
                continue

            z_try = cur_z - al * d
            if not np.all(np.isfinite(z_try)):
                continue

            z_try = z_try - float(np.mean(z_try))
            zspan = float(np.max(z_try) - np.min(z_try))
            if zspan > 120.0:
                continue

            a_try = softmax_from_z(z_try)
            sc = score_unit(a_try)
            if not np.isfinite(sc):
                continue

            if sc < best_try_score:
                best_try_score = sc
                best_try_a = a_try
                best_try_z = z_try
                best_al = al

        if best_try_a is not None and best_try_score < cur_score - 1e-11:
            alpha = min(3.0, alpha * 1.25)
            no_improve = 0
            cur_a = best_try_a
            cur_z = best_try_z
            cur_score = best_try_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        else:
            no_improve += 1
            alpha *= 0.5

            if no_improve >= 30 or alpha < 1e-5:
                if reset_count >= 12:
                    break

                # Perturb the best known solution in log-space.
                z = np.log(best_a + 1e-300)
                z = z - float(np.mean(z))
                z = z + rng.normal(0.0, 0.12, size=n)
                z = z - float(np.mean(z))
                cur_z = z
                cur_a = softmax_from_z(z)
                cur_score = score_unit(cur_a)

                alpha = float(alpha0)
                no_improve = 0
                reset_count += 1

        if verbose and (it == 1 or it % 500 == 0 or best_score < last_print - 1e-7):
            print("  [grad] n=%d tau=%.1f it=%d best=%.6f cur=%.6f alpha=%.4f"
                  % (int(n), float(tau), int(it), float(best_score),
                     float(cur_score), float(alpha)))
            last_print = float(best_score)

    return cur_a, cur_z, cur_score, best_a, best_score


def natural_optimize(a_init, n, time_budget, rng, verbose=True, min_tau=0.0):
    """Run staged natural-gradient line search and return best probability vector."""
    n = int(n)
    a = normalize_to_length(a_init, n, rng)
    z = np.log(a + 1e-300)
    z = z - float(np.mean(z))

    cur_a = a
    cur_z = z
    cur_score = score_unit(a)
    best_a = a.copy()
    best_score = cur_score

    if time_budget <= 1.0:
        return best_a, best_score

    overall_start = time.monotonic()
    overall_deadline = overall_start + float(time_budget)

    phases = standard_phases(min_tau)
    remaining_frac = sum(f for _, f, _ in phases)
    if remaining_frac <= 0.0:
        return best_a, best_score

    if verbose:
        print("natural_optimize n=%d min_tau=%.1f budget=%.1fs"
              % (int(n), float(min_tau), float(time_budget)))

    for tau, frac, alpha0 in phases:
        rem_time = overall_deadline - time.monotonic()
        if rem_time < 2.0:
            break

        phase_budget = rem_time * (frac / max(remaining_frac, 1e-12))
        remaining_frac -= frac

        if phase_budget < 1.0:
            phase_budget = min(1.0, rem_time)

        cur_a, cur_z, cur_score, best_a, best_score = gradient_phase(
            cur_a, cur_z, cur_score, best_a, best_score,
            n, tau, phase_budget, alpha0, rng, verbose=verbose
        )

    # Final very peaky polishing with any remaining time.
    rem_time = overall_deadline - time.monotonic()
    if rem_time > 5.0:
        cur_a, cur_z, cur_score, best_a, best_score = gradient_phase(
            cur_a, cur_z, cur_score, best_a, best_score,
            n, 2000.0, rem_time, 0.01, rng, verbose=verbose
        )

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative height sequence minimizing evaluate_sequence.
    Uses multi-start tapered initializations and staged natural-gradient
    line search in log-density coordinates.
    """
    start_global = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2300.0

    # Reserve a little time; cap to a safe value even if more is available.
    usable = max(60.0, min(budget, 2350.0) - 50.0)
    deadline = start_global + usable

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    # Optional provided previous state-of-the-art sequence.
    provided = None
    try:
        provided = globals().get("height_sequence_1", None)
        if provided is not None:
            provided = np.asarray(provided, dtype=np.float64).ravel()
            provided = np.clip(provided, 0.0, 1000.0)
            provided = np.where(np.isfinite(provided), provided, 0.0)
            if provided.size < 16:
                provided = None
    except Exception:
        provided = None

    # Main search length: use provided length if reasonable, else 8192.
    if provided is not None and provided.size >= 1024:
        n_main = int(provided.size)
        if n_main > 32768:
            n_main = 16384
    else:
        n_main = 8192

    inits = make_inits(n_main, provided, rng)

    global_best = None
    global_best_score = np.inf

    # Initial scoring.
    for name, a in inits:
        sc = score_unit(a)
        print("init n=%d name=%s score=%.6f" % (int(n_main), name, float(sc)))
        if sc < global_best_score:
            global_best_score = float(sc)
            global_best = a.copy()

    if global_best is None:
        global_best = np.ones(n_main, dtype=np.float64) / float(n_main)
        global_best_score = 2.0

    # Allocate main budget.
    main_budget = usable * 0.78
    main_deadline = start_global + main_budget

    n_inits = len(inits)
    has_provided = provided is not None

    if has_provided:
        fracs = [0.70]
        if n_inits > 1:
            rest = 0.30 / float(n_inits - 1)
            fracs.extend([rest] * (n_inits - 1))
        else:
            fracs = [1.0]
    else:
        fracs = [1.0 / float(n_inits)] * n_inits

    for i, (name, a) in enumerate(inits):
        rem_main = main_deadline - time.monotonic() - 2.0
        if rem_main < 30.0:
            break

        budget_i = main_budget * float(fracs[i])
        budget_i = min(float(max(30.0, budget_i)), float(max(0.0, rem_main)))
        if budget_i < 30.0:
            continue

        min_tau = 10.0 if (has_provided and i == 0) else 2.0
        print("=== main init %d/%d name=%s n=%d budget=%.0fs min_tau=%.1f ==="
              % (i + 1, n_inits, name, int(n_main), float(budget_i), float(min_tau)))

        a_out, sc_out = natural_optimize(
            a, n_main, budget_i, rng, verbose=True, min_tau=min_tau
        )
        if sc_out < global_best_score:
            global_best_score = float(sc_out)
            global_best = a_out.copy()
            print("=== new global best score=%.6f (main n=%d) ==="
                  % (float(global_best_score), int(n_main)))

    # Optionally explore one larger length, especially if the provided length is 8192.
    n_large = 0
    if n_main <= 8192:
        n_large = 16384

    if n_large > 0 and time.monotonic() < deadline - 250.0:
        large_budget = min(usable * 0.16, deadline - time.monotonic() - 60.0)
        if large_budget > 120.0:
            init_large = normalize_to_length(global_best, n_large, rng)
            print("=== large length n=%d budget=%.0fs ===" % (int(n_large), float(large_budget)))
            a_large, sc_large = natural_optimize(
                init_large, n_large, large_budget, rng, verbose=True, min_tau=10.0
            )
            if sc_large < global_best_score:
                global_best_score = float(sc_large)
                global_best = a_large.copy()
                print("=== new global best score=%.6f (large n=%d) ==="
                      % (float(global_best_score), int(n_large)))

    # Final short high-tau polish on the globally best sequence.
    rem_final = deadline - time.monotonic() - 15.0
    if global_best is not None and rem_final > 30.0:
        final_budget = min(rem_final, usable * 0.10)
        if final_budget > 30.0:
            print("=== final polish n=%d budget=%.0fs ==="
                  % (int(global_best.size), float(final_budget)))
            a_fin, sc_fin = natural_optimize(
                global_best, int(global_best.size), final_budget,
                rng, verbose=True, min_tau=200.0
            )
            if sc_fin < global_best_score:
                global_best_score = float(sc_fin)
                global_best = a_fin.copy()

    # Sanitize answer.
    ans = np.asarray(global_best, dtype=np.float64).ravel()
    ans = np.where(np.isfinite(ans), ans, 0.0)
    ans = np.clip(ans, 0.0, 1000.0)
    s = float(ans.sum())
    if not np.isfinite(s) or s <= 0.0:
        ans = np.ones(256, dtype=np.float64)
        s = float(ans.sum())
    ans = ans / s
    ans = np.clip(ans, 0.0, 1000.0)

    local_score = score_unit(ans)
    print("FINAL n=%d internal_score=%.6f elapsed=%.1fs"
          % (int(ans.size), float(local_score), time.monotonic() - start_global))

    return [float(x) for x in ans]
```
