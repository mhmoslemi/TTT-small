"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Multi-phase optimization: diverse FISTA exploration, LSE-annealed refinement,
exact peak-shaving polish, and greedy peak elimination.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    delta = (c - cm).max()
    if delta > 0 and beta * delta > 500:
        beta = 500.0 / delta
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta, fsize=None):
    """Gradient of LSE-smoothed max of convolution with respect to a."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def peak_subgradient(a, c, k_top=8, fsize=None):
    """Exact subgradient from the top-k lags."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    k = min(k_top, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.ones(k) / k
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0


def excess_loss_gradient(a, c, T, p=2.0, fsize=None):
    """Gradient of sum((b_k - T)^p_+)."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = (excess > 0).astype(np.float64)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj

def _conv_self(a, fsize):
    """Fast self-convolution using precomputed FFT size."""
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]

def greedy_peak_elimination(a_init, S, max_time, rng, fsize=None):
    """
    Greedy approach: find the peak lag, identify contributing elements,

    """

        # Let me just use the gradient: db_k/da_j = a[(j+k-(n-1)) % ...]

        w_full[k_star] = 1.0

        # Reduce the elements with highest contribution

        reduction = 0.01 * S / n
        a_modified = a.copy()

def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if given_seq is not None:
        gs = np.asarray(given_seq, dtype=np.float64)
        if len(gs) == n:
            inits.append(np.maximum(gs, 0.0))
        elif len(gs) > n:
            # Downsample
            ratio = len(gs) // n
            small = gs[:n * ratio].reshape(n, ratio).mean(axis=1)
            inits.append(np.maximum(small, 0.0))
        else:
            # Upscale
            t_old = np.linspace(0, 1, len(gs))
            inits.append(np.maximum(np.interp(tt, t_old, gs), 0.0))
    
    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(np.sin(np.pi * tt) ** 2)
    
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5) + 0.15 * np.cos(2 * np.pi * 7 * tt))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))
    
    # Block
    for nb in [8, 16, 32]:
        bl = n // nb
        h = rng.uniform(0.3, 1.5, nb)
        bs = np.repeat(h, bl)
        if len(bs) < n:
            bs = np.concatenate([bs, np.full(n - len(bs), h[-1])])
        inits.append(np.maximum(bs[:n], 0.01))
    
    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    fsize = 1 << (2 * n - 1).bit_length()
    b = _conv_self(a, fsize)
    return 2.0 * n * b.max() / (s * s)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)

def _conv_self(a, fsize):
    """Fast self-convolution using precomputed FFT size."""
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Compute LSE-smoothed max."""
    cm = c.max()
    delta = (c - cm).max()
    if delta > 0 and beta * delta > 500:
        beta = 500.0 / delta
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta, fsize):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    delta = (c - cm).max()
    if delta > 0 and beta * delta > 500:
        beta = 500.0 / delta
    scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_subgradient(a, c, k_top=8, fsize=None):
    """Exact subgradient from the top-k lags, equally weighted."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    k = min(k_top, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_peak_subgradient(a, c, k_top=16, fsize=None):
    """Subgradient weighted by excess above mean."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    k = min(k_top, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_backtrack(a_init, S, max_time, rng, beta_init=5.0, beta_end=10000.0,
                    fsize=None, total_iters_hint=10000):
    """
    FISTA with LSE smoothing, geometric beta annealing, and backtracking step.
    """
    n = len(a_init)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()

    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = _conv_self(a, fsize)
    best_val = c.max()
    best_a = a.copy()
    it = 0
    stall = 0
    step = 2.0 / beta_init  # initial step (will be backtracked)

    while time.monotonic() - start < max_time:
        it += 1
        # Geometric beta schedule
        progress = min(1.0, it / max(1, total_iters_hint))
        beta = beta_init * (beta_end / beta_init) ** (progress ** 0.6)

        c = _conv_self(y, fsize)
        cur_max = c.max()
        fval = lse_value(c, beta)

        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = max(step * 1.2, 1.0 / (2.0 * beta * max(y.max(), 1e-10)))
        step = min(step, 5.0 / beta)
        x_new = None
        for _ in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = _conv_self(x_new, fsize)
            f_new = lse_value(c_new, beta)
            # Descent condition (loose: allow small increase)
            if f_new <= fval + 1e-8 * max(1.0, abs(fval)) + (0.5 / step) * np.dot(x_new - y, x_new - y):
                break
            step *= 0.5

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart: if stuck, perturb from best
        if stall > 3000:
            noise_scale = 0.004 * S / np.sqrt(n) * np.exp(-stall / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0
                stall = 0
                step = 2.0 / beta

        if it % 5000 == 0:
            t_mom = 1.0

    # Final check
    c_final = _conv_self(a, fsize)
    if c_final.max() < best_val:
        best_val = c_final.max()
        best_a = a.copy()

    return best_a, best_val


def fista_fast(a_init, S, max_time, rng, beta_init=5.0, beta_end=5000.0,
               fsize=None, total_iters_hint=20000):
    """
    FISTA with fixed step (faster, no backtracking) for quick runs.
    """
    n = len(a_init)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()

    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = _conv_self(a, fsize).max()
    best_a = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, total_iters_hint))
        beta = beta_init * (beta_end / beta_init) ** (progress ** 0.7)

        c = _conv_self(y, fsize)
        cur_max = c.max()

        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: based on beta and signal level
        step = 1.5 / (beta * max(y.max() * 2.0, S / n))
        step = min(step, 3.0 / beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart logic
        if stall > 2000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stall / 4000.0)
            if noise_scale > 1e-10:
                y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0
                stall = 0

        if it % 4000 == 0:
            t_mom = 1.0

    return best_a, best_val


def peak_shave_polish(a_init, S, max_time, rng, fsize=None):
    """Polish using exact subgradient on top peaks with cosine-decay step."""
    n = len(a_init)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()

    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best_val = _conv_self(a, fsize).max()
    best_a = a.copy()
    y = a.copy()
    t_mom = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k_top = max(2, int(4 + 24 * progress ** 0.5))

        c = _conv_self(y, fsize)
        if c.max() < best_val - 1e-12:
            best_val = c.max()
            best_a = y.copy()

        grad = weighted_peak_subgradient(y, c, k_top=k_top, fsize=fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Cosine decay step
        step = (0.01 + 0.04 * (1.0 + np.cos(np.pi * progress))) / (1.0 + 0.001 * it)
        step = min(step, S / (n * 5.0))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

        if it % 2000 == 0:
            t_mom = 1.0

    return best_a, best_val


def spectral_phase_refine(a_init, S, max_time, rng, fsize=None):
    """
    Refine by adjusting the phase of A = fft(a) to flatten ifft(A^2).
    Uses iterative: compute peaks of b, derive phase correction, apply.
    """
    n = len(a_init)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()

    a = normalize_to_sum(a_init, S).copy()
    start = time.monotonic()
    best_val = _conv_self(a, fsize).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = _conv_self(a, fsize)
        if c.max() < best_val - 1e-12:
            best_val = c.max()
            best_a = a.copy()

        # Compute gradient of max via top peaks
        k_top = min(16, 2 * n - 1)
        idx = np.argsort(c)[::-1][:k_top]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k_top

        # The gradient in time domain is 2 * (a convolved with w)
        # But we can also think of it as: reduce a_j where it contributes to peaks
        # A simpler approach: use the subgradient to do a small projected step
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Also try a "deflation" approach: reduce mass at positions that
        # are part of the peak-bearing products
        peak_k = c.argmax()
        # Elements a_j and a_{j + peak_k - (n-1)} (or similar) contribute
        # Just use gradient-based step for simplicity
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.5 / (it * 0.1 + 10.0)
        a_new = project_simplex(a - step * grad, S)

        # Accept if improvement
        c_new = _conv_self(a_new, fsize)
        if c_new.max() < c.max():
            a = a_new
        else:
            # Try smaller step
            a_new2 = project_simplex(a - step * 0.3 * grad, S)
            c_new2 = _conv_self(a_new2, fsize)
            if c_new2.max() < c.max():
                a = a_new2
            elif it % 500 == 0:
                # Perturb
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

    return best_a, best_val


def excess_anneal(a_init, S, max_time, rng, fsize=None):
    """Anneal a threshold T downward, minimizing excess above T."""
    n = len(a_init)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()

    a = normalize_to_sum(a_init, S).copy()
    start = time.monotonic()
    c0 = _conv_self(a, fsize)
    best_val = c0.max()
    best_a = a.copy()
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())

    y = a.copy()
    t_mom = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.8)
        T = max(T, T_end)

        c = _conv_self(y, fsize)
        if c.max() < best_val - 1e-12:
            best_val = c.max()
            best_a = y.copy()

        # Excess gradient
        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            # Use peak subgradient instead
            grad = peak_subgradient(y, c, k_top=8, fsize=fsize)
        else:
            w = 2.0 * excess
            wsum = w.sum()
            if wsum < 1e-30:
                grad = peak_subgradient(y, c, k_top=4, fsize=fsize)
            else:
                w = w / wsum
                W = fft(w, fsize)
                A = fft(y, fsize)
                grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (50.0 * (1.0 + 0.005 * it) * (1.0 + progress))
        step = min(step, S / (n * 10.0))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

        if it % 3000 == 0:
            t_mom = 1.0

    return best_a, best_val


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        gs = np.asarray(given_seq, dtype=np.float64)
        if len(gs) == n:
            inits.append(np.maximum(gs, 0.0))
        elif len(gs) > n:
            ratio = len(gs) // n
            small = gs[:n * ratio].reshape(n, ratio).mean(axis=1)
            inits.append(np.maximum(small, 0.0))
        else:
            t_old = np.linspace(0, 1, len(gs))
            inits.append(np.maximum(np.interp(tt, t_old, gs), 0.0))

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5) + 0.15 * np.cos(2 * np.pi * 7 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    for nb in [8, 16, 32]:
        bl = n // nb
        h = rng.uniform(0.3, 1.5, nb)
        bs = np.repeat(h, bl)
        if len(bs) < n:
            bs = np.concatenate([bs, np.full(n - len(bs), h[-1])])
        inits.append(np.maximum(bs[:n], 0.01))

    # Antisymmetric-like (to reduce even-lag peaks)
    anti = np.ones(n)
    half = n // 2
    anti[:half] = 1.0 + 0.3 * np.sin(np.pi * np.arange(half) / half)
    anti[half:] = 1.0 - 0.3 * np.sin(np.pi * np.arange(len(anti) - half) / max(1, half))
    inits.append(np.maximum(anti, 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    fsize = 1 << (2 * n - 1).bit_length()
    b = _conv_self(a, fsize)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: diverse FISTA exploration → LSE-annealed refinement →
    excess annealing → peak-shave polish → spectral refinement.
    Tries n=8192 primarily, n=16384 secondarily.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    n_primary = 8192
    n_secondary = 16384
    S_p = np.sqrt(2.0 * n_primary)
    S_s = np.sqrt(2.0 * n_secondary)
    fsize_p = 1 << (2 * n_primary - 1).bit_length()
    fsize_s = 1 << (2 * n_secondary - 1).bit_length()

    total_time = deadline - time.monotonic()

    # === PHASE 1: Diverse FISTA exploration (25% time) ===
    print("\n=== Phase 1: Diverse FISTA exploration (n=%d) ===" % n_primary)
    inits = generate_inits(n_primary, rng, given)
    n_explore = min(len(inits), 6)
    explore_time = min(total_time * 0.25, 250.0)
    time_per_init = explore_time / n_explore

    candidates = []
    for i in range(n_explore):
        if time.monotonic() > deadline - total_time * 0.45:
            break
        t_alloc = max(5.0, min(time_per_init, deadline - time.monotonic() - total_time * 0.4))
        if t_alloc < 5:
            continue

        a_opt, val = fista_fast(inits[i], S_p, t_alloc, rng,
                                 beta_init=5.0, beta_end=3000.0,
                                 fsize=fsize_p, total_iters_hint=int(t_alloc * 150))
        obj = 2.0 * n_primary * val / (a_opt.sum() ** 2)
        print("  Init %d: max_b=%.8f, obj=%.8f" % (i, val, obj))
        candidates.append((val, a_opt))

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()

    if not candidates:
        global_a = np.ones(n_primary) * (S_p / n_primary)
        global_best = 2.0

    candidates.sort(key=lambda x: x[0])
    best_a = candidates[0][1].copy()
    best_val = candidates[0][0]
    print("  Best from Phase 1: obj=%.8f" % (2.0 * n_primary * best_val / (best_a.sum() ** 2)))

    # === PHASE 2: Long FISTA with backtracking (30% time) ===
    print("\n=== Phase 2: FISTA with backtracking ===")
    refine_time = min(total_time * 0.30, deadline - time.monotonic() - total_time * 0.25)
    if refine_time > 30:
        a_ref, val_ref = fista_backtrack(
            best_a, S_p, refine_time, rng,
            beta_init=10.0, beta_end=15000.0,
            fsize=fsize_p, total_iters_hint=int(refine_time * 80))
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref
        obj = 2.0 * n_primary * best_val / (best_a.sum() ** 2)
        print("  After Phase 2: max_b=%.8f, obj=%.8f" % (best_val, obj))
        if obj < global_best:
            global_best = obj
            global_a = best_a.copy()

    # === PHASE 3: Excess annealing (15% time) ===
    print("\n=== Phase 3: Excess annealing ===")
    excess_time = min(total_time * 0.15, deadline - time.monotonic() - total_time * 0.15)
    if excess_time > 20:
        a_ex, val_ex = excess_anneal(best_a, S_p, excess_time, rng, fsize=fsize_p)
        if val_ex < best_val:
            best_val = val_ex
            best_a = a_ex
        obj = 2.0 * n_primary * best_val / (best_a.sum() ** 2)
        print("  After Phase 3: max_b=%.8f, obj=%.8f" % (best_val, obj))
        if obj < global_best:
            global_best = obj
            global_a = best_a.copy()

    # === PHASE 4: Peak shaving polish (15% time) ===
    print("\n=== Phase 4: Peak shaving polish ===")
    polish_time = min(total_time * 0.15, deadline - time.monotonic() - total_time * 0.08)
    if polish_time > 10:
        a_pol, val_pol = peak_shave_polish(best_a, S_p, polish_time, rng, fsize=fsize_p)
        if val_pol < best_val:
            best_val = val_pol
            best_a = a_pol
        obj = 2.0 * n_primary * best_val / (best_a.sum() ** 2)
        print("  After Phase 4: max_b=%.8f, obj=%.8f" % (best_val, obj))
        if obj < global_best:
            global_best = obj
            global_a = best_a.copy()

    # === PHASE 5: Try secondary length (10% time) ===
    remaining = deadline - time.monotonic() - 15.0
    if remaining > 100:
        print("\n=== Phase 5: Try n=%d (%.0fs) ===" % (n_secondary, remaining))

        t_old = np.linspace(0, 1, n_primary)
        t_new = np.linspace(0, 1, n_secondary)
        init_sec = np.interp(t_new, t_old, best_a)

        t_sec = remaining * 0.55
        a_sec, val_sec = fista_fast(init_sec, S_s, t_sec, rng,
                                     beta_init=5.0, beta_end=5000.0,
                                     fsize=fsize_s, total_iters_hint=int(t_sec * 100))
        obj_sec = 2.0 * n_secondary * val_sec / (a_sec.sum() ** 2)
        print("  n=%d: max_b=%.8f, obj=%.8f" % (n_secondary, val_sec, obj_sec))

        if obj_sec < global_best:
            global_best = obj_sec
            global_a = a_sec.copy()
            print("  New global best from n=%d!" % n_secondary)

        # Quick peak shave on secondary
        rem2 = deadline - time.monotonic() - 10.0
        if rem2 > 20 and obj_sec < global_best + 0.01:
            a_sec_pol, val_sec_pol = peak_shave_polish(a_sec, S_s, rem2 * 0.5, rng, fsize=fsize_s)
            obj_sec_pol = 2.0 * n_secondary * val_sec_pol / (a_sec_pol.sum() ** 2)
            if obj_sec_pol < global_best:
                global_best = obj_sec_pol
                global_a = a_sec_pol.copy()
                print("  After secondary polish: obj=%.8f" % global_best)

    # === PHASE 6: Final intensive polish on global best ===
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 20 and global_a is not None:
        print("\n=== Phase 6: Final polish (%.0fs) ===" % remaining)
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)
        fsize_fin = 1 << (2 * n_fin - 1).bit_length()

        # High-beta FISTA
        seg1 = remaining * 0.45
        a_fin1, val_fin1 = fista_backtrack(
            global_a, S_fin, seg1, rng,
            beta_init=100.0, beta_end=30000.0,
            fsize=fsize_fin, total_iters_hint=int(seg1 * 60))

        cur_max = _conv_self(global_a, fsize_fin).max()
        if val_fin1 < cur_max:
            global_a = a_fin1
        cur_max = _conv_self(global_a, fsize_fin).max()
        global_best = 2.0 * n_fin * cur_max / (global_a.sum() ** 2)
        print("  After final FISTA: obj=%.8f" % global_best)

        # Peak shave
        rem2 = deadline - time.monotonic() - 5.0
        if rem2 > 10:
            a_ps, val_ps = peak_shave_polish(global_a, S_fin, rem2, rng, fsize=fsize_fin)
            cur_max = _conv_self(global_a, fsize_fin).max()
            if val_ps < cur_max:
                global_a = a_ps
            cur_max = _conv_self(global_a, fsize_fin).max()
            global_best = 2.0 * n_fin * cur_max / (global_a.sum() ** 2)
            print("  After final peak shave: obj=%.8f" % global_best)

    # Verification
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification via evaluate_sequence: %.8f" % ver)
    if not np.isfinite(ver) or ver > 2.0:
        print("  Fallback triggered")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, obj=%.8f" % (len(final_list), global_best))
    return final_list
