"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses FISTA with backtracking, adaptive LSE smoothing, multi-start,
and active-set polishing with QP subproblem solving.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE approximation of max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_backtrack(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                    beta_max=5000.0, mode="lse", step_scale=1.0):
    """
    FISTA with backtracking line search and adaptive LSE.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if stuck_count > 2000:
                k = max(1, k // 8)
            grad = topk_gradient(y, c, k, power=1.5)
        elif mode == "hybrid":
            if stuck_count > 1000 and (it % 10 == 0 or rng.random() < 0.1):
                grad = exact_subgradient(y, c, tol_frac=0.005)
            else:
                grad = lse_gradient(y, c, beta)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        c_val = lse_value(y, c, beta)
        step_bt = step
        x_new = project_simplex(y - step_bt * grad, S)
        prev_val = lse_value(y, c, beta)
        
        # Simple backtracking: try to ensure descent in LSE value
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            val_new = lse_value(x_new, c_new, beta)
            if val_new < prev_val - 1e-10 * gnorm * step_bt:
                break
            step_bt *= 0.5
            x_new = project_simplex(y - step_bt * grad, S)
        
        step = step_bt  # Use the successful step for next iteration
        # But also adapt based on progress
        if stuck_count > 1000:
            step = min(step, 0.1 / beta)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic noise when stuck
        if stuck_count > 5000 and it % 200 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                a = y.copy()
        
        # Restart momentum periodically
        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def active_set_qp(a_init, max_time, rng, k_max=64):
    """
    Active-set QP polishing: minimize t subject to (a*a)[k] <= t for top-k lags,
    a >= 0, sum(a) = S.
    Uses projected gradient on the QP subproblem.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Determine active set size
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 4 + stuck // 200)
        else:
            k = min(k_max, 16 + stuck // 500)
        
        # Get gradient from active set
        if k == 1:
            grad = exact_subgradient(a, c, tol_frac=0.0)
        else:
            # Use a small tolerance to include near-active lags
            tol_frac = min(0.05, 0.001 * (1 + stuck / 1000))
            max_val = c.max()
            active = np.where(c >= max_val * (1 - tol_frac))[0]
            k_actual = min(k, len(active))
            idx = np.argsort(c)[::-1][:k_actual]
            w_full = np.zeros(len(c))
            # Power-weighted
            vals = c[idx]
            w = np.power(np.maximum(vals - max_val * (1 - tol_frac), 0), 0.5) + 1e-6
            w = w / w.sum()
            w_full[idx] = w
            fsize = 1 << (len(w_full) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = S / (gnorm * (1 + 0.01 * it))
        step = min(step, 2 * S / gnorm)
        
        # Try the step
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max + 1e-10:
            a = a_new
        else:
            # Reduce step
            step *= 0.5
            a = project_simplex(a - step * grad, S)
        
        # Noise when stuck
        if stuck > 10000 and it % 500 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def make_symmetric(a):
    """Make a sequence symmetric (palindromic)."""
    n = len(a)
    half = a[:n // 2]
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, [a[n // 2]], a[n // 2 - 1:][::-1]])

Given a sequence a of length n, and a block B of indices {i_1, ..., i_b}, we want to optimize the variables a_{i_1}, ..., a_{i_b} while keeping the rest fixed.

The self-convolution is:

For a given k, the terms involving the free variables are:

- If both i_p and i_q are in B and i_p + i_q = k, the term a_{i_p} * a_{i_q} involves two free variables (bilinear!)

So the formulation is NOT purely linear if both indices of a product fall in the block. For small blocks and large n, the probability of both indices being in the block is small (about b^2/n for random blocks), so we can either:
1. Include the bilinear terms (making it a QP instead of LP)
2. Ignore them (approximation)
3. Choose blocks that are "spread out" so that product indices don't overlap

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Combines FISTA with LSE smoothing for bulk optimization, then block coordinate
descent with LP subproblem solving for precise polishing. Multi-start and
length exploration for robustness.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE approximation of max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_opt(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if stuck_count > 2000:
                k = max(1, k // 8)
            grad = topk_gradient(y, c, k, power=1.5)
        elif mode == "hybrid":
            if stuck_count > 1000:
                tol = min(0.02, 0.001 * (1 + stuck_count / 2000))
                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                grad = lse_gradient(y, c, beta)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        step_base = step_scale / max(beta, 1.0)
        if stuck_count > 2000:
            step_use = step_base * 0.5
        else:
            step_use = step_base

        x_new = project_simplex(y - step_use * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise when stuck
        if stuck_count > 5000 and it % 300 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0

        if it % 15000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
        best_a = best_a.copy()
    return best_a, best_conv


def block_lp_polish(a_init, max_time, rng, block_size=64):
    """
    Block coordinate descent with LP subproblem.
    For each block of variables, solves:
      min t s.t. c_k <= t (linearized), a_i >= 0, sum(a_block) = S_free
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    n_blocks = n // block_size
    remaining_vars = n - n_blocks * block_size

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    no_improve_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-13:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive: if stuck, try smaller blocks or random blocks
        if no_improve_count > 200:
            bs = min(block_size, 32)
        else:
            bs = block_size

        # Pick block (cyclic with some randomization)
        if rng.random() < 0.3:
            block_idx = rng.integers(0, n // bs)
        else:
            block_idx = (it * bs) % (n // bs)

        block_start = block_idx * bs
        block_end = block_start + bs
        block_indices = np.arange(block_start, block_end)

        S_free = a[block_indices].sum()
        if S_free < 1e-10:
            continue

        # Build LP: min t s.t. c_k(a) <= t, a_i >= 0, sum(a_block) = S_free
        # c_k = sum_j a_{block_start + j} * a_{k - (block_start + j)} + (fixed part)
        # Linearize: c_k ≈ sum_j a_j * (fixed contribution at k from j) + bilinear (ignored)
        # For each free variable a_j (j in block), its contribution to c_k is a_j * a_{k-j}
        # where a_{k-j} is FIXED (unless k-j is also in the block - we ignore those)

        # Precompute: for each free var j and each lag k, the coefficient is a[k-j] (if k-j not in block)
        # c_k = sum_{j in block} a_j * a_fixed[k-j] + (terms with both in block, ignored) + c_k_fixed

        # Compute fixed part of convolution (excluding block contributions)
        a_fixed = a.copy()
        a_fixed[block_indices] = 0.0
        c_fixed = fft_convolve_self(a_fixed)

        # c_k_lin = c_fixed[k] + sum_j a_j * a[k-j] (where k-j is in [0,n-1] and not in block)
        # For the LP: c_k_lin <= t means sum_j a_j * a[k-j] <= t - c_fixed[k]

        # Build constraint matrix
        # Variables: [a_0, ..., a_{bs-1}, t] (bs+1 vars)
        # For each k in 0..2n-2: sum_j a_j * a[k-(block_start+j)] - t <= -c_fixed[k]
        # Subject to: a_j >= 0, sum_j a_j = S_free

        # Efficient: only need to consider k values where the constraint is potentially tight
        # Focus on lags near the current maximum
        max_c = c.max()
        # Consider all lags, but we'll trim to keep LP manageable
        # For n=8192, we have 16383 lags - that's fine for LP

        # Build A_ub matrix: (2n-1) x (bs+1)
        # Row k: [a[k-(block_start+0)], a[k-(block_start+1)], ..., 1] (with -1 for t coefficient)
        # Actually: sum_j a_j * coeff[j,k] - t <= -c_fixed[k]
        # So row is: [coeff[0,k], coeff[1,k], ..., -1] and rhs = -c_fixed[k]

        coeff = np.zeros((bs, 2 * n - 1))
        for j in range(bs):
            idx = block_start + j
            # a_j contributes to c_k for k in [idx, idx+n-1] via a_j * a[k-idx]
            # c_k gets term a_j * a[k-idx] where k-idx in [0, n-1]
            # So for k from idx to idx+n-1: coeff[j, k] = a[k-idx]
            # But we should exclude k-idx in block (bilinear terms)
            k_range = np.arange(idx, idx + n)
            valid = (k_range < 2 * n - 1)
            k_valid = k_range[valid]
            a_idx = k_valid - idx
            # Exclude a_idx in block
            in_block = (a_idx >= block_start) & (a_idx < block_end)
            a_idx_valid = a_idx[~in_block]
            k_valid_valid = k_valid[~in_block]
            coeff[j, k_valid_valid] = a[a_idx_valid]

        A_ub = np.zeros((2 * n - 1, bs + 1))
        A_ub[:, :bs] = coeff.T
        A_ub[:, bs] = -1.0
        b_ub = -c_fixed

        # Equality: sum(a_block) = S_free
        A_eq = np.zeros((1, bs + 1))
        A_eq[0, :bs] = 1.0
        b_eq = np.array([S_free])

        # Bounds: a_j >= 0 (no upper bound needed, but cap at S_free), t >= 0
        bounds = [(0, None)] * bs + [(0, None)]

        c_obj = np.zeros(bs + 1)
        c_obj[bs] = 1.0  # Minimize t

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                a_new_block = res.x[:bs]
                a_new_block = np.maximum(a_new_block, 0)
                # Renormalize block to maintain exact sum
                s_block = a_new_block.sum()
                if s_block > 1e-10:
                    a_new_block = a_new_block * (S_free / s_block)
                a[block_indices] = a_new_block
        except Exception:
            pass

        # Check if we should increase block size adaptively
        if no_improve_count > 500 and block_size < 128:
            block_size = min(128, block_size * 2)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def block_qp_polish(a_init, max_time, rng, block_size=32, n_inner_steps=200):
    """
    Block coordinate descent with QP subproblem solved by projected gradient.
    More accurate than LP (keeps bilinear terms) but slower per block.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        block_idx = (it * block_size) % (n - block_size + 1)
        block_start = block_idx
        block_end = block_start + block_size
        block_indices = np.arange(block_start, block_end)

        S_free = a[block_indices].sum()
        if S_free < 1e-10:
            continue

        # Get current block values
        a_block = a[block_indices].copy()

        # Solve subproblem: min max(c_k) s.t. a_block >= 0, sum(a_block) = S_free
        # Use projected gradient with LSE on the small subproblem
        a_b = a_block.copy()
        S_b = S_free

        for inner in range(n_inner_steps):
            # Compute convolution with current a_b
            a_trial = a.copy()
            a_trial[block_indices] = a_b
            c_trial = fft_convolve_self(a_trial)

            # Gradient of LSE w.r.t. a_b only
            beta_b = 50.0 + 10.0 * inner
            cm = c_trial.max()
            shifted = beta_b * (c_trial - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            tot = e.sum()
            if tot < 1e-300:
                break
            w = e / tot

            # Gradient w.r.t. a_b: for each j in block, d/d(a_j) sum_k w_k * c_k
            # = sum_k w_k * (sum_i a_i * a_{k-i} where k-i = block_start+j or i = block_start+j)
            # = 2 * sum_k w_k * a[k - (block_start+j)] (approximately, accounting for symmetry)
            grad_b = np.zeros(block_size)
            for j in range(block_size):
                idx = block_start + j
                # Contribution from a_idx * a_{k-idx} for varying k
                k_vals = np.arange(idx, idx + n)
                valid = (k_vals < len(c_trial))
                k_v = k_vals[valid]
                a_idx = k_v - idx
                grad_b[j] = 2.0 * np.sum(w[k_v[valid]] * a_trial[a_idx])
                # Subtract self-term (a_idx * a_idx counted once, not twice)
                grad_b[j] -= a_trial[idx] * w[idx] if idx < len(c_trial) else 0

            # Projected gradient step
            gnorm = np.linalg.norm(grad_b)
            if gnorm < 1e-12:
                break
            step_b = S_b / (gnorm * (1 + 0.01 * inner))
            a_b = project_simplex(a_b - step_b * grad_b, S_b)

        a[block_indices] = a_b

        c_new = fft_convolve_self(a)
        cur_max = c_new.max()
        if cur_max < best_conv - 1e-13:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

        a1b, _ = fista_opt(global_best_a, remaining1, rng,
                            beta_init=50.0, beta_rate=0.003,
                            beta_max=10000.0, mode="hybrid", step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.6f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Block LP polish (30% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.30
        print("\n=== Phase 2: Block LP polish (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        # Use moderate block size for speed
        a2, _ = block_lp_polish(global_best_a, p2_budget, rng, block_size=64)
        obj2 = compute_obj(a2)
        print("  Block LP: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: FISTA topk + active polish (20% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: TopK + Active (budget %.0fs) ===" % p3_budget)

        a3a, _ = fista_opt(global_best_a, p3_budget * 0.5, rng,
                            beta_init=100.0, beta_rate=0.002,
                            beta_max=20000.0, mode="topk", step_scale=0.3)
        obj3a = compute_obj(a3a)
        print("  TopK: obj=%.6f (t=%.0fs)" % (obj3a, time.monotonic() - t3))
        if obj3a < global_best_obj:
            global_best_obj = obj3a
            global_best_a = a3a.copy()

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 5:
            a3b, _ = fista_opt(global_best_a, remaining3, rng,
                                beta_init=500.0, beta_rate=0.001,
                                beta_max=50000.0, mode="hybrid", step_scale=0.2)
            obj3b = compute_obj(a3b)
            print("  Active: obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Block QP polish (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.10
        print("\n=== Phase 4: Block QP (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, _ = block_qp_polish(global_best_a, p4_budget, rng,
                                 block_size=32, n_inner_steps=100)
        obj4 = compute_obj(a4)
        print("  Block QP: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final intensive block LP (5% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.85
        print("\n=== Phase 5: Final LP polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, _ = block_lp_polish(global_best_a, p5_budget, rng, block_size=48)
        obj5 = compute_obj(a5)
        print("  Final LP: obj=%.6f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
