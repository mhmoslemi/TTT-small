def project_simplex(v, S):
    # Euclidean projection onto {x>=0, sum(x)=S}
    v = np.asarray(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, len(v)+1)
    theta_candidates = (cumsum - S) / idx
    if np.any(theta_candidates <= w):
        rho = np.max(np.where(theta_candidates <= w)[0])
        theta = (cumsum[rho] - S) / (rho+1)
        return np.maximum(v - theta, 0.0)
    else:
        return 0.0 * v
