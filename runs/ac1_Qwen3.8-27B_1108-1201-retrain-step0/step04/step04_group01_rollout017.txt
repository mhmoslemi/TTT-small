def fft_convolve_self(a):

def project_simplex(v, S):

    n = len(v)

        # Step size with adaptive scaling
        base_step = 1.0 / (beta * 0.12) * step_adapt

        if use_ls:
            step = base_step
            f_cur = lse_max_val(c, beta)
            x_new = y
            for ls_it in range(15):
                x_try = project_simplex(y - step * grad, S)
                c_try = fft_convolve_self(x_try)
                f_try = lse_max_val(c_try, beta)
                if f_try <= f_cur - 1e-5 * step * gnorm + 1e-15:
                    x_new = x_try
                    break
                step *= 0.6
            else:
                x_new = project_simplex(y - step * grad, S)
        else:
            x_new = project_simplex(y - base_step * grad, S)

        # Adaptive step size based on observed decrease
        if use_ls:
            c_new_check = fft_convolve_self(x_new)
            if c_new_check.max() < c.max() * 0.99:
                step_adapt = min(step_adapt * 1.1, 3.0)
            elif c_new_check.max() > c.max():
                step_adapt = max(step_adapt * 0.7, 0.1)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.0005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            step_adapt = 1.0
            last_improve = it

        # Periodic restart with decaying noise
        if it % 8000 == 0:
            decay = max(0.005, 1.0 - it / 200000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best_cmax:
        best_cmax = c_final.max()
        best_a = a.copy()

    return best_a, best_cmax


def run_peak_shave_refined(a_init, S, max_time, rng):
    """Refined peak shaving with adaptive k and curvature-aware steps."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_cmax = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        # Adaptive k: start small, grow over time
        k = int(4 + 60 * progress ** 0.5)
        k = min(k, n // 4)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_cmax - 1e-14:
            best_cmax = cur_max
            best_a = y.copy()
            last_improve = it

        # Use excess-weighted subgradient
        grad = subgradient_topk(y, c, k, weight_mode='excess')
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Curvature-aware step: use local quadratic model
        # f(x - alpha*g) ≈ f(x) - alpha*||g||^2 + alpha^2 * L/2 * ||g||^2
        # Optimal alpha ≈ ||g|| / L, where L is local Lipschitz constant
        # Estimate L from the Hessian of the max: L ≈ 2 * max(|a|)
        L_est = 2.0 * max(np.max(np.abs(y)), 1e-6)
        step = gnorm / (L_est * (1.0 + 0.5 * progress))
        step = min(step, S / (n * 0.1))

        x_new = project_simplex(y - step * grad, S)

        # Verify improvement, backtrack if needed
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur_max * 1.001:
            # Backtrack
            for bt in range(5):
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= cur_max:
                    break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart if stuck
        if it - last_improve > 2000:
            noise_scale = 0.0003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best_cmax:
        best_cmax = c_final.max()
        best_a = a.copy()

    return best_a, best_cmax


def run_conjugate_peak_shave(a_init, S, max_time, rng):
    """Peak shaving with conjugate direction (Polak-Ribiere style)."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * S / s

    y = a.copy()
    start = time.monotonic()
    best_cmax = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    c_prev = fft_convolve_self(y)
    grad_prev = subgradient_topk(y, c_prev, 16, weight_mode='excess')
    direction = -grad_prev.copy()

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_cmax - 1e-14:
            best_cmax = cur_max
            best_a = y.copy()
            last_improve = it

        k = int(8 + 40 * progress ** 0.5)
        k = min(k, n // 4)
        grad = subgradient_topk(y, c, k, weight_mode='excess')
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polak-Ribiere beta
        dot = np.dot(grad - grad_prev, direction)
        denom = np.dot(grad_prev, grad_prev)
        if denom < 1e-30:
            beta_pr = 0.0
        else:
            beta_pr = max(0.0, dot / denom)
        beta_pr = min(beta_pr, 1.0)

        if it % 50 == 0 or beta_pr < 0.01:
            direction = -grad
        else:
            direction = -grad + beta_pr * direction

        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            direction = -grad
            dnorm = gnorm

        # Line search
        step = 1.0 / (2.0 * np.max(np.abs(y)) * (1.0 + progress))
        f_cur = cur_max
        x_new = y
        for ls in range(12):
            x_try = project_simplex(y - step * direction, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() < f_cur - 1e-8 * step * dnorm:
                x_new = x_try
                break
            step *= 0.5
        else:
            x_new = project_simplex(y - step * direction, S)

        a = x_new
        y = a.copy()
        grad_prev = grad
        c_prev = c

        if it - last_improve > 3000:
            noise_scale = 0.0003 * S / np.sqrt(n) * max(0.1, 1.0 - progress)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            direction = -grad
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best_cmax:
        best_cmax = c_final.max()
        best_a = a.copy()

    return best_a, best_cmax


def generate_perturbations(a_base, S, rng, n_perturbs=5):
    """Generate diverse perturbations of the base sequence."""
    n = len(a_base)
    S_target = S
    perturbations = []

    for i in range(n_perturbs):
        a = a_base.copy()
        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = a + noise
        elif strategy == 1:
            # Targeted perturbation at the peaks
            c = fft_convolve_self(a_base)
            idx_top = np.argsort(c)[::-1][:50]
            # Perturb the sequence at positions that contribute to top peaks
            for idx in idx_top[:20]:
                j = idx % n
                a[j] *= (1.0 + rng.normal(0, 0.01))
        elif strategy == 2:
            # Frequency domain perturbation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            A *= (1.0 + rng.normal(0, 0.005, fsize))
            a = np.real(ifft(A))[:n]
        else:
            # Block perturbation
            block_size = n // 32
            for b in range(32):
                if rng.random() < 0.3:
                    offset = int(rng.normal(0, 0.002 * S / np.sqrt(n)))
                    a[b * block_size:(b + 1) * block_size] += offset

        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s > 1e-10:
            a = a * S_target / s
        else:
            a = np.full(n, S_target / n)
        perturbations.append(a)

    return perturbations


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase optimization with adaptive methods starting from the
    provided near-optimal sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get the provided sequence
    a_base = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            a_base = np.clip(given, 0, 1000)
            print("Loaded given sequence of length %d" % len(a_base))
            obj_given = objective_val(a_base)
            print("Given sequence objective: %.6f" % obj_given)
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    if a_base is None:
        # Fallback: use a good initial guess
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_base = 0.5 + 0.3 * np.cos(2 * np.pi * tt) + 0.2 * np.sin(4 * np.pi * tt)
        a_base = np.maximum(a_base, 0.01)

    n = len(a_base)
    S = np.sqrt(2.0 * n)

    # Normalize
    a_base = np.clip(a_base, 0, 1000)
    s = a_base.sum()
    if s > 1e-10:
        a_base = a_base * S / s
    else:
        a_base = np.full(n, S / n)

    global_best_a = a_base.copy()
    global_best_cmax = fft_convolve_self(a_base).max()
    global_best_obj = 2.0 * n * global_best_cmax / S**2
    print("Starting objective: %.6f (max_c=%.8f)" % (global_best_obj, global_best_cmax))

    def update_global(a_candidate, label=""):
        nonlocal global_best_a, global_best_cmax, global_best_obj
        a_c = np.clip(a_candidate, 0, 1000)
        s_c = a_c.sum()
        if s_c < 0.01:
            return
        a_c = a_c * S / s_c
        c = fft_convolve_self(a_c)
        cmax = c.max()
        obj = 2.0 * n * cmax / S**2
        if cmax < global_best_cmax:
            global_best_cmax = cmax
            global_best_a = a_c.copy()
            global_best_obj = obj
            print("  NEW BEST [%s]: obj=%.8f, max_c=%.10f" % (label, obj, cmax))

    # ============ Phase 1: Quick FISTA warmup (100s) ============
    print("\n=== Phase 1: FISTA warmup (100s) ===")
    t1 = time.monotonic()
    time_p1 = min(100.0, deadline - t1 - 30.0)
    if time_p1 > 10:
        a_p1, cmax_p1 = run_phase_fista(a_base, S, time_p1, rng,
                                         beta_init=20.0, beta_rate=0.005,
                                         beta_max=2000.0, use_ls=True, k_subgrad=0)
        update_global(a_p1, "P1-FISTA")
    elapsed = time.monotonic() - t1
    print("  Phase 1 done in %.0fs, best=%.6f" % (elapsed, global_best_obj))

    # ============ Phase 2: High-beta FISTA with line search (400s) ============
    print("\n=== Phase 2: High-beta FISTA (400s) ===")
    t2 = time.monotonic()
    time_p2 = min(400.0, deadline - t2 - 30.0)
    if time_p2 > 20:
        a_p2, cmax_p2 = run_phase_fista(global_best_a, S, time_p2, rng,
                                         beta_init=200.0, beta_rate=0.002,
                                         beta_max=10000.0, use_ls=True, k_subgrad=0)
        update_global(a_p2, "P2-HiBeta")
    elapsed = time.monotonic() - t2
    print("  Phase 2 done in %.0fs, best=%.6f" % (elapsed, global_best_obj))

    # ============ Phase 3: Peak shaving with adaptive k (400s) ============
    print("\n=== Phase 3: Peak shaving (400s) ===")
    t3 = time.monotonic()
    time_p3 = min(400.0, deadline - t3 - 30.0)
    if time_p3 > 20:
        a_p3, cmax_p3 = run_peak_shave_refined(global_best_a, S, time_p3, rng)
        update_global(a_p3, "P3-PeakShave")
    elapsed = time.monotonic() - t3
    print("  Phase 3 done in %.0fs, best=%.6f" % (elapsed, global_best_obj))

    # ============ Phase 4: Conjugate peak shaving (300s) ============
    print("\n=== Phase 4: Conjugate peak shaving (300s) ===")
    t4 = time.monotonic()
    time_p4 = min(300.0, deadline - t4 - 30.0)
    if time_p4 > 20:
        a_p4, cmax_p4 = run_conjugate_peak_shave(global_best_a, S, time_p4, rng)
        update_global(a_p4, "P4-Conj")
    elapsed = time.monotonic() - t4
    print("  Phase 4 done in %.0fs, best=%.6f" % (elapsed, global_best_obj))

    # ============ Phase 5: Mixed LSE + subgradient (300s) ============
    print("\n=== Phase 5: Mixed gradient (300s) ===")
    t5 = time.monotonic()
    time_p5 = min(300.0, deadline - t5 - 30.0)
    if time_p5 > 20:
        a_p5, cmax_p5 = run_phase_fista(global_best_a, S, time_p5, rng,
                                         beta_init=500.0, beta_rate=0.001,
                                         beta_max=20000.0, use_ls=True,
                                         k_subgrad=32, blend_rate=0.003)
        update_global(a_p5, "P5-Mixed")
    elapsed = time.monotonic() - t5
    print("  Phase 5 done in %.0fs, best=%.6f" % (elapsed, global_best_obj))

    # ============ Phase 6: Multi-restart perturbation (remaining time) ============
    print("\n=== Phase 6: Multi-restart perturbation ===")
    t6 = time.monotonic()
    remaining = deadline - t6
    print("  Remaining: %.0fs" % remaining)

    if remaining > 60:
        # Generate perturbations and try each briefly
        n_perturbs = min(10, int(remaining / 40))
        perturbations = generate_perturbations(global_best_a, S, rng, n_perturbs)

        for i, a_pert in enumerate(perturbations):
            t_start_pert = time.monotonic()
            time_left = deadline - t_start_pert - 20
            if time_left < 15:
                break
            time_alloc = min(35.0, time_left / (n_perturbs - i))

            # Quick FISTA on perturbation
            a_r, cmax_r = run_phase_fista(a_pert, S, time_alloc, rng,
                                           beta_init=500.0, beta_rate=0.003,
                                           beta_max=20000.0, use_ls=True, k_subgrad=16)
            update_global(a_r, "P6-pert%d" % i)

        # Additional conjugate shaving on best
        if time.monotonic() < deadline - 30:
            time_left = deadline - time.monotonic() - 10
            if time_left > 20:
                a_c, cmax_c = run_conjugate_peak_shave(global_best_a, S, time_left, rng)
                update_global(a_c, "P6-final-conj")

    # ============ Phase 7: Final ultra-fine polishing ============
    if time.monotonic() < deadline - 10:
        print("\n=== Phase 7: Final ultra-fine polishing ===")
        time_left = deadline - time.monotonic() - 5
        if time_left > 10:
            # Very high beta, exact subgradient only
            a_fin, cmax_fin = run_phase_fista(global_best_a, S, time_left, rng,
                                              beta_init=5000.0, beta_rate=0.0005,
                                              beta_max=50000.0, use_ls=True,
                                              k_subgrad=64, blend_rate=0.01)
            update_global(a_fin, "P7-Ultra")

    # ============ Final verification ============
    print("\n=== Final Result ===")
    final_a = np.clip(global_best_a, 0, 1000)
    final_a = np.maximum(final_a, 0.0)
    s_final = final_a.sum()
    if s_final < 0.01:
        print("WARNING: sum too small, using fallback!")
        final_a = np.ones(256) * (np.sqrt(512.0) / 256)
    else:
        final_a = final_a * (np.sqrt(2.0 * len(final_a)) / s_final)

    final_obj = objective_val(final_a)
    print("Final objective (internal): %.8f" % final_obj)
    print("Final length: %d" % len(final_a))

    # Verify with evaluate_sequence
    final_list = [float(x) for x in final_a]
    try:
        ver = evaluate_sequence(final_list)
        print("Verification via evaluate_sequence: %.8f" % ver)
        if not np.isfinite(ver) or ver < 0:
            print("  evaluate_sequence rejected! Using fallback.")
            final_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in final_a]
            ver2 = evaluate_sequence(final_list)
            print("  Fallback verification: %.8f" % ver2)
    except Exception as e:
        print("  evaluate_sequence error: %s" % str(e))

    print("Returning sequence of length %d" % len(final_list))
    return [float(x) for x in np.clip(final_a, 0, 1000.0)]
