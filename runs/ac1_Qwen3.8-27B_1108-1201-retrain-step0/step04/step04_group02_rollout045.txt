"""
Minimize 2*n*max(self_conv)/sum^2 over non-negative sequences.
Novel approach: Adaptive LSE optimization with cubic line search, active-set
peak equalization, multi-length search, and simulated annealing restarts.
Key differences from previous: geometric temperature schedule, cubic line search,
active set tracking with equalization, and explicit multi-length comparison.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_from_conv(c, n, S):
    """Compute objective from convolution and sum."""
    return 2.0 * n * c.max() / (S * S)


def lse_gradient(a, c, beta, n):
    """LSE-smoothed gradient of max(conv)."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: 2 * sum_k w_k * a[k-j] for valid k
    # This is a cross-correlation: grad[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (w reverse-correlated with a)
    # Using FFT: ifft(fft(w) * conj(fft(a))) gives sum_k w[k] * a[k-j]... 
    # Actually we need: for each j, sum over k where 0<=k-j<n: w[k]*a[k-j]
    # Let l = k-j, then sum_l w[j+l]*a[l] = (w_rev * a)_j where w_rev is reversed
    # Equivalently: ifft(conj(fft(w)) * fft(a))... 
    # Let's use: corr[j] = sum_k w[k] * a[k-j] = ifft(fft(w) * conj(fft(a)))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return np.maximum(grad, 0.0)


def topk_gradient(a, c, k, n):
    """Subgradient from top-k peaks."""
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def cubic_line_search(a, d, S, n, c_cur, alpha_init, max_alpha, eval_func):
    """
    Cubic line search: try several alphas and pick the best.
    Returns best alpha and best point.
    """
    alphas = np.array([
        alpha_init,
        alpha_init * 0.5,
        alpha_init * 2.0,
        alpha_init * 0.25,
        alpha_init * 1.5,
        alpha_init * 0.1,
        max_alpha * 0.5,
        max_alpha,
    ])
    alphas = np.clip(alphas, 1e-10, max_alpha)
    # Also add some in between
    alphas = np.unique(np.sort(alphas))
    
    best_obj = obj_from_conv(c_cur, n, S)
    best_a = a.copy()
    best_alpha = 0.0
    
    for alpha in alphas:
        a_try = proj_simplex(a + alpha * d, S)
        if not np.isfinite(a_try).all():
            continue
        s = a_try.sum()
        if s < 0.01:
            continue
        c_try = fft_self_conv(a_try)
        o = obj_from_conv(c_try, n, a_try.sum())
        if o < best_obj - 1e-12:
            best_obj = o
            best_a = a_try
            best_alpha = alpha
    
    return best_a, best_obj, best_alpha


def equalize_peaks(a, n, S, top_m=5, n_inner=10):
    """
    Given the current a, find the top_m peaks of self-convolution
    and try to equalize them by adjusting a in the direction
    that reduces the spread.
    """
    c = fft_self_conv(a)
    peaks = np.argsort(c)[::-1][:top_m]
    
    # For each peak, compute its gradient
    # We want to move in a direction that reduces the maximum
    # but also equalizes the top peaks
    # Direction: negative weighted sum of peak gradients, weighted by (c_k - mean(c_peaks))
    
    peak_vals = c[peaks]
    mean_peak = peak_vals.mean()
    spread = peak_vals - mean_peak
    
    # If peaks are already equal, just use the top-1 gradient
    if np.std(peak_vals) < 1e-10 * peak_vals.mean():
        return topk_gradient(a, c, 1, n)
    
    # Weight peaks by how much they exceed the mean
    weights = np.maximum(spread, 0)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        return topk_gradient(a, c, 1, n)
    weights = weights / w_sum
    
    grad = np.zeros(n)
    for wk, pk in zip(weights, peaks):
        lo = max(0, pk - n + 1)
        hi = min(pk, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += wk * 2.0 * a[pk - i_vals]
    
    return grad


def run_lse_optimizer(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=500.0):
    """
    LSE gradient descent with geometric beta schedule and cubic line search.
    """
    a = np.clip(a_init.astype(np.float64), 0, 1000).copy()
    S_actual = a.sum()
    if S_actual < 0.01:
        a = np.ones(n) * (S / n)
        S_actual = S
    a = a * (S / S_actual)
    
    start = time.monotonic()
    best_obj = obj_from_conv(fft_self_conv(a), n, S)
    best_a = a.copy()
    
    alpha = 1.0
    it = 0
    total_iters = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        total_iters += 1
        
        # Geometric beta schedule
        frac = min(1.0, it / max(1, int(max_time / 0.5)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_self_conv(a)
        cur_obj = obj_from_conv(c, n, S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Choose gradient type
        if it % 50 == 0:
            # Periodically use equalization
            grad = equalize_peaks(a, n, S, top_m=5)
        else:
            grad = lse_gradient(a, c, beta, n)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        max_alpha = 2.0 * S / (n * gnorm + 1e-15)
        max_alpha = min(max_alpha, 100.0)
        
        # Line search
        a_new, new_obj, best_alpha = cubic_line_search(
            a, -grad, S, n, c, alpha, max_alpha, None
        )
        
        if new_obj < cur_obj:
            a = a_new
            alpha = best_alpha if best_alpha > 0 else min(alpha * 1.5, max_alpha)
        else:
            # Reject, reduce step
            alpha *= 0.5
            if alpha < 1e-8:
                # Perturb and restart
                noise = 0.001 * S / np.sqrt(n)
                a = proj_simplex(a + rng.normal(0, noise, n), S)
                alpha = 0.1
        
        if it % 1000 == 0:
            print("    LSE it=%d, obj=%.8f, beta=%.1f, alpha=%.6f" % (it, best_obj, beta, alpha))
    
    return best_a, best_obj, total_iters

        c_try = fft_self_conv(a_try)

                a = a_try

                a = a_try

        if it % 2000 == 0:

def run_active_set_eq(a_init, n, S, max_time, rng):
    """
    Active set peak equalization: iteratively identify top peaks
    and solve a small QP-like subproblem to equalize them.
    Uses iterative water-filling inspired approach.
    """
    a = np.clip(a_init.astype(np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    
    start = time.monotonic()
    best_obj = obj_from_conv(fft_self_conv(a), n, S)
    best_a = a.copy()
    
    alpha = 0.1
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_self_conv(a)
        cur_obj = obj_from_conv(c, n, S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Find active set: peaks within epsilon of max
        c_max = c.max()
        eps = 0.01 * c_max
        active = np.where(c >= c_max - eps)[0]
        n_active = len(active)
        
        if n_active == 0:
            break
        
        # Compute equalization gradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1)
            # Weight by how far below max this peak is (push up low peaks, push down high)
            weight = (c[k] - (c_max - eps)) / eps  # 0 to 1
            grad[i_vals] += 2.0 * a[k - i_vals] * weight
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        max_alpha = min(2.0 * S / (n * gnorm + 1e-15), 50.0)
        tried = np.array([alpha, alpha*0.5, alpha*2, alpha*0.25, max_alpha*0.5])
        tried = np.clip(tried, 1e-8, max_alpha)
        
        best_step_alpha = 0
        best_new_obj = cur_obj
        best_new_a = a.copy()
        
        for sa in tried:
            a_try = proj_simplex(a - sa * grad, S)
            c_try = fft_self_conv(a_try)
            o = obj_from_conv(c_try, n, S)
            if o < best_new_obj:
                best_new_obj = o
                best_new_a = a_try
                best_step_alpha = sa
        
        if best_step_alpha > 0:
            a = best_new_a
            alpha = min(best_step_alpha * 1.3, max_alpha)
        else:
            alpha *= 0.6
            if alpha < 1e-8:
                noise = 0.002 * S / np.sqrt(n)
                a = proj_simplex(a + rng.normal(0, noise, n), S)
                alpha = 0.1
        
        if it % 500 == 0:
            print("    ActiveEq it=%d, obj=%.8f, n_active=%d" % (it, best_obj, n_active))
    
    return best_a, best_obj

    best_obj = obj_from_conv(c, n, S)
    best_a = a.copy()
    
    # Precompute which convolution terms each coordinate affects
    # c_k depends on a_j when 0 <= k-j < n, i.e., j <= k <= j+n-1
    
    it = 0
    n_outer = 0
    while time.monotonic() - start < max_time:
        n_outer += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - start > max_time:
                break
            
            # Try several values for a[j]
            cur_val = a[j]
            candidates = [
                cur_val * 0.7,
                cur_val * 0.85,
                cur_val * 1.15,
                cur_val * 1.3,
                cur_val + S / n,
                cur_val - S / n,
                max(0, cur_val - 0.1 * S / n),
                cur_val + 0.3 * S / n,
            ]
            
            best_cand = cur_val
            best_c = c.max()
            
            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                if cv > 1000:
                    cv = 1000.0
                if abs(cv - cur_val) < 1e-10:
                    continue
                
                a[j] = cv
                # Update affected convolution terms
                # c_k for k in [j, j+n-1] (clamped to [0, 2n-2])
                # c_k = sum_i a[i]*a[k-i]
                # When a[j] changes by delta = cv - cur_val:
                # c_k changes by delta * (a[k-j] + a[j-k] if j != k-j... )
                # Actually: c_k = sum_i a_i * a_{k-i}
                # ∂c_k/∂a_j = a_{k-j} (if valid)
                # So delta_c_k = delta * a_{k-j} for valid k-j
                
                delta = cv - cur_val
                # Affected k: j <= k <= j+n-1, clamped to [0, 2n-2]
                k_lo = j
                k_hi = min(j + n - 1, 2 * n - 2)
                
                if k_lo <= k_hi:
                    # k-j ranges from 0 to k_hi-j
                    idx = np.arange(0, k_hi - k_lo + 1)
                    # c[k_lo + i] += delta * a[i]... wait
                    # c_k = sum_i a_i * a_{k-i}
                    # ∂c_k/∂a_j = a_{k-j}
                    # So delta_c_k = delta * a_{k-j}
                    # k ranges from k_lo to k_hi, k-j ranges from 0 to k_hi-j = min(n-1, 2n-2-j)
                    
                    # Actually c is the FULL convolution of length 2n-1
                    # c[k] = sum_{i=0}^{n-1} a[i] * a[k-i] where a[x]=0 outside [0,n-1]
                    # ∂c[k]/∂a[j] = a[k-j] if 0 <= k-j <= n-1, else 0
                    # So for fixed j, c[k] is affected for k in [j, j+n-1]
                    # delta_c[k] = delta * a[k-j]
                    
                    k_idx = np.arange(k_lo, k_hi + 1)
                    idx_into_a = np.arange(0, k_hi - k_lo + 1)  # k-j values
                    c[k_idx] += delta * a[idx_into_a]
                
                new_max = c.max()
                if new_max < best_c - 1e-12:
                    best_c = new_max
                    best_cand = cv
                    best_c_snapshot = c.copy()
                
                # Revert
                if k_lo <= k_hi:
                    c[k_idx] -= delta * a[idx_into_a]
            
            if best_cand != cur_val:
                a[j] = best_cand
                # Update c
                delta = best_cand - cur_val
                k_lo = j
                k_hi = min(j + n - 1, 2 * n - 2)
                if k_lo <= k_hi:
                    k_idx = np.arange(k_lo, k_hi + 1)
                    idx_into_a = np.arange(0, k_hi - k_lo + 1)
                    c[k_idx] += delta * a[idx_into_a]
                improved = True
        
        cur_obj = obj_from_conv(c, n, S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            print("    CoordDesc outer=%d, obj=%.8f" % (n_outer, best_obj))
        
        if not improved and n_outer > 3:
            # No improvement in a full sweep, try perturbation
            noise = 0.001 * S / np.sqrt(n)
            a = proj_simplex(a + rng.normal(0, noise, n), S)
            c = fft_self_conv(a)
    
    return best_a, best_obj


def try_length(a_src, target_len, n_src, rng):
    """Resample or pad a source sequence to target length."""
    if len(a_src) == target_len:
        return a_src.copy()
    # Use linear interpolation
    x_orig = np.linspace(0, 1, len(a_src), endpoint=False)
    x_new = np.linspace(0, 1, target_len, endpoint=False)
    new_a = np.interp(x_new, x_orig, a_src)
    return np.maximum(new_a, 0.0)


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase, multi-length optimization with LSE gradients, coordinate
    descent, active set equalization, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 40.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    print("Starting. Given length: %s, budget: %.0fs" % (
        len(given) if given is not None else "None", budget_s))
    
    t0 = time.monotonic()
    
    # Define candidate lengths
    candidate_lengths = [4096, 8192, 16384]
    
    # Time allocation: give more time to 8192 (our best so far)
    # and also try 16384
    time_frac = [0.2, 0.45, 0.35]
    
    for length_idx, n in enumerate(candidate_lengths):
        if time.monotonic() > deadline - 60:
            break
        
        n_time = (deadline - time.monotonic()) * time_frac[length_idx] / sum(time_frac[length_idx:])
        n_time = max(30.0, min(n_time, deadline - time.monotonic() - 30))
        S = np.sqrt(2.0 * n)
        
        print("\n=== Length n=%d, time alloc=%.0fs ===" % (n, n_time))
        
        # Generate initializations for this length
        inits = []
        
        if given is not None:
            # Resample given to this length
            given_resampled = try_length(given, n, len(given), rng)
            inits.append(("given", given_resampled))
            # Perturbed versions
            for p in range(2):
                noise = 0.01 * S / np.sqrt(n)
                pert = proj_simplex(given_resampled + rng.normal(0, noise, n), S)
                inits.append(("given_pert%d" % p, pert))
        
        # Standard initializations
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(("cos", np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01)))
        inits.append(("sine_sq", np.sin(np.pi * tt) ** 2 + 0.01))
        inits.append(("rand", rng.random(n) + 0.1))
        
        # Multi-phase optimization for this length
        # Phase A: LSE gradient descent (50% of length time)
        # Phase B: Active set equalization (25%)
        # Phase C: Coordinate descent (15%)
        # Phase D: SA (10%)
        
        phase_times = {
            'lse': n_time * 0.50,
            'active': n_time * 0.25,
            'coord': n_time * 0.15,
            'sa': n_time * 0.10,
        }
        
        best_for_len = None
        best_obj_for_len = np.inf
        
        # Phase A: LSE
        for init_name, init_arr in inits[:3]:  # Limit to top 3 inits
            if time.monotonic() > deadline - 60:
                break
            t_alloc = phase_times['lse'] / max(1, len(inits[:3]))
            t_alloc = min(t_alloc, deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break
            
            a_opt, obj_opt, n_it = run_lse_optimizer(
                init_arr, n, S, t_alloc, rng,
                beta_start=5.0, beta_end=300.0
            )
            a_opt = np.maximum(a_opt, 0)
            obj_real = obj_from_conv(fft_self_conv(a_opt), n, a_opt.sum())
            
            if obj_real < best_obj_for_len:
                best_obj_for_len = obj_real
                best_for_len = a_opt.copy()
                if obj_real < global_best:
                    global_best = obj_real
                    global_a = a_opt.copy()
                    print("  [LSE/%s] n=%d: obj=%.8f (total %.0fs)" % (
                        init_name, n, global_best, time.monotonic() - t0))
        
        # Phase B: Active set equalization from best
        if best_for_len is not None:
            t_alloc = min(phase_times['active'], deadline - time.monotonic() - 30)
            if t_alloc > 10:
                a_opt, obj_opt = run_active_set_eq(
                    best_for_len, n, S, t_alloc, rng
                )
                obj_real = obj_from_conv(fft_self_conv(a_opt), n, a_opt.sum())
                if obj_real < best_obj_for_len:
                    best_obj_for_len = obj_real
                    best_for_len = a_opt.copy()
                if obj_real < global_best:
                    global_best = obj_real
                    global_a = a_opt.copy()
                    print("  [ActiveEq] n=%d: obj=%.8f" % (n, global_best))
        
        # Phase C: Coordinate descent (only feasible for smaller n)
        if best_for_len is not None and n <= 4096:
            t_alloc = min(phase_times['coord'], deadline - time.monotonic() - 30)
            if t_alloc > 10:
                a_opt, obj_opt = run_coordinate_descent(
                    best_for_len, n, S, t_alloc, rng
                )
                obj_real = obj_from_conv(fft_self_conv(a_opt), n, a_opt.sum())
                if obj_real < best_obj_for_len:
                    best_obj_for_len = obj_real
                    best_for_len = a_opt.copy()
                if obj_real < global_best:
                    global_best = obj_real
                    global_a = a_opt.copy()
                    print("  [CoordDesc] n=%d: obj=%.8f" % (n, global_best))
        
        # Phase D: SA polish
        if best_for_len is not None:
            t_alloc = min(phase_times['sa'], deadline - time.monotonic() - 30)
            if t_alloc > 10:
                a_opt, obj_opt = run_sa_optimizer(
                    best_for_len, n, S, t_alloc, rng
                )
                obj_real = obj_from_conv(fft_self_conv(a_opt), n, a_opt.sum())
                if obj_real < best_obj_for_len:
                    best_obj_for_len = obj_real
                    best_for_len = a_opt.copy()
                if obj_real < global_best:
                    global_best = obj_real
                    global_a = a_opt.copy()
                    print("  [SA] n=%d: obj=%.8f" % (n, global_best))
    
    # Final polish: if we have time, do one more LSE pass on the global best
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final polish (remaining %.0fs) ===" % remaining)
        
        a_opt, obj_opt, _ = run_lse_optimizer(
            global_a, n_best, S_best, remaining * 0.6, rng,
            beta_start=50.0, beta_end=1000.0
        )
        obj_real = obj_from_conv(fft_self_conv(a_opt), n_best, a_opt.sum())
        if obj_real < global_best:
            global_best = obj_real
            global_a = a_opt.copy()
            print("  Final LSE: obj=%.8f" % global_best)
        
        # Quick coordinate descent if n is small enough
        remaining2 = deadline - time.monotonic()
        if global_a is not None and remaining2 > 30:
            n_best = len(global_a)
            S_best = np.sqrt(2.0 * n_best)
            if n_best <= 8192:
                a_opt, obj_opt = run_coordinate_descent(
                    global_a, n_best, S_best, remaining2 * 0.8, rng
                )
                obj_real = obj_from_conv(fft_self_conv(a_opt), n_best, a_opt.sum())
                if obj_real < global_best:
                    global_best = obj_real
                    global_a = a_opt.copy()
                    print("  Final CoordDesc: obj=%.8f" % global_best)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("\nReturning sequence of length %d, obj=%.8f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
