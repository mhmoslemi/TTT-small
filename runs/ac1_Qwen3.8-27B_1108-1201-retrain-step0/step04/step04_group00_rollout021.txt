"""

Strategy: Multi-phase optimization combining:
1. LSE-smoothed FISTA with aggressive adaptive beta and backtracking line search
2. Spectral whitening "kicks" to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via L2 projection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_project(v, S, temperature=1.0):
    """Entropy-regularized projection: x_i = S * exp(v_i/t) / sum exp(v_j/t)."""
    v = np.asarray(v, dtype=np.float64)
    shifted = v / temperature - np.max(v) / temperature
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-300:
        return np.full(len(v), S / len(v))
    x = S * e / total
    return x


def lse_value_and_weights(c, beta):
    """Compute LSE value and softmax weights over c."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(len(c))
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return lse_val, w


def gradient_from_weights(w, y):
    """Compute gradient of sum_k w_k * c[k] w.r.t. y."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Y = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(Y)))[:n]


def exact_subgradient(y, c, top_k=1, tol_frac=0.0):
    """Subgradient from lags within tolerance of max (top-k)."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        k = min(top_k, len(c))
        active = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return gradient_from_weights(w_full, y)


def spectral_whiten(a, n_keep=0):
    """Apply spectral whitening: flatten |DFT(a)|^2, return real part."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    # Soft whitening: reduce peaks, boost dips
    A_new = A * np.sqrt(P_mean / np.maximum(P, P_mean * 0.01))
    a_new = np.real(ifft(A_new, fsize)[:n])
    return a_new


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_v2(a_init, max_time, rng, beta_init=20.0, beta_rate=0.004,
                 beta_max=50000.0, step_scale=1.0, use_backtrack=True,
                 spectral_kick_interval=5000, momentum_decay=0.999):
    """
    Improved FISTA with:
    - Adaptive LSE beta (exponential growth with cap)
    - Backtracking line search
    - Spectral whitening kicks
    - Momentum restart
    - Adaptive top-k blend when stuck
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient: blend LSE with top-k when stuck
        lse_val, w_lse = lse_value_and_weights(c, beta)
        if stuck_count > 2000:
            # Blend with exact top-3 subgradient
            active = np.argsort(c)[::-1][:3]
            w_top = np.zeros(len(c))
            w_top[active] = 1.0 / 3.0
            blend = min(0.5, stuck_count / 10000.0)
            w = (1 - blend) * w_lse + blend * w_top
        else:
            w = w_lse
        grad = gradient_from_weights(w, y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size
        step = step_scale / max(beta, 1.0)
        step = min(step, S / (0.5 * gnorm + 1e-15))

        # Backtracking
        if use_backtrack:
            for _ in range(8):
                x_new = project_simplex(y - step * grad, S)
                break
        else:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new * momentum_decay  # Slight momentum decay for stability

        # Spectral kick: periodically flatten the spectrum
        if it % spectral_kick_interval == 0 and stuck_count > 500:
            a_sw = spectral_whiten(y)
            a_sw = project_simplex(a_sw, S)
            # Blend: small perturbation toward whitened version
            alpha_kick = 0.1 * np.exp(-stuck_count / 5000.0)
            y = (1 - alpha_kick) * y + alpha_kick * a_sw
            y = project_simplex(np.maximum(y, 0), S)
            t_mom = 1.0  # Reset momentum
            stuck_count = 0

        # Noise injection when very stuck
        if stuck_count > 5000 and it % 200 == 0:
            noise_scale = max(5e-5, 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stuck_count = 0

        # Periodic momentum restart
        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, np.concatenate([a, [a[0]]]))


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def make_cosine_init(n, freq=1, n_harmonics=1):
    """Create a multi-harmonic cosine initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for h in range(n_harmonics):
        f = (h + 1) * freq
        a += 0.3 / (h + 1) * np.cos(2 * np.pi * f * tt)
    a = np.maximum(a, 0)
    return a


def make_random_phase_init(n, rng):
    """Create initialization from random Fourier phases."""
    phases = rng.uniform(0, 2 * np.pi, n)
    magnitudes = np.ones(n) * (1.0 / np.sqrt(n))
    # Add some low-frequency bias
    freqs = np.arange(n)
    low_freq_mask = freqs < n // 8
    magnitudes[low_freq_mask] *= 2.0
    A = magnitudes * np.exp(1j * phases)
    a = np.real(ifft(A, n=n))
    a = np.maximum(a, 0)
    return a


def make_triangular_init(n):
    """Triangular (Bartlett-like) window initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.where(tt < 0.5, 2 * tt, 2 * (1 - tt))
    a = np.maximum(a, 0.01)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase approach: diverse initialization, FISTA optimization,
    spectral kicks, multi-scale exploration, and intensive polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj(a)
        if np.isfinite(obj) and obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
        return obj

    # Get initial sequence from provided or create default
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_cosine_init(n, freq=1, n_harmonics=3)

    n0 = len(a_start)
    print("Starting: n=%d" % n0)
    update_best(a_start)
    print("Initial objective: %.6f" % global_best_obj)

    # === Phase 1: Multi-start FISTA with diverse initializations (40% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.40
    print("\n=== Phase 1: Multi-start FISTA (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    # Start 1: Provided/best initial
    a1, _ = run_fista_v2(a_start, p1_budget * 0.35, rng,
                          beta_init=20.0, beta_rate=0.005,
                          beta_max=50000.0, step_scale=1.0)
    obj1 = update_best(a1)
    print("  Start 1 (provided): obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t1))

    # Start 2: Cosine with different frequency
    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 30:
        a_cos = make_cosine_init(n0, freq=2, n_harmonics=4)
        a2, _ = run_fista_v2(a_cos, remaining1 * 0.5, rng,
                              beta_init=15.0, beta_rate=0.004,
                              beta_max=30000.0, step_scale=0.8)
        obj2 = update_best(a2)
        print("  Start 2 (cosine f=2): obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t1))

    # Start 3: Random phase
    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 30:
        a_rp = make_random_phase_init(n0, rng)
        a3, _ = run_fista_v2(a_rp, remaining1 * 0.6, rng,
                              beta_init=10.0, beta_rate=0.006,
                              beta_max=40000.0, step_scale=1.2)
        obj3 = update_best(a3)
        print("  Start 3 (random phase): obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t1))

    # Continue best with entropy-regularized FISTA
    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 20:
        a4, _ = run_entropy_fista(global_best_a, remaining1, rng,
                                   temp_init=0.05, temp_decay=0.9998)
        obj4 = update_best(a4)
        print("  Entropy FISTA: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t1))

    # === Phase 2: Continue FISTA on best with aggressive settings (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.25
        print("\n=== Phase 2: Aggressive FISTA on best (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        # High-beta focused
        a5, _ = run_fista_v2(global_best_a, p2_budget * 0.5, rng,
                              beta_init=100.0, beta_rate=0.003,
                              beta_max=100000.0, step_scale=0.5,
                              spectral_kick_interval=3000)
        obj5 = update_best(a5)
        print("  High-beta FISTA: obj=%.6f (t=%.0fs)" % (obj5, time.monotonic() - t2))

        # Lower step, more exploration
        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 20:
            a6, _ = run_fista_v2(global_best_a, remaining2, rng,
                                  beta_init=50.0, beta_rate=0.002,
                                  beta_max=50000.0, step_scale=0.3,
                                  spectral_kick_interval=2000)
            obj6 = update_best(a6)
            print("  Low-step FISTA: obj=%.6f (t=%.0fs)" % (obj6, time.monotonic() - t2))

    # === Phase 3: Multi-scale upsample (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and n0 <= 16384:
        p3_budget = remaining * 0.20
        n2 = n0 * 2
        print("\n=== Phase 3: Upsample to n=%d (budget %.0fs) ===" % (n2, p3_budget))
        t3 = time.monotonic()

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())

        a7, _ = run_fista_v2(a_up, p3_budget * 0.6, rng,
                              beta_init=10.0, beta_rate=0.004,
                              beta_max=30000.0, step_scale=1.0,
                              spectral_kick_interval=4000)
        obj7 = update_best(a7)
        print("  FISTA at 2n: obj=%.6f (t=%.0fs)" % (obj7, time.monotonic() - t3))

        # Downsample back and compare
        a_ds = downsample(a7, factor=2)
        S_ds = np.sqrt(2.0 * n0)
        a_ds = a_ds * (S_ds / a_ds.sum())
        obj_ds = update_best(a_ds)
        print("  Downsampled back to n=%d: obj=%.6f" % (n0, obj_ds))

        # Polish the upsampled version
        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 20:
            a7b, _ = run_active_polish(global_best_a if len(global_best_a) == n2 else a7,
                                        remaining3, rng)
            obj7b = update_best(a7b)
            print("  Polish at 2n: obj=%.6f (t=%.0fs)" % (obj7b, time.monotonic() - t3))

    # === Phase 4: Intensive active-set polishing (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = remaining * 0.50
        print("\n=== Phase 4: Active-set polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a8, _ = run_active_polish(global_best_a, p4_budget * 0.6, rng)
        obj8 = update_best(a8)
        print("  Active polish 1: obj=%.6f (t=%.0fs)" % (obj8, time.monotonic() - t4))

        remaining4 = p4_budget - (time.monotonic() - t4)
        if remaining4 > 10:
            a9, _ = run_active_polish(global_best_a, remaining4, rng)
            obj9 = update_best(a9)
            print("  Active polish 2: obj=%.6f (t=%.0fs)" % (obj9, time.monotonic() - t4))

    # === Phase 5: Final spectral + FISTA combo (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.80
        print("\n=== Phase 5: Final optimization (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # One more aggressive FISTA run
        a10, _ = run_fista_v2(global_best_a, p5_budget * 0.5, rng,
                               beta_init=200.0, beta_rate=0.001,
                               beta_max=200000.0, step_scale=0.2,
                               spectral_kick_interval=2000)
        obj10 = update_best(a10)
        print("  Final FISTA: obj=%.6f (t=%.0fs)" % (obj10, time.monotonic() - t5))

        # Brief active polish
        remaining5 = p5_budget - (time.monotonic() - t5)
        if remaining5 > 5:
            a11, _ = run_active_polish(global_best_a, remaining5, rng)
            obj11 = update_best(a11)
            print("  Final polish: obj=%.6f (t=%.0fs)" % (obj11, time.monotonic() - t5))

    # === Final: Try a few quick random perturbations around best ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Final perturbations (budget %.0fs) ===" % remaining)
        t_final = time.monotonic()
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)

        for trial in range(5):
            if time.monotonic() - t_final > remaining * 0.8:
                break
            noise_scale = 0.005 * S_best / np.sqrt(n_best)
            a_pert = global_best_a + rng.normal(0, noise_scale, n_best)
            a_pert = project_simplex(a_pert, S_best)
            # Quick FISTA from perturbation
            a_pf, _ = run_fista_v2(a_pert, remaining * 0.08, rng,
                                    beta_init=50.0, beta_rate=0.005,
                                    beta_max=50000.0, step_scale=0.8)
            obj_pf = update_best(a_pf)
            print("  Perturb %d: obj=%.6f" % (trial, obj_pf))

    # Final verification
    print("\n=== Final Result ===")
    print("Best objective: %.6f (n=%d)" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("Total time used: %.0fs" % (time.monotonic() - (deadline - budget_s + 30.0)))
    return final_list
