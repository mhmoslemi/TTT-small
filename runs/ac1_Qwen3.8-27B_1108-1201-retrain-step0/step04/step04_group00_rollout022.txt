import numpy as np
import time
from scipy.fft import fft, ifft

# Helper functions (top-level, no closures)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def exact_subgradient(y, c, tol_frac=0.0):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_gradient(y, c, k, power=1.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def structured_perturb(a, c, rng, S, strength=0.01):
    """
    Perturb the sequence in a structured way based on the convolution.
    Positions that contribute to the peak lags get decreased,
    positions that don't get increased.
    """
    n = len(a)
    max_val = c.max()
    active = np.where(c >= max_val * 0.99)[0]

    # Compute contribution of each position to active lags
    contrib = np.zeros(n)
    for k in active:
        for j in range(n):
            other = k - j
            if 0 <= other < n:
                contrib[j] += a[j] * a[other]

    # Normalize contribution
    contrib_max = contrib.max()
    if contrib_max < 1e-15:
        return a.copy()

    # Decrease high-contribution positions, increase low-contribution
    direction = -contrib / contrib_max
    # Add some randomness
    direction += rng.normal(0, 0.5 * strength, n)

    x_new = project_simplex(a + strength * direction, S)
    return x_new


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase strategy: FISTA LSE exploration, multi-length cross-pollination,
subgradient with backtracking line search, and intensive final polishing with
annealed active-set size and multi-direction probing.
Key improvements over baseline: backtracking step sizes for subgradient,
structured multi-direction probes when stuck, annealed active set, and
adaptive multi-length exploration with re-optimization.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 1:
        return np.array([max(0.0, S)])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        return np.zeros(n)
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,

    FISTA with adaptive LSE smoothing and periodic multi-direction probing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

            grad = lse_gradient(y, c, beta)
        elif mode == "hybrid":
            stuck = it - last_improve
            if stuck > 400 and (it % 3 == 0 or rng.random() < 0.3):
                tol = min(0.02, 0.005 * (stuck // 200 + 1))
                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if it - last_improve > 500:

            grad = topk_gradient(y, c, k, power=1.3)
        else:
            grad = lse_gradient(y, c, beta)

        if gnorm < 1e-15:
            break

        # FISTA update

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

        a = x_new
        t_mom = t_new

                    best_a = y_probe.copy()
                    y = y_probe.copy()
                    a = y_probe.copy()
                    t_mom = 1.0
                    last_improve = it

    step = S / (2.0 * max(gnorm0, 1e-10))

        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:

            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 50:
            tol_frac = 0.0
        elif stuck < 200:
            tol_frac = 0.002
        elif stuck < 500:
            tol_frac = 0.005
        elif stuck < 1000:
            tol_frac = 0.01
        else:
            tol_frac = min(0.03, 0.01 + 0.001 * (stuck // 500))

                step = step_try  # keep successful step size
                break
            step_try *= 0.5
            if step_try < 1e-12:
                break

        if not improved:
            # Try without momentum
            v = np.zeros(n)
            for bt in range(5):
                x_try = project_simplex(y - step_try * grad, S)
                c_try = fft_convolve_self(x_try)
                if c_try.max() < cur_max * (1.0 - 1e-8):
                    y = x_try
                    step = step_try
                    improve_flag = True
                    break
                step_try *= 0.5
            else:
                # Stuck, apply restart
                step *= 0.5
                if stuck > 2000:
                    noise_scale = max(1e-6, 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    v = np.zeros(n)
                    # Reset step
                    c_reset = fft_convolve_self(y)
                    g_reset = exact_subgradient(y, c_reset, tol_frac=0.0)
                    gnorm_reset = np.linalg.norm(g_reset)
                    step = S / (2.0 * max(gnorm_reset, 1e-10))
                    mu = 0.5  # reduce momentum after restart

        # Momentum adaptation
        mu = min(0.95, mu + 0.0001)

    return best_a, best_conv


def run_focused_polish(a_init, max_time, rng):
    """
    Intensive polishing: alternate between LSE with high beta and exact subgradient.
    Use very small steps and frequent evaluation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Alternate between modes
        if stuck < 1000:
            # High-beta LSE
            beta = 50.0 + 50.0 * (it // 500)
            grad = lse_gradient(y, c, min(beta, 500.0))
            step = 0.3 / max(beta, 1.0)
        elif stuck < 3000:
            # Exact subgradient with small tol
            grad = exact_subgradient(y, c, tol_frac=0.005)
            gnorm = np.linalg.norm(grad)
            step = S / (10.0 * max(gnorm, 1e-10))
        else:
            # Very small tol, many probes
            grad = exact_subgradient(y, c, tol_frac=0.001)
            gnorm = np.linalg.norm(grad)
            step = S / (20.0 * max(gnorm, 1e-10))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Probe when stuck for a while
        if stuck > 2000 and it % 500 == 0:
            for _ in range(6):
                scale = 0.003 + 0.01 * rng.random()
                y_probe = project_simplex(y + rng.normal(0, scale * S / np.sqrt(n), n), S)
                c_probe = fft_convolve_self(y_probe)
                if c_probe.max() < best_conv - 1e-14:
                    best_conv = c_probe.max()
                    best_a = y_probe.copy()
                    y = y_probe.copy()
                    a = y_probe.copy()
                    t_mom = 1.0
                    last_improve = it
                    break

    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400):
