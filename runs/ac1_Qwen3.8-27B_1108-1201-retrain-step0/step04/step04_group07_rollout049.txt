"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smoothed projected gradient
with log-sum-exp regularization, Nesterov acceleration, and adaptive epsilon.

Key ideas:
1. Log-sum-exp smoothing of the max autocorrelation gives a convex smooth objective
   with well-defined gradient computed via FFT.
2. Adaptive epsilon: start moderate (multi-peak awareness) and decrease (focus on true max).
3. Nesterov-accelerated projected gradient with backtracking line search.
4. Multiple initializations and sequence lengths.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N=None):
    """Full self-convolution of a via FFT."""
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def smooth_objective(c, epsilon):
    """Compute log-sum-exp smoothing of max(c)."""
    c_shifted = c / epsilon - np.max(c) / epsilon
    return float(epsilon * (np.max(c) / epsilon + np.log(np.sum(np.exp(c_shifted)))))


def compute_smoothing_gradient(a, epsilon, N_fft=None):
    """
    Compute gradient of epsilon * log(sum(exp(c[k]/epsilon))) w.r.t. a.
    where c = full convolution of a with itself.
    """
    n = len(a)
    if N_fft is None:
        N_fft = 1 << (3 * n - 2).bit_length()
    
    # Compute autocorrelation
    A = fft(a, N_fft)
    P_spec = np.abs(A) ** 2
    c = np.real(ifft(P_spec))[:2 * n - 1]
    
    # Compute softmax weights
    c_scaled = c / epsilon
    c_max = c_scaled.max()
    exps = np.exp(c_scaled - c_max)
    p = exps / exps.sum()
    
    # Compute gradient via FFT
    # grad = real(ifft(P * conj(A)) + ifft(P * A))[:n]
    P = fft(p, N_fft)
    term1 = ifft(P * np.conj(A))[:n]
    term2 = ifft(P * A)[:n]
    grad = np.real(term1 + term2)
    
    return grad, c, p


def smoothed_projected_gradient(a_init, n, S, max_time, rng,

    y = a.copy()
    t_nest = 1.0

    best_score = compute_score(a)
    best_local = a.copy()
    it = 0

        it += 1

        grad_y, c_y, p_y = compute_smoothing_gradient(y, epsilon, N_fft)

        grad_norm = np.linalg.norm(grad_y)
        if grad_norm < 1e-15:
            break

            val_y = smooth_objective(c_y, epsilon)
            step_bt = step

            c_x = fft_convolve_self(x_candidate, N_fft)

        else:

        # Check actual score
        if it % 10 == 0:

            if sc < best_score:

                best_local = a.copy()
                if verbose and it % 100 == 0:

        best_local = a.copy()

def spectral_equalize(a, N, n_iter, alpha, S):
    """
    Spectral equalization: iteratively push |FFT(a)|^2 toward its geometric mean.
    This is a different angle: instead of gradient on the time-domain objective,
    directly modify the spectrum.
    """
    n = len(a)
    a_cur = a.copy()
    
    for it in range(n_iter):
        A = fft(a_cur, N)
        P = np.abs(A) ** 2
        P[P < 1e-30] = 1e-30
        
        # Target: geometric mean of P
        log_P = np.log(P)
        log_target = np.mean(log_P)
        
        # Push toward target
        log_P_new = log_P + alpha * (log_target - log_P)
        P_new = np.exp(log_P_new)
        
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        # Project to simplex
        a_new = project_simplex(np.maximum(a_new, 0), S)
        a_cur = a_new
    
    return a_cur
