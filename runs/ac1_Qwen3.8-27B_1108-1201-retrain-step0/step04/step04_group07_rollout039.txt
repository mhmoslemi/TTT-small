```python
import numpy as np
import time
from scipy.fft import fft, ifft

def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def score(a):
    """Compute the objective."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (S * S))
```

```python
def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_range = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / rho_range
    valid = (theta_cand <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_fft(a, N):
    """Self-convolution via FFT. N is the FFT size (>= 2*len(a)-1)."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-start with diverse initializations (binary, smooth, chirp, mirror-anti)
2. Phase 1: Spectral flattening via amplitude compression in frequency domain
3. Phase 2: Nesterov-accelerated projected subgradient on top-K convolution lags
   with exact (non-circular) gradient computation and adaptive step size
4. Phase 3: Final gentle refinement interleaving spectral and subgradient steps
5. Multi-length search (8192, 16384, 4096)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_range = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / rho_range
    valid = (theta_cand <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_self(a, N):
    """Self-convolution via FFT. N must be >= 2*len(a)-1."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * len(a) - 1]


def compute_score(a):
    """Compute 2n*max(c)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2 * n - 1).bit_length()
    c = convolve_self(a, N)
    return float(2.0 * n * c.max() / (S * S))


def spectral_flatten_step(a, N, alpha):
    """One step: compress |FFT(a)|^2 toward uniform, preserving phase."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_mean = P.sum() / N
    if P_mean < 1e-30:
        return a.copy()
    beta = 1.0 - alpha
    P_new = P_mean + (P - P_mean) * beta
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return proj_simplex(a_new, S)


def spectral_anneal(a_init, N, max_time, rng, target=1.50, verbose=False):
    """Spectral flattening with alpha annealing from aggressive to gentle."""
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        # Alpha: start at 0.7, decay to 0.03
        alpha = 0.7 * (1.0 - 0.95 * progress) + 0.03

        a = spectral_flatten_step(a, N, alpha)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
        if best_score <= target:
            break

        # Occasional perturbation to escape plateaus
        if it % 300 == 150 and rng.random() < 0.4:
            noise = rng.standard_normal(n) * (0.003 * S / np.sqrt(n))
            a = proj_simplex(a + noise, S)

    return best_a, best_score


def subgradient_step_grad(a, c, top_k, n):
    """
    Compute exact subgradient from top-K lags of convolution c.
    grad[j] = sum_{k in T} w_k * 2 * a[k-j] (with zero extension)
    where w_k are normalized weights from top c values.
    """
    k_use = min(top_k, len(c))
    # Get top-K indices
    if k_use == len(c):
        top_idx = np.arange(len(c))
        top_vals = c.copy()
    else:
        top_idx = np.argpartition(c, -k_use)[-k_use:]
        top_vals = c[top_idx]

    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), top_idx, np.zeros(k_use)

    # Use exponential weighting: higher weight for lags closer to max
    c_max = top_vals.max()
    # weights proportional to exp((top_vals - c_max) / (c_max * 0.1 + 1e-10))
    # This focuses more on the actual peak
    spread = c_max * 0.15 + 1e-10
    weights = np.exp((top_vals - c_max) / spread)
    weights /= weights.sum()

    grad = np.zeros(n)
    for i in range(k_use):
        k = int(top_idx[i])
        wk = weights[i]
        # grad[j] += wk * 2 * a[k-j] for valid j: max(0,k-n+1) <= j <= min(n-1,k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            j_arr = np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] += wk * 2.0 * a[k - j_arr]

    return grad, top_idx, weights


def subgradient_polish(a_init, max_time, rng, top_k_init=50, top_k_max=200, verbose=False):
    """
    Nesterov-accelerated projected subgradient with adaptive step size.
    """
    n = len(a_init)
    S = a_init.sum()
    N = 1 << (2 * n - 1).bit_length()

    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_param = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    step = 2.0 / n
    it = 0
    stall = 0
    best_for_restart = best_score

    while time.monotonic() - start < max_time:
        it += 1
        c = convolve_self(y, N)
        c_max = c.max()
        cur_score = 2.0 * n * c_max / (S * S)

        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > 500:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = proj_simplex(y + rng.standard_normal(n) * noise_scale, S)
            a = y.copy()
            t_param = 1.0
            step = 2.0 / n
            stall = 0
            if verbose:
                print(f"    [restart] it={it}, best={best_score:.6f}")

        # Adaptive top_k: increase over time for better gradient
        k_use = min(top_k_max, top_k_init + it // 20)

        grad, _, _ = subgradient_step_grad(y, c, k_use, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Nesterov step
        x_new = proj_simplex(y - step * grad / gnorm, S)

        # Adaptive step size (check every 20 iterations)
        if it % 20 == 0:
            c_check = convolve_self(x_new, N)
            check_score = 2.0 * n * c_check.max() / (S * S)
            if check_score > cur_score * 1.02:
                step *= 0.7
            elif check_score < cur_score * 0.98:
                step = min(step * 1.15, 20.0 / n)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_param = t_new

        # Slow step decay
        if it % 1000 == 0:
            step *= 0.92

        if verbose and it % 2000 == 0:
            print(f"    [sg] it={it}, score={best_score:.6f}, step={step:.2e}")

    return best_a, best_score

    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth", (0.1 + 0.9 * smooth) * (S / smooth.sum())))

    # Chirp
    chirp = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt ** 2))
    chirp = np.maximum(chirp, 0.05)
    inits.append(("chirp", chirp * (S / chirp.sum())))

    # Multi-sine
    ms = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.sin(2 * np.pi * 7 * tt) + \
           0.1 * np.sin(2 * np.pi * 11 * tt) + 0.05 * np.sin(2 * np.pi * 17 * tt)
    ms = np.maximum(ms, 0.05)
    inits.append(("multisine", ms * (S / ms.sum())))

    # Uniform
    inits.append(("uniform", np.ones(n) * S / n))

    return inits


def optimize_length(n, budget, rng, provided=None, target=1.50):
    """Full optimization pipeline for a given sequence length."""
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()

    best_overall = np.inf
    best_seq = None
    t_start = time.monotonic()

    inits = make_initializations(n, S, rng, provided)
    n_inits = len(inits)

    # Allocate: 15s spectral per init, rest for subgradient
    spec_time_each = min(15.0, budget / (n_inits * 4))
    sg_time_each = min(120.0, budget / (n_inits * 2.5))

    for name, a0 in inits:
        elapsed = time.monotonic() - t_start
        remaining = budget - elapsed
        if remaining < 10:
            break

        # Phase 1: Spectral flattening
        t_spec = min(spec_time_each, remaining * 0.15)
        if t_spec < 2:
            break
        a_spec, sc_spec = spectral_anneal(a0, N, t_spec, rng, target)
        if sc_spec < best_overall:
            best_overall = sc_spec
            best_seq = a_spec.copy()
            print(f"    [{name}] spectral: {sc_spec:.6f}")

        # Phase 2: Subgradient
        rem = budget - (time.monotonic() - t_start)
        if rem < 5:
            break
        t_sg = min(sg_time_each, rem * 0.4)
        if t_sg < 3:
            continue
        a_sg, sc_sg = subgradient_polish(a_spec, t_sg, rng,
                                         top_k_init=min(50, n // 128),
                                         top_k_max=min(200, n // 32))
        if sc_sg < best_overall:
            best_overall = sc_sg
            best_seq = a_sg.copy()
            print(f"    [{name}] subtotal: {sc_sg:.6f}")

    # Phase 3: Final extended polish on best
    if best_seq is not None:
        rem = budget - (time.monotonic() - t_start)
        if rem > 20:
            # Long subgradient run
            t_final_sg = min(rem * 0.6, 300.0)
            a_f1, sc_f1 = subgradient_polish(best_seq, t_final_sg, rng,
                                             top_k_init=min(100, n // 64),
                                             top_k_max=min(300, n // 16))
            if sc_f1 < best_overall:
                best_overall = sc_f1
                best_seq = a_f1.copy()
                print(f"    [final_sg] {sc_f1:.6f}")

            # Gentle spectral pass
            rem2 = budget - (time.monotonic() - t_start)
            if rem2 > 10:
                t_f2 = min(rem2 * 0.3, 60.0)
                a_f2, sc_f2 = spectral_anneal(best_seq, N, t_f2, rng, target)
                if sc_f2 < best_overall:
                    best_overall = sc_f2
                    best_seq = a_f2.copy()
                    print(f"    [final_spec] {sc_f2:.6f}")

            # One more subgradient
            rem3 = budget - (time.monotonic() - t_start)
            if rem3 > 10:
                a_f3, sc_f3 = subgradient_polish(best_seq, min(rem3 * 0.5, 120.0), rng,
                                                 top_k_init=min(150, n // 48),
                                                 top_k_max=min(400, n // 12))
                if sc_f3 < best_overall:
                    best_overall = sc_f3
                    best_seq = a_f3.copy()
                    print(f"    [final_sg2] {sc_f3:.6f}")

    return best_seq, best_overall


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing evaluate_sequence.
    Uses multi-start spectral flattening + Nesterov-accelerated projected
    subgradient with exact gradient and adaptive step size, across multiple
    sequence lengths.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global = time.monotonic()
    print(f"Starting optimization, budget={budget_s:.0f}s")

    # Get provided sequence
    provided = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        print(f"Provided sequence: length={len(provided)}")
    except Exception:
        pass

    # Determine lengths to try
    lengths_to_try = []
    if provided is not None and len(provided) >= 1024:
        lengths_to_try.append(len(provided))
    for L in [8192, 16384, 4096]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Prioritize: provided first, then 8192, then 16384, then 4096
    priority = {}
    if provided is not None:
        priority[len(provided)] = 0
    priority[8192] = 1
    priority[16384] = 2
    priority[4096] = 3
    lengths_to_try.sort(key=lambda x: priority.get(x, 99))

    global_best = np.inf
    global_seq = None
    total_budget = budget_s - 40  # reserve for verification

    # Allocate budget proportional to priority
    # Give more time to primary length
    n_len = len(lengths_to_try)
    allocs = []
    for i, L in enumerate(lengths_to_try):
        if i == 0:
            allocs.append(total_budget * 0.55)
        elif i == 1:
            allocs.append(total_budget * 0.25)
        elif i == 2:
            allocs.append(total_budget * 0.15)
        else:
            allocs.append(total_budget * 0.05)

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_global
        remaining = budget_s - 40 - elapsed
        if remaining < 30:
            break

        alloc = min(allocs[i], remaining - 30)
        if alloc < 30:
            break

        print(f"\n{'='*50}")
        print(f"Optimizing n={n}, budget={alloc:.0f}s")
        print(f"{'='*50}")

        seq, sc = optimize_length(n, alloc, rng, provided=provided, target=1.50)

        if seq is not None and sc < global_best:
            global_best = sc
            global_seq = seq.copy()
            print(f"  >> New global best: {global_best:.6f} (n={n})")
        elif seq is not None:
            print(f"  Best for n={n}: {sc:.6f} (global: {global_best:.6f})")

    # Fallback
    if global_seq is None:
        n_fb = 4096
        global_seq = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence")

    # Final verification
    global_seq = np.clip(global_seq, 0.0, 1000.0)
    if global_seq.sum() < 0.01:
        global_seq = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_seq]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print(f"\nVerification score: {ver:.6f}")
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate returned inf, using fallback")
        global_seq = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_seq]
        ver = evaluate_sequence(final_list)
        print(f"Fallback verification: {ver:.6f}")

    print(f"Final: length={len(final_list)}, score={ver:.6f}")
    return final_list
```
