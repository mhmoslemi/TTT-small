def project_simplex(v, S):
    n = len(v)
    if n == 0: return np.zeros(0)
    if S <= 0: return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u > (css - S) / np.arange(1, n + 1, dtype=np.float64))[0]
    if len(rho) == 0:
        return np.full(n, S / n, dtype=np.float64)
    theta = (css[rho[-1]] - S) / (rho[-1] + 1)
    out = np.maximum(v - theta, 0.0)
    ss = out.sum()
    if ss <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    out = out * (S / ss)
    return out
```

```python
"""
Minimize 2*n*max(self-convolution)/sum^2 for a non-negative sequence.

The program works on a fixed length (usually the length of height_sequence_1,
otherwise 8192).  It normalizes sequences to sum sqrt(2n), so the value being
minimized locally is exactly the coefficient max of the convolution.

Core ideas:
1. Mirror-descent / natural-gradient steps on an LSE-smoothed peak objective.
   This has the fixed-point property that active-lag gradients are equalized
   on the support, which is close to the KKT condition.
2. Alternating spectral-projection moves:
   - clip the convolution spectrum/time-domain coefficients to reduce peaks,
   - take an approximate non-negative square root in the frequency domain,
   - project back to the simplex.
   Two square-root candidates are used: old phase and complex half-angle.
3. A small L-BFGS block on the LSE objective for local polishing.
4. Random basin exploration with short mirror-descent runs.

No file/network IO.  evaluate_sequence is not called inside the search;
all candidates are normalized and clipped to valid non-negative floats.
"""

import numpy as np
import time

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


def get_initial_global():
    """Return the previous best sequence if available, else None."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return np.clip(arr, 0.0, 1000.0)


def safe_normalize(a, n, S):
    """Clip to a valid non-negative vector and rescale to sum S."""
    arr = np.asarray(a, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size != n:
        arr = arr[:n] if arr.size > n else np.pad(arr, (0, n - arr.size))
    s = float(arr.sum())
    if s <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    arr = arr * (S / s)
    return arr


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    valid = u > (css - S) / idx
    rho = np.nonzero(valid)[0]
    if len(rho) == 0:
        return np.full(n, S / float(n), dtype=np.float64)
    theta = (css[int(rho[-1])] - S) / (float(int(rho[-1])) + 1.0)
    out = np.maximum(v - theta, 0.0)
    ss = float(out.sum())
    if ss <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    return out * (S / ss)


def conv_full(a, N):
    """Full linear self-convolution via FFT. Returns (c, A)."""
    n = len(a)
    ap = np.zeros(N, dtype=np.float64)
    ap[:n] = a
    A = fft(ap)
    prod = A * A
    c = np.real(ifft(prod))[:2 * n - 1]
    # Remove tiny negative noise; convolution of nonnegative data should be >= 0.
    c = np.where(c > 0.0, c, 0.0)
    c = np.where(np.isfinite(c), c, 0.0)
    return c, A


def cmax_of(a, N):
    """Return max coefficient of a*a, with NaN/Inf guard."""
    c, _ = conv_full(a, N)
    val = float(c.max())
    return val if np.isfinite(val) else np.inf


def lse_grad(a, N, beta):
    """
    Gradient and LSE-smoothed max of self-convolution.

    Returns (grad, lse_value, c, cmax).
    """
    n = len(a)
    c, A = conv_full(a, N)
    cmax = float(c.max())
    if not np.isfinite(cmax):
        return np.zeros(n, dtype=np.float64), np.inf, c, np.inf
    beta = float(max(1.0, beta))
    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    total = float(e.sum())
    if total <= 0.0 or not np.isfinite(total):
        return np.zeros(n, dtype=np.float64), cmax, c, cmax
    w = e / total
    lse_val = cmax + max(0.0, float(np.log(total))) / beta
    W = fft(w, N)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g, lse_val, c, cmax


def lbfgs_dir(s_hist, y_hist, g):
    """Unit L-BFGS descent direction. Falls back to negative gradient."""
    m = len(s_hist)
    n = len(g)
    gn = float(np.linalg.norm(g))
    if gn < 1e-18:
        return np.zeros(n, dtype=np.float64)
    if m == 0:
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m, dtype=np.float64)
    for i in range(m - 1, -1, -1):
        si = s_hist[i]
        yi = y_hist[i]
        yy = float(np.dot(yi, yi))
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = float(np.dot(si, q)) / yy
            q = q - alphas[i] * yi

    sl = s_hist[-1]
    yl = y_hist[-1]
    yy = float(np.dot(yl, yl))
    if yy > 1e-30:
        gamma = max(1e-12, float(np.dot(sl, yl)) / yy)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        si = s_hist[i]
        yi = y_hist[i]
        yy = float(np.dot(yi, yi))
        if yy < 1e-30:
            continue
        beta_i = float(np.dot(si, yi)) / yy
        r = r + (alphas[i] - beta_i) * yi

    d = -r
    if float(np.dot(g, d)) > 1e-18:
        d = -g
    dn = float(np.linalg.norm(d))
    if dn < 1e-18:
        return -g / gn
    return d / dn


def mirror_lse_block(a_init, n, S, N, budget_s, rng, beta=20.0, eta=1.0):
    """
    Mirror/natural-gradient descent on LSE-smoothed peak.

    Uses update a_i <- a_i * exp(-eta * (g_i - mass_mean(g)) / scale(g)),
    normalized to sum S.  Temperature-controlled line search accepts small
    temporary increases to avoid discrete max stalls.
    """
    a = safe_normalize(a_init, n, S)
    best_m = cmax_of(a, N)
    best_a = a.copy()
    end = time.monotonic() + max(0.0, float(budget_s))
    beta_curr = float(beta)
    eta_curr = float(max(0.05, eta))
    temp = 0.004 * max(1.0, best_m)
    it = 0
    stagnation = 0

    inv_sqrt_n = 1.0 / np.sqrt(float(n))

    while time.monotonic() < end:
        it += 1
        if it % 100 == 0:
            beta_curr = min(500.0, beta_curr * 1.06)
            temp = max(1e-6, temp * 0.985)

        g, _, c, m = lse_grad(a, N, beta_curr)
        if m < best_m - 1e-13:
            best_m = m
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Escape local/contact set stagnation.
        if stagnation > 5000:
            std = 0.02 * S * inv_sqrt_n
            noise = rng.normal(0.0, std, n) + 0.25 * rng.random(n) * std
            a = safe_normalize(a + noise, n, S)
            eta_curr = max(0.2, eta_curr * 0.5)
            stagnation = 0
            continue

        s_inv = 1.0 / S if S > 0 else 0.0
        mean_g = float(np.dot(a, g) * s_inv)
        dev = g - mean_g
        var = float(np.dot(a, dev * dev) * s_inv)
        scale = max(np.sqrt(var), 1e-10)
        if scale < 1e-12:
            a = safe_normalize(a + rng.normal(0.0, 1e-4 * S * inv_sqrt_n, n), n, S)
            continue

        mults = (eta_curr * 0.5, eta_curr, eta_curr * 2.0)
        min_obj = m
        min_cand = None
        min_mult = eta_curr

        for mult in mults:
            arg = -mult * (dev / scale)
            arg = np.clip(arg, -30.0, 30.0)
            up = np.exp(arg)
            cand = a * up
            ssum = float(cand.sum())
            if ssum <= 1e-15:
                continue
            cand = cand * (S / ssum)
            cand = safe_normalize(cand, n, S)
            ob = cmax_of(cand, N)
            if ob < min_obj:
                min_obj = ob
                min_cand = cand
                min_mult = mult

        if min_cand is not None and (min_obj < m - 1e-13 or min_obj - m < temp):
            a = min_cand
            if min_obj < best_m:
                best_m = min_obj
                best_a = a.copy()
                stagnation = 0
            # Adapt step size to the accepted multiplier.
            if min_mult > eta_curr * 1.8:
                eta_curr = min(200.0, eta_curr * 1.1)
            elif min_mult < eta_curr * 0.7:
                eta_curr = max(0.02, eta_curr * 0.8)
        else:
            eta_curr = max(0.02, eta_curr * 0.6)
            if eta_curr <= 0.02 and it % 200 == 0:
                std = 0.005 * S * inv_sqrt_n
                a = safe_normalize(a + rng.normal(0.0, std, n), n, S)
                eta_curr = 1.0

    return best_a, best_m


def spectral_shave_block(a_init, n, S, N, budget_s, rng, frac=0.05, target_m=None):
    """
    Alternating projection / spectral shaping block.

    Alternate between:
      - reducing high convolution coefficients (cap + rescale) in the
        convolution domain,
      - converting the modified sequence back to a candidate height sequence
        by taking an approximate frequency-domain square root,
        respecting the old phase or using a complex square root,
        then projecting to the simplex.
    """
    a = safe_normalize(a_init, n, S)
    best_m = cmax_of(a, N)
    best_a = a.copy()
    end = time.monotonic() + max(0.0, float(budget_s))
    f = float(frac)
    temp = 0.008 * max(1.0, best_m)
    it = 0
    last_improve = 0
    inv_sqrt_n = 1.0 / np.sqrt(float(n))
    s2 = S * S

    while time.monotonic() < end:
        it += 1
        c, A = conv_full(a, N)
        m = float(c.max())
        if not np.isfinite(m):
            m = np.inf
        if m < best_m - 1e-13:
            best_m = m
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 4000:
            std = 0.02 * S * inv_sqrt_n
            noise = rng.normal(0.0, std, n) + 0.25 * rng.random(n) * std
            a = safe_normalize(a + noise, n, S)
            f = max(0.02, f * 0.8)
            temp = max(1e-6, temp * 0.98)
            last_improve = it
            continue

        fr = f * (0.5 + rng.random())
        cap = m * (1.0 - fr)
        if target_m is not None:
            cap = min(cap, float(target_m))

        avg = s2 / float(2 * n - 1)
        floor = 0.5 * m + 0.5 * avg
        if cap < floor:
            cap = floor
        if cap >= m:
            cap = m * 0.98

        b = c.copy()
        over = b > cap
        if np.any(over):
            b[over] = cap

        # small random query to avoid exact fixed points and phase lock-in
        b = b + rng.normal(0.0, 0.001 * max(1.0, m), b.size)
        b = np.maximum(b, 0.0)
        bsum = float(b.sum())
        if bsum <= 1e-15 or not np.isfinite(bsum):
            continue
        b *= (s2 / bsum)

        B = fft(b, N)
        mag = np.sqrt(np.abs(B))

        # Candidate 1: preserve old phase.
        an_old = ifft(mag * np.exp(1j * np.angle(A))).real[:n]
        cand_old = project_simplex(an_old, S)

        # Candidate 2: complex square root, i.e. half-angle of modified B.
        an_half = ifft(np.sqrt(B)).real[:n]
        cand_half = project_simplex(an_half, S)

        min_obj = m
        min_cand = a

        for cand in (cand_old, cand_half):
            if cand is None:
                continue
            ob = cmax_of(cand, N)
            if ob < min_obj:
                min_obj = ob
                min_cand = cand
            # Blends with the current point often give smoother progress.
            for mix in (0.5, 0.25):
                bl = (1.0 - mix) * a + mix * cand
                bl = safe_normalize(bl, n, S)
                obb = cmax_of(bl, N)
                if obb < min_obj:
                    min_obj = obb
                    min_cand = bl

        if min_cand is not a and (min_obj < m - 1e-13 or min_obj - m < temp):
            a = min_cand
            if min_obj < best_m:
                best_m = min_obj
                best_a = a.copy()
                last_improve = it
            f = min(0.25, f * 1.05)
        else:
            f = max(0.01, f * 0.85)
            temp = max(1e-6, temp * 0.99)
            if f <= 0.01 and it % 100 == 0:
                std = 0.005 * S * inv_sqrt_n
                a = safe_normalize(a + rng.normal(0.0, std, n), n, S)
                f = 0.03

    return best_a, best_m


def lbfgs_lse_block(a_init, n, S, N, budget_s, rng, beta=180.0, m_hist=12):
    """
    L-BFGS on the LSE-smoothed objective with exact max line search.
    Useful as a polishing local method.
    """
    a = safe_normalize(a_init, n, S)
    best_m = cmax_of(a, N)
    best_a = a.copy()
    end = time.monotonic() + max(0.0, float(budget_s))
    s_hist = []
    y_hist = []
    alpha = 0.1
    beta_curr = float(beta)
    it = 0
    no_improve = 0
    inv_sqrt_n = 1.0 / np.sqrt(float(n))

    while time.monotonic() < end:
        it += 1
        g, _, c, m = lse_grad(a, N, beta_curr)
        if m < best_m - 1e-12:
            best_m = m
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            std = 0.01 * S * inv_sqrt_n
            a = safe_normalize(a + rng.normal(0.0, std, n), n, S)
            s_hist = []
            y_hist = []
            alpha = 0.1
            no_improve = 0
            continue

        d = lbfgs_dir(s_hist, y_hist, g)
        dn = float(np.linalg.norm(d))
        if dn < 1e-14:
            std = 0.005 * S * inv_sqrt_n
            a = safe_normalize(a + rng.normal(0.0, std, n), n, S)
            s_hist = []
            y_hist = []
            alpha = 0.1
            continue
        d = d / dn

        min_obj = m
        min_cand = None
        min_st = alpha

        for sign in (1.0, -1.0):
            for st in (alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125):
                cand = project_simplex(a + sign * st * d, S)
                ob = cmax_of(cand, N)
                if ob < min_obj:
                    min_obj = ob
                    min_cand = cand
                    min_st = st

        if min_cand is not None and min_obj < m * (1.0 - 1e-12):
            a_new = min_cand
            g_new, _, _, m_new = lse_grad(a_new, N, beta_curr)
            sk = a_new - a
            yk = g_new - g
            ss = float(np.dot(sk, sk))
            sy = float(np.dot(sk, yk))
            if ss > 1e-24 and sy > 1e-12 * ss:
                s_hist.append(sk)
                y_hist.append(yk)
                if len(s_hist) > m_hist:
                    s_hist = s_hist[-m_hist:]
                    y_hist = y_hist[-m_hist:]
            else:
                s_hist = []
                y_hist = []

            a = a_new
            alpha = max(1e-4, float(min_st))
            if m_new < best_m - 1e-12:
                best_m = m_new
                best_a = a.copy()
                no_improve = 0
        else:
            alpha = max(1e-4, alpha * 0.5)
            if alpha < 1e-3 and it % 200 == 0:
                std = 0.005 * S * inv_sqrt_n
                a = safe_normalize(a + rng.normal(0.0, std, n), n, S)
                s_hist = []
                y_hist = []
                alpha = 0.1

    return best_a, best_m


def explore_round(a_init, n, S, N, budget_s, rng, beta=50.0, eta=0.8):
    """
    Short multi-start exploration around a current point using mirror descent.
    """
    a0 = safe_normalize(a_init, n, S)
    best_m = cmax_of(a0, N)
    best_a = a0.copy()
    if budget_s <= 0.0:
        return best_a, best_m

    end = time.monotonic() + float(budget_s)
    inv_sqrt_n = 1.0 / np.sqrt(float(n))
    configs = [
        (0.004, beta),
        (0.008, beta * 1.2),
        (0.016, beta * 1.5),
    ]
    per = max(0.0, (end - time.monotonic()) / float(len(configs)))

    for stdf, bet in configs:
        if time.monotonic() >= end:
            break
        std = stdf * S * inv_sqrt_n
        start = a0 + rng.normal(0.0, std, n) + 0.2 * rng.random(n) * std
        start = safe_normalize(start, n, S)
        a_rel, m_rel = mirror_lse_block(start, n, S, N, per, rng, beta=bet, eta=eta)
        if m_rel < best_m:
            best_m = m_rel
            best_a = a_rel.copy()
        # Continue next perturbation from the best point found so far.
        a0 = best_a.copy()

    return best_a, best_m


def propose_candidate(seed=42, budget_s=2400.0):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.

    This function is invoked by the harness.  It returns a Python list of
    non-negative floats.  It uses the length of height_sequence_1 when that
    length is reasonable; otherwise it defaults to 8192.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0

    rng = np.random.default_rng(seed)

    arr = get_initial_global()
    if arr is not None and 256 <= arr.size <= 65536:
        n = int(arr.size)
        base = arr
    else:
        n = 8192
        base = None

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    if base is None or base.size != n:
        base = rng.random(n) + 0.05

    current = safe_normalize(base, n, S)
    best_a = current.copy()
    best_m = cmax_of(current, N)

    print("n=%d, N(FFT)=%d, S=%.6f, budget=%.1fs" % (n, N, S, budget))
    print("initial internal objective (== evaluate if sum^2=2n): %.6f" % best_m)

    # Leave a small reserve for final cleanup/return.
    reserve = min(60.0, max(1.0, 0.05 * budget))
    work = max(1.0, budget - reserve)
    deadline = time.monotonic() + work

    # Phase 1: moderate-temperature mirror descent from the given point.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        b = min(0.25 * work, rem - 10.0)
        if b > 5.0:
            print("=== Phase 1: mirror LSE beta ramps 20->500 ===")
            a_rel, m_rel = mirror_lse_block(current, n, S, N, b, rng, beta=20.0, eta=1.0)
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase1 best objective = %.6f" % best_m)

    target_m = max(1.44, best_m - 0.0025)

    # Phase 2: spectral alternating projection.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        b = min(0.12 * work, rem - 10.0)
        if b > 5.0:
            print("=== Phase 2: spectral shaving / square-root projection ===")
            a_rel, m_rel = spectral_shave_block(
                current, n, S, N, b, rng, frac=0.06, target_m=target_m
            )
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase2 best objective = %.6f" % best_m)

    # Phase 3: short exploratory mirror runs from noisy perturbations.
    rem = deadline - time.monotonic()
    if rem > 30.0:
        b = min(0.12 * work, rem - 10.0)
        if b > 10.0:
            print("=== Phase 3: exploratory mirror basin searches ===")
            a_rel, m_rel = explore_round(current, n, S, N, b, rng, beta=50.0, eta=0.8)
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase3 best objective = %.6f" % best_m)

    # Phase 4: higher-temperature mirror polishing.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        b = min(0.20 * work, rem - 10.0)
        if b > 5.0:
            print("=== Phase 4: mirror LSE high beta ===")
            a_rel, m_rel = mirror_lse_block(current, n, S, N, b, rng, beta=80.0, eta=0.8)
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase4 best objective = %.6f" % best_m)

    # Phase 5: L-BFGS polishing.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        b = min(0.12 * work, rem - 10.0)
        if b > 5.0:
            print("=== Phase 5: L-BFGS LSE polish ===")
            a_rel, m_rel = lbfgs_lse_block(current, n, S, N, b, rng, beta=180.0, m_hist=12)
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase5 best objective = %.6f" % best_m)

    # Phase 6: second spectral pass with updated target.
    target_m = max(1.44, best_m - 0.0025)
    rem = deadline - time.monotonic()
    if rem > 20.0:
        b = min(0.08 * work, rem - 10.0)
        if b > 5.0:
            print("=== Phase 6: spectral fine spectral shaping ===")
            a_rel, m_rel = spectral_shave_block(
                current, n, S, N, b, rng, frac=0.04, target_m=target_m
            )
            if m_rel < best_m:
                best_m = m_rel
                best_a = a_rel.copy()
            current = a_rel.copy() if m_rel < best_m + 0.005 else best_a.copy()
            print("phase6 best objective = %.6f" % best_m)

    # Final phase: remaining time to high-beta mirror descent.
    rem = deadline - time.monotonic()
    if rem > 10.0:
        b = max(5.0, rem - 5.0)
        print("=== Phase 7: final high-beta mirror polish ===")
        a_rel, m_rel = mirror_lse_block(current, n, S, N, b, rng, beta=250.0, eta=0.5)
        if m_rel < best_m:
            best_m = m_rel
            best_a = a_rel.copy()
        print("final internal objective = %.6f" % best_m)

    final_a = safe_normalize(best_a, n, S)
    out = [float(x) for x in final_a]
    print("returning length %d, final internal objective %.6f" % (len(out), best_m))
    return out
