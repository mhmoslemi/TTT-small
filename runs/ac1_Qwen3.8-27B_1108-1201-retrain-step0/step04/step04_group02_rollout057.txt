def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf

            best_val = val

            best_delta = delta

**Alternative: Smoothed objective optimization**

Instead of coordinate descent on the non-smooth max, I can optimize a smooth approximation:
f_p(a) = 2n * (sum_k c[k]^p)^(1/p) / sum(a)^2

For large p, this approximates the max. The gradient is:
∂f_p/∂a[i] = 2n * (1/p) * (sum c^p)^(1/p - 1) * p * sum_k c[k]^{p-1} * ∂c[k]/∂a[i] / sum(a)^2
             - 2n * (sum c^p)^(1/p) * 2*sum(a) / sum(a)^4 * 1
             = 2n * (sum c^p)^{1/p - 1} * sum_k c[k]^{p-1} * 2*a[k-i] / sum(a)^2
             - 4n * (sum c^p)^{1/p} / sum(a)^3

The first term can be computed efficiently using convolution:
sum_k c[k]^{p-1} * a[k-i] = (c^{p-1} * a)[i] (cross-correlation)

Wait, more precisely: sum_k c[k]^{p-1} * a[k-i] is the cross-correlation of c^{p-1} and a at lag i. This can be computed via FFT.

So the gradient computation is:
1. Compute c = a*a (FFT, O(n log n))
2. Compute c^{p-1} element-wise (O(n))
3. Compute g = cross_correlation(c^{p-1}, a) (FFT, O(n log n))
4. Compute the scalar terms (O(1))

Total: O(n log n) per gradient evaluation. With a good optimizer (e.g., L-BFGS or Adam), this could be very effective.

But there's a catch: the p-norm is only a good approximation of the max when p is large, and for large p, the gradient becomes concentrated at the peaks (similar to the subgradient of max). So the smoothness benefit is lost.

A better approach might be to use a moderate p (e.g., p=4 or p=8) and optimize that, then switch to the exact max at the end.

**Another alternative: Direct optimization of the peak**

The key insight is that the objective is determined by the maximum of the convolution. So I should focus on reducing the peaks.

Algorithm:
1. Find all peaks (local maxima) of c
2. For each peak at position k, the value is c[k] = sum_i a[i]*a[k-i]
3. To reduce c[k], I need to reduce some of the products a[i]*a[k-i]
4. The most effective way is to reduce the largest a[i]*a[k-i] pairs

1. Find the global maximum of c at position k*
2. Identify the pairs (i, k*-i) that contribute most to c[k*]
3. Reduce a[i] for the i's in those pairs
4. Repeat

This is essentially what the subgradient method does, but I can make it more targeted.

```python
"""
Minimizes 2*n*max(self-convolution) / (sum)^2 over non-negative sequences.
Uses coordinate descent with active-set 1D subproblem solving,
combined with periodic FFT re-computation and multi-start strategies.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def solve_1d_subproblem(C, a, i, x_old, S, n):
    """
    Solve the 1D subproblem: find x_new >= 0 that minimizes
    2n * max(C_new) / (S_new)^2
    where C_new is the convolution after changing a[i] from x_old to x_new.
    
    Returns the optimal x_new.
    """
    lo = i
    hi = min(i + n - 1, 2 * n - 2)
    
    # C_out: max of C outside [lo, hi]
    C_out = 0.0
    if lo > 0:
        C_out = max(C_out, C[:lo].max())
    if hi < 2 * n - 2:
        C_out = max(C_out, C[hi + 1:].max())
    
    k_range = np.arange(lo, hi + 1)
    a_vals = a[k_range - i]
    C_seg = C[k_range].copy()
    
    has_mid = (lo <= 2 * i <= hi)
    mid_idx = 2 * i - lo if has_mid else -1
    
    best_x = x_old  # Default: no change
    best_val = np.inf
    
    def eval_at_delta(delta):
        """Evaluate the (partial) objective at a given delta = x_new - x_old."""
        S_new = S + delta
        if S_new <= 1e-15:
            return np.inf
        C_in = C_seg + 2.0 * delta * a_vals
        if has_mid:
            C_in[mid_idx] = C[2 * i] + delta * delta
        max_conv = max(C_out, C_in.max())
        if max_conv < 0:
            max_conv = 0
        return max_conv / (S_new * S_new)
    
    # Try delta = 0
    val0 = eval_at_delta(0.0)
    best_val = val0
    best_x = x_old
    
    # Fixed-point iteration
    delta = 0.0
    for it in range(15):
        # Find argmax at current delta
        S_new = S + delta
        if S_new <= 1e-15:
            break
        
        C_in = C_seg + 2.0 * delta * a_vals
        if has_mid:
            C_in[mid_idx] = C[2 * i] + delta * delta
        
        max_in = C_in.max()
        
        if C_out > max_in:
            # C_out dominates: we want to increase delta
            # Find the smallest delta where some in-range k ties C_out
            # C_seg[j] + 2*delta*a_vals[j] = C_out
            # delta = (C_out - C_seg[j]) / (2*a_vals[j])
            pos = a_vals > 1e-15
            if np.any(pos):
                deltas = (C_out - C_seg[pos]) / (2.0 * a_vals[pos])
                # We want the smallest delta > current delta
                valid = deltas > delta + 1e-12
                if np.any(valid):
                    delta_new = deltas[valid].min()
                else:
                    # All in-range k already exceed C_out at current delta
                    # (shouldn't happen if C_out > max_in, but handle gracefully)
                    delta_new = delta + 0.01 * S
            else:
                delta_new = delta + 0.01 * S
            
            delta_new = min(delta_new, 3.0 * S)
            delta_new = max(delta_new, -x_old)
            
            # Evaluate
            val = eval_at_delta(delta_new)
            if val < best_val - 1e-15:
                best_val = val
                best_x = x_old + delta_new
            
            if abs(delta_new - delta) < 1e-10:
                break
            delta = delta_new
        else:
            # In-range k dominates
            k_star_idx = np.argmax(C_in)
            k_star = k_range[k_star_idx]
            
            if k_star == 2 * i:
                delta_new = C[2 * i] / S if S > 1e-15 else 0.0
            else:
                a_k = a[k_star - i]
                if a_k > 1e-15:
                    delta_new = S - C[k_star] / a_k
                else:
                    delta_new = 0.0
            
            delta_new = max(delta_new, -x_old)
            delta_new = min(delta_new, 3.0 * S)
            
            val = eval_at_delta(delta_new)
            if val < best_val - 1e-15:
                best_val = val
                best_x = x_old + delta_new
            
            if abs(delta_new - delta) < 1e-10:
                delta = delta_new
                # One more evaluation at the converged point
                val = eval_at_delta(delta)
                if val < best_val:
                    best_val = val
                    best_x = x_old + delta
                break
            delta = delta_new
    
    # Also check boundary: x_new = 0
    if x_old > 0:
        val_boundary = eval_at_delta(-x_old)
        if val_boundary < best_val:
            best_x = 0.0
            best_val = val_boundary
    
    return max(0.0, best_x)


def coordinate_descent(a_init, max_time, rng, verbose=True, recompute_every=None):
    """
    Coordinate descent for minimizing 2n*max(C)/S^2.
    """
    a = np.array(a_init, dtype=np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    
    if S < 1e-10:
        a = np.ones(n) * 0.1
        S = a.sum()
    
    # Initial convolution
    C = fft_convolve_self(a)
    
    best_obj = 2.0 * n * C.max() / S ** 2
    best_a = a.copy()
    
    if recompute_every is None:
        recompute_every = max(1, n // 4)
    
    start = time.monotonic()
    sweep = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        sweep += 1
        S = a.sum()
        
        if S < 1e-10:
            break
        
        order = rng.permutation(n)
        
        for count, i in enumerate(order):
            if time.monotonic() - start > max_time:
                break
            
            x_old = a[i]
            x_new = solve_1d_subproblem(C, a, i, x_old, S, n)
            
            delta = x_new - x_old
            if abs(delta) < 1e-15:
                continue
            
            a[i] = x_new
            S += delta
            
            # Update C incrementally
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            k_range = np.arange(lo, hi + 1)
            a_vals = a[k_range - i]  # a[k-i] for k in range (a[i] is now x_new)
            
            # For k ≠ 2i: C[k] += 2 * delta * a[k-i] (a[k-i] is unchanged since k-i ≠ i)
            # For k = 2i: C[2i] += delta * (2*x_old + delta)
            
            # Vectorized update
            C[k_range] += 2.0 * delta * a_vals
            
            # Correct for k = 2i (we added 2*delta*a[2i-i] = 2*delta*x_new, 
            # but the correct update is delta*(2*x_old + delta))
            if lo <= 2 * i <= hi:
                # We added 2*delta*x_new, correct is delta*(2*x_old + delta)
                # Difference: delta*(2*x_old + delta) - 2*delta*x_new = delta*(2*x_old + delta - 2*x_new)
                # = delta*(2*x_old + delta - 2*(x_old + delta)) = delta*(2*x_old + delta - 2*x_old - 2*delta)
                # = delta*(-delta) = -delta^2
                C[2 * i] -= delta * delta
            
            # Periodic recompute
            if (count + 1) % recompute_every == 0:
                C = fft_convolve_self(a)
        
        # End of sweep: recompute for accuracy
        C = fft_convolve_self(a)
        S = a.sum()
        obj = 2.0 * n * C.max() / S ** 2
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_improve = sweep
        else:
            stall_count += 1
        
        if verbose and (sweep % 50 == 0 or sweep <= 5):
            elapsed = time.monotonic() - start
            print(f"  CD sweep {sweep}: obj={obj:.8f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")
        
        # If stalled, try a perturbation
        if stall_count > 20 and rng.random() < 0.3:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = np.maximum(a + rng.normal(0, noise_scale, n), 0.0)
            C = fft_convolve_self(a)
            S = a.sum()
            stall_count = 0
    
    return best_a, best_obj


def peak_shaving(a_init, max_time, rng, verbose=True):
    """
    Peak shaving: iteratively identify the peak of the convolution and
    reduce the coordinates that contribute most to it.
    """
    a = np.array(a_init, dtype=np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    
    if S < 1e-10:
        a = np.ones(n) * 0.1
        S = a.sum()
    
    C = fft_convolve_self(a)
    best_obj = 2.0 * n * C.max() / S ** 2
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        S = a.sum()
        if S < 1e-10:
            break
        
        # Find the peak
        k_star = int(np.argmax(C))
        c_max = C[k_star]
        
        # Find the pairs (i, k_star - i) that contribute to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        i_vals = np.arange(lo, hi + 1)
        pairs = a[i_vals] * a[k_star - i_vals]
        
        # Find the largest contributors
        n_top = min(10, len(i_vals))
        top_idx = np.argsort(pairs)[::-1][:n_top]
        
        # Reduce the top contributors
        for idx in top_idx:
            i = i_vals[idx]
            other = k_star - i
            
            # Reduce a[i] by a small amount
            reduction = 0.01 * a[i]
            a[i] = max(0.0, a[i] - reduction)
            
            # Update C
            delta = -reduction
            S -= reduction
            lo_k = i
            hi_k = min(i + n - 1, 2 * n - 2)
            kr = np.arange(lo_k, hi_k + 1)
            av = a[kr - i]
            C[kr] += 2.0 * delta * av
            if lo_k <= 2 * i <= hi_k:
                C[2 * i] -= delta * delta
        
        # Also boost the edges slightly (to compensate for sum loss)
        edge_boost = reduction * 2.0 / n
        a[0] += edge_boost
        a[-1] += edge_boost
        S += 2 * edge_boost
        
        # Update C for edge changes
        # Too complex to do incrementally for two changes; just recompute periodically
        
        if it % 50 == 0:
            C = fft_convolve_self(a)
        
        S = a.sum()
        obj = 2.0 * n * C.max() / S ** 2
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"  PS iter {it}: obj={obj:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj


def pnorm_descent(a_init, max_time, rng, p=8, verbose=True):

    best_obj = 2.0 * n * C.max() / S ** 2
    best_a = a.copy()

    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        if S < 1e-10:

        # Compute gradient of (sum c^p)^(1/p) / S^2
        # ∂/∂a[i] of (sum c^p)^(1/p) = (1/p) * (sum c^p)^(1/p-1) * p * sum_k c[k]^(p-1) * 2*a[k-i]
        # = (sum c^p)^(1/p-1) * sum_k c[k]^(p-1) * 2*a[k-i]
        # = (sum c^p)^(1/p-1) * 2 * (c^(p-1) * a)[i]  [cross-correlation]
        
        C_pos = np.maximum(C, 0)  # c should be non-negative but clip for safety
        c_pow = C_pos ** (p - 1)
        
        # Cross-correlation: sum_k c_pow[k] * a[k-i]
        # = (c_pow * a)[i] where * is linear cross-correlation
        # Actually: sum_k c_pow[k] * a[k-i] = convolution of c_pow and a at specific lag
        # Let's use FFT: ifft(fft(c_pow) * conj(fft(a))) gives cross-correlation
        
        fsize2 = 1 << (len(c_pow) + n - 1).bit_length()
        CP = fft(c_pow, fsize2)
        AP = fft(a, fsize2)
        cross = np.real(ifft(CP * np.conj(AP)))[:n]
        
        sum_cp = np.sum(C_pos ** p)
        if sum_cp < 1e-30:
            break
        
        # gradient of f_p = 2n * (sum c^p)^(1/p) / S^2
        # = 2n * [ (1/p) * (sum c^p)^(1/p - 1) * p * cross * 2 / S^2 - (sum c^p)^(1/p) * 2 * a / S^3 ]
        # = 2n * [ 2 * cross * (sum c^p)^(1/p - 1) / S^2 - 2 * a * (sum c^p)^(1/p) / S^3 ]
        
        base = sum_cp ** (1.0 / p)
        grad = 2.0 * cross * (sum_cp ** (1.0 / p - 1.0)) / (S * S) - 2.0 * a * base / (S * S * S)
        grad = 2.0 * n * grad
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Check if it improved
        C_new = fft_convolve_self(a_new)
        S_new = a_new.sum()
        if S_new < 1e-10:
            lr *= 0.5
            continue
        
        obj_new = 2.0 * n * C_new.max() / (S_new * S_new)
        
        if obj_new < 2.0 * n * C.max() / (S * S):
            a = a_new
            C = C_new
            S = S_new
            lr = min(lr * 1.05, 10.0 / n)
        else:
            lr *= 0.7
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  pNorm iter {it}: obj={obj_new:.6f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj
