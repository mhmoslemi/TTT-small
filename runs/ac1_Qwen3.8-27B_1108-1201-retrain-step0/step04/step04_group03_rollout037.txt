def fft_convolve_full(f, g):
    """Compute full convolution of f and g using FFT."""
    n1, n2 = len(f), len(g)
    size = n1 + n2 - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    return np.real(ifft(F * G))[:size]

def gaussian_cancellation(f, S, n, max_iter=10000, rng=None):
    """
    Gaussian cancellation algorithm for reducing max autocorrelation.
    """
    if rng is None:
        rng = np.random.default_rng()

    f = np.maximum(f, 0)
    f = f * (S / f.sum())

    best_max = fft_convolve_self(f).max()
    
    for it in range(max_iter):

        k_star = np.argmax(c)
        M = c[k_star]
