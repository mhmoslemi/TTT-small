import numpy as np
import time
from scipy.fft import fft, ifft

N_TOPK = 10  # Number of top peaks to focus on

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def topk_lse_gradient(a, c, beta, top_k):
    """Compute gradient of top-K LSE approximation."""
    n = len(a)
    size = 2 * n - 1
    
    # Get top K indices
    if top_k < size:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(size)
    
    # Compute weights (softmax over top K)
    c_top = c[top_idx]
    cm = c_top.max()
    e = np.exp(beta * (c_top - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # Compute gradient using FFT
    # grad[j] = 2 * sum_k w_k * a[k-j] (for valid k-j)
    W = np.zeros(size)
    for i, k in enumerate(top_idx):
        W[k] = 2.0 * w[i]
    
    # Cross-correlation: grad[j] = sum_k W[k] * a[k-j]
    # = (W ⋆ a)[j] where ⋆ is cross-correlation
    # Using FFT: cross_corr = ifft(conj(fft(W_pad)) * fft(a_pad))
    fsize = 1 << (size + n - 1).bit_length()
    W_pad = np.zeros(fsize)
    W_pad[:size] = W
    a_pad = np.zeros(fsize)
    a_pad[:n] = a
    
    # r[m] = sum_k W[k] * a[k+m] (standard cross-correlation)
    r = np.real(ifft(np.conj(fft(W_pad)) * fft(a_pad)))
    
    # We want grad[j] = sum_k W[k] * a[k-j] = r[-j]
    # r[m] for m = -j means r[fsize - j] (circular)
    # For j = 0: grad[0] = r[0]
    # For j = 1: grad[1] = r[fsize-1]
    # For j = 2: grad[2] = r[fsize-2]
    # ...
    grad = np.zeros(n)
    grad[0] = r[0]
    if n > 1:
        grad[1:] = r[fsize-1 : fsize-n : -1]
    
    return grad

def topk_lse_value(c, beta, top_k):
    """Compute top-K LSE value."""
    size = len(c)
    if top_k < size:
        c_top = np.partition(c, -top_k)[-top_k:]
    else:
        c_top = c
    cm = c_top.max()
    return cm + np.log(c_top.size) / beta  # approximation
    # More accurately: cm + log(mean(exp(beta*(c_top - cm))))/beta

def run_fista_topk(a_init, n, S, max_time, rng, top_k=10, beta_init=20.0, beta_max=3000.0):
    """FISTA with top-K LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    c = fft_convolve_self(y)
    best = c.max()
    
    while time.monotonic() - start < max_time:
        it += 1

        progress = min(1.0, it / 5000.0)
        beta = beta_init + (beta_max - beta_init) * (1 - np.cos(np.pi * progress)) / 2
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:

            best_a = y.copy()
            last_improve = it

        grad = topk_lse_gradient(y, c, beta, top_k)
        gnorm = np.linalg.norm(grad)

        # But with top-K, it's effectively smaller

            noise_scale = 0.001 * S / np.sqrt(n) * decay

                t = 1.0
        
        if it % 200 == 0:
            elapsed = time.monotonic() - start
            print("    FISTA-topK: it=%d, beta=%.0f, best=%.6f, elapsed=%.0fs" % (it, beta, best, elapsed))
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:

        best_a = a.copy()
    return best_a, best

    a = normalize_sum(a_init, n)
    start = time.monotonic()

    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stall = 0

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:

                noise_scale = 0.005 * S / np.sqrt(n) * (1.0 / (1.0 + stall/50))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                stall = 0

        size = 2 * n - 1
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        med = np.median(c)
        weights = np.maximum(c[top_idx] - med, 0.0)
        if weights.sum() < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / weights.sum()

        sens = np.zeros(n)

        # For simplicity, use the linear approximation and verify

                best_d = d
        
        if best_d > 0:
            a[j_idx] += best_d
            a[m_idx] -= best_d
            # Renormalize (shouldn't be needed since we're transferring)
        else:
            # No improvement from this pair, try noise
            stall += 1
    
    return best_a, best
