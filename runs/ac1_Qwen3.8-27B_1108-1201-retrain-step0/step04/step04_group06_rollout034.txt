```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= S:
        # Need to add mass - just distribute
        # Actually, the projection handles this
        pass
    # Standard simplex projection
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n+1)
    theta = (cumsum - S) / rho
    cond = mu > theta
    rho_star = np.max(cond.nonzero()[0]) + 1
    theta = (cumsum[rho_star-1] - S) / rho_star
    return np.maximum(v - theta, 0.0)

def autocorr(a):
    """Compute full autocorrelation (self-convolution) of a."""
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def topk_gradient(a, b, k, beta):
    """
    Compute gradient of top-k soft-minimax.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Get top-k lags
    top_idx = np.argsort(b)[::-1][:k]
    top_vals = b[top_idx]
    
    # Softmax weights over top-k
    max_v = top_vals.max()
    weights = np.exp(beta * (top_vals - max_v))
    weights /= weights.sum()
    
    # Build sparse p vector
    p = np.zeros(2*n-1)
    p[top_idx] = weights
    
    # Gradient: 2 * cross_correlate(a, p)
    # r[m] = sum_j a[j] * p[m+j] for j=0..n-1
    # = ifft(conj(fft(a)) * fft(p))[m]
    size = 1 << (2*n - 1).bit_length()
    A = fft(a, size)
    P = fft(p, size)
    r = np.real(ifft(np.conj(A) * P))[:n]
    
    return 2.0 * r

def soft_minimize(a_init, n, S, max_time, beta_init, beta_final, k_init, k_final, rng):
    """
    Minimize top-k soft-max of autocorrelation using projected gradient with Nesterov.
    """
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = np.inf
    start = time.monotonic()
    
    y = a.copy()  # Nesterov momentum
    t = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Animate parameters
        frac = min(1.0, it / (max_time * 10))  # rough progress
        beta = beta_init + (beta_final - beta_init) * frac
        k = max(k_init, int(k_init + (k_final - k_init) * frac))
        
        b = autocorr(y)
        cur_max = b.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Compute gradient
        grad = topk_gradient(y, b, k, beta)
        
        # Step size (use Lipschitz estimate or backtracking)
        # L ~ 2 * S * beta (rough estimate)
        L = 2.0 * S * min(beta, 50.0) + 2.0 * S
        eta = 1.0 / L
        
        # Grad descent step
        x_new = y - eta * grad
        x_new = project_simplex(x_new, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Periodic check
        if it % 100 == 0:
            b_check = autocorr(y)
            if b_check.max() < best_score:
                best_score = b_check.max()
                best_a = y.copy()
    
    return best_a, best_score
```

"""
Minimize 2*n*max_conv/sum^2 using adaptive top-k subgradient method with
smart coordinate descent. Key innovations:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / rho
    cond = mu > theta
    if not np.any(cond):
        return v - theta[-1]  # all get reduced
    rho_star = int(np.max(np.where(cond)[0]) + 1)
    theta_star = (cumsum[rho_star - 1] - S) / rho_star
    return np.maximum(v - theta_star, 0.0)


def autocorr_fft(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def topk_subgradient(a, b, k):
    """
    Compute subgradient of the top-k mean of b = conv(a,a).
    Returns gradient vector of length n.
    Uses FFT for efficiency: grad = (2/k) * ifft(conj(fft(a)) * fft(p))[:n]
    where p is uniform over top-k lags.
    """
    n = len(a)
    top_idx = np.argsort(b)[::-1][:k]
    
    # Build p (uniform over top-k lags)
    p = np.zeros(2 * n - 1)
    p[top_idx] = 1.0 / k
    
    # grad = (2/k) * cross_correlate(a, p)
    # cross_correlate(a, p)[m] = sum_j a[j] * p[m+j]
    # = ifft(conj(fft(a)) * fft(p))[m]
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    P = fft(p, size)
    r = np.real(ifft(np.conj(A) * P))[:n]
    
    return 2.0 * r


def soft_topk_subgradient(a, b, k, beta):
    """
    Compute gradient of the soft top-k objective:
    f(a) = (1/beta) * log(sum_{j in top_k} exp(beta * b[j]))
    (approximately, focusing on top-k)
    """
    n = len(a)
    top_idx = np.argsort(b)[::-1][:max(k, 10)]
    top_vals = b[top_idx]
    
    # Softmax weights
    max_v = top_vals.max()
    weights = np.exp(beta * (top_vals - max_v))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(top_vals)) / len(top_vals)
    else:
        weights /= wsum
    
    # Build p
    p = np.zeros(2 * n - 1)
    p[top_idx] = weights
    
    # Gradient: 2 * cross_correlate(a, p)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    P = fft(p, size)
    r = np.real(ifft(np.conj(A) * P))[:n]
    
    return 2.0 * r


def subgradient_descent(a_init, n, S, max_time, rng, k_init=5, k_max=200, 
                        beta_init=1.0, beta_max=20.0):
    """
    Adaptive subgradient method with Nesterov acceleration.
    """
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = autocorr_fft(a).max()
    start = time.monotonic()
    
    y = a.copy()  # Nesterov
    t = 1.0
    it = 0
    stall = 0
    last_best = best_score
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive parameters
        progress = min(1.0, 2.0 * (time.monotonic() - start) / max_time)
        k = min(k_max, k_init + int(progress * (k_max - k_init)))
        beta = beta_init + (beta_max - beta_init) * progress
        
        b = autocorr_fft(y)
        cur = b.max()
        
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute subgradient
        if rng.random() < 0.3:
            grad = soft_topk_subgradient(y, b, k, beta)
        else:
            grad = topk_subgradient(y, b, k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking
        eta = 2.0 * S / (n * (1.0 + 0.01 * it))
        
        # Try a few step sizes
        best_x = y.copy()
        best_val = cur
        for factor in [1.0, 0.5, 0.25, 0.1]:
            x_candidate = project_simplex(y - factor * eta * grad, S)
            b_check = autocorr_fft(x_candidate)
            val = b_check.max()
            if val < best_val:
                best_val = val
                best_x = x_candidate
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = best_x + ((t - 1.0) / t_new) * (best_x - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum
        y_sum = y.sum()
        if y_sum > S * 1.01 or y_sum < S * 0.99:
            y = project_simplex(y, S)
        a = best_x
        t = t_new
        
        # Periodic perturbation
        if it % 200 == 0 and stall > 50:
            n_perturb = max(10, n // 100)
            idx = rng.choice(n, size=n_perturb, replace=False)
            y[idx] *= rng.uniform(0.5, 1.5, n_perturb)
            y = project_simplex(y, S)
            t = 1.0
            stall = 0
        
        # Early stopping
        if stall > 500:
            break
    
    return best_a, best_score


def coordinate_descent(a_init, n, S, max_time, rng, n_top=20):

    For each top-contributing coordinate at the peak, try scaling it
    and redistributing mass to "cheap" coordinates.
    """
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = autocorr_fft(a).max()
    start = time.monotonic()
    
    scale_factors = [0.3, 0.5, 0.7, 0.8, 1.2, 1.5]
    
    while time.monotonic() - start < max_time:
        b = autocorr_fft(a)
        c_max = b.max()
        peak_lag = int(np.argmax(b))
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Find coordinates contributing to the peak
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_coords:
            if time.monotonic() - start > max_time:
                break
            i = int(idx)
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue
            
            orig = a[i]
            if orig < 1e-10:
                continue
            
            best_frac = 1.0
            best_local = c_max
            
            for frac in scale_factors:
                new_val = orig * frac
                delta = new_val - orig
                if abs(delta) < 1e-15:
                    continue
                
                # Estimate change in b at peak_lag
                # b[peak_lag] changes by: delta * a[j] + (symmetric term)
                # If i != j: change = delta * a[j] + delta * a[i] ... 
                # Actually: b[peak] includes a[i]*a[j] and a[j]*a[i] (if i≠j)
                # So change = delta*a[j] + delta*a[i_old]...

                #  the term a[j]*a[i] appears once as a[j] contributes...

                #  and the term with i'=j is a[j]*a[i]. So both are present.

                else:
                    est_change = 2.0 * delta * a[j]

            actual_max = actual_b.max()
            if actual_max > c_max * 1.05:

def spectral_perturb(a, n, N, S, sigma, rng):
    """Add random complex noise to FFT, project back to simplex."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

```python
"""
Minimize 2*n*max_conv/sum^2 using:
1. Multi-resolution spectral shaping with lag-cycling
2. Top-k subgradient with Nesterov acceleration (direct time-domain)
3. Partner-aware coordinate descent with mass redistribution
4. Multi-start with diverse structured initializations and n-sweep
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64).copy()
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / rho
    cond = mu > theta
    if not np.any(cond):
        return np.maximum(v - theta[-1], 0.0)
    rho_star = int(np.max(np.where(cond)[0]) + 1)
    theta_star = (cumsum[rho_star - 1] - S) / rho_star
    return np.maximum(v - theta_star, 0.0)


def autocorr_fft(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def topk_subgradient(a, b, k):
    """Subgradient of the mean of top-k autocorrelation values."""
    n = len(a)
    top_idx = np.argsort(b)[::-1][:k]
    p = np.zeros(2 * n - 1)
    p[top_idx] = 1.0 / k
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    P = fft(p, size)
    r = np.real(ifft(np.conj(A) * P))[:n]
    return 2.0 * r


def soft_topk_grad(a, b, k, beta):
    """Gradient of log-sum-exp over top-k lags (smooth surrogate)."""
    n = len(a)
    kk = max(k, 10)
    top_idx = np.argsort(b)[::-1][:kk]
    top_vals = b[top_idx]
    max_v = top_vals.max()
    weights = np.exp(beta * (top_vals - max_v))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(top_vals)) / len(top_vals)
    else:
        weights /= wsum
    p = np.zeros(2 * n - 1)
    p[top_idx] = weights
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    P = fft(p, size)
    r = np.real(ifft(np.conj(A) * P))[:n]
    return 2.0 * r


def spectral_shape_step(a, N, n, top_k, alpha, rng, target):
    """One spectral water-filling step with exponential shaping and lag cycling."""
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    if c_max <= target:
        return a.copy(), c_max
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy(), c_max
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total
    weights = excess / excess_sum
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    alpha_eff = alpha * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.3)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))
    if rng is not None and rng.random() < 0.25:
        phase_noise = rng.normal(0, 0.06 * alpha_eff, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def spectral_loop(a_init, n, N, max_time, rng, target=1.48, top_k_init=5, alpha_init=0.5):
    """Iterative spectral water-filling with adaptive parameters."""
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    best_a = a.copy()
    best_cmax = autocorr_fft(a).max()
    it = 0
    stall = 0
    while time.monotonic() < max_time:
        it += 1
        top_k = min(80, top_k_init + it // 15)
        alpha = alpha_init / (1.0 + 0.03 * it)
        a_new, c_max = spectral_shape_step(a, N, n, top_k, alpha, rng, target)
        actual_cmax = autocorr_fft(a_new).max()
        if actual_cmax < best_cmax:
            best_cmax = actual_cmax
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            a = a_new.copy()
            stall += 1
            if stall > 80:
                a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
                stall = 0
        if best_cmax <= target:
            break
        if it % 300 == 150 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)
    return best_a, best_cmax


def subgradient_loop(a_init, n, S, max_time, rng, k_init=5, k_max=300,
                     beta_init=1.0, beta_max=30.0):
    """Adaptive top-k subgradient with Nesterov acceleration."""
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    best_a = a.copy()
    best_score = autocorr_fft(a).max()
    start = time.monotonic()
    y = a.copy()
    t = 1.0
    it = 0
    stall = 0
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, 2.0 * (time.monotonic() - start) / max_time)
        k = max(3, min(k_max, k_init + int(progress * (k_max - k_init))))
        beta = beta_init + (beta_max - beta_init) * progress
        b = autocorr_fft(y)
        cur = b.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        use_soft = (it % 3 == 0)
        if use_soft:
            grad = soft_topk_grad(y, b, k, beta)
        else:
            grad = topk_subgradient(y, b, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        eta = 2.0 * S / (n * (1.0 + 0.005 * it))
        best_x = y.copy()
        best_val = cur
        for factor in [1.0, 0.5, 0.25, 0.1]:
            x_cand = project_simplex(y - factor * eta * grad, S)
            b_check = autocorr_fft(x_cand)
            val = b_check.max()
            if val < best_val:
                best_val = val
                best_x = x_cand
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = best_x + ((t - 1.0) / t_new) * (best_x - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = best_x
        t = t_new
        if it % 150 == 0 and stall > 40:
            n_pert = max(20, n // 50)
            idx = rng.choice(n, size=n_pert, replace=False)
            y[idx] *= rng.uniform(0.5, 1.5, n_pert)
            y = project_simplex(y, S)
            t = 1.0
            stall = 0
        if stall > 400:
            break
    return best_a, best_score


def coordinate_descent(a_init, n, S, max_time, rng, n_top=30):
    """
    Partner-aware coordinate descent.
    For each top-contributing coordinate at the peak lag, try scaling it
    down and redistributing mass to coordinates with smaller partners.
    """
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    best_a = a.copy()
    best_score = autocorr_fft(a).max()
    start = time.monotonic()
    scale_factors = [0.2, 0.4, 0.6, 0.75, 0.85, 1.15, 1.3, 1.5]
    while time.monotonic() - start < max_time:
        b = autocorr_fft(a)
        c_max = b.max()
        peak_lag = int(np.argmax(b))
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        top_coords = np.argsort(contribs)[::-1][:n_top]
        improved = False
        # Find "cheap" targets: coordinates with small total contribution to top lags
        top_lags = np.argsort(b)[::-1][:min(20, 2 * n - 1)]
        total_contrib = np.zeros(n)
        for lag in top_lags[:10]:
            lo = max(0, int(lag) - n + 1)
            hi = min(int(lag), n - 1)
            for i in range(lo, hi + 1):
                j = int(lag) - i
                if 0 <= j < n:
                    total_contrib[i] += a[i] * a[j]
        cheap_coords = np.argsort(total_contrib)[:n_top]
        for idx in top_coords:
            if time.monotonic() - start > max_time:
                break
            i = int(idx)
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue
            orig = a[i]
            if orig < 1e-12:
                continue
            best_frac = 1.0
            best_local = c_max
            for frac in scale_factors:
                new_val = orig * frac
                delta = new_val - orig
                if abs(delta) < 1e-15:
                    continue
                if i == j:
                    est_change = 2.0 * delta * orig + delta * delta
                else:
                    est_change = 2.0 * delta * a[j]
                est_new_max = c_max + est_change
                if est_new_max < best_local:
                    best_local = est_new_max
                    best_frac = frac
            if best_frac != 1.0:
                a[i] *= best_frac
                # Redistribute to a cheap coordinate
                if best_frac < 1.0:
                    delta_mass = orig * (1.0 - best_frac)
                    # Add to a cheap coordinate
                    for ci in cheap_coords[:5]:
                        ci = int(ci)
                        if ci != i and ci != j:
                            a[ci] += delta_mass / 5.0
                a = project_simplex(a, S)
                improved = True
        if not improved and rng.random() < 0.4:
            n_pert = min(30, max(5, n // 200))
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a[idx_p] *= rng.uniform(0.6, 1.4, n_pert)
            a = project_simplex(a, S)
    return best_a, best_score


def spectral_perturb(a, n, N, S, sigma, rng):
    """Add random complex noise to FFT, project back."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def structured_init(n, rng, method="sin"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t) \
            + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = 0.3 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 4) ** 2) + 0.2 * np.exp(-0.5 * ((t - 0.7) * 4) ** 2)
    elif method == "asym":
        a = 0.3 + 0.4 * np.exp(-0.5 * ((t - 0.2) * 5) ** 2) + 0.1 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.01, 2.0)
    return a


def normalize_seq(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
```
