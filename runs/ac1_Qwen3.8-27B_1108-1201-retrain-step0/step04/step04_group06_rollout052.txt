```python
import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= v_sorted
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def peak_grad_step(a, top_k, step, S, c=None):
    """
    Projected gradient step targeting top-k peaks.
    Returns new a.
    """
    n = len(a)
    if c is None:
        c = fft_convolve(a)
    
    # Get top-k peaks
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    # Weighted gradient
    grad = np.zeros(n)
    for k, w in zip(top_indices, top_vals):
        k = int(k)
        w = w / total
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * w * a[k - (np.arange(lo, hi+1))]
    
    a_new = a - step * grad
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_phase_anneal(a, N, n, S, temperature, rng, n_steps=10):
    """
    Perturb phases of DFT with given temperature, project back.
    """
    best_a = a.copy()
    best_c = fft_convolve(a).max()
    
    for _ in range(n_steps):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        
        # Perturb phases
        phase_perturb = rng.normal(0, temperature, N)
        A_new = np.abs(A) * np.exp(1j * (np.angle(A) + phase_perturb))
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        c_new = fft_convolve(a_new).max()
        if c_new < best_c:
            best_c = c_new
            best_a = a_new.copy()
    
    return best_a, best_c
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-length search, spectral phase
annealing, peak-targeted projected gradient with backtracking, anti-symmetric
initialization, and vectorized coordinate descent.

Key innovations over previous:
1. Anti-symmetric initialization exploiting c[n-1] = sum a[i]*a[n-1-i] structure
2. Multi-n search (8192, 16384) to find best length
3. Backtracking projected gradient for robust step selection
4. Spectral phase annealing with geometric cooling for global exploration
5. Vectorized coordinate descent with efficient delta computation
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(c, n):
    """Objective = 2n*max_c/S^2. With S=sqrt(2n), this equals max_c."""
    return c.max()


def spectral_shape_step(a, N, n, S, top_k, alpha, rng, target=1.45):
    """
    Single spectral shaping step: reduce power at frequencies that
    constructively interfere at the top-k peak lags.
    """
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target:
        return a, c_max

    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = top_vals.sum()
    if excess_sum < 1e-30:
        return a, c_max

    weights = excess / excess_sum

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    alpha_eff = alpha * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.2)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    # Small random phase perturbation
    if rng.random() < 0.15:
        phase += rng.normal(0, 0.03 * alpha_eff, N)

    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def gradient_step_backtrack(a, S, c, top_k, step_init, max_backtrack=10):
    """
    Projected gradient step with backtracking line search.
    Targets top-k peaks with weighted gradient.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a, 0

    weights = top_vals / total
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * w * a[k - js]

    c_max = c.max()
    step = step_init
    for _ in range(max_backtrack):
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve(a_new)
        if c_new.max() < c_max:
            return a_new, step
        step *= 0.5
    return a, 0


def phase_anneal(a, N, n, S, temperature, rng, n_trials=20):
    """
    Spectral phase annealing: perturb phases, project back, keep best.
    """
    best_a = a.copy()
    best_c = fft_convolve(a).max()

    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    base_phase = np.angle(A)

    for _ in range(n_trials):
        perturbed_phase = base_phase + rng.normal(0, temperature, N)
        A_new = np.sqrt(P) * np.exp(1j * perturbed_phase)
        a_try = np.real(ifft(A_new))[:n]
        a_try = project_simplex(a_try, S)
        c_try = fft_convolve(a_try).max()
        if c_try < best_c:
            best_c = c_try
            best_a = a_try.copy()
        # Also try with slightly modified power
        if rng.random() < 0.3:
            P_mod = P * (1.0 + 0.1 * rng.standard_normal(N))
            P_mod = np.maximum(P_mod, 0)
            A_mod = np.sqrt(P_mod) * np.exp(1j * perturbed_phase)
            a_try2 = np.real(ifft(A_mod))[:n]
            a_try2 = project_simplex(a_try2, S)
            c_try2 = fft_convolve(a_try2).max()
            if c_try2 < best_c:
                best_c = c_try2
                best_a = a_try2.copy()

    return best_a, best_c


def coord_descent_fast(a, S, max_time, rng, n_top=25, multipliers=None):
    """
    Vectorized coordinate descent on top-contributing coordinates.
    """
    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.8, 0.85, 1.15, 1.2, 1.3, 0.3, 0.6, 1.4])

    n = len(a)
    start = time.monotonic()
    a_work = a.copy()
    c = fft_convolve(a_work)
    best_a = a_work.copy()
    best_c = c.max()

    while time.monotonic() - start < max_time:
        c_max = c.max()
        k_peak = int(np.argmax(c))

        if c_max < best_c:
            best_c = c_max
            best_a = a_work.copy()

        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        js = np.arange(lo, hi + 1)
        contribs = a_work[js] * a_work[k_peak - js]

        # Aggregate contribution per index
        index_contrib = np.zeros(n)
        for j in js:
            index_contrib[j] += contribs[j - lo]
            partner = k_peak - j
            if partner != j and 0 <= partner < n:
                index_contrib[partner] += contribs[j - lo]

        top_coords = np.argsort(index_contrib)[::-1][:n_top]

        improved = False
        for idx in top_coords:
            j = int(idx)
            if time.monotonic() - start > max_time:
                break
            orig = a_work[j]
            if orig < 1e-15:
                continue

            # Affected k range
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            k_range = np.arange(k_lo, k_hi + 1)
            j2_range = k_range - j  # 0..n-1

            best_frac = 1.0
            best_local = c_max

            for frac in multipliers:
                delta = orig * (frac - 1.0)
                delta_c = 2.0 * delta * a_work[j2_range]
                # Handle the k=2j case (where j==j2)
                mask_2j = (k_range == 2 * j)
                if np.any(mask_2j):
                    delta_c[mask_2j] = (orig * frac) ** 2 - orig ** 2

                new_c_vals = c[k_range].copy()
                new_c_vals += delta_c
                new_max_affected = new_c_vals.max()
                new_max_other = max(c[:k_lo].max() if k_lo > 0 else 0.0,
                                    c[k_hi + 1:].max() if k_hi + 1 < len(c) else 0.0)
                new_cmax = max(new_max_affected, new_max_other)

                if new_cmax < best_local:
                    best_local = new_cmax
                    best_frac = frac

            if best_frac != 1.0:
                delta = orig * (best_frac - 1.0)
                delta_c = 2.0 * delta * a_work[j2_range]
                mask_2j = (k_range == 2 * j)
                if np.any(mask_2j):
                    delta_c[mask_2j] = (orig * best_frac) ** 2 - orig ** 2
                c[k_range] += delta_c
                a_work[j] = orig * best_frac
                improved = True

        a_work = project_simplex(a_work, S)

        if not improved and rng.random() < 0.4:
            # Perturb top contributors
            pert_n = min(8, n_top)
            for idx in top_coords[:pert_n]:
                a_work[int(idx)] *= rng.uniform(0.5, 1.5)
            a_work = project_simplex(a_work, S)
            c = fft_convolve(a_work)

    return best_a, best_c


def make_init_antisym(n, eps, rng):
    """Anti-symmetric (lopsided) initialization to reduce c[n-1]."""
    t = np.arange(n) / n
    a = 1.0 + eps * (2.0 * t - 1.0)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_bump(n, center_frac, width, rng):
    """Gaussian bump off-center."""
    t = np.arange(n) / n
    a = 0.3 + 0.7 * np.exp(-0.5 * ((t - center_frac) / width) ** 2)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_sin(n, freq, rng):
    """Sinusoidal modulation."""
    t = np.arange(n) / n
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_random(n, rng):
    return 0.3 + 0.7 * rng.random(n)


def normalize_for_n(a, n):
    """Clip and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses spectral shaping, projected gradient, phase annealing,
    and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Try multiple lengths
    n_candidates = [8192, 16384]

    # Get previous best if available
    prev_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size > 0 and np.sum(np.clip(given, 0, 1000)) > 0:
            prev_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    global_best = np.inf
    global_a = None
    global_n = 0

    def try_update(a_candidate, n):
        nonlocal global_best, global_a, global_n
        a_candidate = normalize_for_n(a_candidate, n)
        c = fft_convolve(a_candidate)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            global_n = n
            return True
        return False

    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, deadline - time.monotonic()))
    if prev_seq is not None:
        print("Previous best length: %d" % prev_seq.size)

    # === PHASE 1: Multi-start across lengths (35% time) ===
    print("=" * 50)
    print("PHASE 1: Multi-start spectral shaping")
    print("=" * 50)
    t_p1 = time.monotonic()
    p1_budget = (deadline - t_p1) * 0.35
    p1_end = t_p1 + p1_budget

    inits_by_n = {}

    for n in n_candidates:
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        inits = []

        # From previous best (if length matches)
        if prev_seq is not None and prev_seq.size == n:
            inits.append(("prev", prev_seq.copy()))

        # Anti-symmetric
        inits.append(("asym_0.5", make_init_antisym(n, 0.5, rng)))
        inits.append(("asym_1.0", make_init_antisym(n, 1.0, rng)))
        inits.append(("asym_0.8", make_init_antisym(n, 0.8, rng)))

        # Off-center bump
        inits.append(("bump_0.3", make_init_bump(n, 0.3, 0.15, rng)))
        inits.append(("bump_0.7", make_init_bump(n, 0.7, 0.15, rng)))

        # Sinusoidal
        inits.append(("sin_3", make_init_sin(n, 3, rng)))
        inits.append(("sin_7", make_init_sin(n, 7, rng)))

        # Random
        inits.append(("rand", make_init_random(n, rng)))

        inits_by_n[n] = (inits, S, N)

        # Evaluate initializations
        for name, arr in inits:
            a_norm = normalize_for_n(arr, n)
            c = fft_convolve(a_norm)
            sc = c.max()
            if sc < global_best:
                global_best = sc
                global_a = a_norm.copy()
                global_n = n
                print("  [n=%d][%s] init: %.6f" % (n, name, sc))

        print("  n=%d: %d inits, best so far: %.6f" % (n, len(inits), global_best))

    # Run spectral shaping on each init (time-limited)
    n_total_inits = sum(len(inits_by_n[n][0]) for n in n_candidates)
    time_per_init = p1_budget / n_total_inits

    for n in n_candidates:
        inits, S, N = inits_by_n[n]
        for name, arr in inits:
            if time.monotonic() > p1_end:
                break
            a = normalize_for_n(arr, n)
            t0 = time.monotonic()
            t_limit = min(time_per_init, 30.0)

            it = 0
            while time.monotonic() - t0 < t_limit:
                it += 1
                top_k = min(20, 3 + it // 10)
                alpha = 0.5 / (1.0 + 0.03 * it)
                a, _ = spectral_shape_step(a, N, n, S, top_k, alpha, rng, target=1.45)
                if it % 200 == 0:
                    c = fft_convolve(a)
                    sc = c.max()
                    if sc < global_best:
                        global_best = sc
                        global_a = a.copy()
                        global_n = n

            # Final check
            c = fft_convolve(a)
            sc = c.max()
            if sc < global_best:
                global_best = sc
                global_a = a.copy()
                global_n = n
                print("  [n=%d][%s] shape: %.6f" % (n, name, sc))

    print("Phase 1 done. Best: %.6f (n=%d)" % (global_best, global_n))

    if global_a is None:
        global_a = normalize_for_n(make_init_random(8192, rng), 8192)
        global_n = 8192
        global_best = fft_convolve(global_a).max()

    # === PHASE 2: Focused optimization (45% time) ===
    print("=" * 50)
    print("PHASE 2: Focused optimization on n=%d" % global_n)
    print("=" * 50)
    t_p2 = time.monotonic()
    p2_budget = (deadline - t_p2) * 0.45
    p2_end = t_p2 + p2_budget

    n = global_n
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    a_curr = global_a.copy()

    # Alternating: spectral shaping, gradient, phase anneal
    stall = 0
    cycle = 0

    while time.monotonic() < p2_end:
        cycle += 1
        rem = p2_end - time.monotonic()
        if rem < 3:
            break

        # Spectral shaping burst
        t_sub = time.monotonic()
        sub_budget = min(rem * 0.3, 20.0)
        it = 0
        while time.monotonic() - t_sub < sub_budget:
            it += 1
            top_k = min(30, 5 + cycle // 5)
            alpha = 0.3 / (1.0 + 0.02 * it)
            a_curr, _ = spectral_shape_step(a_curr, N, n, S, top_k, alpha, rng, target=1.45)
            if it % 100 == 0:
                c = fft_convolve(a_curr)
                sc = c.max()
                if sc < global_best:
                    global_best = sc
                    global_a = a_curr.copy()
                    stall = 0
                    print("  [P2][cycle %d] shape: %.6f" % (cycle, sc))

        c = fft_convolve(a_curr)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_curr.copy()
            stall = 0
        else:
            stall += 1

        # Projected gradient with backtracking
        if time.monotonic() < p2_end - 5:
            step_init = 0.1 * S / np.sqrt(n)
            a_new, used_step = gradient_step_backtrack(a_curr, S, c, top_k=5,
                                                       step_init=step_init)
            if used_step > 0:
                c_new = fft_convolve(a_new)
                sc_new = c_new.max()
                if sc_new < global_best:
                    global_best = sc_new
                    global_a = a_new.copy()
                    a_curr = a_new.copy()
                    stall = 0
                    print("  [P2][cycle %d] grad: %.6f" % (cycle, sc_new))
                else:
                    a_curr = a_new.copy()  # Still move
                    stall += 1
            else:
                stall += 1

        # Phase annealing when stalled
        if stall >= 3 and time.monotonic() < p2_end - 10:
            temperature = 0.05 + 0.1 * min(stall, 10) / 10.0
            a_try, c_try = phase_anneal(a_curr, N, n, S, temperature, rng, n_trials=10)
            if c_try < global_best:
                global_best = c_try
                global_a = a_try.copy()
                a_curr = a_try.copy()
                stall = 0
                print("  [P2][cycle %d] anneal: %.6f" % (cycle, c_try))
            else:
                a_curr = a_curr + 0.01 * (a_try - a_curr)
                a_curr = project_simplex(a_curr, S)
            stall = 0  # Reset to avoid excessive annealing

        # If very stuck, try a fresh perturbation
        if stall >= 10 and time.monotonic() < p2_end - 20:
            print("  [P2][cycle %d] Very stalled, large perturbation" % cycle)
            a_curr = project_simplex(a_curr + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            # Quick spectral shaping from perturbed state
            t_sub2 = time.monotonic()
            for _ in range(50):
                a_curr, _ = spectral_shape_step(a_curr, N, n, S, 5, 0.4, rng, target=1.45)
            c = fft_convolve(a_curr)
            sc = c.max()
            if sc < global_best:
                global_best = sc
                global_a = a_curr.copy()
            stall = 0

    print("Phase 2 done. Best: %.6f" % global_best)

    # === PHASE 3: Try other n value (10% time) ===
    print("=" * 50)
    print("PHASE 3: Try other length")
    print("=" * 50)
    t_p3 = time.monotonic()
    p3_budget = (deadline - t_p3) * 0.10
    p3_end = t_p3 + p3_budget

    other_n = [x for x in n_candidates if x != global_n]
    if other_n:
        n2 = other_n[0]
        S2 = np.sqrt(2.0 * float(n2))
        N2 = 1 << (2 * n2 - 1).bit_length()

        # Best init for other n
        a2 = normalize_for_n(make_init_antisym(n2, 0.8, rng), n2)
        c2 = fft_convolve(a2)
        sc2 = c2.max()
        print("  n=%d init: %.6f" % (n2, sc2))
        if sc2 < global_best:
            global_best = sc2
            global_a = a2.copy()
            global_n = n2
            n = n2
            S = S2
            N = N2
            a_curr = a2.copy()

        # Spectral shaping
        t_sub = time.monotonic()
        while time.monotonic() < p3_end and time.monotonic() - t_sub < p3_budget:
            a2, _ = spectral_shape_step(a2, N2, n2, S2, 10, 0.3, rng, target=1.45)
            if (time.monotonic() - t_sub) % 10 < 0.1:
                c2 = fft_convolve(a2)
                sc2 = c2.max()
                if sc2 < global_best:
                    global_best = sc2
                    global_a = a2.copy()
                    global_n = n2
                    print("  n=%d shape: %.6f" % (n2, sc2))
            if time.monotonic() > p3_end:
                break

        c2 = fft_convolve(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            global_n = n2

    print("Phase 3 done. Best: %.6f (n=%d)" % (global_best, global_n))

    # === PHASE 4: Final polish (10% time) ===
    print("=" * 50)
    print("PHASE 4: Final polish")
    print("=" * 50)
    t_p4 = time.monotonic()
    p4_budget = deadline - t_p4
    n = global_n
    S = np.sqrt(2.0 * float(n))

    # Coordinate descent
    seg1 = min(p4_budget * 0.5, 60.0)
    if seg1 > 3:
        a_cd, sc_cd = coord_descent_fast(global_a, S, seg1, rng, n_top=30)
        if sc_cd < global_best:
            global_best = sc_cd
            global_a = a_cd.copy()
            print("  coord: %.6f" % global_best)

    # Fine spectral shaping
    rem = deadline - time.monotonic()
    if rem > 5:
        N = 1 << (2 * n - 1).bit_length()
        a_fine = global_a.copy()
        for it in range(200):
            if time.monotonic() > deadline - 5:
                break
            a_fine, _ = spectral_shape_step(a_fine, N, n, S, 10, 0.05, rng, target=1.40)
            if it % 50 == 0:
                c = fft_convolve(a_fine)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_fine.copy()
        c = fft_convolve(a_fine)
        if c.max() < global_best:
            global_best = c.max()
            global_a = a_fine.copy()

    # Final coordinate descent
    rem = deadline - time.monotonic()
    if rem > 3:
        a_cd2, sc_cd2 = coord_descent_fast(global_a, S, min(rem, 30.0), rng, n_top=40)
        if sc_cd2 < global_best:
            global_best = sc_cd2
            global_a = a_cd2.copy()
            print("  final coord: %.6f" % global_best)

    # --- Verification and return ---
    print("=" * 50)
    print("FINAL: n=%d, internal score=%.6f" % (global_n, global_best))
    print("=" * 50)

    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verify evaluate_sequence: %.6f" % ver if np.isfinite(ver) else "Verify: inf (fallback)")
        if not np.isfinite(ver):
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    return [float(x) for x in global_a]
```
