def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    return best_f, best


def solve_lp_best_response(f, n, S, top_k=200, time_limit=30.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    c = fft_convolve_self(f)
    M = c.max()

    if top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    for solver in ['ECOS', 'SCS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue

    if time.monotonic() - start > time_limit:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g

    return None


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    inits.append(rng.random(n) + 0.05)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Bump-like structures
    for center, width in [(0.3, 0.1), (0.7, 0.15), (0.5, 0.3)]:
        inits.append(np.exp(-((tt - center) / width) ** 2))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate():

    rng = np.random.default_rng(12345)

    global_a = None
    global_n = 8192

    given = None

            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)
    print("Main length: %d, budget: %.0fs" % (n, total_budget))

    # Phase 1: Quick FISTA warmup (60s)
    if given is not None:
        given_norm = np.clip(given, 0, 1000)
        s_g = given_norm.sum()
        if s_g > 1e-10:
            given_norm = given_norm * (S / s_g)
        else:
            given_norm = np.ones(n) * S / n

        print("=== Phase 1: FISTA warmup (%.0fs) ===" % 60)
        a1, s1 = run_fista_cd(given_norm, n, S, 60.0, rng,
                               min(deadline, time.monotonic() + 60),
                               beta_init=20.0, beta_rate=0.003, beta_max=500.0,
                               cd_every=100, cd_sweeps=5)
        c1 = fft_convolve_self(a1)
        obj1 = 2.0 * n * c1.max() / (a1.sum() ** 2)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()

        print("  Phase 1: obj=%.8f" % global_best)
    else:
        # Generate initial sequence

        a1, s1 = run_fista_cd(init, n, S, 60.0, rng,

    # Phase 2: Intensive coordinate descent (1200s)

        c_cd = fft_convolve_self(f_cd)
        obj_cd = 2.0 * n_p * c_cd.max() / (f_cd.sum() ** 2)

            global_best = obj_cd
            global_a = f_cd.copy()
            global_n = n_p

    # Phase 3: FISTA + CD cycles (400s)

    print("Returning sequence of length %d, best internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
