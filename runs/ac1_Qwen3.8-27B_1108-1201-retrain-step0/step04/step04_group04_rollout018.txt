def fft_convolve(a):
    """Compute full linear convolution a * a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def exact_subgrad(a, c):
    """Compute an exact subgradient of max(c) at point a.
    
    Returns the subgradient corresponding to the uniform distribution over active lags.
    """
    n = len(a)
    max_val = c.max()
    # Get all active lags (where c is close to max)
    tol = 1e-10 * max_val + 1e-15
    active = np.where(c >= max_val - tol)[0]
    
    # For numerical stability, weight by how close to max
    weights = np.exp(-100 * (max_val - c[active]))
    weights /= weights.sum()
    
    # grad[j] = 2 * sum_{k in active} w_k * a[k-j] (valid)
    # Use FFT
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(np.zeros(len(c)), fsize)
    P[active] = weights
    A = fft(a, fsize)
    corr = np.real(ifft(P * np.conj(A)))
    grad = 2.0 * corr[:n]
    return grad

def run_mirror_descent(a_init, n, S, max_time, rng):

    if c_final.max() < best:

            # Apply and check
            a[j] += delta
            a[m] -= delta
            if a[j] < 0 or a[m] < 0:
                a[j] -= delta
                a[m] += delta
                continue
            
            # Recompute c (approximately)
            c += delta * d
            # Add quadratic corrections
            if 2 * j < 2 * n - 1:
                c[2 * j] += delta * delta  # actually 2*delta^2? Let me just recompute periodically.
            
            new_max = c.max()
            if new_max < best - 1e-12:
                best = new_max
                best_a = a.copy()
                improved = True
            else:
                # Revert
                a[j] -= delta
                a[m] += delta
                c -= delta * d
                if 2 * j < 2 * n - 1:
                    c[2 * j] -= delta * delta
        
        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            elif c.max() > best + 1e-6:
                a = best_a.copy()
                c = fft_convolve(a)
        
        if not improved and it > 20:
            # Add small noise
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 100.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best
