"""
Minimize 2*n*max_conv/sum^2 using Nesterov-accelerated LSE + L-BFGS + coordinate swap polish.

Key innovations over previous approach:
1. Nesterov acceleration on LSE-smoothed objective for faster coarse convergence.
2. Adaptive beta schedule with oscillation to escape local minima.
3. Proper backtracking line search (Armijo) instead of fixed reduction.
4. Coordinate swap polish for final refinement (exploits gradient structure).
5. Warm start from previously found best sequence.
6. Frequency-domain mean-preserving reshaping as complementary global moves.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_and_grad(a, N, beta):
    """Compute LSE value and gradient. Returns (grad, lse_val, c)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -50, 50)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_reshape(a, N, top_k, alpha):
    """Spectral shaping: reduce peak by modifying power spectrum."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy()
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def nesterov_lse_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0):
    """Nesterov-accelerated gradient on LSE-smoothed objective."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    beta = beta_init
    t = 1.0
    a_prev = a.copy()
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Adaptive beta
        if it % 300 == 0:
            beta = min(beta_max, beta * 1.06)

        # Nesterov lookahead
        mu = (t - 1.0) / (t + 1.0)
        y = a + mu * (a - a_prev)
        y = np.maximum(y, 0.0)

        # Gradient at y
        grad_y, lse_y, c_y = lse_and_grad(y, N, beta)
        cmax_y = c_y.max()

        # Step size: 2 / (L * t) where L ~ beta * max_grad_component
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-18:
            break
        step = 2.0 * S / (n * beta * (1.0 + 0.01 * it)) * min(gnorm / (S / n), 5.0)
        step = min(step, 4.0 * S / n)

        # Backtracking line search
        alpha_ls = step
        x_new = a
        for ls in range(12):
            candidate = project_simplex(y - alpha_ls * grad_y, S)
            c_ls = compute_conv(candidate, N)
            if c_ls.max() <= c_y.max() + 0.01:
                x_new = candidate
                break
            alpha_ls *= 0.5
        else:
            x_new = project_simplex(y - alpha_ls * grad_y, S)

        a_prev = a
        a = x_new
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

        c_new = compute_conv(a, N)
        if c_new.max() < best_cmax:
            best_cmax = c_new.max()
            best_a = a.copy()

        # Periodic random perturbation
        if it % 500 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            a_prev = a.copy()
            t = 1.0

    return best_a, best_cmax


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=300.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta and Armijo line search."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    start_time = time.monotonic()
    it = 0

    g, f_val, c = lse_and_grad(a, N, beta)
    c_max = c.max()
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 150 == 0:
            beta = min(beta_max, beta * 1.1)
            g, f_val, c = lse_and_grad(a, N, beta)
            c_max = c.max()
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Armijo backtracking
        step0 = 3.0 * S / (n * max(beta, 1.0))
        slope = np.dot(g, direction)
        if slope >= 0:
            direction = -g
            slope = -np.dot(g, g)
        c_max = compute_conv(a, N).max()

        alpha_ls = step0
        a_new = a
        cmax_new = c_max
        for ls in range(18):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max + 1e-10:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.4

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw = lse_and_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-10):
            s_hist.append(s_k.copy())
            y_hist.append(y_k.copy())

        a = a_new
        g = g_new

        if it % 400 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, f_val, c = lse_and_grad(a, N, beta)

    return best_a, best_cmax


def coordinate_swap_polish(a_init, n, S, N, max_time, rng):
    """Coordinate swap polish: transfer mass from high-gain to low-gain positions at the peak."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        c = compute_conv(a, N)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        # Find peak position(s)
        peak_thresh = cmax * 0.999
        active = np.where(c >= peak_thresh)[0]
        k_star = active[0]

        # Compute gain vector: gain[m] = a[k_star - m] + a[m - k_star] (with 0 for out of range)
        gain = np.zeros(n)
        for m in range(n):
            v1 = k_star - m
            v2 = m - k_star
            if 0 <= v1 < n:
                gain[m] += a[v1]
            if 0 <= v2 < n:
                gain[m] += a[v2]

        # Find high-gain (source) and low-gain (target) positions
        # Among positions with non-zero a, find max gain
        nz = a > 1e-10
        if not np.any(nz):
            break
        gain_nz = gain.copy()
        gain_nz[~nz] = -np.inf
        src = np.argmax(gain_nz)

        # Find target: minimize gain, prefer positions with small a (less disruption)
        gain_target = gain.copy()
        # Add penalty for large a at target (to avoid creating new peaks)
        gain_target = gain_target + 0.1 * a
        tgt = np.argmin(gain_target)

        if src == tgt:
            break

        if gain[src] <= gain[tgt]:
            # No improvement possible
            if rng.random() < 0.5:
                # Try random swap
                candidates = np.where(nz)[0]
                if len(candidates) > 1:
                    src = candidates[rng.integers(len(candidates))]
                    tgt_candidates = np.setdiff1d(np.arange(n), [src])
                    tgt = tgt_candidates[rng.integers(len(tgt_candidates))]
                else:
                    break
            else:
                break

        # Determine step size: transfer so that gain equality is approached
        delta_gain = gain[src] - gain[tgt]
        if delta_gain <= 0:
            break
        # Optimal delta: balance the two gains
        # New gain at src: (a[src]-d) * 2 (approx) -> decreases by 2d
        # New gain at tgt: a[tgt]+d -> gain increases by 2*something
        # Simple: transfer a fraction
        frac = min(0.1, 0.5 * delta_gain / (gain[src] + 1e-10))
        delta = frac * a[src]
        delta = min(delta, 0.5 * a[src])

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 0:
            a_new = a_new * (S / s_new)

        c_new = compute_conv(a_new, N)
        if c_new.max() < cmax:
            a = a_new
        else:
            # Try smaller step
            delta2 = delta * 0.3
            a_new2 = a.copy()
            a_new2[src] -= delta2
            a_new2[tgt] += delta2
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 0:
                a_new2 = a_new2 * (S / s2)
            c_new2 = compute_conv(a_new2, N)
            if c_new2.max() < cmax:
                a = a_new2
            else:
                # Stuck, try random perturbation
                if rng.random() < 0.3:
                    a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
                else:
                    break

        it += 1
        if it % 1000 == 0:
            print("    coord_polish it=%d, cmax=%.6f" % (it, c.max()))

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (7 * tt + 0.5 * tt * tt)) + 0.15 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_asym(n, rng):
    """Asymmetric initialization: high in first half, low in second (complementary)."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a_first = 1.0 + 0.5 * np.sin(2 * np.pi * 5 * tt[:n//2])
    a_second = 0.3 + 0.3 * np.sin(2 * np.pi * 5 * tt[n//2:] + np.pi)
    a = np.concatenate([a_first, a_second])
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

        it += 1
        c = compute_conv(a_sp, N)
        cur = c.max()
        if cur < best_sp:
            best_sp = cur
            global_best = best_sp
            global_a = a_sp.copy()
            print("  [spectral it%d] %.6f" % (it, best_sp))
        excess = max(0.0, cur - 1.0) / max(cur, 1e-10)
        alpha_curr = 0.15 * (0.3 + 0.7 * excess)
        a_sp = spectral_reshape(a_sp, N, top_k=10, alpha=alpha_curr)
        if it % 200 == 100 and rng.random() < 0.2:
            a_sp = project_simplex(a_sp + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    # --- Phase 4: High-beta L-BFGS (fine) ---
    print("=== Phase 4: High-beta L-BFGS (fine) ===")
    t4 = time.monotonic()
    remaining_after4 = (deadline - t4) * 0.4
    phase4_budget = min(400.0, max(50.0, remaining_after4))
    a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, phase4_budget, rng,
                                     beta_init=50.0, beta_max=500.0, m_lbfgs=15)
    update_best(a_fin, "high_beta_LBFGS")

    # --- Phase 5: Coordinate swap polish ---
    print("=== Phase 5: Coordinate swap polish ===")
    t5 = time.monotonic()
    remaining = deadline - t5
    polish_budget = min(300.0, max(20.0, remaining * 0.5))
    a_pol, sc_pol = coordinate_swap_polish(global_a, n, S, N, polish_budget, rng)
    update_best(a_pol, "coord_polish")

    # --- Phase 6: Final L-BFGS ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Phase 6: Final L-BFGS ===")
        a_final, sc_final = lbfgs_lse_block(global_a, n, S, N, min(remaining - 5, 200.0),
                                             rng, beta_init=100.0, beta_max=1000.0, m_lbfgs=15)
        update_best(a_final, "final_LBFGS")

    # --- Phase 7: Final coordinate polish ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pol2, sc_pol2 = coordinate_swap_polish(global_a, n, S, N, min(remaining - 5, 100.0), rng)
        update_best(a_pol2, "coord_polish2")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f (internal: %.6f)" % (ver, global_best))
        else:
            print("Verification FAILED, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d" % len(global_a))
    return [float(x) for x in global_a]
