"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using spectral equalization,
accelerated subgradient descent, and coordinate descent refinement.

Key improvements:
1. Principled spectral equalization via multiplicative weight updates.
2. Nesterov-accelerated subgradient with adaptive step size.
3. Coordinate descent refinement for final polishing.
4. Multi-start with diverse initializations.
5. Focus on n=8192 with budget allocation.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(np.asarray(v, dtype=np.float64), -1e30)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = (theta_candidates <= w) & (idx > 0)
    if not np.any(valid):
        rho = n
        theta = (cumsum[rho - 1] - S) / rho
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def spectral_equalize(a, n_iter, alpha, rng):
    """
    Iteratively flatten the power spectrum using multiplicative weight updates.
    This pushes |FFT(a)|^2 toward uniform, which should reduce the autocorrelation peak.
    """
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        A = fft(a, N)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Multiplicative weight update: reduce high-power bins, boost low-power bins
        # P_new = P * exp(-alpha * (P/P_mean - 1))
        # This is equivalent to: P_new = P * (P_mean/P)^(alpha/2) * exp(alpha/4) (approximately)
        # Simpler: P_new = P * (P_mean / P)^alpha for P > 0
        ratio = P / P_mean
        # Use a smoothed version to avoid division by zero
        P_new = P * np.exp(-alpha * (ratio - 1.0))
        
        # Ensure non-negative
        P_new = np.maximum(P_new, 0.0)
        
        # Preservation of total power (Parseval): sum(P) = N * sum(a^2)
        # We'll re-normalize in time domain after inverse FFT
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        
        # Project onto simplex
        a_new = project_simplex(a_new, S)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using accelerated subgradient descent
with adaptive peak tracking, momentum restarts, and multi-start strategy.

Key improvements over previous approach:
1. Focused FISTA with adaptive top-k (start k=1 for exact subgradient, grow to k=30)
2. Polyak step size using theoretical lower bound
3. Momentum restart when stalled
4. Antisymmetric perturbation initializations exploiting a[j]*a[n-1-j] structure
5. Efficient gradient computation exploiting the shift structure
6. Aggressive time management with early termination

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = (theta_candidates <= w) & (idx > 0)
    if not np.any(valid):
        rho = n
        theta = (cumsum[rho - 1] - S) / rho
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    theta = max(theta, -1e30)
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_gradient(c, a, top_k, n):
    """
    Compute weighted subgradient of max(c) w.r.t. a.
    Uses top-k lags with weights proportional to their values.
    Returns gradient vector of length n.
    """
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(len(a))
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_optimize(a_init, n, S, max_time, rng, start_k=1, end_k=30,
                   f_star=1.0, noise_interval=500, noise_prob=0.25):
    """
    FISTA (Nesterov-accelerated proximal gradient) for minimizing max(convolve(a,a))
    subject to a >= 0, sum(a) = S.
    
    Uses adaptive top-k and Polyak step size with momentum restarts.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_local = a.copy()
    it = 0

    # Track best for Polyak step
    f_best = 2.0 * n * fft_convolve_self(y).max() / (S * S)
    
    # Stall detection
    stall_count = 0
    stall_threshold = 200
    last_improve_it = 0

    N = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive top-k: grow from start_k to end_k over the first 50% of iterations
        frac = min(1.0, it / max(1, int(max_time * 15)))  # rough estimate of total iters
        k = int(start_k + (end_k - start_k) * min(1.0, it / 1000))
        k = min(k, n // 4)

        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)

        if sc < best_score:
            best_score = sc
            best_local = y.copy()
            last_improve_it = it

        # Polyak step size: step = (f(y) - f*) / ||g||^2
        grad = compute_gradient(c, y, k, n)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Use Polyak step with safeguard
        f_current = 2.0 * n * cur_max / (y.sum() ** 2)
        polyak_step = (f_current - f_star) / (gnorm * gnorm / (n * S) + 1e-20)
        polyak_step = max(0.01, min(polyak_step, 2.0 * S / (n * 0.5)))

        # Also compute standard diminishing step
        std_step = 2.0 * S / (n * (1.0 + 0.02 * np.sqrt(it)))

        # Blend: use the larger of the two (Polyak when optimal, std as fallback)
        step = max(polyak_step * 0.8, std_step)
        step = min(step, 4.0 * S / n)

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15) * S * 0.5, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)

        # Check if momentum is helping: compare f(y_new) vs f(x_new)
        sc_x = 2.0 * n * fft_convolve_self(x_new).max() / (x_new.sum() ** 2)
        sc_y = 2.0 * n * fft_convolve_self(y_new).max() / (y_new.sum() ** 2)

        if sc_y > sc_x * 1.01:
            # Momentum is hurting, use x directly and reset t
            y = x_new.copy()
            t = 1.0
        else:
            y = y_new
            t = t_new

        a = x_new

        # Stall detection: if no improvement in stall_threshold iterations, add noise
        if it - last_improve_it > stall_threshold:
            stall_count += 1
            if stall_count > 3:
                # Reset with noise
                noise_scale = 0.005 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t = 1.0
                stall_count = 0
                last_improve_it = it

                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)

                t = 1.0

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0

def spectral_flat_step(a, alpha, N):
    """
    One step of spectral flattening using multiplicative weight update.
    Pushes |FFT(a)|^2 toward uniform.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a

    ratio = P / P_mean
    # Multiplicative update: reduce high, boost low
    P_new = P * np.exp(-alpha * (ratio - 1.0))
    P_new = np.maximum(P_new, 0.0)

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def try_spectral_flatten(a, max_time, rng):
    """Apply spectral flattening for a limited time, keep best result."""
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    alpha = 0.5

    it = 0
    while time.monotonic() - start < max_time:
        # Process in batches
        batch = min(50, int(max_time / max(0.01, time.monotonic() - start + 0.01)))
        for _ in range(batch):
            a = spectral_flat_step(a, alpha, N)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # Anneal alpha
        it += 1
        alpha = max(0.05, 0.5 * (0.95 ** it))

    return best_a, best_score


def make_antisymmetric_init(n, rng, S):
    """
    Create an initialization with antisymmetric structure around the center.
    a[j] = S/n + b[j] where b[j] ≈ -b[n-1-j]
    This minimizes the center-lag autocorrelation.
    """
    base = S / n
    # Generate antisymmetric perturbation
    half = n // 2
    b_half = rng.standard_normal(half) * 0.3
    # Extend to full antisymmetric: b[j] for j < half, b[n-1-j] = -b[j] for j >= half
    b = np.zeros(n)
    b[:half] = b_half
    if n % 2 == 0:
        b[half:] = -b_half[::-1]
    else:
        b[half:] = -b_half[::-1][:n - half]

    # Scale to keep a >= 0
    max_neg = -b.min()
    if max_neg > base * 0.9:
        b = b * (base * 0.9 / max_neg)

    a = base + b
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    return a

    a = np.maximum(a, 0.01)

    inits.append(a)

    a = S / n * (0.5 + smooth)
    a = a * (S / a.sum())

    return inits

        a = a_f

    # Phase 3: Refinement with different parameters (20% budget)
    t_p3 = 0.20 * budget
    elapsed = time.monotonic() - t_start
    remaining = budget - elapsed
    t_p3_actual = min(t_p3, remaining)
    if t_p3_actual > 5:
        a_r, sc_r = fista_optimize(best_a, n, S, t_p3_actual, rng,
                                    start_k=5, end_k=50,
                                    f_star=max(1.0, best_score * 0.85))
        if sc_r < best_score:
            best_score = sc_r
            best_a = a_r.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses FISTA with adaptive subgradient + spectral flattening + multi-start.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    n = 8192
    S = np.sqrt(2.0 * float(n))

    print("Starting optimization: n=%d, S=%.2f, budget=%.0fs" % (n, S, budget_s))

    global_best = np.inf
    global_a = None

    t_global_start = time.monotonic()

    # Collect initializations
    inits = []
    priorities = []

    # Try to use provided sequence (highest priority)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            given_clipped = np.clip(given, 0.0, 1000.0)
            if given_clipped.sum() > 0.01:
                inits.append(given_clipped)
                priorities.append(0)
                print("Using provided sequence as initial, sum=%.2f" % given_clipped.sum())
    except Exception:
        pass

    # Add diverse initializations
    div_inits = make_diverse_inits(n, rng, S)
    for d in div_inits:
        inits.append(d)
        priorities.append(1)

    print("Total initializations: %d" % len(inits))

    # Budget allocation: 50% to provided, rest split among others
    total_budget = budget_s - 40  # leave 40s for final polish and verification
    provided_budget = total_budget * 0.45
    other_budget = total_budget * 0.55
    n_others = max(1, len(inits) - (1 if priorities[0] == 0 else 0))
    per_other = other_budget / n_other

    # Process in priority order
    sorted_indices = sorted(range(len(inits)), key=lambda i: priorities[i])

    for idx in sorted_indices:
        elapsed = time.monotonic() - t_global_start
        remaining = total_budget - elapsed
        if remaining < 15:
            break

        if priorities[idx] == 0:
            alloc = min(provided_budget, remaining - 10)
        else:
            alloc = min(per_other, remaining - 10)

        if alloc < 10:
            break

        print("  Opt init %d (prio=%d) with budget %.0fs" % (idx, priorities[idx], alloc))
        t0 = time.monotonic()

        a_result, score = optimize_single_start(inits[idx], n, S, alloc, rng)

        elapsed_this = time.monotonic() - t0
        if score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("    *** New best: %.6f (took %.0fs) ***" % (global_best, elapsed_this))
        else:
            print("    Score: %.6f (best: %.6f, took %.0fs)" % (score, global_best, elapsed_this))

    # Final polish on the best solution
    if global_a is not None:
        final_budget = max(0, total_budget - (time.monotonic() - t_global_start))
        if final_budget > 10:
            print("Final polish with %.0fs" % final_budget)

            # Try FISTA with very small steps (refinement)
            a_fp, sc_fp = fista_optimize(global_a, n, S, min(final_budget * 0.6, 120.0),
                                         rng, start_k=10, end_k=100,
                                         f_star=max(1.0, global_best * 0.9))
            if sc_fp < global_best:
                global_best = sc_fp
                global_a = a_fp.copy()
                print("  Final polish improved to: %.6f" % global_best)

            # Quick spectral flatten
            rem = max(0, final_budget - (time.monotonic() - t_global_start - (total_budget - final_budget)))
            if rem > 5:
                a_sf, sc_sf = try_spectral_flatten(global_a, min(rem * 0.5, 30.0), rng)
                if sc_sf < global_best:
                    global_best = sc_sf
                    global_a = a_sf.copy()
                    print("  Spectral polish improved to: %.6f" % global_best)

    # Fallback
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(n, dtype=np.float64) * S / n
        global_best = 2.0

    # Ensure valid output
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 0.5

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned non-finite, using fallback")
        global_a = np.ones(n, dtype=np.float64) * S / n
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback verification: %.6f" % global_best)

    total_time = time.monotonic() - t_global_start
    print("Done in %.0fs. Final: n=%d, score=%.6f" % (total_time, len(global_a), global_best))
    return [float(x) for x in global_a]
