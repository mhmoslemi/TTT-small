4. **Perturbation**: Add small random perturbations to escape local minima.

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using alternating minimization with FISTA.
Key idea: alternate between (1) softmax weights on convolution lags (w-step),
and (2) FISTA on the resulting QP f^T H(w) f (f-step). At high beta, use
delta weights for O(n) f-steps. Multiple beta schedules and n values explored.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_Hf(f, w, n, L, W_fft):
    """Compute (Hf)_j = sum_m w[j+m] * f[m] using precomputed FFT of w."""
    F = fft(f, L)
    Hf = np.real(ifft(np.conj(F) * W_fft))[:n]
    return Hf


def compute_Hf_hard(f, k_star, n):
    """Hard Hf: (Hf)_j = f[k_star - j] if valid, else 0."""
    j = np.arange(n)
    m = k_star - j
    valid = (m >= 0) & (m < n)
    result = np.zeros(n)
    result[valid] = f[m[valid]]
    return result


def fista_qp_soft(f_init, w, n, S, n_iter, step, L, W_fft):
    """FISTA for soft QP using precomputed W_fft."""
    f = f_init.copy()
    y = f.copy()
    t = 1.0
    for _ in range(n_iter):
        Hf = compute_Hf(y, w, n, L, W_fft)
        grad = 2.0 * Hf
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new
    return f


def fista_qp_hard(f_init, k_star, n, S, n_iter, step):
    """FISTA for hard QP (delta weight at k_star)."""
    f = f_init.copy()
    y = f.copy()
    t = 1.0
    for _ in range(n_iter):
        Hf = compute_Hf_hard(y, k_star, n)
        grad = 2.0 * Hf
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new
    return f


def run_alternating(f_init, n, S, total_time, rng,
                    beta_start=5.0, beta_end=500.0,
                    fista_iters=80, step=0.5,
                    perturb_every=300, noise_scale=0.005):
    """Alternating minimization: softmax w + FISTA f-step."""
    f = f_init.copy()
    start = time.monotonic()
    best_obj = np.inf
    best_f = f.copy()
    it = 0
    L = 1 << (3 * n - 1).bit_length()

    while time.monotonic() - start < total_time:
        it += 1
        progress = min(1.0, it / max(1, int(total_time / 0.05)))
        beta = beta_start * (beta_end / beta_start) ** progress

        # W-step
        c = fft_convolve_self(f)
        c_max = c.max()
        c_shifted = c - c_max
        e = np.exp(beta * c_shifted)
        w_sum = e.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w = e / w_sum

        # F-step
        if beta > 100.0:
            k_star = int(np.argmax(c))
            f = fista_qp_hard(f, k_star, n, S, fista_iters, step)
        else:
            W_fft = fft(w, L)
            f = fista_qp_soft(f, w, n, S, fista_iters, step, L, W_fft)

        # Evaluate
        c = fft_convolve_self(f)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()

        # Perturbation
        if it % perturb_every == 0:
            ns = noise_scale * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, ns, n), S)

    return best_f, best_obj


def run_hard_polish(f_init, n, S, total_time, rng, fista_iters=200, step=0.5):
    """Polish using only hard (delta) weights at various argmax positions."""
    f = f_init.copy()

    it = 0

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length using linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def generate_chirp_init(n, rng):
    """Generate a chirp (FM) initialization."""
    tt = np.arange(n) / n
    phase = 2.0 * np.pi * (tt + 0.5 * tt * tt)
    f = np.abs(np.cos(phase)) + 0.01
    return f


def generate_random_init(n, rng):
    """Generate a random non-negative initialization."""
    f = rng.random(n) + 0.05

    if s < 0.01:

    return 2.0 * n * c.max() / (s * s)

        print("Final verification: %.6f" % ver if np.isfinite(ver) else "Final verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            global_a *= np.sqrt(2 * 256) / 256
    else:
        global_a = np.ones(256, dtype=np.float64)
        global_a *= np.sqrt(2 * 256) / 256

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
