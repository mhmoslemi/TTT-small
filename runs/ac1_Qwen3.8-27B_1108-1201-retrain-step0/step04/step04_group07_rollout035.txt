def project_simplex(v, S):

    n = len(v)

    idx = np.arange(1, n+1, dtype=np.float64)

So the problem is fundamentally about: given a finite-length non-negative sequence, how flat can its autocorrelation be?

For a sequence that's 1 in the middle and 0 at the edges (like a rectangular window), the autocorrelation is triangular. For a sequence that's 1 everywhere (uniform), same thing. For a sequence with specific structure (like a Barker sequence), the sidelobes can be much lower than the peak.

The key insight is: we want to minimize the PEAK of the autocorrelation relative to the sum. The peak is typically at the center (lag n-1) where all n terms contribute. We want to make this central peak as small as possible while keeping the sum fixed.

For the central lag k = n-1:
`c_{n-1} = sum_j a[j] * a[n-1-j]`

This is the "central overlap" - the dot product of `a` with its reverse. To minimize this while keeping `sum(a) = S`, we want `a[j]` and `a[n-1-j]` to be "anti-correlated" - when one is large, the other is small.

For non-negative sequences, this means: if `a[0]` is large, `a[n-1]` should be small, and vice versa. More generally, the sequence should have a "complementary" structure where the first half and the reversed second half are "anti-correlated."

This is very similar to the construction of Barker sequences or low-autocorrelation binary sequences!

**New approach: Direct optimization of the central peak**

Since the peak is (usually) at the center, let's focus on minimizing `c_{n-1} = sum_j a[j] * a[n-1-j]`.

This is a quadratic form: `c_{n-1} = a^T J a` where `J` is the exchange matrix (J[i,j] = 1 if i+j = n-1).

We want to minimize `a^T J a` subject to `a >= 0, sum(a) = S`.

The minimum of `a^T J a` over the simplex is 0 (achieved when `a[i] = 0` for all i where `a[n-1-i] > 0`, i.e., the support is entirely in one half). But this makes other lags large.

So we need to balance: minimize the central peak while keeping other peaks low.

**Algorithm: Iterative peak trimming**

1. Compute `c = convolve(a, a)`
2. Find the top-k lags (largest values)
3. For each such lag k, compute the "pressure" on each position j:
   `pressure[j] += w_k * a[k-j]` (if valid)
4. Update: reduce a[j] proportionally to pressure[j], increase where pressure is low
5. Project onto simplex

This is a form of "successive convex approximation" or "iterative water-filling."

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def lse_gradient(c, a, tau, fsize):
    """
    Compute gradient of logsumexp(c/tau) w.r.t. a.
    grad[j] = (1/tau) * sum_k softmax(c_k/tau)_k * a[k-j] (valid indices)
    """
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(len(a))
    w /= w_sum
    
    W = np.fft.fft(w, fsize)
    A = np.fft.fft(a, fsize)
    grad = np.real(np.fft.ifft(W * np.conj(A)))[:len(a)] / tau
    return grad


def nesterov_lse_optimize(a0, n, S, budget_s, tau_init, tau_min, rng, target=1.50):

    a = np.asarray(a0, dtype=np.float64).copy()

    y = a.copy()
    t = 1.0
    best_score = compute_score(a)

    start = time.monotonic()

        tau = tau_init * (tau_min / tau_init) ** progress
        tau = max(tau, 1e-4)

        c = fft_convolve_self(y)
        c_max = c.max()
        
        # LSE gradient
        grad = lse_gradient(c, y, tau, fsize)
        
        # Step size: use diminishing step

        if gnorm < 1e-15:
            break
        eta = 1.0 / (it * 0.01 + 1.0)

        x_new = project_simplex(x - eta * grad, S)
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - x)
        y = np.maximum(y, 0.0)
        y = y * S / max(y.sum(), 1e-30)
        x = x_new
        t = t_new

            if sc < best_score:
                best_score = sc
                best_a = x.copy()

                    last_print = it

                break

                noise_scale = 0.05 * S / np.sqrt(n)
                y = project_simplex(x + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    return best_a, best_score


def peak_trimming(a0, n, S, n_iter, rng, top_k=5):
    """
    Iterative peak trimming: identify top lags, apply repulsive force.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a = a * S / max(a.sum(), 1e-30)
    
    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        c = fft_convolve_self(a)
        c_max = c.max()
        
        # Top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        weights = np.exp(top_vals / (0.1 * c_max + 1e-10))
        weights /= weights.sum()
        
        # Apply pressure
        pressure = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            # a[k-j] is the contribution of position j to lag k
            # For position j, a[k-j] is valid when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                pressure[j_lo:j_hi+1] += weights[i] * a[k - j_hi:k - j_lo:-1] if k - j_hi != k - j_lo - 1 else weights[i] * a[k - j_lo:k - j_hi + 1]
        
        # Simpler: just use FFT
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = np.fft.fft(w_full, fsize)
        A = np.fft.fft(a, fsize)
        pressure = np.real(np.fft.ifft(W * np.conj(A)))[:n]
        
        # Update: reduce high-pressure positions
        strength = 0.5 * (1.0 - it / n_iter) + 0.1
        a_new = a - strength * pressure / (c_max + 1e-10) * S
        a_new = project_simplex(a_new, S)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if best_score <= 1.50:
            break
    
    return best_a, best_score

    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def lse_subgradient(c, a, tau, fsize):
    """
    Compute subgradient of logsumexp(c/tau) w.r.t. a.
    Uses FFT cross-correlation: grad[j] = (1/tau) * sum_k softmax_k * a[k-j]
    """
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(len(a))
    w /= w_sum

    W = np.fft.fft(w, fsize)
    A = np.fft.fft(a, fsize)
    grad = np.real(np.fft.ifft(W * np.conj(A)))[:len(a)] / tau
    return grad


def nesterov_lse(a0, n, S, budget_s, tau_init, tau_min, rng, target=1.50):
    """
    Nesterov-accelerated LSE smoothed subgradient descent.
    Minimizes max(convolve(a,a)) subject to a>=0, sum(a)=S.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n)
    a = a * S / a.sum()

    fsize = 1 << (2 * n - 1).bit_length()

    x = a.copy()
    y = a.copy()
    t = 1.0
    best_score = compute_score(x)
    best_a = x.copy()

    start = time.monotonic()
    it = 0
    last_report = 0

    while time.monotonic() - start < budget_s:
        it += 1

        # Anneal tau: exponential decay
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / (budget_s * 0.85))
        tau = tau_init * (tau_min / tau_init) ** frac
        tau = max(tau, 1e-5)

        # Compute convolution at y
        c = fft_convolve_self(y)

        # LSE subgradient
        grad = lse_subgradient(c, y, tau, fsize)

        # Step size: theoretical bound + diminishing
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        a_max = np.max(np.abs(y)) + 1e-10
        eta_theo = 2.0 * tau * tau / (a_max * a_max + 1e-10)
        eta_dim = 8.0 * S / (gnorm * np.sqrt(it) + 1e-10)
        eta = min(eta_theo, eta_dim, 5.0)
        eta = max(eta, 1e-10)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        x_new = project_simplex(x - eta * grad, S)
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - x)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < 1e-30:
            y = x_new.copy()
        else:
            y = y * S / y_sum
        x = x_new
        t = t_new

        # Periodic evaluation
        if it % 100 == 0:
            sc = compute_score(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
                if it - last_report > 3000:
                    print(f"      nesterov it={it:d} score={sc:.6f} tau={tau:.6f}")
                    last_report = it

            if best_score <= target:
                break

            # Periodic perturbation to escape local minima
            if it % 10000 == 5000 and rng.random() < 0.4:
                noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
                y = project_simplex(x + noise, S)
                t = 1.0
                # Reset Nesterov
                x = y.copy()

    return best_a, best_score


def peak_pressure_optimize(a0, n, S, n_iter, rng, top_k=10, strength_init=1.0):
    """
    Iterative peak pressure reduction.
    At each step: compute convolution, identify top-k lags,
    apply repulsive pressure via FFT cross-correlation.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a * S / max(a.sum(), 1e-30)

    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        c = fft_convolve_self(a)
        c_max = c.max()

        # Temperature for softmax over lags (controls how many lags we target)
        temp = 0.15 * c_max * (1.0 - 0.7 * it / n_iter) + 0.01 * c_max
        c_scaled = c / temp
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()

        # Compute pressure via FFT cross-correlation
        W = np.fft.fft(w, fsize)
        A = np.fft.fft(a, fsize)
        pressure = np.real(np.fft.ifft(W * np.conj(A)))[:n]

        # Apply pressure: reduce high-pressure positions
        strength = strength_init * (1.0 - 0.8 * it / n_iter) + 0.05
        scale = c_max / S + 1e-10
        a_new = a - strength * pressure / scale * S
        a_new = project_simplex(a_new, S)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if best_score <= 1.50:
            break

    return best_a, best_score


def multiplicative_update(a0, n, S, n_iter, tau=1.0, top_k=20, rng=None):
    """
    Multiplicative weight update (mirror descent on simplex).
    Alternates between reducing peak lags and the max lag.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a * S / max(a.sum(), 1e-30)

    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        c = fft_convolve_self(a)
        c_max = c.max()

        # Softmax weights with temperature
        temp = tau * c_max / max(top_k, 1)
        c_scaled = c / temp
        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w /= w.sum()

        # Pressure
        W = np.fft.fft(w, fsize)
        A = np.fft.fft(a, fsize)
        pressure = np.real(np.fft.ifft(W * np.conj(A)))[:n]

        # Multiplicative update: a_new = a * exp(-eta * pressure)
        eta = 0.5 * (1.0 - 0.5 * it / n_iter)
        log_a = np.log(np.maximum(a, 1e-30) / S)
        log_a_new = log_a - eta * pressure / (c_max + 1e-10)
        log_a_new -= log_a_new.max()  # prevent overflow
        a_new = np.exp(log_a_new)
        a_new = a_new * S / max(a_new.sum(), 1e-30)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if best_score <= 1.50:
            break

    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse non-negative initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n)))

    # Random
    inits.append(("random", rng.random(n) + 0.01))

    # Binary random
    b = (rng.random(n) >= 0.5).astype(float)
    inits.append(("binary", np.maximum(b, 0.01)))

    # Sine modulated
    inits.append(("sin1", np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * tt), 0.01)))
    inits.append(("sin3", np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * 3 * tt), 0.01)))
    inits.append(("sin7", np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * 7 * tt), 0.01)))

    # Chirp
    inits.append(("chirp", np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * (2 * tt + 2 * tt * tt)), 0.01)))

    # Asymmetric ramp (breaks central symmetry)
    inits.append(("ramp", np.maximum(1.0 - 0.8 * tt, 0.01)))
    inits.append(("ramp2", np.maximum(0.5 + 0.4 * np.sin(np.pi * tt), 0.01)))

    # Low-pass filtered noise
    noise = rng.standard_normal(n)
    ksize = max(3, n // 64)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth", 0.1 + 0.9 * smooth))

    # Two-tone (alternating blocks)
    block = n // 64
    tones = (np.arange(n) // block % 2).astype(float)
    inits.append(("two_tone", 0.3 + 0.7 * tones))

    return inits
