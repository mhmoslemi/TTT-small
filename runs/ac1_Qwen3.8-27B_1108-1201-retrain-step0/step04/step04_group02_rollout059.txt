from scipy.fft import fft, ifft

# Helper functions at top level (no closures)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(v - np.minimum(v, 0))  # Hmm, this isn't right for negative values
    # Actually the standard algorithm:
    # Sort v in descending order
    # Find the largest rho such that v[rho] + (S - sum(v[0:rho]))/rho > 0
    # If all v[i] <= 0, then the projection is uniform S/n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    rho_candidates = (cumsum - S) / idx
    valid = (w > rho_candidates) & (w > 0)
    if not np.any(valid):
        return np.full(n, S/n)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):

    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a ** 2)

def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:

        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += a[j - i_vals]
    return grad / k

def fft_conv_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A*A))[:size]

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= S:
        # All mass is below S, just clip and renormalize
        v = np.maximum(v, 0)
        s = v.sum()
        if s < 1e-15:
            return np.full(n, S/n)
        return v * (S/s)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    rho_vals = (cumsum - S) / idx
    valid = w > rho_vals
    if not np.any(valid):
        return np.full(n, S/n)
    rho = int(np.where(valid)[0].max() + 1)
    theta = (cumsum[rho-1] - S) / rho

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Combined subgradient (top-k peaks) + L2 flattening gradient
- Adaptive step size with backtracking and perturbations
- Multi-length search (8192, 16384) with diverse initializations
- Water-filling gradient for smoother optimization signal

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Self-convolution of a via FFT (full linear convolution)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def project_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho_vals = (cumsum - S) / idx
    valid = w >= rho_vals
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.where(valid)[0].max()) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def topk_grad(a, c, k):
    """Subgradient of max(c) using top-k peaks, averaged."""
    n = len(a)
    k = min(k, len(c))
    if k >= len(c):
        top = np.arange(len(c))
    else:
        top = np.argpartition(c, -k)[-k:]
    grad = np.zeros(n)
    for j in top:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def l2_grad(a, c):
    """Gradient of sum(c[k]^2) w.r.t. a. Computed via FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(C * np.conj(A)))
    return 4.0 * cross[:n]


def waterfill_grad(a, c, k):
    """Weighted subgradient: weight each of top-k peaks by excess over mean."""
    n = len(a)
    k = min(k, len(c))
    mean_c = c.mean()
    excess = np.maximum(c - mean_c, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n)
    if k >= len(c):
        top = np.arange(len(c))
    else:
        top = np.argpartition(excess, -k)[-k:]
    w = excess[top]
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for idx in range(len(top)):
        j = top[idx]
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * w[idx] * a[j - i_vals]
    return grad


def compute_obj(a, n):
    """Compute objective value given a and its convolution."""
    c = fft_conv_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf, c
    return 2.0 * n * c.max() / (S * S), c


def optimize_combined(a_init, n, S, t_budget, rng, mode="combined", l2_w=0.3):
    """
    Main optimization loop.
    mode: 'combined' (peak + L2), 'peak' (top-k only), 'water' (water-filling)
    """
    a = project_simplex(a_init.astype(np.float64).copy(), S)
    t0 = time.monotonic()
    c = fft_conv_self(a)
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    step = 1.0
    fails = 0
    it = 0

    while time.monotonic() - t0 < t_budget:
        it += 1
        cur = 2.0 * n * c.max() / S**2
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a.copy()
            fails = 0

        # Build gradient based on mode
        if mode == "combined":
            k = min(64, 1 + it // 3000)
            g1 = topk_grad(a, c, k)
            g2 = l2_grad(a, c)
            n1 = np.linalg.norm(g1)
            n2 = np.linalg.norm(g2)
            if n1 > 1e-15 and n2 > 1e-15:
                alpha = l2_w * n1 / n2
                g = g1 + alpha * g2
            elif n1 > 1e-15:
                g = g1
            else:
                g = g2
        elif mode == "peak":
            k = min(32, 1 + it // 5000)
            g = topk_grad(a, c, k)
        elif mode == "water":
            k = min(256, 16 + it // 2000)
            g = waterfill_grad(a, c, k)
            g2 = l2_grad(a, c)
            n1 = np.linalg.norm(g)
            n2 = np.linalg.norm(g2)
            if n1 > 1e-15 and n2 > 1e-15:
                alpha = l2_w * n1 / n2
                g = g1 + alpha * g2 if n1 > 1e-15 else g
        else:
            g = topk_grad(a, c, 1)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Cap step size
        max_step = S / (n * gn + 1e-15)
        step = min(step, max_step)

        a_new = project_simplex(a - step * g, S)
        c_new = fft_conv_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2

        if obj_new < cur - 1e-15:
            a, c = a_new, c_new
            step = min(step * 1.04, max_step)
        else:
            fails += 1
            step *= 0.72
            if fails > 50:
                # Structured perturbation to escape local min
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
                elif strategy == 1:
                    freq = rng.uniform(1, 8)
                    mod = 1.0 + 0.2 * np.sin(2 * np.pi * freq * np.arange(n) / n)
                    a = project_simplex(a * mod, S)
                elif strategy == 2:
                    # Block perturbation
                    nb = int(rng.integers(4, 12))
                    bm = rng.uniform(0.7, 1.3, nb)
                    mod = np.repeat(bm, n // nb + 1)[:n]
                    a = project_simplex(a * mod, S)
                else:
                    # Frequency domain perturbation
                    fsize = 1 << (n - 1).bit_length()
                    A = fft(a, fsize)
                    phase_noise = rng.uniform(-0.1, 0.1, len(A))
                    A = A * np.exp(1j * phase_noise)
                    a = project_simplex(np.real(ifft(A))[:n], S)
                c = fft_conv_self(a)
                step = 1.0
                fails = 0

        if it % 30000 == 0:
            print("    it=%d obj=%.6f best=%.6f t=%.0f" % (it, cur, best_obj, time.monotonic() - t0))

    return best_a, best_obj


def generate_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.sin(np.pi * t)**2 + 0.01)
    inits.append(np.maximum(1.0 + 0.3*np.cos(4*np.pi*t) + 0.1*np.cos(8*np.pi*t), 0.01))
    inits.append(np.maximum(1.0 + 0.2*np.sin(2*np.pi*3*t) + 0.1*cos(2*np.pi*7*t), 0.01) if False else np.maximum(1.0 + 0.2*np.sin(2*np.pi*3*t) + 0.1*np.cos(2*np.pi*7*t), 0.01))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(-4.0*(t-0.5)**2/0.25) + 0.01)
    inits.append(np.maximum(1.0 + 0.4*(t**2 + (1-t)**2), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Load previous best
    prev = None
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        prev = np.clip(prev, 0, 1000)
        if prev.size < 64 or prev.sum() < 0.01:
            prev = None
    except Exception:
        prev = None

    print("Previous best available: %s (len=%s)" % (prev is not None, prev.size if prev is not None else 0))
    t_start = time.monotonic()

    tracks = []

    # Track 1: Refine previous best with combined gradient
    if prev is not None and len(prev) >= 512:
        n1 = len(prev)
        S1 = np.sqrt(2.0 * n1)
        t_alloc = min(700.0, (deadline - time.monotonic()) * 0.35)
        if t_alloc > 60:
            print("Track 1: refine prev n=%d, t=%.0fs" % (n1, t_alloc))
            a, obj = optimize_combined(prev, n1, S1, t_alloc, rng, mode="combined", l2_w=0.3)
            a = project_simplex(a, S1)
            real_obj, _ = compute_obj(a, n1)
            tracks.append((real_obj, a, "track1"))
            print("  Track 1 done: %.6f (t=%.0fs)" % (real_obj, time.monotonic() - t_start))
            if real_obj < global_best:
                global_best = real_obj
                global_a = a.copy()

    # Track 2: Upsample to 16384 and optimize
    if prev is not None:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        remaining = deadline - time.monotonic()
        t_alloc = min(600.0, remaining * 0.30)
        if t_alloc > 60:
            # Upsample via interpolation
            x_old = np.linspace(0, 1, len(prev), endpoint=False)
            x_new = np.linspace(0, 1, n2, endpoint=False)
            a2_init = np.interp(x_new, x_old, np.maximum(prev, 1e-6))
            a2_init = np.maximum(a2_init, 0.0)
            print("Track 2: upsample n=%d, t=%.0fs" % (n2, t_alloc))
            a2, obj2 = optimize_combined(a2_init, n2, S2, t_alloc, np.random.default_rng(seed + 100), mode="combined", l2_w=0.2)
            a2 = project_simplex(a2, S2)
            real_obj2, _ = compute_obj(a2, n2)
            tracks.append((real_obj2, a2, "track2"))
            print("  Track 2 done: %.6f (t=%.0fs)" % (real_obj2, time.monotonic() - t_start))
            if real_obj2 < global_best:
                global_best = real_obj2
                global_a = a2.copy()

    # Track 3: Fresh start at n=8192 with water-filling
    n3 = 8192
    S3 = np.sqrt(2.0 * n3)
    remaining = deadline - time.monotonic()
    t_alloc = min(400.0, remaining * 0.25)
    if t_alloc > 60:
        # Use a diverse initialization
        inits3 = generate_inits(n3, rng)
        best_init = None
        best_init_obj = np.inf
        for init in inits3[:4]:
            init = project_simplex(init, S3)
            obj_init, _ = compute_obj(init, n3)
            if obj_init < best_init_obj:
                best_init_obj = obj_init
                best_init = init.copy()

        print("Track 3: fresh n=%d, t=%.0fs (init obj=%.4f)" % (n3, t_alloc, best_init_obj))
        a3, obj3 = optimize_combined(best_init, n3, S3, t_alloc, np.random.default_rng(seed + 200), mode="water", l2_w=0.4)
        a3 = project_simplex(a3, S3)
        real_obj3, _ = compute_obj(a3, n3)
        tracks.append((real_obj3, a3, "track3"))
        print("  Track 3 done: %.6f (t=%.0fs)" % (real_obj3, time.monotonic() - t_start))
        if real_obj3 < global_best:
            global_best = real_obj3
            global_a = a3.copy()

    # Track 4: Peak-only refinement of the best found so far
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 90:
        nb = len(global_a)
        Sb = np.sqrt(2.0 * nb)
        t_alloc = remaining * 0.40
        print("Track 4: peak refine n=%d, t=%.0fs" % (nb, t_alloc))
        a4, obj4 = optimize_combined(global_a.copy(), nb, Sb, t_alloc, np.random.default_rng(seed + 300), mode="peak", l2_w=0.0)
        a4 = project_simplex(a4, Sb)
        real_obj4, _ = compute_obj(a4, nb)
        tracks.append((real_obj4, a4, "track4"))
        print("  Track 4 done: %.6f (t=%.0fs)" % (real_obj4, time.monotonic() - t_start))
        if real_obj4 < global_best:
            global_best = real_obj4
            global_a = a4.copy()

    # Track 5: Final combined refinement
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        nb = len(global_a)
        Sb = np.sqrt(2.0 * nb)
        t_alloc = remaining * 0.85
        print("Track 5: final combined n=%d, t=%.0fs" % (nb, t_alloc))
        a5, obj5 = optimize_combined(global_a.copy(), nb, Sb, t_alloc, np.random.default_rng(seed + 400), mode="combined", l2_w=0.5)
        a5 = project_simplex(a5, Sb)
        real_obj5, _ = compute_obj(a5, nb)
        tracks.append((real_obj5, a5, "track5"))
        print("  Track 5 done: %.6f (t=%.0fs)" % (real_obj5, time.monotonic() - t_start))
        if real_obj5 < global_best:
            global_best = real_obj5
            global_a = a5.copy()

    # Select overall best
    if global_a is None and tracks:
        best_track = min(tracks, key=lambda x: x[0])
        global_best = best_track[0]
        global_a = best_track[1]

    if global_a is None:
        # Fallback: constant sequence
        n_f = 256
        global_a = np.ones(n_f) * np.sqrt(2 * n_f) / n_f
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_f = 256
        global_a = np.ones(n_f) * np.sqrt(2 * n_f) / n_f
        global_best = 2.0

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f (internal: %.6f)" % (ver, global_best))
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")
            n_f = 256
            global_a = np.ones(n_f) * np.sqrt(2 * n_f) / n_f
            final_list = [float(x) for x in global_a]
            global_best = 2.0
    except Exception as e:
        print("Verification error: %s" % str(e))

    print("FINAL: n=%d, obj=%.6f, total_time=%.0fs" % (len(final_list), global_best, time.monotonic() - t_start))
    return final_list
