"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Adaptive LSE-FISTA with aggressive beta schedule, active-set refinement,
multi-length exploration, and surgical peak shaving for the final "last mile."
Key insight: The value equals max(f*f) when sum(f)=sqrt(2n), so we directly minimize
the peak of the self-convolution. Asymmetry in the sequence reduces the center lag.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, beta_at=300.0):
    """Gradient using hard active set (top lags within threshold of max)."""
    n = len(a)
    cm = c.max()
    threshold = cm - 5.0 / beta_at
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200,
                  momentum_restart_gap=500, noise_decay=80000.0):
    """
    FISTA with adaptive LSE smoothing.

    Uses momentum with restart and periodic noise injection.
    Returns (best_a, best_true_max).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    cur_true_max = np.inf

    total_iterations = 0
    while time.monotonic() - start < max_time:
        it += 1
        total_iterations += 1

        # Adaptive beta: logistic schedule
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        # Compute convolution
        c = fft_convolve_self(y)
        cur_true_max = c.max()

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        # Compute LSE gradient
        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/beta for LSE-smoothed function
        step = 1.0 / beta

        # Stagnation handling

        if stagnant > momentum_restart_gap:

            # Inject noise proportional to current precision

        t_mom = t_new

                best_a = y.copy()
                last_improve_it = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def raised_cosine_init(n, center_frac=0.3, width_frac=0.5):
    """Generate a raised-cosine initialization centered at center_frac*n."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)

        inits.append(np.maximum(a, 1e-10))

    # Sinusoidal
    for k in [1, 2, 3]:

def peak_shaving(a, n, S, n_peaks=5, n_steps=50, lr_scale=0.01):
    """
    Surgical peak shaving: identify the top lags and directly reduce contributions.
    For each peak lag k, reduce a[j] where a[j]*a[k-j] is large.
    """
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute "responsibility" for each a[j]
        resp = np.zeros(n)
        for k in top_idx:
            # Contribution of a[j] to c[k] is a[j] * a[k-j]
            # For valid j: 0 <= j < n and 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Add to responsibility (weighted by how much this lag exceeds target)
            weight = max(0, c[k] - cm * 0.99)
            np.add.at(resp, js, contrib * weight)

        # Normalize and apply
        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Multiplicative update: reduce high-responsibility elements
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety margin: %.0fs" % (budget_s, safety))

    # Try to load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()  # Since sum = sqrt(2n), value = max_b
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # === Phase 1: Quick optimization of given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(60.0, (deadline - time.monotonic()) * 0.1)
        print("=== Phase 1: Quick LSE on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng,
                                beta_start=100.0, beta_end=5000.0, beta_steps=50)
        try_sequence(a1)

    # === Phase 2: Main optimization on best found ===
    if global_a is not None:
        n_main = len(global_a)
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.45
        print("=== Phase 2: Main hybrid (n=%d, %.0fs) ===" % (n_main, t2))
        a2, v2 = run_hybrid(global_a, n_main, S_main, t2, rng)
        try_sequence(a2)

    # === Phase 3: Try larger n ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_large = len(global_a) * 2
        if n_large <= 32768:
            S_large = np.sqrt(2.0 * n_large)
            # Interpolate
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_large)
            init_large = np.interp(t_new, t_old, global_a)
            init_large = normalize_sum(init_large, n_large)

            t3 = remaining * 0.3
            print("=== Phase 3: Larger n=%d (%.0fs) ===" % (n_large, t3))
            a3, v3 = run_hybrid(init_large, n_large, S_large, t3, rng)
            try_sequence(a3)

    # === Phase 4: Try completely different initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_try = 8192
        S_try = np.sqrt(2.0 * n_try)
        t4 = remaining * 0.25
        print("=== Phase 4: Diverse inits (n=%d, %.0fs) ===" % (n_try, t4))

        inits = generate_diverse_inits(n_try, rng)
        per_init = max(3.0, t4 / len(inits))

        # Sort inits by initial objective (quick eval)
        init_scores = []
        for i, init in enumerate(inits):
            init = normalize_sum(init, n_try)
            c = fft_convolve_self(init)
            init_scores.append((c.max(), i))

        init_scores.sort()

        for score, idx in init_scores[:6]:
            if time.monotonic() > deadline - 30:
                break
            if per_init < 3:
                break
            init = inits[idx]
            print("  Trying init %d (initial max=%.4f)" % (idx, score))
            ai, vi = run_lse_fista(init, n_try, S_try, per_init, rng,
                                    beta_start=50.0, beta_end=10000.0, beta_steps=80)
            try_sequence(ai)

    # === Phase 5: Peak shaving on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_p = len(global_a)
            S_p = np.sqrt(2.0 * n_p)
            print("=== Phase 5: Peak shaving (remaining %.0fs) ===" % remaining)
            a_p = peak_shaving(global_a, n_p, S_p, n_peaks=8, n_steps=100, lr_scale=0.005)
            try_sequence(a_p)

            if a_p.sum() > 0:
                # Quick LSE polish after shaving
                remaining2 = max(1.0, deadline - time.monotonic())
                if remaining2 > 10:
                    a_p2, v_p2 = run_lse_fista(a_p, n_p, S_p, remaining2 * 0.5, rng,
                                                beta_start=5000.0, beta_end=30000.0, beta_steps=30)
                    try_sequence(a_p2)

    # === Phase 6: Final active-set refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            print("=== Phase 6: Final active refinement (%.0fs) ===" % remaining)
            a_f, v_f = run_active_refinement(global_a, n_f, S_f, remaining * 0.6, rng, n_active=20)
            try_sequence(a_f)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
