The minimum of sum a[i]*a[n-1-i] subject to sum a[i] = S and a[i] >= 0 is achieved when half the mass is at the beginning and half at the end (or more precisely, when the "anti-pairing" is maximized).

Wait, by the rearrangement inequality, sum a[i]*b[perm(i)] is minimized when a and b are oppositely sorted. So if a is increasing, then a[n-1-i] is decreasing, and sum a[i]*a[n-1-i] is the minimum possible pairing.

The minimum of sum a[i]*a[n-1-i] subject to sum a[i] = S, a[i] >= 0 is... let's think. If we put all mass at i=0 and i=n-1 (say a[0] = S/2, a[n-1] = S/2), then c[n-1] = a[0]*a[n-1] + a[n-1]*a[0] = 2*(S/2)*(S/2) = S^2/2. But c[0] = a[0]^2 = S^2/4 and c[2n-2] = a[n-1]^2 = S^2/4. Also c[1] = 2*a[0]*a[n-1] = S^2/2 (if n > 2). So max(c) = S^2/2, giving objective = 2n * S^2/2 / S^2 = n. That's terrible for large n.

So putting mass at the ends is bad because it creates large values at small lags.

The optimal tradeoff is to spread the mass in a way that minimizes the peak of the convolution. This is essentially a "time-frequency" tradeoff.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-phase optimization: subgradient, LSE-FISTA, multiplicative updates,
with adaptive step sizes, multiple lengths, and parametric initialization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / idx
    valid = v_sorted - tau > 0
    if not np.any(valid):
        # All zeros case
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    diff = c - cm
    max_diff = diff.max()
    if beta * max_diff > 500:
        beta = 500.0 / max(1e-15, max_diff)
    terms = np.exp(beta * diff)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    max_diff = diff.max()
    if beta * max_diff > 500:
        beta = 500.0 / max(1e-15, max_diff)
    scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: 2 * cross_correlate(w, a) truncated to n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, k=8):
    """Combined subgradient from top-k lags, weighted by excess above mean."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_lse_fista(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.003,
                  beta_max=50000.0, use_topk_blend=False, topk_frac=0.01):
    """FISTA with LSE smoothing and optional top-k blending."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    target_improve = 500

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > target_improve:
            # Restart with noise
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
            no_improve = 0
            target_improve = min(target_improve + 200, 2000)
            continue
        
        grad = lse_gradient(y, c, beta)
        
        if use_topk_blend:
            k_blend = max(4, int(len(c) * topk_frac))
            grad_topk = topk_subgradient(y, c, k=k_blend)
            grad = 0.7 * grad + 0.3 * grad_topk
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use 1/L where L ~= 2*beta*||a||_inf (rough Lipschitz estimate)
        step = 1.0 / (2.0 * beta * 0.3 * max(1.0, np.max(np.abs(a))))
        step = min(step, S / (n * gnorm))  # Cap to avoid overshooting
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_subgradient(a_init, S, max_time, rng, k=8, eta_init=None):
    """Pure subgradient method on top-k lags with adaptive step size."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    if eta_init is None:
        eta_init = S / (n * 2.0)
    
    eta = eta_init
    it = 0
    no_improve = 0
    min_eta = eta_init * 1e-6
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 1000:
            eta *= 0.5
            if eta < min_eta:
                break
            no_improve = 0
        
        grad = topk_subgradient(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step: try to reduce the peak
        # The peak value is cur_max, gradient norm is gnorm
        # Step that would reduce peak by ~10%: eta * gnorm * cur_max ~ cur_max * 0.1
        # So eta ~ 0.1 / gnorm
        step = min(eta, 0.1 / gnorm)
        
        a_new = project_simplex(a - step * grad, S)
        
        # Line search: try a few step sizes
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.01:
            # Step was too large, try smaller
            for factor in [0.5, 0.25, 0.1]:
                a_try = project_simplex(a - step * factor * grad, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max:
                    a_new = a_try
                    break
            else:
                a_new = a  # Keep old
        
        a = a_new
    
    return best_a, best


def run_multiplicative_peak_shave(a_init, S, max_time, rng, alpha=0.1):
    """Multiplicative update to reduce peak contributions."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
        
        # Find top k lags
        k = min(16, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        
        # Compute "blame" for each element
        blame = np.zeros(n)
        for idx in top_idx:
            weight = c[idx] / c[top_idx[0]]  # Relative weight
            for j in range(n):
                k_j = idx - j
                if 0 <= k_j < n:
                    blame[j] += weight * a[k_j]
        
        # Multiplicative update: reduce elements with high blame
        blame_max = blame.max()
        if blame_max < 1e-30:
            break
        
        ratio = 1.0 - alpha * blame / blame_max
        ratio = np.maximum(ratio, 0.01)  # Don't zero out
        a_new = a * ratio
        a_new = normalize_to_sum(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.05:
            alpha *= 0.5
            if alpha < 0.001:
                break
        else:
            a = a_new
    
    return best_a, best


def generate_parametric_init(n, rng, n_params=20):
    """Generate a parametric initialization: sum of sinusoids + offset."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(n_params):
        freq = rng.integers(1, 50)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.maximum(a, 0.01)


def generate_window_init(n, rng):
    """Generate various window function initializations."""
    tt = np.arange(n, dtype=np.float64) / n
    choices = [
        np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01),
        0.54 + 0.46 * np.cos(2 * np.pi * tt),
        np.exp(-5.0 * (tt - 0.5) ** 2 / 0.05),
        np.sin(np.pi * tt) ** 2,

        np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01),
        np.exp(rng.normal(0, 0.3, n)),
        rng.random(n) + 0.1,
    ]
    return np.maximum(choices[rng.integers(len(choices))], 0.01)

- This gives more weight to the peak and less to the valleys

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-phase optimization with adaptive step sizes, excess annealing,
LSE-FISTA, subgradient with line search, and multiplicative updates.
Explores multiple sequence lengths for diversity.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / idx
    valid = v_sorted - tau > 0
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    max_diff = diff.max()
    if max_diff > 0 and beta * max_diff > 400:
        beta = 400.0 / max_diff
    scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, k=8):
    """Combined subgradient from top-k lags weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_val(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_lse_fista(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.004,
                  beta_max=100000.0, blend_topk=False, topk_k=8):
    """FISTA with LSE smoothing, adaptive step, and restarts."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_kr = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    patience = 400

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_kr = 1.0
            no_improve = 0
            patience = min(patience + 300, 3000)
            continue

        grad = lse_gradient(y, c, beta)
        if blend_topk:
            g2 = topk_subgradient(y, c, k=topk_k)
            grad = 0.6 * grad + 0.4 * g2

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with backtracking
        a_inf = max(1.0, np.max(a))
        step = 1.0 / (1.5 * beta * 0.25 * a_inf)
        step = min(step, 2.0 * S / (n * gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)

        # Backtracking: check if we made progress
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur_max * (1 + 1e-6) and step > 1e-20:
            # Try smaller step
            for trial in [0.5, 0.25, 0.1]:
                x_try = project_simplex(y - step * trial * grad, S)
                c_try = fft_convolve_self(x_try)
                if c_try.max() <= cur_max:
                    x_new = x_try
                    step *= trial
                    break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_kr * t_kr))
        y = x_new + ((t_kr - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_kr = t_new

    return best_a, best


def run_excess_anneal(a_init, S, max_time, rng):
    """Anneal threshold T downward, minimizing excess above T with adaptive step."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    T_start = c0.max()
    T_floor = c0.mean() + 0.02 * (c0.max() - c0.mean())

    y = a.copy()
    t_kr = 1.0
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Anneal T with a bias toward the lower end
        T = T_start + (T_floor - T_start) * (progress ** 0.5)
        T = max(T, T_floor)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_kr = 1.0
            no_improve = 0
            continue

        # Excess gradient
        excess = np.maximum(c - T, 0.0)
        p = 2.0
        w = p * excess ** (p - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            # Use subgradient if no excess
            grad = topk_subgradient(y, c, k=4)
        else:
            w = w / wsum
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        step = 1.0 / (0.5 * max(1.0, S / n))
        step = min(step, S / (n * gnorm + 1e-15) * 0.5)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_kr * t_kr))
        y = x_new + ((t_kr - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_kr = t_new

    return best_a, best


def run_subgradient_ls(a_init, S, max_time, rng, k=6):
    """Subgradient on top-k lags with line search."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    eta = S / (4.0 * n)
    min_eta = eta * 1e-8
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 600:
            eta *= 0.7
            if eta < min_eta:
                break
            no_improve = 0

        grad = topk_subgradient(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Line search: try several step sizes
        best_step = eta
        best_cmax = cur_max
        for factor in [1.0, 0.5, 0.25, 0.1, 2.0]:
            trial_step = min(eta * factor, S / (n * gnorm + 1e-15) * 0.3)
            if trial_step < 1e-20:
                continue
            a_try = project_simplex(a - trial_step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_cmax:
                best_cmax = c_try.max()
                best_step = trial_step
                a = a_try

        eta = best_step * 1.1  # Slightly increase if we found a good step

    return best_a, best


def run_multiplicative(a_init, S, max_time, rng, alpha_init=0.15):
    """Multiplicative update to reduce peak contributions."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    alpha = alpha_init
    it = 0
    k = min(12, n // 16)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:k]
        blame = np.zeros(n)
        for idx in top_idx:
            weight = c[idx] / c[top_idx[0]]
            # blame[j] += weight * a[idx - j] for valid j
            j_start = max(0, idx - n + 1)
            j_end = min(n, idx + 1)
            for j in range(j_start, j_end):
                k_j = idx - j
                if 0 <= k_j < n:
                    blame[j] += weight * a[k_j]

        blame_max = blame.max()
        if blame_max < 1e-30:
            break

        ratio = 1.0 - alpha * blame / blame_max
        ratio = np.maximum(ratio, 0.005)
        a_new = a * ratio
        a_new = normalize_to_sum(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.02:
            alpha *= 0.6
            if alpha < 0.002:
                break
        else:
            a = a_new
            alpha = min(alpha * 1.05, 0.5)

    return best_a, best


def gen_parametric_init(n, rng, n_sines=15):
    """Parametric init: offset + sum of sinusoids."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 1.0
    for _ in range(n_sines):
        freq = rng.integers(1, 40)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.maximum(a, 0.01)


def gen_window_init(n, rng):
    """Various window function inits."""
    tt = np.arange(n, dtype=np.float64) / n
    choices = [
        np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01),
        0.54 + 0.46 * np.cos(2 * np.pi * tt),
        np.exp(-4.0 * (tt - 0.5) ** 2 / 0.04),
        np.sin(np.pi * tt) ** 2,
        np.sin(np.pi * tt) ** 4,
        np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01),
        np.exp(rng.normal(0, 0.3, n)),
        rng.random(n) + 0.2,
        1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.cos(2 * np.pi * 7 * tt + 0.5),
    ]
    return np.maximum(choices[rng.integers(len(choices))], 0.01)
