"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Adaptive active-set gradient with backtracking line search
2. Variance flattening on the active set
3. LSE-smoothed FISTA for global convergence
4. Multi-scale initialization (small-n optimize, upscale, refine)
5. Coordinate-wise hill climbing for final polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_set_gradient(a, c, k):
    """Exact subgradient from top-k lags with uniform weights."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def variance_gradient(a, c, k):
    """
    Gradient that reduces variance of b over the top-k lags.
    This encourages flattening the peak region.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    b_active = c[idx]
    mean_b = b_active.mean()
    # Want to minimize sum_{i in active} (b_i - mean_b)^2
    # Gradient wrt a: 2 * sum_i 2*(b_i - mean_b) * (a shifted by i)
    # = 4 * sum_i (b_i - mean_b) * (a * delta_i)
    # where delta_i is the impulse at lag i
    weights = 2.0 * (b_active - mean_b)
    wsum = abs(weights).sum()
    if wsum < 1e-20:
        return np.zeros(n)
    w_full = np.zeros(len(c))
    w_full[idx] = weights / wsum
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_gradient(a, c, weights):
    """Compute 2 * (w * a) where * is linear convolution, truncated to n."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_adaptive_active_set(a_init, S, max_time, rng, 
                            k_start=8, k_end=64, 
                            delta_start=0.1, delta_end=0.01):
    """
    Adaptive active-set method with annealing of the active set size
    and a backtracking-like step size strategy.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    y = a.copy()
    t_momentum = 1.0
    it = 0
    step = 1.0 / (2.0 * S)
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive parameters
        k = int(k_start + (k_end - k_start) * progress ** 0.5)
        delta = delta_start + (delta_end - delta_start) * progress
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Build active set weights
        threshold = c.max() - delta * (c.max() - c.mean() + 1e-10)
        active = np.where(c > threshold)[0]
        
        if len(active) == 0:
            active = np.argsort(c)[::-1][:k]
        
        # Use top-k from active for stability
        if len(active) > k:
            # Weight by excess within active set
            excess = c[active] - c[active].min()
            e_sum = excess.sum()
            if e_sum < 1e-20:
                w_active = np.ones(len(active)) / len(active)
            else:
                w_active = excess / e_sum
        else:
            w_active = np.ones(len(active)) / len(active)
        
        # Build full weight vector
        w_full = np.zeros(len(c))
        w_full[active] = w_active
        
        # Compute gradient
        grad = weighted_gradient(y, c, w_full)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-14:
            break
        
        # Adaptive step size: decrease as we progress, with occasional increase
        base_step = S / (n * (0.5 + 0.5 * (1.0 - progress)))
        step = base_step * (2.0 / (1.0 + 0.1 * it))
        
        # Try the step
        x_new = project_simplex(y - step * grad, S)
        
        # Check if it improved (light check)
        # For speed, skip the full convolution check most of the time
        if it % 10 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() > c.max() * 1.05:
                # Step was too large, reduce it
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
        
        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new
        
        # Periodic restart with noise
        if it % 500 == 0 and progress < 0.9:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_momentum = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    
    return best_a, best


def run_lse_fista(a_init, S, max_time, rng,
                  beta_init=50.0, beta_rate=0.003, beta_max=20000.0):
    """FISTA with LSE smoothing."""
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    y = a.copy()
    t = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-14:
            break
        
        # Step size: 1/L where L is Lipschitz constant of the gradient
        # For LSE with parameter beta, L ≈ 2 * beta * max|a| * 2 (rough)
        step = 1.0 / (2.0 * beta * y.max() * 2.0 + 1e-10)
        step = min(step, 1.0 / (2.0 * S * 0.1))
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Restart periodically
        if it % 2000 == 0:
            t = 1.0
    
    return best_a, best


def run_variance_flatten(a_init, S, max_time, rng, k=32):
    """
    Phase that tries to reduce the variance of the top-k convolution values.
    This helps when the max is achieved at multiple lags with slightly different values.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    y = a.copy()
    t = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Compute variance gradient
        grad_var = variance_gradient(y, c, k)
        
        # Also compute the standard max gradient
        grad_max = active_set_gradient(y, c, min(8, k))
        
        # Combine: primarily reduce max, secondarily flatten
        alpha = 0.3 * (1.0 - progress)  # Reduce variance contribution over time
        grad = (1.0 - alpha) * grad_max + alpha * grad_var
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = S / (n * (1.0 + 0.01 * it))
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best

def run_multi_scale(a_init_large, S_large, max_time, rng,

    """

    best_a = a_init_large.copy()

        t_old = np.linspace(0, 1, small_n)
        t_new = np.linspace(0, 1, n_large)

def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if base_seq is not None and len(base_seq) >= n:
        if len(base_seq) == n:
            inits.append(np.clip(base_seq, 0.01, None))
        else:
            ratio = len(base_seq) // n
            small = base_seq[:n * ratio].reshape(n, ratio).mean(axis=1)
            inits.append(np.clip(small, 0.01, None))
    
    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt + 0.1) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.cos(2 * np.pi * 7 * tt + 0.3))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 1.5 * tt) + 0.3 * np.cos(2 * np.pi * 4 * tt + 1.0))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))
    
    # Triangle
    inits.append(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01))
    
    # Exponential decay
    inits.append(np.exp(-2.0 * tt))
    
    return [np.maximum(x, 0.01) for x in inits]

The idea is:
1. Compute c = conv(a, a)
2. Define weights w[k] = 1/c[k] (or some function of c[k])
3. Find a' that minimizes sum_k w[k] * c'[k] (a weighted sum) subject to a' >= 0, sum(a') = S
4. This is a QP (quadratic program) that can be solved efficiently
5. Iterate

The weighted sum is convex in a (it's a quadratic form), so each iteration has a unique minimum. The idea is that by weighting the large c[k] more heavily, we push them down.

Actually, minimizing sum_k w[k] * c[k] = sum_k w[k] * (a*a)[k] = a^T (W * circulant) a, which is a quadratic form. If the matrix is positive definite, this is convex.

**Simpler approach: Use the gradient of the weighted sum**

The gradient of f(a) = sum_k w[k] * (a*a)[k] with respect to a is:
∂f/∂a[i] = 2 * sum_k w[k] * (∂(a*a)[k]/∂a[i])

Now (a*a)[k] = sum_j a[j]*a[k-j], so ∂(a*a)[k]/∂a[i] = a[k-i] + a[i] * (if k-i is valid, counting both terms)

Actually, more carefully:
(a*a)[k] = sum_j a[j]*a[k-j]
∂(a*a)[k]/∂a[i] = a[k-i] (from the j=i term) + a[i] (from the k-j=i term, i.e., j=k-i)
But we need to be careful about double-counting when i = k-i, i.e., i = k/2.

So ∂f/∂a[i] = 2 * sum_k w[k] * (a[k-i] + a[i]) (with appropriate bounds and double-counting correction)

= 2 * sum_k w[k] * a[k-i] + 2 * a[i] * sum_k w[k]

The first term is a convolution of w and a (with appropriate reflection), and the second term is just a[i] times a constant.

So the gradient is: grad[i] = 2 * (w_reflected * a)[i] + 2 * W_sum * a[i]

where w_reflected is w reversed (or appropriately shifted) and * denotes convolution.

This can be computed in O(n log n) using FFT!

So the algorithm is:
1. Compute c = conv(a, a)
2. Set w[k] = some function of c[k] (e.g., w[k] = c[k]^p for p > 0, or w[k] = 1/c[k])
3. Compute grad = 2 * conv(w_reflected, a) + 2 * sum(w) * a
4. Take a step in the negative gradient direction
5. Project onto the simplex
6. Repeat

This is a quadratic minimization at each step (the "local" model is the quadratic form with Hessian 2*Q_w). The step size can be chosen via line search.

Actually, this is essentially a Newton's method where the Hessian is approximated by Q_w at the current point. But since Q_w changes at each step, it's more like a "variable metric" method.

def run_quadratic_reweighting(a_init, S, max_time, rng, p=2.0):
    """
    Iterative quadratic reweighting: at each step, minimize a weighted
    quadratic approximation of the max-convolution objective.
    """
    a = normalize_to_sum(a_init, S)
    n = len(a)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        # Compute weights: emphasize large c[k]
        # Use a "soft" weight: w[k] = (c[k] / mean(c))^p
        mean_c = c.mean()
        if mean_c < 1e-20:
            break
        w = (c / mean_c) ** p
        
        # Compute gradient of sum_k w[k] * c[k]
        # grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds) + 2 * sum(w) * a[i]
        # The first term is a "cross-correlation" of w and a
        # In FFT terms: it's ifft(fft(w_flipped) * fft(a))
        
        # w has length 2n-1, a has length n
        # The cross-correlation (w * a)[i] = sum_k w[k] * a[k-i] (with valid bounds)
        # This is: sum_k w[k] * a[k-i] for k-i in [0, n-1]
        # = sum_j w[j+i] * a[j] for j in [0, n-1] and j+i in [0, 2n-2]
        
        # In FFT: ifft(fft(w, 2fsize) * fft(a, 2fsize)) gives circular correlation
        # We need to be careful about the lengths
        
        # Let me use a direct approach with FFT:
        # conv(w, a) has length (2n-1) + n - 1 = 3n - 2
        # The element at index (2n-2+i) is sum_j w[2n-2+i-j] * a[j]
        # = sum_j w[j+i] * a[j] (reindexing)
        # This is what we want!
        
        fsize2 = 1 << (3 * n - 1).bit_length()
        W = fft(w, fsize2)
        A = fft(a, fsize2)
        cross = np.real(ifft(W * A))
        # cross[2n-2+i] = sum_j w[j+i] * a[j]
        cross_part = cross[2*n-2 : 2*n-2+n]
        
        W_sum = w.sum()
        grad = 2.0 * (cross_part + W_sum * a)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use 1/L where L is an estimate of the Lipschitz constant
        # For the quadratic form with Hessian 2*Q_w, L ≈ 2 * lambda_max(Q_w)
        # Rough estimate: L ≈ 2 * (max(w) * n + sum(w))
        # More conservatively: L ≈ 2 * sum(w) * max(a) * 2
        L_est = 2.0 * (W_sum * 2.0 * a.max() + np.max(w) * 2.0 * S)
        step = 1.0 / max(L_est, 1e-10)
        
        # Adaptive: reduce step over time
        step *= 2.0 / (1.0 + 0.01 * it)
        
        a_new = project_simplex(a - step * grad / gnorm * min(gnorm, step * 0.5 * gnorm), S)
        # Simpler:
        a_new = project_simplex(a - step * grad, S)
        
        # Line search (backtracking)
        c_new = fft_convolve_self(a_new)
        if c_new.max() > c.max() * 1.01:
            # Try smaller step
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
        
        a = a_new
    
    return best_a, best
