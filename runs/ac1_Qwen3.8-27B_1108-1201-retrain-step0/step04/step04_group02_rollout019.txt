    """Full self-convolution via FFT."""
    n = len(a)

    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on threshold."""
    n = len(v)
    u = np.sort(v)[::-1]

    start = time.monotonic()
    
    # Initial active set
    max_c = c.max()
    threshold = max_c * (1 - delta_frac)
    K = np.where(c >= threshold)[0]
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Update active set every 50 iterations
        if it % 50 == 1:
            c = fft_convolve_self(a)
            max_c = c.max()
            threshold = max_c * (1 - delta_frac)
            K = np.where(c >= threshold)[0]
            if len(K) < 1:
                K = np.array([np.argmax(c)])
            # Limit K size for efficiency
            if len(K) > 200:
                idx = np.argsort(c[K])[::-1][:200]
                K = K[idx]
        
        # Compute gradient of (1/|K|) * sum_{k in K} (a*a)[k]
        grad = np.zeros(n)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            m_vals = np.arange(lo, hi + 1)
            grad[m_vals] += 2.0 * y[k - m_vals]
        grad /= len(K)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size: use 0.5 (since Lipschitz constant <= 2)
        # But also constrain to not overshoot
        step = 0.5
        max_step = S / gnorm  # don't change the sum too much
        step = min(step, max_step)
        
        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Evaluate
        if it % 10 == 0:
            c = fft_convolve_self(a)
            cur_obj = c.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print("    AS-FISTA it=%d, obj=%.6f, |K|=%d" % (it, best_obj, len(K)))
        
        # Periodic perturbation to escape local minima
        if it % 5000 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            y = a.copy()
            t = 1.0
    
    # Final evaluation
    c = fft_convolve_self(a)
    if c.max() < best_obj:
        best_a = a.copy()
        best_obj = c.max()
    
    return best_a, best_obj


def lse_optimize(a_init, n, S, max_time, rng, verbose=False):
    """LSE-smoothed optimization with annealing."""
    a = np.clip(a_init, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    y = a.copy()
    t = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        # Annealing schedule for beta
        progress = min(1.0, elapsed / max_time)
        beta = 5.0 + 500.0 * progress ** 2
        
        c = fft_convolve_self(y)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        # Gradient of LSE
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size with backtracking
        step = min(0.5, S / gnorm)
        x_new = project_simplex(y - step * grad, S)
        
        # Check if improvement (use actual objective)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < c.max() or rng.random() < 0.1:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            t = 1.0
            y = a.copy()
        
        if it % 20 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_obj:
                best_obj = c_check.max()
                best_a = a.copy()
                if verbose and it % 200 == 0:
                    print("    LSE it=%d, beta=%.1f, obj=%.6f" % (it, beta, best_obj))
    
    c = fft_convolve_self(a)
    if c.max() < best_obj:
        best_a = a.copy()
        best_obj = c.max()
    
    return best_a, best_obj


def coordinate_descent(a_init, n, S, max_time, rng, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, (a*a)[k] is affected for k in [i, i+n-1].
    The optimal a[i] given all others can be found by checking the peaks.
    """
    a = np.clip(a_init, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    # Precompute the "background" contribution to each lag (convolution without a[i])
    # When we change a[i] from old to new, the change in (a*a)[k] for k in [i, i+n-1] is:
    # delta_conv[k] = 2 * a_new[i] * a[k-i] - (a_new[i]^2 - a_old[i]^2) * [k==2i (approximately)]
    # Actually it's: new_conv[k] = old_conv[k] - 2*a_old[i]*a[k-i] + 2*a_new[i]*a[k-i] - (a_new[i]^2 - a_old[i]^2)
    # Wait, let me be more careful.
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # The terms involving a[i] are: a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when k-j=i, j=k-i)
    # If i != k-i (i.e., k != 2i): both terms are a[i]*a[k-i], total contribution = 2*a[i]*a[k-i]
    # If i == k-i (i.e., k == 2i): the term is a[i]*a[i] = a[i]^2, contribution = a[i]^2
    # When we change a[i] from old to new:
    # delta (a*a)[k] = 2*(a_new - a_old)*a[k-i] for k != 2i (and k-i in range)
    # delta (a*a)[k] = (a_new^2 - a_old^2) for k = 2i
    
    # This is complex to optimize exactly. Let me use a simpler approach:
    # For each coordinate, try a few candidate values and keep the best.
    
    order = list(range(n))
    
    while time.monotonic() - start < max_time:
        it += 1
        # Random order for diversity
        if it % 1 == 0:
            rng.shuffle(order)
        
        for i in order:
            if time.monotonic() - start >= max_time:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # The "effective" values at k=i+... that involve a[i]:
            # For k from i to min(i+n-1, 2n-2):
            # contribution = 2 * a[i] * a[k-i] (adjusted for the self-term)
            
            # Try reducing a[i] and redistributing
            # Simple: try a[i] * 0.5, a[i] * 0.8, a[i] * 1.2
            # and redistribute the difference to random other coordinates
            
            best_trial = None
            best_trial_obj = None
            
            for scale in [0.5, 0.7, 0.9, 1.1, 1.3]:
                new_val = old_val * scale
                if new_val < 0:
                    new_val = 0
                diff = old_val - new_val  # amount to redistribute (positive = reduce i)
                
                a_new = a.copy()
                a_new[i] = new_val
                a_new += diff / n  # spread evenly (approximate)
                a_new = np.maximum(a_new, 0)
                # Re-normalize
                s_new = a_new.sum()
                if s_new > 0:
                    a_new = a_new * (S / s_new)
                
                c_new = fft_convolve_self(a_new)
                obj_new = c_new.max()
                
                if best_trial_obj is None or obj_new < best_trial_obj:
                    best_trial_obj = obj_new
                    best_trial = a_new
            
            if best_trial is not None and best_trial_obj < best_obj * 1.001:
                a = best_trial
                if best_trial_obj < best_obj:
                    best_obj = best_trial_obj
                    best_a = a.copy()
    
    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.01))
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(np.sin(np.pi * t) ** 4)
    inits.append(np.sin(np.pi * t) ** 8)
    
    # Flat with edge taper
    flat = np.ones(n)
    taper = 10
    flat[:taper] = np.linspace(0.5, 1, taper)
    flat[-taper:] = np.linspace(1, 0.5, taper)
    inits.append(flat)
    
    # Exponential
    inits.append(np.exp(-3 * (t - 0.5) ** 2 / 0.1))
    
    # Multi-frequency
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.sin(2 * np.pi * 7 * t))
    
    # Random
    inits.append(rng.random(n) + 0.5)
    
    # Ramp
    inits.append(np.abs(t - 0.5) + 0.1)
    
    # Gaussian
    inits.append(np.exp(-10 * (t - 0.5) ** 2))
    
    return [np.maximum(x, 0.01) for x in inits]
