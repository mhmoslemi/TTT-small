def project_simplex(v, S):

    v = np.asarray(v, dtype=np.float64)

"""
Minimize 2*n*max(conv(f,f))/sum(f)^2 using adaptive LP best-response with
active-set constraint selection, symmetric exploitation, and FISTA polishing.
Key improvements: adaptive K for LP constraints, dual-guided constraint selection,
symmetric LP for final polish, and aggressive multi-phase schedule.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (a.sum() ** 2)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_constraint_matrix(a, lags):
    """Build the constraint matrix T where T[k,j] = a[k-j] if valid else 0."""
    n = len(a)
    K = len(lags)
    T = np.zeros((K, n))
    for i, k in enumerate(lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        T[i, js] = a[k - js]
    return T


def solve_lp(a, S, constraint_lags, time_limit=20.0):
    """Solve LP: max sum(g) s.t. T@g <= M, g >= 0. Returns (g, M, dual_vars)."""
    n = len(a)
    c = fft_convolve(a)
    M = c.max()
    K = len(constraint_lags)

    if K == 0:
        return None, M, None

    T = build_constraint_matrix(a, constraint_lags)

    g = cp.Variable(n)
    y = cp.Variable(K)  # dual variables
    constraints = [T @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver='SCS', verbose=False, eps=1e-7, max_iters=200000)
    except Exception:
        try:
            prob.solve(solver='PDDLP', verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                return None, M, None

    if g.value is None:
        return None, M, None

    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return None, M, None

    # Get dual values (constraint multipliers for T@g <= M)
    dual = None
    if prob.value is not None:
        # Try to get dual values from constraint
        try:
            dual = np.maximum(constraints[0].dual_value, 0.0)
        except Exception:
            dual = None

    g_val = g_val * (S / s)
    return g_val, M, dual


def find_top_lags(c, K):
    """Find indices of top K values in convolution."""
    if K >= len(c):
        return np.arange(len(c))
    return np.argsort(c)[::-1][:K]


def line_search_step(a, g, S, best_val, max_evals=12):
    """Find best step size t in [0,1] for convex combination a + t*(g-a)."""
    n = len(a)
    best_t = 0.0
    best_v = best_val

    # Try t=1 first
    a_test = project_simplex(g, S)
    v_test = compute_obj(a_test)
    if v_test < best_v:
        best_v = v_test
        best_t = 1.0

    # Ternary search in [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(max_evals - 1):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        a1 = project_simplex((1 - m1) * a + m1 * g, S)
        a2 = project_simplex((1 - m2) * a + m2 * g, S)
        v1 = compute_obj(a1)
        v2 = compute_obj(a2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        v_mid = compute_obj(project_simplex((1 - (lo+hi)/2) * a + ((lo+hi)/2) * g, S))
        if v_mid < best_v:
            best_v = v_mid
            best_t = (lo + hi) / 2

    if best_t < 0.05:
        return a.copy(), best_val
    a_new = project_simplex((1 - best_t) * a + best_t * g, S)
    return a_new, best_v


def run_fista_polish(a, n, S, max_time, rng, beta_init=30.0, beta_rate=0.003, beta_max=500.0):
    """FISTA with LSE smoothing for quick gradient-based improvement."""
    a = normalize(a, n).copy()
    y = a.copy()
    momentum = 1.0
    start = time.monotonic()
    best = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve(y)
        cur = compute_obj(y)
        if cur < best - 1e-8:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            momentum = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum))
        y = x_new + ((momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        momentum = t_new

    return best_a, best


def run_lp_search(a, n, S, max_time, rng, K_start=80, K_end=400, lp_interval=1):
    """
    Main LP-based search with adaptive constraint count.
    Alternates LP best-response with gradient steps.
    """
    a = normalize(a, n).copy()
    start = time.monotonic()
    best = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    active_lags = find_top_lags(fft_convolve(a), K_start)

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac = elapsed / max_time
        # Adaptive K: increase over time
        K = int(K_start + (K_end - K_start) * min(1.0, frac * 1.5))
        K = max(K, K_start)

        c = fft_convolve(a)
        cur = compute_obj(a)
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it

        # LP best response
        if it % lp_interval == 1:
            # Update active lags
            top_lags = find_top_lags(c, K)
            # Also include previously active lags for stability
            if it > 5:
                # Keep some overlap with previous active set
                combined = np.unique(np.concatenate([active_lags, top_lags]))
                if len(combined) > K * 1.5:
                    combined = combined[:K]
                active_lags = combined
            else:
                active_lags = top_lags

            t_lp_start = time.monotonic()
            g, M, dual = solve_lp(a, S, active_lags, time_limit=min(15.0, max_time * 0.05))
            lp_time = time.monotonic() - t_lp_start

            if g is not None:
                # Line search for best step
                a_new, val = line_search_step(a, g, S, best, max_evals=10)
                if val < best:
                    best = val
                    best_a = a_new.copy()
                    a = a_new
                    last_improve = it
                    if it % 10 == 0:
                        print("    LP iter %d: obj=%.8f, K=%d, lp_time=%.1fs, t_found=%.3f" %
                              (it, best, K, lp_time, 1.0))
                elif it % 20 == 0:
                    print("    LP iter %d: no improvement (best=%.8f, lp_time=%.1fs)" %
                          (it, best, lp_time))

            # If no LP improvement, do a gradient step
            if val >= best and cur - best < 1e-6:
                beta = 50.0 + 100.0 * frac
                grad = lse_gradient(a, c, beta)
                step = 1.0 / (2.0 * beta)
                a = project_simplex(a - step * grad, S)

        # Stuck detection: add noise
        if it - last_improve > 50:
            noise_scale = 0.002 * S / np.sqrt(n) * (0.5 + 0.5 * np.exp(-it / 200.0))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it
            active_lags = find_top_lags(fft_convolve(a), K)

    return best_a, best


def run_symmetric_lp(a, n, S, max_time, K=200):
    """
    Optimize a symmetric sequence using LP.
    Symmetric means a[j] = a[n-1-j].
    """
    if n % 2 != 0:
        return a, compute_obj(a)

    m = n // 2
    # Symmetrize
    a_sym = (a[:m] + a[n-1:n-m-1:-1]) / 2.0
    a_sym = project_simplex(a_sym, S / 2.0)
    # Expand to full sequence
    a_full = np.concatenate([a_sym, a_sym[::-1]])
    a_full = project_simplex(a_full, S)

    start = time.monotonic()
    best = compute_obj(a_full)
    best_a = a_full.copy()

    for it in range(200):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve(a_full)
        cur = compute_obj(a_full)
        if cur < best:
            best = cur
            best_a = a_full.copy()

        # Get top K lags (only need lags 0..n-1 for symmetric)
        top_lags = find_top_lags(c[:n], K)

        # Build symmetric constraint matrix
        # Variables: h_0..h_{m-1}, where a[j] = h[min(j, n-1-j)]
        # For constraint lag k: sum_j a[j]*h[min(k-j, n-1-k+j)] <= M
        M = c.max()
        T = np.zeros((len(top_lags), m))
        for i, k in enumerate(top_lags):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                h_idx = min(j, n - 1 - j)
                if h_idx < m:
                    T[i, h_idx] += a_full[j]

        # LP: max sum(h) s.t. T@h <= M, h >= 0
        h = cp.Variable(m)
        constraints = [T @ h <= M, h >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(h)), constraints)
        try:
            prob.solve(solver='SCS', verbose=False, eps=1e-7, max_iters=200000)
        except:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except:
                break

        if h.value is None:
            break

        h_val = np.maximum(h.value, 0.0)
        s = h_val.sum()
        if s < 1e-10:
            break
        h_val = h_val * (S / (2.0 * s))

        # Expand
        g_full = np.concatenate([h_val, h_val[::-1]])
        g_full = project_simplex(g_full, S)

        # Line search
        a_new, val = line_search_step(a_full, g_full, S, best, max_evals=8)
        if val < best:
            a_full = a_new
            best = val
            best_a = a_new.copy()
        else:
            # Try smaller step
            for t in [0.3, 0.1, 0.05]:
                a_test = project_simplex((1-t)*a_full + t*g_full, S)
                v = compute_obj(a_test)
                if v < best:
                    a_full = a_test
                    best = v
                    best_a = a_test.copy()
                    break

    return best_a, best

        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback evaluate: %.8f" % ver)
    except Exception as e:
        print("  Error in evaluate: %s, using internal value" % str(e))

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), min(final_obj, ver if np.isfinite(ver) else np.inf)))
    return final_list
