where `LSE_beta(c) = (1/beta) * log(sum(exp(beta * c)))` and `S = sum(a)`.

`df/da = (2n/S^2) * dLSE/da + 2n * d(1/S^2)/da * LSE_beta(c)`
`= (2n/S^2) * dLSE/da - (2n * 2 * LSE_beta(c) / S^3) * dS/da`
`= (2n/S^2) * dLSE/da - (4n * LSE_beta(c) / S^3) * 1`

where `dLSE/da = 2 * ifft(FFT(w, N) * conj(FFT(a, N)))[:n]` and `dS/da = 1` (the all-ones vector).

So: `grad_f = (2n/S^2) * grad_LSE - (4n * LSE_val / S^3) * ones`

where `LSE_val = (1/beta) * log(sum(exp(beta*c)))`.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-smoothed L-BFGS-B refinement,
adaptive subgradient descent, and multi-length search (8192, 16384).
Uses "one-sided" structure prior for initializations and frequency-aware
gradient computation via FFT.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def lse_objective_and_grad(a, beta, lambda_scale=0.01):
    """
    Compute LSE-smoothed objective and its gradient.
    Objective: 2n * LSE_beta(c) / S^2 + lambda_scale * (S - S_target)^2
    where c = convolve(a,a), S = sum(a), S_target = sqrt(2n).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        e = np.zeros(len(c))
        e[int(np.argmax(c))] = 1.0
        tot = 1.0
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    
    # Gradient of LSE part: 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_lse = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Full gradient
    S_target = np.sqrt(2.0 * n)
    obj = 2.0 * n * lse_val / (S ** 2) + lambda_scale * (S - S_target) ** 2
    grad = (2.0 * n / (S ** 2)) * grad_lse - (4.0 * n * lse_val / (S ** 3)) * np.ones(n)
    grad += 2.0 * lambda_scale * (S - S_target) * np.ones(n)
    
    return obj, grad

def make_initializations(n, rng):
    """Generate diverse initializations with one-sided structure."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    # Exponential decays (one-sided)
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(-alpha * t))
    
    # Gaussian bumps at different positions
    for center in [0.15, 0.25, 0.35, 0.5]:
        for width in [0.1, 0.15, 0.25]:
            inits.append(np.exp(-((t - center) / width) ** 2))
    
    # Cosine windows on first half
    for p in [1.0, 1.5, 2.0, 3.0]:
        mask = t < 0.5
        val = np.zeros(n)
        val[mask] = np.cos(np.pi * t[mask] * 2) ** p
        inits.append(np.maximum(val, 0.001))
    
    # Raised cosine (full)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    
    # Triangle (one-sided)
    tri = np.zeros(n)
    half = n // 2
    tri[:half] = np.arange(half) / half
    tri[half:] = np.arange(half, 0, -1) / half
    inits.append(tri)
    
    # Power-law decay
    for p in [0.5, 1.0, 1.5, 2.0]:
        inits.append((1 - t) ** p)
    
    # Multi-frequency modulations
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) * np.exp(-2 * t))
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * t) * np.exp(-3 * t))
    
    # Random
    inits.append(rng.random(n) + 0.01)
    
    # Edge-heavy
    inits.append(1.0 + 3.0 * (t ** 6 + (1 - t) ** 6))
    
    # Sinc-like (one-sided)
    inits.append(np.sinc(5 * t) ** 2)
    
    return [np.maximum(x, 0.001) for x in inits]


def structured_perturb(a, n, S, rng):
    """Apply structured perturbation."""
    strategy = rng.integers(0, 4)
    a_new = a.copy()
    t = np.arange(n) / n
    
    if strategy == 0:
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t + phase)
        a_new = a * mod
    elif strategy == 1:
        mod = 1.0 + 0.2 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
        a_new = a * mod
    elif strategy == 2:
        # Widen or narrow the main lobe
        factor = rng.uniform(0.5, 2.0)
        t_new = t * factor
        mask = t_new < 1.0
        a_new = np.zeros(n)
        # Interpolate
        idx_old = np.minimum((t_new * (n - 1)).astype(int), n - 1)
        a_new[mask] = a[idx_old[mask]]
    else:
        # Frequency domain perturbation
        A = fft(a)
        noise = rng.normal(0, 0.03, n) + 1j * rng.normal(0, 0.03, n)
        a_new = np.real(ifft(A * (1 + noise)))
    
    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

    # For each length, try several initializations with subgradient
    n_inits_per_len = 6
    time_per_init = explore_time / (len(lengths_to_try) * n_inits_per_len)
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - total_time * 0.55:
            break
        S = np.sqrt(2.0 * n)
        
        # Build initialization list
        init_list = []
        if given is not None and len(given) == n:
            init_list.append(given.copy())
        elif given is not None:
            # Resample given to length n
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(given))
            x_new = np.linspace(0, 1, n)
            f = interp1d(x_old, given, kind='cubic', fill_value='extrapolate', bounds_error=False)
            init_list.append(np.maximum(f(x_new), 0.0))
        
        gen_inits = make_initializations(n, rng)
        # Pick a subset of the best-looking ones
        for i in range(min(n_inits_per_len, len(gen_inits))):
            idx = min(i * 3, len(gen_inits) - 1)  # spread out selection
            init_list.append(gen_inits[idx])
        
        # Add a structured perturbation of the given
        if given is not None and len(given) == n:
            init_list.append(structured_perturb(given, n, S, rng))
        
        for init_i, init_arr in enumerate(init_list):
            if time.monotonic() > deadline - total_time * 0.55:
                break
            
            t_alloc = min(time_per_init * 2, deadline - time.monotonic() - total_time * 0.55)
            if t_alloc < 3:
                break
            
            a_opt, score = run_subgradient(init_arr, n, S, t_alloc, rng)
            a_norm = normalize_sum(a_opt, n)
            real_score = compute_objective(a_norm)
            
            if real_score < global_best:
                global_best = real_score
                global_a = a_norm.copy()
                global_n = n
                print("  [n=%d, init%d] obj=%.6f (elapsed %.0fs)" % (
                    n, init_i, global_best, time.monotonic() - p1_start))
    
    print("=== After Phase 1: best=%.6f, n=%d ===" % (global_best, global_n))
    
    # ==================== PHASE 2: L-BFGS Refinement ====================
    remaining = deadline - time.monotonic()
    print("=== Phase 2: L-BFGS Refinement (remaining %.0fs) ===" % remaining)
    
    if global_a is not None and remaining > 30:
        n_best = global_n
        # Allocate time between L-BFGS on current best and trying other lengths
        lbfgs_alloc = min(remaining * 0.6, lbfgs_time)
        
        # Run L-BFGS on the best solution
        a_l, obj_l = run_lbfgs(global_a, n_best, lbfgs_alloc, rng)
        if obj_l < global_best:
            global_best = obj_l
            global_a = a_l.copy()
            print("  After L-BFGS (n=%d): obj=%.6f" % (n_best, global_best))
        
        # Also try L-BFGS on a different length if time permits
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            for n_try in lengths_to_try:
                if n_try == n_best:
                    continue
                if remaining2 < 30:
                    break
                S_try = np.sqrt(2.0 * n_try)
                # Quick subgradient to get a starting point
                t_quick = min(20, remaining2 * 0.3)
                init_try = make_initializations(n_try, rng)[2].copy()
                if given is not None:
                    from scipy.interpolate import interp1d
                    x_old = np.linspace(0, 1, len(given))
                    x_new = np.linspace(0, 1, n_try)
                    f = interp1d(x_old, given, kind='cubic', fill_value='extrapolate', bounds_error=False)
                    init_try = np.maximum(f(x_new), 0.0)
                
                a_q, s_q = run_subgradient(init_try, n_try, S_try, t_quick, rng)
                a_q = normalize_sum(a_q, n_try)
                s_q = compute_objective(a_q)
                
                if remaining2 - (time.monotonic() + t_quick) > 20:
                    t_lbfgs = remaining2 * 0.5
                    a_r, s_r = run_lbfgs(a_q, n_try, t_lbfgs, rng)
                    s_r = compute_objective(a_r)
                    if s_r < global_best:
                        global_best = s_r
                        global_a = a_r.copy()
                        global_n = n_try
                        print("  L-BFGS n=%d: obj=%.6f" % (n_try, global_best))
                
                remaining2 = deadline - time.monotonic()
    
    # ==================== PHASE 3: Polish ====================
    remaining = deadline - time.monotonic()
    print("=== Phase 3: Polish (remaining %.0fs) ===" % remaining)
    
    if global_a is not None and remaining > 20:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        
        # Micro-optimization with very small steps
        a_micro = global_a.copy()
        micro_start = time.monotonic()
        step = 0.01
        best_micro = global_best
        it = 0
        stall = 0
        
        while time.monotonic() - micro_start < remaining - 10:
            it += 1
            c = fft_convolve_self(a_micro)
            cur = 2.0 * n_best * c.max() / S_best**2
            
            if cur < best_micro - 1e-12:
                best_micro = cur
                stall = 0
            else:
                stall += 1
            
            if stall > 500:
                break
            
            # Top-k subgradient with varying k
            if it % 100 < 10:
                k = 1
            elif it % 100 < 50:
                k = 8
            else:
                k = 32
            grad = topk_subgradient(a_micro, c, k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            eff_step = step / max(1.0, 0.01 * stall)
            a_try = project_simplex(a_micro - eff_step * grad, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best**2
            
            if obj_try < cur:
                a_micro = a_try
                step = min(step * 1.05, 0.5)
            else:
                step *= 0.9
                if step < 0.001:
                    noise = 0.001 * S_best / np.sqrt(n_best)
                    a_micro = project_simplex(a_micro + rng.normal(0, noise, n_best), S_best)
                    step = 0.02
            
            if it % 2000 == 0:
                print("    micro it=%d, best=%.6f" % (it, best_micro))
        
        a_micro = normalize_sum(a_micro, n_best)
        s_micro = compute_objective(a_micro)
        if s_micro < global_best:
            global_best = s_micro
            global_a = a_micro.copy()
            print("  After micro: %.6f" % global_best)
        
        # One more L-BFGS pass if time
        remaining_final = deadline - time.monotonic()
        if remaining_final > 30:
            a_fin, obj_fin = run_lbfgs(global_a, n_best, remaining_final - 10, rng)
            obj_fin = compute_objective(a_fin)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  Final L-BFGS: %.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("Returning length=%d, obj=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
