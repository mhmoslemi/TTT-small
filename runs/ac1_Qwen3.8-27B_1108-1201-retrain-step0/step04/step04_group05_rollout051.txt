"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach:

3. Random 1D cuts with exact line search for directed improvement

- 1D cuts exploit the quadratic structure of convolution in one direction

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_cross(a, b, N):
    """Cross-convolution a*b via FFT. Returns [2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    b_pad = np.zeros(N)
    b_pad[:n] = b
    A = fft(a_pad)
    B = fft(b_pad)
    c = np.real(ifft(A * B))[:2 * n - 1]
    return c


def lse_smoothing(c, beta):
    """Compute LSE value and soft weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.full(len(c), 1.0 / len(c))
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return lse_val, w


def lse_gradient(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a)."""
    n = len(a)
    c = compute_conv(a, N)
    lse_val, w = lse_smoothing(c, beta)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c.max()


def lbfgs_direction(s_hist, y_hist, g):

    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-30)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):

        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective."""
    a = project_simplex(a_init, S)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_gradient(a, N, beta)
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.10)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_gradient(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        direction = lbfgs_direction(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a.copy()
        cmax_new = c_max
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw, cmax_nw = lse_gradient(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 800 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_gradient(a, N, beta)

    return best_a, best_cmax


def spectral_equalization(a, N, n, S, step_size=0.3, n_iter=50, rng=None):
    """
    POCS-style spectral equalization:
    Target flat power spectrum |A[k]|^2 = S^2 for all k.
    Alternates between frequency-domain correction and time-domain projection.
    """
    a = a.copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()

    for it in range(n_iter):
        # Frequency domain
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        target = S ** 2  # target uniform power

        # Gradually flatten: reduce peaks, boost valleys
        P_ratio = P / (target + 1e-10)
        # Damp peaks (where P_ratio > 1) and leave valleys alone
        correction = 1.0 - step_size * np.clip(P_ratio - 1.0, 0, None)
        correction = np.clip(correction, 0.3, 1.0)
        P_new = P * correction
        # Also boost the DC to maintain sum
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        # Preserve DC
        A_new[0] = A[0]

        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        c = compute_conv(a_new, N)
        cm = c.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = a_new.copy()
        a = a_new

    return best_a, best_cmax


def spectral_equalization_aggressive(a, N, n, S, rng, max_time=60.0):
    """Aggressive spectral equalization with adaptive step and restarts."""
    a = a.copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    step_size = 0.5

    while time.monotonic() - start_time < max_time:
        target = S ** 2
        for sub in range(20):
            a_pad = np.zeros(N)
            a_pad[:n] = a
            A = fft(a_pad)
            P = np.abs(A) ** 2
            P_ratio = P / (target + 1e-10)
            # More aggressive: strong damping of peaks
            correction = 1.0 - step_size * np.clip(P_ratio - 0.5, 0, None)
            correction = np.clip(correction, 0.15, 1.0)
            P_new = P * correction
            phase = np.angle(A)
            A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
            A_new[0] = A[0]
            a_new = np.real(ifft(A_new))[:n]
            a = project_simplex(a_new, S)

        c = compute_conv(a, N)
        cm = c.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = a.copy()
            step_size = max(0.1, step_size * 0.95)
        else:
            step_size = min(0.8, step_size * 1.3)
            if rng is not None and rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def one_dim_cut(a, d, N, n, S, max_step=1.0):
    """
    1D line search along direction d.
    a(t) = project_simplex(a + t*d, S)
    Minimize max_k (c_k + 2t*cross_k + t^2*self_k)
    where cross = conv(a,d), self = conv(d,d).
    """
    c = compute_conv(a, N)
    cross = compute_conv_cross(a, d, N)
    self_c = compute_conv(d, N)

    # Find valid t range (a + t*d >= 0)
    pos_mask = d > 1e-12
    neg_mask = d < -1e-12
    if np.any(pos_mask):
        t_lower = max(-max_step, np.max(-a[pos_mask] / d[pos_mask]))
    else:
        t_lower = -max_step
    if np.any(neg_mask):
        t_upper = min(max_step, np.min(-a[neg_mask] / d[neg_mask]))
    else:
        t_upper = max_step
    t_lower = max(t_lower, -max_step)
    t_upper = min(t_upper, max_step)
    if t_upper <= t_lower:
        return a.copy(), c.max()

    # Evaluate objective at candidate t values
    def obj_at_t(t):
        c_t = c + 2.0 * t * cross + t**2 * self_c
        return c_t.max()

    # Candidate points: 0, vertices of top peaks, and grid
    c_max_cur = c.max()
    top_k = np.argsort(c)[::-1][:50]
    candidates = [0.0, t_lower, t_upper]
    for k in top_k:
        if abs(self_c[k]) > 1e-20:
            t_v = -cross[k] / self_c[k]
            if t_lower <= t_v <= t_upper:
                candidates.append(t_v)

    # Grid points
    n_grid = 30
    candidates.extend(np.linspace(t_lower, t_upper, n_grid).tolist())

    best_t = 0.0
    best_val = obj_at_t(0.0)
    for t in candidates:
        if t_lower - 1e-12 <= t <= t_upper + 1e-12:
            t_clipped = np.clip(t, t_lower, t_upper)
            val = obj_at_t(t_clipped)
            if val < best_val:
                best_val = val
                best_t = t_clipped

    # Refine with golden section around best_t
    span = (t_upper - t_lower) * 0.1
    ga = t_lower + (t_upper - t_lower) * 0.618
    gb = t_lower + (t_upper - t_lower) * 0.382
    a_lo = max(t_lower, best_t - span)
    a_hi = min(t_upper, best_t + span)
    if a_hi > a_lo:
        ga = a_lo + (a_hi - a_lo) * 0.618
        gb = a_lo + (a_hi - a_lo) * 0.382
        fa = obj_at_t(ga)
        fb = obj_at_t(gb)
        for _ in range(40):
            if fa < fb:
                a_hi = gb
                gb = ga
                fb = fa
                ga = a_lo + (a_hi - a_lo) * 0.618
                fa = obj_at_t(ga)
            else:
                a_lo = ga
                ga = gb
                fa = fb
                gb = a_lo + (a_hi - a_lo) * 0.382
                fb = obj_at_t(gb)
        t_star = (a_lo + a_hi) / 2.0
        val_star = obj_at_t(t_star)
        if val_star < best_val:
            best_val = val_star
            best_t = t_star

    a_new = project_simplex(a + best_t * d, S)
    return a_new, best_val


def random_cut_block(a_init, n, S, N, max_time, rng, n_cuts=200):
    """Block of random 1D cuts."""
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()

    for it in range(n_cuts):
        if time.monotonic() - start_time > max_time:
            break

        # Mix of random and gradient-based directions
        if it % 3 == 0:
            # Gradient direction (subgradient of max)
            c = compute_conv(a, N)
            k_star = c.argmax()
            d = np.zeros(n)
            valid = (k_star >= 0) & (k_star < 2 * n - 1)
            if valid:
                j_start = max(0, k_star - n + 1)
                j_end = min(n - 1, k_star)
                for j in range(j_start, j_end + 1):
                    d[j] = -2.0 * a[k_star - j]
            d = d - d.mean()  # zero mean
        elif it % 3 == 1:
            # Multi-peak gradient
            c = compute_conv(a, N)
            top_idx = np.argsort(c)[::-1][:10]
            d = np.zeros(n)
            for k in top_idx:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                w = (c[k] - c.min()) / max(c.max() - c.min(), 1e-10)
                for j in range(j_start, j_end + 1):
                    d[j] -= w * a[k - j]
            d = d - d.mean()
        else:
            # Random direction
            d = rng.standard_normal(n)
            d = d - d.mean()

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            continue
        d = d / d_norm * (S / np.sqrt(n))

        a_new, val = one_dim_cut(a, d, N, n, S, max_step=0.5)
        c_new = compute_conv(a_new, N)
        cm = c_new.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = a_new.copy()
        a = a_new

    return best_a, best_cmax


def level_set_block(a_init, n, S, N, max_time, rng, m_init=None):
    """
    Level-set approach: bisection on target peak M.
    For a given M, try to find a with max(conv) <= M.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()

    if m_init is None:
        m_init = best_cmax * 0.99

    # Bisection: find minimum feasible M
    m_lo = best_cmax * 0.85
    m_hi = best_cmax * 1.01

    for outer in range(15):
        if time.monotonic() - start_time > max_time:
            break
        m_target = 0.5 * (m_lo + m_hi)
        # Try to achieve max(conv) <= m_target
        a_try = a.copy()
        success = False
        for inner in range(300):
            if time.monotonic() - start_time > max_time:
                break
            c = compute_conv(a_try, N)
            cm = c.max()
            if cm <= m_target + 1e-10:
                success = True
                break
            # Reduce peak: subgradient step
            top_k = np.argsort(c)[::-1][:20]
            d = np.zeros(n)
            for k in top_k:
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                w = c[k] - m_target
                if w > 0:
                    for j in range(j_start, j_end + 1):
                        d[j] += w * a_try[k - j]
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-15:
                break
            step = 0.1 * S / (n * d_norm) * min(1.0, max(0.01, (cm - m_target) / m_target))
            a_try = project_simplex(a_try - step * d, S)
            c_try = compute_conv(a_try, N)
            if c_try.max() < best_cmax:
                best_cmax = c_try.max()
                best_a = a_try.copy()

        if success:
            m_hi = m_target
            a = a_try.copy()
        else:
            m_lo = m_target

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.35 * np.sin(2 * np.pi * (7 * tt + 0.5 * tt**2)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 33]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Initialize from random spectrum with controlled flatness."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    # Flatten toward uniform
    P_mean = P.mean()
    P = 0.3 * P + 0.7 * P_mean
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    # Make phase symmetric for real signal
    phase[0] = 0
    phase[N // 2] = 0 if N % 2 == 0 else 1
    for k in range(1, N // 2 + 1):
        if k < N - k:
            phase[N - k] = -phase[k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid search: LSE-LBFGS + Spectral Equalization + 1D Cuts + Level Set.
    Multi-length exploration with focused refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_best(a_candidate, n_cand, label=""):
        nonlocal global_best, global_a, global_n
        S_cand = np.sqrt(2.0 * n_cand)
        N_cand = 1 << (2 * n_cand - 1).bit_length()
        a_norm = safe_normalize(a_candidate, n_cand, S_cand)
        c = compute_conv(a_norm, N_cand)
        cmax = c.max()
        score = 2.0 * n_cand * cmax / (S_cand ** 2)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            global_n = n_cand
            print("  [%s] n=%d score=%.6f cmax=%.6f" % (label, n_cand, score, cmax))
        return score

    # Try to load previous best
    prev_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            prev_a = given
            print("Loaded previous sequence of length %d" % len(given))
    except Exception:
        pass

    # Determine which lengths to try
    n_options = [4096, 8192, 16384]
    if prev_a is not None:
        prev_n = len(prev_a)
        if prev_n not in n_options:
            n_options.insert(0, prev_n)
        # Prioritize the previous length
        n_options = [prev_n] + [x for x in n_options if x != prev_n]

    print("n_options: %s, budget=%.0fs" % (n_options, budget_s))

    # ========== Phase 1: Quick exploration across lengths ==========
    print("=== Phase 1: Multi-length exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(400.0, (deadline - t1) * 0.15)
    per_n = phase1_budget / len(n_options)

    for n in n_options:
        if time.monotonic() > deadline - 300:
            break
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        t0 = time.monotonic()
        rem = max(3.0, per_n - (time.monotonic() - t0))

        # Build initializations
        inits = []
        if prev_a is not None and len(prev_a) == n:
            inits.append(("prev", np.clip(prev_a, 0, 1000)))
        inits.append(("rand", make_init_rand(n, rng)))
        inits.append(("chirp", make_init_chirp(n, rng)))
        inits.append(("fourier", make_init_fourier(n, rng, N)))

        per_init = rem / len(inits)
        for name, arr in inits:
            if time.monotonic() > deadline - 300:
                break
            a_init = safe_normalize(arr, n, S)
            # Quick LSE-LBFGS
            a_l, sc_l = lbfgs_block(a_init, n, S, N, min(per_init, 60.0), rng,
                                     beta_init=5.0, beta_max=80.0, m_lbfgs=8)
            update_best(a_l, n, "P1/%s" % name)

    print("Phase 1 done. Best: %.6f (n=%d)" % (global_best, global_n))

    # ========== Phase 2: Focused optimization on best length ==========
    if global_a is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        global_a = np.ones(n) * S / n
        global_n = n
        global_best = 2.0

    n_best = global_n
    S_best = np.sqrt(2.0 * float(n_best))
    N_best = 1 << (2 * n_best - 1).bit_length()

    print("=== Phase 2: Focused optimization (n=%d) ===" % n_best)
    t2 = time.monotonic()
    phase2_budget = min(1400.0, (deadline - t2) * 0.6)
    a_curr = global_a.copy()
    stall = 0
    last_best = global_best
    block_count = 0

    while time.monotonic() < t2 + phase2_budget:
        block_count += 1
        block_time = min(80.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 5:
            break

        # Choose strategy based on stall count and block number
        strategy = block_count % 4
        if stall >= 3:
            strategy = 3  # Force diversification

        if strategy == 0:
            # LSE-LBFGS
            beta_start = 8.0 + 4.0 * min(stall, 5)
            a_new, sc_new = lbfgs_block(a_curr, n_best, S_best, N_best, block_time,
                                         rng, beta_init=beta_start,
                                         beta_max=min(400.0, 60.0 + 20.0 * stall),
                                         m_lbfgs=12)
            label = "P2/LBFGS"
        elif strategy == 1:
            # Spectral equalization
            a_new, sc_new = spectral_equalization_aggressive(a_curr, N_best, n_best,
                                                             S_best, rng, max_time=block_time)
            label = "P2/Spectral"
        elif strategy == 2:
            # Random 1D cuts
            a_new, sc_new = random_cut_block(a_curr, n_best, S_best, N_best,
                                             block_time, rng)
            label = "P2/Cuts"
        else:
            # Level set + small perturbation
            a_curr = project_simplex(a_curr + rng.normal(0, 0.006 * S_best / np.sqrt(n_best), n_best), S_best)
            a_new, sc_new = level_set_block(a_curr, n_best, S_best, N_best,
                                            max_time=block_time, rng=rng)
            label = "P2/LevelSet"

        if sc_new < global_best - 1e-9:
            score = 2.0 * n_best * sc_new / (S_best ** 2)
            update_best(a_new, n_best, label)
            a_curr = a_new.copy()
            stall = 0
        else:
            # Still adopt if it's not much worse (avoids getting stuck)
            if sc_new < global_best * 1.02:
                a_curr = a_new.copy()
            else:
                a_curr = a_new.copy()
            stall += 1
            if stall >= 4:
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S_best / np.sqrt(n_best), n_best), S_best)
                stall = 0

    # ========== Phase 3: Intensive spectral + cut hybrid ==========
    print("=== Phase 3: Intensive hybrid ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.2)

    # Alternate spectral and cuts in short blocks
    a_h = global_a.copy()
    n_sub = 8
    sub_budget = phase3_budget / n_sub
    for i in range(n_sub):
        if time.monotonic() > deadline - 120:
            break
        if i % 2 == 0:
            a_h, sc = spectral_equalization_aggressive(a_h, N_best, n_best, S_best,
                                                       rng, max_time=sub_budget)
        else:
            a_h, sc = random_cut_block(a_h, n_best, S_best, N_best,
                                       sub_budget, rng)
        update_best(a_h, n_best, "P3/hybrid%d" % i)

    # ========== Phase 4: Final L-BFGS polish ==========
    print("=== Phase 4: Final polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        a_fin, sc_fin = lbfgs_block(global_a, n_best, S_best, N_best,
                                     min(remaining - 10, 300.0), rng,
                                     beta_init=30.0, beta_max=500.0, m_lbfgs=12)
        update_best(a_fin, n_best, "P4/final")

        # One more short cut block
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_cut, sc_cut = random_cut_block(global_a, n_best, S_best, N_best,
                                             min(remaining - 5, 120.0), rng)
            update_best(a_cut, n_best, "P4/cuts")

    # ========== Verification ==========
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
            if not np.isfinite(ver):
                print("  Rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("  Verification error: %s" % str(e))
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, internal_score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
