Combined approach: Active-set subgradient, spectral flattening,

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    # We can compute this via convolution with the indicator of active set

    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    a = normalize_sum(a, n)

        # In frequency domain: flatten non-DC magnitudes

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses adaptive FISTA with LSE smoothing, active-set subgradient,
spectral flattening (Gerchberg-Saxton), and multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def active_subgradient(a, c, rel_eps=0.02):
    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(len(a))
    active = c >= max_c - rel_eps * max_c
    active_f = active.astype(np.float64)
    n = len(a)
    fsize = 1 << (len(active_f) + n - 1).bit_length()
    W = fft(active_f, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    count = max(1, int(active.sum()))
    grad /= count
    return grad

def spectral_flatten(a, n, S, n_iter=30):
    """GS-style spectral flattening."""
    a = np.array(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        A_abs = np.abs(A)
        A_phase = np.ones(fsize, dtype=np.complex128)
        mask = A_abs > 1e-15
        A_phase[mask] = A[mask] / A_abs[mask]
        
        # Flatten non-DC magnitudes toward their mean
        non_dc = A_abs[1:]
        target = np.mean(non_dc) if len(non_dc) > 0 else 0
        
        # Blend: move 50% toward target
        blend = 0.5
        new_mags = A_abs.copy()
        new_mags[0] = S
        new_mags[1:] = (1 - blend) * non_dc + blend * target
        
        A_new = new_mags * A_phase
        A_new[0] = S  # Ensure DC is correct
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = project_simplex(a_new, S)
    
    return a

def multiplicative_repulsion(a, n, S, beta=40.0, lr=0.05):
    c = fft_convolve_self(a)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    a_new = a * np.exp(-lr * rep / max(a.max(), 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    return project_simplex(a_new, S)

def interpolate_sequence(a, new_n):
