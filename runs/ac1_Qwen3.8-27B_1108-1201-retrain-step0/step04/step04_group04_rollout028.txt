"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization:
1. FISTA with high-beta LSE smoothing and backtracking
2. Targeted coordinate descent on top peaks
3. Final polish with very high beta FISTA + coordinate descent
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_conv_dir(a, j, m, n):
    """Compute d[k] = g_j(k) - g_m(k) for all k=0..2n-2.
    g_j(k) = (a[j] + a[k-j]) if max(0,k-n+1) <= j <= min(k,n-1), else 0.
    """
    c_size = 2 * n - 1
    d = np.zeros(c_size)
    
    # For lag k, j is in S(k) iff max(0, k-n+1) <= j <= min(k, n-1)
    # This means: k-n+1 <= j (if k >= n) and j <= k (always if j < n and k >= j)
    # and j >= 0 (always) and j <= n-1 (always)
    # So: j >= max(0, k-n+1) and j <= min(k, n-1)
    # Equivalent: k >= j and k <= j + n - 1
    # So for index j, the lags where it's active are k = j, j+1, ..., j+n-1 (bounded to [0, 2n-2])
    
    lo_j = j  # k >= j
    hi_j = min(j + n - 1, 2 * n - 2)  # k <= j + n - 1
    
    lo_m = m
    hi_m = min(m + n - 1, 2 * n - 2)
    
    # For k in [lo_j, hi_j]: g_j(k) = a[j] + a[k-j]
    # For k in [lo_m, hi_m]: g_m(k) = a[m] + a[k-m]
    
    if lo_j <= hi_j:
        ks = np.arange(lo_j, hi_j + 1)
        d[lo_j:hi_j+1] += a[j] + a[ks - j]
    
    if lo_m <= hi_m:
        ks = np.arange(lo_m, hi_m + 1)
        d[lo_m:hi_m+1] -= a[m] + a[ks - m]
    
    return d


def compute_conv_quad(a, j, m, n):
    """Compute q[k] = I[k=2j] - I[k=2m] (second order terms)."""
    c_size = 2 * n - 1
    q = np.zeros(c_size)
    if 0 <= 2 * j < c_size:
        q[2 * j] = 1.0
    if 0 <= 2 * m < c_size:
        q[2 * m] = -1.0
    return q


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=5000.0,
              beta_ramp_iters=2000, use_hybrid=True):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    c_size = 2 * n - 1
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(c_size)
            w[np.argmax(c)] = 1.0
        else:
            w = e / tot
        
        # grad = 2 * ifft(fft(w) * conj(fft(y)))[:n]
        fsize = 1 << (c_size + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking
        step = 1.0 / (2.0 * beta)
        # If no recent improvement, reduce step
        if it - last_improve > 2000:
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic noise
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coordinate_descent(a_init, n, S, max_time, rng, n_pairs_per_iter=100, target_top_k=5):
    """Efficient coordinate descent targeting top peaks."""
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    c_size = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() - start < max_time:
        improved = False
        no_improve_count = 0
        
        # Identify top k lags
        top_k_indices = np.argsort(c)[-target_top_k:]
        
        # Try pairs targeting top peaks and random pairs
        n_try = n_pairs_per_iter
        for trial in range(n_try):
            if time.monotonic() - start > max_time - 0.5:
                break
            
            # Choose pair
            if trial < n_try * 0.6 and len(top_k_indices) > 0:
                # Target a top peak: find j that minimizes contribution to that peak
                k_star = top_k_indices[trial % len(top_k_indices)]
                # g_j(k_star) = a[j] + a[k_star - j] (if valid)
                # We want to move FROM m (high contribution) TO j (low contribution)
                # Compute contributions for all j
                lo = max(0, k_star - n + 1)
                hi = min(k_star, n - 1)
                if lo > hi:
                    continue
                contribute = a[lo:hi+1] + a[k_star - (np.arange(lo, hi+1))]
                # m should have high contribution, j should have low
                m_candidates = np.where(contribute > contribute.mean())[0] + lo
                j_candidates = np.where(contribute < contribute.mean())[0] + lo
                if len(m_candidates) == 0 or len(j_candidates) == 0:
                    j = rng.integers(0, n)
                    m = rng.integers(0, n)
                else:
                    m = m_candidates[rng.integers(0, len(m_candidates))]
                    j = j_candidates[rng.integers(0, len(j_candidates))]
            else:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            
            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue
            
            # Compute direction
            d = compute_conv_dir(a, j, m, n)
            q = compute_conv_quad(a, j, m, n)
            
            # Minimize max_k [c[k] + delta*d[k] + delta^2*q[k]] over delta in [-a[j], a[m]]
            lo_db = -a[j]
            hi_db = a[m]
            
            # Golden section search
            a_g = lo_db
            b_g = hi_db
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            
            def eval_delta(dlt):
                vals = c + dlt * d
                if q.any():
                    vals = vals + dlt * dlt * q
                return vals.max()
            
            fc = eval_delta(c_g)
            fd = eval_delta(d_g)
            
            for _ in range(50):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = eval_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = eval_delta(d_g)
            
            delta = (a_g + b_g) / 2.0
            delta = max(lo_db, min(hi_db, delta))
            
            if abs(delta) < 1e-12:
                continue
            
            # Apply
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue
            
            # Update convolution

    """Compute gradient of LSE(beta, c) w.r.t. a."""
    c_size = 2 * n - 1
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(c_size)
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    
    # c_rev = flip(w): c_rev[i] = w[c_size-1-i]
    c_rev = w[::-1]
    
    # b[j] = (a * c_rev)[c_size-1-j] for j=0..n-1
    # Compute a * c_rev via FFT
    conv_size = n + c_size - 1  # = n + 2n-1 - 1 = 3n-2
    fsize = 1 << (conv_size - 1).bit_length()
    A = fft(a, fsize)
    CR = fft(c_rev, fsize)
