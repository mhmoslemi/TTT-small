"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating method combined
with FISTA polish and coordinate descent.

Key innovation: Use the inner LP (min T s.t. a*b <= T, b>=0, sum(b)=S) to get
exact best-response directions, much more effective than subgradient methods
for this max-convolution objective. Active-set LP keeps solves fast.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def build_lp_matrix(a, n, lags):
    """Build the constraint matrix for the LP. A[i,j] = a[k_i - j] if valid."""
    m = len(lags)
    A = np.zeros((m, n))
    for i in range(m):
        k = int(lags[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        idx_hi = k - j_lo
        idx_lo = k - j_hi
        A[i, j_lo:j_hi + 1] = a[idx_lo:idx_hi + 1][::-1]
    return A


def solve_lp_for_b(a, n, S, lags, time_limit=20.0):
    """Solve LP: min T s.t. (a*b)_k <= T for k in lags, b>=0, sum(b)=S."""
    m = len(lags)
    if m == 0:
        m = 1
        lags = np.array([n - 1])
    A = build_lp_matrix(a, n, lags)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            b_opt = np.maximum(res.x[:n], 0.0)
            T_opt = float(res.x[n])
            return b_opt, T_opt
    except Exception:
        pass
    return None, None


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=2000.0,
              beta_ramp=3000):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta * 0.1)
        # Cap step to avoid overshooting
        step = min(step, 0.5 / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Periodic noise
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

            tl = min(30.0, max(5.0, max_time * 0.05 - elapsed * 0.01))

        # Solve LP
        b_opt, T_lp = solve_lp_for_b(a, n, S, lags, time_limit=tl)
        if b_opt is None:
            # Fallback: try with fewer lags
            if len(lags) > 100:
                # Keep only the top lags by convolution value
                c_curr = fft_convolve_self(a)
                top_k = np.argsort(c_curr)[::-1][:200]
                lags = top_k
                b_opt, T_lp = solve_lp_for_b(a, n, S, lags, time_limit=tl)
            if b_opt is None:
                break

        # Evaluate the LP solution
        b_opt = np.maximum(b_opt, 0.0)
        s_b = b_opt.sum()
        if s_b < 1e-15:
            break
        b_opt = b_opt * (S / s_b)

        # Blend
        a_new = (1.0 - blend) * a + blend * b_opt
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c_new = fft_convolve_self(a_new)
        cur_max = c_new.max()

        # Update active set
        threshold = T_lp + 0.01 * max(1.0, T_lp)
        active_mask = c_new >= threshold
        if np.sum(active_mask) < 10:
            # Always keep at least the top 50
            top_idx = np.argsort(c_new)[::-1][:50]
            active_mask[top_idx] = True
        active_lags = np.where(active_mask)[0]

        if cur_max < best:
            best = cur_max
            best_a = a_new.copy()
            a = a_new
            # Occasionally reduce blend for finer convergence
            if it > 20:
                blend = max(0.15, blend * 0.995)
        else:
            # No improvement; try smaller blend
            a_revert = (1.0 - blend * 0.5) * a + blend * 0.5 * b_opt
            a_revert = np.maximum(a_revert, 0.0)
            s = a_revert.sum()
            if s > 1e-15:
                a_revert = a_revert * (S / s)
            c_revert = fft_convolve_self(a_revert)
            if c_revert.max() < best:
                best = c_revert.max()
                best_a = a_revert.copy()
                a = a_revert

        # Add noise periodically to escape local minima
        if it % 50 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-it / 300.0)
            if noise_scale > 1e-9:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()

    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng):
    """GS-style spectral flattening as a diversification tool."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    alpha = 0.3

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / 500.0)
        alpha_cur = 0.2 + 0.7 * progress

        P_target = np.ones_like(P)
        P_new = (1.0 - alpha_cur) * P / P_mean + alpha_cur * P_target
        mag_new = np.sqrt(P_new)
        mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
        F_new = phase * mag_new
        a_raw = np.real(ifft(F_new))[:n]
        a_raw = np.maximum(a_raw, 0.0)
        s = a_raw.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_raw * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()
        a = a_new

    return best_a, best


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """Search for a non-negative sequence minimizing 2*n*max_conv/sum^2."""
    global S_global
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 1024 <= len(parent) <= 32768:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    S_global = S

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes" if parent is not None else "no"))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] score=%.6f (elapsed %.0fs, n=%d)" % (label, score, elapsed, len(a_norm)))
        return score

    # Build starting points
    starts = []
    if parent is not None:
        starts.append(("parent", normalize_sum(parent, n)))
        # Palindromic variant
        half = parent[:n // 2]
        if n % 2 == 0:
            pal = np.concatenate([half, half[::-1]])
        else:
            pal = np.concatenate([half, parent[n // 2:n // 2 + 1], half[::-1]])
        starts.append(("palindromic", normalize_sum(pal, n)))
        # Perturbed
        for sc in [0.05, 0.15]:
            noise = rng.normal(0, sc * S / np.sqrt(n), n)
            v = project_simplex(normalize_sum(parent, n) + noise, S)
            starts.append(("perturb_%.2f" % sc, v))

    # Structured starts
    tt = np.linspace(0, 1, n, endpoint=False)
    starts.append(("gaussian", np.maximum(np.exp(-12.0 * (tt - 0.5) ** 2), 0.001)))
    starts.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.001)))
    starts.append(("random", rng.random(n) + 0.05))

    # Phase 1: Quick FISTA on diverse starts to find good initial point
    print("\n=== Phase 1: Initial Exploration (FISTA) ===")
    p1_budget = min(200.0, (deadline - time.monotonic()) * 0.15)
    n_starts = len(starts)
    per_start = p1_budget / n_starts

    for i, (name, arr) in enumerate(starts):
        if time.monotonic() > deadline - (deadline - t_start) * 0.7 - 60:
            break
        t_alloc = min(per_start, deadline - time.monotonic() - 30)
        if t_alloc < 5:
            continue
        a_opt, score = run_fista(arr, n, S, t_alloc, rng,
                                  beta_init=20.0, beta_max=500.0, beta_ramp=500)
        update_best(a_opt, "fista_" + name)

    print("After Phase 1: best=%.6f" % global_best)

    if global_a is None:
        global_a = np.ones(n) * (S / n)
        global_best = 2.0

    # Phase 2: LP-based alternating method (main optimization)
    print("\n=== Phase 2: LP Alternating ===")
    p2_budget = (deadline - time.monotonic()) * 0.55
    if p2_budget > 30:
        a_lp, score_lp = alternating_lp(global_a, n, S, p2_budget, rng)
        update_best(a_lp, "lp_alt")

    print("After Phase 2: best=%.6f" % global_best)

    # Phase 3: Spectral flattening from best
    print("\n=== Phase 3: Spectral Flattening ===")
    p3_budget = (deadline - time.monotonic()) * 0.15
    if p3_budget > 10:
        a_sf, _ = spectral_flatten(global_a, n, S, p3_budget, rng)
        update_best(a_sf, "spectral")

    print("After Phase 3: best=%.6f" % global_best)

    # Phase 4: FISTA polish with high beta
    print("\n=== Phase 4: FISTA Polish ===")
    p4_budget = (deadline - time.monotonic()) * 0.20
    if p4_budget > 20:
        a_fin, _ = run_fista(global_a, n, S, p4_budget, rng,
                              beta_init=100.0, beta_max=3000.0, beta_ramp=2000)
        update_best(a_fin, "fista_polish")

    print("After Phase 4: best=%.6f" % global_best)

    # Phase 5: Coordinate descent
    print("\n=== Phase 5: Coordinate Descent ===")
    p5_budget = max(0.0, (deadline - time.monotonic()) - 30)
    if p5_budget > 15:
        a_cd, _ = coordinate_descent_fast(global_a, n, S, p5_budget, rng)
        update_best(a_cd, "coord_descent")

    print("After Phase 5: best=%.6f" % global_best)

    # Final verification
    print("\n=== Final ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("Verification failed (inf), using fallback")
        global_a = np.ones(512) * (np.sqrt(2 * 512) / 512)
        final_list = [float(x) for x in global_a]

    print("Returning: length=%d, score=%.6f, time=%.0fs" % (
        len(final_list), global_best, time.monotonic() - t_start))
    return final_list
