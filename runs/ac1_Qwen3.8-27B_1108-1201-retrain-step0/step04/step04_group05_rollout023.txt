"""
Minimize 2*n*max_conv/sum^2 using weighted projected gradient with adaptive peak focusing.

Key innovations:
1. Weighted projected gradient: minimize sum_k w[k]*c[k] where w[k]=c[k]^p,
   with adaptive p for progressive peak focusing.
2. Efficient gradient via FFT cross-correlation.
3. Nesterov momentum for accelerated convergence.
4. LSE-LBFGS polishing for final precision.
5. Multi-scale strategy: broad flatness first, then sharp peak reduction.

import numpy as np
import time
from scipy.fft import fft, ifft


def next_pow2(m):
    return 1 << (m - 1).bit_length() if m > 1 else 2


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_weighted_gradient(a, w_vec, N_grad):
    """
    Compute gradient of (2/sum(w)) * sum_k w[k]*c[k] with respect to a.
    g[i] = (2/sum(w)) * sum_m a[m] * w[i+m]
    Computed via FFT: g[i] = (a_flipped * w)[n-1+i]
    """
    n = len(a)
    a_flipped = a[::-1]
    w_padded = np.zeros(N_grad)
    w_padded[:len(w_vec)] = w_vec
    Af = fft(a_flipped, N_grad)
    Wf = fft(w_padded, N_grad)
    conv_result = np.real(ifft(Af * Wf))
    return conv_result[n - 1:2 * n - 1]


def weighted_projected_gradient(a_init, n, S, N_conv, N_grad, max_time, rng,
                                p_init=2.0, p_final=30.0, n_iters_max=50000):
    """
    Weighted projected gradient with Nesterov momentum and adaptive p.
    Minimizes sum_k c[k]^p (smooth approximation to max(c)).
    Returns (best_a, best_cmax).
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = np.inf
    t_param = 1.0
    it = 0
    start_time = time.monotonic()
    no_improve = 0
    step_base = 2.0 * S / (n * 3.0)

    while time.monotonic() - start_time < max_time and it < n_iters_max:
        it += 1
        # Adaptive p: ramp from p_init to p_final
        frac = min(1.0, it / (n_iters_max * 0.8))
        p_curr = p_init + (p_final - p_init) * frac * frac

        # Compute convolution
        c = compute_conv(a, N_conv)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute weights
        c_safe = np.maximum(c, 1e-30)
        w_vec = c_safe ** p_curr
        w_sum = w_vec.sum()
        if w_sum < 1e-30:
            break

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        momentum_factor = (t_param - 1.0) / t_new

        # Compute gradient at current point
        g = compute_weighted_gradient(a, w_vec, N_grad)
        g = 2.0 * g / w_sum

        # Nesterov extrapolation
        if it > 1:
            y = a + momentum_factor * (a - a_prev)
            y = np.maximum(y, 0.0)
            y = y * (S / max(y.sum(), 1e-15))
            g_y = compute_weighted_gradient(y, w_vec, N_grad)
            g_y = 2.0 * g_y / w_sum
        else:
            y = a
            g_y = g

        # Step size with adaptive scaling
        gnorm = np.linalg.norm(g_y)
        if gnorm < 1e-18:
            break
        step = step_base / (1.0 + 0.01 * it)

        # Backtracking line search
        a_prev = a.copy()
        alpha = step
        c_old = cmax
        found = False
        for ls in range(10):
            candidate = y - alpha * g_y
            candidate = project_simplex(candidate, S)
            c_new = compute_conv(candidate, N_conv)
            obj_new = c_new.max()
            if obj_new <= c_old + 1e-12:
                a = candidate
                found = True
                break
            alpha *= 0.5
        if not found:
            a = project_simplex(y - alpha * g_y, S)

        t_param = t_new

        # Periodic restart
        if no_improve > 800 and it % 200 == 0:
            noise_sigma = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_sigma, n), S)
            t_param = 1.0
            no_improve = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print("    WPG: it=%d, cmax=%.8f, p=%.1f, time=%.1fs" % (it, best_cmax, p_curr, elapsed))

    return best_a, best_cmax


def lse_gradient(a, N, N_grad, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    cmax = c.max()
    exp_bc = np.exp(beta * (c - cmax))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), cmax, c
    w = exp_bc / total
    # Gradient: (2/sum(w_eff)) * (a_flipped * w_vector)
    # LSE gradient: sum_k w[k] * dc[k]/da[i] = 2 * sum_k w[k] * a[k-i] * 1_{valid}
    g = compute_weighted_gradient(a, w * len(w), N_grad)
    g = 2.0 * g / (w.sum() * len(w))
    # Actually w already sums to 1, so:
    g = 2.0 * compute_weighted_gradient(a, w, N_grad)
    lse_val = cmax + np.log(total) / beta
    return g, lse_val, c


def lbfgs_lse_block(a_init, n, S, N, N_grad, max_time, rng,
                    beta_init=20.0, beta_max=500.0, m_lbfgs=10):

    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)

    best_cmax = np.inf

    y_hist = []

    cmax = c.max()

        best_cmax = cmax

    step_scale = 4.0 * S / (n * beta)

    while time.monotonic() - start_time < max_time:
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.05)
            step_scale = 4.0 * S / (n * beta)
            g, f_val, c = lse_gradient(a, N, N_grad, beta)
            cmax = c.max()
            if cmax < best_cmax:
                best_cmax = cmax
                best_a = a.copy()

        it += 1
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # L-BFGS direction
        direction = -g
        m = len(s_hist)
        if m > 0:
            gn = np.linalg.norm(g)
            if gn > 1e-18:
                q = g.copy()
                alphas = np.zeros(m)
                for i in range(m - 1, -1, -1):
                    s_i = s_hist[i]
                    y_i = y_hist[i]
                    yy = np.dot(y_i, y_i)
                    if yy > 1e-30:
                        alphas[i] = np.dot(s_i, q) / yy
                        q = q - alphas[i] * y_i
                s_last = s_hist[-1]
                y_last = y_hist[-1]
                yy_last = np.dot(y_last, y_last)
                if yy_last > 1e-30:
                    gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
                else:
                    gamma_use = 1.0
                r = gamma_use * q
                for i in range(m):
                    s_i = s_hist[i]
                    y_i = y_hist[i]
                    yy = np.dot(y_i, y_i)
                    if yy > 1e-30:
                        beta_i = np.dot(s_i, y_i) / yy
                        r = r + (alphas[i] - beta_i) * y_i
                direction = -r
                if np.dot(g, direction) > 0:
                    direction = -g

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = cmax
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= cmax or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= cmax:
                    break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        # Update L-BFGS history
        g_new, f_new, c_new = lse_gradient(a_new, N, N_grad, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss + 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)

        a = a_new
        g = g_new
        cmax = c_new

        if it % 500 == 0 and rng.random() < 0.1:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, f_val, c = lse_gradient(a, N, N_grad, beta)
            cmax = c.max()

    return best_a, best_cmax


def spectral_block(a_init, n, S, N, max_time, rng, top_k=10, alpha=0.15):
    """Spectral shaping refinement."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Compute spectrum
        A = fft(a, N)
        P = np.abs(A) ** 2
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            k_int = int(k)
            if k_int > N // 2:
                k_int = N - k_int
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * k_int / N)

        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    # Make symmetric for real output
    P_sym = np.zeros(N)
    P_sym[0] = P[0]
    for k in range(1, N // 2 + 1):
        val = (P[k] + P[N - k]) / 2.0
        P_sym[k] = val
        P_sym[N - k] = val
    phase = np.zeros(N)
    for k in range(1, N // 2):
        ph = rng.uniform(0, 2 * np.pi)
        phase[k] = ph
        phase[N - k] = -ph
    A = np.sqrt(P_sym) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        rem = max(10.0, remaining3 - (time.monotonic() - t3))
        if rem < 15:
            break
        a_l, sc_l = lbfgs_lse_block(global_a, n, S, N_conv, N_grad, min(rem - 5, 180.0), rng,
                                     beta_init=beta_start, beta_max=beta_start * 5.0, m_lbfgs=10)
        update_best(a_l, "LSE b=%d" % int(beta_start))

    # --- Phase 4: Spectral shaping ---
    print("=== Phase 4: Spectral shaping ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4

    if remaining4 > 30:
        a_sp, sc_sp = spectral_block(global_a, n, S, N_conv, min(120.0, remaining4 * 0.5), rng,
                                     top_k=8, alpha=0.12)
        update_best(a_sp, "spectral")

    if time.monotonic() < deadline - 20:
        a_sp2, sc_sp2 = spectral_block(global_a, n, S, N_conv, min(80.0, deadline - time.monotonic() - 20),
                                       rng, top_k=4, alpha=0.06)
        update_best(a_sp2, "spectral_fine")

    # --- Phase 5: Final LSE polish ---
    remaining5 = deadline - time.monotonic()
    if remaining5 > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N_conv, N_grad, min(remaining5 - 10, 200.0),
                                         rng, beta_init=100.0, beta_max=800.0, m_lbfgs=12)
        update_best(a_fin, "final_LSE")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_obj = 2.0 * n * global_best / (S * S)
    print("Final: length=%d, cmax=%.8f, obj=%.8f" % (len(global_a), global_best, final_obj))
    return [float(x) for x in global_a]
