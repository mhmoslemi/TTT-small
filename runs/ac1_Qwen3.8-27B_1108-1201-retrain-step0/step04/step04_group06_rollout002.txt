def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho

"""
Minimize 2*n*max_conv/sum^2 using adaptive subgradient with peak-lag targeting,
pair-based coordinate descent, multi-length search, and structured perturbations.
Key innovations: (1) adaptive active-set subgradient with gradient normalization,
(2) targeted pair-reduction at peak lags, (3) multi-n exploration,
(4) hybrid block+scalar coordinate descent polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = (v.sum() - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, active_lags):
    """
    Compute subgradient of max_k c_k at current point.
    For each active lag k: grad[i] += 2*a[k-i] for valid i.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo..hi] += 2 * a[k-hi .. k-lo] (reversed)
        # a[k-i] for i=lo is a[k-lo], for i=hi is a[k-hi]
        # We need: grad[lo:hi+1] += 2 * a[k-hi : k-lo+1 : -1]... 
        # Actually a[k-lo], a[k-lo-1], ..., a[k-hi]
        # That's a[k-lo : k-hi-1 : -1] if k-lo > k-hi
        # Let's just index directly
        i_range = np.arange(lo, hi + 1)
        j_vals = k - i_range
        # j_vals should be in [0, n-1] by construction
        grad[lo:hi + 1] += 2.0 * a[j_vals]
    return grad


def subgradient_optimize(a_init, n, S, max_time, rng, target_score=1.50):
    """
    Adaptive subgradient method to minimize max(c_k) subject to a>=0, sum(a)=S.
    """
    a = a_init.copy()
    a = project_simplex(a, S)
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    lr = 0.5
    active_frac = 0.05
    stall = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cmax = c.max()

        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
            last_improve = it
        else:
            stall += 1

        score = 2.0 * n * cmax / (S * S)
        if score <= target_score:
            break

        if stall > 80:
            # Perturb to escape
            sigma = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, sigma, n), S)
            stall = 0
            continue

        # Find active lags (top fraction of max)
        threshold = cmax * (1.0 - active_frac)
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]

        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(c))])
        elif len(active_lags) > 50:

            active_lags = top_idx

            a = a_new

            # Try smaller step

def pair_coordinate_descent(a_init, n, S, max_time, rng, n_top=20, n_rounds=200):
    """
    Coordinate descent targeting peak-lag contributing pairs.
    For the peak lag, identify top-contributing pairs and try reducing them.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        cmax = c.max()
        peak_lag = int(np.argmax(c))

        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        # Find top contributing pairs at peak lag
        lo = max(0, peak_lag - n + 1)
        hi = min(n - 1, peak_lag)
        pairs = []
        contribs = []
        for i in range(lo, hi + 1):
            j = peak_lag - i
            if j >= 0 and j < n and i <= j:
                pairs.append((i, j))
                contribs.append(a[i] * a[j] * (2 if i != j else 1))

        if not pairs:
            break

        # Sort by contribution
        order = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for idx in order:
            if time.monotonic() - start > max_time:
                break
            i, j = pairs[idx]
            orig_i = a[i]
            orig_j = a[j]

            # Try reducing one or both
            best_red = 0.0
            best_cmax_local = cmax

            fracs = [0.3, 0.5, 0.7, 1.3, 1.5]
            for fi in fracs:
                for fj in fracs:
                    a[i] = orig_i * fi
                    a[j] = orig_j * fj
                    # Quick check: recompute only affected lags
                    # Lags affected: those involving i or j
                    # c_k changes for k in [i, i+n-1] U [j, j+n-1]
                    # For speed, just check the peak lag and a few neighbors
                    k_check_lo = max(0, peak_lag - 5)
                    k_check_hi = min(2 * n - 2, peak_lag + 5)
                    # Recompute c at peak_lag
                    c_peak = 0.0
                    klo = max(0, peak_lag - n + 1)
                    khi = min(n - 1, peak_lag)
                    for ii in range(klo, khi + 1):
                        jj = peak_lag - ii
                        c_peak += a[ii] * a[jj]
                    if c_peak < best_cmax_local:
                        # Full check (expensive but necessary)
                        c_full = fft_convolve_self(a)
                        if c_full.max() < best_cmax_local:
                            best_cmax_local = c_full.max()
                            best_red = 1
                            best_fi, best_fj = fi, fj

            if best_red:
                a[i] = orig_i * best_fi
                a[j] = orig_j * best_fj
                improved = True
            else:
                a[i] = orig_i
                a[j] = orig_j

        # Rescale
        a = project_simplex(a, S)

        if not improved and rnd > 20:
            # Perturb
            n_pert = min(20, n)
            idxs = rng.choice(n, n_pert, replace=False)
            for idx in idxs:
                a[idx] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)

    return best_a, best_cmax


def block_optimize(a_init, n, S, max_time, rng, block_size=128):
    """
    Block-level optimization: partition into blocks, for each block try several
    multiplicative patterns and keep the best.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = fft_convolve_self(a).max()

    n_blocks = n // block_size
    patterns = [
        np.ones(block_size),
        np.linspace(0.5, 1.5, block_size),
        np.linspace(1.5, 0.5, block_size),
        0.5 + 0.5 * np.random.RandomState(0).random(block_size),
    ]

    for rnd in range(5):
        if time.monotonic() - start > max_time:
            break

        # Shuffle block order
        block_order = rng.permutation(n_blocks)

        for b in block_order:
            if time.monotonic() - start > max_time:
                break
            sl = slice(b * block_size, (b + 1) * block_size)
            orig = a[sl].copy()
            best_pattern = None
            best_local = best_cmax

            for pat in patterns:
                a[sl] = orig * pat
                a = project_simplex(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_local:
                    best_local = c.max()
                    best_pattern = pat.copy()

            if best_pattern is not None:
                a[sl] = orig * best_pattern
                a = project_simplex(a, S)
                if best_local < best_cmax:
                    best_cmax = best_local
                    best_a = a.copy()
            else:
                a[sl] = orig

        # Occasionally add new random patterns
        patterns.append(0.5 + 0.7 * rng.random(block_size))
        if len(patterns) > 8:
            patterns = patterns[-8:]

    return best_a, best_cmax


def upsequence(a_src, n_target, n_src):
    """Upsample a sequence from n_src to n_target length using linear interpolation."""
    if n_src == n_target:
        return a_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    return np.interp(x_tgt, x_src, a_src)


def structured_init(n, rng, method="random"):
    """Create a structured initialization."""
    t = np.arange(n) / n
    if method == "random":
        a = 0.3 + 0.7 * rng.random(n)
    elif method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t) \
            + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n)
    elif method == "asymmetric":
        # High on one side, low on other (reduces center lag)
        a = 0.3 + 0.7 * (1 - t) + 0.1 * rng.standard_normal(n)
    elif method == "bimodal":
        a = 0.3 + 0.5 * np.exp(-0.5 * ((t - 0.2) * 5) ** 2) + 0.5 * np.exp(-0.5 * ((t - 0.8) * 5) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif method == "alternating_block":
        # Alternating blocks of high/low
        block = max(1, n // 64)
        heights = np.where(np.arange(n // block) % 2 == 0, 1.0, 0.4)
        a = np.repeat(heights, block)
        a = a[:n] + 0.05 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)

    return np.clip(a, 0.01, 3.0)


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_downshift(a, n, N, top_k, alpha, rng=None):
    """
    Reduce peak lags by modifying the power spectrum.
    Targets top-k lags with exponential downshift.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]

    top_indices = np.argsort(c)[::-1][:top_k]
    m = np.arange(N)

    # Weighted cosine
    cos_eff = np.zeros(N)
    for k in top_indices[:10]:  # Limit to top 10 for speed
        cos_eff += np.cos(2.0 * np.pi * m * int(k) / N)
    cos_eff /= min(10, len(top_indices))

    P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))

    # Preserve total power approximately
    power_ratio = P.sum() / max(P_new.sum(), 1e-30)
    P_new *= power_ratio

    a_new = np.real(ifft(np.sqrt(P_new) * np.exp(1j * np.angle(A))))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

        best_init_score = np.inf
        for name, arr in inits:
            c = fft_convolve_self(arr)
            sc = 2.0 * n * c.max() / (arr.sum() ** 2)
            if sc < best_init_score:
                best_init_score = sc
                best_init = arr.copy()

        # Optimization loop for this n
        a_curr = best_init.copy()
        phase_time = time.monotonic()
        stall_global = 0

        while time.monotonic() - phase_time < alloc_time and time.monotonic() < deadline - 30:
            remaining = alloc_time - (time.monotonic() - phase_time)
            if remaining < 5:
                break

            # --- Subgradient phase (60% of remaining) ---
            sg_time = remaining * 0.50
            a_sg, sc_sg = subgradient_optimize(a_curr, n, S, sg_time, rng, target_score=1.49)
            if update_best(a_sg, n):
                print("  [n=%d] subgrad: %.6f" % (n, global_best))
                a_curr = a_sg.copy()
                stall_global = 0
            else:
                stall_global += 1

            if time.monotonic() > deadline - 30:
                break

            # --- Spectral downshift (15% of remaining) ---
            remaining = deadline - time.monotonic()
            if remaining > 10:
                spec_time = min(remaining * 0.10, 30.0)
                t_spec_start = time.monotonic()
                while time.monotonic() - t_spec_start < spec_time:
                    alpha = 0.3 + 0.3 * rng.random()
                    top_k = 5 + int(10 * rng.random())
                    a_spec = spectral_downshift(a_curr, n, N_fft, top_k, alpha, rng)
                    a_spec = safe_normalize(a_spec, n)
                    if update_best(a_spec, n):
                        a_curr = a_spec.copy()
                        stall_global = 0
                        print("  [n=%d] spectral: %.6f" % (n, global_best))
                        break
                    # Combine with current
                    a_mix = 0.5 * a_curr + 0.5 * a_spec
                    a_mix = safe_normalize(a_mix, n)
                    if update_best(a_mix, n):
                        a_curr = a_mix.copy()
                        stall_global = 0
                        break
                    # Small step in spectral direction
                    t_step = 0.1 + 0.3 * rng.random()
                    a_small = (1 - t_step) * a_curr + t_step * a_spec
                    a_small = safe_normalize(a_small, n)
                    if update_best(a_small, n):
                        a_curr = a_small.copy()
                        stall_global = 0
                        break

            if time.monotonic() > deadline - 30:
                break

            # --- Pair coordinate descent (15% of remaining) ---
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_time = min(remaining * 0.15, 40.0)
                a_cd, sc_cd = pair_coordinate_descent(a_curr, n, S, cd_time, rng,
                                                      n_top=15, n_rounds=100)
                if update_best(a_cd, n):
                    print("  [n=%d] pair_cd: %.6f" % (n, global_best))
                    a_curr = a_cd.copy()
                    stall_global = 0
                else:
                    stall_global += 1

            if time.monotonic() > deadline - 30:
                break

            # --- Block optimization (10% of remaining) ---
            remaining = deadline - time.monotonic()
            if remaining > 10 and n <= 16384:
                blk_time = min(remaining * 0.10, 20.0)
                a_blk, sc_blk = block_optimize(a_curr, n, S, blk_time, rng,
                                               block_size=max(64, n // 256))
                if update_best(a_blk, n):
                    print("  [n=%d] block: %.6f" % (n, global_best))
                    a_curr = a_blk.copy()
                    stall_global = 0
                else:
                    stall_global += 1

            if time.monotonic() > deadline - 30:
                break

            # --- Perturbation and restart (5% of remaining) ---
            remaining = deadline - time.monotonic()
            if remaining > 5 and stall_global >= 3:
                # Try several perturbation strengths
                for sigma in [0.02, 0.05, 0.1, 0.2]:
                    a_pert = a_curr + rng.normal(0, sigma * S / np.sqrt(n), n)
                    a_pert = safe_normalize(a_pert, n)
                    if update_best(a_pert, n):
                        a_curr = a_pert.copy()
                        stall_global = 0
                        break
                    # Short subgradient from perturbed state
                    if remaining > 20:
                        a_sg2, _ = subgradient_optimize(a_pert, n, S, 5.0, rng, target_score=1.49)
                        if update_best(a_sg2, n):
                            a_curr = a_sg2.copy()
                            stall_global = 0
                            break
                stall_global = 0

        print("  [n=%d] done. Best so far: %.6f" % (n, global_best))

    # === Final polish on global best ===
    print("\n=== Final polish ===")
    if global_a is not None:
        n_final = len(global_a)
        S_final = global_a.sum()
        N_final = 1 << (2 * n_final - 1).bit_length()

        # Long subgradient
        rem = deadline - time.monotonic()
        if rem > 5:
            t1 = min(rem * 0.3, 60.0)
            a_p, _ = subgradient_optimize(global_a, n_final, S_final, t1, rng, target_score=1.49)
            if update_best(a_p, n_final):
                print("  Final subgrad: %.6f" % global_best)

        # Pair CD
        rem = deadline - time.monotonic()
        if rem > 3:
            t2 = min(rem * 0.25, 30.0)
            a_p2, _ = pair_coordinate_descent(global_a, n_final, S_final, t2, rng,
                                              n_top=30, n_rounds=200)
            if update_best(a_p2, n_final):
                print("  Final pair_cd: %.6f" % global_best)

        # Spectral fine-tune
        rem = deadline - time.monotonic()
        if rem > 3:
            for _ in range(5):
                a_spec = spectral_downshift(global_a, n_final, N_final, 5, 0.1, rng)
                a_spec = safe_normalize(a_spec, n_final)
                if update_best(a_spec, n_final):
                    global_a = a_spec
                    break
                # Blend
                for t_blend in [0.1, 0.2, 0.3]:
                    a_blend = (1 - t_blend) * global_a + t_blend * a_spec
                    a_blend = safe_normalize(a_blend, n_final)
                    if update_best(a_blend, n_final):
                        global_a = a_blend
                        break

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification score: %.6f" % ver)
        else:
            print("\nVerification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
