"""
Minimize 2*n*max_conv/sum^2 using adaptive log-sum-exp spectral shaping,
vectorized coordinate descent with line search, and multi-scale multi-start search.

Key improvements over previous approach:
1. Log-sum-exp (soft-max) surrogate for the peak, with adaptive temperature
   that sharpens over time for more precise peak targeting.
2. Multi-peak importance weighting: simultaneously targets top-K peaks with
   weights proportional to their excess above the current best.
3. Vectorized coordinate descent: O(1) FFT-based convolution update after
   single-coordinate changes (delta * cross-correlation structure).
4. Multi-scale: tries n=4096, 8192, 16384 and picks the best.
5. Bisection feasibility: binary search on target peak to push closer to optimum.
6. Simulated annealing restarts with adaptive cooling.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = 0.0
        # All should be positive
        result = np.maximum(v, 0.0)
        s = result.sum()
        if s > 0:
            result = result * (S / s)
        return result
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the evaluation score 2*n*max_b/S^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)


def log_sum_exp(x, alpha):
    """Compute log-sum-exp with temperature alpha. Returns scalar."""
    max_x = x.max()
    vals = alpha * (x - max_x)
    # Clip for numerical stability
    vals = np.clip(vals, -100, 100)
    return max_x + np.log(np.sum(np.exp(vals))) / alpha


def spectral_lse_shape(a, N, n, top_k, alpha_temp, lam, rng):
    """
    Single spectral shaping step using log-sum-exp objective.
    Returns (new_a, c_max).
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Compute LSE-based gradient (soft assignment to peaks)
    # The gradient of LSE(c, alpha) w.r.t. P is: exp(alpha * c[k]) / sum(exp(alpha * c))
    # And dP/da phase -> we use the frequency-domain gradient
    
    alpha_eff = alpha_temp
    exp_vals = np.exp(alpha_eff * (c - c_max))
    weights_freq = exp_vals / exp_vals.sum()  # weight for each lag
    
    # The gradient of LSE w.r.t. |A(f)|^2 can be computed via:
    # d LSE / d A(f) = 2 * Re(A(f)^* * conj(DFT(weights_freq)))
    # where weights_freq is the vector of lag weights
    
    # Build the full weight vector for the DFT
    w_full = np.zeros(N)
    for k_idx in top_indices:
        k_int = int(k_idx)
        if k_int < N:
            w_full[k_int] += weights_freq[k_int]
    
    # Also account for the conjugate symmetry in convolution
    # For real signals, (a*a)[k] = (a*a)[2n-2-k] (no, that's not right for linear conv)
    # Actually for linear conv, there's no symmetry in the weights.
    
    # Gradient in frequency domain
    W = fft(w_full, N)
    grad_A = 2 * alpha_eff * np.conj(A) * np.conj(W) / (2 * N)  # scaling
    
    # Update A
    step = lam
    A_new = A - step * grad_A
    
    # Project back: take magnitude from A_new, keep phase
    # Actually, let's use a power spectrum update instead
    # P_new = P - step * 2*Re(A * conj(W)/N)
    # Then phase = angle(A_new), magnitude = sqrt(P_new)
    
    P_new = P - step * np.real(np.conj(A) * W) / N
    P_new = np.maximum(P_new, 0.0)
    
    # Add small random phase perturbation
    if rng.random() < 0.1:
        phase_noise = rng.normal(0, 0.02, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def spectral_shaping_loop(a_init, n, N, deadline, rng, top_k=10, 
                          alpha_init=1.0, alpha_final=20.0, lam_init=0.01,
                          lam_final=0.001):

    Iterative spectral shaping with log-sum-exp and adaptive parameters.
    """
    a = a_init.copy()
    S = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Adaptive parameters (exponential schedule)
        progress = min(1.0, it / 500.0)
        alpha = alpha_init * (alpha_final / alpha_init) ** progress
        lam = lam_init * (lam_final / lam_init) ** progress
        cur_top_k = min(64, top_k + it // 50)

        a_new, c_max = spectral_lse_shape(a, N, n, cur_top_k, alpha, lam, rng)

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if it % 200 == 0:
                print("      LSE iter %d: %.6f" % (it, best_score))
        else:
            a = a_new.copy()
            stall += 1
            if stall > 30:
                # Perturb to escape

                a = project_simplex(a + rng.normal(0, sigma, n), S)
                stall = 0

        if it % 1000 == 0:
            print("      LSE iter %d: best=%.6f, cur=%.6f" % (it, best_score, new_score))

    return best_a, best_score
