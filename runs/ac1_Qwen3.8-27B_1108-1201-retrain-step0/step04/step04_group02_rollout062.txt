```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

This program combines three complementary search ideas:

1. Low-dimensional FFT-modulated refinement.
   A candidate is written as a probability p_i = base_i*exp(v_i)/sum,
   where v_i is restricted to a small set of low-frequency Fourier modes.
   Smooth LSE smooth max objective and exact analytic gradients are optimized
   with L-BFGS-B.  This explores wide-shape changes while preserving a base
   sequence (including the previous state of the art if present).

2. Fast spectral capping.
   Build a capped/truncated convolution target and update the DFT of the
   sequence toward a spectral square root of that target, then project to
   the non-negative simplex by clipping and renormalization.

3. Direct subgradient/LSE local polishing.
   Multiplicative (mirror-style) steps using exact peak subgradients and
   softmax-peak (LSE) gradients, with backtracking line search and
   multiplicative perturbations to escape stalls.

The best sequence found before the deadline is returned.
"""

import numpy as np
import time

from scipy.fft import fft, ifft

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_conv_full(a):
    """Full linear self-convolution of a 1D array using FFT."""
    n = int(len(a))
    if n == 0:
        return np.array([], dtype=float)
    size = 2 * n - 1
    L = next_pow2(size)
    A = fft(a, L)
    c = np.real(ifft(A * A))
    c = c[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def normalize_sequence(a, n=None, target=None):
    """Clip to [0,1000], remove non-finite values, and rescale to target sum."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if n is None:
        n = int(arr.size)
    else:
        n = int(n)
    if n <= 0:
        return np.array([], dtype=float)

    if arr.size != n:
        if arr.size < n:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[:arr.size] = arr
            arr = tmp
        else:
            arr = arr[:n]

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    if target is None:
        target = float(np.sqrt(2.0 * n))
    target = float(target)

    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        arr = np.ones(n, dtype=np.float64) * (target / n)
        return arr

    arr = arr * (target / s)
    return arr


def sequence_objective(a):
    """Same objective as evaluate_sequence, but computed locally."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = int(arr.size)
    if n == 0:
        return float("inf")

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")

    c = fft_conv_full(arr)
    if c.size == 0:
        return float("inf")
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return float("inf")

    val = 2.0 * n * mb / (s * s)
    return float(val) if np.isfinite(val) else float("inf")


def lse_weights(c, beta):
    """Weights for the soft maximum logsumexp(beta*c)."""
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(c, dtype=np.float64) / max(1, c.size)
    return e / s


def weighted_subgradient_fft(a, w):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.
    Returns 2 * cross-correlation(a, w), truncated to len(a).
    """
    n = int(len(a))
    L = next_pow2(len(w) + n - 1)
    W = fft(w, L)
    A = fft(a, L)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def make_cap_target(c, threshold):
    """
    Make a target convolution by capping peaks at threshold and redistributing
    the removed mass into currently low lags as evenly as possible.
    """
    csum = float(c.sum())
    target = np.minimum(c, threshold)
    excess = csum - float(target.sum())
    if excess <= 1e-300:
        return target

    slack = threshold - target
    total_slack = float(slack.sum())
    if total_slack > 1e-300:
        if total_slack >= excess:
            target = target + slack * (excess / total_slack)
        else:
            target = target + slack
            excess -= total_slack

    if excess > 1e-300:
        target = target + excess / max(1, target.size)

    target = np.where(np.isfinite(target), target, 0.0)
    target = np.maximum(target, 0.0)
    target_sum = float(target.sum())
    if target_sum <= 0.0:
        target = np.ones_like(c, dtype=np.float64) * (csum / max(1, c.size))
    else:
        target = target * (csum / target_sum)
    return target


def mirror_step(a, grad, step, target_sum):
    """Multiplicative / mirror step for positive sequences."""
    a = np.asarray(a, dtype=np.float64)
    grad = np.asarray(grad, dtype=np.float64)
    tiny = np.finfo(np.float64).tiny
    loga = np.log(np.maximum(a, tiny))
    delta = step * grad
    delta = np.clip(delta, -3.0, 3.0)
    lognew = loga - delta
    lognew -= np.max(lognew)
    new = np.exp(lognew)
    new = np.where(np.isfinite(new), new, 0.0)
    s = float(new.sum())
    if not np.isfinite(s) or s <= tiny:
        return a.copy()
    return new * (target_sum / s)


def local_optimize(a0, max_seconds, rng, initial_eta=0.35, beta_start=20.0, beta_end=180.0):
    """
    Direct local polishing using multiplicative mirror steps.
    Uses exact peak subgradients and LSE-smoothed gradients.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = int(a0.size)
    if n <= 1 or max_seconds <= 0.0:
        a = normalize_sequence(a0, n)
        c = fft_conv_full(a)
        return a, (float(np.max(c)) if c.size else float("inf"))

    S = float(np.sqrt(2.0 * n))
    a = normalize_sequence(a0, n, S)
    c = fft_conv_full(a)
    cur = float(np.max(c))
    best_a = a.copy()
    best = cur

    start = time.monotonic()
    it = 0
    eta = initial_eta
    fail = 0
    no_improve = 0

    while time.monotonic() - start < max_seconds:
        dt = time.monotonic() - start
        if dt >= max_seconds:
            break
        it += 1

        cur = float(np.max(c))
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 6000 and fail > 20:
                break

        r = it % 300
        if r < 50:
            mode = 0
        elif r < 110:
            mode = 1
        else:
            mode = 2

        if mode == 0:
            k = int(np.argmax(c))
            grad = np.zeros(n, dtype=np.float64)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] = 2.0 * a[k - idx]
            gmax = float(np.max(np.abs(grad)))
        elif mode == 1:
            kk = min(3, n)
            idxs = np.argpartition(c, -kk)[-kk:] if kk < c.size else np.arange(c.size)
            grad = np.zeros(n, dtype=np.float64)
            for j in idxs:
                j = int(j)
                lo = max(0, j - n + 1)
                hi = min(j, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                grad[idx] += 2.0 * a[j - idx]
            grad /= float(max(1, kk))
            gmax = float(np.max(np.abs(grad)))
        else:
            beta = min(beta_end, beta_start + 0.0015 * it)
            w = lse_weights(c, beta)
            grad = weighted_subgradient_fft(a, w)
            gmax = float(np.max(np.abs(grad)))

        if not np.isfinite(gmax) or gmax <= 1e-15:
            break

        step = eta / gmax
        accepted = False
        for _try in range(3):
            a_try = mirror_step(a, grad, step, S)
            c_try = fft_conv_full(a_try)
            sc = float(np.max(c_try))
            if np.isfinite(sc) and sc < cur - 1e-12:
                a = a_try
                c = c_try
                cur = sc
                eta = min(5.0, eta * 1.05)
                accepted = True
                break
            step *= 0.5

        if not accepted:
            fail += 1
            eta = max(1e-4, eta * 0.82)
            if fail >= 6:
                noise = min(0.25, 0.035 + 0.004 * fail)
                a = a * np.exp(rng.normal(0.0, noise, n))
                a = normalize_sequence(a, n, S)
                c = fft_conv_full(a)
                cur = float(np.max(c))
                fail = 0
                eta = initial_eta
                if cur > best + 1e-9:
                    a = best_a.copy()
                    c = fft_conv_full(a)
                    cur = best
                no_improve = 0

    best_a = normalize_sequence(best_a, n, S)
    return best_a, best


def fourier_basis(N, K):
    """Precompute cosine and sine basis matrices for low-frequency modulation."""
    N = int(N)
    K = int(K)
    if N <= 0 or K <= 0:
        return np.empty((0, N)), np.empty((0, N))
    i = np.arange(N, dtype=np.float64)
    Fcos = np.empty((K, N), dtype=np.float64)
    Fsin = np.empty((K, N), dtype=np.float64)
    for k in range(1, K + 1):
        arg = (2.0 * np.pi * k * i / float(N))
        Fcos[k - 1] = np.cos(arg)
        Fsin[k - 1] = np.sin(arg)
    return Fcos, Fsin


def compute_mod_p(coeff, N, K, Fcos, Fsin, base_log):
    """Probability p_i = base_i exp(v_i)/sum, v from Fourier coefficients."""
    coeff = np.asarray(coeff, dtype=np.float64).ravel()
    if coeff.size < 2 * K:
        tmp = np.zeros(2 * K, dtype=np.float64)
        tmp[:coeff.size] = coeff
        coeff = tmp

    even = coeff[:K]
    odd = coeff[K:2 * K]

    v = Fcos.dot(even) + Fsin.dot(odd)
    x = base_log + v
    x -= np.max(x)
    p = np.exp(x)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(N, dtype=np.float64) / float(N)
        return p
    p = p / s
    p = np.where(np.isfinite(p), p, 0.0)
    s2 = float(p.sum())
    if s2 <= 0.0:
        p = np.ones(N, dtype=np.float64) / float(N)
    else:
        p = p / s2
    return p


def modulation_objective(coeff, N, K, Fcos, Fsin, base_log, beta):
    """LSE-smoothed objective for a low-frequency modulation."""
    try:
        p = compute_mod_p(coeff, N, K, Fcos, Fsin, base_log)
        c = fft_conv_full(p)
        W = c * (2.0 * float(N))
        cm = float(np.max(W))
        if not np.isfinite(cm):
            return 1e100

        z = beta * (W - cm)
        z = np.clip(z, -745.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0:
            return cm
        return float(cm + np.log(s) / float(beta))
    except Exception:
        return 1e100


def modulation_gradient(coeff, N, K, Fcos, Fsin, base_log, beta):
    """Analytic gradient of LSE objective w.r.t. Fourier coefficients."""
    try:
        coeff = np.asarray(coeff, dtype=np.float64).ravel()
        if coeff.size != 2 * K:
            tmp = np.zeros(2 * K, dtype=np.float64)
            mn = min(coeff.size, 2 * K)
            tmp[:mn] = coeff[:mn]
            coeff = tmp

        p = compute_mod_p(coeff, N, K, Fcos, Fsin, base_log)
        c = fft_conv_full(p)
        W = c * (2.0 * float(N))
        cm = float(np.max(W))
        z = beta * (W - cm)
        z = np.clip(z, -745.0, 0.0)
        e = np.exp(z)
        w = e / float(e.sum())

        grad_c = weighted_subgradient_fft(p, w)       # d c / d p
        grad_W = (2.0 * float(N)) * grad_c             # d W / d p
        h = p * grad_W

        H = fft(h, N)
        P = fft(p, N)
        H0 = float(h.sum())

        g = np.zeros(2 * K, dtype=np.float64)
        for k in range(1, K + 1):
            reH = float(H[k].real)
            imH = float(H[k].imag)
            reP = float(P[k].real)
            imP = float(P[k].imag)
            # even coefficient k
            g[k - 1] = reH - reP * H0
            # odd/sine coefficient k
            g[K + (k - 1)] = -imH + imP * H0

        g = np.where(np.isfinite(g), g, 0.0)
        return g
    except Exception:
        return np.zeros(int(len(coeff)), dtype=np.float64)


def modulation_sequence(coeff, base_p, N, K, Fcos, Fsin):
    """Return a normalized sequence (sum sqrt(2N)) from modulation coefficients."""
    p = compute_mod_p(coeff, N, K, Fcos, Fsin, np.log(np.maximum(base_p, 1e-300)))
    S = float(np.sqrt(2.0 * N))
    return p * S


def run_modulation_lbfgs(base_a, N, K, max_seconds, rng,
                         betas=(10.0, 30.0, 80.0),
                         bound=8.0,
                         base_maxfun=2500,
                         n_starts=3):
    """
    Optimize a low-frequency modulation of base_a with L-BFGS-B.
    Returns best sequence, best exact score, and corresponding coefficients.
    """
    N = int(N)
    K = int(K)
    if N <= 1 or K <= 0 or max_seconds <= 0.0 or scipy_minimize is None:
        seq = normalize_sequence(base_a, N)
        return seq, sequence_objective(seq), np.zeros(0, dtype=np.float64)

    base_p = normalize_sequence(base_a, N, 1.0)
    base_log = np.log(np.maximum(base_p, 1e-300))
    Fcos, Fsin = fourier_basis(N, K)

    best_seq = normalize_sequence(base_a, N)
    best_score = sequence_objective(best_seq)
    best_coeff = np.zeros(2 * K, dtype=np.float64)

    starts = []
    starts.append(np.zeros(2 * K, dtype=np.float64))
    starts.append(rng.normal(0.0, 0.25, 2 * K))
    preset = np.zeros(2 * K, dtype=np.float64)
    if K > 0:
        preset[K] = 0.7
    if K > 2:
        preset[K + 1] = -0.35
    starts.append(preset)

    betas = list(betas)
    start_phase = time.monotonic()

    for si in range(min(n_starts, len(starts))):
        rem_phase = max_seconds - (time.monotonic() - start_phase)
        if rem_phase <= 5.0:
            break
        x_cur = starts[si].copy()

        for bi, beta in enumerate(betas):
            rem_phase = max_seconds - (time.monotonic() - start_phase)
            if rem_phase <= 5.0:
                break

            n_remaining = max(1, (len(betas) - bi) * max(1, n_starts - si))
            budget = max(1.0, rem_phase / float(n_remaining))
            maxfun_i = int(min(base_maxfun, max(200, float(budget) * 10.0)))
            maxiter_i = int(min(100, max(25, maxfun_i // 4)))

            bounds = [(-bound, bound)] * (2 * K)
            try:
                res = scipy_minimize(
                    modulation_objective,
                    x_cur,
                    jac=modulation_gradient,
                    args=(N, K, Fcos, Fsin, base_log, float(beta)),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": maxiter_i,
                        "maxfun": maxfun_i,
                        "ftol": 1e-10,
                        "gtol": 1e-7,
                        "ftol_f": 1e-10,
                        "ftol_x": 1e-10,
                        "maxls": 20,
                        "disp": False,
                    }
                )
                x_try = np.asarray(res.x, dtype=np.float64).ravel()
                if x_try.size != 2 * K:
                    x_try = x_cur.copy()
            except Exception:
                x_try = x_cur.copy()

            x_try = np.where(np.isfinite(x_try), x_try, 0.0)

            seq = modulation_sequence(x_try, base_p, N, K, Fcos, Fsin)
            seq = normalize_sequence(seq, N)
            sc = sequence_objective(seq)

            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = float(sc)
                best_seq = seq.copy()
                best_coeff = x_try.copy()

            x_cur = x_try

    best_seq = normalize_sequence(best_seq, N)
    return best_seq, best_score, best_coeff


def spectral_capping(a0, max_seconds, rng, lower=1.001):
    """
    Spectral capping method: repeatedly cap convolution peaks in the DFT
    target and move the sequence DFT toward a square root of the target.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = int(a0.size)
    if n <= 1 or max_seconds <= 0.0:
        a = normalize_sequence(a0, n)
        c = fft_conv_full(a)
        return a, (float(np.max(c)) if c.size else float("inf"))

    S = float(np.sqrt(2.0 * n))
    a = normalize_sequence(a0, n, S)
    c = fft_conv_full(a)
    score = float(np.max(c))
    best_a = a.copy()
    best_score = score

    size = 2 * n - 1
    L = next_pow2(size)
    start = time.monotonic()

    gap = max(score - lower, 0.05)
    delta = 0.012 * gap
    step = 0.45
    stuck = 0
    it = 0

    while time.monotonic() - start < max_seconds:
        it += 1
        if not np.isfinite(score):
            a = best_a.copy()
            c = fft_conv_full(a)
            score = best_score

        gap = max(score - lower, 0.01)
        delta = min(delta, 0.22 * gap)
        if delta < 1e-6:
            delta = 0.012 * gap
        threshold = max(lower + 1e-4, score - delta)

        if threshold >= score - 1e-10:
            break

        target = make_cap_target(c, threshold)
        Uhat = fft(target, L)
        A = fft(a, L)
        Vhat = A * A
        absV = np.abs(Vhat)
        ratio = np.ones_like(Vhat, dtype=np.complex128)
        mask = absV > 1e-250
        ratio[mask] = Uhat[mask] / Vhat[mask]
        ratio = np.where(np.isfinite(ratio), ratio, 1.0 + 0.0j)

        accepted = False
        for s in (step, step * 0.5, step * 0.25):
            if time.monotonic() - start >= max_seconds:
                break
            mag = np.abs(ratio)
            ang = np.angle(ratio)
            mag = np.clip(mag, 0.001, 1000.0)
            logfactor = 0.5 * s * (np.log(mag) + 1j * ang)
            factor = np.exp(logfactor)
            fmag = np.abs(factor)
            fmag = np.clip(fmag, np.exp(-2.0), np.exp(2.0))
            factor = fmag * np.exp(1j * np.angle(factor))

            Fnew = A * factor
            a_try = np.real(ifft(Fnew)[:n])
            a_try = np.where(np.isfinite(a_try), a_try, 0.0)
            a_try = np.maximum(a_try, 0.0)
            ss = float(a_try.sum())
            if not np.isfinite(ss) or ss <= 1e-300:
                continue
            a_try = a_try * (S / ss)
            c_try = fft_conv_full(a_try)
            sc = float(np.max(c_try))
            if np.isfinite(sc) and sc < score - 1e-10:
                score = sc
                a = a_try
                c = c_try
                accepted = True
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                break

        if accepted:
            stuck = 0
            delta = min(delta * 1.18, 0.18 * gap)
            step = min(1.0, step * 1.03)
        else:
            stuck += 1
            delta *= 0.82
            step *= 0.9
            if stuck >= 25:
                noise = min(0.30, 0.04 + 0.004 * stuck)
                a = a * np.exp(rng.normal(0.0, noise, n))
                a = normalize_sequence(a, n, S)
                c = fft_conv_full(a)
                score = float(np.max(c))
                if score > best_score + 1e-9:
                    a = best_a.copy()
                    c = fft_conv_full(a)
                    score = best_score
                delta = 0.012 * max(score - lower, 0.05)
                step = 0.45
                stuck = 0

        if it % 3000 == 0:
            print("    spectral_capping it=%d current=%.6f best=%.6f time=%.0fs" %
                  (it, score, best_score, time.monotonic() - start))

    best_a = normalize_sequence(best_a, n, S)
    return best_a, best_score


def resample_sequence_periodic(a, m):
    """Periodic linear resampling to length m."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    m = int(m)
    if n <= 0 or m <= 0:
        return np.array([], dtype=float)
    if n == m:
        return a.copy()

    pos = np.arange(m, dtype=np.float64) * (float(n) / float(m))
    i0 = np.floor(pos).astype(np.int64) % n
    i1 = (i0 + 1) % n
    frac = pos - np.floor(pos)
    out = a[i0] * (1.0 - frac) + a[i1] * frac
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return out


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(int(seed))
    budget_s = float(budget_s)

    # --- Obtain if the previous best sequence is available ---------------
    parent = None

        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 128 and p.sum() > 0.01:
            parent = p
    except Exception:
        parent = None

    # Choose a primary working length (prefer powers of two).
    if parent is not None and 512 <= parent.size <= 20000:
        l2 = int(np.rint(np.log2(float(parent.size))))
        l2 = max(11, min(14, l2))
        main_n = 1 << l2
        parent = resample_sequence_periodic(parent, main_n)
    else:
        main_n = 8192
        parent = np.ones(main_n, dtype=np.float64) * (np.sqrt(2.0 * main_n) / main_n)

    main_n = int(main_n)
    best_a = normalize_sequence(parent, main_n)
    best_score = sequence_objective(best_a)
    print("Initial parent n=%d score=%.6f" % (main_n, best_score))

    def try_update(seq, name):
        nonlocal_best = sequence_objective(seq)
        return seq, nonlocal_best

    # Helper is not a closure? To comply strictly, use top-level logic only.
    def _local_update(seq):
        return normalize_sequence(seq), sequence_objective(seq)

    # ---------------- Phase 1: low-frequency L-BFGS on parent -------------
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.32, 1100.0)
    if t1 > 20.0:
        K1 = 48 if main_n <= 8192 else 32
        print("=== Phase 1: L-BFGS low-frequency modulation (n=%d K=%d %.0fs) ===" %
              (main_n, K1, t1))
        seq, sc, _ = run_modulation_lbfgs(
            best_a, main_n, K1, t1, rng,
            betas=(10.0, 30.0, 80.0),
            bound=8.0,
            base_maxfun=3000 if main_n <= 8192 else 1600,
            n_starts=3
        )
        seq = normalize_sequence(seq, main_n)
        sc = sequence_objective(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = seq.copy()
            print("  L-BFGS parent improved: %.6f" % best_score)
        else:
            # keep the potentially useful smoothed candidate as backup only if close
            pass

    # ---------------- Phase 2: direct local polish ------------------------
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.26, 800.0)
    if t2 > 10.0:
        print("=== Phase 2: direct subgradient/LSE local polish (%.0fs) ===" % t2)
        seq, sc = local_optimize(best_a, t2, rng, initial_eta=0.35,

        if np.isfinite(sc) and sc < best_score:

    # ---------------- Phase 3: spectral capping ---------------------------
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 650.0)
    if t3 > 10.0:
        print("=== Phase 3: spectral capping (%.0fs) ===" % t3)
        seq, sc = spectral_capping(best_a, t3, rng, lower=1.001)

    # ---------------- Phase 4: second L-BFGS round on current best --------

            alt_done = True

    # If alternate length did not run, try lower length instead.
    if not alt_done and rem > 300.0 and main_n > 2048:
        alt_n = max(2048, main_n // 2)
        t5 = min(rem * 0.22, 350.0)
        if t5 > 30.0:

            alt_a = resample_sequence_periodic(best_a, alt_n)
            alt_a = normalize_sequence(alt_a, alt_n)

    rem = deadline - time.monotonic()

    # ---------------- Final verification ---------------------------------
    final_a = normalize_sequence(best_a)
    final_a = np.clip(final_a, 0.0, 1000.0)
    final_a = np.where(np.isfinite(final_a), final_a, 0.0)
    if final_a.size == 0 or final_a.sum() < 0.01:
        nfb = 256
        final_a = np.ones(nfb, dtype=np.float64) * (np.sqrt(2.0 * nfb) / nfb)

    final_list = [float(x) for x in final_a]

    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
            if ver < best_score:

        nfb = 256
        final_a = np.ones(nfb, dtype=np.float64) * (np.sqrt(2.0 * nfb) / nfb)
        final_list = [float(x) for x in final_a]

          (len(final_list), best_score, time.monotonic() - (deadline - budget_s + 30.0)))
    return final_list
```
