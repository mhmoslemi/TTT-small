"""
Minimize 2*n*max_conv/sum^2 using iterative LP-based peak reduction with
subgradient refinement. Uses linearized quadratic constraints to compute
descent directions, then refines with adaptive subgradient steps.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import sys

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += a[j - i_vals]
    return 2.0 * grad / k

def lse_subgradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def feat_map(a, K, b):
    """Compute the top-K lags and their values."""
    top_k = np.argsort(b)[::-1][:K]
    return top_k, b[top_k]

4. Update the bundle

"""
Minimize 2*n*max_conv/sum^2 using a hybrid LP-based cutting plane method
with subgradient refinement. The key innovation is maintaining a collection
of linear underestimators (cuts) of the peak self-convolution and solving
a linear program to find a balanced descent direction that reduces multiple
peaks simultaneously. Combined with adaptive subgradient steps and periodic
LP-based direction finding, this approach aims to escape shallow local
minima more effectively than pure subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_sum(a, S):
    """Rescale to have sum S, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)

    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    j_vals = np.arange(lo, hi + 1)
    grad[j_vals] = 2.0 * a[k - j_vals]
    return grad


def topk_subgradient(a, c, k_count):
    """Average subgradient over top-k peaks."""
    n = len(a)
    k_count = min(k_count, len(c))
    idx = np.argsort(c)[::-1][:k_count]
    grad = np.zeros(n)
    for j in idx:
        grad += get_subgradient_at_peak(a, int(j))
    return grad / k_count


def lse_subgradient(a, c, beta):
    """LSE-smoothed gradient of max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def solve_lp_direction(a, bundle, S):
    """
    Solve the cutting-plane LP to find a descent direction.
    bundle: list of (b_k_value, grad_array) where grad is 2*a[k-j] pattern.
    Returns the LP solution for a or None on failure.
    """
    n = len(a)
    m = len(bundle)
    if m == 0:
        return None

    # Variables: [a_0, a_1, ..., a_{n-1}, M]
    # Minimize M
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: 2*(a_0 shifted)^T * a - M <= b_0[k]
    # i.e., grad_k^T * a - M <= b_0[k]
    # where grad_k = 2 * a[k-j] pattern
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, (b_val, grad) in enumerate(bundle):
        A_ub[idx, :n] = grad
        A_ub[idx, n] = -1.0
        b_ub[idx] = b_val

    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Bounds
    bounds = [(0.0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 15})
        if result.success:
            return result.x[:n]
    except Exception:
        pass
    return None


def generate_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine raised
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.001))
    # Sine squared
    inits.append(np.sin(np.pi * t) ** 2)
    # Sine fourth
    inits.append(np.sin(np.pi * t) ** 4)
    # Gaussian
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2 / 0.1))
    # Multi-harmonic
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) +
                            0.2 * np.cos(2 * np.pi * 7 * t + 0.5), 0.01))
    # Random
    inits.append(rng.random(n) * 0.5 + 0.25)
    # Edge-weighted
    inits.append(1.0 + 1.5 * (t ** 3 + (1 - t) ** 3))
    # Blackman-like
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) +
                            0.08 * np.cos(4 * np.pi * t), 0.01))
    # Two-lobe
    inits.append(np.maximum(np.sin(2 * np.pi * t) ** 2 * np.sin(np.pi * t) ** 2 + 0.05, 0.01))

    return inits


def run_lp_subgradient_hybrid(a_init, n, S, max_time, rng, lp_every=30):
    """
    Hybrid optimization: alternating between LP-based direction finding
    and subgradient steps, with adaptive management.
    """
    a = normalize_sum(a_init, S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    # Bundle: (b_value, gradient) pairs from recent peaks
    bundle = []
    max_bundle = 40

    # Parameters
    subgrad_step = 0.3
    consecutive_fail = 0
    lp_success_count = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0

        # Update bundle with current top peaks
        top_count = min(5, 1 + it // 50)
        idx = np.argsort(c)[::-1][:top_count]
        for k in idx:
            bi = int(k)
            b_val = c[bi]
            grad = get_subgradient_at_peak(a, bi)
            bundle.append((b_val, grad))

        # Prune bundle: keep most recent and most diverse
        if len(bundle) > max_bundle:
            bundle = bundle[-max_bundle:]

        # Decide: LP step or subgradient step
        use_lp = (it % lp_every == 0) and (len(bundle) >= 3)

        if use_lp:
            a_lp = solve_lp_direction(a, bundle, S)
            if a_lp is not None:
                a_lp = project_simplex(np.maximum(a_lp, 0), S)
                obj_lp = compute_objective(a_lp)

                    a = a_lp
                    lp_success_count += 1
                    consecutive_fail = 0
                else:

                    for alpha in [0.5, 0.3, 0.1, 0.05]:
                        a_blend = (1 - alpha) * a + alpha * a_lp
                        a_blend = project_simplex(a_blend, S)
                        obj_blend = compute_objective(a_blend)
                        if obj_blend < cur_obj:
                            a = a_blend
                            improved = True
                            break
                    if not improved:
                        consecutive_fail += 1
                        lp_every = min(lp_every + 10, 200)
            else:
                consecutive_fail += 1
                lp_every = min(lp_every + 5, 200)

            if it % 7 == 0:
                # Use LSE gradient periodically

                grad = lse_subgradient(a, c, beta)

                grad = topk_subgradient(a, c, k_use)
