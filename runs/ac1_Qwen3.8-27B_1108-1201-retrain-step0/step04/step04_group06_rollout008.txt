"""
Minimize 2*n*max_conv/sum^2 using a hybrid spectral-time-domain optimization.
Combines improved spectral shaping, direct subgradient methods with away steps,
and adaptive coordinate descent to find near-optimal non-negative sequences.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Sort v in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    # Find rho: largest index where w[rho] > (cumsum[rho] - S) / rho
    theta_candidates = (cumsum - S) / idx
    valid = w > theta_candidates
    if not np.any(valid):
        return v * 0.0 + S / n
    rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def spectral_shape_v2(a, N, n, target_rel, top_k, alpha, rng):
    """
    Improved spectral shaping:
    - Compute the self-convolution
    - Find top-k peaks
    - In the frequency domain, reduce power at frequencies contributing to those peaks
    - Use a smoother, more adaptive shaping function
    """
    S = a.sum()
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Target: we want c_max * n / S^2 < target
    # i.e., c_max < target * S^2 / (2n)
    target_c = target_rel * S * S / (2.0 * n)

        return a, c_max
    
    # Find top peaks
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]

    excess = np.maximum(top_vals - target_c, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals

    weights = excess / excess_sum

    A = fft(a, N)

    # Shape: reduce where cos_weighted > 0

    P_new = P * shape
    
    # Reconstruct

    a_new = np.real(ifft(A_new))[:n]

def subgradient_step(a, S, step_size, top_k_lags):
    """
    One step of projected subgradient descent on max(b).
    Uses top-k lags for a more robust subgradient.
    Includes an "away step" for better convergence.
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # Get top-k lags
    top_indices = np.argsort(c)[::-1][:top_k_lags]
    top_vals = c[top_indices]
    
    # Weight by excess (or just use equal weights if all similar)
    max_val = top_vals[0]
    weights = np.maximum(top_vals - 0.9 * max_val, 0.0)
    if weights.sum() < 1e-30:
        weights = np.ones(top_k_lags) / top_k_lags
    else:
        weights = weights / weights.sum()
    
    # Compute combined subgradient
    g = np.zeros(n)
    for w, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            g[i] += w * 2.0 * a[k - i]
    
    # Projected gradient step with away step
    # Away step: for coordinates where g[i] > 0 and a[i] > 0, 
    # we want to decrease a[i]. The projection handles this.
    
    a_new = a - step_size * g
    a_new = project_simplex(a_new, S)
    
    return a_new


def subgradient_step_fft(a, S, step_size, top_k_lags):
    """
    FFT-accelerated subgradient step.
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    top_indices = np.argsort(c)[::-1][:top_k_lags]
    top_vals = c[top_indices]
    
    max_val = top_vals[0]
    weights = np.maximum(top_vals - 0.85 * max_val, 0.0)
    if weights.sum() < 1e-30:
        weights = np.ones(top_k_lags) / top_k_lags
    else:
        weights = weights / weights.sum()
    
    # Build weighted lag filter
    w_full = np.zeros(2 * n - 1)
    for w, k in zip(weights, top_indices):
        w_full[int(k)] = w
    
    # Subgradient: g[i] = 2 * sum_k w_k * a[k-i] (valid range)
    # This is 2 * (w_full * a) truncated to [0, n-1]
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * A))[:n]
    
    a_new = a - step_size * g
    a_new = project_simplex(a_new, S)
    
    return a_new

def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
