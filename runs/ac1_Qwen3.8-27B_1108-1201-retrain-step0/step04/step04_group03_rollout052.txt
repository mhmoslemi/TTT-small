    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 1:
        return np.array([S])

    theta = (cumsum[rho - 1] - S) / rho

"""
Minimize 2*n*max(f*f)/(sum f)^2 over non-negative sequences.
Strategy: LSE-FISTA with targeted peak deflation, diverse asymmetric initializations,
and multi-phase time-budgeted search. Uses subgradient-informed direction for exact peak handling.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_deflation(a, n, S, sharpness=40.0, lr=0.05, top_fraction=0.05):
    """Reduce a[j] where it contributes most to the convolution peak."""
    c = fft_convolve_self(a)
    cm = c.max()
    # Focus weights on top fraction of lags
    n_lags = len(c)
    top_count = max(1, int(n_lags * top_fraction))
    threshold_idx = np.argsort(c)[::-1][top_count - 1] if top_count < n_lags else c.min()
    # Soft weights: sharp near peak, zero far from peak
    w = np.zeros(n_lags)
    above = c >= threshold_idx
    w[above] = np.exp(sharpness * (c[above] - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w /= tot
    # Compute per-coordinate contribution to weighted peak
    fsize = 1 << (n_lags + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    contrib = np.real(ifft(W * np.conj(A)))[:n]
    # Multiplicative update
    scale = max(a.max(), 1e-10)
    a_new = a * np.exp(-lr * contrib / scale)
    a_new = np.maximum(a_new, 0.0)
    return project_simplex(a_new, S)


def subgrad_direction(a, n, top_k=10):
    """Compute weighted subgradient of max(f*f) using top_k active lags."""
    c = fft_convolve_self(a)
    k_actual = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k_actual]
    # Exponential weighting by closeness to max
    c_sorted = c[idx]
    w = np.exp(20.0 * (c_sorted - c_sorted[0]))
    w /= w.sum()
    grad = np.zeros(n)
    for kk, ww in zip(idx, w):
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        js = np.arange(lo, hi + 1)
        grad[js] += ww * 2.0 * a[kk - js]
    return grad, c.max()

            b0 = rng.uniform(0.1, 0.9)
            phi = rng.uniform(0, 2 * np.pi)
            p = rng.uniform(1.0, 4.0)
            t = np.arange(n) / (n - 1)
            a = np.maximum(a0 + b0 * np.cos(2 * np.pi * t + phi), 0.01) ** p
        elif family == 1:
            # Gaussian
            mu = rng.uniform(0.05, 0.95)
            sigma = rng.uniform(0.05, 0.4)
            offset = rng.uniform(0.0, 0.3)
            t = np.arange(n) / (n - 1)
            a = np.exp(-((t - mu) / sigma)**2) + offset + 0.01
        else:
            # Polynomial
            c1 = rng.uniform(0.0, 2.0)
            c2 = rng.uniform(0.0, 2.0)
            c3 = rng.uniform(0.0, 1.0)
            t = np.arange(n) / (n - 1)
            a = c1 + c2 * t + c3 * t**2 + 0.01
            a = np.maximum(a, 0.01)

        a = project_simplex(a, S)
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (a.sum()**2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 40

    global_best = np.inf
    global_a = None

    def try_update(a):
        nonlocal global_best, global_a
        if a is None or len(a) == 0:
            return
        n = len(a)
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()

    # Try to access the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 1.0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting search. Budget: %ds. Given seq available: %s" %
          (budget_s, given is not None))

    # === Phase 1: Optimize the given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        a_g = project_simplex(given / max(given.sum(), 1e-10) * S_g, S_g)
        t_avail = min(300.0, (deadline - time.monotonic()) * 0.25)
        print("Phase 1: Optimize given seq, n=%d, time=%.0fs" % (n_g, t_avail))
        a1, b1 = run_local_opt(a_g, n_g, S_g, t_avail, rng,
                                beta_init=30.0, beta_rate=0.002, beta_max=2000.0,
                                deflation_freq=150, deflation_lr=0.04,
                                subgrad_freq=400, subgrad_lr=0.003)
        try_update(a1)
        print("  Phase 1 best: %.8f" % global_best)

    # === Phase 2: Diverse asymmetric initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Choose lengths to try
        lengths_to_try = [4096, 8192, 16384]
        if given is not None and len(given) not in lengths_to_try:
            lengths_to_try.append(len(given))

        n_inits = min(14, max(6, int(remaining / 150)))
        per_init = (remaining * 0.45) / n_inits
        print("Phase 2: %d inits, remaining=%.0fs, per_init=%.0fs" %
              (n_inits, remaining, per_init))

        for i in range(n_inits):
            if time.monotonic() > deadline - 60:
                break

            # Cycle through lengths
            n = lengths_to_try[i % len(lengths_to_try)]
            S = np.sqrt(2.0 * n)

            # Generate init
            inits = generate_asymmetric_inits(n, rng, count=14)
            if i < len(inits):
                a_init = inits[i]
            else:
                a_init = rng.random(n) + 0.01

            if len(a_init) != n:
                a_init = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(a_init)), a_init)

            a_init = project_simplex(np.maximum(a_init, 0.0), S)

            t_alloc = min(per_init, deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break

            a_opt, b_opt = run_local_opt(a_init, n, S, t_alloc, rng,
                                          beta_init=15.0, beta_rate=0.003, beta_max=1000.0,
                                          deflation_freq=120, deflation_lr=0.05,
                                          subgrad_freq=300, subgrad_lr=0.003,
                                          stuck_threshold=2000)
            try_update(a_opt)
            print("  Init %d (n=%d): obj=%.8f, global=%.8f" %
                  (i, n, 2.0 * n * b_opt / S**2, global_best))

    # === Phase 3: Parametric search for good starting points ===
    remaining = deadline - time.monotonic()
    if remaining > 300 and global_a is not None:
        n_p = len(global_a)
        S_p = global_a.sum()
        t_param = min(60.0, remaining * 0.1)
        print("Phase 3: Parametric search, n=%d, time=%.0fs" % (n_p, t_param))
        a_param, obj_param = run_parametric_search(n_p, S_p, t_param, rng)
        if a_param is not None:
            try_update(a_param)
            # If parametric solution is close to best, use it for joint optimization
            if obj_param < global_best * 1.02:
                # Blend
                a_blend = project_simplex(0.5 * global_a + 0.5 * a_param, S_p)
                try_update(a_blend)

    # === Phase 4: Intensive multi-strategy polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_pol = len(global_a)
        S_pol = global_a.sum()
        print("Phase 4: Polishing n=%d, remaining=%.0fs, best=%.8f" %
              (n_pol, remaining, global_best))

        strategies = [
            # High beta pure FISTA
            dict(beta_init=200.0, beta_rate=0.0005, beta_max=5000.0,
                 deflation_freq=999999, deflation_lr=0.0,
                 subgrad_freq=999999, subgrad_lr=0.0,
                 noise_freq=10000, stuck_threshold=5000),
            # Medium beta + deflation
            dict(beta_init=50.0, beta_rate=0.002, beta_max=1500.0,
                 deflation_freq=100, deflation_lr=0.03,
                 subgrad_freq=600, subgrad_lr=0.002,
                 noise_freq=8000, stuck_threshold=3000),
            # Low beta + strong deflation
            dict(beta_init=10.0, beta_rate=0.004, beta_max=500.0,
                 deflation_freq=80, deflation_lr=0.08,
                 subgrad_freq=200, subgrad_lr=0.004,
                 noise_freq=5000, stuck_threshold=2000),
            # Very high beta + subgrad
            dict(beta_init=500.0, beta_rate=0.0003, beta_max=8000.0,
                 deflation_freq=999999, deflation_lr=0.0,
                 subgrad_freq=200, subgrad_lr=0.005,
                 noise_freq=8000, stuck_threshold=4000),
        ]

        per_strat = (remaining * 0.5) / len(strategies)
        for si, strat in enumerate(strategies):
            t_alloc = min(per_strat, deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break
            a_pol, b_pol = run_local_opt(global_a, n_pol, S_pol, t_alloc, rng, **strat)
            try_update(a_pol)
            print("  Strategy %d: %.8f (global: %.8f)" % (si, 2.0*n_pol*b_pol/S_pol**2, global_best))

    # === Phase 5: Fine-tuning with small perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_f = len(global_a)
        S_f = global_a.sum()
        n_pert = min(15, int(remaining / 5))
        print("Phase 5: %d perturbations, remaining=%.0fs" % (n_pert, remaining))

        for pi in range(n_pert):
            if time.monotonic() > deadline - 20:
                break
            # Different noise scales
            scales = [0.005, 0.01, 0.02, 0.003, 0.015]
            ns = scales[pi % len(scales)] * S_f / np.sqrt(n_f)
            a_pert = project_simplex(global_a + rng.normal(0, ns, n_f), S_f)

            t_alloc = min(3.0, (deadline - time.monotonic() - 20) / max(1, n_pert - pi))
            if t_alloc < 0.5:
                break
            a_opt, b_opt = run_local_opt(a_pert, n_f, S_f, t_alloc, rng,
                                          beta_init=100.0, beta_rate=0.003, beta_max=3000.0,
                                          deflation_freq=100, deflation_lr=0.03,
                                          stuck_threshold=1000)
            try_update(a_opt)

    # === Phase 6: Try different length (interpolation) ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_curr = len(global_a)
        # Try 2x
        n_new = n_curr * 2
        if n_new <= 65536:
            S_new = np.sqrt(2.0 * n_new)
            t_old = np.linspace(0, 1, n_curr)
            t_new = np.linspace(0, 1, n_new)
            a_interp = np.interp(t_new, t_old, global_a)
            a_interp = project_simplex(a_interp, S_new)
            t_alloc = min(100.0, (deadline - time.monotonic()) * 0.4)
            if t_alloc > 20:
                print("Phase 6: Interpolation to n=%d, time=%.0fs" % (n_new, t_alloc))
                a_new, b_new = run_local_opt(a_interp, n_new, S_new, t_alloc, rng,
                                              beta_init=20.0, beta_rate=0.002, beta_max=1000.0,
                                              deflation_freq=150, deflation_lr=0.04)
                try_update(a_new)
                print("  Phase 6: obj=%.8f (global: %.8f)" %
                      (2.0 * n_new * b_new / S_new**2, global_best))

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("FINAL: %.8f (length=%d)" % (ver, len(global_a)))
            # Check if evaluation matches our internal computation
            n_v = len(global_a)
            c_v = fft_convolve_self(global_a)
            internal = 2.0 * n_v * c_v.max() / (global_a.sum()**2)
            print("  Internal: %.8f, Eval: %.8f" % (internal, ver))
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256.0)
    else:
        print("WARNING: No solution found, using fallback")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256.0)

    result = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    print("Returning sequence of length %d" % len(result))
    return result
