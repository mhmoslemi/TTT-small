"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using multiplicative subgradient
descent with spectral flattening and multi-length search.

Key innovations over previous approach:
1. Multiplicative updates (mirror descent) that naturally maintain non-negativity
   and work in the log-domain, leading to smoother optimization.
2. FFT-based gradient computation for the subgradient (cross-correlation trick).
3. Adaptive active set: focus on peaks that are "close" to the maximum.
4. Spectral domain refinement: directly shape the power spectrum for flatness.
5. Multi-length search with careful time allocation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full linear self-convolution via FFT."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a, c=None, n=None):
    """Compute objective."""
    if n is None:
        n = len(a)
    if c is None:
        N = 1 << (2 * n - 1).bit_length()
        c = fft_convolve_self(a, N)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def multiplicative_step(a, N, n, eta, top_k, peak_fraction=0.95):
    """
    One step of multiplicative subgradient descent on log(objective).
    
    The log-objective (up to constants) is: log(max(c)) - 2*log(sum(a))
    
    Subgradient: (1/M) * grad_max_c - 2/S * ones
    where grad_max_c is a subgradient of the max of c.
    
    Multiplicative update: a <- a * exp(-eta * subgradient)
    """
    c = fft_convolve_self(a, N)
    c_max = c.max()
    S = a.sum()
    
    if c_max < 1e-30:
        return a, c, c_max, S
    
    # Find active peaks (those above peak_fraction * c_max)
    threshold = peak_fraction * c_max
    active_mask = c >= threshold
    n_active = active_mask.sum()
    
    if n_active == 0:
        # Fallback: just the maximum
        top_idx = np.array([int(np.argmax(c))])
        weights = np.array([1.0])
    elif n_active < top_k:
        # Use all active peaks
        top_idx = np.where(active_mask)[0]
        vals = c[top_idx]
        # Weight by how close to max (sharper for more active set)
        weights = (vals - threshold) / (c_max - threshold)
        weights = weights / weights.sum()
    else:
        # Use top_k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        weights = (vals - threshold) / (c_max - threshold + 1e-30)
        weights = weights / weights.sum()
    
    # Compute subgradient via cross-correlation
    # grad_i = (1/M) * sum_k w_k * a[k-i] * 1[0<=k-i<n] - 2/S
    # The first term is the cross-correlation of the weight sequence with a
    
    # Create weight sequence of length 2n-1
    W = np.zeros(2 * n - 1)
    W[top_idx] = weights / c_max  # normalize by M
    
    # Cross-correlation: h_i = sum_k W_k * a[k-i] for valid ranges
    # This equals: (W * flip(a))[i] restricted to [0, n-1]
    # Or: IFFT(FFT(W) * conj(FFT(a)))
    
    # Use FFT
    N2 = 1 << (len(W) + n - 2).bit_length()
    FW = fft(W, N2)
    FA = fft(a, N2)
    grad_corr = np.real(ifft(FW * np.conj(FA)))[:n]
    
    # Full subgradient of log-objective
    subgrad = grad_corr - 2.0 / S
    
    # Multiplicative update with clipping
    update = -eta * subgrad
    update = np.clip(update, -1.0, 1.0)  # prevent overflow
    a_new = a * np.exp(update)
    a_new = np.maximum(a_new, 1e-30)
    
    return a_new, c, c_max, S


def spectral_flatten_step(a, N, alpha):
    """
    One step of spectral flattening.
    Pushes |FFT(a)|^2 toward uniform.
    """
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Target: flat spectrum (all bins equal)
    p_mean = P.mean()
    if p_mean < 1e-30:
        return a
    
    # Interpolate toward flat
    P_new = (1 - alpha) * P + alpha * p_mean
    
    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize to same sum (objective is scale-invariant, but helps stability)
    S = a.sum()
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new * (S / s_new)
    
    return a_new

    Combines multiplicative subgradient with spectral flattening.

    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a_init / a_init.sum()  # normalize to sum=1 (scale invariant)
    
    best_a = a.copy()
    best_score = compute_score(a, n=n)
    
    if verbose:
        print("    Initial score: %.6f" % best_score)
    
    t_start = time.monotonic()
    n_iter = 0

    # Large step size, focus on top few peaks

    while time.monotonic() - t_start < phase1_budget:
        n_iter += 1
        a, c, c_max, S = multiplicative_step(a, N, n, eta, top_k, peak_frac)
        
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Adaptive step size

            eta *= 0.995
            top_k = min(top_k + 1, 100)
            peak_frac = max(peak_frac - 0.001, 0.85)

            # Add noise to escape local minima

            a = np.maximum(a + noise, 1e-30)

            elapsed = time.monotonic() - t_start

    alpha = 0.3

    """
    n = len(a)
    c_max = c.max()
    S = a.sum()
    if c_max < 1e-30 or S < 1e-30:
        return np.zeros(n)
    W = make_grad_weights(c, c_max, top_k, peak_fraction)
    FW = fft(W, N)
    FA = fft(a, N)
    gc = np.real(ifft(FW * np.conj(FA)))[:n]
    return gc / c_max - 2.0 / S


def multiplicative_step(a, c, N, eta, top_k, peak_fraction):
    """One multiplicative (mirror-descent) step on the log-objective."""
    g = subgradient_L_si(a, c, N, top_k, peak_fraction)
    a_new = a * np.exp(np.clip(-eta * g, -1.5, 1.5))
    return np.maximum(a_new, 1e-20)


def spectral_mix_step(a, N, alpha):
    """
    Blend the power spectrum |FFT(a)|^2 toward a flat spectrum by fraction alpha,
    then project back to non-negativity and renormalize.
    """
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    p_target = P.sum() / N
    P_new = (1.0 - alpha) * P + alpha * p_target
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 1e-20)
    S = a.sum()
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new * (S / s_new)
    return a_new


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


# ---------------------------------------------------------------------------
# Optimization pipeline for a single sequence length
# ---------------------------------------------------------------------------

def optimize_single(a_init, n, N, budget_s, rng, verbose=True):
    """
    Multi-phase optimizer for a fixed-length sequence.
    Phase A: aggressive multiplicative subgradient (few active peaks, large step)
    Phase B: spectral mixing (flatten power spectrum)
    Phase C: fine multiplicative subgradient (more peaks, small step)
    Phase D: gentle spectral + multiplicative interleaving
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    S0 = a.sum()
    if S0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S0  # normalize; objective is scale-invariant

    best_a = a.copy()
    c = fft_convolve_self(a, N)
    best_score = compute_score_from_c(a, c, n)

    if verbose:
        print("    Init score: %.6f" % best_score)

    t0 = time.monotonic()

    def elapsed():
        return time.monotonic() - t0

    def remaining_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)

    # ---- Phase A: aggressive multiplicative subgradient ----
    frac_A = 0.35
    t_A_end = t0 + budget_s * frac_A
    eta = 0.6
    top_k = 10
    pf = 0.97
    stall = 0
    best_phase = best_score
    it = 0

    while time.monotonic() < t_A_end:
        it += 1
        a_new = multiplicative_step(a, c, N, eta, top_k, pf)
        c = fft_convolve_self(a_new, N)
        a = a_new
        s = compute_score_from_c(a, c, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            stall = 0
            if s < best_phase:
                best_phase = s
                if verbose and it % 5000 == 0:
                    print("    A[%d] new best %.6f" % (it, best_score))
        else:
            stall += 1

        if it % 2000 == 0:
            eta *= 0.97
            top_k = min(top_k + 2, 80)
            pf = max(pf - 0.003, 0.85)
            if stall > 8000:
                noise = rng.normal(0, 0.5 / np.sqrt(n), n)
                a = np.maximum(a + noise, 1e-20)
                a = a / a.sum()
                stall = 0
                eta = min(eta * 1.5, 1.0)

        if best_score < 1.50:
            break

    if verbose:
        print("    After A (%d iters): %.6f  eta=%.4f tk=%d pf=%.3f" %
              (it, best_score, eta, top_k, pf))

    # ---- Phase B: spectral mixing ----
    frac_B = 0.20
    t_B_end = t0 + budget_s * (frac_A + frac_B)
    alpha = 0.15
    a = best_a.copy()
    stall = 0
    it = 0

    while time.monotonic() < t_B_end:
        it += 1
        a_new = spectral_mix_step(a, N, alpha)
        c_new = fft_convolve_self(a_new, N)
        s_new = compute_score_from_c(a_new, c_new, n)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            stall = 0
        else:
            # occasional accept via annealing
            delta = s_new - best_score
            temp = 0.005 * remaining_frac()
            if temp > 1e-12 and rng.random() < np.exp(-delta / temp):
                a = a_new
                c = c_new
            stall += 1
            alpha *= 0.998

        if it % 1000 == 0:
            # inject a multiplicative step to break symmetry
            a = multiplicative_step(a, c, N, eta * 0.3, top_k, pf)
            c = fft_convolve_self(a, N)
            stall = max(0, stall - 500)

        if best_score < 1.50:
            break

    if verbose:
        print("    After B (%d iters): %.6f  alpha=%.4f" % (it, best_score, alpha))

    # ---- Phase C: fine multiplicative subgradient ----
    frac_C = 0.30
    t_C_end = t0 + budget_s * (frac_A + frac_B + frac_C)
    a = best_a.copy()
    c = fft_convolve_self(a, N)
    eta = 0.15
    top_k = 50
    pf = 0.99
    stall = 0
    it = 0

    while time.monotonic() < t_C_end:
        it += 1
        a = multiplicative_step(a, c, N, eta, top_k, pf)
        c = fft_convolve_self(a, N)
        s = compute_score_from_c(a, c, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            stall = 0
            if verbose and it % 5000 == 0:
                print("    C[%d] new best %.6f" % (it, best_score))
        else:
            stall += 1

        if it % 2000 == 0:
            eta *= 0.99
            top_k = min(top_k + 5, 300)
            pf = max(pf - 0.001, 0.90)
            if stall > 6000:
                noise = rng.normal(0, 0.2 / np.sqrt(n), n)
                a = np.maximum(a + noise, 1e-20)
                a = a / a.sum()
                c = fft_convolve_self(a, N)
                stall = 0

        if best_score < 1.50:
            break

    if verbose:
        print("    After C (%d iters): %.6f  eta=%.4f tk=%d pf=%.3f" %
              (it, best_score, eta, top_k, pf))

    # ---- Phase D: interleaved gentle spectral + multiplicative ----
    t_D_end = t0 + budget_s
    a = best_a.copy()
    c = fft_convolve_self(a, N)
    eta = 0.05
    top_k = 100
    pf = 0.995
    alpha = 0.05
    stall = 0
    it = 0

    while time.monotonic() < t_D_end:
        it += 1
        if it % 3 == 0:
            a = spectral_mix_step(a, N, alpha)
        else:
            a = multiplicative_step(a, c, N, eta, top_k, pf)
        c = fft_convolve_self(a, N)
        s = compute_score_from_c(a, c, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            stall = 0
            if verbose and it % 5000 == 0:
                print("    D[%d] new best %.6f" % (it, best_score))
        else:
            stall += 1

        if it % 3000 == 0:
            eta *= 0.99
            alpha *= 0.995
            top_k = min(top_k + 5, 500)
            if stall > 10000:
                noise = rng.normal(0, 0.1 / np.sqrt(n), n)
                a = np.maximum(a + noise, 1e-20)
                a = a / a.sum()
                c = fft_convolve_self(a, N)
                stall = 0

        if best_score < 1.50:
            break

    if verbose:
        print("    After D (%d iters): %.6f" % (it, best_score))

    return best_a, best_score

def make_initializations(n, rng):
    """Produce a list of (name, array) initializations of length n."""
    inits = []
    tt = np.linspace(0.0, 1.0, n, endpoint=False)

    inits.append(("uniform", np.ones(n) / n))

    ramp = tt + 0.05
    inits.append(("ramp", ramp / ramp.sum()))

    s1 = 0.5 + 0.45 * np.sin(2 * np.pi * tt)
    inits.append(("sin1", s1 / s1.sum()))

    s3 = 0.5 + 0.45 * np.sin(2 * np.pi * 3 * tt)
    inits.append(("sin3", s3 / s3.sum()))

    chirp = 0.5 + 0.35 * np.sin(2 * np.pi * (4 * tt + 2 * tt * tt))
    inits.append(("chirp", chirp / chirp.sum()))

    multi = (0.5 + 0.15 * np.sin(2 * np.pi * 3 * tt)
             + 0.12 * np.sin(2 * np.pi * 7 * tt)
             + 0.10 * np.sin(2 * np.pi * 13 * tt))
    inits.append(("multi_sin", multi / multi.sum()))

    rnd = rng.random(n) + 0.05
    inits.append(("random", rnd / rnd.sum()))

    bump = np.exp(-0.5 * ((tt - 0.5) * 5.0) ** 2) + 0.05
    inits.append(("bump", bump / bump.sum()))

    two_bump = (np.exp(-0.5 * ((tt - 0.25) * 7.0) ** 2)
                + np.exp(-0.5 * ((tt - 0.75) * 7.0) ** 2) + 0.03)
    inits.append(("two_bump", two_bump / two_bump.sum()))

    # Low-pass random
    noise = rng.standard_normal(n)
    ksize = max(3, n // 128)
    kern = np.ones(ksize) / ksize
    smooth = np.convolve(noise, kern, mode="same")
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-12)
    inits.append(("lp_random", (0.1 + 0.9 * smooth) / (0.1 + 0.9 * smooth).sum()))

    # Binary-like (m-sequence style)
    b = (rng.random(n) > 0.5).astype(np.float64) * 0.6 + 0.2
    inits.append(("mseq", b / b.sum()))

    return inits

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Runs over multiple lengths (4096, 8192, 16384) with multi-start
    within each length. Returns the best sequence found as a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global = time.monotonic()

    # Load provided sequence if available
    provided = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if raw.size >= 512:
            raw = np.clip(raw, 0.0, 1000.0)
            if raw.sum() > 0.01:
                provided = raw
                print("Loaded provided sequence, length=%d, sum=%.4f" % (raw.size, raw.sum()))
    except Exception:
        pass

    # Lengths to try, ordered by priority
    lengths = []
    if provided is not None:
        lengths.append("provided")  # placeholder, resolved below
        plen = len(provided)
        if plen not in (4096, 8192, 16384):
            lengths.append(plen)
    for L in [8192, 16384, 4096]:
        if L not in lengths:
            lengths.append(L)

    # Allocate budget per length
    # 8192 gets the most, 16384 second, rest less
    alloc = {}
    remaining = budget_s - 40  # reserve 40s for final polish + verification
    if provided is not None and len(provided) in (4096, 8192, 16384):
        alloc[len(provided)] = remaining * 0.45
        remaining -= alloc[len(provided)]
    if 8192 in lengths and len(alloc) < len(lengths):
        alloc[8192] = remaining * 0.35
    if 16384 in alloc_lengths(alloc, lengths, 16384, remaining):
        pass  # handled below

    # Simpler allocation
    alloc = {}
    total_budget = budget_s - 40
    n_lens = len(lengths)
    # Give 8192 ~40%, 16384 ~30%, provided ~20%, 4096 ~10%
    for i, L in enumerate(lengths):
        if L in (8192,):
            alloc[L] = total_budget * 0.40
        elif L == 16384:
            alloc[L] = total_budget * 0.30
        elif L == 4096:
            alloc[L] = total_budget * 0.10
        else:
            alloc[L] = total_budget * 0.20

    # Deduplicate (if provided length == 8192, combine budgets)
    merged_alloc = {}
    for L in lengths:
        if L in merged_alloc:
            merged_alloc[L] += alloc.get(L, 0)
        else:
            merged_alloc[L] = alloc.get(L, 0)

    global_best = np.inf
    global_a = None
    elapsed_global = 0.0

    unique_lengths = []
    seen = set()
    for L in lengths:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)

    for n in unique_lengths:
        elapsed_global = time.monotonic() - t_global
        remaining = budget_s - 30 - elapsed_global
        if remaining < 20:
            break

        N = 1 << (2 * n - 1).bit_length()
        b_this = min(merged_alloc.get(n, 30.0), remaining - 10)
        if b_this < 15:
            break

        print("\n=== n=%d  FFT N=%d  budget=%.1fs ===" % (n, N, b_this))

        inits = make_initializations(n, rng)

        # Prepend provided sequence if it matches
        if provided is not None and len(provided) == n:
            inits.insert(0, ("provided", provided.astype(np.float64).copy()))

        n_inits = len(inits)
        # Reserve 25% of this length's budget for final polish
        per_init = b_this * 0.75 / n_inits
        local_best = np.inf
        local_best_a = None
        t_len = time.monotonic()

        for name, arr in inits:
            t_init = time.monotonic()
            time_left = b_this * 0.75 - (t_init - t_len)
            if time_left < 5:
                break

            b_init = min(per_init, time_left)
            a0 = np.asarray(arr, dtype=np.float64).copy()
            a0 = np.maximum(a0, 0.0)
            s0 = a0.sum()
            if s0 < 1e-15:
                a0 = np.ones(n) / n
            else:
                a0 = a0 / s0

            print("  [%s] starting (budget=%.1fs)" % (name, b_init))
            a_opt, sc = optimize_single(a0, n, N, b_init, rng, verbose=(n >= 8192))

            if sc < local_best:
                local_best = sc
                local_best_a = a_opt.copy()
                print("  [%s] local best updated: %.6f" % (name, local_best))

            if local_best < 1.50:
                break

        # Final polish on best for this length
        if local_best_a is not None:
            t_polish_start = time.monotonic()
            polish_budget = b_this - (t_polish_start - t_len)
            if polish_budget > 8:
                print("  [polish] budget=%.1fs" % polish_budget)
                a_final, sc_final = optimize_single(local_best_a, n, N, polish_budget,
                                                    rng, verbose=True)
                if sc_final < local_best:
                    local_best = sc_final
                    local_best_a = a_final.copy()

        if local_best_a is not None and local_best < global_best:
            global_best = local_best
            global_a = local_best_a.copy()
            print("  >>> Global best: %.6f at n=%d" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f  (global: %.6f)" %
                  (n, local_best, global_best))

    # ---- Global final polish ----
    if global_a is not None:
        remaining = budget_s - 20 - (time.monotonic() - t_global)
        if remaining > 15:
            n = len(global_a)
            N = 1 << (2 * n - 1).bit_length()
            print("\n=== Global final polish: n=%d budget=%.1fs ===" % (n, remaining))
            a_fin, sc_fin = optimize_single(global_a, n, N, remaining, rng, verbose=True)
            if sc_fin < global_best:
                global_best = sc_fin
                global_a = a_fin.copy()

    # ---- Fallback ----
    if global_a is None or not np.all(np.isfinite(global_a)):
        print("Falling back to uniform sequence.")
        global_a = np.ones(8192, dtype=np.float64)
        global_best = 2.0

    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64) / 8192

    result = [float(x) for x in global_a]

    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(result)
        if np.isfinite(ver):
            print("\nVerification: evaluate_sequence = %.6f  (internal: %.6f)" % (ver, global_best))
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence returned non-finite! Using fallback.")
            global_a = np.ones(8192, dtype=np.float64)
            result = [float(x) for x in global_a]
            ver = evaluate_sequence(result)
            print("Fallback verify: %.6f" % ver)
    except Exception as e:
        print("Verification error: %s" % str(e))

    print("\nFinal: length=%d  score=%.6f" % (len(global_a), global_best))
    return result


def alloc_lengths(alloc, lengths, target, remaining):
    """Helper (placeholder for logic clarity – returns list of lengths to alloc)."""
    out = []
    for L in lengths:
        if L == target and remaining > 10:
            out.append(L)
    return out
