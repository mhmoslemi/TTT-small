def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    cm = c.max()
    e = np.exp(beta * (c - cm))

    if tot < 1e-300:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. FISTA with adaptive LSE smoothing (improved beta scheduling, backtracking)
2. Peak-targeting coordinate descent (focus on reducing largest convolution peaks)

4. Random restarts and L-BFGS polish on the smoothed objective
5. Multiple diverse initializations and length exploration

Key improvements: better gradient computation via FFT correlation,
adaptive step-size with backtracking, peak-focused coordinate descent,
and careful time budget allocation.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """
    Compute gradient of LSE approximation to max(c) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    Returns (lse_value, gradient).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    p = e / tot
    lse_val = cm + np.log(tot) / beta

    # Gradient: grad[j] = (2/beta) * sum_m p[m+j] * a[m] for j=0..n-1
    # This is cross-correlation of p (len 2n-1) and a (len n)
    F = 1 << (2 * n).bit_length()
    P = fft(p, F)
    A = fft(a, F)
    grad = (2.0 / beta) * np.real(ifft(np.conj(P) * A))[:n]
    return lse_val, grad


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    # grad[j] = 2*a[k-j] if 0 <= k-j < n
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if hi >= lo:
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0,
              beta_ramp_frac=0.3, use_hybrid=True, noise_period=2000):
    """FISTA with adaptive LSE smoothing, backtracking, and noise injection."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta_ramp_iters = int(max_time * beta_ramp_frac / 0.05)  # estimate

    c = fft_convolve_self(y)
    if c.max() < best:
        best = c.max()
        best_a = y.copy()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: smooth sigmoid ramp
        progress = min(1.0, it / max(beta_ramp_iters, 100))
        beta = beta_init + (beta_max - beta_init) * (1.0 - np.exp(-3.0 * progress))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-10:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient type
        use_exact = (use_hybrid and it > 500 and
                     rng.random() < min(0.2, 0.001 * it))

        if use_exact:
            grad = exact_subgradient(y, c)
        else:
            _, grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        if use_exact:
            step = S / (gnorm * (1.0 + 0.005 * np.log1p(it / 500.0)))
        else:
            step = 1.0 / (0.5 * beta)
            # Reduce if stuck
            if it - last_improve > 2000:
                step *= 0.5
            # Increase slightly if improving fast
            if it - last_improve < 50 and it > 100:
                step *= 1.2

        # Gradient step with projection
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum with restart
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        if it - last_improve > 3000:
            t_mom = 1.0  # restart momentum
            t_new = 1.0
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new

        # Periodic noise injection
        if it % noise_period == 0:
            decay = np.exp(-it / (10 * noise_period))
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def peak_targeting_cd(a_init, n, S, max_time, rng):
    """
    Coordinate descent that focuses on reducing the largest convolution peaks.
    Uses paired transfers with line search, prioritizing pairs that affect
    the current maximum lags.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    cur = c.max()
    best = cur
    best_a = a.copy()

    # Track top peaks
    n_peaks = min(10, 2 * n - 1)
    peak_indices = np.argsort(c)[::-1][:n_peaks]

    it = 0
    stagnant_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        improved_this_round = False

        # Update peak tracking every 50 iterations
        if it % 50 == 1:
            peak_indices = np.argsort(c)[::-1][:n_peaks]
            cur_max = c.max()
            if cur_max < best:
                best = cur_max
                best_a = a.copy()
                stagnant_count = 0
            else:
                stagnant_count += 1

        # Number of pair tries per round
        n_try = min(30 + it // 20, 150)

        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break

            # Choose pair: bias towards lags near the peaks
            r = rng.random()
            if r < 0.4:
                # Target a peak lag: find pairs that affect it
                k_peak = peak_indices[rng.integers(0, n_peaks)]
                # A pair (j, m) affects lag k if |j-k| or |m-k| is small
                # Choose j near the peak lag
                j = int(np.clip(k_peak + rng.integers(-5, 6), 0, n - 1))
                m = int(np.clip(j + rng.integers(1, max(2, n // 100)), 0, n - 1))
            elif r < 0.7:
                # Local pair
                j = rng.integers(0, n - 1)
                m = int(np.clip(j + rng.integers(1, 10), 0, n - 1))
            else:
                # Random pair
                j = rng.integers(0, n)
                m = rng.integers(0, n)

            if j == m:
                continue
            if a[j] < 1e-12 or a[m] < 1e-12:
                continue

            # Compute direction of change in convolution
            # Transfer delta from m to j: a[j]+=delta, a[m]-=delta
            # d_c[k] = a[j+k_valid] + a[j-k_valid] - a[m+k_valid] - a[m-k_valid]
            # More precisely: d_c[k] = (a[k-j] if valid) - (a[k-m] if valid)
            #                    + (a[j-k] if valid) - (a[m-k] if valid)
            # But with the correction for the diagonal...
            # Simple approach: d_c[k] = a[k-j] - a[k-m] + a[j-k] - a[m-k]
            # (with out-of-range = 0)

            # For efficiency, compute only for a subset of lags
            # Actually let's compute for all lags but vectorize
            ks = np.arange(2 * n - 1)
            kj = ks - j  # k - j
            km = ks - m  # k - m
            jk = j - ks  # j - k
            mk = m - ks  # m - k

            val_kj = np.where((kj >= 0) & (kj < n), a[kj], 0.0)
            val_km = np.where((km >= 0) & (km < n), a[km], 0.0)
            val_jk = np.where((jk >= 0) & (jk < n), a[jk], 0.0)
            val_mk = np.where((mk >= 0) & (mk < n), a[mk], 0.0)

            d = val_kj - val_km + val_jk - val_mk

            # For k=0: there's a quadratic term (delta^2) in c[0]
            # c[0] = a[0]^2. If j=0: (a[0]+d)^2 - a[0]^2 = 2*a[0]*d + d^2
            # If m=0: (a[0]-d)^2 - a[0]^2 = -2*a[0]*d + d^2
            # Quadratic coefficient in c[0]:
            quad_0 = 0.0
            if j == 0:
                quad_0 += 1.0
            if m == 0:
                quad_0 += 1.0

            # Also for the lag where j and m "meet": lag = j+m or |j-m|
            # Actually the quadratic terms appear at lags where both j and m
            # contribute to the same term. This is复杂... let's just handle k=0
            # and use linear approximation elsewhere (good for small delta).

            # Bisection/ternary search for optimal delta
            lo = -a[j] * 0.99
            hi = a[m] * 0.99

            def f_delta(dlt):
                vals = c + dlt * d
                if quad_0 > 0:
                    vals[0] += quad_0 * dlt * dlt
                return vals.max()

            # Golden section search (20 iterations is enough)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = f_delta(c_g)
            fd = f_delta(d_g)

            for _ in range(30):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = np.clip(delta, lo, hi)

            if abs(delta) < 1e-12:
                continue

            # Apply and check
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            if a_trial[j] < -1e-12 or a_trial[m] < -1e-12:
                continue

            c_trial = c + delta * d
            if quad_0 > 0:
                c_trial[0] += quad_0 * delta * delta

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-10:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved_this_round = True

        # If no improvement, add noise
        if not improved_this_round and it > 3:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant_count / 20.0)
            if noise_scale < 1e-10:
                if stagnant_count > 100:
                    break
            else:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()

    return best_a, best

            # Hmm let me just compute it directly.
            # The gradient of the smoothed objective w.r.t. a is grad_base.
            # By chain rule: grad_params[j] = sum_i grad_base[i] * da_i/dp_j
            # da_i/dp_j = (S/s)(delta_ij - a_i/S)
            # grad_params[j] = sum_i grad_base[i] * (S/s)(delta_ij - a_i/S)
            #               = (S/s) * grad_base[j] - (S/s) * (1/S) * sum_i grad_base[i] * a_i
            #               = (S/s) * grad_base[j] - (1/s) * sum_i grad_base[i] * a_i

            s_val = s
            sum_ga = np.dot(grad_base, a)
            grad = (S / s_val) * grad_base - (1.0 / s_val) * sum_ga
            # For L-BFGS-B with bounds, we can just return grad_base
            # since we clip to non-negative. The bound constraint is handled.
            # But the normalization makes it tricky. Let's use a simpler approach:
            # just return grad_base and let the bounds handle non-negativity.
            # The normalization is approximate (we re-normalize each step).

    # Simpler approach: just do gradient descent with the projected gradient
    # Since L-BFGS-B doesn't handle the sum constraint well.
    # Let's do a simple accelerated gradient method instead.

    a = a0.copy()
    y = a.copy()
    t_mom = 1.0
    step = 1.0 / (0.5 * beta)

    for it in range(200):
        if time.monotonic() - start > max_time - 1:
            break

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        if it < 50:
            beta_cur = beta * 0.5
        elif it < 150:
            beta_cur = beta
        else:
            beta_cur = beta * 2.0

        _, grad = lse_gradient(y, c, beta_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Linesearch (backtracking)
        step_cur = step
        for ls in range(10):
            x_new = project_simplex(y - step_cur * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur - 1e-12 * step_cur * gnorm:
                break
            step_cur *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new
        step = step_cur * 1.05  # slight increase if linesearch accepted

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def make_variants(a, n, rng, S):
    """Generate diverse variants of a sequence for exploration."""
    variants = [a.copy()]

    # Palindromic
    half = a[:n // 2]
    if n % 2 == 0:
        pal = np.concatenate([half, half[::-1]])
    else:
        pal = np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])
    variants.append(normalize_sum(pal, n))

    # Small noise perturbations
    for scale in [0.01, 0.05, 0.1, 0.2]:
        noise = rng.normal(0, scale * S / np.sqrt(n), n)
        v = project_simplex(a + noise, S)
        variants.append(v)

    # Low-pass filtered
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    cutoff = 0.2
    F_lp = F * np.exp(-20.0 * (freqs / cutoff) ** 2)
    a_lp = np.real(ifft(F_lp))[:n]
    a_lp = np.maximum(a_lp, 0)
    s = a_lp.sum()
    if s > 1e-15:
        variants.append(a_lp * (S / s))

    # Spectral rotation (add linear phase)
    for rot in [0.1, -0.1, 0.3]:
        phases = 1.0 / np.maximum(np.abs(F), 1e-30)
        F_rot = F * np.exp(2j * np.pi * freqs * rot)
        a_rot = np.real(ifft(F_rot))[:n]
        a_rot = np.maximum(a_rot, 0)
        s = a_rot.sum()
        if s > 1e-15:
            variants.append(a_rot * (S / s))

    return variants


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses multi-phase optimization: FISTA, coordinate descent, spectral, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 128 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent=%s" % (
        n, budget_s, "yes" if parent is not None else "no"))
    if parent is not None:
        c_p = fft_convolve_self(normalize_sum(parent, n))
        print("  Parent score: %.6f" % c_p.max())

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: score=%.6f (elapsed %.0fs)" % (
                label, len(a_norm), score, elapsed))
        return score

    # Build starting points
    starts = []
    if parent is not None:
        starts.append(("parent", normalize_sum(parent, n)))
        # Generate variants
        variants = make_variants(normalize_sum(parent, n), n, rng, S)
        for i, v in enumerate(variants[:5]):
            starts.append(("var_%d" % i, v))

    # Structured starts
    tt = np.linspace(0, 1, n, endpoint=False)
    starts.append(("sinc", np.maximum(np.sinc(tt - 0.5) ** 2, 0.01)))
    starts.append(("gauss", np.exp(-6.0 * (tt - 0.5) ** 2)))
    starts.append(("cos", np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01)))
    starts.append(("rand", rng.random(n) + 0.1))

    # ===== Phase 1: FISTA on diverse starts (35% budget) =====
    print("\n=== Phase 1: FISTA ===")
    p1_budget = (deadline - t_start) * 0.35
    n_starts = len(starts)
    # Weight parent more heavily
    weights = np.ones(n_starts)
    if parent is not None:
        weights[0] = 4.0
        if n_starts > 1:
            weights[1:4] = 2.0  # variants get more weight
    weights = weights / weights.sum()
    times = p1_budget * weights

    for i, (name, arr) in enumerate(starts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(times[i], remaining * 0.3)
        if t_alloc < 5:
            continue
        a_opt, score = run_fista(arr, n, S, t_alloc, rng,
                                  beta_init=5.0, beta_max=300.0,
                                  beta_ramp_frac=0.4, use_hybrid=True)
        update_best(a_opt, "fista_" + name)

    print("After Phase 1: best=%.6f" % global_best)

    # ===== Phase 2: Peak-targeting coordinate descent (30% budget) =====
    print("\n=== Phase 2: Peak-Targeting CD ===")
    if global_a is not None:
        p2_budget = (deadline - time.monotonic()) * 0.30
        if p2_budget > 10:
            a_cd, _ = peak_targeting_cd(global_a, n, S, p2_budget, rng)
            update_best(a_cd, "cd")

    print("After Phase 2: best=%.6f" % global_best)

    # ===== Phase 3: Spectral refinement (15% budget) =====
    print("\n=== Phase 3: Spectral ===")
    if global_a is not None:
        p3_budget = (deadline - time.monotonic()) * 0.15
        if p3_budget > 10:
            a_sp, _ = spectral_refine(global_a, n, S, p3_budget * 0.6, rng)
            update_best(a_sp, "spectral")
            # Run CD after spectral
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cd2, _ = peak_targeting_cd(global_a, n, S, remaining * 0.7, rng)
                update_best(a_cd2, "cd_after_spec")

    print("After Phase 3: best=%.6f" % global_best)

    # ===== Phase 4: Final FISTA polish with high beta (15% budget) =====
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fin, _ = run_fista(global_a, n, S, remaining * 0.5, rng,
                                  beta_init=50.0, beta_max=500.0,
                                  beta_ramp_frac=0.3, use_hybrid=True,
                                  noise_period=5000)
            update_best(a_fin, "final_fista")

    # ===== Phase 5: Local coordinate descent polish =====
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print("\n=== Phase 5: Local CD Polish ===")
        a_pol, _ = peak_targeting_cd(global_a, n, S, remaining * 0.8, rng)
        update_best(a_pol, "final_cd")

    # ===== Phase 6: Try alternate length if time permits =====
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Phase 6: Alternate Length ===")
        # Try doubling or halving the length
        for alt_factor in [2, 0.5]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            alt_n = int(n * alt_factor)
            if alt_n < 1024 or alt_n > 65536:
                continue
            alt_S = np.sqrt(2.0 * alt_n)

            # Resample
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                nblocks = n // alt_n
                if nblocks < 1:
                    continue
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alt = remaining * 0.35
            a_alt, _ = run_fista(alt_start, alt_n, alt_S, t_alt, rng,
                                  beta_init=5.0, beta_max=200.0)
            update_best(a_alt, "alt_n_%d" % alt_n)

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        # Fallback
        fb_n = 256
        fb_S = np.sqrt(2.0 * fb_n)
        global_a = np.ones(fb_n) * (fb_S / fb_n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        fb_n = 256
        fb_S = np.sqrt(2.0 * fb_n)
        global_a = np.ones(fb_n) * (fb_S / fb_n)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        fb_n = 256
        fb_S = np.sqrt(2.0 * fb_n)
        global_a = np.ones(fb_n) * (fb_S / fb_n)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, total_time=%.0fs" % (
        len(global_a), time.monotonic() - t_start))
    return final_list
