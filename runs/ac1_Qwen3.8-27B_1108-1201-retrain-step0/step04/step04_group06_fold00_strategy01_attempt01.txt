def spectral_exp_shape(a, target):
    """Convolve two sequences and update by weighting top contributing frequencies."""
    n = len(a)

        return None, np.inf

    A = fft(a)
    c = np.real(ifft(A * A))
    
    # Target the top-k peak lags
    top_k = min(50, n//2)
    peak_lags = np.argsort(c[:n-1])[::-1][:top_k]
    
    # If all peak lags are below target, no improvement
    if np.max(c) < target:
        return a, np.max(c)
    
    # Find which coordinates contribute to these peak lags
    # We have c[k] = sum_{i} a[i] * a[i+k] for k < n
    # and similarly for k>=n using symmetric view
    total_energy = np.array([c[k] if k < n else c[2*n-2-k] for k in range(top_k)])
    effective_contrib = np.zeros(n)
    for i in range(n):
        if i >= n:
            break
        # For each peak lag considered, a[i] contributes to a[i+k]
        lag_contrib = 0.0
        high_contribs = set(peak_lags)
        for k in peak_lags:
            if k + i < n:
                lag_contrib += a[i] * c[k]
        effective_contrib[i] = lag_contrib

    # Exclude zeros to avoid massive changes
    tmp_contrib = effective_contrib[effective_contrib > 0]
    if tmp_contrib.size == 0:
        return a, np.max(c)

    lag_indices = set(peak_lags)
    better_contribs = []
    for j in range(n):

        if j == 0 or lag_indices.isdisjoint({j-1}):

        # Check that j-1 is contributing to the lags we care about
