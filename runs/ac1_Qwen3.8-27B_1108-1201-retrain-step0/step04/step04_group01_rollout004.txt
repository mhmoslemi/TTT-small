"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Refined FISTA with adaptive LSE smoothing, backtracking line search,
exact subgradient polishing, and multi-phase beta scheduling. Starts from the
given sequence and refines through progressively higher beta values.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Smooth approximation of max(c)."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    return cm + np.log(np.exp(scaled).sum()) / beta


def compute_weighted_gradient(y, w, fsize):
    """Compute gradient: 2 * cross_correlation(w, y) truncated to len(y)."""
    n = len(y)
    A = fft(y, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def lse_weights(c, beta):
    """Compute softmax weights for LSE."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def topk_weights(c, k):
    """Weights concentrated on top-k peaks, proportional to excess above mean."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.zeros(len(c))
    vals = np.maximum(c[idx] - mean_c, 0.0)
    vs = vals.sum()
    if vs < 1e-30:
        weights[idx] = 1.0 / k
    else:
        weights[idx] = vals / vs
    return weights

                    use_backtrack=True, restart_interval=2000, noise_base=0.003):

    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    it = 0
    best_a = a.copy()
    best_val = fft_convolve_self(a).max()
    step = 1.0 / (4.0 * (1.0 + beta_start))
    last_improve = 0
    stall_count = 0

    n = len(a)

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        w = lse_weights(c, beta)
        grad = compute_weighted_gradient(y, w, fsize)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            f_y = lse_value(c, beta)
            ls_step = step
            for _ in range(20):
                x_new = project_simplex(y - ls_step * grad, S)

            n_target = len(given)
            print("Using given sequence of length %d, sum=%.4f" % (n_target, a_start.sum()))
    except Exception:
        pass

    if a_start is None:
        a_start = np.ones(n_target) * 0.5
        print("No given sequence, using uniform init of length %d" % n_target)

    n = len(a_start)
    S = np.sqrt(2.0 * n)
    a_start = normalize_to_sum(a_start, S)
    fsize = 1 << (3 * n - 1).bit_length()

    # Initial evaluation
    c0 = fft_convolve_self(a_start)
    obj0 = 2.0 * n * c0.max() / (a_start.sum() ** 2)
    print("Initial: n=%d, max_b=%.10f, obj=%.8f" % (n, c0.max(), obj0))

    best_a = a_start.copy()
    best_obj = obj0

    def check_and_update(a_cand, label):
        nonlocal best_a, best_obj
        a_cand = normalize_to_sum(a_cand, S)
        c_cand = fft_convolve_self(a_cand)
        obj_cand = 2.0 * n * c_cand.max() / (a_cand.sum() ** 2)
        if obj_cand < best_obj:
            best_a = a_cand.copy()
            best_obj = obj_cand
            print("  %s: NEW BEST obj=%.8f (max_b=%.10f)" % (label, best_obj, c_cand.max()))
        return a_cand, obj_cand

    # Phase 1: Moderate beta FISTA (400s)
    t1 = min(400.0, deadline - time.monotonic() - 20.0)
    if t1 > 30:
        print("\n=== Phase 1: FISTA beta 10->2000 (%.0fs) ===" % t1)
        a1, _ = run_fista_phase(a_start, S, t1, 10.0, 2000.0, rng, fsize,
                                use_backtrack=True, restart_interval=1500, noise_base=0.005)
        a1, _ = check_and_update(a1, "Phase1")

    # Phase 2: High beta FISTA (1000s)
    t2 = min(1000.0, deadline - time.monotonic() - 20.0)
    if t2 > 30:
        print("\n=== Phase 2: FISTA beta 2000->50000 (%.0fs) ===" % t2)
        a2, _ = run_fista_phase(best_a, S, t2, 2000.0, 50000.0, rng, fsize,
                                use_backtrack=True, restart_interval=2000, noise_base=0.003)
        a2, _ = check_and_update(a2, "Phase2")

    # Phase 3: Exact subgradient polishing (500s)
    t3 = min(500.0, deadline - time.monotonic() - 20.0)
    if t3 > 30:
        print("\n=== Phase 3: Exact subgradient (%.0fs) ===" % t3)
        a3, _ = run_exact_subgradient(best_a, S, t3, rng, fsize, k_schedule=(4, min(32, n // 256)))
        a3, _ = check_and_update(a3, "Phase3")

    # Phase 4: Adaptive refinement with remaining time
    t4 = deadline - time.monotonic() - 10.0
    if t4 > 60:
        print("\n=== Phase 4: Adaptive refinement (%.0fs) ===" % t4)
        a4, _ = run_fista_phase(best_a, S, t4 * 0.5, 10000.0, 200000.0, rng, fsize,
                                use_backtrack=True, restart_interval=3000, noise_base=0.001)
        a4, _ = check_and_update(a4, "Phase4a")

        t4b = deadline - time.monotonic() - 10.0
        if t4b > 30:
            a4b, _ = run_exact_subgradient(best_a, S, t4b, rng, fsize, k_schedule=(8, min(64, n // 128)))
            a4b, _ = check_and_update(a4b, "Phase4b")

    # Final: try a few random perturbations of the best
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 30:
        print("\n=== Final perturbations (%.0fs) ===" % remaining)
        per_try = max(5.0, remaining / 10)
        for trial in range(10):
            if time.monotonic() > deadline - 5:
                break
            noise_scale = 0.001 * S / np.sqrt(n)
            perturbed = normalize_to_sum(best_a + rng.normal(0, noise_scale, n), S)
            a_p, _ = run_fista_phase(perturbed, S, per_try, 5000.0, 50000.0, rng, fsize,
                                     use_backtrack=True, restart_interval=500, noise_base=0.0)
            a_p, _ = check_and_update(a_p, "Perturb%d" % trial)

    # Verify
    final_a = np.clip(best_a, 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    print("\nFinal verification: obj=%.8f (n=%d)" % (ver, len(final_a)))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        final_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in final_a]

    print("Returning sequence of length %d, obj=%.8f" % (len(final_a), ver))
    return final_list
