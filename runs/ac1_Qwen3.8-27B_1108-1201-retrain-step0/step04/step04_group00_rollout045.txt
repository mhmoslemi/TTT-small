Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-scale optimization combining:

2. Spectral flattening (GS-like) for diverse exploration

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective: 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def make_init_cosine(n, alpha=0.5):
    """Raised cosine: a[i] = 1 - alpha*(1-cos(2*pi*i/n))/2"""
    t = np.arange(n) / n
    return 1.0 - alpha * 0.5 * (1.0 - np.cos(2 * np.pi * t))


def make_init_hamming(n):
    """Hamming window."""
    t = np.arange(n) / n
    return 0.54 + 0.46 * np.cos(2 * np.pi * t)


def make_init_flat_noise(n, rng, noise_frac=0.1):
    """Flat with small noise."""
    a = np.ones(n)
    a += rng.normal(0, noise_frac, n)
    return np.maximum(a, 0.01)


def make_init_bimodal(n, rng):
    """Two Gaussian bumps."""
    t = np.arange(n) / n
    sigma = 0.15
    a = np.exp(-((t - 0.25) / sigma) ** 2) + np.exp(-((t - 0.75) / sigma) ** 2)
    return a


def make_init_random_exp(n, rng):
    """Exponential random."""
    return rng.exponential(1.0, n)


def normalize_for_obj(a, n):
    """Normalize so sum(a) = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) * S / n
    return a * (S / s)


def run_projected_gradient(a_init, max_time, rng, beta_init=20.0, beta_rate=0.004,
                           beta_max=20000.0, use_hybrid=True, noise_adaptive=True):
    """
    Projected gradient with backtracking and adaptive LSE/exact hybrid.
    Returns (best_a, best_conv_max).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / beta_init

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive gradient selection
        if use_hybrid:
            if stuck < 100:
                grad = lse_gradient(y, c, beta)
            elif stuck < 500:
                # Blend LSE and top-k
                g_lse = lse_gradient(y, c, beta)
                k = max(4, min(64, n // 16))
                g_topk = topk_gradient(y, c, k, power=1.5)
                blend = min(1.0, (stuck - 100) / 400.0)
                grad = (1 - blend) * g_lse + blend * g_topk
            else:
                # Focus on top few
                k = min(32, max(2, stuck // 100))
                grad = topk_gradient(y, c, k, power=2.0)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / max(beta, 1.0)
        max_bt = 8

        def obj_at(x):
            cx = fft_convolve_self(x)
            return cx.max()

        base_obj = obj_at(y)
        accepted = False
        for bt in range(max_bt):
            x_new = project_simplex(y - step * grad, S)
            new_obj = obj_at(x_new)
            if new_obj <= base_obj + 1e-14:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Just do a small step
            x_new = project_simplex(y - step * grad, S)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if noise_adaptive and stuck > 2000:
            noise_scale = max(1e-5, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

        # Periodic restart
        if it % 20000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_gs_spectral(a_init, max_time, rng, n_iters=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Iterates between: (1) flatten |DFT|^2, (2) enforce a >= 0.
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    if n_iters is None:
        n_iters = 100000

    target_spec = a.sum() / np.sqrt(n)

    while time.monotonic() - start < max_time and it < n_iters:
        it += 1

        # A = DFT of a
        A = fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        # Blend current magnitude toward flat target
        # Use soft blending: new_mag = (1-blend)*mag + blend*target
        blend = 0.1 + 0.3 * min(1.0, it / 5000.0)
        new_mag = (1 - blend) * mag + blend * target_spec
        # Ensure DC component stays large (it's sum(a))
        new_mag[0] = a.sum()

        # Reconstruct
        a_new = np.real(ifft(new_mag * np.exp(1j * phase)))
        a_new = np.maximum(a_new, 0.0)

        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)

        a = a_new

        # Check improvement
        cur_obj = compute_obj(a)
        if cur_obj < best_obj - 1e-8:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Add perturbation if stuck
        if it - last_improve > 500 and it % 100 == 0:
            A = fft(a)
            A += rng.normal(0, 0.01 * target_spec, n) + 1j * rng.normal(0, 0.01 * target_spec, n)
            A[0] = np.abs(A[0])  # Keep DC real positive
            a = np.real(ifft(A))
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())

    return best_a, best_obj

def upsample(a, factor=2):
    """Upsample with linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        print("\n=== Phase 4: Final polish at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        # Active-set polish
        a_pol4, _ = run_active_polish(global_best_a, p4_budget * 0.4, rng)
        obj_pol4 = compute_obj(a_pol4)
        print("  active polish: obj=%.6f (t=%.0fs)" % (obj_pol4, time.monotonic() - t4))
        if obj_pol4 < global_best_obj:
            global_best_obj = obj_pol4
            global_best_a = a_pol4.copy()

        # High-beta FISTA-style
        remaining4 = p4_budget - (time.monotonic() - t4)
        if remaining4 > 20:
            a_hi, _ = run_projected_gradient(global_best_a, remaining4 * 0.5, rng,
                                              beta_init=100.0, beta_rate=0.002,
                                              beta_max=50000.0)
            obj_hi = compute_obj(a_hi)
            print("  high-beta: obj=%.6f (t=%.0fs)" % (obj_hi, time.monotonic() - t4))
            if obj_hi < global_best_obj:
                global_best_obj = obj_hi
                global_best_a = a_hi.copy()

        # One more round of active polish
        remaining4b = p4_budget - (time.monotonic() - t4)
        if remaining4b > 10:
            a_pol4b, _ = run_active_polish(global_best_a, remaining4b, rng)
            obj_pol4b = compute_obj(a_pol4b)
            print("  final active: obj=%.6f" % obj_pol4b)
            if obj_pol4b < global_best_obj:
                global_best_obj = obj_pol4b
                global_best_a = a_pol4b.copy()

    # === Phase 5: Try n=32768 if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(global_best_a) < 32768:
        p5_budget = remaining * 0.8
        n5 = 32768
        print("\n=== Phase 5: n=%d (budget %.0fs) ===" % (n5, p5_budget))
        t5 = time.monotonic()

        a_up5 = upsample(global_best_a, factor=n5 // len(global_best_a))
        a_up5 = normalize_for_obj(a_up5, n5)

        a_res5, _ = run_projected_gradient(a_up5, p5_budget * 0.6, rng,
                                            beta_init=5.0, beta_rate=0.003,
                                            beta_max=20000.0)
        obj_res5 = compute_obj(a_res5)
        print("  n=%d: obj=%.6f (t=%.0fs)" % (n5, obj_res5, time.monotonic() - t5))
        if obj_res5 < global_best_obj:
            global_best_obj = obj_res5
            global_best_a = a_res5.copy()

        remaining5 = p5_budget - (time.monotonic() - t5)
        if remaining5 > 10:
            a_pol5, _ = run_active_polish(global_best_a, remaining5, rng)
            obj_pol5 = compute_obj(a_pol5)
            if obj_pol5 < global_best_obj:
                global_best_obj = obj_pol5
                global_best_a = a_pol5.copy()
                print("  after polish: obj=%.6f" % obj_pol5)

    # Final verification
    print("\n=== Final Result ===")
    print("Best objective: %.6f (n=%d)" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, trying fallback")
            # Fallback: use the initial

                fb = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
                fb = np.clip(fb, 0, 1000)
                final_list = [float(x) for x in fb]
            except Exception:
                n_fb = 8192
                final_list = [float(x) for x in np.ones(n_fb) * np.sqrt(2.0 / n_fb)]
    else:
        n_fb = 8192
        final_list = [float(x) for x in np.ones(n_fb) * np.sqrt(2.0 / n_fb)]
