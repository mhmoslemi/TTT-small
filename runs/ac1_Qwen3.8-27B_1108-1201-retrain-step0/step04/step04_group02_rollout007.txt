"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses Nesterov-accelerated mirror descent with annealed log-sum-exp smoothing,
adaptive step size (BB + backtracking), multi-start from diverse asymmetric
initializations, and multi-length exploration (8192, 16384).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_step(a, grad, eta, S):
    """Multiplicative (mirror) descent maintaining sum=S, a>=0."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """LSE-smoothed gradient of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def run_mirror_nesterov(a_init, n, S, max_time, rng, use_asymmetric_init=True):
    """
    Nesterov-accelerated mirror descent with annealed LSE smoothing.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0
    consecutive_fail = 0
    beta_start = 5.0
    beta_end = 400.0
    beta_anneal_rate = 0.015  # per iteration

    # BB step size components
    prev_grad = None
    prev_a = None
    bb_gamma = 1.0

    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            consecutive_fail = 0

        # Anneal beta
        beta = min(beta_end, beta_start * np.exp(beta_anneal_rate * it))

        # Compute gradient at y (Nesterov lookahead)
        c_y = fft_convolve_self(y)
        if beta > 50:
            grad = lse_gradient(y, c_y, beta)
        else:
            # Blend: mix LSE and top-k for stability at low beta
            k_use = min(16, max(2, int(beta)))
            grad_k = topk_gradient(y, c_y, k_use)
            grad_lse = lse_gradient(y, c_y, beta)
            mix = min(1.0, beta / 40.0)
            grad = mix * grad_lse + (1 - mix) * grad_k

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: BB if available, else adaptive
        if prev_grad is not None:
            diff_a = y - prev_a
            diff_g = grad - prev_grad
            sq = float(np.dot(diff_g, diff_g))
            dy = float(np.dot(diff_a, diff_g))
            if sq > 1e-20 and abs(dy) > 1e-20:
                bb_step = abs(dy) / sq
                bb_step = min(bb_step, 5.0 / gnorm)
                step = 0.7 * step + 0.3 * bb_step
        else:
            step = min(step, 2.0 * S / (n * gnorm + 1e-15))

        # Mirror descent step
        eta = step / max(gnorm, 1e-15)
        eta = min(eta, 10.0)
        x_new = mirror_step(y, grad, eta, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 1e-300)
        y = y * (S / y.sum())
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = x_new
        t = t_new

        # Recompute c and check
        c = fft_convolve_self(a)
        obj_new = 2.0 * n * c.max() / S**2

        if obj_new >= cur_obj and consecutive_fail == 0:
            # No improvement, try backtracking
            step *= 0.8

        consecutive_fail += 1 if obj_new >= cur_obj else 0
        if obj_new < cur_obj:
            step = min(step * 1.03, 5.0 / max(gnorm, 1e-15))
            consecutive_fail = 0

        # Periodic perturbation if stuck
        if consecutive_fail > 30:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t = 1.0
            c = fft_convolve_self(a)
            step = 1.0
            consecutive_fail = 0

        if it % 10000 == 0:
            print("    it=%d, obj=%.6f, beta=%.1f, step=%.4f" % (it, best_obj, beta, step))

    return best_a, best_obj


def run_exact_polish(a_init, n, S, max_time, rng):
    """Fine polishing with exact (top-1) subgradient and tiny steps."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    step = 0.05
    it = 0
    consecutive_fail = 0

    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0

        # Exact top-1 subgradient
        grad = topk_gradient(a, c, 1)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try both Euclidean and mirror steps
        eff_step = step / (1.0 + 0.0001 * it)

        # Euclidean projection step
        a_eucl = project_simplex(a - eff_step * grad, S)
        c_eucl = fft_convolve_self(a_eucl)
        obj_eucl = 2.0 * n * c_eucl.max() / S**2

        # Mirror step
        eta = eff_step / max(gnorm, 1e-15)
        a_mirror = mirror_step(a, grad, eta, S)
        c_mirror = fft_convolve_self(a_mirror)
        obj_mirror = 2.0 * n * c_mirror.max() / S**2

        # Pick the better
        if obj_eucl <= obj_mirror:
            a_new, c_new, obj_new = a_eucl, c_eucl, obj_eucl
        else:
            a_new, c_new, obj_new = a_mirror, c_mirror, obj_mirror

        if obj_new < cur_obj:
            a = a_new
            c = c_new
            step = min(step * 1.02, 0.5)
            consecutive_fail = 0
        else:
            step *= 0.9
            consecutive_fail += 1
            if consecutive_fail > 20:
                noise_scale = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                step = 0.03
                consecutive_fail = 0

        if it % 5000 == 0:
            print("    polish it=%d, obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations, emphasizing asymmetric structures."""
    inits = []

    inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * tt), 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * tt + 0.3 * np.pi), 0.01))

    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * 3 * tt + 1.0) + 0.2 * np.cos(2 * np.pi * 7 * tt))

    inits.append(1.0 + 2.0 * (tt**4 + (1 - tt)**4))

def structured_perturb(a, n, S, rng):
    """Apply structured perturbation."""
    strategy = rng.integers(0, 6)
    a_new = a.copy()
    tt = np.arange(n) / n

    if strategy == 0:
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt + phase)
        a_new = a * mod
    elif strategy == 1:
        # Asymmetric modulation
        mod = 1.0 + 0.4 * np.sin(np.pi * tt + rng.uniform(-0.5, 0.5))
        a_new = a * mod
    elif strategy == 2:
        # Block modulation
        nblocks = rng.integers(4, 12)
        block_mod = rng.uniform(0.5, 1.5, nblocks)
        mod = np.repeat(block_mod, n // nblocks + 1)[:n]
        a_new = a * mod
    elif strategy == 3:
        # Frequency domain perturbation
        A = fft(a)
        noise = rng.normal(0, 0.03, len(A)) + 1j * rng.normal(0, 0.03, len(A))
        A = A * (1 + noise)
        a_new = np.real(ifft(A))
    elif strategy == 4:
        # Shift and blend (creates asymmetry)
        shift = rng.integers(1, n // 4)
        a_shifted = np.roll(a, shift)
        a_new = 0.7 * a + 0.3 * a_shifted
    else:
        # Pure Gaussian noise
        a_new = a + rng.normal(0, 0.02 * S / np.sqrt(n), n)

    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

        if real_score_pol < global_best:
            global_best = real_score_pol
            global_a = a_pol.copy()
            print("  After polish: %.6f" % global_best)

    # === Phase 4: Try longer length if time permits ===
    remaining = deadline - time.monotonic()
    print("=== Phase 4: Multi-length try (remaining %.0fs) ===" % remaining)

    if remaining > 120:
        # Try n=16384
        n_long = 16384
        S_long = np.sqrt(2.0 * n_long)

        # Upsample the current best as initialization
        if global_a is not None:
            # Simple upsampling by repetition
            ratio = n_long // len(global_a)
            a_long_init = np.repeat(global_a, ratio)[:n_long]
            a_long_init = normalize_sum(a_long_init, n_long)
        else:
            a_long_init = normalize_sum(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n_long, endpoint=False)), 0.01), n_long)

        t_alloc_long = min(remaining * 0.5, 600)
        a_long, score_long = run_mirror_nesterov(a_long_init, n_long, S_long, t_alloc_long, rng)
        a_long = normalize_sum(a_long, n_long)
        real_score_long = compute_objective(a_long)
        print("  n=%d: obj=%.6f" % (n_long, real_score_long))

        if real_score_long < global_best:
            global_best = real_score_long
            global_a = a_long.copy()
            print("  New best from longer sequence: %.6f" % global_best)

        # Polish the longer sequence
        remaining2 = deadline - time.monotonic()
        if real_score_long < 1.52 and remaining2 > 30:
            a_pol_long, _ = run_exact_polish(a_long, n_long, S_long, remaining2 * 0.5, rng)
            a_pol_long = normalize_sum(a_pol_long, n_long)
            real_score_pol_long = compute_objective(a_pol_long)
            if real_score_pol_long < global_best:
                global_best = real_score_pol_long
                global_a = a_pol_long.copy()
                print("  After polish (long): %.6f" % global_best)

    # === Final: One more polish on the absolute best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        a_fin, _ = run_exact_polish(global_a, n_final, S_final, remaining * 0.7, rng)
        a_fin = normalize_sum(a_fin, n_final)
        real_score_fin = compute_objective(a_fin)
        if real_score_fin < global_best:
            global_best = real_score_fin
            global_a = a_fin.copy()
            print("  Final polish: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
