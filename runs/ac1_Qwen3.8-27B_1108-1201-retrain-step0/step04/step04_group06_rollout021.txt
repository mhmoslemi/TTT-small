```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
        theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def self_conv(a):
    """Compute linear self-convolution."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
```

```python
4. Multi-method ensemble with adaptive switching: Combines spectral, time-domain,

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        nz = np.where(result > 0)[0]
        if len(nz) > 0:
            result[nz] += diff / len(nz)
            result = np.maximum(result, 0.0)
    return result


def self_conv(a):
    """Compute linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score_full(a):
    """Compute the full score: 2*n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * c.max() / (S * S), c.max(), c


def spectral_shape_amplitude(a, N, n, top_k, alpha, target_cmax, rng):
    """
    Amplitude spectral shaping: reduce |A| at frequencies contributing to peak lags.
    Returns new sequence.
    """
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Compute current convolution
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target_cmax:
        return a

    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    # Weight by excess
    excess = np.maximum(top_vals - target_cmax, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total
    weights = excess / excess_sum

    # Compute effective "cosine" (contribution pattern)
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Amplitude shaping
    alpha_eff = alpha * min(1.0, (c_max - target_cmax) / max(c_max, 1e-10) + 0.2)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    a_new = np.real(ifft(np.sqrt(P_new) * np.exp(1j * phase)))[:n]
    S = a.sum()
    a_new = project_simplex(a_new, S)
    return a_new


def phase_shift_optimize(a, N, n, S, n_trials, rng):
    """
    Try random phase shifts and keep the best one.
    The self-conv DFT is A^2, so phase of A directly affects the result.
    """
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A)
    phase = np.angle(A)

    # Current score
    c_cur = np.real(ifft(A * A))[:2 * n - 1]
    best_cmax = c_cur.max()
    best_a = a.copy()

    m = np.arange(N)

    for trial in range(n_trials):
        # Generate a structured phase perturbation
        # Shift phase proportional to a random frequency
        freq = rng.integers(1, N // 2)
        delta = rng.uniform(-0.3, 0.3)
        phase_pert = phase + delta * np.sin(2.0 * np.pi * m * freq / N)

        # Ensure Hermitian symmetry for real output
        # A[N-k] should be conjugate of A[k]
        A_pert = mag * np.exp(1j * phase_pert)
        # Enforce Hermitian symmetry
        for k in range(1, N // 2):
            avg_re = (A_pert[k].real + A_pert[N - k].real) / 2
            avg_im = (A_pert[k].im - A_pert[N - k].im) / 2
            A_pert[k] = avg_re + 1j * avg_im
            A_pert[N - k] = avg_re - 1j * avg_im
        A_pert[0] = mag[0]
        if N % 2 == 0:
            A_pert[N // 2] = mag[N // 2] + 0j

        a_pert = np.real(ifft(A_pert))[:n]
        a_pert = project_simplex(a_pert, S)

        c_pert = self_conv(a_pert)
        cmax_pert = c_pert.max()

        if cmax_pert < best_cmax:
            best_cmax = cmax_pert
            best_a = a_pert.copy()

    return best_a, best_cmax

def nesterov_subgradient(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=128):
    """Nesterov-accelerated subgradient method."""
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = self_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Compute subgradient from active set
        threshold = cur * (1.0 - 0.05)
        active = np.where(c >= threshold)[0][:top_k]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        g = np.zeros(n)
        for k in active:
            k = int(k)
            lo1 = max(0, k - (n - 1))
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                g[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
            lo2 = max(k, 0)
            hi2 = min(n - 1, k + (n - 1))
            if lo2 <= hi2:
                g[lo2:hi2 + 1] += a[0:hi2 - lo2 + 1]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * g / max(g_norm, 1e-15), S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        y_new = project_simplex(y_new, S)
        a = x_new
        y = y_new
        t = t_new

        # Occasional restart of momentum
        if it % 500 == 0 and stall > 100:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            stall = 0

    return best_a, best_score


def coordinate_descent_v2(a_init, n, S, max_time, rng, n_top=20, n_rounds=200):
    """
    Improved coordinate descent: for each top-contributing coordinate,
    try multiple multipliers and use exact evaluation of affected lags.
    """
    a = a_init.copy()
    start = time.monotonic()

    best_a = a.copy()

    for rnd in range(n_rounds):

            break

        c = self_conv(a)
        c_max = c.max()

            best_score = c_max
            best_a = a.copy()

        # Find top contributing coordinates at the peak lag
        peak_lag = int(np.argmax(c))
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]

        improved = False

            if time.monotonic() - start > max_time:
                break

            if orig_val < 1e-15:

                c_test = self_conv(a)
                if c_test.max() < best_score:
                    best_score = c_test.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.0
                continue

            best_cmax = c_max

                a[i] = orig_val * frac
                # Only recompute affected lags for speed
                # Lags affected: k where a[i] contributes: k in [i, i+n-1] and k in [i-n+1, i]
                # But for self-conv, k ranges [0, 2n-2]

                    if 0 <= j2 < n and j2 != j1:

                    best_cmax = new_max
                    best_frac = frac

            a[i] = orig_val * best_frac
            if best_frac != 1.0:
                improved = True

        # Rescale
        a = project_simplex(a, S)

        if not improved and rng.random() < 0.3:

def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            a = a_new.copy()
            stall += 1
            if stall > 50:
                # Phase perturbation to escape
                a_pert, _ = phase_shift_optimize(a, N, n, S, 5, rng)
                c_pert = self_conv(a_pert)
                if c_pert.max() < best_cmax:
                    best_cmax = c_pert.max()
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                else:
                    a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                stall = 0

        if best_cmax <= target * S * S / (2 * n):
            break

    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses phase-aware spectral optimization, subgradient methods,
    and adaptive multi-method ensemble.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 16384

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = safe_normalize(a_candidate, n)
        c = self_conv(a_candidate)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            return True
        return False

    # --- Build initializations ---
    inits = []

    # Try to use previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
        elif given.size > 0:
            # Resample to length n
            new_given = np.interp(np.linspace(0, given.size - 1, n),
                                  np.arange(given.size), given)
            inits.append(("given_resampled", np.clip(new_given, 0, 1000)))
    except Exception:
        pass

    # Structured initializations
    for method in ["asymmetric_sin", "front_loaded", "back_loaded",
                   "chirp_asym", "bump_asym", "power_law", "random_multi"]:
        inits.append((method, structured_init_v2(n, rng, method)))

    n_inits = len(inits)
    print("  %d initializations" % n_inits)

    # Score all initializations
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c0 = self_conv(a_norm)
        sc0 = c0.max()
        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: %.6f (score=%.4f)" % (name, sc0, 2.0 * n * sc0 / S**2))

    if global_a is None:
        global_a = safe_normalize(structured_init_v2(n, rng, "asymmetric_sin"), n)
        global_best = self_conv(global_a).max()

    print("Initial best: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

    # === Phase 1: Spectral water-filling from best initializations (30% time) ===
    print("=== Phase 1: Spectral water-filling ===")
    t_phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.30

    # Focus on the best 3 initializations
    scored_inits = []
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c0 = self_conv(a_norm)
        scored_inits.append((c0.max(), name, a_norm))
    scored_inits.sort(key=lambda x: x[0])

    for rank in range(min(4, len(scored_inits))):
        if time.monotonic() > t_phase1_end - 10:
            break
        sc0, name, a_norm = scored_inits[rank]
        time_budget = min(40.0, (t_phase1_end - time.monotonic()) / max(1, 4 - rank))

        a_sw, sc_sw = spectral_waterfill_loop_v2(a_norm, N, n,
                                                  time.monotonic() + time_budget,
                                                  rng, target=1.42,
                                                  top_k_init=5, alpha_init=0.5)
        if update_best(a_sw):
            print("  [%s] waterfill: %.6f (score=%.4f)" % (name, global_best, 2.0 * n * global_best / S**2))

    print("=== Phase 1 done. Best: %.6f (score=%.4f) ===" % (global_best, 2.0 * n * global_best / S**2))

    # === Phase 2: Subgradient + phase optimization (35% time) ===
    print("=== Phase 2: Subgradient + phase optimization ===")
    t_phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    a_curr = global_a.copy()
    stall_count = 0

    while time.monotonic() < t_phase2_end:
        # Subgradient step
        rem = t_phase2_end - time.monotonic()
        if rem < 5:
            break
        time_budget = min(rem * 0.3, 25.0)
        a_ng, sc_ng = nesterov_subgradient(a_curr, n, S, time_budget, rng,
                                            top_k_init=10, top_k_max=200)
        if update_best(a_ng):
            a_curr = a_ng.copy()
            stall_count = 0
            if global_best < 1.51 * S * S / (2 * n):
                print("  [P2] subgrad: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))
        else:
            stall_count += 1

        # Phase optimization
        rem = t_phase2_end - time.monotonic()
        if rem > 3:
            a_ph, sc_ph = phase_shift_optimize(a_curr, N, n, S, 10, rng)
            if update_best(a_ph):
                a_curr = a_ph.copy()
                stall_count = 0
                print("  [P2] phase: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

        # Spectral water-filling
        rem = t_phase2_end - time.monotonic()
        if rem > 5:
            time_budget = min(rem * 0.3, 20.0)
            a_sw, sc_sw = spectral_waterfill_loop_v2(a_curr, N, n,
                                                      time.monotonic() + time_budget,
                                                      rng, target=1.42,
                                                      top_k_init=10, alpha_init=0.3)
            if update_best(a_sw):
                a_curr = a_sw.copy()
                stall_count = 0
                print("  [P2] waterfill: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))
            else:
                stall_count += 1

        # Random perturbation when stuck
        rem = t_phase2_end - time.monotonic()
        if rem > 10 and stall_count >= 4:
            for sigma in [0.03, 0.05, 0.1, 0.15]:
                a_pert = random_spectral_perturb(a_curr, N, n, S, sigma, rng)
                # Quick water-filling from perturbed state
                a_sw2, _ = spectral_waterfill_loop_v2(a_pert, N, n,
                                                       time.monotonic() + 5,
                                                       rng, target=1.42,
                                                       top_k_init=5, alpha_init=0.4)
                if update_best(a_sw2):
                    a_curr = a_sw2.copy()
                    stall_count = 0
                    print("  [P2] perturb+wf(%.2f): %.6f (score=%.4f)" % (sigma, global_best, 2.0 * n * global_best / S**2))
                    break
            stall_count = 0

    print("=== Phase 2 done. Best: %.6f (score=%.4f) ===" % (global_best, 2.0 * n * global_best / S**2))

    # === Phase 3: Coordinate descent + fine polishing (30% time) ===
    print("=== Phase 3: Coordinate descent + fine polishing ===")
    t_phase3_end = time.monotonic() + (deadline - time.monotonic()) * 0.30

    # Coordinate descent
    rem = t_phase3_end - time.monotonic()
    if rem > 5:
        time_budget = min(rem * 0.4, 60.0)
        a_cd, sc_cd = coordinate_descent_v2(global_a, n, S, time_budget, rng,
                                             n_top=30, n_rounds=300)
        if update_best(a_cd):
            print("  [P3] coord: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

    # More subgradient with high top-k
    rem = t_phase3_end - time.monotonic()
    if rem > 5:
        time_budget = min(rem * 0.3, 40.0)
        a_ng2, _ = nesterov_subgradient(global_a, n, S, time_budget, rng,
                                         top_k_init=50, top_k_max=300)
        if update_best(a_ng2):
            print("  [P3] subgrad high-k: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

    # Fine spectral water-filling
    rem = t_phase3_end - time.monotonic()
    if rem > 3:
        time_budget = min(rem * 0.3, 20.0)
        a_sw3, _ = spectral_waterfill_loop_v2(global_a, N, n,
                                               time.monotonic() + time_budget,
                                               rng, target=1.40,
                                               top_k_init=20, alpha_init=0.1)
        if update_best(a_sw3):
            print("  [P3] fine wf: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

    # Final coordinate descent
    rem = t_phase3_end - time.monotonic()
    if rem > 3:
        time_budget = min(rem * 0.5, 40.0)
        a_cd2, _ = coordinate_descent_v2(global_a, n, S, time_budget, rng,
                                          n_top=50, n_rounds=200)
        if update_best(a_cd2):
            print("  [P3] final coord: %.6f (score=%.4f)" % (global_best, 2.0 * n * global_best / S**2))

    print("=== Phase 3 done. Best: %.6f (score=%.4f) ===" % (global_best, 2.0 * n * global_best / S**2))

    # === Phase 4: Final intensive polish (5% time) ===
    print("=== Phase 4: Final polish ===")

    # Multiple rounds of coordinate descent with increasing precision
    while time.monotonic() < deadline - 5:
        rem = deadline - 5 - time.monotonic()
        if rem < 2:
            break
        time_budget = min(rem * 0.5, 15.0)
        a_final, _ = coordinate_descent_v2(global_a, n, S, time_budget, rng,
                                            n_top=50, n_rounds=100)
        if not update_best(a_final):
            break

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * n * global_best / S**2
    print("Returning sequence of length %d, score=%.6f" % (len(global_a), final_score))
    return [float(x) for x in global_a]
```
