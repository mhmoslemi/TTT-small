So total: `db[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]` (if both j and k-j are in [0,n-1]).

If j and k-j are the same (i.e., j = k/2, which requires k even), then we have only one term: `a[k/2]^2`, and the derivative is `2*a[k/2]`. But this is consistent with the formula above (2*a[k-j] = 2*a[k/2]).

So in general: `db[k]/da[j] = 2*a[k-j]` if 0 ≤ k-j ≤ n-1, else 0.

This means the gradient of b[k] with respect to a is: `grad[k] = 2 * flip(a)[k]` (circularly shifted and flipped, with zeros for out-of-bounds).

More precisely, if we define `g[j] = 2*a[k-j]` for j in [0,n-1] where k-j is in [0,n-1], and 0 otherwise, then `g` is the gradient of b[k] with respect to a.

In numpy: `g = np.zeros(n); idx = range(max(0,k-n+1), min(k+1, n)); g[idx] = 2*a[k-idx]`

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Multi-phase approach: FISTA-LSE with high beta, peak-guided coordinate descent,
simulated annealing for global exploration, and multi-start strategy.
Key improvements: higher LSE beta for accuracy, hybrid exact-subgradient corrections,
peak-lag-guided coordinate descent, and block perturbation for escaping local minima.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Normalize a to sum=S, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_grad_and_val(a, c, beta):
    """Compute LSE value and gradient. c = convolve(a,a)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm / beta, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    grad *= 2.0
    return lse_val, grad


def exact_subgrad(a, c):
    """Exact subgradient at argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad, k


def run_fista_lse(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=3000.0,
                  ramp_frac=0.3, hybrid_frac=0.2):

    a = normalize(a_init, S)
    y = a.copy()
    t_acc = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    total_iters_est = max_time * 8000  # rough estimate

    while time.monotonic() - start < max_time:
        it += 1
        if it < ramp_iters and ramp_iters > 0:
            beta = beta_start + (beta_end - beta_start) * (it / ramp_iters)
        else:
            beta = beta_end

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-15:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient source
        use_exact = (it > 2000 and rng.random() < hybrid_frac * min(1.0, it / 10000.0))
        if use_exact:
            grad, _ = exact_subgrad(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            # Scale for exact subgradient (it's larger than LSE grad for same beta)
            step *= 0.5
        else:
            _, grad = lse_grad_and_val(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
        y = x_new + ((t_acc - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_acc = t_new

        # Noise to escape local minima

            decay = np.exp(-it / 40000.0)
            ns = 0.003 * S / np.sqrt(n) * decay

        # Stagnation handling

def run_coord_descent(a_init, n, S, max_time, rng):
    """Peak-guided coordinate descent: move mass from high-contributing to
    low-contributing positions relative to the current peak lag."""
    a = normalize(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    b = fft_convolve_self(a)
    best = b.max()
    best_a = a.copy()

    it = 0
    no_improve = 0
    n_blocks = 20

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        k_star = int(np.argmax(b))
        cur_max = b[k_star]

        # Compute contribution of each position to b[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        contrib = np.zeros(n)
        idx_range = np.arange(lo, hi + 1)
        contrib[idx_range] = a[idx_range] * a[k_star - idx_range]

        # Compute "target attractiveness": positions j where a[k_star - j] is small
        # (moving mass there would reduce b[k_star] most)
        target_score = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                target_score[j] = a[kj]
            else:
                target_score[j] = 0.0  # Moving here doesn't affect b[k_star] at all (best!)

        # Try several (source, target) pairs
        n_try = 40
        bd = 0.0
        bc = np.inf
        bp = None

        for _ in range(n_try):
            # Source: weighted by contribution (prefer high contributors)
            total_cw = contrib.sum()
            if total_cw > 1e-15:
                probs = contrib / total_cw
                i = int(rng.choice(n, p=probs))
            else:
                i = int(rng.integers(0, n))

            # Target: prefer low target_score (prefer positions that don't add to b[k_star])
            # Add some randomness
            if rng.random() < 0.5:
                # Sample proportional to (1 - target_score)
                inv_score = np.maximum(1.0 - target_score / max(target_score.max(), 1e-10), 0.01)
                probs_t = inv_score / inv_score.sum()
                j = int(rng.choice(n, p=probs_t))
            else:
                j = int(rng.integers(0, n))

            if j == i or a[i] < 1e-12:
                continue

            # Compute first-order change in b
            db = np.zeros(2 * n - 1)
            # Contribution from moving delta from i to j:
            # db[k] = delta * (a[k-j] - a[k-i]) for each k
            for k in range(2 * n - 1):
                val = 0.0
                kj = k - j
                ki = k - i
                if 0 <= kj < n:
                    val += a[kj]
                if 0 <= ki < n:
                    val -= a[ki]
                db[k] = val

            # Minimize max(b + d*db) for d in [-a[i], a[j]]
            lo_d = -a[i]
            hi_d = a[j]
            # Coarse search
            n_samp = 16
            d_samps = np.linspace(lo_d, hi_d, n_samp)
            max_vals = np.empty(n_samp)
            for si, d in enumerate(d_samps):
                max_vals[si] = np.max(b + d * db)
            d_coarse = d_samps[int(np.argmin(max_vals))]

            # Refine
            margin = (hi_d - lo_d) / (n_samp * 1.5)
            lo_r = max(lo_d, d_coarse - margin)
            hi_r = min(hi_d, d_coarse + margin)
            if hi_r <= lo_r:
                continue
            d_fine = np.linspace(lo_r, hi_r, 16)
            max_fine = np.empty(len(d_fine))
            for fi, d in enumerate(d_fine):
                max_fine[fi] = np.max(b + d * db)
            d_best = d_fine[int(np.argmin(max_fine))]
            val_best = max_fine.min()

            if val_best < bc:
                bc = val_best
                bd = d_best
                bp = (i, j)

        if bp is not None and bc < cur_max - 1e-13:
            i, j = bp
            a[i] -= bd
            a[j] += bd
            a = np.maximum(a, 0.0)
            a = normalize(a, S)
            b = fft_convolve_self(a)
            if b.max() < best:
                best = b.max()
                best_a = a.copy()
            improved = True
            no_improve = 0

        if not improved:
            no_improve += 1
            # Block perturbation to escape
            block_size = max(8, n // n_blocks)
            if no_improve % 10 == 0 or no_improve < 10:
                bstart = int(rng.integers(0, n // block_size)) * block_size
                bend = min(bstart + block_size, n)
                block_sum = a[bstart:bend].sum()
                if block_sum > 1e-10:
                    # Redistribute within block using random Dirichlet-like
                    w = rng.random(bend - bstart) + 0.1
                    w = w / w.sum() * block_sum
                    a[bstart:bend] = w
                    a = normalize(a, S)
                    b = fft_convolve_self(a)
                    if b.max() < best:
                        best = b.max()
                        best_a = a.copy()

            if no_improve > 200:
                # Larger restart
                ns = 0.02 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, ns, n), S)
                b = fft_convolve_self(a)
                no_improve = 0

        if it % 100 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def run_sim_anneal(a_init, n, S, max_time, rng):
    """Simulated annealing with block-level moves."""
    a = normalize(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    b = fft_convolve_self(a)
    best = b.max()
    cur = best
    T = 0.005 * best
    T_min = 1e-10 * best
    total_budget = max_time

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = elapsed / total_budget
        T = T_min + (T - T_min) * np.exp(-8.0 * progress)

        # Random move: pick 2-4 positions and redistribute
        n_move = int(rng.integers(2, min(5, n // 4)))
        idx = rng.choice(n, n_move, replace=False)
        move_sum = a[idx].sum()
        if move_sum < 1e-12:
            continue
        w = rng.random(n_move) + 0.05
        w = w / w.sum() * move_sum

        a_trial = a.copy()
        a_trial[idx] = w
        a_trial = normalize(a_trial, S)

        b_trial = fft_convolve_self(a_trial)
        cur_trial = b_trial.max()
        delta_E = cur_trial - cur

        if delta_E < 0 or rng.random() < np.exp(-max(delta_E, 0) / max(T, 1e-30)):
            a = a_trial
            cur = cur_trial
            b = b_trial
            if cur < best:
                best = cur
                best_a = a.copy()

    return best_a, best


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Polished subgradient descent with adaptive step and periodic FISTA restart."""
    a = normalize(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    b = fft_convolve_self(a)
    best = b.max()
    best_a = a.copy()

    t = 1
    step = 0.005 * S / max(a.max(), S / n)
    last_improve = 0
    fista_counter = 0

    while time.monotonic() - start < max_time:
        k_star = int(np.argmax(b))
        cur_max = b[k_star]

        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            last_improve = t

        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        grad = np.zeros(n)
        idx_range = np.arange(lo, hi + 1)
        grad[idx_range] = 2.0 * a[k_star - idx_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_new = project_simplex(a - step * grad, S)
        b_new = fft_convolve_self(a_new)

        if b_new.max() <= cur_max + 1e-14 * cur_max:
            a = a_new
            b = b_new
            step *= 1.02  # Small increase on success
        else:
            step *= 0.5
            a_new2 = project_simplex(a - step * grad, S)
            b_new2 = fft_convolve_self(a_new2)
            a = a_new2
            b = b_new2

        step = min(step, 0.05 * S / max(b.max(), 1e-10))
        t += 1

        # Occasional noise
        if t % 500 == 0 and t < 20000:
            ns = 0.002 * S / np.sqrt(n) * np.exp(-t / 20000.0)
            if ns > 1e-14:
                a = project_simplex(a + rng.normal(0, ns, n), S)
                b = fft_convolve_self(a)

        # FISTA restart every 5000 iters
        fista_counter += 1
        if fista_counter >= 5000:
            fista_counter = 0
            # Brief FISTA burst
            y = a.copy()
            t_acc = 1.0
            beta_local = 2000.0
            for fi in range(200):
                c_f = fft_convolve_self(y)
                _, grad_f = lse_grad_and_val(y, c_f, beta_local)
                step_f = 1.0 / (2.0 * beta_local)
                x_f = project_simplex(y - step_f * grad_f, S)
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
                y = x_f + ((t_acc - 1.0) / t_new) * (x_f - a)
                y = np.maximum(y, 0.0)
                a = x_f
                t_acc = t_new
            b = fft_convolve_self(a)
            if b.max() < best:
                best = b.max()
                best_a = a.copy()

        # Stagnation
        if t - last_improve > 50000 and t > 10000:
            break

    return best_a, best


def generate_starts(n, S, rng, parent):
    """Generate diverse starting points."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        starts.append(("parent", normalize(parent, S)))
        # Variants
        starts.append(("parent_rev", normalize(parent[::-1], S)))
        # Small perturbations
        for scale in [0.02, 0.05, 0.1]:
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            starts.append(("pert_%d" % int(scale * 100), normalize(parent + noise, S)))
        # Palindromic version
        half = parent[:n // 2]
        pal = np.concatenate([half, parent[n // 2:n // 2 + 1], half[::-1]]) if n % 2 == 1 \
            else np.concatenate([half, half[::-1]])
        starts.append(("palindromic", normalize(pal, S)))
        # Low-pass
        fsize = 1 << (2 * n - 1).bit_length()
        F = fft(normalize(parent, S), fsize)
        freqs = np.fft.fftfreq(fsize)
        F_lp = F * np.exp(-20.0 * (freqs / 0.2) ** 2)
        a_lp = np.real(ifft(F_lp))[:n]
        if a_lp.sum() > 1e-10:
            starts.append(("lowpass", normalize(a_lp, S)))
        # High-pass complement
        a_hp = np.real(ifft(F - F_lp))[:n]
        if a_hp.sum() > 1e-10:
            starts.append(("highpass", normalize(a_hp, S)))

    # Structured starts (always include for diversity)
    starts.append(("sinc", normalize(np.maximum(np.sinc(tt - 0.5) ** 2, 0.01), S)))
    starts.append(("gauss", normalize(np.exp(-6.0 * (tt - 0.5) ** 2), S)))
    starts.append(("cosine", normalize(np.maximum(0.5 * (1.0 + np.cos(2 * np.pi * tt)), 0.01), S)))
    starts.append(("ramp", normalize(np.linspace(0.1, 1.0, n), S)))
    starts.append(("decay", normalize(np.exp(-3.0 * tt), S)))
    starts.append(("bimodal", normalize(np.maximum(np.exp(-20*(tt-0.3)**2) + np.exp(-20*(tt-0.7)**2), 0.01), S)))
    starts.append(("random", normalize(rng.random(n) + 0.05, S)))
    starts.append(("triangular", normalize(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01), S)))

    return starts


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Multi-phase: diverse starts -> FISTA-LSE -> coordinate descent ->
    simulated annealing -> subgradient polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    # Get parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        parent = np.clip(parent, 0, 1000)
        if parent.size < 64 or parent.sum() < 0.01:
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print("Setup: n=%d, S=%.2f, parent=%s, budget=%.0fs" % (
        n, S, "yes" if parent is not None else "no", budget_s))

    global_best = np.inf
    global_a = None

    def try_update(a_cand):
        nonlocal global_best, global_a
        a_cand = normalize(a_cand, S)
        b = fft_convolve_self(a_cand)
        score = b.max()
        eval_score = 2.0 * n * score / (S * S)
        if score < global_best:
            global_best = score
            global_a = a_cand.copy()
            elapsed = time.monotonic() - t_start
            print("  NEW BEST: max_b=%.10f, score=%.6f (t=%.0fs)" % (score, eval_score, elapsed))

    # Phase 1: Generate and quickly evaluate starts
    print("\n=== Phase 1: Multi-start evaluation ===")
    starts = generate_starts(n, S, rng, parent)
    scored = []
    for name, arr in starts:
        b = fft_convolve_self(arr)
        scored.append((b.max(), name, arr))
        try_update(arr)
    scored.sort(key=lambda x: x[0])
    for i, (sc, nm, _) in enumerate(scored[:5]):
        print("  Start %d: %s -> %.6f" % (i, nm, sc))

    # Phase 2: FISTA-LSE on top starts
    print("\n=== Phase 2: FISTA-LSE optimization ===")
    p2_budget = (deadline - time.monotonic()) * 0.35
    top_k = min(4, len(scored))
    # Allocate more time to better starts
    weights = np.array([3.0, 2.0, 1.5, 1.0][:top_k])
    weights /= weights.sum()
    for i in range(top_k):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = p2_budget * weights[i]
        if t_alloc < 10:
            continue
        print("  FISTA on %s (alloc %.0fs)..." % (scored[i][1], t_alloc))
        a_opt, sc = run_fista_lse(scored[i][2], n, S, t_alloc, rng,
                                   beta_start=50.0, beta_end=4000.0,
                                   ramp_frac=0.25, hybrid_frac=0.3)
        try_update(a_opt)

    # Phase 3: Coordinate descent on best
    print("\n=== Phase 3: Coordinate descent ===")
    if global_a is not None:
        p3_budget = (deadline - time.monotonic()) * 0.30
        if p3_budget > 20:
            print("  Running CD (budget %.0fs)..." % p3_budget)
            a_cd, sc = run_coord_descent(global_a, n, S, p3_budget, rng)
            try_update(a_cd)

    # Phase 4: Simulated annealing (exploration)
    print("\n=== Phase 4: Simulated annealing ===")
    if global_a is not None:
        p4_budget = (deadline - time.monotonic()) * 0.20
        if p4_budget > 15:
            # Run SA from best and from a different start
            t_sa = p4_budget * 0.6
            a_sa, sc = run_sim_anneal(global_a, n, S, t_sa, rng)
            try_update(a_sa)
            # Also from second best
            if len(scored) > 1:
                t_sa2 = p4_budget * 0.4
                a_sa2, sc2 = run_sim_anneal(scored[1][2], n, S, t_sa2, rng)
                try_update(a_sa2)

    # Phase 5: Final subgradient polish
    print("\n=== Phase 5: Final polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_pol = remaining * 0.7
            a_pol, sc = run_subgradient_polish(global_a, n, S, t_pol, rng)
            try_update(a_pol)
            # One more CD pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_cd2, sc2 = run_coord_descent(global_a, n, S, remaining2 * 0.8, rng)
                try_update(a_cd2)

    # Final check
    print("\n=== Final ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %s (len=%d)" % (ver, len(final_list)))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    total_time = time.monotonic() - t_start
    print("Done. Final score=%.6f, total_time=%.0fs" % (ver, total_time))
    return final_list
