import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta):
    """LSE-smoothed gradient of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to prevent overflow
    scaled = np.clip(scaled, -100, 100)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def mirror_descent_step(a, grad, eta, S):
    """Multiplicative descent step maintaining sum = S and a >= 0."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)

        n_mod = int(frac * n)
        idx = rng.choice(n, n_mod, replace=False)
        a_new[idx] *= rng.uniform(0.5, 1.5, n_mod)

    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)


def run_optimization(a_init, n, S, max_time, rng, use_mirror=False, momentum_decay=0.9):
    """
    Main optimization loop with LSE gradient, adaptive beta, Nesterov momentum,
    and periodic perturbations.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 5.0
    step = 1.0
    c = fft_convolve_self(a)

    # Track convergence
    no_improve_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: increase over time for more focus on peak
        if it < 1000:
            beta = 5.0 + it * 0.01
        elif it < 5000:
            beta = 15.0 + (it - 1000) * 0.02
        else:
            beta = 95.0 + min(500.0, (it - 5000) * 0.01)

        # Compute gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Nesterov momentum
        step_size = S / (n * gnorm * np.sqrt(1.0 + it / 1000.0))

        x_new = y - step_size * grad
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

        a = project_simplex(x_new, S)
        t = t_new

        # Recompute convolution
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Periodic perturbations to escape local minima
        if no_improve_count > 2000 and it - last_improve > 2000:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - it / 50000.0)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t = 1.0
            c = fft_convolve_self(a)
            no_improve_count = 0

            # Occasionally try structured perturbation
            if rng.random() < 0.3:
                a = structured_perturb(a, n, S, rng)
                y = a.copy()
                c = fft_convolve_self(a)

        # Every 10000 iterations, try mirror descent for a while
        if use_mirror and it % 10000 < 2000:
            eta = 0.1 / max(gnorm, 1e-10)
            a_try = mirror_descent_step(a, grad, eta, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < cur_obj:
                a = a_try
                c = c_try
                y = a.copy()
                t = 1.0

    return best_a, best_obj


def run_exact_polish(a_init, n, S, max_time, rng):
    """Polish with exact top-1 subgradient and adaptive step."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    step = 0.1
    c = fft_convolve_self(a)
    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Top-1 subgradient
        grad = topk_subgradient(a, c, 1)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        eff_step = step / (1.0 + 0.005 * stagnation)
        a_try = project_simplex(a - eff_step * grad, S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2

        if obj_try < cur_obj:
            a = a_try
            c = c_try
            step = min(step * 1.03, 0.5)
        else:
            # Try top-k for diversity
            improved = False
            for k in [4, 16, 64]:
                grad_k = topk_subgradient(a, c, k)
                a_try_k = project_simplex(a - eff_step * grad_k, S)
                c_try_k = fft_convolve_self(a_try_k)
                obj_try_k = 2.0 * n * c_try_k.max() / S**2
                if obj_try_k < cur_obj:
                    a = a_try_k
                    c = c_try_k
                    improved = True
                    break

            if not improved:
                step *= 0.85
                if step < 0.001:
                    noise_scale = 0.002 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    c = fft_convolve_self(a)
                    step = 0.05

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Previous best available: %s" % ("yes, len=%d" % len(given) if given is not None else "no"))

    # Define candidate lengths
    candidate_ns = [8192, 16384, 32768]

    # If we have a previous sequence, include its length
    if given is not None and len(given) not in candidate_ns:
        candidate_ns.append(len(given))

    # Sort by priority: prefer the given length first, then larger
    if given is not None:
        candidate_ns.sort(key=lambda x: (x != len(given), -x))
    else:
        candidate_ns.sort(key=lambda x: -x)

    print("Candidate lengths: %s" % candidate_ns)
    t0 = time.monotonic()

    # Phase 1: Quick exploration across lengths (30% of time)
    phase1_time = (deadline - time.monotonic()) * 0.30
    per_length = phase1_time / len(candidate_ns)

    print("=== Phase 1: Exploration (total %.0fs) ===" % phase1_time)

    for n in candidate_ns:
        if time.monotonic() > deadline - (deadline - t0) * 0.65:
            break

        S = np.sqrt(2.0 * n)
        time_alloc = min(per_length, deadline - time.monotonic() - 20)
        if time_alloc < 10:
            continue

        # Generate initializations
        inits = generate_inits(n, rng)

        # Add the given sequence if appropriate length
        if given is not None and len(given) == n:
            inits.insert(0, given.copy())
            inits.insert(1, structured_perturb(given, n, S, rng))
            inits.insert(2, structured_perturb(given, n, S, rng))

        # Try top few initializations
        n_try = min(4, len(inits))
        per_init_time = time_alloc / n_try

        best_this_n = np.inf
        best_a_this_n = None

        for i in range(n_try):
            if time.monotonic() > deadline - (deadline - t0) * 0.60:
                break
            arr = normalize_sum(inits[i], n)
            a_opt, score = run_optimization(arr, n, S, per_init_time * 0.8, rng,
                                             use_mirror=(i % 2 == 0))
            score = compute_objective(a_opt)

            if score < best_this_n:
                best_this_n = score
                best_a_this_n = a_opt.copy()

        if best_a_this_n is not None and best_this_n < global_best:
            global_best = best_this_n
            global_a = best_a_this_n.copy()
            print("  n=%d: obj=%.6f (elapsed %.0fs)" % (
                n, global_best, time.monotonic() - t0))

    print("=== After Phase 1: best=%.6f, elapsed=%.0fs ===" % (
        global_best, time.monotonic() - t0))

    # Phase 2: Focused optimization on best length (45% of time)
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        phase2_time = remaining * 0.50

        print("=== Phase 2: Focused optimization n=%d (total %.0fs) ===" % (n_best, phase2_time))

        # Run multiple optimization variants
        seg_time = phase2_time / 4

        # Variant 1: Standard LSE with momentum
        a1, s1 = run_optimization(global_a, n_best, S_best, seg_time, rng,
                                   use_mirror=False)
        s1 = compute_objective(a1)
        if s1 < global_best:
            global_best = s1
            global_a = a1.copy()
            print("  After opt1: %.6f" % global_best)

        remaining = deadline - time.monotonic()
        if remaining < 10:
            return _finalize(global_a, global_best)

        # Variant 2: Mirror descent
        seg_time = remaining * 0.25
        a2, s2 = run_optimization(global_a, n_best, S_best, seg_time, rng,
                                   use_mirror=True)
        s2 = compute_objective(a2)
        if s2 < global_best:
            global_best = s2
            global_a = a2.copy()
            print("  After opt2 (mirror): %.6f" % global_best)

        remaining = deadline - time.monotonic()
        if remaining < 10:
            return _finalize(global_a, global_best)

        # Variant 3: Exact polish
        seg_time = remaining * 0.30
        a3, s3 = run_exact_polish(global_a, n_best, S_best, seg_time, rng)
        s3 = compute_objective(a3)
        if s3 < global_best:
            global_best = s3
            global_a = a3.copy()
            print("  After exact polish: %.6f" % global_best)

    # Phase 3: Try larger n if we have time (15% of remaining)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try the next larger n
        target_n = 32768
        if global_a is not None and len(global_a) >= target_n:
            target_n = len(global_a) * 2

        if target_n <= 65536:
            S_tgt = np.sqrt(2.0 * target_n)
            time_alloc = remaining * 0.40
            print("=== Phase 3: Trying n=%d (time %.0fs) ===" % (target_n, time_alloc))

            inits = generate_inits(target_n, rng)
            # Upsample from best if available
            if global_a is not None:
                from scipy.interpolate import interp1d
                x_old = np.linspace(0, 1, len(global_a))
                x_new = np.linspace(0, 1, target_n)
                f = interp1d(x_old, global_a, kind='cubic', fill_value='extrapolate')
                upsampled = np.maximum(f(x_new), 0)
                if upsampled.sum() > 0:
                    inits.insert(0, upsampled)

            best_big = np.inf
            best_a_big = None
            per_init = time_alloc / min(3, len(inits))

            for i in range(min(3, len(inits))):
                if time.monotonic() > deadline - 20:
                    break
                arr = normalize_sum(inits[i], target_n)
                a_opt, score = run_optimization(arr, target_n, S_tgt, per_init * 0.8, rng)
                score = compute_objective(a_opt)
                if score < best_big:
                    best_big = score
                    best_a_big = a_opt.copy()

            if best_a_big is not None and best_big < global_best:
                global_best = best_big
                global_a = best_a_big.copy()
                print("  n=%d: obj=%.6f" % (target_n, global_best))

    # Phase 4: Final micro-polish
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Phase 4: Final polish (time %.0fs) ===" % remaining)

        a_final, s_final = run_exact_polish(global_a, n_best, S_best, remaining * 0.7, rng)
        s_final = compute_objective(a_final)
        if s_final < global_best:
            global_best = s_final
            global_a = a_final.copy()
            print("  Final: %.6f" % global_best)

    # Final verification
    return _finalize(global_a, global_best)


def _finalize(global_a, global_best):
    """Final validation and return."""
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, obj=%.6f" % (len(final_list), global_best))
    return final_list
