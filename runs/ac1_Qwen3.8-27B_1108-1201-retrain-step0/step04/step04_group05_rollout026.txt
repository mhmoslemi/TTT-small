from scipy.fft import fft, ifft

def compute_conv(a, N):
    """Compute self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    b = np.real(ifft(A * A))[:2*n-1]
    return b

def solve_slp_iteration(a, N, S, top_k, max_time_lp=5.0):
    """
    One SLP iteration: linearize the max-conv problem and solve the LP.
    Returns the LP solution (a_lp) and the LP objective.
    """
    n = len(a)
    b = compute_conv(a, N)
    b_max = b.max()
    
    # Find top-k active constraints
    top_idx = np.argsort(b)[::-1][:top_k]
    
    # Build LP
    # Variables: [a[0], ..., a[n-1], t]
    # Minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: for each top constraint k
    # 2*sum_i a[i]*a[k-i] - t <= b[k]
    A_ub = np.zeros((top_k, n + 1))
    b_ub = np.zeros(top_k)
    
    for ci, k in enumerate(top_idx):
        k = int(k)
        # Coefficient of a[i] is 2*a[k-i] for valid i
        for i in range(max(0, k-(n-1)), min(k, n-1)+1):
            A_ub[ci, i] = 2.0 * a[k - i]
        A_ub[ci, n] = -1.0  # coefficient of t
        b_ub[ci] = b[k]
    
    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: a[i] >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    # Solve
    t_start = time.monotonic()
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if result.success:

        return a_lp, t_lp
    else:
        return a, t_lp

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Successive Linear Programming (SLP)
with adaptive active-set constraint selection, combined with Fourier-domain
spectral exploration and gradient polishing.

Key innovations over previous L-BFGS/LSE approach:
1. SLP directly linearizes the max-conv problem, solving LPs that naturally
   handle the non-smooth "max" structure without smoothing approximations.
2. Adaptive constraint management: only the most binding constraints are
   included, keeping LPs small and fast while maintaining convergence.
3. Cut-plane recovery: violated constraints from dropped sets are detected
   and re-added, ensuring valid descent.
4. Fourier-domain global moves provide diverse exploration to escape
   local minima that gradient methods get stuck in.
5. Multi-scale initialization and periodic restarts for robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def _compute_conv(a, N):
    """Compute self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    b = np.real(ifft(A * A))[:2 * n - 1]
    return b


def _objective(a, N):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = _compute_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_candidates = (css - S) / np.arange(1, n + 1, dtype=np.float64)
    valid = rho_candidates <= u
    if not np.any(valid):
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
        theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def _slp_iteration(a, N, S, top_k, max_lp_time=3.0):
    """
    One SLP iteration. Linearize max-conv around a, solve LP, return LP solution.
    Returns (a_lp, b_current, top_idx, b_max).
    """
    n = len(a)
    b = _compute_conv(a, N)
    b_max = b.max()
    top_idx = np.argsort(b)[::-1][:top_k]

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []

    for ci, k in enumerate(top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            a_idx = k - i_range
            rows.append(np.full(len(i_range), ci, dtype=np.int32))
            cols.append(i_range.astype(np.int32))
            vals.append(2.0 * a[a_idx])
        # t variable (column n)
        rows.append(np.array([ci], dtype=np.int32))
        cols.append(np.array([n], dtype=np.int32))
        vals.append(np.array([-1.0]))

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(top_k, n + 1))
    b_ub = b[top_idx].astype(np.float64)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, None)] * n + [(b_max * 0.5, None)]

    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_lp_time, 'presolve': True})
    except Exception:
        return a.copy(), b, top_idx, b_max

    if res.success and res.x is not None:
        a_lp = np.maximum(res.x[:n], 0.0)
        a_lp = _project_simplex(a_lp, S)
        return a_lp, b, top_idx, b_max
    else:
        return a.copy(), b, top_idx, b_max


def _slp_phase(a_init, N, S, max_time, rng, start_k=50, end_k=512, m_slp=20):
    """
    SLP optimization phase with adaptive constraint growth and restarts.
    Returns (best_a, best_f).
    """
    n = len(a_init)
    a = _project_simplex(a_init.copy(), S)
    b = _compute_conv(a, N)
    b_max = b.max()
    best_a = a.copy()
    best_f = _objective(a, N)

    t_start = time.monotonic()
    it = 0
    stall = 0
    no_improve = 0
    top_k = start_k

    while time.monotonic() - t_start < max_time - 5.0:
        it += 1

        # Grow top_k gradually
        progress = min(1.0, it / (m_slp * 20))
        top_k_curr = int(start_k + (end_k - start_k) * progress)
        top_k_curr = min(top_k_curr, 2 * n - 1)

        # Solve LP
        a_lp, b_cur, top_idx, b_max_cur = _slp_iteration(
            a, N, S, top_k_curr, max_lp_time=2.0)

        # Line search: try decreasing steps toward a_lp
        best_alpha = 0.0
        best_f_try = _objective(a, N)
        alphas = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * a_lp
            a_try = _project_simplex(a_try, S)
            f_try = _objective(a_try, N)
            if f_try < best_f_try:
                best_f_try = f_try
                best_alpha = alpha

        if best_alpha > 0:
            a_new = (1.0 - best_alpha) * a + best_alpha * a_lp
            a_new = _project_simplex(a_new, S)
            f_new = _objective(a_new, N)

            if f_new < best_f - 1e-10:
                best_f = f_new
                best_a = a_new.copy()
                stall = 0
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            stall += 1

        # Reset if stalled
        if stall >= 15:
            # Random perturbation to escape
            noise_scale = 0.02 * S / np.sqrt(n)
            a = _project_simplex(a + rng.normal(0, noise_scale, n), S)
            stall = 0
            no_improve = 0

        # Periodic larger restarts
        if no_improve >= 80:
            noise_scale = 0.08 * S / np.sqrt(n)
            a = _project_simplex(a + rng.normal(0, noise_scale, n), S)
            # Also reduce top_k to allow more freedom
            top_k = start_k
            no_improve = 0
            stall = 0

        # Time check
        if time.monotonic() - t_start > max_time - 10.0:

    # Final evaluation
    fin_f = _objective(best_a, N)
    return best_a, fin_f

            mean_mag = mag.mean()
            high_mask = mag > 2 * mean_mag
            A_new = A.copy()

        a_new = np.abs(a_new) + 1e-6
        a_new = _project_simplex(a_new, S)
        f_new = _objective(a_new, N)

            best_a = a_new.copy()

        else:

            a_blend = (1.0 - alpha) * a + alpha * a_new

                best_a = a_blend.copy()

def _gradient_polish(a_init, N, S, max_time, rng, beta_init=20.0, beta_max=500.0):
    """
    Gradient-based polishing using Adam on LSE-smoothed objective.
    Returns (best_a, best_f).
    """
    n = len(a_init)
    a = _project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_f = _objective(a, N)

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0
    lr = 0.5 * S / n
    beta = beta_init

    t_start = time.monotonic()
    patience = 0

    while time.monotonic() - t_start < max_time - 5.0:
        t_adam += 1

        # Compute LSE-smoothed gradient
        b = _compute_conv(a, N)
        b_max = b.max()
        exp_beta = np.exp(beta * (b - b_max))
        total = exp_beta.sum()
        if total < 1e-300:
            w = np.full(len(b), 1.0 / len(b))
        else:
            w = exp_beta / total

        # Gradient of LSE via FFT

        W = fft(w, fsize)
        Af = fft(a, fsize)

Minimize 2*n*max(conv(a,a))/sum(a)^2 using Successive Linear Programming (SLP)
with adaptive active-set constraint selection, combined with Fourier-domain
spectral exploration and gradient polishing.

Key innovations over previous L-BFGS/LSE approach:
1. SLP directly linearizes the max-convolution problem at each iteration,

import numpy as np

def _compute_conv(a, N):
    """Compute self-convolution of a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    b = np.real(ifft(A * A))[:2 * n - 1]
    return b


def _objective(a, N):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = _compute_conv(a, N)
    return 2.0 * n * b.max() / (S * S)


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_candidates = (css - S) / np.arange(1, n + 1, dtype=np.float64)
    valid = rho_candidates <= u
    if not np.any(valid):
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
        theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def _slp_solve(a, N, S, top_k, lp_time_limit=2.0):
    """
    One SLP iteration: build and solve the linearized LP.
    Returns (a_lp, b_current, top_idx, b_max).
    """
    n = len(a)
    b = _compute_conv(a, N)
    b_max = b.max()
    top_idx = np.argsort(b)[::-1][:top_k]

    # Build sparse constraint matrix for the LP
    # Variables: [a[0], ..., a[n-1], t]
    # Minimize t
    # Constraints: 2*sum_i a[i]*a[k-i] - b[k] <= t  (for top_k constraints)
    #              sum(a) = S
    #              a >= 0, t >= b_max * 0.5

    row_parts = []
    col_parts = []
    val_parts = []

    for ci in range(len(top_idx)):
        k = int(top_idx[ci])
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            i_idx = np.arange(i_start, i_end + 1)
            a_idx = k - i_idx
            row_parts.append(np.full(len(i_idx), ci, dtype=np.int32))
            col_parts.append(i_idx.astype(np.int32))
            val_parts.append(2.0 * a[a_idx])
        # t variable column
        row_parts.append(np.array([ci], dtype=np.int32))
        col_parts.append(np.array([n], dtype=np.int32))
        val_parts.append(np.array([-1.0]))

    all_rows = np.concatenate(row_parts)
    all_cols = np.concatenate(col_parts)
    all_vals = np.concatenate(val_parts)

    A_ub = csr_matrix((all_vals, (all_rows, all_cols)), shape=(len(top_idx), n + 1))
    b_ub = b[top_idx].astype(np.float64)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0.0, None)] * n + [(b_max * 0.3, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': lp_time_limit})
        if res.success and res.x is not None:
            a_lp = np.maximum(res.x[:n], 0.0)
            a_lp = _project_simplex(a_lp, S)
            return a_lp, b, top_idx, b_max
    except Exception:
        pass

    return a.copy(), b, top_idx, b_max

        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Internal best: %.6f, Verified: %.6f" % (global_best, ver), flush=True)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.", flush=True)
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, internal=%.6f" % (len(global_a), global_best), flush=True)
    return [float(x) for x in global_a]
