import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_active_subgradient(x, c, eps_frac=0.01):
    """Compute subgradient of max(c) using active set."""
    n = len(x)
    max_c = c.max()
    # Active set: lags where c_k is close to max
    threshold = max_c * (1 - eps_frac)
    active = np.where(c >= threshold)[0]
    
    # (grad c_k)_p = 2 * x_{k-p} for valid indices
    # Sum over active lags, then normalize
    g = np.zeros(n)
    for k in active:
        # (grad c_k)_p = 2 * x_{k-p}
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        p = np.arange(lo, hi + 1)
        g[p] += 2.0 * x[k - p]
    g /= len(active)
    return g, max_c

def run_active_fista(x_init, n, S, max_time, rng, 
                      eps_frac=0.01, step_scale=1.0):

    x = project_simplex(x_init, S)
    y = x.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_x = x.copy()
    it = 0
    last_improve = 0
    
    # Adaptive step size
    step_size = 0.1
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_x = y.copy()
            last_improve = it
        
        g, max_c = compute_active_subgradient(y, c, eps_frac)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:

        # Adaptive step: if no improvement, reduce
        if it - last_improve > 500:

        g = g / gnorm  # normalize

                t = 1.0

    if c_final.max() < best:
        best = c_final.max()

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def active_subgradient(x, c, frac=0.005):
    """Compute subgradient of max(c) using active set (lags near max)."""
    n = len(x)

        return np.zeros(n), max_c

    active = np.where(c >= threshold)[0]

        hi = min(k, n - 1)
        p = np.arange(lo, hi + 1)
        g[p] += 2.0 * x[k - p]
    g /= len(active)
    return g, max_c

    if c_final.max() < best:
