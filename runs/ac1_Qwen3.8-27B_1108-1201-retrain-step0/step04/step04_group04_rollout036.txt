2. This means `A(ω)^2 ≈ C(ω)` where `IFFT(C)` is flat

4. But we also need `Re(IFFT(A)) >= 0` and `sum(Re(IFFT(A))) = S`

This is a "two-signal" phase retrieval problem:
- Constraint 1: `|A(ω)|^2` should be such that `IFFT(A^2)` is flat
- Constraint 2: `Re(IFFT(A)) >= 0` in the first n samples

5. Modify C: reduce peaks in c by modifying |C| (e.g., C_new = C * |C|^{-β} normalized)

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the evaluation objective for a normalized sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def spectral_gs(a_init, n, S, max_time, rng, alpha_init=0.3, alpha_max=0.9):
    """Improved Gerchberg-Saxton spectral flattening for linear self-convolution.
    
    Key insight: conv(a,a) has DTFT = A(omega)^2. We iteratively:
    1. Go to time domain, enforce non-negativity and sum constraint
    2. Go to frequency domain, flatten A^2 (the convolution spectrum)
    3. Take complex square root to get new A
    """
    a = normalize_sum(a_init, n)
    fsize = next_fast_len(2 * n - 1)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    c0 = fft_convolve_self(a)
    best = c0.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Time domain: enforce constraints
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
            s = S
        a = a * (S / s)
        
        # Frequency domain: flatten A^2
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        A = fft(a_pad)
        C = A * A  # Spectrum of the convolution
        
        # Compute current convolution in time domain
        c = np.real(ifft(C))[:2 * n - 1]
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Adaptive alpha
        progress = min(1.0, it / 3000.0)
        alpha = alpha_init + (alpha_max - alpha_init) * progress
        
        # Flatten C: push towards flat spectrum
        # The "flat" convolution has C_flat = FFT of a flat sequence
        c_target = np.full(2 * n - 1, S * S / (2 * n - 1))
        c_target_pad = np.zeros(fsize)
        c_target_pad[:2 * n - 1] = c_target
        C_target = fft(c_target_pad)
        
        # Blend in frequency domain
        C_new = (1.0 - alpha) * C + alpha * C_target
        
        # Also apply peak reduction: scale down high-magnitude bins
        mag_C = np.abs(C_new)
        mag_mean = mag_C.mean()
        if mag_mean > 0:
            # Soft thresholding: reduce peaks in the spectrum
            peak_factor = np.minimum(1.0, (mag_mean / np.maximum(mag_C, 1e-30)) ** 0.3)
            C_new = C_new * peak_factor
        
        # Complex square root: A_new = sqrt(C_new)
        # Use principal branch
        A_new = np.sqrt(C_new)
        
        # Ensure the result is "reasonable" - check if time domain is mostly positive
        a_candidate = np.real(ifft(A_new))[:n]
        
        # If too many negatives, try the other square root (negative of principal)
        a_candidate_alt = -np.real(ifft(A_new))[:n]
        pos_frac = np.mean(a_candidate > 0)
        pos_frac_alt = np.mean(a_candidate_alt > 0)
        if pos_frac_alt > pos_frac:
            a_candidate = a_candidate_alt
        
        # Project to non-negative simplex
        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-15:
            a_candidate = np.ones(n) * S / n
        else:
            a_candidate = a_candidate * (S / s_cand)
        
        # Blend with current for stability
        a = 0.7 * a_candidate + 0.3 * a
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
        
        # Periodic noise to escape local minima
        if it % 500 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-9:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = next_fast_len(len(w) + n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=3000.0,
              beta_ramp=2000, use_hybrid=True):
    """FISTA with adaptive LSE smoothing and hybrid subgradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if it < beta_ramp:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp) ** 0.7
        else:
            beta = beta_max
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # Choose gradient method
        use_exact = use_hybrid and it > 500 and (rng.random() < min(0.4, 0.001 * it))
        if use_exact:
            grad = exact_subgradient(y, c)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        if use_exact:
            step = S / (gnorm * (1.0 + 0.02 * np.log1p(it / 100.0)))
        else:
            step = 1.0 / (2.0 * beta + 1e-10)
        
        # Reduce step if stagnating
        if it - last_improve > 2000:
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic noise
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

                # Local neighbors
                j = int(rng.integers(0, n - 1))
                m = (j + int(rng.integers(1, min(10, n - 1)))) % n
            else:
                j = int(rng.integers(0, n))
                m = int(rng.integers(0, n))
            
            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue
            
            # Compute d_k = change in c_k per unit delta
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val_j = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val_j += a[jk]
                if 0 <= jmk < n:
                    val_j += a[jmk]
                val_m = 0.0
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val_m += a[mk]
                if 0 <= mmk < n:
                    val_m += a[mmk]
                d[k] = val_j - val_m
            
            # c_0 has quadratic term: c_0 = sum a[i]^2
            # dc_0/ddelta = 2*a[j]*delta + 2*delta^2 ... correction
            # Actually: new c_0 = c_0 + (a[j]+d)^2 - a[j]^2 + (a[m]-d)^2 - a[m]^2
            # = c_0 + 2*a[j]*d + d^2 - 2*a[m]*d + d^2 = c_0 + 2*(a[j]-a[m])*d + 2*d^2
            # But d[0] already accounts for linear term, so add 2*d^2 as quadratic
            
            lo = -a[j]
            hi = a[m]
            
            # Golden section search for min max(c + delta*d)
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            
            def f_obj(dlt):
                vals = c + dlt * d
                vals[0] += 2.0 * dlt ** 2
                return vals.max()
            
            fc = f_obj(c_g)
            fd = f_obj(d_g)
            
            for _ in range(35):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_obj(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_obj(d_g)
            
            delta = (a_g + b_g) / 2.0
            delta = max(lo, min(hi, delta))
            
            if abs(delta) < 1e-14:
                continue
            
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue
            
            # Compute new max efficiently
            c_trial = c + delta * d
            c_trial[0] = c[0] + 2.0 * (a[j] - a[m]) * delta + 2.0 * delta ** 2
            cur_trial = c_trial.max()
            
            if cur_trial < best - 1e-12:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True
                stagnation = 0
        
        if not improved:
            stagnation += 1
            # Add noise to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnation / 20.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
        
        # Periodic full recompute
        if _ % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    
    return best_a, best


def block_coord_descent(a_init, n, S, max_time, rng, block_size=4):
    """Block coordinate descent: optimize blocks of coordinates using scipy."""
    from scipy.optimize import minimize as scipy_minimize
    
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        improved_any = False
        
        # Try several random blocks
        n_blocks = min(20, max(5, n // block_size // 10))
        for bi in range(n_blocks):
            if time.monotonic() - start > max_time - 5:
                break
            
            # Choose a block of consecutive coordinates
            block_start = int(rng.integers(0, n - block_size))
            block = np.arange(block_start, block_start + block_size)
            
            # Fix other coordinates
            a_fixed = a.copy()
            
            def block_obj(x_block):
                a_trial = a_fixed.copy()
                a_trial[block] = x_block
                c_trial = fft_convolve_self(a_trial)
                return c_trial.max()
            
            x0 = a[block].copy()
            bounds = [(0.0, S)] * block_size
            # Also constraint: sum(x_block) + sum(others) = S
            # sum(others) = S - sum(x0)
            others_sum = S - x0.sum()
            
            def constraint_val(x_block):
                return x_block.sum() - x0.sum()
            
            cons = {'type': 'eq', 'fun': constraint_val}
            
            res = scipy_minimize(block_obj, x0, method='SLSQP', bounds=bounds,
                                  constraints=cons, options={'maxiter': 50, 'ftol': 1e-10})
            
            if res.success and res.fun < best - 1e-12:
                a[block] = np.maximum(res.x, 0.0)
                # Renormalize
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * (S / s)
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()
                    improved_any = True
        
        if not improved_any:
            # Add noise
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            else:
                break
    
    return best_a, best


def generate_starts(parent, n, rng, S):
    """Generate diverse starting sequences."""
    starts = []
    names = []
    
    if parent is not None and len(parent) == n:
        starts.append(parent.copy())
        names.append("parent")
        
        # Small perturbations
        for scale in [0.01, 0.05, 0.1, 0.2]:
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            v = project_simplex(parent + noise, S)
            starts.append(v)
            names.append("pert_%g" % scale)
    
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Gaussian
    g = np.exp(-8.0 * (tt - 0.5) ** 2)
    starts.append(normalize_sum(g, n))
    names.append("gaussian")
    
    # Sinc-like
    s = np.maximum(np.sinc(tt - 0.5) ** 2, 0.01)
    starts.append(normalize_sum(s, n))
    names.append("sinc")
    
    # Cosine raised
    c = np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01)
    starts.append(normalize_sum(c, n))
    names.append("cosine")
    
    # Power law
    p = np.maximum(1.0 - tt, 0.01) ** 0.5
    starts.append(normalize_sum(p, n))
    names.append("power05")
    
    # Bimodal
    b = np.exp(-20.0 * (tt - 0.3) ** 2) + np.exp(-20.0 * (tt - 0.7) ** 2)
    starts.append(normalize_sum(b, n))
    names.append("bimodal")
    
    # Random
    r = rng.random(n) + 0.1
    starts.append(normalize_sum(r, n))
    names.append("random")
    
    # Low-pass filtered parent (if available)
    if parent is not None and len(parent) == n:
        fsize = next_fast_len(2 * n - 1)
        F = fft(parent, fsize)
        freqs = np.abs(np.fft.fftfreq(fsize))
        for cutoff in [0.1, 0.2, 0.4]:
            F_f = F * np.exp(-5.0 * (freqs / cutoff) ** 2)
            a_f = np.real(ifft(F_f))[:n]
            a_f = np.maximum(a_f, 0)
            s_f = a_f.sum()
            if s_f > 1e-15:
                starts.append(a_f * (S / s_f))
                names.append("lowpass_%g" % cutoff)
    
    return starts, names


def propose_candidate(seed=42, budget_s=2100):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)
    
    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 512 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None
    
    # Determine length
    if parent is not None and 2048 <= len(parent) <= 16384:
        n = len(parent)
    else:
        n = 8192
    
    S = np.sqrt(2.0 * n)
    
    print("Starting: n=%d, S=%.4f, budget=%.0fs, parent=%s" % (
        n, S, budget_s, "yes" if parent is not None else "no"))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        if a_candidate is None:
            return
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        ratio = compute_objective(a_norm)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [%s] n=%d: max_conv=%.8f, ratio=%.6f (elapsed %.0fs)" % (
                label, len(a_norm), score, ratio, elapsed))
        return score
    
    # Generate starts
    starts, names = generate_starts(parent, n, rng, S)
    print("Generated %d starting sequences" % len(starts))
    
    # === Phase 1: Spectral GS on top candidates (30% time) ===
    print("\n=== Phase 1: Spectral GS ===")
    p1_budget = (deadline - time.monotonic()) * 0.25
    n_top = min(4, len(starts))
    t_per_start = p1_budget / n_top
    
    for i in range(n_top):
        if time.monotonic() > deadline - (deadline - t_start) * 0.5 - 30:
            break
        a_opt, _ = spectral_gs(starts[i], n, S, t_per_start, rng,
                               alpha_init=0.2, alpha_max=0.8)
        update_best(a_opt, "gs1_%s" % names[i])
        
        # Also try v2
        remaining_p1 = deadline - (deadline - t_start) * 0.5 - 30 - time.monotonic()
        if remaining_p1 > 5:
            a_opt2, _ = spectral_gs_v2(starts[i], n, S, min(t_per_start * 0.5, remaining_p1), rng)
            update_best(a_opt2, "gs2_%s" % names[i])
    
    print("After Phase 1: best=%.8f" % global_best)
    
    # === Phase 2: FISTA on best + variants (25% time) ===
    print("\n=== Phase 2: FISTA ===")
    p2_budget = (deadline - time.monotonic()) * 0.25
    
    if global_a is not None:
        # Run FISTA from the best
        t_fista1 = p2_budget * 0.4
        a_opt, _ = run_fista(global_a, n, S, t_fista1, rng,
                             beta_init=5.0, beta_max=2000.0,
                             beta_ramp=1500, use_hybrid=True)
        update_best(a_opt, "fista_best")
        
        # Run from a perturbed version
        t_fista2 = p2_budget * 0.3
        perturbed = project_simplex(global_a + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
        a_opt2, _ = run_fista(perturbed, n, S, t_fista2, rng,
                              beta_init=10.0, beta_max=3000.0,
                              beta_ramp=1000, use_hybrid=True)
        update_best(a_opt2, "fista_pert")
        
        # Run from a different start
        t_fista3 = p2_budget * 0.3
        if len(starts) > n_top:
            a_opt3, _ = run_fista(starts[n_top], n, S, t_fista3, rng,
                                  beta_init=5.0, beta_max=1500.0,
                                  beta_ramp=1500, use_hybrid=True)
            update_best(a_opt3, "fista_alt")
    
    print("After Phase 2: best=%.8f" % global_best)
    
    # === Phase 3: Peak peeling (20% time) ===
    print("\n=== Phase 3: Peak Peeling ===")
    p3_budget = (deadline - time.monotonic()) * 0.15
    
    if global_a is not None:
        a_opt, _ = peak_peel(global_a, n, S, p3_budget, rng, n_transfers=10)
        update_best(a_opt, "peak_peel")
    
    print("After Phase 3: best=%.8f" % global_best)
    
    # === Phase 4: Coordinate descent polish (20% time) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    p4_budget = (deadline - time.monotonic()) * 0.20
    
    if global_a is not None:
        a_opt, _ = coord_descent(global_a, n, S, p4_budget, rng, n_pairs=150)
        update_best(a_opt, "coord_desc")
    
    print("After Phase 4: best=%.8f" % global_best)
    
    # === Phase 5: Final polish with short FISTA + CD (remaining time) ===
    print("\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 15:
        # Quick FISTA with high beta
        t_fin1 = min(remaining * 0.4, 120)
        a_opt, _ = run_fista(global_a, n, S, t_fin1, rng,
                             beta_init=200.0, beta_max=5000.0,
                             beta_ramp=500, use_hybrid=True)
        update_best(a_opt, "final_fista")
        
        # Quick coordinate descent
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_opt2, _ = coord_descent(global_a, n, S, remaining2 * 0.7, rng, n_pairs=200)
            update_best(a_opt2, "final_cd")
        
        # One more spectral pass
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            a_opt3, _ = spectral_gs(global_a, n, S, remaining3 * 0.8, rng,
                                     alpha_init=0.5, alpha_max=0.9)
            update_best(a_opt3, "final_gs")
    
    # === Try alternate length if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and global_a is not None:
        print("\n=== Phase 6: Alternate Length ===")
        # Try n*2
        alt_n = n * 2
        if alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            # Upsample
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, alt_n)
            alt_start = np.interp(t_new, t_old, global_a)
            alt_start = normalize_sum(alt_start, alt_n)
            
            t_alt = min(remaining * 0.3, 60)
            a_alt, _ = run_fista(alt_start, alt_n, S_alt, t_alt, rng,
                                  beta_init=5.0, beta_max=1000.0)
            
            # Check ratio with correct n
            ratio_alt = compute_objective(a_alt)
            ratio_cur = compute_objective(global_a)
            print("  Alt n=%d: ratio=%.6f vs current ratio=%.6f" % (alt_n, ratio_alt, ratio_cur))
            if ratio_alt < compute_objective(global_a):
                global_a = a_alt
                global_best = fft_convolve_self(a_alt).max()
                print("  Updated to alternate length!")
    
    # === Final verification ===
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.8f" % ver)
    else:
        print("Verification: inf - using fallback")
        global_a = np.ones(512) * (np.sqrt(2 * 512) / 512)
        final_list = [float(x) for x in global_a]
    
    print("Returning length=%d, final_score=%.8f, total_time=%.0fs" % (
        len(global_a), ver if np.isfinite(ver) else float('inf'),
        time.monotonic() - t_start))
    return final_list
