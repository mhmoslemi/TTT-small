"""
Minimize 2*n*max(a*a)/(sum(a))^2 over non-negative sequences a of length n.
Uses LSE-smoothed subgradient with FFT-based gradient computation,
adaptive step size, and multi-scale optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. a.
    Using FFT for efficiency.
    grad[j] = sum_k w_k * dc_k/da[j]
    where w_k = exp(beta*(c_k - c_max)) / sum_j exp(beta*(c_j - c_max))
    and dc_k/da[j] = 2*a[k-j] (with correction for j = k/2)
    """
    n = len(a)
    size = 2 * n - 1
    
    # Compute softmax weights
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # Compute cross-correlation: corr[j] = sum_k w[k] * a[k-j]
    # This is ifft(fft(w) * conj(fft(a)))
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:size]
    
    # grad[j] = 2 * corr[j] for most j
    # Correction: for even k, when j = k/2, the coefficient is 1 instead of 2
    # corr[j] = sum_k w[k] * a[k-j]
    # The "double counting" happens when k-j = j, i.e., k = 2j
    # So the correction is: subtract w[2j] * a[j] for even 2j
    grad = 2.0 * corr[:n]
    
    # Apply correction for j where 2j < size
    # For each j, if 2j < size, subtract w[2j] * a[j]
    valid_j = np.arange(n)
    two_j = 2 * valid_j
    mask = two_j < size
    grad[mask] -= w[two_j[mask]] * a[mask]
    
    return grad


def excess_gradient(a, c):
    """
    Gradient weighted by excess above mean.
    w_k = max(c_k - mean, 0) / sum(max(c_j - mean, 0))
    """
    n = len(a)
    size = 2 * n - 1
    
    mean_c = c.mean()
    excess = np.maximum(c - mean_c, 0)
    tot = excess.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = excess / tot
    
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:size]
    
    grad = 2.0 * corr[:n]
    valid_j = np.arange(n)
    two_j = 2 * valid_j
    mask = two_j < size
    grad[mask] -= w[two_j[mask]] * a[mask]
    
    return grad


def topk_gradient(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    c = fft_self_conv(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def run_lse_optimize(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0):
    """
    LSE-smoothed subgradient optimization.
    """
    a = project_onto_simplex(np.maximum(a_init, 0), S)
    start = time.monotonic()
    c = fft_self_conv(a)
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    it = 0
    step = 1.0
    beta = beta_init
    consecutive_fail = 0
    last_improve_time = start
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve_time = time.monotonic()
            consecutive_fail = 0
        
        # Adaptive beta: increase as we get closer to convergence
        progress = 1.0 - (best_obj - 1.0) / (2.0 - 1.0)  # 0 at start, 1 at optimal
        beta = beta_init + (beta_max - beta_init) * min(1.0, max(0, progress))
        # Also keep increasing with iterations
        beta = min(beta_max, beta * (1.0 + 0.001))
        
        # Compute gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        max_step = 2.0 * S / (n * gnorm + 1e-15)
        step = min(step, max_step)
        
        # Take step
        a_new = project_onto_simplex(a - step * grad, S)
        c_new = fft_self_conv(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new < cur_obj:
            a = a_new
            c = c_new
            step = min(step * 1.03, max_step)
        else:
            consecutive_fail += 1
            step *= 0.75
            if consecutive_fail > 15:
                # Perturbation
                noise_scale = 0.005 * S / np.sqrt(n)
                a = project_onto_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_self_conv(a)
                step = min(1.0, max_step)
                consecutive_fail = 0
            if consecutive_fail > 40:
                noise_scale = 0.02 * S / np.sqrt(n)
                a = project_onto_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_self_conv(a)
                step = 1.0
                consecutive_fail = 0
        
        # Periodic print
        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print("    it=%d, obj=%.6f, beta=%.1f, elapsed=%.0fs" % (it, best_obj, beta, elapsed))
    
    return best_a, best_obj

def make_polynomial_init(n, rng):
    """
    Create initialization based on the 'flat convolution' structure.
    The ideal has c[k] = constant for all k, which gives a specific structure.
    """
    S = np.sqrt(2.0 * n)
    # The ideal flat solution has a[0] ~ S/sqrt(2n), a[1]=a[2] ~ S/(2*sqrt(2n)), etc.
    # This is roughly a * S/sqrt(2n) where a = [1, 0.5, 0.5, 0.25, ...]
    # But this decays too fast. Let's use a slower decay.
    t = np.arange(n) / n
    # Try various shapes
    shape = 1.0
    # Cosine window (good for reducing edge effects)
    shape = 0.5 * (1 + np.cos(2 * np.pi * t))
    # Or try a raised cosine variant
    shape = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    shape = np.maximum(shape, 0.01)
    a = shape * (S / shape.sum())
    return a

"""
Minimize 2*n*max(a*a)/(sum(a))^2 over non-negative sequences using
LSE-smoothed subgradient with FFT-based gradient computation,
adaptive step sizing, mixed gradient strategies, and multi-length optimization.
Key improvement: correct FFT-based LSE gradient (no spurious corrections),
excess-weighted gradient for multi-peak flattening, and careful beta scheduling.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. a via FFT.
    grad[j] = 2 * sum_k w_k * a[k-j] for valid (k-j), i.e. 2*(w cross a)[j].
    """
    n = len(a)
    size = 2 * n - 1
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def excess_gradient(a, c):
    """
    Gradient weighted by excess above mean of convolution.
    Targets all peaks above average simultaneously.
    """
    n = len(a)
    size = 2 * n - 1
    mean_c = c.mean()
    excess = np.maximum(c - mean_c, 0.0)
    tot = excess.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = excess / tot
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def topk_gradient(a, c, k):
    """Subgradient from top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_obj(a):
    """Compute objective 2*n*max(a*a)/(sum a)^2."""
    n = len(a)
    c = fft_self_conv(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def project_simplex(v, S):

    n = len(v)
    v = np.asarray(v, dtype=np.float64)
