def next_power_of_2(n):

def fft_convolve_self(a):

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() <= S:
        # Already under the sum, just clip negatives
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 0:
            v *= S / s
        else:
            v = np.ones(n) * S / n
        return v
    # Standard simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def water_filling_clip(c, target_sum, max_iter=50):
    """Find M such that sum(min(c, M)) = target_sum, assuming c >= 0."""
    c = np.maximum(c, 0)
    lo, hi = 0.0, c.max() * 1.01
    if hi < target_sum / len(c):
        # All values are below target_sum/len, so M > max(c), no clipping needed
        return c.copy()
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        clipped = np.minimum(c, mid)
        s = clipped.sum()
        if abs(s - target_sum) < 1e-10 * target_sum:
            break
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    clipped = np.minimum(c, (lo + hi) / 2)
    # Rescale to exactly hit target_sum
    s = clipped.sum()
    if s > 0:
        clipped *= target_sum / s
    return clipped

    cross = np.real(ifft(W * np.conj(A)))[:n]
    
    return 2.0 * cross


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def run_iap_optimization(a_init, n, max_time, rng, target_ratio=1.0):
    """
    Iterated Alternating Projections optimization.
    Alternates between:
    1. Computing Q = a*a and "flattening" it (water-filling clip).
    2. Taking frequency-domain square root to get new a.
    Combined with LSE subgradient as a secondary mechanism.
    """
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    fsize = next_power_of_2(2 * n - 1)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    lse_eta = 0.1
    stall_count = 0
    
    c = fft_convolve_self(a)
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            stall_count = 0
        
        if it - last_improve > 500:
            stall_count += 1
        
        if stall_count > 5:
            # Add perturbation
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            stall_count = 0
            continue
        
        # Step 1: Flatten the convolution
        target_sum = S**2
        # Use a "soft" target: we want max(c) to be as low as possible
        # Water-filling: find M such that sum(min(c, M)) = target_sum
        M_target = target_sum / (2 * n - 1)  # The average
        # We want to clip above some level
        c_clipped = water_filling_clip(c, target_sum)
        
        # Step 2: Frequency domain square root
        a_new = freq_domain_step(a, c_clipped, fsize)
        
        # Step 3: LSE subgradient correction
        # Compute gradient
        beta = min(200.0, 5.0 + 0.1 * it)
        if c.max() > 0:
            grad = lse_gradient(a, c, beta, fsize)
        else:
            grad = np.zeros(n)
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            # Blend the IAP result with gradient step
            a_iap = np.maximum(a_new, 0)
            s_iap = a_iap.sum()
            if s_iap > 1e-15:
                a_iap = a_iap * S / s_iap
            else:
                a_iap = a.copy()
            
            # Gradient step
            step = lse_eta / gnorm
            a_grad = project_simplex(a - step * grad, S)
            
            # Blend
            blend = 0.7 if (cur_obj > 1.6) else 0.5  # More IAP when far from optimal
            a_new = blend * a_iap + (1 - blend) * a_grad
        
        a = a_new
        c = fft_convolve_self(a)
        
        if it % 1000 == 0:
            print("    IAP it=%d, obj=%.6f, beta=%.1f" % (it, best_obj, beta))
        
        # Adaptive learning rate for LSE
        if cur_obj < best_obj:
            lse_eta = min(lse_eta * 1.01, 1.0)
        else:
            lse_eta = max(lse_eta * 0.95, 0.01)
    
    return best_a, best_obj


def run_refined_subgrad(a_init, n, max_time, rng):
    """
    Refined LSE subgradient with adaptive beta and Nesterov acceleration.
    """
    S = np.sqrt(2.0 * n)
    n_local = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n_local) * S / n_local
    else:
        a = a * S / s
    
    fsize = next_power_of_2(2 * n_local - 1)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Nesterov momentum
    y = a.copy()
    t = 1.0
    beta_start = 2.0
    beta_end = 100.0
    
    c_y = fft_convolve_self(y)
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        progress = min(1.0, it / 50000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        if c_y.max() > 0:
            grad = lse_gradient(y, c_y, beta, fsize)
        else:
            grad = np.zeros(n_local)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: for LSE, the Lipschitz constant is approximately 2*beta*max(a)
        L = 2.0 * beta * a.max() + gnorm
        step = 1.0 / max(L, 1e-10)
        step = min(step, 2.0 * S / (n_local * gnorm + 1e-10))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        a = x_new
        t = t_new
        
        cur_obj = 2.0 * n_local * fft_convolve_self(a).max() / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

            noise = rng.normal(0, 0.005 * S / np.sqrt(n_local), n_local)
