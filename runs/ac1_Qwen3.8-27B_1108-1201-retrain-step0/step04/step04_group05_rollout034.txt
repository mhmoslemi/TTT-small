import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Compute full convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_and_grad(a, N, beta):
    """Compute LSE-smoothed max of conv and its gradient."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    # Gradient of LSE with respect to a
    # d/d a_i [sum_k w_k * c_k] where c_k = sum_j a_j * a_{k-j}
    # = 2 * sum_k w_k * a_{k-i} (for valid k-i)
    # This is 2 * (w * a_reversed) evaluated appropriately
    # Using FFT: grad = 2 * ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
    fsize = 1 << (2 * n).bit_length()  # need at least 2n-1
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c

    """L-BFGS two-loop recursion."""

    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:

def spectral_equality(a, N, top_k, alpha):
    """Spectral equalization step."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

            y_use = []

                best_c_candidate = c_ls
                best_alpha = 0.01 * step_scale

            y_k = g_new - g
            sy = np.dot(s_k, y_k)

            g = g_new
            f_val = f_new
            c = c_new

"""
Minimize 2*n*max_conv/sum_a^2 using enhanced L-BFGS on LSE-smoothed objective.
Key improvements: higher initial beta (starting from good sequence), longer memory,
adaptive step sizing via LSE curvature, aggressive spectral equalization,
and multi-phase optimization with increasing precision.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute full self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_and_grad(a, N, beta):
    """
    Compute LSE-smoothed max of conv(a,a) and its gradient.
    Returns (grad, lse_value, conv_array).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    # Gradient: 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val, c


def lbfgs_two_loop(s_hist, y_hist, g, m):
    """L-BFGS two-loop recursion. Returns descent direction."""
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_equality(a, N, top_k, alpha):
    """Spectral equalization: reduce spectral power at frequencies corresponding to top peaks."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def topk_grad_descent(a, N, S, top_k, step_size):
    """One step of top-k subgradient descent on max convolution."""
    n = len(a)
    c = compute_conv(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    A = fft(a, N)
    W = fft(w_full, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-18:
        return a
    a_new = project_simplex(a - step_size * grad / gnorm, S)
    return a_new


def optimize_block(a_init, n, S, N, max_time, rng, beta_start, beta_end, m_lbfgs=20):
    """
    L-BFGS optimization of LSE-smoothed max convolution.
    Returns (best_a, best_cmax).
    """
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    beta = beta_start
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c = lse_and_grad(a, N, beta)
    c_max = c.max()
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    gnorm = np.linalg.norm(g)
    if gnorm > 1e-18:
        step_scale = max(1e-6, f_val / (gnorm * gnorm) * 0.5)
    else:
        step_scale = S / n

    start_time = time.monotonic()
    it = 0
    # Beta growth rate per iteration
    total_iters_estimate = max(100, int(max_time / 0.3))
    beta_multiplier = (beta_end / max(beta_start, 1e-6)) ** (1.0 / total_iters_estimate)

    no_improve_count = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Update beta
        if it % 30 == 0:
            old_beta = beta
            beta = min(beta_end, beta * (beta_multiplier ** 30))
            if beta != old_beta:
                g, f_val, c = lse_and_grad(a, N, beta)
            c_max = c.max()
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                no_improve_count

        # Truncate history
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # Compute direction
        m_use = len(s_hist)
        if m_use > 0:
            direction = lbfgs_two_loop(s_hist, y_hist, g, m_use)
        else:
            gn = np.linalg.norm(g)
            if gn < 1e-18:
                break
            direction = -g / gn

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Backtracking line search using LSE
        alpha = step_scale
        accepted = False
        for ls in range(25):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cmax_ls = c_ls.max()

            # Compute LSE at candidate
            if cmax_ls < best_cmax - 1e-9:
                # Definitely better
                accepted = True
                break
            # Check LSE improvement
            exp_bc_ls = np.exp(beta * (c_ls - cmax_ls))
            total_ls = exp_bc_ls.sum()
            if total_ls > 1e-300:
                lse_ls = cmax_ls + np.log(total_ls) / beta
            else:
                lse_ls = cmax_ls

            if lse_ls < f_val * 0.999:
                accepted = True
                break

            alpha *= 0.6

        if accepted:
            g_new, f_new, c_new = lse_and_grad(candidate, N, beta)
            cmax_new = c_new.max()

            # Update L-BFGS history
            s_k = candidate - a
            y_k = g_new - g
            sy = np.dot(s_k, y_k)
            ss = np.dot(s_k, s_k)
            if sy > 1e-12 * max(ss, 1e-30) and ss > 1e-20:
                s_hist.append(s_k)
                y_hist.append(y_k)

            a = candidate
            g = g_new
            f_val = f_new

            if cmax_new < best_cmax:
                best_cmax = cmax_new
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1

            # Adapt step scale
            step_scale = alpha * 1.2
        else:
            no_improve_count += 1
            step_scale *= 0.5
            if step_scale < 1e-10:
                break

        # Periodic perturbation to escape local minima
        if it % 400 == 0:
            noise_scale = 0.004 * S / np.sqrt(n) * (1.0 + no_improve_count * 0.3)
            a_pert = project_simplex(a + rng.normal(0, min(noise_scale, 0.05 * S / np.sqrt(n)), n), S)
            c_pert = compute_conv(a_pert, N)
            if c_pert.max() < best_cmax:
                best_cmax = c_pert.max()
                best_a = a_pert.copy()
                a = a_pert
            elif it % 800 == 0:
                a = a_pert
            s_hist = []
            y_hist = []
            g, f_val, c = lse_and_grad(a, N, beta)
            c_max = c.max()
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        # Periodic spectral step
        if it % 250 == 125:
            a_sp = spectral_equality(a, N, 8, 0.1)
            c_sp = compute_conv(a_sp, N)
            if c_sp.max() < best_cmax:
                best_cmax = c_sp.max()
                best_a = a_sp.copy()
                a = a_sp
                s_hist = []
                y_hist = []
                g, f_val, c = lse_and_grad(a, N, beta)

    return best_a, best_cmax


def nesterov_topk_polish(a_init, n, S, N, max_time, rng, top_k_init=10, top_k_max=200):
    """Nesterov-accelerated top-k subgradient for polishing."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()

        # Compute top-k gradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        A = fft(y, N)
        W = fft(w_full, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t = t_new

        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_a, best_cmax

            a_fin, sc_fin = optimize_block(a_try, n, S, N, bt, rng,
                                            beta_start=100.0, beta_end=800.0, m_lbfgs=25)
            update_global(a_fin, "P4/try%d" % trial)
            remaining = deadline - time.monotonic()

    # === Final top-k direct gradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_curr = global_a.copy()
        step = 0.5 * S / n
        t_polish = time.monotonic()
        while time.monotonic() - t_polish < remaining - 5:
            a_curr = topk_grad_descent(a_curr, N, S, 5, step)
            c_p = compute_conv(a_curr, N)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_curr.copy()
            step *= 0.98

    # === Verification ===
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Final verification: R = %.8f (length=%d)" % (ver, len(final_list)))
    else:
        print("Verification failed, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning length %d, internal R=%.8f" % (len(final_list), 2.0*n*global_best/S**2))
    return final_list
