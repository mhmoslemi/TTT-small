"""

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the evaluation score for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def safe_normalize(a, n):

    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def mirror_descent_step(a, S, top_k, alpha, rng):
    """
    One mirror-descent step: find peak lags, compute anti-mirror direction,
    take a step, project onto simplex.
    Returns (new_a, c_max).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices].copy()
    # Weight by excess
    excess = np.maximum(top_vals - 0.9 * c_max, 1e-10)
    weights = excess / excess.sum()
    
    # Compute weighted anti-mirror direction
    # For lag k, the subgradient at position j is a[k-j] (for valid j)
    direction = np.zeros(n)
    for i, k in enumerate(top_indices):
        k = int(k)
        w = weights[i]
        # Valid j: 0 <= j < n and 0 <= k-j < n => max(0, k-n+1) <= j <= min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            direction[j_lo:j_hi+1] += w * a[k - j_hi:k - j_lo + 1][::-1]
    
    # Center the direction to preserve sum approximately
    direction -= direction.mean()
    
    # Adaptive step size
    dnorm = np.linalg.norm(direction)

        return a.copy(), c_max
    
    step = alpha * S / (n * dnorm)
    a_new = a - step * direction
    a_new = project_simplex(a_new, S)
    
    return a_new, c_max

        # The contribution of frequency f to lag k is cos(2*pi*f*k/N)
        # We want to reduce frequencies where this cosine is positive
        cos_val = np.cos(2.0 * np.pi * m * k / N)
        weight = (top_vals[i] / c_max)
        suppression -= beta * weight * np.maximum(cos_val, 0)
    
    # Ensure suppression is positive
    suppression = np.maximum(suppression, 0.01)
    
    P_new = P * suppression
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    
    c_new = fft_convolve_self(a_new)
    return a_new, c_new.max()


def frequency_concentration(a, n, N, n_peaks, rng):
    """
    Try to concentrate the power spectrum into fewer frequency bins.
    This makes the autocorrelation more "flat" (all lags similar).
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Find dominant frequency bins
    P_sorted_indices = np.argsort(P)[::-1]
    
    # Create a "soft mask" that keeps top frequencies
    mask = np.zeros(N)
    for i in range(min(N, 200)):
        frac = 1.0 - i / 200.0
        mask[P_sorted_indices[i]] = max(mask[P_sorted_indices[i]], frac)
    
    # Blend between original and masked
    blend = 0.3
    P_new = P * (blend + (1 - blend) * mask / max(mask.max(), 1e-10))
    
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    
    c_new = fft_convolve_self(a_new)
    return a_new, c_new.max()


def coordinate_descent(a_init, n, S, max_time, rng, n_top=20, n_rounds=100):
    """Coordinate descent on top-contributing coordinates at peak lags."""
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    N = 1 << (2 * n - 1).bit_length()
    
    multipliers = [0.3, 0.5, 0.7, 0.8, 1.2, 1.5]
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Find top 3 peak lags
        top_lags = np.argsort(c)[::-1][:3]
        
        improved = False
        for lag in top_lags:
            lag = int(lag)
            if time.monotonic() - start > max_time:
                break
            
            # Find top-contributing coordinates at this lag
            low = max(0, lag - n + 1)
            high = min(lag, n - 1)
            contribs = np.zeros(n)
            for i in range(low, high + 1):
                j = lag - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            
            top_coords = np.argsort(contribs)[::-1][:n_top]
            
            for idx in top_coords:
                i = int(idx)
                if time.monotonic() - start > max_time:
                    break
                j = lag - i
                if j < 0 or j >= n or i == j:
                    continue
                
                orig_val = a[i]
                best_frac = 1.0
                best_cmax_local = c_max
                
                for frac in multipliers:
                    a[i] = orig_val * frac
                    # Quick recompute of affected convolution values
                    delta = a[i] - orig_val
                    # c[k] changes by delta * a[k-i] for each k
                    # We only need to check if the max decreases
                    k_range = range(max(0, i), min(2*n-1, i+n))
                    min_gain = np.inf
                    for k in k_range:
                        j2 = k - i
                        if 0 <= j2 < n:
                            new_val = c[k] + delta * a[j2]
                            # Also the symmetric term if j2 != i
                            if j2 != i:
                                new_val += delta * a[k - j2] if 0 <= k - j2 < n else 0
                            if new_val > best_cmax_local:
                                break
                    else:
                        # All affected values are below best_cmax_local
                        # Check if this is actually better
                        max_affected = max(c[k] + delta * a[k-i] + (delta * a[k-(k-i)] if 0 <= k-(k-i) < n and (k-i) != i else 0) for k in k_range if 0 <= k-i < n)
                        other_max = max(c[:max(0,i)].max() if max(0,i) > 0 else 0, 
                                       c[min(2*n-1, i+n):].max() if min(2*n-1, i+n) < 2*n-1 else 0)
                        new_cmax = max(max_affected, other_max)
                        if new_cmax < best_cmax_local:
                            best_cmax_local = new_cmax
                            best_frac = frac
                
                a[i] = orig_val * best_frac
                if best_frac != 1.0:
                    improved = True
        
        # Rescale to maintain sum
        a = project_simplex(a, S)
        
        # Perturb if stuck
        if not improved:
            n_perturb = min(20, n // 10)
            idxs = rng.choice(n, n_perturb, replace=False)
            for idx in idxs:
                a[int(idx)] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)
    
    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k_init=5):
    """
    Nesterov-accelerated subgradient method using the mirror structure.
    The subgradient of max_k conv(a,a)[k] at lag k* is 2*a reversed/shifted.
    """
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_cmax = np.inf
    best_local = a.copy()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(64, top_k_init + it // 15)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        
        # Weight by excess above mean
        excess = np.maximum(top_vals, 1e-10)
        weights = excess / total
        
        # Compute weighted subgradient (mirror direction)
        grad = np.zeros(n)
        for i, k in enumerate(top_indices):
            k = int(k)
            w = 2.0 * weights[i]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += w * a[k - j_hi:k - j_lo + 1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size (Lipschitz constant estimate)
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if stall > 50 and it % 50 == 0:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            stall = 0
        
        if best_cmax < 1.0 and stall > 200:
            break
    
    return best_local, best_cmax


def iterative_mirror_optimize(a_init, n, S, max_time, rng, target=1.48):
    """
    Main iterative optimization combining mirror descent, spectral shaping,
    and subgradient methods with adaptive switching.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    N = 1 << (2 * n - 1).bit_length()
    stall = 0
    last_improve_time = start
    
    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        
        # Adaptive parameters
        top_k = min(100, 3 + it // 10)
        alpha = 0.5 / (1.0 + 0.02 * it)
        beta = 0.3 / (1.0 + 0.01 * it)
        
        # Choose method based on stall count
        method = "mirror"
        if stall > 30:
            method = rng.choice(["mirror", "spectral", "subgrad"], p=[0.3, 0.4, 0.3])
        elif stall > 10:
            method = rng.choice(["mirror", "spectral"], p=[0.5, 0.5])
        
        if method == "mirror":
            a_new, c_max = mirror_descent_step(a, S, top_k, alpha, rng)
        elif method == "spectral":
            a_new, c_max = spectral_peak_suppress(a, n, N, top_k, beta, rng)
        else:
            # Short subgradient burst
            a_new, c_max = subgradient_polish(a, n, S, min(remaining * 0.1, 5.0), rng, top_k)
        
        # Verify
        c_check = fft_convolve_self(a_new)
        actual_cmax = c_check.max()
        
        if actual_cmax < best_cmax:
            best_cmax = actual_cmax
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            last_improve_time = time.monotonic()
            if it % 100 == 0:
                score = 2.0 * n * best_cmax / (S * S)
                print("    iter %d: cmax=%.6f, score=%.6f" % (it, best_cmax, score))
        else:
            a = a_new.copy()
            stall += 1
        
        # Escalate perturbation if stuck too long
        if stall > 100:
            sigma = min(0.1, 0.01 * (stall // 100 + 1))
            a = project_simplex(a + rng.normal(0, sigma * S / np.sqrt(n), n), S)
            stall = 0
        
        # Check if we've hit target
        current_score = 2.0 * n * best_cmax / (S * S)
        if current_score <= target:
            break
        
        # Time to reduce step size towards end
        if remaining < 60:
            alpha *= 0.3
            beta *= 0.3
    
    return best_a, best_cmax


def multi_scale_optimize(a_init, n, S, max_time, rng, target=1.48):
    """
    Multi-scale approach: optimize a downsampled version first, then upsample and refine.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    start = time.monotonic()
    
    # Try several downsample factors
    for factor in [4, 2, 1]:
        n_small = n // factor
        if n_small < 64:
            continue
        
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        
        if factor == 1:
            a_work = best_a.copy()
        else:
            # Downsample by averaging
            n_chunks = n // factor
            a_work = a_init[:n_chunks * factor].reshape(n_chunks, factor).mean(axis=1)
            S_small = a_work.sum()
        
        S_work = a_work.sum() if factor > 1 else S
        time_budget = min(remaining * 0.5, 60.0) if factor > 1 else remaining * 0.4
        
        a_opt, cmax_small = iterative_mirror_optimize(
            a_work, len(a_work), S_work, time_budget, rng, target)
        
        if factor > 1:
            # Upsample
            a_up = np.repeat(a_opt, factor)[:n]
            a_up = project_simplex(a_up, S)
            c_up = fft_convolve_self(a_up)
            score_up = 2.0 * n * c_up.max() / (S * S)
            if score_up < best_score:
                best_a = a_up
                best_score = score_up
        else:
            c_opt = fft_convolve_self(a_opt)
            score_opt = 2.0 * n * c_opt.max() / (S * S)
            if score_opt < best_score:
                best_a = a_opt
                best_score = score_opt
    
    return best_a, best_score


def structured_init(n, rng, method="mixed"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "ramp":
        a = 0.1 + 0.9 * t
    elif method == "ramp_rev":
        a = 0.1 + 0.9 * (1 - t)
    elif method == "asym":
        a = 0.05 + 0.95 * t**2
    elif method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * t) + 0.2 * np.sin(2 * np.pi * 5 * t)
    elif method == "quadratic":
        a = 0.1 + 0.8 * t * (1 - t) * 4  # Peak in middle
    elif method == "one_side":
        a = np.where(t < 0.5, 1.0, 0.1)
    elif method == "mixed":
        a = 0.2 + 0.6 * t + 0.2 * np.sin(4 * np.pi * t)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    
    a = np.clip(a, 0.01, None)
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses mirror-descent, spectral shaping, subgradient methods,
    multi-scale optimization, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    global_best_score = np.inf
    global_a = None
    global_n = 0
    
    def update_best(a_candidate, n):
        nonlocal global_best_score, global_a, global_n
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        S = a_candidate.sum()
        if S < 0.01:
            return False
        c = fft_convolve_self(a_candidate)
        score = 2.0 * n * c.max() / (S * S)
        if score < global_best_score:
            global_best_score = score
            global_a = a_candidate.copy()
            global_n = n
            return True
        return False
    
    # Try multiple sequence lengths
    n_candidates = [8192, 4096, 16384]
    
    # Try to use previous best as initialization
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if prev.size > 100 and prev.sum() > 0:
            # Find the closest n_candidate
            prev_n = len(prev)
            closest_n = min(n_candidates, key=lambda x: abs(x - prev_n))
            if prev_n in n_candidates:
                n_candidates.remove(prev_n)
                n_candidates.insert(0, prev_n)
            # Adjust to fit
            for n_c in n_candidates:
                if prev_n != n_c:
                    # Resize
                    new_t = np.linspace(0, 1, n_c)
                    old_t = np.linspace(0, 1, prev_n)
                    resized = np.interp(new_t, old_t, prev)
                else:
                    resized = prev.copy()
                if prev_n in n_candidates:
                    break
            else:
                resized = prev[:n_candidates[0]] if prev_n > n_candidates[0] else prev
        else:
            resized = None
    except Exception:
        resized = None
    
    print("Starting optimization. Budget: %.0fs" % budget_s)
    print("n candidates: %s" % n_candidates)
    
    # Phase 1: Quick screening of initializations for the primary n
    n_primary = 8192
    S_primary = np.sqrt(2.0 * float(n_primary))
    N_primary = 1 << (2 * n_primary - 1).bit_length()
    
    print("\n=== Phase 1: Quick screening (n=%d) ===" % n_primary)
    t0 = time.monotonic()
    phase1_time = min(60.0, (deadline - t0) * 0.08)
    
    inits = []
    if resized is not None and len(resized) == n_primary:
        inits.append(("prev", np.clip(resized, 0.01, 1000)))
    
    for method in ["ramp", "ramp_rev", "asym", "sin", "quadratic", "one_side", "mixed"]:
        inits.append((method, structured_init(n_primary, rng, method)))
    inits.append(("rand", 0.2 + 0.8 * rng.random(n_primary)))
    
    best_init_score = np.inf
    best_init_a = None
    for name, arr in inits:
        a_norm = safe_normalize(arr, n_primary)
        score = compute_score(a_norm)
        if score < best_init_score:
            best_init_score = score
            best_init_a = a_norm.copy()
        if score < global_best_score:
            global_best_score = score
            global_a = a_norm.copy()
            global_n = n_primary
        print("  [%s] score=%.6f" % (name, score))
        if time.monotonic() - t0 > phase1_time:
            break
    
    print("  Best init: %.6f" % best_init_score)
    
    # Phase 2: Main optimization loop (70% of remaining time)
    print("\n=== Phase 2: Main optimization ===")
    t1 = time.monotonic()
    phase2_budget = (deadline - t1) * 0.65
    
    a_curr = best_init_a.copy()
    stall_global = 0
    iteration = 0
    
    while time.monotonic() < t1 + phase2_budget:
        iteration += 1
        remaining = (t1 + phase2_budget) - time.monotonic()
        if remaining < 10:
            break
        
        # Alternate between different strategies
        strategy = iteration % 5
        time_budget = min(remaining * 0.3, 30.0)
        
        if strategy == 0:
            # Mirror descent
            a_new, cmax = iterative_mirror_optimize(a_curr, n_primary, S_primary,
                                                    time_budget, rng, target=1.48)
        elif strategy == 1:
            # Subgradient polish
            a_new, cmax = subgradient_polish(a_curr, n_primary, S_primary,
                                             time_budget, rng, top_k_init=10)
        elif strategy == 2:
            # Spectral shaping burst
            N = N_primary
            S = S_primary
            n = n_primary
            a_work = a_curr.copy()
            for _ in range(20):
                if time.monotonic() > t1 + time_budget:
                    break

                beta = 0.2 / (1.0 + 0.05 * _)

                c_check = fft_convolve_self(a_work)
                if c_check.max() < cmax:
                    cmax = c_check.max()
            a_new = a_work
        elif strategy == 3:
            # Coordinate descent
            a_new, cmax = coordinate_descent(a_curr, n_primary, S_primary,
                                             min(time_budget, 20.0), rng,
                                             n_top=15, n_rounds=30)
        else:
            # Random restart + short optimization
            if stall_global > 15:
                a_restart = structured_init(n_primary, rng, method=rng.choice(
                    ["ramp", "ramp_rev", "asym", "sin", "quadratic", "mixed"]))
                a_restart = safe_normalize(a_restart, n_primary)
                # Blend with current
                blend = 0.3
                a_restart = (1 - blend) * a_curr + blend * a_restart
                a_restart = project_simplex(a_restart, S_primary)
                a_new, cmax = iterative_mirror_optimize(a_restart, n_primary, S_primary,
                                                        min(time_budget, 15.0), rng)
            else:
                a_new, cmax = mirror_descent_step(a_curr, S_primary, 10, 0.3, rng)
                a_new = project_simplex(a_new, S_primary)
                cmax = fft_convolve_self(a_new).max()
        
        # Update
        score_new = 2.0 * n_primary * cmax / (S_primary * S_primary)
        if score_new < global_best_score:
            global_best_score = score_new
            global_a = a_new.copy()
            global_n = n_primary
            a_curr = a_new.copy()
            stall_global = 0
            print("  [iter %d] score=%.6f (strategy %d)" % (iteration, global_best_score, strategy))
        else:
            a_curr = a_new.copy()
            stall_global += 1
        
        # Occasionally try different n
        if iteration % 10 == 5 and len(n_candidates) > 1:
            n_try = n_candidates[1]
            S_try = np.sqrt(2.0 * float(n_try))
            a_try = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_primary), a_curr)
            a_try = safe_normalize(a_try, n_try)
            score_try = compute_score(a_try)
            if score_try < global_best_score:
                global_best_score = score_try
                global_a = a_try.copy()
                global_n = n_try
                print("  [n=%d] score=%.6f" % (n_try, global_best_score))
    
    print("  Phase 2 best: %.6f" % global_best_score)
    
    # Phase 3: Focused polishing (20% of remaining time)
    print("\n=== Phase 3: Focused polishing ===")
    t2 = time.monotonic()
    phase3_budget = (deadline - t2) * 0.5
    
    if global_a is not None and phase3_budget > 5:
        # Use the best found
        best_n = global_n
        best_S = np.sqrt(2.0 * float(best_n))
        a_polish = np.interp(np.linspace(0, 1, best_n), np.linspace(0, 1, len(global_a)), global_a) if len(global_a) != best_n else global_a.copy()
        a_polish = safe_normalize(a_polish, best_n)
        
        # Long subgradient
        time_budget = min(phase3_budget * 0.4, 60.0)
        a_p1, _ = subgradient_polish(a_polish, best_n, best_S, time_budget, rng, top_k_init=20)
        if update_best(a_p1, best_n):
            print("  [P3 subgrad] score=%.6f" % global_best_score)
        
        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 10:
            time_budget = min(rem * 0.3, 30.0)
            a_p2, _ = coordinate_descent(global_a, best_n, np.sqrt(2.0 * best_n),
                                         time_budget, rng, n_top=30, n_rounds=100)
            if update_best(a_p2, best_n):
                print("  [P3 coord] score=%.6f" % global_best_score)
        
        # Fine mirror descent
        rem = deadline - time.monotonic()
        if rem > 5:
            time_budget = min(rem * 0.3, 20.0)
            a_p3, _ = iterative_mirror_optimize(global_a, best_n, np.sqrt(2.0 * best_n),
                                                time_budget, rng, target=1.45)
            if update_best(a_p3, best_n):
                print("  [P3 mirror] score=%.6f" % global_best_score)
    
    # Phase 4: Final verification and safety
    print("\n=== Phase 4: Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verified score: %.6f" % ver)
            print("Sequence length: %d" % len(global_a))
            print("Min value: %.4f, Max value: %.4f" % (global_a.min(), global_a.max()))
        else:
            print("evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(512, dtype=np.float64)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(512, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    
    print("Returning sequence of length %d" % len(final_list))
    return final_list
