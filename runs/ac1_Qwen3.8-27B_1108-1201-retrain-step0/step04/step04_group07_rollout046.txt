"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using soft-max gradient descent
with FFT-accelerated gradient computation, backtracking line search,
and periodic restarts.

Key innovations:
1. Soft-max gradient: uses softmax weighting over all lags (not just top-K)
   for a smoother, more informative gradient direction.
2. FFT-accelerated combined gradient: O(N log N) per iteration.
3. Backtracking line search for adaptive step size.
4. Beta annealing: start with smooth (low beta), end with peaky (high beta).
5. Multi-start with diverse initializations.
6. Spectral flattening as a secondary refinement tool.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - S) / rho > 0
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N=None):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:size]
    return c


def compute_score_fast(a):
    """Fast score computation."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2 * n - 2).bit_length()
    A = fft(a, N)
    c_max = np.real(ifft(A * A)).max()
    return float(2.0 * n * c_max / (S * S))


def soft_gradient_step(a, S, beta, alpha, N):
    """
    One step of soft-max gradient descent.
    Returns (a_new, c, c_max, score_new).
    """
    n = len(a)
    # Compute convolution
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    # Soft-max weights
    shifted = c - c_max  # avoid overflow
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    # Combined gradient via FFT
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    # Gradient descent step with simplex projection
    a_new = project_simplex(a - alpha * grad, S)
    a_new = np.maximum(a_new, 0.0)
    # Re-normalize to maintain sum
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * (S / s_new)
        # Re-project to ensure both constraints
        a_new = project_simplex(a_new, S)

    return a_new, c, c_max


def backtrack_line_search(a, S, beta, alpha_init, N, max_iter=20):
    """
    Backtracking line search.
    Returns (a_best, best_score, step_used).
    """
    n = len(a)
    current_score = 2.0 * n * np.real(ifft(fft(a, N) * fft(a, N))).max() / (S * S)

    alpha = alpha_init
    best_a = a.copy()
    best_score = current_score

    for _ in range(max_iter):
        a_new, c, c_max = soft_gradient_step(a, S, beta, alpha, N)
        new_score = 2.0 * n * c_max / (S * S)

        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score

        if new_score < current_score * (1 - 1e-4 * alpha):
            # Armijo condition
            return a_new, new_score, alpha

        alpha *= 0.5
        if alpha < 1e-10:
            break

    return best_a, best_score, alpha


def spectral_flatten(a, S, N, alpha, n_iter):
    """Spectral flattening for diversity."""
    n = len(a)
    for _ in range(n_iter):
        A = fft(a, N)
        P = np.abs(A) ** 2
        p_max = P.max()
        p_min = P.min()
        if p_max < 1e-30:
            break
        # Compress dynamic range
        P_new = P ** (1.0 - alpha)  # toward flat
        # Actually, let's use a different approach:
        # P_new = P * exp(-alpha * (P - P.mean()) / P.std())
        # This is more stable
        p_mean = P.mean()
        p_std = P.std()
        if p_std < 1e-30:
            break
        P_new = P * np.exp(-alpha * (P - p_mean) / p_std)
        P_new = np.maximum(P_new, 0.0)
        # Preserve total power (to maintain sum(a))
        P_new = P_new * (p_mean / P_new.mean()) if P_new.mean() > 0 else P
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
    return a

    Main optimization loop with soft-max gradient, line search, and restarts.

    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a * (S / s)
    a = project_simplex(a, S)

    best_a = a.copy()
    best_score = compute_score_fast(a)

    t_start = time.monotonic()
    it = 0
    alpha = S / (n * 10.0)  # initial step size
    beta = 5.0  # initial beta

    restart_interval = 500  # iterations between restarts
    restart_prob = 0.3

    no_improve_count = 0
    patience = 2000

    while time.monotonic() - t_start < budget_s:
        it += 1

        # Anneal beta: increase over time
        progress = it / max(int(budget_s * 500), 1)  # assume ~500 iters/sec
        beta = 5.0 + 50.0 * min(1.0, it / 50000.0)

        # Anneal alpha: decrease over time
        alpha = max(1e-6 * S, alpha * 0.9999)

        # Gradient step

        new_score = 2.0 * n * c_max / (S * S)

            best_a = a_new.copy()
            no_improve_count = 0
        else:

        else:
