Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Hybrid population-based search combining:
1. Diverse asymmetric initializations (Barker-like, chirp-modulated, random)
2. Spectral-domain genetic crossover (recombine FFT magnitudes/phases)
3. Local FISTA optimization with adaptive LSE smoothing
4. Simulated annealing for escape from local minima
5. Multi-scale exploration (different sequence lengths)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((b_k - T)^p_+)."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.004, beta_max=5000.0, mode="lse"):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (2.0 * beta * 0.3)
        elif mode == "excess":
            T = c.mean() + 0.3 * (c.max() - c.mean())
            grad, _ = excess_gradient(y, c, T, p=2.0)
            step = 0.5 / (0.3 * c.max() + 1e-10)
        else:  # peak
            k = min(32, n // 8)
            idx = np.argsort(c)[::-1][:k]
            weights = np.maximum(c[idx] - c.mean(), 0.0)
            wsum = weights.sum()
            if wsum < 1e-30:
                break
            weights = weights / wsum
            w_full = np.zeros(len(c))
            w_full[idx] = weights
            fsize = 1 << (len(w_full) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A = fft(a, fsize)  # Use a instead of y for stability
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            step = 1.0 / (0.1 * c.max() + 1e-10)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Stall detection and restart
        if it - last_improve > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart with decreasing noise
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Hybrid search combining:
1. Diverse asymmetric initializations (ramps, modulated, spectral)
2. FISTA local optimization with adaptive LSE smoothing
3. Population-based genetic search with spectral crossover
4. Excess annealing for peak flattening
5. Final high-beta polishing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=32):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.maximum(c[idx] - c.mean(), 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.004, beta_max=10000.0):
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        step = 1.0 / (2.0 * beta * 0.3)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if it - last_improve > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_excess_anneal(a_init, S, max_time, rng):
    """Anneal a threshold, minimizing excess above T."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.5)
        T = max(T, T_end)

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if c.max() <= T * (1 + 1e-6):
            T2 = max(c.mean(), T - 0.01 * (c.max() - c.mean() + 1e-10))
            excess = np.maximum(c - T2, 0.0)
        else:
            excess = np.maximum(c - T, 0.0)

        if excess.max() < 1e-15:
            # Use LSE gradient as fallback
            grad = lse_gradient(y, c, 50.0)
            step = 0.01 / (c.max() + 1e-10)
        else:
            w = 2.0 * excess
            wsum = w.sum()
            if wsum < 1e-30:
                break
            w = w / wsum
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            step = 0.5 / (np.sum(excess) + 1e-10) * (1.0 + 0.5 * (1.0 - progress))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if it - last_improve > 1500:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom  # Keep momentum

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

            prev_grad = None
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 4000 == 0:
            decay = np.exp(-it / 40000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                prev_y = y.copy()
                prev_grad = None

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_excess_anneal(a_init, S, max_time, rng):
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.6)
        T = max(T, T_end)

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        excess = np.maximum(c - T, 0.0)
        if excess.max() < 1e-12:
            grad = lse_gradient(y, c, 100.0)
            step = 0.005 / (c.max() + 1e-10)
        else:
            w = 2.0 * excess
            wsum = w.sum()
            if wsum < 1e-30:
                break
            w = w / wsum
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            step = 0.3 * (1.0 + (1.0 - progress)) / (np.sum(excess) + 1e-10)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if it - last_improve > 1000:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress + 0.1)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_peak_shave(a_init, S, max_time, rng):
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(8 + 24 * progress)

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = peak_shave_gradient(y, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.8 / (c.max() * c.max() + 1e-10) * (1.0 + 0.5 * (1.0 - progress))

        if it - last_improve > 1000:
            noise_scale = 0.004 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def generate_asymmetric_inits(n, rng, base_seq=None):
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None and len(base_seq) == n:
        inits.append(np.asarray(base_seq, dtype=np.float64).copy())

    # Strong linear ramp (most asymmetric)
    inits.append(0.1 + 0.9 * tt)
    # Exponential decay
    inits.append(np.exp(-2.0 * tt))
    # Power law decay
    inits.append(0.05 + 0.95 * (1.0 - tt) ** 0.5)
    # Asymmetric Gaussian (skewed)
    inits.append(np.exp(-5.0 * (tt - 0.3) ** 2 / 0.1))
    # S-curve (sigmoid shifted)
    inits.append(0.1 + 0.9 / (1.0 + np.exp(-10 * (tt - 0.4))))
    # Inverse S-curve
    inits.append(0.1 + 0.9 / (1.0 + np.exp(-10 * (0.6 - tt))))
    # Modulated ramp: ramp * (1 + 0.3*sin)
    inits.append((0.1 + 0.5 * tt) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt)))
    inits.append((0.1 + 0.5 * tt) * (1.0 + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5)))
    # Two humps (asymmetric)
    inits.append(np.exp(-20 * (tt - 0.25) ** 2) + 0.6 * np.exp(-15 * (tt - 0.7) ** 2))
    # Cubic polynomial
    inits.append(0.1 + 0.3 * tt + 0.3 * tt ** 2 + 0.3 * tt ** 3)
    # Random with linear trend
    inits.append(0.1 + 0.8 * tt + 0.2 * rng.standard_normal(n))
    # Random exponential
    inits.append(np.exp(rng.normal(0, 0.5, n) - 1.0 * tt))
    # Sine with phase (asymmetric envelope)
    inits.append(0.2 + 0.5 * tt + 0.3 * np.sin(2 * np.pi * 5 * tt + 1.0))
    # Chirp-like
    freq = 3.0
    inits.append(0.1 + 0.5 * tt + 0.3 * np.sin(2 * np.pi * (freq * tt + 0.5 * freq * tt ** 2)))

    # All clipped to non-negative
    return [np.maximum(np.asarray(x, dtype=np.float64), 0.01) for x in inits]


def spectral_crossover(p1, p2, rng):
    n = len(p1)
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(p1, fsize)
    A2 = fft(p2, fsize)

    alpha = rng.uniform(0.2, 0.8)
    mag = np.abs(A1) ** alpha * np.abs(A2) ** (1.0 - alpha)

    if rng.random() < 0.5:
        phase = np.angle(A1)
    else:
        phase = np.angle(A2)

    A = mag * np.exp(1j * phase)
    child = np.real(ifft(A))[:n]
    return np.maximum(child, 0.0)


def spectral_crossover_v2(p1, p2, rng):
    """Frequency split crossover: low freq from one, high from other."""
    n = len(p1)
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(p1, fsize)
    A2 = fft(p2, fsize)

    split = int(rng.uniform(0.2, 0.8) * fsize)
    if rng.random() < 0.5:
        A = np.concatenate([A1[:split], A2[split:]])
    else:
        A = np.concatenate([A2[:split], A1[split:]])

    child = np.real(ifft(A))[:n]
    return np.maximum(child, 0.0)


def upscale_sequence(a_small, n_large):
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def mutate_sequence(a, S, rng, scale=0.02):
    """Apply Gaussian mutation and reproject."""
    noise = rng.normal(0, scale, len(a))
    return project_simplex(a + noise, S)


def jump_mutation(a, S, rng):
    """Large perturbation to escape local minima."""
    n = len(a)
    # Randomly select a block to perturb heavily
    block_size = int(n * rng.uniform(0.1, 0.3))
    start_idx = int(rng.integers(0, n - block_size))
    a_perturbed = a.copy()
    a_perturbed[start_idx:start_idx + block_size] += rng.normal(0, 0.1 * a.sum() / n, block_size)
    return project_simplex(a_perturbed, S)


def ga_select(pop, objs, pop_size, rng, n_elite=2, tournament_size=3):
    """Select next generation using elitism + tournament."""
    order = np.argsort(objs)
    parents = []
    # Elitism
    for i in range(min(n_elite, len(order))):
        parents.append(order[i])
    # Tournament for the rest
    while len(parents) < pop_size:
        idx = rng.integers(0, len(order), tournament_size)
        best_idx = idx[np.argmin(np.array([objs[i] for i in idx]))]
        parents.append(best_idx)
    return parents


def run_genetic_algorithm(pop_init, S, n, max_time, rng, n_elite=2):
    """Main GA loop with periodic local optimization."""
    pop_size = len(pop_init)
    objs = []
    for ind in pop_init:
        c = fft_convolve_self(ind)
        objs.append(c.max())
    objs = np.array(objs)

    best_obj = objs.min()
    best_idx = np.argmin(objs)
    best_a = pop_init[best_idx].copy()

    start = time.monotonic()
    gen = 0

    while time.monotonic() - start < max_time:
        gen += 1
        elapsed = time.monotonic() - start
        remaining_frac = 1.0 - elapsed / max_time

        # Selection
        parent_indices = ga_select(list(range(pop_size)), objs, pop_size, rng, n_elite)

        # Crossover and mutation to create offspring
        offspring = []
        for i in range(pop_size - n_elite):
            p1 = parent_indices[i % len(parent_indices)]
            p2 = parent_indices[(i + 1) % len(parent_indices)]
            if p1 == p2:
                p2 = (p2 + 1) % pop_size

            rng_choice = rng.random()
            if rng_choice < 0.4:
                child = spectral_crossover(pop_init[p1], pop_init[p2], rng)
            elif rng_choice < 0.7:
                child = spectral_crossover_v2(pop_init[p1], pop_init[p2], rng)
            else:
                child = pop_init[p1].copy()

            # Mutation
            mut_scale = 0.01 + 0.02 * remaining_frac
            if rng.random() < 0.15:
                child = jump_mutation(child, S, rng)
            else:
                child = mutate_sequence(child, S, rng, scale=mut_scale)
            child = normalize_to_sum(child, S)
            offspring.append(child)

        # New population: elites + offspring
        elite_indices = np.argsort(objs)[:n_elite]
        new_pop = [pop_init[i].copy() for i in elite_indices]
        new_pop.extend(offspring)
        pop_init = new_pop[:pop_size]

        # Evaluate
        for i, ind in enumerate(pop_init):
            c = fft_convolve_self(ind)
            objs[i] = c.max()

        # T
        # rack best
        if objs.min() < best_obj:
            best_obj = objs.min()
            best_idx = np.argmin(objs)
            best_a = pop_init[best_idx].copy()

        # Periodic local optimization on top candidates
        if gen % 15 == 0 and remaining_frac > 0.05:
            top_k = 3
            top_indices = np.argsort(objs)[:top_k]
            for ti in top_indices:
                t_left = max_time - (time.monotonic() - start) - 5
                if t_left < 10:
                    break
                t_alloc = min(30.0, t_left / (top_k - int(top_indices.tolist().index(ti)) - 1 + 1))
                t_alloc = min(30.0, t_left)
                a_opt, val_opt = run_fista(pop_init[ti], S, t_alloc, rng,
                                           beta_init=50.0, beta_rate=0.005, beta_max=5000.0)
                if val_opt < objs[ti]:
                    pop_init[ti] = a_opt
                    objs[ti] = val_opt
                if val_opt < best_obj:
                    best_obj = val_opt
                    best_a = a_opt.copy()
            # Re-evaluate all after local opt
            for i in range(pop_size):
                c = fft_convolve_self(pop_init[i])
                objs[i] = c.max()

        # Print progress periodically
        if gen % 50 == 0:
            print("  GA gen %d: best_obj=%.6f, elapsed=%.0fs, remaining=%.0fs" %
                  (gen, best_obj, elapsed, max_time - elapsed))

    return best_a, best_obj


def run_multi_phase_polish(a_init, S, max_time, rng):
    """Run multiple polishing phases in sequence."""
    n = len(a_init)
    a = a_init.copy()
    S = np.sqrt(2.0 * n)
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    elapsed = 0.0
    start = time.monotonic()

    # Phase 1: FISTA with medium beta
    t1 = max_time * 0.35
    if max_time > 20:
        a, val = run_fista(a, S, t1, rng, beta_init=20.0, beta_rate=0.005, beta_max=5000.0)
        if val < best:
            best = val
            best_a = a.copy()
        elapsed = time.monotonic() - start

    # Phase 2: Excess annealing
    t2 = max_time * 0.3
    if max_time - elapsed > 15:
        a, val = run_excess_anneal(best_a, S, t2, rng)
        if val < best:
            best = val
            best_a = a.copy()
        elapsed = time.monotonic() - start

    # Phase 3: Peak shaving
    t3 = max_time * 0.2
    if max_time - elapsed > 10:
        a, val = run_peak_shave(best_a, S, t3, rng)
        if val < best:
            best = val
            best_a = a.copy()
        elapsed = time.monotonic() - start

    # Phase 4: High-beta FISTA
    t4 = max_time - elapsed - 5
    if t4 > 5:
        a, val = run_fista(best_a, S, t4, rng, beta_init=200.0, beta_rate=0.003, beta_max=20000.0)
        if val < best:
            best = val
            best_a = a.copy()

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-scale hybrid: diverse asymmetric inits -> FISTA -> GA -> polish.
    """
    rng = np.random.default_rng(seed)
    total_budget = budget_s
    deadline = time.monotonic() + total_budget - 30.0  # Safety margin

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, deadline - time.monotonic()))

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Main working length
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)

    # Generate initializations for main length
    inits_main = []
    if given is not None:
        if len(given) == n_main:
            inits_main.append(given.copy())
        elif len(given) > n_main:
            ratio = len(given) // n_main
            small = given[:n_main * ratio].reshape(n_main, ratio).mean(axis=1)
            inits_main.append(small)
        else:
            inits_main.append(upscale_sequence(given, n_main))
        # Also add perturbed version of given
        inits_main.append(mutate_sequence(normalize_to_sum(given if len(given) == n_main else upscale_sequence(given, n_main), S_main), S_main, rng, 0.03))

    gen_inits = generate_asymmetric_inits(n_main, rng)
    inits_main.extend(gen_inits[:10])

    # Normalize all inits
    inits_main = [normalize_to_sum(x, S_main) for x in inits_main]
    print("Generated %d initializations for n=%d" % (len(inits_main), n_main))

    # Phase 1: Quick FISTA on diverse inits
    n_init_to_try = min(6, len(inits_main))
    t_per_init = 40.0
    print("\n=== Phase 1: Quick FISTA on %d inits (%.0fs each) ===" % (n_init_to_try, t_per_init))

    phase1_results = []
    for i in range(n_init_to_try):
        if time.monotonic() > deadline - 1200:
            break
        t_alloc = min(t_per_init, deadline - time.monotonic() - 1000)
        if t_alloc < 10:
            break
        a_opt, val = run_fista(inits_main[i], S_main, t_alloc, rng,
                               beta_init=10.0, beta_rate=0.005, beta_max=5000.0)
        obj = 2.0 * n_main * val / (a_opt.sum() ** 2)
        phase1_results.append((obj, a_opt.copy()))
        print("  Init %d: max_b=%.6f, obj=%.6f" % (i, val, obj))
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()

    # Sort and pick top for GA
    phase1_results.sort(key=lambda x: x[0])
    print("Phase 1 best: obj=%.6f" % phase1_results[0][0])

    # Phase 2: Genetic Algorithm
    print("\n=== Phase 2: Genetic Algorithm ===")
    ga_time = min(1200.0, deadline - time.monotonic() - 400)
    if ga_time > 60:
        # Build initial population for GA
        ga_pop = [phase1_results[i][1] for i in range(min(6, len(phase1_results)))]
        # Add some diverse members
        for i in range(6, min(12, len(inits_main))):
            if time.monotonic() > deadline - 400:
                break
            ga_pop.append(inits_main[i])
            # Quick eval
            c = fft_convolve_self(ga_pop[-1])
            # Don't add if too bad
            if 2.0 * n_main * c.max() / (ga_pop[-1].sum() ** 2) > 2.0:
                ga_pop.append(mutate_sequence(ga_pop[0], S_main, rng, 0.1))

        ga_pop = ga_pop[:12]
        print("  GA population size: %d, time budget: %.0fs" % (len(ga_pop), ga_time))

        a_ga, val_ga = run_genetic_algorithm(ga_pop, S_main, n_main, ga_time, rng)
        obj_ga = 2.0 * n_main * val_ga / (a_ga.sum() ** 2)
        print("  GA best: max_b=%.6f, obj=%.6f" % (val_ga, obj_ga))
        if obj_ga < global_best:
            global_best = obj_ga
            global_a = a_ga.copy()

    # Phase 3: Multi-phase polishing
    print("\n=== Phase 3: Multi-phase Polishing ===")
    t_polish = min(400.0, deadline - time.monotonic() - 30)
    if t_polish > 30 and global_a is not None:
        print("  Polishing with %.0fs" % t_polish)
        a_fin, val_fin = run_multi_phase_polish(global_a, S_main, t_polish, rng)
        obj_fin = 2.0 * n_main * val_fin / (a_fin.sum() ** 2)
        print("  After polishing: max_b=%.6f, obj=%.6f" % (val_fin, obj_fin))
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()

    # Phase 4: Try multi-scale approach if time allows
    t_left = deadline - time.monotonic()
    if t_left > 100 and global_a is not None:
        print("\n=== Phase 4: Multi-scale refinement (%.0fs) ===" % t_left)
        # Downsample to n=2048, optimize, upscale back
        n_small = 2048
        S_small = np.sqrt(2.0 * n_small)
        t_old = np.linspace(0, 1, n_main)
        t_new = np.linspace(0, 1, n_small)
        a_small = np.interp(t_new, t_old, global_a)
        a_small = normalize_to_sum(a_small, S_small)

        t_small = min(40.0, t_left * 0.3)
        a_small_opt, val_small = run_fista(a_small, S_small, t_small, rng,
                                           beta_init=20.0, beta_rate=0.006, beta_max=5000.0)
        obj_small = 2.0 * n_small * val_small / (a_small_opt.sum() ** 2)
        print("  Small scale: obj=%.6f" % obj_small)

        # Upscale and refine
        a_up = upscale_sequence(a_small_opt, n_main)
        a_up = normalize_to_sum(a_up, S_main)

        t_up = min(60.0, t_left * 0.4)
        a_up_opt, val_up = run_fista(a_up, S_main, t_up, rng,
                                     beta_init=50.0, beta_rate=0.004, beta_max=10000.0)
        obj_up = 2.0 * n_main * val_up / (a_up_opt.sum() ** 2)
        print("  Upscaled: obj=%.6f" % obj_up)
        if obj_up < global_best:
            global_best = obj_up
            global_a = a_up_opt.copy()

    # Final verification
    print("\n=== Final Result ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
        print("Final: n=%d, objective=%.6f" % (len(global_a), ver))
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        print("Fallback: n=256, objective=%.6f" % evaluate_sequence(final_list))

    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
