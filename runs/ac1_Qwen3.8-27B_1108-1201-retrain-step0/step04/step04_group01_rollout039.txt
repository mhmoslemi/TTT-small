import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0)
    sv = v.sum()
    if sv <= 0:
        return np.full(n, S / n)
    # Bisection for the threshold
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)

    e = np.exp(scaled)

    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: subgradient descent with top-k lags, LSE polishing,
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, verbose_msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            if verbose_msg:
                print(f"  NEW GLOBAL BEST: obj={obj:.6f} ({verbose_msg})")

    # === Phase 1: Primary optimization on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(700, (deadline - time.monotonic()) * 0.35)
        print(f"\nPhase 1: Subgradient on provided, n={n1}, time={t1:.0f}s")
        a_opt, best_max = optimize_subgradient(a1, S1, t1, rng)
        update_global(a_opt, "Phase 1 subgradient")

        # LSE polish on the result
        remaining = deadline - time.monotonic()
        t1b = min(300, remaining * 0.25)
        if t1b > 20:
            print(f"Phase 1b: LSE polish, time={t1b:.0f}s")
            a_pol, _ = optimize_lse_polish(a_opt, S1, t1b, rng,
                                           beta_init=100.0, beta_max=8000.0)
            update_global(a_pol, "Phase 1b LSE polish")

    # === Phase 2: Try alternative lengths ===
    remaining = deadline - time.monotonic()
    lengths_to_try = []
    if global_a is not None:
        lengths_to_try = [8192, 16384]
    else:
        lengths_to_try = [4096, 8192]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        if global_a is not None and len(global_a) == n:
            continue

        S = np.sqrt(2.0 * n)
        t_alloc = min(400, (deadline - time.monotonic()) * 0.25)
        if t_alloc < 30:
            continue

        print(f"\nPhase 2: New length n={n}, time={t_alloc:.0f}s")
        init = generate_init(n, rng, base=global_a)
        init = project_simplex(init, S)

        a_opt, best_max = optimize_subgradient(init, S, t_alloc * 0.6, rng,
                                               verbose=False)
        update_global(a_opt, f"Phase 2 n={n} subgrad")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_pol, _ = optimize_lse_polish(a_opt, S, min(150, rem * 0.4), rng,
                                           beta_init=200.0, beta_max=10000.0)
            update_global(a_pol, f"Phase 2 n={n} polish")

    # === Phase 3: Intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t3 = remaining - 20
        print(f"\nPhase 3: Final refinement, n={n}, time={t3:.0f}s")

        # Alternate between subgradient and LSE
        a_curr = global_a.copy()
        for phase_i in range(4):
            if time.monotonic() > deadline - 10:
                break
            t_phase = min(t3 / 4, deadline - time.monotonic() - 10)
            if t_phase < 5:
                break

            if phase_i % 2 == 0:
                a_curr, _ = optimize_subgradient(a_curr, S, t_phase, rng,
                                                 verbose=False)
            else:
                a_curr, _ = optimize_lse_polish(a_curr, S, t_phase, rng,
                                                beta_init=500.0, beta_max=20000.0)
            update_global(a_curr, f"Phase 3 iter {phase_i}")

    # === Phase 4: Multiple random restarts on best solution ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = 3
        t_per = (remaining - 20) / n_restarts

        print(f"\nPhase 4: {n_restarts} perturbations+refines, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Perturb the global best
            noise_scale = 0.02 * S / np.sqrt(n)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            a_pert, _ = optimize_subgradient(a_pert, S, t_per, rng, verbose=False)
            update_global(a_pert, f"Phase 4 restart {ri}")

    # === Final verification and return ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
