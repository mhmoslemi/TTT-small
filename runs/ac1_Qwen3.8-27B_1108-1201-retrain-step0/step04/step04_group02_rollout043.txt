def project_simplex(v,S=1.0):
   v=np.array(v,dtype=float)
   n=v.size
   u=np.sort(v)[::-1]
   css=np.cumsum(u)
   rho=np.nonzero(u*(np.arange(1,n+1)-css+S)>0)[0]
   if len(rho)==0: return np.full(n,S/n)
   rho=rho[-1]
   theta=(css[rho]-S)/(rho+1)
   w=np.maximum(v-theta,0)
   s=w.sum()
   if s<=0: return np.full(n,S/n)
   return w*(S/s)

def spectral_flatten_target(a, target, relax, fsize, rng=None, phase_noise=0.0):
    a=np.asarray(a, float)
    F=rfft(a, fsize)
    G=rfft(target, fsize)
    absG=np.abs(G)
    absF=np.abs(F)
    ph=np.ones(F.shape, dtype=np.complex128)
    mask=absF > 1e-30
    ph[mask]=F[mask]/absF[mask]
    if phase_noise is not None and phase_noise > 0.0 and rng is not None:
        ph = ph * np.exp(1j * phase_noise * rng.standard_normal(F.shape))
    Fnew = np.sqrt(absG) * ph
    if Fnew.size > 0:
        Fnew[0] = abs(Fnew[0])
    if Fnew.size > 1:
        Fnew[-1] = abs(Fnew[-1])
    b=irfft(Fnew, fsize)[:a.size]
    b=np.maximum(b,0.0)
    b=normalize_sum(b,1.0)
    out=(1.0-relax)*a + relax*b
    return normalize_sum(out,1.0)
```

```python
"""
Search for non-negative step-function heights minimizing

    2*n*max_k((a*a)[k]) / (sum(a))^2

using a scale-invariant mirror-descent optimizer on the probability-simplex
normalization sum(a)=1.  The method combines:

- top-k subgradient mirror updates (with adaptive log-scale step alpha),
- periodic spectral flattening of the self-convolution spectrum,
- random phase jumps to escape basins,
- multiple starts from the supplied construction, skewed variants,
  downsampled variants, and random-phase seeds built from the supplied
  convolution spectrum,
- optional upsampled refinements up to length 16384.

The returned list is normalized to sum 1, so the objective is evaluated
directly without rescaling issues.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft


STOP_TARGET = 1.5025


def next_pow2_int(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_sum(a, S=1.0):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(a.size, S / float(max(1, a.size)), dtype=np.float64)
    return a * (S / s)


def mix_uniform(a, frac):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    out = (1.0 - frac) * a + (frac / float(n))
    return normalize_sum(out, 1.0)


def conv_self(a, fsize=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if fsize is None:
        fsize = next_pow2_int(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)


def fast_objective(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    n = a.size
    if n == 0 or s < 1e-12:
        return np.inf
    c = conv_self(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def topk_subgradient_slice(a, c, k):
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    k = int(k)
    if k <= 0 or n == 0:
        return grad

    if k == 1:
        idx = [int(np.argmax(c))]
    elif k >= c.size:
        idx = list(range(c.size))
    else:
        idx = np.argpartition(c, -k)[-k:]

    for jj in idx:
        j = int(jj)
        lo = j - n + 1
        if lo < 0:
            lo = 0
        hi = j
        if hi > n - 1:
            hi = n - 1
        if lo > hi:
            continue
        start = j - hi
        stop = j - lo
        grad[lo:hi + 1] += 2.0 * a[start:stop + 1][::-1]

    grad /= float(k)
    return grad


def hankel_hv(lam, a):
    n = a.size
    m = lam.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    size = m + n - 1
    fsize = next_pow2_int(size)
    L = rfft(lam, fsize)
    R = rfft(a[::-1], fsize)
    out = irfft(L * R, fsize)
    return out[n - 1:n - 1 + n]


def topk_gradient(a, c, k):
    n = a.size
    k = int(k)
    if k <= 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    if k == 1:
        return topk_subgradient_slice(a, c, 1)
    if k <= 64:
        return topk_subgradient_slice(a, c, k)
    if k >= c.size:
        lam = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
        return 2.0 * hankel_hv(lam, a)
    idx = np.argpartition(c, c.size - k)[-k:]
    lam = np.zeros(c.size, dtype=np.float64)
    lam[idx] = 1.0 / float(k)
    return 2.0 * hankel_hv(lam, a)


def log_perturb(a, temp, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    loga = np.log(np.maximum(a, 1e-300))
    loga = loga + float(temp) * rng.standard_normal(n)
    out = np.exp(np.clip(loga - loga.max(), -700, 700))
    s = out.sum()
    if s < 1e-300:
        return normalize_sum(a, 1.0)
    return normalize_sum(out / s, 1.0)


def make_target_from_c(c, delta):
    target = np.maximum(c, 0.0).copy()
    if target.size == 0:
        return target
    maxc = float(target.max())
    if maxc <= 0.0:
        return target
    level = maxc * (1.0 - float(delta))
    minc = float(target.min())
    if level < minc:
        level = minc
    target[target > level] = level
    st = float(target.sum())
    if st > 0.0:
        target *= 1.0 / st
    return target


def make_target_power(c, p):
    target = np.maximum(c, 0.0)
    if target.size == 0:
        return target
    p = float(p)
    if p <= 0.0 or not np.isfinite(p):
        p = 1.0
    target = target ** p
    st = float(target.sum())
    if st > 0.0:
        target *= 1.0 / st
    return target


def spectral_flatten_target(a, target, relax, fsize, rng=None, phase_noise=0.0):
    a = np.asarray(a, dtype=np.float64)
    F = rfft(a, fsize)
    G = rfft(target, fsize)
    absG = np.abs(G)
    absF = np.abs(F)

    ph = np.ones(F.shape, dtype=np.complex128)
    mask = absF > 1e-30
    ph[mask] = F[mask] / absF[mask]

    if rng is not None and phase_noise is not None and phase_noise > 0.0:
        ph = ph * np.exp(1j * float(phase_noise) * rng.standard_normal(F.shape))

    Fnew = np.sqrt(absG) * ph
    if Fnew.size > 0:
        Fnew[0] = abs(Fnew[0])
    if Fnew.size > 1:
        Fnew[-1] = abs(Fnew[-1])

    b = irfft(Fnew, fsize)[:a.size]
    b = np.maximum(b, 0.0)
    b = normalize_sum(b, 1.0)

    out = (1.0 - float(relax)) * a + float(relax) * b
    return normalize_sum(out, 1.0)


def spectral_flatten_candidate(a, c, delta, relax, fsize, rng=None):
    target = make_target_from_c(c, delta)
    return spectral_flatten_target(a, target, relax, fsize, rng, 0.0)


def spectral_phase_jump(a, noise, rng, fsize):
    a = np.asarray(a, dtype=np.float64)
    F = rfft(a, fsize)
    absF = np.abs(F)
    ph = np.exp(1j * float(noise) * rng.standard_normal(F.shape))
    if ph.size > 0:
        ph[0] = 1.0 + 0.0j
    if ph.size > 1:
        ph[-1] = 1.0 + 0.0j
    b = irfft(absF * ph, fsize)[:a.size]
    b = np.maximum(b, 0.0)
    return normalize_sum(b, 1.0)


def random_phase_seed_from_target(n, target, rng, fsize=None):
    if fsize is None:
        fsize = next_pow2_int(2 * int(n) - 1)
    G = rfft(target, fsize)
    absG = np.abs(G)
    ph = np.exp(1j * rng.uniform(-np.pi, np.pi, size=G.shape))
    if ph.size > 0:
        ph[0] = 1.0 + 0.0j
    if ph.size > 1:
        ph[-1] = 1.0 + 0.0j
    F = np.sqrt(absG) * ph
    if F.size > 0:
        F[0] = abs(F[0])
    if F.size > 1:
        F[-1] = abs(F[-1])
    a = irfft(F, fsize)[:int(n)]
    a = np.maximum(a, 0.0)
    return normalize_sum(a, 1.0)


def resample_sequence(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = a.size
    n_new = int(n_new)
    if n_new <= 0:
        return a.copy()
    if n_new == n_old:
        return normalize_sum(a, 1.0)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    out = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    return normalize_sum(out, 1.0)


def make_initializations(n, rng):
    out = []
    n = int(n)
    tt = np.linspace(0.0, 1.0, n, endpoint=False)

    shapes = []
    shapes.append(("uniform", np.ones(n, dtype=np.float64)))
    shapes.append(("exp_up", np.exp(5.0 * tt)))
    shapes.append(("exp_down", np.exp(-5.0 * tt)))
    shapes.append(("bump_neg", np.exp(4.0 * tt * (tt - 1.0))))
    shapes.append(("bump_pos", np.exp(-4.0 * tt * (tt - 1.0))))

    centers = (0.15, 0.3, 0.45, 0.55, 0.7, 0.85)
    for ci, mu in enumerate(centers):
        shapes.append(("gauss_%d" % ci, np.exp(-2.0 * ((tt - mu) / 0.15) ** 2)))

    powers = (0.5, 1.0, 2.0)
    for pi, p in enumerate(powers):
        shapes.append(("poly_l_%d" % pi, tt ** p))
        shapes.append(("poly_r_%d" % pi, (1.0 - tt) ** p))

    shapes.append(("wavy1", np.maximum(0.0, np.sin(2.0 * np.pi * tt) + 0.35)))
    shapes.append(("wavy2", np.maximum(0.0, np.sin(4.0 * np.pi * tt + 1.1) + 0.55)))
    shapes.append(("wavy3", 1.0 + 0.4 * np.sin(6.0 * np.pi * tt) +
                   0.2 * np.cos(10.0 * np.pi * tt + 0.7)))

    for name, arr in shapes:
        out.append((name, normalize_sum(arr, 1.0)))

    out.append(("random", normalize_sum(rng.random(n) + 0.01, 1.0)))
    return out


def run_mirror(a_init, budget, rng, max_k=128, use_spectral=True, name=""):
    a = normalize_sum(a_init, 1.0)
    n = a.size
    if n == 0:
        return a, np.inf
    if budget <= 0:
        return a, fast_objective(a)

    fsize = next_pow2_int(2 * n - 1)
    start = time.monotonic()
    deadline = start + float(budget)

    c = conv_self(a, fsize)
    cur_obj = float(2.0 * n * c.max())
    best_obj = cur_obj
    best_a = a.copy()

    alpha = 0.15
    it = 0
    stall = 0
    last_print = start

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(float(budget), 1e-9))

        cur_obj = float(2.0 * n * c.max())
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0

        if best_obj < STOP_TARGET:
            break

        if max_k <= 1:
            k_base = 1
        else:
            k_base = 1 + int(round((float(max_k) - 1.0) * (progress ** 0.7)))
        k = max(1, int(k_base * rng.uniform(0.6, 1.4)))
        if k > c.size:
            k = c.size

        grad = topk_gradient(a, c, k)
        maxg = float(np.max(grad)) if grad.size else 0.0
        if maxg <= 1e-18:
            break

        base_alpha = 0.25 * (1.0 - progress) + 0.02
        if alpha < base_alpha * 0.25:
            alpha = base_alpha * 0.5
        alpha = min(max(alpha, 1e-4), 0.5)

        scaled = grad / maxg
        loga = np.log(np.maximum(a, 1e-300))
        new_log = loga - alpha * scaled
        new_log = np.clip(new_log, -700, 700)
        new_a = np.exp(new_log - new_log.max())
        snew = float(new_a.sum())

        if snew > 1e-300:
            new_a = new_a / snew
            new_c = conv_self(new_a, fsize)
            new_obj = float(2.0 * n * new_c.max())
            if new_obj < cur_obj - 1e-12:
                a = new_a
                c = new_c
                cur_obj = new_obj
                alpha = min(alpha * 1.05, 0.5)
                stall = 0
            else:
                alpha = max(alpha * 0.75, 1e-4)
                stall += 1
        else:
            stall += 10

        if use_spectral and it % 500 == 0 and c.size > 0:
            improved_sp = False

            for delta in (0.08, 0.2):
                target = make_target_from_c(c, delta)
                cand = spectral_flatten_target(a, target, 0.5, fsize)
                cc = conv_self(cand, fsize)
                co = float(2.0 * n * cc.max())
                if co < cur_obj - 1e-12:
                    a = cand
                    c = cc
                    cur_obj = co
                    improved_sp = True
                    stall = 0
                    break

            if not improved_sp:
                for p in (0.85, 0.65):
                    target = make_target_power(c, p)
                    cand = spectral_flatten_target(a, target, 0.4, fsize)
                    cc = conv_self(cand, fsize)
                    co = float(2.0 * n * cc.max())
                    if co < cur_obj - 1e-12:
                        a = cand
                        c = cc
                        cur_obj = co
                        improved_sp = True
                        stall = 0
                        break

            if not improved_sp:
                for noise in (0.3, 1.0, 2.5):
                    cand = spectral_phase_jump(a, noise, rng, fsize)
                    cc = conv_self(cand, fsize)
                    co = float(2.0 * n * cc.max())
                    if co < cur_obj - 1e-12:
                        a = cand
                        c = cc
                        cur_obj = co
                        improved_sp = True
                        stall = 0
                        break

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()

        if stall > 150:
            temp = 0.12 * (1.0 - progress) + 0.02
            a = log_perturb(a, temp, rng)
            c = conv_self(a, fsize)
            cur_obj = float(2.0 * n * c.max())
            stall = 0
            alpha = base_alpha
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
        elif stall > 600:
            a = mix_uniform(a, 0.1)
            c = conv_self(a, fsize)
            cur_obj = float(2.0 * n * c.max())
            stall = 0
            alpha = base_alpha
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()

        it += 1
        if it % 5000 == 0 and time.monotonic() - last_print > 30.0:
            print("  %s iter=%d best=%.6f cur=%.6f alpha=%.4f k=%d" %
                  (name, it, best_obj, cur_obj, alpha, k))
            last_print = time.monotonic()

    return normalize_sum(best_a, 1.0), best_obj


def propose_candidate(seed=42, budget_s=2300):

            parent = normalize_sum(arr, 1.0)
            if parent.size > 16384:
                parent = resample_sequence(parent, 16384)

                              normalize_sum(parent * np.exp(s * tt), 1.0)))

        pc = conv_self(parent, fsize_p)
        pc_clip = make_target_from_c(pc, 0.2)
        pc_pow = make_target_power(pc, 0.85)

        for si in range(10):

            elif si % 3 == 1:
                target = pc_clip
            else:
                target = pc_pow
            init_list.append(("phase_seed_%d" % si,
                              random_phase_seed_from_target(p_len, target, rng, fsize_p)))
    else:
        n0 = 4096
        init_list.extend(make_initializations(n0, rng))
        flat = np.full(2 * n0 - 1, 1.0 / float(2 * n0 - 1), dtype=np.float64)
        for si in range(12):
            init_list.append(("seed_flat_%d" % si,
                              random_phase_seed_from_target(n0, flat, rng)))

    scored = []
    for idx, (cname, a) in enumerate(init_list):
        a = normalize_sum(a, 1.0)
        if a.size < 16:
            continue
        obj = fast_objective(a)
        if np.isfinite(obj):
            scored.append((obj, idx, cname, a))

    if not scored:
        best_a = np.ones(1024, dtype=np.float64)
        best_a = normalize_sum(best_a, 1.0)
        best_obj = fast_objective(best_a)
        print("No valid initial constructions; using uniform fallback, obj=%.6f" % best_obj)
    else:
        scored.sort()
        best_obj = scored[0][0]
        best_a = scored[0][3]
        print("Initial best: name=%s n=%d obj=%.6f" % (scored[0][2], best_a.size, best_obj))

    # Quick local refinement of the few best starts.
    rem = deadline - time.monotonic()
    if rem > 300:
        quick_total = min(rem * 0.25, 600.0)
        top_count = min(3, len(scored))
        if top_count > 0 and quick_total > 30:
            per = quick_total / float(top_count)
            for qi in range(top_count):
                if time.monotonic() > deadline - 200:
                    break
                a_opt, obj = run_mirror(scored[qi][3], per, rng,
                                        max_k=64, use_spectral=True,
                                        name="quick_%s" % scored[qi][2])
                if obj < best_obj:
                    best_obj = obj
                    best_a = normalize_sum(a_opt, 1.0)
                    print("Quick improved: n=%d obj=%.6f" % (best_a.size, best_obj))

    # Main long refinement.
    rem = deadline - time.monotonic()
    if rem > 120:
        main_budget = min(rem * 0.5, 1200.0)
        a_opt, obj = run_mirror(best_a, main_budget, rng,
                                max_k=128, use_spectral=True,
                                name="main_n%d" % best_a.size)
        if obj < best_obj:
            best_obj = obj
            best_a = normalize_sum(a_opt, 1.0)
            print("Main improved: n=%d obj=%.6f" % (best_a.size, best_obj))

    # Upsample refinement, capped at 16384 for safe evaluation cost.
    rem = deadline - time.monotonic()
    ups_done = 0
    while best_a is not None and best_a.size < 16384 and rem > 300 and ups_done < 2:
        up_n = min(int(best_a.size) * 2, 16384)
        up_init = resample_sequence(best_a, up_n)

        trial_budget = min(240.0, (deadline - time.monotonic()) * 0.15)
        a_up, obj_up = run_mirror(up_init, trial_budget, rng,
                                  max_k=96, use_spectral=True,
                                  name="up_trial_n%d" % up_n)

        if obj_up < best_obj - 1e-6 and (deadline - time.monotonic()) > 200:
            cont_budget = min((deadline - time.monotonic()) * 0.40, 900.0)
            a_up2, obj_up2 = run_mirror(a_up, cont_budget, rng,
                                        max_k=128, use_spectral=True,
                                        name="up_cont_n%d" % up_n)
            if obj_up2 < best_obj:
                best_obj = obj_up2
                best_a = normalize_sum(a_up2, 1.0)
                print("Upsample improved: n=%d obj=%.6f" % (best_a.size, best_obj))
            elif obj_up < best_obj:
                best_obj = obj_up
                best_a = normalize_sum(a_up, 1.0)
                print("Upsample trial improved: n=%d obj=%.6f" % (best_a.size, best_obj))
            ups_done += 1
            rem = deadline - time.monotonic()
        else:
            break

    # Final polish.
    rem = deadline - time.monotonic()
    if rem > 45:
        final_budget = rem * 0.90
        a_opt, obj = run_mirror(best_a, final_budget, rng,
                                max_k=64, use_spectral=True,
                                name="final_n%d" % best_a.size)
        if obj < best_obj:
            best_obj = obj
            best_a = normalize_sum(a_opt, 1.0)
            print("Final polish improved: n=%d obj=%.6f" % (best_a.size, best_obj))

    # Safety: make sure final sequence is valid and within evaluator bounds.
    best_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = normalize_sum(best_a, 1.0)

    final_list = [float(x) for x in best_a]

    try:
        verified = evaluate_sequence(final_list)
    except Exception:
        verified = np.inf

    if np.isfinite(verified):
        if verified < best_obj:
            best_obj = verified
        print("Verified objective: %.6f, length=%d" % (best_obj, len(final_list)))
    else:
        print("evaluate_sequence rejected final sequence; using uniform fallback")
        final_list = [1.0 / 1024.0] * 1024
        best_obj = 2.0

    print("Returning sequence of length %d with estimated objective %.6f" %
          (len(final_list), best_obj))
    print("Total time: %.0fs" % (time.monotonic() - start))
    return final_list
