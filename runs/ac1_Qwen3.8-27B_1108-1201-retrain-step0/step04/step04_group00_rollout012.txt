def fft_conv(a, b, size=None):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    if size is None:
        size = n
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:n]


def self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / rho
    valid = w > tau
    if not np.any(valid):
        k = n
    else:
        k = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[k - 1] - S) / k
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE of c with parameter beta."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_grad(y, c, beta):
    """LSE gradient w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgrad_random(y, c, rng, tol_frac=0.0):
    """Random subgradient: pick random lag(s) near the max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    # Pick a random subset of active lags
    k = min(len(active), max(1, int(np.random.default_rng(rng.integers(10**9)).integers(1, min(8, len(active) + 1)))))
    chosen = rng.choice(active, size=k, replace=False)
    w_full = np.zeros(len(c))
    w_full[chosen] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgrad_topk(y, c, k, power=1.0):
    """Top-k subgradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute objective: 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:

    c = self_conv(a)
    return 2.0 * n * c.max() / (s * s)

                step *= 0.5
        else:
            x_trial = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - a)
        y = np.maximum(y, 0.0)
        a = x_trial
        t_mom = t_new

        # Noise injection and restart
        if stuck > noise_interval and it % max(1, noise_interval // 10) == 0:
            noise_scale = noise_base * S / np.sqrt(n) * np.exp(-stuck / (noise_interval * 2))
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                restart_count += 1
                stuck = 0

        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def randomized_subgrad(a_init, max_time, rng):
    """
    Randomized subgradient method with momentum.
    Uses random selections of near-max lags for the subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    base_step = S / (2.0 * np.sqrt(n))  # initial step

    while time.monotonic() - start < max_time:
        it += 1
        c = self_conv(y)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance for "near-max"
        if stuck < 500:
            tol_frac = 0.0
        elif stuck < 2000:
            tol_frac = 0.002
        elif stuck < 5000:
            tol_frac = 0.005
        else:
            tol_frac = 0.01

        grad = subgrad_random(y, c, rng, tol_frac=tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step size with restarts
        step = base_step / (1.0 + 0.01 * it)
        if stuck > 3000:
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Aggressive restart when very stuck
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    return best_a, best_obj


def active_polish(a_init, max_time, rng):
    """
    Polishing with exact subgradient on the active set.
    Iteratively identifies and uses the tight constraints.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Track active set history
    active_history = []

    while time.monotonic() - start < max_time:
        it += 1
        c = self_conv(y)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        max_val = c.max()

        # Identify active set with widening tolerance
        if stuck < 100:
            tol = 1e-10
        elif stuck < 500:
            tol = 1e-8
        elif stuck < 2000:
            tol = max_val * 0.001
        elif stuck < 5000:
            tol = max_val * 0.005
        else:
            tol = max_val * 0.02

        active = np.where(c >= max_val - tol)[0]
        k = len(active)

        if k == 1:
            grad = subgrad_topk(y, c, 1, power=1.0)
        else:
            # Use top-k with the active set
            kk = min(k, 32)
            grad = subgrad_topk(y, c, kk, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Very conservative step
        step = S / (2.0 * gnorm) * (1.0 / (1.0 + 0.02 * stuck))
        step = max(step, 1e-10 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Occasional small perturbation
        if stuck > 3000 and it % 2000 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    return best_a, best_obj


def local_search(a_init, max_time, rng):
    """
    Local search: try small random modifications to subsets of coordinates.
    Uses a simulated annealing-like acceptance criterion.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = self_conv(a).max()
    best_a = a.copy()
    cur_obj = best_obj

    # Track which lags are at the max (for gradient info)
    c = self_conv(a)

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = self_conv(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        # Determine how many coordinates to modify
        # Start with larger modifications, decrease over time
        frac = (time.monotonic() - start) / max_time
        n_mod = max(1, int(n * 0.01 * (1.0 - frac)))  # 1% to 0% of n

        # Select random coordinates
        idx = rng.choice(n, size=n_mod, replace=False)

        # Compute a perturbation
        # Use the subgradient to guide the direction
        max_val = c.max()
        active = np.where(c >= max_val * (1.0 - 0.01))[0]
        k = len(active)
        kk = min(k, 16)
        idx_top = np.argsort(c)[::-1][:kk]
        w_full = np.zeros(len(c))
        w_full[idx_top] = 1.0 / kk
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Perturbation: move in -grad direction for selected coords
        scale = 0.05 * S / np.sqrt(n) * np.exp(-frac * 3.0)
        delta = -grad[idx] * scale * rng.uniform(0.5, 1.5, n_mod)

        a_trial = a.copy()
        a_trial[idx] += delta
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = a_trial * (S / a_trial.sum())

        c_trial = self_conv(a_trial)
        obj_trial = c_trial.max()

        # Accept with annealing criterion
        if obj_trial < cur_obj:
            a = a_trial
            cur_obj = obj_trial
        elif rng.random() < np.exp(-(obj_trial - cur_obj) / (0.001 * (1.0 - frac) + 1e-10)):
            a = a_trial
            cur_obj = obj_trial

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging."""
    n = len(a)
    n_new = n // factor
    return a[:n_new * factor].reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
