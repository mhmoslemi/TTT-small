For the line search: after getting g from the LP, I want to find the best α ∈ [0, 1] such that:

Since both f and g have sum S (after normalization), f_new also has sum S. The score is:
score(α) = 2n * max_k [(1-α)^2 * c_f[k] + 2α(1-α) * c_fg[k] + α^2 * c_g[k]] / S^2

I can precompute c_f, c_fg, c_g (using FFT), then do a 1D minimization over α.

The function max_k [quadratic in α] is convex in α, so I can use golden section search or scipy's minimize_scalar.

import time

from scipy.sparse import csr_matrix
from scipy.optimize import minimize_scalar
import cvxpy as cp

def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def normalize_sum(a, S):
    """Scale a to have sum S, clipping to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)

def score_of(a):
    """Compute the score 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # Column j has n non-zeros at rows j, j+1, ..., j+n-1 with values f[0],...,f[n-1]
    n2 = n * n
    rows = np.empty(n2, dtype=np.int32)
    cols = np.empty(n2, dtype=np.int32)
    vals = np.empty(n2, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        rows[idx:idx+n] = np.arange(j, j+n)
        cols[idx:idx+n] = j
        vals[idx:idx+n] = f
        idx += n
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp_cross(f, S, solver=None):
    """Solve min t s.t. T(f)g <= t, sum(g) = S, g >= 0.
    Returns (t_opt, g_opt)."""
    n = len(f)
    A = build_conv_matrix(f)

    t = cp.Variable()
    objective = cp.Minimize(t)
    constraints = [A @ g <= t, cp.sum(g) == S]
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference

    if solver is not None:
        solvers_to_try = [solver]
    else:

        except:

        except:
            pass
        solvers_to_try.extend([cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC])
    
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                g_val = np.asarray(g.value).flatten()
                t_val = float(t.value)
                g_val = np.clip(g_val, 0, None)
                return t_val, g_val
            solvers_tried.append((sol, prob.status))
        except Exception as e:
            solvers_tried.append((sol, str(e)))
    
    print("  LP solve failed. Tried: %s" % [(s, st) for s, st in solvers_tried])
    return None, None

def line_search_mixed(f, g, c_f, c_fg, c_g, S, n):
    """Find best alpha to mix f and g: f_new = (1-alpha)*f + alpha*g."""
    def score_alpha(alpha):
        c = (1-alpha)**2 * c_f + 2*alpha*(1-alpha) * c_fg + alpha**2 * c_g
        return c.max()
    
    res = minimize_scalar(score_alpha, bounds=(0, 1), method='bounded',
                          options={'xatol': 1e-6})
    return res.x

def run_lp_alternating(f_init, S, max_time, rng, verbose=True):
    """Run the LP alternating method."""
    n = len(f_init)
    f = normalize_sum(f_init.copy(), S)
    start = time.monotonic()
    best_score = score_of(f)
    best_f = f.copy()
    it = 0
    
    if verbose:
        print("  LP Alt start: n=%d, score=%.8f" % (n, best_score))

        it += 1
        if time.monotonic() - start > max_time - 30:
            break  # Need time for at least one more LP solve
        
        # Solve LP
        t_opt, g = solve_lp_cross(f, S)
        if g is None:
            break
        
        g = normalize_sum(g, S)
        
        # Line search
        c_f = fft_convolve_self(f)
        c_fg = fft_convolve(f, g)
        c_g = fft_convolve_self(g)
        
        alpha = line_search_mixed(f, g, c_f, c_fg, c_g, S, n)
        
        f_new = (1 - alpha) * f + alpha * g
        f_new = normalize_sum(f_new, S)
        
        new_score = score_of(f_new)
        
        if verbose and it <= 20:
            print("  LP Alt it %d: t_opt=%.6f, alpha=%.4f, score=%.8f" % (it, t_opt, alpha, new_score))
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
        
        # Convergence check

            break
    
    if verbose:
        print("  LP Alt done: %d iters, best score=%.8f" % (it, best_score))
    
    return best_f, best_score

    """Build sparse matrix A where A[i,j] = f[lags[i]-j] if 0 <= lags[i]-j < n."""

    rows = []
    cols = []
    vals = []

        j_hi = min(k, n - 1)

            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(lags), n))
    return A
