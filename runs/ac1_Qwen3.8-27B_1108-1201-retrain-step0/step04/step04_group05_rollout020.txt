def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    # Water-filling
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.sum(sorted_v > (cumsum - S) / np.arange(1, n+1))
    if rho == 0:
        return np.zeros(n)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    # Use sorting-based water-filling
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_fft(a, N):
    """Full self-convolution via FFT. Returns (c[2n-1], A[N])."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def compute_topk_grad(a, A, N, top_idx, weights):
    """Compute gradient of weighted sum of top-k convolution peaks."""
    n = len(a)
    m = np.arange(N, dtype=np.float64)
    cos_sum = np.zeros(N, dtype=np.float64)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = weights[i]
        if w > 1e-15:
            cos_sum += w * np.cos(2.0 * np.pi * m * k / N)
    grad_fft = 2.0 * A * cos_sum
    grad = np.real(ifft(grad_fft))[:n]
    return grad


def spectral_step(a, A, N, top_k, alpha):
    """One spectral shaping step: reduce contribution of top peaks."""
    n = len(a)
    S = a.sum()
    c = np.real(ifft(A * A))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, A
    weights = top_vals / total
    m = np.arange(N, dtype=np.float64)
    cos_eff = np.zeros(N, dtype=np.float64)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_proj = project_simplex(a_new, S)
    # Recompute A for the projected version
    a_pad = np.zeros(N)
    a_pad[:n] = a_proj
    A_proj = fft(a_pad)
    return a_proj, A_proj

    cmax = c.max()
    best_cmax = cmax
    best_a = a.copy()

    it = 0
    last_improve = 0
    top_k = 1

import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_fft(a, N):
    """Full self-convolution via FFT. Returns (c of length 2n-1, A of length N)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def compute_topk_grad(a, A, N, top_idx, weights):
    """Compute gradient of weighted sum of specified convolution peaks."""
    n = len(a)
    m = np.arange(N, dtype=np.float64)
    cos_sum = np.zeros(N, dtype=np.float64)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = weights[i]
        if w > 1e-15:
            cos_sum += w * np.cos(2.0 * np.pi * m * k / N)
    grad_fft = 2.0 * A * cos_sum
    grad = np.real(ifft(grad_fft))[:n]
    return grad


def spectral_step(a, A, N, top_k, alpha):
    """Spectral shaping: reduce spectral content contributing to top peaks."""
    n = len(a)
    S = a.sum()
    c = np.real(ifft(A * A))[:2 * n - 1]
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = np.maximum(c[top_idx], 0.0)
    total = top_vals.sum()
    if total < 1e-30:
        return a, A
    weights = top_vals / total
    m = np.arange(N, dtype=np.float64)
    cos_eff = np.zeros(N, dtype=np.float64)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * k / N)
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_proj = project_simplex(a_new, S)
    a_pad = np.zeros(N)
    a_pad[:n] = a_proj
    A_proj = fft(a_pad)
    return a_proj, A_proj

def make_init_binary_pattern(n, rng):
    """Binary-like pattern based on quadratic residue (Paley-like)."""
    p = 1

    for x in range(p):
        residues.add((x * x) % p)
    a = np.zeros(n)
    for i in range(n):
        a[i] = 1.0 if (i % p) in residues else 0.3
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)

            a_opt, cmax_opt = peak_chase_optimize(a_norm, n, S, N, rem, rng, gamma=2.0)
            update_state(state, a_opt, n, "P1/%s/n%d" % (name, n))

    print("Phase 1 best: %.6f" % state['best'])

    # Fallback if nothing found
    if state['best_a'] is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        state['best_a'] = np.ones(n) * S / n
        state['best_n'] = n
        state['best'] = 2.0
        print("  Using uniform fallback, R=2.0")

    # --- Phase 2: Focused optimization on best (1600s) ---
    print("\n=== Phase 2: Focused peak-chasing ===")
    t2 = time.monotonic()
    n_best = state['best_n']
    S = np.sqrt(2.0 * n_best)
    N = 1 << (2 * n_best - 1).bit_length()

    a_curr = state['best_a'].copy()
    stall = 0
    block_count = 0

    while time.monotonic() < deadline - 400:
        block_time = min(180.0, (deadline - 400) - time.monotonic())
        if block_time < 15:
            break
        block_count += 1

        # Vary gamma: start high (focus on top peak), decrease over time
        gamma_val = max(1.0, 3.0 - 0.3 * block_count)

        a_new, cmax_new = peak_chase_optimize(a_curr, n_best, S, N, block_time, rng, gamma=gamma_val)
        R_new = eval_internal(a_new, n_best, S, N)

        if R_new < state['best'] - 1e-9:
            state['best'] = R_new
            state['best_a'] = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2 blk%d] R=%.6f (gamma=%.1f)" % (block_count, R_new, gamma_val))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                # Perturb to escape
                scale = 0.01 * stall * S / np.sqrt(n_best)
                a_curr = project_simplex(a_curr + rng.normal(0, min(scale, 0.1 * S / np.sqrt(n_best)), n_best), S)
                stall = 0

    # --- Phase 3: Spectral polish (250s) ---
    print("\n=== Phase 3: Spectral polish ===")
    t3 = time.monotonic()
    remaining = deadline - t3
    if remaining > 60:
        a_sp, cmax_sp = spectral_polish(state['best_a'], n_best, S, N, min(250.0, remaining - 20), rng)
        R_sp = eval_internal(a_sp, n_best, S, N)
        if R_sp < state['best']:
            state['best'] = R_sp
            state['best_a'] = a_sp.copy()
            print("  [P3] R=%.6f (spectral)" % R_sp)
        else:
            print("  [P3] No improvement (spectral R=%.6f)" % R_sp)

    # --- Phase 4: Final peak-chase with low gamma (200s) ---
    print("\n=== Phase 4: Final polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 40:
        a_fin, cmax_fin = peak_chase_optimize(state['best_a'], n_best, S, N,
                                               min(200.0, remaining - 10), rng, gamma=1.0)
        R_fin = eval_internal(a_fin, n_best, S, N)
        if R_fin < state['best']:
            state['best'] = R_fin
            state['best_a'] = a_fin.copy()
            print("  [P4] R=%.6f (final)" % R_fin)
        else:
            print("  [P4] No improvement (R=%.6f)" % R_fin)

    # --- Phase 5: Try a different length if time remains ---
    remaining = deadline - time.monotonic()
    if remaining > 120 and n_best != 16384:
        print("\n=== Phase 5: Try n=16384 ===")
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        N2 = 1 << (2 * n2 - 1).bit_length()
        # Initialize from best sequence (upsample via interpolation)
        a_up = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n_best), state['best_a'])
        a_up = safe_normalize(a_up, n2, S2)
        a2, cmax2 = peak_chase_optimize(a_up, n2, S2, N2, min(remaining - 30, 100), rng, gamma=2.0)
        R2 = update_state(state, a2, n2, "P5/n16384")

    # --- Verification ---
    print("\n=== Verification ===")
    final_a = state['best_a']
    if final_a is not None:
        final_a = np.clip(final_a, 0.0, 1000.0)
        if final_a.sum() < 0.01:
            final_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in final_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified R = %.6f" % ver)
        else:
            print("VERIFICATION FAILED - using fallback")
            final_a = np.ones(256, dtype=np.float64)
            ver = evaluate_sequence([float(x) for x in final_a])
            print("Fallback R = %.6f" % ver)
    else:
        final_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in final_a])
        print("Fallback R = %.6f" % ver)

    print("\nFinal: length=%d, internal_best=%.6f" % (len(final_a), state['best']))
    return [float(x) for x in final_a]
