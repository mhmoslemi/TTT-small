def project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u >= (cssv - S) / rho
    if not np.any(cond):
        return np.maximum(v - (v.sum() - S) / n, 0.0)
    rho_max = int(np.max(rho[cond]))
    theta = (cssv[rho_max - 1] - S) / rho_max

    # Nesterov state
    y = a.copy()
    t_nesterov = 1.0

    while time.monotonic() - start < budget_s:
        iter_count += 1

        # Compute convolution at y
        c = fft_convolve_self(y)
        c_max = c.max()

        # Soft-max weights with adaptive temperature
        temp = c_max / max(beta, 1e-10)
        exponent = (c - c_max) / temp
        exponent = np.maximum(exponent, -50.0)  # avoid underflow
        w = np.exp(exponent)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient via FFT
        grad = compute_subgradient_fft(y, w, fsize_grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        typical = S / n
        step = lr * typical / gnorm
        step = min(step, 2.0 * typical)
        step = max(step, 1e-10 * typical)

        # Nesterov update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
        y_new = x_new + ((t_nesterov - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        # Re-project y_new to maintain feasibility (Nesterov can go off-manifold)
        y_new = project_simplex(y_new, S)

        y = y_new
        t_nesterov = t_new

        # Evaluate
        sc = compute_score(y)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        # Adapt parameters
        # Increase beta (sharper focus) slowly
        beta = min(200.0, beta * 1.002)
        # Decrease learning rate
        lr = base_lr / (1.0 + 0.0005 * iter_count)

        # Handle getting stuck
        if stuck > 200:
            if stuck % 100 == 0:
                r = rng.random()
                if r < 0.3:
                    # Spectral clean
                    y = spectral_clean(y, S, clip_frac=0.03, fsize=fsize_conv)
                    stuck = 0
                elif r < 0.6:
                    # Spectral flatten
                    y = spectral_flatten_step(y, S, alpha=0.3, fsize=fsize_conv)
                    stuck = 0
                else:
                    # Noise kick + reset Nesterov
                    noise_scale = 0.005 * typical
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_nesterov = 1.0
                    stuck = 0
                    lr = base_lr * (0.5 + rng.random())

        # Check convergence
        if best_score <= target:
            break

        # Periodic progress report
        if iter_count % 1000 == 0:
            elapsed = time.monotonic() - start
            print("    iter=%d best=%.6f elapsed=%.0fs beta=%.1f stuck=%d" %
                  (iter_count, best_score, elapsed, beta, stuck))

    # Final gentle polish
    remaining = budget_s - (time.monotonic() - start)
    if remaining > 2 and best_score > target:
        t_polish = time.monotonic()
        y = best_a.copy()
        t_nesterov = 1.0
        beta_polish = min(200.0, beta * 2.0)
        while time.monotonic() - t_polish < remaining * 0.5:
            c = fft_convolve_self(y)
            c_max = c.max()
            temp = c_max / max(beta_polish, 1e-10)
            exponent = (c - c_max) / temp
            exponent = np.maximum(exponent, -50.0)
            w = np.exp(exponent)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum
            grad = compute_subgradient_fft(y, w, fsize_grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 0.3 * (S / n) / gnorm
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
            y_new = project_simplex(x_new + ((t_nesterov - 1.0) / t_new) * (x_new - y), S)
            y = y_new
            t_nesterov = t_new
            sc = compute_score(y)
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
            if best_score <= target:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov-accelerated projected subgradient with soft-max lag weighting,
    spectral cleaning, and adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)

    print("Starting optimization with budget %.0fs" % budget_s)
    t_start = time.monotonic()

    # Get provided sequence
    provided_seq = None
    try:
        pseq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if pseq.size >= 512 and pseq.sum() > 0.01:
            provided_seq = pseq
            print("Using provided sequence of length %d, sum=%.2f" % (pseq.size, pseq.sum()))
    except Exception:
        pass

    # Determine lengths to try (prioritize provided length)
    lengths_to_try = []
    if provided_seq is not None:
        lengths_to_try.append(len(provided_seq))
    for L in [8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Total budget (leave 30s for verification)
    total_budget = budget_s - 30.0

    # Allocate: 75% to primary, 25% to secondary
    if len(lengths_to_try) == 1:
        budgets = [total_budget]
    elif len(lengths_to_try) == 2:
        budgets = [total_budget * 0.75, total_budget * 0.25]
    else:
        budgets = [total_budget * 0.5, total_budget * 0.3, total_budget * 0.2]

    global_best = np.inf
    global_a = None

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_start
        remaining = total_budget - elapsed
        if remaining < 15:
            break

        alloc = min(budgets[i], remaining)
        if alloc < 10:
            break

        print("\n=== Optimizing n=%d (budget %.0fs) ===" % (n, alloc))

        pseq_n = None
        if provided_seq is not None and len(provided_seq) == n:
            pseq_n = provided_seq

        a_result, score = optimize_single(n, alloc, rng, pseq_n, target=1.502)

        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  New global best: %.6f at n=%d" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

    # Fallback
    if global_a is None:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.ones(n_fb, dtype=np.float64) * S_fb / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence")

    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * (np.sqrt(2048.0) / 1024)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nVerification: evaluate_sequence = %.6f" % ver)
        global_best = min(global_best, ver)
    else:
        print("WARNING: evaluation returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(16384.0) / 8192)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0

    print("Final: length=%d, score=%.6f, total_time=%.0fs" %
          (len(global_a), global_best, time.monotonic() - t_start))
    return [float(x) for x in global_a]
