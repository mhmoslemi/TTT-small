Strategy: LSE-smoothed gradient descent with adaptive beta, exact line search,

Key innovations over previous approach: (1) proper line search with golden-section

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rc = (cumsum - S) / idx
    valid = rc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv[k]))."""
    n = len(a)
    size = 2 * n - 1
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize2 = 1 << (size + n - 1).bit_length()
    W2 = fft(w, fsize2)
    A2 = fft(a, fsize2)
    h = np.real(ifft(W2 * np.conj(A2)))[:n]
    return 2.0 * h

def normalize_sum(a, n):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:

    return a * (S / s)


def golden_section_line_search(a, direction, S, f0, t_max, n_evals=5):
    """Golden section search for best step size along -direction."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = t_max
    a_pts = [a - lo * direction]
    a_pts.append(a - hi * direction)
    f_lo = f0
    f_hi = 2.0 * len(a) * fft_convolve_self(project_simplex(a_pts[1], S)).max() / S**2
    
    for _ in range(n_evals):
        t1 = lo + phi * (hi - lo)
        t2 = hi - phi * (hi - lo)
        p1 = project_simplex(a - t1 * direction, S)
        f1 = 2.0 * len(a) * fft_convolve_self(p1).max() / S**2
        p2 = project_simplex(a - t2 * direction, S)
        f2 = 2.0 * len(a) * fft_convolve_self(p2).max() / S**2
        if f1 < f2:
            hi = t2
            f_hi = f2
        else:
            lo = t1
            f_lo = f1
    t_best = 0.5 * (lo + hi)
    return t_best


def peak_directed_move(a, c, n, S, rng, n_moves=10):
    """
    Try to reduce the peak by redistributing mass from peak-contributing
    positions to non-contributing positions.
    Returns (improved_a, improved_obj) or None.
    """
    best_obj = 2.0 * n * c.max() / S**2
    best_a = a.copy()
    size = 2 * n - 1
    
    peak_lag = int(np.argmax(c))
    
    # Find positions that contribute most to the peak
    lo = max(0, peak_lag - n + 1)
    hi = min(peak_lag, n - 1)
    i_vals = np.arange(lo, hi + 1)
    contrib = a[i_vals] * a[peak_lag - i_vals]
    
    # Find "expensive" positions (high contribution) and "cheap" positions
    # (positions that contribute little to the peak)
    total_contrib = contrib.sum()
    if total_contrib < 1e-15:
        return None
    
    # Rank positions by contribution to peak
    contrib_rank = np.argsort(contrib)[::-1]
    
    # For each n_moves, try moving mass from a high-contrib position to a random low-contrib position
    for _ in range(n_moves):
        # Pick a source (high contributor)
        src_local = contrib_rank[rng.integers(0, min(20, len(contrib_rank)))]
        src = lo + src_local
        
        # Pick a destination (low contributor or non-contributor)
        # Non-contributing positions: those outside [lo, hi] or with a[peak_lag - j] = 0
        non_contrib = np.where((np.arange(n) < lo) | (np.arange(n) > hi))[0]
        if len(non_contrib) > 0:
            dst = non_contrib[rng.integers(0, len(non_contrib))]
        else:
            dst_lo = contrib_rank[-min(20, len(contrib_rank)):]
            dst = lo + dst_lo[rng.integers(0, len(dst_lo))]
        
        if src == dst:
            continue
        
        # Try several step sizes
        delta = 0.001 * S / np.sqrt(n)
        a_try = a.copy()
        a_try[src] -= delta
        a_try[dst] += delta
        if a_try[src] < 0 or a_try[dst] < 0:
            continue
        if a_try.sum() != S:
            a_try = project_simplex(a_try, S)
        
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
    
    if best_obj < 2.0 * n * c.max() / S**2:
        return best_a, best_obj
    return None

    elif strategy == 1:

        mask[n - f_hi:n - f_lo] = True

        A[f_lo:f_hi] += noise

            A[n - k] = np.conj(A[k])

        a_try = np.maximum(a_try, 0)
        return project_simplex(a_try, S)
    
    a_try = np.real(ifft(A))
    a_try = np.maximum(a_try, 0)
    return project_simplex(a_try, S)


def generate_initializations(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Standard windows
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(np.pi * tt) ** 0.5)
    
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * tt) + 0.15 * np.sin(6 * np.pi * tt + 0.3))
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * 3 * tt + 1.0) + 0.2 * np.sin(2 * np.pi * 7 * tt))
    
    # Edge-heavy
    inits.append(1.0 + 2.0 * (tt ** 4 + (1 - tt) ** 4))
    inits.append(1.0 + 3.0 * (tt ** 8 + (1 - tt) ** 8))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    
    # Gaussian centered
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.1))
    
    # Two bumps
    inits.append(np.exp(-20.0 * (tt - 0.25) ** 2) + np.exp(-20.0 * (tt - 0.75) ** 2))
    
    if given is not None and len(given) == n:
        inits.append(given.copy())
    
    return [np.maximum(x, 0.01) for x in inits]


def run_optimization(a_init, n, S, max_time, rng, beta_start=10.0, beta_end=200.0):
    """
    Main optimization loop combining LSE gradient, top-K subgradient,
    line search, and peak-directed moves.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / S**2
    best_obj = cur_obj
    best_a = a.copy()
    
    it = 0
    last_improve = 0
    beta = beta_start
    step = 1.0
    consecutive_fail = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time
        
        # Adaptive beta: start low, increase over time, but keep some exploration
        if it < 100:
            beta = beta_start
        else:
            beta_frac = min(1.0, (it - 100) / max(1, max_time * 10))
            beta = beta_start + (beta_end - beta_start) * beta_frac
            # Occasionally reset beta to explore
            if it % 500 == 0 and rng.random() < 0.1:
                beta = beta_start * 2
        
        # Choose gradient type
        # Mix: mostly LSE, sometimes top-K for exactness
        use_topk = False
        if it % 20 == 0:
            use_topk = True
            k_top = min(8, 1 + it // 200)
        elif it % 50 == 0:
            use_topk = True
            k_top = min(16, 2 + it // 100)
        
        if use_topk:
            grad = topk_subgradient(a, c, k_top)
            gnorm = np.linalg.norm(grad)
        else:
            grad = lse_gradient(a, c, beta)
            gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Step size: scale by 1/gnorm, with adaptive factor
        base_step = S / (n * 0.01 + gnorm)
        t_max = min(5.0 * base_step, 2.0 * S / (n * gnorm + 1e-15))
        t_max = max(t_max, 1e-6)
        
        # Direction: negative gradient, projected
        direction = grad / gnorm
        
        # Try line search (limited evaluations)
        n_evals = 4 if it < 500 else 6
        t_best = golden_section_line_search(a, direction, S, cur_obj, t_max, n_evals)
        
        # Apply step
        a_new = project_simplex(a - t_best * direction, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new < cur_obj - 1e-12:
            a = a_new
            c = c_new
            cur_obj = obj_new
            consecutive_fail = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
        else:
            # Try a smaller step
            t_small = t_best * 0.1
            a_small = project_simplex(a - t_small * direction, S)
            c_small = fft_convolve_self(a_small)
            obj_small = 2.0 * n * c_small.max() / S**2
            if obj_small < cur_obj:
                a = a_small
                c = c_small
                cur_obj = obj_small
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    last_improve = it
                consecutive_fail = 0
            else:
                consecutive_fail += 1
        
        # Periodic peak-directed move
        if it % 50 == 0 and consecutive_fail > 5:
            result = peak_directed_move(a, c, n, S, rng, n_moves=5)
            if result is not None:
                a_cand, obj_cand = result
                if obj_cand < cur_obj:
                    a = a_cand
                    c = fft_convolve_self(a)
                    cur_obj = obj_cand
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        last_improve = it
                    consecutive_fail = 0
        
        # Stuck detection: apply perturbation
        if consecutive_fail > 30:
            pert_scale = 1.0 + 0.5 * (consecutive_fail // 30)
            a = structured_perturbation(a, n, S, rng, scale=pert_scale)
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / S**2
            consecutive_fail = 0
            step = 1.0
        
        if it % 500 == 0:
            print("    it=%d, obj=%.6f, best=%.6f, beta=%.1f, fail=%d" % (
                it, cur_obj, best_obj, beta, consecutive_fail))
    
    return best_a, best_obj


def run_mirror_descent(a_init, n, S, max_time, rng):
    """Mirror (multiplicative) descent as an alternative optimization path."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / S**2
    best_obj = cur_obj
    best_a = a.copy()
    
    it = 0
    eta = 0.5
    consecutive_fail = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Use LSE gradient
        beta = min(100.0, 5.0 + 0.05 * it)
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Mirror descent step: a_new[j] = a[j] * exp(-eta * grad[j] / gnorm)
        log_a = np.log(np.maximum(a, 1e-300))
        new_log = log_a - eta * grad / (gnorm + 1e-15)
        new_log = np.clip(new_log, -500, 500)
        a_new = np.exp(new_log - new_log.max())
        a_new = a_new * (S / a_new.sum())
        
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2
        
        if obj_new < cur_obj:
            a = a_new
            c = c_new
            cur_obj = obj_new
            eta = min(eta * 1.05, 5.0)
            consecutive_fail = 0
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            consecutive_fail += 1
            eta *= 0.8
            if consecutive_fail > 20:
                # Reset with perturbation
                a = structured_perturbation(a, n, S, rng, scale=1.0)
                c = fft_convolve_self(a)
                cur_obj = 2.0 * n * c.max() / S**2
                eta = 0.5
                consecutive_fail = 0
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Main length: use given length if reasonable, else 8192
    if given is not None and 2048 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192
    
    n = main_len
    S = np.sqrt(2.0 * n)
    
    print("Starting: n=%d, budget=%.0fs, given=%s" % (n, budget_s, "yes" if given is not None else "no"))
    t0 = time.monotonic()
    
    # Phase 1: Quick multi-start exploration (30% of time)
    phase1_time = (deadline - time.monotonic()) * 0.30
    print("=== Phase 1: Multi-start exploration (%.0fs) ===" % phase1_time)
    
    inits = generate_initializations(n, rng, given)
    n_inits = len(inits)
    per_init_time = phase1_time / max(1, n_inits)
    
    for i in range(n_inits):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.65:
            break
        if time.monotonic() - t0 > budget_s - 60:
            break
        
        init = inits[i]
        arr_n = len(init)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        
        # Alternate between methods
        if i % 3 == 0:
            a_opt, score = run_optimization(init, arr_n, S_i, per_init_time, rng)
        elif i % 3 == 1:
            a_opt, score = run_mirror_descent(init, arr_n, S_i, per_init_time, rng)
        else:
            a_opt, score = run_optimization(init, arr_n, S_i, per_init_time, rng,
                                           beta_start=5.0, beta_end=80.0)
        
        a_norm = normalize_sum(a_opt, arr_n)
        real_score = compute_objective(a_norm)
        
        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  init %d (n=%d): %.6f" % (i, arr_n, global_best))
    
    print("After Phase 1: best=%.6f" % global_best)
    
    # Phase 2: Deep optimization on best (50% of time)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.55
    print("=== Phase 2: Deep optimization (%.0fs) ===" % phase2_time)
    
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        
        # Run main optimizer with more time
        a_opt, score = run_optimization(global_a, n_best, S_best, phase2_time * 0.6, rng,
                                       beta_start=20.0, beta_end=300.0)
        a_norm = normalize_sum(a_opt, n_best)
        score = compute_objective(a_norm)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            print("  After deep opt: %.6f" % global_best)
        
        # Run mirror descent
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a_opt2, score2 = run_mirror_descent(global_a, n_best, S_best, remaining2 * 0.4, rng)
            a_norm2 = normalize_sum(a_opt2, n_best)
            score2 = compute_objective(a_norm2)
            if score2 < global_best:
                global_best = score2
                global_a = a_norm2.copy()
                print("  After mirror: %.6f" % global_best)
    
    # Phase 3: Final polishing with peak-directed moves (20% of time)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.5
    print("=== Phase 3: Final polishing (%.0fs) ===" % phase3_time)
    
    if global_a is not None and phase3_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_polish = global_a.copy()
        c_polish = fft_convolve_self(a_polish)
        cur_obj = 2.0 * n_best * c_polish.max() / S_best**2
        best_polish = cur_obj
        best_polish_a = a_polish.copy()
        
        polish_start = time.monotonic()
        it = 0
        while time.monotonic() - polish_start < phase3_time:
            it += 1
            elapsed = time.monotonic() - polish_start
            
            # Alternate between gradient step and peak-directed move
            if it % 3 == 0:
                # Peak-directed move
                result = peak_directed_move(a_polish, c_polish, n_best, S_best, rng, n_moves=20)
                if result is not None:
                    a_cand, obj_cand = result
                    if obj_cand < cur_obj:
                        a_polish = a_cand
                        c_polish = fft_convolve_self(a_polish)
                        cur_obj = obj_cand
                        if cur_obj < best_polish:
                            best_polish = cur_obj
                            best_polish_a = a_polish.copy()
            else:
                # Gradient step with line search
                beta = min(500.0, 50.0 + it * 0.5)
                grad = lse_gradient(a_polish, c_polish, beta)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    break
                
                direction = grad / gnorm
                t_max = S_best / (n_best * 0.001 + gnorm)
                t_best = golden_section_line_search(a_polish, direction, S_best, cur_obj, t_max, 4)
                
                a_new = project_simplex(a_polish - t_best * direction, S_best)
                c_new = fft_convolve_self(a_new)
                obj_new = 2.0 * n_best * c_new.max() / S_best**2
                
                if obj_new < cur_obj:
                    a_polish = a_new
                    c_polish = c_new
                    cur_obj = obj_new
                    if cur_obj < best_polish:
                        best_polish = cur_obj
                        best_polish_a = a_polish.copy()
                elif it % 50 == 0:
                    # Small perturbation
                    a_polish = structured_perturbation(a_polish, n_best, S_best, rng, scale=0.5)
                    c_polish = fft_convolve_self(a_polish)
                    cur_obj = 2.0 * n_best * c_polish.max() / S_best**2
            
            if it % 2000 == 0:
                print("    polish it=%d, obj=%.6f, best=%.6f" % (it, cur_obj, best_polish))
        
        if best_polish < global_best:
            global_best = best_polish
            global_a = best_polish_a.copy()
            print("  After polishing: %.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("Returning length %d, obj=%.6f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
