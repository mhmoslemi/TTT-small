import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smooth_max(c, beta):
    """Smooth approx of max(c) via log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    return cm + np.log(e.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lag(a, k):
    """Exact subgradient of c[k] = sum_j a[j]*a[k-j] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    # c[k] = sum_j a[j] * a[k-j], so d c[k]/d a[j] = a[k-j] + a[j-k] (with zero padding)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        g[j] = val
    return g


def subgradient_top_k(a, c, k_count):
    """Weighted subgradient from top-k lags, weighted by excess over mean."""
    n = len(a)
    k_count = min(k_count, len(c))
    idx = np.argsort(c)[::-1][:k_count]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    # Compute weighted subgradient using FFT
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a)
    s = a.sum()
    if s < 0.01:

    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_to_sum(a, S):

    if s < 1e-15:

    return a * (S / s)

            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Conservative step with backtracking
        step = 0.5 / (beta * 0.1)
        for _ in range(15):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max - 1e-14 or step < 1e-14:
                break
            step *= 0.4

        c_new = fft_convolve_self(x_new)
        if c_new.max() <= cur_max + 1e-12:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_k = t_new

        # Very occasional perturbation
        if it - last_improve > 5000:
            noise_scale = 0.0005 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_k = 1.0
            last_improve = it

    return best_a, best_c


def run_direct_peak_reduction(a_init, S, max_time, rng):
    """Direct peak reduction: identify dominant pairs at peak lag and reduce them."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best_c = fft_convolve_self(a).max()
    best_a = a.copy()

    y = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_c - 1e-13:
            best_c = cur_max
            best_a = y.copy()

        # Find peak lag
        k_star = int(np.argmax(c))

        # Find dominant pairs contributing to c[k_star]
        # c[k] = sum_j a[j] * a[k-j] for valid j
        j_range = range(max(0, k_star - n + 1), min(n, k_star + 1))
        products = []
        for j in j_range:
            if 0 <= k_star - j < n:
                products.append((j, k_star - j, y[j] * y[k_star - j]))

        if not products:
            break

        products.sort(key=lambda x: -x[2])

        # Reduce the top contributing pair
        if products:
            j1, j2, prod_val = products[0]
            if prod_val < 1e-15:
                break

            # Reduce the larger of y[j1], y[j2]
            if y[j1] >= y[j2]:
                source = j1
                target_lag_idx = j2
            else:
                source = j2
                target_lag_idx = j1

            # Amount to reduce: proportional to the excess
            excess = cur_max - c.mean()
            if excess <= 0:
                break

            # Reduce by a fraction of the source value
            frac = 0.01 * (1.0 + progress)
            delta = frac * y[source]

            # Find target: a position that minimally increases the peak
            # Heuristic: spread to a random position far from the peak-contributing positions
            target_source = rng.integers(0, n)
            while target_source == source or abs(target_source - target_lag_idx) < n // 4:
                target_source = rng.integers(0, n)

            y[source] -= delta
            y[target_source] += delta
            y = np.maximum(y, 0.0)
            y = normalize_to_sum(y, S)

        # Every 1000 iterations, do a gradient step for smoothing
        if it % 1000 == 0:
            grad = subgradient_top_k(y, c, k_count=3)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = 0.5 / (n * (1.0 + 0.01 * it))
                y = project_simplex(y - step * grad / gnorm, S)

        # Periodic noise
        if it % 5000 == 0 and progress < 0.9:
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best_c:
        best_c = c_final.max()
        best_a = a.copy()

    return best_a, best_c


def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.3))
    inits.append(rng.random(n) + 0.1)

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(convolve(a,a))/sum(a)^2.
    Multi-phase: LSE gradient -> cyclic peak shave -> fine polish -> direct reduction.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    n = 8192
    S = np.sqrt(2.0 * n)

    print("Starting optimization, budget=%.0fs, n=%d" % (budget_s, n))

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Prepare initializations
    inits = []
    if given is not None:
        if len(given) == n:
            inits.append(given.copy())
        elif len(given) > n:
            # Downsample by averaging
            ratio = len(given) // n
            if ratio >= 1:
                small = given[:n * ratio].reshape(n, ratio).mean(axis=1)
                inits.append(small)
            else:
                inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(given)), given))
        else:
            inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(given)), given))

    gen_inits = generate_inits(n, rng, base_seq=None)
    inits.extend(gen_inits[:4])

    # Phase 1: LSE gradient descent on best init (60% of time)
    time_p1 = (deadline - time.monotonic()) * 0.55
    print("\n=== Phase 1: LSE Gradient Descent (%.0fs) ===" % time_p1)

    # First, quickly find the best starting point
    best_init = None
    best_init_c = np.inf
    time_per_init = max(5.0, time_p1 * 0.3 / len(inits))

    for i, init in enumerate(inits):
        if time.monotonic() > deadline - time_p1 * 0.5:
            break
        t_alloc = min(time_per_init, deadline - time.monotonic() - time_p1 * 0.4)
        if t_alloc < 5:
            break

        a_tmp, c_tmp = run_lse_phase(init, S, t_alloc, rng,
                                      beta_start=2.0, beta_end=50.0,
                                      name="init_%d" % i)
        obj_tmp = 2.0 * n * c_tmp / (a_tmp.sum() ** 2)
        print("  Init %d: obj=%.6f, max_c=%.6f" % (i, obj_tmp, c_tmp))
        if c_tmp < best_init_c:
            best_init_c = c_tmp
            best_init = a_tmp.copy()

    if best_init is None:
        best_init = inits[0].copy()
        best_init_c = fft_convolve_self(best_init).max()

    # Main LSE phase on best init
    time_main_lse = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.3
    time_main_lse = max(30.0, min(time_main_lse, (deadline - time.monotonic()) * 0.4))
    print("  Main LSE phase: %.0fs" % time_main_lse)

    a_p1, c_p1 = run_lse_phase(best_init, S, time_main_lse, rng,
                                beta_start=10.0, beta_end=1000.0,
                                name="main_lse")
    obj_p1 = 2.0 * n * c_p1 / (a_p1.sum() ** 2)
    print("  Phase 1 result: obj=%.6f" % obj_p1)

    if c_p1 < best_init_c:
        best_init = a_p1.copy()
        best_init_c = c_p1
    global_best = 2.0 * n * best_init_c / (best_init.sum() ** 2)
    global_a = best_init.copy()
    print("  Current global best: obj=%.6f" % global_best)

    # Phase 2: Cyclic peak shaving (25% of remaining time)
    remaining = deadline - time.monotonic()
    time_p2 = remaining * 0.4
    if time_p2 > 30:
        print("\n=== Phase 2: Cyclic Peak Shaving (%.0fs) ===" % time_p2)
        a_p2, c_p2 = run_cyclic_peak_shave(global_a, S, time_p2, rng, k_count=5)
        obj_p2 = 2.0 * n * c_p2 / (a_p2.sum() ** 2)
        print("  Phase 2 result: obj=%.6f" % obj_p2)
        if c_p2 < best_init_c:
            best_init_c = c_p2
            global_a = a_p2.copy()
            global_best = obj_p2
            print("  New global best: obj=%.6f" % global_best)

    # Phase 3: Fine polishing (15% of remaining time)
    remaining = deadline - time.monotonic()
    time_p3 = remaining * 0.3
    if time_p3 > 20:
        print("\n=== Phase 3: Fine Polishing (%.0fs) ===" % time_p3)
        a_p3, c_p3 = run_fine_polish(global_a, S, time_p3, rng)
        obj_p3 = 2.0 * n * c_p3 / (a_p3.sum() ** 2)
        print("  Phase 3 result: obj=%.6f" % obj_p3)
        if c_p3 < best_init_c:
            best_init_c = c_p3
            global_a = a_p3.copy()
            global_best = obj_p3
            print("  New global best: obj=%.6f" % global_best)

    # Phase 4: Direct peak reduction (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 4: Direct Peak Reduction (%.0fs) ===" % remaining)
        a_p4, c_p4 = run_direct_peak_reduction(global_a, S, remaining, rng)
        obj_p4 = 2.0 * n * c_p4 / (a_p4.sum() ** 2)
        print("  Phase 4 result: obj=%.6f" % obj_p4)
        if c_p4 < best_init_c:
            best_init_c = c_p4
            global_a = a_p4.copy()
            global_best = obj_p4
            print("  New global best: obj=%.6f" % global_best)

    # Final: one more LSE polish with whatever time is left
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        print("\n=== Final Polish (%.0fs) ===" % remaining)
        a_fin, c_fin = run_lse_phase(global_a, S, remaining, rng,
                                      beta_start=200.0, beta_end=5000.0,
                                      name="final")
        obj_fin = 2.0 * n * c_fin / (a_fin.sum() ** 2)
        print("  Final result: obj=%.6f" % obj_fin)
        if c_fin < best_init_c:
            best_init_c = c_fin
            global_a = a_fin.copy()
            global_best = obj_fin
            print("  New global best: obj=%.6f" % global_best)

    # Verify and return
    if global_a is None:
        global_a = np.ones(n) * S / n

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification via evaluate_sequence: %.6f" % ver)

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(8192) * (S / 8192)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver)

    print("Returning sequence of length %d, final obj=%.6f" % (len(global_a), ver))
    return final_list
