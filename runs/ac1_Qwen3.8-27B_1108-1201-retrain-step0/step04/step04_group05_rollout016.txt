def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

"""

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_grad_topk(a, N, top_idx, weights):
    """
    Efficient gradient of sum_k w_k * c[k] where c = conv(a,a).
    Uses the identity: grad = ifft(A_fft * 2*Re(W_fft)) where A=fft(a_pad), W=fft(w).
    Only 2 FFTs + 1 IFFT.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A_fft = fft(a_pad)

    w = np.zeros(N)
    for i, k in enumerate(top_idx):
        w[int(k)] = weights[i]
    W_fft = fft(w)

    grad_full = ifft(A_fft * 2.0 * np.real(W_fft))
    return grad_full.real[:n]


def spectrum_flatten(a, N, alpha=0.3, target='median'):
    """
    Frequency-domain spectrum flattening (water-filling).
    Reduces spectral peaks and boosts valleys to flatten |FFT(a)|^2.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2

    # Only consider non-DC, non-Nyquist bins
    mask = np.ones(N, dtype=bool)
    mask[0] = False
    if N % 2 == 0:
        mask[N // 2] = False

    P_work = P.copy()
    P_vals = P[mask]
    if target == 'median':
        P_target = np.median(P_vals)
    else:
        P_target = np.mean(P_vals)

    # Water-filling: reduce peaks, boost valleys
    excess = np.maximum(P - P_target, 0)
    deficit = np.maximum(P_target - P, 0)
    total_excess = excess[mask].sum()
    total_deficit = deficit[mask].sum()

    if total_excess < 1e-30:
        return a

    # Scale factor for redistribution
    scale = min(1.0, alpha * total_excess / (total_deficit + 1e-30))

    P_new = P.copy()
    P_new[mask] = P[mask] - scale * excess[mask]
    P_new = np.maximum(P_new, 0.0)

    # Reconstruct with original phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]

    # Ensure real symmetry (numerical cleanup)
    S = a.sum()
    a_new = project_simplex(a_new, S)
    return a_new

        y = np.maximum(y, 0.0)

        if S_y > 1e-15:

        t = t_new

def population_search(a_init, n, S, N, max_time, rng, pop_size=4,
                      K_init=1, K_max=256):
    """
    Population-based search: maintain a small population, create variants
    via crossover and mutation, evaluate and select.
    """
    start_time = time.monotonic()
    population = [project_simplex(a_init.copy(), S)]
    scores = []
    for i in range(pop_size - 1):
        # Create variant via mutation
        variant = a_init + rng.normal(0, 0.01 * S / np.sqrt(n), n)
        population.append(project_simplex(variant, S))

    # Initial evaluation
    for i in range(pop_size):
        c = compute_conv(population[i], N)
        scores.append(c.max())

    best_idx = int(np.argmin(scores))
    best_score = scores[best_idx]
    best_a = population[best_idx].copy()

    gen = 0
    while time.monotonic() - start_time < max_time:
        gen += 1
        # Create new candidate from best via crossover + mutation
        parent = population[best_idx]
        other_idx = rng.integers(0, pop_size)
        if other_idx == best_idx:
            other_idx = (other_idx + 1) % pop_size
        other = population[other_idx]

        # Crossover: mix with random ratio
        ratio = rng.uniform(0.3, 0.7)
        child = ratio * parent + (1 - ratio) * other
        # Mutation
        mut_scale = 0.008 * S / np.sqrt(n) * (1.0 + 0.5 * gen / max(1, max_time / 10))
        child = child + rng.normal(0, mut_scale, n)
        child = project_simplex(child, S)

        # Evaluate child
        c = compute_conv(child, N)
        child_score = c.max()

        # Also try spectrum flattening on best
        if gen % 5 == 0:
            flat = spectrum_flatten(population[best_idx], N, alpha=0.15 + 0.2 * min(1.0, gen / 50.0))
            c_flat = compute_conv(flat, N)
            flat_score = c_flat.max()
            if flat_score < best_score:
                best_score = flat_score
                best_a = flat.copy()
                population[best_idx] = flat.copy()
                scores[best_idx] = flat_score
            # Replace worst with flat if better
            worst_idx = int(np.argmax(scores))
            if flat_score < scores[worst_idx]:
                population[worst_idx] = flat.copy()
                scores[worst_idx] = flat_score
                if flat_score < best_score:
                    best_score = flat_score
                    best_a = flat.copy()
                    best_idx = worst_idx

        # Select: replace worst if child is better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child.copy()
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_a = child.copy()
                best_idx = worst_idx

        # Occasionally do a short Nesterov run on best
        if gen % 10 == 0:
            rem = max_time - (time.monotonic() - start_time)
            if rem > 2:
                block_time = min(2.0, rem * 0.1)
                a_n, sc_n = nesterov_block(population[best_idx], n, S, N,
                                           block_time, rng,
                                           K_init=max(50, K_max // 4),
                                           K_max=K_max,
                                           mu_init=0.95, mu_end=0.999)
                if sc_n < best_score:
                    best_score = sc_n
                    best_a = a_n.copy()
                    population[best_idx] = a_n.copy()
                    scores[best_idx] = sc_n

    return best_a, best_score


def fine_polish(a_init, n, S, N, max_time, rng):
    """
    Final polishing: very small steps targeting individual peaks.
    Uses a hybrid of gradient steps and coordinate updates.
    """
    a = project_simplex(a_init.copy(), S)
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()

    it = 0
    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        # Find the single dominant peak
        k_star = int(np.argmax(c))
        # Also find second peak if close
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.95 * c_sorted[0]:
            K = 3
        else:
            K = 1
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]
        total = top_vals.sum()
        weights = top_vals / total

        grad = compute_grad_topk(a, N, top_idx, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Very small step
        step = 0.3 * S / (n * (1.0 + 0.01 * it))
        a_new = project_simplex(a - step * grad / gnorm, S)

        # Check improvement
        c_new = compute_conv(a_new, N)
        if c_new.max() <= cmax + 1e-12:
            a = a_new
        else:
            # Try half step
            a_half = project_simplex(a - 0.5 * step * grad / gnorm, S)
            c_half = compute_conv(a_half, N)
            if c_half.max() <= cmax + 1e-12:
                a = a_half
            else:
                # Skip this step
                break

    return best_a, best_cmax


def make_init_from_given(n, rng):
    """Try to use the given height_sequence_1."""
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            return np.clip(given, 0, 1000).astype(np.float64)
        elif given.size > n:
            return np.clip(given[:n], 0, 1000).astype(np.float64)
        elif given.size > 0:
            # Resample
            from numpy import interp
            x_old = np.linspace(0, 1, given.size)
            x_new = np.linspace(0, 1, n)
            return np.clip(interp(x_new, x_old, given), 0, 1000)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov momentum peak-tracking as primary optimizer,
    frequency-domain spectrum flattening for global moves,
    population-based search for exploration, and fine polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine length
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 16384

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # Initialize
    a_init = make_init_from_given(n, rng)
    if a_init is None:
        # Fourier-based random init
        P = rng.exponential(1.0, N)
        P[0] = float(S)
        P = np.maximum(P, 1e-6)
        phase = rng.uniform(0, 2 * np.pi, N)
        A = np.sqrt(P) * np.exp(1j * phase)
        a_init = np.abs(np.real(ifft(A)))[:n] + 0.01
    a_init = project_simplex(a_init, S)

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = project_simplex(a_candidate, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            if label:
                print("  [%s] cmax=%.6f -> obj=%.6f" % (label, sc, 2.0 * n * sc / S**2))
        return sc

    # Evaluate initial
    print("Initial cmax: %.6f" % compute_conv(a_init, N).max())

    # === Phase 1: Nesterov momentum (main optimizer) ===
    print("=== Phase 1: Nesterov momentum (adaptive K) ===")
    t1 = time.monotonic()
    phase1_budget = min(900.0, (deadline - t1) * 0.4)

    # Split into multiple Nesterov runs with increasing K
    n_blocks = 4
    per_block = phase1_budget / n_blocks
    a_curr = a_init.copy()
    for bi in range(n_blocks):
        rem = max(1.0, (t1 + phase1_budget) - time.monotonic())
        block_time = min(per_block, rem)
        K_start = [1, 5, 32, 128][bi]
        K_end = [16, 64, 128, 256][bi]
        print("  Block %d: K=%d->%d, time=%.0fs" % (bi + 1, K_start, K_end, block_time))

        a_new, sc_new = nesterov_block(a_curr, n, S, N, block_time, rng,
                                       K_init=K_start, K_max=K_end,
                                       mu_init=0.9, mu_end=0.999)
        update_best(a_new, "nesterov_%d" % (bi + 1))
        a_curr = global_a.copy() if global_a is not None else a_new.copy()

    print("Phase 1 best: %.6f" % global_best)

    # === Phase 2: Population-based exploration ===
    print("=== Phase 2: Population search ===")
    t2 = time.monotonic()
    phase2_budget = min(400.0, (deadline - t2) * 0.2)

    a_pop, sc_pop = population_search(global_a, n, S, N, phase2_budget, rng,
                                      pop_size=4, K_init=50, K_max=256)
    update_best(a_pop, "population")

    # === Phase 3: More Nesterov with fine parameters ===
    print("=== Phase 3: Refined Nesterov ===")
    t3 = time.monotonic()
    phase3_budget = min(500.0, (deadline - t3) * 0.25)

    a_ref, sc_ref = nesterov_block(global_a, n, S, N, phase3_budget, rng,
                                    K_init=128, K_max=512,
                                    mu_init=0.95, mu_end=0.9999)
    update_best(a_ref, "refined_nesterov")

    # === Phase 4: Spectrum flattening refinement ===
    print("=== Phase 4: Spectrum flattening ===")
    t4 = time.monotonic()
    phase4_budget = min(200.0, (deadline - t4) * 0.15)

    a_curr = global_a.copy()
    it_count = 0
    while time.monotonic() - t4 < phase4_budget:
        it_count += 1
        progress = min(1.0, it_count / 1000.0)
        alpha = 0.1 + 0.3 * (1.0 - progress)  # Start aggressive, end gentle
        a_flat = spectrum_flatten(a_curr, N, alpha=alpha)
        c_flat = compute_conv(a_flat, N)
        if c_flat.max() < global_best:
            global_best = c_flat.max()
            global_a = a_flat.copy()
            a_curr = a_flat.copy()
            if it_count % 100 == 0:
                print("  [flatten_%d] %.6f" % (it_count, global_best))
        # Occasionally interleave with Nesterov
        if it_count % 50 == 0:
            a_n, sc_n = nesterov_block(a_curr, n, S, N, 2.0, rng,
                                        K_init=64, K_max=128,
                                        mu_init=0.95, mu_end=0.999)
            if sc_n < global_best:
                global_best = sc_n
                global_a = a_n.copy()
                a_curr = a_n.copy()

    # === Phase 5: Fine polishing ===
    print("=== Phase 5: Fine polishing ===")
    t5 = time.monotonic()
    remaining = deadline - t5
    if remaining > 10:
        a_pol, sc_pol = fine_polish(global_a, n, S, N, min(remaining - 5, 300.0), rng)
        update_best(a_pol, "polish")

    # === Phase 6: Final Nesterov squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fin, sc_fin = nesterov_block(global_a, n, S, N, min(remaining - 3, 120.0), rng,
                                        K_init=256, K_max=512,
                                        mu_init=0.98, mu_end=0.9999,
                                        step_init=0.5 * S / (n * 5.0))
        update_best(a_fin, "final_squeeze")

    # === Verification and output ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
    else:
        print("Verification: inf, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning length %d, final score=%.6f" % (len(final_list), ver))
    return final_list
