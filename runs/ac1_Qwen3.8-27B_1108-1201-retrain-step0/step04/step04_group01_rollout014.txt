"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 using dual ascent.
Key idea: alternate between:
1. Minimizing a^T Q(λ) a over a >= 0, sum(a) = S (a QP solvable by projected gradient)
2. Updating λ to concentrate on peak lags of conv(a,a)
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_Qa(a, lam, fsize):
    """Compute (Q*a)_i = sum_m a_m * lam_{i+m}"""
    A = fft(a, fsize)
    L = fft(lam, fsize)
    C = np.real(ifft(np.conj(A) * L))
    return C[:len(a)]

def lse_smooth_max(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    return cm + np.log(e.sum()) / beta

def lse_gradient(a, c, beta, fsize):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    # Gradient: 2 * sum_k w_k * shift(a, k)
    # = 2 * Q(w) * a where Q(w)_{i,m} = w_{i+m}
    # Use the same compute_Qa trick
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad

def dual_ascent_method(a_init, S, max_time, rng, beta_start=1.0, beta_end=50.0):
    """
    Dual ascent: alternate between QP solve (for a) and lambda update.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Initialize lambda
    lam = np.ones(size) / size
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Pre-compute FFT of lambda (will be updated)
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Beta for lambda update (annealing)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # === QP Solve: minimize a^T Q(λ) a subject to a >= 0, sum(a) = S ===
        # Use projected gradient with a few iterations
        L = fft(lam, fsize)
        
        for inner in range(20):
            A = fft(a, fsize)
            Qa = np.real(ifft(np.conj(A) * L))[:n]
            grad = 2.0 * Qa
            
            # Step size: eigenvalues of Q are in [0, ~1], so grad eigenvalues in [0, ~2]
            eta = 0.4  # safe step size (need eta < 1/L where L <= 2)
            
            a = project_simplex(a - eta * grad, S)
        
        # === Lambda Update ===
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Update lambda: concentrate on peaks
        # Use softmax with increasing beta
        scaled = beta * (c - c.max())
        lam_new = np.exp(scaled)
        lam_new /= lam_new.sum()
        
        # Mix with old lambda for stability
        mix = 0.3
        lam = mix * lam + (1 - mix) * lam_new
        lam /= lam.sum()
    
    return best_a, best

def lse_fista(a_init, S, max_time, rng, beta_start=10.0, beta_end=500.0):
    """LSE-smoothed FISTA."""
    n = len(a_init)
    fsize = 1 << (2*n - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size for FISTA: 1/L where L is Lipschitz constant of gradient
        # L for LSE gradient is approximately 2*beta*Lipschitz(conv)
        # Lipschitz of conv gradient = 2*||a||_2
        L = 2.0 * beta * 2.0 * np.linalg.norm(a)
        step = 1.0 / max(L, 1e-10)
        
        x = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
    
    return best_a, best

**Refined dual ascent algorithm:**

The key insight is that we're solving:
min_a max_k b_k(a)
s.t. a >= 0, sum(a) = S

By strong duality (the problem is convex), the optimal value equals:
max_{λ: sum=1, λ>=0} min_{a: a>=0, sum=S} a^T Q(λ) a

The algorithm:
1. Start with some a and λ
2. Inner loop: solve min_{a: a>=0, sum=S} a^T Q(λ) a using projected gradient
3. Outer loop: update λ based on the current b = conv(a,a)

For the inner loop, I'll use FISTA (since the QP objective is smooth and convex).
The Lipschitz constant of ∇(a^T Q a) = 2*Q*a is 2*λ_max(Q) <= 2 (since Q is PSD with trace <= 1).

Wait, I need to be more careful. Q_{i,j} = λ_{i+j}. The trace of Q is sum_i λ_{2i} <= sum all λ = 1. But the largest eigenvalue could be up to 1 (if all λ mass is at one point). In practice, if λ is spread out, λ_max(Q) is smaller.

For safety, I'll use step size 0.4 for the QP solve (which is < 1/2 = 1/L where L <= 2).

For the outer loop (λ update), I'll use:
- λ = softmax(β * b) where β increases over time
- This concentrates λ on the peak lags
- As β → ∞, λ → uniform over the peak lags

Let me also add a "dual gap" check: the dual gap is max(b) - g(λ) where g(λ) = min_a a^T Q(λ) a. When the gap is small, we're close to optimal.

Actually, computing g(λ) = min_a a^T Q(λ) a exactly requires solving the QP to convergence, which is expensive. I'll just use the current a's value as an upper bound on g(λ).

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 using a combination of:
1. LSE-smoothed FISTA (proven effective)
2. Dual ascent (alternating QP solve and lambda update)
3. Peak shaving with exact subgradients
4. Multi-scale and multi-length search
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_gradient(a, c, beta, fsize):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    mx = scaled.max()
    if mx > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled - scaled.max())
    w = e / e.sum()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad

def run_lse_fista(a_init, S, max_time, rng, beta_start=10.0, beta_end=1000.0):
    n = len(a_init)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        if it - last_improve > 2000 and it > 100:
            # Restart with noise
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
            continue
        
        grad = lse_gradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Lipschitz constant estimate
        L = 4.0 * beta * np.linalg.norm(a)
        step = 1.0 / max(L, 1e-10)
        # Cap step to prevent huge updates
        step = min(step, S / (n * 0.1))
        
        x = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best

def run_dual_ascent(a_init, S, max_time, rng):
    """
    Dual ascent: alternate between
    1. Solving min a^T Q(λ) a subject to a >= 0, sum(a) = S (QP via FISTA)
    2. Updating λ = softmax(β * conv(a,a))
    """
    n = len(a_init)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    lam = np.ones(size) / size
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    beta_start = 2.0
    beta_end = 100.0
    inner_iters = 30
    mix_rate = 0.5  # how much new lambda to mix in
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        
        # === Inner: Solve QP min a^T Q(λ) a s.t. a>=0, sum(a)=S ===
        L_ft = fft(lam, fsize)
        
        for inner in range(inner_iters):
            A_ft = fft(a, fsize)
            Qa = np.real(ifft(np.conj(A_ft) * L_ft))[:n]
            grad = 2.0 * Qa
            
            # Step size: L = 2 * λ_max(Q) <= 2 * max_eig of Q
            # For safety, use 0.4 (since L <= 2, need step < 1/L)
            eta = 0.4
            a = project_simplex(a - eta * grad, S)
        
        # === Outer: Update λ ===
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Softmax update
        scaled = beta * (c - c.max())
        lam_new = np.exp(scaled)
        lam_new /= lam_new.sum()
        
        # Mix
        lam = mix_rate * lam + (1 - mix_rate) * lam_new
        lam /= lam.sum()
        
        # Periodic: also try exact subgradient step
        if it % 100 == 0:
            k_star = np.argmax(c)
            # Subgradient: 2 * shift(a, k_star)
            grad_exact = np.zeros(n)
            # shift(a, k_star)_i = a_{i - k_star} if 0 <= i-k_star < n else 0
            # So grad_exact_i = 2 * a_{i - k_star} for i >= k_star
            if k_star < n:
                grad_exact[k_star:] = 2.0 * a[:n - k_star]
            elif k_star >= n:
                # k_star >= n: shift(a, k_star)_i = a_{i-k_star}
                # For i from k_star to k_star+n-1, a_{i-k_star} = a_0, ..., a_{n-1}
                # But i only goes up to n-1, so if k_star >= n, no valid i
                pass
            # Projected step
            step = S / (n * 5.0)
            a = project_simplex(a - step * grad_exact, S)
        
        # Periodic perturbation
        if it % 500 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * (1 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def run_peak_shave(a_init, S, max_time, rng):
    """Direct subgradient method on the max."""
    n = len(a_init)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    a = a_init.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Use multiple subgradients (top-K lags)
    K = 4
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Increase K over time
        K_cur = min(64, K + int(60 * progress))
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Get top-K lags
        K_use = min(K_cur, size)
        top_idx = np.argsort(c)[::-1][:K_use]
        
        # Weight by excess above mean
        mean_c = c.mean()
        weights = np.maximum(c[top_idx] - mean_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(K_use) / K_use
        else:
            weights = weights / wsum
        
        # Compute weighted subgradient: 2 * sum_k w_k * shift(a, k)
        # shift(a, k)_i = a_{i-k} if 0 <= i-k < n else 0
        # So weighted_subgrad_i = 2 * sum_k w_k * a_{i-k}
        # This is a correlation: 2 * corr(weights_padded, a)[i]
        
        # Build full weight vector
        w_full = np.zeros(size)
        w_full[top_idx] = weights
        
        # Compute correlation: result_i = sum_m w_full[m] * a[m + i]... 
        # Wait, I need: 2 * sum_k w_k * a_{i-k} for each i
        # Let j = i-k, so k = i-j: 2 * sum_j a_j * w_{i-j}
        # This is the cross-correlation of a and w_full
        # corr[i] = sum_j a[j] * w_full[i-j] = ifft(conj(fft(a)) * fft(w_full))[i]
        
        A_ft = fft(a, fsize)
        W_ft = fft(w_full, fsize)
        grad = 2.0 * np.real(ifft(np.conj(A_ft) * W_ft))[:n]
        
        # Step size: decreasing
        step = S / (n * (2.0 + it * 0.001))
        
        a = project_simplex(a - step * grad, S)
        
        # Periodic perturbation
        if it % 2000 == 0 and progress < 0.9:
            noise_scale = 0.001 * S / np.sqrt(n) * (1 - progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def upscale_sequence(a_small, n_large):
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)
