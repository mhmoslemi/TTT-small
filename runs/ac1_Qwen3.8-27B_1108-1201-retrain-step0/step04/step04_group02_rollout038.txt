"""
Minimize 2*n*max(self_convolution)/sum^2 for non-negative sequences.
Uses p-norm smoothed objective with Nesterov acceleration, adaptive step size,
pole-apaper periodic perturbations, and multi-length exploration.
Key innovation: uses polynomial (p-norm) weighting instead of exponential (LSE)
for better numerical stability and different optimization landscape exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def pnorm_weights(c, p):
    """Compute normalized p-norm weights w_k proportional to c_k^(p-1)."""
    c_safe = np.maximum(c, 1e-300)
    if p <= 2:
        w = c_safe / c_safe.sum()
        if w.sum() < 1e-300:
            w = np.ones(len(c)) / len(c)
        return w
    log_w = (p - 1.0) * np.log(c_safe)
    log_w -= log_w.max()
    w = np.exp(log_w)
    total = w.sum()
    if total < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= total
    return w


def gradient_from_weights(w, a):
    """
    Compute grad direction: grad[i] = 2 * sum_j w[i+j] * a[j] for j in [0,n).
    Equivalent to 2 * (w * flip(a))[n-1 : 2n-2].
    """
    n = len(a)
    a_flip = a[::-1].copy()
    # Linear conv of w (len 2n-1) and a_flip (len n) has length 3n-2
    conv_len = (2 * n - 1) + n - 1
    N = 1 << (conv_len - 1).bit_length()
    W = fft(w, N)
    AF = fft(a_flip, N)
    conv = np.real(ifft(W * AF))
    return 2.0 * conv[n - 1:2 * n - 2]


def compute_objective(a):
    """Compute 2n * max_conv / sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (sum_a ** 2)


def topk_gradient(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def normalize_seq(a, n):
    """Normalize to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def structured_perturb(a, rng):
    """Apply a structured perturbation."""
    n = len(a)
    S = a.sum()
    strategy = rng.integers(0, 6)
    t = np.arange(n) / n
    if strategy == 0:
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
    elif strategy == 1:
        mod = 1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.sin(8 * np.pi * t + 0.7)
    elif strategy == 2:
        mod = 1.0 + 0.5 * np.cos(np.pi * t) ** 2
    elif strategy == 3:
        nblocks = rng.integers(4, 20)
        block_mod = rng.uniform(0.4, 1.6, nblocks)
        mod = np.repeat(block_mod, n // nblocks + 1)[:n]
    elif strategy == 4:
        # Frequency domain perturbation
        A = fft(a, 1 << (n - 1).bit_length())
        noise = rng.normal(0, 0.03, len(A)) + 1j * rng.normal(0, 0.03, len(A))
        new_a = np.real(ifft(A * (1 + noise)))
        return project_simplex(np.maximum(new_a, 0), S)
    else:
        # Gaussian noise
        noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
        return project_simplex(a + noise, S)
    return project_simplex(np.maximum(a * mod, 0), S)


def run_pnorm_optimize(a_init, n, S, max_time, rng, p_start=3, p_max=80):
    """
    Main optimizer using p-norm smoothed objective with Nesterov acceleration.
    """
    a = project_simplex(np.maximum(a_init, 0), S)
    y = a.copy()
    t = 1.0
    p = float(p_start)
    eta = S / (n * 0.5)
    eta_max = S / (n * 0.1)
    eta_min = S / (n * 200.0)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    consec_fail = 0
    p_increase_rate = 1.002

    while time.monotonic() - start < max_time:
        it += 1

        # Compute convolution
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S ** 2)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            consec_fail = 0

        # Compute p-norm weights
        w = pnorm_weights(c, p)

        # Compute gradient direction
        grad = gradient_from_weights(w, y)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Step size (adaptive)
        step = eta
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y to maintain sum constraint approximately
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t = t_new

        # Update p (increase for sharper approximation)
        p = min(p * p_increase_rate, p_max)
        # Slow down p increase after a while
        if it > 10000:
            p_increase_rate = max(1.001, p_increase_rate * 0.999)

        # Check improvement via quick eval every 50 iterations
        if it % 50 == 0:
            c_check = fft_convolve_self(a)
            obj_check = 2.0 * n * c_check.max() / (S ** 2)
            if obj_check < best_obj - 1e-12:
                best_obj = obj_check
                best_a = a.copy()
                last_improve = it
                consec_fail = 0
            else:
                consec_fail += 1

            # Adaptive step size
            if it - last_improve < 200:
                eta = min(eta * 1.03, eta_max)
            elif it - last_improve < 1000:
                eta *= 0.9
            else:
                eta *= 0.8

            eta = max(eta, eta_min)

        # Periodic perturbation to escape local minima
        if consec_fail > 50 and it - last_improve > 500:
            noise_scale = min(0.02, 0.005 * (consec_fail / 50.0)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t = 1.0
            eta = S / (n * 0.5)
            consec_fail = 0

        if consec_fail > 100:
            # Larger perturbation
            a = structured_perturb(a, rng)
            y = a.copy()
            t = 1.0
            eta = S / (n * 0.3)
            consec_fail = 0

    # Final check
    final_obj = compute_objective(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj


def run_exact_polish(a_init, n, S, max_time, rng):
    """
    Polish using exact top-1 subgradient with very small steps.
    """
    a = project_simplex(np.maximum(a_init, 0), S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    step = S / (n * 10.0)
    it = 0
    consec_fail = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S ** 2)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            consec_fail = 0

        # Top-1 subgradient
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[i_vals] = 2.0 * a[k_star - i_vals]

        # Try the step
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S ** 2)

        if obj_new < cur_obj:
            a = a_new
            step = min(step * 1.02, S / (n * 0.5))
            consec_fail = 0
        else:
            step *= 0.85
            consec_fail += 1
            if consec_fail > 100:
                # Small perturbation
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                step = S / (n * 5.0)
                consec_fail = 0

    return best_a, best_obj


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.arange(n, dtype=np.float64) / n

    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Sine squared
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    # Two-peak
    inits.append(np.exp(-20 * (tt - 0.25) ** 2) + np.exp(-20 * (tt - 0.75) ** 2) + 0.01)
    # Sawtooth-like
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.01)
    # Random smooth
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * tt + rng.uniform(0, 6))
                 + 0.1 * np.sin(2 * np.pi * rng.uniform(5, 15) * tt) + 0.01)
    # Block pattern
    nblocks = rng.integers(8, 32)
    block_vals = rng.uniform(0.3, 1.5, nblocks)
    mod = np.repeat(block_vals, n // nblocks + 1)[:n]
    inits.append(mod + 0.01)
    # Pure random (smoothed)
    raw = rng.random(n) + 0.1
    inits.append(raw)

    return [np.maximum(x, 0.001) for x in inits]
