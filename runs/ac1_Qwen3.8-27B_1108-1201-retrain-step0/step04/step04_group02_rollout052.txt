    """Compute cross[j] = sum_m a[m] * b[m+j] for j=0..len(a)-1."""
    n = len(a)
    size = n + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    result = np.real(ifft(B * np.conj(A)))
    return result[:n]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def normalize_to_sum(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        n = len(a)
        return np.full(n, S / n)
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    grad = 2.0 * fft_cross_corr_ab(a, w)
    return grad


def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        if lo <= hi:
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def lp_direction_find(a, M, max_iters=80, tol=1e-5):
    """
    Find g >= 0 maximizing sum(g) subject to (a*g)[k] <= M for all k.
    Uses projected gradient on the primal LP with adaptive step size.
    Returns g normalized to have the same sum as a.
    """
    n = len(a)
    target_sum = a.sum()
    g = a.copy().astype(np.float64)
    g = g / max(g.sum(), 1e-300) * target_sum

    best_g = g.copy()
    best_obj = g.sum()

    for it in range(max_iters):
        c = fft_convolve_cross(a, g)
        viol = np.maximum(c - M, 0.0)
        max_viol = viol.max()
        if max_viol < tol * M:
            break

        # Compute pressure: p[j] = sum_k viol[k] * a[k-j]
        p = fft_cross_corr_ab(a, viol)

        pnorm = np.linalg.norm(p)
        if pnorm < 1e-15:
            break

        # Adaptive step size: ensure we don't over-correct
        # The "driving force" is the violation; step proportional to it
        eta = min(0.5 * max_viol / (pnorm + 1e-15), target_sum / (n * pnorm + 1e-15))

        g_new = np.maximum(g - eta * p, 0.0)
        s = g_new.sum()
        if s < 1e-15:
            g_new = g.copy()
        else:
            g_new = g_new * (target_sum / s)

        # Check if this is feasible (approximately)
        c_new = fft_convolve_cross(a, g_new)
        new_max_viol = np.maximum(c_new - M, 0.0).max()

        if new_max_viol <= max_viol + 1e-10:
            g = g_new
        else:
            # Halve the step
            g_new = np.maximum(g - 0.5 * eta * p, 0.0)
            s = g_new.sum()
            if s < 1e-15:
                break
            g_new = g_new * (target_sum / s)
            g = g_new

        if g.sum() > best_obj:
            best_obj = g.sum()
            best_g = g.copy()

    return best_g


def lp_direction_mult(a, M, max_iters=30):
    """
    Alternative LP direction using multiplicative update on g directly.
    More conservative but sometimes more stable.
    """
    n = len(a)
    target_sum = a.sum()
    g = a.copy().astype(np.float64)
    g = g / max(g.sum(), 1e-300) * target_sum

    for it in range(max_iters):
        c = fft_convolve_cross(a, g)
        viol = np.maximum(c - M, 0.0)
        max_viol = viol.max()
        if max_viol < 1e-5 * M:
            break

        # Compute which positions of g are "responsible" for violations
        # g[j] contributes to c[k] through a[k-j]
        # Reduction signal: r[j] = sum_k viol[k] * a[k-j]
        r = fft_cross_corr_ab(a, viol)

        # Multiplicative reduction: g[j] *= exp(-eta * r[j])
        eta = min(0.1, max_viol / (max(r.max(), 1e-15)))
        log_g = np.log(np.maximum(g, 1e-300))
        log_g -= eta * r
        log_g = np.clip(log_g, -700, 700)
        g = np.exp(log_g - log_g.max())
        s = g.sum()
        if s < 1e-300:
            g = a.copy()
            break
        g = g * (target_sum / s)

    return g


def run_lp_sota(a_init, max_time, rng):
    """
    Main LP-based SOTA optimization loop.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    M = c.max()
    it = 0
    stall_count = 0
    last_improve_obj = best_obj

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * M / (S ** 2)

        if cur_obj < best_obj - 1e-10:
            if it % 50 == 0:
                print("    LP SOTA it=%d: obj=%.6f" % (it, best_obj))
            best_obj = cur_obj
            best_a = a.copy()
            last_improve_obj = cur_obj
            stall_count = 0
        else:
            stall_count += 1

        # Find LP direction
        k = 1 if it % 2 == 0 else 2
        if k == 1 and it > 20:
            g = lp_direction_find(a, M, max_iters=60)
        else:
            g = lp_direction_find(a, M, max_iters=40)

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g * (S / gs)

        # Line search: try t = 1, 0.5, 0.25, 0.1
        improved = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_to_sum(a_try, S)
            c_try = fft_convolve_self(a_try)
            M_try = c_try.max()
            obj_try = 2.0 * n * M_try / (S ** 2)

            if obj_try < cur_obj - 1e-12:
                a = a_try
                c = c_try
                M = M_try
                improved = True
                break

        if not improved:
            # Try mirror-style update
            log_a = np.log(np.maximum(a, 1e-300))
            log_g = np.log(np.maximum(g, 1e-300))
            for t in [0.1, 0.05, 0.02]:
                log_mix = (1.0 - t) * log_a + t * log_g
                a_try = np.exp(log_mix - log_mix.max())
                a_try = normalize_to_sum(a_try, S)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n * c_try.max() / (S ** 2)
                if obj_try < cur_obj - 1e-12:
                    a = a_try
                    c = c_try
                    M = c_try.max()
                    improved = True
                    break

        if not improved:
            # Perturbation to escape
            noise_scale = 0.005 * S / np.sqrt(n) * max(0.1, 1.0 - it / 500.0)
            a = normalize_to_sum(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            M = c.max()

    return best_a, best_obj


def run_lse_fista(a_init, max_time, rng):
    """LSE-smoothed gradient with Nesterov acceleration and adaptive step."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)

    start = time.monotonic()
    y = a.copy()
    t_mom = 1.0
    beta = 5.0
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c_y = fft_convolve_self(y)
        grad = lse_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Step size: balance between smoothness (beta) and gradient magnitude
        eta = step_scale * S / (n * gnorm + 1e-15)
        eta = min(eta, 2.0 / (beta + 1.0))

        x_new = project_simplex(y - eta * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic re-normalization of y to prevent drift
        if it % 100 == 0:
            y = normalize_to_sum(y, S)

        c_a = fft_convolve_self(a)
        cur_obj = 2.0 * n * c_a.max() / (S ** 2)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            step_scale = min(step_scale * 1.02, 5.0)
        else:
            step_scale *= 0.95

        # Slowly increase beta for sharper peak targeting
        beta = min(200.0, 5.0 + 0.01 * it)

        # Periodic perturbation
        if it - last_improve > 2000 and it % 500 == 0:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_a, best_obj


def run_adaptive_subgrad(a_init, max_time, rng, method="hybrid"):
    """Adaptive subgradient with backtracking, multiple gradient strategies."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    it = 0
    step = 0.5
    consecutive_fail = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S ** 2)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Choose gradient
        phase = it % 400
        if method == "hybrid":
            if phase < 10:
                grad = topk_subgradient(a, c, 1)
            elif phase < 60:
                grad = topk_subgradient(a, c, 4)
            elif phase < 150:
                grad = topk_subgradient(a, c, 16)
            elif phase < 300:
                grad = topk_subgradient(a, c, 64)
            else:
                beta = min(100.0, 5.0 + 0.02 * it)
                grad = lse_gradient(a, c, beta)
        else:
            beta = min(50.0, 2.0 + 0.01 * it)
            grad = lse_gradient(a, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        max_step = 2.0 * S / (n * gnorm + 1e-15)
        step = min(step, max_step)
        step = max(step, 1e-6)

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S ** 2)

        if obj_new < cur_obj - 1e-12:
            a = a_new
            c = c_new
            step = min(step * 1.1, max_step)
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            step *= 0.7

            if consecutive_fail > 30:
                noise = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                step = 0.3
                consecutive_fail = 0

            if consecutive_fail > 80:
                noise = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                step = 0.5
                consecutive_fail = 0

    return best_a, best_obj


def run_peak_flatten(a_init, max_time, rng):
    """
    Targeted peak-flattening: iteratively identify peaks and redistribute mass.
    Uses a "water-filling" metaphor.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    alpha = 0.05  # redistribution strength

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S ** 2)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top peaks
        n_peaks = min(32, max(2, it))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_thresh = c[peak_idx[-1]] * 0.9

        # Compute "pressure map": which a[j] contribute most to peaks
        pressure = np.zeros(n)
        for k in peak_idx:
            weight = (c[k] - peak_thresh) / n_peaks
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                pressure[lo:hi+1] += weight * a[k - (np.arange(lo, hi+1))]

        # Subtract pressure (reduce contributors to peaks)
        pnorm = np.linalg.norm(pressure)
        if pnorm < 1e-15:
            break

        # Scale pressure to have the right magnitude
        max_red = alpha * S * 0.1
        if pnorm > max_red:
            pressure = pressure * (max_red / pnorm)

        a_new = a - pressure

        # Find "safe" positions to add mass back (low contribution to peaks)
        safe_mask = pressure < np.median(pressure)
        if a_new.sum() < S * 1e-10:
            a_new = a.copy()
        else:
            # Redistribute to safe positions
            deficit = S - a_new.sum()
            if deficit > 0:
                safe_vals = np.where(safe_mask, a_new, 0.0)
                s_safe = safe_vals.sum()
                if s_safe > 1e-15:
                    a_new[safe_mask] += deficit * (safe_vals[safe_mask] / s_safe)
                else:
                    a_new += deficit / n
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, S)

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S ** 2)

        if obj_new < cur_obj - 1e-12:
            a = a_new
        else:
            # Backtrack
            for back_t in [0.5, 0.25, 0.1]:
                a_try = (1.0 - back_t) * a + back_t * a_new
                a_try = normalize_to_sum(a_try, S)
                obj_try = 2.0 * len(a_try) * fft_convolve_self(a_try).max() / (S ** 2)
                if obj_try < cur_obj:
                    a = a_try
                    break
            else:
                # No improvement, add noise
                noise = 0.002 * S / np.sqrt(n)
                a = normalize_to_sum(a + rng.normal(0, noise, n), S)
                alpha *= 0.9

        # Decay alpha slowly
        alpha = max(0.001, alpha * 0.999)

    return best_a, best_obj


def generate_initializations(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None and len(given) == n:
        inits.append(("given", given.copy().astype(np.float64)))

    inits.append(("cos", 0.5 + 0.5 * np.cos(2 * np.pi * tt)))
    inits.append(("raised_cos", 0.54 + 0.46 * np.cos(2 * np.pi * tt)))
    inits.append(("sinc_like", np.maximum(np.sinc(4 * (tt - 0.5)), 0.01)))
    inits.append(("gauss", np.exp(-8.0 * (tt - 0.5) ** 2)))
    inits.append(("flat_top", 1.0 + 0.05 * np.cos(2 * np.pi * 3 * tt)))
    inits.append(("multi_harm", 1.0 + 0.2 * np.cos(2*np.pi*2*tt) + 0.1*(1-np.exp(-50.0*tt*(1-tt)))) )
    inits.append(("random", rng.random(n) + 0.1))
    inits.append(("bimodal", np.exp(-40*(tt-0.25)**2) + np.exp(-40*(tt-0.75)**2) + 0.1))

    # Structured: alternating high/low
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits.append(("alternating", alt))

    # Tukey window (flat-top with smooth edges)
    r = 0.3
    tukey = np.ones(n)
    edge = int(r * n)
    for i in range(edge):
        tukey[i] = 0.5 + 0.5 * np.cos(np.pi * (2 * i / edge - 1))
        tukey[n-1-i] = tukey[i]
    inits.append(("tukey", tukey))

    # Given with perturbations
    if given is not None and len(given) == n:
        for i in range(3):
            pert = given.copy() * (1.0 + 0.1 * rng.standard_normal(n))
            prop = ("given_pert_%d" % i, np.maximum(pert, 0.0))
            inits.append(prop)

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses LP-based SOTA, LSE+FISTA, adaptive subgradient, and peak flattening.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization. Budget: %.0fs" % budget_s)
    if given is not None:
        print("Previous best length: %d" % len(given))

    # Determine lengths to try
    lengths = []
    if given is not None and 4096 <= len(given) <= 16384:
        lengths.append(len(given))
    lengths.append(8192)
    lengths.append(16384)
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    print("Trying lengths: %s" % lengths)

    # Time allocation: give more time to the length of the previous best
    # and to 8192
    time_weights = []
    for L in lengths:
        if given is not None and L == len(given):
            time_weights.append(3.0)
        elif L == 8192:
            time_weights.append(2.5)
        else:
            time_weights.append(1.5)
    total_w = sum(time_weights)
    time_weights = [w / total_w for w in time_weights]

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * n)
        t_alloc = (deadline - time.monotonic()) * time_weights[li]
        # Cap to not exceed remaining time
        t_alloc = min(t_alloc, (deadline - time.monotonic()) * 0.6)
        t_alloc = max(t_alloc, 30.0)

        print("\n=== n=%d, allocated %.0fs ===" % (n, t_alloc))
        t0 = time.monotonic()

        # Get initializations for this length
        if given is not None and len(given) == n:
            given_n = given
        else:
            given_n = None

        inits = generate_initializations(n, rng, given_n)
        print("  %d initializations" % len(inits))

        # Split time among methods
        # 50% LP SOTA, 20% LSE FISTA, 15% peak flatten, 15% mixed
        n_inits = len(inits)
        per_init = t_alloc / (n_inits + 4)  # +4 for dedicated method runs

        # Phase A: Run LP SOTA on best initializations
        # Sort initializations by their objective (best first)
        init_scores = []
        for name, arr in inits:
            arr = normalize_to_sum(arr, S)
            obj = compute_objective(arr)
            init_scores.append((obj, name, arr))
        init_scores.sort(key=lambda x: x[0])

        # Run LP SOTA on top initializations
        lp_time = t_alloc * 0.45
        n_lp = min(4, len(init_scores))
        per_lp = lp_time / n_lp

        for i in range(n_lp):
            if time.monotonic() > deadline - 40:
                break
            obj_i, name, arr = init_scores[i]
            t_rem = deadline - time.monotonic() - 30
            t_this = min(per_lp, t_rem)
            if t_this < 10:
                break

            print("  LP SOTA [%s] (init obj=%.6f, %.0fs)" % (name, obj_i, t_this))
            a_opt, score = run_lp_sota(arr, t_this, rng)
            a_opt = normalize_to_sum(a_opt, n)
            score = compute_objective(a_opt)

            if score < global_best:
                global_best = score
                global_a = a_opt.copy()
                print("    *** New best: %.6f (n=%d) ***" % (global_best, n))

        # Phase B: LSE FISTA on the best so far
        if time.monotonic() < deadline - 30:
            fista_time = min(t_alloc * 0.20, deadline - time.monotonic() - 30)
            if fista_time > 20:
                if global_a is not None and len(global_a) == n:
                    start_arr = global_a
                else:
                    start_arr = init_scores[0][2]
                print("  LSE FISTA (%.0fs)" % fista_time)
                a_opt, score = run_lse_fista(start_arr, fista_time, rng)
                a_opt = normalize_to_sum(a_opt, n)
                score = compute_objective(a_opt)
                if score < global_best:
                    global_best = score
                    global_a = a_opt.copy()
                    print("    *** New best: %.6f (n=%d) ***" % (global_best, n))

        # Phase C: Peak flattening
        if time.monotonic() < deadline - 30:
            pf_time = min(t_alloc * 0.15, deadline - time.monotonic() - 30)
            if pf_time > 20:
                if global_a is not None and len(global_a) == n:
                    start_arr = global_a
                else:
                    start_arr = init_scores[0][2]
                print("  Peak flatten (%.0fs)" % pf_time)
                a_opt, score = run_peak_flatten(start_arr, pf_time, rng)
                a_opt = normalize_to_sum(a_opt, n)
                score = compute_objective(a_opt)
                if score < global_best:
                    global_best = score
                    global_a = a_opt.copy()
                    print("    *** New best: %.6f (n=%d) ***" % (global_best, n))

        # Phase D: More LP SOTA iterations on the best
        if time.monotonic() < deadline - 30:
            lp2_time = min(t_alloc * 0.20, deadline - time.monotonic() - 30)
            if lp2_time > 20 and global_a is not None and len(global_a) == n:
                print("  LP SOTA refine (%.0fs)" % lp2_time)
                a_opt, score = run_lp_sota(global_a, lp2_time, rng)
                a_opt = normalize_to_sum(a_opt, n)
                score = compute_objective(a_opt)
                if score < global_best:
                    global_best = score
                    global_a = a_opt.copy()
                    print("    *** New best: %.6f (n=%d) ***" % (global_best, n))

        print("  n=%d done. Best so far: %.6f (elapsed %.0fs)" % (
            n, global_best, time.monotonic() - t0))

    # Final phase: intensive polishing of the global best
    if global_a is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final polish: n=%d, remaining %.0fs ===" % (n_best, remaining))

        # Run LP SOTA one more time with fresh RNG
        rng2 = np.random.default_rng(seed + 999)
        polish_time = remaining * 0.6
        if polish_time > 20:
            a_opt, score = run_lp_sota(global_a, polish_time, rng2)
            a_opt = normalize_to_sum(a_opt, n_best)
            score = compute_objective(a_opt)
            if score < global_best:
                global_best = score
                global_a = a_opt.copy()
                print("  After final LP: %.6f" % global_best)

        # Very fine subgradient
        if time.monotonic() < deadline - 10:
            fine_time = min(remaining * 0.3, deadline - time.monotonic() - 5)
            if fine_time > 10:
                a_opt, score = run_adaptive_subgrad(global_a, fine_time, rng2, method="hybrid")
                a_opt = normalize_to_sum(a_opt, n_best)
                score = compute_objective(a_opt)
                if score < global_best:
                    global_best = score
                    global_a = a_opt.copy()
                    print("  After fine subgrad: %.6f" % global_best)

    # Final verification
    if global_a is None:
        # Fallback: use previous best or a simple sequence
        if given is not None:
            global_a = given[:8192] if len(given) >= 8192 else given
        else:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence returned non-finite, using fallback")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
    except Exception as e:
        print("Verification error: %s" % str(e))

    print("Final: length=%d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - (deadline - budget_s + 30)))
    return [float(x) for x in global_a]
