import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def smooth_max_grad(c, beta):
    """Compute smooth max and its gradient (weights) on c."""
    # Shift for numerical stability
    c_shifted = c - c.max()
    logits = beta * c_shifted
    # Clip for numerical stability
    logits = np.clip(logits, -500, 0)
    exp_logits = np.exp(logits)
    weights = exp_logits / exp_logits.sum()
    smooth_max_val = c.max() + np.log(exp_logits.sum()) / beta
    return smooth_max_val, weights

                     lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):

    s = a.sum()

        a = a_new
        
        # Periodic noise injection

            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

def compute_score(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:

    return float(2 * n * c.max() / (S * S))

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smooth-max ADAM optimization
with adaptive beta annealing, top-k focusing, and periodic restarts.

Key improvements over previous approach:
1. Smooth max objective with beta annealing (replaces discontinuous peak tracking).
2. Proper ADAM optimizer with momentum for faster convergence.
3. Efficient O(n log n) subgradient via FFT cross-correlation.
4. Periodic noise injection to escape local minima.
5. Multi-start with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_subgrad(a, weights):
    """Compute 2 * (weights cross-correlate a) for the smooth max subgradient."""
    n = len(a)
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    xcorr = np.real(ifft(np.conj(W) * A))
    return 2.0 * xcorr[:n]


def get_weights(c, beta, top_k=None):
    """Compute softmax weights for the smooth max approximation."""
    if top_k is not None and top_k < len(c):
        # Focus on top-k peaks
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        # Zero out all but top-k
        c_masked = np.full_like(c, -np.inf)
        c_masked[top_idx] = c[top_idx]
    else:
        c_masked = c.copy()
    
    # Numerical stability
    c_max = c_masked.max()
    logits = beta * (c_masked - c_max)
    logits = np.clip(logits, -50, 0)
    exp_logits = np.exp(logits)
    total = exp_logits.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_logits / total


def adam_smooth_max(a_init, n, S, max_time, rng, 
                     beta_init=5.0, beta_final=500.0,
                     lr=0.05, top_k=50,
                     noise_period=500, noise_prob=0.3,
                     stall_threshold=300):
    """ADAM optimization on smooth max of self-convolution with projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    a = project_simplex(a, S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1_adam = 0.9
    beta2_adam = 0.999
    eps = 1e-8
    
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    last_improve_t = 0
    last_improve_score = best_score
    
    print("    Initial score: %.6f" % best_score)
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Anneal beta (exponential)
        beta = beta_init * (beta_final / beta_init) ** progress
        beta = max(beta, 1.0)
        
        # Adaptive top-k: start broad, narrow down
        tk = max(10, int(top_k * (1.0 + 2.0 * (1 - progress))))
        
        # Compute convolution
        c = fft_convolve_self(a)
        
        # Get weights
        weights = get_weights(c, beta, top_k=tk)
        
        # Compute subgradient
        grad = compute_subgrad(a, weights)
        
        # ADAM update
        m = beta1_adam * m + (1 - beta1_adam) * grad
        v = beta2_adam * v + (1 - beta2_adam) * grad ** 2
        m_hat = m / (1 - beta1_adam ** t)
        v_hat = v / (1 - beta2_adam ** t)
        
        # Adaptive learning rate: decrease over time
        lr_eff = lr * (1.0 - 0.5 * progress)
        
        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)
        a_new = project_simplex(a_new, S)
        
        # Evaluate (every few steps to save time, or every step for accuracy)
        if t % 5 == 0 or t < 100:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve_t = t
                last_improve_score = sc
                if t % 1000 == 0:
                    print("    t=%d, score=%.6f, beta=%.1f" % (t, best_score, beta))
        
        a = a_new
        
        # Stall detection and restart
        if t - last_improve_t > stall_threshold and t > 1000:
            noise_scale = 0.05 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            print("    Restart at t=%d (score=%.6f)" % (t, best_score))
    
    return best_a, best_score


def make_diverse_inits(n, rng, provided_seq=None):
    """Generate diverse initializations."""
    inits = []
    
    if provided_seq is not None and len(provided_seq) == n:
        inits.append(("provided", np.clip(provided_seq, 0, 1000).astype(np.float64)))
    
    tt = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(("uniform", np.ones(n) * 0.5))
    inits.append(("sin1", 0.5 + 0.4 * np.sin(2 * np.pi * tt)))
    inits.append(("sin3", 0.5 + 0.4 * np.sin(2 * np.pi * 3 * tt)))
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt * tt))))
    inits.append(("random", rng.random(n) + 0.05))
    
    # Smooth noise
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 128)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_noise", 0.2 + 0.6 * smooth))
    
    return inits

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using smooth-max ADAM optimization
with adaptive beta annealing, top-k peak focusing, and periodic structured restarts.

Key improvements over previous approach:
1. Smooth max objective with continuous beta annealing (5 to 2000) for smooth gradient.

3. ADAM optimizer with adaptive learning rate and momentum.
4. Top-k peak focusing that narrows adaptively over optimization.
5. Structured restarts (Gaussian + spectral perturbations) to escape local minima.
6. Spectral flattening post-processing for final polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value 2*n*max(c)/S^2."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def compute_subgrad(a, weights):
    """Compute 2*(weights cross-correlate a) for smooth max subgradient.
    a: length n, weights: length 2n-1. Returns length n.
    """
    n = len(a)
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    xcorr = np.real(ifft(np.conj(W) * A))
    return 2.0 * xcorr[:n]


def get_weights(c, beta, top_k=None):
    """Compute softmax weights for smooth max approximation with optional top-k."""
    if top_k is not None and top_k < len(c):
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        c_masked = np.full_like(c, -1e18)
        c_masked[top_idx] = c[top_idx]
    else:
        c_masked = c

    c_max = c_masked.max()
    logits = beta * (c_masked - c_max)
    logits = np.clip(logits, -50.0, 0.0)
    exp_logits = np.exp(logits)
    total = exp_logits.sum()
    if total < 1e-300:
        return np.ones(len(c), dtype=np.float64) / len(c)
    return exp_logits / total


def adam_smooth_max(a_init, n, S, max_time, rng,
                    beta_init, beta_final, lr, top_k,
                    stall_threshold=400, verbose=True):
    """ADAM optimization on smooth max of self-convolution with simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    a = project_simplex(a, S)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    last_improve_t = 0

    if verbose:
        print("    Init score: %.6f, n=%d, S=%.2f" % (best_score, n, S))

    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Anneal beta exponentially
        beta = beta_init * (beta_final / beta_init) ** progress
        beta = max(beta, 1.0)

        # Adaptive top-k: broader early, narrower late
        tk = max(5, int(top_k * (1.5 - 1.0 * progress)))

        # Compute convolution
        c = fft_convolve_self(a)

        # Get softmax weights
        weights = get_weights(c, beta, top_k=tk)

        # Compute subgradient via FFT cross-correlation
        grad = compute_subgrad(a, weights)

        # ADAM update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        # Adaptive learning rate
        lr_eff = lr * (1.0 - 0.6 * progress ** 1.5)

        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)

        # Evaluate periodically

            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve_t = t

        a = a_new

        # Stall detection: structured restart

            noise_scale = 0.03 * S / np.sqrt(n)
            a = a + rng.normal(0, noise_scale, n)

            A = fft(a, fsize)

            for bn in bins:

                A[fsize - bn] *= np.exp(-1j * phase_shift)
            a = np.real(ifft(A))[:n]

            if s2 > 1e-15:

            v = np.zeros(n, dtype=np.float64)
            t = 0
            if verbose:
                print("    Restart (score=%.6f)" % best_score)

    return best_a, best_score


def spectral_flatten_pass(a, n_iter, alpha, rng):
    """Apply spectral flattening steps to reduce convolution peak."""
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Flatten: compress dynamic range
        p_median = np.median(P[P > 0]) if np.any(P > 0) else 1.0
        if p_median < 1e-30:
            break
        ratio = P / p_median
        ratio = np.clip(ratio, 1e-10, 1e10)

        # Power compression
        P_new = p_median * np.power(ratio, alpha)
        P_new = np.maximum(P_new, 0.0)

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new_full, S)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def make_inits(n, rng, provided):
    """Generate diverse initializations."""
    inits = []
    if provided is not None:
        inits.append(("provided", np.clip(provided, 0, 1000).astype(np.float64)))

    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("uniform", np.ones(n) * 0.5))
    inits.append(("sin1", 0.5 + 0.4 * np.sin(2 * np.pi * tt + rng.uniform(0, 2 * np.pi))))
    inits.append(("sin3", 0.5 + 0.35 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.sin(2 * np.pi * 7 * tt)))
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (4 * tt + 2 * tt * tt))))

    inits.append(("smooth_noise", 0.15 + 0.7 * smooth))

    # Triangular
    inits.append(("triangle", 1.0 - np.abs(2.0 * tt - 1.0)))

    return inits
