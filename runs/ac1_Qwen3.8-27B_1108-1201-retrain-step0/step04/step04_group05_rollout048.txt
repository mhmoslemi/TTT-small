"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using combined spectral gradient descent and L-BFGS.

Key innovations:
1. Spectral-domain gradient descent: directly optimizes the power spectrum to flatten
   the autocorrelation peak, exploiting the natural FFT structure of the problem.
2. L-BFGS on LSE-smoothed max-conv for time-domain local refinement.
3. Spectral equalization as global exploration moves.
4. Adaptive beta ramping and step size control.
5. Multi-start with diverse initializations including Fourier-based and cosine-modulated.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. a has length n, N >= 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_values(c, beta, c_max):
    """Compute LSE weights and value."""
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c), c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return w, lse_val


def spectral_gradient_step(a, N, beta, eta, S):
    """
    One step of spectral-domain gradient descent on LSE of autocorrelation peak.
    Works by modifying the power spectrum to reduce the LSE.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    
    w, lse_val = lse_values(c, beta, c_max)
    
    # Gradient of LSE w.r.t. P[m]: (1/N) * FFT(w_padded)[m]
    w_full = np.zeros(N)
    w_full[:2 * n - 1] = w
    grad_P = np.real(fft(w_full)) / N
    
    # Update P
    P_new = P - eta * grad_P
    P_new = np.maximum(P_new, 0.0)
    # Enforce Hermitian symmetry (for real a)
    P_sym = np.zeros(N)
    P_sym[0] = (P_new[0] + P_new[0]) / 2.0  # DC is its own conjugate
    for m in range(1, N // 2 + 1):
        P_sym[m] = (P_new[m] + P_new[N - m]) / 2.0
        P_sym[N - m] = P_sym[m]
    # Maintain DC component
    P_sym[0] = S * S
    # Ensure non-negative
    P_sym = np.maximum(P_sym, 0.0)
    
    # Reconstruct A from new P, keeping old phase
    phase = np.angle(A)
    A_new = np.sqrt(P_sym) * np.exp(1j * phase)
    # Enforce Hermitian symmetry on A
    A_new[0] = np.abs(A_new[0])  # DC is real and non-negative for non-negative a
    for m in range(1, N // 2 + 1):
        A_new[N - m] = np.conj(A_new[m])
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a, c_max
    a_new = a_new * (S / s_new)
    return a_new, lse_val


def spectral_block(a_init, n, S, N, max_time, rng, beta_init=20.0, beta_end=200.0, eta_init=None):
    """
    Block of spectral gradient descent steps with adaptive parameters.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    if eta_init is None:
        eta_init = 2.0 * S * S / N
    
    while time.monotonic() - start_time < max_time:
        it += 1
        frac = min(1.0, it / max(1, int(max_time * 5)))
        beta = beta_init + (beta_end - beta_init) * frac
        eta = eta_init * (1.0 + 0.5 * frac)
        
        # Try the step with backtracking
        a_new, val = spectral_gradient_step(a, N, beta, eta, S)
        c_new, _ = compute_conv(a_new, N)
        cmax_new = c_new.max()
        
        # Backtrack if objective increased significantly
        if cmax_new > best_cmax * 1.05:
            # Try smaller step
            for attempt in range(3):
                eta *= 0.5
                a_new, val = spectral_gradient_step(a, N, beta, eta, S)
                c_new, _ = compute_conv(a_new, N)
                cmax_new = c_new.max()
                if cmax_new <= best_cmax * 1.02:
                    break
        
        a = a_new
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a.copy()
        
        # Periodic perturbation to escape local minima
        if it % 300 == 150 and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Periodic spectral equalization (global move)
        if it % 500 == 250:
            a_pad = np.zeros(N)
            a_pad[:n] = a
            A = fft(a_pad)
            P = np.abs(A) ** 2
            P_mean = P.mean()
            # Compress dynamic range
            alpha_eq = 0.15
            P_eq = P ** (1.0 - alpha_eq) * (P_mean ** alpha_eq)
            P_eq[0] = S * S
            phase = np.angle(A)
            A_eq = np.sqrt(np.maximum(P_eq, 0)) * np.exp(1j * phase)
            A_eq[0] = np.abs(A_eq[0])
            for m in range(1, N // 2 + 1):
                A_eq[N - m] = np.conj(A_eq[m])
            a_eq = np.real(ifft(A_eq))[:n]
            a_eq = np.maximum(a_eq, 0.0)
            s_eq = a_eq.sum()
            if s_eq > 1e-10:
                a_eq = a_eq * (S / s_eq)
                c_eq, _ = compute_conv(a_eq, N)
                if c_eq.max() < best_cmax:
                    best_cmax = c_eq.max()
                    best_a = a_eq.copy()
                    a = a_eq
    
    return best_a, best_cmax


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lse_grad_time(a, N, beta):
    """
    Gradient of LSE-smoothed peak of conv(a,a) w.r.t. a (time domain).
    Returns (grad, c, c_max).
    """
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    w, _ = lse_values(c, beta, c_max)
    
    # grad_j = 2 * sum_k w[k] * a[k-j] (shifted)
    # Using FFT: grad = 2 * IFFT(FFT(w_padded) * conj(A_padded))[:n]
    w_full = np.zeros(N)
    w_full[:2 * n - 1] = w
    W = fft(w_full)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c, c_max


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=300.0, m_lbfgs=10):
    """
    L-BFGS on LSE-smoothed max-conv objective (time domain).
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, c, c_max = lse_grad_time(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * np.sqrt(beta))
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.05)
            step_scale = 2.0 * S / (n * np.sqrt(beta))
            g, c, c_max = lse_grad_time(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        if it % 50 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        alpha_ls = step_scale
        best_alpha = 0.0
        a_new = a
        cmax_new = c_max
        
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * 1.001 or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max * 1.001:
                    best_alpha = alpha_ls
                    break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, c_nw, cmax_nw = lse_grad_time(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c = c_nw
        c_max = cmax_nw
        
        if it % 400 == 200 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, c, c_max = lse_grad_time(a, N, beta)
    
    return best_a, best_cmax


def nesterov_polish(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=400):
    """Nesterov accelerated top-k subgradient for final polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 20)
        c, A = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(N)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full)
        Af = fft(np.concatenate([y, np.zeros(N - n)]))
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        if it % 3 == 0:
            c_x, _ = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = y * (S / max(y.sum(), 1e-15))
        a = x_new
        t = t_new
        
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_cosine(n, rng, alpha=0.5):
    """Cosine-modulated initialization: 1 + alpha*cos(2*pi*j/(n-1) + phase)."""
    j = np.arange(n, dtype=np.float64)
    phase = rng.random() * 2 * np.pi
    a = 1.0 + alpha * np.cos(2 * np.pi * j / max(n - 1, 1) + phase)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Fourier-based: random phase, shaped power spectrum."""
    # Use a power spectrum that decays slowly (flatter than sinc^2)
    m = np.arange(N)
    # Target: roughly flat with slight decay
    P_target = 1.0 / (1.0 + 0.1 * np.abs(m - N // 2) / (N // 4))
    P = P_target * (1.0 + 0.3 * rng.random(N))
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    # Enforce symmetry
    P_sym = np.zeros(N)
    P_sym[0] = P[0]
    for m in range(1, N // 2 + 1):
        P_sym[m] = (P[m] + P[N - m]) / 2
        P_sym[N - m] = P_sym[m]
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P_sym) * np.exp(1j * phase)
    A[0] = np.sqrt(P_sym[0])
    for m in range(1, N // 2 + 1):
        A[N - m] = np.conj(A[m])
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_multi_sine(n, rng):
    """Multi-frequency sine initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for _ in range(5):
        freq = rng.uniform(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.15 * np.sin(2 * np.pi * freq * tt + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Combines spectral gradient descent, L-BFGS, and Nesterov polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c, _ = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            if label:
                print("  [%s] %.6f" % (label, global_best))
        return sc
    
    # --- Build initializations ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
        elif given.size > n:
            inits.append(("given_trunc", np.clip(given[:n], 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("cosine", make_init_cosine(n, rng, 0.5)))
    inits.append(("cosine2", make_init_cosine(n, rng, 0.8)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("multisine", make_init_multi_sine(n, rng)))
    
    n_inits = len(inits)
    print("Initializations: %s" % [x[0] for x in inits])
    
    # --- Phase 1: Quick spectral gradient on each init ---
    print("=== Phase 1: Spectral gradient exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(480.0, (deadline - t1) * 0.25)
    per_init = phase1_budget / n_inits
    
    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        
        a_sp, sc_sp = spectral_block(a_norm, n, S, N, rem, rng,
                                      beta_init=15.0, beta_end=100.0)
        update_best(a_sp, name + "/spec")
    
    print("Phase 1 done. Best: %.6f" % global_best)
    
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0
    
    # --- Phase 2: Focused optimization alternating spectral and L-BFGS ---
    print("=== Phase 2: Alternating spectral + L-BFGS ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.5)
    
    a_curr = global_a.copy()
    stall = 0
    n_rounds = 0
    
    while time.monotonic() < t2 + phase2_budget:
        n_rounds += 1
        block_time = min(60.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 5:
            break
        
        # Alternate between spectral and L-BFGS
        if n_rounds % 2 == 1:
            # Spectral step
            beta_s = 20.0 + 10.0 * n_rounds
            beta_s_end = min(300.0, 50.0 + 20.0 * n_rounds)
            a_new, sc_new = spectral_block(a_curr, n, S, N, min(block_time, 40.0), rng,
                                            beta_init=beta_s, beta_end=beta_s_end)
            method = "spec"
        else:
            # L-BFGS step
            beta_l = 10.0 + 5.0 * stall
            beta_l_max = min(400.0, 50.0 + 20.0 * stall)
            a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, min(block_time, 50.0), rng,
                                             beta_init=beta_l, beta_max=beta_l_max, m_lbfgs=10)
            method = "lbfgs"
        
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2 r%d/%s] %.6f" % (n_rounds, method, global_best))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 4:
                # Perturb to escape local minimum
                noise_scale = 0.01 * S / np.sqrt(n)
                a_curr = project_simplex(a_curr + rng.normal(0, noise_scale, n), S)
                stall = 0
    
    # --- Phase 3: High-precision L-BFGS ---
    print("=== Phase 3: High-precision L-BFGS ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.25)
    
    a_curr = global_a.copy()
    stall = 0
    while time.monotonic() < t3 + phase3_budget:
        block_time = min(80.0, (t3 + phase3_budget) - time.monotonic())
        if block_time < 5:
            break
        beta_start = 50.0 + 10.0 * stall
        beta_end = min(500.0, 100.0 + 30.0 * stall)
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end, m_lbfgs=12)
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P3] %.6f" % global_best)
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 3:
                a_curr = project_simplex(a_curr + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                stall = 0
    
    # --- Phase 4: Spectral final refinement ---
    print("=== Phase 4: Spectral refinement ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20:
        a_sp, sc_sp = spectral_block(global_a, n, S, N, min(remaining * 0.4, 200.0), rng,
                                      beta_init=50.0, beta_end=300.0)
        update_best(a_sp, "spec_final")
    
    # --- Phase 5: Nesterov polish ---
    print("=== Phase 5: Nesterov polish ===")
    t5 = time.monotonic()
    remaining = deadline - t5
    if remaining > 30:
        a_ng, sc_ng = nesterov_polish(global_a, n, S, N, min(remaining - 10, 300.0),
                                       rng, top_k_init=30, top_k_max=500)
        update_best(a_ng, "nesterov")
    
    # --- Phase 6: Final L-BFGS polish ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining - 5, 150.0),
                                         rng, beta_init=100.0, beta_max=500.0, m_lbfgs=10)
        update_best(a_fin, "final_lbfgs")
    
    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    print("Returning length %d, internal=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
