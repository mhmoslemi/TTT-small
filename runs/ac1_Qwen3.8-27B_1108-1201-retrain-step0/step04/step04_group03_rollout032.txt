"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Targeted peak-reduction via multiplicative updates with QP corrections,
periodic LP best-response, and adaptive noise for exploration.
Key insight: The optimal sequence must be highly asymmetric (symmetric sequences
have objective >= 2), so we focus on reducing the peak lags of the self-convolution.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def fft_convolve_self(a):
    """Full self-convolution via FFT (linear, not circular)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(a, b, top_k=100, beta=50.0):
    """
    Compute a 'pressure' gradient from the top-k peak lags of b = a*a.
    Returns gradient vector of length n.
    """
    n = len(a)
    n_b = len(b)
    max_b = b.max()
    
    # Get top-k lags
    top_lags = np.argsort(b)[::-1][:top_k]
    
    # Compute weights (softmax over top peaks)
    weights = np.exp(beta * (b[top_lags] - max_b))
    weights /= weights.sum()
    
    # Accumulate gradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_lags):
        # b[k] = sum_i a[i] * a[k-i] for valid i
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # db[k]/da[i] = a[k-i] for i in [lo, hi]
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * a[k - i_range]
    
    return grad

def multiplicative_update(a, grad, lr, S):
    """Apply multiplicative update and project to simplex."""
    max_grad = grad.max()
    if max_grad < 1e-15:
        return a.copy()
    # Clip exponent to avoid underflow
    scale = lr * grad / max(max_grad, 1e-15)
    scale = np.clip(scale, -50, 0)
    a_new = a * np.exp(scale)
    a_new = np.maximum(a_new, 1e-15)
    a_new = project_simplex(a_new, S)
    return a_new

def try_qp_correction(a, b, S, top_k=50, M_tol=0.02):
    """
    QP correction: linearize the top-k peak constraints and solve.
    Returns corrected a or None if failed.
    """
    n = len(a)
    max_b = b.max()
    M = max_b * (1 - M_tol)  # Try to reduce the peak by M_tol fraction
    
    # Get top-k lags
    top_lags = np.argsort(b)[::-1][:top_k]
    
    # Build constraint matrix (sparse would be better, but dense for simplicity)
    # For lag k: sum_i a_old[k-i] * a[i] <= M
    # This is a row vector times a
    
    # Actually, let's use a different formulation.
    # We want to minimize ||a - a_old||^2 subject to:
    #   sum a = S
    #   a >= 0
    #   For each top lag k_j: sum_i a_old[k_j-i] * a[i] <= M
    
    # The constraint matrix A has top_k rows, each row is a shifted version of a_old.
    # Let's build it efficiently.
    
    A_rows = []
    for k in top_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        row[i_range] = a[k - i_range]  # a_old[k - i]
        A_rows.append(row)
    
    A_mat = np.array(A_rows)  # (top_k, n)
    M_vec = np.full(top_k, M)
    
    # Solve QP using cvxpy
    a_var = cp.Variable(n)
    objective = cp.Minimize(cp.sum Squared(a_var - a))
    constraints = [
        cp.sum(a_var) == S,
        a_var >= 0,
        A_mat @ a_var <= M_vec
    ]
    
    prob = cp.Problem(objective, constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
    except Exception:
        return None
    
    if time.monotonic() - start > 10:
        return None
    
    if a_var.value is not None:
        a_new = np.maximum(a_var.value, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new * (S / s)
            return a_new
    return None

def try_lp_correction(a, S, top_k=100, time_limit=5.0):
    """
    LP best-response: find g that maximizes sum(g) subject to (a*g)(k) <= M for top-k lags.
    """
    n = len(a)
    b = fft_convolve_self(a)
    M = b.max()
    
    top_lags = np.argsort(b)[::-1][:top_k]
    
    A_rows = []
    for k in top_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        row[i_range] = a[k - i_range]
        A_rows.append(row)
    
    A_mat = np.array(A_rows)
    M_vec = np.full(top_k, M)
    
    g_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [A_mat @ g_var <= M_vec, g_var >= 0]
    
    prob = cp.Problem(objective, constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=20000)
    except Exception:
        return None
    
    if time.monotonic() - start > time_limit * 3:
        return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        s = g.sum()
        if s > 1e-10:
            g = g * (S / s)
            return g
    return None

def optimize_sequence(a_init, n, S, max_time, rng, verbose=False):

        a = a * (S / s)

    best_val = np.inf
    b = fft_convolve_self(a)
    best_val = b.max()
    
    start = time.monotonic()
    it = 0
    last_improve = 0
    q = 0
    
    # Parameters
    lr_start = 0.3
    lr_end = 0.01
    beta_start = 30.0
    beta_end = 200.0
    qp_every = 300  # Do QP every this many iterations
    lp_every = 2000  # Do LP every this many iterations
    noise_every = 5000  # Add noise every this many iterations
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 2)))  # Rough progress estimate
        
        # Adaptive learning rate and beta
        lr = lr_start * (lr_end / lr_start) ** progress
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute convolution
        b = fft_convolve_self(a)
        cur = b.max()
        
        if cur < best_val:
            best_val = cur
            best_a = a.copy()

            if verbose and it % 1000 == 0:

        # QP correction periodically

            if a_qp is not None:

                    a = a_qp
                    cur = b_qp.max()
                    if cur < best_val:
                        best_val = cur
                        best_a = a.copy()
                        last_improve = it
                    continue
        
        # LP correction (less frequent)
        if it % lp_every == 0 and (time.monotonic() - start) < max_time * 0.7:
            g_lp = try_lp_correction(a, S, top_k=80)
            if g_lp is not None:
                # Try convex combinations

                    a_test = (1 - frac) * a + frac * g_lp
                    a_test = np.maximum(a_test, 0)
                    a_test = project_simplex(a_test, S)
                    b_test = fft_convolve_self(a_test)
                    if b_test.max() < cur:

                        cur = b_test.max()
                        if cur < best_val:
                            best_val = cur
                            best_a = a.copy()
                            last_improve = it
                        break
        
        # Multiplicative gradient update
        grad = compute_peak_gradient(a, b, top_k=100, beta=beta)
        a = multiplicative_update(a, grad, lr, S)
        
        # Add noise periodically to escape local minima
        if it % noise_every == 0:
            age = it - last_improve
            if age > qp_every * 3:
                # Stronger noise when stuck
                noise_scale = 0.01 * S / np.sqrt(n) * min(1.0, age / (qp_every * 10))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Small noise always (for exploration)

            noise_scale = 0.001 * S / np.sqrt(n) * decay

    # Final check
    b_final = fft_convolve_self(best_a)
    if b_final.max() < best_val:
        best_val = b_final.max()
    
    return best_a, best_val

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

Looking at this problem, I need to minimize `2*n*max(f*f)/sum(f)^2` where `f*f` is the full self-convolution. Key insights:

1. **Symmetric sequences have objective ≥ 2** (by Cauchy-Schwarz on `sum(f[i]*f[n-1-i])`), so the optimal must be highly asymmetric
2. The convolution has `2n-1` lags that should be as flat as possible
3. The previous code achieved 1.507; target is 1.5027 (a ~0.3% improvement needed)

My approach uses **targeted peak reduction** with multiplicative updates (natural for non-negative variables), combined with **QP corrections** (linearizing peak constraints) and **LP best-response**, with adaptive parameters and noise injection.

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Targeted peak-reduction via multiplicative updates with QP corrections,
LP best-response, and adaptive noise. Key insight: optimal sequences must be
highly asymmetric; we focus on reducing the peak lags of the self-convolution.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT (linear, not circular)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(a, b, top_k=100, beta=50.0):
    """Compute weighted gradient pressure from top-k peak lags."""
    n = len(a)
    max_b = b.max()
    top_lags = np.argsort(b)[::-1][:top_k]
    weights = np.exp(beta * (b[top_lags] - max_b))
    tot = weights.sum()
    if tot < 1e-300:
        return np.zeros(n)
    weights /= tot
    grad = np.zeros(n)
    for w, k in zip(weights, top_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * a[k - i_range]
    return grad


def multiplicative_update(a, grad, lr, S):
    """Apply multiplicative update and project to simplex."""
    max_grad = grad.max()
    if max_grad < 1e-15:
        return a.copy()
    scale = lr * grad / max_grad
    scale = np.clip(scale, -40.0, 0.0)
    a_new = a * np.exp(scale)
    a_new = np.maximum(a_new, 1e-15)
    return project_simplex(a_new, S)


def try_qp_correction(a, b, S, top_k=50, M_tol=0.008):
    """QP: minimize ||a-a_old||^2 s.t. linearized peak constraints, sum=S."""
    n = len(a)
    max_b = b.max()
    M = max_b * (1.0 - M_tol)
    top_lags = np.argsort(b)[::-1][:top_k]
    A_rows = []
    for k in top_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            row[i_range] = a[k - i_range]
        A_rows.append(row)
    A_mat = np.array(A_rows)
    M_vec = np.full(top_k, M)
    a_var = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(a_var - a))
    constraints = [cp.sum(a_var) == S, A_mat @ a_var <= M_vec]
    prob = cp.Problem(objective, constraints)
    start_t = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=20000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if time.monotonic() - start_t > 15.0:
        return None
    if a_var.value is not None:
        a_new = np.maximum(a_var.value, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new * (S / s)
            return a_new
    return None


def try_lp_correction(a, S, top_k=80, time_limit=8.0):
    """LP best-response: max sum(g) s.t. (a*g)(k) <= M for top-k lags, g>=0."""
    n = len(a)
    b = fft_convolve_self(a)
    M = b.max()
    top_lags = np.argsort(b)[::-1][:top_k]
    A_rows = []
    for k in top_lags:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            row[i_range] = a[k - i_range]
        A_rows.append(row)
    A_mat = np.array(A_rows)
    M_vec = np.full(top_k, M)
    g_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [A_mat @ g_var <= M_vec, g_var >= 0]
    prob = cp.Problem(objective, constraints)
    start_t = time.monotonic()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=30000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if time.monotonic() - start_t > time_limit * 3:
        return None
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g * (S / s)
    return None


def optimize_sequence(a_init, n, S, max_time, rng, verbose=False):

    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    b = fft_convolve_self(a)
    best_val = b.max()
    start = time.monotonic()
    it = 0
    last_improve = 0
    n_grad = 0
    n_qp = 0
    n_lp = 0

    # Parameters
    lr_start = 0.4
    lr_end = 0.005
    beta_start = 20.0
    beta_end = 300.0
    qp_every = 200
    lp_every = 1500
    noise_period = 3000

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1.0, max_time * 1.5))

        lr = lr_start * (lr_end / lr_start) ** progress
        beta = beta_start * (beta_end / beta_start) ** progress

        b = fft_convolve_self(a)
        cur = b.max()

        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            last_improve = it
            if verbose and it % 2000 == 0:
                obj = 2.0 * n * cur / (S * S)
                print("    it=%d: obj=%.6f (grads=%d, qp=%d, lp=%d)" % (it, obj, n_grad, n_qp, n_lp))

            a_qp = try_qp_correction(a, b, S, top_k=50, M_tol=0.005 + 0.005 * (1.0 - progress))
            if a_qp is not None:
                n_qp += 1
                b_qp = fft_convolve_self(a_qp)
                if b_qp.max() < cur * 1.0001:
                    a = a_qp
                    cur = b_qp.max()
                    if cur < best_val:
                        best_val = cur
                        best_a = a.copy()
                        last_improve = it
                    continue

        # LP correction (less frequent)
        if it % lp_every == 0 and (time.monotonic() - start) < max_time * 0.65:
            g_lp = try_lp_correction(a, S, top_k=60)
            if g_lp is not None:
                n_lp += 1
                for frac in [1.0, 0.7, 0.4, 0.2, 0.1]:
                    a_test = (1.0 - frac) * a + frac * g_lp
                    a_test = np.maximum(a_test, 0.0)
                    a_test = project_simplex(a_test, S)
                    b_test = fft_convolve_self(a_test)
                    if b_test.max() < cur:
                        a = a_test
                        cur = b_test.max()
                        if cur < best_val:
                            best_val = cur
                            best_a = a.copy()
                            last_improve = it
                        break

        # Multiplicative gradient update
        grad = compute_peak_gradient(a, b, top_k=100, beta=beta)
        a = multiplicative_update(a, grad, lr, S)
        n_grad += 1

        # Noise: bigger when stuck
        age = it - last_improve
        if age > qp_every * 5 and it % 100 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * min(3.0, float(age) / (qp_every * 10))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

        # Small persistent noise (decaying)
        if it % 20 == 0:
            decay = max(0.0, 1.0 - progress) ** 2
            ns = 0.002 * S / np.sqrt(n) * decay
            if ns > 1e-12:
                a = project_simplex(a + rng.normal(0, ns, n), S)

    b_final = fft_convolve_self(best_a)
    if b_final.max() < best_val:
        best_val = b_final.max()
    return best_a, best_val


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def generate_asymmetric_init(n, rng):
    """Generate an asymmetric initialization biased toward one end."""
    t = np.linspace(0, 1, n)
    # Exponential bias toward beginning
    init = np.exp(-3.0 * t) + 0.1
    # Add some structure
    init += 0.2 * np.abs(np.sin(2 * np.pi * 3 * t)) * np.exp(-t)
    init += rng.normal(0, 0.05, n)
    return np.maximum(init, 0.01)
