def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def top_k_gradient(a, c, k=20, temperature=3.0):
    """Gradient targeting top-k peaks via cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    top_idx = np.argsort(c)[::-1][:k]
    values = c[top_idx]
    max_v = values.max()
    w = np.exp((values - max_v) / temperature)
    w /= w.sum()
    
    W_arr = np.zeros(fsize)
    for l in range(k):
        W_arr[top_idx[l]] = w[l]
    
    A = fft(a, fsize)
    W_fft = fft(W_arr, fsize)
    
    # cross-correlation: IFFT(FFT(W) * conj(FFT(A)))
    cross = np.real(ifft(W_fft * np.conj(A)))
    grad = 2.0 * cross[:n]
    
    return grad

    for solver in ['SCS', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue
    
    if time.monotonic() - start > time_limit * 2:
        return None
    
    if g_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    
    return None


def run_topk_fista(a_init, n, S, max_time, rng, k_peaks=30, temp_init=5.0, temp_rate=0.999,
                   temp_min=0.5, use_coord_descent=True, cd_every=500):
    """FISTA with top-k peak gradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_nest = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    temp = temp_init
    eval_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
            eval_count += 1
            if eval_count % 10 == 1:
                elapsed = time.monotonic() - start
                obj = 2.0 * n * cur / (y.sum() ** 2)
                print("    topk-FISTA it %d (%.0fs/%.0fs): obj=%.6f" % (it, elapsed, max_time, obj))
        
        grad = top_k_gradient(y, c, k=k_peaks, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = 1.0 / (2.0 * k_peaks)  # fixed step based on k
        # Decay step if stuck
        stuck = it - last_improve
        if stuck > 2000:
            step *= 0.5
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_nest = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2))
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_nest = t_new
        
        # Decay temperature
        temp = max(temp_min, temp * temp_rate)
        
        # Coordinate descent
        if use_coord_descent and it % cd_every == 0:
            y = coordinate_descent_step(y, fft_convolve_self(y), n, S, 
                                        n_coords=min(500, n), rng=rng)
            t_nest = 1.0
            a = y.copy()
        
        # Periodic repulsion
        if it % 300 == 0:
            c_rep = fft_convolve_self(y)
            cm = c_rep.max()
            # Find peaks
            peak_idx = np.argsort(c_rep)[::-1][:10]
            w_rep = np.exp(20.0 * (c_rep[peak_idx] - cm))
            w_rep /= w_rep.sum()
            # Repulsion gradient
            fsize = 1 << (2 * n - 1).bit_length()
            W_arr = np.zeros(fsize)
            for l in range(10):
                W_arr[peak_idx[l]] = w_rep[l]
            A = fft(y, fsize)
            W_fft = fft(W_arr, fsize)
            rep_grad = 2.0 * np.real(ifft(W_fft * np.conj(A)))[:n]
            # Multiplicative update
            y = y * np.exp(-0.05 * rep_grad / max(y.max(), 1e-10))
            y = np.maximum(y, 1e-15)
            y = project_simplex(y, S)
            t_nest = 1.0
            a = y.copy()
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = best_a.copy()
    
    return best_a, best


def run_adaptive_optimization(a_init, n, S, max_time, rng):
    """
    Adaptive optimization: alternates between top-k gradient, LP, and coordinate descent.
    Adaptively adjusts which method to use based on progress.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    method_history = []
    
    # Track which method is working best
    improvement_by_method = {'gradient': 0.0, 'lp': 0.0, 'cd': 0.0}
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        obj_cur = 2.0 * n * cur / (a.sum() ** 2)
        
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it
            if (it % 50 == 1) or cur < best * 1.001:
                elapsed = time.monotonic() - start
                print("    Adaptive it %d (%.0fs/%.0fs): obj=%.6f" % (it, elapsed, max_time, obj_cur))
        
        # Choose method based on progress
        stuck = it - last_improve
        if stuck < 100:
            method = 'gradient'
        elif stuck < 500:
            method = 'gradient'
        elif stuck < 1500:
            method = 'lp' if (time.monotonic() - start) < max_time * 0.7 else 'cd'
        else:
            method = 'cd'
        
        if method == 'gradient':
            # Top-k gradient step with Nesterov
            y = a.copy()
            k_peaks = min(50, max(10, n // 200))
            temp = max(1.0, 5.0 * np.exp(-it / 2000.0))
            grad = top_k_gradient(y, c, k=k_peaks, temperature=temp)
            step = 1.0 / (2.0 * k_peaks)
            if stuck > 500:
                step *= 0.3
            
            # Nesterov step
            x_new = project_simplex(y - step * grad, S)
            # Simple momentum
            if not hasattr(run_adaptive_optimization, '_prev'):
                run_adaptive_optimization._prev = a.copy()
            momentum = 0.9
            y_mom = a + momentum * (a - run_adaptive_optimization._prev)
            x_new = project_simplex(y_mom - step * grad, S)
            run_adaptive_optimization._prev = a
            a = x_new
            
        elif method == 'lp':
            # LP best response
            t0 = time.monotonic()
            g_lp = lp_best_response(a, n, S, top_k=min(500, 2*n-1), time_limit=5.0)
            lp_time = time.monotonic() - t0
            
            if g_lp is not None:
                c_lp = fft_convolve_self(g_lp)
                obj_lp = c_lp.max()
                
                # Try full or partial replacement
                for frac in [1.0, 0.5, 0.3, 0.1]:
                    a_test = (1 - frac) * a + frac * g_lp
                    a_test = np.maximum(a_test, 0)
                    a_test = project_simplex(a_test, S)
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < c.max():
                        a = a_test
                        improvement_by_method['lp'] += c.max() - c_test.max()
                        break
        
        elif method == 'cd':
            # Coordinate descent
            a_new = coordinate_descent_step(a, c, n, S, n_coords=min(1000, n), rng=rng)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c.max():
                a = a_new
                improvement_by_method['cd'] += c.max() - c_new.max()
            else:
                # Add noise to escape
                noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-it / 5000.0)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Reset if very stuck
        if stuck > 3000:
            noise_scale = 0.02 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it
            run_adaptive_optimization._prev = a.copy()
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    
    return best_a, best


def generate_diverse_inits(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Start from given if available
    if given is not None and len(given) == n:
        inits.append(given.copy())
    
    # Various structured inits
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.exp(-2.0 * (tt - 0.5) ** 2 / 0.1) + 0.01)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 13 * tt) * np.cos(2 * np.pi * 5 * tt))
    
    # Random with structure
    inits.append(np.exp(rng.normal(0, 0.3, n)))
    inits.append(rng.random(n) ** 0.5 + 0.01)
    
    # Block structure
    nblock = 17
    block_len = n // nblock
    heights = rng.uniform(0.2, 2.0, nblock)
    block_seq = np.repeat(heights, block_len)
    if len(block_seq) < n:
        block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
    inits.append(block_seq[:n])
    
    # "Chirp" structure
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * (tt + 0.5 * tt ** 2)))
    
    return [np.maximum(x, 0.01) for x in inits]


def interpolate_sequence(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def final_polish(a, n, S, max_time, rng):
    """Final intensive polishing with very focused gradient."""
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    
    it = 0
    last_improve = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it
        
        # Very focused: only top 5 peaks
        k = 5
        temp = 0.5
        grad = top_k_gradient(a, c, k=k, temperature=temp)
        
        # Small step size for precision
        step = 0.1 / k
        a_new = project_simplex(a - step * grad, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c.max():
            a = a_new
        else:
            # Try smaller step
            step *= 0.3
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c.max():
                a = a_new
            else:
                # Coordinate descent fallback
                a = coordinate_descent_step(a, c, n, S, n_coords=200, rng=rng)
        
        if it - last_improve > 1000:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it
    
    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses top-peak gradient Nesterov, adaptive LP, coordinate descent,
    multi-length exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Main length
    if given is not None and 2048 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192
    
    print("=== Starting optimization ===")
    print("Main length: %d, budget: %.0fs" % (main_len, budget_s))
    if given is not None:
        given_obj = 2.0 * len(given) * fft_convolve_self(given).max() / (given.sum() ** 2)
        print("Given sequence objective: %.6f" % given_obj)
    
    n = main_len
    S = np.sqrt(2.0 * n)
    
    # Phase 1: Quick top-k FISTA from given (or best init)
    t0 = time.monotonic()
    phase1_time = min(120.0, (deadline - t0) * 0.15)
    print("\n=== Phase 1: Top-k FISTA warmup (%.0fs) ===" % phase1_time)
    
    a1 = given if given is not None else np.ones(n) * S / n
    a1 = normalize_sum(a1, n)
    a1, s1 = run_topk_fista(a1, n, S, phase1_time, rng, k_peaks=30, temp_init=5.0)
    obj1 = 2.0 * n * s1 / (a1.sum() ** 2)
    print("  Phase 1 best: %.6f" % obj1)
    if obj1 < global_best:
        global_best = obj1
        global_a = a1.copy()
    
    # Phase 2: Adaptive optimization (gradient + LP + CD)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.45
    print("\n=== Phase 2: Adaptive optimization (%.0fs) ===" % phase2_time)
    
    if global_a is not None:
        a2, s2 = run_adaptive_optimization(global_a, n, S, phase2_time, rng)
        obj2 = 2.0 * n * s2 / (a2.sum() ** 2)
        print("  Phase 2 best: %.6f" % obj2)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
    
    # Phase 3: Try diverse initializations
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase3_time = remaining * 0.2
        print("\n=== Phase 3: Diverse initializations (%.0fs) ===" % phase3_time)
        
        inits = generate_diverse_inits(n, rng, given=global_a)
        per_init = phase3_time / min(4, len(inits))
        
        for i in range(min(4, len(inits))):
            if time.monotonic() > deadline - 60:
                break
            if per_init < 10:
                break
            ai, si = run_topk_fista(inits[i], n, S, per_init, rng, k_peaks=20, temp_init=8.0)
            obji = 2.0 * n * si / (ai.sum() ** 2)
            print("  Diverse init %d: %.6f" % (i, obji))
            if obji < global_best:
                global_best = obji
                global_a = ai.copy()
    
    # Phase 4: Try different length
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 32768 and alt_n != n:
            S_alt = np.sqrt(2.0 * alt_n)
            phase4_time = remaining * 0.2
            print("\n=== Phase 4: Alternative length n=%d (%.0fs) ===" % (alt_n, phase4_time))
            
            if global_a is not None:
                alt_init = interpolate_sequence(global_a, alt_n)
                alt_init = normalize_sum(alt_init, alt_n)
            else:
                alt_init = np.ones(alt_n) * S_alt / alt_n
            
            a4, s4 = run_topk_fista(alt_init, alt_n, S_alt, phase4_time, rng, k_peaks=30)
            obj4 = 2.0 * alt_n * s4 / (a4.sum() ** 2)
            print("  Alt length %d: %.6f" % (alt_n, obj4))
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
    
    # Phase 5: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        phase5_time = remaining * 0.5
        print("\n=== Phase 5: Final polish (%.0fs, n=%d) ===" % (phase5_time, n_p))
        
        a5, s5 = final_polish(global_a, n_p, S_p, phase5_time, rng)
        obj5 = 2.0 * n_p * s5 / (a5.sum() ** 2)
        print("  Final polish: %.6f" % obj5)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
        
        # One more round of adaptive on the polished result
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a6, s6 = run_adaptive_optimization(global_a, n_p, S_p, remaining2 * 0.6, rng)
            obj6 = 2.0 * n_p * s6 / (a6.sum() ** 2)
            print("  Post-polish adaptive: %.6f" % obj6)
            if obj6 < global_best:
                global_best = obj6
                global_a = a6.copy()
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("\nFinal verification FAILED, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = evaluate_sequence([float(x) for x in global_a])
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = evaluate_sequence([float(x) for x in global_a])
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("\nReturning sequence of length %d, best score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
