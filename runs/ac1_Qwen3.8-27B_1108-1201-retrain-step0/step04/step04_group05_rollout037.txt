"""
Minimize 2*n*max_conv/sum^2 using exact top-k gradient with Newton equalization,
Nesterov acceleration, periodic spectral domain moves, and backtracking line search.

Key differences from previous approach:
1. Exact max-gradient (top-k weighted average) instead of LSE smoothing
2. Newton equalization to directly reduce peak spread among active positions
3. Spectral domain equalization leveraging FFT structure
4. Adaptive top-k that grows as we converge
5. Nesterov momentum for faster convergence
6. Aggressive periodic restarts with structured perturbations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_topk_gradients(a, N, top_indices, k):
    """Compute gradients of c(m) for m in top_indices. Returns (k, n) array."""
    n = len(a)
    grads = np.zeros((k, n))
    for i, m in enumerate(top_indices):
        m = int(m)
        if m == 0:
            grads[i] = 2.0 * a
        elif m < n:
            # c(m) = sum_{j=0}^{n-1-m} a[j]*a[j+m]
            # dc/da[j] = a[j+m] (if j < n-m) + a[j-m] (if j >= m)
            if m > 0 and m < n:
                grads[i, m:] = a[:n - m]
                grads[i, :n - m] += a[m:]
            else:
                grads[i] = 0
        else:
            # m >= n, use symmetry: c(m) = c(2n-2-m)
            m_sym = 2 * n - 2 - m
            if m_sym == 0:
                grads[i] = 2.0 * a
            elif 0 < m_sym < n:
                grads[i, m_sym:] = a[:n - m_sym]
                grads[i, :n - m_sym] += a[m_sym:]
    return grads


def newton_equalize_direction(a, top_indices, k, top_vals):
    """
    Compute Newton direction to equalize top-k peaks.
    Returns a direction d such that c(m_i) + grad_i . d ≈ mean for all i.
    """
    n = len(a)
    if k < 2:
        return None

    grads = compute_topk_gradients(a, 0, top_indices, k)
    # grads shape: (k, n)

    # Use last as reference
    ref = k - 1
    target_diffs = top_vals[ref] - top_vals[:ref]  # (k-1,)

    # G[j, i] = grad_j . (grad_i - grad_ref) for j, i in 0..k-2
    G = np.zeros((k - 1, k - 1))
    for j in range(k - 1):
        diff_j = grads[j] - grads[ref]
        for i in range(k - 1):
            G[j, i] = np.dot(grads[j], grads[i]) - np.dot(grads[j], grads[ref])

    # Regularize for numerical stability
    G += 1e-10 * np.eye(k - 1)

    try:
        alpha = np.linalg.solve(G, target_diffs)
    except np.linalg.LinAlgError:
        alpha = np.linalg.lstsq(G, target_diffs, rcond=None)[0]

    # d = sum_i alpha_i * (grad_i - grad_ref)
    d = np.zeros(n)
    for i in range(k - 1):
        d += alpha[i] * (grads[i] - grads[ref])

    # Project out the constant component (to maintain sum)
    d = d - np.mean(d)

    # Check direction is meaningful
    if np.linalg.norm(d) < 1e-15:
        return None
    return d


def spectral_equalize_step(a, N, top_k, alpha):
    """Spectral domain equalization: reduce power at frequencies that feed into peaks."""
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]

    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k_val in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k_val) / N)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return project_simplex(a_new_full, S)


def backtracking_ls(a, direction, c_max, n, N, S, max_ls=20):
    """Backtracking line search on the simplex."""
    step = 1.0
    for _ in range(max_ls):
        candidate = project_simplex(a + step * direction, S)
        c = compute_conv(candidate, N)
        cm = c.max()
        if cm <= c_max + 1e-12:
            return candidate, cm
        step *= 0.5
    return candidate, cm


def optimize_loop(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=500):
    """
    Main optimization loop combining:
    - Top-k exact gradient descent with Nesterov momentum
    - Periodic Newton equalization
    - Periodic spectral equalization
    - Periodic random perturbations
    """
    a = project_simplex(a_init.copy(), S)
    y = a.copy()  # Nesterov momentum variable
    t_mom = 1.0
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    prev_cmax = np.inf
    stall_count = 0

    c = compute_conv(a, N)
    c_max = c.max()
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive top-k: grow as we converge
        effective_k = min(top_k_max, top_k_init + it // 20)

        # Compute current top-k
        top_idx = np.argsort(c)[::-1][:effective_k]
        top_vals = c[top_idx]
        c_max = top_vals[0]

        if c_max < best_cmax - 1e-10:
            best_cmax = c_max
            best_a = c.copy() if False else a.copy()
            stall_count = 0
        elif c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        else:
            stall_count += 1

        # Compute gradient: weighted average of top-k gradients
        # Use softmax weights for smoothness
        diff = top_vals - top_vals[0]
        if top_vals[0] > 1e-10:
            relative = np.maximum(diff / top_vals[0], -1.0)
            beta_w = 10.0 + 2.0 * (it % 200)  # adaptive sharpness
            w = np.exp(beta_w * relative)
            w = w / w.sum()
        else:
            w = np.ones(effective_k) / effective_k

        grads = compute_topk_gradients(a, N, top_idx, effective_k)
        # Weighted gradient
        grad = np.zeros(n)
        for i in range(effective_k):
            grad += w[i] * grads[i]

        # Newton equalization (every 5 iterations)
        if it % 5 == 0 and effective_k >= 3:
            d_newton = newton_equalize_direction(a, top_idx, effective_k, top_vals)
            if d_newton is not None and np.linalg.norm(d_newton) > 1e-10:
                # Combine gradient and Newton directions
                grad_dir = -grad / (np.linalg.norm(grad) + 1e-15)
                newton_dir = d_newton / (np.linalg.norm(d_newton) + 1e-15)
                # Check if Newton direction helps
                nw_dot = np.dot(newton_dir, -grad)
                if nw_dot > 0:

                else:
                    direction = grad_dir
            else:
                direction = -grad
        else:
            direction = -grad

        # Nesterov momentum update
        step_size = 2.0 / (1.0 + 0.5 * np.linalg.norm(grad) / max(c_max, 1e-10))
        step_size = min(step_size, 1.0)

        # Nesterov: update y
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        beta_mom = (t_mom - 1.0) / t_new

        # Take step from y (Nesterov lookahead)
        y_candidate = project_simplex(y + beta_mom * (a - y) + step_size * direction, S)

        # Line search
        a_new, cmax_new = backtracking_ls(y_candidate, 0.0, c_max, n, N, S, max_ls=10)

        # If line search failed to improve, try smaller step from a
        if cmax_new > c_max + 1e-10:
            a_new, cmax_new = backtracking_ls(a, 0.5 * direction, c_max, n, N, S, max_ls=10)
            if cmax_new > c_max + 1e-10:
                a_new, cmax_new = backtracking_ls(a, 0.1 * direction, c_max, n, N, S, max_ls=10)

        a = a_new
        t_mom = t_new

        # Recompute convolution
        c = compute_conv(a, N)
        c_max = c.max()

        if c_max < best_cmax - 1e-10:
            best_cmax = c_max
            best_a = a.copy()
            stall_count = 0
            if it % 50 == 0:
                elapsed = time.monotonic() - start_time
                print(f"  it={it}, c_max={c_max:.8f}, time={elapsed:.0f}s")

        # Spectral equalization (every 50 iterations)

            alpha_spec = min(alpha_spec, 0.5)

            c_spec = compute_conv(a_spec, N)
            if c_spec.max() < c_max:
                a = a_spec
                c = c_spec
                c_max = c.max()
                if c_max < best_cmax:

                    best_a = a.copy()

def make_init_ramp(n, rng):

    return np.clip(a, 0.01, None)


def make_init_cosine(n, rng):
    """Raised cosine window initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_bimodal(n, rng):

    a += 0.03 * rng.standard_normal(n)

def make_init_fourier_flat(n, rng, N):
    """Try to initialize with flat spectrum."""

    P[0] = n
    noise = rng.exponential(1.0, N)

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= len(given) <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print(f"n={n}, S={S:.4f}, N(FFT)={N}, budget={budget_s:.0f}s")

    global_best = np.inf
    global_a = None

    def report(val, label=""):
        nonlocal global_best, global_a
        if val < global_best:
            global_best = val
            if label:
                print(f"  [{label}] {val:.8f}")

    # Prepare initializations
    inits = []
    if given is not None and len(given) == n:
        g = np.clip(given, 0, 1000).astype(np.float64)
        if g.sum() > 0.01:
            inits.append(("given", g))

    inits.append(("ramp", make_init_ramp(n, rng)))
    inits.append(("cosine", make_init_cosine(n, rng)))
    inits.append(("bimodal", make_init_bimodal(n, rng)))
    inits.append(("random", make_init_random(n, rng)))
    inits.append(("fourier", make_init_fourier_flat(n, rng, N)))

    # --- Phase 1: Initial exploration (20% of time) ---
    print("=== Phase 1: Initial exploration ===")
    t_phase1 = time.monotonic()
    phase1_budget = min(400.0, (deadline - t_phase1) * 0.20)
    n_inits = len(inits)
    per_init = phase1_budget / n_inits

    best_from_phase1 = None
    best_val_phase1 = np.inf

    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        a_norm = project_simplex(arr, S)
        rem = max(2.0, per_init - (time.monotonic() - t_phase1) / n_inits * (inits.index((name, arr))) - 1.0)
        rem = min(rem, phase1_budget - (time.monotonic() - t_phase1) - 5.0)
        if rem < 2:
            rem = 2.0

        a_opt, val_opt = optimize_loop(a_norm, n, S, N, rem, rng,
                                        top_k_init=20, top_k_max=200)
        if val_opt < best_val_phase1:
            best_val_phase1 = val_opt
            best_from_phase1 = a_opt.copy()
        report(val_opt, f"P1/{name}")

    if best_from_phase1 is not None:
        global_a = best_from_phase1
    else:
        # Fallback: use given or random
        if given is not None and len(given) == n:
            global_a = project_simplex(np.clip(given, 0, 1000), S)
        else:
            global_a = project_simplex(make_init_ramp(n, rng), S)
        c_check = compute_conv(global_a, N)
        global_best = c_check.max()

    print(f"Phase 1 best: {global_best:.8f}")

    # --- Phase 2: Main optimization (50% of time) ---
    print("=== Phase 2: Main optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.50)

    # Run multiple rounds with increasing top-k
    a_curr = global_a.copy()
    n_rounds = 4
    round_budget = phase2_budget / n_rounds

    for r in range(n_rounds):
        if time.monotonic() > deadline - 180:
            break
        rem = min(round_budget, (deadline - 180) - time.monotonic())
        if rem < 5:
            break

        top_k_init_r = 30 + r * 30
        top_k_max_r = 150 + r * 100

        a_new, val_new = optimize_loop(a_curr, n, S, N, rem, rng,
                                        top_k_init=top_k_init_r,
                                        top_k_max=top_k_max_r)
        if val_new < global_best:
            global_best = val_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            print(f"  [P2/r{r}] {global_best:.8f}")
        else:
            # Try from the old best too
            a_alt, val_alt = optimize_loop(global_a, n, S, N, min(rem * 0.5, 60.0), rng,
                                            top_k_init=top_k_init_r,
                                            top_k_max=top_k_max_r)
            if val_alt < global_best:
                global_best = val_alt
                global_a = a_alt.copy()
                a_curr = a_alt.copy()
                print(f"  [P2/r{r}/alt] {global_best:.8f}")
            else:
                a_curr = a_new.copy()

    # --- Phase 3: Fine polishing (20% of time) ---
    print("=== Phase 3: Fine polishing ===")
    t3 = time.monotonic()
    phase3_budget = min(480.0, (deadline - t3) * 0.20)

    a_pol, val_pol = fine_polish(global_a, n, S, N, phase3_budget, rng)
    if val_pol < global_best:
        global_best = val_pol
        global_a = a_pol.copy()
        print(f"  [P3/polish] {global_best:.8f}")

    # --- Phase 4: Spectral refinement (10% of time) ---
    print("=== Phase 4: Spectral refinement ===")
    t4 = time.monotonic()
    remaining = deadline - t4 - 10.0
    if remaining > 10:
        a = project_simplex(global_a, S)
        best_sp = a.copy()
        best_sp_val = compute_conv(a, N).max()
        start_sp = time.monotonic()

        while time.monotonic() - start_sp < remaining:
            for alpha in [0.05, 0.1, 0.15, 0.2]:
                for tk in [10, 20, 40, 80]:
                    a_trial = spectral_equalize_step(a, N, tk, alpha)
                    c_trial = compute_conv(a_trial, N)
                    if c_trial.max() < best_sp_val:
                        best_sp_val = c_trial.max()
                        best_sp = a_trial.copy()
                        a = a_trial.copy()
            # Small random perturbation
            if rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

        if best_sp_val < global_best:
            global_best = best_sp_val
            global_a = best_sp.copy()
            print(f"  [P4/spectral] {global_best:.8f}")

        # Final polish after spectral
        rem_final = deadline - time.monotonic() - 5.0
        if rem_final > 10:
            a_fin, val_fin = fine_polish(global_a, n, S, N, rem_final, rng)
            if val_fin < global_best:
                global_best = val_fin
                global_a = a_fin.copy()
                print(f"  [P4/final] {global_best:.8f}")

    # --- Verification and return ---
    print(f"=== Final: internal best = {global_best:.8f} ===")

    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"Verification: {ver:.8f}")
        else:
            print("Verification failed, using fallback")
            final_list = [1.0] * 256
            ver = evaluate_sequence(final_list)
            print(f"Fallback: {ver:.8f}")
    else:
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)

    print(f"Returning length {len(final_list)}, eval={ver:.8f}")
    return final_list
