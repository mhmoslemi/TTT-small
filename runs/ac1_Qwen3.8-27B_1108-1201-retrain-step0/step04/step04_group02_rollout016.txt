"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Hybrid of Nesterov-accelerated LSE-smoothed gradient descent,
LP-inspired direction finding via multiplicative weights, and cyclic beta
annealing to escape local minima. Exploits FFT-based operations for
efficiency. Key innovation: alternating between gradient-based and
LP-direction steps, with the LP direction providing a more globally-aware
search direction that respects the non-convex structure.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_fft(a, c, beta, fsize, A_cached):
    """Compute LSE-smoothed gradient of max(a*a) w.r.t. a using FFT."""
    n = len(a)
    size = 2 * n - 1
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: grad[j] = sum_k w[k] * a[k-j] for valid k-j
    # This is cross-correlation (w _ a)[j] = sum_m w[m+j] * a[m]
    # Using FFT: ifft(fft(w) * conj(fft(a)))[:n]
    V = fft(w, fsize)
    corr = np.real(ifft(V * np.conj(A_cached)))[:n]
    return corr


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_direction_fft(a, c, M, n, max_iter=50, eta_decay=0.95):
    """
    LP-inspired direction: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k.
    Uses multiplicative weights on the violated constraints.
    Returns the normalized direction (with sum = sqrt(2n)).
    """
    S = np.sqrt(2.0 * n)
    g = a.copy()
    size = 2 * n - 1
    fsize = 1 << (size + n - 1).bit_length()
    A_cached = fft(a, fsize)

    eta = 1.0
    for it in range(max_iter):
        # Compute a * g
        G = fft(g, fsize)
        ag = np.real(ifft(A_cached * G))[:size]
        violations = np.maximum(ag - M, 0.0)
        max_viol = violations.max()
        if max_viol < 1e-12 * M:
            break

        # Update g: reduce where it contributes to violations
        # red[j] = sum_k violations[k] * a[k-j] = (violations _ a)[j]
        V = fft(violations, fsize)
        red = np.real(ifft(V * np.conj(A_cached)))[:n]

        # Multiplicative update: g *= exp(-eta * red / max_viol)
        # This is equivalent to reducing g more where it causes more violations
        scale = eta * red / (max_viol + 1e-300)
        scale = np.clip(scale, -10, 10)
        g = g * np.exp(-scale)

        # Clamp to prevent underflow
        g = np.maximum(g, 1e-300)
        eta *= eta_decay

    gs = g.sum()
    if gs < 1e-15:
        return a.copy()
    return g * (S / gs)


def run_gradient_descent(a_init, n, S, max_time, rng, use_nesterov=True,
                         beta_min=5.0, beta_max=500.0, beta_cycle=500):
    """
    Nesterov-accelerated LSE-smoothed gradient descent.
    Cyclic beta schedule for exploration-exploitation balance.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    a = project_simplex(a, S)
    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    c = fft_convolve_self(a)

    it = 0
    last_improve = 0
    step_base = 5.0 / S

    while time.monotonic() - start < max_time:
        it += 1

        # Cyclic beta schedule
        phase = it % beta_cycle
        if phase < beta_cycle // 4:
            beta = beta_min + (beta_max - beta_min) * phase / (beta_cycle // 4)
        elif phase < beta_cycle // 2:
            beta = beta_max
        elif phase < 3 * beta_cycle // 4:
            beta = beta_max - (beta_max - beta_min) * (phase - beta_cycle // 2) / (beta_cycle // 4)
        else:
            beta = beta_min

        # Compute LSE gradient at y (Nesterov lookahead)
        A_cached = fft(y, fsize)
        c_y = fft_convolve_self(y)
        grad = lse_gradient_fft(y, c_y, beta, fsize, A_cached)

            y = x_new + ((t - 1.0) / t_new) * (x_new - a)

            t = t_new

    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    a = project_simplex(a, S)
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    c = fft_convolve_self(a)

    it = 0
    step = 0.05
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Every 10 iterations, try a balancing move
        if it % 10 == 0:
            # Get top peak lags
            top_lags = np.argsort(c)[::-1][:5]
            a_bal = a.copy()
            for k in top_lags:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                # Find the most imbalanced pair
                i_vals = np.arange(lo, hi + 1)
                pairs = a_bal[i_vals] * a_bal[k - i_vals]
                # Equalize the pair with max product
                idx_max = np.argmax(pairs)
                i = i_vals[idx_max]
                j = k - i
                if i != j and 0 <= j < n:
                    diff = a_bal[i] - a_bal[j]
                    transfer = 0.3 * diff
                    a_bal[i] -= transfer
                    a_bal[j] += transfer
                    a_bal = np.maximum(a_bal, 0)
            a_bal = project_simplex(a_bal, S)
            c_bal = fft_convolve_self(a_bal)
            obj_bal = 2.0 * n * c_bal.max() / S**2
            if obj_bal < cur_obj:
                a = a_bal
                c = c_bal
                continue

        # Exact subgradient (top-1)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[i_vals] = a[k_star - i_vals]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2

        if obj_try < cur_obj:
            a = a_try
            c = c_try
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.8

        # Perturbation if stuck
        if it - last_improve > 1000:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            step = 0.05

    return best_a, best_obj


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Cosine-based (close to optimal structure)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))

    # Asymmetric (exploits the non-symmetry of the problem)
    inits.append(np.maximum(np.exp(-3 * tt), 0.01))
    inits.append(np.maximum(np.exp(-3 * (1 - tt)), 0.01))
    inits.append(np.maximum(1 + np.sin(np.pi * tt), 0.01))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.cos(6 * np.pi * tt))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 3 * tt) + 0.1 * np.sin(2 * np.pi * 5 * tt + 0.3))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.1)

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max_conv/sum^2.
    Uses multi-phase optimization: gradient descent, LP-hybrid, and fine tuning.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 16384:
        n = len(given)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))
    t0 = time.monotonic()

    # Collect initializations
    inits = []
    if given is not None:
        inits.append(("given", given.copy()))
        # Also try a perturbed version
        pert = project_simplex(given + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
        inits.append(("given_pert", pert))

    gen_inits = generate_diverse_inits(n, rng)
    for i, g in enumerate(gen_inits[:4]):
        inits.append(("gen_%d" % i, g))

    n_inits = len(inits)
    elapsed_check = time.monotonic() - t0
    usable_time = max(120.0, deadline - time.monotonic())

    # Phase 1: Quick evaluation of initializations (10% of time)
    phase1_time = usable_time * 0.10
    print("=== Phase 1: Quick eval (%d inits, %.0fs) ===" % (n_inits, phase1_time))

    p1_start = time.monotonic()
    best_init = None
    best_init_obj = np.inf

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - usable_time * 0.5:
            break
        arr_n = len(arr)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        a_norm = project_simplex(arr, S_i)
        obj = compute_objective(a_norm)
        print("  [%s] n=%d: obj=%.6f" % (name, arr_n, obj))
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
            global_best = obj
            global_a = a_norm.copy()

    if best_init is None:
        best_init = np.ones(n) * S / n
        global_a = best_init.copy()
        global_best = compute_objective(best_init)

    print("Phase 1 done: best=%.6f, elapsed=%.0fs" % (
        global_best, time.monotonic() - p1_start))

    # Phase 2: Nesterov gradient descent (35% of time)
    phase2_time = usable_time * 0.35
    remaining = deadline - time.monotonic()
    phase2_time = min(phase2_time, remaining * 0.5)
    print("=== Phase 2: Nesterov gradient descent (%.0fs) ===" % phase2_time)

    a_opt, obj_opt = run_gradient_descent(
        global_a, n, S, phase2_time, rng,
        use_nesterov=True, beta_min=5.0, beta_max=400.0, beta_cycle=1000)

    if obj_opt < global_best:
        global_best = obj_opt
        global_a = a_opt.copy()
        print("  After Nesterov: %.6f" % global_best)

    # Phase 3: LP-hybrid (30% of time)
    remaining = deadline - time.monotonic()
    phase3_time = min(usable_time * 0.30, remaining * 0.6)
    print("=== Phase 3: LP-hybrid (%.0fs) ===" % phase3_time)

    if phase3_time > 20:
        a_opt, obj_opt = run_lp_hybrid(
            global_a, n, S, phase3_time, rng,
            lp_iters=80, beta_min=50.0, beta_max=600.0)

        if obj_opt < global_best:
            global_best = obj_opt
            global_a = a_opt.copy()
            print("  After LP-hybrid: %.6f" % global_best)

    # Phase 4: Fine tuning (15% of time)
    remaining = deadline - time.monotonic()
    phase4_time = min(usable_time * 0.15, remaining * 0.7)
    print("=== Phase 4: Fine tuning (%.0fs) ===" % phase4_time)

    if phase4_time > 15:
        a_opt, obj_opt = run_fine_tuning(global_a, n, S, phase4_time, rng)
        if obj_opt < global_best:
            global_best = obj_opt
            global_a = a_opt.copy()
            print("  After fine tuning: %.6f" % global_best)

    # Phase 5: One more round of gradient with different parameters
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("=== Phase 5: Final gradient pass (%.0fs) ===" % remaining)
        # Use different beta range for exploration
        a_opt, obj_opt = run_gradient_descent(
            global_a, n, S, remaining * 0.7, rng,
            use_nesterov=True, beta_min=20.0, beta_max=1000.0, beta_cycle=500)
        if obj_opt < global_best:
            global_best = obj_opt
            global_a = a_opt.copy()
            print("  After final pass: %.6f" % global_best)

    # Final fine-tune with whatever time is left
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, obj_opt = run_fine_tuning(global_a, n, S, remaining * 0.8, rng)
        if obj_opt < global_best:
            global_best = obj_opt
            global_a = a_opt.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
