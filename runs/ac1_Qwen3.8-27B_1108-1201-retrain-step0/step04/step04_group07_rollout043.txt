```
def project_simplex(v, S):
    v=np.asarray(v,dtype=np.float64).copy()
    n=v.size
    if n==0: return v
    w=np.sort(v)[::-1]
    csw=np.cumsum(w)
    rho_candidates=(csw-S)/np.arange(1,n+1)
    valid=rho_candidates <= w
    if not np.any(valid):
        return np.full(n, S/n)
    rho_idx=np.max(np.where(valid)[0])
    theta=(csw[rho_idx]-S)/(rho_idx+1)
    return np.maximum(v-theta,0)
```

```
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    cs = c.sum()
    if not np.isfinite(cs) or cs <= 0:
        return a.copy()
    c = c / cs
    if mode == 1:
       mean_c = c.mean()
       cap = max(1e-12, float(param) * mean_c)
       col = c / (1.0 + c / cap)
    elif mode == 2:
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for nonnegative sequences.

The optimizer combines:
1. Several strong structured starts, including endpoint/taper parametric guesses,
   the finite square-root of a flat target polynomial, and a central-binomial start.
2. True spectral flattening in the A^2 domain (not |A|^2), including soft/hard
   peak caps on the coefficient vector of a*a.
3. Smoothed log-sum-exp maximization using a multiplicative projected gradient
   update on the simplex, with annealed temperature, line search, and random
   perturbations to escape local minima.
4. Multi-length search and interpolation between lengths.
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


def normalize_nonnegative(v, S=1.0):
    v = np.array(v, dtype=np.float64, copy=True)
    if v.size == 0:
        return v
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(v.size, S / v.size, dtype=np.float64)
    v *= (S / s)
    return v


def next_fft_len(n):
    L = 2 * n - 1
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def fft_convolve_self(a, N):
    a = np.asarray(a, dtype=np.float64)
    A = fft(a, N)
    c = ifft(A * A)
    return c.real[:2 * len(a) - 1]


def score_normalized(a, N):
    """Score assuming sum(a) == 1 and N valid FFT length."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = fft_convolve_self(a, N)
    if c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.inf
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax)


def score_sequence(a, N=None):
    """Safe score for arbitrary finite sequence."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = normalize_nonnegative(a, 1.0)
    if N is None:
        N = next_fft_len(a.size)
    return score_normalized(a, N)


def compute_c_grad(a, N, tau):
    """
    Compute c = a*a (linear convolution via FFT), normalized log-sum-exp weights
    for c, and the gradient of that log-sum-exp objective w.r.t. a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    A = fft(a, N)
    c = ifft(A * A).real[:L]
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)

    cmax = float(c.max()) if L > 0 else 0.0

    if L <= 0:
        weights = np.array([], dtype=np.float64)
    elif cmax <= 1e-300 or tau <= 0.0:
        weights = np.ones(L, dtype=np.float64) / L
    else:
        ex = tau * (c - cmax)
        ex = np.clip(ex, -745.0, 0.0)
        w = np.exp(ex)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            w = np.zeros(L, dtype=np.float64)
            w[0] = 1.0
        else:
            w /= ws
        weights = w

    W = fft(weights, N)
    grad = 2.0 * ifft(np.conj(W) * A).real[:n]
    grad[~np.isfinite(grad)] = 0.0
    np.maximum(grad, 0.0, out=grad)

    score = float(2.0 * n * cmax)
    return c, grad, score


def make_mul_candidate(a, grad, eta0, damp):
    """
    Multiplicative simplex update in the direction -grad.
    grad is expected nonnegative; all coordinates are damped, with larger
    damping for larger gradient components.
    """
    a = np.asarray(a, dtype=np.float64)
    gmax = float(np.max(grad)) if grad.size else 0.0
    if gmax <= 1e-300:
        return a.copy()

    x = grad / gmax
    x = np.clip(x, -10.0, 10.0)

    cond = 1.0 + eta0 * x
    cond = np.maximum(cond, 1e-8)

    new = a / cond
    s = float(new.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    new /= s

    out = (1.0 - damp) * a + damp * new
    np.maximum(out, 0.0, out=out)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    return out / s


def sqrt_flatten_step(a, N, mode, param, base, blend):
    """
    One global step:
      c = a*a,
      transform c to a flatter target,
      take the FFT square root (A^2 = target FFT),
      project back to nonnegative/support, and blend.
    mode:
      0: power compression target = c^p (p=param)
      1: soft cap target = c / (1 + c / cap)
      2: hard cap target = min(c, cap)
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    c = fft_convolve_self(a, N)
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)

    cs = float(c.sum())
    if not np.isfinite(cs) or cs <= 0.0:
        return a.copy()
    c = c / cs

    if mode == 1:
        mean_c = float(c.mean())
        cap = max(1e-12, float(param) * mean_c)
        col = c / (1.0 + c / cap)
    elif mode == 2:
        mean_c = float(c.mean())
        cap = max(1e-12, float(param) * mean_c)
        col = np.minimum(c, cap)
    else:
        p = float(param)
        if p <= 0.0:
            col = np.ones(L, dtype=np.float64) / L
        else:
            with np.errstate(divide="ignore", invalid="ignore"):
                col = np.power(c, p)
            col[~np.isfinite(col)] = 0.0

    s = float(col.sum())
    if not np.isfinite(s) or s <= 0.0:
        col = np.ones(L, dtype=np.float64) / L
    else:
        col = col / s

    if base > 0.0:
        col = (1.0 - base) * col + (base / L)

    C = fft(col, N)
    C[0] = C[0].real
    if N % 2 == 0:
        C[N // 2] = C[N // 2].real

    A = np.sqrt(C)
    A[0] = abs(A[0])

    raw = np.real(ifft(A))[:n]
    proj = normalize_nonnegative(raw, 1.0)

    if blend >= 1.0:
        return proj

    out = (1.0 - blend) * a + blend * proj
    return normalize_nonnegative(out, 1.0)


def run_sqrt_phase(a, N, budget, rng):
    """Run several spectral square-root flattening steps."""
    a = normalize_nonnegative(a, 1.0)
    best_a = a.copy()
    best_score = score_normalized(a, N)
    cur_score = best_score

    if budget <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()
    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= budget:
            break
        progress = min(1.0, elapsed / max(1e-9, budget))

        mode = int(rng.integers(0, 3))
        if mode == 0:
            if progress < 0.3:
                param = float(rng.choice((0.0, 0.25, 0.5)))
            elif progress < 0.6:
                param = float(rng.choice((0.25, 0.5, 0.75)))
            else:
                param = float(rng.choice((0.5, 0.75, 0.9)))
        elif mode == 1:
            param = float(rng.uniform(1.0, 1.0 + 20.0 * progress))
        else:
            param = float(rng.uniform(1.0, 1.0 + 15.0 * progress))

        base = 0.02 + 0.25 * (1.0 - progress)
        blend = 0.25 + 0.45 * (1.0 - progress)

        a_new = sqrt_flatten_step(a, N, mode, param, base, blend)
        sc_new = score_normalized(a_new, N)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        allow_worsening = 1.0 + 0.02 * (1.0 - progress)
        if sc_new <= cur_score * allow_worsening or progress < 0.15:
            a = a_new
            cur_score = sc_new
        else:
            if rng.random() < 0.6:
                a = best_a.copy()
                cur_score = best_score
            else:
                a = normalize_nonnegative((1.0 - blend) * a + blend * a_new, 1.0)
                cur_score = score_normalized(a, N)

        if best_score <= 1.499:
            break

    return best_a, best_score


def multiply_smooth_optimize(a, N, budget, rng):
    """
    Smooth max-coefficient minimization using log-sum-exp weights and a
    multiplicative projected update on the simplex.
    """
    a = normalize_nonnegative(a, 1.0)
    n = a.size

    best_a = a.copy()
    best_score = score_normalized(a, N)
    cur_score = best_score

    if budget <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()
    stale = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= budget:
            break
        progress = min(1.0, elapsed / max(1e-9, budget))

        # Temperature must be roughly proportional to n because c values scale like 1/n.
        tau = max(1000.0, (2.0 + 200.0 * progress) * n)
        if tau > 1.0e7:
            tau = 1.0e7

        eta0 = 0.55 * (1.0 - 0.45 * progress) + 0.05
        damp = 0.65 - 0.35 * progress

        c, grad, cur_score = compute_c_grad(a, N, tau)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stale = 0

        gmax = float(np.max(grad)) if grad.size else 0.0
        if gmax <= 1e-300:
            break

        cand_best = a
        cand_best_score = cur_score
        moved = False

        # Line search over a few step sizes.  Only immediate improvements are
        # generally accepted; a small amount of simulated annealing is allowed.
        for mult in (0.35, 0.7, 1.4):
            if time.monotonic() - t0 >= budget:
                break
            cand = make_mul_candidate(a, grad, eta0 * mult, damp)
            sc = score_normalized(cand, N)
            if sc < cand_best_score:
                cand_best_score = sc
                cand_best = cand
                moved = True

        if not moved and rng.random() < 0.05:
            mult = float(rng.choice((0.2, 0.5, 1.0, 2.0)))
            cand = make_mul_candidate(a, grad, eta0 * mult, damp)
            sc = score_normalized(cand, N)
            tolerant = 1.0 + 0.005 * (1.0 - progress)
            if sc < cur_score * tolerant:
                cand_best_score = sc
                cand_best = cand
                moved = True

        if moved:
            a = cand_best
            cur_score = cand_best_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1

        if cur_score > best_score * 1.003 and stale > 30:
            a = best_a.copy()
            cur_score = best_score
            stale = 0

        if stale > 150:
            if rng.random() < 0.65:
                noise_std = 0.08 + 0.15 * (1.0 - progress)
                a = normalize_nonnegative(a * np.exp(rng.normal(0.0, noise_std, n)), 1.0)
                cur_score = score_normalized(a, N)
            else:
                a_new = sqrt_flatten_step(a, N, 0, 0.5, 0.1, 0.3)
                sc_new = score_normalized(a_new, N)
                if sc_new < cur_score or rng.random() < 0.2:
                    a = a_new
                    cur_score = sc_new
            stale = 0

        if best_score <= 1.499:
            break

    return best_a, best_score


def interpolate_sequence(x, m):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.full(m, 1.0 / max(1, m), dtype=np.float64)
    if n == m:
        return normalize_nonnegative(x, 1.0)
    if n < 2:
        val = float(x[0]) if np.isfinite(x[0]) and x[0] > 0.0 else 1.0 / max(1, m)
        return np.full(m, val, dtype=np.float64)

    if n > max(100000, 10 * m):
        idx = np.linspace(0, n - 1, m, dtype=np.float64).round().astype(np.int64)
        y = x[idx]
    else:
        old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        new = np.linspace(0.0, 1.0, m, dtype=np.float64)
        y = np.interp(new, old, x, 0.0, 0.0)

    return normalize_nonnegative(y, 1.0)


def uniform_sqrt_init(n, N):
    """Take the Fourier square root of a flat target convolution."""
    L = 2 * n - 1
    target = np.ones(L, dtype=np.float64) / L
    C = fft(target, N)
    C[0] = C[0].real
    if N % 2 == 0:
        C[N // 2] = C[N // 2].real

    A = np.sqrt(C)
    A[0] = abs(A[0])
    raw = np.real(ifft(A))[:n]
    s_pos = float(np.maximum(raw, 0.0).sum())
    if s_pos < 1e-7:
        return np.ones(n, dtype=np.float64) / n
    return normalize_nonnegative(raw, 1.0)


def central_binomial_init(n):
    """First n formal square-root coefficients of 1/(1-z): binom(2k,k)/4^k."""
    p = np.empty(n, dtype=np.float64)
    p[0] = 1.0
    if n > 1:
        k = np.arange(1, n, dtype=np.float64)
        ratios = (2.0 * k - 1.0) / (2.0 * k)
        p[1:] = np.cumprod(ratios)
    return normalize_nonnegative(p, 1.0)


def generate_parametric(n, max_time, rng):
    """
    Candidate densities of the form
       a_i ∝ (i+0.5)^(-alpha) * (n-i+0.5)^(beta) * exp(gamma*x + ...),
    with endpoint singular/taper behavior and mild global modulation.
    """
    out = []
    i = np.arange(n, dtype=np.float64) + 0.5
    rn = n + 0.5 - i
    x = i / float(n)

    left_log = np.log(i)
    right_log = np.log(rn)

    alphas = (
        0.0, 0.05, 0.10, 0.15, 0.20, 0.25,
        0.30, 0.35, 0.40, 0.42, 0.45,
        0.47, 0.48, 0.485, 0.49, 0.495
    )
    betas = (
        -0.15, -0.05, 0.0, 0.02,
        0.05, 0.10, 0.20, 0.40
    )
    gammas = (-0.40, -0.15, 0.0, 0.15, 0.40)

    t0 = time.monotonic()
    count = 0
    hard_cap = 180

    for alpha in alphas:
        left = -float(alpha) * left_log
        for beta in betas:
            right = float(beta) * right_log
            logbase = left + right
            for gamma in gammas:
                if count >= hard_cap or (max_time > 0.0 and time.monotonic() - t0 > max_time):
                    return out
                logw = logbase + float(gamma) * (x - 0.5)
                m = float(logw.max())
                w = np.exp(logw - m)
                out.append(normalize_nonnegative(w, 1.0))
                count += 1

    # A few random low-dimensional parameter settings.
    for _ in range(50):
        if count >= hard_cap + 50 or (max_time > 0.0 and time.monotonic() - t0 > max_time):
            break
        alpha = float(rng.uniform(0.0, 0.495))
        beta = float(rng.uniform(-0.20, 0.50))
        gamma = float(rng.uniform(-0.60, 0.60))
        delta = float(rng.uniform(-1.0, 1.0))

        logw = -alpha * left_log + beta * right_log
        logw = logw + gamma * (x - 0.5) + delta * (x - 0.5) * (x - 0.5)
        m = float(logw.max())
        w = np.exp(logw - m)
        out.append(normalize_nonnegative(w, 1.0))
        count += 1

    return out


def optimize_length(n, budget, rng, extra_inits=None):
    """Full search pipeline for one fixed sequence length n."""
    N = next_fft_len(n)
    budget = max(0.0, float(budget))

    candidates = []

    if extra_inits is not None:
        for arr in extra_inits:
            arr = np.asarray(arr, dtype=np.float64)
            if arr.size == n:
                candidates.append(normalize_nonnegative(arr, 1.0))
            elif arr.size >= 2:
                candidates.append(interpolate_sequence(arr, n))

    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if provided.size == n:
            candidates.append(normalize_nonnegative(provided, 1.0))
        elif provided.size >= 512:
            candidates.append(interpolate_sequence(provided, n))
    except Exception:
        pass

    candidates.append(np.ones(n, dtype=np.float64) / float(n))

    usq = uniform_sqrt_init(n, N)
    candidates.append(usq)
    candidates.append(normalize_nonnegative(0.5 * (usq + usq[::-1]), 1.0))

    cb = central_binomial_init(n)
    candidates.append(cb)
    candidates.append(normalize_nonnegative(0.5 * (cb + cb[::-1]), 1.0))

    max_time_grid = max(0.3, min(4.0, 0.05 * budget + 0.3))
    candidates.extend(generate_parametric(n, max_time_grid, rng))

    scored = []
    score_arr = []
    t_init = time.monotonic()
    max_time_init = max(0.5, min(5.0, 0.05 * budget + 0.5))
    max_eval = 160 if n <= 16384 else 110

    for idx, arr in enumerate(candidates):
        if idx >= max_eval:
            break
        if time.monotonic() - t_init > max_time_init:
            break
        a = np.asarray(arr, dtype=np.float64)
        if a.size != n:
            continue
        sc = score_normalized(a, N)
        if np.isfinite(sc):
            scored.append(a)
            score_arr.append(float(sc))

    if not scored:
        a = np.ones(n, dtype=np.float64) / float(n)
        return a, score_normalized(a, N)

    score_array = np.array(score_arr, dtype=np.float64)
    order = np.argsort(score_array)

    top_pairs = []
    for idx in order:
        if len(top_pairs) >= 6:
            break
        ii = int(idx)
        top_pairs.append((float(score_array[ii]), scored[ii]))

    best_score = top_pairs[0][0]
    best_a = top_pairs[0][1].copy()
    print("  n=%d initial best=%.6f" % (n, best_score))

    if budget > 5.0:
        t_opt = time.monotonic()
        fractions = (0.42, 0.24, 0.18, 0.12, 0.04)[:len(top_pairs)]

        for idx, frac in enumerate(fractions):
            elapsed_local = time.monotonic() - t_opt
            rem = budget - elapsed_local
            if rem < 20.0:
                break

            start_b = min(budget * frac, rem * 0.9)
            if start_b < 5.0:
                break

            init_score, init_a = top_pairs[idx]
            start_a = init_a.copy()

            b_sqrt = start_b * 0.18
            if b_sqrt > 2.0:
                a_sqrt, s_sqrt = run_sqrt_phase(start_a, N, b_sqrt, rng)
                if s_sqrt < best_score:
                    best_score = s_sqrt
                    best_a = a_sqrt.copy()
                if s_sqrt < init_score:
                    start_for_mul = a_sqrt.copy()
                else:
                    start_for_mul = start_a
            else:
                start_for_mul = start_a

            b_mul = start_b - (b_sqrt if b_sqrt > 2.0 else 0.0)
            if b_mul > 1.0:
                a_mul, s_mul = multiply_smooth_optimize(start_for_mul, N, b_mul, rng)
                if s_mul < best_score:
                    best_score = s_mul
                    best_a = a_mul.copy()

            if best_score <= 1.499:
                break

        # Final polishing on the best object found at this length.
        rem = budget - (time.monotonic() - t_opt)
        if rem > 20.0:
            a_final, s_final = multiply_smooth_optimize(best_a, N, rem * 0.70, rng)
            if s_final < best_score:
                best_score = s_final
                best_a = a_final.copy()

            rem2 = budget - (time.monotonic() - t_opt)
            if rem2 > 15.0:
                a_sqrt2, s_sqrt2 = run_sqrt_phase(best_a, N, rem2 * 0.35, rng)
                if s_sqrt2 < best_score:
                    best_score = s_sqrt2
                    best_a = a_sqrt2.copy()

                rem3 = budget - (time.monotonic() - t_opt)
                if rem3 > 10.0:
                    a_final2, s_final2 = multiply_smooth_optimize(best_a, N, rem3 * 0.80, rng)
                    if s_final2 < best_score:
                        best_score = s_final2
                        best_a = a_final2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    t_start = time.monotonic()
    safe_budget = max(60.0, float(budget_s) - 80.0)
    deadline = t_start + safe_budget
    rng = np.random.default_rng(seed)

    provided = None
    provided_size = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if provided.size > 0:
            provided_size = int(provided.size)
    except Exception:
        provided = None
        provided_size = None

    best_global = None
    best_score = np.inf
    best_by_length = {}
    score_by_length = {}

    # Use any provided previous construction as a safe baseline.
    if provided is not None and provided.size > 0:
        p = normalize_nonnegative(provided, 1.0)
        L = p.size
        if 512 <= L <= 65536:
            sc = score_sequence(p)
            if np.isfinite(sc):
                best_score = float(sc)
                best_global = p.copy()
                best_by_length[L] = p.copy()
                score_by_length[L] = float(sc)
                print("Provided baseline length=%d score=%.6f" % (L, sc))

    lengths = set()
    for L in (4096, 8192, 16384, 32768):
        lengths.add(L)
    if provided_size is not None and 2048 <= provided_size <= 32768:
        lengths.add(provided_size)

    if float(budget_s) < 300:
        ordered = []
        for L in (4096, 8192):
            if L in lengths and L not in ordered:
                ordered.append(L)
        if not ordered:
            ordered = [8192]
    else:
        ordered = []
        for L in (8192, 16384, 32768):
            if L in lengths and L not in ordered:
                ordered.append(L)
        if provided_size is not None and 2048 <= provided_size <= 32768:
            if provided_size not in ordered:
                ordered.append(provided_size)
        for L in sorted(lengths):
            if L not in ordered and 2048 <= L <= 32768 and L >= 8192:
                ordered.append(L)
        if not ordered:
            ordered = [8192]

    frac_map = {
        4096: 0.25,
        8192: 0.20,
        16384: 0.45,
        32768: 0.30,
    }

    total_frac = 0.0
    for L in ordered:
        if L in frac_map:
            total_frac += frac_map[L]
        elif provided_size is not None and L == provided_size:
            total_frac += 0.25
        else:
            total_frac += 0.15
    if total_frac <= 0.0:
        total_frac = 1.0

    for L in ordered:
        now = time.monotonic()
        rem = deadline - now
        if rem < 30.0:
            break

        if len(ordered) == 1:
            alloc = max(0.0, rem - 20.0)
        else:
            frac = frac_map.get(L)
            if frac is None:
                frac = 0.25 if (provided_size is not None and L == provided_size) else 0.15
            alloc = min(rem - 20.0, safe_budget * frac / total_frac)

        if alloc < 15.0:
            continue

        extra = []
        if provided is not None and provided.size >= 64:
            extra.append(interpolate_sequence(provided, L))
        for prev_L, arr in best_by_length.items():
            if prev_L != L and arr is not None and arr.size >= 128:
                extra.append(interpolate_sequence(arr, L))

        print("Optimizing length %d with budget %.1fs" % (L, alloc))
        res = optimize_length(L, alloc, rng, extra_inits=extra)
        if res is None:
            continue

        a_res, s_res = res
        if np.isfinite(s_res):
            s_res = float(s_res)
            a_res = normalize_nonnegative(a_res, 1.0)
            best_by_length[L] = a_res.copy()
            score_by_length[L] = s_res
            if s_res < best_score:
                best_score = s_res
                best_global = a_res.copy()
        print("  length=%d best score=%.6f" % (L, s_res))

        if best_score <= 1.498:
            # Still finish quickly if possible; more unlikely to matter much.
            if time.monotonic() >= deadline - 60.0:
                break

    if best_global is None:
        n_fb = 8192 if safe_budget > 300.0 else 4096
        best_global = np.ones(n_fb, dtype=np.float64) / float(n_fb)
        best_score = score_sequence(best_global)
        best_by_length[n_fb] = best_global.copy()
        score_by_length[n_fb] = float(best_score)

    # Final selection: prefer the best score, but if a smaller length is
    # essentially tied, choose the smaller one to keep future evaluation safer.
    final_candidates = []
    seen_lengths = set()
    for L in sorted(score_by_length.keys()):
        arr = best_by_length.get(L)
        if arr is None:
            continue
        if L in seen_lengths:
            continue
        final_candidates.append((float(score_by_length[L]), int(L), arr))
        seen_lengths.add(L)

    if best_global is not None:
        Lg = int(best_global.size)
        if Lg not in seen_lengths:
            final_candidates.append((float(best_score), Lg, best_global))
            seen_lengths.add(Lg)

    if len(final_candidates) == 1:
        chosen = final_candidates[0]
    else:
        score_list = []
        for item in final_candidates:
            score_list.append(float(item[0]))

        order = np.argsort(np.array(score_list, dtype=np.float64))
        best_candidate_score = float(score_list[int(order[0])])
        chosen_idx = int(order[0])

        for idx in order:
            ii = int(idx)
            if score_list[ii] <= best_candidate_score + 0.001:
                L_i = final_candidates[ii][1]
                L_c = final_candidates[chosen_idx][1]
                if L_i < L_c:
                    chosen_idx = ii

        chosen = final_candidates[chosen_idx]

    final_arr = normalize_nonnegative(chosen[2], 1.0)
    if final_arr.sum() < 0.01:
        final_arr = np.ones(8192, dtype=np.float64)

    out = []
    for val in final_arr:
        out.append(float(val))

    print("Selected length=%d, internal score=%.6f" % (len(final_arr), chosen[0]))
    return out
```
