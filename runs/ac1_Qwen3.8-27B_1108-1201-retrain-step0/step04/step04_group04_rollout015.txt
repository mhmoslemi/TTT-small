"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses hierarchical multigrid construction + FISTA + peak shaving.
Under normalization sum=sqrt(2n), this equals minimizing max_conv.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_smooth_max(c, beta):
    """Log-sum-exp approximation to max(c), returns value and gradient."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros_like(c)
    w = e / tot
    return cm + np.log(tot) / beta, w

def fista_minimize(a_init, n, max_time, rng, beta_init=50.0, beta_max=2000.0):
    """FISTA to minimize max(conv(a)) subject to a>=0, sum(a)=sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.05 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot  # weights on the convolution
        
        # Gradient of LSE w.r.t. a: 2 * (w * a) circular (truncated to n)
        # More precisely: grad_j = 2 * sum_k w_k * (conv_derivative)_k_j
        # conv_k = sum_i a_i * a_{k-i}, so d(conv_k)/da_j = a_{k-j} + a_{j-k} (with bounds)
        # Using FFT: grad = 2 * ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
        fsize = 1 << (2 * n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = S / (gnorm * 1.0)
        c_old = fft_convolve_self(y)
        max_old = c_old.max()
        
        for _ in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            max_new = c_new.max()
            if max_new < max_old - 1e-10:
                break
            step *= 0.5
        
        if step < 1e-10:
            # Can't improve, add noise
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            x_new = project_simplex(y + noise, S)
            t_mom = 1.0
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
        
        # Adaptive restart
        if it - last_improve > 1000:
            t_mom = 1.0
            noise = rng.normal(0, 0.005 * S / np.sqrt(n) * np.exp(-it / 5000.0), n)
            y = project_simplex(y + noise, S)
        
        # Periodic noise
        if it % 2000 == 0:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n) * np.exp(-it / 10000.0), n)
            if noise.std() > 1e-8:
                y = project_simplex(y + noise, S)
                t_mom = 1.0

    return best_a, best

def upsample(a, factor=2):
    """Upsample by inserting zeros and interpolating."""
    n_old = len(a)
    n_new = n_old * factor
    t_old = np.linspace(0, 1, n_old, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    return a_new

def peak_shave(a, n, max_time, rng):
    """Target peak-shaving: identify top convolution peaks and reduce them."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top peaks
        top_k = np.argsort(c)[-min(5, len(c)):]
        min_peak_idx = top_k[0]  # The peak that's hardest to reduce
        
        # For the peak at lag k, the gradient w.r.t. a[i] is a[k-i] + a[i-k]
        # We want to reduce c[k], so we should decrease a in the direction of this gradient
        k = min_peak_idx
        grad = np.zeros(n)
        for i in range(n):
            gi = 0.0
            if 0 <= (k - i) < n:
                gi += a[k - i]
            if 0 <= (i - k) < n:
                gi += a[i - k]
            grad[i] = gi
        
        # Reduce in the direction that lowers c[k] the most
        # Project: a_new = a - step * grad, projected to simplex
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 0.01 * S / gnorm
        a_new = project_simplex(a - step * grad, S)
        
        # Verify improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur + 1e-10:
            a = a_new
        else:
            # Try adding noise
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Hierarchical multigrid construction + FISTA with backtracking +
targeted peak shaving + coordinate descent polish.
Under normalization sum=sqrt(2n), the objective equals max_conv directly.
Key innovations: ramp-based initialization, hierarchy for basin finding,
adaptive LSE with restarts, and peak-directed coordinate updates.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE approximation to max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_core(a_init, n, max_time, rng, beta_init=50.0, beta_max=3000.0,
               bt_max_iter=15, noise_period=1500, noise_scale=0.003):
    """FISTA with backtracking line search and adaptive restarts."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 2.0 * S / (gnorm * (1.0 + 0.1 * np.log1p(it / 100.0)))
        x_new = None
        for _ in range(bt_max_iter):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur - 1e-12:
                break
            step *= 0.5

        if x_new is None or step < 1e-12:
            # Stuck: inject noise and restart momentum
            noise = rng.normal(0, noise_scale * S / np.sqrt(n) * np.exp(-it / 20000.0), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

        # Adaptive restart if stuck
        if it - last_improve > 300:
            t_mom = 1.0

        # Periodic noise injection
        if it % noise_period == 0:
            ns = noise_scale * 0.5 * S / np.sqrt(n) * np.exp(-it / 30000.0)
            if ns > 1e-9:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0

    return best_a, best


def peak_shave(a_init, n, max_time, rng):
    """Targeted peak shaving: reduce the top convolution peaks specifically."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Find top K peaks
        K = min(3, len(c))
        top_peaks = np.argsort(c)[-K:]

        # For each peak, compute the "anti-gradient" (direction to reduce it)
        # d(c[k])/da[j] = 2*a[k-j] if 0<=k-j<n
        # To reduce c[k], decrease a[j] where a[k-j] is large
        combined_grad = np.zeros(n)
        for k in top_peaks:
            # Vectorized: grad_k[j] = 2*a[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                combined_grad[j_min:j_max + 1] += 2.0 * a[k - j_max:k - j_min + 1][::-1]

        gnorm = np.linalg.norm(combined_grad)
        if gnorm < 1e-14:
            break

        # Take a step in the direction that reduces peaks
        # But we need to be careful: reducing one peak might increase another
        # Use a small step and verify
        step = 0.5 * S / gnorm
        a_new = project_simplex(a - step * combined_grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur + 0.5 * (cur - best):
            a = a_new
        else:
            # Try smaller step or noise
            if it % 10 == 0:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * np.exp(-it / 500.0), n)
                a = project_simplex(a + noise, S)
                if noise.std() < 1e-10:
                    break

    return best_a, best


def coord_descent_focus(a_init, n, max_time, rng):
    """Coordinate descent focusing on peak-reducing transfers."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Identify the peak lag
        k_peak = int(np.argmax(c))

        # For reduction of c[k_peak]:
        # d(c[k])/da[j] = 2*a[k-j] (if valid)
        # We want to transfer mass from high-gradient positions to low-gradient positions
        grad = np.zeros(n)
        j_min = max(0, k_peak - n + 1)
        j_max = min(k_peak, n - 1)
        if j_min <= j_max:
            grad[j_min:j_max + 1] = 2.0 * a[k_peak - j_max:k_peak - j_min + 1][::-1]

        # Find good (source, dest) pairs
        n_try = min(30 + it // 5, 80)
        improved = False

        for _ in range(n_try):
            if time.monotonic() - start > max_time - 2:
                break

            # Source: random, biased towards high gradient
            if rng.random() < 0.6:
                # Sample from positions with high gradient
                probs = grad + 1e-10
                probs = probs / probs.sum()
                j = rng.choice(n, p=probs)
            else:
                j = rng.integers(0, n)

            m = rng.integers(0, n)
            if j == m or a[j] < 1e-12:
                continue

            # Compute change in c for transferring delta from j to m
            # c[k] changes by: delta * (d(c[k])/da[j] - d(c[k])/da[m])
            # = delta * (2*a[k-j] - 2*a[k-m]) (approximately, for the peak lag)
            # But we need to check ALL lags to ensure max doesn't increase

            # Quick rejection: check if this transfer helps the peak
            d_j = 0.0
            d_m = 0.0
            if 0 <= k_peak - j < n:
                d_j = 2.0 * a[k_peak - j]
            if 0 <= k_peak - m < n:
                d_m = 2.0 * a[k_peak - m]

            if d_m >= d_j:
                # Transfer from j to m doesn't help reduce c[k_peak]
                continue

            delta = a[j] * 0.5  # Try transferring half
            a[j] -= delta
            a[m] += delta

            c_new = c - delta * _compute_conv_diff(a, j, m, k_peak, n)
            # Actually this is complex; just recompute for the peak lag
            # and check if overall max improved
            if a[j] < 0 or a[m] < 0:
                a[j] += delta
                a[m] -= delta
                continue

            c_check = c + 0  # We'll just check the peak lag approximately
            # For efficiency, check if peak decreased
            new_peak_val = c[k_peak] - delta * (d_j - d_m)
            if new_peak_val < cur - 1e-12:
                # Full verification (expensive but infrequent)
                c_full = fft_convolve_self(a)
                if c_full.max() < best:
                    c = c_full
                    best = c_full.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] += delta
                    a[m] -= delta
            else:
                a[j] += delta
                a[m] -= delta

        if not improved and it > 10:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n) * np.exp(-it / 200.0), n)
            if noise.std() < 1e-10:
                break
            a = project_simplex(a + noise, S)

    return best_a, best


def _compute_conv_diff(a, j, m, k, n):
    """Compute how c changes at each lag when transferring mass from m to j."""
    # This is a helper - actually let's skip it and use full recompute
    return np.zeros(2 * n - 1)


def upsample_interp(a, factor=2):
    """Upsample via linear interpolation."""
    n_old = len(a)
    n_new = n_old * factor
    t_old = np.linspace(0, 1, n_old + 1)
    a_ext = np.concatenate([a, [a[0]]])  # Wrap for interpolation
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(t_new, t_old, a_ext)
    return a_new


def downsample_block(a, factor=2):
    """Downsample by block averaging."""
    n_old = len(a)
    n_new = n_old // factor
    a_blocks = a[:n_new * factor].reshape(n_new, factor)
    return a_blocks.mean(axis=1)


def make_ramp(n, alpha=2.0):
    """Create a ramp-like initialization."""
    t = np.linspace(0, 1, n)
    return 1.0 + alpha * t


def make_ramp_exp(n, alpha=2.0):
    """Exponential ramp."""
    t = np.linspace(0, 1, n)
    return np.exp(alpha * t)


def make_bell(n, center=0.7, width=0.3):
    """Bell-shaped (offset from center for asymmetry)."""
    t = np.linspace(0, 1, n)
    return np.exp(-((t - center) / width) ** 2)

        normalize_sum(make_ramp_exp(n_cur, 1.5), n_cur),

    ]
    best_score = np.inf
    for cand in candidates:
        c = fft_convolve_self(cand)

            best_score = c.max()

def multi_start_refine(a_starts, n, max_time, rng):
    """Run FISTA from multiple starts and return best."""
    best_a = a_starts[0]
    best_score = np.inf

    c0 = fft_convolve_self(normalize_sum(best_a, n))
    best_score = c0.max()

    # Time per start
    t_per = max_time / max(len(a_starts), 1) * 0.8
    t_start = time.monotonic()

    for i, a_s in enumerate(a_starts):
        if time.monotonic() - t_start > max_time - 5:
            break
        a_s = normalize_sum(a_s, n)
        t_alloc = min(t_per * (1.0 + 0.5 * (len(a_starts) - 1 - i) / len(a_starts)),
                      max_time - (time.monotonic() - t_start) - 5)
        if t_alloc < 5:
            continue

        a_opt, score = fista_core(a_s, n, t_alloc, rng,
                                   beta_init=50.0, beta_max=2000.0,
                                   bt_max_iter=12)
        if score < best_score:
            best_score = score
            best_a = a_opt
            print("  Multi-start %d/%d: improved to %.6f" % (i + 1, len(a_starts), score))

    return best_a, best_score

                alt_n = n_best * 2 if n_best <= 8192 else n_best // 2
                if 2048 <= alt_n <= 65536:
                    t_old = np.linspace(0, 1, n_best)
                    t_new = np.linspace(0, 1, alt_n)
                    alt_start = np.interp(t_new, t_old, global_best_a)
                    alt_start = normalize_sum(alt_start, alt_n)
                    a_alt, _ = fista_core(alt_start, alt_n, remaining3 * 0.5, rng,
                                           beta_init=50.0, beta_max=1500.0)
                    update_best(a_alt, "alt_n=%d" % alt_n)

    # === Final verification ===
    print("\n=== Final Verification ===")
    if global_best_a is None:
        global_best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        global_best_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_best_a]

    print("Returning length=%d, score=%.6f, total_time=%.0fs" % (
        len(global_best_a), ver if np.isfinite(ver) else float('inf'),
        time.monotonic() - t_start))
    return final_list
