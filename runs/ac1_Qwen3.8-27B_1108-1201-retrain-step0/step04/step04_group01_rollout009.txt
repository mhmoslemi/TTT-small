```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Handle edge case
    if v.sum() < 1e-15:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    """Smooth approximation of max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    terms = np.exp(shifted)
    return cm + np.log(terms.sum()) / beta

def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution wrt a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]
```

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

        c = fft_convolve_self(a)
        cur_max = c.max()

            best_max = cur_max
            best_a = a.copy()
            last_improve = time.monotonic()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

import numpy as np
import time
from scipy.fft import fft, ifft

# Helper functions (top-level)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_lse_and_grad(a, c, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    cm = c.max()
    shifted = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad

            best_a = y.copy()

            stall_counter = 0
        else:

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            print(f"  Converged at iter {it}, max_b = {best_max:.8f}")
            break
        
        # Step size
        max_a = np.max(np.abs(y)) + 1e-10
        step = 1.0 / (2.0 * beta * max_a + 1.0)
