        #   T2 = ifft(fft(w, N) * fft(a, N))
        T2 = np.real(ifft(W * A_y))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Nesterov step
        step = step_base / (1.0 + 0.05 * np.sqrt(it))
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t = t_new
        
        # Periodic noise injection
        if it % 1000 == 500 and rng.random() < 0.3:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score

def spectral_water_filling(a, S, n_iter, eta_init, rng, target_obj=1.49):
    """
    Spectral water-filling: iteratively flatten |FFT(a)|^2 to reduce peak of auto-correlation.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a = np.clip(a, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        eta = eta_init * (1.0 - 0.95 * it / max(n_iter - 1, 1))
        eta = max(eta, 0.01)
        
        # Compute spectrum
        A = fft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compute auto-correlation to find peaks
        c = np.real(ifft(P))[:2*n-1]
        c_max = c.max()
        
        # Target: make P flat. The "excess" at each frequency bin contributes to the peak.
        # Use a soft water-filling: P_new[m] = P[m] - eta * max(0, P[m] - P_mean)
        P_mean = P.mean()
        excess = np.maximum(P - P_mean, 0)
        deficit = np.maximum(P_mean - P, 0)
        
        # Water-filling: move power from excess to deficit
        # P_new = P - eta * (excess - deficit) (approximately)
        P_new = P - eta * (excess - deficit)
        P_new = np.maximum(P_new, 0)
        
        # Preserve energy (approximately)
        P_orig_sum = P.sum()
        P_new_sum = P_new.sum()
        if P_new_sum > 1e-30:
            P_new = P_new * (P_orig_sum / P_new_sum)
        
        # Reconstruct
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Accept/reject (simulated annealing style)
        if sc <= compute_score(a) + 0.01 * rng.exponential(1.0/n):
            a = a_new
        
        if best_score <= target_obj:
            break
    
    return best_a, best_score

Minimize 2*n*max(conv(a,a))/sum(a)^2 using proximal subgradient with Nesterov
acceleration, multi-peak spectral weighting, and adaptive water-filling.

Key techniques:
1. Multi-peak subgradient: weights multiple near-peak lags for smooth gradients
2. Nesterov acceleration with adaptive restarts
3. Spectral water-filling: flatten |FFT(a)|^2 to reduce auto-correlation peaks
4. Multi-start with diverse structured initializations
5. Multi-length search (4096-16384)
6. Backtracking-style adaptive step size
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))

    w_full: weights for each lag (length 2n-1)

    T2 = np.real(ifft(W * A))[:len(a)]
    return T1 + T2

        top_vals = c[sorted_idx[:k_use]]

        # Exponential decay weighting
        weights = top_vals * np.exp(-decay_rate * np.arange(k_use))
        total_w = weights.sum()
        if total_w < 1e-30:
            break
        weights = weights / total_w

        w_full = np.zeros(2 * n - 1)
        for i in range(k_use):
            w_full[int(sorted_idx[i])] = weights[i]

        grad = compute_gradient_multi(y, w_full, N_grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = step_scale * S / (np.sqrt(n) * (1.0 + 0.08 * np.sqrt(it)))

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        momentum_coeff = (t_mom - 1.0) / t_new
        y = x_new + momentum_coeff * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new

        # Periodic noise to escape local minima
        if it % noise_interval < noise_interval // 2 and rng.random() < noise_prob:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            no_improve = 0

        # Restart if stuck for too long
        if no_improve > improve_threshold * 5:
            noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
            y = project_simplex(best_local + noise, S)
            a = y.copy()
            t_mom = 1.0
            no_improve = 0

    return best_local, best_score


def spectral_water_filling(a_init, S, n_iter, eta_init, rng, target_obj=1.49):
    """
    Iterative spectral water-filling: flatten |FFT(a)|^2 to reduce peak of auto-correlation.
    Preserves phase and normalizes to maintain sum constraint.
    """
    n = len(a_init)
    N = 1 << (2 * n - 1).bit_length()
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        eta = max(0.01, eta_init * (1.0 - 0.9 * progress))

        A = fft(a, N)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute auto-correlation
        c = np.real(ifft(P))[:2 * n - 1]

        # Water-filling: reduce excess power, boost deficit
        P_mean = P.mean()
        excess = np.maximum(P - P_mean, 0.0)
        deficit = np.maximum(P_mean - P, 0.0)

        # Soft water-filling with nonlinear response
        P_new = P - eta * (excess - deficit)

        # Additional flattening: compress dynamic range
        P_min_pos = np.percentile(P_new[P_new > 0], 1) if np.any(P_new > 0) else 0
        P_max = P_new.max()
        if P_max > 1e-30 and P_min_pos > 0:
            ratio = P_new / P_min_pos
            # Compress: P_compressed = P_min * ratio^alpha where alpha < 1 for high values
            alpha_comp = 0.85
            mask_high = ratio > 1.0
            P_new[mask_high] = P_min_pos * np.power(ratio[mask_high], alpha_comp)

        P_new = np.maximum(P_new, 0.0)

        # Preserve total power
        P_sum_orig = P.sum()
        P_sum_new = P_new.sum()
        if P_sum_new > 1e-30:
            P_new = P_new * (P_sum_orig / P_sum_new)

        # Reconstruct with original phase
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        elif rng.random() < 0.3:
            # Metropolis acceptance for exploration
            delta = sc - compute_score(a)
            if delta < 0.01 * eta:
                a = a_new

        if best_score <= target_obj:
            break

    return best_a, best_score


def generate_initializations(n, S, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    arr = np.ones(n) * S / n
    inits.append(("uniform", arr))

    # Smooth noise
    noise = rng.standard_normal(n)
    k = max(3, n // 128)
    kernel = np.ones(k) / k
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() > 1e-10:
        smooth = smooth * (S / smooth.sum())
    inits.append(("smooth", smooth))

    # Chirp
    chirp = 0.5 + 0.3 * np.sin(2 * np.pi * (5 * tt + 3 * tt * tt))
    chirp = np.maximum(chirp, 0.01)
    inits.append(("chirp", chirp * (S / chirp.sum())))

    # Multi-sine
    ms = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.sin(2 * np.pi * 7 * tt) + \
         0.1 * np.sin(2 * np.pi * 13 * tt) + 0.05 * np.sin(2 * np.pi * 29 * tt)
    ms = np.maximum(ms, 0.01)
    inits.append(("multisin", ms * (S / ms.sum())))

    # Random (smoothed)
    rand = rng.random(n) + 0.1
    k2 = max(3, n // 256)
    kernel2 = np.ones(k2) / k2
    rand = np.convolve(rand, kernel2, mode='same')
    inits.append(("random", rand * (S / rand.sum())))

    # Two-bump
    bump = np.exp(-0.5 * ((tt - 0.25) * 5) ** 2) + np.exp(-0.5 * ((tt - 0.75) * 5) ** 2) + 0.1
    inits.append(("bump", bump * (S / bump.sum())))

    # Anti-symmetric perturbation from uniform (to reduce center lag)
    base = np.ones(n) * S / n
    anti = np.zeros(n)
    for j in range(n // 2):
        anti[j] = 0.1 * S / n
        anti[n - 1 - j] = -0.1 * S / n
    perturbed = np.maximum(base + anti, 0.0)
    if perturbed.sum() > 1e-10:
        inits.append(("anti_sym", perturbed * (S / perturbed.sum())))

    # High-frequency perturbation
    base2 = np.ones(n) * S / n
    freq_pert = 0.05 * S / n * np.sin(2 * np.pi * (n // 3) * tt)
    perturbed2 = np.maximum(base2 + freq_pert, 0.0)
    if perturbed2.sum() > 1e-10:
        inits.append(("freq_pert", perturbed2 * (S / perturbed2.sum())))

    return inits


def optimize_length(n, budget, rng, provided_seq=None, target_obj=1.49):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    best_score = np.inf
    best_a = None

    t_start = time.monotonic()

    # Build initialization list
    inits = []
    if provided_seq is not None:
        g = np.clip(provided_seq, 0, 1000).astype(np.float64)
        if len(g) == n and g.sum() > 0.01:
            inits.append(("provided", g * (S / g.sum())))

    gen_inits = generate_initializations(n, S, rng)
    inits.extend(gen_inits)

    n_inits = len(inits)
    # Reserve 30% budget for final polish
    search_budget = budget * 0.70
    per_init = search_budget / n_inits

    for idx, (name, init_a) in enumerate(inits):
        elapsed = time.monotonic() - t_start
        remaining = search_budget - elapsed
        if remaining < 3:
            break

        alloc = min(per_init, remaining / max(n_inits - idx, 1) * 1.5)
        alloc = max(alloc, 2.0)

        a = init_a.copy()

        # Stage 1: Spectral water-filling (short burst)
        rem1 = min(alloc * 0.35, remaining * 0.5)
        if rem1 > 2:
            # Estimate iterations
            N_est = 1 << (2 * n - 1).bit_length()
            time_per_iter = N_est * 8 * 1e-10  # rough estimate
            n_iter_wf = int(min(rem1 / max(time_per_iter, 0.0005), 2000))
            n_iter_wf = max(n_iter_wf, 50)
            a_wf, _ = spectral_water_filling(a, S, n_iter_wf, 0.4, rng, target_obj)
            sc = compute_score(a_wf)
            if sc < best_score:
                best_score = sc
                best_a = a_wf.copy()
            a = a_wf

        # Stage 2: Nesterov subgradient
        elapsed = time.monotonic() - t_start
        remaining = search_budget - elapsed
        rem2 = min(alloc * 0.65, remaining * 0.5)
        if rem2 > 2:
            a_sg, sc_sg = nesterov_subgradient(a, S, rem2, rng,
                                               top_k=min(80, n // 128),
                                               decay_rate=0.25,
                                               step_scale=2.5)
            if sc_sg < best_score:
                best_score = sc_sg
                best_a = a_sg.copy()

    # Final polish phase
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = budget - elapsed
        if remaining > 5:
            # Subgradient with more peaks
            rem = remaining * 0.6
            a_pol, sc_pol = nesterov_subgradient(best_a, S, rem, rng,
                                                 top_k=min(200, n // 64),
                                                 decay_rate=0.15,
                                                 step_scale=2.0)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol.copy()

            # Gentle spectral polishing
            elapsed = time.monotonic() - t_start
            remaining = budget - elapsed
            if remaining > 3:
                n_final = max(100, int(remaining * 0.5 / max(1 << (2 * n - 1).bit_length() * 8e-10, 0.001)))
                a_wf2, sc_wf2 = spectral_water_filling(best_a, S, min(n_final, 500),
                                                       0.1, rng, target_obj)
                if sc_wf2 < best_score:
                    best_score = sc_wf2
                    best_a = a_wf2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov subgradient + spectral water-filling with multi-length search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, budget_s - 25))

    global_best = np.inf
    global_a = None

    # Get provided sequence
    provided = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if provided is not None and provided.size >= 1024:
        lengths.append(provided.size)
    for L in [8192, 4096, 16384, 32768]:
        if L not in lengths:
            lengths.append(L)
    lengths = list(dict.fromkeys(lengths))

    # Sort: provided first, then prefer 8192, 4096, 16384
    def length_priority(L):
        if provided is not None and L == provided.size:
            return 0
        return abs(L - 8192) + L // 100000
    lengths.sort(key=length_priority)

    t_global = time.monotonic()
    total_budget = max(0, budget_s - 35)

    # Allocate budget: favor provided length and 8192
    weights = []
    for L in lengths:
        if provided is not None and L == provided.size:
            weights.append(2.0)
        elif L == 8192:
            weights.append(1.5)
        elif L == 4096:
            weights.append(1.2)
        elif L == 16384:
            weights.append(1.0)
        else:
            weights.append(0.6)
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]

    for i, n in enumerate(lengths):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc = min(budgets[i], remaining - 25)
        if alloc < 20:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        t0 = time.monotonic()

        prov = None
        if provided is not None and provided.size == n:
            prov = provided

        a_res, score = optimize_length(n, alloc, rng, provided_seq=prov)

        elapsed_len = time.monotonic() - t0
        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  New global best: %.6f (n=%d, took %.0fs)" % (global_best, n, elapsed_len))
        elif a_res is not None:
            print("  Best for n=%d: %.6f (global: %.6f, took %.0fs)" % (n, score, global_best, elapsed_len))
        else:
            print("  No result for n=%d (took %.0fs)" % (n, elapsed_len))

    # Fallback
    if global_a is None:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.ones(n_fb, dtype=np.float64) * S_fb / n_fb
        global_best = 2.0
        print("Using fallback uniform, score=%.4f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
    except Exception as e:
        print("Verification error: %s" % str(e))
        ver = np.inf

    if np.isfinite(ver):
        print("Verified: evaluate_sequence = %.6f" % ver)
        global_best = min(global_best, ver)
    else:
        print("WARNING: verification returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback score: %.6f" % global_best)

    print("FINAL: length=%d, score=%.6f, total_time=%.0fs" % (
        len(global_a), global_best, time.monotonic() - t_global))
    return [float(x) for x in global_a]
