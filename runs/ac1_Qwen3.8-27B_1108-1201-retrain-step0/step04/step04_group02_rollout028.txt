"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses LSE-smoothed gradient with Nesterov momentum, backtracking line search,
multi-resolution warm start, KKT active-set refinement, and multi-start.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(w, a):
    """Compute sum_k w[k]*a[k-j] for j=0..n-1 (cross-correlation)."""
    n = len(a)
    size = len(w)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, c, beta):
    """Gradient of LSE smoothing of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    return 2.0 * fft_cross_correlate(w, a)


def topk_grad(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s ** 2)


def lse_smoothed_obj(a, beta):
    """LSE smooth approximation of the objective (for line search)."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    smooth_max = cm + np.log(e.sum()) / beta
    s = a.sum()
    return 2.0 * n * smooth_max / (s * s)

    Nesterov-accelerated gradient descent on LSE-smoothed objective
    with backtracking line search (Armijo condition).

    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    c = fft_convolve_self(a)
    obj_smooth = lse_smoothed_obj(a, beta_start)

        it += 1

        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max(1, int(max_time * 10)))

        # Compute gradient at y
        c_y = fft_convolve_self(y)
        grad = lse_grad(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        step = 1.0 / (beta * S + 1.0)
        obj_y = lse_smoothed_obj(y, beta)
        cde = 1e-4  # Armijo slope
        rho = 0.5
        x_new = None
        for _ in range(30):
            x_try = project_simplex(y - step * grad, S)
            obj_try = lse_smoothed_obj(x_try, beta)
            if obj_try <= obj_y - cde * step * gnorm * gnorm:
                x_new = x_try
                break
            step *= rho
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t = t_new

        # Track best
        c_a = fft_convolve_self(a)
        cur_obj = 2.0 * n * c_a.max() / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Periodic perturbation to escape stagnation
        if it % 5000 == 0:
            noise = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            y = a.copy()
            t = 1.0

    return best_a, best_obj


def topk_momentum_optimize(a_init, n, S, max_time, rng):
    """
    Gradient descent on top-k subgradient with momentum and
    adaptive step size.
    """
    a = normalize_sum(a_init, n)
    v = np.zeros(n)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    momentum = 0.9
    lr = 0.5 / np.sqrt(n)
    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive k
        k = min(64, max(1, it // 100))
        grad = topk_grad(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Adam-like adaptive learning rate
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        step = lr / (1.0 + 0.01 * it)
        a_try = project_simplex(a - step * v / max(v_norm / gnorm, 1e-10), S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2
        obj_cur = 2.0 * n * c.max() / S**2

        if obj_try < obj_cur:
            a = a_try
            c = c_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
        else:
            lr *= 0.95
            if it % 2000 == 0 and lr < 1e-6:
                noise = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                v = np.zeros(n)
                lr = 0.3 / np.sqrt(n)

    return best_a, best_obj


def kkt_refine(a_init, n, S, max_time, rng):
    """
    KKT active-set refinement.
    Alternates between:
    - Given a: find w (dual weights on active peaks) making cross_corr(w,a) constant
    - Given w: solve linear system for a making cross_corr(w,a) constant
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    size = 2 * n - 1

    # Initialize w as softmax of current peaks
    beta = 50.0
    cm = c.max()
    w = np.exp(beta * (c - cm))
    w /= w.sum()

    for outer in range(100):
        if time.monotonic() - start > max_time - 5:
            break

        # Step 1: Update a given w
        # Solve: T_w * a = lambda * 1, sum(a) = S, a >= 0
        # T_w[j,m] = w[j+m] (Hankel matrix)
        # Use iterative method: a_new = project(a + alpha * (T_w^T * r))
        # where r = T_w * a - mean(T_w * a)

        ta = fft_cross_correlate(w, a)  # (T_w a)[j] = sum_k w[k] a[k-j]
        # We want ta to be constant. Gradient of ||ta - mean(ta)||^2 w.r.t. a:
        # d/da [sum_j (ta[j] - mu)^2] = 2 * T_w^T * (ta - mu)
        # T_w^T[j] applied to r: (T_w^T r)[m] = sum_j r[j] w[j+m]
        # = fft_cross_correlate(w, r) ... let's check
        # (T_w^T r)[m] = sum_j (T_w)[j,m] r[j] = sum_j w[j+m] r[j]
        # fft_cross_correlate(w, r)[m] = sum_k w[k] r[k-m]
        # These are different! We need sum_j w[j+m] r[j] = sum_j r[j] w[j+m]
        # Let's define: adjoint[m] = sum_j w[j+m] * r[j]
        # = sum_j r[j] w[j+m]
        # This is the "cross-correlation" but with w shifted FORWARD.
        # adjoint[m] = (r * w_rev_padded)[m] ... 
        # Actually: adjoint[m] = sum_j r[j] w[j+m] = cross_corr_extended
        # In FFT: this is ifft(fft(r) * fft(w_shifted))... 
        # Let's just compute it directly via the same FFT trick:
        # adjoint[m] = sum_j r[j] w[j+m]
        # = sum_j r[j] w[j+m]
        # Let's reverse w: w_rev[k] = w[size-1-k]
        # adjoint[m] = sum_j r[j] w_rev[size-1-j-m]
        # = (r * w_rev)[size-1-m] (convolution)
        # So: adjoint = ifft(fft(r) * fft(w_rev))[size-1-m] for m=0..n-1
        # Which is: adjoint[m] = conv[m'] where m' = size-1-m
        # Let's just compute: conv = ifft(fft(r, fsize) * fft(w_rev, fsize))
        # adjoint[m] = conv[size-1-m]

        r = ta - ta.mean()
        # Compute T_w^T * r
        w_rev = np.zeros(size)
        w_rev[:len(w)] = w[::-1] if len(w) <= size else w[:size][::-1]
        # Actually w has length size (2n-1)
        w_rev = w[::-1].copy()
        fsize = 1 << (n + size - 1).bit_length()
        R = fft(r, fsize)
        WR = fft(w_rev, fsize)
        conv_rw = np.real(ifft(R * WR))
        # adjoint[m] = sum_j r[j] w[j+m]
        # conv_rw[k] = sum_j r[j] w_rev[k-j] = sum_j r[j] w[size-1-(k-j)] = sum_j r[j] w[size-1-k+j]
        # We want sum_j r[j] w[j+m]. Set size-1-k+j = j+m => size-1-k = m => k = size-1-m
        # So adjoint[m] = conv_rw[size-1-m]
        adjoint = conv_rw[size - 1 - np.arange(n)]

        # Update a
        step = S / (n * (np.linalg.norm(adjoint) + 1e-10))
        a_try = project_simplex(a - step * adjoint, S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2

        if obj_try < compute_obj(a) + 1e-10:
            a = a_try
            c = c_try

        # Step 2: Update w given a
        # We want cross_corr(w, a) to be constant. 
        # cross_corr(w, a)[j] = sum_k w[k] a[k-j]
        # Gradient of ||cross_corr(w,a) - mean||^2 w.r.t. w:
        # = 2 * T_a^T * (cross_corr(w,a) - mean)
        # T_a^T applied to s: (T_a^T s)[k] = sum_j s[j] a[k-j]
        # = fft_cross_correlate(a, s) ... let's verify
        # (T_a^T s)[k] = sum_j (T_a)[j,k] s[j] = sum_j a[k-j] s[j]
        # = sum_j s[j] a[k-j]
        # fft_cross_correlate(a, s)[k] = sum_m a[m] s[m-k]
        # Hmm, not quite the same. Let me just compute directly.
        # (T_a^T s)[k] = sum_j s[j] a[k-j] (for 0<=k-j<n, 0<=j<n)
        # = sum_j s[j] a[k-j]
        # This is the convolution (s * a)[k]! (with appropriate bounds)
        # In FFT: ifft(fft(s) * fft(a))[:size]

        cw = fft_cross_correlate(w, a)  # (T_a w)[j] = sum_k w[k] a[k-j]
        r2 = cw - cw.mean()
        fsize2 = 1 << (n + size - 1).bit_length()
        S2 = fft(r2, fsize2)
        A2 = fft(a, fsize2)
        conv_sa = np.real(ifft(S2 * A2))
        # (T_a^T r2)[k] = (r2 * a)[k] = conv_sa[k] for k in 0..size-1
        grad_w = conv_sa[:size]

        # Update w with multiplicative weights (stay on probability simplex)
        eta_w = 1.0 / (np.linalg.norm(grad_w) + 1e-10)
        w_new = w * np.exp(-eta_w * grad_w)
        w_new = np.maximum(w_new, 1e-30)
        w_new /= w_new.sum()

        # Check improvement
        # The improvement is in making cross_corr(w,a) more uniform
        a_check = a  # a hasn't changed in this step
        # Recompute objective with new w only affects the gradient direction for next iteration
        w = w_new

        # Track best
        c_cur = fft_convolve_self(a)
        cur = 2.0 * n * c_cur.max() / S**2
        if cur < best_obj:
            best_obj = cur
            best_a = a.copy()

        # Convergence check
        if outer > 10 and abs(c.max() - c_try.max()) < 1e-8:
            break

    return best_a, best_obj


def coordinate_descent(a_init, n, S, max_time, rng):
    """
    Cyclic coordinate descent. For each pair (i,j), find the optimal
    transfer of mass from one to the other to reduce max convolution.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    n_iter = 0

    while time.monotonic() - start < max_time:
        n_iter += 1
        improved = False

        # Random subset of coordinates to try
        n_try = min(2000, n)
        idxs = rng.integers(0, n, size=n_try)

        for idx in idxs:
            if time.monotonic() - start > max_time - 2:
                break

            j = int(idxs[idx])
            # Try transferring mass from j to a random other position
            m = int(rng.integers(0, n))
            if m == j:
                continue

            # Try small transfers: delta from j to m
            # a[j] -> a[j] - delta, a[m] -> a[m] + delta
            # Change in c[k]: 
            #   For k where both j and m contribute: c[k] += delta*(a[k-j_old] - a[k-m_old])
            #   Plus quadratic terms at k=2j (delta^2 removed) and k=2m (delta^2 added)
            
            # Simple approach: try a few delta values
            delta_base = 0.01 * S / np.sqrt(n)
            for sign in [1, -1]:
                delta = sign * delta_base * rng.uniform(0.5, 2.0)
                if a[j] - delta < 0 or a[m] + delta < 0:
                    continue
                a_try = a.copy()
                a_try[j] -= delta
                a_try[m] += delta
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n * c_try.max() / S**2
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    c = c_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    delta_base *= 1.1  # increase step if improving
                    break

            if not improved and rng.random() < 0.01:
                delta_base *= 0.95

        if n_iter % 50 == 0:
            print("    CD iter %d, obj=%.8f" % (n_iter, best_obj))

    return best_a, best_obj


def multi_resolution_optimize(a_init, n, S, max_time, rng):
    """
    Multi-resolution: optimize at n/4, n/2, then n.
    """
    # Start at full resolution if n is small
    if n <= 256:
        return nesterov_lse_optimize(a_init, n, S, max_time, rng)

    # Downsample and optimize
    n2 = n // 2
    S2 = np.sqrt(2.0 * n2)
    a_half = (a_init[n//2::2] + a_init[:n//2:2]) / 2.0  # average pairs
    a_half = normalize_sum(a_half, n2)

    # Optimize at half resolution
    t_half = max_time * 0.3
    a_half_opt, _ = nesterov_lse_optimize(a_half, n2, S2, t_half, rng)

    # Upsample to full resolution
    a_full_init = np.zeros(n)
    a_full_init[::2] = a_half_opt * 2  # split mass to both positions
    a_full_init[1::2] = a_half_opt * 2
    a_full_init = normalize_sum(a_full_init, n)

    # Optimize at full resolution
    t_full = max_time * 0.7
    return nesterov_lse_optimize(a_full_init, n, S, t_full, rng)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Parent (previous best)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    # Cosine
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01))
    # Raised cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Sine squared
    inits.append(np.maximum(np.sin(np.pi * tt)**2, 0.01))
    # Multi-frequency
    inits.append(1.0 + 0.3*np.sin(2*np.pi*2*tt) + 0.2*np.cos(2*np.pi*3*tt + 0.5) + 0.1*np.sin(2*np.pi*7*tt))
    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    amps = rng.uniform(0.1, 0.4, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    sig = 1.0
    for f, amp, ph in zip(freqs, amps, phases):
        sig += amp * np.sin(2*np.pi*f*tt + ph)
    inits.append(np.maximum(sig, 0.01))
    # Edge-weighted
    inits.append(1.0 + 0.5*(tt**2 + (1-tt)**2))
    # Plateau
    inits.append(np.where(np.abs(tt - 0.5) < 0.3, 1.0, 0.5))
    
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    t0 = time.monotonic()

    global_best = np.inf
    global_a = None

    # Try to get previous best
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            parent = np.clip(parent, 0, 1000)
    except Exception:
        pass

    print("Starting search, budget=%.0fs, parent_len=%s" % (budget_s, parent.size if parent is not None else 'None'))

    # Main optimization at n = 8192 (or parent length)
    n_main = 8192
    if parent is not None and 1024 <= len(parent) <= 16384:
        n_main = len(parent)

    S_main = np.sqrt(2.0 * n_main)

    # Phase 1: Multi-resolution from parent (30% time)
    t1 = (deadline - time.monotonic()) * 0.30
    print("Phase 1: Multi-resolution optimization, n=%d, time=%.0fs" % (n_main, t1))
    if t1 > 30:
        if parent is not None and len(parent) == n_main:
            a1, s1 = multi_resolution_optimize(parent, n_main, S_main, t1, rng)
            a1 = normalize_sum(a1, n_main)
            s1 = compute_obj(a1)
            print("  Phase 1 result: %.8f" % s1)
            if s1 < global_best:
                global_best = s1
                global_a = a1.copy()
        else:
            a1, s1 = nesterov_lse_optimize(np.ones(n_main), n_main, S_main, t1, rng)
            a1 = normalize_sum(a1, n_main)
            s1 = compute_obj(a1)
            print("  Phase 1 result: %.8f" % s1)
            if s1 < global_best:
                global_best = s1
                global_a = a1.copy()

    # Phase 2: KKT refinement (25% time)
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.30
    print("Phase 2: KKT refinement, time=%.0fs" % t2)
    if global_a is not None and t2 > 20:
        a2, s2 = kkt_refine(global_a, n_main, S_main, t2, rng)
        a2 = normalize_sum(a2, n_main)
        s2 = compute_obj(a2)
        print("  Phase 2 result: %.8f" % s2)
        if s2 < global_best:
            global_best = s2
            global_a = a2.copy()

    # Phase 3: Top-k momentum optimization (20% time)
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.35
    print("Phase 3: Top-k momentum, time=%.0fs" % t3)
    if global_a is not None and t3 > 20:
        a3, s3 = topk_momentum_optimize(global_a, n_main, S_main, t3, rng)
        a3 = normalize_sum(a3, n_main)
        s3 = compute_obj(a3)
        print("  Phase 3 result: %.8f" % s3)
        if s3 < global_best:
            global_best = s3
            global_a = a3.copy()

    # Phase 4: Coordinate descent (15% time)
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.40
    print("Phase 4: Coordinate descent, time=%.0fs" % t4)
    if global_a is not None and t4 > 15:
        a4, s4 = coordinate_descent(global_a, n_main, S_main, t4, rng)
        a4 = normalize_sum(a4, n_main)
        s4 = compute_obj(a4)
        print("  Phase 4 result: %.8f" % s4)
        if s4 < global_best:
            global_best = s4
            global_a = a4.copy()

    # Phase 5: Final Nesterov LSE polish (remaining time)
    remaining = deadline - time.monotonic()
    print("Phase 5: Final polish, time=%.0fs" % remaining)
    if global_a is not None and remaining > 20:
        # Use higher beta for sharper max approximation
        a5, s5 = nesterov_lse_optimize(global_a, n_main, S_main, remaining * 0.6, rng,
                                        beta_start=50.0, beta_end=500.0)
        a5 = normalize_sum(a5, n_main)
        s5 = compute_obj(a5)
        print("  Phase 5a result: %.8f" % s5)
        if s5 < global_best:
            global_best = s5
            global_a = a5.copy()

        # One more KKT pass
        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            a6, s6 = kkt_refine(global_a, n_main, S_main, rem2 * 0.7, rng)
            a6 = normalize_sum(a6, n_main)
            s6 = compute_obj(a6)
            print("  Phase 5b result: %.8f" % s6)
            if s6 < global_best:
                global_best = s6
                global_a = a6.copy()

    # Try a different length if we have time
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_alt = 16384
        S_alt = np.sqrt(2.0 * n_alt)
        print("Phase 6: Trying n=%d, time=%.0fs" % (n_alt, remaining))
        # Initialize from parent by interpolation or tiling
        if parent is not None:
            # Upsample parent to n_alt
            from scipy.interpolate import interp1d
            p_len = len(parent)
            p_norm = normalize_sum(parent, p_len)
            x_orig = np.linspace(0, 1, p_len)
            x_new = np.linspace(0, 1, n_alt)
            f = interp1d(x_orig, p_norm, kind='cubic', fill_value='extrapolate')
            a_alt_init = np.maximum(f(x_new), 0.01)
            a_alt_init = normalize_sum(a_alt_init, n_alt)
        else:
            a_alt_init = np.ones(n_alt) * S_alt / n_alt

        a_alt, s_alt = nesterov_lse_optimize(a_alt_init, n_alt, S_alt, remaining * 0.6, rng,
                                              beta_start=5.0, beta_end=100.0)
        a_alt = normalize_sum(a_alt, n_alt)
        s_alt = compute_obj(a_alt)
        print("  n=%d result: %.8f" % (n_alt, s_alt))
        if s_alt < global_best:
            global_best = s_alt
            global_a = a_alt.copy()

        # KKT on the new length
        rem3 = deadline - time.monotonic()
        if rem3 > 20:
            a_alt2, s_alt2 = kkt_refine(a_alt, n_alt, S_alt, rem3 * 0.5, rng)
            a_alt2 = normalize_sum(a_alt2, n_alt)
            s_alt2 = compute_obj(a_alt2)
            print("  n=%d KKT result: %.8f" % (n_alt, s_alt2))
            if s_alt2 < global_best:
                global_best = s_alt2
                global_a = a_alt2.copy()

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.8f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("FINAL: length=%d, obj=%.8f, time=%.0fs" % (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
