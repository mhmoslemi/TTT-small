"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines top-K subgradient with exact line search, cyclic peak reduction,
Fourier-phase perturbation for global moves, and L-BFGS polish.
Key innovation: direct peak-targeted coordinate updates that exploit the
quadratic structure of individual convolution terms.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (css - S) / rho > 0
    if not np.any(cond):
        rho_val = n
    else:
        rho_val = np.max(rho[cond])
    theta = (css[rho_val - 1] - S) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c of length 2n-1, A of length N)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def conv_value_at(a, p, n):
    """Compute c_p = sum_k a_k * a_{p-k} for a single p."""
    k_min = max(0, p - (n - 1))
    k_max = min(n - 1, p)
    if k_min > k_max:
        return 0.0
    return np.dot(a[k_min:k_max + 1], a[p - k_max:p - k_min + 1][::-1])


def gradient_at_peak(a, p, n):
    """Gradient of c_p w.r.t. a: grad[j] = 2*a[p-j] if 0<=p-j<n."""
    grad = np.zeros(n)
    k_min = max(0, p - (n - 1))
    k_max = min(n - 1, p)
    if k_min <= k_max:
        indices = np.arange(k_min, k_max + 1)
        grad[indices] = 2.0 * a[p - indices]
    return grad


def top_k_peaks(c, k):
    """Return indices of top k peaks in convolution."""
    return np.argsort(c)[::-1][:k]


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_smoothing(a, N, beta, n):
    """Compute LSE-smoothed max of conv and its gradient."""
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n)
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    # Gradient via FFT
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad


def lbfgs_two_loop(s_hist, y_hist, g, m_max=10):
    """L-BFGS two-loop recursion. Returns descent direction."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        return -g / gn if gn > 1e-18 else np.zeros(n)
    m = min(m, m_max)
    s_hist = s_hist[-m:]
    y_hist = y_hist[-m:]
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        yy = np.dot(y_hist[i], y_hist[i])
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_hist[i], q) / yy
            q = q - alphas[i] * y_hist[i]
    s_last, y_last = s_hist[-1], y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    gamma = max(1e-10, np.dot(s_last, y_last) / yy_last) if yy_last > 1e-30 else 1.0
    r = gamma * q
    for i in range(m):
        yy = np.dot(y_hist[i], y_hist[i])
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_hist[i], y_hist[i]) / yy
        r = r + (alphas[i] - beta_i) * y_hist[i]
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def toppk_gradient_step(a, c, n, S, N, top_k, step_scale, rng):
    """One step of top-K subgradient descent. Returns (new_a, new_cmax)."""
    k = min(top_k, 2 * n - 1)
    peak_idx = np.argsort(c)[::-1][:k]
    # Weighted gradient: weight peaks by their value
    peak_vals = c[peak_idx]
    weights = peak_vals / peak_vals.sum()
    grad = np.zeros(n)
    for i, p in enumerate(peak_idx):
        p = int(p)
        grad += weights[i] * gradient_at_peak(a, p, n)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-18:
        return a.copy(), c.max()
    a_new = project_simplex(a - step_scale * grad / gnorm, S)
    c_new, _ = compute_conv(a_new, N)
    return a_new, c_new.max(), c_new


def cyclic_peak_reduction(a, n, S, N, target_reduction, max_iters=50):
    """Cyclically reduce each top peak toward a target. Returns improved a."""
    c, _ = compute_conv(a, N)
    cmax = c.max()
    target = cmax - target_reduction
    it = 0
    while it < max_iters and time.monotonic() % 1000 < 999:
        it += 1
        c, _ = compute_conv(a, N)
        cmax = c.max()
        if cmax <= target:
            break
        # Find current peak
        p = int(np.argmax(c))
        cp = c[p]
        if cp <= target:
            break
        # Compute descent direction for this peak
        grad = gradient_at_peak(a, p, n)
        # Project to maintain sum
        grad -= np.mean(grad)
        # Don't decrease already-zero coords too much
        grad = np.where(a < 1e-10, np.maximum(grad, 0), grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        # Find step to reduce cp by approximately (cp - target)
        # c_p(a + alpha*d) ≈ cp + alpha * dot(grad, d) + alpha^2 * (d^T Q_p d)
        # For small alpha: reduction ≈ alpha * |cp - target|/cp * cp
        # Use a fraction of the full reduction
        frac = 0.5
        alpha = frac * (cp - target) / max(gnorm * cp / (2 * S / np.sqrt(n)), 1e-10)
        alpha = min(alpha, 2.0 * S / (n * gnorm + 1e-10))
        a_try = project_simplex(a + alpha * grad, S)
        c_try = conv_value_at(a_try, p, n)
        if c_try < cp:
            a = a_try
        else:
            alpha *= 0.3
            a_try = project_simplex(a + alpha * grad, S)
            a = a_try
    return a


def fourier_phase_perturb(a, n, N, rng, theta_scale=0.3):
    """Perturb the phase of A = fft(a) while keeping |A| roughly fixed."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    magnitude = np.abs(A)
    phase = np.angle(A)
    # Random phase perturbation (smooth in frequency)
    dphase = rng.normal(0, theta_scale, N)
    # Smooth it a bit
    dphase = np.convolve(dphase, np.ones(5) / 5, mode='same')
    phase_new = phase + dphase
    A_new = magnitude * np.exp(1j * phase_new)
    a_new = np.real(ifft(A_new))[:n]
    return safe_normalize(a_new, n, S_global)


def mirror_perturb(a, n, S, rng, scale=0.1):

    mirror = a[::-1]
    diff = a - mirror
    diff -= np.mean(diff)
    a_new = a + scale * rng.random() * diff

    print("Phase 1: L-BFGS polish (300s)")
    t1 = time.monotonic()
    p1_budget = min(300.0, (deadline - t1) * 0.15)
    a1, sc1 = lbfgs_block(global_a, n, S, N, p1_budget, rng, beta_init=10.0, beta_max=150.0)
    update_best(a1, "P1_LBFGS")

    # --- Phase 2: Top-K gradient + cyclic peak reduction (700s) ---
    print("Phase 2: Top-K + cyclic (700s)")
    t2 = time.monotonic()
    p2_budget = min(700.0, (deadline - t2) * 0.35)
    a_curr = global_a.copy()
    c_curr, _ = compute_conv(a_curr, N)
    cmax_curr = c_curr.max()
    top_k = 5
    stall = 0
    start_p2 = time.monotonic()

    while time.monotonic() - start_p2 < p2_budget:
        it_block = int((time.monotonic() - start_p2) // 2)
        top_k = min(30, 5 + it_block // 50)
        step = 2.5 * S / (n * max(top_k, 3))
        a_new, cmax_new, c_curr = toppk_gradient_step(a_curr, c_curr, n, S, N, top_k, step, rng)

        # Also try cyclic reduction
        a_cyc = cyclic_peak_reduction(a_new, n, S, N, 0.003 * cmax_new, max_iters=5)
        c_cyc, _ = compute_conv(a_cyc, N)
        if c_cyc.max() < cmax_new:
            a_new = a_cyc
            cmax_new = c_cyc.max()
            c_curr = c_cyc

        if cmax_new < global_best:
            global_best = cmax_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            if int((time.monotonic() - start_p2) // 10) % 50 == 0:
                print("    [P2 it~%d] %.6f (k=%d)" % (it_block, global_best, top_k))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall > 20:
                # Perturb to escape
                a_curr = project_simplex(a_curr + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                c_curr, _ = compute_conv(a_curr, N)
                cmax_curr = c_curr.max()
                stall = 0

    # --- Phase 3: Stochastic exploration (600s) ---
    print("Phase 3: Stochastic exploration (600s)")
    t3 = time.monotonic()
    p3_budget = min(600.0, (deadline - t3) * 0.30)
    a3, sc3 = stochastic_exploration(global_a, n, S, N, p3_budget, rng, top_k=10)
    update_best(a3, "P3_stoch")

    # --- Phase 4: Final L-BFGS polish (remaining time) ---
    print("Phase 4: Final L-BFGS polish")
    t4 = time.monotonic()
    remaining = max(5.0, deadline - t4 - 10)
    a4, sc4 = lbfgs_block(global_a, n, S, N, remaining, rng, beta_init=30.0, beta_max=500.0, m_lbfgs=12)
    update_best(a4, "P4_final")

    # --- Phase 5: Ultra-high-beta final squeeze ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a5, sc5 = lbfgs_block(global_a, n, S, N, min(remaining - 5, 200.0), rng,
                              beta_init=80.0, beta_max=1000.0, m_lbfgs=8)
        update_best(a5, "P5_ultra")

    # --- Verification ---
    print("Final verification...")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f (internal: %.6f)" % (ver, global_best))
            if ver < global_best:
                global_best = ver
        else:
            print("Verification: inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, value=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
