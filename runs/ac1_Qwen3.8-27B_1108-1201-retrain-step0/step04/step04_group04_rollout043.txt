```python
"""
Search for a non-negative step-function height sequence minimizing

    2*n*max_k conv(a,a)_k / (sum a)^2

The optimizer combines several complementary local/generative steps:

1. FISTA with softmax-smoothed max-convolution, using either full-spectrum
   gradients or cheap exact gradients when one or a few lags dominate.
2. A square-root target-projection iteration in frequency domain:
   shape the convolution coefficients toward a lower peak, then recover a new
   non-negative coefficient sequence by square-rooting the shaped spectrum.
3. Spectral flattening projected back to non-negative palindromic sequences.
4. A fast coordinate-transfer polishing method that updates one pair of
   coordinates at a time and maintains the convolution exactly.

The program uses multiple candidate lengths and spends time polishing the best
sequence found so far.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def _fft_len(n):
    """Power-of-two FFT length large enough for linear self-convolution."""
    size = 2 * int(n) - 1
    return 1 << (size - 1).bit_length()


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    S = float(S)
    n = v.size
    if n == 0:
        return v
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)

    w = np.sort(v)[::-1]
    idx = np.arange(1, n + 1, dtype=np.float64)
    cumsum = np.cumsum(w)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    u = np.maximum(v - theta, 0.0)
    u_sum = float(u.sum())
    if u_sum < 1e-300:
        u = np.ones(n, dtype=np.float64)
        u *= S / n
    else:
        u *= S / u_sum
    return u


def _normalize(a, n, S):
    """Clip to [0, 1000] and rescale to sum S."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(n)
    S = float(S)
    if a.size != n:
        if a.size == 0:
            return np.ones(n, dtype=np.float64) * (S / max(1, n))
        a = _resize_sequence(a, n, S)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        a = np.ones(n, dtype=np.float64)
        a *= S / max(1, n)
    else:
        a *= S / s
    return a


def _uniform_n(n):
    return np.ones(int(n), dtype=np.float64)


def _symmetrize(a, n, S):
    """Return the palindromic symmetrization of a, normalized."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(n)
    if a.size != n:
        a = _resize_sequence(a, n, S)
    a = 0.5 * (a + a[::-1])
    return _normalize(a, n, S)


def _conv_from_A(A, L, n):
    """Convenience: recover linear self-convolution from FFT of zero-padded a."""
    b = ifft(A * A, L)
    b = np.real(b[:2 * n - 1])
    b[b < 0.0] = 0.0
    return b


def _self_conv(a, L=None):
    """Linear self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if L is None:
        L = _fft_len(n)
    A = rfft(a, L)
    b = irfft(A * A, L)
    b = np.real(b[:2 * n - 1])
    b[b < 0.0] = 0.0
    return b


def _full_score(a):
    """Fast FFT version of the evaluation objective."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    b = _self_conv(a)
    return float(2.0 * n * float(b.max()) / (s * s))


def _resize_sequence(parent, new_n, new_S):
    """Resize a sequence by linear interpolation and normalize."""
    p = np.asarray(parent, dtype=np.float64).reshape(-1)
    p = np.clip(p, 0.0, 1000.0)
    new_n = int(new_n)
    new_S = float(new_S)
    if p.size == 0:
        return np.ones(new_n, dtype=np.float64) * (new_S / max(1, new_n))
    if p.size == new_n:
        return _normalize(p, new_n, new_S)

    x_old = np.linspace(0.0, new_n, p.size, dtype=np.float64)
    x_new = np.arange(new_n, dtype=np.float64)
    arr = np.interp(x_new, x_old, p)
    return _normalize(arr, new_n, new_S)


def _as_length(seq, n, S):
    """Return seq as a normalized sequence of length n."""
    n = int(n)
    if seq is None:
        return np.ones(n, dtype=np.float64) * (float(S) / max(1, n))
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    if seq.size == n:
        return _normalize(seq, n, S)
    return _resize_sequence(seq, n, S)


def _direct_lag_grad(a, n, k):
    """Exact subgradient for a single convolution coefficient b_k."""
    n = int(n)
    k = int(k)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        # For r in [lo, hi], derivative is 2*a[k-r].
        g[lo:hi + 1] = 2.0 * np.flip(a[k - hi:k - lo + 1])
    return g


def _weighted_top_grad(a, n, w, idx):
    """Weighted sum of exact gradients for a small set of lags."""
    a = np.asarray(a, dtype=np.float64)
    n = int(n)
    g = np.zeros(n, dtype=np.float64)
    for i in range(len(idx)):
        wk = float(w[i])
        if wk <= 1e-15:
            continue
        k = int(idx[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * wk * np.flip(a[k - hi:k - lo + 1])
    return g


def _fista_smooth(a, n, S, budget, rng,
                  beta0=10.0, beta1=2000.0, ramp=3000,
                  top_thresh=0.90, use_direct=True):
    """
    FISTA-style smooth approximation of max convolution.

    Returns (best_sequence, best_peak) where best_peak is the normalized
    objective for sequences with sum sqrt(2n).
    """
    n = int(n)
    S = float(S)
    L = _fft_len(n)
    budget = max(1e-6, float(budget))

    a = _normalize(a, n, S)
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    it = 0
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    step_scale = 1.0
    max_terms = min(6, 2 * n - 1)

    while time.monotonic() - start < budget:
        it += 1
        frac = min(1.0, it / max(1, ramp))
        beta = beta0 + (beta1 - beta0) * frac

        A = rfft(y, L)
        b = _conv_from_A(A, L, n)
        cm = float(b.max())
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        # Softmax weights over convolution coefficients.
        e = np.exp(beta * (b - cm))
        e_sum = float(e.sum())
        if e_sum < 1e-300:
            w = np.zeros_like(b)
            w[kmax] = 1.0 if b.size else 0.0
            top_weight = 1.0
        else:
            w = e / e_sum
            top_weight = float(w.max())

        use_top = bool(use_direct and top_weight > top_thresh)

        if use_top:
            nt = min(max_terms, w.size)
            if nt <= w.size // 2:
                idx = np.argpartition(w, -nt)[-nt:]
            else:
                idx = np.argsort(w)[-nt:]
            wi = w[idx].astype(np.float64)
            s = float(wi.sum())
            if s <= 0.0:
                wi = np.ones(nt, dtype=np.float64) / max(1, nt)
            else:
                wi = wi / s
            grad = _weighted_top_grad(y, n, wi, idx)
        else:
            W = rfft(w, L)
            grad = 2.0 * irfft(W * np.conj(A), L)
            grad = np.ascontiguousarray(grad[:n])

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        if use_top:
            exact_step = S / (gnorm * (1.0 + 0.005 * np.log1p(it / 200.0)))
            step = max(step_scale / (2.0 * beta + 1e-14),
                       min(exact_step, 10.0 / max(1.0, beta)))
        else:
            step = step_scale / (2.0 * beta + 1e-14)

        # Safeguard against large moves.
        max_step = 0.05 * S / max(1e-14, gnorm)
        if step > max_step:
            step = max_step
        if step <= 1e-15:
            continue

        x = _project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x + ((t_mom - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        ys = float(y.sum())
        if ys > 1e-300:
            y *= S / ys
        else:
            y = x.copy()

        a = x
        t_mom = t_new

        # Adaptive restart/noise.
        if it - last_improve > 2000:
            step_scale = max(0.3, step_scale * 0.7)
            if it % 1000 == 0:
                noise = 0.003 * S / np.sqrt(n) * np.exp(-it / 20000.0)
                if noise > 1e-10 and rng is not None:
                    y = _project_simplex(best_a + rng.normal(0.0, noise, n), S)
                    a = _project_simplex(y, S)
                    t_mom = 1.0
        else:
            step_scale = min(1.2, step_scale * 1.01)

    b_a = _self_conv(a, L)
    if float(b_a.max()) < best:
        best = float(b_a.max())
        best_a = a.copy()
    return best_a, best


def _square_root_target(a, n, S, budget, rng, T_target=None,
                        mix_start=0.20, mix_end=0.60):
    """
    Alternating target-projection / polynomial square-root method.

    Shape the current convolution coefficients toward a lower peak T, take a
    frequency-domain square root aligned with the current sequence, project
    back to non-negative coefficients, and normalize.
    """
    n = int(n)
    S = float(S)
    L = _fft_len(n)
    budget = max(1e-6, float(budget))

    a = _normalize(a, n, S)
    start = time.monotonic()
    it = 0
    best = np.inf
    best_a = a.copy()
    last_improve = 0

    # Initial peak and target floor.
    Af = fft(a, L)
    b = _conv_from_A(Af, L, n)
    cur = float(b.max())
    best = cur
    best_a = a.copy()
    if T_target is None:
        floor_T = max(1.0, 0.95 * cur)
    else:
        floor_T = max(1.0, float(T_target))
    T = max(cur, floor_T * 1.05)

    while time.monotonic() - start < budget:
        it += 1

        # Current convolution and best.
        Af = fft(a, L)
        b = _conv_from_A(Af, L, n)
        cur = float(b.max())
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it

        # Shape convolution toward T.
        bt = b.copy()
        over = bt > T
        if np.any(over):
            bt[over] = T + 0.25 * (bt[over] - T)
        bt = np.maximum(bt, 0.0)
        bt_sum = float(bt.sum())
        if bt_sum < 1e-15:
            bt = b.copy()
            bt_sum = float(bt.sum())
        if bt_sum < 1e-15:
            break
        bt *= (S * S) / bt_sum

        # Frequency-domain square root aligned with current A.
        B = fft(bt, L)
        At = np.empty_like(B, dtype=np.complex128)
        mask = np.abs(Af) > 1e-10
        if np.any(mask):
            # One Newton step from current root Af to sqrt(B).
            At[mask] = 0.5 * (Af[mask] + B[mask] / Af[mask])
        if np.any(~mask):
            At[~mask] = np.sqrt(B[~mask].astype(np.complex128))

        a1 = ifft(At, L)
        a1 = np.real(a1[:n])
        if float(a1.sum()) < 0.0:
            a1 = -a1
        a1 = np.maximum(a1, 0.0)
        a1_sum = float(a1.sum())

        if a1_sum < 1e-15:
            a_new = a.copy()
        else:
            # Mix with the rooted candidate.
            frac = min(1.0, it / 2500.0)
            mix = mix_start + (mix_end - mix_start) * frac
            a1n = a1 * (S / a1_sum)
            a_new = (1.0 - mix) * a + mix * a1n
            a_new = _project_simplex(a_new, S)

            # Occasional noise to escape local minima.
            if rng is not None and rng.random() < 0.03:
                ns = 0.004 * S / np.sqrt(n)
                a_new = _project_simplex(a_new + rng.normal(0.0, ns, n), S)

        # Lower T gradually.
        T = max(floor_T, T * 0.9995)

        # If stuck, restart from best.
        if it - last_improve > 600:
            a = best_a.copy()
            T = max(floor_T, best * 0.985)
            last_improve = it - 200

        a = a_new

    b_final = _self_conv(a)
    if float(b_final.max()) < best:
        best = float(b_final.max())
        best_a = a.copy()
    return best_a, best


def _spectral_flatten(a, n, S, budget, rng, alpha0=0.40, alpha1=0.90):
    """
    Gerchberg-Saxton style flattening for palindromic sequences.

    This optimizes the spectral magnitude (autocorrelation-like) and is useful
    as an additional basin-hopping projection.
    """
    n = int(n)
    S = float(S)
    L = _fft_len(n)
    budget = max(1e-6, float(budget))

    a = _symmetrize(a, n, S)
    start = time.monotonic()
    it = 0
    alpha = alpha0
    best = np.inf
    best_a = a.copy()
    last_improve = 0

    while time.monotonic() - start < budget:
        it += 1

        A = rfft(a, L)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        P = mag * mag
        P_mean = float(P.mean())
        if P_mean < 1e-30:
            break

        P_pow = np.power(np.maximum(P, 1e-300), alpha)
        P_pow_sum = float(P_pow.sum())
        if P_pow_sum < 1e-30:
            P_pow = np.ones_like(P)
        else:
            P_pow = P_pow / P_pow_sum

        P_new = (1.0 - alpha) * (P / P_mean) + alpha * (P_pow * P_mean)
        P_new_sum = float(P_new.sum())
        P_total = float(P.sum())
        if P_new_sum > 1e-30 and P_total > 1e-30:
            P_new *= P_total / P_new_sum
        else:
            P_new = P

        mag_new = np.sqrt(np.maximum(P_new, 0.0))
        A_new = phase * mag_new
        a_raw = irfft(A_new, L)
        a_raw = np.real(a_raw[:n])
        a_raw = 0.5 * (a_raw + a_raw[::-1])
        a_new = _normalize(a_raw, n, S)

        b = _self_conv(a_new)
        cur = float(b.max())
        if cur < best:
            best = cur
            best_a = a_new.copy()
            last_improve = it

        a = a_new
        alpha = min(alpha1, alpha + 0.001)

        if it - last_improve > 800:
            noise = 0.002 * S / np.sqrt(n)
            if rng is not None:
                a = _project_simplex(best_a + rng.normal(0.0, noise, n), S)
            else:
                a = best_a.copy()
            last_improve = it

    b_final = _self_conv(a)
    if float(b_final.max()) < best:
        best = float(b_final.max())
        best_a = a.copy()
    return best_a, best


def _coord_descent(a, n, S, budget, rng, active=6, tops=6):
    """
    Fast polishing by coordinate transfers.

    Maintains the convolution vector exactly under pairwise transfers and
    periodically recomputes it to remove drift.
    """
    n = int(n)
    S = float(S)
    L = _fft_len(n)
    budget = max(1e-6, float(budget))

    a = _normalize(a, n, S)
    b = _self_conv(a, L)
    cur = float(b.max())
    best = cur
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    no_improve = 0
    accept_count = 0
    active = max(1, int(min(active, 2 * n - 1)))
    kk = min(20, n)

    while time.monotonic() - start < budget:
        it += 1

        # Active lags: top convolution coefficients with mild softmax.
        top_idx = np.argpartition(b, -active)[-active:] if active < b.size else np.arange(b.size)
        vals = b[top_idx]
        wv = np.exp(3.0 * (vals - vals.max()))
        wv = wv / wv.sum()

        # Weighted subgradient of active lags.
        g = np.zeros(n, dtype=np.float64)
        for i, k in enumerate(top_idx):
            wk = float(wv[i])
            if wk <= 1e-12:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * wk * np.flip(a[k - hi:k - lo + 1])

        if kk == n:
            pos_idx = np.argsort(g)
            neg_idx = np.argsort(-g)
        else:
            pos_idx = np.argpartition(g, -kk)[-kk:]
            neg_idx = np.argpartition(-g, -kk)[-kk:]

        pos = [int(i) for i in pos_idx if g[i] > 1e-10 and a[i] > 1e-10][:12]
        neg = [int(i) for i in neg_idx if g[i] < -1e-10][:12]

        if len(pos) == 0 or len(neg) == 0:
            no_improve += 5
        else:
            pairs = []
            for j in neg:
                gj = float(g[j])
                for m in pos:
                    if j == m:
                        continue
                    D = gj - float(g[m])
                    if D < -1e-12:
                        pairs.append((D, j, m))

            if len(pairs) == 0:
                no_improve += 1
            else:
                pairs.sort()  # sorts by D first
                improved = False
                for D, j, m in pairs[:int(tops)]:
                    bound = min(0.99 * float(a[j]), 0.99 * float(a[m]))
                    if bound <= 1e-12:
                        continue

                    # Candidate deltas: fractions of the feasible bound,
                    # plus a D-scaled estimate.
                    cand = []
                    for f in (0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90):
                        d = bound * f
                        if 0.0 < d < bound:
                            cand.append(d)
                    base = -D / (0.2 + abs(D))
                    if 0.0 < base < bound:
                        cand.append(base)

                    if len(cand) == 0:
                        continue

                    # Linear derivative vector for transferring from m to j.
                    dvec = np.zeros(2 * n - 1, dtype=np.float64)
                    dvec[j:j + n] += 2.0 * a
                    dvec[m:m + n] -= 2.0 * a

                    best_delta = 0.0
                    best_val = cur
                    for delta in cand:
                        delta2 = delta * delta
                        base_vals = b + delta * dvec
                        val = float(np.max(base_vals))

                        # Exact quadratic corrections at j, m, j+m.
                        if j < len(b):
                            vv = b[j] + delta * dvec[j] + delta2
                            if vv > val:
                                val = vv
                        if m < len(b):
                            vv = b[m] + delta * dvec[m] + delta2
                            if vv > val:
                                val = vv
                        jm = j + m
                        if jm < len(b):
                            vv = b[jm] + delta * dvec[jm] - 2.0 * delta2
                            if vv > val:
                                val = vv

                        if val < best_val:
                            best_val = val
                            best_delta = delta

                    if best_delta > 0.0 and best_val < cur - 1e-11:
                        delta = best_delta
                        delta2 = delta * delta
                        a[j] += delta
                        a[m] -= delta
                        if a[m] < 0.0:
                            a[m] = 0.0

                        b = b + delta * dvec
                        if j < len(b):
                            b[j] += delta2
                        if m < len(b):
                            b[m] += delta2
                        jm = j + m
                        if jm < len(b):
                            b[jm] -= 2.0 * delta2
                        b[b < 0.0] = 0.0

                        cur = best_val
                        if cur < best:
                            best = cur
                            best_a = a.copy()
                        improved = True
                        accept_count += 1

                if not improved:
                    no_improve += 1

        if no_improve > 40:
            if rng is not None:
                ns = 0.002 * S / np.sqrt(n)
                a = _project_simplex(best_a + rng.normal(0.0, ns, n), S)
            else:
                a = best_a.copy()
            b = _self_conv(a, L)
            cur = float(b.max())
            if cur < best:
                best = cur
                best_a = a.copy()
            no_improve = 0

        # Periodic exact recomputation to prevent drift.
        if it % 50 == 0 or accept_count % 200 == 0:
            b = _self_conv(a, L)
            cur = float(b.max())
            if cur < best:
                best = cur
                best_a = a.copy()

    b_final = _self_conv(a, L)
    if float(b_final.max()) < best:
        best = float(b_final.max())
        best_a = a.copy()
    return best_a, best


def _consider(state, seq, label):
    """Update the global state with a candidate sequence if it improves."""
    if seq is None:
        return
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    if seq.size == 0:
        return
    score = _full_score(seq)
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = float(score)
        state['best_a'] = seq.copy()
        print("[consider] %-22s n=%d score=%.6f" % (label, seq.size, score))


def propose_candidate(seed=42, budget_s=2400.0):
    """
    Search for a high-quality non-negative sequence.

    Returns a list of floats. The global ``height_sequence_1`` is used as a
    parent if available, but the search also explores other initializations
    and lengths.
    """
    t0 = time.monotonic()
    total_budget = float(budget_s)
    if total_budget < 60.0:
        total_budget = 60.0
    T = max(60.0, total_budget - 30.0)
    deadline = t0 + T

    rng = np.random.default_rng(int(seed))
    state = {"best_a": None, "best_score": np.inf}

    MAX_N = 16384
    MIN_N = 2048

    # Load parent if present.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 64 and float(p.sum()) > 0.01:
            if p.size <= MAX_N:
                parent = p
                _consider(state, parent, "parent")
            else:
                parent = _resize_sequence(p, MAX_N, np.sqrt(2.0 * MAX_N))
                _consider(state, parent, "parent_resized")
    except Exception:
        parent = None

    # Main lengths to explore.
    if parent is not None:
        Np = len(parent)
        n1 = int(1 << max(0, (min(Np, MAX_N) - 1).bit_length()))
    else:
        n1 = 8192
    if n1 < MIN_N:
        n1 = MIN_N
    if n1 > MAX_N:
        n1 = MAX_N
    if n1 not in (2048, 4096, 8192, 16384):
        n1 = 8192
    n2 = 16384 if n1 < 16384 else 8192
    if n1 == n2:
        n2 = 0

    S1 = float(np.sqrt(2.0 * n1))
    a_main = _as_length(parent, n1, S1)
    a_sym = _symmetrize(a_main, n1, S1)
    _consider(state, a_main, "start_main")
    _consider(state, a_sym, "start_sym")

    print("propose_candidate: n1=%d n2=%d budget=%.0fs start_score=%s" %
          (n1, n2, total_budget,
           ("%.6f" % state["best_score"]) if np.isfinite(state["best_score"]) else "inf"))

    # ---------------- Stage 1: FISTA on main length ----------------
    rem = deadline - time.monotonic()
    b1 = max(5.0, min(T * 0.24, rem - 10.0))
    if rem > 20.0 and b1 > 5.0:
        a_opt, _ = _fista_smooth(a_main, n1, S1, b1, rng,
                                 beta0=10.0, beta1=3000.0, ramp=4000)
        _consider(state, a_opt, "fista_main")

    # ---------------- Stage 2: FISTA on symmetrized start ----------------
    b2 = max(5.0, min(T * 0.06, deadline - time.monotonic() - 10.0))
    if b2 > 5.0:
        a_opt, _ = _fista_smooth(a_sym, n1, S1, b2, rng,
                                 beta0=20.0, beta1=2500.0, ramp=3000)
        _consider(state, a_opt, "fista_sym")

    # ---------------- Stage 3: square-root target on main length ----------------
    if np.isfinite(state["best_score"]):
        target1 = max(1.35, state["best_score"] - 0.015)
    else:
        target1 = 1.9
    b3 = max(5.0, min(T * 0.12, deadline - time.monotonic() - 10.0))
    if b3 > 5.0:
        start = _as_length(state["best_a"] if state["best_a"] is not None else a_main, n1, S1)
        a_opt, _ = _square_root_target(start, n1, S1, b3, rng, T_target=target1)
        _consider(state, a_opt, "square_main")

    # ---------------- Stage 4: spectral flattening on main length ----------------
    b4 = max(5.0, min(T * 0.06, deadline - time.monotonic() - 10.0))
    if b4 > 5.0:
        start = _as_length(state["best_a"] if state["best_a"] is not None else a_main, n1, S1)
        a_opt, _ = _spectral_flatten(start, n1, S1, b4, rng)
        _consider(state, a_opt, "spectral_main")

    # ---------------- Stage 5: coordinate polish on main length ----------------
    b5 = max(5.0, min(T * 0.10, deadline - time.monotonic() - 10.0))
    if b5 > 5.0:
        start = _as_length(state["best_a"] if state["best_a"] is not None else a_main, n1, S1)
        a_opt, _ = _coord_descent(start, n1, S1, b5, rng)
        _consider(state, a_opt, "cd_main")

    # ---------------- Alt length stages ----------------
    if n2 > 0:
        S2 = float(np.sqrt(2.0 * n2))

        b6 = max(5.0, min(T * 0.22, deadline - time.monotonic() - 10.0))
        if b6 > 5.0:
            src = state["best_a"] if state["best_a"] is not None else a_main
            start = _as_length(src, n2, S2)
            a_opt, _ = _fista_smooth(start, n2, S2, b6, rng,
                                     beta0=10.0, beta1=3000.0, ramp=4000)
            _consider(state, a_opt, "fista_alt")

        if np.isfinite(state["best_score"]):
            target2 = max(1.35, state["best_score"] - 0.010)
        else:
            target2 = 1.9
        b7 = max(5.0, min(T * 0.08, deadline - time.monotonic() - 10.0))
        if b7 > 5.0:
            src = state["best_a"] if state["best_a"] is not None else a_main
            start = _as_length(src, n2, S2)
            a_opt, _ = _square_root_target(start, n2, S2, b7, rng, T_target=target2)
            _consider(state, a_opt, "square_alt")

        b8 = max(5.0, min(T * 0.04, deadline - time.monotonic() - 10.0))
        if b8 > 5.0:
            src = state["best_a"] if state["best_a"] is not None else a_main
            start = _as_length(src, n2, S2)
            a_opt, _ = _coord_descent(start, n2, S2, b8, rng)
            _consider(state, a_opt, "cd_alt")

    # ---------------- Final polish on globally best sequence ----------------
    final_seq = state["best_a"]
    if final_seq is None:
        final_seq = np.ones(2048, dtype=np.float64)
        _consider(state, final_seq, "fallback_uniform")

    nb = len(final_seq)
    Sb = float(np.sqrt(2.0 * nb))

    rem = max(0.0, deadline - time.monotonic())
    if rem > 10.0:
        # FISTA final.
        tf = max(5.0, rem * 0.40)
        a_opt, _ = _fista_smooth(final_seq, nb, Sb, tf, rng,
                                 beta0=50.0, beta1=5000.0, ramp=2000)
        _consider(state, a_opt, "final_fista")

        # Square-root final.
        rem = max(0.0, deadline - time.monotonic())
        if rem > 10.0:
            tq = max(5.0, rem * 0.30)
            if np.isfinite(state["best_score"]):
                tq_target = max(1.30, state["best_score"] - 0.020)
            else:
                tq_target = 1.6
            a_opt, _ = _square_root_target(final_seq, nb, Sb, tq, rng, T_target=tq_target)
            _consider(state, a_opt, "final_square")

        # Coordinate final.
        rem = max(0.0, deadline - time.monotonic())
        if rem > 10.0:
            tc = max(5.0, rem * 0.30)
            a_opt, _ = _coord_descent(final_seq, nb, Sb, tc, rng)
            _consider(state, a_opt, "final_cd")

    if state["best_a"] is None:
        final_seq = np.ones(2048, dtype=np.float64)
    final_seq = np.asarray(state["best_a"], dtype=np.float64).reshape(-1)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    fs = float(final_seq.sum())
    if fs < 0.01:
        final_seq = np.ones(2048, dtype=np.float64)

    final_list = [float(x) for x in final_seq]

    # Final verification with the official evaluator, if available.
    try:
        off = evaluate_sequence(final_list)
        if not np.isfinite(off) or off > state["best_score"] + 0.05:
            # If the official evaluator disagrees badly (e.g. length/perf issue),
            # fall back to a smaller normalized parent-like sequence.
            fb = _as_length(parent, 8192, float(np.sqrt(2.0 * 8192))) if parent is not None else _uniform_n(8192)
            fb = np.clip(fb, 0.0, 1000.0)
            fb_list = [float(x) for x in fb]
            try:
                off2 = evaluate_sequence(fb_list)
                if np.isfinite(off2):
                    final_list = fb_list
            except Exception:
                final_list = [1.0] * 8192
        print("final: n=%d score_fft=%.6f score_official=%s total_time=%.0fs" %
              (len(final_list),
               state["best_score"] if np.isfinite(state["best_score"]) else np.inf,
               ("%.6f" % off) if np.isfinite(off) else "inf",
               time.monotonic() - t0))
    except Exception as e:
        print("final: n=%d score_fft=%.6f evaluation_error=%s total_time=%.0fs" %
              (len(final_list),
               state["best_score"] if np.isfinite(state["best_score"]) else np.inf,
               repr(e),
               time.monotonic() - t0))

    return final_list
```
