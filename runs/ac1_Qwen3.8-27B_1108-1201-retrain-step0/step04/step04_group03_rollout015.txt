"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: FISTA with LSE smoothing at high beta, targeted repulsion at
hot peaks, LP best-response with many constraints, and exact subgradient
polishing. Uses n=16384 primary with multi-scale exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad_topk(a, c, k=5):
    """Exact subgradient using top-k peaks of convolution."""
    n = len(a)
    cm = c.max()
    # Weight peaks by how close they are to max
    weights = np.exp(50.0 * (c - cm))
    tot = weights.sum()
    if tot < 1e-300:
        return np.zeros(n)
    weights /= tot
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def targeted_repulsion(a, n, S, c, lr=0.1, top_frac=0.05):
    """Reduce elements that most contribute to the top peaks."""
    n_peaks = max(3, int(top_frac * len(c)))
    peak_idx = np.argsort(c)[::-1][:n_peaks]
    # Compute contribution of each a[i] to the top peaks
    contrib = np.zeros(n)
    for k in peak_idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        contrib[js] += a[k - js]
    # Scale by how much each peak exceeds the median
    peak_vals = c[peak_idx]
    median_val = np.median(c)
    excess = np.maximum(peak_vals - median_val, 0)
    # Normalize excess
    if excess.sum() > 0:
        excess /= excess.sum()
    # Weighted contribution
    total_contrib = np.zeros(n)
    for i, k in enumerate(peak_idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        total_contrib[js] += excess[i] * a[k - js]
    # Multiplicative reduction
    max_contrib = total_contrib.max()
    if max_contrib > 0:
        reduction = np.exp(-lr * total_contrib / max_contrib)
        a_new = a * reduction
        a_new = np.maximum(a_new, 1e-15)
        return project_simplex(a_new, S)
    return a.copy()


def run_fista_aggressive(a_init, n, S, max_time, rng, beta_init=50.0,

                         repulsion_freq=200, repulsion_lr=0.08,
                         noise_freq=4000, exact_grad_after=5000):

    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        if it > exact_grad_after:
            beta = beta_max
        else:
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Use exact subgradient after warmup

            grad = exact_subgrad_topk(y, c, k=3)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta + gnorm * 1e-6)

        # Stuck detection
        if it - last_improve_it > 3000:
            step *= 0.3
            # Targeted repulsion
            y = targeted_repulsion(y, n, S, c, lr=0.15, top_frac=0.03)
            t = 1.0
            last_improve_it = it
            # Also add noise

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic targeted repulsion
        if it % repulsion_freq == 0 and it > 1000:
            cur_lr = repulsion_lr * np.exp(-it / 50000.0)
            if cur_lr > 0.001:

                t = 1.0

def solve_lp_best_response(f, n, S, top_k=500, time_limit=15.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]

    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver='SCS', verbose=False, max_iters=50000)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False, max_iters=50000)
        except Exception:
            try:
                prob.solve(solver='CLARABEL', verbose=False)
            except Exception:
                return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def coordinate_polish(a, n, S, max_time, rng):
    """Coordinate-wise local search for final polishing."""
    a = a.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    improved = True
    attempts = 0

    while time.monotonic() - start < max_time and improved:
        improved = False
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_obj:
            best_obj = cur

        # Find the hottest lag
        k_star = np.argmax(c)
        # Elements contributing to this peak
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        partners = a[k_star - js]

        # Try reducing top contributors and redistributing
        contrib = a[js] * partners
        top_contrib_idx = np.argsort(contrib)[::-1][:50]

        for idx2 in top_contrib_idx[:20]:
            if time.monotonic() - start > max_time * 0.9:
                break
            j = js[idx2]
            # Try reducing a[j] by various amounts
            for frac in [0.05, 0.1, 0.2, 0.3]:
                a_test = a.copy()
                reduction = a_test[j] * frac
                a_test[j] -= reduction
                # Redistribute to a random cool position
                cool_k = np.argmin(c)
                # Find where to add
                add_pos = rng.integers(0, n)
                a_test[add_pos] += reduction
                a_test = np.maximum(a_test, 0)
                a_test = project_simplex(a_test, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur * 0.999:
                    a = a_test.copy()
                    improved = True
                    break
            if improved:
                break
        attempts += 1

    return a, best_obj


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])
    return [np.maximum(x, 0.01) for x in inits]


def run_lp_assisted(a_init, n, S, max_time, rng, lp_every=150, top_k=400):
    """FISTA with periodic LP corrections and aggressive restarts."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    lp_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(2000.0, 50.0 * np.exp(0.002 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # LP correction
        elapsed = time.monotonic() - start
        if (it % lp_every == 0 and elapsed < max_time * 0.85
                and (max_time - elapsed) > 30):
            t_lp_start = time.monotonic()
            g_lp = solve_lp_best_response(y, n, S, top_k=top_k,
                                          time_limit=min(15.0, (max_time - elapsed) * 0.1))
            lp_time = time.monotonic() - t_lp_start
            lp_count += 1

            if g_lp is not None:
                c_lp = fft_convolve_self(g_lp)
                if c_lp.max() < best * 1.005:
                    # Good LP solution - use it
                    y = g_lp.copy()
                    a = g_lp.copy()
                    t = 1.0
                    if c_lp.max() < best:
                        best = c_lp.max()
                        best_a = y.copy()
                        last_improve_it = it
                        print("    LP %d: %.6f (t=%.1fs)" % (lp_count, best, lp_time))
                    continue
                else:
                    # Try convex combinations
                    for frac in [0.5, 0.3, 0.2, 0.1]:
                        y_test = (1 - frac) * y + frac * g_lp
                        y_test = np.maximum(y_test, 0)
                        y_test = project_simplex(y_test, S)
                        c_test = fft_convolve_self(y_test)
                        if c_test.max() < best:
                            best = c_test.max()
                            best_a = y_test.copy()
                            last_improve_it = it
                            y = y_test.copy()
                            a = y_test.copy()
                            t = 1.0
                            print("    LP %d (f=%.1f): %.6f" % (lp_count, frac, best))
                            break

        # FISTA step
        if it > 8000:
            grad = exact_subgrad_topk(y, c, k=3)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 4000:
            # Restart from best with perturbation
            y = best_a + rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(np.maximum(y, 0), S)
            a = y.copy()
            t = 1.0
            last_improve_it = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Targeted repulsion
        if it % 300 == 0 and it > 2000:
            c_rep = fft_convolve_self(y)
            lr = 0.06 * np.exp(-it / 100000.0)
            if lr > 0.001:
                y = targeted_repulsion(y, n, S, c_rep, lr=lr, top_frac=0.04)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best, lp_count


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best = np.inf
    global_a = None

    # Get initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Budget: %.0fs" % budget_s)

    # Primary length: try 16384
    n = 16384
    S = np.sqrt(2.0 * n)

    # Prepare initializations
    if given is not None:
        if len(given) == n:
            init_primary = given.copy()
        else:
            init_primary = interpolate_sequence(given, n)
        init_primary = normalize_sum(init_primary, n)
    else:
        init_primary = np.maximum(0.5 * (1 + np.cos(2 * np.pi * np.linspace(0, 1, n, endpoint=False))), 0.01)
        init_primary = normalize_sum(init_primary, n)

    # === Phase 1: Quick FISTA warmup (15% time) ===
    phase1_time = max(30.0, (deadline - time.monotonic()) * 0.15)
    print("=== Phase 1: FISTA warmup (n=%d, %.0fs) ===" % (n, phase1_time))
    a1, s1 = run_fista_aggressive(init_primary, n, S, phase1_time, rng,
                                   beta_init=50.0, beta_rate=0.003, beta_max=2000.0,
                                   repulsion_freq=200, repulsion_lr=0.08,
                                   exact_grad_after=3000)
    c1 = fft_convolve_self(a1)
    obj1 = 2.0 * n * c1.max() / (a1.sum() ** 2)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
    print("  Phase 1: max_b=%.6f, obj=%.6f" % (global_best, obj1))

    # === Phase 2: LP-assisted optimization (45% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        phase2_time = remaining * 0.45
        print("=== Phase 2: LP-assisted (n=%d, %.0fs) ===" % (n, phase2_time))
        a2, s2, nlp = run_lp_assisted(global_a, n, S, phase2_time, rng,
                                       lp_every=150, top_k=400)
        c2 = fft_convolve_self(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            print("  Phase 2: max_b=%.6f (LP calls: %d)" % (global_best, nlp))
        else:
            print("  Phase 2: no improvement (best=%.6f, p2=%.6f, LP: %d)" %
                  (global_best, c2.max(), nlp))

    # === Phase 3: Try different n ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Try n=8192
        n_alt = 8192
        S_alt = np.sqrt(2.0 * n_alt)
        if global_a is not None:
            alt_init = interpolate_sequence(global_a, n_alt)
            alt_init = normalize_sum(alt_init, n_alt)
        else:
            alt_init = np.ones(n_alt) * 0.01

        phase3_time = remaining * 0.3
        print("=== Phase 3: n=%d (%.0fs) ===" % (n_alt, phase3_time))
        a3, s3 = run_fista_aggressive(alt_init, n_alt, S_alt, phase3_time, rng,
                                       beta_init=50.0, beta_rate=0.003, beta_max=2000.0,
                                       repulsion_freq=200, repulsion_lr=0.08,
                                       exact_grad_after=3000)
        c3 = fft_convolve_self(a3)
        obj3 = 2.0 * n_alt * c3.max() / (a3.sum() ** 2)
        # Compare objectives (different n values)
        # We need to compare the actual objective values
        current_obj = 2.0 * len(global_a) * global_best / (global_a.sum() ** 2) if global_a is not None else np.inf
        if obj3 < current_obj:
            global_best = c3.max()
            global_a = a3.copy()
            print("  Phase 3 (n=%d): obj=%.6f (improved!)" % (n_alt, obj3))
        else:
            print("  Phase 3 (n=%d): obj=%.6f (vs %.6f)" % (n_alt, obj3, current_obj))

    # === Phase 4: Final intensive polish on best ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        print("=== Phase 4: Final polish (n=%d, %.0fs) ===" % (n_p, remaining))

        # High-beta exact subgradient FISTA
        polish_time = remaining * 0.5
        a4, s4 = run_fista_aggressive(global_a, n_p, S_p, polish_time, rng,
                                       beta_init=500.0, beta_rate=0.001, beta_max=5000.0,
                                       repulsion_freq=500, repulsion_lr=0.03,
                                       exact_grad_after=100,
                                       noise_freq=10000)
        c4 = fft_convolve_self(a4)
        if c4.max() < global_best:
            global_best = c4.max()
            global_a = a4.copy()
            print("  Phase 4a: max_b=%.6f" % global_best)
        else:
            print("  Phase 4a: no improvement (%.6f vs %.6f)" % (c4.max(), global_best))

        # Coordinate polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            print("  Phase 4b: Coordinate polish (%.0fs)" % remaining2)
            a5, s5 = coordinate_polish(global_a, n_p, S_p, remaining2, rng)
            c5 = fft_convolve_self(a5)
            if c5.max() < global_best:
                global_best = c5.max()
                global_a = a5.copy()
                print("  Phase 4b: max_b=%.6f" % global_best)
            else:
                print("  Phase 4b: no improvement (%.6f)" % c5.max())

    # === Phase 5: Diverse restarts if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_d = len(global_a)
        S_d = np.sqrt(2.0 * n_d)
        print("=== Phase 5: Diverse restarts (%.0fs) ===" % remaining)
        inits = generate_initializations(n_d, rng)
        current_obj = 2.0 * n_d * global_best / (global_a.sum() ** 2)
        per_init = max(10.0, (remaining * 0.5) / len(inits))
        for i, init in enumerate(inits[:5]):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 5:
                break
            ai, si = run_fista_aggressive(init, n_d, S_d, per_init, rng,
                                           beta_init=30.0, beta_rate=0.004, beta_max=2000.0,
                                           repulsion_freq=150, repulsion_lr=0.1,
                                           exact_grad_after=2000)
            ci = fft_convolve_self(ai)
            obji = 2.0 * n_d * ci.max() / (ai.sum() ** 2)
            if obji < current_obj:
                global_best = ci.max()
                global_a = ai.copy()
                current_obj = obji
                print("  Restart %d: obj=%.6f" % (i, obji))

    # Final
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final verification: %.6f" % ver if np.isfinite(ver) else "Final verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_obj = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / (global_a.sum() ** 2)
    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), final_obj))
    return [float(x) for x in global_a]
