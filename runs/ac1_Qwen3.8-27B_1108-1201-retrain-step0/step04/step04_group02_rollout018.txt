LP-based refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

            break

        a_try = project_simplex(a - step * grad, S)
        obj_try = compute_obj(a_try)
        
        if obj_try < cur_obj:
            a = a_try
            step = min(step * 1.03, 2.0 * S / (n * gnorm + 1e-15))
        else:
            step *= 0.7
            if step < 0.001:
                noise_scale = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 0.3

    return best_a, best_obj


def generate_flat_conv_init(n, rng):
    """
    Generate initialization based on 'flat convolution' principle.
    For non-negative a, edges should be larger.
    We construct a sequence whose convolution is approximately flat.
    """
    # Start from the recurrence: if conv(a,a) = c (constant),
    # then a[k] can be determined recursively.
    # a[0]^2 = c, a[1] = c/(2*a[0]), etc.
    # For a finite sequence, we can't make conv exactly constant,
    # but we can get close by starting from this shape.
    
    # Use the "edge-weighted" shape: a[k] ~ 1/sqrt(k+1) for small k
    # and symmetric on the other side
    t = np.arange(n)
    # Edge-weighted: large at edges, small in middle
    # Use form: a[k] = 1/sqrt(min(k, n-1-k) + alpha)
    alpha = 0.5
    dist_to_edge = np.minimum(t, n - 1 - t).astype(float)
    a = 1.0 / np.sqrt(dist_to_edge + alpha)
    
    # Add some random perturbation
    a = a * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    return a


def generate_varied_init(n, rng, variant=0):
    """Generate various initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    
    if variant == 0:
        # Edge-weighted (flat convolution heuristic)
        dist = np.minimum(t, 1 - t)
        return 1.0 / (dist + 0.05)
    elif variant == 1:
        # Uniform
        return np.ones(n)
    elif variant == 2:
        # Cosine window
        return 0.5 * (1 + np.cos(2 * np.pi * t))
    elif variant == 3:
        # Sinusoidal
        return np.sin(np.pi * t) ** 2
    elif variant == 4:
        # Random
        return rng.random(n) + 0.1
    elif variant == 5:
        # Inverse cosine (peaked at edges)
        return 1.0 / (np.abs(np.cos(np.pi * t)) + 0.1)
    elif variant == 6:
        # Two-hump
        return np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
    elif variant == 7:
        # Power law
        return (t + 0.01) ** (-0.3) + ((1 - t) + 0.01) ** (-0.3)
    elif variant == 8:
        # Sinusoidal with multiple frequencies
        return 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t)
    elif variant == 9:
        # Geometric decay from both edges
        return np.exp(-3 * t) + np.exp(-3 * (1 - t))
    else:
        return np.ones(n) * 0.5


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based direction finding with active-set constraint management.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    # Try to get previous best
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        prev_seq = np.clip(prev_seq, 0, 1000)
        if prev_seq.size < 64 or prev_seq.sum() < 0.01:
            prev_seq = None
    except Exception:
        prev_seq = None

    global_best = np.inf
    global_a = None

    print("Starting. Budget: %.0fs" % budget_s)
    t0 = time.monotonic()

    # Determine sequence length(s) to try
    lengths_to_try = []
    if prev_seq is not None and 1024 <= len(prev_seq) <= 32768:
        lengths_to_try.append(len(prev_seq))
    
    for L in [4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Phase 1: Exploration (30% of time)
    expl_time = (deadline - time.monotonic()) * 0.30
    print("Phase 1: Exploration (%.0fs)" % expl_time)

    candidates = []
    
    # Add previous best if available
    if prev_seq is not None:
        candidates.append(("prev", prev_seq.copy()))
    
    # Generate various initializations for the main length
    main_n = lengths_to_try[0]
    for v in range(10):
        init = generate_varied_init(main_n, rng, v)
        candidates.append(("init_%d" % v, init))

    # Flat convolution init
    cand_flat = generate_flat_conv_init(main_n, rng)
    candidates.append(("flat_conv", cand_flat))

    # Try a different length too
    if len(lengths_to_try) > 1:
        other_n = lengths_to_try[1]
        for v in [0, 4, 7]:
            init = generate_varied_init(other_n, rng, v)
            candidates.append(("n%d_v%d" % (other_n, v), init))

    # Quick refinement of each candidate
    per_cand_time = expl_time / max(1, len(candidates))
    
    for name, arr in candidates:
        if time.monotonic() > deadline - (deadline - t0) * 0.55:
            break
        n = len(arr)
        S = np.sqrt(2.0 * n)
        arr = np.maximum(arr, 0)
        s = arr.sum()
        if s < 0.01:
            continue
        arr = arr * (S / s)
        
        t_alloc = max(5, per_cand_time)
        if t_alloc < 3:
            continue
        
        a_opt, score = run_subgrad_refine(arr, n, S, t_alloc, rng)
        score = compute_obj(a_opt)
        
        if score < global_best:
            global_best = score
            global_a = a_opt.copy()
            print("  [%s] n=%d: obj=%.8f (elapsed %.0fs)" % (
                name, n, global_best, time.monotonic() - t0))

    print("After Phase 1: best=%.8f" % global_best)

    # Phase 2: LP Refinement (50% of time)
    remaining = deadline - time.monotonic()
    lp_time = remaining * 0.50
    print("Phase 2: LP Refinement (%.0fs)" % lp_time)

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        
        # Split LP time: main run + potential second run at different length
        main_lp_time = lp_time * 0.7
        a_lp, score_lp = run_lp_refine(global_a, n_best, S_best, main_lp_time, rng)
        score_lp = compute_obj(a_lp)
        if score_lp < global_best:
            global_best = score_lp
            global_a = a_lp.copy()
            print("  After LP refine: %.8f" % global_best)
        
        # Try LP at a different length if time permits
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            other_n = 8192 if n_best != 8192 else 4096
            S_other = np.sqrt(2.0 * other_n)
            init_other = generate_flat_conv_init(other_n, rng)
            a_o, score_o = run_lp_refine(init_other, other_n, S_other, remaining2 * 0.5, rng)
            score_o = compute_obj(a_o)
            if score_o < global_best:
                global_best = score_o
                global_a = a_o.copy()
                print("  After LP at n=%d: %.8f" % (other_n, global_best))

    # Phase 3: Final polish (20% of time)
    remaining = deadline - time.monotonic()
    polish_time = remaining * 0.60
    print("Phase 3: Polish (%.0fs)" % polish_time)

    if global_a is not None and polish_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        
        # Another round of LP refinement (should converge further)
        a_pol, score_pol = run_lp_refine(global_a, n_best, S_best, polish_time, rng)
        score_pol = compute_obj(a_pol)
        if score_pol < global_best:
            global_best = score_pol
            global_a = a_pol.copy()
            print("  After polish: %.8f" % global_best)

    # Final: try subgradient micro-steps
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_sub, score_sub = run_subgrad_refine(global_a, n_best, S_best, remaining, rng)
        score_sub = compute_obj(a_sub)
        if score_sub < global_best:
            global_best = score_sub
            global_a = a_sub.copy()
            print("  After final subgrad: %.8f" % global_best)

    # Verification
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.8f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning n=%d, best=%.8f, time=%.0fs" % (len(final_list), global_best, time.monotonic() - t0))
    return final_list
