import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 1:
        return np.array([min(max(S, 0), 1000.0)])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    """LSE smooth max."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        return cm + np.log(len(c)) / beta  # approximation for very large beta
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta

def lse_gradient(a, c, beta):

    n = len(a)
    cm = c.max()

    if scaled.max() > 500:

        idx = np.argmax(c)
        w = np.zeros(len(c))
        # Use top few for numerical stability
        k = min(4, len(c))
        top = np.argsort(c)[::-1][:k]
        for i in top:
            w[i] = 1.0
        w = w / w.sum()
    else:
        e = np.exp(scaled)
        w = e / e.sum()

    g2 = np.real(ifft(A * np.conj(W)))[:n]  # cross-corr of a and w

        best_a = a.copy()

def run_coordinate_descent(a_init, S, max_time, rng):
    """Simple coordinate descent: try small changes to individual coordinates."""
    a = project_simplex(a_init.copy(), S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # Precompute the full convolution
    c = fft_convolve_self(a)
    
    while time.monotonic() - start < max_time:
        # Pick a random coordinate to modify
        i = rng.integers(0, n)
        
        # Try a few different amounts
        for delta in rng.choice([-1, 1]) * rng.uniform(0.001, 0.01) * S / n:
            a_try = a.copy()
            a_try[i] = max(0, a_try[i] + delta)
            
            # Compensate by adjusting another coordinate
            j = rng.integers(0, n)
            if j == i:
                continue
            a_try[j] = max(0, a_try[j] - delta)
            
            # Renormalize
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try * (S / s)
            
            # Check if it helped (use a fast approximation)
            # Only recompute convolution if the change might help
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best:
                best = c_try.max()
                best_a = a_try.copy()
                a = a_try
                c = c_try
                break
    
    return best_a, best

def run_excess_annealing(a_init, S, max_time, rng):
    """Anneal a threshold, minimizing excess above threshold."""
    a = project_simplex(a_init.copy(), S)
    n = len(a)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    c0 = fft_convolve_self(a)
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    
    y = a.copy()
    t = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = T_start + (T_end - T_start) * (progress ** 0.5)
        T = max(T, T_end)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        if cur_max <= T * (1 + 1e-6):
            T_use = max(c.mean(), cur_max - 0.001 * (cur_max - c.mean() + 1e-10))
        else:
            T_use = T
        
        # Excess gradient
        excess = np.maximum(c - T_use, 0.0)
        ex_sum = excess.sum()
        if ex_sum < 1e-15:
            # Use LSE as fallback
            grad = lse_gradient(y, c, 100.0)
        else:
            w = excess / ex_sum  # normalize to probability
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(y, fsize)
            g1 = np.real(ifft(W * np.conj(A)))[:n]
            g2 = np.real(ifft(A * np.conj(W)))[:n]
            grad = g1 + g2
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 0.5 / (beta_max_est(c, S, n) + 1e-10)
        step = step * (1.0 + 0.5 * (1.0 - progress))
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 2000 == 0 and rng.random() < 0.05:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def beta_max_est(c, S, n):
    """Estimate Lipschitz constant."""
    return 2.0 * c.max() + S * S / n

def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * ((tt - 0.5) ** 2) / (0.09)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt + 0.1))
    inits.append(rng.random(n) + 0.1)
    
    return [np.maximum(x, 0.01) for x in inits]

def upscale_sequence(a_small, n_large):
    """Upscale a sequence by interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)

def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: LSE-FISTA, top-k FISTA, excess annealing, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    # Main length: use the given one if reasonable, else 8192
    n_main = 8192
    if given is not None and 2048 <= len(given) <= 16384:
        n_main = len(given)
    
    S = np.sqrt(2.0 * n_main)
    
    print("Main length: %d, S = %.2f" % (n_main, S))
    print("Initial objective: %.6f" % objective_value(given) if given is not None else "No given sequence")
    
    # Phase 1: LSE-FISTA on the given sequence (main workhorse)
    if given is not None:
        a_work = np.clip(given.copy(), 0, 1000)
        s_work = a_work.sum()
        if s_work > 0:
            a_work = a_work * (S / s_work)
        
        time_phase1 = min(600, (deadline - time.monotonic()) * 0.3)
        print("\n=== Phase 1: LSE-FISTA (%.0fs) ===" % time_phase1)
        
        a1, val1 = run_lse_fista(a_work, S, time_phase1, rng,
                                   beta_init=20.0, beta_max=8000.0, beta_rate=0.003)
        obj1 = objective_value(a1)
        print("Phase 1 result: obj=%.6f" % obj1)
        
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
    
    # Phase 2: Top-k focused FISTA
    if global_a is not None:
        remaining = deadline - time.monotonic()
        time_phase2 = min(400, remaining * 0.25)
        if time_phase2 > 30:
            print("\n=== Phase 2: Top-k FISTA (%.0fs) ===" % time_phase2)
            a2, val2 = run_topk_fista(global_a, S, time_phase2, rng, k=8, alpha=50.0, alpha_max=500.0)
            obj2 = objective_value(a2)
            print("Phase 2 result: obj=%.6f" % obj2)
            if obj2 < global_best:
                global_best = obj2
                global_a = a2.copy()
    
    # Phase 3: Excess annealing
    if global_a is not None:
        remaining = deadline - time.monotonic()
        time_phase3 = min(400, remaining * 0.25)
        if time_phase3 > 30:
            print("\n=== Phase 3: Excess annealing (%.0fs) ===" % time_phase3)
            a3, val3 = run_excess_annealing(global_a, S, time_phase3, rng)
            obj3 = objective_value(a3)
            print("Phase 3 result: obj=%.6f" % obj3)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()
    
    # Phase 4: Another round of LSE-FISTA with higher beta
    if global_a is not None:
        remaining = deadline - time.monotonic()
        time_phase4 = min(300, remaining * 0.3)
        if time_phase4 > 30:
            print("\n=== Phase 4: High-beta LSE-FISTA (%.0fs) ===" % time_phase4)
            a4, val4 = run_lse_fista(global_a, S, time_phase4, rng,
                                       beta_init=200.0, beta_max=20000.0, beta_rate=0.002)
            obj4 = objective_value(a4)
            print("Phase 4 result: obj=%.6f" % obj4)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
    
    # Phase 5: Try a different length if time allows
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 200:

            n_alt = 2 * n_main if n_main < 8192 else n_main // 2
            S_alt = np.sqrt(2.0 * n_alt)
            
            # Upscale or downscale
            if n_alt > n_main:
                a_alt_init = upscale_sequence(global_a, n_alt)
            else:

                a_alt_init = global_a[:n_alt * ratio].reshape(n_alt, ratio).mean(axis=1)
            
            time_phase5 = min(200, remaining * 0.5)

            print("Phase 5 result: obj=%.6f" % obj5)
            if obj5 < global_best:

                global_a = a5.copy()

        final_list = [float(x) for x in global_a]
