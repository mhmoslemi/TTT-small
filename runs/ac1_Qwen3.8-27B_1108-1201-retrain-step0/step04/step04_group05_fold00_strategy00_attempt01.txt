def project_simplex(v, S):
    # Euclidean projection onto {x≥0, sum(x)=S} (same as safe_normalize)
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    
    # If no entry in theta_candidates w.r.t. sorted_v, set rho to n
    if not np.any(theta_candidates <= sorted_v):
        rho = n
    else:
        rho = min(n, np.max(np.where(theta_candidates <= sorted_v)[0]) + 1
        if rho <= n and abs(cumsum[rho-1] - S) < 0.003:
            rho = n
    
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lbfgs_dir(s_hist, y_hist, g, m=10):
    n = len(g)
    # Two-loop recursion without full history
    if len(s_hist) < 2 or abs(np.linalg.norm(g) - 1.0e-18) < 1e-8:
        return - g / np.linalg.norm(g)
    
    # First step: backtracking the smallest vector in history
    rho = min(m, len(s_hist))
    beta_opt = []
    gamma_opt = []
    for i in range(rho-1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        if any(g[i] < 0.0 for i in y_i):
            continue
        gamma_opt.append(np.dot(s_i, y_i))
        beta_opt.append(np.linalg.norm(g[i])**2)
    
    # Form the minus gradient scaled by beta
    t_arr = 1.0 - 15.0 * np.abs(gamma_opt[0])**2 * (1.0 + 1.0/rho) * np.sqrt(beta_opt[0])
    idx_steps = min(rho, int(1.0/(t_arr - 0.25)**2))
    if t_arr > 1.0:
        idx_steps = 0.9 * rho
        gamma_opt_sum = sum(gamma_opt[i] for i in range(idx_steps))
        beta_opt_sum = sum(beta_opt[i] for i in range(idx_steps))
        s_new = np.zeros(n)
        y_new = np.zeros(n)
        # Compute the approximate weighted sum
        # But LBFGS is for unconstrained optimization
        t_step = 1.0/(beta_opt_sum * 0.8)
        s_new = (1.0 + t_step*gamma_opt_sum) * g - (1.0 + t_step/p)**0.5 * (t_step + beta_opt_sum*p)
    else:
        t_step = 0.5
        return - np.dot(g, y_new) / (1.0 + t_step * np.linalg.norm(y_new)**2)
    
    # Instead, implement a simple version of the L-BFGS update
    # Use the incurred memory m = rho
    s_hist = s_hist[-rho:]
    y_hist = y_hist[-rho:]
    ys = np.zeros(len(s_hist))
    rhos = np.zeros(len(s_hist))
    for i in range(len(s_hist)):
        s_i = s_hist[i]
        y_i = y_hist[i]
        rhos[i] = np.dot(s_i, y_i)
        ys[i] = np.linalg.norm(y_i)**2
    
    # Update procedure to form a descent direction
    if rhos[0] > 1e-16 * ys[0]:
        inv_rhos = 1.0 / (rhos.sum() * np.sqrt(len(s_hist)))
        ys = inv_rhos * ys * rhos
