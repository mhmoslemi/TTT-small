"""
Minimize 2*n*max_conv/sum^2 using multi-strategy optimization:
1. Spectral initialization (flat magnitude, random phase)
2. LSE-smoothed L-BFGS with adaptive beta
3. Peak-targeted coordinate updates
4. Multi-length search (try n=4096, 8192, 16384)
5. Basin hopping with re-optimization

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_objective_and_grad(a, N, beta):
    """Compute LSE-smoothed peak and its gradient."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    
    # Gradient: d/d a_i [LSE] = sum_k w_k * 2 * a_{k-i}
    # This is a cross-correlation of w with a
    fsize = 1 << (3 * n + 2).bit_length()
    w_full = np.zeros(2 * n - 1)
    w_full[:] = w
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def spectral_init(n, N, rng, method="random_phase"):
    """Generate initial sequence using spectral methods."""
    if method == "random_phase":
        # Flat magnitude, random phase
        mag = np.ones(N)
        phase = rng.uniform(0, 2 * np.pi, N)
        # Ensure Hermitian symmetry for real output
        A = mag * np.exp(1j * phase)
        A = (A + np.conj(A[::-1])) / 2  # Make Hermitian (approximate)
        A[0] = np.abs(A[0])
        if N % 2 == 0:
            A[N//2] = np.abs(A[N//2])
        a = np.real(ifft(A))[:n]
    elif method == "raised_cos":
        # Raised cosine window
        t = np.arange(n)
        a = (1 + np.cos(2 * np.pi * t / n)) / 2
    elif method == "gaussian":
        # Gaussian window
        t = (np.arange(n) - n/2) / (n/4)
        a = np.exp(-t**2)
    elif method == "sinc_like":
        # Sinc-like (truncated)
        t = np.arange(n) - n/2
        a = np.sinc(t / (n/4))
    elif method == "binary_random":
        a = (rng.random(n) > 0.5).astype(float)
    elif method == "quadratic_residue":
        # Based on quadratic residues mod p
        p = 1
        while p < n:
            p *= 2
        # Simple pseudo-random based on quadratic residues
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 if ((i*i) % 7 < 3) else 0.0
    else:
        a = rng.random(n)
    
    return np.abs(a) + 0.01


def lbfgs_two_loop(s_hist, y_hist, g):
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.1)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
        
        if it % 50 == 0:

                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a_new = a

        c_new = c

            if cm_ls <= c_max or ls == 11:

def peak_targeted_update(a, n, S, N, top_m, eta, iters):
    """Directly target the top-m peaks for reduction."""
    a = a.copy()
    c, A = compute_conv(a, N)
    
    for _ in range(iters):
        peak_idx = np.argsort(c)[::-1][:top_m]
        peak_vals = c[peak_idx]
        c_max = c.max()
        # Weights proportional to excess above mean
        mean_c = c.mean()
        excess = np.maximum(0, peak_vals - mean_c)
        if excess.sum() < 1e-30:
            break
        w = excess / excess.sum()
        
        # Compute pressure on each coordinate
        # pressure_i = sum_k w_k * a_{k-i} (with valid indices)
        # Use FFT
        fsize = 1 << (3 * n + 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(peak_idx):
            w_full[int(k)] = w[i]
        W = fft(w_full)
        Af = fft(a, fsize)
        pressure = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        # Update: reduce coordinates with high pressure
        step = eta * S / n
        a_new = a - step * pressure / max(np.max(np.abs(pressure)), 1e-30)
        a_new = project_simplex(a_new, S)
        
        c, A = compute_conv(a_new, N)
        if c.max() <= c_max:
            a = a_new
        else:
            # Try smaller step
            step = eta * S / (4 * n)
            a_new = a - step * pressure / max(np.max(np.abs(pressure)), 1e-30)
            a_new = project_simplex(a_new, S)
            c, A = compute_conv(a_new, N)
            if c.max() < c_max:
                a = a_new
    
    return a


def iterative_spectral_flatness(a_init, n, N, eta, iters):
    """Iteratively push the power spectrum toward flatness."""
    a = safe_normalize(a_init, n, np.sqrt(2.0 * n))
    S = a.sum()
    
    for _ in range(iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        # Push P toward flat: P_new = P * exp(-eta * (P - P_mean) / P_mean)
        P_new = P * np.exp(-eta * (P - P_mean) / max(P_mean, 1e-30))
        P_new = np.maximum(P_new, 1e-30)
        
        # Keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.abs(a_new)  # Non-negativity
        a = safe_normalize(a_new, n, S)
    
    return a

"""
Minimize 2*n*max_conv/sum^2 using multi-strategy optimization combining:
1. Low-frequency spectral parameterization (reduced-space global optimization)
2. LSE-smoothed L-BFGS for local refinement
3. Peak-targeted coordinate updates
4. Basin hopping for global exploration
5. Multi-length search
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, N, beta):
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    fsize = 1 << (3 * n + 2).bit_length()
    w_full = np.zeros(2 * n - 1)
    w_full[:] = w
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return c_max, grad, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def optimize_lbfgs(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    _, g, c, c_max = lse_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.1)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            _, g, c, c_max = lse_grad(a, N, beta)
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        if np.linalg.norm(direction) < 1e-18:
            break
        
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax - 1e-10:
            best_cmax = cmax_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                a_new = project_simplex(a_new + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                s_hist = []
                y_hist = []
                no_improve = 0
        
        _, g_new, c_new, cmax_nw = lse_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        if np.dot(s_k, y_k) > 1e-14 * np.dot(s_k, s_k):
            s_hist.append(s_k)
            y_hist.append(y_k)
        
        a = a_new
        g = g_new
        c = c_new
        c_max = cmax_nw
    
    return best_a, best_cmax


def low_freq_sequence(params, n, S):
    """Generate sequence from low-frequency parameters."""
    K = params.size // 2
    alphas = params[:K]
    phis = params[K:2*K]
    t = np.arange(n)
    a = np.ones(n)
    for k in range(1, K + 1):
        a += alphas[k-1] * np.cos(2.0 * np.pi * k * t / n + phis[k-1])
    a = np.maximum(a, 0.05)
    return a * S / a.sum()


def low_freq_score(params, n, N, S):
    """Score for low-frequency parameterized sequence."""
    a = low_freq_sequence(params, n, S)
    c, _ = compute_conv(a, N)
    return c.max()


def spectral_init_flat(n, N, rng):
    """Initialize with flat magnitude, random phase."""
    mag = np.ones(N)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = mag * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    A[0] = np.abs(A[0])
    if N % 2 == 0:
        A[N//2] = np.abs(A[N//2])
    for i in range(1, (N+1)//2):
        A[N-i] = np.conj(A[i])
    a = np.real(ifft(A))[:n]
    return np.abs(a) + 0.01


def iterative_spectral_flatness(a, n, N, eta, iters):
    """Push power spectrum toward flatness iteratively."""
    S = a.sum()
    for _ in range(iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        P_new = P * np.exp(-eta * (P - P_mean) / max(P_mean, 1e-30))
        P_new = np.maximum(P_new, 1e-30)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        # Enforce Hermitian
        A_new[0] = np.abs(A_new[0])
        if N % 2 == 0:
            A_new[N//2] = np.abs(A_new[N//2])
        for i in range(1, (N+1)//2):
            A_new[N-i] = np.conj(A_new[i])
        a_new = np.real(ifft(A_new))[:n]
        a = safe_normalize(np.abs(a_new), n, S)
    return a


def peak_targeted_update(a, n, S, N, top_m, eta, iters):
    """Target top peaks for direct reduction."""
    a = a.copy()
    c, _ = compute_conv(a, N)
    
    for _ in range(iters):
        peak_idx = np.argsort(c)[::-1][:top_m]
        peak_vals = c[peak_idx]
        c_max = c.max()
        mean_c = c.mean()
        excess = np.maximum(0, peak_vals - mean_c)
        if excess.sum() < 1e-30:
            break
        w = excess / excess.sum()
        
        fsize = 1 << (3 * n + 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(peak_idx):
            w_full[int(k)] = w[i]
        W = fft(w_full)
        Af = fft(a, fsize)
        pressure = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        step = eta * S / n
        a_new = project_simplex(a - step * pressure / max(np.max(np.abs(pressure)), 1e-30), S)
        c, _ = compute_conv(a_new, N)
        if c.max() <= c_max:
            a = a_new
        else:
            step = eta * S / (4 * n)
            a_new = project_simplex(a - step * pressure / max(np.max(np.abs(pressure)), 1e-30), S)
            c, _ = compute_conv(a_new, N)
            if c.max() < c_max:
                a = a_new
    
    return a

    global_n = 0

        nonlocal global_best, global_a, global_n
        a = safe_normalize(a, n, np.sqrt(2.0 * float(n)))
        c, _ = compute_conv(a, N)
        score = c.max()
        if score < global_best:
