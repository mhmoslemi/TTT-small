import numpy as np
import time

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_candidates
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def time_domain_refine(a, n, S, max_time, rng, n_top=20):
    """
    Time-domain refinement: for the peak lag, adjust the contributing coordinates.
    """
    a = a.copy()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        k_peak = int(np.argmax(c))
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Find top-contributing pairs at the peak lag
        # c[k_peak] = sum_i a[i] * a[k_peak - i]
        contribs = np.zeros(n)
        low = max(0, k_peak - n + 1)
        high = min(k_peak, n - 1)
        for i in range(low, high + 1):
            j = k_peak - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_coords:
            i = int(idx)
            j = k_peak - i
            if j < 0 or j >= n or i == j:
                continue
            
            orig_i = a[i]
            orig_j = a[j]
            
            # Try reducing a[i] and increasing a[j] (or vice versa)
            for frac in [0.8, 0.6, 0.4, 1.2, 1.4]:
                a[i] = orig_i * frac
                # Simple check: recompute only the affected lags
                # For speed, just check the peak lag
                new_peak = c[k_peak] + (a[i] - orig_i) * a[j]
                # Also check adjacent lags
                if new_peak < best_score:
                    improved = True
                    break
                a[i] = orig_i
            
            if not improved:
                # Try the other direction
                for frac in [0.8, 0.6, 0.4]:
                    a[j] = orig_j * frac
                    new_peak = c[k_peak] + (a[j] - orig_j) * a[i]
                    if new_peak < best_score:
                        improved = True
                        break
                    a[j] = orig_j
            
            a = project_simplex(a, S)
        
        if not improved:
            # Perturb
            idx = rng.integers(n)
            a[idx] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)
    
    return best_a, best_score


def structured_init(n, rng, method="gaussian"):
    """Create structured initializations."""
    t = np.arange(n) / n
    if method == "gaussian":
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif method == "sin":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * np.sin(2 * np.pi * 11 * t) + 0.1 * rng.standard_normal(n)
    elif method == "bimodal":
        a = 0.5 + 0.8 * np.exp(-((t - 0.3) * 5) ** 2) + 0.8 * np.exp(-((t - 0.7) * 5) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.5, 0.5) + 0.1 * rng.standard_normal(n)
    elif method == "smoothed_random":
        a = rng.random(n)
        # Apply a simple smoothing
        kernel = np.ones(8) / 8
        a = np.convolve(a, kernel, mode='same')
        a += 0.1 * rng.standard_normal(n)
    else:
        a = 1.0 + 0.5 * rng.standard_normal(n)
    
    return np.clip(a, 0.01, None)

"""
Minimize 2*n*max_conv/sum^2 using iterative spectral peak shaving with
momentum-based frequency modification, adaptive multi-lag targeting,
and strategic restarts. Uses n=16384 for better attack surface.

Key ideas:
1. Spectral momentum: accumulate cosine patterns across iterations for smoother shaping
2. Adaptive top-k: focus on main peak first, then handle secondary peaks
3. Multi-start with structured and random initializations
4. Time-domain polish on top-contributing coordinates
5. Strategic perturbations to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_arr
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_c_from_P(P, N, n):
    """Compute convolution from power spectrum."""
    c = np.real(ifft(P))[:2 * n - 1]
    return c


def spectral_optimize(a_init, n, N, S, max_time, rng,
                      initial_top_k=3, max_top_k=60,
                      initial_alpha=0.35, target=1.44,
                      momentum_decay=0.93, name=""):
    """
    Main spectral optimization loop.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_score = np.inf
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad, N)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    best_score = c.max()
    
    P_momentum = np.zeros(N)
    m = np.arange(N)
    it = 0
    stall = 0
    last_improve_it = 0
    
    while time.monotonic() < max_time:
        it += 1
        
        # Find top-k lags
        top_k = min(max_top_k, initial_top_k + it // 25)
        top_indices = np.argsort(c)[::-1][:top_k]
        c_max = c[top_indices[0]]
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
            stall = 0
            last_improve_it = it
        
        if c_max <= target:
            break
        
        # Weights for top lags
        # Emphasize the gap above target
        gaps = np.maximum(c[top_indices] - target, 0.0)
        if gaps.sum() < 1e-10:
            gaps = c[top_indices]
        weights = gaps / gaps.sum()
        
        # Adaptive alpha
        alpha = initial_alpha / (1.0 + 0.008 * it)
        # Scale by how far above target we are
        alpha *= min(3.0, max(0.3, (c_max - target) / (0.1 * c_max + 1e-10)))
        
        # Compute effective cosine
        cos_eff = np.zeros(N)
        for i in range(top_k):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(top_indices[i]) / N)
        
        # Update momentum
        P_momentum = momentum_decay * P_momentum + (1.0 - momentum_decay) * np.maximum(cos_eff, 0.0)
        P_momentum[0] = 0
        
        # Apply spectral modification
        P_new = P * np.exp(-alpha * P_momentum)
        P_new[0] = P[0]
        P_new = np.maximum(P_new, 0.0)
        
        # Occasional random perturbation
        if rng.random() < 0.08:
            sigma = 0.008 * alpha
            P_new[1:] += rng.normal(0, sigma, N - 1)
            P_new = np.maximum(P_new, 0.0)
            P_new[0] = P[0]
        
        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Evaluate
        a_pad[:n] = a_new
        A_new_full = fft(a_pad, N)
        P_new_full = np.abs(A_new_full) ** 2
        c_new = np.real(ifft(P_new_full))[:2 * n - 1]
        new_cmax = c_new.max()
        
        if new_cmax < c_max:
            a = a_new
            A = A_new_full
            P = P_new_full
            c = c_new
        else:
            stall += 1
            if stall > 40:
                # Try larger perturbation
                pert_scale = 0.003 * S / np.sqrt(n) * min(3.0, stall / 20.0)
                a = project_simplex(a + rng.normal(0, pert_scale, n), S)
                P_momentum = np.zeros(N) * rng.random(N) * 0.1
                a_pad[:n] = a
                A = fft(a_pad, N)
                P = np.abs(A) ** 2
                c = np.real(ifft(P))[:2 * n - 1]
                stall = 0
    
    return best_a, best_score


def td_refine(a_init, n, S, max_time, rng, n_top=15):
    """
    Time-domain refinement on top-contributing coordinates at the peak lag.
    """
    a = a_init.copy()
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c_max = c.max()
        k_peak = int(np.argmax(c))
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Contributions at peak lag
        low = max(0, k_peak - n + 1)
        high = min(k_peak, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = k_peak - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_coords = np.argsort(contribs)[::-1][:n_top]
        
        improved_this_round = False
        for idx in top_coords:
            i = int(idx)
            j = k_peak - i
            if j < 0 or j >= n or i == j:
                continue
            
            orig_i = a[i]
            best_new_cmax = c_max
            best_frac = 1.0
            
            for frac in [0.7, 0.85, 0.5, 1.3, 0.3]:
                new_val = orig_i * frac
                # Estimate change in c[k_peak]
                # c[k_peak] changes by (new_val - orig_i) * a[j]
                # Also affects other lags, but we approximate
                delta = (new_val - orig_i) * a[j]
                approx_new_peak = c_max + delta
                
                if approx_new_peak < best_new_cmax:
                    best_new_cmax = approx_new_peak
                    best_frac = frac
            
            if best_frac != 1.0:
                a[i] = orig_i * best_frac
                improved_this_round = True
        
        if improved_this_round:
            a = project_simplex(a, S)
            # Periodic full recompute
            c = fft_convolve_self(a)
        else:
            # Perturb a few coordinates
            for _ in range(3):
                idx = rng.integers(n)
                a[idx] *= rng.uniform(0.6, 1.4)
            a = project_simplex(a, S)
            c = fft_convolve_self(a)
    
    return best_a, best_score


def make_init(n, rng, method):
    """Create a structured initialization."""
    t = np.arange(n) / n
    if method == "gaussian":
        a = 1.0 + 0.4 * rng.standard_normal(n)
    elif method == "sinusoidal":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) \
                  + 0.2 * np.sin(2 * np.pi * 7 * t) \
                  + 0.15 * np.sin(2 * np.pi * 13 * t) \
                  + 0.05 * rng.standard_normal(n)
    elif method == "bimodal":
        a = 0.4 + 0.9 * np.exp(-((t - 0.25) * 4) ** 2) \
                    + 0.9 * np.exp(-((t - 0.75) * 4) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif method == "alternating":
        base = np.where(np.arange(n) % 2 == 0, 1.4, 0.6)
        a = base + 0.1 * rng.standard_normal(n)
    elif method == "piecewise":
        n_pieces = 128
        piece_len = n // n_pieces
        heights = 0.3 + 1.0 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, 0.5 + 0.5 * rng.random(n - len(a))])
        a = a[:n]
    else:
        a = 0.5 + 1.0 * rng.random(n)
    
    return np.clip(a, 0.001, None)

                prev_resized = prev.copy()
