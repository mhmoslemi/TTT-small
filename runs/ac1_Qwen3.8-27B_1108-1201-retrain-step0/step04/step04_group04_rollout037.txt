import numpy as np

from scipy.fft import fft, ifft, next_fast_len


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_normalized(a):
    """Compute score for a sequence (normalizes internally)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_norm = normalize_sum(a, n)
    c = fft_convolve_self(a_norm)
    return c.max(), a_norm


def spectral_gs(a_init, n, S, max_time, rng, n_restarts=3):
    """
    Improved spectral Gerchberg-Saxton with:
    - Geometric interpolation between current and target spectrum
    - Soft non-negativity projection
    - Phase randomization for diversity
    - Adaptive flattening strength
    """
    a = normalize_sum(a_init, n)
    L = next_fast_len(2 * n - 1)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    for restart in range(n_restarts):
        # For restarts > 0, randomize phases
        if restart > 0:
            a_pad = np.zeros(L)
            a_pad[:n] = a
            F = fft(a_pad)
            mag = np.abs(F)
            phase = np.exp(1j * rng.uniform(0, 2 * np.pi, L))
            # Keep DC component
            F[0] = F[0].real  # DC is real and positive
            F_new = phase * mag
            a = np.real(ifft(F_new))[:n]
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * (S / s)

        it = 0
        alpha = 0.3 + 0.2 * restart  # Start with less aggressive flattening
        soft_beta = 5.0

        while time.monotonic() - start < max_time:
            it += 1

            a_pad = np.zeros(L)
            a_pad[:n] = a
            F = fft(a_pad)
            P = np.abs(F) ** 2
            P_mean = P.mean()
            if P_mean < 1e-30:
                break

            # Geometric interpolation towards flat
            # P_new = P^alpha * (P_mean)^((1-alpha) - alpha) ... 
            # Actually: P_new = P^alpha * P_mean^(1-alpha) gives geometric interp
            P_target = np.full(L, P_mean)
            # Geometric interpolation (in log domain for stability)
            log_P = np.log(np.maximum(P, 1e-30))
            log_target = np.log(P_mean)
            log_new = (1.0 - alpha) * log_P + alpha * log_target
            P_new = np.exp(log_new)

            # Ensure total power is conserved
            P_scale = P.sum() / P_new.sum()
            P_new *= P_scale

            mag_new = np.sqrt(P_new)
            phase = F / np.maximum(np.abs(F), 1e-30)
            F_new = phase * mag_new

            a_raw = np.real(ifft(F_new))[:n]

            # Soft non-negativity: use power mapping
            # a_proj = a_raw^p normalized (for p>1, suppresses small values)
            if soft_beta > 1.01:
                # Soft projection: a * sigmoid-style
                a_clipped = np.maximum(a_raw, 0.0)
                s = a_clipped.sum()
                if s < 1e-15:
                    a = np.ones(n) * S / n
                else:
                    a = a_clipped * (S / s)
            else:
                a = np.maximum(a_raw, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = np.ones(n) * S / n
                else:
                    a = a * (S / s)

            # Evaluate
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best:
                best = cur
                best_a = a.copy()

            # Adapt alpha: increase over time for more aggressive flattening
            alpha = min(0.95, alpha + 0.003)
            # Periodically reduce alpha to allow exploration
            if it % 200 == 0:
                alpha = max(0.3, alpha * 0.7)
                soft_beta = max(1.0, soft_beta * 0.8)

            if it > 500 and cur - best > 0.01:
                # No improvement, break this restart
                break

    return best_a, best


def spectral_gs_v2(a_init, n, S, max_time, rng):
    """
    Alternative spectral GS using direct magnitude optimization.
    Uses a windowed flat target and momentum.
    """
    a = normalize_sum(a_init, n)
    L = next_fast_len(2 * n - 1)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    a_prev = a.copy()
    momentum = 0.7
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(L)
        a_pad[:n] = a
        F = fft(a_pad)
        P = np.abs(F) ** 2
        P_mean = P.max()  # Use max as reference for "peaks"

        if P_mean < 1e-30:
            break

        # Target: flatten peaks while preserving overall shape
        # Use soft thresholding in spectral domain
        threshold = 0.5 * P_mean
        P_soft = np.where(P > threshold, threshold + (P - threshold) * 0.5, P)
        
        # Blend
        progress = min(1.0, it / 1000.0)
        blend = 0.3 + 0.5 * progress
        P_new = (1.0 - blend) * P + blend * P_soft

        # Normalize to preserve energy
        P_new *= P.sum() / max(P_new.sum(), 1e-30)

        mag_new = np.sqrt(P_new)
        phase = F / np.maximum(np.abs(F), 1e-30)
        F_new = phase * mag_new
        a_new = np.real(ifft(F_new))[:n]

        # Project
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        # Momentum
        a_new = momentum * a_new + (1.0 - momentum) * a_prev
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a_prev = a
        a = a_new

        # Adapt momentum
        momentum = min(0.9, momentum + 0.001)

    return best_a, best


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = next_fast_len(len(w) + n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista_improved(a_init, n, S, max_time, rng, beta_init=50.0,
                       beta_max=5000.0, beta_ramp=2000):
    """Improved FISTA with better hybrid scheduling and adaptive steps."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    patience = 0

    while time.monotonic() - start < max_time:
        it += 1
        if it > 50000:
            break

        # Adaptive beta with plateaus
        progress = min(1.0, it / beta_ramp)
        beta = beta_init + (beta_max - beta_init) * progress
        # Periodic beta reduction to escape
        if it % 3000 == 0 and it > 1000:
            beta = max(beta_init, beta * 0.3)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-8:
            best = cur
            best_a = y.copy()
            last_improve = it
            patience = 0
        else:
            patience += 1

        # Choose gradient: hybrid approach
        use_exact = (beta > 500 and rng.random() < min(0.5, 0.1 * (beta / 1000.0)))
        if use_exact:
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for subgradient: use diminishing steps
            step = S / (gnorm * np.sqrt(1.0 + 0.01 * it))
        else:
            grad, _ = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            # Reduce step if stuck
            if patience > 500:
                step *= 0.5
            if patience > 1500:
                step *= 0.5

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize to prevent drift
        s = y.sum()
        if s > 0:
            y = y * (S / s)
        a = x_new
        t = t_new

        # Noise injection to escape local minima
        if patience > 2000 and patience % 500 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-patience / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                patience = 0

        if patience > 10000:
            break

    return best_a, best


def peak_directed_cd(a_init, n, S, max_time, rng, top_k=5):
    """
    Peak-directed coordinate descent.
    Focuses on reducing the actual peaks in the convolution.
    Uses structured pair selection based on peak contributions.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        improved_this_round = False

        # Identify top peaks
        peak_lags = np.argsort(c)[::-1][:top_k]

        # Generate candidate pairs
        n_pairs = min(100 + it * 2, 400)
        tried_pairs = set()

        for _ in range(n_pairs):
            if time.monotonic() - start > max_time - 2:
                break

            # Select pair: bias towards indices that contribute to peaks
            peak = peak_lags[rng.integers(0, len(peak_lags))]
            
            # Indices contributing to this peak
            lo = max(0, int(peak) - n + 1)
            hi = min(int(peak), n - 1)
            contributing = np.arange(lo, hi + 1)
            
            if len(contributing) < 2:
                continue

            # Pick j from contributing (high a value) and m elsewhere
            j_candidates = contributing[np.argsort(a[contributing])[::-1][:max(1, len(contributing)//4)]]
            m_candidates = np.array(list(set(range(n)) - set(contributing)))
            if len(m_candidates) == 0:
                m_candidates = np.array([rng.integers(0, n)])
            
            # Prefer high a[m] for m (to have more to give)
            m_candidates = m_candidates[np.argsort(a[m_candidates])[::-1][:max(1, len(m_candidates)//4)]]

            j = int(rng.choice(j_candidates))
            m = int(rng.choice(m_candidates))

            if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                continue

            pair_key = (min(j, m), max(j, m))
            if pair_key in tried_pairs:
                continue
            tried_pairs.add(pair_key)

            # Compute derivative direction
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    val += a[jk]
                if 0 <= jmk < n:
                    val += a[jmk]
                mk = m + k
                mmk = m - k
                if 0 <= mk < n:
                    val -= a[mk]
                if 0 <= mmk < n:
                    val -= a[mmk]
                d[k] = val

            # Golden section search for optimal delta
            lo_d = -a[j] * 0.99
            hi_d = a[m] * 0.99

            if hi_d - lo_d < 1e-12:
                continue

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_d
            b_g = hi_d
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)

            def f_delta(dlt):
                vals = c + dlt * d
                # Quadratic correction for k=0 (a_j^2 changes)
                # Actually the d vector already handles linear terms
                # For k=0: conv[0] = a[0]^2, change is only if j==0 or m==0
                # But d[0] already captures the linear part
                # The quadratic term: 2*dlt^2 if j==m (not possible here)
                return vals.max()

            fc = f_delta(c_g)
            fd = f_delta(d_g)

            for _ in range(25):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_delta(d_g)

            delta_opt = (a_g + b_g) / 2.0
            delta_opt = max(lo_d, min(hi_d, delta_opt))

            if abs(delta_opt) < 1e-12:
                continue

            # Apply and check
            a_trial = a.copy()
            a_trial[j] += delta_opt
            a_trial[m] -= delta_opt
            if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                continue

            c_trial = c + delta_opt * d
            trial_max = c_trial.max()

            if trial_max < best - 1e-10:
                a = a_trial
                c = c_trial
                best = trial_max
                best_a = a.copy()
                improved_this_round = True

        # Full recompute periodically
        if it % 10 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        # No improvement: inject noise
        if not improved_this_round and it > 3:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 200.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
            it -= 3  # Give more chances before breaking

    return best_a, best


def pair_transfer_cd(a_init, n, S, max_time, rng):
    """
    Faster coordinate descent using precomputed peak structure.
    Focuses on the single worst peak and finds best transfer to reduce it.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        if time.monotonic() - start > max_time - 3:
            break

        peak_k = int(np.argmax(c))

        # For this peak, find the pair (j, m) transfer that most reduces it
        # while not increasing other peaks too much
        # The change in c[peak_k] from transferring delta from m to j:
        # dc[peak_k] = delta * (a[j+peak_k] + a[j-peak_k] - a[m+peak_k] - a[m-peak_k]) (with bounds)
        
        # Compute for all j, m efficiently:
        # For peak_k, the "weight" of index i is:
        # w[i] = a[i + peak_k] + a[i - peak_k] (with bounds)
        # Then dc[peak_k] = delta * (w[j] - w[m])
        # To decrease c[peak_k], we want w[j] < w[m], i.e., transfer from high-w to low-w

        w = np.zeros(n)
        for i in range(n):
            val = 0.0
            if 0 <= i + peak_k < n:
                val += a[i + peak_k]
            if 0 <= i - peak_k < n:
                val += a[i - peak_k]
            w[i] = val

        # Best pair: j with smallest w (and a[j] > 0), m with largest w (and a[m] > 0)
        # But we also need to check other lags
        # Simplified: try top few (j, m) pairs

        # Sort by w
        w_order = np.argsort(w)
        candidates_j = w_order[:20]  # Low w: good targets
        candidates_m = w_order[::-1][:20]  # High w: good sources

        improved = False
        for j in candidates_j:
            if a[j] > 100.0 * S / n:  # Don't overload
                continue
            for m in candidates_m:
                if m == j or a[m] < 1e-12:
                    continue

                # Compute full derivative
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    val = 0.0
                    jk = j + k
                    jmk = j - k
                    if 0 <= jk < n:
                        val += a[jk]
                    if 0 <= jmk < n:
                        val += a[jmk]
                    mk = m + k
                    mmk = m - k
                    if 0 <= mk < n:
                        val -= a[mk]
                    if 0 <= mmk < n:
                        val -= a[mmk]
                    d[k] = val

                # Binary search for optimal delta
                lo_d = -a[j]
                hi_d = a[m]

                if hi_d - lo_d < 1e-12:
                    continue

                # The objective is max_k (c[k] + delta * d[k])
                # which is convex in delta (max of linear functions)
                # Use bisection on the subgradient
                # Subgradient at delta: d[k*] where k* = argmax(c + delta*d)
                
                # Golden section
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_g = lo_d
                b_g = hi_d
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = (c + c_g * d).max()
                fd = (c + d_g * d).max()

                for _ in range(20):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = (c + c_g * d).max()
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = (c + d_g * d).max()

                delta_opt = (a_g + b_g) / 2.0
                delta_opt = max(lo_d, min(hi_d, delta_opt))

                if abs(delta_opt) < 1e-12:
                    continue

                a_trial = a.copy()
                a_trial[j] += delta_opt
                a_trial[m] -= delta_opt
                if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                    continue

                c_trial = c + delta_opt * d
                trial_max = c_trial.max()

                if trial_max < best - 1e-10:
                    a = a_trial
                    c = c_trial
                    best = trial_max
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random pairs
            for _ in range(50):
                j = rng.integers(0, n)
                m = rng.integers(0, n)
                if j == m or a[j] < 1e-12 or a[m] < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    val = 0.0
                    if 0 <= j + k < n:
                        val += a[j + k]
                    if 0 <= j - k < n:
                        val += a[j - k]
                    if 0 <= m + k < n:
                        val -= a[m + k]
                    if 0 <= m - k < n:
                        val -= a[m - k]
                    d[k] = val

                lo_d = -a[j] * 0.9
                hi_d = a[m] * 0.9
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_g = lo_d
                b_g = hi_d
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = (c + c_g * d).max()
                fd = (c + d_g * d).max()

                for _ in range(15):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = (c + c_g * d).max()
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = (c + d_g * d).max()

                delta_opt = (a_g + b_g) / 2.0
                delta_opt = max(lo_d, min(hi_d, delta_opt))

                if abs(delta_opt) < 1e-12:
                    continue

                a_trial = a.copy()
                a_trial[j] += delta_opt
                a_trial[m] -= delta_opt
                if a_trial[j] < -1e-10 or a_trial[m] < -1e-10:
                    continue

                c_trial = c + delta_opt * d
                trial_max = c_trial.max()

                if trial_max < best - 1e-10:
                    a = a_trial
                    c = c_trial
                    best = trial_max
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 5:
            # Noise to escape
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-it / 100.0)
            if noise_scale < 1e-9:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best


def multi_scale_optimize(a_init, n, S, max_time, rng):
    """
    Multi-scale optimization: downsample, optimize, upsample, refine.
    """
    start = time.monotonic()
    best = np.inf
    best_a = a_init.copy()

    # Try multiple scales
    scales = []
    ns = n
    while ns >= 256:
        scales.append(ns)
        ns //= 2
    scales.append(n)  # Original scale at the end

    time_per_scale = max_time / len(scales)

    for scale_n in scales:
        if time.monotonic() - start > max_time - 10:
            break

        # Downsample or upsample to target scale
        if scale_n < n:
            # Block average
            n_blocks = n // scale_n
            a_scaled = a_init[:n_blocks * scale_n].reshape(scale_n, n_blocks).mean(axis=1)
        elif scale_n > n:
            # Interpolate
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, scale_n)
            a_scaled = np.interp(t_new, t_old, a_init)
        else:
            a_scaled = a_init.copy()

        S_scaled = np.sqrt(2.0 * scale_n)
        a_scaled = normalize_sum(a_scaled, scale_n)

        # Optimize at this scale
        t_alloc = min(time_per_scale, max_time - (time.monotonic() - start) - 10)
        if t_alloc < 5:
            continue

        # Quick FISTA
        a_opt, score = run_fista_improved(a_scaled, scale_n, S_scaled, t_alloc, rng,
                                           beta_init=20.0, beta_max=2000.0, beta_ramp=500)
        
        # Upsample back to original n
        if scale_n < n:
            t_old = np.linspace(0, 1, scale_n)
            t_new = np.linspace(0, 1, n)
            a_up = np.interp(t_new, t_old, a_opt)
            a_up = normalize_sum(a_up, n)
            c_chk = fft_convolve_self(a_up)
            if c_chk.max() < best:
                best = c_chk.max()
                best_a = a_up.copy()
                a_init = a_up.copy()  # Update for next scale

    return best_a, best


def random_phase_starts(n, S, rng, n_starts=10):
    """Generate diverse starting points from random phases."""
    starts = []
    L = next_fast_len(2 * n - 1)

    for i in range(n_starts):
        # Random phases
        if i == 0:
            # Almost flat: small perturbation from uniform
            phases = np.exp(1j * 0.01 * rng.normal(0, 1, L))
        elif i == 1:
            # Linear phase (shifted uniform)
            phases = np.exp(1j * np.linspace(0, 2 * np.pi, L))
        elif i == 2:
            # Quadratic phase
            t = np.arange(L) / L
            phases = np.exp(1j * 2 * np.pi * 2 * t * t)
        else:
            # Random phases
            phases = np.exp(1j * rng.uniform(0, 2 * np.pi, L))

        # DC component
        mag = np.ones(L)
        F = mag * phases
        F[0] = np.abs(F[0])  # Keep DC positive

        a = np.real(ifft(F))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        else:
            a = np.ones(n) * S / n
        starts.append(a)

    return starts


def propose_candidate(seed=42, budget_s=2100):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Multi-phase approach:
    1. Spectral GS from parent and random phase starts
    2. Peak-directed coordinate descent
    3. FISTA polish
    4. Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 128 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            parent = np.clip(parent, 0, 1000)
        else:
            parent = None
    except Exception:
        parent = None

    # Determine length
    if parent is not None and 1024 <= len(parent) <= 65536:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print("Starting: n=%d, budget=%.0fs, parent_len=%s" % (
        n, budget_s, len(parent) if parent is not None else "None"))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label):
        nonlocal global_best, global_a
        a_norm = normalize_sum(a_candidate, len(a_candidate))
        c_chk = fft_convolve_self(a_norm)
        score = c_chk.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print("  [*] %s: score=%.6f (elapsed %.0fs, score_pct=%.2f%%)" % (
                label, score, elapsed, (score - 1.0) / (2.0 - 1.0) * 100))
        return score

    # Initialize from parent
    if parent is not None:
        a_start = normalize_sum(parent, n)
        _, init_score = score_normalized(a_start)
        print("  Parent initial score: %.6f" % init_score)
        update_best(a_start, "parent_init")
    else:
        a_start = np.ones(n) * S / n
        update_best(a_start, "uniform_init")

    # Phase 1: Spectral GS (35% of time)
    print("\n=== Phase 1: Spectral GS ===")
    p1_budget = (deadline - time.monotonic()) * 0.35
    if p1_budget > 20:
        # From parent
        t_sf = p1_budget * 0.5
        a_sf1, _ = spectral_gs(a_start, n, S, t_sf, rng, n_restarts=3)
        update_best(a_sf1, "spec_gs_v1")

        # Alternative spectral method
        t_sf2 = p1_budget * 0.3
        remaining_p1 = p1_budget - t_sf - t_sf2 - 5
        if remaining_p1 > 10:
            a_sf2, _ = spectral_gs_v2(a_start, n, S, remaining_p1, rng)
            update_best(a_sf2, "spec_gs_v2")

        # From random phase starts (only if time permits)
        remaining_p1 = deadline - time.monotonic()
        if remaining_p1 > 30 and global_a is not None:
            starts = random_phase_starts(n, S, rng, n_starts=5)
            t_per = min(remaining_p1 / len(starts), p1_budget / 10)
            for i, s in enumerate(starts):
                if time.monotonic() > deadline - (deadline - t_start) * 0.5:
                    break
                if t_per < 5:
                    break
                a_rs, _ = spectral_gs(s, n, S, t_per, rng, n_restarts=1)
                update_best(a_rs, "rand_phase_%d" % i)

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: Peak-directed coordinate descent (35% of time)
    print("\n=== Phase 2: Peak-Directed CD ===")
    if global_a is not None:
        p2_budget = (deadline - time.monotonic()) * 0.35
        if p2_budget > 20:
            n_cd = len(global_a)
            S_cd = np.sqrt(2.0 * n_cd)
            
            # Main CD run
            t_cd1 = p2_budget * 0.6
            a_cd1, _ = peak_directed_cd(global_a, n_cd, S_cd, t_cd1, rng, top_k=5)
            update_best(a_cd1, "peak_cd_1")

            # Faster CD variant
            remaining_p2 = p2_budget - t_cd1 - 5
            if remaining_p2 > 10:
                a_cd2, _ = pair_transfer_cd(global_a, n_cd, S_cd, remaining_p2, rng)
                update_best(a_cd2, "pair_cd")

    print("After Phase 2: best=%.6f" % global_best)

    # Phase 3: FISTA polish (20% of time)
    print("\n=== Phase 3: FISTA Polish ===")
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        p3_budget = (deadline - time.monotonic()) * 0.20
        if p3_budget > 15:
            a_f, _ = run_fista_improved(global_a, n_f, S_f, p3_budget, rng,
                                         beta_init=100.0, beta_max=8000.0,
                                         beta_ramp=3000)
            update_best(a_f, "fista_polish")

    print("After Phase 3: best=%.6f" % global_best)

    # Phase 4: Final alternating polish (10% of time)
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Alternate between CD and small FISTA
            n_rounds = max(2, int(remaining / 30))
            t_per_round = remaining / n_rounds
            
            for r in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                
                if r % 2 == 0:
                    a_r, _ = pair_transfer_cd(global_a, n_p, S_p, t_per_round * 0.8, rng)
                    update_best(a_r, "final_cd_%d" % r)
                else:
                    a_r, _ = run_fista_improved(global_a, n_p, S_p, t_per_round * 0.8, rng,
                                                 beta_init=200.0, beta_max=10000.0,
                                                 beta_ramp=500)
                    update_best(a_r, "final_fista_%d" % r)

    # Phase 5: Try different length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print("\n=== Phase 5: Alternate Length ===")
        for alt_n in [n * 2, n // 2]:
            if time.monotonic() > deadline - 30:
                break
            if not (1024 <= alt_n <= 32768):
                continue
            
            S_alt = np.sqrt(2.0 * alt_n)
            # Interpolate
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, alt_n)
            alt_start = np.interp(t_new, t_old, global_a)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alt = min(remaining * 0.4, 60)
            a_alt, score_alt = run_fista_improved(alt_start, alt_n, S_alt, t_alt, rng,
                                                   beta_init=20.0, beta_max=3000.0,
                                                   beta_ramp=1000)
            update_best(a_alt, "alt_n_%d" % alt_n)

            # Quick CD on the result
            remaining_alt = deadline - time.monotonic()
            if remaining_alt > 15:
                a_alt2, _ = pair_transfer_cd(a_alt, alt_n, S_alt, remaining_alt * 0.5, rng)
                update_best(a_alt2, "alt_n_cd_%d" % alt_n)

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    
    if np.isfinite(ver):
        print("Verification score: %.6f" % ver)
    else:
        print("Verification: inf, falling back")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(global_a), ver if np.isfinite(ver) else np.inf, time.monotonic() - t_start))
    return final_list
