"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining frequency-domain initialization,
LSE-smoothed subgradient with adaptive step size, Barzilai-Borwein step,
and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_gradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k

def normalize(a, n, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def gen_gauss_init(n, sigma, S):
    t = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-t**2 / (2.0 * sigma**2))
    return np.clip(a, 0, None) * (S / a.sum())

def gen_cos_init(n, S):
    t = np.arange(n) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    return a * (S / a.sum())

def gen_cos2_init(n, S):
    t = np.arange(n) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * t))**2
    return a * (S / a.sum())

def gen_blackman_init(n, S):
    t = np.arange(n) / (n - 1)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

def gen_trapezoid_init(n, alpha, S):
    t = np.arange(n) / n
    a = np.ones(n)
    ramp = int(n * alpha / 2)
    if ramp > 0:
        a[:ramp] = np.linspace(0.1, 1, ramp)
        a[-ramp:] = np.linspace(1, 0.1, ramp)
    return a * (S / a.sum())

def gen_flat_with_edges(n, edge_frac, S):
    t = np.arange(n) / n
    a = np.ones(n)
    m = int(n * edge_frac)
    decay = np.linspace(1, 0.3, m)
    a[:m] *= decay
    a[-m:] *= decay[::-1]
    return a * (S / a.sum())

def run_lse_subgrad(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=200.0, method='lse'):
    a = normalize(a_init, n, S)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    step = 0.5
    consec_fail = 0
    c = fft_convolve_self(a)
    
    # Track previous step for BB step
    prev_a = a.copy()
    prev_grad = None
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            consec_fail = 0
        
        # Compute gradient
        if method == 'lse':
            beta = beta_start + (beta_end - beta_start) * min(1.0, it / 5000.0)
            grad = lse_gradient(a, c, beta)
        elif method == 'topk':
            k = min(32, 2 + (it // 100))
            grad = topk_gradient(a, c, k)
        else:  # hybrid
            if it % 100 < 20:
                grad = topk_gradient(a, c, 1)
            elif it % 100 < 60:
                grad = topk_gradient(a, c, 8)
            else:
                beta = min(150.0, 5.0 + 0.03 * it)
                grad = lse_gradient(a, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size
        # Barzilai-Borwein
        if prev_grad is not None and it > 2:
            s_prev = a - prev_a
            y_prev = grad - prev_grad
            sTs = np.dot(s_prev, s_prev)
            sy = np.dot(s_prev, y_prev)
            if sTs > 1e-30 and abs(sy) > 1e-30:
                bb_step = sTs / abs(sy)
                step = 0.5 * (step + bb_step)
        
        max_step = 4.0 * S / (n * gnorm + 1e-15)
        step = min(step, max_step)
        step = max(step, 1e-10)
        
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)
        
        if obj_new < cur_obj:
            prev_a = a.copy()
            prev_grad = grad.copy()
            a = a_new
            c = c_new
            step = min(step * 1.1, max_step)
        else:
            consec_fail += 1
            step *= 0.6
            if consec_fail > 30:
                # Perturb and restart
                noise = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                step = 0.5
                consec_fail = 0
                prev_grad = None
            if consec_fail > 80:
                noise = 0.02 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                c = fft_convolve_self(a)
                step = 1.0
                consec_fail = 0
                prev_grad = None
    
    return best_a, best_obj

def coord_descent(a, n, S, max_time, rng):
    """Coordinate descent: try adjusting pairs of elements to reduce max(c)."""
    a = normalize(a.copy(), n, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Find the peak
        k_star = int(np.argmax(c))
        
        # Gradient at peak: grad[i] = 2*a[k_star - i] for valid i
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        
        # Try reducing the contribution at the peak by moving mass
        # from a[i] to a[j] where a[k_star-j] is small
        # Simple approach: try a random coordinate pair
        
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving a small amount from i to j
        for eps in [0.001, 0.0005, 0.0002]:
            a_try = a.copy()
            move = eps * a[i]
            if a[i] - move < 0:
                continue
            a_try[i] -= move
            a_try[j] += move
            a_try = normalize(a_try, n, S)
            
            # Quick check: only compute the changed part of c
            # c changed at positions affected by i and j
            # For efficiency, just compute full c for now
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            
            if obj_try < best_obj - 1e-12:
                a = a_try
                c = c_try
                best_obj = obj_try
                best_a = a.copy()
                break
    
    return best_a, best_obj

def spectral_init(n, S, spectrum_type, rng):
    """Initialize by designing the spectrum and taking IFFT."""
    N = 1 << (2 * n - 2).bit_length()
    k = np.arange(N)
    
    if spectrum_type == 'gauss':
        sigma = rng.uniform(5, 30)
        A = np.exp(-k**2 / (2 * sigma**2))
    elif spectrum_type == 'lorentz':
        gamma = rng.uniform(3, 15)
        A = 1.0 / (1.0 + (k / gamma)**2)
    elif spectrum_type == 'power':
        p = rng.uniform(1.5, 4.0)
        A = 1.0 / (1.0 + np.abs(k))**p
    else:
        A = np.ones(N)
    
    # Make it conjugate symmetric for real IFFT
    # A[k] should satisfy A[N-k] = conj(A[k])
    # For real A, just make it symmetric
    A = (A + A[::-1].copy()) / 2.0
    A[0] = A[0]  # DC term
    if N % 2 == 0:
        A[N//2] = A[N//2]  # Nyquist
    
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
