```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated projection
with active-set subgradients, spectral rebalancing, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Full self-convolution via FFT. N is FFT size >= 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def nesterov_block(a_init, n, S, N, max_time, rng, 
                    top_k_init=50, top_k_max=512,
                    step_scale=1.0, lb=1.0):
    """
    Nesterov-accelerated projected subgradient method.
    Uses top-k active constraints for subgradient.
    lb is the known lower bound on max_conv (for Polyak step).
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = np.inf
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 10)
        
        # Evaluate at y (Nesterov extrapolation point)
        c = compute_conv(y, N)
        cur_val = c.max()
        
        if cur_val < best_val:
            best_val = cur_val
            best_a = y.copy()
        
        # Also check a (might be better than y)
        if it % 5 == 0:
            c_a = compute_conv(a, N)
            if c_a.max() < best_val:
                best_val = c_a.max()
                best_a = a.copy()
        
        # Compute subgradient using top-k
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Weight by excess over lower bound
        excess = np.maximum(top_vals - lb, 0)
        if excess.sum() < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = excess / excess.sum()
        
        # Subgradient: sum_k w_k * d(conv_k)/d(a)
        # d(conv(a,a)[k])/d(a_i) = a_{i+k} + a_{i-k} (with boundary handling)
        # More precisely: conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/d(a_i) = a[k-i] + a[k-i] if k-i != i, else... 
        # Actually: d/d(a_i) sum_j a[j]*a[k-j] = a[k-i] + a[i+k-n+1]... 
        # Let me think more carefully.
        # conv(a,a)[k] = sum_j a[j] * a[k-j] where a is defined on [0, n-1]
        # d/d(a_i): the term a[i] appears as a[j] when j=i, giving a[k-i]
        #           and as a[k-j] when k-j=i, i.e., j=k-i, giving a[i]
        #           Wait, a[k-j] with k-j = i means j = k-i, and the term is a[k-i]*a[i]
        #           So d/d(a_i) = a[k-i] + a[i] (if both indices are in range)
        #           Actually it's: a[k-i] (from the j=i term) + a[i] * (coefficient of a[i] in the second position)
        #           Let me just use: d conv_k / d a_i = a[(k-i) mod stuff] + a[(i+k - (n-1)) mod stuff]
        
        # Simpler: use FFT-based gradient computation
        # The gradient of conv(a,a) w.r.t. a is: 2 * ifft(fft(w) * conj(fft(a)))
        # where w is the weight vector on the convolution lags
        
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        fsize = N  # Use same FFT size
        W = fft(w_full, fsize)
        Af = fft(a, fsize)  # Use a (not y) for the gradient point
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        # Nesterov step
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-18:
            break
        
        # Polyak-like step size: use gap between current and lower bound
        gap = cur_val - lb
        if gap > 1e-10:
            step = step_scale * S * gap / (norm_g * norm_g)
            step = np.clip(step, 0.1 * S / np.sqrt(n), 10.0)
        else:
            step = step_scale * 2.0 * S / (n * (1.0 + 0.01 * it))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic spectral rebalancing
        if it % 500 == 0 and rng.random() < 0.3:
            c_sp = compute_conv(a, N)
            top_sp = np.argsort(c_sp)[::-1][:10]
            m_arr = np.arange(N)
            cos_eff = np.zeros(N)
            for k in top_sp:
                cos_eff += np.cos(2.0 * np.pi * m_arr * int(k) / N)
            cos_eff /= len(top_sp)
            A_full = fft(a, N)
            P = np.abs(A_full) ** 2
            P_new = np.maximum(P * (1.0 - 0.1 * cos_eff), 0.0)
            phase = np.angle(A_full)
            A_new = np.sqrt(P_new) * np.exp(1j * phase)
            a_new = np.real(ifft(A_new))[:n]
            a = project_simplex(a_new, S)
            y = a.copy()
            t_mom = 1.0
    
    return best_a, best_val


def coordinate_descent_block(a_init, n, S, N, max_time, rng):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, given other coordinates fixed, the convolution values
    that depend on a_i are:
    conv[k] = fixed_part[k] + a_i * (a[k-i] + a[i+k-(n-1)])
    (approximately, for the relevant k values)
    
    To minimize max(conv), we want to choose a_i to minimize the maximum
    of these linear functions of a_i, subject to a_i >= 0 and sum constraint.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)
    best_a = a.copy()
    best_val = compute_conv(a, N).max()
    start_time = time.monotonic()
    
    # Precompute: for each i, which convolution positions are affected
    # conv[k] = sum_j a[j] * a[k-j]
    # If we change a[i], then conv[k] changes by: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] (if k != 2i)
    # Actually: conv[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
    #   j=i: a[i]*a[k-i]
    #   k-j=i => j=k-i: a[k-i]*a[i]
    # So d(conv[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] (if k-i != i)
    # or a[k-i] + a[i] = 2*a[i] (if k-i = i, i.e., k = 2i)
    # Wait, that's the derivative. The actual change when a[i] changes by delta is:
    # delta * a[k-i] + a[k-i] * delta = 2*delta*a[k-i] (for k-i != i)
    # or delta * a[i] + a[i] * delta... hmm this is getting confusing.
    
    # Let me just do: for each i, compute the "partial convolution" without a[i],
    # then the full convolution is partial + a[i] * (contribution from a[i])
    
    # Actually, the simplest approach: 
    # conv(a)[k] = sum_j a[j] * a[k-j]
    # = a[i]*a[k-i] + sum_{j != i} a[j]*a[k-j]
    # = a[i]*a[k-i] + (conv_without_i)[k]
    # where conv_without_i is the convolution with a[i] set to 0.
    # But this is expensive to compute for each i.
    
    # Alternative: use the fact that changing a[i] by delta changes conv[k] by:
    # delta * a[k-i] + a[k-i] * delta (if k-i is in range and k-i != i)
    # or delta * a[k-i] + a[i+delta] * ... 
    # This is getting too complicated. Let me use a simpler approach:
    
    # For each coordinate i, the effect on conv is:
    # conv_new[k] = conv_old[k] + delta * (a[k-i] + a[i+k])  [with appropriate indexing]
    # where delta = a_new[i] - a_old[i]
    
    # To minimize max(conv_new), we want to find a[i] that minimizes
    # max_k (conv_old[k] + (a[i]-0) * contrib[i,k])
    # where contrib[i,k] = a[k-(n-1-i)] (approximately)
    
    # This is a 1D minimax problem. The optimal a[i] is where the
    # "ramp" from the affected conv positions is balanced.
    
    # Simplified approach: just try a few values of a[i] and pick the best.
    
    n_passes = 0
    while time.monotonic() - start_time < max_time:
        n_passes += 1
        improved = False
        
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start_time > max_time - 1:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Compute the "linear part" of conv as a function of a[i]
            # conv[k] = base[k] + a[i] * lin[k]
            # where lin[k] = a[k - (n-1-i)] (the other factor in the product)
            
            # The affected k values are those where both i and k-j are in [0, n-1] and j != i... 
            # Actually, let me just compute the gradient and do a small line search.
            
            # Cheap approach: compute conv, then for coordinate i, 
            # the derivative of conv[k] w.r.t. a[i] is a[k-i] (with range check)
            # Plus the "reflection" term.
            
            # Let me use the FFT-based approach: compute the full conv, 
            # then for the specific coordinate, do a binary search on the optimal value.
            
            # Even simpler: try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5, S / n * 2]
            
            best_candidate_val = np.inf
            best_candidate = old_val
            
            for cand in candidates:
                if cand < 0 or cand > 1000:
                    continue
                a[i] = cand
                # Renormalize (approximately - just adjust one other element)
                # Actually, for coordinate descent, we should keep sum fixed.
                # But this is complicated. Let's just clip and project.
                a[i] = cand
                a[i+1 if i+1 < n else 0] = max(0, a[i+1 if i+1 < n else 0] - (cand - old_val))
                
                c = compute_conv(a, N)
                val = c.max()
                if val < best_candidate_val:
                    best_candidate_val = val
                    best_candidate = cand
            
            a[i] = best_candidate
            if best_candidate_val < best_val - 1e-12:
                best_val = best_candidate_val
                best_a = a.copy()
                improved = True
        
        if not improved and n_passes > 3:
            break
    
    return best_a, best_val
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Nesterov-accelerated projected subgradient with large adaptive active set
2. Fourier-mode parameterized optimization for global structure search
3. Dual-gap adaptive step sizing (Polyak-style)
4. Final fine-grained polishing with coordinate-aware updates
Key difference from previous: uses much larger active sets for subgradient accuracy,
Fourier-mode reparameterization for global moves, and dual-based step control.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Full self-convolution via FFT. N is FFT size >= 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def subgrad_topk(a, N, n, top_idx, weights):
    """Compute weighted subgradient of max conv using given active set."""
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    W = fft(w_full, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def nesterov_large_active(a_init, n, S, N, max_time, rng,
                          top_k_init=100, top_k_max=2048,
                          lb=1.0, step_mult=1.0):
    """Nesterov with large active set and Polyak-style step size."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    patience = 500

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it * 2)

        c = compute_conv(y, N)
        cur_val = c.max()

        if cur_val < best_val:
            best_val = cur_val
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        if it % 5 == 0:
            c_a = compute_conv(a, N)
            if c_a.max() < best_val:
                best_val = c_a.max()
                best_a = a.copy()
                stall_count = 0

        if stall_count > patience:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            y = a.copy()
            t_mom = 1.0
            stall_count = 0
            continue

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        excess = np.maximum(top_vals - lb, 1e-10)
        weights = excess / excess.sum()

        grad = subgrad_topk(a, N, n, top_idx, weights)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-18:
            break

        gap = max(cur_val - lb, 1e-10)
        step = step_mult * S * gap / (norm_g * norm_g + 1e-30)
        step = np.clip(step, 0.05 * S / np.sqrt(n), 5.0 * S / n)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 1000 == 0 and rng.random() < 0.2:
            A_full = fft(a, N)
            P = np.abs(A_full) ** 2
            P_smooth = np.convolve(P, np.ones(20) / 20, mode='same')
            P_new = 0.7 * P + 0.3 * P_smooth
            phase = np.angle(A_full)
            A_new = np.sqrt(P_new) * np.exp(1j * phase)
            a_new = np.real(ifft(A_new))[:n]
            a = project_simplex(a_new, S)
            y = a.copy()
            t_mom = 1.0

    return best_a, best_val


def fourier_mode_optimize(a_init, n, S, N, max_time, rng, n_modes=12):
    """
    Optimize using Fourier mode parameterization.
    a_i = c + sum_m A_m * cos(2*pi*f_m*i/n + phi_m)
    Optimize (A_m, phi_m) to minimize max_conv subject to a_i >= 0.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)
    c_mean = S / n

    best_a = a.copy()
    best_val = compute_conv(a, N).max()
    start_time = time.monotonic()

    # Pick frequencies that are "useful" (not DC, not too high)
    max_freq = min(n_modes + 5, n // 4)
    freqs = np.arange(1, max_freq + 1, dtype=np.float64)
    # Use a subset
    if len(freqs) > n_modes:
        freqs = freqs[rng.choice(len(freqs), n_modes, replace=False)]
    freqs = np.sort(freqs)

    tt = np.arange(n, dtype=np.float64)
    basis = np.zeros((n, n_modes))
    for m, f in enumerate(freqs):
        basis[:, m] = np.cos(2.0 * np.pi * f * tt / n)

    # Initial amplitudes from current a
    resid = a - c_mean
    amps = np.linalg.lstsq(basis, resid, rcond=None)[0]
    phases = np.zeros(n_modes)

    # Simple coordinate descent on (amps, phases)
    for it in range(2000):
        if time.monotonic() - start_time > max_time - 1:
            break

        # Update amplitudes (keep phases fixed)
        for m in range(n_modes):
            best_amp = amps[m]
            best_v = np.inf
            base = a - amps[m] * basis[:, m]
            for scale in [0.0, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                trial_amp = best_amp * scale if m > 0 else amps[m] * scale
                a_trial = base + trial_amp * basis[:, m]
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial * (S / s)
                c_val = compute_conv(a_trial, N).max()
                if c_val < best_v:
                    best_v = c_val
                    best_amp = trial_amp
            amps[m] = best_amp

        # Reconstruct
        a = c_mean + basis @ amps
        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)

        # Update phases
        for m in range(n_modes):
            best_phi = phases[m]
            best_v = np.inf
            base = a - amps[m] * basis[:, m]
            for dphi in [0.0, 0.1, -0.1, 0.2, -0.2, 0.5, -0.5, 1.0, -1.0]:
                trial_phi = phases[m] + dphi
                col_trial = np.cos(2.0 * np.pi * freqs[m] * tt / n + trial_phi)
                a_trial = base + amps[m] * col_trial
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial * (S / s)
                c_val = compute_conv(a_trial, N).max()
                if c_val < best_v:
                    best_v = c_val
                    best_phi = trial_phi
            phases[m] = best_phi
            basis[:, m] = np.cos(2.0 * np.pi * freqs[m] * tt / n + phases[m])

        a = c_mean + basis @ amps
        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)
        cur_val = compute_conv(a, N).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


def fine_polish(a_init, n, S, N, max_time, rng, top_k=50):
    """Very fine Nesterov with small top_k for precision."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_val = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0

    # Use measured lower bound: average of conv values
    c_init = compute_conv(a, N)
    lb_measured = c_init.mean()  # This is S^2/(2n-1), a valid lower bound

    while time.monotonic() - start_time < max_time:
        it += 1

        c = compute_conv(y, N)
        cur_val = c.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = y.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        lb_use = max(lb_measured, top_vals.min() * 0.99)
        excess = np.maximum(top_vals - lb_use, 1e-12)
        weights = excess / excess.sum()

        grad = subgrad_topk(a, N, n, top_idx, weights)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-18:
            break

        gap = max(cur_val - lb_use, 1e-12)
        step = S * gap / (norm_g * norm_g + 1e-30)
        step = np.clip(step, 0.01 * S / np.sqrt(n), 1.0 * S / n)
        # Decay step over time
        step *= max(0.1, 1.0 - 0.3 * it / max(it, 50000))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            y = a.copy()
            t_mom = 1.0

    return best_a, best_val


def active_set_lp_direction(a, c, t_cur, n, N, top_idx):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= t_cur for k in top_idx, b >= 0.
    Use scipy linprog. Returns direction b / sum(b) - a/sum(a).
    """
    from scipy.optimize import linprog

    # Constraints: for each k in top_idx, sum_j b_j * a[k-j] <= t_cur
    # where a[k-j] means the convolution kernel
    # conv(a,b)[k] = sum_j a[j]*b[k-j] = sum_j b[j]*a[k-j]
    # So constraint k: sum_j b[j] * a[k-j] <= t_cur
    
    n_constraints = len(top_idx)
    # A_ub[i, j] = a[k_i - j] (with a[m] = 0 for m outside [0, n-1])
    A_ub = np.zeros((n_constraints, n))
    for i, k in enumerate(top_idx):
        k_int = int(k)
        for j in range(n):
            idx = k_int - j
            if 0 <= idx < n:
                A_ub[i, j] = a[idx]
    
    c_obj = -np.ones(n)  # maximize sum(b) = minimize -sum(b)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(n_constraints, t_cur),
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    b = res.x
    s = b.sum()
    if s < 1e-15:
        return None
    
    return b / s
```
