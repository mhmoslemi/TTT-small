"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using a multi-phase
combining spectral flattening (Gerchberg-Saxton style), adaptive FISTA with
LSE smoothing, LP-inspired subgradient polishing, and multi-scale exploration.
Key innovations: frequency-domain convolution flattening, adaptive top-k
subgradient with momentum restart, and periodic cross-scale transfers.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE-smoothed gradient of max(c) w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm) - beta * (c.max() - cm)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - vals[-1], 0.0) + 1e-30, power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c):
    """Exact subgradient at the argmax."""
    n = len(y)
    idx = int(np.argmax(c))
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_flatten(f, max_time, rng, S=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively modifies the frequency content to flatten the convolution.
    """
    n = len(f)
    if S is None:
        S = np.sum(f)
    N = 1 << (2 * n - 1).bit_length()

    f = np.clip(f.copy(), 0, None)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f * (S / s)

    start = time.monotonic()
    best_f = f.copy()
    best_obj = compute_obj(f)
    alpha = 0.2
    no_improve = 0

    while time.monotonic() - start < max_time:
        F = fft(f, N)
        C = F * F
        c = np.real(ifft(C))[:2 * n - 1]

        # Build target: partial flattening
        c_mean = c.mean()
        c_target = (1.0 - alpha) * c + alpha * c_mean
        c_target = np.maximum(c_target, 0.0)

        # Frequency-domain update: match magnitude to target, keep phase
        C_target = fft(c_target, N)
        abs_F = np.abs(F)
        abs_Ct = np.abs(C_target)
        # Avoid division by zero
        mask = abs_F > 1e-30
        phase_F = np.zeros_like(F)
        phase_F[mask] = F[mask] / abs_F[mask]

        # Blend: partially update magnitude
        blend = 0.5  # fraction of the way to target magnitude
        new_mag = abs_F * (1 - blend) + np.sqrt(np.maximum(abs_Ct, 0.0)) * blend
        F_new = new_mag * phase_F

        f_new = np.real(ifft(F_new))[:n]
        f_new = project_simplex(f_new, S)

        obj_new = compute_obj(f_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            alpha = min(0.85, alpha * 1.02)
        else:
            no_improve += 1
            alpha = max(0.05, alpha * 0.97)
            if no_improve > 200:
                # Perturb and reset
                noise_scale = 0.02 * S / np.sqrt(n)
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)
                alpha = 0.15
                no_improve = 0

    return best_f, best_obj


def run_fista_adaptive(a_init, max_time, rng, mode="auto"):
    """
    FISTA with adaptive strategy:
    - Starts with LSE smoothing at moderate beta
    - Transitions to top-k when LSE gets stuck
    - Uses periodic restarts and noise injection
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * (S / a.sum())

    y = a.copy()
    x_prev = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    no_improve = 0
    beta = 20.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Choose gradient based on phase
        if no_improve < 500:
            # LSE phase
            grad = lse_gradient(y, c, beta)
            beta = min(5000.0, beta * 1.003)
            step = 0.8 / max(beta * 2.0, 1.0)
        elif no_improve < 2000:
            # Transition: mix LSE and top-k
            k = max(4, n // 64)
            g1 = lse_gradient(y, c, max(beta, 100.0))
            g2 = topk_gradient(y, c, k, power=1.5)
            grad = 0.5 * g1 + 0.5 * g2
            step = 0.6 / max(np.linalg.norm(grad) * 0.5, 1e-10)
        else:
            # Pure top-k / exact subgradient
            if no_improve < 5000:
                k = min(64, max(4, no_improve // 200))
                grad = topk_gradient(y, c, k, power=1.0)
            else:
                k = min(256, max(8, no_improve // 100))
                grad = topk_gradient(y, c, k, power=0.5)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (gnorm * (1.0 + 0.01 * no_improve))
            step = min(step, 2.0 * S / (0.5 * gnorm + 1e-10))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # FISTA update
        x_new = project_simplex(y - step * grad, S)

        # Momentum restart when stuck
        if no_improve > 3000 and no_improve % 500 == 0:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-no_improve / 8000.0))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                x_prev = y.copy()
                no_improve = 0
                continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        x_prev = x_new
        t_mom = t_new

        # Periodic full restart
        if it % 10000 == 0:
            t_mom = 1.0
            x_prev = y.copy()

    return best_a, best_obj


def run_lp_direction(a_init, max_time, rng):
    """
    LP-based direction finding.
    Solves: max sum(b) s.t. f*b <= M, b >= 0
    Uses the result as a direction to move.
    For large n, uses a downsampled version.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    n_iter = 0

    # Use a smaller size for LP if n is large
    n_lp = min(n, 4096)
    if n_lp < n:
        # Downsample
        ratio = n / n_lp
        idx = (np.arange(n_lp) * ratio).astype(int)
        idx = np.minimum(idx, n - 1)
        f_lp = a[idx]
    else:
        f_lp = a.copy()

    m_lp = len(f_lp)
    S_lp = np.sqrt(2.0 * m_lp)
    f_lp = f_lp * (S_lp / f_lp.sum())

    while time.monotonic() - start < max_time:
        n_iter += 1
        c_lp = fft_convolve_self(f_lp)
        M = c_lp.max() * (1.0 + 0.01)  # Slightly relaxed for feasibility

        # Build constraint matrix A for f*b <= M
        # A[k, j] = f_lp[k-j] if 0 <= k-j < m_lp, else 0
        # Size: (2*m_lp - 1) x m_lp
        n_cons = 2 * m_lp - 1
        # Use sparse representation
        rows = []
        cols = []
        data = []
        for k in range(n_cons):
            j_start = max(0, k - m_lp + 1)
            j_end = min(m_lp - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                data.append(f_lp[k - j])

        A_sparse = csr_matrix((data, (rows, cols)), shape=(n_cons, m_lp))

        # Objective: maximize sum(b) -> minimize -sum(b)
        c_obj = -np.ones(m_lp)

        # Constraints: A @ b <= M, b >= 0
        # Reduce constraints: only keep those where M is tight
        # (skip very loose constraints to speed up LP)
        # For now, use all constraints

        try:
            result = linprog(c_obj, A_ub=A_sparse, b_ub=np.full(n_cons, M),
                             bounds=[(0, None)] * m_lp,
                             method='highs',
                             options={'time_limit': 5, 'presolve': True})
            if result.success and result.x is not None:
                b = result.x
                b_sum = b.sum()
                if b_sum < 1e-10:
                    break
                # Normalize b to sum S_lp
                b_norm = b * (S_lp / b_sum)
                # Move a small step toward b
                t_step = min(0.1, 0.5 / (1.0 + 0.01 * n_iter))
                f_lp_new = (1.0 - t_step) * f_lp + t_step * b_norm
                f_lp_new = np.maximum(f_lp_new, 0.0)
                f_lp_new = f_lp_new * (S_lp / f_lp_new.sum())

                # Check if improved (at full size if downsampled)
                if n_lp < n:
                    # Check the full-size objective by interpolating
                    # (approximation: check at LP size)
                    obj_lp = 2.0 * m_lp * fft_convolve_self(f_lp_new).max() / (S_lp ** 2)
                    if obj_lp < 2.0 * m_lp * c_lp.max() / (S_lp ** 2) * 0.999:
                        f_lp = f_lp_new
                        # Upsample to full size
                        ratio_full = n / m_lp
                        idx_full = (np.arange(n) * ratio_full).astype(int)
                        idx_full = np.minimum(idx_full, m_lp - 1)
                        a_new = f_lp[idx_full]
                        a_new = a_new * (S / a_new.sum())
                        obj_full = compute_obj(a_new)
                        if obj_full < best_obj:
                            best_obj = obj_full
                            best_a = a_new.copy()
                else:
                    obj_lp = compute_obj(f_lp_new)
                    if obj_lp < best_obj:
                        best_obj = obj_lp
                        best_a = f_lp_new.copy()
                        f_lp = f_lp_new

        except Exception as e:
            print("  LP error: %s" % str(e)[:80])
            break

        if n_iter > 200:
            break

    print("  LP: %d iterations, best_obj=%.6f" % (n_iter, best_obj))
    return best_a, best_obj


def upsample_interp(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample_avg(a, factor=2):
    """Downsample by averaging."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: spectral flattening, adaptive FISTA, LP directions,
    multi-scale exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # === Get initial sequence ===
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n0 = 8192
        tt = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    def update_global(a_candidate):
        nonlocal global_best_obj, global_best_a
        if a_candidate is None:
            return
        obj_c = compute_obj(a_candidate)
        if obj_c < global_best_obj:
            global_best_obj = obj_c
            global_best_a = a_candidate.copy()
            print("  NEW BEST: obj=%.6f (n=%d, t=%.0fs)" % (
                obj_c, len(a_candidate), time.monotonic() - (deadline - budget_s + 30.0)))

    # === Phase 1: Spectral flattening (200s) ===
    remaining = deadline - time.monotonic()
    p1_budget = min(200.0, remaining * 0.15)
    if p1_budget > 20:
        print("\n=== Phase 1: Spectral flattening (budget %.0fs) ===" % p1_budget)
        t_phase = time.monotonic()
        a1, obj1 = spectral_flatten(global_best_a, p1_budget, rng, S0)
        print("  Spectral: obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t_phase))
        update_global(a1)

    # === Phase 2: Adaptive FISTA (400s) ===
    remaining = deadline - time.monotonic()
    p2_budget = min(400.0, remaining * 0.2)
    if p2_budget > 30:
        print("\n=== Phase 2: Adaptive FISTA (budget %.0fs) ===" % p2_budget)
        t_phase = time.monotonic()
        a2, obj2 = run_fista_adaptive(global_best_a, p2_budget, rng)
        print("  FISTA: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t_phase))
        update_global(a2)

    # === Phase 3: LP direction finding (300s) ===
    remaining = deadline - time.monotonic()
    p3_budget = min(300.0, remaining * 0.15)
    if p3_budget > 30 and len(global_best_a) <= 16384:
        print("\n=== Phase 3: LP direction (budget %.0fs) ===" % p3_budget)
        t_phase = time.monotonic()
        a3, obj3 = run_lp_direction(global_best_a, p3_budget, rng)
        print("  LP: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t_phase))
        update_global(a3)

    # === Phase 4: Multi-scale exploration (400s) ===
    remaining = deadline - time.monotonic()
    p4_budget = min(400.0, remaining * 0.2)
    if p4_budget > 50:
        print("\n=== Phase 4: Multi-scale (budget %.0fs) ===" % p4_budget)
        t_phase = time.monotonic()

        # Try upsample to 2n
        n_cur = len(global_best_a)
        if n_cur <= 8192:
            n_up = n_cur * 2
            S_up = np.sqrt(2.0 * n_up)
            a_up = upsample_interp(global_best_a, factor=2)
            a_up = a_up * (S_up / a_up.sum())
            print("  Upsampled to n=%d, initial obj=%.6f" % (n_up, compute_obj(a_up)))

            # Spectral flatten at new size
            sub_budget = p4_budget * 0.3
            a_up1, obj_up1 = spectral_flatten(a_up, sub_budget, rng, S_up)
            print("  Spectral at 2n: obj=%.6f" % obj_up1)
            update_global(a_up1)

            # FISTA at new size
            remaining4 = p4_budget - (time.monotonic() - t_phase)
            if remaining4 > 30:
                a_up2, obj_up2 = run_fista_adaptive(a_up1, remaining4, rng)
                print("  FISTA at 2n: obj=%.6f" % obj_up2)
                update_global(a_up2)

        # Also try downsample if current is large
        if n_cur > 4096:
            n_ds = n_cur // 2
            S_ds = np.sqrt(2.0 * n_ds)
            a_ds = downsample_avg(global_best_a, factor=2)
            a_ds = a_ds * (S_ds / a_ds.sum())
            obj_ds = compute_obj(a_ds)
            print("  Downsampled to n=%d, obj=%.6f" % (n_ds, obj_ds))
            update_global(a_ds)

    # === Phase 5: Intensive subgradient polish (400s) ===
    remaining = deadline - time.monotonic()
    p5_budget = min(400.0, remaining * 0.25)
    if p5_budget > 30:
        print("\n=== Phase 5: Intensive polish (budget %.0fs) ===" % p5_budget)
        t_phase = time.monotonic()
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)

        # Run FISTA with high beta, focused mode
        a5, obj5 = run_fista_adaptive(global_best_a, p5_budget * 0.5, rng)
        print("  Polish 1: obj=%.6f (t=%.0fs)" % (obj5, time.monotonic() - t_phase))
        update_global(a5)

        remaining5 = p5_budget - (time.monotonic() - t_phase)
        if remaining5 > 20:
            # Second pass with different rng (different noise pattern)
            rng2 = np.random.default_rng(seed + 12345)
            a5b, obj5b = run_fista_adaptive(global_best_a, remaining5, rng2)
            print("  Polish 2: obj=%.6f" % obj5b)
            update_global(a5b)

    # === Phase 6: Final spectral + FISTA combo (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Final combo (budget %.0fs) ===" % remaining)
        t_phase = time.monotonic()

        # Spectral polish
        sub_budget = remaining * 0.4
        a6, obj6 = spectral_flatten(global_best_a, sub_budget, rng,
                                     np.sqrt(2.0 * len(global_best_a)))
        print("  Final spectral: obj=%.6f" % obj6)
        update_global(a6)

        remaining6 = remaining - (time.monotonic() - t_phase)
        if remaining6 > 20:
            a6b, obj6b = run_fista_adaptive(global_best_a, remaining6, rng)
            print("  Final FISTA: obj=%.6f" % obj6b)
            update_global(a6b)

    # === Final verification ===
    print("\n=== Final: obj=%.6f (n=%d) ===" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Basic sanity check
        total = sum(final_list)
        if total < 0.01:
            print("WARNING: sum too small, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
