"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-method: FISTA with oscillating beta, L-BFGS-B penalty polishing,
DCT-domain structured restarts, and diverse multi-start.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_objective(a, beta):
    """LSE-smoothed max of convolution."""
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.exp(shifted).sum().__float__() / beta


def run_fista_osc(a_init, max_time, rng, beta_start=10.0, beta_osc_amp=1.5,
                  beta_osc_period=3000, step_scale=1.0, mode="lse"):
    """
    FISTA with oscillating beta schedule to escape local minima.
    Beta oscillates between beta_start*beta_osc_amp^-1 and beta_start*beta_osc_amp.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Oscillating beta: ramps up then periodically drops
        base_beta = beta_start * (1.0 + 0.003 * it)
        osc = np.sin(2.0 * np.pi * it / beta_osc_period)
        beta = base_beta * (1.0 + beta_osc_amp * 0.3 * osc)
        beta = max(5.0, min(beta, 50000.0))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 1000:
                k = max(2, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:  # hybrid
            if stuck_count > 800 and (it % 5 == 0 or rng.random() < 0.3):
                tol = min(0.05, 0.001 * stuck_count)
                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # DCT-domain noise when stuck (structured exploration)
        if stuck_count > 2000:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-7:
                # Generate noise in DCT domain (smooth perturbations)
                noise_time = rng.normal(0, 1.0, n)
                # Apply low-pass in frequency domain for smooth noise
                F = fft(noise_time)
                cutoff = max(5, n // 50)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0
                stuck_count = 0

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=5):
    """
    L-BFGS-B polishing with penalty method for sum constraint.
    Minimizes LSE(conv(a,a), beta) + penalty*(sum(a)-S)^2 subject to a>=0.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [20.0, 50.0, 100.0, 300.0, 800.0][:n_beta_levels]
    penalties = [5.0, 10.0, 20.0, 50.0, 100.0][:n_beta_levels]

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        c = fft_convolve_self(a)
        cm = c.max()

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            # LSE
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            # Penalty
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            # Gradient
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)  # penalty gradient (same for all)
            return obj, grad

        time_budget_this = remaining * 0.85
        t0 = time.monotonic()

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 100, 'maxls': 30, 'ftol': 1e-14, 'gtol': 1e-10,
                     'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, pen, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """Intensive polishing using exact subgradient with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 100:
            k = 1
            tol_frac = 0.0
        elif stuck < 500:
            k = 3
            tol_frac = 0.003
        elif stuck < 2000:
            k = min(16, 4 + stuck // 200)
            tol_frac = 0.008
        else:
            k = min(64, 16 + stuck // 500)
            tol_frac = 0.02

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def generate_diverse_inits(n, rng, S):
    """Generate diverse initial sequences for multi-start."""
    inits = []

    # Cosine with different frequencies
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Smooth random (Gaussian kernel)
    for _ in range(3):
        raw = rng.random(n)
        # Smooth via FFT low-pass
        F = fft(raw)
        cutoff = max(3, n // 100)
        F[cutoff:-cutoff] *= 0.05
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() > 1e-10:
            smooth = smooth * (S / smooth.sum())
            inits.append(smooth.copy())

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    if tri.sum() > 0:
        tri = tri * (S / tri.sum())
        inits.append(tri.copy())

    # Two-peak
    peak1 = n // 4
    peak2 = 3 * n // 4
    sigma = n // 20
    a = np.exp(-0.5 * ((np.arange(n) - peak1) / sigma) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - peak2) / sigma) ** 2)
    a = a * (S / a.sum())
    inits.append(a.copy())

    return inits

        obj5 = compute_obj(a5)
        print("  Active polish: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.8, rng, n_beta_levels=3)
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            # Re-optimize if verification gives different value
            if ver < global_best_obj + 1e-10:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
