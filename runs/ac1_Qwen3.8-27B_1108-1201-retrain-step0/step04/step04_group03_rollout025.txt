"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-mode FISTA with LSE smoothing,
exact top-K subgradient steps, multiplicative repulsion, and adaptive noise.
Key innovations: higher beta ceiling, periodic exact subgradient exploitation,
dual-length optimization (8192 + 16384), and aggressive restart strategy.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(y, c, beta):
    """Gradient of LSE(beta * c) with respect to y."""
    n = len(y)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, top_k=10, smooth_eps=0.0):
    """Compute subgradient using top-K lags with optional smoothing."""
    n = len(y)
    num_c = len(c)
    # Get top-K indices
    if top_k >= num_c:
        top_idx = np.arange(num_c)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Weight: emphasize lags close to the max
    c_max = c.max()
    c_min_top = c[top_idx].min()
    if c_max - c_min_top < 1e-12:
        weights = np.ones(top_k) / top_k
    else:
        # Exponential weighting within top-K
        raw_w = np.exp(20.0 * (c[top_idx] - c_min_top) / (c_max - c_min_top + 1e-12))
        weights = raw_w / raw_w.sum()
    
    # Build full weight vector
    w_full = np.zeros(num_c)
    w_full[top_idx] = weights
    
    # Gradient = 2 * cross-correlation(w_full, y)
    fsize = 1 << (num_c + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def multiplicative_repulsion(y, c, lr=0.05, beta_rep=30.0):
    """Multiplicative repulsion: reduce components that contribute to high lags."""
    n = len(y)
    cm = c.max()
    w = np.exp(beta_rep * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return y.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    # Multiplicative update: reduce where repulsion is high
    y_new = y * np.exp(-lr * rep / max(y.max(), 1e-10))
    y_new = np.maximum(y_new, 1e-15)
    return y_new


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

                c = c_topk
                cur = c.max()
                if cur < best:
                    best = cur
                    best_a = y.copy()
                    last_improve_it = it
                continue
        
        if use_repulsion:
            # Multiplicative repulsion step
            y_rep = multiplicative_repulsion(y, c, lr=repulsion_lr, beta_rep=30.0)
            y_rep = project_simplex(y_rep, S)
            c_rep = fft_convolve_self(y_rep)
            if c_rep.max() < cur * 1.001:  # Accept if not too bad
                y = y_rep
                a = y_rep
                t_mom = 1.0
                c = c_rep
                cur = c.max()
                if cur < best:
                    best = cur
                    best_a = y.copy()
                    last_improve_it = it
                continue
        
        # Standard LSE FISTA step
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step: scale by inverse gradient magnitude for stability
        # The theoretical safe step is 1/(2*beta), but we can be more aggressive
        # when the gradient is well-behaved
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic small noise to maintain diversity
        if it % noise_every == 0:
            decay = np.exp(-it / (3 * max_time * 60))  # Very slow decay
            ns = noise_scale_factor * 0.5 * S / np.sqrt(n) * decay
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0
        
        # Progress logging
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print("    it=%d, beta=%.0f, best=%.6f, cur=%.6f, elapsed=%.0fs" %
                  (it, beta, best, cur, elapsed))
    
    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    c_final2 = fft_convolve_self(best_a)
    best = min(best, c_final2.max())
    
    return best_a, best


def run_exact_subgradient_polish(a_init, n, max_time, rng, top_k=5, lr=0.001):
    """
    Polish using exact top-K subgradient with very small steps.
    Focuses intensely on reducing the peak lags.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Adaptive learning rate
    current_lr = lr
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # Get subgradient from top-K lags
        grad = topk_subgradient(y, c, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Normalized step
        step = current_lr * S / (gnorm + 1e-10)
        x_new = project_simplex(y - step * grad, S)
        
        # Check improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            y = x_new
        else:
            # Reduce learning rate
            current_lr *= 0.95
        
        # Periodically increase top_k to smooth the gradient
        if it % 500 == 0:
            top_k = min(50, top_k + 2)
    
    c_final = fft_convolve_self(best_a)
    best = min(best, c_final.max())
    
    return best_a, best


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Smooth families
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.05)
    inits.append(np.exp(rng.normal(0, 0.3, n)))
    
    # V-shaped (high at edges)
    inits.append(np.abs(2 * tt - 1) + 0.1)
    # U-shaped
    inits.append(4 * (tt - 0.5) ** 2 + 0.1)
    # Two bumps
    inits.append(np.exp(-20 * (tt - 0.25) ** 2) + np.exp(-20 * (tt - 0.75) ** 2) + 0.05)
    
    # Block structures
    for nblock in [7, 13, 31]:
        block_len = max(1, n // nblock)
        heights = rng.uniform(0.3, 1.5, nblock)
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])
    
    return [np.maximum(x, 0.01) for x in inits]


def score_sequence(a, n):
    """Compute the objective function value."""
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase strategy: FISTA with LSE + top-K subgradient + repulsion,
    dual-length optimization, and final polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)
    
    global_best = np.inf
    global_a = None
    global_n = 0
    
    # Get the given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Determine main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192
    
    print("Main length: %d, budget: %.0fs" % (main_len, budget_s))
    
    # Phase 1: Quick FISTA warmup on given sequence
    n = main_len
    S = np.sqrt(2.0 * n)
    remaining = deadline - time.monotonic()
    
    if given is not None and remaining > 60:
        t1 = min(60.0, remaining * 0.1)
        print("=== Phase 1: FISTA warmup (n=%d, %.0fs) ===" % (n, t1))
        a1, s1 = run_multi_mode_fista(given, n, t1, rng,
                                       beta_start=20.0, beta_rate=0.002, beta_max=500.0,
                                       topk_every=300, topk_k=5, topk_lr_factor=0.5,
                                       repulsion_every=500, repulsion_lr=0.05,
                                       stuck_threshold=1500, verbose=False)
        a1 = normalize_sum(a1, n)
        obj1 = score_sequence(a1, n)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
            global_n = n
        print("  Phase 1 best: %.6f" % global_best)
    
    if global_a is None:
        # Fallback: use a cosine initialization
        n = main_len
        tt = np.linspace(0, 1, n, endpoint=False)
        global_a = np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01)
        global_a = normalize_sum(global_a, n)
        global_n = n
        global_best = score_sequence(global_a, n)
        print("  Fallback init: %.6f" % global_best)
    
    n = global_n
    S = np.sqrt(2.0 * n)
    
    # Phase 2: Main optimization (bulk of the time)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        t2 = remaining * 0.5
        print("=== Phase 2: Main optimization (n=%d, %.0fs) ===" % (n, t2))
        a2, s2 = run_multi_mode_fista(global_a, n, t2, rng,
                                       beta_start=50.0, beta_rate=0.003, beta_max=5000.0,
                                       topk_every=200, topk_k=10, topk_lr_factor=0.5,
                                       repulsion_every=800, repulsion_lr=0.04,
                                       stuck_threshold=2000, verbose=True)
        a2 = normalize_sum(a2, n)
        obj2 = score_sequence(a2, n)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            global_n = n
        print("  Phase 2 best: %.6f" % global_best)
    
    # Phase 3: Exact subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_p = len(global_a)
        t3 = min(120.0, remaining * 0.2)
        print("=== Phase 3: Subgradient polish (n=%d, %.0fs) ===" % (n_p, t3))
        a3, s3 = run_exact_subgradient_polish(global_a, n_p, t3, rng, top_k=5, lr=0.002)
        a3 = normalize_sum(a3, n_p)
        obj3 = score_sequence(a3, n_p)
        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()
            global_n = n_p
        print("  Phase 3 best: %.6f" % global_best)
    
    # Phase 4: Try larger length
    remaining = deadline - time.monotonic()
    if remaining > 180 and global_a is not None:
        alt_n = global_n * 2 if global_n <= 8192 else global_n // 2
        if 4096 <= alt_n <= 32768:
            t4 = remaining * 0.35
            print("=== Phase 4: Alternate length n=%d (%.0fs) ===" % (alt_n, t4))
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            a4, s4 = run_multi_mode_fista(alt_init, alt_n, t4, rng,
                                           beta_start=30.0, beta_rate=0.002, beta_max=3000.0,
                                           topk_every=300, topk_k=8, topk_lr_factor=0.5,
                                           repulsion_every=1000, repulsion_lr=0.03,
                                           stuck_threshold=2500, verbose=False)
            a4 = normalize_sum(a4, alt_n)
            obj4 = score_sequence(a4, alt_n)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                global_n = alt_n
            print("  Phase 4 (n=%d): %.6f %s" % (alt_n, obj4,
                  "(IMPROVED)" if obj4 < global_best else ""))
    
    # Phase 5: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        t5 = remaining * 0.6
        print("=== Phase 5: Final polish (n=%d, %.0fs) ===" % (n_f, t5))
        
        # High-beta FISTA
        a5, s5 = run_multi_mode_fista(global_a, n_f, t5 * 0.5, rng,
                                       beta_start=200.0, beta_rate=0.001, beta_max=10000.0,
                                       topk_every=100, topk_k=5, topk_lr_factor=0.7,
                                       repulsion_every=2000, repulsion_lr=0.02,
                                       stuck_threshold=3000, verbose=False)
        a5 = normalize_sum(a5, n_f)
        obj5 = score_sequence(a5, n_f)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
        
        # Exact subgradient with very small steps
        remaining5 = deadline - time.monotonic()
        if remaining5 > 20:
            a5b, s5b = run_exact_subgradient_polish(global_a, n_f, remaining5 * 0.5, rng,
                                                     top_k=3, lr=0.001)
            a5b = normalize_sum(a5b, n_f)
            obj5b = score_sequence(a5b, n_f)
            if obj5b < global_best:
                global_best = obj5b
                global_a = a5b.copy()
        
        print("  Phase 5 best: %.6f" % global_best)
    
    # Phase 6: Try a few diverse restarts if time remains
    remaining = deadline - time.monotonic()
    if remaining > 90 and global_a is not None:
        n_d = len(global_a)
        S_d = np.sqrt(2.0 * n_d)
        print("=== Phase 6: Diverse restarts (remaining %.0fs) ===" % remaining)
        inits = generate_initializations(n_d, rng)
        per_init = (remaining * 0.4) / min(3, len(inits))
        for i, init in enumerate(inits[:3]):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 10:
                break
            init_n = normalize_sum(init, n_d)
            ai, si = run_multi_mode_fista(init_n, n_d, per_init, rng,
                                           beta_start=30.0, beta_rate=0.003, beta_max=2000.0,
                                           topk_every=200, topk_k=8, topk_lr_factor=0.5,
                                           repulsion_every=600, repulsion_lr=0.04,
                                           stuck_threshold=1500, verbose=False)
            ai = normalize_sum(ai, n_d)
            obji = score_sequence(ai, n_d)
            if obji < global_best:
                global_best = obji
                global_a = ai.copy()
                print("  Diverse init %d: %.6f (IMPROVED)" % (i, global_best))
            else:
                print("  Diverse init %d: %.6f" % (i, obji))
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification score: %.6f" % ver)
            # Update global best with verified score
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("Returning sequence of length %d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
