def full_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()  # next power of 2
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u - (css - S) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1] + 1
    theta = (css[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max of convolution c."""
    n = len(a)

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm:
1. LSE-smoothed gradient descent with adaptive beta and FISTA acceleration
2. Top-k exact subgradient polishing for final convergence
3. Multi-phase approach with diverse inits and progressive refinement

Key innovations over previous approach:
- Clean gradient computation (verified correct via FFT cross-correlation)
- Adaptive step size via gradient magnitude monitoring
- Phased optimization: smooth → precise → fine
- Avoids momentum instability by using restarts more aggressively
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def full_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}. O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho_mask = u - (css - S) / np.arange(1, n + 1) > 0
    rho_idx = np.where(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.full(n, S / n)
    rho = rho_idx[-1] + 1
    theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max of convolution c with parameter beta.
    
    Returns grad[m] = 2 * sum_l a[l] * w[l+m] where w = softmax(beta*(c - max(c))).
    """
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # Cross-correlation: grad[m] = 2 * sum_{l=0}^{n-1} a[l] * w[l+m]
    N = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def topk_grad(a, c, k=8):
    """Gradient using top-k peaks weighted by excess above mean."""
    n = len(a)
    idx = np.argsort(c)[::-1][:k]
    excess = np.maximum(c[idx] - c.mean(), 0.0)
    wsum = excess.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = excess / wsum
    
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    
    N = 1 << (n + len(c) - 1).bit_length()
    A = fft(a, N)
    W = fft(w_full, N)
    return 2.0 * np.real(ifft(np.conj(A) * W))[:n]


def single_peak_grad(a, c):
    """Gradient from just the single maximum peak."""
    n = len(a)
    k_max = np.argmax(c)
    w_full = np.zeros(len(c))
    w_full[k_max] = 1.0
    
    N = 1 << (n + len(c) - 1).bit_length()
    A = fft(a, N)
    W = fft(w_full, N)
    return 2.0 * np.real(ifft(np.conj(A) * W))[:n]

                    beta_max=50000.0, restart_interval=300):

    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    
    c = full_conv(a)
    best_cmax = c.max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = full_conv(y)
        cmax = c.max()
        if cmax < best_cmax - 1e-12:
            best_cmax = cmax
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: 1/(L) where L ≈ 2*beta*cmax (conservative)
        # Use 0.5 factor for safety
        alpha = 1.0 / (2.0 * beta * max(cmax, 1e-10))
        
        x_new = project_simplex(y - alpha * grad, S)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        x_prev = a
        a = x_new
        t = t_new
        
        # Restart if stuck
        if no_improve >= restart_interval:
            y = best_a.copy()
            x_prev = y.copy()
            t = 1.0
            no_improve = 0
            # Small perturbation to escape
            if beta < beta_max * 0.5:
                noise_scale = 0.003 * S / np.sqrt(len(a))
                y = project_simplex(y + rng.normal(0, noise_scale, len(a)), S)
                x_prev = y.copy()
    
    # Final evaluation
    c_final = full_conv(a)
    if c_final.max() < best_cmax:
        best_cmax = c_final.max()
        best_a = a.copy()
    
    return best_a, best_cmax


def phase_topk_subgrad(a_init, S, max_time, rng, k=8):
    """Subgradient method using top-k peaks with line search."""
    a = project_simplex(a_init.copy(), S)
    n = len(a)
    start = time.monotonic()
    
    c = full_conv(a)
    best_cmax = c.max()
    best_a = a.copy()
    
    # Initial step: based on problem scale
    step = S / (n * 5.0)
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = full_conv(a)
        cmax = c.max()
        
        if cmax < best_cmax - 1e-12:
            best_cmax = cmax
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 2000:
            # Try different k
            k = rng.integers(2, 32) if k.equals is None else int(rng.integers(2, 32))
            a = best_a.copy()
            step = S / (n * 5.0)
            stuck = 0
            continue
        
        if cmax <= c.mean() * (1 + 1e-5):
            break
        
        grad = topk_grad(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Line search (backtracking)
        alpha = step
        improved = False
        for _ in range(10):
            a_try = project_simplex(a - alpha * grad, S)
            c_try = full_conv(a_try)
            if c_try.max() < cmax - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try smaller step or different direction
            step *= 0.9
            if step < 1e-18:
                break
            # Add small noise
            a = project_simplex(a + rng.normal(0, 1e-6 * S / np.sqrt(n), n), S)
        
        # Gradually decrease step
        step *= 0.9995
    
    return best_a, best_cmax
