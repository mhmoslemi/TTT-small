"""

Strategy: Multi-algorithm ensemble with:
1. FISTA with adaptive LSE + Barzilai-Borwein step size

3. Active-set coordinate descent with exact 1D minimization
4. Multi-scale refinement (n -> 2n -> 4n)
5. Aggressive restart policy with diversity

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_sum_gradient(y, c, p):
    n = len(y)
    c_pos = np.maximum(c, 1e-30)
    w = np.power(c_pos, p - 1.0)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_bb(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                 beta_max=10000.0, mode="lse", step_scale=1.0):
    """FISTA with LSE smoothing and Barzilai-Borwein adaptive step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    # BB step size state
    prev_x = a.copy()
    prev_y = y.copy()
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "power":
            p = min(200.0, 2.0 * beta)
            grad = power_sum_gradient(y, c, p)
        elif mode == "hybrid":
            if stuck_count > 200:
                k = min(32, 4 + stuck_count // 100)
                grad = topk_gradient(y, c, k, power=1.5)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(2, min(64, n // 64))
            grad = topk_gradient(y, c, k, power=1.0)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size
        s_vec = y - prev_y
        y_vec = grad - (prev_y - y + (y - a) * 0)  # simplified
        sy = np.dot(s_vec, s_vec)
        yg = np.dot(s_vec, grad)
        if sy > 1e-15 and abs(yg) > 1e-15:
            step_bb = sy / abs(yg)
            step_bb = np.clip(step_bb, 0.1 * step_scale / max(beta, 1.0),
                             5.0 * step_scale / max(beta, 1.0))
            step = 0.5 * step + 0.5 * step_bb
        else:
            step = step_scale / max(beta, 1.0)

        # Adaptive: reduce step when stuck
        if stuck_count > 2000:
            step *= 0.5
            if stuck_count > 5000 and rng.random() < 0.1:
                noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
                if noise_scale > 1e-7:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    a = y.copy()
                    t_mom = 1.0
                    stuck_count = 0

        prev_y = y.copy()
        prev_x = a.copy()

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_mirror_descent(a_init, max_time, rng, beta_init=20.0, beta_rate=0.004,
                       beta_max=5000.0):
    """Mirror descent with multiplicative weight updates and LSE gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Ensure strictly positive for log
    epsilon = 1e-10
    a = np.maximum(a, epsilon)
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = 1.0 / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent update: multiplicative
        # a_new[i] = a[i] * exp(-step * grad[i]) / sum_j a[j]*exp(-step*grad[j])
        exponent = -step * grad
        exponent -= exponent.max()  # numerical stability
        weights = a * np.exp(exponent)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        x_new = weights / wsum * S
        x_new = np.maximum(x_new, 0.0)
        x_new = project_simplex(x_new, S)

        # FISTA-like momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, epsilon)
        y = y * (S / y.sum())
        a = x_new
        t_mom = t_new

        # Restart when stuck
        stuck = it - last_improve
        if stuck > 3000 and it % 500 == 0:
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            perturbation = 1.0 + rng.normal(0, noise_scale / (S / n), n)
            perturbation = np.maximum(perturbation, epsilon)
            y = a * perturbation
            y = y * (S / y.sum())
            a = y.copy()
            t_mom = 1.0

        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

            # Let me just use the full range.

            # Simpler: compute the gradient-like vector
            # dc/da[j] at current point: for each k, the coefficient of delta
            # u[k] = a[k-j] + a[j] if 0<=j<n and 0<=k-j<n
            # But for 1D search, we need the full quadratic structure.

            # Let me use a simpler approach: try a few candidate values
            # and pick the best one.

            # Candidate deltas: reduce v by various fractions
            # Also try increasing v slightly
            candidates = [v * 0.5, v * 0.7, v * 0.8, v * 0.9, v * 1.1, v * 1.3]
            # Add: move mass to/from neighbors
            if j > 0:
                candidates.append(v * 0.3)  # big reduction
            if j < n - 1:
                candidates.append(v * 0.3)

            best_val = v
            best_local = cur_max

            for new_v in candidates:
                new_v = max(0.0, new_v)
                a[j] = new_v
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = v
                    continue
                scale = S / s_new
                # Instead of full renormalization (which changes all coords),
                # just check the objective proportionally
                # obj = 2n * max(c) / S^2, and max(c) scales as scale^2
                # So obj doesn't change under scaling! We just need max(c)/S^2
                c_check = fft_convolve_self(a)
                local_max = c_check.max()
                if local_max < best_local - 1e-12:
                    best_local = local_max
                    best_val = new_v

            a[j] = best_val
            s_new = a.sum()
            if abs(s_new - S) > 1e-10 * S:
                a = a * (S / s_new)

        # Full pass done, check improvement
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S2
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1

        if no_improve > 5:
            # Try perturbation
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """Intensive active-set polishing with adaptive gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive gradient strategy
        if stuck < 100:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        elif stuck < 500:
            grad = topk_gradient(y, c, k=3, power=1.0)
        elif stuck < 2000:
            k = min(64, 4 + stuck // 100)
            grad = topk_gradient(y, c, k, power=1.2)
        else:
            beta = min(5000.0, 50.0 * np.exp(0.001 * stuck))
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step with oscillation
        base_step = S / (gnorm * (1.0 + 0.002 * it))
        step = min(base_step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000 and it % 200 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample_cubic(a, factor=2):
    """Upsample using cubic interpolation with OHLC (optimal harmonic)."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)

    # Use scipy-like cubic interpolation via FFT
    # Upsample in frequency domain (zero-padding)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # The upsampled sequence is the lowpass-filtered version
    # Zeros out frequencies above the new Nyquist
    # New Nyquist in old frequency units: n_new / (2 * n_new) = 1/2
    # Old frequencies: 0, 1, ..., fsize-1 mapped to 0, ..., n-1 (original)
    # We keep frequencies that correspond to the original band
    n_keep = n  # keep first n frequency bins (and symmetric)
    A_up = np.zeros(fsize, dtype=np.complex128)
    A_up[:n_keep] = A[:n_keep]
    A_up[fsize - n_keep + 1:] = A[fsize - n_keep + 1:]

    a_up = np.real(ifft(A_up))[:n_new]
    a_up = np.maximum(a_up, 0.0)
    if a_up.sum() < 1e-15:
        a_up = np.interp(t_new, t_old, a)
    return a_up


def upsample_linear(a, factor=2):
    """Simple linear upsampling."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    n_new = len(a) // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-algorithm ensemble: FISTA-BB, mirror descent, active polish,
    multi-scale refinement, and random restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    t_start = time.monotonic()

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = np.ones(n) * (np.sqrt(2.0 * n) / n)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Intensive FISTA at original size (20% time) ===
    p1_budget = (deadline - time.monotonic()) * 0.20
    print("\n=== Phase 1: FISTA-BB at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_bb(global_best_a, p1_budget * 0.4, rng,
                         beta_init=20.0, beta_rate=0.006,
                         beta_max=8000.0, mode="lse", step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  FISTA-BB lse: obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem = p1_budget - (time.monotonic() - t1)
    if rem > 10:
        a1b, _ = run_fista_bb(global_best_a, rem, rng,
                              beta_init=100.0, beta_rate=0.003,
                              beta_max=20000.0, mode="hybrid", step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-BB hybrid: obj=%.6f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Mirror descent (15% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15
    if p2_budget > 15:
        print("\n=== Phase 2: Mirror descent at n=%d (budget %.0fs) ===" % (n0, p2_budget))
        t2 = time.monotonic()

        a2, _ = run_mirror_descent(global_best_a, p2_budget, rng,
                                   beta_init=30.0, beta_rate=0.003,
                                   beta_max=5000.0)
        obj2 = compute_obj(a2)
        print("  Mirror descent: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Multi-scale upsample (30% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.30
    if p3_budget > 60 and n0 <= 32768:
        n2 = n0 * 2
        print("\n=== Phase 3: Multi-scale to n=%d (budget %.0fs) ===" % (n2, p3_budget))
        t3 = time.monotonic()

        # Upsample
        a2_init = upsample_cubic(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())

        # Check if upsampled version is already good
        obj2_init = compute_obj(a2_init)
        print("  Upsampled init: obj=%.6f" % obj2_init)
        if obj2_init < global_best_obj:
            global_best_obj = obj2_init
            global_best_a = a2_init.copy()

        # Optimize at new size
        a3, _ = run_fista_bb(a2_init, p3_budget * 0.4, rng,
                             beta_init=10.0, beta_rate=0.005,
                             beta_max=5000.0, mode="lse", step_scale=1.0)
        obj3 = compute_obj(a3)
        print("  FISTA at 2n: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        rem3 = p3_budget - (time.monotonic() - t3)
        if rem3 > 15:
            a3b, _ = run_fista_bb(global_best_a, rem3 * 0.5, rng,
                                  beta_init=50.0, beta_rate=0.003,
                                  beta_max=10000.0, mode="hybrid", step_scale=0.5)
            obj3b = compute_obj(a3b)
            print("  Hybrid at 2n: obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

            rem3b = rem3 - (time.monotonic() - t3 - (p3_budget * 0.4 + rem3 * 0.5))
            # Just use remaining time
            rem3b = p3_budget - (time.monotonic() - t3)
            if rem3b > 5:
                a3c, _ = run_active_polish(global_best_a, rem3b, rng)
                obj3c = compute_obj(a3c)
                print("  Active polish at 2n: obj=%.6f (t=%.0fs)" % (obj3c, time.monotonic() - t3))
                if obj3c < global_best_obj:
                    global_best_obj = obj3c
                    global_best_a = a3c.copy()

    # === Phase 4: Active-set polish at best size (25% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.25
    if p4_budget > 20:
        n_best = len(global_best_a)
        print("\n=== Phase 4: Active polish at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        a4, _ = run_active_polish(global_best_a, p4_budget * 0.5, rng)
        obj4 = compute_obj(a4)
        print("  Active polish: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

        rem4 = p4_budget - (time.monotonic() - t4)
        if rem4 > 10:
            # Very high beta FISTA
            a4b, _ = run_fista_bb(global_best_a, rem4, rng,
                                  beta_init=200.0, beta_rate=0.002,
                                  beta_max=50000.0, mode="topk", step_scale=0.3)
            obj4b = compute_obj(a4b)
            print("  Topk FISTA: obj=%.6f (t=%.0fs)" % (obj4b, time.monotonic() - t4))
            if obj4b < global_best_obj:
                global_best_obj = obj4b
                global_best_a = a4b.copy()

    # === Phase 5: Random restarts on best (10% time) ===
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    if p5_budget > 15:
        n_best = len(global_best_a)
        print("\n=== Phase 5: Random restarts at n=%d (budget %.0fs) ===" % (n_best, p5_budget))
        t5 = time.monotonic()

        n_restarts = min(5, int(p5_budget / 20))
        per_restart = p5_budget / max(1, n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 5:
                break
            # Perturb the best solution
            noise_scale = 0.01 * np.sqrt(2.0 * n_best) / np.sqrt(n_best)
            a_pert = global_best_a + rng.normal(0, noise_scale, n_best)
            S_best = np.sqrt(2.0 * n_best)
            a_pert = project_simplex(a_pert, S_best)

            a_r, _ = run_fista_bb(a_pert, per_restart * 0.7, rng,
                                  beta_init=50.0, beta_rate=0.005,
                                  beta_max=10000.0, mode="lse", step_scale=0.8)
            obj_r = compute_obj(a_r)
            print("  Restart %d: obj=%.6f" % (r + 1, obj_r))
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_r.copy()

    # === Phase 6: Final polish with remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        n_best = len(global_best_a)
        print("\n=== Phase 6: Final polish (remaining %.0fs) ===" % remaining)
        t6 = time.monotonic()

        a6, _ = run_active_polish(global_best_a, remaining * 0.6, rng)
        obj6 = compute_obj(a6)
        print("  Final active: obj=%.6f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

        rem6 = remaining - (time.monotonic() - t6)
        if rem6 > 5:
            a6b, _ = run_fista_bb(global_best_a, rem6, rng,
                                  beta_init=500.0, beta_rate=0.001,
                                  beta_max=100000.0, mode="hybrid", step_scale=0.2)
            obj6b = compute_obj(a6b)
            print("  Final FISTA: obj=%.6f (t=%.0fs)" % (obj6b, time.monotonic() - t6))
            if obj6b < global_best_obj:
                global_best_obj = obj6b
                global_best_a = a6b.copy()

    # === Final verification ===
    print("\nBest found: obj=%.6f (n=%d)" % (global_best_obj, len(global_best_a)))
    print("Total time: %.0fs" % (time.monotonic() - t_start))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluation rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
