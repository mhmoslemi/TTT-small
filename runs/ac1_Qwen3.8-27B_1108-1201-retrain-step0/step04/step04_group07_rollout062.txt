```python
"""
Search for a non-negative sequence minimizing

    2 * n * max(convolve(a, a)) / sum(a)^2

using a combination of:
- logarithmic additively-weighted minimization with FFT-based subgradients,
- target-convolution spectral phase-randomization (random square roots),
- multi-length seeding and periodic basin hopping.

The previous construction is used as one seed when available.  The objective is
scale-invariant, so all internal optimizers maintain sum(a) = 1.
"""

import time
import numpy as np

try:
    from scipy.fft import fft as _sc_fft, ifft as _sc_ifft

    def my_fft(x, n=None):
        try:
            return _sc_fft(x, n, workers=1)
        except Exception:
            return _sc_fft(x, n)

    def my_ifft(x, n=None):
        try:
            return _sc_ifft(x, n, workers=1)
        except Exception:
            return _sc_ifft(x, n)

except Exception:
    from numpy.fft import fft as my_fft
    from numpy.fft import ifft as my_ifft


def fft_size_for_sequence(n):
    n = int(n)
    L = 2 * n - 1
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def norm_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0.0:
        a = a / s
    else:
        a.fill(1.0 / float(a.size))
    return a


def resize_sequence(a, n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = norm_nonnegative(a)
    if a.size == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 1e-300:
        out = out / s
    else:
        out.fill(1.0 / float(n))
    return out


def compute_conv_full(q, N):
    q = np.asarray(q, dtype=np.float64)
    n = int(q.size)
    L = 2 * n - 1
    if N is None or int(N) < L:
        N = fft_size_for_sequence(n)
    Q = my_fft(q, int(N))
    c = my_ifft(Q * Q).real
    if c.size > L:
        c = c[:L]
    elif c.size < L:
        c = np.pad(c, (0, L - c.size), mode="constant")
    return c, Q


def conv_max_from_q(q, N=None):
    q = np.asarray(q, dtype=np.float64)
    n = int(q.size)
    if n == 0:
        return 0.0
    c, _ = compute_conv_full(q, N)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def softmax_theta(theta, floor=1e-15):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return theta
    if not np.all(np.isfinite(theta)):
        return np.full(theta.size, 1.0 / float(theta.size), dtype=np.float64)
    t = theta - float(np.max(theta))
    v = np.exp(t)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    v = v / s
    if floor > 0.0:
        v = np.maximum(v, float(floor))
        s = float(v.sum())
        if s > 0.0:
            v = v / s
        else:
            v.fill(1.0 / float(v.size))
    return v


def log_from_q(q, floor=1e-15):
    q = norm_nonnegative(q)
    if q.size == 0:
        return q
    q = np.maximum(q, float(floor))
    s = float(q.sum())
    if s > 0.0:
        q = q / s
    return np.log(q)


def fill_to_sum(d, U):
    d = np.asarray(d, dtype=np.float64).copy()
    L = int(d.size)
    if L == 0:
        return d
    avg = 1.0 / float(L)
    U = max(float(U), avg)

    deficit = 1.0 - float(d.sum())
    if deficit <= 1e-13:
        return np.minimum(d, U)

    cap = np.maximum(U - d, 0.0)
    total_cap = float(cap.sum())

    if total_cap <= 1e-300 or deficit >= total_cap:
        out = np.full(L, U, dtype=np.float64)
        s = float(out.sum())
        if s > 0.0:
            out = out / s
        return out

    idx = np.argsort(d)
    d_sorted = d[idx]
    cap_sorted = np.maximum(U - d_sorted, 0.0)
    cum = np.cumsum(cap_sorted)
    k = int(np.searchsorted(cum, deficit, side="right"))

    out = d.copy()
    if k > 0:
        out[idx[:k]] = U
        rem = deficit - float(cum[k - 1])
    else:
        rem = deficit

    if k < L:
        out[idx[k]] = d[idx[k]] + rem

    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 0.0 and abs(s - 1.0) > 1e-12:
        out = out / s
    out = np.minimum(out, U)
    return out


def make_bounded_target(c, U):
    c = np.asarray(c, dtype=np.float64).copy()
    L = int(c.size)
    if L == 0:
        return c

    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 1e-300:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    c = c / s

    avg = 1.0 / float(L)
    U = max(float(U), avg)

    d = np.minimum(c, U)
    d = fill_to_sum(d, U)

    for _ in range(3):
        d = 0.5 * (d + d[::-1])
        d = np.maximum(d, 0.0)
        s = float(d.sum())
        if s > 0.0:
            d = d / s
        if float(np.max(d)) <= U * 1.0005:
            break
        d = np.minimum(d, U)
        d = fill_to_sum(d, U)

    d = np.maximum(d, 0.0)
    s = float(d.sum())
    if s > 0.0:
        d = d / s

    if float(np.max(d)) > U:
        d = np.minimum(d, U)
        d = fill_to_sum(d, U)
        d = 0.5 * (d + d[::-1])
        d = np.maximum(d, 0.0)
        s = float(d.sum())
        if s > 0.0:
            d = d / s
    return d


def project_spectral_raw(r, mode):
    r = np.asarray(r, dtype=np.float64).reshape(-1)
    n = int(r.size)
    if n == 0:
        return None

    if mode == 0:
        q = r.copy()
        q[q < 0.0] = 0.0
        s = float(q.sum())
        if s <= 1e-300:
            return None
        q = q / s
        return q

    if mode == 1:
        q = r - float(np.min(r))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s <= 1e-300:
            return None
        q = q / s
        return q

    if mode == 2:
        std = float(np.std(r))
        if std <= 1e-300:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
        z = (r - float(np.max(r))) / std * 2.0
        z = z - float(np.max(z))
        q = np.exp(z)
        s = float(q.sum())
        if s <= 1e-300:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
        q = q / s
        return q

    return None


def spectral_from_target(n, d, rng, max_trials, max_time, modes, N=None):
    n = int(n)
    L = 2 * n - 1
    d = np.asarray(d, dtype=np.float64).reshape(-1)
    if d.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        m = min(d.size, L)
        if m > 0:
            tmp[:m] = d[:m]
        d = tmp
    s = float(d.sum())
    if s <= 1e-300:
        d = np.full(L, 1.0 / float(L), dtype=np.float64)
    else:
        d = d / s

    if N is None or int(N) < L:
        N = fft_size_for_sequence(n)
    N = int(N)

    D = my_fft(d, N)
    A = np.sqrt(D.astype(np.complex128))
    if D[0].real > 0.0:
        A[0] = np.sqrt(float(D[0].real))

    half = N // 2
    best_q = None
    best_score = np.inf
    deadline = time.monotonic() + max(float(max_time), 0.0)

    for _trial in range(int(max_trials)):
        if time.monotonic() > deadline:
            break

        At = A.copy()
        if half > 1:
            signs = 1.0 - 2.0 * rng.integers(0, 2, half - 1).astype(np.float64)
            At[1:half] *= signs
            At[N - half + 1:N] *= signs[::-1]
        if N % 2 == 0:
            if half < N and rng.random() < 0.5:
                At[half] = -At[half]

        raw = my_ifft(At).real
        r = raw[:n]
        pos = float(np.sum(r[r > 0.0]))
        if pos <= 1e-300:
            continue

        for mode in modes:
            q = project_spectral_raw(r, mode)
            if q is None:
                continue
            sc = 2.0 * float(n) * conv_max_from_q(q, N)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_q = q.copy()

    return best_q, best_score


def spectral_jump(q, n, target_score, rng, max_trials=60, max_time=8.0,
                  modes=(0, 1), N=None):
    n = int(n)
    q = norm_nonnegative(q)
    if q.size != n:
        q = resize_sequence(q, n)

    if N is None or int(N) < 2 * n - 1:
        N = fft_size_for_sequence(n)
    N = int(N)

    L = 2 * n - 1
    c, _ = compute_conv_full(q, N)
    c = np.maximum(c[:L], 0.0)
    s = float(c.sum())
    if s > 0.0:
        c = c / s
    else:
        c = np.full(L, 1.0 / float(L), dtype=np.float64)

    score = 2.0 * float(n) * float(c.max())
    avg = 1.0 / float(L)

    U_target = float(target_score) / (2.0 * float(n))
    if U_target < avg:
        U_target = avg

    U_current = score * 0.95 / (2.0 * float(n))
    if U_current < avg:
        U_current = avg

    U_list = [U_target]
    if abs(U_current - U_target) > 0.001 * max(U_target, avg):
        U_list.append(U_current)

    best_q = q.copy()
    best_score = score
    trials_each = max(10, int(max_trials) // max(1, len(U_list)))
    time_each = max(0.5, float(max_time) / float(len(U_list)))

    for U in U_list:
        d = make_bounded_target(c, U)
        qj, scj = spectral_from_target(n, d, rng, trials_each, time_each, modes, N)
        if qj is not None and np.isfinite(scj) and scj < best_score:
            best_q = qj.copy()
            best_score = float(scj)

    return best_q, best_score


def local_log_minimize(q0, n, target_score, max_time, rng, verbose_interval=0, N=None):
    n = int(n)
    q = norm_nonnegative(q0)
    if q.size != n:
        q = resize_sequence(q, n)

    if N is None or int(N) < 2 * n - 1:
        N = fft_size_for_sequence(n)
    N = int(N)
    L = 2 * n - 1

    if not np.isfinite(float(target_score)) or float(target_score) <= 0.0:
        target_score = 1.5000
    target_score = float(target_score)

    theta = log_from_q(q, 1e-15)
    best_q = q.copy()
    current_score = 2.0 * float(n) * conv_max_from_q(q, N)
    best_score = current_score

    if not np.isfinite(current_score):
        return best_q, best_score

    deadline = time.monotonic() + max(float(max_time), 0.0)
    base_eta = 0.8
    stall = 0
    it = 0
    floor = 1e-15

    while time.monotonic() < deadline:
        it += 1

        Q = my_fft(q, N)
        c = my_ifft(Q * Q).real[:L]
        c = np.maximum(c, 0.0)
        R = float(c.max())
        current_score = 2.0 * float(n) * R

        if not np.isfinite(current_score):
            break

        if current_score < best_score - 1e-10:
            best_score = float(current_score)
            best_q = q.copy()
            stall = 0

        if current_score <= target_score:
            break

        if current_score > best_score + 1e-8 and stall > 20:
            q = best_q.copy()
            theta = log_from_q(q, floor)
            stall = 0
            base_eta = 0.8
            continue

        if verbose_interval > 0 and it % verbose_interval == 0:
            print("    n=%d it=%d score=%.7f best=%.7f stall=%d" %
                  (n, it, current_score, best_score, stall))

        if R <= 1e-300:
            break

        gamma = 2.0 + 6.0 * min(1.0, float(it) / 5000.0)
        x = (c - R) / R
        w = np.exp(gamma * x)

        if target_score > 0.0:
            U_t = target_score / (2.0 * float(n))
            if U_t > 0.0:
                wt = np.maximum(c - U_t, 0.0)
                s = float(wt.sum())
                if s > 1e-300:
                    wt = wt / s
                    w = 0.5 * w + 0.5 * wt

        if it % 4 == 0:
            thresh = R * 0.80
            w2 = np.maximum(c - thresh, 0.0)
            s2 = float(w2.sum())
            if s2 > 1e-300:
                w2 = w2 / s2
                w = 0.5 * w + 0.5 * w2

        w = w + 1e-6
        ws = float(w.sum())
        if ws <= 1e-300:
            w = np.ones(L, dtype=np.float64) / float(L)
        else:
            w = w / ws

        W = my_fft(w, N)
        G = my_ifft(W * np.conj(Q)).real[:n]
        G = np.maximum(G, 0.0)
        Gmax = float(G.max())
        if Gmax <= 1e-300:
            break
        Gs = G / Gmax

        etas = []
        for f in (1.0, 0.5, 0.25, 2.0):
            e = float(base_eta) * f
            if e < 0.02:
                e = 0.02
            if e > 2.0:
                e = 2.0
            add = True
            for old in etas:
                if abs(old - e) <= 1e-6 * max(1.0, abs(e)):
                    add = False
                    break
            if add:
                etas.append(e)

        best_trial_q = None
        best_trial_score = current_score
        best_eta = base_eta

        for eta in etas:
            th_try = theta - eta * Gs
            q_try = softmax_theta(th_try, floor)
            if q_try is None:
                continue
            sc_try = 2.0 * float(n) * conv_max_from_q(q_try, N)
            if np.isfinite(sc_try) and sc_try < best_trial_score - 1e-12:
                best_trial_score = float(sc_try)
                best_trial_q = q_try
                best_eta = float(eta)

        if best_trial_q is not None:
            q = best_trial_q.copy()
            theta = log_from_q(q, floor)
            base_eta = max(0.05, min(2.0, best_eta))
            stall = 0
        else:
            stall += 1
            if stall <= 50 and base_eta > 0.05:
                base_eta *= 0.7
            else:
                kick = 0.02 + 0.10 * min(1.0, float(stall) / 100.0)
                theta = theta + rng.normal(0.0, kick, n)
                q = softmax_theta(theta, floor)
                if q is None:
                    q = best_q.copy()
                theta = log_from_q(q, floor)

            if stall >= 100:
                jj_target = min(target_score, best_score * 0.999)
                qj, scj = spectral_jump(q, n, jj_target, rng,
                                        max_trials=40, max_time=5.0,
                                        modes=(0, 1), N=N)
                if qj is not None and np.isfinite(scj) and scj < current_score + 1e-8:
                    q = norm_nonnegative(qj)
                    theta = log_from_q(q, floor)
                    stall = 0
                    base_eta = 0.8
                else:
                    stall = 50

            if stall > 200:
                theta = theta + rng.normal(0.0, 0.20, n)
                q = softmax_theta(theta, floor)
                if q is None:
                    q = best_q.copy()
                theta = log_from_q(q, floor)
                stall = 80

    return best_q, best_score


def sort_by_score(items):
    m = len(items)
    for i in range(m):
        for j in range(i + 1, m):
            if items[j][0] < items[i][0]:
                items[i], items[j] = items[j], items[i]
    return items


def select_top_candidates(cands, n, N, k=3):
    n = int(n)
    top = []
    for q in cands:
        if q is None:
            continue
        q2 = norm_nonnegative(q)
        if q2.size != n:
            q2 = resize_sequence(q2, n)
        sc = 2.0 * float(n) * conv_max_from_q(q2, N)
        if not np.isfinite(sc):
            continue
        if len(top) < int(k):
            top.append((float(sc), q2))
        else:
            worst = 0
            for i in range(1, len(top)):
                if top[i][0] > top[worst][0]:
                    worst = i
            if sc < top[worst][0]:
                top[worst] = (float(sc), q2)
    sort_by_score(top)
    return top


def make_initial_candidates(n, hs_arr, best_q, rng, target_score, spectral_time, N=None):
    n = int(n)
    cands = []

    if N is None:
        N = fft_size_for_sequence(n)

    if hs_arr is not None:
        cands.append(resize_sequence(hs_arr, n))

    if best_q is not None:
        cands.append(resize_sequence(best_q, n))

    base = np.full(n, 1.0 / float(n), dtype=np.float64)

    for amp in (0.05, 0.15, 0.30):
        q = base * (1.0 + amp * rng.standard_normal(n))
        cands.append(norm_nonnegative(q))

    t = np.linspace(0.0, 1.0, n, endpoint=False)
    for freq, amp in ((1, 0.30), (1, 0.60), (3, 0.40)):
        q = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
        cands.append(norm_nonnegative(q))

    q = 1.0 + 0.40 * np.sin(2.0 * np.pi * t) + 0.20 * np.sin(4.0 * np.pi * t)
    cands.append(norm_nonnegative(q))

    q = np.exp(-0.5 * ((t - 0.5) * 4.0) ** 2) + 0.05
    cands.append(norm_nonnegative(q))

    sp_time = float(spectral_time)
    if sp_time > 2.0:
        q_unif = norm_nonnegative(base)
        cu, _ = compute_conv_full(q_unif, N)
        U_init = 1.8 / (2.0 * float(n))
        d = make_bounded_target(cu, U_init)
        trials = min(500, max(50, int(sp_time * 30)))
        qsp, ssp = spectral_from_target(n, d, rng, trials, sp_time, (0, 1, 2), N)
        if qsp is not None and np.isfinite(ssp):
            cands.append(qsp)

    return cands


def threshold_improve(q):
    q = norm_nonnegative(q)
    n = int(q.size)
    if n == 0:
        return q, np.inf

    best_q = q.copy()
    best_score = 2.0 * float(n) * conv_max_from_q(q)
    if not np.isfinite(best_score):
        return best_q, best_score

    maxv = float(np.max(q))
    thresholds = [1e-15, 1e-12, 1e-10, 1e-8, 1e-6, 0.001 * maxv, 0.01 * maxv]

    for th in thresholds:
        if not np.isfinite(th) or th <= 0.0:
            continue
        q2 = q.copy()
        q2[q2 < th] = 0.0
        s = float(q2.sum())
        if s <= 0.01:
            continue
        q2 = q2 / s
        sc = 2.0 * float(n) * conv_max_from_q(q2)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_q = q2.copy()

    return best_q, best_score


def propose_candidate(seed=42, budget_s=2380.0):
    start = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2380.0
    if not np.isfinite(budget_s) or budget_s < 120.0:
        budget_s = 2380.0

    work_deadline = start + max(60.0, budget_s - 100.0)
    rng = np.random.default_rng(int(seed))

    hs_raw = globals().get("height_sequence_1", None)
    hs_arr = None
    if hs_raw is not None:
        try:
            tmp = np.asarray(hs_raw, dtype=np.float64).reshape(-1)
            if tmp.size > 0 and bool(np.any(np.isfinite(tmp))):
                hs_arr = norm_nonnegative(tmp)
        except Exception:
            hs_arr = None

    best_q = None
    best_score = np.inf
    best_short_q = None
    best_short_score = np.inf

    if hs_arr is not None:
        nhs = int(hs_arr.size)
        Nhs = fft_size_for_sequence(nhs)
        sc_h = 2.0 * float(nhs) * conv_max_from_q(hs_arr, Nhs)
        if np.isfinite(sc_h):
            best_q = hs_arr.copy()
            best_score = float(sc_h)
            if nhs <= 16384:
                best_short_q = hs_arr.copy()
                best_short_score = float(sc_h)

    lengths = []
    if hs_arr is not None and 512 <= int(hs_arr.size) <= 16384:
        lengths.append(int(hs_arr.size))
    for L in (8192, 16384):
        if L not in lengths:
            lengths.append(L)
    if len(lengths) > 2:
        lengths = lengths[:2]
    if not lengths:
        lengths = [8192, 16384]

    if len(lengths) == 1:
        fractions = [1.0]
    else:
        fractions = [0.70, 0.30]

    search_total = max(60.0, work_deadline - time.monotonic())

    for idx, n in enumerate(lengths):
        n = int(n)
        rem = work_deadline - time.monotonic()
        if rem < 90.0:
            break
        alloc = min(search_total * fractions[idx], rem - 20.0)
        if alloc < 60.0:
            break

        print("=== Optimizing n=%d with %.0fs ===" % (n, alloc))
        N = fft_size_for_sequence(n)

        sp_time = min(25.0, max(2.0, alloc * 0.02))
        cands = make_initial_candidates(n, hs_arr, best_q, rng, 1.5000, sp_time, N)
        selected = select_top_candidates(cands, n, N, k=3)

        if not selected:
            q_unif = np.full(n, 1.0 / float(n), dtype=np.float64)
            sc_unif = 2.0 * float(n) * conv_max_from_q(q_unif, N)
            selected = [(float(sc_unif), q_unif)]

        local_best_q = None
        local_best_score = np.inf

        search_part = alloc * 0.80
        weights = [0.50, 0.30, 0.20]
        wsum = 0.0
        for rank in range(len(selected)):
            if rank < len(weights):
                wsum += weights[rank]
            else:
                wsum += 0.10
        if wsum <= 0.0:
            wsum = 1.0

        for rank in range(len(selected)):
            rem2 = work_deadline - time.monotonic()
            if rem2 < 20.0:
                break
            if rank < len(weights):
                wrank = weights[rank]
            else:
                wrank = 0.10
            t_slot = min(search_part * wrank / wsum, rem2 - 10.0)
            if t_slot < 10.0:
                break

            if np.isfinite(best_score) and best_score > 0.0:
                target = max(1.0, min(1.5000, best_score * 0.998))
            else:
                target = 1.5000

            qloc, scloc = local_log_minimize(selected[rank][1], n, target,
                                             t_slot, rng, verbose_interval=300, N=N)
            if qloc is not None and np.isfinite(scloc):
                if scloc < local_best_score:
                    local_best_score = float(scloc)
                    local_best_q = qloc.copy()
                if scloc < best_score:
                    best_score = float(scloc)
                    best_q = qloc.copy()
                    if n <= 16384:
                        best_short_q = qloc.copy()
                        best_short_score = float(scloc)
                    print("  n=%d new global best=%.7f" % (n, best_score))

        rem3 = work_deadline - time.monotonic()
        final_t = min(alloc * 0.20, max(0.0, rem3 - 10.0))
        if final_t > 10.0:
            start_q = local_best_q
            if start_q is None:
                start_q = selected[0][1]
            if np.isfinite(best_score) and best_score > 0.0:
                target = max(1.0, min(1.5000, best_score * 0.998))
            else:
                target = 1.5000
            qf, scf = local_log_minimize(start_q, n, target, final_t,
                                         rng, verbose_interval=300, N=N)
            if qf is not None and np.isfinite(scf):
                if scf < local_best_score:
                    local_best_score = float(scf)
                    local_best_q = qf.copy()
                if scf < best_score:
                    best_score = float(scf)
                    best_q = qf.copy()
                    if n <= 16384:
                        best_short_q = qf.copy()
                        best_short_score = float(scf)
                    print("  n=%d polish global best=%.7f" % (n, best_score))

    time_left = max(0.0, budget_s - (time.monotonic() - start) - 60.0)
    if time_left > 20.0 and best_q is not None:
        nb = int(best_q.size)
        Nb = fft_size_for_sequence(nb)
        if np.isfinite(best_score) and best_score > 0.0:
            target = max(1.0, min(1.5000, best_score * 0.998))
        else:
            target = 1.5000
        qt, sct = local_log_minimize(best_q, nb, target, min(time_left - 15.0, 60.0),
                                     rng, verbose_interval=300, N=Nb)
        if qt is not None and np.isfinite(sct) and sct < best_score:
            best_score = float(sct)
            best_q = qt.copy()
            if nb <= 16384:
                best_short_q = qt.copy()
                best_short_score = float(sct)

    if best_q is None:
        best_q = np.ones(8192, dtype=np.float64) / 8192.0
        best_score = 2.0
    else:
        if int(best_q.size) > 20000:
            if best_short_q is not None and np.isfinite(best_short_score):
                best_q = best_short_q.copy()
                best_score = float(best_short_score)
            else:
                best_q = resize_sequence(best_q, 8192)
                best_score = 2.0 * float(best_q.size) * conv_max_from_q(best_q)

    best_q, best_score = threshold_improve(best_q)

    final_q = norm_nonnegative(best_q)
    final_q = np.where(np.isfinite(final_q), final_q, 0.0)
    final_q = np.clip(final_q, 0.0, 1000.0)
    final_list = [float(x) for x in final_q]

    if int(final_q.size) <= 20000:
        try:
            ver = evaluate_sequence(final_list)
        except Exception:
            ver = np.nan
        if not np.isfinite(ver):
            fb = np.ones(8192, dtype=np.float64) / 8192.0
            final_list = [float(x) for x in fb]
            try:
                ver = evaluate_sequence(final_list)
            except Exception:
                ver = np.nan
        if np.isfinite(ver) and ver < best_score:
            best_score = float(ver)

    print("Final answer: length=%d, score=%.7f" % (int(final_q.size), best_score))
    return final_list
```
