"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: 
1. Multi-length exploration (n=8192, 16384, 32768) with upscaled initializations
2. Power-sum smoothing FISTA with adaptive step size and Nesterov restart

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_cross_convolve(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # First, project to non-negative
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s == S:
        return v
    # Need to adjust: find theta such that sum(max(v - theta, 0)) = S
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def power_sum_max(c, p):
    """Approximate max(c) using power sum: (sum c^p)^(1/p)."""
    c_pos = np.maximum(c, 1e-30)
    log_val = np.log(c_pos).sum() / p  # Wait, this isn't right
    # Actually: (sum c_k^p)^(1/p)
    # For numerical stability: = exp(log(sum c_k^p) / p)
    # = exp(max(c) + log(sum exp(p*c_k - p*max(c))) / p)
    cm = c.max()
    if cm < 1e-30:
        return 0.0
    scaled = p * (c - cm)
    # Clip for numerical stability
    scaled = np.maximum(scaled, -500.0)
    log_sum = np.log(np.exp(scaled).sum())  # log(sum exp(p*(c-cm)))
    return cm + log_sum / p

def power_sum_gradient(a, c, p):
    """
    Gradient of (sum c_k^p)^(1/p) with respect to a.
    c = convolve(a, a)
    ∂f/∂a[m] = 2 * sum_k w_k * a[k-m] where w_k = f^(1-p) * c_k^(p-1) / (sum c^p)^(1/p) * ... 
    Actually: f = (sum c^p)^(1/p)
    ∂f/∂c_k = (1/p) * (sum c^p)^(1/p - 1) * p * c_k^(p-1) = (sum c^p)^(1/p - 1) * c_k^(p-1)
    Let f_val = (sum c^p)^(1/p)
    ∂f/∂c_k = f_val^(1-p) * c_k^(p-1)
    Then ∂f/∂a[m] = 2 * sum_{k: 0<=k-m<n} (∂f/∂c_k) * a[k-m]
    """
    n = len(a)
    c = np.maximum(c, 1e-30)
    f_val = power_sum_max(c, p)
    if f_val < 1e-30:
        return np.zeros(n)
    # w_k = f_val^(1-p) * c_k^(p-1)
    log_w = (1 - p) * np.log(f_val) + (p - 1) * np.log(c)
    log_w = np.clip(log_w, -500, 500)
    w = np.exp(log_w)
    # Normalize w to sum to 1 (for numerical stability in the FFT)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    # Compute 2 * cross_correlation(w, a) restricted to [0, n)
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * w_sum * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lse_max(c, beta):
    """LSE smooth max."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    return cm + np.log(np.exp(scaled).sum()) / beta

def lse_gradient(a, c, beta):
    """Gradient of LSE smooth max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def peak_shave_gradient(a, c, k=32):
    """Exact subgradient from top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.maximum(c[idx] - c.mean(), 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista_adaptive(a_init, S, max_time, rng, 
                       p_init=5.0, p_growth=1.05, p_max=100.0,
                       use_lse_after=20.0, beta_init=50.0):
    """
    FISTA with power-sum smoothing transitioning to LSE, 
    adaptive step size with backtracking, and Nesterov restart.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    # Track for Nesterov restart
    prev_y = y.copy()
    
    # Adaptive step size
    L = 2.0 * S * np.max(a)  # Initial Lipschitz estimate
    step = 1.0 / L
    
    consecutive_fails = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        
        # Choose smoothing: power-sum early, LSE later
        if elapsed < use_lse_after:
            p = min(p_max, p_init * (p_growth ** it))
            c = fft_convolve_self(y)
            fval = power_sum_max(c, p)
            grad = power_sum_gradient(y, c, p)
            # Step size heuristic for power-sum
            L = 2.0 * S * (c.max() / fval + 0.5)
        else:
            beta = beta_init * (1.0 + 0.02 * it)
            beta = min(beta, 50000.0)
            c = fft_convolve_self(y)
            fval = lse_max(c, beta)
            grad = lse_gradient(y, c, beta)
            L = 2.0 * beta * 0.5  # Lipschitz of LSE gradient
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Check and update best
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Backtracking line search (simplified)
        step = 1.0 / L
        x_new_raw = y - step * grad
        
        # Project
        x_new = project_nonneg_sum(x_new_raw, S)
        
        # Check if objective improved (using the smooth surrogate)
        c_new = fft_convolve_self(x_new)
        if elapsed < use_lse_after:
            p_cur = min(p_max, p_init * (p_growth ** (it + 1)))
            f_new = power_sum_max(c_new, p_cur)
        else:
            beta_cur = min(50000.0, beta_init * (1.0 + 0.02 * (it + 1)))
            f_new = lse_max(c_new, beta_cur)
        
        if f_new > fval * 1.01 + 1e-10:
            # Step was too large, reduce
            L *= 1.5
            step = 1.0 / L
            x_new = project_nonneg_sum(y - step * grad, S)
            consecutive_fails += 1
        else:
            # Step was OK, maybe increase L slightly
            if consecutive_fails > 5:
                L = max(L / 1.2, 1e-10)
            consecutive_fails = 0
        
        # Nesterov restart
        if np.linalg.norm(y - x_new) >= np.linalg.norm(x_new - prev_y) + 1e-10 * S:
            t = 1.0
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        prev_y = y.copy()
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure y has the right sum (approximately)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t = t_new
        
        # Periodic perturbation to escape local minima
        if it % 5000 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_nonneg_sum(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            prev_y = y.copy()
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def run_peak_polish(a_init, S, max_time, rng):

            t = 1.0

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

        best_a = a.copy()

def run_lp_direction(a, M, max_iter=50):
    """
    Find the LP best-response direction using dual subgradient.
    Returns a direction d such that moving towards d reduces max convolution.
    Uses projected subgradient on the dual.
    """
    n = len(a)
    # Dual: min M * sum(u) s.t. F^T u >= 1, u >= 0
    # Subgradient of the constraint violation: 1 - F^T u (where F^T u = cross-correlation)
    # We want to find u >= 0 with F^T u >= 1 minimizing sum(u)
    
    # Start with u = 0
    u = np.zeros(2 * n - 1)
    
    # Subgradient method on the Lagrangian dual
    # Actually, let's use a simpler approach: 
    # The optimal b satisfies: for the active constraints (where (a*b)_k = M),
    # the gradient of the objective (which is 1^T) is in the span of the active constraint gradients.
    
    # Simpler: use projected gradient on the primal with the constraint (a*b)_k <= M
    # Objective: max sum(b) = min -sum(b)
    # Projection: b >= 0, (a*b)_k <= M for all k
    
    # This projection is hard. Instead, use an iterative approach:
    # b_{k+1} = max(0, b_k + step * (1 - lambda * (a*b_k)_shifted))
    # where lambda is adjusted to maintain the constraint.
    
    # Actually, let me just use a simpler heuristic:
    # Find the "most violated" constraint direction and move against it.
    
    # b that maximizes sum(b) subject to convolve(a,b) <= M
    # This is like: b should put mass where a is small (to keep convolve small)
    # while being non-negative.
    
    # A good heuristic: b ∝ 1/a (where a > 0), clipped and normalized
    # More precisely: b_j = min(1, M / max(a_j, eps)) roughly
    
    # Let me try: b_j = M / (sum_k a_k * a_{k-j} + eps) ... no that's convolve(a,a)
    
    # Actually the constraint is convolve(a, b)_k <= M for all k.
    # convolve(a, b)_k = sum_j a_j * b_{k-j}
    # For a given b, the max over k is what we want to bound.
    
    # Simple heuristic: b_j = M / (a_j + eps) (inverse of a)
    # This makes convolve(a, b) ≈ M * sum(1/a_j * a_j) ... not quite.
    
    # Let me just use: b = M / (a + eps) renormalized
    eps = 1e-10
    b = M / (a + eps)
    b = np.maximum(b, 0)
    b_sum = b.sum()
    if b_sum > 0:
        b = b / b_sum
    else:
        b = np.ones(n) / n
    
    return b

def ups

    """Upscale by interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)

def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)
