import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rhat = rho.max()
    theta = (cssv[rhat] - S) / (rhat + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    e_sum = e.sum()
    lse_val = c_max + np.log(e_sum) / beta
    w = e / e_sum
    # Gradient: (2/beta) * cross-correlation(w, a)
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n] / beta
    return grad, lse_val, c


def peak_subgradient(a, c, k=1):
    """Subgradient from the top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above mean
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (2 * n).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    a = a_init.copy()
    n = len(a)
    S = a.sum()
    a = project_simplex(a, S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        # Adaptive beta
        progress = t / max(t, 1)
        # Use log-linear schedule

        # Simpler: linear in log-space over the duration

        grad, lse_val, c = lse_objective_and_grad(a, beta)
        
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            last_improve = t
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        gnorm = np.sqrt(np.mean(v_hat)) + eps
        lr = lr_base / (1.0 + 0.1 * frac)

        a = project_simplex(a_new, S)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Diverse initialization with quick exploration
2. Adam optimizer with LSE-smoothed max-convolution objective (increasing sharpness)
3. Peak shaving with exact subgradient on dominant lags
4. High-beta fine-tuning with very small steps
5. Multi-length exploration for safety
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, S / n)
    rhat = idx.max()
    theta = (cssv[rhat] - S) / (rhat + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad_val(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient.
    Returns (gradient, lse_value, convolution)."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    e_sum = e.sum()
    lse_val = c_max + np.log(e_sum) / beta
    w = e / e_sum
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n] / beta
    return grad, lse_val, c


def peak_subgrad(a, c, k=1):
    """Subgradient from top-k lags weighted by excess above mean."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 1e-15)
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (2 * n).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def obj_norm(a):
    """Normalized objective: max(conv) when sum(a) = sqrt(2n)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return c.max() * (2.0 * n) / (S * S)


def adam_lse_phase(a_init, max_time, beta_start, beta_end, lr_base=0.02,
                   beta1=0.9, beta2=0.999, eps=1e-8,
                   restart_interval=500, noise_base=0.015):
    """Adam optimization with LSE-smoothed objective and increasing beta."""
    a = project_simplex(a_init.copy(), a_init.sum())
    n = len(a)
    S = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    last_imp = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < max_time:
        t += 1
        frac = min(1.0, t / max(1, (max_time * 300)))
        # Log-linear beta schedule
        beta = np.exp(np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start)))

        grad, lse_val, c = lse_grad_val(a, beta)

        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            last_imp = t

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive step: scale by typical element magnitude
        elem_scale = S / n
        step = lr_base * elem_scale / (1.0 + 0.5 * frac)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        # Restart if stuck
        if t - last_imp > restart_interval and t > restart_interval * 2:
            ns = noise_base * elem_scale * (1.0 - frac * 0.5)
            if ns > 1e-12:
                a = project_simplex(a + rng.normal(0, ns, n), S)
                m = np.zeros(n)
                v = np.zeros(n)
                last_imp = t

    return best_a, best


def peak_shave_phase(a_init, max_time, k_start=1, k_end=12):
    """Exact subgradient peak shaving with decreasing step."""
    a = project_simplex(a_init.copy(), a_init.sum())
    n = len(a)
    S = a.sum()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    t = 0
    elem_scale = S / n

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        frac = min(1.0, (time.monotonic() - start) / max_time)
        k = int(k_start + (k_end - k_start) * frac)

        grad = peak_subgrad(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step
        step = elem_scale * 0.05 / (1.0 + 0.02 * t + 3.0 * frac)
        a = project_simplex(a - step * grad, S)

    return best_a, best


def fine_tune_phase(a_init, max_time, rng):
    """Very fine-tuning: alternating between high-beta LSE and exact peak shave."""
    a = project_simplex(a_init.copy(), a_init.sum())
    n = len(a)
    S = a.sum()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    elem_scale = S / n

    while time.monotonic() - start < max_time:
        t += 1
        frac = min(1.0, t / max(1, max_time * 200))

        # Alternate: even steps use LSE, odd steps use peak subgrad
        if t % 3 == 0:
            c = fft_convolve_self(a)
            grad = peak_subgrad(a, c, k=2)
        else:
            beta = 3000.0 + 7000.0 * frac
            grad, _, _ = lse_grad_val(a, beta)

        # Simple momentum (not full Adam for speed at this stage)
        m = 0.85 * m + 0.15 * (grad - 0.85 * m)
        v = 0.99 * v + 0.01 * grad * grad

        gnorm = np.sqrt(np.mean(v)) + 1e-12
        step = elem_scale * 0.005 * (1.0 - 0.5 * frac)
        a = project_simplex(a - step * m / gnorm, S)

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Tiny perturbation every 500 steps
        if t % 500 == 0:
            ns = elem_scale * 0.001 * (1.0 - frac)
            if ns > 1e-14:
                a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best


def make_inits(n, rng, base=None):
    """Generate diverse initializations for length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base is not None and len(base) == n:
        inits.append(base.copy())

    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)), 0.01))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.04))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt + 0.3) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.05)
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    return [np.maximum(x, 0.01) for x in inits]


def resample_to_length(a, n_target):
    """Resample sequence to target length via interpolation or averaging."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    if n_src > n_target:
        ratio = n_src // n_target
        if ratio > 0:
            trunc = a[:n_target * ratio].reshape(n_target, ratio).mean(axis=1)
            return trunc
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)
