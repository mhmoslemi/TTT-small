                    update_c_after_coord_change(c, a, j, a[j], t_opt, n)
                    a[j] = t_opt

                    new_max = c.max()
                    if new_max < best_cmax - 1e-13:
                        best_cmax = new_max
                        best_a = a.copy()
                        improved_this_round = True

        # Rescale
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n

        c_max = c.max()

        if not improved_this_round:
            stuck_rounds += 1
            if stuck_rounds >= perturb_interval:
                # Perturb to escape
                sigma = 0.005 / np.sqrt(n)
                a = a + rng.normal(0, sigma, n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                c = fft_convolve_self(a)
                c_max = c.max()
                stuck_rounds = 0

    return best_a, best_cmax


def mirror_descent_phase(a_init, max_time, rng, n_bisect=45):
    """
    Mirror descent with multiplicative updates for global exploration.
    Uses soft-max gradient with annealed temperature.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = float(n)
    a = a / s

    c = fft_convolve_self(a)
    best_a = a.copy()
    best_cmax = c.max()

    start = time.monotonic()
    tau = 5.0  # temperature for soft-max
    eta = 0.1  # step size

    N_fft = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        # Compute soft-max weights
        c_shifted = c - c.max()
        weights = np.exp(c_shifted / tau)
        weights /= weights.sum()

        # Compute gradient via cross-correlation with weights
        # grad[i] ~ sum_k weights[k] * a[k-i] (the "adjoint" of convolution)
        w_padded = np.zeros(N_fft)
        w_padded[:2 * n - 1] = weights
        W = fft(w_padded)
        A = fft(a, N_fft)
        grad = np.real(ifft(W * np.conj(A)))[:n]

        # Add diagonal correction: for each k, extra a[k/2] if k is even
        # (approximate: just scale grad slightly)
        grad *= 1.5  # Rough correction for the diagonal terms

        # Mirror descent update: a_new = a * exp(-eta * grad) / Z
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        a_new = a * np.exp(-eta * grad)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a

        # Check improvement
        c_check = fft_convolve_self(np.maximum(a_new, 0.0))
        if c_check.max() < best_cmax:
            best_cmax = c_check.max()
            best_a = np.maximum(a_new, 0.0).copy()
            a = best_a.copy()
        else:
            a = np.maximum(a_new, 0.0)

        # Anneal
        tau *= 0.97
        eta *= 0.995

        # Occasional larger perturbation
        if rng.random() < 0.02:
            a = a + rng.normal(0, 0.002 / np.sqrt(n), n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_cmax


def spectral_perturb(a, sigma, rng):
    """Add random spectral noise, project to non-negative, normalize."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_initialization(n, rng, method="uniform"):
    """Create diverse initializations."""
    t = np.arange(n) / n
    if method == "uniform":
        a = np.ones(n)
    elif method == "hann":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 0.01)
    elif method == "sin3":
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * t) + 0.1 * np.sin(2.0 * np.pi * 7.0 * t)
        a = np.maximum(a, 0.01)
    elif method == "smooth_rand":
        a = rng.random(n)
        # Low-pass filter via FFT
        N = 1 << (n - 1).bit_length()
        A = fft(a, N)
        cutoff = N // 8
        mask = np.ones(N)
        mask[cutoff:-cutoff] = 0.0
        A *= mask
        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.ones(n) / n
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.3, 0.7)
    elif method == "bimodal":
        a = 1.0 + 0.4 * np.exp(-((t - 0.3) / 0.1) ** 2) + 0.4 * np.exp(-((t - 0.7) / 0.1) ** 2)
    elif method == "given":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64).flatten()
            a = np.clip(a, 0, 1000)
            if len(a) != n:
                # Resample
                old_n = len(a)
                new_t = np.linspace(0, old_n - 1, n)
                a = np.interp(new_t, np.arange(old_n), a)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n)
            else:
                a = a / s
        except Exception:
            a = np.ones(n)
    else:
        a = 0.5 + 0.5 * rng.random(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def safe_normalize_for_eval(a, n):
    """Normalize a for the evaluate_sequence call."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(256, dtype=np.float64)
    # Scale so that sum is sqrt(2n) (gives objective = max_b directly)
    target = np.sqrt(2.0 * float(n))
    a = a * (target / s)
    a = np.clip(a, 0.0, 1000.0)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses exact coordinate descent with bisection, mirror descent exploration,
    and multi-start diversification.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))

    print("n=%d, S=%.4f, budget=%.0fs" % (n, S, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = np.maximum(np.asarray(a_candidate, dtype=np.float64), 0.0)
        s = a_candidate.sum()
        if s < 1e-15:
            return False
        a_norm = a_candidate / s
        c = fft_convolve_self(a_norm)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_candidate.copy()
            print("  NEW BEST: %.6f" % global_best)
            return True
        return False

    # === Build initializations ===
    methods = ["given", "uniform", "sin3", "smooth_rand", "bimodal", "alternating"]
    inits = []
    for m in methods:
        inits.append((m, make_initialization(n, rng, m)))

    # Add a few random smooth ones
    for i in range(3):
        inits.append(("rand%d" % i, make_initialization(n, rng, "smooth_rand")))

    print("  %d initializations" % len(inits))

    # === Phase 1: Quick evaluation and short CD from each init (30% time) ===
    print("=== Phase 1: Multi-start CD ===")
    t0 = time.monotonic()
    phase1_budget = (deadline - t0) * 0.30
    per_init = phase1_budget / len(inits)

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        if not update_best(arr):
            pass  # Still try to improve from it

        time_budget = min(per_init, 30.0)
        a_cd, sc_cd = exact_coord_descent(arr, time_budget, rng, n_coords=100)
        update_best(a_cd)

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    if global_a is None:
        global_a = np.ones(n)
        global_best = fft_convolve_self(global_a / global_a.sum()).max()

    # === Phase 2: Alternating CD and mirror descent (45% time) ===
    print("=== Phase 2: Alternating optimization ===")
    t1 = time.monotonic()
    phase2_budget = (deadline - t1) * 0.45
    a_curr = global_a.copy()
    stall = 0

    while time.monotonic() < t1 + phase2_budget:
        rem = (t1 + phase2_budget) - time.monotonic()
        if rem < 10:
            break

        # CD burst
        cd_budget = min(rem * 0.5, 60.0)
        a_cd, sc_cd = exact_coord_descent(a_curr, cd_budget, rng, n_coords=150)
        if update_best(a_cd):
            a_curr = a_cd.copy()
            stall = 0
        else:
            stall += 1

        # Mirror descent burst
        rem = (t1 + phase2_budget) - time.monotonic()
        if rem > 10:
            md_budget = min(rem * 0.3, 40.0)
            a_md, sc_md = mirror_descent_phase(a_curr, md_budget, rng)
            if update_best(a_md):
                a_curr = a_md.copy()
                stall = 0
            else:
                stall += 1

        # Random perturbation when stuck
        rem = (t1 + phase2_budget) - time.monotonic()
        if rem > 10 and stall >= 4:
            for sigma in [0.02, 0.05, 0.1, 0.2]:
                a_pert = spectral_perturb(a_curr, sigma, rng)
                if update_best(a_pert):
                    a_curr = a_pert.copy()
                    stall = 0
                    break
                # Short CD from perturbed
                a_cd2, _ = exact_coord_descent(a_pert, min(rem * 0.1, 10.0), rng, n_coords=50)
                if update_best(a_cd2):
                    a_curr = a_cd2.copy()
                    stall = 0
                    break
            stall = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # === Phase 3: Long focused CD on best (20% time) ===
    print("=== Phase 3: Long CD polish ===")
    t2 = time.monotonic()
    phase3_budget = (deadline - t2) * 0.20

    # Gradually increase n_coords for finer optimization
    segments = [
        (0.4, 100),
        (0.3, 200),
        (0.3, 300),
    ]
    for frac, nc in segments:
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        seg_budget = min(phase3_budget * frac, rem - 5)
        a_cd, _ = exact_coord_descent(global_a, seg_budget, rng, n_coords=nc)
        update_best(a_cd)

    print("=== Phase 3 done. Best: %.6f ===" % global_best)

    # === Phase 4: Final fine-tune (5% time) ===
    print("=== Phase 4: Final polish ===")
    rem = deadline - time.monotonic()
    if rem > 5:
        a_final, _ = exact_coord_descent(global_a, rem, rng, n_coords=500, n_bisect=50)
        update_best(a_final)

    # === Verification ===
    if global_a is not None:
        eval_a = safe_normalize_for_eval(global_a, len(global_a))
        final_list = [float(x) for x in eval_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [1.0] * 256
    else:
        final_list = [1.0] * 256

    print("Returning sequence of length %d, internal score=%.6f" % (len(final_list), global_best))
    return final_list
