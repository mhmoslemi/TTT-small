def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    m = 1 << (2 * n - 1).bit_length()
    A = fft(a, m)
    return np.real(ifft(A * A))[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    return v * (S / s)


def lse_and_grad(a, c, beta):
    """Compute LSE value and its gradient."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clamp to avoid overflow
    max_scaled = np.max(scaled)
    if max_scaled > 500:
        scaled = scaled - max_scaled + 500  # shift to keep max at 500
        # Actually just recompute
        scaled = beta * (c - c_max)
        scale = 500.0 / max(1e-30, scaled.max())
        if scale < beta:
            beta = scale
            scaled = beta * (c - c_max)
    
    e = np.exp(scaled - scaled.max())
    w = e / e.sum()
    lse_val = c_max + np.log(e.sum()) / beta
    
    m = 1 << (2 * n - 1).bit_length()
    W = fft(w, m)
    A = fft(a, m)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return lse_val, grad, beta

    a = project_simplex(a_init.copy(), S)

    best = fft_convolve_self(a).max()
    best_a = a.copy()

    total_iters = 0

        if time.monotonic() - start > max_time:
            break

            noise_scale = noise_decay ** restart * S / np.sqrt(n) * 0.1
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
            best_cur = fft_convolve_self(a).max()
            if best_cur < best:
                best = best_cur
                best_a = a.copy()
        
        restart_deadline = min(start + max_time, start + max_time * (restart + 1) / (n_restarts + 1))
        # Actually, let's just run until time runs out, with beta increasing

            # Adaptive beta: exponential increase

"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 over non-negative sequences.
Uses FISTA with adaptive LSE, excess minimization, and direct peak shaving.
Employs m-sequence-based initializations for good spectral structure.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    m = 1 << (2 * n - 1).bit_length()
    A = fft(a, m)
    return np.real(ifft(A * A))[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    return v * (S / s)


def lse_gradient_fast(a, c, beta):
    """Gradient of LSE(beta, c) where c = convolve(a,a). Returns (lse_val, grad, beta_used)."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    if scaled.max() > 400.0:
        beta = 400.0 / max(1e-30, scaled.max())
        scaled = beta * (c - c_max)
    shifted = scaled - scaled.max()
    e = np.exp(shifted)
    w = e / e.sum()
    lse_val = c_max + (shifted.max() + np.log(e.sum())) / beta

    m = 1 << (2 * n - 1).bit_length()
    W = fft(w, m)
    A = fft(a, m)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, beta


def excess_gradient(a, c, T, power=2.0):
    """Gradient of sum((c_k - T)^power_+). Returns (grad, objective_value)."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if power == 1.0:
        w = excess.copy()
    else:
        w = power * excess * (excess ** (power - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), 0.0
    obj = np.sum(excess ** power)
    m = 1 << (2 * n - 1).bit_length()
    W = fft(w, m)
    A = fft(a, m)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, obj


def peak_shave_gradient(a, c, k_peaks=8):
    """Subgradient from top-k peaks, weighted by excess above mean."""
    n = len(a)
    k = min(k_peaks, len(c))
    idx = np.argpartition(c, -k)[-k:]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    weights = weights / w_sum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    m = 1 << (2 * n - 1).bit_length()
    W = fft(w_full, m)
    A = fft(a, m)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def generate_m_sequence(length_order):
    """Generate a binary m-sequence using a primitive polynomial LFSR."""
    # Primitive polynomials for various lengths (from 2^m - 1)
    # x^m + ... + 1 (represented as list of tap positions)
    taps_dict = {
        4: [4, 3],
        5: [5, 3],
        6: [6, 5],
        7: [7, 6],
        8: [8, 6],
        9: [9, 5],
        10: [10, 7],
        11: [11, 9],
        12: [12, 6],
        13: [13, 12],
        14: [14, 12],
        15: [15, 14],
    }
    m = length_order
    if m not in taps_dict:
        # Fall back to a random binary sequence
        return np.array([1.0 if (i % 2 == 0) else 0.0 for i in range((1 << m) - 1)])
    
    taps = taps_dict[m]
    # Initialize LFSR state
    state = [1] * m
    seq = []
    out_length = (1 << m) - 1
    
    for _ in range(out_length):
        seq.append(state[0])
        # Compute feedback
        feedback = 0
        for t in taps:
            feedback ^= state[t - 1]
        # Shift right
        state = [feedback] + state[:-1]
    
    return np.array(seq, dtype=np.float64)

        # Convert to smooth values

            inits.append(np.maximum(base * alpha + (1.0 - alpha) * 0.5, 0.01))

def run_fista_lse(a_init, n, S, max_time, rng,
                  beta_start=30.0, beta_end=8000.0,
                  n_restarts=4, noise_scale_init=0.05):
    """FISTA with adaptive LSE smoothing."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    best_it = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        progress = min(1.0, it / max(1000, int(max_time / 0.005)))
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.8)
        beta = min(beta, beta_end)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1
        
        lse_val, grad, beta_used = lse_gradient_fast(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Step size estimation
        # L ~ 2 * beta * max |c| (rough Lipschitz bound for LSE gradient)
        step = S / (2.0 * beta_used * max(1.0, c.max()))
        step = min(step, 1.0 / max(1e-10, step * 0.1))
        step = max(step, 1e-8)
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_k_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_k_new
        
        # Restart if stuck
        if stall_count > 2000:
            noise = noise_scale_init * (0.7 ** (stall_count // 2000 - 1)) * S / np.sqrt(n)
            if noise > 1e-8:
                y = project_simplex(y + rng.normal(0, noise, n), S)
                a = y.copy()
                t_k = 1.0
                stall_count = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_excess_min(a_init, n, S, max_time, rng):
    """Minimize excess above a decreasing threshold."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    T_start = c0.max()
    T_end = c0.mean() + 0.02 * (c0.max() - c0.mean())
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Anneal threshold
        T = T_start + (T_end - T_start) * (progress ** 0.6)
        T = max(T, T_end)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        if cur_max <= T * (1 + 1e-8):
            T_eff = max(c.mean(), cur_max - 0.005 * (cur_max - c.mean() + 1e-10))
        else:
            T_eff = T
        
        grad, obj_excess = excess_gradient(y, c, T_eff, power=2.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Step size
        max_w = max(T_eff, 1e-10)
        step = 1.0 / (4.0 * max_w * max(c.max(), 1e-10))
        step = min(step, S / (n * 0.05))
        step = max(step, 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        t_k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_k_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_k_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_peak_shave(a_init, n, S, max_time, rng):
    """Direct peak shaving with adaptive number of peaks."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k_peaks = int(4 + 20 * progress)
        k_peaks = min(k_peaks, n // 8)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        grad = peak_shave_gradient(y, c, k_peaks=k_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        step = min(step, 0.1)
        
        x_new = project_simplex(y - step * grad, S)
        t_k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_k_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_k_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_direct_reduction(a_init, n, S, max_time, rng):
    """Directly reduce top peaks by solving small subproblems."""
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find top peaks
        n_peaks = min(16, n // 16)
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        peak_vals = c[peak_idx]
        
        # For each peak at position k, the contribution to a[i] is proportional to a[k-i]
        # Build a "pressure" vector
        pressure = np.zeros(n)
        for k in peak_idx:
            weight = peak_vals[list(peak_idx).index(k)] - c.mean()
            if weight > 0:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    if 0 <= k - i < n:
                        pressure[i] += weight * a[k - i]
        
        if pressure.max() < 1e-15:
            break
        
        # Apply pressure (reduce a where it contributes to peaks)
        eps = 0.01 * S / (n * (1.0 + 0.005 * it))
        a_new = a - eps * pressure
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c.max():
            a = a_new
        else:
            # Try smaller step
            eps *= 0.3
            a_new = project_simplex(a - eps * pressure, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c.max():
                a = a_new
            else:
                # Add noise to escape
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=2380):
    """
    Search for non-negative sequence minimizing 2n*max(convolve(a,a))/sum(a)^2.
    Multi-phase: FISTA-LSE -> excess minimization -> peak shaving -> direct reduction.
    Multi-length exploration with m-sequence initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print(f"Using given sequence of length {len(given)}")
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    # Lengths to try
    lengths = [8192, 4096, 16384]
    if given is not None and len(given) != 8192:
        lengths.insert(0, len(given))
        if len(given) > 8192:
            lengths.append(8192)
    
    # Deduplicate and sort by priority
    lengths = list(dict.fromkeys(lengths))
    
    print(f"Trying lengths: {lengths}")
    print(f"Total budget: {budget_s:.0f}s")
    
    # Allocate time: give more time to n=8192 (proven length)
    time_weights = {}
    for n in lengths:
        if n == 8192:
            time_weights[n] = 0.45
        elif n == 4096:
            time_weights[n] = 0.25
        elif n == 16384:
            time_weights[n] = 0.20
        else:
            time_weights[n] = max(0.1, 0.3 / max(1, len(lengths)))
    total_weight = sum(time_weights.values())
    for n in time_weights:
        time_weights[n] = time_weights[n] / total_weight * (budget_s - 120)
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        t_alloc = min(time_weights.get(n, 200), deadline - time.monotonic() - 30)
        if t_alloc < 30:
            continue
        
        print(f"\n{'='*50}")
        print(f"Optimizing n={n}, allocated {t_alloc:.0f}s")
        print(f"{'='*50}")
        
        S = np.sqrt(2.0 * n)
        
        # Generate initializations
        inits = generate_inits(n, rng, given)
        max_init_count = min(len(inits), max(4, int(t_alloc / 20)))
        inits = inits[:max_init_count]
        t_per_init = t_alloc / (max_init_count + 2)  # +2 for polishing phases
        
        best_a_n = None
        best_val_n = np.inf
        
        for i, init in enumerate(inits):
            t_remaining = deadline - time.monotonic() - 30
            if t_remaining < 20:
                break
            t_init = min(t_per_init, t_remaining * 0.6)
            if t_init < 10:
                break
            
            # Choose method based on init index
            method = ["fista", "excess", "fista", "peak"][i % 4]
            
            if method == "fista":
                a_opt, val = run_fista_lse(init, len(init), S, t_init, rng,
                                            beta_start=30.0, beta_end=8000.0,
                                            n_restarts=3, noise_scale_init=0.04)
            elif method == "excess":
                a_opt, val = run_excess_min(init, len(init), S, t_init, rng)
            else:
                a_opt, val = run_peak_shave(init, len(init), S, t_init, rng)
            
            val_check = fft_convolve_self(a_opt).max()
            if val_check < best_val_n:
                best_val_n = val_check
                best_a_n = a_opt.copy()
                obj = 2.0 * len(a_opt) * val_check / (a_opt.sum() ** 2)
                print(f"  Init {i} ({method}): obj={obj:.6f}, max_b={val_check:.6f}")
        
        if best_a_n is None:
            best_a_n = np.ones(n)
            best_val_n = fft_convolve_self(best_a_n).max()
        
        # Polishing phases
        n_best = len(best_a_n)
        S_best = np.sqrt(2.0 * n_best)
        t_remaining = deadline - time.monotonic() - 30
        
        if t_remaining > 30:
            # Phase: High-beta FISTA
            t_pol1 = min(t_remaining * 0.3, 120)
            a_p1, val_p1 = run_fista_lse(best_a_n, n_best, S_best, t_pol1, rng,
                                          beta_start=200.0, beta_end=15000.0,
                                          n_restarts=2, noise_scale_init=0.02)
            if val_p1 < best_val_n:
                best_val_n = val_p1
                best_a_n = a_p1.copy()
                obj = 2.0 * n_best * best_val_n / (best_a_n.sum() ** 2)
                print(f"  After high-beta FISTA: obj={obj:.6f}")
        
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining > 20:
            # Phase: Excess minimization
            t_pol2 = min(t_remaining * 0.3, 100)
            a_p2, val_p2 = run_excess_min(best_a_n, n_best, S_best, t_pol2, rng)
            if val_p2 < best_val_n:
                best_val_n = val_p2
                best_a_n = a_p2.copy()
                obj = 2.0 * n_best * best_val_n / (best_a_n.sum() ** 2)
                print(f"  After excess min: obj={obj:.6f}")
        
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining > 15:
            # Phase: Peak shaving
            t_pol3 = min(t_remaining * 0.3, 80)
            a_p3, val_p3 = run_peak_shave(best_a_n, n_best, S_best, t_pol3, rng)
            if val_p3 < best_val_n:
                best_val_n = val_p3
                best_a_n = a_p3.copy()
                obj = 2.0 * n_best * best_val_n / (best_a_n.sum() ** 2)
                print(f"  After peak shave: obj={obj:.6f}")
        
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining > 10:
            # Phase: Direct reduction
            t_pol4 = min(t_remaining * 0.4, 60)
            a_p4, val_p4 = run_direct_reduction(best_a_n, n_best, S_best, t_pol4, rng)
            if val_p4 < best_val_n:
                best_val_n = val_p4
                best_a_n = a_p4.copy()
                obj = 2.0 * n_best * best_val_n / (best_a_n.sum() ** 2)
                print(f"  After direct reduction: obj={obj:.6f}")
        
        # Update global best
        n_final = len(best_a_n)
        obj_final = 2.0 * n_final * best_val_n / (best_a_n.sum() ** 2)
        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_n.copy()
            print(f"  *** New global best: obj={global_best:.6f} (n={n_final}) ***")
    
    # Final intensive polishing
    if global_a is not None:
        remaining = deadline - time.monotonic() - 10
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        
        if remaining > 20:
            print(f"\n=== Final polishing ({remaining:.0f}s remaining) ===")
            
            # Very high beta FISTA for peak concentration
            seg = remaining * 0.4
            a_fin, val_fin = run_fista_lse(global_a, n_final, S_final, seg, rng,
                                            beta_start=500.0, beta_end=20000.0,
                                            n_restarts=2, noise_scale_init=0.01)
            if val_fin < fft_convolve_self(global_a).max():
                global_a = a_fin.copy()
                global_best = 2.0 * n_final * val_fin / (global_a.sum() ** 2)
                print(f"  After final FISTA: obj={global_best:.6f}")
            
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                a_ps, val_ps = run_peak_shave(global_a, n_final, S_final, remaining * 0.5, rng)
                if val_ps < fft_convolve_self(global_a).max():
                    global_a = a_ps.copy()
                    global_best = 2.0 * n_final * val_ps / (global_a.sum() ** 2)
                    print(f"  After final peak shave: obj={global_best:.6f}")
    
    # Verify and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(8192)
        
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print(f"\nVerification via evaluate_sequence: {ver:.6f}")
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected! Using fallback.")
                global_a = np.ones(8192)
                final_list = [float(x) for x in global_a]
        except Exception as e:
            print(f"  Verification error: {e}")
    else:
        # Fallback: uniform
        global_a = np.ones(8192)
        final_list = [float(x) for x in global_a]
        print("Using fallback uniform sequence!")
    
    print(f"Returning sequence of length {len(global_a)}")
    return final_list
