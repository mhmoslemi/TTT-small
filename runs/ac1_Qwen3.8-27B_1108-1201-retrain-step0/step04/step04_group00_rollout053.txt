                noise = noise_scale * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck = 0
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def run_lp_alternating(a_init, max_time, rng, top_k=50):
    """
    LP-based alternating minimization.
    Fix a, find best g that minimizes max(a*g) subject to sum(g)=S, g>=0.
    Blend and repeat.
    """
    from scipy.optimize import linprog
    
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = a.copy()
        
        # Find active lags
        max_val = c.max()
        active = np.where(c >= max_val * 0.999)[0]
        if len(active) > top_k:
            # Keep top k
            top_idx = np.argsort(c)[::-1][:top_k]
            active = top_idx
        elif len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Build LP: min t s.t. (a*g)[k] <= t for k in active, sum(g)=S, g>=0
        # Variables: [g_0, ..., g_{n-1}, t]
        # (a*g)[k] = sum_i a[i] * g[k-i] (with g[j]=0 for j not in [0,n-1])
        
        # c_mat[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # We need c_mat[k, j] <= t, i.e., c_mat[k, j] - t <= 0
        # So A_ub row: [c_mat[k, :], -1] with b_ub = 0
        
        # For efficiency, only use active constraints
        A_rows = []
        for k in active:
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            row = np.append(row, -1.0)
            A_rows.append(row)
        
        A_ub = np.array(A_rows)
        b_ub = np.zeros(len(active))
        
        # Equality: sum(g) = S
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        
        # Objective: minimize t (last variable)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        bounds = [(0, None)] * n + [(0, None)]
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            g = g * S / g.sum()
            
            # Blend
            alpha = max(0.01, 0.1 / (1 + it * 0.01))
            a = (1 - alpha) * a + alpha * g
            a = np.maximum(a, 0)
            a = a * S / a.sum()
        else:
            # Fall back to subgradient step
            grad = subgradient_topk(a, c, len(active))
            step = S / (np.linalg.norm(grad) + 1e-10)
            a = project_simplex(a - step * grad, S)
    
    return best_a, best_conv

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Combines LSE-FISTA, LP-based alternating minimization, multi-scale exploration,
structured noise perturbations, and adaptive subgradient polishing.
Key new elements: LP directional search for escaping plateaus, multi-restart

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(y, c, k):
    """Subgradient using top-k lags."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

                    freq = noise_freq
                else:
                    freq = rng.integers(1, max(2, n // 16))
                phase = rng.uniform(0, 2 * np.pi)
                noise = noise_scale * np.sin(2 * np.pi * freq * t + phase)
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lp_direction(a_init, max_time, rng, top_k=30, blend_rate=0.15):
    """
    LP-based alternating: fix a, find g minimizing max(a*g) s.t. sum(g)=S, g>=0.
    Blend a toward g. Repeat.
    """
    from scipy.optimize import linprog

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    fail_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-13:
            best_conv = cur_max
            best_a = a.copy()
            fail_count = 0

        # Find active lags (top-k by value)
        idx = np.argsort(c)[::-1][:top_k]
        active = idx

        # Build LP: min t s.t. (a*g)[k] <= t for k in active, sum(g)=S, g>=0
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        A_rows = []
        for k in active:
            row = np.zeros(n + 1)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                row[j] = a[k - j]
            row[n] = -1.0  # coefficient of t
            A_rows.append(row)

        A_ub = np.array(A_rows)
        b_ub = np.zeros(len(active))

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        bounds = [(0, None)] * n + [(0, None)]

        t0 = time.monotonic()
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        solve_time = time.monotonic() - t0

        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-10:
                g = np.ones(n) * S / n
            else:
                g = g * S / gs
            # Check improvement
            c_test = fft_convolve_self(0.5 * a + 0.5 * g)
            if c_test.max() < cur_max * 1.01:  # accept if not much worse
                alpha = blend_rate / (1.0 + 0.05 * fail_count)
                a = (1.0 - alpha) * a + alpha * g
                a = np.maximum(a, 0)
                a = a * S / a.sum()
                fail_count = 0
            else:
                fail_count += 1
                # Try smaller step
                alpha = 0.05
                a2 = (1.0 - alpha) * a + alpha * g
                a2 = np.maximum(a2, 0)
                a2 = a2 * S / a2.sum()
                c2 = fft_convolve_self(a2)
                if c2.max() < cur_max:
                    a = a2
                    fail_count = 0
                else:
                    a = a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a, S)
        else:
            fail_count += 1
            # Fallback: subgradient step
            grad = subgradient_topk(a, c, top_k)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                step = S / (gnorm * (1.0 + 0.02 * it))
                a = project_simplex(a - step * grad, S)

        # If LP is too slow, stop
        if solve_time > max_time * 0.3:
            break

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_subgradient_polish(a_init, max_time, rng):
    """Intensive subgradient polishing with adaptive top-k and momentum."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = min(8, 2 + stuck // 100)
        elif stuck < 5000:
            k = min(32, 8 + stuck // 300)
        else:
            k = min(128, 32 + stuck // 200)

        grad = subgradient_topk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-like step with decay
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, S / (0.2 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Escape mechanism: structured perturbation
        if stuck > 4000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            if noise_scale > 1e-9:
                t = np.linspace(0, 1, n)
                freq1 = rng.integers(1, max(2, n // 8))
                freq2 = rng.integers(1, max(2, n // 8))
                noise = noise_scale * (np.sin(2 * np.pi * freq1 * t) +
                                       0.5 * np.cos(2 * np.pi * freq2 * t))
                y = project_simplex(y + noise, S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: exploration, FISTA, LP alternating, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 35.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 0: Quick multi-restart exploration (10% time) ===
    remaining = deadline - time.monotonic()
    p0 = remaining * 0.10
    print("\n=== Phase 0: Multi-restart exploration (%.0fs) ===" % p0)

    # Try different initializations
    inits = [global_best_a.copy()]

    # Raised cosine with different parameters
    n_try = n0
    t = np.linspace(0, 1, n_try, endpoint=False)
    # Generalized raised cosine
    for p_val in [0.8, 1.2, 1.5, 2.0]:
        a_rc = np.maximum(0, (1 + np.cos(2 * np.pi * t)) ** p_val)
        if a_rc.sum() > 0:
            inits.append(a_rc)

    # Tapered versions
    a_taper = a_start.copy()
    edge = max(1, n_try // 16)
    for frac in [0.3, 0.5, 0.7]:
        a_t = a.start.copy() if hasattr(a, 'start') else a_start.copy()
        ramp = np.linspace(0, 1, edge)
        a_t[:edge] *= ramp
        a_t[-edge:] *= ramp[::-1]
        inits.append(a_t)

    time_per_init = p0 / len(inits)
    for i, a_init in enumerate(inits):
        if time.monotonic() >= deadline:
            break
        a_init = np.asarray(a_init, dtype=np.float64)
        n_init = len(a_init)
        S_init = np.sqrt(2.0 * n_init)
        a_init = a_init * S_init / a_init.sum()

        budget_i = min(time_per_init, deadline - time.monotonic())
        if budget_i < 2:
            break

        a_res, _ = run_fista_lse(a_init, budget_i, rng,
                                  beta_init=5.0, beta_rate=0.01, beta_max=2000.0)
        obj_res = compute_obj(a_res)
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()
            print("  Init %d: obj=%.6f (improved!)" % (i, obj_res))
        else:
            print("  Init %d: obj=%.6f" % (i, obj_res))

    # === Phase 1: Main FISTA at best size (35% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.35
    n_best = len(global_best_a)
    print("\n=== Phase 1: Main FISTA at n=%d (%.0fs) ===" % (n_best, p1))

    # Run with increasing beta
    a1, _ = run_fista_lse(global_best_a, p1 * 0.6, rng,
                           beta_init=10.0, beta_rate=0.006, beta_max=10000.0)
    obj1 = compute_obj(a1)
    print("  FISTA main: obj=%.6f" % obj1)
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # Continue with higher beta
    remaining1 = deadline - time.monotonic()
    if remaining1 > 10:
        a1b, _ = run_fista_lse(global_best_a, min(remaining1, p1 * 0.4), rng,
                                beta_init=50.0, beta_rate=0.004, beta_max=50000.0)
        obj1b = compute_obj(a1b)
        print("  FISTA high-beta: obj=%.6f" % obj1b)
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Upsample and optimize (20% time) ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.20

    if n_best <= 16384 and remaining > 60:

            global_best_a = a_up_res.copy()
            print("  Improved with upsample!")

    # === Phase 3: LP-based alternating (20% time) ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.20
    n_best = len(global_best_a)
    print("\n=== Phase 3: LP alternating at n=%d (%.0fs) ===" % (n_best, p3))

    if n_best <= 16384:
        # Use LP direction finding
        top_k = min(20, n_best // 256)
        a3, _ = run_lp_direction(global_best_a, p3 * 0.6, rng, top_k=top_k)

            global_best_a = a3.copy()

        # Polish after LP
        remaining3 = deadline - time.monotonic()
        if remaining3 > 5:
            a3b, _ = run_fista_lse(global_best_a, min(remaining3, p3 * 0.4), rng,
                                    beta_init=100.0, beta_rate=0.003, beta_max=100000.0)
            obj3b = compute_obj(a3b)
            print("  Post-LP FISTA: obj=%.6f" % obj3b)
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()
    else:
        # Too large for efficient LP, use FISTA
        a3, _ = run_fista_lse(global_best_a, p3, rng,
                               beta_init=100.0, beta_rate=0.003, beta_max=100000.0)
        obj3 = compute_obj(a3)
