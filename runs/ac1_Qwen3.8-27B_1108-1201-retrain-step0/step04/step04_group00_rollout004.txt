Strategy: Multi-method optimization combining:
1. Entropy-regularized mirror descent (multiplicative weights)
2. FISTA with LSE smoothing (as backup)

4. Multi-scale exploration (upsample/downsample)
5. Active-set polishing with exact subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-15, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_init
    eta = eta_init

        it += 1
        step *= step_decay
        eta = min(eta_max, eta_init * np.exp(eta_rate * it))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        grad = lse_gradient(a, c, eta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Scale gradient to unit norm for stable step
        grad = grad / gnorm
        
        # Mirror descent update (multiplicative weights)
        log_a = np.log(a)
        new_log = log_a - step * grad
        new_log -= new_log.max()
        new_a = np.exp(new_log)
        new_a = S * new_a / new_a.sum()
        
        a = new_a
        
        # Noise injection when stuck
        stuck = it - last_improve
        if stuck > 2000 and it % 500 == 0:
            noise_scale = 0.01 * np.exp(-stuck / 5000.0)
            if noise_scale > 1e-5:
                log_a = np.log(a) + rng.normal(0, noise_scale, n)
                log_a -= log_a.max()
                a = S * np.exp(log_a) / np.exp(log_a).sum()
                last_improve = it  # reset stuck counter
    
    return best_a, best_obj


def run_fista_v2(a_init, max_time, rng, mode="lse", momentum=0.9):
    """
    FISTA with constant momentum (instead of Nesterov).
    Different step size strategy: use the gradient norm for adaptive stepping.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Adaptive parameters
    beta = 10.0
    beta_rate = 0.006
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(5000.0, 10.0 * np.exp(0.006 * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1
        
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk_small":
            grad = topk_gradient(y, c, max(2, n // 256), power=2.0)
        elif mode == "topk_large":
            grad = topk_gradient(y, c, min(256, n // 16), power=1.0)
        elif mode == "exact":
            grad = exact_subgradient(y, c, tol_frac=0.01)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step: based on gradient norm and stuck counter
        base_step = S / (gnorm * np.sqrt(it + 1))
        if stuck_count > 1000:
            base_step *= 0.5
        step = min(base_step, S / (0.1 * gnorm))
        
        # FISTA update with constant momentum
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic: try different momentum by restarting
        if it % 10000 == 0:
            t_mom = 1.0
            y = a.copy()
        
        # Noise injection
        if stuck_count > 3000 and it % 1000 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample_zero_insert(a, factor=2):
    """Upsample by inserting zeros."""
    n = len(a)
    n_new = n * factor
    a_new = np.zeros(n_new)
    a_new[::factor] = a
    return a_new


def upsample_interpolate(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample_average(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)
