"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP best-response iterations with full constraint set (or top-k),
multi-scale optimization, and aggressive blending.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def solve_lp_best_response(f, n, S, top_k=None, solver='CBC', time_limit=30.0):
    """
    Solve the LP: min M s.t. (f*g)(k) <= M for all k (or top-k), sum(g) = S, g >= 0.
    Returns (g, M_opt) or (None, None) if failed.
    """
    c_ff = fft_convolve_self(f)
    M_current = c_ff.max()
    
    if top_k is None or top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(c_ff)[::-1][:top_k]
    
    k_constraints = len(idx)
    
    # Build constraint matrix A (k_constraints x n) sparsely
    row_idx = []
    col_idx = []
    data = []
    
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            row_idx.append(i)
            col_idx.append(j)
            data.append(f[k - j])

    g_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == S, g_var >= 0]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except Exception:
                return None, None
    
    if time.monotonic() - start > time_limit:
        return None, None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None, None
        g = g * (S / g_sum)
        M_opt = float(M_var.value) if M_var.value is not None else M_current
        return g, M_opt
    
    return None, None

    # values: f[k - col] for col in range

    cols_start = np.maximum(0, k_vals - n + 1)
    cols_end = np.minimum(k_vals, n - 1)
    
    total_nz = 0
    for i in range(k_constraints):
        total_nz += cols_end[i] - cols_start[i] + 1
    
    row_idx = np.empty(total_nz, dtype=np.int32)
    col_idx = np.empty(total_nz, dtype=np.int32)
    data = np.empty(total_nz, dtype=np.float64)
    
    offset = 0
    for i in range(k_constraints):
        s = cols_start[i]
        e = cols_end[i]
        length = e - s + 1
        row_idx[offset:offset+length] = i
        col_idx[offset:offset+length] = np.arange(s, e+1)
        data[offset:offset+length] = f[k_vals[i] - np.arange(s, e+1)]
        offset += length
    
    return csr_matrix((data, (row_idx, col_idx)), shape=(k_constraints, n))

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Iterated LP best-response with top-k constraints, line search blending,
multi-scale optimization, subgradient polishing, and diverse restarts.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def build_conv_rows(f, n, idx):
    """Build constraint rows for the LP. Returns (row_data, row_starts) for efficient use."""
    k_vals = np.asarray(idx, dtype=np.int64)
    k_c = len(k_vals)
    rows = []
    for i in range(k_c):
        k = int(k_vals[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        vals = f[k - js]
        rows.append((js, vals))
    return rows


def solve_lp_br(f, n, S, top_k=2000, solver='CBC'):
    """Solve LP best response. Returns (g, M_opt) or (None, None)."""
    c_ff = fft_convolve_self(f)
    M_cur = float(c_ff.max())
    
    if top_k is None or top_k >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argsort(c_ff)[::-1][:top_k]
    
    k_c = len(idx)
    
    # Build sparse matrix
    row_list = []
    col_list = []
    data_list = []
    k_vals = np.asarray(idx, dtype=np.int64)
    
    for i in range(k_c):
        k = int(k_vals[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        row_list.append(np.full(len(js), i, dtype=np.int32))
        col_list.append(js.astype(np.int32))
        data_list.append(f[k - js])
    
    row_idx = np.concatenate(row_list)
    col_idx = np.concatenate(col_list)
    data = np.concatenate(data_list)
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(k_c, n))
    
    g_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [
        A @ g_var <= M_var,
        cp.sum(g_var) == S,
        g_var >= 0
    ]
    
    try:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return None, None
    
    if g_var.value is None:
        return None, None
    
    g = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None, None
    g = g * (S / g_sum)
    M_opt = float(M_var.value) if M_var.value is not None else M_cur
    return g, M_opt


def lp_iteration(f, n, S, top_k=2000, solver='CBC', rng=None):
    """
    One LP best-response iteration with line search.
    Returns (f_new, objective_new, objective_old).
    """
    c = fft_convolve_self(f)
    obj_old = float(c.max())
    
    g, M_opt = solve_lp_br(f, n, S, top_k=top_k, solver=solver)
    if g is None:
        return f.copy(), obj_old, obj_old
    
    # Try full replacement first
    c_g = fft_convolve_self(g)
    obj_g = float(c_g.max())
    
    best_obj = obj_old
    best_f = f.copy()
    
    if obj_g < best_obj:
        best_obj = obj_g
        best_f = g.copy()
    
    # Try blends
    for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.8, 0.9]:
        f_test = (1.0 - t) * f + t * g
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        obj_test = float(c_test.max())
        if obj_test < best_obj:
            best_obj = obj_test
            best_f = f_test.copy()
    
    return best_f, best_obj, obj_old

def subgradient_step_simple(f, n, S, lr=0.005, top_k=5, rng=None):
    """Simple subgradient step using average of top-k lag gradients."""
    c = fft_convolve_self(f)
    obj = float(c.max())
    
    # Get top-k lags
    k_vals = np.argsort(c)[::-1][:top_k]
    
    grad = np.zeros(n)
    for k in k_vals:
        lo = max(0, int(k) - n + 1)
        hi = min(int(k), n - 1)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * f[int(k) - j]
    grad /= top_k
    
    # Scale learning rate
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return f.copy(), obj
    
    f_new = project_simplex(f - lr * grad / gnorm * np.sqrt(2 * n), S)
    return f_new, obj
