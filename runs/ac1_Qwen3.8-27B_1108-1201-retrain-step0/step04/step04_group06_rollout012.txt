import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand <= w
    if not np.any(valid):
        return np.full(n, S / n, dtype=np.float64)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, w, fsize):
    """
    Compute gradient given softmax weights w (length 2n-1).
    grad[m] = 2 * sum_j w[m+j] * a[j]
    Uses FFT cross-correlation.
    """
    n = len(a)
    W = fft(w, fsize)
    A = fft(a, fsize)
    R = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * R


def softmax_weights(c, T):
    """Compute softmax weights exp(c/T)/sum(exp(c/T)) with numerical stability."""
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / T)
    return exp_c / exp_c.sum()


def adam_update(a, grad, m, v, t, lr, S, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update with simplex projection."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad ** 2
    m_hat = m / (1.0 - beta1 ** t)
