def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, float)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=float)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

    s = a.sum()
    if s < 1e-30:

"""
Minimize 2*n*max(self_conv)/sum^2 for non-negative sequences.
Multi-phase: Spectral GS (novel) → Adaptive LSE-FISTA → Multi-scale local search → Final polish.
Key innovations: self-convolution spectral method exploiting A^2 (not |A|^2),
annealed LSE scheduling, and block-aware coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def self_conv_fft(a, fsize):
    """Compute full linear self-convolution via FFT. c[k] = sum_i a[i]*a[k-i]."""
    A = fft(a, fsize)
    C = A * A
    c = np.real(ifft(C))
    return c


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using the O(n log n) algorithm."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clip to non-negative and rescale to sum = S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_score(a, n, fsize):
    """Compute the score 2*n*max_c/sum^2."""
    c = self_conv_fft(a, fsize)
    peak = c.max()
    s = a.sum()
    return 2.0 * n * peak / (s * s)


def spectral_gs(a_init, n, S, max_time, rng, alpha_init=0.65, alpha_min=0.15,
                relax=0.7, use_momentum=True):
    """
    Gerchberg-Saxton style iteration for self-convolution flattening.
    Exploits the fact that DFT(self_conv) = A^2 (not |A|^2).
    """
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_sum(a_init, S)
    best_a = a.copy()
    best_peak = np.inf
    start = time.monotonic()
    it = 0
    alpha = alpha_init
    a_prev = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        A = fft(a, fsize)
        C = A * A
        c = np.real(ifft(C))[:2 * n - 1]
        c = np.maximum(c, 0.0)

        peak = c.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        if it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SpecGS it={it}: peak={peak:.10f}, alpha={alpha:.4f}, t={elapsed:.0f}s")

        c_max = c.max()
        if c_max < 1e-30:
            break

        # Flatten via power compression
        c_norm = c / c_max
        c_target = np.power(c_norm, alpha)
        if c_target.sum() > 1e-30:
            c_target = c_target * (c.sum() / c_target.sum())
        else:
            c_target = c.copy()

        # Recover via square root
        c_full = np.zeros(fsize)
        c_full[:2 * n - 1] = c_target
        C_new = fft(c_full, fsize)

        # Add tiny epsilon for numerical stability
        C_new = C_new + 1e-14 * (1.0 + 0.1j)
        A_new = np.sqrt(C_new)
        a_candidate = np.real(ifft(A_new))[:n]

        # Project to non-negative
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s < 1e-15:
            alpha = min(alpha_init + 0.1, 1.0)
            continue
        a_candidate = a_candidate * (S / s)

        # Relaxation and momentum
        if use_momentum:
            a_new = relax * a_candidate + (1.0 - relax) * a_prev
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-30:
                a_new = a_new * (S / s)
            else:
                a_new = a_candidate
        else:
            a_new = a_candidate

        # Check if improvement; if not, try without relaxation
        c_check = self_conv_fft(a_new, fsize)
        if c_check.max() > peak * 1.5:
            # Too much disruption, use pure candidate
            a_new = a_candidate

        a_prev = a
        a = a_new
        alpha = max(alpha_min, alpha * 0.9998)

    # Final evaluation
    c_final = self_conv_fft(a, fsize)
    if c_final.max() < best_peak:
        best_peak = c_final.max()
        best_a = a.copy()

    return best_a, best_peak


def lse_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=3000.0,
              ramp_iters=2000, anneal_period=3000, anneal_factor=0.5):
    """
    FISTA with LSE approximation to max. Features annealing to escape local minima.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_peak = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    current_beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta with annealing
        if it < ramp_iters:
            base_beta = beta_init + (beta_max - beta_init) * (it / ramp_iters)
        else:
            base_beta = beta_max

        # Periodic annealing
        if it > ramp_iters and (it - ramp_iters) % anneal_period < anneal_period * 0.2:
            current_beta = base_beta * anneal_factor
        else:
            current_beta = base_beta

        beta = current_beta

        # Compute convolution of y
        c = self_conv_fft(y, fsize)
        cur_peak = c.max()
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = cur_peak
        beta_eff = beta
        e = np.exp(beta_eff * (c[:2*n-1] - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        # Gradient: 2 * cross_conv(w, a) truncated
        fsize_g = 1 << (2 * n - 1).bit_length()
        W = fft(w, fsize_g)
        A = fft(y, fsize_g)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(beta * L) where L ~ 2*max|a|
        step = 1.0 / (0.5 * beta_eff + gnorm * 0.1)
        step = min(step, S * 0.5)

        # Adaptive: reduce step if stuck
        if it - last_improve > 500:
            step *= 0.5
        if it - last_improve > 2000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Noise injection during annealing
        if current_beta < base_beta * 0.8 and it % 50 == 0:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

    # Final
    c_final = self_conv_fft(a, fsize)
    if c_final.max() < best_peak:
        best_peak = c_final.max()
        best_a = a.copy()

    return best_a, best_peak
