def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

Since `M_k` is a permutation matrix (it's the matrix that reverses indices around position k/2), the eigenvalues of `M_k` are +1 and -1 (depending on the dimension). So `a^T M_k a` is NOT convex in general.

This makes the QP non-convex. However, we can still use it in a local optimization sense.

**Alternative: Use the fact that we're near a local optimum.**

Near the current solution, the active peaks are stable. We can linearize the constraints:
`c_k(a) ≈ c_k(a_0) + 2 * grad_c_k(a_0) · (a - a_0)`

Then the problem becomes a QP with linear constraints, which is convex!

Let me implement this:

1. At the current point `a_0`, identify active peaks (those within epsilon of the max).
2. Linearize: `c_k(a) ≈ c_k(a_0) + 2 * g_k · (a - a_0)` where `g_k = conv(indicator_k, a_0)`.

4. Move to the solution.
5. Repeat.

This is essentially a **sequential quadratic programming (SQP)** approach!

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    A = fft(a_pad)

"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.ascontiguousarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def grad_topk(a, N, top_idx, n):
    """Gradient of sum_{k in top_idx} c_k with respect to a."""
    ind = np.zeros(2 * n - 1)
    for k in top_idx:
        ind[int(k)] = 1.0
    fsize = 1 << (3 * n).bit_length()
    W = fft(ind, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def lse_gradient(a, N, beta, n):
    """Gradient of LSE-smoothed max of conv(a,a)."""
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max, c
    w = exp_bc / total
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c_max, c


def spectral_equalize(a, N, alpha, top_k, n):
    """Spectral equalization: flatten |A|^2 in dominant frequency regions."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    p_mean = P.mean()
    # Compute equalization target: where P is high, reduce it
    excess = np.maximum(P - p_mean, 0)
    P_new = P - alpha * excess
    P_new = np.maximum(P_new, 1e-10)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    S = a.sum()
    return project_simplex(a_new, S)


def bb_step(s_prev, y_prev, step_current):
    """Barzilai-Borwein step size update with safeguards."""
    if s_prev is None or y_prev is None:
        return step_current
    ss = np.dot(s_prev, s_prev)
    sy = np.dot(s_prev, y_prev)
    if ss < 1e-30:
        return step_current
    if sy < 1e-14 * ss:
        # Not a descent direction, keep current step
        return step_current
    step_bb = ss / sy
    # Safeguards: limit to [0.01, 100] * current step
    step_new = max(0.01 * step_current, min(100.0 * step_current, step_bb))
    return step_new


def nesterov_bb_block(a_init, n, S, N, max_time, rng, K_start=10, K_end=2000,
                      spectral_interval=100, spectral_alpha=0.3):
    """
    Nesterov-accelerated subgradient on sum-of-top-K peaks with BB step size.
    Periodic spectral equalization to escape local minima.
    """
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = np.inf
    start = time.monotonic()
    it = 0

    # BB history
    s_hist = None
    y_hist = None
    step = 2.0 * S / (n * 3.0)

    # Nesterov restart counter
    nesterov_restart_interval = 1000
    no_improve = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1

        # Time-varying K
        frac = min(1.0, it / max(1, max_time / 0.02))
        K = int(K_start + (K_end - K_start) * frac)
        K = min(K, 2 * n - 1)

        # Compute convolution at Nesterov point
        c = compute_conv(y, N)
        c_max = c.max()

        if c_max < best_val:
            best_val = c_max
            best_a = y.copy()
            no_improve = 0
            if it % 100 == 0:
                print("    NB iter %d: max=%.6f (K=%d)" % (it, c_max, K))
        else:
            no_improve += 1

        # Get top-K indices and gradient
        top_idx = np.argsort(c)[::-1][:K]
        grad = grad_topk(y, N, top_idx, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Update BB step
        if s_hist is not None and y_hist is not None:
            step = bb_step(s_hist, y_hist, step)

        # Compute search direction: project -grad onto tangent space of sum=S
        mean_g = grad.mean()
        direction = -(grad - mean_g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Step size: scale by step / dir_norm to get unit-ish movement
        alpha = step / dir_norm

        # Line search (backtracking)
        a_new = None
        for ls in range(12):
            candidate = project_simplex(y - alpha * direction, S)
            c_cand = compute_conv(candidate, N)
            if c_cand.max() <= c_max + 1e-12:
                a_new = candidate
                break
            alpha *= 0.5
        if a_new is None:
            a_new = project_simplex(y - alpha * direction, S)

        # Store BB history
        s_hist = a_new - a
        c_at_new = compute_conv(a_new, N)
        top_idx_new = np.argsort(c_at_new)[::-1][:K]
        grad_new = grad_topk(a_new, N, top_idx_new, n)
        y_hist = grad_new - grad

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y, S)
        a = a_new
        t = t_new

        # Nesterov restart
        if it % nesterov_restart_interval == 0:
            t = 1.0
            y = a.copy()
            s_hist = None
            y_hist = None

        # Periodic spectral equalization
        if it % spectral_interval == 0:
            a = spectral_equalize(a, N, spectral_alpha, K, n)
            y = a.copy()
            t = 1.0
            s_hist = None
            y_hist = None

        # Random perturbation if stuck
        if no_improve > 500 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t = 1.0
            s_hist = None
            y_hist = None
            no_improve = 0
            last_best = best_val

    return best_a, best_val


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=30.0, beta_max=300.0, m_lbfgs=15):
    """L-BFGS on LSE-smoothed objective."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, c_max, c = lse_gradient(a, N, beta, n)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    start = time.monotonic()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.05)
            g, c_max, c = lse_gradient(a, N, beta, n)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        if it % 30 == 0:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # L-BFGS direction
        direction = _lbfgs_direction(s_hist, y_hist, g, n)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Line search
        alpha = step_scale
        a_new = a
        c_new_max = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 14:
                a_new = candidate
                c_new_max = cm_ls
                if cm_ls <= c_max:
                    break
            alpha *= 0.5

        if c_new_max < best_cmax:
            best_cmax = c_new_max
            best_a = a_new.copy()
            if it % 50 == 0:
                print("    LBFGS iter %d: max=%.6f (beta=%.0f)" % (it, best_cmax, beta))

        g_new, _, _ = lse_gradient(a_new, N, beta, n)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        step_scale = max(0.01, min(10.0, step_scale * (1.0 + 0.001)))

        a = a_new
        g = g_new
        c = compute_conv(a, N)
        c_max = c.max()

        if it % 300 == 0 and rng.random() < 0.15:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            g, c_max, c = lse_gradient(a, N, beta, n)

    return best_a, best_cmax


def _lbfgs_direction(s_hist, y_hist, g, n):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def fine_polish_block(a_init, n, S, N, max_time, rng):
    """Ultra-fine polishing: exact max subgradient with very small steps."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    best_a = a.copy()
    best_val = np.inf
    start = time.monotonic()
    it = 0
    step = 0.5 * S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_val:
            best_val = c_max
            best_a = a.copy()
            if it % 200 == 0:
                print("    Polish iter %d: max=%.6f" % (it, c_max))

        # Find all positions at or near the max
        threshold = c_max - 0.01 * c_max
        top_mask = c >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(c)])
        # Limit to top 200 for efficiency
        if len(top_idx) > 200:
            order = np.argsort(c[top_idx])[::-1][:200]
            top_idx = top_idx[order]

        grad = grad_topk(a, N, top_idx, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        mean_g = grad.mean()
        direction = -(grad - mean_g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha = step / dir_norm
        improved = False
        for ls in range(8):
            a_new = project_simplex(a - alpha * direction, S)
            c_new = compute_conv(a_new, N)
            if c_new.max() < c_max - 1e-14:
                a = a_new
                improved = True
                break
            alpha *= 0.5

        if not improved and it % 200 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_val


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-phase optimization starting from the given good solution.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # Start from given solution
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a = np.clip(given, 0, 1000).astype(np.float64)
            a = a * (S / a.sum())
            c = compute_conv(a, N)
            start_val = c.max()
            print("Starting from given: max_conv=%.6f" % start_val)
        else:
            raise ValueError("size mismatch")
    except Exception:
        a = np.ones(n) * S / n
        start_val = compute_conv(a, N).max()
        print("Using flat initialization: max_conv=%.6f" % start_val)

    global_best = start_val
    global_a = a.copy()

    def try_update(a_cand, label):
        nonlocal global_best, global_a
        a_norm = project_simplex(a_cand, S)
        c = compute_conv(a_norm, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()
            print("  [%s] %.6f" % (label, global_best))
        return sc

    # === Phase 1: Nesterov BB with spectral moves (60% of time) ===
    print("=== Phase 1: Nesterov BB + Spectral ===")
    t1 = time.monotonic()
    phase1_budget = min(1400.0, (deadline - t1) * 0.6)

    a_curr = global_a.copy()
    best_p1, val_p1 = nesterov_bb_block(
        a_curr, n, S, N, phase1_budget, rng,
        K_start=10, K_end=1500,
        spectral_interval=150, spectral_alpha=0.25
    )
    try_update(best_p1, "Phase1_NesterovBB")

    # === Phase 2: L-BFGS on LSE (25% of time) ===
    print("=== Phase 2: L-BFGS LSE ===")
    t2 = time.monotonic()
    phase2_budget = min(600.0, (deadline - t2) * 0.3)

    if phase2_budget > 10:
        # Multiple L-BFGS blocks with increasing beta
        a_l = global_a.copy()
        while time.monotonic() < t2 + phase2_budget:
            block_time = min(200.0, (t2 + phase2_budget) - time.monotonic())
            if block_time < 10:
                break
            beta_s = 30.0 + 10.0 * (t2 + phase2_budget - time.monotonic()) / phase2_budget
            a_l, sc_l = lbfgs_lse_block(
                a_l, n, S, N, block_time, rng,
                beta_init=beta_s, beta_max=min(500.0, beta_s * 10.0), m_lbfgs=15
            )
            try_update(a_l, "L-BFGS")
            a_l = global_a.copy()

    # === Phase 3: Fine polishing (15% of remaining time) ===
    print("=== Phase 3: Fine Polish ===")
    t3 = time.monotonic()
    remaining = deadline - t3
    phase3_budget = min(remaining * 0.5, 400.0)

    if phase3_budget > 10:
        a_pol, val_pol = fine_polish_block(global_a, n, S, N, phase3_budget, rng)
        try_update(a_pol, "Polish")

    # === Phase 4: Final Nesterov with very small K ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("=== Phase 4: Final Nesterov (small K) ===")
        a_fin, val_fin = nesterov_bb_block(
            global_a, n, S, N, min(remaining - 10, 200.0), rng,
            K_start=1, K_end=50,
            spectral_interval=200, spectral_alpha=0.1
        )
        try_update(a_fin, "Final_Nesterov")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fin2, val_fin2 = fine_polish_block(global_a, n, S, N, min(remaining - 5, 100.0), rng)
        try_update(a_fin2, "Final_Polish")

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: %.6f" % ver)
    else:
        print("Verification: inf (using fallback)")
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
