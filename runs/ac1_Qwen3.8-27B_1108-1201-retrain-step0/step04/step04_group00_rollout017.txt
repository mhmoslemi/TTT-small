"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Multi-phase strategy:
1. Multi-start FISTA with adaptive LSE smoothing and backtracking at original size.
2. Diversification: perturb best solution, try different spectral initializations.
3. Multi-scale: upsample to 2x for more degrees of freedom.
4. Intensive polishing with exact active-set subgradients and coordinate refinement.

Key improvements over baseline: backtracking line search, adaptive active set growth,
spectral initialization, and coordinate descent polishing for final precision.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE (log-sum-exp) gradient of smoothed max-convolution."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Gradient using top-k lags with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - c.min(), 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within relative tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val < 1e-30:
        return np.zeros(n)
    if tol_frac > 0:
        threshold = max_val * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the objective value 2n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_backtrack(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                        beta_max=5000.0, step_scale=1.0, use_backtrack=True):
    """
    FISTA with adaptive LSE smoothing and optional backtracking.
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_conv(y)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient: blend LSE with exact as beta grows
        if beta < 50.0:
            grad = lse_gradient(y, c, beta)
        elif beta < 500.0:
            w_blend = min(1.0, (beta - 50.0) / 450.0)
            grad_lse = lse_gradient(y, c, beta)
            grad_exact = exact_subgradient(y, c, tol_frac=0.005)
            grad = (1.0 - w_blend) * grad_lse + w_blend * grad_exact
        else:
            tol = min(0.05, 0.005 * (1.0 + stuck_count / 5000.0))
            grad = exact_subgradient(y, c, tol_frac=tol)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search (simplified: no inner loop for speed)
        if use_backtrack and it % 10 == 0 and stuck_count > 100:
            # Try a larger step
            step_try = step * 2.0
            x_try = project_simplex(y - step_try * grad, S)
            c_try = fft_conv(x_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < cur_obj:
                step = step_try

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if stuck_count > 3000 and it % 200 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

        # Periodic restart
        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_polish(a_init, max_time, rng):
    """
    Intensive polishing using adaptive active-set subgradient.
    Focuses on making the convolution as flat as possible over the active set.
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv(y)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        max_val = c.max()
        if max_val < 1e-30:
            break

        # Adaptive tolerance: grow active set when stuck
        if stuck < 100:
            tol_frac = 0.0
        elif stuck < 500:
            tol_frac = 0.002
        elif stuck < 2000:
            tol_frac = 0.005
        elif stuck < 5000:
            tol_frac = 0.01
        else:
            tol_frac = 0.02

        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        k_active = len(active)

        # Weighted subgradient: weight by how close each active lag is to max
        if k_active <= 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            # Power-weighted: emphasize lags closer to max
            diffs = (max_val - c[active]) / max_val
            w = np.exp(-50.0 * diffs)
            w /= w.sum()
            w_full = np.zeros(len(c))
            w_full[active] = w
            fsize = 1 << (len(w_full) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-type step size
        step = S / (gnorm * np.sqrt(1.0 + it * 0.01))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Escape from local minima
        if stuck > 5000 and it % 300 == 0:
            noise_scale = max(1e-5, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def coordinate_descent_polish(a_init, max_time, rng):
    """
    Simple coordinate descent: update pairs of coordinates to maintain sum.
    For each pair (i, j), find the optimal redistribution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_conv(a)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()

        # Pick pairs to update: coordinates where a_i is large and contribute to max conv
        max_lag = int(np.argmax(c))
        # Focus on coordinates that affect the max lag
        affected = np.arange(max(0, max_lag - n + 1), min(n, max_lag + 1))
        if len(affected) < 2:
            affected = np.random.choice(n, min(n, 100), replace=False)

        # Update a random subset
        n_updates = min(len(affected), 50)
        idx = rng.choice(affected, n_updates, replace=False)

        for i in idx:
            # Find partner j: coordinate with most overlap in convolution
            # Simplified: pick j = (i + n//2) % n (anti-symmetric partner)
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n

            # Current values
            ai = a[i]
            aj = a[j]
            if ai + aj < 1e-15:
                continue

            # Binary search for optimal delta (move mass from j to i or vice versa)
            # We want to minimize max(conv) after moving delta from j to i
            # a_i -> a_i + delta, a_j -> a_j - delta
            # delta in [-aj, ai]
            lo = -aj
            hi = ai
            # Ternary search on the convex function
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Compute max conv for delta = m1
                a[i] = ai + m1
                a[j] = aj - m1
                c1 = fft_conv(a)
                obj1 = c1.max()

                # Compute max conv for delta = m2
                a[i] = ai + m2
                a[j] = aj - m2
                c2 = fft_conv(a)
                obj2 = c2.max()

                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1

            # Set to best
            delta = (lo + hi) / 2.0
            a[i] = ai + delta
            a[j] = aj - delta
            a[i] = max(0.0, a[i])
            a[j] = max(0.0, a[j])

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def make_init_cosine(n, alpha=0.5):
    """Raised cosine initialization: 1 + alpha*cos(2*pi*t)."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + alpha * np.cos(2 * np.pi * t)
    return np.maximum(a, 0.0)


def make_init_hann(n):
    """Hann window."""
    t = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t / n))
    return np.maximum(a, 1e-10)


def make_init_spectral(n, rng, target_flat=True):
    """Initialize with a flat spectral magnitude (random phase)."""
    # Random phase with flat magnitude
    fsize = 1 << (2 * n - 1).bit_length()
    mag = np.ones(fsize)
    phase = rng.uniform(0, 2 * np.pi, fsize)
    # Enforce Hermitian symmetry for real output
    phase[0] = 0
    if fsize % 2 == 0:
        phase[fsize // 2] = 0
    spectrum = mag * np.exp(1j * phase)
    spectrum[fsize // 2:] = np.conj(spectrum[1:fsize // 2 + 1][::-1]) if fsize % 2 == 0 else np.conj(spectrum[1:fsize // 2][::-1])
    a = np.real(ifft(spectrum)[:n])
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = make_init_cosine(n)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing 2n*max_conv/sum^2.
    Phases: multi-start FISTA, diversification, multi-scale, polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_start = time.monotonic()
    deadline = total_start + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a):
        nonlocal global_best_obj, global_best_a
        if a is not None and len(a) > 0:
            obj = compute_obj(a)
            if np.isfinite(obj) and obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a.copy()
                print("  NEW BEST: obj=%.6f (n=%d) at t=%.0fs" % (obj, len(a), time.monotonic() - total_start))

    # Get initial sequence from provided construction
    a_start = None
    try:
        a_cand = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_cand = np.clip(a_cand, 0, 1000)
        if a_cand.sum() >= 0.01:
            a_start = a_cand
    except Exception:
        pass

    if a_start is None:
        n_def = 8192
        a_start = make_init_cosine(n_def, alpha=0.5)
        print("No provided sequence, using cosine init n=%d" % n_def)

    n0 = len(a_start)
    # Normalize to standard form
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    update_best(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, global_best_obj))

    # === Phase 1: Multi-start FISTA at original size (35% budget) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: Multi-start FISTA at n=%d (budget %.0fs) ===" % (n0, p1_budget))

    # Start 1: From provided sequence with LSE FISTA
    t1 = time.monotonic()
    a1, obj1 = run_fista_backtrack(a_start, p1_budget * 0.35, rng,
                                    beta_init=10.0, beta_rate=0.008,
                                    beta_max=5000.0, step_scale=1.0)
    update_best(a1)
    print("  Start1 (provided): obj=%.6f (t=%.0fs)" % (obj1, time.monotonic() - t1))

    # Start 2: Cosine init with different alpha
    remaining_p1 = p1_budget - (time.monotonic() - t1)
    if remaining_p1 > 30:
        a_cos = make_init_cosine(n0, alpha=0.3)
        a_cos = a_cos * (S0 / a_cos.sum())
        t2 = time.monotonic()
        a2, obj2 = run_fista_backtrack(a_cos, remaining_p1 * 0.4, rng,
                                        beta_init=5.0, beta_rate=0.006,
                                        beta_max=3000.0, step_scale=1.0)
        update_best(a2)
        print("  Start2 (cosine a=0.3): obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))

        # Start 3: Perturbed best
        remaining_p1b = remaining_p1 - (time.monotonic() - t2)
        if remaining_p1b > 20:
            a_pert = global_best_a + rng.normal(0, 0.01 * S0 / np.sqrt(n0), n0)
            a_pert = project_simplex(a_pert, S0)
            t3 = time.monotonic()
            a3, obj3 = run_fista_backtrack(a_pert, remaining_p1b, rng,
                                            beta_init=20.0, beta_rate=0.005,
                                            beta_max=5000.0, step_scale=0.8)
            update_best(a3)
            print("  Start3 (perturbed): obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))

    if remaining > 60:
        p2_budget = remaining * 0.15

        for n_try in [n0 // 2, n0, n0 * 2]:

        remaining_p3 = p3_budget - (time.monotonic() - t_up)

            a_u2, obj_u2 = run_polish(global_best_a, remaining_p3, rng)

    # === Phase 4: Intensive polishing on best (20% budget) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.25
        n_best = len(global_best_a)
        print("\n=== Phase 4: Intensive polish at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        # Active-set polish
        p4a_budget = min(p4_budget * 0.5, deadline - time.monotonic() - 20)
        if p4a_budget > 10:

        remaining_p4 = p4_budget - (time.monotonic() - t4)
        if remaining_p4 > 10:
            a_p4b, obj_p4b = run_fista_backtrack(global_best_a, remaining_p4, rng,
                                                  beta_init=100.0, beta_rate=0.002,
                                                  beta_max=20000.0, step_scale=0.5)

            print("  High-beta: obj=%.6f (t=%.0fs)" % (obj_p4b, time.monotonic() - t4))

    # === Phase 5: Coordinate descent polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(global_best_a) <= 16384:

        t5 = time.monotonic()

    if remaining > 10:

    # Final verification
    if global_best_a is not None:

        # Verify with evaluate_sequence

            print("Total time: %.0fs" % (time.monotonic() - total_start))

            final_list = [float(x) for x in make_init_cosine(n_fb)]
    else:
        n_fb = 8192
        final_list = [float(x) for x in make_init_cosine(n_fb)]
