"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimization with adaptive strategies:
1. FISTA with LSE smoothing and adaptive beta
2. Active-set focused gradient with top-k lags
3. Stochastic polishing with random active lag sampling
4. Multi-length exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def stochastic_gradient(y, c, k, rng):
    """Random k lags gradient."""
    n = len(y)
    c_idx = np.random.choice(len(c), size=k, replace=False, p=np.maximum(c, 1e-30) / np.maximum(c, 1e-30).sum())
    w_full = np.zeros(len(c))
    w_full[c_idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        step = min(step, 2.0 / max(gnorm, 1e-10))

def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

- Total: O(n * 20) per coordinate, O(n^2 * 20) total
- For n=8192: about 1.3B operations. Still too slow.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimization with:
1. FISTA with adaptive LSE/topk gradient and backtracking line search
2. Multi-length exploration (n, 2n)
3. Active-set intensive polishing
4. Stochastic polishing with random lag sampling
Key innovations over baseline: line search for better step sizes,
adaptive mode switching, and periodic warm restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def stochastic_top_gradient(y, c, k, rng):
    """Sample k lags proportional to their values."""
    n = len(y)
    probs = np.maximum(c, 1e-30)
    probs = probs / probs.sum()
    idx = rng.choice(len(c), size=k, replace=False, p=probs)
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_ls(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
                 beta_max=10000.0, mode="auto", step_scale=1.0,
                 do_linsearch=True, ls_intervals=(1.0, 0.7, 0.5, 0.3, 0.1)):
    """
    FISTA with optional backtracking line search and adaptive mode.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        if mode == "auto":
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        else:
            beta = float(beta_init)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Choose gradient mode
        if mode == "auto":
            if stuck_count < 300:
                grad = lse_gradient(y, c, beta)
            elif stuck_count < 2000:
                k = min(64, max(8, stuck_count // 50))
                grad = topk_gradient(y, c, k, power=1.0)
            elif stuck_count < 8000:
                k = min(128, 64 + stuck_count // 200)
                grad = topk_gradient(y, c, k, power=1.2)
            else:
                k = min(256, 128 + stuck_count // 500)
                grad = stochastic_top_gradient(y, c, k, rng)
        elif mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = min(128, max(8, n // 32))
            grad = topk_gradient(y, c, k, power=1.3)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size determination
        base_step = step_scale / max(beta, 1.0)

        if do_linsearch and (it <= 50 or it % 10 == 0 or stuck_count < 50):
            # Line search: try multiple step sizes
            best_x = y.copy()
            best_c = cur_max
            for factor in ls_intervals:
                step = base_step * factor
                x_trial = project_simplex(y - step * grad, S)
                c_trial = fft_convolve_self(x_trial)
                if c_trial.max() < best_c:
                    best_c = c_trial.max()
                    best_x = x_trial
            x_new = best_x
        else:
            # Fixed step (cheaper)
            x_new = project_simplex(y - base_step * grad, S)

        # Momentum restart when stuck
        if stuck_count > 3000 and it % 500 == 0:
            t_mom = 1.0
            noise_scale = 0.004 * S / np.sqrt(n) * np.exp(-stuck_count / 8000.0)
            if noise_scale > 1e-7:
                y = np.maximum(y + rng.normal(0, noise_scale, n), 0.0)
                y = project_simplex(y, S)
                a = y.copy()

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic full restart
        if it % 10000 == 0:
            t_mom = 1.0
            a = best_a.copy()
            y = a.copy()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """
    Focused polishing on the active (maximum) lags only.
    Uses exact subgradient from the top few lags with decreasing steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive number of active lags
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 100)
        elif stuck < 10000:
            k = min(32, 8 + stuck // 200)
        else:
            k = min(64, 32 + stuck // 500)

        if k == 1:
            idx = np.array([int(np.argmax(c))])
        else:
            idx = np.argsort(c)[::-1][:k]

        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / len(idx)
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polished step size: decrease slowly
        step = min(2.0 / max(gnorm, 1e-10), S / (gnorm * (1.0 + 0.002 * it)))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck for a long time
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            a = y.copy()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_stochastic_polish(a_init, max_time, rng, batch_size=16):
    """
    Stochastic gradient polishing: sample random subsets of active lags.
    Good for escaping flat regions.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    # Precompute weights for sampling (based on initial convolution)
    c_init = fft_convolve_self(y)
    probs = np.maximum(c_init, 1e-30)
    probs = probs / probs.sum()

    while time.monotonic() - start < max_time:
        it += 1

        # Sample a batch of lags
        batch_idx = rng.choice(len(c_init), size=batch_size, replace=False, p=probs)
        w_full = np.zeros(len(c_init))
        w_full[batch_idx] = 1.0 / batch_size
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / max(gnorm, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic full evaluation
        if it % 200 == 0:
            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv:
                best_conv = cur_max
                best_a = y.copy()
            # Update sampling weights
            probs = np.maximum(c, 1e-30)
            probs = probs / probs.sum()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)
