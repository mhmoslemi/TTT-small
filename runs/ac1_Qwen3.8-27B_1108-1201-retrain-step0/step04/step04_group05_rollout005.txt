"""
Minimize 2*n*max_conv/sum^2 using Adam optimizer on LSE-smoothed objective
with log-parameterization, peak-targeted coordinate descent, and spectral refinement.

Key innovations:
1. Adam optimizer with per-coordinate adaptive learning rates (different from L-BFGS)
2. Log-parameterization for better conditioning near zero

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv_full(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_value_and_grad(a, N, beta):
    """Compute LSE-smoothed max of conv and its gradient."""
    n = len(a)
    c = compute_conv_full(a, N)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Numerical stability
    exp_bc = np.exp(np.minimum(shifted, 50))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    
    # Gradient via FFT
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max

def adam_step(a, grad, m_v, v_v, t, lr, beta1, beta2, eps, S):
    """One Adam step with simplex projection."""
    n = len(a)
    m_v[:] = beta1 * m_v + (1 - beta1) * grad
    v_v[:] = beta2 * v_v + (1 - beta2) * grad * grad
    m_hat = m_v / (1 - beta1 ** t)
    v_hat = v_v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a - step, S)
    return a_new, m_v, v_v

def peak_targeted_descent(a, S, N, n, top_k, step_size, n_iter):
    """
    For each of the top_k peaks, compute the gradient contribution
    and take a step to reduce it.
    """
    a = a.copy()
    for it in range(n_iter):
        c = compute_conv_full(a, N)
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:top_k]
        
        # Compute gradient from top peaks only
        w = np.zeros(2 * n - 1)
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        w[top_idx] = top_vals / total
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        
        a = project_simplex(a - step_size * grad, S)
    
    return a

def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g if gn >= 1e-18 else np.zeros(n)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last) if yy_last > 1e-30 else 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    shifted = beta * (c - c_max)
    exp_bc = np.exp(np.minimum(shifted, 50.0))
    total = exp_bc.sum()

        return np.zeros(n), c_max, c, c_max

    lse_val = c_max + np.log(total) / beta

    W = fft(w, fsize)
    Af = fft(a, fsize)
