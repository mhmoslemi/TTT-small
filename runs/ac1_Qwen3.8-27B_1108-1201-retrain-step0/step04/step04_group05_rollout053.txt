"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Mirror descent (exponential family) for fast initial progress on non-negative variables
2. L-BFGS on log-sum-exp smoothed objective with adaptive beta for refinement
3. Subgradient with exact 1D line search for final polishing
4. Spectral flattening as periodic global moves
Uses n=16384 for more degrees of freedom, with tapered initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_and_grad(a, N, beta):
    """LSE-smoothed peak of conv(a,a) and its gradient."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def mirror_descent_block(a_init, n, S, N, max_time, rng):
    """Mirror descent with exponential family updates + Nesterov momentum."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    momentum = np.zeros(n)

    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = compute_conv(a, N)
        c_max = c.mean()
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Weighted subgradient: weight peaks by exceedance above mean
        excess = np.maximum(c - c_mean, 0) if (c_mean := c.mean()) > 0 else np.zeros_like(c)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = excess / total_excess

        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        # Adaptive step size
        eta = 0.5 * S / (np.max(np.abs(grad)) + 1e-10) * (1.0 / (1.0 + 0.01 * it))
        eta = min(eta, 0.1)

        # Mirror descent update (exponential tilting)
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        update = np.exp(log_update)
        a_new = a * update
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / a_new_sum)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))

        # Momentum (HTML - Heavy Ball)
        momentum = 0.9 * momentum + 0.1 * (a_new - a)
        a = np.maximum(a + momentum, 0.0)
        a = a * (S / max(a.sum(), 1e-15))

        # Periodic noise
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_cmax


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta ramping."""
    a = project_simplex(a_init.copy(), S)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    _, g, c, c_max = lse_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = S / np.max(np.abs(g)) * 0.5 if np.max(np.abs(g)) > 1e-15 else 1.0
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            _, g, c, c_max = lse_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
            step_scale = S / (np.max(np.abs(g)) + 1e-10) * 0.5

        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Backtracking line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        found = False
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1 + 1e-8) or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max * (1 + 1e-8):
                    found = True
                    break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        _, g_new, c_new, cmax_new2 = lse_and_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)

        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c = c_new
        c_max = cmax_new2

        # Periodic perturbation
        if it % 300 == 0 and rng.random() < 0.25:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            _, g, c, c_max = lse_and_grad(a, N, beta)

    return best_a, best_cmax


def subgrad_exact_ls_block(a_init, n, S, N, max_time, rng):
    """Subgradient method with exact 1D line search along the direction."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    momentum_beta = 0.5

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Compute subgradient: weighted average of active peak gradients
        c_mean = c.mean()
        threshold = c_mean + 0.1 * (cur - c_mean)
        active = c > threshold
        n_active = active.sum()
        if n_active == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[np.argmax(c)] = True

        # Compute gradient using FFT
        fsize = 1 << (3 * n - 2).bit_length()
        w = active.astype(np.float64) / n_active
        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        # Direction: -grad projected to sum-zero
        direction = -grad
        direction -= direction.mean()  # ensure sum = 0 for line search

        # Find alpha_max such that a + alpha*direction >= 0
        neg_mask = direction < -1e-15
        if neg_mask.any():
            alpha_max = (a[neg_mask] / (-direction[neg_mask])).min()
            alpha_max = max(alpha_max, 0.0)
        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 10.0 * S / (np.max(np.abs(direction)) + 1e-10))

        # Exact 1D line search: minimize max(c + alpha * v) where v = direction · grad_k
        # Compute v_k = directional derivative of c_k along 'direction'
        d_pad = np.zeros(N)
        d_pad[:n] = direction
        D = fft(d_pad)
        v = np.real(ifft(D * np.conj(fft(a, N))))[:2 * n - 1]
        # Actually v_k = sum_i direction_i * (d(c_k)/d(a_i)) = sum_i direction_i * a_{k-i} (symmetric)
        # v = conv(direction, a) + conv(a, direction) but since it's the same, v = 2*conv(direction, a) for the cross terms
        # Let's just compute it directly via FFT
        v = 2.0 * np.real(ifft(D * fft(a, N)))[:2 * n - 1]

        # Minimize max_k (c_k + alpha * v_k) over alpha in [0, alpha_max]
        # This is a 1D convex optimization
        lo, hi = 0.0, alpha_max
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            # Derivative at mid: max v_k among k where c_k + mid*v_k is near the max
            vals = c + mid * v
            max_val = vals.max()
            active_at_mid = vals > max_val - 1e-10 * max(1.0, max_val)
            if active_at_mid.any():
                derivative = v[active_at_mid].max()
            else:
                derivative = 0.0
            if derivative > 0:
                hi = mid
            else:
                lo = mid
        alpha_opt = 0.5 * (lo + hi)

        # Apply momentum
        new_pos = a + alpha_opt * direction
        momentum = momentum_beta * momentum + (1 - momentum_beta) * (new_pos - a)
        a_new = project_simplex(a + momentum, S)

        # Verify and update
        c_new, _ = compute_conv(a_new, N)
        cm_new = c_new.max()
        if cm_new < best_cmax:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Mirror descent (exponential family) for fast initial progress on non-negative variables
2. L-BFGS on log-sum-exp smoothed objective with adaptive beta for refinement
3. Subgradient with exact 1D line search for final polishing
4. Spectral flattening as periodic global moves
Uses tapered initializations and adaptive problem sizing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_and_grad(a, N, beta):
    """LSE-smoothed peak of conv(a,a) and its gradient."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n), c, c_max
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def mirror_descent_block(a_init, n, S, N, max_time, rng):
    """Mirror descent with exponential family updates + heavy-ball momentum."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    mom_beta = 0.7

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        c_mean = c.mean()
        excess = np.maximum(c - c_mean, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = (excess ** 2) / (excess ** 2).sum()

        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        eta = 0.3 * S / (np.max(np.abs(grad)) + 1e-10)
        eta = min(eta, 0.05)

        log_update = -eta * grad
        log_update -= log_update.max()
        update = np.exp(log_update)
        a_new = a * update
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / a_new_sum)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))

        momentum = mom_beta * momentum + (1.0 - mom_beta) * (a_new - a)
        a = np.maximum(a + momentum, 0.0)
        a = a * (S / max(a.sum(), 1e-15))

        if it % 300 == 0 and rng.random() < 0.3:

            momentum = np.zeros(n)

    return best_a, best_cmax

def subgrad_exact_ls_block(a_init, n, S, N, max_time, rng):
    """Subgradient with exact 1D line search and heavy-ball momentum."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    mom_beta = 0.4

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv(a, N)
        cur = c.max()

            best_cmax = cur
            best_a = a.copy()

        c_mean = c.mean()
        spread = cur - c_mean
        if spread < 1e-15:
            break
        threshold = c_mean + 0.05 * spread
        active = c > threshold
        n_active = active.sum()
        if n_active == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True

        weighted = active.astype(np.float64) * (c - threshold)
        total_w = weighted.sum()
        if total_w < 1e-30:
            w = active.astype(np.float64) / n_active
        else:
            w = weighted / total_w

        W = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        direction = -grad
        direction -= direction.mean()

        neg_mask = direction < -1e-15

            alpha_max = (a[neg_mask] / (-direction[neg_mask])).min()
            alpha_max = max(alpha_max, 0.0)

        d_pad[:n] = direction

            else:

            best_a = a_new.copy()
        a = a_new

        if it % 150 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            momentum = np.zeros(n)

        mom_beta = max(0.2, mom_beta * 0.9995)

    return best_a, best_cmax


def spectral_flatten_block(a_init, n, S, N, max_time, rng, top_k=20, alpha_init=0.1):
    """Spectral flattening to reduce peak contrast."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = compute_conv(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)

        P = np.abs(A) ** 2
        alpha_curr = alpha_init * (1.0 + 0.3 * it / 50.0)
        alpha_curr = min(alpha_curr, 0.5)
        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        if it % 150 == 75 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def make_init_dome(n, rng):
    """Dome-shaped initialization."""
    tt = np.linspace(0, 1, n)
    a = 1.0 + 0.4 * np.cos(np.pi * tt)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_bowl(n, rng):
    """Bowl-shaped initialization (higher at edges)."""
    tt = np.linspace(0, 1, n)
    a = 1.0 - 0.4 * np.cos(np.pi * tt)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
