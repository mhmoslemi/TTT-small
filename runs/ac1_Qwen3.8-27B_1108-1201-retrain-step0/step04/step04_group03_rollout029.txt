def run_mwua(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002, eta_init=0.1, eta_decay=0.999):
    """Multiplicative Weights Update Algorithm for minimizing max(f*f)."""
    a = normalize_sum(a_init, n)
    best = np.inf
    best_a = a.copy()
    eta = eta_init
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(1000.0, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Compute softmax weights over the convolution
        cm = c.max()
        w = np.exp(beta * (c - cm))
        tot = w.sum()
        if tot < 1e-300:
            w = np.zeros_like(c)
            w[cm.argmax()] = 1.0
        else:
            w /= tot
        
        # Compute repulsion force
        rep = fft_cross_correlate(w, a)
        rep = np.maximum(rep, 1e-10)  # avoid log of 0
        
        # Multiplicative update
        a_new = a * np.exp(-eta * rep / max(a.max(), 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        a_new = project_simplex(a_new, S)
        
        if np.all(np.isfinite(a_new)):
            a = a_new
        else:
            a = best_a.copy()
            eta *= 0.5
        
        eta *= eta_decay
    
    return best_a, best

def line_search(f, g, S, n, max_iter=40):
    """Find alpha in [0,1] that minimizes max(conv((1-a)f + a*g))."""
    from scipy.optimize import minimize_scalar
    
    def objective(alpha):
        if alpha < 0 or alpha > 1:
            return np.inf
        f_test = (1 - alpha) * f + alpha * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c = fft_convolve_self(f_test)
        return c.max()
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if objective(m1) < objective(m2):
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-6:
            break
    
    best_alpha = (lo + hi) / 2
    return best_alpha, objective(best_alpha)

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multiplicative Weights Update (MWUA) for fast convergence,
followed by LP best-response iterations with line search for precision.
Constraint generation for the LP to handle all relevant lags.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.sparse import csr_matrix, lil_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(w, f):
    """
    Compute grad[j] = sum_k w[k] * f[k-j] for all j in [0, n-1].
    w has length 2n-1, f has length n.
    """
    n = len(f)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    F = fft(f, fsize)
    return np.real(ifft(W * np.conj(F)))[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max(f*f)/sum(f)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def run_mwua(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.003,
             eta_init=0.05, eta_decay=0.9995):
    """
    Multiplicative Weights Update Algorithm.
    Fast O(n log n) per iteration.
    """
    a = normalize_sum(a_init, n)
    best_obj = compute_objective(a)
    best_a = a.copy()
    eta = eta_init
    start = time.monotonic()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(2000.0, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(a)
        cur = c.max()
        cur_obj = 2.0 * n * cur / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Softmax weights
        cm = c.max()
        if cm < 1e-30:
            break
        w = np.exp(beta * (c - cm))
        tot = w.sum()
        if tot < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= tot
        
        # Repulsion force (gradient-like)
        rep = 2.0 * fft_cross_correlate(w, a)
        rep = np.maximum(rep, 1e-10)
        
        # Multiplicative update
        scale = max(a.max(), 1e-10)
        a_new = a * np.exp(-eta * rep / scale)
        a_new = np.maximum(a_new, 1e-15)
        a_new = project_simplex(a_new, S)
        
        if not np.all(np.isfinite(a_new)) or a_new.sum() < 1e-10:
            a = best_a.copy()
            eta *= 0.5
        else:
            a = a_new
        
        # Adaptive eta: reduce if not improving
        if it - last_improve > 500:
            eta *= 0.7
            # Add small noise to escape
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = it
        
        eta *= eta_decay
        if eta < 1e-6:
            eta = 1e-6
    
    return best_a, best_obj


def solve_lp_best_response(f, n, S, K=200, solver='SCS', time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-K lags, g >= 0.
    Returns normalized g (sum = S) or None.
    """
    c = fft_convolve_self(f)
    M = c.max()
    if M < 1e-15:
        return None
    
    # Select top-K lags (most binding constraints)
    idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix A (K x n): A[i, j] = f[idx[i] - j] if valid
    A = np.zeros((K, n))
    for i in range(K):
        k = idx[i]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        A[i, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    
    # Solve LP: max sum(g) s.t. A @ g <= M, g >= 0
    g_var = cp.Variable(n)
    constraints = [A @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    start = time.monotonic()
    solved = False
    for sol in [solver, 'SCS', 'MOSEK', 'ECOS']:
        try:
            prob.solve(solver=sol, verbose=False)
            if time.monotonic() - start > time_limit * 2:
                continue
            solved = True
            break
        except Exception as e:
            continue
    
    if not solved or g_var.value is None:
        return None
    
    g = np.maximum(g_var.value, 0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g = g * (S / g_sum)
    return g


def line_search_convex(f, g, S, n):
    """
    Find alpha in [0,1] minimizing max(conv((1-a)*f + a*g)).
    Uses golden section search on a convex 1D function.
    """
    def eval_alpha(alpha):
        f_t = (1 - alpha) * f + alpha * g
        f_t = np.maximum(f_t, 0)
        f_t = project_simplex(f_t, S)
        c = fft_convolve_self(f_t)
        return c.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = eval_alpha(c1)
    f2 = eval_alpha(c2)
    
    for _ in range(40):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_alpha(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_alpha(c2)
        if hi - lo < 1e-7:
            break
    
    best_alpha = (lo + hi) / 2
    best_val = eval_alpha(best_alpha)
    return best_alpha, best_val


def run_lp_iterations(a_init, n, S, max_time, K=200, solver='SCS'):
    """
    Iterative LP best-response with line search.
    """
    f = normalize_sum(a_init, n)
    best_obj = compute_objective(f)
    best_a = f.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 15:
            break
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_best_response(f, n, S, K=K, solver=solver)
        lp_time = time.monotonic() - t0
        
        if g is None:
            # Try smaller K
            g = solve_lp_best_response(f, n, S, K=K//2, solver='ECOS')
            if g is None:
                break
        
        # Line search
        alpha, val = line_search_convex(f, g, S, n)
        
        if alpha > 0.01:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0)
            f_new = project_simplex(f_new, S)
            obj_new = compute_objective(f_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = f_new.copy()
                f = f_new
                print("  LP iter %d: obj=%.6f, alpha=%.3f, lp_time=%.1fs" % (it, best_obj, alpha, lp_time))
            else:
                # Try smaller step
                for frac in [0.3, 0.1, 0.05]:
                    f_t = (1 - frac) * f + frac * g
                    f_t = np.maximum(f_t, 0)
                    f_t = project_simplex(f_t, S)
                    obj_t = compute_objective(f_t)
                    if obj_t < best_obj:
                        best_obj = obj_t
                        best_a = f_t.copy()
                        f = f_t
                        print("  LP iter %d (frac=%.2f): obj=%.6f" % (it, frac, best_obj))
                        break
        else:
            # Alpha is small, try full replacement
            c_g = fft_convolve_self(g)
            obj_g = 2.0 * n * c_g.max() / (S * S)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
                f = g.copy()
                print("  LP iter %d (full replace): obj=%.6f" % (it, best_obj))
    
    return best_a, best_obj
