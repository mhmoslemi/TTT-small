def fft_convolve_self(a):

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    valid = tc <= w

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE-smoothed gradient of max(conv) w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()

    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()

def topk_subgradient(y, c, k=1, tol_frac=0.0):
    """Subgradient from top-k or tolerance-based active set."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        k = len(active)
    else:
        active = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)

def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.004,

    n = len(a_init)

def run_coordinate_polish(a_init, max_time, rng, n_passes_max=50):
    """
    Coordinate descent polishing: iterate over elements, try moving mass
    from high-convolution contributors to low-convolution ones.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    x = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    x = x * (S / x.sum())
    start = time.monotonic()
    best_obj = compute_obj(x)
    best_x = x.copy()
    c = fft_convolve_self(x)
    max_c = c.max()

    for _ in range(n_passes_max):
        if time.monotonic() - start > max_time:
            break
        improved = False
        # Try random subsets of coordinate pairs
        n_trials = min(n, 2000)
        idx_i = rng.integers(0, n, n_trials)
        idx_j = rng.integers(0, n, n_trials)
        mask = idx_i != idx_j
        idx_i = idx_i[mask]
        idx_j = idx_j[mask]

        for i, j in zip(idx_i, idx_j):
            if x[i] <= 0:
                continue
            # Try moving delta from i to j
            delta = 0.01 * x[i]
            x_old = x.copy()
            x[i] -= delta
            x[j] += delta
            if x[i] < 0:
                x[:] = x_old
                continue
            c_new = fft_convolve_self(x)
            max_new = c_new.max()
            if max_new < max_c - 1e-12:
                max_c = max_new
                improved = True
            else:
                x[:] = x_old

        if improved:
            obj = 2.0 * n * max_c / S**2
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_x = x.copy()
        else:
            break

    return best_x

    perturbed = np.maximum(perturbed, 0)
    if perturbed.sum() < 1e-10:

    return perturbed

    # === Phase 2: Diverse random starts (15% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.15
    print("\n=== Phase 2: Diverse starts (budget %.0fs) ===" % p2_budget)

    if p2_budget > 20:
        # Try an exponential decay start (strongly asymmetric)
        t = np.linspace(0, 1, n0)
        a_exp = np.exp(-2.0 * t)
        a_exp = a_exp / a_exp.sum() * np.sqrt(2.0 * n0)

        a_opt2, _ = run_fista(a_exp, p2_budget * 0.5, rng,
                              beta_init=5.0, beta_rate=0.004,
                              beta_max=50000.0, step_scale=1.0,
                              mode="auto")
        obj2 = compute_obj(a_opt2)
        print("  Exp-decay start: obj=%.6f" % obj2)
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a_opt2.copy()

        # Try a random start
        if remaining > deadline - time.monotonic() - 10:
            a_rand = np.abs(rng.normal(0, 1, n0))
            a_rand = a_rand / a_rand.sum() * np.sqrt(2.0 * n0)
            rem2 = deadline - time.monotonic()
            if rem2 > 10:
                a_opt2b, _ = run_fista(a_rand, rem2, rng,
                                       beta_init=3.0, beta_rate=0.005,
                                       beta_max=30000.0, step_scale=1.0,
                                       mode="auto")
                obj2b = compute_obj(a_opt2b)
                print("  Random start: obj=%.6f" % obj2b)
                if obj2b < global_best_obj:
                    global_best_obj = obj2b
                    global_best_a = a_opt2b.copy()

    # === Phase 3: Upsample to 2n (30% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.30
    print("\n=== Phase 3: Upsample to 2n (budget %.0fs) ===" % p3_budget)

    if p3_budget > 60 and n0 <= 16384:
        n2 = n0 * 2
        print("  Target n=%d" % n2)

        a2_init = upsample_asymmetric(global_best_a, factor=2, rng=rng)
        a2_init2 = upsample_asymmetric(global_best_a, factor=2, rng=None)

        # Optimize the perturbed upsample
        a_opt3, _ = run_fista(a2_init, p3_budget * 0.6, rng,
                              beta_init=3.0, beta_rate=0.003,
                              beta_max=80000.0, step_scale=1.0,
                              mode="auto", restart_interval=5000)
        obj3 = compute_obj(a_opt3)
        print("  Upsampled (perturbed): obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t_global_start))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a_opt3.copy()

        # Continue with the cleaner upsample
        rem3 = deadline - time.monotonic()
        if rem3 > 20:
            a_opt3b, _ = run_fista(a_opt3 if obj3 < global_best_obj else a2_init2,
                                   rem3, rng,
                                   beta_init=20.0, beta_rate=0.002,
                                   beta_max=150000.0, step_scale=0.7,
                                   mode="auto")
            obj3b = compute_obj(a_opt3b)
            print("  Upsampled (continued): obj=%.6f" % obj3b)
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a_opt3b.copy()

    # === Phase 4: Intensive polishing (15% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15
    print("\n=== Phase 4: Intensive polish (budget %.0fs) ===" % p4_budget)

    if p4_budget > 10:
        # Very high beta FISTA
        a_opt4, _ = run_fista(global_best_a, p4_budget * 0.6, rng,
                              beta_init=100.0, beta_rate=0.001,
                              beta_max=200000.0, step_scale=0.3,
                              mode="topk", restart_interval=2000,
                              noise_base=0.001)
        obj4 = compute_obj(a_opt4)
        print("  High-beta polish: obj=%.6f" % obj4)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a_opt4.copy()

        # Coordinate descent polish
        rem4 = deadline - time.monotonic()
        if rem4 > 5:
            a_pol = run_coordinate_polish(global_best_a, rem4, rng)
            obj_pol = compute_obj(a_pol)
            print("  Coordinate polish: obj=%.6f" % obj_pol)
            if obj_pol < global_best_obj:
                global_best_obj = obj_pol
                global_best_a = a_pol.copy()

    # === Phase 5: Final squeeze (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 5: Final squeeze (budget %.0fs) ===" % remaining)

        # One more pass with asymmetric perturbation + high beta
        a_pert = asymmetric_perturb(global_best_a, rng, scale=0.1)
        a_opt5, _ = run_fista(a_pert, remaining * 0.7, rng,
                              beta_init=50.0, beta_rate=0.001,
                              beta_max=100000.0, step_scale=0.5,
                              mode="auto")
        obj5 = compute_obj(a_opt5)
        print("  Final perturbed: obj=%.6f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a_opt5.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - t_global_start))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")

    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
    final_list = [float(x) for x in a_fb]
    return final_list
