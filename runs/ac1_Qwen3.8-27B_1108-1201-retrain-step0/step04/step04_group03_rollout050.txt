Strategy: FISTA with adaptive LSE smoothing, subgradient switching,
deflation steps, and multi-scale exploration.

def fft_convolve_self(a):
    """Full self-convolution via FFT. a has length n, result has length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    v = np.maximum(v, 0)  # First clamp negatives (crude but fast)
    # Actually, proper simplex projection:
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """

        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * a[k - js]
    grad /= len(top_idx)
    return grad
