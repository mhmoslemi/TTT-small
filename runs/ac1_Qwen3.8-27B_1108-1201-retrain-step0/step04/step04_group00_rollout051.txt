```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum a)^2  over non-negative sequences.

Key mathematical insight used here (different from the FISTA/LSE SOTA):
The objective is scale-invariant, so with p = a/sum(a) (a probability
vector, sum 1) we must minimize   F(p) = 2 * n * max_k (p * p)_k ,
where (p*p) is the LINEAR self-convolution (length 2n-1).  This is the
"minimum peak factor" problem for a non-negative, one-sided sequence:
we want its linear self-convolution to be as flat as possible.

  * Lower bound: max_k (p*p)_k >= mean_k (p*p)_k = 1/(2n-1)  =>  F >= ~1.
  * A symmetric p forces the centre lag = sum p_i^2 >= 1/n ~ 2*mean, so
    symmetric profiles are essentially capped at F ~ 2.  Good profiles are
    strongly ASYMMETRIC / skewed (and mildly oscillatory).

Algorithm (deliberately different from the previous FISTA+LSE program):
  1. Multiplicative / negative-entropy MIRROR descent on the simplex
     (z <- z - lr * g;  p = softmax(z))  with stochastic sub-gradients
     taken from a small set of near-maximum convolution lags.  This kernel
     keeps p strictly positive and naturally produces the skewed shapes
     that flatten the linear convolution.
  2. A large bank of structurally-biased, ASYMMETRIC parametric inits
     (power-law skew x^a(1-x)^b times a low-order cosine ripple) plus
     random mutations of the current best, each optimised in a separate
     short mirror-descent run -> wide, diverse coverage of the landscape.
  3. A short Euclidean-projected active-set FISTA polish on the best
     profile (simplex projection, exact top-lag sub-gradient) to squeeze
     the last bit, run at a couple of sizes.
  4. Multi-size selection: the profile is resampled to a few lengths and
     briefly re-optimised; the minimum-2n*maxc result is returned.

All helpers are top level; no closures, no lambdas, no fs/network IO.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


# ----------------------------- low-level helpers -----------------------------

def fft_convolve_self(a):
    """Full linear self-convolution of a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(z):
    """Stable probability vector from logits (always positive, sum 1)."""
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60.0, 60.0))
    e = e / e.sum()
    return e


def obj_of(p):
    """Objective 2*n*max(p*p) for p with sum 1 (equals evaluate_sequence)."""
    n = len(p)
    c = fft_convolve_self(p)
    return 2.0 * n * c.max()


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum x = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def grad_active(p, active, weights):
    """Exact gradient of sum_k weights_k * (p*p)_k  w.r.t. p.

    d(p*p)_k / dp_j = 2 * p_{k-j}  (zero if k-j out of [0,n)).
    """
    n = len(p)
    g = np.zeros(n, dtype=np.float64)
    j = np.arange(n)
    for k, w in zip(active, weights):
        s = k - j
        m = (s >= 0) & (s < n)
        g[m] += (2.0 * w) * p[s[m]]
    return g


def active_weights(c, idx, pow=6.0):
    """Soft weighting of the chosen active lags (emphasises the peak)."""
    vals = c[idx]
    base = np.clip(vals / (vals.max() + 1e-30), 0.0, 1.0)
    w = base ** pow
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones_like(w) / len(w)
    else:
        w = w / wsum
    return w


def resample_profile(p, newn):
    """Resample a unit-sum profile to a different length (kept unit-sum)."""
    old = len(p)
    t_old = np.linspace(0.0, 1.0, old, endpoint=False)
    t_new = np.linspace(0.0, 1.0, newn, endpoint=False)
    q = np.interp(t_new, t_old, p)
    q = np.clip(q, 0.0, None)
    s = q.sum()
    if s < 1e-300:
        q = np.ones(newn) / newn
    else:
        q = q / s
    return q


def parametric_init(n, rng):
    """Asymmetric, mildly oscillatory unit-sum profile (biased, not symmetric)."""
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    a = rng.uniform(0.3, 6.0)
    b = rng.uniform(0.3, 6.0)
    f = rng.uniform(0.4, 3.5)
    e = rng.uniform(0.0, 0.55)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    core = (t + 1e-9) ** a * (1.0 - t + 1e-9) ** b
    core = core * (1.0 + e * np.cos(2.0 * np.pi * f * t + ph))
    core = np.clip(core, 1e-12, None)
    core = core / core.sum()
    return core


def mutate_best(p, rng, scale=0.35):
    """Multiplicative log-noise mutation of a profile, re-normalised."""
    z = np.log(np.clip(p, 1e-12, None))
    z = z + rng.normal(0.0, scale, size=len(z))
    return softmax(z)


# ----------------------------- optimisers -----------------------------

def run_mirror(p0, deadline, rng, topm=4, momentum=0.85, pow=6.0, nstep=None):
    """Negative-entropy mirror descent minimising 2n*max(p*p).

    State is logits z; p = softmax(z).  Sub-gradient from the top active
    lags of the current linear convolution.  Returns (best_obj, best_p).
    """
    n = len(p0)
    p = np.clip(np.asarray(p0, dtype=np.float64), 1e-12, None)
    p = p / p.sum()
    z = np.log(p) - np.log(p.mean())
    v = np.zeros(n, dtype=np.float64)
    best_obj = obj_of(p)
    best_p = p.copy()
    it = 0
    last_improve = 0
    j = np.arange(n)
    while True:
        if time.monotonic() > deadline:
            break
        it += 1
        c = fft_convolve_self(p)
        cur = 2.0 * n * c.max()
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_p = p.copy()
            last_improve = it
        stuck = it - last_improve
        # grow the active set when stuck, with a cap
        m = min(40, topm + stuck // 250)
        idx = np.argsort(c)[::-1][:m]
        w = active_weights(c, idx, pow=pow)
        g = grad_active(p, idx, w)
        g = g * n  # scale: p entries are O(1/n), logits are O(1)
        v = momentum * v + (1.0 - momentum) * g
        lr = 0.55 * np.exp(-0.5e-4 * stuck)
        z = z - lr * v
        z = np.clip(z, -55.0, 55.0)
        p = softmax(z)
        # occasional gentle log-space nudge to escape flat spots
        if stuck > 1500 and it % 400 == 0:
            z = z + rng.normal(0.0, max(1e-3, 0.02 * np.exp(-stuck / 4000.0)), size=n)
    c_f = fft_convolve_self(best_p)
    fo = 2.0 * len(best_p) * c_f.max()
    if fo < best_obj:
        best_obj, best_p = fo, best_p
    return best_obj, best_p


def run_fista_active(p0, deadline, rng, topm=4, pow=6.0):
    """Short Euclidean-projected active-set FISTA polish (sum=1 simplex).

    Uses the exact top-lag sub-gradient; cheap and good for the final squeeze.
    """
    n = len(p0)
    a = np.clip(np.asarray(p0, dtype=np.float64), 1e-12, None)
    a = a / a.sum()
    y = a.copy()
    t_mom = 1.0
    best_obj = obj_of(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    j = np.arange(n)
    while True:
        if time.monotonic() > deadline:
            break
        it += 1
        c = fft_convolve_self(y)
        cur = 2.0 * n * c.max()
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve
        m = min(64, topm + stuck // 300)
        idx = np.argsort(c)[::-1][:m]
        w = active_weights(c, idx, pow=pow)
        g = grad_active(y, idx, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break
        # RGR-type step on the simplex (sum 1)
        step = (2.0 / (1.0 + 0.01 * it)) / (gnorm + 1e-12) * np.sqrt(2.0 * n)
        x = project_simplex(y - step * g, 1.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x + ((t_mom - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        s = y.sum()
        y = y / s if s > 1e-300 else x
        a = x
        t_mom = t_new
        if stuck > 4000 and it % 500 == 0:
            y = project_simplex(y + rng.normal(0.0, 0.004 * np.exp(-stuck / 8000.0), n), 1.0)
            t_mom = 1.0
    c_f = fft_convolve_self(best_a)
    fo = 2.0 * n * c_f.max()
    if fo < best_obj:
        best_obj, best_a = fo, best_a
    return best_obj, best_a
```
