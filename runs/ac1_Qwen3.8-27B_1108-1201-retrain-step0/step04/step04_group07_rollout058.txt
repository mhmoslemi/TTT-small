def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}"""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using projected subgradient descent
with adaptive multi-peak targeting, spectral guidance, and simulated annealing.

Key ideas:
1. Scale invariance: fix sum(a) = sqrt(2n), minimize max(conv(a,a)).
2. Multi-peak subgradient: target top-k convolution peaks simultaneously.
3. Adaptive step size with backtracking.
4. Periodic spectral flattening in frequency domain.
5. Simulated annealing for exploration.
6. Multi-length search (n=8192 primary, n=16384 secondary).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        return np.full(n, S / n)
    # Standard projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Self-convolution via FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    n = len(a)
    return c[:2*n - 1]


def compute_score(a):
    """Compute objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 1 << (2*n - 1).bit_length()
    c = fft_convolve_self(a, N)
    return 2.0 * n * c.max() / (S * S)


def compute_gradient_direct(a, c, top_k):
    """Compute weighted gradient of max(c) using top-k peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx].copy()
    top_vals = np.maximum(top_vals, 0.0)
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), top_idx, np.zeros(top_k)
    weights = top_vals / total
    
    grad = np.zeros(n)
    for i in range(top_k):
        k = int(top_idx[i])
        w = weights[i]
        if w < 1e-15:
            continue
        # grad[m] += 2*w*a[k-m] for valid m
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            indices = k - np.arange(m_lo, m_hi + 1)
            grad[m_lo:m_hi + 1] += 2.0 * w * a[indices]
    
    return grad, top_idx, weights


def gradient_step(a, N, top_k, step_size):
    """One projected gradient step."""
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a, N)
    grad, _, _ = compute_gradient_direct(a, c, top_k)
    
    a_new = project_simplex(a - step_size * grad, S)
    return a_new, c


def spectral_flatten(a, N, alpha):
    """Flatten the magnitude spectrum."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    # Use A*A for convolution (not autocorrelation)
    # The "spectral" quantity related to convolution is A^2, not |A|^2
    # But for flattening purposes, |A|^2 is a reasonable proxy
    # (minimizing max of ifft(|A|^2) correlates with minimizing max of ifft(A^2))
    
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a
    
    ratio = P / P_mean
    # Compress: new_ratio = ratio^(1/alpha) for ratio > 1, ratio^alpha for ratio < 1
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * np.power(ratio[above], 1.0/alpha - 1.0)
    P_new[~above] = P[~above] * np.power(ratio[~above], alpha - 1.0)
    # Actually simpler: P_new = P * ratio^(alpha-1) for all
    # This gives: if ratio > 1, P_new = P * ratio^(alpha-1) < P (since alpha < 1)
    # If ratio < 1, P_new = P * ratio^(alpha-1) > P
    P_new = P * np.power(np.maximum(ratio, 1e-10), alpha - 1.0)
    
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def optimize_subgradient(a_init, n, budget_s, rng, top_k_range=(5, 50), target=1.502):

    """
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2*n - 1).bit_length()
    
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a, N)
    best_score = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    start = time.monotonic()
    eta = 0.5  # initial step size
    patience = 0
    max_patience = 2000
    
    it = 0
    while time.monotonic() - start < budget_s:
        it += 1
        
        # Adaptive top_k: start small, increase
        progress = (time.monotonic() - start) / budget_s
        top_k = min(top_k_range[1], top_k_range[0] + int(progress * (top_k_range[1] - top_k_range[0])))
        
        # Compute gradient and take step
        c = fft_convolve_self(a, N)
        grad, _, _ = compute_gradient_direct(a, c, top_k)
        
        # Adaptive step size: decrease over time
        eta_cur = eta * (1.0 / (1.0 + 0.01 * it * progress))
        
        a_new = project_simplex(a - eta_cur * grad, S)
        
        # Check score
        c_new = fft_convolve_self(a_new, N)
        score_new = 2.0 * n * c_new.max() / (S * S)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            # Slightly increase step if improving
            eta = min(eta * 1.01, 2.0)
        else:
            patience += 1
            eta *= 0.95
        
        a = a_new
        
        # Periodic perturbations
        if patience > 500 and rng.random() < 0.3:
            noise_scale = 0.01 * S / np.sqrt(n) * (1 + patience / 500)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            eta = 0.5
            patience = 0
        
        if patience > max_patience:
            # Larger restart
            a = project_simplex(a + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
            eta = 1.0
            patience = 0
        
        if best_score <= target:
            break
        
        if it % 10000 == 0:
            print("    subgrad it=%d score=%.6f eta=%.4f top_k=%d" % (it, best_score, eta, top_k))
    
    return best_a, best_score

This gives: |A[m]| should be reduced where cos(2*phase(A[m]) + 2*pi*m*k*/N) > 0.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated subgradient
with frequency-domain peak targeting and simulated annealing exploration.

Key innovations over prior approach:
1. Nesterov momentum for faster subgradient convergence
2. Frequency-domain peak reduction: directly modify |FFT(a)| at bins that
   constructively contribute to the target peak lag
3. Exponential peak weighting for sharper focus on dominant peaks
4. Adaptive phase cycling between time-domain and frequency-domain operations
5. Multi-start with the provided sequence as primary initialization

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 1e-15:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Self-convolution via FFT of size N >= 2n-1."""
    A = fft(a, N)
    c = np.real(ifft(A * A))
    n = len(a)
    return c[:2 * n - 1]


def compute_score_from_c(c, n, S):
    """Compute objective from pre-computed convolution."""
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def gradient_topk(a, c, top_k, exp_power=2.0):
    """Compute subgradient using top-k peaks with exponential weighting."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = np.maximum(c[top_idx], 0.0).copy()
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), None
    # Exponential weighting: sharper focus on top peaks
    weights = np.power(top_vals / max(top_vals.max(), 1e-15), exp_power)
    weights /= weights.sum()

    grad = np.zeros(n)
    for i in range(top_k):
        k = int(top_idx[i])
        w = weights[i]
        if w < 1e-12:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            indices = k - np.arange(m_lo, m_hi + 1)
            grad[m_lo:m_hi + 1] += 2.0 * w * a[indices]
    return grad, top_idx


def freq_peak_reduce(a, N, k_target, strength):
    """Reduce convolution peak at lag k_target by modifying frequency magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    ph = np.angle(A)
    mag = np.abs(A)

    # Contribution of bin m to c[k_target]:
    # (1/N) * |A[m]|^2 * cos(2*ph[m] + 2*pi*m*k_target/N)
    # Reduce |A[m]| where this is positive
    m_idx = np.arange(N)
    cos_term = np.cos(2.0 * ph + 2.0 * np.pi * m_idx * k_target / N)

    mag_new = mag.copy()
    mask = cos_term > 0.0
    mag_new[mask] *= (1.0 - strength * cos_term[mask])
    mag_new = np.maximum(mag_new, 0.0)

    A_new = mag_new * np.exp(1j * ph)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_compress(a, N, alpha):
    """Compress the magnitude spectrum toward uniformity."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a
    ratio = np.maximum(P / P_mean, 1e-15)
    # Compress: P_new = P * ratio^(alpha-1), alpha in (0,1)
    P_new = P * np.power(ratio, alpha - 1.0)
    phase = np.angle(A)
    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def normalize_to_S(a, S):
    """Scale a to have sum S, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    Nesterov-accelerated projected subgradient for minimizing max(conv(a,a)).

    N = 1 << (2 * n - 1).bit_length()

    a = normalize_to_S(a_init.copy(), S)
    xm1 = a.copy()  # previous iterate for momentum
    t = 1.0

    c = fft_convolve_self(a, N)
    best_score = compute_score_from_c(c, n, S)
    best_a = a.copy()

    start = time.monotonic()
    eta_base = 1.0
    stall = 0
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        progress = (time.monotonic() - start) / budget_s

        # Adaptive top_k
        top_k = min(top_k_end, top_k_start + int(
            (top_k_end - top_k_start) * min(1.0, progress * 1.5)))

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y = a + beta * (a - xm1)
        y = np.maximum(y, 0.0)

        # Compute gradient at y
        c_y = fft_convolve_self(y, N)
        grad, _ = gradient_topk(y, c_y, top_k, exp_power)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Step size: standard subgradient rate with Nesterov scaling
        step = eta_base * S / (gnorm * t_new)
        # Cap step to avoid overshooting
        step = min(step, S * 0.1 / max(gnorm, 1e-15))

        x_new = project_simplex(y - step * grad, S)

        # Evaluate
        c_new = fft_convolve_self(x_new, N)
        score_new = compute_score_from_c(c_new, n, S)

        if score_new < best_score:
            best_score = score_new
            best_a = x_new.copy()
            stall = 0
            eta_base = min(eta_base * 1.005, 3.0)
        else:
            stall += 1
            if stall % 100 == 0:
                eta_base *= 0.9

        xm1 = a
        a = x_new
        t = t_new

        # Periodic kick to escape local minima
        if stall > 500 and rng.random() < 0.15:
            kick = 0.003 * S / np.sqrt(n) * (1.0 + stall / 500.0)
            a = project_simplex(a + rng.normal(0, kick, n), S)
            xm1 = a.copy()
            t = 1.0
            stall = 0
            eta_base = 1.0

        if best_score <= target:
            break

        if it % 10000 == 0:
            print("      NS it=%d score=%.6f eta=%.3f top_k=%d stall=%d" % (
                it, best_score, eta_base, top_k, stall))

    return best_a, best_score


def freq_domain_optimize(a_init, n, budget_s, rng, target=1.502):
    """
    Frequency-domain optimization: cycle through top peaks, reducing each
    by adjusting frequency magnitudes.
    """
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    a = normalize_to_S(a_init.copy(), S)
    c = fft_convolve_self(a, N)
    best_score = compute_score_from_c(c, n, S)
    best_a = a.copy()

    start = time.monotonic()
    strength = 0.05
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        progress = (time.monotonic() - start) / budget_s

        # Find current top peaks
        c = fft_convolve_self(a, N)
        score = compute_score_from_c(c, n, S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if best_score <= target:
            break

        # Pick the current max peak and reduce it
        k_max = int(np.argmax(c))

        # Anneal strength
        s_cur = strength * (1.0 - 0.5 * progress)

        a_new = freq_peak_reduce(a, N, k_max, s_cur)

        # Also occasionally do spectral compression
        if it % 10 == 0:
            a_new = spectral_compress(a_new, N, alpha=0.7)

        a = a_new

        # Small noise
        if it % 50 == 0:
            a = project_simplex(
                a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

        if it % 5000 == 0:
            print("      FDO it=%d score=%.6f" % (it, best_score))

    return best_a, best_score


def sa_refine(a_init, n, budget_s, rng, target=1.502):
    """
    Simulated annealing with gradient-guided proposal.
    """
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    a = normalize_to_S(a_init.copy(), S)
    c = fft_convolve_self(a, N)
    best_score = compute_score_from_c(c, n, S)
    best_a = a.copy()

    start = time.monotonic()
    T = 0.005
    T_min = 0.0001
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / budget_s
        T_cur = max(T_min, T * (1.0 - progress) ** 2)

        # Proposal: small gradient step + noise
        c = fft_convolve_self(a, N)
        top_k = 5
        grad, _ = gradient_topk(a, c, top_k, exp_power=2.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            grad = np.zeros(n)

        noise_scale = 0.002 * S / np.sqrt(n)
        step = 0.1 * S / max(gnorm, 1e-15)
        a_prop = project_simplex(
            a - step * grad + rng.normal(0, noise_scale, n), S)

        # Evaluate
        c_prop = fft_convolve_self(a_prop, N)
        score_prop = compute_score_from_c(c_prop, n, S)
        score_cur = compute_score_from_c(c, n, S)

        delta = score_prop - score_cur
        if delta < 0 or rng.random() < np.exp(-delta / T_cur):
            a = a_prop
            if score_prop < best_score:
                best_score = score_prop
                best_a = a.copy()

        if best_score <= target:
            break

        if it % 20000 == 0:
            print("      SA it=%d score=%.6f T=%.6f" % (it, best_score, T_cur))

    return best_a, best_score


def final_polish(a_init, n, budget_s, rng, target=1.502):
    """Very careful final polishing with small steps."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    a = normalize_to_S(a_init.copy(), S)
    c = fft_convolve_self(a, N)
    best_score = compute_score_from_c(c, n, S)
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        progress = (time.monotonic() - start) / budget_s

        # Use very small top_k and step
        top_k = 3
        c = fft_convolve_self(a, N)
        grad, _ = gradient_topk(a, c, top_k, exp_power=3.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.02 * S / (gnorm * (1.0 + progress))
        a_new = project_simplex(a - step * grad, S)

        c_new = fft_convolve_self(a_new, N)
        score_new = compute_score_from_c(c_new, n, S)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # Accept: keep going in same direction
            a = a_new
        else:
            # Reject: try a slightly different direction
            a = project_simplex(
                a + rng.normal(0, 0.0005 * S / np.sqrt(n), n), S)

        if best_score <= target:
            break

        if it % 20000 == 0:
            print("      POLISH it=%d score=%.6f" % (it, best_score))

    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("uniform", np.ones(n) * 0.5))
    inits.append(("ramp", np.linspace(0.1, 1.0, n)))
    inits.append(("sin1", 0.5 + 0.4 * np.sin(2 * np.pi * tt)))
    inits.append(("sin3", 0.5 + 0.4 * np.sin(6 * np.pi * tt)))
    inits.append(("chirp", 0.5 + 0.3 * np.sin(2 * np.pi * (4 * tt + 3 * tt**2))))
    inits.append(("multi_sin", 0.5 + 0.2 * np.sin(6 * np.pi * tt) +
                                    0.15 * np.sin(14 * np.pi * tt) +
                                    0.1 * np.sin(26 * np.pi * tt)))
    inits.append(("random", rng.random(n) * 0.7 + 0.15))
    inits.append(("asym", 0.3 + 0.5 * np.exp(-0.5 * ((tt - 0.35) * 4)**2) +
                                  0.2 * np.exp(-0.5 * ((tt - 0.7) * 4)**2)))

    # High-low alternating (exploits convolution structure)
    hl = np.tile([0.8, 0.3], n // 2 + 1)[:n]
    inits.append(("high_low", hl))

    # Smoothed random
    noise = rng.standard_normal(n)
    ks = max(3, n // 16)
    kernel = np.ones(ks) / ks
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(("smooth_rand", 0.2 + 0.6 * smooth))

    return inits


def optimize_single_length(n, budget_s, rng, provided_a=None, target=1.502):
    """Full optimization pipeline for a single sequence length."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    best_score = np.inf
    best_a = None

    t_start = time.monotonic()

    # Build initialization list
    inits = make_initializations(n, rng)

    # Provided sequence first
    if provided_a is not None and len(provided_a) == n:
        inits.insert(0, ("provided", provided_a.copy()))

    # Global best if available at this length
    # (handled by caller)

    n_inits = len(inits)

    # Phase budgets
    # Phase 1: Nesterov subgradient (50% of per-init time)
    # Phase 2: Frequency domain (20%)
    # Phase 3: SA (15%)
    # Phase 4: Final polish (15%)

    for name, arr in inits:
        elapsed = time.monotonic() - t_start
        remaining_total = budget_s - elapsed - 30
        if remaining_total < 20:
            break

        # Allocate time for this init
        inits_remaining = n_inits - inits.index((name, arr))
        per_init = remaining_total / max(inits_remaining, 1)
        per_init = min(per_init, budget_s * 0.15)  # cap per init

        if per_init < 5:
            break

        a = normalize_to_S(np.clip(arr, 0.0, 1000.0), S)

        # Phase 1: Nesterov subgradient
        t1 = min(per_init * 0.5, remaining_total * 0.4)
        if t1 > 3:
            a1, s1 = nesterov_subgradient(a, n, t1, rng, target=target,
                                          exp_power=2.0, top_k_start=3, top_k_end=30)
            if s1 < best_score:
                best_score = s1
                best_a = a1.copy()

        # Phase 2: Frequency domain
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed - 30
        t2 = min(per_init * 0.2, remaining * 0.2)
        if t2 > 3:
            a2, s2 = freq_domain_optimize(a1, n, t2, rng, target=target)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()

        # Phase 3: SA
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed - 30
        t3 = min(per_init * 0.15, remaining * 0.2)
        if t3 > 3:
            a3, s3 = sa_refine(a2, n, t3, rng, target=target)
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()

        # Phase 4: Final polish
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed - 30
        t4 = min(per_init * 0.15, remaining * 0.1)
        if t4 > 2:
            a4, s4 = final_polish(a3, n, t4, rng, target=target)
            if s4 < best_score:
                best_score = s4
                best_a = a4.copy()

        print("    [%s n=%d] best=%.6f (elapsed %.0fs)" % (
            name, n, best_score, time.monotonic() - t_start))

        if best_score <= target:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov subgradient + frequency-domain peak reduction + SA + polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30

    print("Starting optimization, budget=%ds" % int(budget_s))

    global_best = np.inf
    global_a = None

    # Try to use provided sequence
    provided_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            provided_a = np.clip(given, 0, 1000).astype(np.float64)
            print("Using provided sequence, length=%d" % len(provided_a))
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if provided_a is not None:
        lengths.append(len(provided_a))
    if 8192 not in lengths:
        lengths.append(8192)
    if 16384 not in lengths and budget_s > 2000:
        lengths.append(16384)

    # Budget allocation
    total_budget = budget_s - 60
    n_lens = len(lengths)
    budgets = []
    for L in lengths:
        if L == 8192:
            budgets.append(1.0)
        elif L == 16384:
            budgets.append(0.4)
        else:
            budgets.append(0.6)
    tw = sum(budgets)
    budgets = [b / tw * total_budget for b in budgets]

    for i, n in enumerate(lengths):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(budgets[i], remaining - 30)
        if alloc < 20:
            break

        print("=== n=%d, alloc=%.0fs ===" % (n, alloc))

        # Use global best as additional init if it matches this length
        prov = provided_a if (provided_a is not None and len(provided_a) == n) else None

        a_res, score = optimize_single_length(n, alloc, rng,
                                               provided_a=prov, target=1.502)

        if a_res is not None and score < global_best:
            global_best = score
            global_a = a_res.copy()
            print("  New global best: %.6f (n=%d)" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

    # Final extended polish on global best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(global_a)
            print("=== Final polish (budget=%.0fs, n=%d) ===" % (remaining, n_best))
            a_fin, s_fin = final_polish(global_a, n_best, min(remaining - 10, 120.0),
                                        rng, target=1.502)
            if s_fin < global_best:
                global_best = s_fin
                global_a = a_fin.copy()
                print("  After final polish: %.6f" % global_best)

            # One more Nesterov pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_nes, s_nes = nesterov_subgradient(global_a, n_best,
                                                    min(remaining2 - 10, 120.0),
                                                    rng, target=1.502,
                                                    exp_power=3.0,
                                                    top_k_start=2, top_k_end=10)
                if s_nes < global_best:
                    global_best = s_nes
                    global_a = a_nes.copy()
                    print("  After Nesterov pass: %.6f" % global_best)

    # Fallback
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64) * np.sqrt(16384) / 8192
        global_best = 2.0
        print("Using uniform fallback, score=2.0")

    # Verify
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: verification returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback score: %.6f" % global_best)

    print("FINAL: n=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
