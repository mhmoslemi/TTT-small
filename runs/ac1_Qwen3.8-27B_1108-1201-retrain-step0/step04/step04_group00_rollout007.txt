"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. FISTA with adaptive LSE smoothing at original and upsampled sizes
2. L-BFGS-B polish using LSE-smoothed objective for precise local optimization
3. Active-set focused gradient with top-k lag selection
4. Multi-start with diverse initializations to escape local minima
Key improvements: L-BFGS cross-validation, top-k adaptive gradient,
larger search space via upsampled dimensions, and constraint-aware steps.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute normalized LSE weights over convolution lags."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c), cm
    return e / tot, cm


def compute_gradient(y, w):
    """Compute gradient of LSE-smoothed max_conv w.r.t. y."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_weights(c, k, power=1.0):
    """Create weights on top-k lags."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return w_full


def compute_obj(a):
    """Compute objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, step_scale=1.0, use_topk=False, topk_k=8):
    """
    FISTA with adaptive LSE smoothing and optional top-k focus.
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        S_y = y.sum()
        obj_y = 2.0 * n * cur_max / (S_y * S_y) if S_y > 0.01 else np.inf

        if obj_y < best_obj - 1e-14:
            best_obj = obj_y
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient computation
        if use_topk and stuck_count > 500:
            k = min(topk_k, len(c))
            w = topk_weights(c, k, power=1.2)
        else:
            w, _ = lse_weights(c, beta)

        grad = compute_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        if stuck_count > 2000:
            step *= 0.5
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 5000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 30000.0)
            if decay > 0.05:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    return best_a, best_obj


def run_lbfgs(a_init, max_time, beta=50.0):
    """
    L-BFGS-B optimization using LSE-smoothed objective.
    Works in the normalized space (sum = sqrt(2n)).
    Returns (best_a, best_obj).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a0 = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a0 = a0 * (S / a0.sum())

    start = time.monotonic()
    best_obj = compute_obj(a0)
    best_a = a0.copy()
    call_count = [0]
    max_calls = 5000  # Safety limit

    def fun(x):
        call_count[0] += 1
        if time.monotonic() - start > max_time * 0.95 or call_count[0] > max_calls:
            return 1e10
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 0.01:
            return 1e10
        c = fft_convolve_self(x)
        _, cm = lse_weights(c, beta)
        val = cm + np.log((np.exp(beta * (c - cm) - (beta * (c - cm)).max()).sum())) / beta
        obj = 2.0 * n * val / (s * s)
        return obj

    def fun_and_grad(x):
        call_count[0] += 1
        if time.monotonic() - start > max_time * 0.95 or call_count[0] > max_calls:
            return 1e10, np.zeros(n)
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 0.01:
            return 1e10, np.zeros(n)
        c = fft_convolve_self(x)
        w, cm = lse_weights(c, beta)
        val = cm + np.log((np.exp(beta * (c - cm) - (beta * (c - cm)).max()).sum())) / beta
        obj = 2.0 * n * val / (s * s)
        grad = compute_gradient(x, w)
        obj_grad = 2.0 * n / (s * s) * grad - 4.0 * n * val / (s ** 3) * np.ones(n)
        return obj, obj_grad

    try:
        res = scipy_minimize(fun_and_grad, a0, method='L-BFGS-B',
                             bounds=[(0, 1000)] * n,
                             options={'maxiter': max_calls, 'ftol': 1e-12,
                                      'gtol': 1e-10, 'maxfun': max_calls})
        if res.x is not None and res.success is not False:
            a_res = np.clip(res.x, 0, 1000)
            obj_res = compute_obj(a_res)
            if obj_res < best_obj:
                best_obj = obj_res
                best_a = a_res.copy()
    except Exception as e:
        pass

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact subgradient with adaptive top-k.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    start = time.monotonic()

    it = 0
    last_improve = 0

        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        S_y = y.sum()
        obj_y = 2.0 * n * cur_max / (S_y * S_y) if S_y > 0.01 else np.inf

        if obj_y < best_obj - 1e-14:
            best_obj = obj_y
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 3 + stuck // 100)
        else:
            k = min(64, 16 + stuck // 500)

        if k <= 1:
            active = np.argwhere(c == cur_max).flatten()
            w = np.zeros(len(c))
            w[active] = 1.0 / len(active)
        else:
            w = topk_weights(c, k, power=1.0)

        grad = compute_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, S / (0.5 * gnorm + 1e-30))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if stuck > 3000 and it % 200 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample using interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 10:
            a3b, obj3b = run_lbfgs(global_best_a, remaining3, beta=80.0)
            print("  L-BFGS (beta=80): obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Active-set polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Active-set polish at n=%d (budget %.0fs) ===" % (len(global_best_a), p4_budget))
        t4 = time.monotonic()

        a4a, _ = run_active_polish(global_best_a, p4_budget, rng)
        obj4a = compute_obj(a4a)
        print("  Active polish: obj=%.6f (t=%.0fs)" % (obj4a, time.monotonic() - t4))
        if obj4a < global_best_obj:
            global_best_obj = obj4a
            global_best_a = a4a.copy()

    # === Phase 5: Final FISTA with high beta (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.12
        print("\n=== Phase 5: High-beta FISTA (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5a, _ = run_fista(global_best_a, p5_budget * 0.5, rng,
                           beta_init=200.0, beta_rate=0.002,
                           beta_max=50000.0, step_scale=0.3,
                           use_topk=True, topk_k=4)
        obj5a = compute_obj(a5a)
        print("  High-beta: obj=%.6f (t=%.0fs)" % (obj5a, time.monotonic() - t5))
        if obj5a < global_best_obj:
            global_best_obj = obj5a
            global_best_a = a5a.copy()

        remaining5 = p5_budget - (time.monotonic() - t5)
        if remaining5 > 5:
            a5b, _ = run_active_polish(global_best_a, remaining5, rng)
            obj5b = compute_obj(a5b)
            print("  Final active: obj=%.6f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Try downsampled version if we went to 2n ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(global_best_a) > 8192:
        n_ds = len(global_best_a) // 2
        a_ds = downsample(global_best_a, factor=2)
        S_ds = np.sqrt(2.0 * n_ds)
        a_ds = a_ds * (S_ds / a_ds.sum())
        obj_ds = compute_obj(a_ds)
        print("\n=== Phase 6: Downsampled n=%d: obj=%.6f ===" % (n_ds, obj_ds))
        if obj_ds < global_best_obj:
            global_best_obj = obj_ds
            global_best_a = a_ds.copy()
            # Polish
            a_ds2, _ = run_active_polish(a_ds, remaining * 0.5, rng)
            obj_ds2 = compute_obj(a_ds2)
            if obj_ds2 < global_best_obj:
                global_best_obj = obj_ds2
                global_best_a = a_ds2.copy()
                print("  After polish: obj=%.6f" % obj_ds2)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver) and ver < global_best_obj + 0.01:
            global_best_obj = ver
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("\nWARNING: eval=%.6f differs from tracked %.6f" % (ver, global_best_obj))
            if np.isfinite(ver):
                pass  # use as-is
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        global_best_obj = 2.0
        final_list = [float(x) for x in global_best_a]

    print("Total time: %.0fs" % (time.monotonic() - t_total_start))
    return final_list
