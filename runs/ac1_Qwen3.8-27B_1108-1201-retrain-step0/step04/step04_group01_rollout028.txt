"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Log-parameterized Adam optimization with LSE smoothing,
top-k peak shaving, and multi-restart exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient_a(a, c, beta):
    """Gradient of LSE-smoothed max of conv(a,a) with respect to a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of w^T b with respect to a, where b = conv(a,a)
    # db_k/da_i = a_{i-k} + a_{i+k} (with boundary handling)
    # Equivalent to 2 * conv(w, a) truncated to length n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient_a(a, c, k=16):
    """Subgradient from top-k autocorrelation lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.ones(k)  # uniform weight on top-k
    w_full = np.zeros(len(c))
    w_full[idx] = weights / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Combines log-parameterized Adam optimization with LSE smoothing,
spectral flattening (Gerchberg-Saxton), direct peak shaving, and multi-restart
exploration. Uses the provided height_sequence_1 as primary starting point.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient_a(a, c, beta):
    """Gradient of LSE-smoothed max of conv(a,a) with respect to a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient_a(a, c, k=16):
    """Subgradient from top-k autocorrelation lags, weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 1e-10)
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_flatten(a, n_iter=200, n=0):
    """
    Gerchberg-Saxton style spectral flattening.
    Tries to make |A_k| constant for k != 0 while keeping a >= 0.
    """
    if n == 0:
        n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    a = np.asarray(a, dtype=np.float64).copy()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        # Target: |A_k| = constant for k != 0
        pwr = np.abs(A)
        dc = pwr[0]
        target = pwr[1:].mean()
        # Blend: partially flatten
        phase = A / (np.abs(A) + 1e-300)
        new_mag = np.abs(A).copy()
        new_mag[1:] = 0.7 * new_mag[1:] + 0.3 * target
        new_mag[0] = dc  # keep DC
        A_new = new_mag * phase
        a = np.real(ifft(A_new, fsize))[:2*n-1]
        # Take the first n elements (or pad if needed)
        a = a[:n]
        # Project to non-negative
        a = np.maximum(a, 0.0)
    
    return a


def optimize_log_adam(a_init, max_time, rng, beta_init=20.0, beta_growth=0.003,
                      beta_max=50000.0, lr_init=0.04, use_topk=False, topk_interval=1000):
    """
    Optimize in log-parameterization using Adam.
    a = exp(x), minimize LSE-smoothed max of conv(a,a).
    Scale-invariant, so we fix mean(x)=0.
    """
    n = len(a_init)
    a = np.clip(a_init, 1e-10, 1000.0)
    x = np.log(a)
    x = x - x.mean()
    
    m1 = np.zeros(n)
    v1 = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    def safe_obj(a):
        s = a.sum()
        if s < 1e-15:
            return np.inf
        b = fft_convolve_self(a)
        return 2.0 * n * b.max() / (s * s)
    
    best_obj = safe_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * t))
        
        a = np.exp(x)
        # Numerical safety
        if not np.all(np.isfinite(a)):
            x = np.log(np.clip(a_init, 1e-10, 1000.0))
            x = x - x.mean()
            a = np.exp(x)
            m1 = np.zeros(n)
            v1 = np.zeros(n)
            t = 0
            continue
        
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (a.sum() ** 2)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Gradient in a-space
        if use_topk and t % topk_interval == 0:
            grad_a = topk_gradient_a(a, c, k=min(32, n // 64))
        else:
            grad_a = lse_gradient_a(a, c, beta)
        
        # Chain rule to x-space
        grad_x = a * grad_a
        
        # Adam update
        m1 = b1 * m1 + (1 - b1) * grad_x
        v1 = b2 * v1 + (1 - b2) * grad_x ** 2
        m1_hat = m1 / (1.0 - b1 ** t)
        v1_hat = v1 / (1.0 - b2 ** t)
        
        lr = lr_init / (1.0 + 0.0005 * t)
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        if no_improve > 2000:
            # Restart with perturbation
            noise = rng.normal(0, 0.01, n)
            x = x + noise
            x = x - x.mean()
            m1 = np.zeros(n)
            v1 = np.zeros(n)
            t = 0
            no_improve = 0
            continue
        
        x = x - lr * m1_hat / (np.sqrt(v1_hat) + eps)
        x = x - x.mean()
        
        # Prevent extreme x ranges
        if x.max() - x.min() > 30:
            x = 0.5 * x
            m1 = np.zeros(n)
            v1 = np.zeros(n)
            t = 0
    
    # Convert back
    a_final = np.exp(best_a - best_a.max())
    a_final = np.clip(a_final, 0.0, 1000.0)
    if a_final.sum() < 0.01:
        a_final = np.ones(n) / n
    return a_final, best_obj


def optimize_peak_shave(a_init, max_time, rng):
    """
    Direct peak shaving using top-k subgradient.
    Works in a-space with simplex projection.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    
    best_obj = objective_value(a)
    best_a = a.copy()
    start = time.monotonic()
    
    step = 2.0 / n
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (a.sum() ** 2)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Use top-k gradient with increasing k
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = max(4, int(4 + 60 * progress))
        k = min(k, n // 8)
        
        grad = topk_gradient_a(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step
        eff_step = step / (1.0 + 0.001 * it)
        if no_improve > 500:
            eff_step *= 0.5
            no_improve = 0
        if no_improve > 3000:
            # Perturb and restart
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            step = 2.0 / n
            no_improve = 0
            continue
        
        a_new = project_simplex(a - eff_step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() <= c.max() + 1e-10:
            a = a_new
        else:
            # Halve step
            step *= 0.8
            if step < 1e-12:
                step = 2.0 / n
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_a, best_obj


def optimize_spectral(a_init, max_time, rng):
    """
    Spectral flattening + refinement.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    
    best_obj = objective_value(a)
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        # Spectral flattening step
        t_spectral = min(5.0, (max_time - (time.monotonic() - start)) * 0.3)
        a_spec = spectral_flatten(a, n_iter=50)
        a_spec = normalize_to_sum(a_spec, S)
        
        # Refine with a few Adam steps
        x = np.log(np.maximum(a_spec, 1e-10))
        x = x - x.mean()
        m1 = np.zeros(n)
        v1 = np.zeros(n)
        t = 0
        b1, b2 = 0.9, 0.999
        beta = 50.0
        
        for _ in range(200):
            if time.monotonic() - start >= max_time - 5:
                break
            t += 1
            a_curr = np.exp(x)
            c = fft_convolve_self(a_curr)
            cur_obj = 2.0 * n * c.max() / (a_curr.sum() ** 2)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a_curr.copy()
            
            grad_a = lse_gradient_a(a_curr, c, beta)
            grad_x = a_curr * grad_a
            
            m1 = b1 * m1 + (1 - b1) * grad_x
            v1 = b2 * v1 + (1 - b2) * grad_x ** 2
            m1_hat = m1 / (1.0 - b1 ** t)
            v1_hat = v1 / (1.0 - b2 ** t)
            
            lr = 0.02
            x = x - lr * m1_hat / (np.sqrt(v1_hat) + 1e-8)
            x = x - x.mean()
            beta *= 1.01
        
        # Check if improved
        a_curr = np.exp(x)
        cur_obj = 2.0 * n * fft_convolve_self(a_curr).max() / (a_curr.sum() ** 2)
        if cur_obj >= best_obj * 1.01:
            # Not improving much, perturb
            a = normalize_to_sum(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
    
    return best_a, best_obj


def generate_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())
    
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(1.0 + 0.15 * np.sin(2 * np.pi * 3 * tt) + 0.1 * np.cos(2 * np.pi * 7 * tt + 0.5))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    # Multi-frequency
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 2 * tt) + 0.15 * np.cos(2 * np.pi * 5 * tt + 1.0)
                + 0.1 * np.sin(2 * np.pi * 9 * tt + 2.0))
    
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Combines log-Adam, spectral flattening, and peak shaving with
    multi-restart and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            given_obj = objective_value(given)
            print("Given sequence: length=%d, obj=%.6f" % (len(given), given_obj))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    # Determine primary length
    primary_n = 8192
    if given is not None:
        primary_n = len(given)
    
    # Time allocation
    total_time = deadline - time.monotonic()
    
    # Phase 1: Optimize at primary length (60% of time)
    phase1_time = total_time * 0.55
    print("\n=== Phase 1: Primary optimization (n=%d, %.0fs) ===" % (primary_n, phase1_time))
    
    inits = []
    if given is not None and len(given) == primary_n:
        inits.append(given.copy())
    gen_inits = generate_inits(primary_n, rng)
    inits.extend(gen_inits[:4])
    
    best_a_p1 = None
    best_obj_p1 = np.inf
    
    # Strategy A: Log-Adam from best init (40% of phase1)
    t_a = phase1_time * 0.40
    best_init = inits[0]
    a_a, obj_a = optimize_log_adam(best_init, t_a, rng,
                                   beta_init=20.0, beta_growth=0.003,
                                   beta_max=30000.0, lr_init=0.04)
    print("  Log-Adam A: obj=%.6f" % obj_a)
    if obj_a < best_obj_p1:
        best_obj_p1 = obj_a
        best_a_p1 = a_a.copy()
    
    # Strategy B: Log-Adam with top-k mixing (25% of phase1)
    if time.monotonic() < deadline - 60:
        t_b = min(phase1_time * 0.25, deadline - time.monotonic() - 60)
        start_b = best_a_p1 if best_a_p1 is not None else best_init
        a_b, obj_b = optimize_log_adam(start_b, t_b, rng,
                                       beta_init=100.0, beta_growth=0.005,
                                       beta_max=80000.0, lr_init=0.02,
                                       use_topk=True, topk_interval=500)
        print("  Log-Adam B (topk): obj=%.6f" % obj_b)
        if obj_b < best_obj_p1:
            best_obj_p1 = obj_b
            best_a_p1 = a_b.copy()
    
    # Strategy C: Peak shaving (20% of phase1)
    if time.monotonic() < deadline - 30:
        t_c = min(phase1_time * 0.20, deadline - time.monotonic() - 30)
        start_c = best_a_p1 if best_a_p1 is not None else best_init
        a_c, obj_c = optimize_peak_shave(start_c, t_c, rng)
        print("  Peak shave: obj=%.6f" % obj_c)
        if obj_c < best_obj_p1:
            best_obj_p1 = obj_c
            best_a_p1 = a_c.copy()
    
    # Strategy D: Spectral flattening (15% of phase1)
    if time.monotonic() < deadline - 20:
        t_d = min(phase1_time * 0.15, deadline - time.monotonic() - 20)
        start_d = best_a_p1 if best_a_p1 is not None else best_init
        a_d, obj_d = optimize_spectral(start_d, t_d, rng)
        print("  Spectral: obj=%.6f" % obj_d)
        if obj_d < best_obj_p1:
            best_obj_p1 = obj_d
            best_a_p1 = a_d.copy()
    
    if best_a_p1 is not None:
        n_chk = len(best_a_p1)
        obj_chk = objective_value(best_a_p1)
        if obj_chk < global_best:
            global_best = obj_chk
            global_a = best_a_p1.copy()
            print("  Phase 1 best: obj=%.6f (n=%d)" % (global_best, n_chk))
    
    # Phase 2: Try secondary lengths (25% of time)
    phase2_time = total_time * 0.25
    secondary_lengths = [4096, 16384]
    if primary_n not in secondary_lengths:
        pass  # already tried
    
    for n_sec in secondary_lengths:
        if time.monotonic() > deadline - 45:
            break
        t_sec = min(phase2_time / len(secondary_lengths), deadline - time.monotonic() - 30)
        if t_sec < 20:
            continue
        
        print("\n=== Phase 2: n=%d (%.0fs) ===" % (n_sec, t_sec))
        
        # Get init: upscale/downscale from global best
        if global_a is not None:
            if len(global_a) > n_sec:
                ratio = len(global_a) // n_sec
                init_sec = global_a[:n_sec * ratio].reshape(n_sec, ratio).mean(axis=1)
            else:
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_sec)
                init_sec = np.interp(t_new, t_old, global_a)
        else:
            init_sec = np.ones(n_sec) * 0.1
        
        a_sec, obj_sec = optimize_log_adam(init_sec, t_sec * 0.6, rng,
                                           beta_init=20.0, beta_growth=0.003,
                                           beta_max=30000.0, lr_init=0.04)
        print("  Secondary log-Adam: obj=%.6f" % obj_sec)
        
        remaining_sec = deadline - time.monotonic() - 15
        if remaining_sec > 10 and obj_sec < global_best:
            a_sec2, obj_sec2 = optimize_peak_shave(a_sec, remaining_sec * 0.5, rng)
            print("  Secondary peak shave: obj=%.6f" % obj_sec2)
            if obj_sec2 < obj_sec:
                a_sec = a_sec2
                obj_sec = obj_sec2
        
        obj_sec_chk = objective_value(a_sec)
        if obj_sec_chk < global_best:
            global_best = obj_sec_chk
            global_a = a_sec.copy()
            print("  New global best: obj=%.6f (n=%d)" % (global_best, n_sec))
    
    # Phase 3: Final intensive polishing (20% of time)
    phase3_time = total_time * 0.20
    if global_a is not None and time.monotonic() < deadline - 20:
        t_pol = min(phase3_time, deadline - time.monotonic() - 15)
        n_pol = len(global_a)
        print("\n=== Phase 3: Final polishing (n=%d, %.0fs) ===" % (n_pol, t_pol))
        
        # Very high beta log-Adam
        t_p3a = t_pol * 0.5
        a_p3a, obj_p3a = optimize_log_adam(global_a, t_p3a, rng,
                                           beta_init=500.0, beta_growth=0.002,
                                           beta_max=200000.0, lr_init=0.01)
        print("  High-beta log-Adam: obj=%.6f" % obj_p3a)
        if obj_p3a < global_best:
            global_best = obj_p3a
            global_a = a_p3a.copy()
        
        # Peak shave
        t_p3b = min(t_pol * 0.3, deadline - time.monotonic() - 15)
        if t_p3b > 5:
            a_p3b, obj_p3b = optimize_peak_shave(global_a, t_p3b, rng)
            print("  Final peak shave: obj=%.6f" % obj_p3b)
            if obj_p3b < global_best:
                global_best = obj_p3b
                global_a = a_p3b.copy()
        
        # One more spectral pass
        t_p3c = min(t_pol * 0.2, deadline - time.monotonic() - 10)
        if t_p3c > 5:
            a_p3c, obj_p3c = optimize_spectral(global_a, t_p3c, rng)
            print("  Final spectral: obj=%.6f" % obj_p3c)
            if obj_p3c < global_best:
                global_best = obj_p3c
                global_a = a_p3c.copy()
    
    # Final verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        print("  No valid result found, using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
    
    final_obj = objective_value(global_a)
    print("Final result: length=%d, obj=%.6f" % (len(global_a), final_obj))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
