import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

def cross_corr(w, a, fsize=None):
    """Linear cross-correlation of w and a, first len(a) elements."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]

def qp_gradient(a, p, fsize):
    """Gradient of a^T Q_p a = 2 * (Q_p a) = 4 * cross_corr(p, a)"""
    return 4.0 * cross_corr(p, a, fsize)

def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Solve min a^T Q_p a s.t. a >= 0, sum(a) = S using FISTA."""
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t = 1.0
    
    # Estimate Lipschitz constant L of the QP gradient
    if L is None:

        v = np.random.randn(len(a))
        v = project_simplex(v, S)
        for _ in range(20):

        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return a

"""
Minimize 2*n*max_conv/sum^2 using adaptive FISTA with Polyak step sizes,
multi-scale active-set tracking, and targeted deflation.
Key improvements: no counterproductive upsampling, Polyak-guided steps,
 aggressive high-beta polishing, and near-size exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def cross_corr_weights(w, a, fsize=None):
    """Compute sum_k w[k] * a[k-j] + a[j-k] (gradient of weighted conv)."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_weights(c, beta):
    """Compute LSE softmax weights over lags."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def topk_weights(c, k, tol_frac=0.0):
    """Weights on the top-k lags (or all lags within tol_frac of max)."""
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        k_eff = len(active)
    else:
        k_eff = min(k, len(c))
        active = np.argsort(c)[::-1][:k_eff]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    return w


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value 2*n*max_c/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_optimize(a_init, max_time, rng, beta_init=20.0, beta_rate=0.012,
                   beta_max=50000.0, use_polyak=True, f_lb=1.0,
                   topk_switch=500, step_scale=1.0, restart_interval=3000):
    """
    FISTA with adaptive LSE smoothing and optional Polyak step size.
    Returns (best_a, best_conv_max).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient: hybrid LSE + topk
        if stuck_count > topk_switch and (it % 4 == 0 or stuck_count > topk_switch + 2000):
            k = 1 if stuck_count > topk_switch + 5000 else 3
            if stuck_count > topk_switch + 10000:
                k = 5
            w = topk_weights(c, k, tol_frac=0.005 if k > 1 else 0.0)
        else:
            w = lse_weights(c, beta)

        grad = cross_corr_weights(w, y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size selection
        if use_polyak:
            obj_cur = cur_max  # normalized: sum^2 = 2n, so obj = 2n*cur_max/(2n) = cur_max
            # Actual normalized objective: 2n*cur_max/S^2 = cur_max (since S^2=2n)
            gap = obj_cur - f_lb
            if gap > 1e-10:
                step = step_scale * gap / (gnorm * gnorm)
                step = min(step, step_scale / max(beta * 0.1, 1.0))
                step = max(step, 1e-10 / gnorm)
            else:
                step = step_scale / max(beta, 1.0)
        else:
            step = step_scale / max(beta, 1.0)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with structured noise
        if stuck_count > restart_interval and it % 200 == 0:
            decay = np.exp(-stuck_count / (2.0 * restart_interval))
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it

        # Hard restart from best if very stuck
        if stuck_count > 4 * restart_interval:
            y = best_a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y, S)
            a = y.copy()
            t_mom = 1.0
            stuck_count = 0
            last_improve = it

    return best_a, best_conv

def game_inspired_optimize(a_init, max_time, rng):
    """
    Game-theoretic approach: alternate between optimizing a (inner QP)
    and updating lag weights (outer exponential weights).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    csize = 2 * n - 1
    fsize = 1 << (csize + n - 1).bit_length()

    p = np.ones(csize) / csize
    best_conv = np.inf
    best_a = a.copy()
    start = time.monotonic()
    eta = 0.5
    it = 0
    inner_iters = 8

    while time.monotonic() - start < max_time:
        it += 1

        # Inner: solve min a^T Q_p a s.t. a>=0, sum(a)=S
        # Use a few FISTA steps
        a_inner = a.copy()
        y_inner = a_inner.copy()
        t_inner = 1.0
        step_inner = 1.0 / (2.0 * n)  # conservative initial step

        for _ in range(inner_iters):
            # Gradient of a^T Q_p a is 2*Q_p*a = 2*cross_corr(p, a) * 2
            grad = 2.0 * cross_corr_weights(p, y_inner, fsize=fsize)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Backtracking: reduce step if needed
            step = step_inner
            x_trial = project_simplex(y_inner - step * grad, S)
            # Check if objective decreased
            obj_trial = x_trial @ np.zeros(n)  # placeholder, just project
            x_inner = x_trial
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_inner * t_inner))
            y_inner = x_inner + ((t_inner - 1.0) / t_new) * (x_inner - a_inner)
            y_inner = np.maximum(y_inner, 0.0)
            a_inner = x_inner
            t_inner = t_new

        a = project_simplex(a_inner, S)

        # Evaluate
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()

        # Outer: update p via exponential weights
        c_shifted = c - c.max()
        p *= np.exp(eta * c_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(csize) / csize
        else:
            p /= p_sum

        # Decay eta slowly
        eta *= 0.9995

    return best_a, best_conv


def resize_sequence(a, n_new):
    """Resize sequence to new length via linear interpolation, renormalize."""
    n_old = len(a)
    S_new = np.sqrt(2.0 * n_new)
    t_old = np.linspace(0, 1, n_old, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(t_new, t_old, a.values if hasattr(a, 'values') else a)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S_new / s)
    else:
        a_new = np.ones(n_new) * S_new / n_new
    return a_new

        # Run 2: Higher initial beta, different schedule
        a1b, _ = fista_optimize(global_best_a, remaining1 * 0.5, rng,
                                 beta_init=100.0, beta_rate=0.005,
                                 beta_max=80000.0, use_polyak=True,
                                 step_scale=0.7, topk_switch=300)
        obj1b = compute_obj(a1b)
        print("  FISTA-2: obj=%.6f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

        remaining1b = p1_budget - (time.monotonic() - t1)
        if remaining1b > 15:
            # Run 3: Game-inspired
            a1c, _ = game_inspired_optimize(global_best_a, remaining1b, rng)
            obj1c = compute_obj(a1c)
            print("  Game: obj=%.6f (t=%.0fs)" % (obj1c, time.monotonic() - t1))
            if obj1c < global_best_obj:
                global_best_obj = obj1c
                global_best_a = a1c.copy()

    # === Phase 2: Active-set polish (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.30
        n_best = len(global_best_a)
        print("\n=== Phase 2: Active-set polish at n=%d (budget %.0fs) ===" % (n_best, p2_budget))
        t2 = time.monotonic()

        a2, _ = active_set_polish(global_best_a, p2_budget * 0.5, rng)
        obj2 = compute_obj(a2)
        print("  Polish-1: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 10:
            # Second polish with different parameters
            a2b, _ = fista_optimize(global_best_a, remaining2, rng,
                                     beta_init=500.0, beta_rate=0.001,
                                     beta_max=100000.0, use_polyak=True,
                                     step_scale=0.3, topk_switch=100)
            obj2b = compute_obj(a2b)
            print("  Polish-2: obj=%.6f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Near-size exploration (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Near-size exploration (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        # Try a few nearby sizes
        try_sizes = [n0 - 200, n0 + 200]
        # Filter to valid sizes
        try_sizes = [s for s in try_sizes if 1000 <= s <= 20000]
        # Remove duplicates and the original size
        try_sizes = [s for s in try_sizes if s != n0]

        per_size = p3_budget / max(len(try_sizes), 1)
        for n_try in try_sizes:
            if time.monotonic() > deadline - 30:
                break
            a_try = resize_sequence(global_best_a, n_try)
            obj_try = compute_obj(a_try)
            if obj_try < global_best_obj:
                print("  n=%d initial: obj=%.6f" % (n_try, obj_try))
                global_best_obj = obj_try
                global_best_a = a_try.copy()

            # Brief optimization
            a_try2, _ = fista_optimize(a_try, per_size * 0.6, rng,
                                         beta_init=50.0, beta_rate=0.01,
                                         beta_max=20000.0, use_polyak=True,
                                         step_scale=0.8)
            obj_try2 = compute_obj(a_try2)
            if obj_try2 < global_best_obj:
                global_best_obj = obj_try2
                global_best_a = a_try2.copy()
                print("  n=%d after FISTA: obj=%.6f" % (n_try, obj_try2))

        print("  Phase 3 done (t=%.0fs)" % (time.monotonic() - t3))

    # === Phase 4: Final intensive polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p4_budget = remaining * 0.80
        n_best = len(global_best_a)
        print("\n=== Phase 4: Final polish at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        a4, _ = active_set_polish(global_best_a, p4_budget * 0.6, rng)
        obj4 = compute_obj(a4)
        print("  Final-1: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

        remaining4 = p4_budget - (time.monotonic() - t4)
        if remaining4 > 5:
            # Last-ditch FISTA with very high beta
            a4b, _ = fista_optimize(global_best_a, remaining4, rng,
                                     beta_init=1000.0, beta_rate=0.0,
                                     beta_max=1000.0, use_polyak=False,
                                     step_scale=0.15, topk_switch=50)
            obj4b = compute_obj(a4b)
            print("  Final-2: obj=%.6f (t=%.0fs)" % (obj4b, time.monotonic() - t4))
            if obj4b < global_best_obj:
                global_best_obj = obj4b
                global_best_a = a4b.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    total_time = time.monotonic() - (deadline - budget_s + 30.0)
    print("Total time: %.0fs" % total_time)
    return final_list
