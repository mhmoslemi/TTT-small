import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.maximum(v, 0.0)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(u, top_indices, weights, n):
    """Compute weighted subgradient for top lags."""
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        k = int(k)
        w = weights[i]
        # grad[m] += 2 * w * u[k-m] for valid m
        # Valid m: 0 <= m < n and 0 <= k-m < n
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            # u[k-m] for m in [m_lo, m_hi]
            # k-m ranges from k-m_lo to k-m_hi
            idx = k - np.arange(m_lo, m_hi + 1)
            grad[m_lo:m_hi+1] += 2.0 * w * u[idx]
    return grad

def subgradient_step(u, eta, top_k, n, softmax_temp=1.0):
    """One subgradient step with top-k lags and softmax weighting."""
    c = fft_convolve_self(u)
    c_max = c.max()
    
    # Get top-k lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Softmax weighting (sharper = more focus on top)
    if softmax_temp > 0:
        exp_vals = np.exp(top_vals / softmax_temp - top_vals.max() / softmax_temp)
        weights = exp_vals / exp_vals.sum()
    else:
        # Use excess above average
        avg = c.sum() / len(c)
        excess = np.maximum(top_vals - avg, 1e-30)
        weights = excess / excess.sum()
    
    # Compute subgradient
    grad = compute_subgradient(u, top_indices, weights, n)
    
    # Step
    u_new = u - eta * grad
    u_new = project_simplex(u_new, 1.0)
    
    return u_new, c_max, top_indices

    top_vals = c[top_indices]

        weights = exp_vals / exp_vals.sum()

        weights = excess / excess.sum()

    u_new = np.exp(log_u_new)
    u_new /= u_new.sum()
    
    return u_new, c_max

def objective_from_cmax(cmax, n):
    return 2.0 * n * cmax


def fast_subgrad_step(u, n, top_k, eta, temp_scale):
    c = fft_convolve_self(u)
    c_max = float(c.max())
    if c_max < 1e-30:
        return u, c_max
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    temp = max(c_max * temp_scale, 1e-15)
    shifted = top_vals / temp - top_vals[0] / temp
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    grad = compute_multi_lag_gradient(u, sorted_idx, weights, n)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-30:
        return u, c_max
    step = eta * gnorm
    u_new = project_simplex(u - step * grad / gnorm, 1.0)
    return u_new, c_max


def kl_mirror_step(u, n, top_k, eta, temp_scale):
    c = fft_convolve_self(u)
    c_max = float(c.max())
    if c_max < 1e-30:
        return u, c_max
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    temp = max(c_max * temp_scale, 1e-15)
    shifted = top_vals / temp - top_vals[0] / temp
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    grad = compute_multi_lag_gradient(u, sorted_idx, weights, n)
    log_u = np.log(np.maximum(u, 1e-40)) - eta * grad
    log_u -= log_u.max()
    u_new = np.exp(log_u)
    s = u_new.sum()
    if s < 1e-30:
        u_new = np.ones(n) / n
    else:
        u_new /= s
    return u_new, c_max


def convex_combo_step(u, n, top_k, eta, temp_scale, p_kl=0.4):
    c = fft_convolve_self(u)
    c_max = float(c.max())
    if c_max < 1e-30:
        return u, c_max
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    temp = max(c_max * temp_scale, 1e-15)
    shifted = top_vals / temp - top_vals[0] / temp
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    grad = compute_multi_lag_gradient(u, sorted_idx, weights, n)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-30:
        return u, c_max
    step = eta * gnorm
    u_eucl = project_simplex(u - step * grad / gnorm, 1.0)
    log_u = np.log(np.maximum(u, 1e-40)) - step
    log_u -= log_u.max()
    u_kl = np.exp(log_u)
    s = u_kl.sum()
    if s < 1e-30:
        u_kl = np.ones(n) / n
    else:
        u_kl /= s
    u_new = (1.0 - p_kl) * u_eucl + p_kl * u_kl
    u_new = project_simplex(u_new, 1.0)
    return u_new, c_max


def peak_shave_step(u, n, num_peaks=5, shave_frac=0.1):
    c = fft_convolve_self(u)
    c_max = float(c.max())
    avg = c.sum() / len(c)
    ratio = c_max / max(avg, 1e-30)
    if ratio < 1.05:
        return u, c_max, ratio
    sorted_idx = np.argsort(c)[::-1][:num_peaks]
    for k in sorted_idx:
        k = int(k)
        if c[k] <= avg * 1.02:
            break
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo > m_hi:
            continue
        contribs = np.zeros(n)
        idx = k - np.arange(m_lo, m_hi + 1)
        contribs[m_lo:m_hi + 1] = u[m_lo:m_hi + 1] * u[idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-30:
            continue
        target_reduce = shave_frac * (c[k] - avg)
        if target_reduce <= 0:
            continue
        top_positions = np.argsort(contribs)[::-1][:min(20, m_hi - m_lo + 1)]
        mass_to_move = 0.0
        for pos in top_positions:
            pos = int(pos)
            if contribs[pos] < 1e-30:
                continue
            frac = min(shave_frac, target_reduce / max(contribs[pos] * n, 1e-30))
            delta = u[pos] * frac
            u[pos] -= delta
            mass_to_move += delta
            target_reduce -= delta * contribs[pos] / max(u[pos] + delta, 1e-30) if u[pos] + delta > 0 else 0
            if target_reduce <= 0:
                break
        if mass_to_move > 1e-15:
            low_contrib_mask = contribs < np.percentile(contribs[m_lo:m_hi + 1], 25)
            low_positions = np.where(low_contrib_mask)[0]
            if len(low_positions) > 0:
                per_pos = mass_to_move / len(low_positions)
                for lp in low_positions:
                    u[lp] += per_pos
            else:
                u[m_lo:m_hi + 1] += mass_to_move / (m_hi - m_lo + 1)
        u = project_simplex(u, 1.0)
    c_new = fft_convolve_self(u)
    return u, float(c_new.max()), ratio


def multi_scale_optimize(u_init, n, max_time, rng, target_obj=1.48):
    u = np.asarray(u_init, dtype=np.float64).copy()
    u = np.clip(u, 0, None)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(n) / n
    else:
        u = u / s

    best_u = u.copy()
    best_cmax = float(fft_convolve_self(u).max())
    best_obj = 2.0 * n * best_cmax

    eta = 0.3
    top_k = 20
    temp_scale = 0.3
    stall = 0
    it = 0
    start = time.monotonic()
    cooldown = 0

    while time.monotonic() - start < max_time:
        it += 1

        top_k = min(300, 20 + it // 50)
        eta = 0.3 / (1.0 + 0.02 * it)
        temp_scale = 0.3 * (1.0 + 0.1 * (it // 200))

        if cooldown > 0:
            cooldown -= 1
            continue

        if it % 7 == 0:
            u_new, c_max = peak_shave_step(u, n, num_peaks=min(10, top_k // 5), shave_frac=0.15)
        else:
            u_new, c_max = convex_combo_step(u, n, top_k, eta, temp_scale, p_kl=0.3)

        c_check = fft_convolve_self(u_new)
        actual_cmax = float(c_check.max())
        actual_obj = 2.0 * n * actual_cmax

        if actual_obj < best_obj:
            best_obj = actual_obj
            best_u = u_new.copy()
            u = u_new.copy()
            best_cmax = actual_cmax
            stall = 0
            if it % 2000 == 0:
                print("    iter %d: obj=%.6f" % (it, best_obj))
        else:
            u = u_new.copy()
            stall += 1

        if stall > 30:
            noise = rng.normal(0, 0.02 / np.sqrt(n), n)
            u = project_simplex(u + noise, 1.0)
            stall = 0
            cooldown = 1

        if stall > 150:
            noise = rng.normal(0, 0.1 / np.sqrt(n), n)
            u = project_simplex(u + noise, 1.0)
            stall = 0
            cooldown = 5
            eta = max(eta * 2, 0.1)

        if best_obj <= target_obj:
            break

    return best_u * np.sqrt(2.0 * n), best_obj


def coordinate_descent(u, n, max_time, rng, top_k_coords=30):
    a = u.copy()
    S = a.sum()
    start = time.monotonic()
    best_cmax = float(fft_convolve_self(a).max())
    best_a = a.copy()
    stall = 0

    multipliers = [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 1.8, 0.1]

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = float(c.max())
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 20:
            break

        peak_lag = int(np.argmax(c))
        m_lo = max(0, peak_lag - n + 1)
        m_hi = min(peak_lag, n - 1)
        if m_lo > m_hi:
            break

        contribs = np.zeros(n)
        idx = peak_lag - np.arange(m_lo, m_hi + 1)
        contribs[m_lo:m_hi + 1] = a[m_lo:m_hi + 1] * a[idx]

        top_coords = np.argsort(contribs)[::-1][:top_k_coords]
        improved = False

        for ci in range(len(top_coords)):
            if time.monotonic() - start > max_time:
                break
            m = int(top_coords[ci])
            j = peak_lag - m
            if j < 0 or j >= n or j == m:
                continue
            orig = a[m]
            if orig < 1e-15:
                continue
            best_frac = 1.0
            best_local_cmax = c_max
            for frac in multipliers:
                new_val = orig * frac
                delta = new_val - orig
                a[m] = new_val
                affected_lo = max(0, m)
                affected_hi = min(2 * n - 2, m + n - 1)
                c_local = c.copy()
                for k in range(affected_lo, affected_hi + 1):
                    j2 = k - m
                    if 0 <= j2 < n:
                        c_local[k] += delta * a[j2]
                    j3 = m
                    k2 = k - j3
                    if 0 <= k2 < n and k2 != j2:
                        c_local[k] += delta * a[k2]
                new_cmax = float(c_local.max())
                if new_cmax < best_local_cmax:
                    best_local_cmax = new_cmax
                    best_frac = frac
            a[m] = orig * best_frac
            if best_frac != 1.0:
                improved = True

        a = project_simplex(a, S)

        if not improved:
            top5 = top_coords[:5]
            for m in top5:
                a[int(m)] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)
            stall += 1

    return best_a, float(2.0 * n * best_cmax)


def spectral_equalize(u, n, N, max_time, rng, alpha_init=0.2):
    """Reduce peaks by shaping the spectrum to flatten the convolution."""
    a = u.copy()
    S = a.sum()
    best_a = a.copy()
    best_cmax = float(fft_convolve_self(a).max())
    it = 0
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        c_max = float(c.max())
        avg = c.sum() / len(c)
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        if c_max / max(avg, 1e-30) < 1.01:
            break

        target = avg * 1.05
        top_idx = np.where(c > target)[0]
        if len(top_idx) == 0:
            top_idx = np.argsort(c)[::-1][:5]

        weights = np.maximum(c[top_idx] - target, 0)
        if weights.sum() < 1e-30:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights /= weights.sum()

        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)

        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        alpha = alpha_init / (1.0 + 0.05 * it)
        P_new = np.abs(A) ** 2 * np.exp(-alpha * np.maximum(cos_eff, 0))
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        c_check = fft_convolve_self(a_new)
        if c_check.max() < c_max:
            a = a_new

    return best_a, float(2.0 * n * best_cmax)


def try_length(a_init, n, max_time, rng, target_obj=1.48):
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()

    u = np.asarray(a_init, dtype=np.float64).copy()
    u = np.clip(u, 0, None)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(n) / n
    else:
        u = u / s

    # Phase 1: Main subgradient optimization (60% time)
    t0 = time.monotonic()
    phase1_time = max_time * 0.55
    best_a, best_obj = multi_scale_optimize(u, n, phase1_time, rng, target_obj)
    print("    P1: obj=%.6f (t=%.0fs)" % (best_obj, time.monotonic() - t0))

    # Phase 2: Spectral equalization (20% time)
    u_best = best_a / S
    t1 = time.monotonic()
    phase2_time = max(5.0, max_time * 0.2)
    best_a2, best_obj2 = spectral_equalize(u_best, n, N, phase2_time, rng)
    if best_obj2 < best_obj:
        best_a = best_a2
        best_obj = best_obj2
        print("    P2 (spectral): obj=%.6f (t=%.0fs)" % (best_obj, time.monotonic() - t1))

    # Phase 3: Coordinate descent polish (25% time)
    u_best = best_a / S
    t2 = time.monotonic()
    phase3_time = max(5.0, max_time * 0.25)
    best_a3, best_obj3 = coordinate_descent(u_best, n, phase3_time, rng, top_k_coords=40)
    if best_obj3 < best_obj:
        best_a = best_a3
        best_obj = best_obj3
        print("    P3 (coord): obj=%.6f (t=%.0fs)" % (best_obj, time.monotonic() - t2))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to use previous best
    prev_a = None
    prev_n = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and np.sum(np.clip(given, 0, 1000)) > 0:
            prev_a = np.clip(given, 0, 1000)
            prev_n = len(prev_a)
            print("Using previous best: n=%d" % prev_n)
    except Exception:
        pass

    # Determine which lengths to try
    candidates = []

    if prev_a is not None and prev_n >= 1024:
        candidates.append((prev_n, prev_a, "prev"))

    # Add random initializations for n=8192
    n_main = 8192
    inits = []

    # Sinusoidal
    t = np.arange(n_main) / n_main
    a1 = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * np.sin(2 * np.pi * 11 * t) \
        + 0.1 * np.sin(2 * np.pi * 23 * t) + 0.05 * rng.standard_normal(n_main)
    inits.append(("sin", np.clip(a1, 0.01, None)))

    # Asymmetric ramp + noise
    ramp = 1.0 + 0.5 * np.linspace(0, 1, n_main)
    a2 = ramp + 0.3 * rng.standard_normal(n_main)
    inits.append(("ramp", np.clip(a2, 0.01, None)))

    # Bimodal
    a3 = np.exp(-0.5 * ((t - 0.3) * 5) ** 2) * 0.7 + np.exp(-0.5 * ((t - 0.7) * 5) ** 2) * 0.3 + 0.1 * rng.standard_normal(n_main)
    inits.append(("bimodal", np.clip(a3, 0.01, None)))

    # Pure random
    inits.append(("rand", 0.1 + 0.9 * rng.random(n_main)))

    # Concentrated (mass on one side)
    a5 = np.exp(-np.linspace(0, 3, n_main)) + 0.05 * rng.standard_normal(n_main)
    inits.append(("conc", np.clip(a5, 0.01, None)))

    # High-frequency oscillation (reduces autocorrelation)
    a6 = 0.5 + 0.4 * np.sin(2 * np.pi * 37 * t + 0.5) + 0.1 * rng.standard_normal(n_main)
    inits.append(("osc", np.clip(a6, 0.01, None)))

    for name, arr in inits:
        candidates.append((n_main, arr, name))

    # Also try n=16384 with previous best (upsampled) or random
    if prev_a is not None:
        n_big = 16384
        # Interpolate previous best to larger size
        from numpy import interp
        x_old = np.linspace(0, 1, len(prev_a))
        x_new = np.linspace(0, 1, n_big)
        interp_a = interp(x_new, x_old, prev_a)
        interp_a = np.clip(interp_a, 0.01, None)
        candidates.append((n_big, interp_a, "prev_interp"))

    print("Total candidates: %d" % len(candidates))

    # Allocate time: give more time to previous best and main length
    # Strategy: quick eval of all, then focus on best
    quick_time = min(15.0, (deadline - time.monotonic()) * 0.05)
    per_quick = quick_time / len(candidates) if len(candidates) > 0 else 0

    print("=== Quick evaluation phase ===")
    for cn, carr, cname in candidates:
        if time.monotonic() > deadline - 60:
            break
        carr_c = np.clip(carr, 0, 1000).astype(np.float64)
        c = fft_convolve_self(carr_c)
        c_max = float(c.max())
        s = carr_c.sum()
        if s < 0.01:
            continue
        obj = 2.0 * cn * c_max / (s ** 2)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = carr_c.copy()
            print("  [%s] n=%d: obj=%.6f" % (cname, cn, obj))

    print("Quick phase best: %.6f" % global_best_obj)

    # Now focus remaining time on the best candidates
    remaining = deadline - time.monotonic()
    print("Remaining time: %.0fs" % remaining)

    # Sort candidates by their quick score (recompute if needed)
    scored = []
    for cn, carr, cname in candidates:
        carr_c = np.clip(carr, 0, 1000).astype(np.float64)
        c = fft_convolve_self(carr_c)
        c_max = float(c.max())
        s = carr_c.sum()
        if s < 0.01:
            continue
        obj = 2.0 * cn * c_max / (s ** 2)
        scored.append((obj, cn, carr_c, cname))

    scored.sort(key=lambda x: x[0])

    # Allocate remaining time to top candidates
    top_n_candidates = min(5, len(scored))
    # Give 50% to best, 25% to second, etc.
    allocations = []
    total_alloc = 0
    for i in range(top_n_candidates):
        if i == 0:
            alloc = remaining * 0.40
        elif i == 1:
            alloc = remaining * 0.25
        elif i == 2:
            alloc = remaining * 0.15
        elif i == 3:
            alloc = remaining * 0.10
        else:
            alloc = remaining * 0.07
        alloc = min(alloc, remaining - 30.0)
        allocations.append(max(10.0, alloc))
        total_alloc += alloc

    # Scale if total exceeds remaining
    scale_factor = min(1.0, (remaining - 30.0) / total_alloc) if total_alloc > 0 else 1.0
    allocations = [max(10.0, a * scale_factor) for a in allocations]

    print("=== Focused optimization phase ===")
    for i in range(top_n_candidates):
        if time.monotonic() > deadline - 30:
            break
        obj_init, cn, carr, cname = scored[i]
        alloc = allocations[i]
        print("  Optimizing [%s] n=%d (init obj=%.6f, time=%.0fs)" % (cname, cn, obj_init, alloc))

        best_a, best_obj = try_length(carr, cn, alloc, rng, target_obj=1.48)

        if best_obj < global_best_obj:
            global_best_obj = best_obj
            global_best_a = best_a.copy()
            print("  => New best: %.6f (from %s, n=%d)" % (global_best_obj, cname, cn))

    # Final: coordinate descent on the overall best
    if global_best_a is not None and time.monotonic() < deadline - 10:
        final_time = deadline - time.monotonic() - 5
        if final_time > 15:
            nf = len(global_best_a)
            Sf = global_best_a.sum()
            u_final = global_best_a / Sf
            print("=== Final coordinate descent (t=%.0fs) ===" % final_time)
            best_a_f, best_obj_f = coordinate_descent(u_final, nf, final_time, rng, top_k_coords=50)
            if best_obj_f < global_best_obj:
                global_best_obj = best_obj_f
                global_best_a = best_a_f * Sf
                print("  Final coord: %.6f" % global_best_obj)

    # Verify and return
    if global_best_a is None:
        global_best_a = np.ones(256, dtype=np.float64)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: FAILED")
    if not np.isfinite(ver):
        print("  Using fallback")
        global_best_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver)

    print("Returning n=%d, obj=%.6f" % (len(final_list), global_best_obj))
    return final_list
