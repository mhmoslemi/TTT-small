Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: FISTA with exact subgradient + Nesterov restart + interleaved
spectral flattening + coordinate descent polish. Try multiple lengths.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def exact_subgradient_wall(a, c):
    """Subgradient of max(c) where c = f*f. Returns gradient w.r.t. f."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def subgradient_uniform(a, c):
    """Uniform subgradient: average over all argmax positions."""
    n = len(a)
    max_val = c.max()
    # Find all positions where c is close to max
    tol = 1e-6 * max_val
    peaks = np.where(c >= max_val - tol)[0]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += a[k - idx]
    grad *= 2.0 / len(peaks)
    return grad

    a = normalize_sum(a_init, n)

    # For uniform a, grad norm = 2*sqrt(n*(S/n)^2) = 2*S/sqrt(n) = 2*sqrt(2)/n^0.25...

        else:

def spectral_step(a, n, S, alpha=0.3):
    """One step of spectral flattening."""
    fsize = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(fsize)
    a_pad[:n] = a
    F = fft(a_pad)
    mag = np.abs(F)
    phase = F / np.maximum(mag, 1e-30)
    P = mag ** 2
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a
    P_new = (1.0 - alpha) * P / P_mean + alpha
    mag_new = np.sqrt(P_new)
    mag_new = mag_new * np.sqrt(P.sum() / max(P_new.sum(), 1e-30))
    F_new = phase * mag_new
    a_new = np.real(ifft(F_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) * S / n
    else:
        a_new = a_new * (S / s)
    return a_new

                continue

                val = c_try.max()
                if val < best_val:
                    best_val = val
                    best_delta = dlt
            
            if best_delta != 0 and best_val < best - 1e-12:
                a[j] += best_delta
                a[m] -= best_delta
                c += best_delta * d
                c[0] += 2.0 * (a[j] - a[m] - best_delta) * best_delta + 2.0 * best_delta ** 2
                # Recompute c[0] properly
                # Actually let's just track best and recompute periodically
                if best_val < best:
                    best = best_val
                    best_a = a.copy()
                improved = True
        
        # Periodic full recompute
        if it % 100 == 0 or True:  # recompute every pass with improvements
            pass
        
        # Add noise if stuck

            it += 1

Strategy: Gerchberg-Saxton spectral flattening + FISTA with exact subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    """Subgradient of max(c) where c = a*a. Gradient w.r.t. a."""
    n = len(a)
