We want to minimize: `2 * n * max(f*f) / sum(f)^2`

where `f*f` is the full convolution of `f` with itself.

The key insight is that we want to minimize the maximum value of the auto-correlation while keeping the sum fixed. This is related to "low autocorrelation" sequences.

The previous approach achieved 1.507018 and the target is 1.5027.

2. **Key insight**: The optimum should have the property that the max of f*f is achieved at many points (ideally, the "flat" ones). If we can enforce that many lag values are all equal to the maximum, we might get better solutions.

3. **Different approach - direct optimization of the max**: Instead of LSE approximation, we could try a proximal method that directly targets the max. Or we could use a "water-filling" approach where we try to equalize the peak values.

4. **Another idea**: The problem is equivalent to finding a probability distribution (after normalization) that minimizes n * max(f*f). This is like finding a distribution with minimal peak autocorrelation.

5. **Important observation**: For the uniform distribution of length n, f*f has a triangular shape with max = sum(f)^2/n. So the objective would be 2*n*(sum^2/n)/sum^2 = 2. We're doing much better than 2 already.

So we want to minimize n * max(p*p) where p is a non-negative sequence with sum 1.

24. **Let me think about what the optimal sequence looks like**:

    For the uniform sequence, f*f is a triangle. The peak is at lag 0.
    
    To reduce the peak, we need to reduce the "concentration" of f. But f >= 0 and sum(f) is fixed, so we can't make f too spread out.

26. **Let me also think about the step size more carefully**:

    After solving the LP and getting g, we want to find the best t in [0,1] such that (1-t)f + tg has a smaller max((1-t)f+tg)*((1-t)f+tg)).

    Actually, since the LP guarantees that f*g <= M (the current max), we know that for the new point h = (1-t)f + tg:
    (h*h)(k) = (1-t)^2(f*f)(k) + 2t(1-t)(f*g)(k) + t^2(g*g)(k)
    
    The first two terms are bounded by (1-t)^2 M + 2t(1-t) M = M(1-t+t) = M(1-t). Wait:
    (1-t)^2 M + 2t(1-t) M = M[(1-t)^2 + 2t(1-t)] = M(1-t)[(1-t) + 2t] = M(1-t)(1+t) = M(1-t^2)

    In practice, g is obtained from the LP which constrains f*g <= M. Since f and g are both non-negative with the same sum, and f*g is bounded, g can't be too different from f. So max(g*g) should be close to M.

"""
Minimize 2*n*max(f*f)/sum(f)^2 using LP best-response iterations with full constraints.
Uses scipy HiGHS solver for speed, with periodic FISTA polishing and multi-scale exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix
import copy


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_objective(a):
    """Compute the objective value 2*n*max(f*f)/sum(f)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def solve_lp_full(f, n, S, M):
    """
    Solve the LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Uses scipy linprog with HiGHS.
    Returns g or None.
    """
    # Build constraint matrix A (2n-1 x n) where A[k, j] = f[k-j] for valid (k,j)
    # (f*g)(k) = sum_j f[k-j] * g[j] for valid j
    # Valid j: max(0, k-n+1) <= j <= min(k, n-1)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max sum(g) s.t. A @ g <= M (vector of M's), g >= 0
    # Convert to min: min -sum(g) s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n,
                  method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            g = g * (S / g_sum)
            return g
    return None


def solve_lp_topk(f, n, S, M, top_k=200):
    """
    Solve LP with only top-k most binding constraints.
    """
    c = fft_convolve_self(f)
    idx = np.argsort(c)[::-1][:top_k]
    
    rows = []
    cols = []
    vals = []
    b = np.zeros(top_k)
    
    for i, k in enumerate(idx):
        b[i] = M
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csc_matrix((vals, (rows, cols)), shape=(top_k, n))
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            g = g * (S / g_sum)
            return g
    return None


def line_search_step(f, g, n, S, M_current):
    """
    Find the best step size t for h = (1-t)f + t*g.
    Returns (best_t, best_val, best_h).
    """
    best_t = 0.0
    best_val = M_current
    best_h = f.copy()
    
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
        h = (1 - t) * f + t * g
        h = np.maximum(h, 0)
        h = project_simplex(h, S)
        c_h = fft_convolve_self(h)
        val = c_h.max()
        if val < best_val:
            best_val = val
            best_t = t
            best_h = h.copy()
    
    return best_t, best_val, best_h

def run_fista_polish(f_init, n, S, max_time, rng, beta_init=50.0, beta_max=2000.0):
    """FISTA polishing with LSE."""
    f = normalize_sum(f_init, n)
    y = f.copy()
    t_fista = 1.0
    start = time.monotonic()
    best = fft_convolve_self(f).max()
    best_f = f.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.002 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_f = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t_fista = t_new
    
    return best_f, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max(f*f)/sum(f)^2 using hybrid LP best-response (HiGHS, full/top-k constraints)
with FISTA LSE polishing, peak-shaving, and multi-scale exploration.
Key improvements: full-constraint LP for accuracy, aggressive line search, adaptive noise,
and a "peak balancing" heuristic that directly targets the top convolution lags.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c) with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_objective(a):
    """Compute 2*n*max(f*f)/sum(f)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def build_conv_matrix_rows(f, k_list, n):
    """Build sparse rows for constraints (f*g)(k) for k in k_list."""
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(k_list):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    return rows, cols, vals


def solve_lp_best_response(f, n, S, M, k_list, method='highs'):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in k_list, g >= 0.
    Returns normalized g or None.
    """
    rows, cols, vals = build_conv_matrix_rows(f, k_list, n)
    if len(vals) == 0:
        return None
    A = csc_matrix((vals, (rows, cols)), shape=(len(k_list), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(k_list), M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
        if res.status == 0:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g * (S / g_sum)
    except Exception:
        pass
    return None


def line_search(f, g, n, S, M_current, candidate_ts=None):
    """Find best t in candidate_ts for h=(1-t)f+tg minimizing max(h*h)."""
    if candidate_ts is None:
        candidate_ts = [1.0, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
    best_t = 0.0
    best_val = M_current
    best_h = f.copy()
    for t in candidate_ts:
        if t <= 0:
            continue
        h = (1.0 - t) * f + t * g
        h = np.maximum(h, 0.0)
        h = project_simplex(h, S)
        c_h = fft_convolve_self(h)
        val = c_h.max()
        if val < best_val:
            best_val = val
            best_t = t
            best_h = h.copy()
    return best_t, best_val, best_h


def run_lp_phase(f_init, n, S, max_time, rng, top_k=400, full_constraints=False,
                 max_iter=300, noise_freq=15, noise_scale_base=0.005):
    """
    LP best-response iterations with periodic noise.
    """
    f = normalize_sum(f_init, n)
    start = time.monotonic()
    c = fft_convolve_self(f)
    best_M = c.max()
    best_f = f.copy()
    it = 0
    last_improve = 0
    stall_count = 0
    
    all_lags = list(range(2 * n - 1))
    
    while time.monotonic() - start < max_time and it < max_iter:
        it += 1
        elapsed = time.monotonic() - start
        c = fft_convolve_self(f)
        M_current = c.max()
        
        if M_current < best_M - 1e-12:
            best_M = M_current
            best_f = f.copy()
            last_improve = it
            stall_count = 0
            if it % 3 == 1:
                obj = 2.0 * n * best_M / (best_f.sum() ** 2)
                print(f"    LP it {it}: M={best_M:.8f}, obj={obj:.6f} (t={elapsed:.0f}s)")
        else:
            stall_count += 1
        
        # Stall handling: inject noise
        if stall_count >= noise_freq:
            scale = noise_scale_base * S / np.sqrt(n) * (1.0 + 0.5 * np.sin(0.3 * it))
            f = project_simplex(f + rng.normal(0, scale, n), S)
            stall_count = 0
            continue
        
        # Select constraints
        if full_constraints:
            k_list = all_lags
        else:
            # Use top-k lags by current convolution value, plus lag 0 always
            c_vals = fft_convolve_self(f)
            top_idx = np.argsort(c_vals)[::-1][:top_k]
            if 0 not in top_idx:
                top_idx = np.concatenate([top_idx[:top_k - 1], [0]])
            k_list = list(top_idx)
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_best_response(f, n, S, M_current, k_list)
        lp_time = time.monotonic() - t0
        
        if g is None:
            print(f"    LP it {it}: solver failed ({lp_time:.1f}s)")
            stall_count += 3
            continue
        
        # Check if g is substantially different from f
        diff = np.linalg.norm(g - f) / max(np.linalg.norm(f), 1e-10)
        if diff < 0.005:
            # LP thinks we're at a fixed point; perturb
            scale = noise_scale_base * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, scale, n), S)
            stall_count += 2
            continue
        
        # Line search
        best_t, best_val, best_h = line_search(f, g, n, S, M_current)
        
        if best_t > 0 and best_val < M_current:
            f = best_h
            print(f"    LP it {it}: M {M_current:.6f} -> {best_val:.6f} (t={best_t:.2f}, lp={lp_time:.1f}s)")
        else:
            # Try a very small step
            small_t, small_val, small_h = line_search(f, g, n, S, M_current,
                                                       candidate_ts=[0.02, 0.01])
            if small_t > 0 and small_val < M_current:
                f = small_h
            else:
                stall_count += 1
    
    return best_f, best_M, it


def run_fista(f_init, n, S, max_time, rng, beta_init=30.0, beta_rate=0.003, beta_max=3000.0,
              repulsion_freq=300, repulsion_lr=0.04):
    """FISTA with LSE smoothing and periodic repulsion."""
    f = normalize_sum(f_init, n)
    y = f.copy()
    t_fista = 1.0
    start = time.monotonic()
    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_f = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        # Adaptive: if stuck, reduce step and add noise
        if it - last_improve > 2000:
            step *= 0.5
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_fista = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t_fista = t_new
        
        # Periodic multiplicative repulsion
        if repulsion_freq > 0 and it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(30.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 0:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 1e-15)
                y = project_simplex(y, S)
                t_fista = 1.0
    
    return best_f, best

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

        print(f"\n=== Phase 1: Combined opt (n={n1}, time={t1:.0f}s) ===")
        a1, m1 = run_combined(global_a, n1, S1, t1, rng)
        obj1 = compute_objective(a1)
        print(f"  Phase 1: obj={obj1:.8f}")
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_a = a1
    
    # Phase 2: Extended LP on best so far
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n2 = len(global_a)
        S2 = np.sqrt(2.0 * n2)
        t2 = min(remaining * 0.4, 500.0)
        print(f"\n=== Phase 2: Extended LP (n={n2}, time={t2:.0f}s) ===")
        a2a, m2a, it2a = run_lp_phase(global_a, n2, S2, t2 * 0.5, rng,
                                        top_k=500, max_iter=100,
                                        noise_freq=10, noise_scale_base=0.003)
        obj2a = compute_objective(a2a)
        print(f"  Phase 2a (top-500, {it2a} it): obj={obj2a:.8f}")
        if obj2a < global_best_obj:
            global_best_obj = obj2a
            global_a = a2a
        
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            # Try full constraints (slower)
            t2b = min(remaining2 * 0.4, 200.0)
            a2b, m2b, it2b = run_lp_phase(global_a, n2, S2, t2b, rng,
                                           top_k=0, full_constraints=True, max_iter=30,
                                           noise_freq=8, noise_scale_base=0.002)
            obj2b = compute_objective(a2b)
            print(f"  Phase 2b (full, {it2b} it): obj={obj2b:.8f}")
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_a = a2b
    
    # Phase 3: Peak shaving + FISTA polish
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        t3 = min(remaining * 0.3, 120.0)
        print(f"\n=== Phase 3: Shave+Polish (n={n3}, time={t3:.0f}s) ===")
        
        # Peak shaving
        a3a = peak_shaving(global_a, n3, S3, num_peaks=30, lr=0.015, iters=200)
        obj3a = compute_objective(a3a)
        print(f"  Shaving: obj={obj3a:.8f}")
        if obj3a < global_best_obj:
            global_best_obj = obj3a
            global_a = a3a
        
        # FISTA polish
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a3b, m3b = run_fista(global_a, n3, S3, remaining3 * 0.6, rng,
                                  beta_init=500.0, beta_rate=0.0005, beta_max=8000.0,
                                  repulsion_freq=0)
            obj3b = compute_objective(a3b)
            print(f"  Polish: obj={obj3b:.8f}")
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_a = a3b
    
    # Phase 4: Try larger n
    remaining = deadline - time.monotonic()
    if remaining > 150 and global_a is not None:
        cur_n = len(global_a)
        alt_n = cur_n * 2 if cur_n <= 8192 else cur_n
        if alt_n > cur_n and alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            t4 = min(remaining * 0.5, 300.0)
            print(f"\n=== Phase 4: Larger n={alt_n} (time={t4:.0f}s) ===")
            a4, m4 = run_combined(alt_init, alt_n, S_alt, t4, rng)
            obj4 = compute_objective(a4)
            print(f"  Phase 4: obj={obj4:.8f}")
            if obj4 < global_best_obj:
                global_best_obj = obj4
                global_a = a4
    
    # Phase 5: Diverse restarts
    remaining = deadline - time.monotonic()
    if remaining > 90 and global_a is not None:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        t5_total = remaining * 0.5
        print(f"\n=== Phase 5: Diverse restarts (time={t5_total:.0f}s) ===")
        
        tt = np.linspace(0, 1, n5, endpoint=False)
        inits = [
            np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * 5 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt), 0.01),
            np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.5) + 0.2 * np.cos(2 * np.pi * 11 * tt), 0.01),
            np.exp(-3.0 * (tt - 0.25) ** 2) + 0.7 * np.exp(-3.0 * (tt - 0.75) ** 2) + 0.3,
            np.maximum(0.8 + 0.4 * np.sin(2 * np.pi * 17 * tt + 2.0), 0.01),
        ]
        
        per_init = t5_total / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 15:
                break
            init = np.maximum(init, 0.01)
            ai, mi = run_fista(init, n5, S5, per_init, rng,
                                beta_init=20.0, beta_rate=0.004, beta_max=1000.0,
                                repulsion_freq=200, repulsion_lr=0.06)
            obji = compute_objective(ai)
            print(f"  Diverse {i}: obj={obji:.8f}")
            if obji < global_best_obj:
                global_best_obj = obji
                global_a = ai
    
    # Final: one more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n6 = len(global_a)
        S6 = np.sqrt(2.0 * n6)
        t6 = remaining * 0.7
        print(f"\n=== Final LP (time={t6:.0f}s) ===")
        a6, m6, it6 = run_lp_phase(global_a, n6, S6, t6, rng,
                                     top_k=400, max_iter=50,
                                     noise_freq=15, noise_scale_base=0.002)
        obj6 = compute_objective(a6)
        print(f"  Final: obj={obj6:.8f}")
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_a = a6
    
    # Final verification
    print(f"\n{'='*50}")
    print(f"Best objective: {global_best_obj:.8f}")
    print(f"Sequence length: {len(global_a)}")
    
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print(f"evaluate_sequence: {ver:.8f}")
    
    if not np.isfinite(ver):
        print("  Rejected! Using fallback.")
        n_fb = 512
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"  Fallback obj: {ver:.8f}")
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 512
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
    
    print(f"Returning length={len(global_a)}, obj={compute_objective(global_a):.8f}")
    return [float(x) for x in global_a]
