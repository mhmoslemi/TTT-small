"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using refined Nesterov subgradient
descent with soft top-k weighting and spectral-domain polishing.

Improvements over previous approach:
1. More efficient gradient computation (direct, no FFT for gradient).
2. Soft top-k weighting with exponential decay for smoother landscape.
3. Adaptive step size with restart mechanism.
4. Periodic spectral domain "flattening" to escape local minima.
5. Multi-start with the provided sequence and structured alternatives.
6. Block-refinement for final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2 * n * c.max() / (S * S))


def subgradient_topk(a, c, k, temperature=1.0):
    """
    Compute weighted subgradient using top-k lags with soft weighting.
    Returns (gradient, peak_value).
    """
    n = len(a)
    # Get top-k indices
    # For efficiency, use argpartition for large arrays
    if k < len(c):
        top_idx = np.argpartition(c, -k)[-k:]
    else:
        top_idx = np.arange(len(c))
    
    # Sort by value descending
    order = np.argsort(c[top_idx])[::-1]
    top_idx = top_idx[order]
    top_vals = c[top_idx]
    
    # Soft weighting: use exponential weighting based on temperature
    # Higher temperature = more uniform weighting
    max_val = top_vals[0] if top_vals[0] > 0 else 1.0
    weights = np.exp((top_vals - max_val) / (temperature * max_val + 1e-30))
    weights /= weights.sum()
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for i in range(k):
        K = int(top_idx[i])
        w = weights[i]
        if w < 1e-10:
            continue
        if K < n:
            grad[:K+1] += 2.0 * w * a[K::-1][:K+1]
        else:
            lo = K - n + 1
            grad[lo:n] += 2.0 * w * a[lo:n][::-1]
    
    return grad, max_val


def nesterov_subgradient(a_init, n, S, max_time, rng, k=20, base_temp=1.0,
                          step_scale=1.0, noise_freq=200, noise_scale=0.005):
    """
    Nesterov-accelerated subgradient method for minimizing max(c).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()  # Nesterov lookahead
    t_param = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()
    
    # Adaptive parameters
    patience = 0
    max_patience = 5000
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute convolution of y
        A = fft(y, fsize)
        c = np.real(ifft(A * A))[:2*n-1]
        
        cur_score = 2.0 * n * c.max() / (S * S)
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
        
        # Periodic restart
        if patience > max_patience:
            y = project_simplex(y + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            t_param = 1.0
            patience = 0
        
        # Compute subgradient
        grad, peak_val = subgradient_topk(y, c, k, temperature=base_temp / (1 + 0.01 * it))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        step = step_scale * S / (np.sqrt(n) * np.sqrt(t_param))
        
        # Projected gradient step
        x = project_simplex(y - step * grad / gnorm, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x + ((t_param - 1.0) / t_new) * (x - y)
        y = np.maximum(y, 0.0)
        # Re-normalize
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y * (S / s_y)
        
        a = x
        t_param = t_new
        
        # Periodic noise injection
        if it % noise_freq == 0 and rng.random() < 0.3:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t_param = 1.0
    
    return best_a, best_score


def spectral_flatten(a, N, alpha, n_lags=20):
    """
    Spectral flattening step: reduce the peak of the autocorrelation
    by adjusting the power spectrum.
    
    Returns modified a (same sum).
    """
    n = len(a)
    S = a.sum()
    
    # Current convolution
    c = fft_convolve_self(a)
    
    # Find top lags
    top_idx = np.argpartition(c, -n_lags)[-n_lags:]
    
    # For each top lag K, we want to reduce c[K].
    # In frequency domain: c[K] = (1/N) * sum_m |A(m)|^2 * exp(2*pi*i*m*K/N)
    # To reduce c[K], we want to reduce |A(m)|^2 where cos(2*pi*m*K/N) > 0
    
    # Combined "pressure" from all top lags
    pressure = np.zeros(N)
    A = fft(a, N)
    P = np.abs(A) ** 2
    m = np.arange(N) * 2 * np.pi / N
    
    for K in top_idx:
        K = int(K)
        pressure += np.real(np.exp(1j * m * K))
    
    # Reduce P where pressure > 0, increase where pressure < 0
    P_new = P.copy()
    pressure_norm = pressure / (np.abs(pressure).max() + 1e-30)
    P_new = P * (1.0 - alpha * pressure_norm)
    P_new = np.maximum(P_new, 0.0)
    
    # Reconstruct with same phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Project
    a_new = project_simplex(a_new, S)
    return a_new, c


def spectral_polish(a, N, n_steps, alpha_init, rng, target_obj=1.50):
    """Multiple spectral flattening steps with annealing."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_steps):
        progress = it / max(n_steps - 1, 1)
        alpha = max(0.005, alpha_init * np.exp(-3.0 * progress))
        
        a_new, c = spectral_flatten(a, N, alpha, n_lags=min(20, 2*n-1))
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if best_score <= target_obj:
            break
        
        # Add noise periodically
        if it % 100 == 50 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_score

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using Nesterov-accelerated subgradient
descent with soft top-k weighting, adaptive step sizes, spectral-domain polishing,
and multi-start strategy.

Key improvements:

2. Softmax-weighted top-k subgradient for smoother optimization landscape.
3. Adaptive Nesterov momentum with automatic reset on stagnation.
4. Alternating time-domain (subgradient) and frequency-domain (water-filling) updates.
5. Multi-start from provided sequence and structured alternatives.
6. Final intensive polishing phase.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2 * n * c.max() / (S * S))


def compute_subgradient(a, n, c, top_k, temp):
    """
    Compute weighted subgradient from top-k peaks of convolution.
    For lag K, subgradient of c[K] w.r.t. a[i] = 2*a[K-i] if 0<=K-i<n else 0.
    Uses softmax weighting over top-k peaks.
    """
    size = len(c)
    if top_k < size:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        top_idx = np.arange(size)
    top_vals = c[top_idx]
    
    # Softmax weighting
    max_val = top_vals.max()
    if max_val <= 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        exponents = (top_vals - max_val) / (temp + 1e-30)
        weights = np.exp(exponents)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights /= wsum
    
    # Compute weighted subgradient directly
    grad = np.zeros(n)
    for i in range(top_k):
        w = weights[i]
        if w < 1e-10:
            continue
        K = int(top_idx[i])
        if K < n:
            # i ranges from 0 to K, grad[i] += 2*w*a[K-i]
            grad[:K + 1] += 2.0 * w * a[K::-1][:K + 1]
        else:
            lo = K - n + 1
            # i ranges from lo to n-1, grad[i] += 2*w*a[K-i]
            # K - i for i=lo is n-1, for i=n-1 is K-n+1=lo
            grad[lo:n] += 2.0 * w * a[lo:n][::-1]
    
    return grad


def spectral_waterfill_step(a, N, n_lags=10, alpha=0.1):
    """
    One step of spectral water-filling: reduce the power spectrum at frequencies
    that contribute positively to the peak lags, increase at others.
    """
    n = len(a)
    S = a.sum()
    
    c = fft_convolve_self(a)
    if n_lags < len(c):
        top_idx = np.argpartition(c, -n_lags)[-n_lags:]
    else:
        top_idx = np.arange(len(c))
    
    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    m = np.arange(N)
    # Compute "pressure" from top lags
    pressure = np.zeros(N)
    for K in top_idx:
        K = int(K)
        pressure += np.cos(2.0 * np.pi * m * K / N)
    
    # Normalize pressure
    pmax = np.abs(pressure).max()
    if pmax < 1e-30:
        return a
    
    pressure_norm = pressure / pmax
    
    # Modify spectrum: reduce where pressure is positive, increase where negative
    P_new = P * (1.0 - alpha * pressure_norm)
    P_new = np.maximum(P_new, 0.0)
    
    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    
    return a_new

def spectral_polish(a_init, N, max_time, rng, n_lags=15, alpha_init=0.2):
    """
    Iterative spectral water-filling with annealing.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = it / max(1, int((time.monotonic() - start) / 10))
        alpha = max(0.01, alpha_init * np.exp(-2.0 * min(progress, 3.0)))
        
        a_new = spectral_waterfill_step(a, N, n_lags=n_lags, alpha=alpha)
        
        # Check improvement
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Keep a, add small noise
            if it % 50 == 0:
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
        
        if best_score < 1.50:
            break
    
    return best_a, best_score

    starts.append(("uniform", np.ones(n) * S / n))
    
    # 3. Ramp (linear increase)
    tt = np.linspace(0, 1, n, endpoint=False)
    ramp = 0.1 + 0.9 * tt
    starts.append(("ramp", ramp * S / ramp.sum()))
    
    # 4. Inverse ramp
    iramp = 0.1 + 0.9 * (1.0 - tt)
    starts.append(("iramp", iramp * S / iramp.sum()))
    
    # 5. Multi-frequency
    mf = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.sin(2 * np.pi * 7 * tt)
    mf = np.maximum(mf, 0.05)
    starts.append(("multifreq", mf * S / mf.sum()))
    
    # 6. Smooth random
    noise = rng.standard_normal(n)
    kw = max(5, n // 64)
    kernel = np.exp(-0.5 * (np.arange(-kw // 2, kw // 2 + 1) / (kw // 4.0 + 0.1)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    sr = 1.0 + 0.4 * smooth / (np.std(smooth) + 1e-10)
    sr = np.maximum(sr, 0.05)
    starts.append(("smooth_rand", sr * S / sr.sum()))
    
    # 7. Bump (Gaussian)
    bump = np.exp(-0.5 * ((tt - 0.5) * 3.0) ** 2) + 0.1
    starts.append(("bump", bump * S / bump.sum()))
    
    # 8. Two bumps
    two_b = (np.exp(-0.5 * ((tt - 0.25) * 5.0) ** 2) +
             np.exp(-0.5 * ((tt - 0.75) * 5.0) ** 2) + 0.05)
    starts.append(("two_bump", two_b * S / two_b.sum()))
    
    return starts


def optimize_length(n, budget_s, rng):
    """Full optimization pipeline for a given sequence length."""
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()
    
    global_best = np.inf
    global_a = None
    t_start = time.monotonic()
    
    starts = generate_starts(n, S, rng)
    n_starts = len(starts)
    
    # Allocate time: 70% for multi-start optimization, 30% for final polish
    main_budget = budget_s * 0.70
    per_start = main_budget / n_starts
    
    print("  %d starts, ~%.0fs each (main phase %.0fs)" % (n_starts, per_start, main_budget))
    
    for name, arr in starts:
        elapsed = time.monotonic() - t_start
        remaining = main_budget - elapsed
        if remaining < 5:
            break
        
        alloc = min(per_start, remaining - 2)
        if alloc < 3:
            break
        
        a_copy = arr.astype(np.float64).copy()
        
        # Phase 1: Nesterov subgradient (main workhorse)
        t1 = time.monotonic()
        a1, sc1 = nesterov_subgradient(a_copy, n, S, alloc * 0.75, rng,
                                        top_k=30, base_temp=1.0, step_scale=1.5)
        
        if sc1 < global_best:
            global_best = sc1
            global_a = a1.copy()
        
        # Phase 2: Spectral polish
        rem = main_budget - (time.monotonic() - t_start)
        if rem > 3 and sc1 < global_best + 0.05:
            alloc2 = min(alloc * 0.25, rem - 2)
            a2, sc2 = spectral_polish(a1, N, alloc2, rng, n_lags=15, alpha_init=0.15)
            if sc2 < global_best:
                global_best = sc2
                global_a = a2.copy()
        
        elapsed_total = time.monotonic() - t_start
        print("  [%-12s] score=%.6f (t=%.0fs, best=%.6f)" % (name, sc1, elapsed_total, global_best))
    
    # Final intensive polish on the best found
    if global_a is not None:
        elapsed = time.monotonic() - t_start
        final_budget = budget_s - elapsed
        if final_budget > 5:
            print("  Final polish: %.0fs available" % final_budget)
            
            # Long Nesterov run on best
            fb = min(final_budget * 0.6, 600.0)
            a3, sc3 = nesterov_subgradient(global_a, n, S, fb, rng,
                                            top_k=50, base_temp=0.5, step_scale=0.5)
            if sc3 < global_best:
                global_best = sc3
                global_a = a3.copy()
                print("  After final Nesterov: %.6f" % global_best)
            
            # Spectral polish
            rem = budget_s - (time.monotonic() - t_start)
            if rem > 3:
                a4, sc4 = spectral_polish(global_a, N, min(rem, 120.0), rng,
                                           n_lags=20, alpha_init=0.08)
                if sc4 < global_best:
                    global_best = sc4
                    global_a = a4.copy()
                    print("  After spectral polish: %.6f" % global_best)
            
            # One more Nesterov pass
            rem = budget_s - (time.monotonic() - t_start)
            if rem > 5:
                a5, sc5 = nesterov_subgradient(global_a, n, S, rem, rng,
                                                top_k=50, base_temp=0.3, step_scale=0.3)
                if sc5 < global_best:
                    global_best = sc5
                    global_a = a5.copy()
                    print("  After final Nesterov 2: %.6f" % global_best)
    
    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov subgradient + spectral water-filling with multi-start.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    
    print("=" * 60)
    print("Starting optimization, budget = %.0fs" % budget_s)
    print("=" * 60)
    
    t_global = time.monotonic()
    global_best = np.inf
    global_a = None
    
    # Primary: n = 8192
    n_primary = 8192
    alloc_primary = budget_s * 0.75
    print("\n=== n = %d (budget %.0fs) ===" % (n_primary, alloc_primary))
    
    a_res, sc_res = optimize_length(n_primary, alloc_primary, rng)
    if a_res is not None and sc_res < global_best:
        global_best = sc_res
        global_a = a_res.copy()
        print("  Best for n=%d: %.6f" % (n_primary, global_best))
    
    # Secondary: try n = 16384 if time remains
    elapsed = time.monotonic() - t_global
    remaining = budget_s - elapsed - 30
    if remaining > 200:
        n_sec = 16384
        alloc_sec = min(remaining, budget_s * 0.20)
        print("\n=== n = %d (budget %.0fs) ===" % (n_sec, alloc_sec))
        
        a_res2, sc_res2 = optimize_length(n_sec, alloc_sec, rng)
        if a_res2 is not None and sc_res2 < global_best:
            global_best = sc_res2
            global_a = a_res2.copy()
            print("  NEW GLOBAL BEST: %.6f at n=%d" % (global_best, n_sec))
        else:
            print("  Best for n=%d: %.6f (global still %.6f)" % (n_sec, sc_res2, global_best))
    
    # Tertiary: try n = 4096 if time remains
    elapsed = time.monotonic() - t_global
    remaining = budget_s - elapsed - 30
    if remaining > 100 and global_best > 1.505:
        n_ter = 4096
        alloc_ter = min(remaining, 150.0)
        print("\n=== n = %d (budget %.0fs) ===" % (n_ter, alloc_ter))
        
        a_res3, sc_res3 = optimize_length(n_ter, alloc_ter, rng)
        if a_res3 is not None and sc_res3 < global_best:
            global_best = sc_res3
            global_a = a_res3.copy()
            print("  NEW GLOBAL BEST: %.6f at n=%d" % (global_best, n_ter))
    
    # Final verification
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("FALLBACK: uniform sequence, score=%.4f" % global_best)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)
        global_best = 2.0
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nVerification: evaluate_sequence = %.8f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluation failed, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        global_best = ver if np.isfinite(ver) else 2.0
    
    total_time = time.monotonic() - t_global
    print("\n" + "=" * 60)
    print("FINAL: length=%d, score=%.8f, time=%.0fs" % (len(global_a), global_best, total_time))
    print("=" * 60)
    
    return [float(x) for x in global_a]
