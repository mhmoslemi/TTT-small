def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)

    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((b_k - T)^p_+) — the 'excess' above threshold T."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj


def peak_shave_gradient(a, c, k=16):
    """Exact subgradient from the top-k lags, weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by how much above the mean
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_excess_annealing(a_init, n, S, max_time, rng):

def beta_max_est(c, S, n):
    """Estimate a reasonable Lipschitz constant."""
    return 2.0 * S * S / (2 * n)

def generate_diverse_inits(n, rng, base_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if base_seq is not None and len(base_seq) == n:
        inits.append(base_seq.copy())

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / (0.3 * 0.3)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)

    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt + 0.1) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.cos(2 * np.pi * 7 * tt + 0.3))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Block structures
    for nblock in [4, 8, 16]:
        block_len = n // nblock
        heights = rng.uniform(0.4, 1.2, nblock)
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])

    # Triangle
    inits.append(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def objective_value(a):
