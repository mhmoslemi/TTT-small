"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Nesterov-accelerated subgradient with exact gradient computation
2. Minimax 1D coordinate descent on active lags
3. Spectral equalization for global exploration
4. Multi-start with diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * S / n
    return v * (S / s)


def convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_at_lags(a, lags):
    """Compute sum of gradients at given lags. dc[k]/da[j] = 2*a[k-j]."""
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi+1] += 2.0 * a[idx]
    return g


def nesterov_subgradient(a_init, S, n, fsize, max_time, rng, k_init=10, k_max=512):
    """
    Nesterov-accelerated projected subgradient method.
    Minimizes max(convolve(a,a)) subject to a>=0, sum(a)=S.
    """
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    lr = 1.0
    best_a = a.copy()
    best_cmax = convolve_self(a).max()
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // 10)
        
        c = convolve_self(y)
        cur_max = c.max()
        if cur_max < best_cmax - 1e-10:
            best_cmax = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if no_improve > 100:
            # Reset and perturb
            y = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            no_improve = 0
            continue
        
        # Top-k lags
        top_idx = np.argsort(c)[::-1][:k]
        g = subgradient_at_lags(y, top_idx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = lr * S / (gnorm * np.sqrt(it + 1))
        
        x_new = project_simplex(y - step * g, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Rescale to maintain sum constraint
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        
        a = x_new
        t = t_new
        
        # Occasionally reduce learning rate
        if it % 500 == 0:
            lr *= 0.9
    
    # Final check
    c_final = convolve_self(best_a)
    return best_a, c_final.max()


def coord_descent_minimax(a, S, n, max_time, rng, n_rounds=100, n_coords=50):
    """
    Coordinate descent with minimax 1D optimization.
    For each coordinate, find the value that minimizes the max convolution.
    """
    a = a.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = convolve_self(a).max()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        c = convolve_self(a)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Find peak lag
        peak_k = int(np.argmax(c))
        
        # Find top-contributing coordinates at the peak lag
        lo = max(0, peak_k - (n - 1))
        hi = min(n - 1, peak_k)
        contribs = np.zeros(n)
        for j in range(lo, hi + 1):
            contribs[j] = a[j] * a[peak_k - j]
        
        # Select top coordinates to optimize
        top_coords = np.where(contribs > 0)[0]
        if len(top_coords) > n_coords:
            # Sample from top contributors
            vals = contribs[top_coords]
            top_vals_idx = np.argsort(vals)[::-1][:n_coords]
            chosen = top_coords[top_vals_idx]
        else:
            chosen = top_coords
        
        rng.shuffle(chosen)
        
        for j in chosen:
            if time.monotonic() - start > max_time:
                break
            j = int(j)
            orig = a[j]
            if orig < 1e-10:
                continue
            
            # The lags affected by a[j]: from j to j+n-1
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            
            # Current values at affected lags
            c_affected = c[k_lo:k_hi+1].copy()
            
            # Change in c when a[j] changes by delta: 
            # c[k] += (new_a_j - orig) * (a[k-j] + a[j - (k-j)])
            # Actually: c[k] = sum_i a[i]*a[k-i]
            # If a[j] changes by delta, c[k] changes by delta * (a[k-j] + ...)
            # More precisely: c[k] new = c[k] old + delta * a[k-j] + delta * a[k-j]
            # Wait: c[k] = sum_i a[i] * a[k-i]
            # Term with a[j]: when i=j, term is a[j]*a[k-j]
            # Term with a[k-j]: when i=k-j, term is a[k-j]*a[j]
            # If j = k-j (k=2j), both terms are the same: a[j]^2, derivative is 2*a[j]
            # If j != k-j, two separate terms, each contributes a[k-j]*delta
            # So total change: delta * (a[k-j] + a[k-j]) = 2*delta*a[k-j] (if both valid)
            # Or delta * a[k-j] + delta * a[j] ... no.
            
            # Let me be more careful:
            # c[k] = sum_i a[i] * a[k-i]
            # When a[j] -> a[j] + delta:
            #   c[k] -> sum_i (a[i] + delta*[i==j]) * (a[k-i] + delta*[k-i==j])
            #   = c[k] + delta * a[k-j] + delta * a[k-j] + delta^2 * [k==2j]
            #   = c[k] + 2*delta*a[k-j] + delta^2 (if k=2j)
            #   = c[k] + 2*delta*a[k-j] (if k != 2j)
            # (assuming both j and k-j are valid indices)
            
            # For k in [j, j+n-1]: k-j is in [0, n-1], so a[k-j] is valid.
            # Also need j to be valid (it is, since 0<=j<n).
            # So the change is 2*delta*a[k-j] for all affected k, PLUS delta^2 if k=2j.
            
            # Let's find the optimal delta by 1D minimax:
            # minimize max_k (c[k_lo + (k-k_lo)] + 2*delta*a[k-j])
            # subject to: 0 <= a[j] + delta <= 1000, i.e., -a[j] <= delta <= 1000-a[j]
            
            delta_lo = -orig  # a[j] + delta >= 0
            delta_hi = 1000.0 - orig
            
            # Bisection on delta
            # f(delta) = max_k (c_affected[k-k_lo] + 2*delta*a[k-j])
            # We want to minimize f(delta)
            
            def f_of_delta(delta):
                if k_hi - k_lo + 1 == 0:
                    return 0.0
                base = c_affected.copy()
                # a[k-j] for k in [k_lo, k_hi]
                k_range = np.arange(k_lo, k_hi + 1)
                a_vals = a[k_range - j]  # a[k-j] for valid k
                # But we also need to add the delta^2 term if k=2j
                new_vals = base + 2.0 * delta * a_vals
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    new_vals[2 * j - k_lo] += delta * delta
                return new_vals.max()
            
            # For efficiency, precompute a_vals
            k_range = np.arange(k_lo, k_hi + 1)
            a_vals = a[k_range - j]
            has_diag = (2 * j >= k_lo and 2 * j <= k_hi)
            diag_pos = 2 * j - k_lo if has_diag else -1
            
            # Use bisection to find optimal delta
            # The function f(delta) is convex (max of linear + quadratic terms)
            # Actually, if has_diag, there's a delta^2 term, making it quadratic
            # If not, it's the max of linear functions (convex, piecewise linear)
            
            # For the linear case: minimize max_i (b_i + d_i * delta)
            # Optimal delta is where the "tightest" constraints intersect
            
            # Simple approach: bisection on the derivative
            # f'(delta) = d_{i*(delta)} where i* is the argmax
            # At optimum, the argmax switches, so f(delta - eps) < f(delta) and f(delta + eps) < f(delta)
            
            # Even simpler: just try a few candidate delta values
            # The optimal delta should be near where the peak lag's contribution is reduced
            # Current contribution at peak: 2*orig*a[peak_k - j]
            # We want to reduce this while not increasing other lags too much
            
            # Try: reduce a[j] by a fraction of its value
            best_delta = 0.0
            best_f = f_of_delta(0.0)
            
            # Try several multipliers
            for frac in [0.5, 0.7, 0.8, 1.3, 1.5, 2.0, 0.3]:
                delta = (frac - 1.0) * orig
                if delta < delta_lo or delta > delta_hi:
                    continue
                f_val = (c_affected + 2.0 * delta * a_vals).max()
                if has_diag:
                    f_val = max(f_val, c_affected[diag_pos] + 2.0 * delta * a_vals[diag_pos] + delta * delta)
                if f_val < best_f:
                    best_f = f_val
                    best_delta = delta
            
            if best_delta != 0.0:
                a[j] = orig + best_delta
        
        # Re-normalize to maintain sum
        a = project_simplex(a, S)
    
    c_final = convolve_self(best_a)
    return best_a, c_final.max()


def spectral_equipower(a, S, n, fsize, alpha=0.1, n_iter=100):
    """
    Iteratively equalize the power spectrum to flatten the convolution.
    This is a "water-filling" approach in the frequency domain.
    """
    a = a.copy()
    best_a = a.copy()
    best_cmax = convolve_self(a).max()
    
    for it in range(n_iter):
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        c = np.real(ifft(A * A))[:2*n-1]
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Find the peak lag
        peak_k = int(np.argmax(c))
        
        # The peak lag k corresponds to a specific frequency "beat"
        # To reduce the peak, we want to reduce the power at frequencies
        # that constructively interfere at lag peak_k
        
        # The contribution of frequency f to c[peak_k] is:
        # (1/N) * |A[f]|^2 * cos(2*pi*f*peak_k/N + 2*angle(A[f]))
        # (approximately, for real signals)
        
        # Simplified: reduce power where cos(2*pi*f*peak_k/N) > 0
        m = np.arange(fsize)
        cos_k = np.cos(2.0 * np.pi * m * peak_k / fsize)
        
        # Apply shaping
        P_new = P * np.exp(-alpha * np.maximum(cos_k, 0.0))
        
        # Preserve DC
        P_new[0] = P[0]
        # Preserve Hermitian symmetry (for real signal)
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        
        # Inverse FFT and project
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Check if it actually helped
        c_check = convolve_self(a_new)
        if c_check.max() < cmax:
            a = a_new
        
    return best_a, best_cmax


def structured_init(n, rng, method="hann"):
    """Generate structured initializations."""
    t = np.arange(n) / n
    if method == "hann":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif method == "gauss":
        a = np.exp(-0.5 * ((t - 0.5) * 3) ** 2)
    elif method == "alternating":
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.2)
    elif method == "sine_mix":
        a = 0.5 + 0.3 * np.sin(2*np.pi*3*t) + 0.2 * np.sin(2*np.pi*7*t) + 0.1 * np.sin(2*np.pi*13*t)
    elif method == "random":
        a = 0.3 + 0.7 * rng.random(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2*np.pi*(2*t + 3*t*t))
    elif method == "bimodal":
        a = 0.3 + 0.7 * (np.exp(-0.5*((t-0.25)*5)**2) + np.exp(-0.5*((t-0.75)*5)**2))
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)
    a = np.clip(a, 0.001, 2.0)
    return a
