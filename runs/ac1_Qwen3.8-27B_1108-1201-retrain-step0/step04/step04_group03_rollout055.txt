"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP-based best-response with golden-section step search,
interleaved with FISTA polishing. Focus on high-quality LP iterations.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_candidates
    valid = w > theta_candidates
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_step(y, a_old, t, grad, S, step):
    x_new = project_simplex(y - step * grad, S)
    t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
    y_new = x_new + ((t - 1.0) / t_new) * (x_new - a_old)
    y_new = np.maximum(y_new, 0.0)
    return x_new, y_new, t_new


def solve_lp(f, n, S, K, solver='SCS', time_limit=15.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K lags, g >= 0."""
    c = fft_convolve_self(f)
    M = c.max()
    
    # Get top K lags
    idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix
    rows = np.zeros((K, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]
    
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    obj = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(obj, constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    
    if time.monotonic() - start > time_limit * 2:
        return None
    
    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def golden_section_search(f, g, S, M_old, n_evals=12):
    """Find best t in [0,1] minimizing max(((1-t)f+tg)*((1-t)f+tg))."""
    def eval_t(t):
        if t <= 0:
            return M_old
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max()
    
    # Golden section
    phi = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    a, b = 0.0, 1.0
    c_ = b - phi * (b - a)
    d_ = a + phi * (b - a)
    fc = eval_t(c_)
    fd = eval_t(d_)
    
    for _ in range(n_evals - 2):
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - phi * (b - a)
            fc = eval_t(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + phi * (b - a)
            fd = eval_t(d_)
    
    t_best = (a + b) / 2
    f_best = eval_t(t_best)
    return t_best, f_best


def run_fista_polish(a_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.002, beta_max=2000.0):
    """FISTA with LSE, focused on polishing."""
    a = normalize(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new, y_new, t_new = fista_step(y, a, t, grad, S, step)
        a = x_new
        y = y_new
        t = t_new
    
    return best_a, best

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP best-response iterations with golden-section step search,
interleaved with adaptive FISTA polishing. Uses increasing constraint
counts for tighter LP relaxations and periodic restarts to escape local
stagnation.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_candidates
    valid = w > theta_candidates
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002, beta_max=2000.0):
    a = normalize(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-it / 30000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_a, best


def solve_lp(f, n, S, K, solver='SCS', time_limit=12.0):
    c = fft_convolve_self(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:K]

    rows = np.zeros((K, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 2.5:
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g
    return None


def golden_section(f, g, S, M_old, n_evals=10):
    def eval_t(t):
        if t <= 1e-6:
            return M_old
        ft = (1.0 - t) * f + t * g
        ft = project_simplex(np.maximum(ft, 0.0), S)
        return fft_convolve_self(ft).max()

    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c1 = b - phi * (b - a)
    d1 = a + phi * (b - a)
    fc = eval_t(c1)
    fd = eval_t(d1)

    for _ in range(n_evals - 2):
        if fc < fd:
            b = d1
            d1 = c1
            fd = fc
            c1 = b - phi * (b - a)
            fc = eval_t(c1)
        else:
            a = c1
            c1 = d1
            fc = fd
            d1 = a + phi * (b - a)
            fd = eval_t(d1)

    t_best = (a + b) / 2.0
    val_best = eval_t(t_best)
    return t_best, val_best
