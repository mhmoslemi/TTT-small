    scaled = beta * (c - cm)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([max(S, 0.0)])

def compute_lse(a, c, beta, fsize_grad):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    np.clip(scaled, -80, 80, out=scaled)
    e = np.exp(scaled)
    ess = e.sum()
    if ess < 1e-300:
        return cm, np.zeros(n)
    w = e / ess
    val = cm + np.log(ess) / beta
    W = fft(w, fsize_grad)
    A = fft(a, fsize_grad)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return val, grad


def compute_topk_grad(a, c, k, fsize_grad):
    n = len(a)
    order = np.argsort(c)[::-1][:k]
    vals = c[order]
    min_val = c.min()
    diff = vals - min_val
    scale = diff.max()
    if scale > 1e-15:
        weights = diff / scale
    else:
        weights = np.ones(k) / k
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[order] = weights
    W = fft(w_full, fsize_grad)
    A = fft(a, fsize_grad)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad


def objective_value(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_init(n, rng, n_iter=80):
    a = rng.random(n) + 0.5
    fsize = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = fft(a, fsize)
        A_abs = np.abs(A)
        dc = A_abs[0]
        target = dc / np.sqrt(fsize - 1)
        new_mag = target + 0.1 * A_abs
        new_mag[0] = dc
        new_phase = np.exp(1j * np.angle(A))
        A_new = new_mag * new_phase
        a = np.real(ifft(A_new))[:n]
        a = np.maximum(a, 0.0)
        a = a / a.sum() * np.sqrt(2.0 * n)
    return a


def generate_inits(n, rng):
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.09))
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.15 * np.cos(10 * np.pi * tt + 0.5))
    inits.append(rng.random(n) + 0.1)
    inits.append(spectral_init(n, rng, n_iter=60))
    return [np.maximum(x, 0.01) for x in inits]


def run_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=5000.0,
              use_topk=False, topk=4):
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_grad = 1 << (3 * n - 2).bit_length()
    c = fft_convolve_self(a, fsize_conv)
    best_max = c.max()
    best_a = a.copy()
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    it = 0
    stagnant = 0
    step = 0.01
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, it * 2))
        beta = min(beta, beta_end)

        c_y = fft_convolve_self(y, fsize_conv)
        val, grad = compute_lse(y, c_y, beta, fsize_grad)
        if c_y.max() < best_max:
            best_max = c_y.max()
            best_a = y.copy()
            stagnant = 0
        else:
            stagnant += 1

        if use_topk and it % 10 == 0:
            grad = compute_topk_grad(y, c_y, topk, fsize_grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_new = project_simplex(y - step * grad / gnorm * min(gnorm, step * 0.1 + 0.001 * S / n), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_k = t_new

        if stagnant > 500:
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_k = 1.0
            stagnant = 0
            step = max(0.002 * S / n, step * 0.5)

        step *= (1.0 + 0.001)
        if step > 0.5 * S / n:
            step = 0.5 * S / n

    c_final = fft_convolve_self(a, fsize_conv)
    if c_final.max() < best_max:
        best_max = c_final.max()
        best_a = a.copy()
    return best_a, best_max


def run_polish(a_init, n, S, max_time, rng):
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_grad = 1 << (3 * n - 2).bit_length()
    c = fft_convolve_self(a, fsize_conv)
    best_max = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    step = 0.001 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a, fsize_conv)
        if c.max() < best_max:
            best_max = c.max()
            best_a = a.copy()

        k_top = min(8, n // 4)
        if it < 100:
            grad = compute_topk_grad(a, c, k_top, fsize_grad)
        else:
            beta = min(5000.0, 50.0 * (1.05 ** it))
            _, grad = compute_lse(a, c, beta, fsize_grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        direction = -grad / gnorm
        alpha = step
        for _ in range(15):
            candidate = project_simplex(a + alpha * direction * gnorm, S)
            c_test = fft_convolve_self(candidate, fsize_conv)
            if c_test.max() < c.max() - 1e-12:
                break
            alpha *= 0.5
        else:
            alpha = step * 0.1
            candidate = project_simplex(a + alpha * direction * gnorm, S)

        a = candidate
        step = alpha * 1.05

        if it % 500 == 0:
            noise = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    c_final = fft_convolve_self(a, fsize_conv)
    if c_final.max() < best_max:
        best_max = c_final.max()
        best_a = a.copy()
    return best_a, best_max


def run_excess_anneal(a_init, n, S, max_time, rng):
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_grad = 1 << (3 * n - 2).bit_length()
    c = fft_convolve_self(a, fsize_conv)
    best_max = c.max()
    best_a = a.copy()
    c_mean = c.mean()
    c_peak = c.max()
    start = time.monotonic()
    y = a.copy()
    t_k = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = c_peak + (c_mean + 0.02 * (c_peak - c_mean) - c_peak) * (progress ** 0.6)
        T = max(T, c_mean)

        c_y = fft_convolve_self(y, fsize_conv)
        if c_y.max() < best_max:
            best_max = c_y.max()
            best_a = y.copy()

        excess = np.maximum(c_y - T, 0.0)
        if excess.max() < 1e-10:
            T2 = c_y.max() - 0.005 * (c_y.max() - c_y.mean() + 1e-10)
            excess = np.maximum(c_y - max(T2, c_y.mean()), 0.0)

        w = excess / max(excess.sum(), 1e-30)
        W = fft(w, fsize_grad)
        A = fft(y, fsize_grad)
        grad = 2.0 * np.real(ifft(W * A))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.005 * S / (n * (1.0 + 0.003 * it))
        a_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_k = t_new

        if it % 1000 == 0 and rng.random() < 0.05:
            noise = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
                t_k = 1.0

    c_final = fft_convolve_self(a, fsize_conv)
    if c_final.max() < best_max:
        best_max = c_final.max()
        best_a = a.copy()
    return best_a, best_max


def upscale(a_small, n_large):
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)
