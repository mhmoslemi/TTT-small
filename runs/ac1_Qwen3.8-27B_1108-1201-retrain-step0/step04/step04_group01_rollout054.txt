"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LSE-FISTA with adaptive beta (high-resolution max approximation)
2. L-BFGS on LSE-smoothed objective with softmax parameterization
3. Dual ascent (exponentiated gradient on constraint weights)
4. Multi-length exploration (8192, 16384)
5. Peak shaving with decreasing steps
Starts from previous best (height_sequence_1) when available.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w  # strict inequality for the standard algorithm
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_value(c, beta):
    cm = c.max()
    x = beta * (c - cm)
    x = np.clip(x, -500, 500)
    return cm + np.log(np.exp(x).sum()) / beta

def lse_grad_a(a, c, beta):
    """Gradient of LSE-smoothed max(conv) w.r.t. a."""
    n = len(a)
    cm = c.max()
    x = beta * (c - cm)
    x = np.clip(x, -500, 500)
    e = np.exp(x)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # grad = 2 * (w * a) via FFT
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0

def hankel_matvec(a, lam):
    """Compute Q @ a where Q[i,j] = lam[i+j] for 0<=i,j<n."""
    n = len(a)
    a_rev = a[::-1]
    total_len = len(a_rev) + len(lam) - 1
    fsize = 1 << (total_len - 1).bit_length()
    result = np.real(ifft(fft(a_rev, fsize) * fft(lam, fsize)))
    return result[n-1:n-1+n]

def fista_lse(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.003, beta_max=50000.0, restart_interval=5000):
    """FISTA with LSE smoothing and adaptive beta."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        grad = lse_grad_a(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: for LSE, Lipschitz constant ~ 4*beta*max(a) approximately
        # More accurately: L = 4 * beta * S / n (rough bound)
        step = 1.0 / (4.0 * beta * S / max(n, 1))
        # Actually, the Lipschitz constant of the LSE gradient is bounded by
        # 4 * beta * ||a||_inf * (something)... let me use a safer bound
        # The gradient is 2*(w*a) where sum(w)=1, so ||grad|| <= 2*||a||_1 = 2S
        # The Hessian involves w*w convolved with a... it's complex.
        # Use adaptive: start with conservative step, increase if objective decreases.
        step = 1.0 / (2.0 * beta * S / n * 2.0)  # conservative
        
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking if needed (check if objective improved)
        # For now, just use the fixed step
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        # Periodic restart with noise
        if it % restart_interval == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / (5 * restart_interval))
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0
        
        # Stall detection
        if it - last_improve > restart_interval:
            t_k = 1.0
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)

def lbfgs_lse(a_init, S, max_time, beta=100.0, max_iter=500):
    """L-BFGS on LSE-smoothed objective with softmax parameterization."""
    n = len(a_init)
    a0 = normalize_to_sum(a_init, S)
    # Softmax parameterization: a = S * softmax(x)
    x0 = np.log(np.maximum(a0, 1e-15)) - np.log(a0).mean()
    
    def objective(x):
        # Compute a from x
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a)
        return lse_value(c, beta)
    
    def gradient(x):
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a)
        grad_a = lse_grad_a(a, c, beta)
        # Chain rule: df/dx_j = grad_a_j * a_j - (a_j/S) * sum_i grad_a_i * a_i
        dot = np.dot(grad_a, a)
        grad_x = grad_a * a - (a / S) * dot
        return grad_x
    
    start = time.monotonic()
    best_val = objective(x0)
    best_a = a0.copy()
    
    try:
        result = minimize(objective, x0, method='L-BFGS-B', 
                         jac=gradient,
                         options={'maxiter': max_iter, 'maxfun': 2000,
                                  'max_time': max_time, 'gtol': 1e-8})
        x_opt = result.x
        x_shifted = x_opt - x_opt.max()
        exp_x = np.exp(x_shifted)
        a_opt = S * exp_x / exp_x.sum()
        
        c = fft_convolve_self(a_opt)
        val = c.max()
        if val < best_val:
            best_a = a_opt
            best_val = val
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    # Check time
    elapsed = time.monotonic() - start
    print(f"  L-BFGS done in {elapsed:.1f}s, best max_b = {best_val:.6f}")
    
    return best_a, best_val

def peak_shave(a_init, S, max_time, rng, k_init=8, k_end=128):
    """Iterative peak shaving with top-k subgradient."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    y = a.copy()
    t_k = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(k_init + (k_end - k_init) * progress)
        k = min(k, max(4, n // 16))
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
        
        # Top-k subgradient
        idx = np.argsort(c)[::-1][:k]
        weights = c[idx] - c.mean()
        weights = np.maximum(weights, 0)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        w_full = np.zeros(len(c))
        w_full[idx] = weights
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = np.real(ifft(W * np.conj(A)))[:n] * 2.0
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / (2.0 * S / n * (1.0 + 0.1 * progress))
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

        # Gradient of a^T Q a is 2*Q*a
        y = a.copy()
        t_k = 1.0
        for inner_it in range(inner_iters):
            Qa = hankel_matvec(y, lam)
            grad = 2.0 * Qa
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Lipschitz constant of 2Q: bound by 4 * max row sum of Q
            # Q[i,j] = lam[i+j], row sum = sum_j lam[i+j]
            # For uniform lam, this is about 1
            step = 1.0 / (4.0)  # conservative
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_k = t_new
        
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        # Update lambda
        t = c.max()
        # Exponentiated gradient: focus on peaks
        log_lam = np.log(lam + 1e-30) + eta * (c - 0.5 * t)
        log_lam -= log_lam.max()
        lam = np.exp(log_lam)
        lam = lam / lam.sum()
    
    return best_a, best

def cosine_init(n, n_bins=5, rng=None):
    """Generate a cosine-based initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n) / n

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * m * t + phase)
    a = np.maximum(a, 0.01)
    return a

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LSE-FISTA with adaptive beta and backtracking
2. L-BFGS on LSE-smoothed objective with softmax parameterization
3. Iterative peak shaving with adaptive k
4. Dual ascent (exponentiated gradient on constraint weights)
5. Stochastic coordinate descent for fine-tuning
6. Multi-length exploration (8192, 16384)
Starts from previous best (height_sequence_1) when available.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """LSE-smoothed max."""
    cm = c.max()
    x = beta * (c - cm)
    x = np.clip(x, -500, 500)
    return cm + np.log(np.exp(x).sum()) / beta


def lse_grad_a(a, c, beta):
    """Gradient of LSE-smoothed max(conv) w.r.t. a."""
    n = len(a)
    cm = c.max()
    x = beta * (c - cm)
    x = np.clip(x, -500, 500)
    e = np.exp(x)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0


def hankel_matvec(a, lam):
    """Compute Q @ a where Q[i,j] = lam[i+j] for 0<=i,j<n."""
    n = len(a)
    a_rev = a[::-1]
    total_len = len(a_rev) + len(lam) - 1
    fsize = 1 << (total_len - 1).bit_length()
    result = np.real(ifft(fft(a_rev, fsize) * fft(lam, fsize)))
    return result[n - 1:n - 1 + n]


def fista_lse(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.002,
              beta_max=50000.0, restart_interval=4000):
    """FISTA with LSE smoothing and adaptive beta."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best - 1e-12:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_grad_a(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Conservative step: L ~ 4*beta*max|a| roughly, use 2*S/n as proxy
        step = 1.0 / (4.0 * beta * max(S / n, 1e-6))
        # Cap step to avoid overshooting
        step = min(step, S / (n * gnorm + 1e-15) * 0.5)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Stall detection with restart
        if it - last_improve > restart_interval:
            t_k = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / (10.0 * restart_interval))
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

        # Periodic small perturbation
        if it % (restart_interval * 2) == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def lbfgs_lse(a_init, S, max_time, beta=200.0, max_iter=300):
    """L-BFGS on LSE-smoothed objective with softmax parameterization."""
    n = len(a_init)
    a0 = normalize_to_sum(a_init, S)
    x0 = np.log(np.maximum(a0, 1e-30) + 1e-30)
    x0 = x0 - x0.mean()

    eval_count = [0]

    def objective(x):
        eval_count[0] += 1
        if time.monotonic() > start + max_time:
            return 1e10
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a)
        return lse_value(c, beta)

    def gradient(x):
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a)
        grad_a = lse_grad_a(a, c, beta)
        dot = np.dot(grad_a, a)
        grad_x = grad_a * a - (a / S) * dot
        return grad_x

    start = time.monotonic()
    best_val = objective(x0)
    best_a = a0.copy()

    try:
        result = minimize(objective, x0, method='L-BFGS-B',
                          jac=gradient,
                          options={'maxiter': max_iter, 'maxfun': 500,
                                   'maxls': 50, 'gtol': 1e-6,
                                   'ftol': 1e-10})
        x_opt = result.x
        x_shifted = x_opt - x_opt.max()
        exp_x = np.exp(x_shifted)
        a_opt = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a_opt)
        val = c.max()
        if val < best_val:
            best_a = a_opt
            best_val = val
        print(f"    L-BFGS: {result.nit} iters, {result.nfev} f-evals, "
              f"status={result.status}, max_b={val:.6f}")
    except Exception as e:
        print(f"    L-BFGS error: {e}")

    elapsed = time.monotonic() - start
    print(f"    L-BFGS done in {elapsed:.1f}s, best max_b = {best_val:.6f}")
    return best_a, best_val


def peak_shave(a_init, S, max_time, rng, k_init=4, k_end=128):
    """Iterative peak shaving with top-k subgradient and FISTA momentum."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    y = a.copy()
    t_k = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(k_init + (k_end - k_init) * (progress ** 0.5))
        k = min(k, max(4, n // 32))

        c = fft_convolve_self(y)
        if c.max() < best - 1e-12:
            best = c.max()
            best_a = y.copy()

        idx = np.argsort(c)[::-1][:k]
        weights = c[idx] - c[idx].min()
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        w
