"""
Hybrid optimizer for minimum-peak autocorrelation of non-negative sequences.
Combines LSE-subgradient descent, periodic spectral re-shaping, simulated
annealing, anti-symmetric structure exploitation, and multi-length search
to minimize 2n*max(a*a)/(sum a)^2 below the 1.5027 target.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n*max(a*a)/(sum a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_gradient(a, c, beta, n):
    """LSE-smoothed gradient of max(c)."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k, n):
    """Subgradient using top-k peaks."""
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def threshold_subgradient(a, c, n, frac=0.1):
    """Subgradient using all lags above a fraction of max."""
    thresh = c.max() * frac
    idx = np.where(c >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(c)])
    grad = np.zeros(n)
    weights = c[idx] - thresh + 1e-10
    for j, w in zip(idx, weights):
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += w * a[j - i_vals]
    return grad


def spectral_reshape(a, n, N, n_iters, rng, blend=0.15):
    """
    Periodic spectral re-shaping: adjust DFT magnitudes to flatten the
    convolution spectrum, then project back to non-negativity.
    """
    a_work = a.copy().astype(np.float64)
    S = a_work.sum()
    best_obj = compute_objective(a_work)
    best_a = a_work.copy()

    a_pad = np.zeros(N)
    a_pad[:n] = a_work
    A = fft(a_pad)

    for it in range(n_iters):
        # Target: make |A(k)| more uniform for k != 0
        mags = np.abs(A[1:])
        mean_mag = np.mean(mags)
        # Blend toward uniform magnitude
        target = (1 - blend) * mags + blend * mean_mag
        # Don't let any magnitude go to zero
        target = np.maximum(target, 0.01 * mean_mag)
        
        phases = A[1:] / np.maximum(np.abs(A[1:]), 1e-30)
        A_new = A.copy()
        A_new[1:] = phases * target
        A_new[0] = A[0]  # Keep DC fixed

        # Inverse transform and project
        a_candidate = np.real(ifft(A_new))
        a_candidate[:n] = np.maximum(a_candidate[:n], 0.0)
        a_candidate[n:] = 0.0

        s = a_candidate[:n].sum()
        if s < 1e-15:
            break
        a_candidate[:n] *= (S / s)

        # Check
        obj = compute_objective(a_candidate[:n])
        if obj < best_obj:
            best_obj = obj
            best_a = a_candidate[:n].copy()

        # Update state
        a_work = a_candidate[:n]
        a_pad[:] = 0.0
        a_pad[:n] = a_work
        A = fft(a_pad)

    return best_a, best_obj


def anti_symmetric_perturb(a, n, rng, scale=0.1):
    """Add an anti-symmetric perturbation to reduce center autocorrelation peak."""
    half = (n + 1) // 2
    d = np.zeros(n)
    d[:half] = rng.normal(0, scale, half)
    # Mirror anti-symmetrically
    d[half:] = -d[:n - half][::-1] if n % 2 == 0 else -d[:(n - 1) // 2][::-1]
    if n % 2 == 1:
        d[n // 2] = 0.0
    a_new = a + d * a.sum() / n
    return project_simplex(np.maximum(a_new, 0), a.sum())


def simulate_annealing_step(a, n, S, rng, temperature, obj_cur):
    """One SA step: perturb and accept/reject."""
    # Random perturbation
    scale = temperature * S / np.sqrt(n)
    a_perturbed = a + rng.normal(0, scale, n)
    a_new = project_simplex(np.maximum(a_perturbed, 0), S)
    obj_new = compute_objective(a_new)

    if obj_new < obj_cur or rng.random() < np.exp(-(obj_new - obj_cur) / max(temperature, 1e-10)):
        return a_new, obj_new, True
    return a, obj_cur, False

        it += 1

        # Check improvement

            best_a = a.copy()
            last_improve_it = it

        # Choose gradient type based on iteration phase

def generate_inits(n, S, rng, given=None):

    inits.append(("sin_sq", np.maximum(S / n * np.sin(np.pi * tt) ** 2, 1e-8)))

    inits.append(("anti_sym", project_simplex(S/n * np.ones(n) + S/n * 0.3 * np.sin(2*np.pi*tt), S)))
    inits.append(("multi", np.maximum(S/n * (1 + 0.2*np.sin(4*np.pi*tt) + 0.1*np.sin(9*np.pi*tt)), 0)))
    inits.append(("random", np.maximum(rng.random(n), 0.01)))
    inits.append(("gauss_center", S * np.exp(-5*(tt - 0.5)**2) / np.exp(-5*(tt - 0.5)**2).sum()))

        arr = np.clip(arr, 0, 1000)
        if arr.sum() < 1e-10:
            continue
        arr = arr * (S / arr.sum())
        refined.append((name, arr))
    return refined


def upsample_sequence(a_small, n_small, n_large):
    """Upsample a sequence to a larger length using interpolation."""
    t_old = np.linspace(0, 1, n_small, endpoint=False)
    t_new = np.linspace(0, 1, n_large, endpoint=False)
    a_interp = np.interp(t_new, t_old, a_small, period=1)
    S_large = np.sqrt(2.0 * n_large)
    if a_interp.sum() < 1e-10:
        return np.ones(n_large) * S_large / n_large
    return a_interp * (S_large / a_interp.sum())

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    t0 = time.monotonic()

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs, given length=%s" % (
        budget_s, "None" if given is None else str(len(given))))

    # Define lengths to try
    lengths_to_try = []
    if given is not None and 1024 <= len(given) <= 16384:
        lengths_to_try.append(len(given))
    for L in [4096, 8192, 16384]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)
    # Sort by expected value (prefer the given length first, then larger)
    lengths_to_try.sort(key=lambda x: (0 if x == (len(given) if given is not None else -1) else 1, -x))

    # Time allocation: give more time to the primary length
    n_lengths = len(lengths_to_try)
    primary_time_factor = 0.5  # 50% to primary
    secondary_time = (1 - primary_time_factor) / max(n_lengths - 1, 1)

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * n)
        N_fft = 1 << (2 * n - 1).bit_length()

        if li == 0:
            t_alloc = (deadline - time.monotonic()) * primary_time_factor
        else:
            t_alloc = min((deadline - time.monotonic()) * 0.4, 300.0)
            if t_alloc < 30:
                continue

        print("\n=== Length n=%d (S=%.4f), time_alloc=%.0fs ===" % (n, S, t_alloc))
        lt_start = time.monotonic()

        # Generate initializations for this length
        given_n = None
        if given is not None:
            if len(given) == n:
                given_n = given.copy()
            elif len(given) < n:
                given_n = upsample_sequence(given, len(given), n)
            else:
                # Downsample: take every k-th element
                k = len(given) // n
                given_n = given[::k][:n]
                S_tmp = np.sqrt(2.0 * n)
                given_n = given_n * (S_tmp / given_n.sum()) if given_n.sum() > 0 else np.ones(n) * S_tmp / n

        inits = generate_inits(n, S, rng, given_n)

        # Phase 1: Try each initialization with core optimization
        per_init_time = t_alloc * 0.6 / max(len(inits), 1)
        per_init_time = max(per_init_time, 10.0)

        for name, init_arr in inits:
            if time.monotonic() > deadline - 60:
                break
            remaining_for_init = min(per_init_time, deadline - time.monotonic() - 60)
            if remaining_for_init < 10:
                break

            a_opt, score = run_core_optimize(init_arr, n, S, remaining_for_init, rng, label="%s/%s" % (n, name))
            a_opt = project_simplex(np.maximum(a_opt, 0), S)
            score = compute_objective(a_opt)

            if score < global_best:
                global_best = score
                global_a = a_opt.copy()
                print("  [n=%d, %s] NEW BEST: obj=%.8f" % (n, name, global_best))

        # Phase 2: Fine-tuning on the best for this length
        remaining = deadline - time.monotonic()
        if remaining > 60 and global_a is not None and len(global_a) == n:
            tune_time = min(remaining * 0.35, t_alloc * 0.3)
            print("  Fine-tuning (time=%.0fs)..." % tune_time)
            a_tuned, score_tuned = run_fine_tuning(global_a, n, S, tune_time, rng)
            a_tuned = project_simplex(np.maximum(a_tuned, 0), S)
            score_tuned = compute_objective(a_tuned)
            if score_tuned < global_best:
                global_best = score_tuned
                global_a = a_tuned.copy()
                print("  After fine-tune: obj=%.8f" % global_best)

        # Phase 3: Spectral reshaping pass
        remaining = deadline - time.monotonic()
        if remaining > 30 and global_a is not None and len(global_a) == n:
            a_sp, score_sp = spectral_reshape(global_a, n, N_fft, 100, rng, blend=0.05)
            a_sp = project_simplex(a_sp, S)
            score_sp = compute_objective(a_sp)
            if score_sp < global_best:
                global_best = score_sp
                global_a = a_sp.copy()
                print("  After spectral reshape: obj=%.8f" % global_best)

        print("  [n=%d] best so far: %.8f, elapsed=%.0fs" % (n, global_best, time.monotonic() - lt_start))

    # Final phase: intensive micro-optimization
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final micro-optimization (time=%.0fs) ===" % remaining)

        # Alternate between LSE gradient and spectral reshape
        a_micro = global_a.copy()
        micro_start = time.monotonic()
        best_micro = global_best
        step = 0.02
        it = 0

        c_micro = fft_convolve_self(a_micro)
        cur_micro = 2.0 * n_best * c_micro.max() / S_best**2

        while time.monotonic() - micro_start < remaining - 20:
            it += 1

            if it % 100 == 1 and it > 1:
                # Periodic spectral reshape
                a_sp, _ = spectral_reshape(a_micro, n_best, 1 << (2*n_best-1).bit_length(),
                                             30, rng, blend=0.03)
                a_sp = project_simplex(a_sp, S_best)
                obj_sp = compute_objective(a_sp)
                if obj_sp < cur_micro:
                    a_micro = a_sp
                    c_micro = fft_convolve_self(a_micro)
                    cur_micro = obj_sp

            if cur_micro < best_micro - 1e-14:
                best_micro = cur_micro
                global_best = best_micro
                global_a = a_micro.copy()

            # LSE gradient with very high beta
            beta = min(3000.0, 200.0 + 2.0 * it)
            grad = lse_gradient(a_micro, c_micro, beta, n_best)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            eff_step = step / (1.0 + 0.001 * it)
            a_try = project_simplex(a_micro - eff_step * grad, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best**2

            if obj_try < cur_micro - 1e-15:
                a_micro = a_try
                c_micro = c_try
                cur_micro = obj_try
                step = min(step * 1.005, 0.3)
            else:
                step *= 0.97
                # Tiny perturbation every 50 failures
                if it % 50 == 0:
                    a_micro = anti_symmetric_perturb(a_micro, n_best, rng, scale=0.005)
                    a_micro = project_simplex(a_micro, S_best)
                    c_micro = fft_convolve_self(a_micro)
                    cur_micro = 2.0 * n_best * c_micro.max() / S_best**2
                    step = 0.02

            if it % 10000 == 0:
                print("    micro it=%d, obj=%.8f, best=%.8f" % (it, cur_micro, best_micro))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = compute_objective(global_a)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = compute_objective(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("\nReturning sequence of length %d, obj=%.8f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
