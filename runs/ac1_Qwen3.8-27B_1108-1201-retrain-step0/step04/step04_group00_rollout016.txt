from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    # Standard simplex projection
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho = np.arange(1, n + 1)
    cond = w > (cumsum - S) / rho
    rho_final = np.max(np.where(cond)[0]) + 1 if np.any(cond) else n
    theta = (cumsum[rho_final - 1] - S) / rho_final
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def freq_domain_iterate(a, n_iter=100, flatten_power=0.7, rng=None):
    """Frequency-domain iterative method for low peak autocorrelation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_obj = compute_obj(a)

        A = fft(a, fsize)
        P = np.abs(A) ** 2
        # Flatten: reduce dynamic range
        P_flat = P ** flatten_power
        # Reconstruct with original phase
        phase = np.angle(A)
        magnitude = np.sqrt(P_flat)
        A_new = magnitude * np.exp(1j * phase)
        a_new = np.abs(ifft(A_new))[:n]

        a_new = a_new * (S / a_new.sum())
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Combined frequency-domain iteration + subgradient with Polyak averaging.
- Phase 1: Frequency-domain iterative method (error reduction style)
- Phase 2: FISTA with LSE smoothing for refinement
- Phase 3: Subgradient method with Polyak averaging for final polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE smooth gradient of max(c) w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c):
    """Exact subgradient at the unique max lag."""
    n = len(y)
    k = int(np.argmax(c))
    w_full = np.zeros(len(c))
    w_full[k] = 1.0
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_subgradient(y, c, k_count=5):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(y)
    top_k = np.argsort(c)[::-1][:k_count]
    vals = c[top_k]
    # Softmax weighting
    beta_w = 10.0
    w = np.exp(beta_w * (vals - vals.max()))
    w = w / w.sum()
    w_full = np.zeros(len(c))
    w_full[top_k] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_freq_domain(a_init, max_time, flatten_power=0.5):

    n = len(a_init)
    S = a_init.sum()

    start = time.monotonic()

    best_obj = compute_obj(a)
    best_time = start

        A = fft(a, fsize)
        P = np.abs(A) ** 2
        # Flatten
        P_max = P.max()
        P_min = P.min()
        if P_max < 1e-30:
            break
        # Adaptive flattening: stronger when spectrum is spiky
        dynamic_range = P_max / (P_min + 1e-30)
        power = flatten_power * min(1.0, 10.0 / dynamic_range)
        power = max(0.1, min(1.0, power))
        P_flat = P ** power
        # Reconstruct
        phase = np.angle(A)
        magnitude = np.sqrt(P_flat)
        A_new = magnitude * np.exp(1j * phase)
        a_new = np.abs(ifft(A_new))[:n]

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w

    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):

    n = len(a)

    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE smooth gradient of max(c) w.r.t. y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lag(y, k):
    """Subgradient contribution from a single lag k."""
    n = len(y)
    fsize = 1 << (2 * n).bit_length()
    # The gradient of c[k] w.r.t. y[i] is 2*y[k-i] (with bounds)
    # This equals 2 * ifft(FFT(e_k) * conj(FFT(y)))[:n] where e_k is delta at k
    w_full = np.zeros(2 * n - 1)
    if 0 <= k < 2 * n - 1:
        w_full[k] = 1.0
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_subgradient(y, c, lags, weights):
    """Weighted subgradient using specified lags and weights."""
    n = len(y)
    w_full = np.zeros(len(c))
    for k, w in zip(lags, weights):
        if 0 <= k < len(c):
            w_full[k] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_freq_domain(a_init, max_time, rng, base_power=0.5, damping=0.5):
    """
    Frequency-domain iterative method (error reduction style).
    Flattens the power spectrum to reduce peak autocorrelation.
    """
    n = len(a_init)
    S = a_init.sum()
    fsize = 1 << (2 * n).bit_length()
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    best_time = start
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break
        
        # Adaptive flattening power
        P_min_nonzero = np.min(P[P > P_max * 1e-10]) if np.any(P > P_max * 1e-10) else P.max() * 1e-10
        dynamic_range = P_max / (P_min_nonzero + 1e-30)
        # Stronger flattening when spectrum is spiky
        power = base_power * min(1.5, max(0.3, 20.0 / (dynamic_range + 1)))
        
        P_flat = P ** power
        phase = np.angle(A)
        magnitude = np.sqrt(P_flat)
        A_new = magnitude * np.exp(1j * phase)
        a_hard = np.abs(ifft(A_new))[:n]
        
        if a_hard.sum() < 1e-15:
            break
        
        # Damped update
        a_candidate = (1.0 - damping) * a + damping * a_hard
        a_candidate = a_candidate * (S / a_candidate.sum())
        
        obj = compute_obj(a_candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = a_candidate.copy()
            best_time = time.monotonic()
        
        a = a_candidate
        
        # Convergence check
        elapsed_since_best = time.monotonic() - best_time
        if elapsed_since_best > 60 and (time.monotonic() - start) > 120:
            break
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_max=5000.0, 
              beta_rate=0.008, use_hybrid=False):
    """FISTA with LSE smoothing and adaptive restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = 1.0 / max(beta, 1.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        # Gradient: LSE or hybrid
        if use_hybrid and (it - last_improve) > 2000:
            # Use exact subgradient at top few lags
            top_k = min(8, 1 + (it - last_improve) // 1000)
            k_idx = np.argsort(c)[::-1][:top_k]
            vals = c[k_idx]
            w = np.exp(5.0 * (vals - vals.max()))
            w = w / w.sum()
            grad = weighted_subgradient(y, c, k_idx, w)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart
        if it % 4000 == 0:
            t_mom = 1.0
        
        # Noise injection when stuck
        stuck = it - last_improve
        if stuck > 4000 and it % 2000 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 8000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_subgrad_avg(a_init, max_time, rng, alpha0=0.3, decay_rate=0.5):
    """
    Subgradient method with Polyak (jittered) averaging.
    Uses cycling active sets to escape degeneracy.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    avg_a = a.copy()
    it = 0
    last_improve = 0
    prev_a = a.copy()
    prev_step = None
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
        
        # Check average iterate
        c_avg = fft_convolve_self(avg_a)
        if c_avg.max() < best_conv - 1e-12:
            best_conv = c_avg.max()
            best_a = avg_a.copy()
            last_improve = it
        
        # Determine active lags (cycle through options)
        stuck = it - last_improve
        max_val = c.max()
        near_max = np.where(c >= max_val * 0.999)[0]
        
        if len(near_max) <= 4:
            lags = near_max
            weights = np.ones(len(lags)) / len(lags)
        elif stuck < 1000:
            # Use top 1
            lags = np.array([np.argmax(c)])
            weights = np.array([1.0])
        elif stuck < 5000:
            # Use top 3
            top3 = np.argsort(c)[::-1][:3]
            lags = top3
            w = np.exp(3.0 * (c[top3] - c[top3].max()))
            weights = w / w.sum()
        else:
            # Use more lags with broader weighting
            k = min(16, 4 + stuck // 1000)
            topk = np.argsort(c)[::-1][:k]
            lags = topk
            w = np.exp(1.0 * (c[topk] - c[topk].max()))
            weights = w / w.sum()
        
        grad = weighted_subgradient(a, c, lags, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: diminishing with jitter
        step = alpha0 * S / (gnorm * (1.0 + decay_rate * np.log1p(it)))
        step = min(step, 2.0 * S / max(gnorm, 1e-15))
        
        # Occasionally take a larger "exploration" step
        if stuck > 8000 and it % 5000 == 0:
            step *= 3.0
        
        a_new = project_simplex(a - step * grad, S)
        
        # Barzilai-Borwein style: if step caused increase, halve it
        if it > 1 and prev_step is not None:
            # Simple safeguard
            pass
        
        # Update average
        avg_a = ((it - 1) / it) * avg_a + (1.0 / it) * a_new
        
        prev_a = a.copy()
        prev_step = step
        a = a_new
        
        # Shake when very stuck
        if stuck > 15000 and it % 5000 == 0:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            avg_a = a.copy()
    
    # Compare final a vs avg_a
    c_a = fft_convolve_self(a).max()
    c_avg = fft_convolve_self(avg_a).max()
    if c_avg < best_conv:
        best_conv = c_avg
        best_a = avg_a.copy()
    if c_a < best_conv:
        best_conv = c_a
        best_a = a.copy()
    
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, target_len):
    """Downsample by averaging blocks to target length."""
    n = len(a)
    if target_len >= n:
        return a[:target_len].copy()
    n_new = target_len
    a_padded = np.zeros(n_new * (n // n_new + 1))
    a_padded[:n] = a
    a_padded = a_padded[:n_new * (n // n_new + 1)]
    n_blocks = n // n_new
    if n_blocks < 1:
        return a[:target_len].copy()
    a_trunc = a[:n_new * n_blocks]
    return a_trunc.reshape(n_new, n_blocks).mean(axis=1)
