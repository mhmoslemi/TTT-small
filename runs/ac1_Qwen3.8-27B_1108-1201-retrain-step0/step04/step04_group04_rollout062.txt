```
def resample_sequence(arr,n):
   arr=np.asarray(arr,float)
   if len(arr)==n: return arr.copy()
   if len(arr) > n:
      blocks=len(arr)//n
      if blocks<=0: blocks=1
      # average contiguous blocks and drop remainder
      m=blocks*n
      return arr[:m].reshape(n,blocks).mean(axis=1)
   else:
      t_old=np.linspace(0,1,len(arr),endpoint=False)
      t_new=np.linspace(0,1,n,endpoint=False)
      return np.interp(t_new,t_old,arr)
```

```python
"""
Minimize 2*n*max_conv / sum^2 for non-negative sequences.

This program combines:
  - parent/resampling starts and many structured/randomized inits
  - smoothed maximum-convolution FISTA (LSE/log-sum-exp)
  - fast exact 2-coordinate descent with corrected linear-convolution updates
  - batch projected-gradient moves
  - optional active-set LP direction polish (HiGHS if available)
  - multi-length probing and final polishing

No nested helper functions and no lambda functions are used.
"""

import time
import numpy as np
import scipy.fft as spfft

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

# Global best state
G_T0 = 0.0
G_DEADLINE = 0.0
G_BEST_SCORE = np.inf
G_BEST_ARR = None

PAIR_FRACS = (0.0005, 0.002, 0.008, 0.03, 0.1, 0.3, 1.0)


def next_pow2(x):
    if x <= 1:
        return 1
    p = 1
    while p < x:
        p <<= 1
    return p


def clean_array(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def normalize_to_sum(a, S):
    a = clean_array(a)
    if a is None or a.size == 0:
        return a
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, S / float(a.size), dtype=np.float64)
    return (a * (S / s)).astype(np.float64, copy=False)


def fix_simplex(a, S):
    a = clean_array(a)
    if a is None or a.size == 0:
        return a
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, S / float(a.size), dtype=np.float64)
    return (a * (S / s)).astype(np.float64, copy=False)


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    x = np.where(np.isfinite(v), v, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s0 = float(x.sum())
    if s0 <= 1e-300:
        return np.full(n, S / float(n), dtype=np.float64)

    r = np.sort(x)[::-1]
    cs = np.cumsum(r)
    idx = np.arange(1, n + 1, dtype=np.float64)
    valid = (cs - S) < idx * r
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cs[rho - 1] - S) / float(rho)
    y = np.maximum(x - theta, 0.0)
    y = np.clip(y, 0.0, 1000.0)
    sy = float(y.sum())
    if sy <= 1e-300:
        return np.full(n, S / float(n), dtype=np.float64)
    return (y * (S / sy)).astype(np.float64, copy=False)


def conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_pow2(L)
    A = spfft.rfft(a, size)
    c = spfft.irfft(A * A, size)
    c = c[:L]
    c = np.where(c > 0.0, c, 0.0)
    return c.astype(np.float64, copy=False)


def max_conv(a):
    c = conv_self(a)
    if c.size == 0:
        return np.inf
    return float(c.max())


def fast_score(a):
    a = clean_array(a)
    if a is None or a.size == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = conv_self(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * c.max() / (s * s))


def update_global(a, label=""):
    global G_BEST_SCORE, G_BEST_ARR
    a = clean_array(a)
    if a is None or a.size == 0:
        return
    sc = fast_score(a)
    if np.isfinite(sc) and sc < G_BEST_SCORE:
        G_BEST_SCORE = float(sc)
        G_BEST_ARR = a.copy()
        if label:
            print("  new global bound %.6f len=%d %s" % (G_BEST_SCORE, a.size, label))


def hankel_grad(a, w):
    """
    Gradient factor for sum_k w_k c_k:
    2 * corr(w, a) where (corr)_i = sum_l w_l a_{l-i}; one-sided linear convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return a.copy()
    size = next_pow2(3 * n - 2)
    W = spfft.rfft(w, size)
    A = spfft.rfft(a, size)
    corr = spfft.irfft(W * np.conj(A), size)
    return corr[:n].astype(np.float64, copy=False)


def lse_softmax_value(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.zeros(0), np.inf
    cmax = float(c.max())
    z = beta * (c - cmax)
    zmax = float(z.max())
    ex = np.exp(z - zmax)
    s = float(ex.sum())
    if s < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return p, cmax
    p = ex / s
    F = cmax + (zmax + float(np.log(s))) / float(beta)
    return p, float(F)


def make_weights(c, beta, topK=None):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w
    cmax = float(c.max())

    if topK is not None and 0 < topK < L:
        idx = np.argpartition(c, -topK)[-topK:]
        z = beta * (c[idx] - cmax)
        zmax = float(z.max())
        ex = np.exp(z - zmax)
        s = float(ex.sum())
        if s > 1e-300:
            w[idx] = ex / s
        else:
            w[idx] = 1.0 / float(idx.size)
    else:
        z = beta * (c - cmax)
        zmax = float(z.max())
        ex = np.exp(z - zmax)
        s = float(ex.sum())
        if s > 1e-300:
            w = ex / s
        else:
            w[int(np.argmax(c))] = 1.0

    tot = float(w.sum())
    if not np.isfinite(tot) or tot <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= tot
    return w


def fista_beta(a, S, seg_deadline, beta):
    a = normalize_to_sum(a, S)
    n = a.size
    if n == 0 or not np.isfinite(seg_deadline) or seg_deadline <= time.monotonic():
        return a, max_conv(a)

    x = a.copy()
    y = a.copy()
    t = 1.0
    L = 2.0
    best = max_conv(x)
    best_a = x.copy()
    no_improve = 0
    noise_done = 0

    while time.monotonic() < seg_deadline - 0.05:
        cy = conv_self(y)
        cmax_y = float(cy.max())
        p, Fy = lse_softmax_value(cy, beta)
        g = 2.0 * hankel_grad(y, p)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        x_try = None
        cx = None
        for _ in range(8):
            step = 1.0 / max(float(L), 1e-12)
            x_try = project_simplex(y - step * g, S)
            cx = conv_self(x_try)
            Fx = lse_softmax_value(c_x, beta)[1] if (c_x := cx) is not None else np.inf
            diff = x_try - y
            cond = Fx <= Fy + float(np.dot(g, diff)) + 0.5 * float(np.dot(diff, diff)) / max(float(L), 1e-12) + 1e-12
            if cond:
                break
            L *= 2.0
            if L > 1e7:
                break

        if x_try is None or cx is None:
            x_try = project_simplex(y - g / max(1e8, L), S)
            cx = conv_self(x_try)
            x_try = project_simplex(y - g / max(1e8, L), S)
            cx = conv_self(x_try)

        t_old = t
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mom = (t_old - 1.0) / t_new
        y_new = x_try + mom * (x_try - x)
        y_new = fix_simplex(y_new, S)

        t = t_new
        x = x_try
        Mx = float(cx.max())
        if Mx < best - 1e-12:
            best = Mx
            best_a = x_try.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1200 and noise_done < 12:
            scale = 0.004 * S / np.sqrt(float(n))
            y = project_simplex(y + np.random.default_rng(int(no_improve + 1)).normal(0.0, scale, n), S)
            x = project_simplex(x + np.random.default_rng(int(no_improve + 101)).normal(0.0, 0.5 * scale, n), S)
            t = 1.0
            L = 2.0
            no_improve = 0
            noise_done += 1

        if no_improve > 400 or t > 250.0:
            t = 1.0
            y = x.copy()
            L = 2.0
            no_improve = 0

    c_best = conv_self(best_a)
    Mb = float(c_best.max())
    return best_a, Mb


def lse_fista(a, S, budget, rng=None):
    budget = float(budget)
    if budget <= 1.0:
        return normalize_to_sum(a, S), max_conv(a)

    a = normalize_to_sum(a, S)
    best = max_conv(a)
    best_a = a.copy()
    t_start = time.monotonic()
    deadline = t_start + budget

    betas = (0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 1024.0, 3000.0)
    weights = (0.03, 0.06, 0.10, 0.16, 0.22, 0.20, 0.16, 0.07)
    wsum = float(sum(weights))
    weights = tuple(w / wsum for w in weights)

    for beta, fr in zip(betas, weights):
        if time.monotonic() > deadline:
            break
        seg_budget = budget * fr
        if seg_budget <= 1.0:
            continue
        seg_deadline = min(deadline, time.monotonic() + seg_budget)
        a_cur, m_cur = fista_beta(best_a, S, seg_deadline, beta)
        if m_cur < best - 1e-13:
            best = m_cur
            best_a = a_cur.copy()
        # If this beta did not help, still keep state from best_a.

    c_best = conv_self(best_a)
    return best_a, float(c_best.max())


def batch_gradient_move(a, S, c, M, beta=20.0, topK=128):
    a = fix_simplex(a, S)
    n = a.size
    w = make_weights(c, beta, topK)
    g = 2.0 * hankel_grad(a, w)
    d = -g
    d = d - float(np.mean(d))
    l1 = float(np.abs(d).sum())
    if l1 <= 1e-15:
        return None
    target_var = 0.05 * S
    d = d * (target_var / l1)

    for factor in (1.0, 0.6, 0.4, 0.25, 0.12, 0.05):
        trial = project_simplex(a + factor * d, S)
        ct = conv_self(trial)
        mt = float(ct.max())
        if mt < M - 1e-12:
            return trial, ct, mt
    return None


def pair_candidate_deltas(pos, neg, max_r=14):
    vals = [0.0]
    if pos > 0.0:
        for f in PAIR_FRACS:
            vals.append(float(pos * f))
    if neg > 0.0:
        for f in PAIR_FRACS:
            vals.append(float(-neg * f))
    arr = np.asarray(vals, dtype=np.float64)
    lo = -neg
    hi = pos
    arr = arr[(arr >= lo - 1e-15) & (arr <= hi + 1e-15)]
    if arr.size == 0:
        return np.asarray([0.0], dtype=np.float64)
    arr = np.unique(arr)
    if arr.size > max_r:
        idx = np.linspace(0, arr.size - 1, int(max_r)).round().astype(int)
        arr = arr[idx]
    return arr


def coordinate_descent(a, S, budget, rng):
    a = normalize_to_sum(a, S)
    n = a.size
    if n == 0 or float(budget) <= 1e-6:
        return a, max_conv(a)

    t_start = time.monotonic()
    deadline = t_start + float(budget)
    L = 2 * n - 1
    c = conv_self(a)
    M = float(c.max())
    best = M
    best_a = a.copy()

    active_K = 128 if n <= 16384 else 80
    grad_beta = 10.0
    g = 2.0 * hankel_grad(a, make_weights(c, grad_beta, active_K))
    g = g - float(np.mean(g))
    g_order = np.argsort(g)

    top_count = max(20, min(220, n // 12))
    d_arr = np.zeros(L, dtype=np.float64)
    q_arr = np.zeros(L, dtype=np.float64)

    trial = 0
    accept = 0
    no_improve = 0
    last_batch = t_start
    last_full = t_start

    while time.monotonic() < deadline - 0.05:
        now = time.monotonic()

        if now - last_batch > 1.0:
            last_batch = now
            res = batch_gradient_move(a, S, c, M, beta=20.0, topK=active_K)
            if res is not None:
                a, c, M = res
                if M < best - 1e-13:
                    best = M
                    best_a = a.copy()
                no_improve = 0
                g = 2.0 * hankel_grad(a, make_weights(c, grad_beta, active_K))
                g = g - float(np.mean(g))
                g_order = np.argsort(g)

        if now - last_full > 3.0 or accept > 1500:
            last_full = now
            accept = 0
            a = fix_simplex(a, S)
            c = conv_self(a)
            M = float(c.max())
            if M < best - 1e-13:
                best = M
                best_a = a.copy()
            g = 2.0 * hankel_grad(a, make_weights(c, grad_beta, active_K))
            g = g - float(np.mean(g))
            g_order = np.argsort(g)

        top_idx = g_order[-top_count:]
        bot_idx = g_order[:top_count]

        if top_idx.size > 0 and bot_idx.size > 0 and rng.random() < 0.72:
            src_idx = int(top_idx[int(rng.integers(top_idx.size))])
            dst_idx = int(bot_idx[int(rng.integers(bot_idx.size))])
        else:
            src_idx = int(rng.integers(0, n))
            dst_idx = int(rng.integers(0, n))

        if src_idx == dst_idx or a[src_idx] <= 1e-14:
            src_idx = int(rng.integers(0, n))
            dst_idx = int(rng.integers(0, n))
            if src_idx == dst_idx or a[src_idx] <= 1e-14:
                no_improve += 1
                trial += 1
                if no_improve > 4000:
                    scale = 0.003 * S / np.sqrt(float(n))
                    a = project_simplex(a + rng.normal(0.0, scale, n), S)
                    c = conv_self(a)
                    M = float(c.max())
                    if M < best - 1e-13:
                        best = M
                        best_a = a.copy()
                    no_improve = 0
                    trial = 0
                    g = 2.0 * hankel_grad(a, make_weights(c, grad_beta, active_K))
                    g = g - float(np.mean(g))
                    g_order = np.argsort(g)
                continue

        pos = float(a[src_idx])
        neg = float(a[dst_idx])
        if pos <= 1e-16 and neg <= 1e-16:
            continue

        d_arr.fill(0.0)
        q_arr.fill(0.0)

        # Exact first-order d for linear self-convolution:
        # target dst increases, source src decreases.
        d_arr[dst_idx:dst_idx + n] += 2.0 * a
        d_arr[src_idx:src_idx + n] -= 2.0 * a

        q_arr[2 * dst_idx] += 1.0
        q_arr[2 * src_idx] += 1.0
        if dst_idx != src_idx:
            q_arr[dst_idx + src_idx] -= 2.0

        deltas = pair_candidate_deltas(pos, neg, 14)
        if deltas.size == 0:
            continue

        if deltas.size == 1:
            bd = float(deltas[0])
            bv = float((c + bd * d_arr + bd * bd * q_arr).max())
            vals = np.asarray([bv])
        else:
            D = deltas[:, None]
            mat = c[None, :] + D * d_arr[None, :] + (D * D) * q_arr[None, :]
            vals = mat.max(axis=1)

        bi = int(np.argmin(vals))
        bd = float(deltas[bi])
        bv = float(vals[bi])

        if bd != 0.0 and bv < M - 1e-12:
            a[dst_idx] += bd
            a[src_idx] -= bd
            if a[dst_idx] < -1e-12 or a[src_idx] < -1e-12:
                a[dst_idx] = max(0.0, float(a[dst_idx]))
                a[src_idx] = max(0.0, float(a[src_idx]))
            c += bd * d_arr + bd * bd * q_arr
            if float(c.min()) < -1e-7:
                c = np.where(c > 0.0, c, 0.0)
            Mnew = float(c.max())
            if Mnew < best - 1e-13:
                best = Mnew
                best_a = a.copy()
            M = Mnew
            no_improve = 0
            accept += 1
        else:
            no_improve += 1

        trial += 1
        if no_improve > 4500:
            scale = 0.004 * S / np.sqrt(float(n)) * (0.5 + 0.75 * float(rng.random()))
            a = project_simplex(a + rng.normal(0.0, scale, n), S)
            c = conv_self(a)
            M = float(c.max())
            if M < best - 1e-13:
                best = M
                best_a = a.copy()
            no_improve = 0
            trial = 0
            g = 2.0 * hankel_grad(a, make_weights(c, grad_beta, active_K))
            g = g - float(np.mean(g))
            g_order = np.argsort(g)

    cb = conv_self(best_a)
    Mb = float(cb.max())
    return best_a, Mb


def lp_direction(a, c, active_k, upper):
    if linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    K = active_k.size
    if n == 0 or K == 0:
        return None

    A = np.zeros((K, n + 1), dtype=np.float64)
    for r in range(K):
        k = int(active_k[r])
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            A[r, idx] = 2.0 * a[k - idx]
        A[r, n] = -1.0

    b = np.zeros(K, dtype=np.float64)
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    bounds = []
    for i in range(n):
        bounds.append((-float(a[i]), float(upper)))
    bounds.append((-np.inf, np.inf))

    try:
        sol = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, bounds=bounds, method="highs")
    except Exception:
        return None
    if sol is None or not sol.success:
        return None
    h = np.asarray(sol.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return None
    h = np.clip(h, -a + 1e-15, float(upper))
    h -= float(np.mean(h))
    if float(np.abs(h).sum()) < 1e-12:
        return None
    return h


def lp_polish(a, S, budget, rng):
    if float(budget) <= 2.0 or linprog is None:
        return a, max_conv(a)
    a = normalize_to_sum(a, S)
    n = a.size
    c = conv_self(a)
    M = float(c.max())
    best = M
    best_a = a.copy()
    t_start = time.monotonic()
    deadline = t_start + float(budget)
    upper = 0.10 * S
    K = 48 if n <= 10000 else 28
    calls = 0
    max_calls = 6 if n <= 10000 else 3
    no_improve = 0

    while calls < max_calls and time.monotonic() < deadline - 1.0:
        calls += 1
        if c.size == 0:
            break
        act = np.argpartition(c, -K)[-K:]
        h = lp_direction(a, c, act, upper)
        if h is None:
            break

        best_t = 0.0
        best_m = M
        for t in (1.0, 0.65, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02):
            trial = a + t * h
            trial = fix_simplex(trial, S)
            ct = conv_self(trial)
            mt = float(ct.max())
            if mt < best_m - 1e-13:
                best_m = mt
                best_t = float(t)
                if mt < 1e-14:
                    break

        if best_t > 0.0 and best_m < M - 1e-12:
            a = fix_simplex(a + best_t * h, S)
            c = conv_self(a)
            M = float(c.max())
            if M < best - 1e-13:
                best = M
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    cb = conv_self(best_a)
    return best_a, float(cb.max())


def run_local(a, S, budget, rng, tag="local"):
    a = normalize_to_sum(a, S)
    if a is None or a.size == 0 or float(budget) <= 0.5:
        return a, np.inf

    t_start = time.monotonic()
    deadline = t_start + float(budget)
    best_a = a.copy()
    c = conv_self(a)
    best_M = float(c.max())

    def report(arr):
        nonlocal best_a, best_M
        if arr is not None:
            sc = fast_score(arr)
            if np.isfinite(sc) and sc < G_BEST_SCORE:
                update_global(arr, tag)
        return arr

    budget = max(0.0, deadline - time.monotonic())

    if budget > 30.0:
        fista_alloc = min(0.28 * (deadline - t_start), max(5.0, 0.55 * budget))
        fista_alloc = min(fista_alloc, max(0.0, deadline - time.monotonic() - 5.0))
        if fista_alloc > 4.0:
            a_f, m_f = lse_fista(best_a, S, fista_alloc, rng)
            if m_f < best_M - 1e-13:
                best_M = m_f
                best_a = a_f.copy()
            best_a = report(best_a)
    else:
        fista_alloc = 0.0

    remaining = deadline - time.monotonic()
    coord_alloc = min(0.50 * (deadline - t_start), max(2.0, 0.75 * remaining))
    coord_alloc = min(coord_alloc, max(0.0, remaining - 5.0))
    if coord_alloc > 2.0:
        a_c, m_c = coordinate_descent(best_a, S, coord_alloc, rng)
        if m_c < best_M - 1e-13:
            best_M = m_c
            best_a = a_c.copy()
        best_a = report(best_a)

    remaining = deadline - time.monotonic()
    if best_a.size <= 16384 and remaining > 20.0:
        lp_alloc = min(0.18 * (deadline - t_start), max(5.0, 0.45 * remaining))
        lp_alloc = min(lp_alloc, max(0.0, remaining - 5.0))
        if lp_alloc > 5.0:
            a_l, m_l = lp_polish(best_a, S, lp_alloc, rng)
            if m_l < best_M - 1e-13:
                best_M = m_l
                best_a = a_l.copy()
            best_a = report(best_a)

    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        final_alloc = min(0.25 * (deadline - t_start), max(2.0, 0.60 * remaining))
        if final_alloc > 2.0:
            a_final, m_final = coordinate_descent(best_a, S, final_alloc, rng)
            if m_final < best_M - 1e-13:
                best_M = m_final
                best_a = a_final.copy()
            best_a = report(best_a)

    c_best = conv_self(best_a)
    return best_a, float(c_best.max())


def resample_sequence(arr, n):
    arr = clean_array(arr)
    if arr is None or arr.size == 0:
        return np.ones(n, dtype=np.float64)
    if arr.size == n:
        return arr.copy()
    if arr.size > n:
        blocks = max(1, arr.size // n)
        m = blocks * n
        if m <= 0:
            return np.ones(n, dtype=np.float64)
        return arr[:m].reshape(n, blocks).mean(axis=1)
    t_old = np.linspace(0.0, 1.0, arr.size, endpoint=False)
    t_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(t_new, t_old, arr).astype(np.float64)


def random_phase_start(n, rng):
    S = np.sqrt(2.0 * float(n))
    fsize = next_pow2(2 * n - 1)
    F = np.exp(1j * rng.uniform(0.0, 2.0 * np.pi, fsize))
    freq = spfft.fftfreq(fsize)
    F *= np.exp(-1.2 * (freq / 0.45) ** 2)
    F[0] = S
    a = np.real(spfft.ifft(F))[:n]
    a = np.maximum(a, 0.0)
    a = normalize_to_sum(a, S)
    if a is None or float(a.sum()) <= 1e-15:
        a = np.ones(n, dtype=np.float64) * S / float(n)
    return a


def generate_starts(parent, n, rng):
    S = np.sqrt(2.0 * float(n))
    T = np.linspace(0.0, 1.0, n, endpoint=False)
    starts = []

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    base = rng.random(n) + 0.08
    starts.append(base)

    for alpha in (0.35, 1.0, 3.0):
        r = rng.random(n) + 0.08
        floor = 0.12 * float(np.mean(base))
        a = np.maximum(0.0, r + alpha * (r - r[::-1]) + floor)
        starts.append(a)

    for freq, amp in ((0.5, 1.6), (1.0, 1.2), (2.0, 0.9), (4.0, 0.6)):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        shape = np.maximum(0.05, 1.0 + amp * np.cos(2.0 * np.pi * freq * T + phase))
        starts.append(base * shape)

    edge = 1.0 + 5.0 * (T - 0.5) ** 2 * 4.0
    starts.append(base * edge)

    r2 = rng.random(n) + 0.08
    starts.append(np.maximum(0.0, r2 + 1.8 * (r2 - r2[::-1]) + 0.18 * float(np.mean(base)) * edge))

    r3 = rng.random(n) + 0.08
    left = np.exp(-6.0 * T)
    right = np.exp(-6.0 * (1.0 - T))
    starts.append(np.maximum(0.0, r3 * (0.25 + 2.2 * (left + right)) + 0.05))

    for _ in range(4):
        starts.append(random_phase_start(n, rng))

    clean = []
    for a in starts:
        a2 = normalize_to_sum(a, S)
        if a2 is not None and a2.size == n and float(a2.sum()) > 1e-12:
            clean.append(a2)
    if not clean:
        clean.append(np.ones(n, dtype=np.float64) * S / float(n))
    return clean


def propose_candidate(seed=42, budget_s=2400):
    global G_T0, G_DEADLINE, G_BEST_SCORE, G_BEST_ARR

    t0 = time.monotonic()
    G_T0 = t0
    G_BEST_SCORE = np.inf
    G_BEST_ARR = None
    budget_s = float(budget_s)
    G_DEADLINE = t0 + max(0.0, budget_s - 60.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = height_sequence_1
    except Exception:
        parent = None
    parent = clean_array(parent)
    if parent is not None:
        if parent.size < 64 or float(parent.sum()) < 1e-9:
            parent = None

    if parent is not None:
        base_n = int(parent.size)
    else:
        base_n = 8192
    base_n = int(min(max(base_n, 1024), 16384))

    cand = [base_n]
    if 2 * base_n <= 16384:
        cand.append(2 * base_n)
    if base_n // 2 >= 1024:
        cand.append(base_n // 2)
    for x in (4096, 8192, 16384):
        if 1024 <= x <= 16384:
            cand.append(x)

    lengths = []
    for x in cand:
        x = int(x)
        if x not in lengths:
            lengths.append(x)
    if len(lengths) > 5:
        lengths = lengths[:5]

    print("propose_candidate start: budget=%.1fs lengths=%s parent_len=%s" % (
        budget_s, lengths, parent.size if parent is not None else None))

    best_by_len = {}
    starts_by_len = {}

    probe_total = min(180.0, max(30.0, (G_DEADLINE - t0) * 0.10))
    if G_DEADLINE > t0:
        probe_total = min(probe_total, G_DEADLINE - t0)

    if len(lengths) and probe_total > 10.0:
        per_len = probe_total / float(len(lengths))
        for n in lengths:
            if time.monotonic() > G_DEADLINE:
                break
            S = np.sqrt(2.0 * float(n))
            arrs = generate_starts(parent, n, rng)
            scored = []
            for idx, a in enumerate(arrs):
                a = normalize_to_sum(a, S)
                if a is None or a.size != n:
                    continue
                m = max_conv(a)
                if np.isfinite(m):
                    scored.append((m, idx))
            if not scored:
                a0 = np.ones(n, dtype=np.float64) * S / float(n)
                best_by_len[n] = [max_conv(a0), a0.copy()]
                starts_by_len[n] = [a0.copy()]
                continue

            scored.sort()
            top_idx = [i for _, i in scored[:3]]
            top_arrs = [arrs[i] for i in top_idx]
            starts_by_len[n] = [x.copy() for x in top_arrs]
            cur_best_sc = float(scored[0][0])
            cur_best_arr = top_arrs[0].copy()
            best_by_len[n] = [cur_best_sc, cur_best_arr]

            # Quick refinement on top probe starts
            if len(top_arrs) and per_len > 5.0:
                allocs = [per_len * 0.45, per_len * 0.35, per_len * 0.20][:len(top_arrs)]
                asum = float(sum(allocs))
                if asum > 0:
                    allocs = [x / asum * per_len for x in allocs]
                for a_init, alloc in zip(top_arrs, allocs):
                    if alloc <= 2.0:
                        continue
                    if time.monotonic() > G_DEADLINE:
                        break
                    a_out, m_out = run_local(a_init, S, alloc, rng, "probe_n%d" % n)
                    sc_out = fast_score(a_out)
                    if np.isfinite(sc_out) and sc_out < best_by_len[n][0]:
                        best_by_len[n] = [sc_out, a_out.copy()]
                        update_global(a_out, "probe_n%d" % n)

    items = []
    for n, val in best_by_len.items():
        if np.isfinite(val[0]):
            items.append((float(val[0]), int(n)))
    items.sort()

    if not items:
        n = 8192
        S = np.sqrt(2.0 * float(n))
        a0 = np.ones(n, dtype=np.float64) * S / float(n)
        update_global(a0, "fallback_uniform")
        items = [(fast_score(a0), n)]

    remaining = G_DEADLINE - time.monotonic()
    print("After probe: best=%.6f items=%s remaining=%.1fs" % (
        G_BEST_SCORE if np.isfinite(G_BEST_SCORE) else float('inf'),
        [(round(s, 5), n) for s, n in items], remaining))

    usable = max(0.0, remaining - 45.0)
    weights = []
    for pos in range(min(4, len(items))):
        if pos == 0:
            weights.append(0.56)
        elif pos == 1:
            weights.append(0.25)
        elif pos == 2:
            weights.append(0.14)
        else:
            weights.append(0.05)
    if not weights:
        weights = [1.0]
    wsum = float(sum(weights))
    weights = [w / wsum for w in weights]

    for (sc, n), wt in zip(items[:4], weights):
        if time.monotonic() > G_DEADLINE:
            break
        alloc = usable * float(wt)
        if alloc <= 3.0:
            continue
        S = np.sqrt(2.0 * float(n))
        arrs = []
        if n in best_by_len and best_by_len[n][1] is not None:
            arrs.append(np.asarray(best_by_len[n][1], dtype=np.float64).copy())
        if n in starts_by_len:
            for a in starts_by_len[n]:
                if a is not None and a.size == n:
                    arrs.append(np.asarray(a, dtype=np.float64).copy())
                if len(arrs) >= 3:
                    break
        if not arrs:
            arrs.append(np.ones(n, dtype=np.float64) * S / float(n))

        arr_weights = [0.62, 0.26, 0.12][:len(arrs)]
        awsum = float(sum(arr_weights))
        if awsum > 0:
            arr_weights = [w / awsum for w in arr_weights]

        for a_init, a_w in zip(arrs, arr_weights):
            if time.monotonic() > G_DEADLINE:
                break
            t_alloc = alloc * float(a_w)
            if t_alloc <= 2.0:
                continue
            a_out, m_out = run_local(a_init, S, t_alloc, rng, "deep_n%d" % n)
            sc_out = fast_score(a_out)
            if np.isfinite(sc_out) and sc_out < best_by_len[n][0]:
                best_by_len[n] = [sc_out, a_out.copy()]
            update_global(a_out, "deep_n%d" % n)

    # Final polish directly on the current global best.
    remaining = G_DEADLINE - time.monotonic()
    if G_BEST_ARR is not None and remaining > 20.0:
        arr = np.asarray(G_BEST_ARR, dtype=np.float64).copy()
        mfinal = arr.size
        Sfinal = np.sqrt(2.0 * float(mfinal))
        final_alloc = min((G_DEADLINE - time.monotonic()) * 0.65, max(8.0, 120.0))
        if final_alloc > 5.0:
            a_pol, m_pol = coordinate_descent(arr, Sfinal, final_alloc, rng)
            update_global(a_pol, "final_coord")
            remaining = G_DEADLINE - time.monotonic()
            if remaining > 25.0 and a_pol.size <= 16384:
                a_lp, m_lp = lp_polish(a_pol, Sfinal, min(remaining * 0.5, 60.0), rng)
                update_global(a_lp, "final_lp")
            remaining = G_DEADLINE - time.monotonic()
            if remaining > 15.0:
                a_fin, m_fin = lse_fista(a_pol, Sfinal, min(remaining * 0.5, 60.0), rng)
                update_global(a_fin, "final_fista")

    # Choose final array. Prefer global best; otherwise fallback.
    final_a = G_BEST_ARR
    if final_a is None or len(final_a) == 0:
        n = 1024
        S = np.sqrt(2.0 * float(n))
        final_a = np.ones(n, dtype=np.float64) * S / float(n)
    else:
        final_a = np.asarray(final_a, dtype=np.float64).copy()
        S = np.sqrt(2.0 * float(final_a.size))
        final_a = normalize_to_sum(final_a, S)
        update_global(final_a, "final_choice")

    final_a = np.clip(final_a, 0.0, 1000.0)
    if final_a.size == 0 or float(final_a.sum()) < 0.01:
        n = 256
        S = np.sqrt(2.0 * float(n))
        final_a = np.ones(n, dtype=np.float64) * S / float(n)

    final_list = [float(x) for x in final_a]
    try:
        ver = evaluate_sequence(final_list)
        ver = float(ver)
    except Exception:
        ver = np.inf

    if not np.isfinite(ver):
        n = 256
        S = np.sqrt(2.0 * float(n))
        final_a = np.ones(n, dtype=np.float64) * S / float(n)
        final_list = [float(x) for x in final_a]
        try:
            ver = float(evaluate_sequence(final_list))
        except Exception:
            ver = np.inf

    print("propose_candidate done: returned_len=%d verify=%.6f total_time=%.1fs global_best=%.6f" % (
        len(final_list), ver, time.monotonic() - t0, G_BEST_SCORE if np.isfinite(G_BEST_SCORE) else float('inf')))
    return final_list
```
