"""
Minimize 2*n*max_conv/sum^2 using multi-n search, L-BFGS on LSE, 
peak peeling, and KKT-guided refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def convolve_fft(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A

def lse_of_topk(c, k, beta):
    """LSE of the top-k values of c."""
    top_k_vals = np.sort(c)[::-1][:k]
    m = top_k_vals.max()
    exp_bc = np.exp(beta * (top_k_vals - m))
    total = exp_bc.sum()
    if total < 1e-300:
        return m
    return m + np.log(total) / beta

def topk_subgradient(a, N, k):
    """Subgradient of the sum of top-k convolution values."""
    n = len(a)
    c, A = convolve_fft(a, N)
    top_idx = np.argsort(c)[::-1][:k]
    # Subgradient is sum of 2*a shifted for each top-k position
    # c_k = sum_i a_i * a_{k-i}, so d c_k / d a_i = 2 * a_{k-i} (with bounds)
    # Using FFT: grad = IFFT( sum_{j in topk} 2 * conj(A) * exp(2j*pi*j*m/N) )
    # Actually, let's use the direct formula via FFT
    # The gradient of c_k w.r.t. a is: 2 * (a circularly shifted by k) restricted to [0,n-1]
    # In FFT domain: grad_freq = 2 * conj(A) * exp(2j*pi*k/N) for each k in top_idx
    # But we need to be careful with the linear vs circular convolution.
    
    # Simpler approach: use the fact that grad of c_k w.r.t. a_i = 2*a_{k-i} (if 0 <= k-i < n)
    # So grad = sum_{j in top_idx} 2 * shift(a, j)
    # where shift(a, j)_i = a_{j-i} if 0 <= j-i < n else 0
    # This is equivalent to: grad = 2 * (indicator_topk * a) where * is cross-correlation
    # In FFT: grad = real(IFFT(FFT(indicator_topk, N) * conj(A)))
    
    # Actually, let me think more carefully.
    # c_k = sum_i a_i * a_{k-i} (with a_m = 0 for m outside [0,n-1])
    # d c_k / d a_i = a_{k-i} + (1 if k-i == i else 0) * a_i... 
    # No, d c_k / d a_i = sum over all terms involving a_i.
    # c_k = sum_j a_j * a_{k-j}
    # The terms involving a_i are: j=i gives a_i * a_{k-i}, and j=k-i gives a_{k-i} * a_i.
    # If k-i != i: both terms are present, giving 2 * a_{k-i}
    # If k-i == i (k=2i): only one term a_i^2, giving 2 * a_i = 2 * a_{k-i}
    # So d c_k / d a_i = 2 * a_{k-i} in all cases.
    
    # So grad of sum of top-k c's = sum_{j in top_idx} 2 * a_{j-i} for each i
    # = 2 * sum_{j in top_idx} a_{j-i}
    # This is 2 * (h * a) where h is the indicator of top_idx, and * is cross-correlation
    # (h * a)(i) = sum_j h_j * a_{j-i}
    
    # In FFT: cross-correlation h * a = IFFT(FFT(h, N) * conj(FFT(a, N)))
    # Wait, cross-correlation is IFFT(conj(FFT(h)) * FFT(a)) or IFFT(FFT(h) * conj(FFT(a)))?
    # Cross-correlation: (h ⋆ a)(i) = sum_j h(j) * conj(a)(j-i) = IFFT(conj(FFT(h)) * FFT(a))
    # But we want sum_j h_j * a_{j-i} which is (h ⋆ a)(i) for real signals = IFFT(FFT(h) * conj(FFT(a)))
    # Hmm, let me just verify: if h is delta at position p, then (h ⋆ a)(i) = a_{p-i}.
    # FFT(h) = exp(-2j*pi*p/N) (for DFT convention). 
    # IFFT(FFT(h) * conj(FFT(a))) at position i = ?
    # Actually, let me use the correlation theorem:
    # corr(h, a) = IFFT(conj(FFT(h)) * FFT(a))
    # (corr(h,a))(i) = sum_j h(j) * a((j-i) mod N)
    # For our case, we want sum_j h_j * a_{j-i} (with a_m = 0 for m outside [0,n-1])
    # This is the circular correlation if we pad a to length N.
    
    # So: grad = 2 * real(IFFT(conj(FFT(h, N)) * A))[:n]
    # where h is the indicator of top_idx (length 2n-1, padded to N)
    
    h = np.zeros(N)
    for idx in top_idx:
        h[int(idx)] = 1.0
    H = fft(h)
    # Cross-correlation: IFFT(conj(H) * A)
    grad_full = np.real(ifft(np.conj(H) * A))
    grad = 2.0 * grad_full[:n]
    
    return grad, c, top_idx

def lbfgs_solve(g, s_hist, y_hist, m):
    """L-BFGS two-loop recursion."""
    n = len(g)
    gn = np.linalg.norm(g)
    if len(s_hist) == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    
    q = g.copy()
    alphas = np.zeros(len(s_hist))
    for i in range(len(s_hist) - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    
    for i in range(len(s_hist)):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def optimize_lse(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10, top_k=50):
    """L-BFGS on top-k LSE-smoothed objective."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()

    else:
        a = a * (S / s)
    
    beta = beta_init
    best_a = a.copy()

    s_hist = []
    y_hist = []
    
    grad, c, _ = topk_subgradient(a, N, top_k)
    c_max = c.max()

        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = S / (n * np.sqrt(top_k))

    while time.monotonic() - start_time < max_time:

        if it % 100 == 0:

            c_max = c.max()
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_solve(grad, s_hist, y_hist, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Backtracking line search
        alpha_ls = step_scale
        a_new = a

                    best_alpha = alpha_ls
                    break

            best_a = a_new.copy()
        
        grad_new, c_nw, _ = topk_subgradient(a_new, N, top_k)
        c_max = c_nw.max()
        
        s_k = a_new - a
        y_k = grad_new - grad
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        
        if sy > 1e-14 * max(ss, 1e-30):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        grad = grad_new
        
        # Periodic perturbation
        if it % 300 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            s_hist = []
            y_hist = []
            grad, c, _ = topk_subgradient(a, N, top_k)
            c_max = c.max()
    
    return best_a, best_cmax

def peak_peel(a_init, n, S, N, max_time, rng, top_k=10, n_rounds=50):
    """Iteratively reduce the top peaks by adjusting coordinates."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = convolve_fft(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        # For each top peak, compute the gradient direction
        # and take a small step to reduce it
        grad = np.zeros(n)
        for idx in top_idx:
            idx = int(idx)
            # d c_idx / d a_i = 2 * a_{idx - i} for 0 <= idx - i < n
            # So the gradient is 2 * a reversed and shifted
            lo = max(0, idx - n + 1)
            hi = min(n - 1, idx)
            # For i in [lo, hi]: d c_idx / d a_i = 2 * a_{idx - i}
            # a_{idx - i} for i in [lo, hi] is a[hi:lo-1:-1] if hi >= lo
            if hi >= lo:
                grad[lo:hi+1] += 2.0 * a[hi:lo-1:-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step: reduce the peak by moving against the gradient
        # But we need to be careful not to increase other peaks
        step = 0.1 * S / (n * gnorm * np.sqrt(top_k))
        a_new = project_simplex(a - step * grad / gnorm, S)
        
        c_new, _ = convolve_fft(a_new, N)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try a smaller step
            step *= 0.5
            a_new = project_simplex(a - step * grad / gnorm, S)
            c_new2, _ = convolve_fft(a_new, N)
            if c_new2.max() < cur:
                a = a_new
            else:
                # Perturb and continue
                if rng.random() < 0.5:
                    a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

def spectral_shaping(a_init, n, S, N, max_time, rng, top_k=15, alpha=0.15):
    """Spectral shaping to flatten the convolution."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = convolve_fft(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
        
        P = np.abs(A) ** 2
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)
        
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

def make_init_chirp(n, rng):

"""
Minimize 2*n*max_conv/sum^2 using multi-n search, L-BFGS on top-k LSE,
peak peeling, spectral shaping, and asymmetric initializations.
Key improvements: tries multiple sequence lengths, uses smooth LSE-topk gradient,
and exploits the fact that optimal sequences are asymmetric (one-sided).

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def convolve_fft(a, N):
    """Self-convolution via FFT. Returns (c of length 2n-1, A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_topk_gradient(a, N, top_k, beta):
    """Smooth LSE of top-k convolution values and its gradient."""
    n = len(a)
    c, A = convolve_fft(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    c_max = top_vals.max()
    exp_beta = np.exp(beta * (top_vals - c_max))
    total = exp_beta.sum()
    if total < 1e-300:
        weights = np.ones(top_k) / top_k
        lse_val = c_max
    else:
        weights = exp_beta / total
        lse_val = c_max + np.log(total) / beta
    h = np.zeros(N)
    for i in range(top_k):
        h[int(top_idx[i])] = weights[i]
    H = fft(h)
    grad_full = np.real(ifft(np.conj(H) * A))
    grad = 2.0 * grad_full[:n]
    return grad, lse_val, c


def lbfgs_direction(g, s_hist, y_hist):
    """L-BFGS two-loop recursion. Returns descent direction."""
    n = len(g)
    gn = np.linalg.norm(g)
    m = len(s_hist)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def normalize_seq(a, n, S):
    """Clip and normalize to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def optimize_lbfgs_lse(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10, top_k=40):
    """L-BFGS on LSE-smoothed top-k objective."""

    best_cmax = c_max

    step_scale = S / (np.sqrt(n * top_k) + 1.0)

            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_direction(grad, s_hist, y_hist)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max

        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        ss = np.dot(s_k, s_k)

            s_hist.append(s_k.copy())
            y_hist.append(y_k.copy())

        a = a_new
        grad = grad_new

    return best_a, best_cmax


def peak_peel(a_init, n, S, N, max_time, rng, top_k=8):
    """Directly reduce top peaks via subgradient steps."""
    a = normalize_seq(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    stall = 0

    while time.monotonic() - start_time < max_time:
        c, A = convolve_fft(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                stall = 0

        top_idx = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            idx = int(idx)
            lo = max(0, idx - n + 1)
            hi = min(n - 1, idx)
            if hi >= lo:
                grad[lo:hi + 1] += 2.0 * a[hi:lo - 1:-1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 0.5 * S / (n * max(gnorm, 1e-10))
        for _ in range(3):
            a_new = project_simplex(a - step * grad / gnorm, S)
            c_new, _ = convolve_fft(a_new, N)
            if c_new.max() < cur:
                a = a_new
                break
            step *= 0.5

    return best_a, best_cmax


def spectral_shaping(a_init, n, S, N, max_time, rng, top_k=12, alpha_base=0.12):
    """Spectral domain shaping to flatten convolution peaks."""
    a = normalize_seq(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c, A = convolve_fft(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        c_mean = c.mean()
        excess = max(0.0, cur - c_mean) / max(cur, 1e-10)
        alpha = alpha_base * (0.5 + 0.5 * excess)

        P = np.abs(A) ** 2
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        m_arr = np.arange(N)
        cos_eff = np.zeros(N)
        for i in range(top_k):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(top_idx[i]) / N)
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        if it % 300 == 150 and rng.random() < 0.25:
            a = project_simplex(a + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_ramp_up(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.1 + 0.9 * tt + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_ramp_down(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 - 0.9 * tt + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_exp_decay(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.exp(-2.0 * tt) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_chirp_asym(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    envelope = 0.3 + 0.7 * np.sin(np.pi * tt * 0.8) ** 2
    a = envelope * (0.7 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_init_multi_sine(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 11, 17, 29]:
        a += 0.1 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    a *= (0.5 + 0.5 * np.cos(np.pi * tt))
    return np.clip(a, 0.001, None)
