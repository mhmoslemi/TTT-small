import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient(a, c, beta):
    """Compute gradient of log(sum(exp(beta*c))) with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = beta * e / tot  # This is d/d(c) of LSE
    
    # grad[j] = sum_k w[k] * a[k-j] (with bounds)
    # This is cross-correlation: grad = ifft(FFT(w) * conj(FFT(a)))
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def adam_projected(mat, grad, lr, m, v, t, S, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam step with projection onto {x >= 0, sum = S}."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    new_a = project_nonneg_sum(mat - update, S)
    return new_a, m, v

def lse_optimize(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=200.0):

    a = a_init.copy()

    it = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S**2
    best_a = a.copy()
    lr = 0.01
    patience = 0
    max_patience = 5000
    
    while time.monotonic() - start < max_time:
        it += 1
        t += 1
        
        # Anneal beta
        progress = min(1.0, it / (2 * max_patience))
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Restart with perturbation
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_nonneg_sum(a + noise, S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = 0.01
            patience = 0
            beta = beta_start
            continue
        
        grad = lse_gradient(a, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        grad = grad / gnorm  # Normalize to unit norm

def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)

        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k

    # Enforce Hermitian symmetry
    A_real = np.zeros(n, dtype=complex)
    A_real[0] = A_complex[0]  # DC

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

    else:

"""
Minimize 2*n*max_conv/sum^2 using LogSumExp-smoothed optimization with
Adam in log-space (softmax parameterization), multi-scale warm start,
and final exact-subgradient polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_grad_in_a(a, c, beta):
    """
    Gradient of LSE(beta, c) with respect to a, where c = a*a (self-conv).
    Returns grad such that grad[j] = d/d a[j] of log(sum_k exp(beta * c[k])).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = beta * e / tot  # w[k] = d LSE / d c[k]
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 2 * cross_correlation(w, a)[j]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad_a


def softmax_param(z, S):
    """Convert log-parameters z to a non-negative sequence with sum S."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return S * e / e.sum()


def grad_z_from_grad_a(grad_a, a, S):
    """
    Chain rule: grad_z[k] = d(a @ grad_a)/dz[k]
    where a = softmax(z) * S.
    grad_z[k] = a[k] * (grad_a[k] - dot(a, grad_a) / S)
    """
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot / S)


def lse_z_objective(z, n, S, beta):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_param(z, S)
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    obj = np.log(e.sum()) + beta * cm
    
    grad_a = lse_grad_in_a(a, c, beta)
    grad_z = grad_z_from_grad_a(grad_a, a, S)
    return obj, grad_z, a, c

    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    return m, v, m_hat, v_hat

def exact_subgradient_polish(a_init, n, S, max_time, rng):
    """Fine polishing with exact top-1 subgradient and aggressive line search."""
    a = a_init.copy()
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S**2
    best_a = a.copy()
    step = 0.005
    it = 0
    c = fft_convolve_self(a)
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Top-1 subgradient
        idx = np.argmax(c)
        lo = max(0, idx - n + 1)
        hi = min(idx, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad = np.zeros(n)
        grad[i_vals] = 2.0 * a[idx - i_vals]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search: try several step sizes
        best_step = 0
        best_obj_try = cur_obj
        for s in [step, step*0.5, step*0.25, step*2.0]:
            a_try = a - s * grad
            a_try = np.maximum(a_try, 0)
            s_sum = a_try.sum()
            if s_sum < 1e-10:
                continue
            a_try = a_try * S / s_sum
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_step = s
        
        if best_step > 0:
            a_try = a - best_step * grad
            a_try = np.maximum(a_try, 0)
            a_try = a_try * S / a_try.sum()
            a = a_try
            c = fft_convolve_self(a)
            step = best_step * 1.1
        else:
            step *= 0.5
            if step < 1e-5:
                # Perturb
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0)
                a = a * S / a.sum()
                c = fft_convolve_self(a)
                step = 0.005
    
    return best_a, best_obj

"""

(softmax parameterization) with Adam optimizer, multi-scale warm starts, and
exact top-k subgradient polishing. Key innovations: (1) unconstrained
optimization in z-space via softmax, (2) LSE annealing for smooth landscape,
(3) frequency-domain flat-spectrum initializations, (4) aggressive line search
in final polish phase.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lse_grad_in_a(a, c, beta):
    """
    Gradient of log(sum_k exp(beta*c[k])) w.r.t. a where c = a*self-convolve.
    grad_a[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation, truncated to n).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = beta * e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad_a


def softmax_param(z, S):
    """Convert log-parameters z to non-negative sequence with sum S."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) * S / len(z)
    return S * e / s


def grad_z_from_grad_a(grad_a, a, S):
    """Chain rule for softmax: grad_z[k] = a[k]*(grad_a[k] - <a,grad_a>/S)."""
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    it = 0
    start = time.monotonic()

    c_cur = fft_convolve_self(a_cur)
    best_obj = 2.0 * n * c_cur.max() / S**2
    best_a = a_cur.copy()
    lr = lr_start
    consecutive_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        t += 1

        progress = min(1.0, it / 12000.0)
        beta = 1.0 + (beta_max - 1.0) * (progress ** 0.6)

        a_cur = softmax_param(z, S)
        c_cur = fft_convolve_self(a_cur)
        true_obj = 2.0 * n * c_cur.max() / S**2

        if true_obj < best_obj - 1e-10:
            best_obj = true_obj
            best_a = a_cur.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        cm = c_cur.max()
        e = np.exp(beta * (c_cur - cm))
        tot = e.sum()
        if tot < 1e-300:
            break

        fsize = 1 << (len(w) + n - 1).bit_length()

        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-15:

        m = 0.9 * m + 0.1 * grad_z

def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def exact_polish(a_init, n, S, max_time, rng):
    """
    Fine polishing with exact subgradient and multi-candidate line search.
    Uses top-1 to top-16 subgradients adaptively.
    """
    a = a_init.copy()

    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive k: start small, grow when stuck
        k = 1
        if stall > 200:
            k = 4
        if stall > 1000:
            k = 16
        if stall > 5000:
            k = 64

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Multi-candidate line search
        candidates = [step, step * 0.5, step * 0.25, step * 2.0, step * 4.0]
        best_try_obj = cur_obj
        best_try_a = None
        best_try_c = None

        for s in candidates:
            if s < 1e-8 or s > 10.0:
                continue
            a_try = project_simplex(a - s * grad, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / S**2
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_try_a = a_try
                best_try_c = c_try

        if best_try_a is not None:
            a = best_try_a
            c = best_try_c
            step = min(step * 1.15, 1.0)
        else:
            step *= 0.6
            if step < 1e-6:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
                step = 0.003
                stall = 0

        if it % 20000 == 0:
            print(f"    polish it={it}, obj={best_obj:.6f}, step={step:.6f}, k={k}")

    return best_a, best_obj


def run_adapter_subgrad(a_init, n, S, max_time, rng):

        if gnorm < 1e-15:

                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)

                momentum = np.zeros(n)

        if it % 20000 == 0:
            elapsed = time.monotonic() - start
            print(f"    adapt it={it}, obj={best_obj:.6f}, step={step:.4f}, "

        for k in range(n):
            dist = min(k, n - k) / (n / 2)
            target_power[k] = 1.0 / (1.0 + 0.5 * dist**2)
    else:
        target_power[0] = 1.5
        target_power[n // 2] = 0.5 if n % 2 == 0 else 1.0

    phase = rng.uniform(0, 2 * np.pi, n)
    phase[0] = 0.0

    A = np.zeros(n, dtype=complex)
    for k in range(n):
        mag = np.sqrt(target_power[k] * S**2 / (n * np.sum(target_power)))
        A[k] = mag * np.exp(1j * phase[k])

    # Enforce Hermitian symmetry
    A_real = np.zeros(n, dtype=complex)
    A_real[0] = complex(A[0].real, 0)
    half = n // 2
    for k in range(1, half + 1):
        if k < n:
            A_real[k] = A[k]
            A_real[n - k] = np.conj(A[k])
    if n % 2 == 0:
        A_real[half] = complex(A[half].real, 0)

    a = np.real(ifft(A_real))
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    return a


def generate_initializations(n, rng, S):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Cosine window
    a = np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01)
    inits.append(("cos", a * S / a.sum()))

    # Sinc-like
    a = np.sinc(2.5 * (tt - 0.5))
    a = np.maximum(a, 0.01)
    inits.append(("sinc", a * S / a.sum()))

    # Gaussian
    a = np.exp(-6 * (tt - 0.5)**2 / 0.25)
    inits.append(("gauss", a * S / a.sum()))

    # Flat + noise
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    inits.append(("flat_noisy", a * S / a.sum()))

    # Interleaved
    a = np.ones(n)
    a[::2] = 1.4
    a[1::2] = 0.6
    inits.append(("interleave", a * S / a.sum()))

    # Multi-lobe
    a = 1.0 + 0.35 * np.cos(2*np.pi*3*tt) + 0.15 * np.cos(2*np.pi*7*tt + 0.3)
    a = np.maximum(a, 0.01)
    inits.append(("multi_lobe", a * S / a.sum()))

    # Frequency-domain flat
    inits.append(("freq_flat", generate_freq_init(n, rng, S, "flat")))

    # Frequency-domain tapered
    inits.append(("freq_taper", generate_freq_init(n, rng, S, "tapered")))

    # Smooth random (low-pass filtered)
    a_rand = rng.standard_normal(n)
    fsize = 1 << (2*n - 1).bit_length()
    Ar = fft(a_rand, fsize)
    Ar[n//4:3*n//4] *= 0.1  # Attenuate high frequencies
    a = np.real(ifft(Ar))[:n]
    a = 1.0 + 0.4 * a / (np.std(a) + 1e-10)
    a = np.maximum(a, 0.01)
    inits.append(("smooth_rand", a * S / a.sum()))

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    1. Multi-start LSE optimization in log-space (softmax) with Adam
    2. Adaptive subgradient with momentum for intermediate refinement
    3. Exact top-k subgradient polishing with line search
    4. Final high-beta LSE pass
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None

    # Get previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size < 64 or given.sum() < 0.01:
            given = None
    except Exception:
        pass

    # Choose length
    if given is not None and 4096 <= len(given) <= 32768:
        n = len(given)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)

    print(f"Start: n={n}, S={S:.4f}, budget={budget_s:.0f}s")
    t0 = time.monotonic()

    # Build initializations
    inits = []
    if given is not None:
        if len(given) == n:
            inits.append(("prev_best", given * S / given.sum()))
        else:
            old_t = np.linspace(0, 1, len(given))
            new_t = np.linspace(0, 1, n)
            resampled = np.interp(new_t, old_t, given)
            resampled = np.maximum(resampled, 0.01)
            inits.append(("prev_resampled", resampled * S / resampled.sum()))

    gen_inits = generate_initializations(n, rng, S)
    inits.extend(gen_inits)

    n_inits = len(inits)

    # Phase 1: LSE optimization on diverse starts (50% of time)
    usable = deadline - time.monotonic()
    phase1_time = usable * 0.50
    per_init = max(5.0, phase1_time / n_inits)

    print(f"=== Phase 1: LSE multi-start ({n_inits} inits, {phase1_time:.0f}s) ===")

    for i, (name, arr) in enumerate(inits):
        remaining_p1 = deadline - time.monotonic() - (deadline - time.monotonic()) * 0.45
        t_alloc = min(per_init, max(5.0, remaining_p1))
        if t_alloc < 5:
            break

        a_opt, obj = run_lse_optimization(arr, n, S, t_alloc, rng,
                                          beta_max=400.0, lr_start=0.1, name=name)
        a_opt = np.clip(a_opt, 0.01, 1000)
        a_opt = a_opt * S / a_opt.sum()
        real_obj = compute_objective(a_opt)

        if real_obj < global_best:
            global_best = real_obj
            global_a = a_opt.copy()
            print(f"  NEW BEST [{name}]: {global_best:.6f} (elapsed {time.monotonic()-t0:.0f}s)")

    print(f"After Phase 1: best={global_best:.6f}")

    if global_a is None:
        global_a = inits[0][1].copy()
        global_best = compute_objective(global_a)

    # Phase 2: Adaptive subgradient with momentum (25% of remaining)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.25
    print(f"=== Phase 2: Adaptive subgrad ({phase2_time:.0f}s) ===")

    if phase2_time > 15:
        a_ad, obj_ad = run_adapter_subgrad(global_a, n, S, phase2_time, rng)
        a_ad = np.clip(a_ad, 0.01, 1000)
        a_ad = a_ad * S / a_ad.sum()
        real_obj = compute_objective(a_ad)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_ad.copy()
            print(f"  After adaptive subgrad: {global_best:.6f}")

    # Phase 3: Exact polish (20% of remaining)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.25
    print(f"=== Phase 3: Exact polish ({phase3_time:.0f}s) ===")

    if phase3_time > 15:
        a_pol, obj_pol = exact_polish(global_a, n, S, phase3_time, rng)
        a_pol = np.clip(a_pol, 0.01, 1000)
        a_pol = a_pol * S / a_pol.sum()
        real_obj = compute_objective(a_pol)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_pol.copy()
            print(f"  After exact polish: {global_best:.6f}")

    # Phase 4: High-beta LSE refinement (remaining)
    remaining = deadline - time.monotonic()
    phase4_time = remaining * 0.55
    print(f"=== Phase 4: High-beta LSE ({phase4_time:.0f}s) ===")

    if phase4_time > 15:
        a_hb, obj_hb = run_lse_optimization(global_a, n, S, phase4_time, rng,
                                            beta_max=800.0, lr_start=0.05, name="high_beta")
        a_hb = np.clip(a_hb, 0.01, 1000)
        a_hb = a_hb * S / a_hb.sum()
        real_obj = compute_objective(a_hb)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_hb.copy()
            print(f"  After high-beta LSE: {global_best:.6f}")

    # Final micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_mp, obj_mp = exact_polish(global_a, n, S, remaining * 0.7, rng)
        a_mp = np.clip(a_mp, 0.01, 1000)
        a_mp = a_mp * S / a_mp.sum()
        real_obj = compute_objective(a_mp)
        if real_obj < global_best:
            global_best = real_obj
            global_a = a_mp.copy()
            print(f"  After micro-polish: {global_best:.6f}")

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"Verification: {ver:.6f}")
        else:
            print("WARNING: evaluate_sequence rejected!")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print(f"DONE: length={len(global_a)}, obj={global_best:.6f}, "
          f"time={time.monotonic()-t0:.0f}s")
    return [float(x) for x in global_a]
