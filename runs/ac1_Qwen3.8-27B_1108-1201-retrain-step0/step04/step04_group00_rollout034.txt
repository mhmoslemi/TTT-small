def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)

    idx = np.arange(1, n + 1, dtype=np.float64)

    valid = tc <= w

        rho = n
    else:

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def structured_perturb(a, rng, strength=0.01):
    """Apply structured perturbations based on frequency modes."""
    n = len(a)
    S = a.sum()
    # Random frequency perturbations
    n_modes = rng.integers(2, 8)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation = strength * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * (1.0 + perturbation / a.max())
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new * (S / s)
            if compute_obj(a_new) < compute_obj(a):
                a = a_new
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-method: Mirror Descent, FISTA with cycling, IPM polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Mirror Descent (35% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: Mirror Descent at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    # Instance 1: Standard mirror descent
    a1a, _ = run_mirror_descent(global_best_a, p1_budget * 0.4, rng,
                                 beta_init=15.0, beta_rate=0.005,
                                 beta_max=5000.0, step_scale=0.8,
                                 use_cycling=False, damping=0.7)
    obj1a = compute_obj(a1a)
    print("  MD-std: obj=%.6f (t=%.0fs)" % (obj1a, time.monotonic() - t1))
    if obj1a < global_best_obj:
        global_best_obj = obj1a
        global_best_a = a1a.copy()

    # Instance 2: Cycling mirror descent
    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_mirror_descent(global_best_a, rem1 * 0.6, rng,
                                     beta_init=20.0, beta_rate=0.004,
                                     beta_max=8000.0, step_scale=1.0,
                                     use_cycling=True, cycle_period=3000, damping=0.6)
        obj1b = compute_obj(a1b)
        print("  MD-cycle: obj=%.6f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

        # Instance 3: FISTA with cycling (as comparison)
        rem1b = rem1 - (time.monotonic() - t1)
        if rem1b > 5:
            a1c, _ = run_fista_v2(global_best_a, rem1b, rng,
                                   beta_init=30.0, beta_rate=0.003,
                                   beta_max=10000.0, mode="hybrid",
                                   step_scale=0.6, use_cycling=True,
                                   cycle_period=4000)
            obj1c = compute_obj(a1c)
            print("  FISTA-cycle: obj=%.6f (t=%.0fs)" % (obj1c, time.monotonic() - t1))
            if obj1c < global_best_obj:
                global_best_obj = obj1c
                global_best_a = a1c.copy()

    # === Phase 2: FISTA LSE at original size (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: FISTA LSE at n=%d (budget %.0fs) ===" % (n0, p2_budget))
        t2 = time.monotonic()

        a2, _ = run_fista_v2(global_best_a, p2_budget * 0.5, rng,
                              beta_init=50.0, beta_rate=0.004,
                              beta_max=10000.0, mode="lse", step_scale=0.8)
        obj2 = compute_obj(a2)
        print("  FISTA-LSE: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        rem2 = p2_budget - (time.monotonic() - t2)
        if rem2 > 5:
            a2b, _ = run_fista_v2(global_best_a, rem2, rng,
                                   beta_init=100.0, beta_rate=0.002,
                                   beta_max=20000.0, mode="topk", step_scale=0.5)
            obj2b = compute_obj(a2b)
            print("  FISTA-TopK: obj=%.6f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Upsample and optimize (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and n0 <= 8192:
        p3_budget = remaining * 0.25
        n3 = n0 * 2
        print("\n=== Phase 3: Upsample to n=%d (budget %.0fs) ===" % (n3, p3_budget))
        t3 = time.monotonic()

        a3_init = upsample(global_best_a, factor=2)
        S3 = np.sqrt(2.0 * n3)
        a3_init = a3_init * (S3 / a3_init.sum())

        # Mirror descent on upscaled
        a3, _ = run_mirror_descent(a3_init, p3_budget * 0.4, rng,
                                    beta_init=5.0, beta_rate=0.005,
                                    beta_max=5000.0, step_scale=1.0,
                                    use_cycling=False, damping=0.7)
        obj3 = compute_obj(a3)
        print("  MD at 2n: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        rem3 = p3_budget - (time.monotonic() - t3)
        if rem3 > 10:
            a3b, _ = run_fista_v2(global_best_a, rem3, rng,
                                   beta_init=10.0, beta_rate=0.004,
                                   beta_max=10000.0, mode="hybrid",
                                   step_scale=0.7, use_cycling=True,
                                   cycle_period=3000)
            obj3b = compute_obj(a3b)
            print("  FISTA at 2n: obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: IPM Polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.15
        n4 = len(global_best_a)
        print("\n=== Phase 4: IPM Polish at n=%d (budget %.0fs) ===" % (n4, p4_budget))
        t4 = time.monotonic()

        a4, _ = run_ipm_polish(global_best_a, p4_budget, rng)
        obj4 = compute_obj(a4)
        print("  IPM: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final intensive polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.5
        n5 = len(global_best_a)
        print("\n=== Phase 5: Final polish at n=%d (budget %.0fs) ===" % (n5, p5_budget))
        t5 = time.monotonic()

        # Very high beta, small steps
        a5, _ = run_fista_v2(global_best_a, p5_budget * 0.5, rng,
                              beta_init=500.0, beta_rate=0.001,
                              beta_max=50000.0, mode="topk", step_scale=0.3)
        obj5 = compute_obj(a5)
        print("  High-beta: obj=%.6f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 5:
            a5b, _ = run_ipm_polish(global_best_a, rem5, rng)
            obj5b = compute_obj(a5b)
            print("  Final IPM: obj=%.6f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Try structured perturbations ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Structured perturbations ===")
        t6 = time.monotonic()
        n_try = min(10, int(remaining / 5))
        for i in range(n_try):
            if time.monotonic() > deadline:
                break
            a_pert = structured_perturb(global_best_a.copy(), rng,
                                         strength=0.005 * (i + 1))
            obj_pert = compute_obj(a_pert)
            if obj_pert < global_best_obj:
                global_best_obj = obj_pert
                global_best_a = a_pert.copy()
                print("  Perturb %d: obj=%.6f (improved!)" % (i, obj_pert))
        print("  Done (t=%.0fs)" % (time.monotonic() - t6))

    # === Phase 7: Try downsampling ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(global_best_a) > 4096:
        a_ds = downsample(global_best_a, factor=2)
        S_ds = np.sqrt(2.0 * len(a_ds))
        a_ds = a_ds * (S_ds / a_ds.sum())
        obj_ds = compute_obj(a_ds)
        print("\nPhase 7: Downsampled n=%d: obj=%.6f" % (len(a_ds), obj_ds))
        if obj_ds < global_best_obj:
            global_best_obj = obj_ds
            global_best_a = a_ds.copy()
            # Quick polish
            a_ds2, _ = run_ipm_polish(a_ds, min(remaining * 0.5, 60), rng)
            obj_ds2 = compute_obj(a_ds2)
            if obj_ds2 < global_best_obj:
                global_best_obj = obj_ds2
                global_best_a = a_ds2.copy()
                print("  After polish: obj=%.6f" % obj_ds2)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
