"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses adaptive LSE-smoothed FISTA with multi-peak gradient targeting,
multiplicative mirror descent, peak deflation, and multi-scale exploration.
Key improvements over baseline: more aggressive peak-specific updates,
higher max beta, better time allocation, and structured deflation steps.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def lse_gradient(a, c, beta, p):
    """
    Gradient of LSE approximation of max(c).
    Uses the identity: d(c[m])/d(a[i]) = 2*a[m-i] (if valid).
    grad[i] = (2/beta) * sum_j p[j+i] * a[j]
    This is a cross-correlation computable via FFT.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    # cross_correlation(p, a)[i] = sum_j p[j+i] * a[j] for valid j
    # This equals ifft(conj(fft(p, fsize)) * fft(a, fsize)) truncated
    P = fft(p, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(np.conj(P) * A))[:n]
    return (2.0 / beta) * cc


def multiplicative_repulsion(a, c, beta, lr):
    """Multiplicative update to reduce peaks."""
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w /= tot
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    a_max = max(a.max(), 1e-10)
    a_new = a * np.exp(-lr * rep / a_max)
    a_new = np.maximum(a_new, 1e-15)
    S = a.sum()
    a_new = project_simplex(a_new, S)
    return a_new


def peak_deflation(a, c, n, S, top_k=20, strength=0.1):
    """
    Targeted peak deflation: identify top peaks and reduce contributing elements.
    For each peak lag k, the contribution from index i is a[k-i] (if valid).
    Total blame for index i = sum over top peaks of a[|k-i|] (if valid).
    """
    cm = c.max()
    # Get top_k lags
    idx = np.argsort(c)[::-1][:top_k]
    blame = np.zeros(n)
    for k in idx:
        weight = c[k] - cm * 0.99  # Only blame peaks near the max
        if weight <= 0:
            continue
        # For lag k, index i contributes a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            js = np.arange(lo, hi + 1)
            blame[js] += weight * a[k - js]
    
    max_blame = blame.max()
    if max_blame < 1e-15:
        return a
    a_new = a * np.exp(-strength * blame / max_blame)
    a_new = np.maximum(a_new, 1e-15)
    a_new = project_simplex(a_new, S)
    return a_new


def run_optimization(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.003,
                     beta_max=5000.0, repulsion_freq=300, repulsion_lr=0.05,
                     deflation_freq=500, deflation_lr=0.08, noise_period=8000):
    """
    Main FISTA optimization loop with LSE smoothing, multiplicative repulsion,
    and peak deflation.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_fista = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    c = fft_convolve_self(a)
    cur_val = c.max()
    best = cur_val

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        if it % 10 == 1 or it == 1:
            c = fft_convolve_self(y)
            cur_val = c.max()
            if cur_val < best * (1 - 1e-9):
                best = cur_val
                best_a = y.copy()
                last_improve_it = it

        # Compute LSE weights and gradient
        p = compute_lse_weights(c, beta)
        grad = lse_gradient(y, c, beta, p)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        
        # Adaptive: if stuck, perturb and reset
        stuck = it - last_improve_it
        if stuck > 4000:
            step *= 0.3
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_fista = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_fista = t_new

        # Periodic multiplicative repulsion
        if repulsion_freq > 0 and it % repulsion_freq == 0:
            c = fft_convolve_self(y)
            beta_rep = min(beta, 200.0)
            y = multiplicative_repulsion(y, c, beta_rep, repulsion_lr)
            t_fista = 1.0
            cur_val = c.max()
            if cur_val < best:
                best = cur_val
                best_a = y.copy()
                last_improve_it = it

        # Periodic peak deflation
        if deflation_freq > 0 and it % deflation_freq == 0:
            c = fft_convolve_self(y)
            y = peak_deflation(y, c, n, S, top_k=30, strength=deflation_lr)
            t_fista = 1.0
            cur_val = c.max()
            if cur_val < best:
                best = cur_val
                best_a = y.copy()
                last_improve_it = it

        # Update c periodically
        if it % 10 == 0:
            c = fft_convolve_self(y)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length using cubic-like smoothing."""
    n = len(a)
    if new_n == n:
        return a.copy()
    # Use FFT-based resampling for better quality
    fsize = max(2 * n, 2 * new_n) * 2
    fsize = 1 << fsize.bit_length()
    A = fft(a, fsize)
    t_old = np.arange(n) / n
    t_new = np.arange(new_n) / new_n
    # Reconstruct at new points via inverse DFT sampling
    phase = 2j * np.pi * np.outer(t_new.astype(np.float64), np.arange(fsize))
    # Actually simpler: just use np.interp with the values
    return np.interp(np.linspace(0, 1, new_n, endpoint=False),
                     np.linspace(0, 1, n, endpoint=False), a)


def direct_peak_iteration(a, n, S, n_iters=1000, rng=None):
    """
    Direct iterative peak reduction:
    1. Find the maximum lag k*.
    2. Compute blame: for index i, blame[i] = a[k*-i] (if valid).
    3. Multiply a[i] by exp(-eta * blame[i] / max_blame).
    4. Renormalize.
    Repeat.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a = normalize_sum(a, n)
    best = np.inf
    best_a = a.copy()
    S = np.sqrt(2.0 * n)
    
    for it in range(n_iters):
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top few peaks
        top_k = min(10, 2 * n - 1)
        peak_lags = np.argsort(c)[::-1][:top_k]
        cm = c.max()
        
        # Compute combined blame
        blame = np.zeros(n)
        for k in peak_lags:
            w = max(0, c[k] - cm * 0.95)  # Weight by how close to peak
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if hi >= lo:
                js = np.arange(lo, hi + 1)
                blame[js] += w * a[k - js]
        
        max_blame = blame.max()
        if max_blame < 1e-15:
            break
        
        # Adaptive learning rate
        eta = 0.05 + 0.05 * (1 - it / n_iters)
        a_new = a * np.exp(-eta * blame / max_blame)
        a_new = np.maximum(a_new, 1e-15)
        a = project_simplex(a_new, S)
        
        # Occasional noise to escape
        if it % 200 == 0 and it > 0:
            noise_scale = 0.01 * S / np.sqrt(n) * (1 - it / n_iters)
            if noise_scale > 1e-8:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses multi-phase optimization: FISTA warmup, main loop with peak deflation,
    direct peak iteration, multi-scale exploration, and final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192
    
    n = main_len
    S = np.sqrt(2.0 * n)
    print("Main length: %d, budget: %.0fs" % (main_len, budget_s))
    
    # Initialize
    if given is not None:
        a_curr = normalize_sum(given, n)
    else:
        # Generate a good initial sequence
        tt = np.linspace(0, 1, n, endpoint=False)
        a_curr = 1.0 + 0.3 * np.cos(2 * np.pi * tt) + 0.2 * np.cos(4 * np.pi * tt) + 0.1 * np.cos(6 * np.pi * tt)
        a_curr = np.maximum(a_curr, 0.01)
        a_curr = normalize_sum(a_curr, n)
    
    # Evaluate initial
    c0 = fft_convolve_self(a_curr)
    obj0 = 2.0 * n * c0.max() / (a_curr.sum() ** 2)
    global_best = obj0
    global_a = a_curr.copy()
    print("Initial objective: %.6f" % obj0)
    
    # Phase 1: FISTA warmup (15% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t1 = min(remaining * 0.15, 300.0)
        print("=== Phase 1: FISTA warmup (%.0fs) ===" % t1)
        a1, s1 = run_optimization(a_curr, n, S, t1, rng,
                                   beta_init=10.0, beta_rate=0.005, beta_max=200.0,
                                   repulsion_freq=200, repulsion_lr=0.06,
                                   deflation_freq=400, deflation_lr=0.06)
        c1 = fft_convolve_self(a1)
        obj1 = 2.0 * n * c1.max() / (a1.sum() ** 2)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
            print("  Phase 1 best: %.6f" % global_best)
        else:
            print("  Phase 1: no improvement (%.6f vs %.6f)" % (obj1, global_best))
    
    # Phase 2: Main optimization with higher beta (45% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t2 = remaining * 0.45
        print("=== Phase 2: Main FISTA (%.0fs, beta up to 3000) ===" % t2)
        a2, s2 = run_optimization(global_a, n, S, t2, rng,
                                   beta_init=100.0, beta_rate=0.002, beta_max=3000.0,
                                   repulsion_freq=300, repulsion_lr=0.04,
                                   deflation_freq=500, deflation_lr=0.05,
                                   noise_period=10000)
        c2 = fft_convolve_self(a2)
        obj2 = 2.0 * n * c2.max() / (a2.sum() ** 2)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            print("  Phase 2 best: %.6f" % global_best)
        else:
            print("  Phase 2: no improvement (%.6f vs %.6f)" % (obj2, global_best))
    
    # Phase 3: Direct peak iteration (15% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t3 = remaining * 0.15
        n_direct = int(t3 * 2000)  # ~2000 iterations per second estimate
        n_direct = max(1000, min(n_direct, 50000))
        print("=== Phase 3: Direct peak iteration (%d iters) ===" % n_direct)
        
        # Run multiple direct iterations from slightly perturbed starts
        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            if trial == 0:
                start_a = global_a.copy()
            else:
                start_a = global_a + rng.normal(0, 0.002 * S / np.sqrt(n), n)
                start_a = np.maximum(start_a, 0)
                start_a = project_simplex(start_a, S)
            
            a3, s3 = direct_peak_iteration(start_a, n, S, n_iters=n_direct // 3, rng=rng)
            c3 = fft_convolve_self(a3)
            obj3 = 2.0 * n * c3.max() / (a3.sum() ** 2)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()
                print("  Phase 3 trial %d: %.6f" % (trial, global_best))
            else:
                print("  Phase 3 trial %d: %.6f (no improvement)" % (trial, obj3))
    
    # Phase 4: Multi-scale exploration (15% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t4 = remaining * 0.15
        print("=== Phase 4: Multi-scale (%.0fs) ===" % t4)
        
        # Try 2x length
        alt_n = n * 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            alt_init = interpolate_sequence(global_a, alt_n)
            alt_init = normalize_sum(alt_init, alt_n)
            
            a4a, s4a = run_optimization(alt_init, alt_n, S_alt, t4 * 0.5, rng,
                                         beta_init=50.0, beta_rate=0.003, beta_max=1000.0,
                                         repulsion_freq=200, repulsion_lr=0.05,
                                         deflation_freq=400, deflation_lr=0.05)
            c4a = fft_convolve_self(a4a)
            obj4a = 2.0 * alt_n * c4a.max() / (a4a.sum() ** 2)
            if obj4a < global_best:
                global_best = obj4a
                global_a = a4a.copy()
                print("  Phase 4a (n=%d): %.6f" % (alt_n, global_best))
            else:
                print("  Phase 4a (n=%d): %.6f (no improvement vs %.6f)" % (alt_n, obj4a, global_best))
        
        # Try n/2 length
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alt_n2 = n // 2
            if alt_n2 >= 1024:
                S_alt2 = np.sqrt(2.0 * alt_n2)
                alt_init2 = interpolate_sequence(global_a, alt_n2)
                alt_init2 = normalize_sum(alt_init2, alt_n2)
                
                a4b, s4b = run_optimization(alt_init2, alt_n2, S_alt2, remaining * 0.4, rng,
                                             beta_init=50.0, beta_rate=0.004, beta_max=2000.0,
                                             repulsion_freq=200, repulsion_lr=0.05,
                                             deflation_freq=300, deflation_lr=0.06)
                c4b = fft_convolve_self(a4b)
                obj4b = 2.0 * alt_n2 * c4b.max() / (a4b.sum() ** 2)
                if obj4b < global_best:
                    global_best = obj4b
                    global_a = a4b.copy()
                    print("  Phase 4b (n=%d): %.6f" % (alt_n2, global_best))
                else:
                    print("  Phase 4b (n=%d): %.6f (no improvement)" % (alt_n2, obj4b))
    
    # Phase 5: Final intensive polish (remaining time minus 20s buffer)
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        t5 = remaining * 0.8
        print("=== Phase 5: Final polish (%.0fs, n=%d) ===" % (t5, n_p))
        
        # Very high beta FISTA
        a5, s5 = run_optimization(global_a, n_p, S_p, t5 * 0.5, rng,
                                   beta_init=500.0, beta_rate=0.001, beta_max=10000.0,
                                   repulsion_freq=500, repulsion_lr=0.02,
                                   deflation_freq=1000, deflation_lr=0.03)
        c5 = fft_convolve_self(a5)
        obj5 = 2.0 * n_p * c5.max() / (a5.sum() ** 2)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("  Phase 5a: %.6f" % global_best)
        
        # Another direct peak pass
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_dp = max(500, int(remaining * 1000))
            a5b, s5b = direct_peak_iteration(global_a, n_p, S_p, n_iters=n_dp, rng=rng)
            c5b = fft_convolve_self(a5b)
            obj5b = 2.0 * n_p * c5b.max() / (a5b.sum() ** 2)
            if obj5b < global_best:
                global_best = obj5b
                global_a = a5b.copy()
                print("  Phase 5b: %.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("Returning sequence of length %d, best score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
