    # (a * delta)_j = sum_i a_i * delta_{j-i} (for full convolution)

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = scaled * (500.0 / scaled.max())
        beta_eff = beta * (500.0 / (beta * (c - cm).max()))
    else:
        beta_eff = beta
    terms = np.exp(scaled)
    return cm + np.log(terms.sum()) / beta_eff


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scale_factor = 500.0 / scaled.max()
        scaled = scaled * scale_factor
        beta_eff = beta * scale_factor
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    min_val = peak_vals[-1]
    range_val = max_val - min_val
    if range_val < 1e-15:
        weights = np.ones(k) / k
    else:
        normalized = (peak_vals - min_val) / range_val
        weights = normalized ** sharpness
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[sorted_idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def direct_peak_equalization(a, c, k=8, mu=0.01):
    """
    Solve for perturbation delta that equalizes top-k peaks.
    Uses the linearized system: 2*(a*delta)_lag_i = target - c[lag_i]
    with a soft sum=0 constraint.
    Returns the perturbation delta (before projection).
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    target = c[idx].mean()
    rhs = target - c[idx]
    
    # Build system matrix: for each active lag, (a * delta)_lag = sum_i a[i]*delta[lag-i]
    # We solve: A_sys @ delta = rhs, where A_sys has k+n rows (k data + soft sum)
    # For efficiency, use the FFT-based approach:
    # delta = ifft(FFT(a) * conj(FFT(w))) where w is the weight vector
    # But we want a specific target, so let's use the adjoint approach.
    
    # The gradient of (c - target)^2/2 at the active peaks gives us the direction.
    # Let's just use a weighted subgradient approach with very sharp weights.
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Scale the gradient to aim for the right magnitude
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return np.zeros(n)
    
    # Target reduction per peak
    typical_rhs = np.abs(rhs).mean()
    if typical_rhs < 1e-15:
        return np.zeros(n)
    
    # Scale so that the expected reduction at each peak is about 'mu' fraction of the excess
    scale = mu * typical_rhs / (grad_norm ** 2)
    scale = min(scale, 0.1)  # Cap to avoid huge steps
    
    return -grad * scale


def spectral_shape(a, damping=0.1, budget=0.9):
    """
    Modify the power spectrum to reduce concentration.
    Dampen high-power frequency components slightly.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_power = power.mean()
    
    # Create a damped version: reduce peaks in power spectrum
    A_damped = A.copy()
    for i in range(fsize):
        if power[i] > budget * mean_power * 3:
            damp = np.sqrt(budget * mean_power * 3 / power[i])
            A_damped[i] *= damp
    
    a_new = np.real(ifft(A_damped))[:n]
    return a_new


def run_fista_backtrack(a_init, S, max_time, rng, beta_init=50.0,
                        beta_max=50000.0, beta_growth=0.002,
                        use_direct=False, direct_k=8):
    """
    FISTA with backtracking line search and adaptive LSE beta.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    step = 0.02
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it
        
        # Compute gradient
        if use_direct and it % 5 == 0:
            grad = direct_peak_equalization(y, c, k=direct_k, mu=0.05)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        alpha = step
        base_obj = lse_smooth_max(c, beta)
        found_decrease = False
        for backtrack in range(10):
            x_candidate = project_simplex(y + alpha * (-grad / max(gnorm, 1e-15)), S)
            c_new = fft_convolve_self(x_candidate)
            new_obj = lse_smooth_max(c_new, beta)
            if new_obj < base_obj - 1e-12:
                found_decrease = True
                break
            alpha *= 0.5
        
        if not found_decrease:
            # Just take a small step
            x_new = project_simplex(y - 0.001 * grad / max(gnorm, 1e-15), S)
            step = max(step * 0.5, 0.0001)
        else:
            # Update step size (increase if we didn't use all backtracks)
            if backtrack < 3:
                step = min(step * 1.1, 0.5)
            else:
                step = max(step * 0.7, 0.0001)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart with noise
        if it % 2000 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                step = 0.02
        
        # If stuck, try a different approach
        if it - last_improve_it > 3000:
            t = 1.0
            step = 0.01
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_ultra_polish(a_init, S, max_time, rng):
    """
    Ultra-precise polishing with very high beta and tiny steps.
    Focuses exclusively on the single maximum peak.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Use very high beta for near-exact max
        beta = 5000.0 + 100.0 * it
        
        # Gradient from the top few peaks with extreme sharpness
        k = min(4, n)
        sorted_idx = np.argsort(c)[::-1][:k]
        peak_vals = c[sorted_idx]
        max_val = peak_vals[0]
        
        # Only use peaks within 0.1% of max
        threshold = max_val * (1 - 0.001)
        active = [i for i in sorted_idx if c[i] >= threshold]
        if not active:
            active = [sorted_idx[0]]
        
        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / len(active)
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Very small adaptive step
        progress = (time.monotonic() - start) / max_time
        step = 0.0005 * (1.0 - progress * 0.5)
        
        x_new = project_simplex(a - step * grad / gnorm, S)
        c_new = fft_convolve_self(x_new)
        
        # Only accept if it improves
        if c_new.max() < cur_max:
            a = x_new
        
        # Very occasional perturbation to escape local minima
        if it % 5000 == 0 and rng.random() < 0.3:
            noise_scale = 0.0001 * S / np.sqrt(n)
            a_trial = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best:
                a = a_trial
                best = c_trial.max()
                best_a = a.copy()
    
    return best_a, best


def run_peak_shaving_direct(a_init, S, max_time, rng):
    """
    Iterative peak shaving: repeatedly target the single worst peak.
    Uses the exact subgradient (shifted copy of a).
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    base_step = 0.01
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find the argmax lag
        lag = int(np.argmax(c))
        
        # Subgradient: 2 * (a convolved with delta at lag)
        # (a * delta_lag)_j = a[j - lag] (padded)
        # So the subgradient at input index i is 2*a[lag - i] (within bounds)
        grad = np.zeros(n)
        for i in range(n):
            j = lag - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step: decrease over time
        progress = min(1.0, it / 100000.0)
        step = base_step * (1.0 - 0.5 * progress)
        
        x_new = project_simplex(a - step * grad / gnorm, S)
        c_new = fft_convolve_self(x_new)
        
        if c_new.max() < cur_max:
            a = x_new
        else:
            # Try smaller step
            x_new2 = project_simplex(a - step * 0.1 * grad / gnorm, S)
            c_new2 = fft_convolve_self(x_new2)
            if c_new2.max() < cur_max:
                a = x_new2
                base_step = max(base_step * 0.8, 0.0001)
            else:
                base_step = max(base_step * 0.7, 0.00005)
                # Add tiny noise to escape
                noise_scale = 0.0005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Periodic restart
        if it % 10000 == 0:
            base_step = 0.005
            noise_scale = 0.001 * S / np.sqrt(n)
            a_trial = project_simplex(a + rng.normal(0, noise_scale, n), S)
            if fft_convolve_self(a_trial).max() < best:
                a = a_trial
    
    return best_a, best


def generate_inits(n, rng, base=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if base is not None and len(base) == n:
        inits.append(base.copy())
    
    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.09))
    
    # Multi-sinusoidal
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * 2 * tt + 0.3) - 0.3 * np.cos(2 * np.pi * 5 * tt), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.05)
    
    # Random smooth
    freqs = rng.integers(1, 10, 5)
    amps = rng.uniform(0.1, 0.5, 5)
    phases = rng.uniform(0, 2 * np.pi, 5)
    sig = np.ones(n)
    for f, a_m, ph in zip(freqs, amps, phases):
        sig += a_m * np.sin(2 * np.pi * f * tt + ph)
    inits.append(np.maximum(sig, 0.01))
    
    return inits


def upscale_sequence(a_small, n_large):
    """Upscale a sequence by linear interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: explore with diverse inits, then focus on best with
    ultra-precise polishing using multiple complementary methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    # Primary length
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)
    
    # ========== PHASE 1: Quick exploration with diverse inits ==========
    print("\n=== Phase 1: Exploration ===")
    t1_start = time.monotonic()
    t1_budget = min(300.0, (deadline - t1_start) * 0.2)
    
    inits = []
    if given is not None:
        if len(given) == n_main:
            inits.append(normalize_to_sum(given, S_main))
        elif len(given) > n_main:
            ratio = len(given) // n_main
            small = given[:n_main * ratio].reshape(n_main, ratio).mean(axis=1)
            inits.append(normalize_to_sum(small, S_main))
        else:
            inits.append(normalize_to_sum(upscale_sequence(given, n_main), S_main))
    
    gen_inits = generate_inits(n_main, rng)
    inits.extend(gen_inits[:5])
    
    best_a_phase1 = None
    best_val_phase1 = np.inf
    
    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic() - (deadline - t1_start) * 0.8
        t_alloc = min(t1_budget / max(1, len(inits) - i), remaining)
        if t_alloc < 10:
            break
        
        use_direct = (i > 0)  # Only use direct method for non-given inits
        a_opt, val = run_fista_backtrack(init, S_main, t_alloc, rng,
                                          beta_init=30.0,
                                          beta_max=5000.0,
                                          beta_growth=0.003,
                                          use_direct=use_direct,
                                          direct_k=8)
        obj = objective_value(a_opt)
        print("  Init %d: obj=%.6f, max_b=%.6f (t=%.0fs)" % (i, obj, val, time.monotonic() - t1_start))
        
        if val < best_val_phase1:
            best_val_phase1 = val
            best_a_phase1 = a_opt.copy()
    
    if best_a_phase1 is None:
        best_a_phase1 = np.ones(n_main) * S_main / n_main
        best_val_phase1 = fft_convolve_self(best_a_phase1).max()
    
    obj_check = objective_value(best_a_phase1)
    if obj_check < global_best:
        global_best = obj_check
        global_a = best_a_phase1.copy()
    print("  Phase 1 best: obj=%.6f" % global_best)
    
    # ========== PHASE 2: Focused FISTA polishing ==========
    print("\n=== Phase 2: FISTA Polishing ===")
    t2_start = time.monotonic()
    t2_budget = min(600.0, (deadline - t2_start) * 0.35)
    
    a_polished, val2 = run_fista_backtrack(best_a_phase1, S_main, t2_budget, rng,
                                            beta_init=100.0,
                                            beta_max=20000.0,
                                            beta_growth=0.002,
                                            use_direct=False)
    obj2 = objective_value(a_polished)
    print("  After FISTA polish: obj=%.6f, max_b=%.6f" % (obj2, val2))
    
    if val2 < best_val_phase1:
        best_val_phase1 = val2
        best_a_phase1 = a_polished.copy()
    if obj2 < global_best:
        global_best = obj2
        global_a = a_polished.copy()
    
    # ========== PHASE 3: Direct peak shaving ==========
    print("\n=== Phase 3: Direct Peak Shaving ===")
    t3_start = time.monotonic()
    t3_budget = min(400.0, (deadline - t3_start) * 0.3)
    
    a_shaved, val3 = run_peak_shaving_direct(best_a_phase1, S_main, t3_budget, rng)
    obj3 = objective_value(a_shaved)
    print("  After peak shaving: obj=%.6f, max_b=%.6f" % (obj3, val3))
    
    if val3 < best_val_phase1:
        best_val_phase1 = val3
        best_a_phase1 = a_shaved.copy()
    if obj3 < global_best:
        global_best = obj3
        global_a = a_shaved.copy()
    
    # ========== PHASE 4: Ultra-precise final polish ==========
    print("\n=== Phase 4: Ultra Polish ===")
    t4_start = time.monotonic()
    t4_remaining = deadline - t4_start
    
    if t4_remaining > 30:
        a_ultra, val4 = run_ultra_polish(global_a, S_main, t4_remaining, rng)
        obj4 = objective_value(a_ultra)
        print("  After ultra polish: obj=%.6f, max_b=%.6f" % (obj4, val4))
        
        if obj4 < global_best:
            global_best = obj4
            global_a = a_ultra.copy()
    
    # ========== PHASE 5: Try a slightly different length ==========
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 5: Try n=16384 ===")
        n_alt = 16384
        S_alt = np.sqrt(2.0 * n_alt)
        a_up = upscale_sequence(global_a, n_alt)
        a_up = normalize_to_sum(a_up, S_alt)
        
        t_alt = min(remaining - 30, 150.0)
        a_alt, val_alt = run_fista_backtrack(a_up, S_alt, t_alt, rng,
                                              beta_init=50.0,
                                              beta_max=10000.0,
                                              beta_growth=0.003)
        obj_alt = objective_value(a_alt)
        print("  n=16384: obj=%.6f" % obj_alt)
        
        if obj_alt < global_best:
            global_best = obj_alt
            global_a = a_alt.copy()
    
    # ========== FINAL: Verify and return ==========
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
    
    print("Returning sequence of length %d, internal obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
