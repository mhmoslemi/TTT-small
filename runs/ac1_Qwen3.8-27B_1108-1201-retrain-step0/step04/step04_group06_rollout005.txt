def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(a, top_indices, weights, fsize):
    """Compute weighted subgradient of max conv via FFT cross-correlation."""
    n = len(a)
    w_full = np.zeros(fsize)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    W = fft(w_full)
    A = fft(a, fsize)
    # Cross-correlation: ifft(conj(A) * W) gives sum_k w[k] * a[k-m]
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad

def softmax_weights(values, temperature):
    """Compute softmax weights with given temperature."""
    if temperature < 1e-10:
        # Hard selection
        max_val = values.max()
        mask = values == max_val
        if not mask.any():
            mask[0] = True
        w = np.zeros(len(values))
        w[mask] = 1.0 / mask.sum()
        return w
    shifted = (values - values.max()) / temperature
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()

    Modifies the power spectrum to reduce the weighted peak of the autocorrelation.

    m = np.arange(N)

    if sh_max > 1e-10:

def safe_normalize(a, n):
