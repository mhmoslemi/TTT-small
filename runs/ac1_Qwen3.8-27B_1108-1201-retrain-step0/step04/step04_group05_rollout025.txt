        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = a_new * (S / max(a_new.sum(), 1e-30))
        a = a_new

        if it % 5 == 0 or it < 10:
            c_a = compute_conv(a, N)
            actual_cmax = c_a.max()
            if actual_cmax < best_cmax:
                best_cmax = actual_cmax
                best_a = a.copy()

        # Adaptive eta: decrease if no improvement
        if it % 5000 == 0:
            eta *= 0.95

    return best_a, best_cmax
