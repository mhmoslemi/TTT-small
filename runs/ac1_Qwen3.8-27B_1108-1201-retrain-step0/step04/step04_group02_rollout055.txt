    n = len(x)
    # Softmax
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    a = S * exp_x / exp_x.sum()
    
    # Self-convolution
    c = fft_convolve_self(a)  # length 2n-1
    
    # LSE smoothing
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    lse_sum = exp_bc.sum()
    F = (bc_max + np.log(lse_sum)) / beta
    
    # Weights
    w = exp_bc / lse_sum  # length 2n-1
    
    # Gradient w.r.t. a: g = 2 * xcorr(a, w)
    N = 1 << (3 * n).bit_length()  # >= 3n-2
    FA = fft(a, N)
    FW = fft(w, N)
    g = 2.0 * np.real(ifft(FW * np.conj(FA)))[:n]
    
    # C = 2 * sum(w * c) (Euler's theorem)
    C = 2.0 * np.dot(w, c)
    
    # Gradient w.r.t. x: dF/dx = a * (g - C/S)
    grad_x = a * (g - C / S)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using LSE-smoothed
objective with L-BFGS optimization in softmax parameterization, combined with
beta annealing, multi-start, and subgradient polishing.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_from_log(x, S):
    """Convert log-space variables to simplex with sum S."""
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    total = exp_x.sum()
    if total < 1e-300:
        return np.ones_like(x) * S / len(x)
    return S * exp_x / total


def lse_objective_AND_gradient(x, S, beta):
    """
    LSE-smoothed objective and gradient w.r.t. x (log-space variables).
    F = (1/beta)*log(sum_k exp(beta*c_k)), c = a*a, a = S*softmax(x)
    """
    n = len(x)
    a = softmax_from_log(x, S)
    c = fft_convolve_self(a)
    
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    lse_sum = exp_bc.sum()
    F = (bc_max + np.log(lse_sum)) / beta
    
    w = exp_bc / lse_sum
    
    # Gradient w.r.t. a: g = 2 * xcorr(a, w)
    # xcorr[m] = sum_j a[j] * w[m+j]
    N = 1 << (3 * n).bit_length()
    FA = fft(a, N)
    FW = fft(w, N)
    g = 2.0 * np.real(ifft(FW * np.conj(FA)))[:n]
    
    # C = sum_j g[j] * a[j] = 2 * sum_k w[k]*c[k] (Euler for degree-2)
    C = 2.0 * np.dot(w, c)
    
    # dF/dx = a * (g - C/S)
    grad_x = a * (g - C / S)
    
    return F, grad_x


def run_lse_optimization(a_init, n, beta_schedule, max_time_per_beta, rng):
    """
    Run LSE-smoothed L-BFGS optimization with beta annealing.
    """
    S = np.sqrt(2.0 * n)
    a_init = np.clip(a_init, 1e-10, 1000.0)
    a_init = a_init * (S / a_init.sum())
    x = np.log(a_init)
    
    best_obj = np.inf
    best_a = a_init.copy()
    start_total = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - start_total > max_time_per_beta * len(beta_schedule):
            break  # Over total budget
        
        # Define objective and gradient for this beta
        def objfun(x_in):
            val, _ = lse_objective_AND_gradient(x_in, S, beta)
            return val
        
        def gradfun(x_in):
            _, g = lse_objective_AND_gradient(x_in, S, beta)
            return g
        
        # Run L-BFGS-B
        t_start = time.monotonic()
        max_iter_sec = max_time_per_beta
        result = minimize(
            objfun, x, jac=gradfun, method='L-BFGS-B',
            options={'maxiter': 5000, 'maxfun': 5000, 'maxls': 50,
                     'ftol': 1e-12, 'gtol': 1e-8, 'max_feasible': 1000}
        )
        
        x = result.x
        a = softmax_from_log(x, S)
        
        # Compute actual (unsmoothed) objective
        c = fft_convolve_self(a)
        actual_obj = 2.0 * n * c.max() / S**2
        
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a.copy()
        
        print("    beta=%d: LSE=%.8f, actual=%.8f, time=%.1fs" % (
            beta, result.fun, actual_obj, time.monotonic() - t_start))
    
    return best_a, best_obj


def subgradient_polish(a, n, max_time, rng):
    """Polish with exact subgradient and tiny steps."""
    S = np.sqrt(2.0 * n)
    a = a.copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S**2
    best_a = a.copy()
    step = 0.01
    stall = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = 2.0 * n * c.max() / S**2
        
        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Exact subgradient at the max
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k_star - i]
        grad /= (hi - lo + 1)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Projected step
        a_try = a - step * grad
        # Project onto simplex {x >= 0, sum = S}
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try * (S / s)
        
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2
        
        if obj_try < cur:
            a = a_try
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.8
            if stall > 100:
                # Perturb
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0)
                a = a * (S / a.sum())
                step = 0.01
                stall = 0
    
    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using LSE-smoothed
objective with L-BFGS in softmax parameterization, beta annealing, multi-start,
and subgradient polishing. This differs from the previous approach by using
a smooth optimization method (L-BFGS) on a log-sum-exp smoothed objective
with softmax reparameterization, enabling much faster convergence than
pure subgradient methods.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_from_log(x, S):
    """Convert log-space variables to simplex with sum S."""
    x_shift = x - x.max()
    exp_x = np.exp(x_shift)
    total = exp_x.sum()
    if total < 1e-300:
        return np.ones_like(x) * S / len(x)
    return S * exp_x / total


def lse_objective_and_gradient(x, S, beta):
    """
    LSE-smoothed objective and gradient w.r.t. x (log-space variables).
    F = (1/beta)*log(sum_k exp(beta*c_k)), c = a*a, a = S*softmax(x)
    Returns (F, grad_x).
    """
    n = len(x)
    a = softmax_from_log(x, S)
    c = fft_convolve_self(a)

    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    lse_sum = exp_bc.sum()
    F = (bc_max + np.log(lse_sum)) / beta

    w = exp_bc / lse_sum

    # Gradient w.r.t. a: g = 2 * xcorr(a, w)
    # xcorr[m] = sum_j a[j] * w[m+j]
    N = 1 << (3 * n).bit_length()
    FA = fft(a, N)
    FW = fft(w, N)
    g = 2.0 * np.real(ifft(FW * np.conj(FA)))[:n]

    # Euler: sum_j g[j]*a[j] = 2*sum_k w[k]*c[k]
    C = 2.0 * np.dot(w, c)

    # Chain rule through softmax: dF/dx = a * (g - C/S)
    grad_x = a * (g - C / S)

    return F, grad_x


def run_lse_optimization(a_init, n, beta_schedule, total_time, rng):
    """
    Run LSE-smoothed L-BFGS optimization with beta annealing.
    Returns (best_a, best_score).
    """
    S = np.sqrt(2.0 * n)
    a_init = np.clip(a_init.astype(np.float64), 1e-10, 1000.0).flatten()
    if a_init.sum() < 1e-15:
        a_init = np.ones(n)
    a_init = a_init * (S / a_init.sum())
    x = np.log(a_init)

    best_obj = 2.0 * n * fft_convolve_self(a_init).max() / S**2
    best_a = a_init.copy()
    start_total = time.monotonic()
    n_betas = len(beta_schedule)

    for bi, beta in enumerate(beta_schedule):
        elapsed = time.monotonic() - start_total
        remaining = total_time - elapsed
        if remaining < 5:
            break

        def objfun(x_in):
            val, _ = lse_objective_and_gradient(x_in, S, beta)
            return val

        def gradfun(x_in):
            _, g = lse_objective_and_gradient(x_in, S, beta)
            return g

        # Estimate max iterations based on remaining time
        max_iter = max(50, int(remaining * 50 / n_betas))

        result = minimize(
            objfun, x, jac=gradfun, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter, 'maxls': 30,
                     'ftol': 1e-14, 'gtol': 1e-10, 'max_feasible': 500}
        )

        x = result.x
        a = softmax_from_log(x, S)
        c = fft_convolve_self(a)
        actual_obj = 2.0 * n * c.max() / S**2

        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a.copy()

        print("    beta=%g: LSE=%.8f, actual=%.8f, time=%.1fs" % (
            beta, result.fun, actual_obj, time.monotonic() - start_total))

    return best_a, best_obj


def subgradient_polish(a, n, max_time, rng):
    """Polish with exact subgradient and adaptive tiny steps."""
    S = np.sqrt(2.0 * n)
    a = a.copy() * (S / a.sum())
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / S**2
    best_a = a.copy()
    step = 0.005
    stall = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = 2.0 * n * c.max() / S**2

        if cur < best_obj - 1e-12:
            best_obj = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Top-2 subgradient
        top2 = np.argsort(c)[::-1][:2]
        grad = np.zeros(n)
        for k_star in top2:
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k_star - i]
        grad /= len(top2)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_try = a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            a_try = a.copy()
        else:
            a_try = a_try * (S / s)

        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S**2

        if obj_try < cur:
            a = a_try
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.85
            if stall > 80:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0)
                a = a * (S / a.sum())
                step = 0.005
                stall = 0

    return best_a, best_obj


def generate_initializations(n, rng, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None:
        g = given.flatten()
        if len(g) == n:
            inits.append(("given", np.clip(g, 1e-6, 1000.0)))
        elif len(g) >= 64:
            # Resample to length n
            g_resampled = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, len(g)),
                g
            )
            inits.append(("given_res", np.clip(g_resampled, 1e-6, 1000.0)))

    inits.append(("cos1", np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * tt), 0.01)))
    inits.append(("cos_edge", np.maximum(1.0 + 0.8 * np.cos(np.pi * tt), 0.01)))
    inits.append(("gauss", np.exp(-3.0 * (tt - 0.5) ** 2) + 0.01))
    inits.append(("sin2", np.sin(np.pi * tt) ** 2 + 0.01))
    inits.append(("sin4", np.sin(np.pi * tt) ** 4 + 0.01))
    inits.append(("multi1", np.maximum(
        1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt + 0.5), 0.01)))
    inits.append(("multi2", np.maximum(
        1.0 + 0.2 * np.sin(2 * np.pi * 2 * tt + 1.0) + 0.3 * np.cos(2 * np.pi * 5 * tt + 0.3), 0.01)))
    inits.append(("multi3", np.maximum(
        1.0 + 0.15 * np.sin(2 * np.pi * 4 * tt) + 0.15 * np.sin(2 * np.pi * 11 * tt + 0.7)
        + 0.1 * np.cos(2 * np.pi * 6 * tt + 1.2), 0.01)))
    inits.append(("rand", rng.random(n) * 0.8 + 0.2))
    inits.append(("bimodal", np.exp(-20 * (tt - 0.25) ** 2) + np.exp(-20 * (tt - 0.75) ** 2) + 0.01))
    inits.append(("plateau", np.maximum(1.0 - 0.7 * np.abs(tt - 0.5), 0.01)))

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LSE-smoothed L-BFGS optimization with beta annealing and multi-start.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs" % budget_s)
    t0 = time.monotonic()

    # Main optimization at n=8192
    n = 8192
    S = np.sqrt(2.0 * n)

    # Generate initializations
    inits = generate_initializations(n, rng, given)

    usable = deadline - time.monotonic()
    # Phase 1: Multi-start LSE (65% of time)
    # Phase 2: Focused LSE polish (20% of time)
    # Phase 3: Subgradient polish + length exploration (15% of time)
    phase1_time = usable * 0.65
    phase2_time = usable * 0.20
    phase3_time = usable * 0.15

    print("=== Phase 1: Multi-start LSE (n=%d, %.0fs, %d inits) ===" % (
        n, phase1_time, len(inits)))

    beta_schedule_coarse = [5, 15, 40, 100, 200]
    time_per_init = phase1_time / len(inits)
    time_per_beta = time_per_init / len(beta_schedule_coarse)

    best_from_phase1 = None
    best_score_phase1 = np.inf

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > t0 + usable * 0.65:
            print("  Time running out in Phase 1")
            break

        print("  Init %d/%d: %s" % (i + 1, len(inits), name))
        t_init = time.monotonic()

        a_opt, score = run_lse_optimization(
            arr, n, beta_schedule_coarse, time_per_init * 1.1, rng
        )

        a_norm = a_opt * (S / a_opt.sum())
        c = fft_convolve_self(a_norm)
        real_score = 2.0 * n * c.max() / S**2

        if real_score < best_score_phase1:
            best_score_phase1 = real_score
            best_from_phase1 = a_norm.copy()

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] NEW BEST: %.8f (elapsed %.0fs)" % (
                name, global_best, time.monotonic() - t_init))
        else:
            print("  [%s] score=%.8f (elapsed %.0fs)" % (
                name, real_score, time.monotonic() - t_init))

    print("=== After Phase 1: best=%.8f ===" % global_best)

    # Phase 2: Focused LSE with higher beta on the best
    remaining = deadline - time.monotonic()
    phase2_time = min(phase2_time, remaining)
    print("=== Phase 2: Focused LSE polish (%.0fs) ===" % phase2_time)

    if global_a is not None and phase2_time > 20:
        beta_fine = [50, 100, 200, 400, 800, 1500]
        a2, score2 = run_lse_optimization(
            global_a, n, beta_fine, phase2_time, rng
        )
        a2 = a2 * (S / a2.sum())
        c2 = fft_convolve_self(a2)
        score2 = 2.0 * n * c2.max() / S**2
        if score2 < global_best:
            global_best = score2
            global_a = a2.copy()
            print("  After Phase 2 LSE: %.8f" % global_best)

        # Also try from the given sequence if it was different
        if given is not None and len(given) == n:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a3, score3 = run_lse_optimization(
                    given, n, [100, 200, 500], remaining2 * 0.5, rng
                )
                a3 = a3 * (S / a3.sum())
                c3 = fft_convolve_self(a3)
                score3 = 2.0 * n * c3.max() / S**2
                if score3 < global_best:
                    global_best = score3
                    global_a = a3.copy()
                    print("  From given (high beta): %.8f" % global_best)

    # Phase 3: Subgradient polish + try n=16384
    remaining = deadline - time.monotonic()
    phase3_time = min(phase3_time, remaining)
    print("=== Phase 3: Polish + explore (%.0fs) ===" % phase3_time)

    if global_a is not None and phase3_time > 20:
        # Subgradient polish
        sg_time = phase3_time * 0.5
        a_poly, score_poly = subgradient_polish(global_a, n, sg_time, rng)
        c = fft_convolve_self(a_poly)
        score_poly = 2.0 * n * c.max() / S**2
        if score_poly < global_best:
            global_best = score_poly
            global_a = a_poly.copy()
            print("  After subgrad polish: %.8f" % global_best)

        # Try n=16384 if time remains
        remaining2 = deadline - time.monotonic()
        if remaining2 > 90:
            print("  Trying n=16384...")
            n2 = 16384
            S2 = np.sqrt(2.0 * n2)
            # Upsample
            a_up = np.interp(
                np.linspace(0, 1, n2),
                np.linspace(0, 1, len(global_a)),
                global_a
            )
            a_up = np.maximum(a_up, 1e-6)
            a_up = a_up * (S2 / a_up.sum())

            inits2 = generate_initializations(n2, rng, a_up)
            beta_sched2 = [10, 30, 80, 200]
            time_per_init2 = (remaining2 - 30) / min(3, len(inits2)) / len(beta_sched2)

            for name2, arr2 in inits2[:3]:
                if time.monotonic() > deadline - 20:
                    break
                a_opt2, sc2 = run_lse_optimization(
                    arr2, n2, beta_sched2, time_per_init2 * len(beta_sched2) * 1.2, rng
                )
                a_opt2 = a_opt2 * (S2 / a_opt2.sum())
                c2b = fft_convolve_self(a_opt2)
                real2 = 2.0 * n2 * c2b.max() / S2**2
                if real2 < global_best:
                    global_best = real2
                    global_a = a_opt2.copy()
                    print("  [n=%d, %s] NEW BEST: %.8f" % (n2, name2, real2))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            n = 256
            global_a = np.ones(n, dtype=np.float64) * (np.sqrt(2 * n) / n)
    else:
        n = 256
        global_a = np.ones(n, dtype=np.float64) * (np.sqrt(2 * n) / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.8f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
