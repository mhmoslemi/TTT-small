"""

Key innovations:

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def upsample_coarse_to_fine(a_coarse, n_fine):
    """Upsample by nearest neighbor interpolation."""
    n_coarse = len(a_coarse)
    ratio = n_fine / n_coarse
    indices = (np.arange(n_fine) * ratio).astype(int)
    indices = np.clip(indices, 0, n_coarse - 1)
    a_fine = a_coarse[indices]
    return a_fine

def fista_block(a_init, n, S, N, max_time, rng, top_k_init=50, top_k_max=1024, restart_interval=200):
    """FISTA with adaptive top-K subgradient and backtracking."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, None)
    a = project_simplex(a, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    step = 0.1 * S / np.sqrt(n)
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 10)
        
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute subgradient
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-18:
            break
        
        # Backtracking line search
        x_new = a
        for bt in range(20):
            step_bt = step * (0.9 ** bt)
            x_candidate = project_simplex(y - step_bt * grad / gnorm * gnorm, S)
            # Wait, I should use the actual step size
            x_candidate = project_simplex(y - step_bt * grad, S)
            c_x = compute_conv(x_candidate, N)
            cx_max = c_x.max()
            if cx_max <= cur:
                x_new = x_candidate
                step = min(step * 1.1, 10.0 * S / n)
                break
        else:
            x_new = project_simplex(y - step * grad, S)
            step *= 0.5
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint
        y_sum = y.sum()
        if y_sum > S * 1.01 or y_sum < S * 0.99:
            y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Periodic restart
        if it % restart_interval == 0:
            c_a = compute_conv(a, N)
            if c_a.max() < best_cmax:
                best_cmax = c_a.max()
                best_a = a.copy()
            a = np.clip(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), 0, None)
            a = project_simplex(a, S)
            y = a.copy()
            t = 1.0
            step = 0.1 * S / np.sqrt(n)
    
    return best_a, best_cmax

def coordinate_descent(a_init, n, S, N, max_time, rng):
    """Coordinate descent: optimize one element at a time."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, None)
    a = project_simplex(a, S)
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()

    # For element i, it affects c_k for k in [i, i+n-1] (approximately)

    # c_k depends on a_i through the term a_i * a_{k-i} (and a_{k-i} * a_i = same)

    # More precisely: c_k = sum_j a_j * a_{k-j}

    # Wait, that's not quite right for the self-convolution.
    
    # Actually for linear convolution: c_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a_j * a_{k-j}
    # The term involving a_i is: a_i * a_{k-i} if k-i is in [0,n-1]
    # But also a_{k-i} * a_i if i is in [0,n-1] and k-i is in [0,n-1]
    # Since it's the same term (j=i gives a_i * a_{k-i}), and j=k-i gives a_{k-i} * a_i,

    # So c_k = a_i * a_{k-i} + rest_k (if k != 2i and 0 <= k-i < n)
    # And c_{2i} = a_i^2 + rest_{2i}
    
    # To minimize max(c_k), for a given a_i value, the convolution is:
    # c_k = a_i * a_{k-i} + rest_k for k where k-i is valid
    # c_k is linear in a_i (or quadratic for k=2i)
    
    # This is complex to optimize exactly. Let's use a simple approach:
    # For each element a_i, try values around its current value and keep the best.

    patience = 3 * n  # iterations without improvement before stopping
    
    last_improve = 0
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - start_time >= max_time:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            # Try a few values
            best_local = a.copy()
            best_local_max = best_cmax
            
            for trial in range(n_inner):
                # Try: slightly less, slightly more, zero, half
                if trial == 0:
                    new_val = old_val * 0.8
                elif trial == 1:
                    new_val = old_val * 1.2
                elif trial == 2:
                    new_val = 0.0
                elif trial == 3:
                    new_val = old_val * 0.5
                else:
                    new_val = old_val * 1.5
                
                new_val = max(0.0, new_val)
                a[i] = new_val
                
                # Check sum constraint (approximately - we'll re-project at the end)
                c = compute_conv(a, N)
                cm = c.max()
                if cm < best_local_max:
                    best_local_max = cm
                    best_local = a.copy()
            
            a = best_local.copy()
            a[i] = best_local[i]
            
            if best_local_max < best_cmax - 1e-12:
                best_cmax = best_local_max
                best_a = a.copy()
                improved = True
                last_improve = it
            
            # Re-project to simplex
            a = project_simplex(a, S)

def spectral_equalize(a_init, n, S, N, max_time, rng):

        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        
        c = np.real(ifft(A * A))[:2*n-1]
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def topk_subgradient(c, a, N, K):
    """Compute (1/K) * sum_{k in topK} 2*Q_k*a, the gradient of top-K mean."""
    n = len(a)
    top_idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / K
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, top_idx


def fista_topk(a_init, n, S, N, max_time, rng, K_init=5, K_max=None, step=0.4):
    """
    FISTA on top-K mean objective.
    The top-K mean is 2-smooth, so step <= 0.5 is safe.
    """
    if K_max is None:
        K_max = max(K_init, n // 8)
    
    a = project_simplex(a_init, S)
    y = a.copy()
    t_val = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    it = 0
    no_improve = 0
    last_best = best_cmax
    
    while time.monotonic() - start_time < max_time:
        it += 1
        K = min(K_max, K_init + it // 50)
        
        c = compute_conv(y, N)
        cur_max = c.max()
        
        if cur_max < best_cmax - 1e-12:
            best_cmax = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        grad, top_idx = topk_subgradient(c, y, N, K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Gradient step with projection
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        
        # Re-project to maintain sum
        y_sum = y.sum()
        if abs(y_sum - S) > 0.01 * S:
            y = project_simplex(y, S)
        
        a = x_new
        t_val = t_new
        
        # Periodic restart on stall
        if no_improve > 500:
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            y = a.copy()
            t_val = 1.0
            no_improve = 0
            last_best = best_cmax
        
        # Adaptive step: reduce if not improving
        if no_improve > 1000:
            step = max(0.1, step * 0.5)
            no_improve = 0
        
        # Periodic spectral reshaping
        if it % 1000 == 0 and rng.random() < 0.3:
            a_sp = spectral_step_local(a, n, S, N, K=min(20, K))
            c_sp = compute_conv(a_sp, N)
            if c_sp.max() < best_cmax:
                best_cmax = c_sp.max()
                best_a = a_sp.copy()
                a = a_sp
                y = a.copy()
                t_val = 1.0
                no_improve = 0
    
    return best_a, best_cmax


def spectral_step_local(a, n, S, N, K=10):
    """One spectral shaping step to reduce peaks."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    
    top_idx = np.argpartition(c, -K)[-K:]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    weights = top_vals / total
    m_arr = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    
    alpha = 0.3
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def lse_nesterov(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0, K_max=512):
    """
    Nesterov FISTA on LSE-smoothed max with adaptive beta.
    """
    a = project_simplex(a_init, S)
    y = a.copy()
    t_val = 1.0
    beta = beta_init
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptively increase beta
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.3)
        
        c = compute_conv(y, N)
        c_max = c.max()
        
        if c_max < best_cmax - 1e-12:
            best_cmax = c_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # LSE gradient: (1/beta) * sum_k w_k * 2*Q_k*a where w_k = softmax(beta*(c_k - c_max))
        shifted = beta * (c - c_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 0)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Only use top K for efficiency
        K = min(K_max, max(10, it // 10))
        top_K_idx = np.argpartition(c, -K)[-K:]
        w_top = np.zeros(2 * n - 1)
        w_top[top_K_idx] = w[top_K_idx]
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            break
        w_top = w_top / w_top_sum  # renormalize
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w_top, fsize)
        Af = fft(y, fsize)
        grad = (2.0 / beta) * np.real(ifft(W * np.conj(Af)))[:n]
        
        # Wait, the LSE gradient with respect to a is:
        # LSE = c_max + (1/beta)*log(sum exp(beta*(c_k - c_max)))
        # d LSE / d a_i = (1/beta) * sum_k [beta * exp(beta*(c_k-c_max)) / sum_j exp(beta*(c_j-c_max))] * dc_k/da_i
        # = sum_k w_k * dc_k/da_i
        # where w_k = exp(beta*(c_k-c_max)) / sum_j exp(...)
        # and dc_k/da_i = 2*a_{k-i} (or a_i if k=2i)
        # So grad = sum_k w_k * 2*Q_k*a = 2 * conv(w, a) [approximate]
        # The 1/beta factor is already in the definition of LSE but cancels in the gradient.
        # Wait no:
        # LSE_beta(a) = (1/beta) * log(sum_k exp(beta * c_k(a)))
        # d/da_i LSE = (1/beta) * sum_k exp(beta*c_k) * beta * dc_k/da_i / sum_j exp(beta*c_j)
        # = sum_k [exp(beta*c_k) / sum_j exp(beta*c_j)] * dc_k/da_i
        # = sum_k w_k * dc_k/da_i
        # where w_k = exp(beta*c_k) / sum_j exp(beta*c_j)
        # 
        # So the gradient does NOT have a 1/beta factor! It's just the weighted sum of 
        # the individual gradients, weighted by the softmax weights.
        # 
        # But the Lipschitz constant of this gradient is:
        # L = sum_k w_k * ||2*Q_k||_2 = 2 * sum_k w_k = 2
        # So the step size should be at most 1/2, regardless of beta!
        # 
        # That's interesting. So the LSE gradient has the same Lipschitz constant as 
        # the top-K mean gradient (which is also 2). The difference is in the direction:
        # LSE spreads the weight more smoothly, while top-K concentrates on the top K.
        
        # So the step size is 0.4 (slightly less than 0.5 for safety)
        step = 0.4
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]  # no 1/beta factor!
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if abs(y_sum - S) > 0.01 * S:
            y = project_simplex(y, S)
        
        a = x_new
        t_val = t_new
        
        if no_improve > 500:
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            y = a.copy()
            t_val = 1.0
            no_improve = 0
        
        if no_improve > 2000:
            step = 0.3
            no_improve = 0
    
    return best_a, best_cmax

    """Upsample by nearest neighbor."""

    if n_c == n_fine:
