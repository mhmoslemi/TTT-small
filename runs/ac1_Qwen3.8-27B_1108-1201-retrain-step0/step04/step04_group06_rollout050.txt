"""

Strategy:
1. Spectral water-filling: iteratively reshape power spectrum to flatten autocorrelation peaks

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand < w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_selfconv(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def spectral_step(a, N, n, S, top_k, alpha, rng, target=1.45):
    """One spectral shaping step targeting top-k peak lags. Returns new a."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target:
        return a

    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    # Weight by excess above mean
    mean_c = c.mean()
    excess = np.maximum(top_vals - mean_c, 0.0)
    if excess.sum() < 1e-30:
        excess = top_vals
    weights = excess / excess.sum()

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Adaptive alpha
    ratio = (c_max - target) / max(c_max, 1e-10)
    alpha_eff = alpha * (0.5 + 0.5 * min(1.0, ratio + 0.2))

    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    # Small random phase for exploration
    if rng is not None and rng.random() < 0.15:
        phase_noise = rng.normal(0, 0.03 * alpha_eff, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def nesterov_subgrad(a_init, n, S, max_time, rng, top_k_init=30, top_k_max=256):

    it = 0
    N = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 10)

        c = fft_selfconv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if best_score < 2.001:
            break

        # Top-k subgradient via FFT
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Build weight vector for lags
        w_vec = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_vec[int(k)] = weights[i]

        # Gradient: d(sum_k w_k * conv[k]) / da = 2 * cross_correlation(a, w_vec_reversed)
        # conv[k] = sum_i a[i]*a[k-i], d(conv[k])/da[j] = 2*a[k-j] (if valid)
        # So gradient[j] = sum_k w_k * 2*a[k-j] = 2 * (w_vec_convolved_with_reversed_a)[j]
        # This is 2 * sum_k w_k * a[k-j] = 2 * (w * a_reversed)[n-1-j+... ]
        # More precisely: grad[j] = 2 * sum_k w_k * a[k-j] where 0<=k-j<n
        # = 2 * sum_{m} w_{j+m} * a[m] (where m=k-j, so k=j+m)
        # = 2 * (w_vec cross-correlated with a)[j]
        # We can compute this via FFT: ifft(FFT(w_vec_extended) * conj(FFT(a_extended)))

        fsize = 1 << (len(w_vec) + n - 2).bit_length()
        W = fft(w_vec, fsize)
        A = fft(y, fsize)
        # grad[j] = 2 * sum_{m=0}^{n-1} w_vec[j+m] * a[m] for valid j+m
        # This is the cross-correlation of w_vec and a: (w_vec ⊛ a)[j] = sum_m w_vec[j+m]*a[m]
        # = ifft(conj(FFT(w_vec)) * FFT(a)) ... let me think more carefully
        # Actually: ifft(FFT(w_vec) * conj(FFT(a)))[j] = sum_m w_vec[m] * conj(a[m-j])
        # For real: = sum_m w_vec[m] * a[m-j] (with circular indexing)
        # We want sum_m w_vec[j+m] * a[m] = sum_p w_vec[p] * a[p-j] (p=j+m)
        # = ifft(FFT(w_vec) * conj(FFT(a)))[j] (circular, padded)

        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_selfconv(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Occasional restart
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def coord_descent(a_init, n, S, max_time, rng, n_top=25, n_rounds=200):

    best_score = np.inf

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:

        c = fft_selfconv(a)
        c_max = c.max()
        peak_lag = int(np.argmax(c))

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if best_score < 2.001:
            break

        # Find top-contributing coordinates at peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]
        improved = False

        for idx in top_coords:

            if time.monotonic() - start > max_time:
                break
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue

            orig_val = a[i]
            if orig_val < 1e-15:
                continue
            best_frac = 1.0
            best_cmax = c_max

            # Quick evaluation: only recompute affected lags
            delta_idx = [i, j]
            k_lo = max(0, min(i, j))
            k_hi = min(2 * n - 2, max(i, j) + n - 1)

            for frac in multipliers:
                new_val = orig_val * frac
                new_a = a.copy()
                new_a[i] = new_val
                # Only check a window of lags around the peak
                check_lo = max(0, peak_lag - n)
                check_hi = min(2 * n - 2, peak_lag + n)
                # Fast check: recompute just the affected lags
                new_max = c_max
                for k in range(k_lo, k_hi + 1):
                    j2 = k - i
                    j3 = i
                    k2 = k - j3
                    # Original contribution of a[i] to conv[k]:
                    # a[i]*a[k-i] (if valid) + a[k-i]*a[i] (if valid, i.e., a[i]*a[k-i])
                    orig_contrib = 0.0
                    if 0 <= j2 < n:
                        orig_contrib += a[i] * a[j2]
                    if 0 <= k2 < n and k2 != j2:
                        orig_contrib += a[i] * a[k2]
                    # New contribution
                    new_contrib = 0.0
                    if 0 <= j2 < n:
                        new_contrib += new_val * a[j2]
                    if 0 <= k2 < n and k2 != j2:
                        new_contrib += new_val * a[k2]
                    new_c_k = c[k] - orig_contrib + new_contrib
                    if new_c_k > new_max:
                        new_max = new_c_k
                        if new_max > best_cmax + 0.01:
                            break

                if new_max < best_cmax - 1e-10:
                    best_cmax = new_max
                    best_frac = frac

            if best_frac != 1.0:
                a[i] = orig_val * best_frac
                improved = True

        # Renormalize
        a = project_simplex(a, S)

        if not improved and rng.random() < 0.4:

            n_perturb = min(10, len(top_coords))
            for idx in top_coords[:n_perturb]:
                a[int(idx)] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)

def spectral_waterfill_loop(a_init, n, N, S, deadline, rng, target=1.45,
                            top_k_init=5, alpha_init=0.5):
    """Iterative spectral water-filling until deadline. Returns (best_a, best_score)."""
    a = a_init.copy()
    best_a = a.copy()
    best_score = fft_selfconv(a).max()
    it = 0
    stall = 0
    last_best = best_score

    while time.monotonic() < deadline:
        it += 1
        top_k = min(64, top_k_init + it // 15)
        alpha = alpha_init / (1.0 + 0.03 * it)

        a_new = spectral_step(a, N, n, S, top_k, alpha, rng, target)

        c_new = fft_selfconv(a_new)
        actual_max = c_new.max()

        if actual_max < best_score - 1e-8:
            best_score = actual_max
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
        else:
            a = a_new.copy()
            stall += 1
            if stall > 80:
                # Perturb to escape
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                stall = 0

        if best_score <= target:
            break

        if it % 200 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)

    return best_a, best_score


def structured_init(n, rng, method="sin"):

    elif method == "chirp":

    elif method == "alt":

    elif method == "pw":

        piece_len = n // n_pieces
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)[:n]
        if len(a) < n:

    elif method == "barker_like":
        # Try to mimic Barker code structure (alternating with varying amplitude)

        a = np.zeros(n)
        for i in range(n):
            s = signs[i % 7]
            a[i] = 0.7 + 0.3 * s
        a += 0.1 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)

    return np.clip(a, 0.01, 2.0)


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        if sc0 < global_best:
            global_best = sc0
            global_a = a_norm.copy()
            print("  [%s] init: %.6f" % (name, global_best))

        time_budget = min(per_init_time, 45.0)
        a_sw, sc_sw = spectral_waterfill_loop(
            a_norm, n, N, S,
            time.monotonic() + time_budget,
            rng, target=1.45, top_k_init=5, alpha_init=0.5)

        if update_best(a_sw):
            print("  [%s] waterfill: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f (time used: %.0fs) ===" % (global_best, time.monotonic() - t0))

    if global_a is None:
        global_a = safe_normalize(np.ones(n), n)
        global_best = fft_selfconv(global_a).max()

    # === Phase 2: Nesterov subgradient (30% time) ===
    print("=== Phase 2: Nesterov subgradient ===")
    t1 = time.monotonic()
    phase2_budget = (deadline - t1) * 0.30

    a_curr = global_a.copy()
    n_iter = 0

    while time.monotonic() < t1 + phase2_budget:
        rem = t1 + phase2_budget - time.monotonic()
        if rem < 5:
            break

        seg = min(rem * 0.5, 60.0)
        a_ng, sc_ng = nesterov_subgrad(a_curr, n, S, seg, rng,
                                       top_k_init=30, top_k_max=200)
        n_iter += 1
        if update_best(a_ng):
            a_curr = a_ng.copy()
            print("  [P2 iter%d] nesterov: %.6f" % (n_iter, global_best))
        else:
            # Perturb and try spectral
            a_pert = project_simplex(
                a_curr + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)
            seg2 = min(rem * 0.3, 30.0)
            a_sw, _ = spectral_waterfill_loop(a_pert, n, N, S,
                                              time.monotonic() + seg2,
                                              rng, target=1.45,
                                              top_k_init=10, alpha_init=0.4)
            if update_best(a_sw):
                a_curr = a_sw.copy()
                print("  [P2 iter%d] perturb+wf: %.6f" % (n_iter, global_best))

    print("=== Phase 2 done. Best: %.6f (time used: %.0fs) ===" % (global_best, time.monotonic() - t1))

    # === Phase 3: Coordinate descent + fine polish (25% time) ===
    print("=== Phase 3: Coordinate descent & fine polish ===")
    t2 = time.monotonic()
    phase3_deadline = deadline

    # Coordinate descent
    rem = phase3_deadline - time.monotonic()
    if rem > 5:
        seg = min(rem * 0.4, 120.0)
        a_cd, sc_cd = coord_descent(global_a, n, S, seg, rng,
                                    n_top=30, n_rounds=500)
        if update_best(a_cd):
            print("  [P3] coord: %.6f" % global_best)

    # Fine spectral shaping
    rem = phase3_deadline - time.monotonic()
    if rem > 5:
        seg = min(rem * 0.3, 60.0)
        a_fine, _ = spectral_waterfill_loop(global_a, n, N, S,
                                            time.monotonic() + seg,
                                            rng, target=1.45,
                                            top_k_init=20, alpha_init=0.15)
        if update_best(a_fine):
            print("  [P3] fine wf: %.6f" % global_best)

    # Final Nesterov with high top_k
    rem = phase3_deadline - time.monotonic()
    if rem > 3:
        seg = min(rem * 0.3, 60.0)
        a_fin, _ = nesterov_subgrad(global_a, n, S, seg, rng,
                                    top_k_init=100, top_k_max=400)
        if update_best(a_fin):
            print("  [P3] final nesterov: %.6f" % global_best)

    # One last coordinate descent
    rem = phase3_deadline - time.monotonic()
    if rem > 3:
        a_last, _ = coord_descent(global_a, n, S, rem, rng,
                                  n_top=50, n_rounds=300)
        if update_best(a_last):
            print("  [P3] final coord: %.6f" % global_best)

    print("=== Phase 3 done. Best internal score: %.6f ===" % global_best)

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using uniform fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
