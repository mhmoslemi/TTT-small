```python
"""
Search for non-negative sequences minimizing

    2 * n * max_k (a * a)_k / (sum_j a_j)^2

by working on the normalized log-simplex state z (a = softmax(z)).

The main local optimizer uses a direct-search / multiplicative gradient scheme:
- compute the convolution c = a*a;
- choose a probability p over large lags (active-set / soft-max over c);
- compute a subgradient-like vector g for q(a) = p^T c;
- try several multiplicative updates z <- z - m * normalized(g) and several
  two-coordinate log-ratio moves;
- accept the best trial by exact re-evaluation using FFT convolution;
- restart from the best local point with moderated log-noise / interpolation
  when stuck.

Multiple starting points are used:
- the supplied parent sequence if present;
- a structured tail from the power series of (1-x)^(-1/2);
- skew/interpolated variants.

This is a fully self-contained search procedure.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


# ---------------------------------------------------------------------------
# Global best bookkeeping
# ---------------------------------------------------------------------------
_best_score = np.inf
_best_z = None


# ---------------------------------------------------------------------------
# Search parameters
# ---------------------------------------------------------------------------
M_TRIALS = (0.005, 0.02, 0.07, 0.25, 0.8, 2.5, 8.0, 25.0)
SOFT_M_TRIALS = (0.05, 0.25, 1.0, 4.0)
DELTA_TRIALS = (0.02, 0.07, 0.25, 1.0)

RESTART_AFTER = 30
RESTART_SCALES = (0.03, 0.08, 0.2, 0.5, 1.0, 2.0, 4.0)
RESTARTS_MAX = 30


# ---------------------------------------------------------------------------
# Basic helpers
# ---------------------------------------------------------------------------
def next_pow2(m):
    """Smallest power of two >= m (for m >= 1)."""
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_z(z):
    """Normalize log-simplex state so max(z)=0 and z is within a safe range."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z.copy()

    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, -600.0)

    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.zeros(z.shape, dtype=np.float64)

    z = z - zmax
    z = np.maximum(z, -600.0)
    return z


def softmax_z(z):
    """Convert log-simplex state to a probability vector."""
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return z.copy()

    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, -600.0)

    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.full(n, 1.0 / n, dtype=np.float64)

    zc = z - zmax
    if zc.size and float(np.min(zc)) < -650.0:
        zc = np.maximum(zc, -650.0)

    e = np.exp(zc)
    s = float(np.sum(e))
    if not np.isfinite(s) or s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def z_from_a(a):
    """Convert a non-negative sequence to a normalized log-simplex state."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([], dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-300:
        return np.zeros(a.shape, dtype=np.float64)

    a = a / s
    floor = np.exp(-600.0)
    a = np.maximum(a, floor)
    a = a / float(np.sum(a))

    z = np.log(a)
    return normalize_z(z)


def interpolate_z(z, new_n):
    """Interpolate a log-simplex state to a new length."""
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    n = z.size
    if n == 0:
        return np.zeros(int(new_n), dtype=np.float64)
    if n == 1:
        return normalize_z(np.full(int(new_n), z[0]))
    old_t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    new_t = np.linspace(0.0, 1.0, int(new_n), dtype=np.float64)
    return normalize_z(np.interp(new_t, old_t, z))


def make_base_z(n, skew=0.0, right=0.0):
    """
    Structured starting log-state.

    The base coefficients are the first n coefficients of (1-x)^(-1/2),
    i.e. binom(2i, i) / 4^i.  Optional skew and right-tail terms break
    symmetry and create different basins.
    """
    n = int(max(1, n))
    b = np.ones(n, dtype=np.float64)
    if n > 1:
        i = np.arange(1, n, dtype=np.float64)
        ratio = (2.0 * i - 1.0) / (2.0 * i)
        b[1:] = np.cumprod(ratio)

    logb = np.log(np.maximum(b, 1e-300))
    z = logb.copy()

    if skew != 0.0:
        idx = np.linspace(-0.5, 0.5, n, dtype=np.float64)
        z = z + float(skew) * idx

    if right != 0.0:
        z = z + float(right) * logb[::-1]

    return normalize_z(z)


# ---------------------------------------------------------------------------
# FFT convolution helpers
# ---------------------------------------------------------------------------
def fft_conv_self(a, N):
    """Linear self-convolution a*a truncated to length 2*len(a)-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    F = rfft(a, N)
    c = irfft(F * F, N)
    c = c[:L]
    return np.maximum(c, 0.0)


def score_from_z(zt, n, N):
    """Return (score, a, c) from a z-state; score is the target function."""
    a = softmax_z(zt)
    c = fft_conv_self(a, N)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-300:
        return np.inf, a, c
    return float(2.0 * n * float(np.max(c)) / (s * s)), a, c


def gradient_from_Far(Far, p, N, n):
    """
    Gradient of q(a)=sum_k p_k (a*a)_k with respect to a.

    If Far = rfft(reversed(a), N), then the slice
    irfft(Far*rfft(p,N,N))[n-1:n-1+n] gives sum_j a_j p_{u+j}.
    The true gradient is twice that.
    """
    Fp = rfft(p, N)
    conv = irfft(Far * Fp, N)
    return 2.0 * conv[n - 1:n - 1 + n]


# ---------------------------------------------------------------------------
# Lag probability p over convolution coefficients
# ---------------------------------------------------------------------------
def make_lag_p(c, mode="top"):
    """
    Make a probability distribution over lags k=0..2n-2.

    mode = "top"  : uniform over near-maximal coefficients, with a fallback
                    to a top-argpartition set.
    mode = "soft" : smooth weights concentrated near the maximum.
    mode = "broad": proportional to c_k.
    """
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)

    T = float(np.max(c))
    if not np.isfinite(T) or T <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[0] = 1.0
        return p

    if mode == "top":
        p = np.zeros(L, dtype=np.float64)
        tol = max(1e-15, 1e-8 * T)
        mask = c >= (T - tol)
        cnt = int(np.count_nonzero(mask))

        if cnt == 0:
            p[int(np.argmax(c))] = 1.0
        else:
            if cnt < 8:
                m = max(8, L // 64)
                if m < L:
                    idx = np.argpartition(c, -m)[-m:]
                    p[idx] = 1.0
                else:
                    p[:] = 1.0
                p[mask] = 1.0
            else:
                p[mask] = 1.0

        s = float(np.sum(p))
        if s > 0.0 and np.isfinite(s):
            p /= s
        else:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0

    elif mode == "soft":
        width = 0.25 * T
        x = (c - T) / width
        x = np.clip(x, -50.0, 0.0)
        p = np.exp(x)
        s = float(np.sum(p))
        if s > 0.0 and np.isfinite(s):
            p /= s
        else:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0

    else:  # broad
        s = float(np.sum(c))
        if s > 0.0 and np.isfinite(s):
            p = c / s
        else:
            p = np.full(L, 1.0 / L, dtype=np.float64)

    return p


# ---------------------------------------------------------------------------
# Global best update
# ---------------------------------------------------------------------------
def _update_global(z, score, tag=""):
    """Update module-level best if this z-state is better."""
    global _best_score, _best_z
    if z is None:
        return
    if not np.isfinite(score):
        return
    if score < _best_score - 1e-15:
        _best_score = float(score)
        _best_z = np.asarray(z, dtype=np.float64).copy()
        print("global best: tag=%s score=%.10f n=%d" % (tag, _best_score, len(_best_z)), flush=True)


# ---------------------------------------------------------------------------
# Main local optimizer from a z-state
# ---------------------------------------------------------------------------
def optimize_z(z0, max_time, rng, tag="opt", ref_z=None, m_trials=None, delta_trials=None):
    """
    Local optimizer in log-simplex space.

    z0       : initial log-state
    max_time : seconds
    rng      : numpy Generator
    tag      : string for logging
    ref_z    : optional reference log-state of the same length used in restarts
    """
    if m_trials is None:
        m_trials = M_TRIALS
    if delta_trials is None:
        delta_trials = DELTA_TRIALS

    z0 = np.asarray(z0, dtype=np.float64).reshape(-1)
    n = z0.size
    if n < 2:
        return z0.copy(), np.inf

    L = 2 * n - 1
    N = next_pow2(3 * n - 2)

    z = normalize_z(z0)
    a = softmax_z(z)
    c = fft_conv_self(a, N)
    cur = float(2.0 * n * float(np.max(c)))
    s = float(np.sum(a))
    if s > 0.0:
        cur = cur / (s * s)

    local_score = cur
    local_z = z.copy()
    t_end = time.monotonic() + max(0.0, float(max_time))

    print("optimize start: tag=%s n=%d init_score=%.10f" % (tag, n, local_score), flush=True)

    if np.isfinite(local_score) and local_score < _best_score:
        _update_global(z, local_score, tag + "/init")

    no_improve = 0
    restarts = 0
    it = 0

    while time.monotonic() < t_end:
        it += 1

        T = float(np.max(c))
        best_trial_score = cur
        best_trial_z = None
        best_trial_a = None
        best_trial_c = None

        # --- Primary active-set / top-lag gradient reads -----------------
        p = make_lag_p(c, "top")
        Far = rfft(a[::-1], N)
        g = gradient_from_Far(Far, p, N, n)

        gmax = float(np.max(g))
        gmin = float(np.min(g))
        grange = gmax - gmin

        if np.isfinite(grange) and grange > 1e-14 * max(1.0, abs(gmax)):
            rel = (g - gmin) / grange

            # Multiplicative log-simplex trials.
            for m in m_trials:
                zt = z - float(m) * rel
                zt = normalize_z(zt)
                sc, at, ct = score_from_z(zt, n, N)
                if np.isfinite(sc) and sc < best_trial_score:
                    best_trial_score = sc
                    best_trial_z = zt
                    best_trial_a = at
                    best_trial_c = ct

            # Two-coordinate log-ratio trials.
            contrib = np.maximum(0.0, a * g)
            hi = int(np.argmax(contrib))
            lo = int(np.argmin(contrib))

            if hi != lo:
                use_opposite = no_improve > 8
                for delta in delta_trials:
                    # Increase low-contribution coordinate, decrease high one.
                    zt = z.copy()
                    zt[lo] += float(delta)
                    zt[hi] -= float(delta)
                    zt = normalize_z(zt)
                    sc, at, ct = score_from_z(zt, n, N)
                    if np.isfinite(sc) and sc < best_trial_score:
                        best_trial_score = sc
                        best_trial_z = zt
                        best_trial_a = at
                        best_trial_c = ct

                    if use_opposite:
                        zt = z.copy()
                        zt[lo] -= float(delta)
                        zt[hi] += float(delta)
                        zt = normalize_z(zt)
                        sc, at, ct = score_from_z(zt, n, N)
                        if np.isfinite(sc) and sc < best_trial_score:
                            best_trial_score = sc
                            best_trial_z = zt
                            best_trial_a = at
                            best_trial_c = ct

        # --- Secondary soft-lag multiplier trials -------------------------
        if (it % 3 == 0) or (no_improve > 5):
            p2 = make_lag_p(c, "soft")
            g2 = gradient_from_Far(Far, p2, N, n)
            g2max = float(np.max(g2))
            g2min = float(np.min(g2))
            g2range = g2max - g2min

            if np.isfinite(g2range) and g2range > 1e-14 * max(1.0, abs(g2max)):
                rel2 = (g2 - g2min) / g2range
                for m in SOFT_M_TRIALS:
                    zt = z - float(m) * rel2
                    zt = normalize_z(zt)
                    sc, at, ct = score_from_z(zt, n, N)
                    if np.isfinite(sc) and sc < best_trial_score:
                        best_trial_score = sc
                        best_trial_z = zt
                        best_trial_a = at
                        best_trial_c = ct

        # --- Accept or restart -------------------------------------------
        if best_trial_z is not None and best_trial_score < cur - 1e-13:
            z = best_trial_z
            a = best_trial_a
            c = best_trial_c
            cur = best_trial_score

            if cur < local_score - 1e-15:
                local_score = cur
                local_z = z.copy()
                _update_global(z, local_score, tag)

            no_improve = 0
        else:
            no_improve += 1

            if no_improve >= RESTART_AFTER:
                restarts += 1
                if restarts > RESTARTS_MAX:
                    break

                scale = RESTART_SCALES[min(restarts - 1, len(RESTART_SCALES) - 1)]
                z = local_z.copy()

                if ref_z is not None and len(ref_z) == n:
                    alpha = min(0.7, 0.1 + 0.1 * restarts)
                    z = (1.0 - alpha) * z + alpha * ref_z

                z = z + rng.normal(0.0, scale, n)
                z = normalize_z(z)

                a = softmax_z(z)
                c = fft_conv_self(a, N)
                s = float(np.sum(a))
                cur = float(2.0 * n * float(np.max(c)) / (s * s)) if s > 0.0 else np.inf
                no_improve = 0

                if np.isfinite(cur) and cur < local_score - 1e-15:
                    local_score = cur
                    local_z = z.copy()
                    _update_global(z, local_score, tag + "/restart")

    print("optimize end:   tag=%s n=%d local_score=%.10f" % (tag, n, local_score), flush=True)
    return local_z, local_score

def propose_candidate(seed=42, budget_s=2300):
    """
    Search for the best non-negative sequence within the time budget.

    Returns a Python list of floats.
    """
    global _best_score, _best_z

    _best_score = np.inf
    _best_z = None

    t0 = time.monotonic()
    total = max(120.0, min(float(budget_s), 2350.0))
    deadline = t0 + total - 60.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Try to use the supplied parent construction, if present.
    # ------------------------------------------------------------------
    parent_z = None
    n_main = 8192

    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        raw = np.clip(raw, 0.0, 1000.0)
        sraw = float(np.sum(raw))
        if raw.size >= 64 and np.isfinite(sraw) and sraw > 0.0:
            p = raw / sraw

            if p.size > 20000:
                old_t = np.linspace(0.0, 1.0, p.size, dtype=np.float64)
                new_t = np.linspace(0.0, 1.0, 16384, dtype=np.float64)
                p = np.interp(new_t, old_t, p)
            elif p.size < 2048:
                old_t = np.linspace(0.0, 1.0, p.size, dtype=np.float64)
                new_t = np.linspace(0.0, 1.0, 4096, dtype=np.float64)
                p = np.interp(new_t, old_t, p)

            if p.size >= 256:
                parent_z = z_from_a(p)
                if 2048 <= p.size <= 16384:
                    n_main = p.size
    except Exception:
        parent_z = None

    base_main = make_base_z(n_main, 0.0)
    base_skew_main = make_base_z(n_main, 0.5)

    # ------------------------------------------------------------------
    # Phase 1: optimize the strongest known starting point.
    # ------------------------------------------------------------------
    if parent_z is not None:
        z0 = parent_z
        ref = base_main
        tag = "phase1_parent"
    else:
        z0 = base_main
        ref = base_skew_main
        tag = "phase1_base"

    now = time.monotonic()
    alloc1 = min(0.55 * total, 0.70 * max(0.0, deadline - now))
    if alloc1 < 60.0:
        alloc1 = max(30.0, 0.40 * max(0.0, deadline - now))

    if alloc1 > 20.0:
        optimize_z(z0, alloc1, rng, tag, ref)

    # ------------------------------------------------------------------
    # Phase 2: skew/noise variant of the current best.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 40.0:
        if _best_z is not None and 256 <= len(_best_z) <= 16384:
            nb = len(_best_z)
            idx = np.linspace(-0.5, 0.5, nb, dtype=np.float64)
            z0 = normalize_z(_best_z + 0.5 * idx + rng.normal(0.0, 0.05, nb))
            ref = make_base_z(nb, 0.0)
        else:
            z0 = base_skew_main
            ref = base_main

        alloc2 = min(0.12 * total, 0.40 * max(0.0, deadline - now))
        if alloc2 > 25.0:
            optimize_z(z0, alloc2, rng, "phase2_variant", ref)

    # ------------------------------------------------------------------
    # Phase 3: larger length / interpolation from the best state.
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline - 90.0:
        if _best_z is not None and 256 <= len(_best_z) < 16384:
            z0 = interpolate_z(_best_z, 16384)
            ref = make_base_z(16384, 0.0)
        else:
            # If we already have a 16k best, try a more skewed 16k start.
            if (_best_z is not None and len(_best_z) == 16384 and
                    _best_score < 1.52 and now < deadline - 240.0):
                pass  # defer to final polish in this case
            else:
                z0 = make_base_z(16384, 0.0)
                ref = make_base_z(16384, 0.7)

        if ('z0' in locals()) and ('ref' in locals()):
            alloc3 = min(0.30 * total, 0.55 * max(0.0, deadline - now))
            if alloc3 > 50.0:
                optimize_z(z0, alloc3, rng, "phase3_large", ref)

    # ------------------------------------------------------------------
    # Phase 4: final polish from the global best.
    # ------------------------------------------------------------------
    if _best_z is not None:
        nb = len(_best_z)
        ref = None
        if 256 <= nb <= 16384:
            ref = make_base_z(nb, 0.0)

        now = time.monotonic()
        remaining = max(0.0, deadline - now - 10.0)
        if remaining > 15.0:
            optimize_z(_best_z.copy(), remaining, rng, "phase4_final", ref)

    # ------------------------------------------------------------------
    # Fallback and final verification.
    # ------------------------------------------------------------------
    if _best_z is None or len(_best_z) == 0 or not np.all(np.isfinite(_best_z)):
        a = np.ones(256, dtype=np.float64) / 256.0
        return [float(x) for x in a]

    a = softmax_z(_best_z)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        a = np.ones(256, dtype=np.float64) / 256.0
    else:
        a = a / s

    final_list = [float(x) for x in a]

    try:
        verified = evaluate_sequence(final_list)
        if not np.isfinite(verified):
            print("final evaluate returned inf, using fallback", flush=True)
            a = np.ones(256, dtype=np.float64) / 256.0
            final_list = [float(x) for x in a]
        else:
            print("final verified score=%.10f n=%d" % (verified, len(final_list)), flush=True)
    except Exception:
        # If evaluation is unavailable in this environment for some reason,
        # still return the internal best.
        print("final evaluation unavailable, returning internal best", flush=True)

    return final_list
```
