from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def softmax_seq(u, S):
    u_shifted = u - u.max()
    exp_u = np.exp(u_shifted)
    return S * exp_u / exp_u.sum()

def objective_from_a(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s*s)

def compute_grad_u(u, S, beta):
    a = softmax_seq(u, S)
    n = len(a)
    c = fft_convolve_self(a)
    # LSE weights
    c_max = c.max()
    bc = beta * (c - c_max)
    # Clip to avoid overflow
    bc = np.clip(bc, -700, 700)
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    # Compute h = w cross-correlated with a
    # h[m] = sum_k w[k] * a[k-m]

    return grad_u, f_value, a, c

    v_hat = v / (1 - beta2**t)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)

    A = fft(a, fsize)

def softmax_seq(u, S):
    """Convert u-space to a-space via softmax scaling."""
    u_shifted = u - u.max()
    exp_u = np.exp(u_shifted)
    s = exp_u.sum()
    if s < 1e-300:
        return np.ones(len(u)) * S / len(u)
    return S * exp_u / s


def lse_objective_and_grad(u, S, beta):
    """
    Compute the LSE-smoothed objective and its gradient w.r.t. u.
    Returns (grad_u, lse_value, a, c) where c is the convolution.
    """
    a = softmax_seq(u, S)
    n = len(a)
    c = fft_convolve_self(a)
    
    # LSE: (1/beta) * log(sum exp(beta * c_k))
    c_max = c.max()

def adam_step(u, grad, state, lr):
    """In-place Adam update."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    if 'm' not in state:
        state['m'] = np.zeros_like(u)
        state['v'] = np.zeros_like(u)
        state['t'] = 0
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad * grad
    m_hat = state['m'] / (1 - b1**t)
    v_hat = state['v'] / (1 - b2**t)
    u -= lr * m_hat / (np.sqrt(v_hat) + eps)

    b_rev = a[::-1]
    conv_len = len(w) + n - 1
    fsize = 1 << (conv_len - 1).bit_length()
    W = fft(w, fsize)
    B = fft(b_rev, fsize)
    conv = np.real(ifft(W * B))
    h = conv[n - 1:2 * n - 1]

    C = np.dot(a, h)
    grad_u = 2.0 * a * (h - C / S)

    return grad_u, a, c


def adam_step(u, grad, state, lr):
    """Adam optimizer step, modifying u in place."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    if 't' not in state:
        state['m'] = np.zeros_like(u)
        state['v'] = np.zeros_like(u)
        state['t'] = 0
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad * grad
    m_hat = state['m'] / (1 - b1 ** t)
    v_hat = state['v'] / (1 - b2 ** t)
    u -= lr * m_hat / (np.sqrt(v_hat) + eps)


def reset_adam(state):
    """Reset Adam state."""
    if len(state) > 0 and len(state.get('m', [])) > 0:
        state['m'] = np.zeros_like(state['m'])
        state['v'] = np.zeros_like(state['v'])
        state['t'] = 0


def run_smooth_opt(u_init, n, S, max_time, beta_max=3000.0, lr0=2.0):
    """Adam optimization on LSE-smoothed objective with annealing beta."""
    u = u_init.copy()
    state = {}
    start = time.monotonic()
    best_obj = np.inf
    best_a = None
    it = 0
    no_improve = 0
    best_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Annealing: exponential increase in beta
        beta = min(beta_max, np.exp(0.015 * it))

        grad_u, a, c = lse_grad(u, S, beta)
        true_obj = 2.0 * n * c.max() / (S * S)

        if true_obj < best_obj - 1e-12:
            best_obj = true_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Learning rate schedule: decay with iteration count
        lr = lr0 * np.exp(-0.0003 * it)
        lr = max(lr, 0.01)

        adam_step(u, grad_u, state, lr)

        # Periodic re-randomization to escape local minima
        if no_improve > 20000 and no_improve % 5000 == 0:
            noise_scale = 0.5 * np.exp(-no_improve / 10000.0)
            if noise_scale > 0.01:
                u += noise_scale * np.random.default_rng(no_improve).standard_normal(n)
            reset_adam(state)
            no_improve = 0

        if it % 50000 == 0:
            elapsed = time.monotonic() - start
            print("      it=%d, beta=%.1f, obj=%.6f, best=%.6f, t=%.0fs" % (
                it, beta, true_obj, best_obj, elapsed))

    return best_a, best_obj


def topk_conv_grad(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_refine(a_init, n, S, max_time, rng):
    """Subgradient refinement with adaptive top-k and backtracking."""
    a = project_simplex(a_init.copy(), S)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    step = 1.0
    consecutive_fail = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Cycle through k values
        phase = it % 200
        if phase < 20:
            k = 1
        elif phase < 60:
            k = 4
        elif phase < 120:
            k = 16
        else:
            k = 64

        grad = topk_conv_grad(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step_eff = min(step, 2.0 * S / (n * gnorm + 1e-15))
        a_new = project_simplex(a - step_eff * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            step = min(step * 1.03, 10.0)
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            step *= 0.85
            if consecutive_fail > 50:
                noise_scale = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                step = 2.0
                consecutive_fail = 0
            if consecutive_fail > 150:
                noise_scale = 0.02 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                step = 5.0
                consecutive_fail = 0

        if it % 20000 == 0:
            print("      subgrad it=%d, obj=%.6f, best=%.6f" % (it, obj_new, best_obj))

    return best_a, best_obj


def local_perturbation_search(a_init, n, S, max_time, rng):
    """Local search: randomly perturb subsets of the sequence."""
    a = project_simplex(a_init.copy(), S)
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        # Pick a random subset to perturb
        n_perturb = max(10, n // 50)
        indices = rng.choice(n, size=n_perturb, replace=False)
        scale = 0.05 * S / n
        a[indices] += rng.normal(0, scale, n_perturb)
        a = project_simplex(a, S)
        c = fft_convolve_self(a)
        new_obj = 2.0 * n * c.max() / (S * S)

        if new_obj < cur_obj:
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a = project_simplex(a - 0.5 * (a - best_a) * rng.random(), S)
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / (S * S)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2300):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses smooth optimization in softmax parameterization with adaptive LSE,
    combined with subgradient refinement and local perturbation search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    given_len = 0
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            given_len = len(given)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs, given_len=%d" % (budget_s, given_len))
    t0 = time.monotonic()

    # Determine main length
    if given is not None and 2048 <= given_len <= 32768:
        main_n = given_len
    else:
        main_n = 8192

    n = main_n
    S = np.sqrt(2.0 * n)

    # Convert given to u-space
    u_given = None
    if given is not None and len(given) == n:
        a_g = given.copy()
        s_g = a_g.sum()
        if s_g > 0.01:
            a_g = a_g * (S / s_g)
            u_given = np.log(np.maximum(a_g, 1e-300))
            # Center it
            u_given = u_given - u_given.mean()

    # Phase 1: Smooth optimization from multiple starts (60% of time)
    phase1_time = (deadline - time.monotonic()) * 0.60
    print("=== Phase 1: Smooth optimization (%.0fs) ===" % phase1_time)

    # Build list of initializations
    u_inits = []
    if u_given is not None:
        u_inits.append(("given", u_given.copy()))
        # Perturbed versions of given
        for i in range(3):
            u_pert = u_given + 0.1 * rng.standard_normal(n)
            u_inits.append(("given_pert%d" % i, u_pert))

    # Structured initializations
    t = np.arange(n) / n
    u_inits.append(("asym_linear", np.linspace(3.0, -3.0, n)))
    u_inits.append(("sin_asym", 2.0 * np.sin(2 * np.pi * t) + np.linspace(1.0, -1.0, n)))
    u_inits.append(("cos2", 1.5 * np.cos(4 * np.pi * t) + 0.5 * np.cos(2 * np.pi * 3 * t)))
    u_inits.append(("ramp_sin", np.linspace(0, 1, n) * 2.0 * np.sin(2 * np.pi * 5 * t)))
    u_inits.append(("rand", rng.standard_normal(n)))

    n_starts = len(u_inits)
    time_per_start = phase1_time / n_starts

    for i, (name, u_init) in enumerate(u_inits):
        remaining = deadline - time.monotonic() - phase1_time * 0.4
        alloc = min(time_per_start, remaining, phase1_time - i * time_per_start)
        if alloc < 10:
            continue

        print("  Start %d/%d: %s (alloc=%.0fs)" % (i + 1, n_starts, name, alloc))
        a_opt, obj_opt = run_smooth_opt(u_init, n, S, alloc, beta_max=3000.0, lr0=2.0)

        if a_opt is not None and obj_opt < global_best:
            global_best = obj_opt
            global_a = a_opt.copy()
            print("  -> obj=%.6f (elapsed %.0fs)" % (global_best, time.monotonic() - t0))

    print("=== Phase 1 done: best=%.6f ===" % global_best)

    # If we didn't find anything, use fallback
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    # Phase 2: Subgradient refinement (25% of remaining time)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.40
    if phase2_time > 20:
        print("=== Phase 2: Subgradient refinement (%.0fs) ===" % phase2_time)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_ref, obj_ref = subgradient_refine(global_a, n_best, S_best, phase2_time, rng)
        if obj_ref < global_best:
            global_best = obj_ref
            global_a = a_ref.copy()
            print("  After subgradient: %.6f" % global_best)

    # Phase 3: Go back to smooth optimization from the refined point (15% of remaining)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.40
    if phase3_time > 20 and global_a is not None:
        print("=== Phase 3: Smooth refinement from best (%.0fs) ===" % phase3_time)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_best = global_a.copy()
        s_best = a_best.sum()
        if s_best > 0.01:
            a_norm = a_best * (S_best / s_best)
            u_from_best = np.log(np.maximum(a_norm, 1e-300))
            u_from_best = u_from_best - u_from_best.mean()

            a_opt3, obj3 = run_smooth_opt(u_from_best, n_best, S_best, phase3_time,
                                          beta_max=5000.0, lr0=0.5)
            if a_opt3 is not None and obj3 < global_best:
                global_best = obj3
                global_a = a_opt3.copy()
                print("  After smooth refine 2: %.6f" % global_best)

    # Phase 4: Local perturbation search (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("=== Phase 4: Local perturbation (%.0fs) ===" % remaining)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_lp, obj_lp = local_perturbation_search(global_a, n_best, S_best, remaining, rng)
        if obj_lp < global_best:
            global_best = obj_lp
            global_a = a_lp.copy()
            print("  After local search: %.6f" % global_best)

    # Final: one more subgradient polish if time remains
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Final polish (%.0fs) ===" % remaining)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_fin, obj_fin = subgradient_refine(global_a, n_best, S_best, remaining, rng)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
