import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta, variance_weight=0.0):
    """Compute gradient of LSE(conv) + variance_weight * Var(conv)."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = e / tot

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad_lse = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    if variance_weight > 0.0:
        # Gradient of sum(conv^2) / (2 * len(conv))
        # d(sum conv[k]^2)/d(a_i) = 2 * sum_k conv[k] * d(conv[k])/d(a_i)
        # d(conv[k])/d(a_i) = a[k-i] + a[i+k] (valid range)
        # = 2 * [ (conv cross-correlate a)[i] + (conv correlate a)[i] ]
        C = fft(c, fsize)
        # cross-correlation: sum_k c[k] * a[k-i] -> ifft(fft(c) * conj(fft(a)))
        cc1 = np.real(ifft(C * np.conj(A)))[:n]
        # other direction: sum_k c[k] * a[i+k] -> ifft(conj(fft(c)) * fft(a))
        cc2 = np.real(ifft(np.conj(C) * A))[:n]
        grad_var = 2.0 * (cc1 + cc2) / (2.0 * len(c))
        grad_lse = grad_lse + variance_weight * grad_var

    return grad_lse


def exact_subgradient(y, c, topk=1, tol_frac=0.0):
    """Subgradient from top-k lags."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        active = active[np.argsort(c[active])[::-1][:topk]]
    else:
        k = min(topk, len(c))
        active = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_var(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
                  beta_max=5000.0, var_init=0.5, var_decay=0.001,
                  var_min=0.0, step_scale=1.0, restart_period=4000):
    """
    FISTA with LSE + variance penalty.
    The variance penalty encourages a flat convolution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        var_w = max(var_min, var_init * np.exp(-var_decay * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        grad = lse_gradient(y, c, beta, variance_weight=var_w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if stuck_count > 3000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % restart_period == 0:
            t_mom = 1.0
            decay = np.exp(-float(it) / (50000.0 * (n // 4096 + 1)))
            if decay > 0.05:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_coord_descent(a_init, max_time, rng, n_sweeps_target=200):
    """

    """

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    start = time.monotonic()
    best_conv = fft_convolve_self(a).max()
    best_a = a.copy()
    sweep = 0

    # Precompute timing
    t0 = time.monotonic()
    _ = fft_convolve_self(a[:min(1024, n)])
    t_conv_small = time.monotonic() - t0
    time_per_update = 0.01  # rough estimate for 1D search per coord

    while time.monotonic() - start < max_time:
        sweep += 1
        # Random order for diversity
        order = rng.permutation(n)
        improved = False

        for k in order:
            if (sweep * n + k) % 512 == 0:
                if time.monotonic() - start >= max_time:
                    break

            # 1D search for optimal a[k]
            # conv as function of a[k] is convex (max of linear/quadratic functions)
            old_val = a[k]
            # Try a range of values
            lo = 0.0
            hi = max(2.0 * old_val, S / 4.0)

            # Evaluate at a few points to bracket

                a[k] = val
                c = fft_convolve_self(a)
                return c.max()

            val0 = old_val
            f_mid = val0 * 0.5
            f_hi2 = min(hi, val0 * 2.0 + 0.1)

            c0 = eval_at_k(val0)
            c_mid = eval_at_k(f_mid)
            c_hi = eval_at_k(f_hi2)

            # Simple golden section on [0, f_hi2]
            if c_mid < c0 and c_mid < c_hi:
                lo, hi = 0.0, f_hi2
                best_local = f_mid
            elif c_hi < c0 and c_hi < c_mid:
                lo, hi = f_mid, f_hi2
                best_local = f_hi2
            else:
                lo, hi = 0.0, f_mid
                best_local = val0

            # Golden section search (15 iterations)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(15):
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                f1 = eval_at_k(x1)
                f2 = eval_at_k(x2)
                if f1 < f2:
                    hi = x2
                    best_local = x1
                else:
                    lo = x1
                    best_local = x2

            a[k] = best_local

            # Check improvement
            c_check = eval_at_k(best_local)
            a[k] = best_local
            if c_check < best_conv - 1e-10:
                best_conv = c_check
                best_a = a.copy()
                improved = True

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)

        if not improved and sweep > 5:
            # Try a perturbation
            idx = rng.integers(0, n, size=max(10, n // 100))
            a[idx] = np.maximum(a[idx] * (1.0 + rng.normal(0, 0.1, len(idx))), 0.0)
            a = project_simplex(a, S)

    return best_a, best_conv


def spectral_flatten(a, flatten_strength=0.3, n_band=None):
    """
    Spectral shaping: flatten the power spectrum to reduce convolution peaks.
    This modifies the frequency content to make |A(f)|^2 more uniform.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)

    # Compute power spectrum
    power = np.abs(A) ** 2

    if n_band is None:
        n_band = n // 4

    # Get magnitude and phase
    mag = np.abs(A)
    phase = np.angle(A)

    # Flatten: interpolate log-magnitude towards its mean
    mean_log_mag = np.mean(np.log(mag + 1e-30))
    log_mag = np.log(mag + 1e-30)
    new_log_mag = (1.0 - flatten_strength) * log_mag + flatten_strength * mean_log_mag
    new_mag = np.exp(new_log_mag)

    # Reconstruct with new magnitude, old phase
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)

    # Renormalize
    s = a_new.sum()
    if s > 1e-15:
        S = np.sqrt(2.0 * n)
        a_new = a_new * (S / s)

    return a_new


def perturb_structured(a, rng, method="block_swap"):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_out = a.copy()

    if method == "block_swap":
        # Swap two random blocks
        block_size = max(4, min(n // 32, 256))
        i1 = rng.integers(0, n - 2 * block_size)
        i2 = rng.integers(0, n - 2 * block_size)
        if abs(i1 - i2) > block_size:
            a_out[i1:i1 + block_size], a_out[i2:i2 + block_size] = \
                a_out[i2:i2 + block_size].copy(), a_out[i1:i1 + block_size].copy()
    elif method == "mirror_flip":
        # Flip a random sub-sequence
        length = max(4, min(n // 8, 1024))
        start = rng.integers(0, n - length)
        a_out[start:start + length] = a_out[start:start + length][::-1]
    elif method == "notch":
        # Zero out a random middle section and redistribute
        width = max(8, min(n // 16, 512))
        start = rng.integers(0, n - width)
        removed = a_out[start:start + width].sum()
        a_out[start:start + width] = 0.0
        # Redistribute to the rest
        rest = np.delete(a_out, range(start, start + width))
        rest = rest + removed / len(rest)
        a_out = np.concatenate([a_out[:start], rest[:n - start - width] if start < n - width else np.array([]),
                                a_out[start + width:start + width + max(0, n - start - width)]])
        # Simpler: just add back proportionally
        a_out[start:start + width] = 0.0
        mask = np.ones(n, dtype=bool)
        mask[start:start + width] = False
        a_out[mask] += removed / mask.sum()
    else:
        # Gaussian noise
        a_out = a_out + rng.normal(0, 0.01 * a.mean(), n)
        a_out = np.maximum(a_out, 0.0)

    s = a_out.sum()
    if s > 1e-15:
        a_out = a_out * (S / s)
    return a_out


def upsample(a, factor=2):
    """Upsample using sinc-like interpolation via FFT."""
    n = len(a)
    n_new = n * factor
    fsize_old = 1 << (n - 1).bit_length()
    fsize_new = 1 << (n_new - 1).bit_length()

    A_old = fft(a, fsize_new)
    # Zero-stuff in frequency domain (sinc interpolation)
    A_new = np.zeros(fsize_new, dtype=np.complex128)
    half = fsize_new // 2
    A_new[:half] = A_old[:min(half, fsize_old // 2)]
    A_new[half + 1:] = A_old[-min(half, fsize_old // 2 - 1):] if fsize_old > 2 else 0

    a_new = np.real(ifft(A_new))[:n_new]
    a_new = np.maximum(a_new, 0.0)
    return a_new

        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        pass

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.4 * np.cos(2 * np.pi * tt) + 0.1 * np.sin(2 * np.pi * tt)
        a_start = np.maximum(a_start, 0.0)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))
    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: FISTA with variance penalty (35% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: FISTA+Var at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    # Run with different variance weights
    configs = [
        (10.0, 0.006, 5000.0, 0.5, 0.0005, 1.0),
        (30.0, 0.004, 8000.0, 1.0, 0.0003, 0.7),
        (5.0, 0.008, 3000.0, 0.3, 0.001, 1.0),
    ]

    per_config = p1_budget / len(configs)
    for i, (bi, br, bm, vi, vd, ss) in enumerate(configs):
        if time.monotonic() >= deadline:
            break
        a1, c1 = run_fista_var(global_best_a, per_config, rng,
                               beta_init=bi, beta_rate=br, beta_max=bm,
                               var_init=vi, var_decay=vd, step_scale=ss)
        obj1 = compute_obj(a1)
        print("  Config %d: obj=%.6f (t=%.0fs)" % (i + 1, obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: Multi-restart with structured perturbations (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.25
        print("\n=== Phase 2: Multi-restart (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        methods = ["block_swap", "mirror_flip", "notch", "noise"]
        n_restarts = min(8, max(3, int(p2_budget / 20)))
        per_restart = p2_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() >= deadline:
                break
            method = methods[r % len(methods)]
            a_pert = perturb_structured(global_best_a, rng, method)

            a2, c2 = run_fista_var(a_pert, per_restart, rng,
                                   beta_init=20.0 + r * 5.0, beta_rate=0.005,
                                   beta_max=5000.0, var_init=0.8,
                                   var_decay=0.0008, step_scale=0.8)
            obj2 = compute_obj(a2)
            print("  Restart %d (%s): obj=%.6f" % (r + 1, method, obj2))
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_best_a = a2.copy()

    # === Phase 3: Spectral shaping + re-optimization (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Spectral shaping (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        for strength in [0.1, 0.2, 0.3, 0.5]:
            if time.monotonic() >= deadline:
                break
            a_spec = spectral_flatten(global_best_a, flatten_strength=strength)
            # Re-optimize from spectrally shaped starting point
            a3, c3 = run_fista_var(a_spec, p3_budget / 4.0, rng,
                                   beta_init=15.0, beta_rate=0.006,
                                   beta_max=5000.0, var_init=0.5,
                                   var_decay=0.0005, step_scale=1.0)
            obj3 = compute_obj(a3)
            print("  Strength %.1f: obj=%.6f" % (strength, obj3))
            if obj3 < global_best_obj:
                global_best_obj = obj3
                global_best_a = a3.copy()

    # === Phase 4: Coordinate descent polishing (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        n_best = len(global_best_a)
        print("\n=== Phase 4: Coord descent at n=%d (budget %.0fs) ===" % (n_best, p4_budget))
        t4 = time.monotonic()

        a4, c4 = run_coord_descent(global_best_a, p4_budget, rng)
        obj4 = compute_obj(a4)
        print("  Coord descent: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final intensive FISTA polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.7
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, c5 = run_fista_var(global_best_a, p5_budget, rng,
                               beta_init=100.0, beta_rate=0.003,
                               beta_max=30000.0, var_init=0.2,
                               var_decay=0.0001, step_scale=0.3)
        obj5 = compute_obj(a5)
        print("  Final: obj=%.6f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Bonus: Try slight size change ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(global_best_a) > 4096:
        # Try downsampled version
        n_ds = len(global_best_a) // 2
        a_ds = global_best_a[:n_ds * 2].reshape(n_ds, 2).mean(axis=1)
        S_ds = np.sqrt(2.0 * n_ds)
        a_ds = a_ds * (S_ds / a_ds.sum())
        obj_ds = compute_obj(a_ds)
        print("\n=== Bonus: Downsampled n=%d: obj=%.6f ===" % (n_ds, obj_ds))
        if obj_ds < global_best_obj:
            global_best_obj = obj_ds
            global_best_a = a_ds.copy()
            # Polish
            a_ds2, _ = run_fista_var(a_ds, remaining * 0.6, rng,
                                     beta_init=30.0, beta_rate=0.004,
                                     beta_max=8000.0, var_init=0.5,
                                     var_decay=0.0005, step_scale=1.0)
            obj_ds2 = compute_obj(a_ds2)
            if obj_ds2 < global_best_obj:
                global_best_obj = obj_ds2
                global_best_a = a_ds2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
