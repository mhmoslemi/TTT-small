Let me think about a specific new approach:

**Approach: Iterative Proportional Fitting (IPF) / Sinkhorn-like**

The idea: we want c[k] ≈ K for all k. This is like a matrix scaling problem.

Define a "cross-correlation matrix" M where M[i,j] = a[i]*a[j]. Then c[k] = sum_{i+j=k} M[i,j] (for the appropriate range).

We want all these "diagonal sums" to be equal. This is similar to making a matrix have equal row and column sums, but for diagonals.

An iterative approach:
1. Initialize a
2. Compute c = conv(a, a)
3. For each k, compute the desired scaling: r_k = K / c[k] where K = S^2/(2n-1)
4. Update: for each i, multiply a[i] by the geometric mean of r_k for all k where a[i] contributes
5. Repeat

But this is complicated because each a[i] contributes to multiple c[k].

A simpler variant:
1. Compute c = conv(a, a)
2. Find the peak lag k* = argmax(c)
3. For all pairs (i, j) with i+j = k*, reduce a[i]*a[j]
4. This can be done by: for each i, reduce a[i] by a factor proportional to a[k*-i]
5. Renormalize

This is essentially what the coordinate descent does, but we can make it more systematic.

**Alternative: Multiplicative Updates**

Inspired by matrix scaling:
1. Compute c = conv(a, a)
2. Compute target: K = S^2/(2n-1)
3. For each lag k, compute ratio r_k = K / c[k]
4. For each element a[i], it contributes to lags k = i, i+1, ..., i+n-1 (approximately)
5. Update a[i] *= (product of r_k for relevant k)^(1/(number of relevant k))
6. Renormalize to sum = S

This is like a "diagonal scaling" to equalize the lag sums.

Let me formalize: a[i] appears in c[k] for k from i to i+n-1 (when k-i ranges from 0 to n-1). More precisely, a[i] appears in c[k] with coefficient a[k-i] for k-i ∈ [0, n-1], i.e., k ∈ [i, i+n-1].

So: a[i] contributes to c[k] for k ∈ [i, i+n-1] (clipped to [0, 2n-2]).

The multiplicative update would be:
a[i] ← a[i] * (Π_{k=i}^{i+n-1} (K/c[k])^{w_{i,k}})^{1/Σw}

where w_{i,k} is the weight of a[i] in c[k], which is a[k-i]/c[k] (the fraction of c[k] contributed by a[i]).

import numpy as np

from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}"""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.maximum(v - v.mean(), 0.0) * (S / max(v.sum(), 1e-30))
    rho = int(np.max(np.where(valid)[0])) + 1
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve(a, N):
    """Compute full convolution a*a via FFT of size N."""
    A = fft(a, N)
    return np.real(ifft(A * A))[:2*len(a)-1]

def softmax_weights(c, T):
    """Compute softmax weights from c with temperature T."""
    c_shifted = c - c.max()
    p = np.exp(c_shifted / T)
    return p / p.sum()

def grad_softmax(a, c, T, n, N):
    """Compute gradient of T*log(sum(exp(c/T))) w.r.t. a."""
    p = softmax_weights(c, T)
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    # = 2 * (reverse(p) conv a)[i + 2n-2]
    p_rev = p[::-1]
    # Need FFT size >= (2n-1) + n - 1 = 3n-2
    N2 = 1 << (3*n - 2).bit_length()
    P_rev = fft(p_rev, N2)
    A2 = fft(a, N2)
    conv = np.real(ifft(P_rev * A2))
    grad = 2.0 * conv[2*n-2 : 2*n-2+n]
    return grad

def dirichlet_power(N, n):
    """Compute the target power spectrum (FT of box of length 2n-1)."""
    m = np.arange(N)
    # D[m] = |sum_{k=0}^{2n-2} exp(-2*pi*i*m*k/N)|^2
    # = |exp(-pi*i*(2n-2)*m/N) * sin(pi*(2n-1)*m/N) / sin(pi*m/N)|^2
    # = sin^2(pi*(2n-1)*m/N) / sin^2(pi*m/N)
    # Handle m=0 separately

    mask = m != 0
    # Also handle m = N/2 if N is even (sin(pi*m/N) = 0)
    sin_den = np.sin(np.pi * m[mask] / N)
    sin_num = np.sin(np.pi * (2*n-1) * m[mask] / N)

    with np.errstate(divide='ignore', invalid='ignore'):

def project_simplex(v, S):

    P_rev = fft(p_rev, N2)
    A2 = fft(a, N2)
    conv = np.real(ifft(P_rev * A2))
    return 2.0 * conv[2 * n - 2:2 * n - 2 + n]


def grad_softmax_direct(a, p, n):
    """Direct computation of gradient (for small top-k or verification)."""
    grad = np.zeros(n)
    for j in range(n):
        aj = a[j]
        if aj < 1e-15:
            continue
        # grad[i] += 2 * p[i+j] * a[j] for i = 0..n-1
        # p[i+j] is valid for i+j in [0, 2n-2], i.e., i in [-j, 2n-2-j]
        i_lo = max(0, 0)
        i_hi = min(n - 1, 2 * n - 2 - j)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * aj * p[i_lo + j:i_hi + 1 + j]
    return grad


def compute_c_fast(a, N):
    """Compute convolution, returning both c and the FFT of a."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def spectral_shape_dirichlet(a, A, n, N, S, alpha, target_power=None):
    """
    Spectral shaping: blend current power spectrum toward Dirichlet kernel target.
    Preserves phase of A.
    """
    if target_power is None:
        target_power = compute_dirichlet_target(N, n)
    P = np.abs(A) ** 2
    # Normalize target to have same total power
    P_sum = P.sum()
    T_sum = target_power.sum()
    P_new = (1 - alpha) * P + alpha * target_power * (P_sum / T_sum)
    # Ensure non-negative
    P_new = np.maximum(P_new, 0)
    # Reconstruct with preserved phase
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def compute_dirichlet_target(N, n):
    """
    Compute the target power spectrum proportional to the squared Dirichlet kernel.
    D[m] = sin^2(pi*(2n-1)*m/N) / sin^2(pi*m/N)
    This is |FFT(box of length 2n-1)|^2.
    """
    m = np.arange(N, dtype=np.float64)
    D = np.zeros(N)
    # m = 0
    D[0] = float(2 * n - 1) ** 2
    # m != 0
    mask = m > 0
    sin_den = np.sin(np.pi * m[mask] / N)
    sin_num = np.sin(np.pi * (2 * n - 1) * m[mask] / N)
    # Handle potential zero denominators (m = N/2 when N is even)
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = sin_num / sin_den
    ratio = np.where(np.abs(sin_den) < 1e-15, 0.0, ratio)
    D[mask] = ratio ** 2
    # Symmetrize (for real signal, D should be symmetric)
    D = 0.5 * (D + D[::-1])
    D[0] = float(2 * n - 1) ** 2
    return D


def coordinate_descent_step(a, c, n, N, S, top_k=20):
    """
    One round of coordinate descent: adjust top-contributing elements at peak lags.
    """
    a_new = a.copy()
    peak_lags = np.argsort(c)[::-1][:min(top_k, 5)]

    for lag in peak_lags:
        lag = int(lag)
        # Find top-contributing pairs at this lag
        i_lo = max(0, lag - n + 1)
        i_hi = min(lag, n - 1)
        if i_lo > i_hi:
            continue

        # Contribution of a[i] to c[lag] is a[i] * a[lag-i]
        # Try scaling a[i] by various factors
        best_improvement = 0
        best_change = None
        for i in range(i_lo, i_hi + 1):
            j = lag - i
            if j < 0 or j >= n or i == j:
                continue
            orig = a_new[i]
            if orig < 1e-15:
                continue
            # Try reducing a[i]
            for frac in [0.5, 0.7, 0.85]:
                delta = orig * (frac - 1.0)
                # Estimate new c[lag]
                new_c_lag = c[lag] + delta * (a_new[j] + (a_new[i] - delta) if i != j else a_new[i] - delta)
                # Actually: c[lag] = sum_i a[i]*a[lag-i]
                # If we change a[i] by delta, c[lag] changes by delta*a[lag-i] + a[i]*delta (if i != lag-i)
                # = delta * a[lag-i] + (if i != lag-i) delta * a[lag-i] = 2*delta*a[lag-i] (for i != lag-i)
                # = delta * a[i] (for i == lag-i, i.e., lag = 2*i)
                if i == j:
                    new_c_lag = c[lag] + 2 * orig * delta + delta * delta
                else:
                    new_c_lag = c[lag] + 2 * delta * a_new[j]

                # Check if this reduces the max (rough estimate)
                max_reduction = c.max() - max(new_c_lag, np.delete(c, lag).max() if len(c) > 1 else 0)
                if max_reduction > best_improvement:
                    best_improvement = max_reduction
                    best_change = (i, frac)

        if best_change is not None:
            i, frac = best_change
            a_new[i] *= frac

    return project_simplex(a_new, S)


def random_spectral_perturb(a, N, n, S, sigma, rng):
    """Add random noise to the spectrum and project back."""
    A = fft(a, N)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def structured_init(n, rng, method="smooth_bump"):
    """Create a structured initialization."""
    t = np.arange(n) / n
    if method == "smooth_bump":
        # Gaussian-like bump, asymmetric
        center = 0.4 + 0.2 * rng.random()
        width = 0.15 + 0.15 * rng.random()
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.3 * np.exp(-0.5 * ((t - (1 - center)) / (width * 1.5)) ** 2)
    elif method == "cosine_mod":
        freq = rng.uniform(1, 5)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    elif method == "piecewise_linear":
        n_pieces = 8 + 16 * int(rng.random())
        knots = np.sort(rng.random(n_pieces))
        heights = 0.5 + 0.8 * rng.random(n_pieces)
        a = np.interp(t, [0] + list(knots) + [1], [heights[0]] + list(heights) + [heights[-1]])
    elif method == "multi_bump":
        n_bumps = 2 + int(3 * rng.random())
        a = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.random()
            w = 0.1 + 0.2 * rng.random()
            amp = 0.5 + rng.random()
            a += amp * np.exp(-0.5 * ((t - c) / w) ** 2)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)

    a = np.clip(a, 0.01, 5.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_from_c(c):
    """Compute the objective from the convolution (assuming S = sqrt(2n))."""
    n = (len(c) + 1) // 2
    return c.max()


def main_search(n, S, N, rng, budget, target_power=None):
    """
    Main optimization loop for a given sequence length n.
    Uses soft-max gradient descent + spectral shaping + coordinate descent.
    """
    if target_power is None:
        target_power = compute_dirichlet_target(N, n)

    N2 = 1 << (3 * n - 2).bit_length()  # For gradient computation

    # Initialize from multiple starts
    inits = []
    # Try previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(safe_normalize(given, n))
        elif given.size > 0:
            # Resample to length n
            from numpy import interp
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, n)
            resampled = interp(t_new, t_old, np.clip(given, 0, 1000))
            inits.append(safe_normalize(resampled, n))
    except Exception:
        pass

    for method in ["smooth_bump", "cosine_mod", "piecewise_linear", "multi_bump"]:
        inits.append(safe_normalize(structured_init(n, rng, method), n))
    inits.append(safe_normalize(0.5 + 0.5 * rng.random(n), n))

    # Evaluate initializations
    best_a = None
    best_score = np.inf
    for a_init in inits:
        c = compute_c(a_init, N)[0]
        sc = c.max()
        if sc < best_score:
            best_score = sc
            best_a = a_init.copy()

    print("    Initial best: %.6f" % best_score)

    a = best_a.copy()
    c, A = compute_c_fast(a, N)
    cur_score = c.max()

    start = time.monotonic()
    it = 0
    stall = 0
    last_improve = start
    T_init = max(0.5, cur_score * 0.1)  # Initial temperature
    T_min = 0.01
    T_decay = 0.9995
    T = T_init

    # Barzilai-Borwein state
    prev_grad = None
    prev_a = None
    lr = 1.0 / (n * 0.1)  # Initial learning rate

    while time.monotonic() - start < budget:
        it += 1

        # Anneal temperature
        T = max(T_min, T_init * (T_decay ** it))

        # Compute current state
        c, A = compute_c_fast(a, N)
        cur_score = c.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            stall = 0
            if it % 200 == 0:
                print("    iter %d: %.6f (T=%.3f)" % (it, best_score, T))
        else:
            stall += 1

        # --- Soft-max gradient descent step ---
        p = softmax_weights(c, T)
        grad = grad_softmax_fft(a, p, n, N2)

        # Adaptive step size (simple BB-like)
        if prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_norm_sq = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if s_norm_sq > 1e-15 and abs(sy) > 1e-15:
                lr_bb = s_norm_sq / abs(sy)
                lr = 0.5 * lr + 0.5 * np.clip(lr_bb, 0.1 * lr, 10 * lr)
        prev_a = a.copy()
        prev_grad = grad.copy()

        # Gradient step with projection
        a_try = project_simplex(a - lr * grad, S)

        # Check if it improved
        c_try = compute_c(a_try, N)
        score_try = c_try.max()

        if score_try < cur_score:
            a = a_try
            c, A = compute_c_fast(a, N)
        else:
            # Try smaller step
            a_try2 = project_simplex(a - 0.5 * lr * grad, S)
            c_try2 = compute_c(a_try2, N)
            if c_try2.max() < cur_score:
                a = a_try2
                c, A = compute_c_fast(a, N)
            else:
                # Reduce learning rate
                lr *= 0.9

        # --- Spectral shaping (every 10 iterations) ---
        if it % 10 == 0:
            alpha_spec = min(0.3, 0.05 + 0.01 * (it // 100))
            a_spec = spectral_shape_dirichlet(a, A, n, N, S, alpha_spec, target_power)
            c_spec = compute_c(a_spec, N)
            if c_spec.max() < c.max():
                a = a_spec
                c, A = compute_c_fast(a, N)

        # --- Coordinate descent (every 20 iterations) ---
        if it % 20 == 0:
            a_cd = coordinate_descent_step(a, c, n, N, S, top_k=10)
            c_cd = compute_c(a_cd, N)
            if c_cd.max() < c.max():
                a = a_cd
                c, A = compute_c_fast(a, N)

        # --- Random perturbation (when stalled) ---
        if stall > 100 and it % 50 == 0:
            sigma = 0.02 + 0.03 * (stall // 100)
            a_pert = random_spectral_perturb(a, N, n, S, sigma, rng)
            c_pert = compute_c(a_pert, N)
            if c_pert.max() < best_score:
                a = a_pert
                c, A = compute_c_fast(a, N)
            elif rng.random() < min(0.3, 0.1 * (stall // 100)):
                a = a_pert
                c, A = compute_c_fast(a, N)
            stall = 0

        # Larger perturbation when very stalled
        if stall > 300 and it % 100 == 0:
            # Try a completely new initialization blended with current
            a_new_init = safe_normalize(structured_init(n, rng, "smooth_bump"), n)
            a_blend = project_simplex(0.5 * a + 0.5 * a_new_init, S)
            c_blend = compute_c(a_blend, N)
            if c_blend.max() < best_score:
                a = a_blend
                c, A = compute_c_fast(a, N)
            stall = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses soft-max gradient descent, spectral shaping, and coordinate descent
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Try multiple lengths
    lengths = [4096, 8192, 12288, 16384]

    global_best = np.inf
    global_a = None

    # Allocate time: give more time to longer sequences
    # But start with 8192 (known good length)
    time_alloc = {
        8192: 0.40 * (deadline - time.monotonic()),
        16384: 0.25 * (budget_s - 20),
        4096: 0.15 * (budget_s - 20),
        12288: 0.15 * (budget_s - 20),
    }

    # Process in order: 8192 first, then 16384, then others
    for n in [8192, 16384, 4096, 12288]:
        if time.monotonic() > deadline - 30:
            break

        N = 1 << (2 * n - 1).bit_length()
        S = np.sqrt(2.0 * float(n))
        remaining = min(deadline - time.monotonic(), time_alloc.get(n, 60))

        if remaining < 10:
            continue

        print("  Starting n=%d, N=%d, budget=%.0fs" % (n, N, remaining))

        # Precompute Dirichlet target
        target_power = compute_dirichlet_target(N, n)

        # Run main search
        t0 = time.monotonic()
        a_best, sc_best = main_search(n, S, N, rng, remaining, target_power)

        elapsed = time.monotonic() - t0
        print("  n=%d: score=%.6f (time=%.0fs)" % (n, sc_best, elapsed))

        if sc_best < global_best:
            global_best = sc_best
            global_a = a_best.copy()
            print("  New global best: %.6f" % global_best)

    # Final polish on the best solution
    if global_a is not None and time.monotonic() < deadline - 10:
        n = len(global_a)
        N = 1 << (2 * n - 1).bit_length()
        S = np.sqrt(2.0 * float(n))
        remaining = deadline - time.monotonic()

        print("  Final polish: n=%d, remaining=%.0fs" % (n, remaining))

        target_power = compute_dirichlet_target(N, n)
        N2 = 1 << (3 * n - 2).bit_length()

        a = global_a.copy()

        # Rapid coordinate descent + gradient steps
        it = 0
        while time.monotonic() < deadline - 5:
            it += 1
            c, A = compute_c_fast(a, N)
            cur = c.max()

            # Gradient step with very small T (focus on peaks)
            T = max(0.05, 0.5 * (1 - it / 1000))
            p = softmax_weights(c, T)
            grad = grad_softmax_fft(a, p, n, N2)
            lr = S / (np.linalg.norm(grad) + 1e-10) * 0.5
            a_try = project_simplex(a - lr * grad, S)
            c_try = compute_c(a_try, N)
            if c_try.max() < cur:
                a = a_try
            else:
                # Coordinate descent
                a_cd = coordinate_descent_step(a, c, n, N, S, top_k=15)
                c_cd = compute_c(a_cd, N)
                if c_cd.max() < cur:
                    a = a_cd

            if a.max() < global_best:
                pass  # a.max() is not the score, c.max() is
            if c.max() < global_best:
                global_best = c.max()
                global_a = a.copy()

    # --- Verification and cleanup ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            print("  Warning: sum too small, using fallback")
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
    else:
        print("  No valid solution found, using fallback")
        final_list = [1.0] * 256

    # Verify
    ver = evaluate_sequence(final_list)
    print("Final verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        # Fallback: uniform sequence
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback verification: %.6f" % ver)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(final_list), global_best))
    return final_list
