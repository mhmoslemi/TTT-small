"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using Adam optimizer
with LSE-smoothed max, beta annealing, palindromic constraint option,
multi-length exploration, and adaptive peak targeting.
Key innovation: per-coordinate adaptive learning rates (Adam) combined with
log-sum-exp smoothing of the max, which converges much faster than the
previous fixed-step subgradient method.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def enforce_palindromic(a):
    """Make a palindromic by averaging with its reverse."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def lse_gradient(a, beta):
    """
    Compute gradient of LSE approximation to max(conv(a,a)).
    Returns (gradient, lse_value, max_value).
    """
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Numerical stability
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    # Gradient of LSE w.r.t. a: 2 * (w cross-correlate a)
    # grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val, cm


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def adam_lse_optimize(a_init, n, S, max_time, rng, palindromic=False,
                      beta_start=5.0, beta_end=300.0, lr_start=0.01,
                      lr_end=0.0001, beta0=0.9, beta1=0.999, eps=1e-8):
    """
    Adam optimizer on LSE-smoothed max-convolution objective.
    Minimizes LSE_beta(conv(a,a)) subject to a >= 0, sum(a) = S.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        t += 1
        elapsed_frac = (time.monotonic() - start) / max_time
        # Anneal beta and learning rate
        frac = min(1.0, elapsed_frac * 1.3)  # Ramp up faster than linear
        beta = beta_start + (beta_end - beta_start) * (frac ** 1.5)
        lr = lr_start * (lr_end / lr_start) ** frac

        if palindromic:
            a = enforce_palindromic(a)

        grad, lse_val, max_val = lse_gradient(a, beta)

        if palindromic:
            grad = enforce_palindromic(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adam update
        m = beta0 * m + (1 - beta0) * grad
        v = beta1 * v + (1 - beta1) * (grad * grad)
        m_hat = m / (1 - beta0 ** t)
        v_hat = v / (1 - beta1 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if palindromic:
            a_new = enforce_palindromic(a_new)

        a_new = project_simplex(a_new, S)

        # Check objective every few steps (full conv is expensive)
        if t % 50 == 0 or t < 10:
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / S ** 2
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Periodic perturbation to escape local minima
            if no_improve > 500:
                noise_scale = 0.005 * S / np.sqrt(n)
                a_new = project_simplex(a_new + rng.normal(0, noise_scale, n), S)
                if palindromic:
                    a_new = project_simplex(enforce_palindromic(a_new), S)
                no_improve = 0
                # Reset momentum
                m = np.zeros(n)
                v = np.zeros(n)

        a = a_new

        if t % 2000 == 0:
            print("    adam it=%d, beta=%.1f, lr=%.5f, obj=%.6f" % (t, beta, lr, best_obj))

    return best_a, best_obj


def conjugate_gradient_smooth(a_init, n, S, max_time, rng, palindromic=False,
                              beta=50.0):
    """
    Nonlinear conjugate gradient on LSE-smoothed objective.
    Uses Polak-Ribiere momentum.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()

    grad, _, _ = lse_gradient(a, beta)
    d = -grad
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        if palindromic:
            a = enforce_palindromic(a)
            d = enforce_palindromic(d)

        # Line search (backtracking)
        alpha = 0.1
        c_cur = fft_convolve_self(a)
        f_cur = c_cur.max()

        improved = False
        for _ in range(20):
            a_try = a + alpha * d
            if palindromic:
                a_try = enforce_palindromic(a_try)
            a_try = project_simplex(a_try, S)
            c_try = fft_convolve_self(a_try)
            f_try = c_try.max()
            if f_try < f_cur - 1e-10:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Reinitialize direction
            grad, _, _ = lse_gradient(a, beta)
            d = -grad
            if palindromic:
                d = enforce_palindromic(d)
            no_improve += 1
            if no_improve > 200:
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                if palindromic:
                    a = project_simplex(enforce_palindromic(a), S)
                no_improve = 0
                continue
        else:
            no_improve = 0

        c_new = fft_convolve_self(a)
        obj_new = 2.0 * n * c_new.max() / S ** 2
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a.copy()

        # Compute new gradient and update direction
        grad_new, _, _ = lse_gradient(a, beta)
        if palindromic:
            grad_new = enforce_palindromic(grad_new)

        # Polak-Ribiere
        dg = np.dot(grad_new - grad, d)
        gn = np.dot(grad_new, grad_new)
        beta_pr = max(0.0, dg / max(gn, 1e-15))
        d = -grad_new + beta_pr * d

        grad = grad_new

        if it % 2000 == 0:
            print("    cg it=%d, obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def peak_shaving(a_init, n, S, max_time, rng, palindromic=False):
    """
    Direct peak shaving: iteratively find the peak and reduce it
    by adjusting the contributing elements.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    no_improve = 0
    step = S / n * 0.1

    while time.monotonic() - start < max_time:
        it += 1
        if palindromic:
            a = enforce_palindromic(a)

        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S ** 2

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            if palindromic:
                a = project_simplex(enforce_palindromic(a), S)
            no_improve = 0
            step = S / n * 0.05
            continue

        # Get top-k subgradient (k increases over time)
        k = min(1 + it // 500, 32)
        grad = topk_subgradient(a, c, k)
        if palindromic:
            grad = enforce_palindromic(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with backtracking
        a_try = project_simplex(a - step * grad / gnorm, S)
        if palindromic:
            a_try = project_simplex(enforce_palindromic(a_try), S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S ** 2

        if obj_try < cur_obj:
            a = a_try
            step = min(step * 1.03, S / n)
        else:
            step *= 0.7

        if it % 1000 == 0:
            print("    peakshave it=%d, obj=%.6f, k=%d" % (it, best_obj, k))

    return best_a, best_obj

                    best_local_a = a_norm.copy()

                if real_score < global_best:
                    global_best = real_score
                    global_a = a_norm.copy()
                    global_n = n
                    print("    [%s, pal=%s] n=%d: obj=%.6f (total elapsed %.0fs)" % (
                        name, pal, n, global_best, time.monotonic() - t0))

    print("=== After Phase 1: best=%.6f, n=%d, elapsed=%.0fs ===" % (
        global_best, global_n, time.monotonic() - t0))

    # Phase 2: Refined optimization on best candidate (30% of time)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.35
    print("=== Phase 2: Refined optimization (time=%.0fs) ===" % phase2_time)

    if global_a is not None and phase2_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # 2a: Adam with higher beta range
        t2a = phase2_time * 0.4
        a_opt, score = adam_lse_optimize(
            global_a, n_best, S_best, t2a, rng, palindromic=False,
            beta_start=50.0, beta_end=500.0,
            lr_start=0.001, lr_end=0.00001
        )
        a_norm = np.maximum(a_opt, 0.0)
        sa = a_norm.sum()
        if sa > 1e-15:
            a_norm = a_norm * (S_best / sa)
            real_score = compute_objective(a_norm)
            if real_score < global_best:
                global_best = real_score
                global_a = a_norm.copy()
                print("  After high-beta Adam: %.6f" % global_best)

        # 2b: Conjugate gradient
        t2b = phase2_time * 0.3
        if time.monotonic() < deadline - 30:
            a_opt, score = conjugate_gradient_smooth(
                global_a, n_best, S_best, t2b, rng, palindromic=False, beta=100.0
            )
            a_norm = np.maximum(a_opt, 0.0)
            sa = a_norm.sum()
            if sa > 1e-15:
                a_norm = a_norm * (S_best / sa)
                real_score = compute_objective(a_norm)
                if real_score < global_best:
                    global_best = real_score
                    global_a = a_norm.copy()
                    print("  After CG: %.6f" % global_best)

        # 2c: Peak shaving
        t2c = phase2_time * 0.3
        if time.monotonic() < deadline - 30:
            a_opt, score = peak_shaving(
                global_a, n_best, S_best, t2c, rng, palindromic=False
            )
            a_norm = np.maximum(a_opt, 0.0)
            sa = a_norm.sum()
            if sa > 1e-15:
                a_norm = a_norm * (S_best / sa)
                real_score = compute_objective(a_norm)
                if real_score < global_best:
                    global_best = real_score
                    global_a = a_norm.copy()
                    print("  After peak shaving: %.6f" % global_best)

    # Phase 3: Intensive micro-optimization (15% of time)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.35
    print("=== Phase 3: Micro-optimization (time=%.0fs) ===" % phase3_time)

    if global_a is not None and phase3_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_micro = global_a.copy()
        micro_start = time.monotonic()
        step = 0.005 * S_best / np.sqrt(n_best)
        it = 0
        best_micro = global_best
        c_micro = fft_convolve_self(a_micro)

        while time.monotonic() - micro_start < phase3_time:
            it += 1
            cur_obj = 2.0 * n_best * c_micro.max() / S_best ** 2

            if cur_obj < best_micro - 1e-10:
                best_micro = cur_obj

            # Very small steps with exact peak subgradient
            k = min(1 + it // 100, 8)
            grad = topk_subgradient(a_micro, c_micro, k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            a_try = project_simplex(a_micro - step * grad / gnorm, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best ** 2

            if obj_try < cur_obj:
                a_micro = a_try
                c_micro = c_try
                step = min(step * 1.01, 0.05 * S_best / np.sqrt(n_best))
            else:
                step *= 0.8
                if step < 1e-10:
                    # Perturb
                    noise_scale = 0.001 * S_best / np.sqrt(n_best)
                    a_micro = project_simplex(
                        a_micro + rng.normal(0, noise_scale, n_best), S_best)
                    c_micro = fft_convolve_self(a_micro)
                    step = 0.002 * S_best / np.sqrt(n_best)

            if it % 2000 == 0:
                print("    micro it=%d, obj=%.6f" % (it, 2.0 * n_best * c_micro.max() / S_best ** 2))

        a_micro = np.maximum(a_micro, 0.0)
        sa = a_micro.sum()
        if sa > 1e-15:
            a_micro = a_micro * (S_best / sa)
            s_micro = compute_objective(a_micro)
            if s_micro < global_best:
                global_best = s_micro
                global_a = a_micro.copy()
                print("  After micro: %.6f" % global_best)

    # Phase 4: Try a different length if time remains
    remaining = deadline - time.monotonic()
    print("=== Phase 4: Alternative lengths (remaining=%.0fs) ===" % remaining)

    if remaining > 60:
        # Try a length we haven't fully explored
        tried = {len(global_a)} if global_a is not None else set()
        alt_lengths = [l for l in [2048, 4096, 8192, 16384, 32768] if l not in tried]

        for n_alt in alt_lengths[:2]:
            if time.monotonic() > deadline - 30:
                break
            S_alt = np.sqrt(2.0 * n_alt)
            t_alt = min(60.0, deadline - time.monotonic() - 20)
            if t_alt < 10:
                break

            # Quick Adam run
            a_init_alt = generate_initializations(n_alt, rng)[0]
            a_opt, score = adam_lse_optimize(
                a_init_alt, n_alt, S_alt, t_alt, rng,
                palindromic=False, beta_start=10.0, beta_end=150.0,
                lr_start=0.003, lr_end=0.0001
            )
            a_norm = np.maximum(a_opt, 0.0)
            sa = a_norm.sum()
            if sa > 1e-15:
                a_norm = a_norm * (S_alt / sa)
                real_score = compute_objective(a_norm)
                if real_score < global_best:
                    global_best = real_score
                    global_a = a_norm.copy()
                    global_n = n_alt
                    print("  Alt length n=%d: obj=%.6f" % (n_alt, global_best))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification via evaluate_sequence: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    result = [float(x) for x in global_a]
    print("Returning sequence of length %d, obj=%.6f" % (len(result), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return result
