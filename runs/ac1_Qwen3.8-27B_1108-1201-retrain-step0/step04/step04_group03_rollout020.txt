"""
Minimizes 2*n*max(f*f)/sum(f)^2 over non-negative sequences using
multi-scale FISTA with adaptive LSE smoothing, multiplicative updates,
and L-BFGS polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as sp_minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, target_sum):
    """Clip to [0, 1000] and rescale to target sum."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * target_sum / n
    return a * (target_sum / s)


def lse_value(c, beta):
    """Log-sum-exp value: (1/beta)*log(sum(exp(beta*c)))."""
    cm = c.max()
    return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta * (a*a)) with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_from_conv(c, n, s):
    """Compute objective from convolution, n, and sum."""
    return 2.0 * n * c.max() / (s * s)


def multiplicative_update(a, n, S, c, beta=50.0, lr=0.05):
    """Multiplicative (mirror descent) update."""
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w = w / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    # Multiplicative update
    a_new = a * np.exp(-lr * rep / max(a.max(), 1e-10))
    a_new = np.maximum(a_new, 1e-30)
    return project_simplex(a_new, S)


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002,
              beta_max=5000.0, use_mult=False, mult_freq=200, mult_lr=0.03):
    """FISTA with LSE smoothing and optional multiplicative updates."""
    a = normalize_sum(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        s = y.sum()
        cur_obj = objective_from_conv(c, n, s)
        if cur_obj < best:
            best = cur_obj
            best_a = y.copy()
            last_improve = it

        # Adaptive step based on beta
        step = 1.0 / (2.0 * beta)

        # Compute gradient
        grad = lse_gradient(y, c, beta)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic multiplicative update
        if use_mult and it % mult_freq == 0:
            c_check = fft_convolve_self(y)
            y = multiplicative_update(y, n, S, c_check, beta=40.0, lr=mult_lr)
            t_mom = 1.0

        # Periodic restart if stuck
        if it - last_improve > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it

        # Very periodic larger noise
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0

    return best_a, best


def run_lbfgs_polish(a_init, n, S, max_time, beta=200.0):
    """Polish with L-BFGS-B on the LSE-smoothed objective."""
    a = normalize_sum(a_init.copy(), S)
    start = time.monotonic()
    
    def obj_and_grad(z):
        # Work directly with the vector (no reparameterization)
        # Add penalty for constraint violation
        f = np.maximum(z, 0.0)
        s = f.sum()
        if s < 1e-15:
            return 1e10, np.zeros_like(z)
        c = fft_convolve_self(f)
        val = lse_value(c, beta)
        # Penalty for not matching target sum
        penalty = 1e6 * (s - S) ** 2
        total = val + penalty
        grad = lse_gradient(f, c, beta)
        grad_penalty = 2e6 * (s - S) * np.ones_like(z)
        # Only gradient for positive components
        grad = np.where(z > 0, grad + grad_penalty, 0.0)
        return total, grad

    best_a = a.copy()
    best_val = obj_and_grad(a)[0]
    
    result = sp_minimize(
        obj_and_grad, a, method='L-BFGS-B',
        options={'maxiter': 500, 'maxfun': 1000, 'ftol': 1e-12, 'gtol': 1e-8,
                 'maxls': 50}
    )
    
    if result.success or result.fun < best_val:
        f = np.maximum(result.x, 0.0)
        f = normalize_sum(f, S)
        c = fft_convolve_self(f)
        obj = objective_from_conv(c, n, f.sum())
        if obj < best_val:
            best_a = f
    
    return best_a


def upscale(a, new_n):
    """Upsample sequence to new length via interpolation."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)


def cosine_init(n, freq=1.0, phase=0.0, base=0.5, amp=0.5):
    """Cosine initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    return np.maximum(base + amp * np.cos(2 * np.pi * freq * t + phase), 0.01)


def random_init(n, rng):
    """Random initialization."""
    return np.maximum(rng.random(n) * 0.5 + 0.5, 0.01)


def block_init(n, rng, nblocks=10):
    """Block structure initialization."""
    heights = rng.uniform(0.3, 1.5, nblocks)
    seq = np.repeat(heights, n // nblocks + 1)[:n]
    return np.maximum(seq, 0.01)

"""
Minimizes 2*n*max(f*f)/sum(f)^2 over non-negative sequences using
multi-scale FISTA with adaptive LSE smoothing, multiplicative mirror-descent
updates, L-BFGS polishing, and multi-restart exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as sp_minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, target_sum):
    """Clip to [0,1000] and rescale to target sum."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * target_sum / n
    return a * (target_sum / s)


def lse_value(c, beta):
    """Smooth approx to max: (1/beta)*log(sum(exp(beta*c)))."""
    cm = c.max()
    return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta*(a*a)) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_from_conv(c, n, s):
    """Compute 2*n*max(c)/s^2."""
    return 2.0 * n * c.max() / (s * s)


def mult_update(a, n, S, beta=50.0, lr=0.05):
    """Multiplicative mirror-descent step."""
    c = fft_convolve_self(a)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return a.copy()
    w = w / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    a_new = a * np.exp(-lr * rep / max(a.max(), 1e-10))
    a_new = np.maximum(a_new, 1e-30)
    return project_simplex(a_new, S)


def run_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002,
              beta_max=5000.0, use_mult=False, mult_freq=200, mult_lr=0.04,
              stuck_threshold=2000, noise_base=0.003):
    """FISTA with LSE smoothing and optional multiplicative updates."""
    a = normalize_sum(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_obj = objective_from_conv(c, n, y.sum())
        if cur_obj < best:
            best = cur_obj
            best_a = a.copy()
            last_improve = it

        step = 1.0 / (2.0 * beta)
        grad = lse_gradient(y, c, beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if use_mult and it % mult_freq == 0:
            a = mult_update(a, n, S, beta=40.0, lr=mult_lr)
            y = a.copy()
            t_mom = 1.0

        if it - last_improve > stuck_threshold:
            noise_scale = noise_base * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t_mom = 1.0
            last_improve = it

        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.01 * S / np.sqrt(n) * decay
            if noise_scale > 1e-9:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                y = a.copy()
                t_mom = 1.0

    return best_a, best


def _lbfgs_obj_grad(z, n, S, beta):
    """Objective and gradient for L-BFGS (top-level, no closure)."""
    f = np.maximum(z, 0.0)
    s = f.sum()
    if s < 1e-15:
        return 1e10, np.zeros_like(z)
    c = fft_convolve_self(f)
    val = lse_value(c, beta)
    penalty = 1e6 * (s - S) ** 2
    total = val + penalty
    grad = lse_gradient(f, c, beta)
    grad_pen = 2e6 * (s - S)
    grad = grad + grad_pen
    grad = np.where(z > 0, grad, 0.0)
    return total, grad


def run_lbfgs_polish(a_init, n, S, max_time, beta=500.0):
    """Polish with L-BFGS-B on LSE-smoothed objective."""
    a = normalize_sum(a_init.copy(), S)
    start = time.monotonic()

    c_init = fft_convolve_self(a)
    best_obj = objective_from_conv(c_init, n, a.sum())
    best_a = a.copy()

    try:
        result = sp_minimize(
            _lbfgs_obj_grad, a.copy(), args=(n, S, beta),
            method='L-BFGS-B', jac=True,
            options={'maxiter': 300, 'maxfun': 500,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40}
        )
        if time.monotonic() - start > max_time:
            pass
        else:
            f_new = np.maximum(result.x, 0.0)
            f_new = normalize_sum(f_new, S)
            c_new = fft_convolve_self(f_new)
            obj_new = objective_from_conv(c_new, n, f_new.sum())
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = f_new
    except Exception as e:
        print("  L-BFGS error: %s" % str(e))

    return best_a, best_obj


def upscale_seq(a, new_n):
    """Upsample via linear interpolation."""
    n = len(a)
    if new_n <= n:
        return a[:new_n].copy()
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(t_new, t_old, a)


def cosine_init_seq(n, freq=1.0, phase=0.0, base=0.54, amp=0.46):
    """Cosine-shape initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    return np.maximum(base + amp * np.cos(2 * np.pi * freq * t + phase), 0.01)


def random_init_seq(n, rng):
    """Random initialization."""
    return np.maximum(rng.random(n) * 0.4 + 0.6, 0.01)


def block_init_seq(n, rng, nblocks=13):
    """Block structure initialization."""
    heights = rng.uniform(0.3, 1.7, nblocks)
    seq = np.repeat(heights, n // nblocks + 1)[:n]
    return np.maximum(seq, 0.01)


def targeted_reduction(a, n, S, top_k=5, strength=0.5):
    """Reduce contributions at the top-k convolution lags."""
    c = fft_convolve_self(a)
    peak_idx = np.argsort(c)[::-1][:top_k]
    a_new = a.copy()
    for k in peak_idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        repulsion = 0.0
        for j in js:
            p = k - j
            if 0 <= p < n:
                repulsion += a[j] * a[p]
        if repulsion > 1e-15:
            for j in js:
                p = k - j
                if 0 <= p < n:
                    a_new[j] *= (1.0 - strength * a[j] * a[p] / repulsion)
                    a_new[p] *= (1.0 - strength * 0.5 * a[j] * a[p] / repulsion)
    a_new = np.maximum(a_new, 0.0)
    return project_simplex(a_new, S)
