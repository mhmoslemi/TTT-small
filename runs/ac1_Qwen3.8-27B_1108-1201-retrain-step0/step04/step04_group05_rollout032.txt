"""
Minimize 2*n*max_conv/sum^2 using multi-n exploration + adaptive L-BFGS with 
progressive LSE smoothing + top-k focused gradient + LP-inspired global moves.

Key improvements over previous:
1. Multi-n exploration (4096, 8192, 12288) to find optimal length
2. Progressive beta ramping with higher max for precision
3. Top-k focused gradient in late stages (faster convergence on active peaks)
4. LP-inspired direction as periodic global move
5. Structure-aware random mutations for escaping local minima
6. Aggressive final polish phase
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad_val(a, N, beta, c):
    """Compute LSE-smoothed max value and gradient."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        lse_val = c_max
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        lse_val = c_max + np.log(total) / beta
        w = exp_bc / total
    fsize = N
    w_full = np.zeros(fsize)
    w_full[:len(w)] = w
    W = fft(w_full)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val


def topk_grad(a, N, c, k):
    """Gradient of mean of top-k convolution values."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    m_arr = np.arange(N)
    W = np.zeros(N, dtype=np.complex128)
    for j in range(len(top_idx)):
        W += top_vals[j] * np.exp(-2j * np.pi * int(top_idx[j]) * m_arr / N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, top_idx


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def optimize_lbfgs(a_init, n, S, N, max_time, rng, beta_init=10.0,
                   beta_max=300.0, m_lbfgs=12, use_topk_after=300, topk_size=20):

    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptively increase beta

            c = compute_conv(a, N)

                best_cmax = c_max
                best_a = a.copy()
            if it >= use_topk_after:

            else:

        # Trim history

            y_hist = y_hist[-m_lbfgs:]

        dir_norm = np.linalg.norm(direction)

            break

        a_new = a

        cmax_new = c_max

                c_new = c_ls
                cmax_new = cm_ls

            y_hist.append(y_k)

                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c = c_new
        c_max = cmax_new

            else:

    return best_a, best_cmax


def lp_inspired_direction(a, n, N, top_k=300):
    """
    Compute an LP-inspired direction: find g that maximizes sum(g)
    subject to a*g <= max(a*a) and g >= 0 (approximately).
    Returns a direction vector.
    """
    c = compute_conv(a, N)
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)

    # Approximate best response using iterative thresholding
    # Start with g = 1
    g = np.ones(n)
    S_a = a.sum()
    g = g * S_a / n

    for _ in range(30):
        # Compute a*g (cross-convolution)
        ag = np.convolve(a, g, mode='full')
        # Violations
        violations = np.maximum(0, ag - M)
        if violations.max() < 1e-10:
            break
        # Gradient of violations w.r.t. g
        grad_v = np.convolve(a, violations, mode='full')[:n]
        # Scale
        gv = g.sum()
        if gv < 1e-15:
            break
        step = 0.1 * gv / max(np.linalg.norm(grad_v), 1e-10)
        g = np.maximum(0, g - step * grad_v)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * S_a / g_sum

    # Direction: move toward g
    S_a = a.sum()
    g = safe_normalize(g, n, S_a)
    direction = g - a
    return direction


def make_init_rand(n, rng):
    return rng.random(n) + 0.02


def make_init_smooth(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt**2))
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_tapered(n, rng):
    """Tapered edges to reduce edge effects."""
    tt = np.linspace(0, 1, n)
    taper = np.sin(np.pi * tt) ** 0.5
    a = taper * (1.0 + 0.2 * rng.standard_normal(n))
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    """Fourier-based: random magnitude, random phase."""
    P = 1.0 + rng.exponential(0.5, N)
    P[0] = n
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01
