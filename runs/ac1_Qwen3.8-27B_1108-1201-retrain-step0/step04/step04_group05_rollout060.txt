Minimize 2*n*max_conv/sum^2 using Nesterov-accelerated exact subgradient 
with adaptive active set, spectral rebalancing, and coordinate descent polish.

Key differences from previous approach:
1. Exact subgradient (no LSE bias) with adaptive weighting.
2. Polyak step size for guaranteed convergence rate.
3. Spectral rebalancing that directly targets C-flatness.
4. Coordinate descent with 1D subproblem solving for final polish.
5. Multi-timescale: fast Nesterov + slow spectral + final coordinate descent.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute a*a (full self-convolution) via FFT. Returns C of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    C = np.real(ifft(A * A))[:2 * n - 1]
    return C


def compute_subgradient(a, N, active_weights):
    """Compute subgradient of max(C) using weighted active set.
    active_weights: 1D array of length 2n-1 with weights at active lags.
    Returns gradient vector of length n.
    """
    n = len(a)
    W = fft(active_weights, N)
    A = fft(a, N)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g


def nesterov_exact_subgrad(a_init, n, S, N, max_time, rng, 
                             K_init=50, K_max=500, 
                             noise_scale=0.005,

    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    y = a.copy()  # Nesterov lookahead
    t = 1.0
    best_a = a.copy()
    best_val = np.inf
    
    start_time = time.monotonic()
    it = 0
    f_history = []
    
    while time.monotonic() - start_time < max_time:
        it += 1

        C = compute_conv(y, N)

        if total_excess < 1e-30:

        else:

        alpha = gap / (g_norm * g_norm)
        # Cap the step

"""
Minimize 2*n*max_conv/sum^2 using Nesterov-accelerated exact subgradient with 
adaptive top-K weighting, spectral anti-peak shaping, and coordinate descent polish.

Key innovations over previous approach:
1. Exact subgradient with adaptive peak-emphasis weights (no LSE bias).
2. Multi-timescale: fast Nesterov + periodic spectral reshaping + coordinate polish.
3. Non-monotone acceptance to escape shallow local minima.
4. Frequency-domain peak suppression targeting specific C(lag) contributors.
5. Cyclic coordinate descent with 1D minimax subproblem for final refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute a*a (full self-convolution) via FFT. Returns C of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    C = np.real(ifft(A * A))[:2 * n - 1]
    return C


def compute_subgrad_weighted(a, N, w_active):
    """Compute weighted subgradient: g[i] = 2 * sum_k w[k] * a[k-i].
    w_active: array of length 2n-1 with weights at active lags.
    Uses FFT: g = 2 * real(ifft(conj(FFT(w)) * FFT(a)))[:n]
    """
    n = len(a)
    W = fft(w_active, N)
    A = fft(a, N)
    g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return g


def nesterov_block(a_init, n, S, N, max_time, rng,
                   K_init=30, K_max=400, p_weight=2.0,
                   noise_scale=0.003, spectral_period=600,
                   restart_period=1500, max_step_frac=0.05):
    """Nesterov accelerated subgradient with adaptive top-K peak weighting."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = compute_conv(a, N).max()

    start_time = time.monotonic()
    it = 0

    # Adaptive parameters
    step_floor = 0.5 * S / (n * np.sqrt(n))
    step_cap = max_step_frac * S / np.sqrt(n)

    while time.monotonic() - start_time < max_time:
        it += 1

        # Compute convolution at Nesterov point y
        C = compute_conv(y, N)
        C_max = C.max()
        C_min = C.min()
        C_range = C_max - C_min

        if C_range < 1e-12:
            break

        # Adaptive K: more peaks as we converge
        K = min(K_max, K_init + it // 50)
        K = min(K, 2 * n - 1)

        # Top-K peaks with emphasis weights
        if K < (2 * n - 1):
            top_idx = np.argpartition(-C, K)[:K]
        else:
            top_idx = np.arange(2 * n - 1)

        # Weight by excess^p relative to median
        C_med = np.median(C)
        excess = np.maximum(0.0, C - C_med)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            w_active = np.zeros(2 * n - 1)
            w_active[top_idx] = 1.0 / len(top_idx)
        else:
            w_full = excess ** p_weight
            w_total = w_full.sum()
            if w_total < 1e-30:
                w_active = np.zeros(2 * n - 1)
                w_active[top_idx] = 1.0 / len(top_idx)
            else:
                w_active = w_full / w_total

        # Subgradient
        g = compute_subgrad_weighted(y, N, w_active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        # Step size: Polyak-like with adaptive target
        # Target: slightly below current best (encourage progress)
        f_target = best_val * 0.999
        gap = max(C_max - f_target, 0.001)
        alpha = gap / (g_norm * g_norm)
        alpha = np.clip(alpha, step_floor, step_cap)

        # Nesterov update
        x_new = y - alpha * g
        x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)

        # Occasionally evaluate at a (not just y) for best tracking
        if it % 20 == 0:
            C_a = compute_conv(x_new, N)
            val_a = C_a.max()
            if val_a < best_val:
                best_val = val_a
                best_a = x_new.copy()

        a_old = a
        a = x_new
        t = t_new

        # Momentum restart if stuck
        if it % 100 == 0:
            C_check = compute_conv(a, N)
            if C_check.max() > best_val * 1.02:
                y = a.copy()
                t = 1.0

        # Spectral reshaping (global move)
        if it % spectral_period == 0:
            a = spectral_anti_peak(a, N, n, S, rng, strength=0.05)
            y = a.copy()
            t = 1.0
            C_a = compute_conv(a, N)
            if C_a.max() < best_val:
                best_val = C_a.max()
                best_a = a.copy()

        # Noise injection for exploration
        if it % restart_period == 0:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            y = a.copy()
            t = 1.0

    # Final check
    C = compute_conv(a, N)
    if C.max() < best_val:
        best_val = C.max()
        best_a = a.copy()

    return best_a, best_val


def spectral_anti_peak(a, N, n, S, rng, strength=0.05):
    """Suppress frequency components that contribute to C peaks.
    
    The C(k) peak at lag k is related to frequency content at specific bins.
    We identify which frequencies create constructive interference at peak lags
    and reduce their magnitude.
    """
    C = compute_conv(a, N)
    C_max = C.max()
    C_mean = C.mean()
    
    if C_max <= C_mean * 1.01:
        return a  # Already flat enough

    A = fft(a, N)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    # Identify top C peaks and their frequency contributions
    n_peaks = min(15, 2 * n - 1)
    top_idx = np.argpartition(-C, n_peaks)[:n_peaks]
    top_vals = C[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    weights = (top_vals / total) ** 1.5  # Emphasize worst peaks

    # For each peak lag k, the frequency bins that contribute constructively
    # are those where exp(2*pi*i*l*k/N) aligns with the phase of the peak.
    # Simplified: reduce |A| at bins where multiple peaks have aligned phases.
    m_arr = np.arange(N, dtype=np.float64)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * float(k) / N)

    # Suppress where cos_eff is positive (constructive interference)
    suppression = 1.0 - strength * np.maximum(0, cos_eff)
    A_new = A * np.sqrt(np.maximum(suppression, 0.1))

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)

    # Verify it's an improvement, otherwise use a small random perturbation
    C_new = compute_conv(a_new, N)
    if C_new.max() > C_max:
        # Fallback: small random perturbation
        a_new = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return a_new


def coordinate_descent_block(a_init, n, S, N, max_time, rng, n_sweeps_max=50):
    """Cyclic coordinate descent with 1D minimax subproblem per coordinate."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_val = compute_conv(a, N).max()

    start_time = time.monotonic()
    sweep = 0

    while time.monotonic() - start_time < max_time and sweep < n_sweeps_max:
        sweep += 1
        C = compute_conv(a, N)
        C_max = C.max()

        # Process coordinates in random order (or Cyclic for stability)
        order = np.arange(n)  # Cyclic for better convergence
        # Or: order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start_time > max_time:
                    break

            # 1D subproblem: find delta for a[i] that minimizes max(C)
            # C(k) changes as:
            #   C(k) -> C(k) + 2*a[k-i]*delta  (for k != 2i, where 0<=k-i<n)
            #   C(2i) -> C(2i) + 2*a[i]*delta + delta^2

            # Precompute d(k) = dC(k)/da[i] for relevant k
            # Only track top ~30 peaks for efficiency
            n_track = min(30, 2 * n - 1)
            if n_track < (2 * n - 1):
                track_idx = np.argpartition(-C, n_track)[:n_track]
            else:
                track_idx = np.arange(2 * n - 1)

            d_vals = np.zeros(len(track_idx))
            quad_mask = np.zeros(len(track_idx), dtype=bool)
            for j, k in enumerate(track_idx):
                ki = int(k) - i
                if 0 <= ki < n:
                    d_vals[j] = 2.0 * a[ki]
                if int(k) == 2 * i:
                    d_vals[j] = 2.0 * a[i]
                    quad_mask[j] = True

            # Track values
            track_vals = C[track_idx].copy()

            # Minimize max_j [track_vals[j] + d_vals[j]*delta + delta^2*quad_mask[j]]
            # subject to a[i] + delta >= 0
            # Use subgradient method on the 1D problem
            delta = 0.0
            for _ in range(10):
                new_vals = track_vals + d_vals * delta
                new_vals[quad_mask] += delta * delta
                max_j = np.argmax(new_vals)
                grad_1d = d_vals[max_j]
                if quad_mask[max_j]:
                    grad_1d += 2.0 * delta
                if abs(grad_1d) < 1e-15:
                    break
                # Bisection-style step
                lr = 0.5 / max(abs(grad_1d), 1e-12)
                lr = min(lr, 0.05)
                delta -= lr * grad_1d
                # Clip to maintain feasibility
                delta = np.clip(delta, -a[i] + 1e-15, S - a[i] - 1e-15)

            # Apply update
            a[i] += delta
            a[i] = max(0.0, a[i])

        # Renormalize after full sweep
        s = a.sum()
        if s > 1e-15:
            a = a * S / s

        # Check improvement
        C = compute_conv(a, N)
        if C.max() < best_val:
            best_val = C.max()
            best_a = a.copy()

    return best_a, best_val


def lse_polish(a_init, n, S, N, max_time, rng, beta=200.0):
    """Quick L-BFGS polish using LSE-smoothed objective (for final refinement)."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = project_simplex(a, S)

    best_a = a.copy()
    best_val = compute_conv(a, N).max()

    s_hist = []
    y_hist = []
    m_lbfgs = 8

    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # LSE gradient
        C = compute_conv(a, N)
        C_max = C.max()
        exp_bc = np.exp(beta * (C - C_max))
        total = exp_bc.sum()
        if total < 1e-300:
            g = np.zeros(n)
        else:
            w = exp_bc / total
            W = fft(w, N)
            A = fft(a, N)
            g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break

        # L-BFGS direction
        if len(s_hist) == 0:
            direction = -g
        else:
            q = g.copy()
            alphas = np.zeros(len(s_hist))
            for ii in range(len(s_hist) - 1, -1, -1):
                s_i = s_hist[ii]
                y_i = y_hist[ii]
                yy = np.dot(y_i, y_i)
                if yy < 1e-30:
                    alphas[ii] = 0.0
                else:
                    alphas[ii] = np.dot(s_i, q) / yy
                    q = q - alphas[ii] * y_i
            s_last = s_hist[-1]
            y_last = y_hist[-1]
            yy_last = np.dot(y_last, y_last)
            if yy_last > 1e-30:
                gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
            else:
                gamma_use = 1.0
            r = gamma_use * q
            for ii in range(len(s_hist)):
                s_i = s_hist[ii]
                y_i = y_hist[ii]
                yy = np.dot(y_i, y_i)
                if yy < 1e-30:
                    continue
                beta_i = np.dot(s_i, y_i) / yy
                r = r + (alphas[ii] - beta_i) * y_i
            direction = -r
            if np.dot(g, direction) > 0:
                direction = -g

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Line search
        alpha_ls = 2.0 * S / (n * beta)
        a_new = a
        cmax_new = C_max
        for ls in range(12):
            candidate = project_simplex(a + alpha_ls * direction, S)
            C_ls = compute_conv(candidate, N)
            cm_ls = C_ls.max()
            if cm_ls <= C_max or ls == 11:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= C_max:
                    break
            alpha_ls *= 0.5

        # Update L-BFGS memory
        g_new = g  # approximate (recomputing is expensive)
        s_k = a_new - a
        y_k = g  # use same gradient as approximation
        sy = np.dot(s_k, y_k)
        if sy > 1e-14 * np.dot(s_k, s_k):
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        C_max = cmax_new

        if C_max < best_val:
            best_val = C_max
            best_a = a_new.copy()

    return best_a, best_val
