"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy: Multi-phase optimization with:
1. Averaged-subgradient FISTA with LSE smoothing (top-peak weighting)
2. Periodic spectral domain reweighting to escape local minima
3. Simulated-annealing restarts
4. Coordinate descent final polish
5. Multi-length Exploration (n=4096, 8192, 16384)

Key improvements:
- Top-k subgradient averaging (addresses multiple peaks simultaneously)
- Backtracking line search on actual objective
- Spectral "peak shaving" in frequency domain
- Adaptive restart strategy

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum (fix floating point)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        # Distribute among positive entries
        pos = result > 0
        if pos.any():
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient_topk(a, c, k=3):
    """
    Compute weighted subgradient from top-k peaks of convolution.
    Returns gradient vector and the peak values.
    """
    n = len(a)
    # Get top-k indices
    # Use partial sort for efficiency
    if k >= len(c):
        top_idx = np.argsort(c)[::-1]
    else:
        # Get top-k values first, then find all indices >= (k-th value)
        # For simplicity, use full sort (n is not too large)
        top_idx = np.argsort(c)[::-1][:k * 2]  # extra for ties
    
    # Weight by peak value (higher peaks get more weight)
    peak_values = c[top_idx[:k]]
    weights = peak_values / peak_values.sum()
    
    grad = np.zeros(n)
    for i in range(k):
        kk = top_idx[i]
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        grad[lo:hi + 1] += weights[i] * 2.0 * a[kk - hi:kk - lo + 1][::-1]
    
    return grad

    best = np.inf

        if s_y > 0:

def spectral_peak_shave(a_init, n, S, max_time, rng, n_iter=200):
    """
    Spectral domain peak shaving: modify the power spectrum to reduce
    time-domain convolution peaks.
    
    The convolution c = conv(a,a) has DFT (zero-padded) = A^2.
    We want to minimize max(c), which means making A^2 "flat" in the sense
    that its inverse DFT has minimal peak.
    
    Strategy: iteratively reduce the magnitude of A at frequencies that
    contribute most to the peak.
    """
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    for it in range(n_iter):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find the peak location
        k_peak = np.argmax(c)
        
        # In frequency domain: c has DFT = A^2 (zero-padded)
        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        A = fft(a_pad)
        A_sq = A * A  # This is the DFT of c (zero-padded to fsize)
        
        # We want to reduce the peak of c. The peak at position k_peak
        # is related to the "focus" of A at that spatial frequency.
        # Reduce |A| at frequencies that constructively interfere at k_peak.
        
        # The contribution of frequency m to c[k_peak] is proportional to
        # A[m] * exp(-2πi*m*k_peak/fsize)
        # So we should reduce |A[m]| for m that have exp(-2πi*m*k_peak/fsize) ≈ 1
        
        # Simpler approach: just flatten |A| (make it more uniform)
        mag = np.abs(A)
        mag_mean = mag.mean()
        
        # Soft flattening with adaptive strength
        alpha = 0.1 + 0.3 * (it / max(n_iter, 1))
        mag_new = (1 - alpha) * mag + alpha * mag_mean
        
        # Preserve the DC component (sum of a)
        mag_new[0] = mag[0]
        
        # Reconstruct
        phase = A / np.maximum(mag, 1e-30)
        A_new = phase * mag_new
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative with sum S
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)
        
        a = a_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coordinate_descent_v2(a_init, n, S, max_time, rng):
    """
    Efficient coordinate descent using precomputed convolution structure.
    Uses the fact that for a transfer from j to m, the change in c[k] is
    a[k-j] + a[k-m] ... etc.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        improved = False
        n_try = min(100 + it, 500)
        
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 0.5:
                break
            
            # Choose pair
            r = rng.random()
            if r < 0.6:
                # Random pair
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            elif r < 0.8:
                # Local pair
                j = rng.integers(0, n - 1)
                m = (j + rng.integers(1, 10)) % n
            else:
                # Targeted: move from high-value to low-value position
                # that contributes to the peak
                k_peak = np.argmax(c)
                # Positions that contribute to c[k_peak]
                lo = max(0, k_peak - n + 1)
                hi = min(k_peak, n - 1)
                # Pick from the "hot" region
                if hi - lo > 0:
                    j = rng.integers(lo, hi + 1)
                    m = rng.integers(0, n)
                else:
                    j = rng.integers(0, n)
                    m = rng.integers(0, n)
            
            if j == m or a[j] < 1e-10:
                continue
            
            # Compute the change vector d[k] = c_new[k] - c_old[k] for transfer delta from j to m
            # c[k] = sum_i a[i]*a[k-i]
            # d[k]/d(delta) = a[k-j] + a[j-k] - a[k-m] - a[m-k] (with bounds)
            
            # For efficiency, compute d using the structure:
            # Contribution from j: a appears at position j, so c[k] gains a[k-j] for the "first" a
            # and a[j] for the "second" a (when k-i=j, i=k-j)
            # Net: d[k] = (a[k-j] + a[j-k]) - (a[k-m] + a[m-k]) where out-of-range = 0
            
            # Vectorized computation
            k_vals = np.arange(2 * n - 1)
            d = np.zeros(2 * n - 1)
            
            # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
            lo_j = j
            hi_j = j + n - 1
            valid_j = (k_vals >= lo_j) & (k_vals <= hi_j)
            d[valid_j] += a[k_vals[valid_j] - j]
            
            # a[j-k] is valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
            lo_j2 = j - n + 1
            hi_j2 = j
            valid_j2 = (k_vals >= lo_j2) & (k_vals <= hi_j2)
            d[valid_j2] += a[j - k_vals[valid_j2]]
            
            # Similarly for m (with negative sign)
            lo_m = m
            hi_m = m + n - 1
            valid_m = (k_vals >= lo_m) & (k_vals <= hi_m)
            d[valid_m] -= a[k_vals[valid_m] - m]
            
            lo_m2 = m - n + 1
            hi_m2 = m
            valid_m2 = (k_vals >= lo_m2) & (k_vals <= hi_m2)
            d[valid_m2] -= a[m - k_vals[valid_m2]]
            
            # Also quadratic term for k=0 (c[0] = sum a[i]^2)
            # d[0] should include 2*(a[j]-a[m])*delta + 2*delta^2
            # But our linear d already captures the linear part
            # The quadratic part: c[0] changes by 2*delta^2 (from (a[j]+d)^2 - a[j]^2 + (a[m]-d)^2 - a[m]^2)
            # = 2*a[j]*d + d^2 - 2*a[m]*d + d^2 = 2*(a[j]-a[m])*d + 2*d^2
            # The linear part 2*(a[j]-a[m])*d is already in d[0]
            # So we just need to add 2*delta^2 to c[0]
            
            # Find optimal delta using the fact that c[k] + delta*d[k] should be minimized in max
            # c_new[k] = c[k] + delta * d[k] (plus quadratic correction at k=0)

            # Bounds on delta

            hi_delta = a[m] if m < n else 0  # a[m] - delta >= 0 (if m is valid)
            # Actually m is always < n since we chose it from [0, n)
            hi_delta = a[m]
            
            if lo_delta >= hi_delta:
                continue
            
            # The function to minimize: f(delta) = max_k (c[k] + delta * d[k]) (+ 2*delta^2 at k=0)
            # Use golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo_delta
            b_g = hi_delta

                fc = (c + c_g * d).max()

                a[m] += delta

    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def lse_gradient(a, c, beta):
    """Compute LSE gradient efficiently."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_v2(a_init, n, S, max_time, rng):
    """
    Improved FISTA with:
    - Top-k weighted subgradient
    - Adaptive step with backtracking
    - Periodic spectral correction
    - Restart strategy
    """
    a = normalize_sum(a_init, n)

    start = time.monotonic()

    best_a = a.copy()

    step = 1.0 / 100.0

    spectral_interval = 500

    while time.monotonic() - start < max_time:

            c_chk = fft_convolve_self(y)
            print("    FISTA2 it=%d, best=%.6f, cur=%.6f, elapsed=%.0fs" % (
                it, best, c_chk.max(), time.monotonic() - start))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            if cur < best - 1e-10:
                last_improve = it
            best = cur
            best_a = y.copy()

        # LSE gradient

        grad_lse = lse_gradient(y, c, beta_cur)

            grad = grad_lse

        if gnorm < 1e-14:
            break

                step *= 0.7

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

            mag_mean = mag.mean()
            alpha = 0.05  # gentle

            if s_sp > 1e-15:

            s_y = y.sum()

        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def generate_initializations(n, rng, S):
    """Generate diverse initializations."""
    init_list = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    init_list.append(("uniform", np.ones(n) * S / n))
    
    # Gaussian
    init_list.append(("gaussian", normalize_sum(np.exp(-5.0 * (tt - 0.5) ** 2 / 0.1), n)))
    
    # Sine (positive part)
    init_list.append(("sine", normalize_sum(np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * tt), 0.01), n)))
    
    # Cosine window
    init_list.append(("cosine", normalize_sum(0.5 * (1 + np.cos(2 * np.pi * tt)), n)))
    
    # Triangular
    tri = np.minimum(tt, 1 - tt) * 2
    init_list.append(("triangle", normalize_sum(np.maximum(tri, 0.01), n)))
    
    # Random
    init_list.append(("random", normalize_sum(rng.random(n) + 0.01, n)))
    
    # Para-central (high at edges)
    edge = np.maximum(1 - tt * 4, 0) + np.maximum((tt - 0.75) * 4, 0)
    init_list.append(("edge", normalize_sum(np.maximum(edge, 0.01), n)))
    
    # Two bumps
    bump = np.exp(-50 * tt**2) + np.exp(-50 * (tt-1)**2)
    init_list.append(("twobump", normalize_sum(bump, n)))
    
    return init_list
