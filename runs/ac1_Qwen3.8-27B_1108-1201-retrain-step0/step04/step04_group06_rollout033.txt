"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using log-sum-exp smoothed
subgradient descent with momentum, periodic spectral flattening,
and coordinate descent polish.

Key innovations:
1. Log-sum-exp smoothing of max(conv) gives a differentiable objective
   whose gradient uses ALL lags (weighted by exponential of their values).
2. Adaptive temperature increases focus on peaks as optimization progresses.
3. Spectral flattening step periodically reshapes the power spectrum
   to escape local minima.
4. Incremental coordinate descent for final polish.
5. Multiple restarts with different n and initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    # Lagrangian approach
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Correct for floating point
    w = w / w.sum() * S
    return w


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """
    Compute the gradient of the log-sum-exp objective.
    w: weights for each lag (length 2n-1), should sum to 1
    a: sequence (length n)
    N: FFT size (>= 3n-2)
    Returns: gradient vector of length n
    """
    n = len(a)
    W = fft(w, N)
    A = fft(a, N)
    # Cross-correlation: grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * IFFT(FFT(w) * conj(FFT(a)))[i]
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten_step(a, n, N, gamma):
    """
    One step of spectral flattening.
    Reduces the magnitude of A where the excess spectrum is large.
    Returns the new a (projected to simplex).
    """
    S = a.sum()
    c = fft_convolve_self(a)
    c_target = c.mean()
    excess = np.maximum(c - c_target, 0.0)
    
    if excess.max() < 1e-15:
        return a
    
    # Spectrum of the excess
    E = fft(excess, N)
    
    # Modify A: reduce where |E| is large relative to |A|^2
    A = fft(a, N)
    absA = np.abs(A)
    
    # Avoid division by zero
    ratio = gamma * np.abs(E) / (absA * absA + 1e-30)
    # Cap the reduction
    ratio = np.minimum(ratio, 0.5)
    
    # Modify: A_new = A * (1 - ratio) (in magnitude)
    A_new = A * (1.0 - ratio)
    
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_cpp_shape(a, N, n, target_lag_weight, alpha, rng):
    """
    Targeted spectral shaping: reduce specific peak lags.
    """
    S = a.sum()
    c = fft_convolve_self(a)
    
    # Find peak lags
    k_star = int(np.argmax(c))
    c_peak = c[k_star]
    
    if c_peak <= 1.01:  # Already close to optimal
        return a
    
    # Effective cosine at the peak lag
    m = np.arange(N)
    cos_eff = np.cos(2.0 * np.pi * m * k_star / N)
    
    # Only reduce where cos_eff > 0 (constructive interference at peak)
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_new = P * np.exp(-alpha * np.maximum(cos_eff, 0.0))
    
    a_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(a_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def coord_descent_polish(a_init, n, S, max_time, n_top=20, n_rounds=1000):
    """
    Incremental coordinate descent on top-contributing coordinates.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # Precompute
    c = fft_convolve_self(a)
    c_max = c.max()
    if c_max < best_score:
        best_score = c_max
        best_a = a.copy()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        # Find peak lag
        k_star = int(np.argmax(c))
        c_peak = c[k_star]
        
        # Find top-contributing coordinates
        # c[k_star] = sum_i a[i]*a[k_star-i] for valid i
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        
        # Contribution of a[i] to c[k_star] is a[k_star-i]
        contribs = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        top_idxs = np.argsort(contribs)[::-1][:n_top]
        
        improved = False
        for idx in top_idxs:
            i = int(idx)
            if time.monotonic() - start > max_time:

            if j < 0 or j >= n:
