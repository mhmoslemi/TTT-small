"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences using advanced
FISTA with correct cross-correlation gradients, multi-phase optimization,
multiplicative repulsion, and diverse length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def cross_correlation(a, w):
    """Compute cross-correlation R[m] = sum_j a[j]*w[j+m] via FFT."""
    n = len(a)
    lw = len(w)
    N = 1 << (n + lw - 1).bit_length()
    A_N = fft(a, N)
    W_N = fft(w, N)
    return np.real(ifft(np.conj(A_N) * W_N))[:n]


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max auto-convolution."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    return 2.0 * cross_correlation(a, w)


def lse_value(c, beta):
    """LSE approximation of max."""
    cm = c.max()
    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta


def normalize_to_sum(a, S):
    """Clamp non-negative and rescale to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * (S / len(a))
    return a * (S / s)


def multiplicative_repulsion_update(a, n, S, beta=50.0, lr=0.1):
    """Multiplicative repulsion: reduce values near convolution peaks."""
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return a.copy()
    w = e / tot
    rep = cross_correlation(a, w)
    # Multiplicative decay
    max_rep = rep.max()
    if max_rep < 1e-15:
        return a.copy()
    a_new = a * np.exp(-lr * rep / max_rep)
    a_new = np.maximum(a_new, 1e-15)
    return normalize_to_sum(a_new, S)


def single_fISTA_run(a_init, n, S, max_time, rng, beta_init=20.0, beta_rate=0.002,
                     beta_max=2000.0, use_repulsion=True, repulsion_freq=300,
                     repulsion_lr=0.05, noise_freq=5000, noise_scale_init=0.003):
    """
    FISTA with LSE smoothing, periodic repulsion, and noise injection.
    Returns (best_a, best_objective).
    """
    a = normalize_to_sum(a_init.copy(), S)
    y = a.copy()
    t_fista = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        # Stuck detection: inject noise and reset momentum
        if it - last_improve > 3000:
            ns = noise_scale_init * S / np.sqrt(n) * np.exp(-it / 50000.0)
            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_fista = 1.0
                last_improve = it
            else:
                break

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y = x_new + ((t_fista - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_fista = t_new

        # Periodic multiplicative repulsion
        if use_repulsion and it % repulsion_freq == 0 and it > 200:
            y = multiplicative_repulsion_update(y, n, S, beta=40.0, lr=repulsion_lr)
            a = y.copy()
            t_fista = 1.0

        # Periodic Gaussian noise

            if ns > 1e-8:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_fista = 1.0

    return best_a, best_obj


def diverse_initialization(n, rng):
    """Generate a diverse set of initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Smooth periodic
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t + rng.uniform(0, 2 * np.pi)))

    # Gaussian bump
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2 / 0.3))

    # Sine squared
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(np.sin(np.pi * t) ** 4)

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * t) + 0.2 * np.cos(10 * np.pi * t))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * rng.randint(2, 7) * t))

    # Random smooth
    freqs = rng.choice(np.arange(1, 10), size=5, replace=False)
    amps = rng.uniform(0.1, 0.4, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    sig = np.ones(n)
    for f, a, p in zip(freqs, amps, phases):
        sig += a * np.sin(2 * np.pi * f * t + p)
    inits.append(np.maximum(sig, 0.01))

    # Pure random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Block structure
    nblocks = rng.choice([3, 5, 7, 13])
    block_len = max(1, n // nblocks)
    heights = rng.uniform(0.3, 1.5, nblocks)
    block_seq = np.repeat(heights, block_len)
    if len(block_seq) < n:
        block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
    inits.append(block_seq[:n])

    return [np.maximum(x, 0.01) for x in inits]


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def targeted_peak_reduction(a, n, S, rng, num_peaks=5):
    """
    Targeted reduction: identify top peaks in convolution and try to
    reduce them by adjusting nearby sequence elements.
    """
    c = fft_convolve_self(a)
    # Find top peaks
    peak_idx = np.argsort(c)[::-1][:num_peaks]
    a_new = a.copy()

    for pk in peak_idx:
        # The convolution at pk involves a[j]*a[pk-j] for valid j
        # Try reducing elements that contribute most to this peak
        lo = max(0, pk - n + 1)
        hi = min(pk, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        contributions = a[js] * a[pk - js]
        # Reduce the largest contributors slightly
        top_contrib = np.argsort(contributions)[::-1][:3]
        for tc_idx in top_contrib:
            j = js[tc_idx]
            a_new[j] *= 0.95  # Reduce by 5%

    return normalize_to_sum(a_new, S)

            if 1024 <= nl <= 32768 and nl != n3:
                candidate_lengths.append(nl)

        for nl in candidate_lengths[:3]:
            if time.monotonic() > deadline - 60:
                break
            S_nl = np.sqrt(2.0 * nl)
            t_nl = min(40.0, (deadline - time.monotonic() - 60) * 0.4)
            if t_nl < 10:
                break
            a_nl, obj_nl = try_length_change(global_a, global_n, global_best_obj,
                                             nl, S_nl, t_nl, rng)
            if obj_nl < global_best_obj:
                global_best_obj = obj_nl
                global_a = a_nl.copy()
                global_n = nl
                print("  Length change to n=%d: %.6f" % (nl, obj_nl))

    # ===== Phase 4: Final polishing =====
    remaining = deadline - time.monotonic()
    print("\n=== Phase 4: Final polishing (%.0fs) ===" % remaining)

    if global_a is not None and remaining > 20:
        n4 = global_n
        S4 = np.sqrt(2.0 * n4)

        # High-beta FISTA for sharp peak targeting
        t_polish = remaining * 0.5
        a4, obj4 = single_fISTA_run(global_a, n4, S4, t_polish, rng,
                                     beta_init=200.0, beta_rate=0.0005, beta_max=10000.0,
                                     use_repulsion=True, repulsion_freq=100, repulsion_lr=0.03,
                                     noise_freq=inf, noise_scale_init=0.0)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_a = a4.copy()
            print("  Polish A: %.6f" % obj4)

        # Another round with different parameters
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a5, obj5 = single_fISTA_run(global_a, global_n, S4, remaining2 * 0.8, rng,
                                         beta_init=500.0, beta_rate=0.0003, beta_max=20000.0,
                                         use_repulsion=True, repulsion_freq=150, repulsion_lr=0.02)
            if obj5 < global_best_obj:
                global_best_obj = obj5
                global_a = a5.copy()
                print("  Polish B: %.6f" % obj5)

    # ===== Phase 5: Diverse restarts if time remains =====
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        print("\n=== Phase 5: Diverse restarts (%.0fs) ===" % remaining)
        n5 = global_n
        S5 = np.sqrt(2.0 * n5)
        inits5 = diverse_initialization(n5, rng)
        per_restart = max(5.0, remaining * 0.3 / len(inits5))

        for i, init in enumerate(inits5[:5]):
            if time.monotonic() > deadline - 30:
                break
            if per_restart < 3:
                break
            # Blend with current best for warm start
            blend = 0.5 * global_a + 0.5 * normalize_to_sum(init, S5)
            ai, obji = single_fISTA_run(blend, n5, S5, per_restart, rng,
                                         beta_init=30.0, beta_rate=0.003, beta_max=2000.0)
            if obji < global_best_obj:
                global_best_obj = obji
                global_a = ai.copy()
                print("  Restart %d: %.6f" % (i, obji))

    # ===== Final verification =====
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verification score: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            final_list = [float(x) for x in np.ones(256) * (np.sqrt(512.0) / 256)]
            ver = evaluate_sequence(final_list)
            print("Fallback score: %.6f" % ver)
    else:
        final_list = [float(x) for x in np.ones(256) * (np.sqrt(512.0) / 256)]
        print("No solution found, using fallback")

    return final_list
