So with `fsize >= len(a) + len(w) - 1` to avoid wraparound:
`ifft(conj(fft(a, fsize)) * fft(w, fsize))[m] = sum_k a[k] * w[k+m]` (for m in the valid range)

This gives `(a ⋆ w)[m] = sum_k a[k] * w[k+m]`.

But I want `dLSE/da[m] = 2 * sum_k w[k] * a[k-m]`.

Let me define `s[m] = sum_k w[k] * a[k-m]`. This is `(w ⋆ a)[m]` in the "unshifted" convention.

Note that `s[m] = sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j]` (substituting j = k-m).
= `sum_j a[j] * w[j+m]`
= `(a ⋆ w)[m]` (circular cross-correlation of a and w at lag m)

So `s = ifft(conj(fft(a)) * fft(w))` (with appropriate padding).

And the gradient is `2 * s`.

Let me verify with the previous code:
```python
W = fft(w, fsize)
A = fft(y, fsize)
return 2.0 * np.real(ifft(W * np.conj(A)))[:n]
```

This computes `2 * ifft(W * conj(A))[:n]`.

`ifft(W * conj(A))[m] = sum_k w[k] * conj(a[k-m])` (circular, for real: `= sum_k w[k] * a[k-m]`)

So this is `2 * sum_k w[k] * a[k-m] = 2 * s[m]`. ✓

Great, so the previous code's gradient computation is correct.

"""
Minimize 2*n*max(self_conv)/sum^2 over non-negative sequences.
Strategy: Multi-phase with diverse initializations, FISTA+LSE, multi-scale upsample,
and targeted peak reduction.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_gradient(y, c, mode, beta=10.0, k=8):
    """Compute gradient of the smoothed max convolution."""
    n = len(y)
    if mode == 'lse':
        cm = c.max()
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            return np.zeros(n)
        w = e / tot
    elif mode == 'topk':
        k = min(k, len(c))
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k
    elif mode == 'topk_power':
        k = min(k, len(c))
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        vals = c[idx]
        pw = np.power(np.maximum(vals - c.min(), 1e-10), 0.5)
        w[idx] = pw / pw.sum()
    else:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_optimize(a_init, max_time, rng, mode='lse', beta_init=10.0,

    FISTA optimization with LSE/topk smoothing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1e6).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        grad = compute_gradient(y, c, mode, beta=beta, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step

        if stuck > 2000:

                stuck = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if it % restart_every == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def upsample(a, factor=2, method='linear'):
    """Upsample sequence."""
    n = len(a)
    n_new = n * factor
    if method == 'linear':
        t_old = np.linspace(0, 1, n, endpoint=False)
        t_new = np.linspace(0, 1, n_new, endpoint=False)
        return np.interp(t_new, t_old, a)
    elif method == 'repeat':
        return np.repeat(a, factor)
    else:
        return np.interp(np.linspace(0, 1, n_new, endpoint=False),
                        np.linspace(0, 1, n, endpoint=False), a)


def downsample(a, factor=2):
    """Downsample by block averaging."""
    n = len(a)
    n_new = n // factor
    return a[:n_new * factor].reshape(n_new, factor).mean(axis=1)


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    t = np.linspace(0, 1, n)
    return np.exp(-0.5 * ((t - 0.5) / sigma_frac) ** 2)


def make_cosine(n, freq=1.0, power=1.0):
    """Cosine-shaped sequence."""
    t = np.linspace(0, 1, n)
    return np.maximum(0, np.cos(np.pi * freq * (t - 0.5))) ** power


def make_uniform(n):
    return np.ones(n)


def make_bimodal(n):
    """Two peaks."""
    t = np.linspace(0, 1, n)
    return np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2)
