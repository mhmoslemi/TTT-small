```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2 * n * c.max() / (S * S))

def convolve_fft(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]
```

```python
def subgradient_descent(a_init, n, S, max_time, rng, top_k=50, tau_init=1.0):
    """
    Subgradient descent with Nesterov acceleration for minimizing max(conv(a,a)).
    a is normalized so sum(a) = S.
    """
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()  # Nesterov momentum point
    t_val = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = compute_score(a)
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Compute convolution at y
        c = np.convolve(y, y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (S * S)
        if sc < best_score:
            best_score = sc
            best = y.copy()
        
        # Softmax weights over top-k lags
        # Use temperature based on current max
        tau = tau_init * S / max(cur_max, 1e-10)  # relative temperature
        # Actually, let's use a fixed relative temperature
        c_rel = c / cur_max  # normalize
        top_idx = np.argsort(c_rel)[::-1][:top_k]
        top_vals = c_rel[top_idx]
        weights = np.exp(top_vals / 0.1 - top_vals.max() / 0.1)  # softmax with temp 0.1
        weights /= weights.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            # g[j] += 2 * weights[i] * y[k-j] for valid j
            # y[k-j] is valid when 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                g[j_min:j_max+1] += 2.0 * weights[i] * y[k - j_max:k - j_min:-1]
        
        # Nesterov step
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Step size: O(1/sqrt(t))
        step = 2.0 * S / (n * np.sqrt(t_val + 1.0))
        # Actually, let's use a simpler step size
        step = S / (n * (1.0 + 0.1 * np.sqrt(it)))
        
        x_new = project_simplex(y - step * g / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val**2))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y * (S / ys)
        a = x_new
        t_val = t_new
        
        # Periodic perturbation
        if it % 200 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_val = 1.0  # Reset momentum
    
    return best, best_score
```

Let me focus on the "water-filling" subgradient:

Instead of just pushing down the max lag, I'll push down all lags that are above a threshold T. The subgradient is:

This is like a "soft" version of the max, where all peaks above T are pushed down proportionally to how much they exceed T.

The threshold T can be set to, say, the median of c, or the mean, or a quantile.

```python
def waterfill_subgradient(a, n, S, threshold_quantile=0.5, top_fraction=0.3):
    """Compute water-filling subgradient."""
    c = np.convolve(a, a)
    # Find threshold
    T = np.quantile(c, 1 - top_fraction)
    # Weights: proportional to excess above T
    excess = np.maximum(c - T, 0)
    total = excess.sum()
    if total < 1e-15:
        return None, c, c.max()
    weights = excess / total
    
    # Compute subgradient
    g = np.zeros(n)
    # For efficiency, only include lags with significant weight
    significant = weights > 1e-6
    for k in np.where(significant)[0]:
        w = weights[k]
        if w < 1e-8:
            continue
        j_min = max(0, int(k) - n + 1)
        j_max = min(int(k), n - 1)
        if j_min <= j_max:
            g[j_min:j_max+1] += 2.0 * w * a[int(k) - j_max : int(k) - j_min : -1]
    
    return g, c, c.max()
```

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)
