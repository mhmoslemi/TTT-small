            contrib = P * np.cos(phase)
            # Only reduce where contrib > 0
            mask = contrib > P.mean()
            P *= np.where(mask, 1.0 - 0.01 * T, 1.0)
        
        # Renormalize to maintain sum
        A_new = np.sqrt(np.maximum(P, 0)) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Accept with Metropolis criterion
        c_new = fft_convolve_self(a_new)
        c_max_new = c_new.max()
        delta = c_max_new - c_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T * S**2 / n, 1e-10)):
            a = a_new
        
        # Occasionally add random perturbation
        if it % 100 == 0:
            noise = rng.normal(0, T * 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best_cmax
