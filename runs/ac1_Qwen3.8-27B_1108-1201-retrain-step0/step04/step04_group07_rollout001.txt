def fft_convolve_self(a):

def compute_subgradient(a, N, temperature):
    """
    Compute subgradient of max(convolve(a,a)) with softmax weighting.
    Returns (gradient, convolution, c_max, peak_lag_ratio).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    c_min = c.min()
    c_mean = c.mean()

    # Softmax weights with adaptive temperature
    spread = max(c_max - c_mean, 1e-10)
    tau = max(temperature * spread, 1e-10)
    shifted = c - c_max
    weights = np.exp(shifted / tau)
    weights /= weights.sum()

    # Compute gradient via FFT
    # g[i] = sum_k w[k] * (a[k-i] + a[i-k])  (with bounds)
    # Approximated via circular correlation/convolution with large FFT
    W = fft(weights, N)
    A = fft(a, N)
    # Cross-correlation term: sum_k w[k] * a[k-i]
    T1 = np.real(ifft(np.conj(W) * A))[:n]
    # Convolution term: sum_k w[k] * a[i-k]
    T2 = np.real(ifft(W * A))[:n]
    g = T1 + T2

    return g, c, c_max


def lp_direction_step(f, M, max_time=30.0):
    """
    Solve the LP to find the best direction g >= 0 maximizing sum(g)
    subject to (f*g)[k] <= M for all k.
    Returns g or None if the LP fails or is too slow.
    """
    n = len(f)
    t0 = time.monotonic()

    # Build constraint matrix A_lp: (2n-1) x n
    # Row k: A_lp[k, i] = f[i] if 0 <= k-i < n, else 0
    # This means: for row k, the valid columns i satisfy max(0, k-n+1) <= i <= min(n-1, k)

    # Instead of building the full matrix (which is 16383 x 8192 ~ 134M entries),
    # use the sparse structure or a different approach.
    # For large n, this LP is too big to build explicitly.
    # Use a reduced version: only include the top constraints.
    
    c = fft_convolve_self(f)
    # Get top constraints (lags where convolution is close to M)
    threshold = M * 0.95
    active_lags = np.where(c >= threshold)[0]
    
    if len(active_lags) < 10:
        # Add some random lags for diversity
        extra = np.random.choice(2*n-1, size=min(50, 2*n-1), replace=False)
        active_lags = np.unique(np.concatenate([active_lags, extra]))
    
    if len(active_lags) > 500:
        # Sample 500 active lags
        idx = np.random.choice(len(active_lags), size=500, replace=False)
        active_lags = active_lags[idx]

    # Build sparse constraint matrix for these lags
    # A_lp[k, i] = f[i] if 0 <= k-i < n
    from scipy.sparse import csr_matrix

    rows = []
    cols = []
    vals = []
    for lag in active_lags:
        lag = int(lag)
        i_min = max(0, lag - n + 1)
        i_max = min(n - 1, lag)
        if i_min <= i_max:
            for i in range(i_min, i_max + 1):
                rows.append(lag)
                cols.append(i)
                vals.append(f[i])

    if len(vals) == 0:
        return None

    A_lp = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    b_lp = np.full(len(active_lags), M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    # Solve with cvxpy
    g_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [A_lp @ g_var <= b_lp]
    problem = cp.Problem(objective, constraints)

    try:
        # Try multiple solvers
        for solver in [cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS, cp.CBC]:
            try:
                problem.solve(solver=solver)
                if g_var.value is not None:
                    elapsed = time.monotonic() - t0
                    if elapsed < max_time:
                        return np.maximum(g_var.value, 0.0)
                    break
            except Exception:
                continue
    except Exception:
        pass

    return None


def spectral_reshape(a, rng, strength=0.3):
    """
    Apply spectral reshaping: partially randomize the phase of the DFT
    while preserving approximately the magnitude, then project back.
    """
    n = len(a)
    S = a.sum()
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    magnitude = np.abs(A)
    phase = np.angle(A)

    # Partially randomize phase
    new_phase = phase + rng.standard_normal(N) * strength
    A_new = magnitude * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def spectral_flatten(a, N, alpha, rng=None):
    """
    Flatten the power spectrum |FFT(a)|^2 toward uniform.
    Reduces high-magnitude bins, boosts low ones.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_mean = P.mean()

    if P_mean < 1e-30:
        return a

    ratio = P / P_mean
    # Compression: push toward 1
    P_new = np.zeros_like(P)
    above = ratio >= 1.0
    P_new[above] = P[above] * np.power(ratio[above], -alpha)
    P_new[~above] = P[~above] * np.power(ratio[~above], 1.0 - alpha)
    P_new = np.maximum(P_new, 0.0)

    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def subgradient_optimize(a_init, n, S, max_time, rng, target_obj=1.50,
                         use_lp=False, lp_interval=200):
    """
    Main subgradient optimization with Nesterov acceleration,
    softmax weighting, adaptive step size, and optional LP steps.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (2 * n - 1).bit_length()

    y = a.copy()  # Nesterov momentum variable
    t = 1.0
    start = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()
    it = 0

    # Temperature schedule: start wide, narrow down
    tau_start = 0.5
    tau_end = 0.05
    # Step size parameters
    step_scale = 2.0 / n
    step_decay = 0.9995

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()
            if best_score <= target_obj:
                break

        # Adaptive temperature: start wide, narrow
        progress = min(1.0, it / max(int(max_time * 2), 1))
        tau = tau_start + (tau_end - tau_start) * progress

        # Compute subgradient
        g, c, c_max = compute_subgradient(y, N, tau)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        # Use the gap between max and mean to scale the step
        gap = c_max - c.mean()
        if gap < 1e-10:
            break
        step = step_scale * S / (np.sqrt(t) * gap) * S
        step = min(step, S * 0.1 / n)  # Cap the step

        # Projected gradient step
        x_new = project_simplex(y - step * g / max(gnorm, 1e-15), S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % 1000 == 500:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 + 0.5 * np.sin(it * 0.01))
            y = project_simplex(y + rng.standard_normal(n) * noise_scale, S)
            a = y.copy()
            t = 1.0

        # Periodic spectral reshaping
        if it % 2000 == 1000:
            a_shaped = spectral_flatten(a, N, alpha=0.1 + 0.2 * (1.0 - progress), rng=rng)
            sc_shaped = compute_score(a_shaped)
            if sc_shaped < best_score:
                best_score = sc_shaped
                best_a = a_shaped.copy()
                a = a_shaped
                y = a_shaped
                t = 1.0

        # Optional LP step (expensive, use sparingly)
        if use_lp and it % lp_interval == lp_interval - 1:
            M = c_max
            g_dir = lp_direction_step(a, M, max_time=15.0)
            if g_dir is not None and g_dir.sum() > 1e-10:
                g_dir = g_dir / g_dir.sum() * S
                # Blend with current
                blend = 0.1 + 0.2 * (1.0 - progress)
                a_new = (1.0 - blend) * a + blend * g_dir
                a_new = project_simplex(a_new, S)
                sc_new = compute_score(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    y = a_new
                    t = 1.0

    return best_a, best_score


def make_good_initializations(n, rng):
    """Generate diverse, structured initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Approximate with a periodic binary pattern

    if period < 7:
        period = 7
    pattern = np.zeros(n)
    for i in range(n):
        # Simple LFSR-like pattern
        pattern[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / period) * np.cos(2 * np.pi * 3 * i / (2 * period))
    inits.append(("lfsr", np.clip(pattern, 0.05, 0.95)))

    # Chirp with multiple frequencies
    chirp = 0.5 + 0.2 * np.sin(2 * np.pi * (3 * tt + 2 * tt**2)) + \
                  0.15 * np.sin(2 * np.pi * 7 * tt) + \
                  0.1 * np.sin(2 * np.pi * 13 * tt)
    inits.append(("chirp_multi", np.clip(chirp, 0.05, 0.95)))

    # Smoothed random (low-pass filtered)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 128)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / max(smooth.max() - smooth.min(), 1e-10)
    inits.append(("smooth_rand", 0.2 + 0.6 * smooth))

    # Periodic variation (like a windowed repetition)
    period_len = max(8, n // 16)
    base = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(period_len) / period_len * 3)
    periodic = np.tile(base, n // period_len + 1)[:n]
    inits.append(("periodic", periodic))

    # Two-scale: large slow variation + small fast variation
    slow = 0.5 + 0.2 * np.sin(2 * np.pi * 2 * tt)
    fast = 0.1 * np.sin(2 * np.pi * 50 * tt)
    inits.append(("two_scale", np.clip(slow + fast, 0.05, 0.95)))

    # Random binary (0 or 1, centered)
    binary = (rng.random(n) > 0.5).astype(float)
    inits.append(("binary", 0.3 + 0.4 * binary))

    return inits

        if name not in seen:
            seen.add(name)
            unique_inits.append((name, arr))
    init_list = unique_inits

    t_start = time.monotonic()
    n_inits = len(init_list)

    for idx, (name, arr) in enumerate(init_list):
        elapsed = time.monotonic() - t_start
        remaining = budget_s - elapsed
        if remaining < 5:
            break

        # Allocate time: give more to the first (provided) init
        if idx == 0:
            budget_this = min(remaining * 0.4, budget_s * 0.4)
        elif idx == 1:
            budget_this = min(remaining * 0.25, budget_s * 0.25)
        else:
            budget_this = min(remaining / max(n_inits - idx, 1), budget_s * 0.15)
        budget_this = max(budget_this, 3.0)

        a = np.clip(arr, 0.0, 1000.0).astype(np.float64)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        sc = update_best(a)
        if sc <= target_obj:
            break

        # Run subgradient optimization
        a_opt, sc_opt = subgradient_optimize(a, n, S, budget_this, rng,
                                             target_obj=target_obj,
                                             use_lp=(n <= 8192 and idx == 0))
        update_best(a_opt)

        if idx < 3:
            print("    [%s n=%d] score=%.6f (elapsed %.0fs, remaining %.0fs)" %
                  (name, n, global_best, elapsed, budget_s - elapsed))

    # Final polish on best found with more aggressive settings
    if global_a is not None:
        final_budget = max(0, budget_s - (time.monotonic() - t_start))
        if final_budget > 5:
            a_polish, sc_polish = subgradient_optimize(global_a, n, S, min(final_budget, 60.0),
                                                       rng, target_obj=target_obj)
            update_best(a_polish)

            # Gentle spectral flattening
            rem = max(0, budget_s - (time.monotonic() - t_start))
            if rem > 3:
                n_flatten = max(50, int(rem / max(0.001, N * 14 * 1e-9)))
                for fl_it in range(min(n_flatten, 200)):
                    a_flat = spectral_flatten(global_a, N, alpha=0.05, rng=rng)
                    sc_flat = compute_score(a_flat)
                    if sc_flat < global_best:
                        global_best = sc_flat
                        global_a = a_flat.copy()
                    elif rem - (time.monotonic() - t_start) < 1:
                        break

    return global_a, global_best


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses softmax subgradient with Nesterov acceleration, LP steps,
    and spectral reshaping across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    deadline = t_global_start + max(0.0, budget_s - 20.0)

    print("Starting optimization with budget %.0fs" % budget_s)

    # Determine starting sequence
    start_sequence = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.sum() > 0.01:
            start_sequence = np.clip(given, 0, 1000)
            print("Using provided sequence of length %d as starting point" % len(given))
    except Exception:
        pass

    # Determine which lengths to try
    lengths_to_try = []
    if start_sequence is not None:
        lengths_to_try.append(len(start_sequence))
    for L in [8192, 16384, 4096]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)

    # Sort: start_sequence length first, then by priority
    if start_sequence is not None:
        sl = len(start_sequence)
        lengths_to_try.remove(sl)
        lengths_to_try = [sl] + lengths_to_try

    # Deduplicate
    seen = set()
    unique_lengths = []
    for L in lengths_to_try:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths_to_try = unique_lengths

    global_best = np.inf
    global_a = None

    # Allocate budget proportional to expected value
    total_budget = budget_s - 40  # leave 40s for verification and fallback

    # Prioritize the starting length
    weights = []
    for i, L in enumerate(lengths_to_try):
        if i == 0:
            w = 1.5  # starting length gets most time
        elif L == 16384:
            w = 1.0
        elif L == 8192:
            w = 0.8
        else:
            w = 0.5
        weights.append(w)
    total_w = sum(weights)
    budgets = [total_budget * w / total_w for w in weights]

    for i, n in enumerate(lengths_to_try):
        elapsed = time.monotonic() - t_global_start
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(budgets[i], remaining - 30)
        if alloc < 10:
            break

        print("=== Optimizing n=%d with budget %.0fs ===" % (n, alloc))

        # Pass starting sequence only for the first length
        start_a = None
        if i == 0 and start_sequence is not None and len(start_sequence) == n:
            start_a = start_sequence.copy()

        a_result, score = optimize_single_length(n, alloc, rng,
                                                 target_obj=1.50,
                                                 start_a=start_a)

        if a_result is not None and score < global_best:
            global_best = score
            global_a = a_result.copy()
            print("  New global best: %.6f at n=%d" % (global_best, n))
        else:
            print("  Best for n=%d: %.6f (global: %.6f)" % (n, score, global_best))

    # If we have a good result, do one final intensive polish
    if global_a is not None and global_best > 1.50:
        final_rem = deadline - time.monotonic()
        if final_rem > 10:
            print("=== Final polish with %.0fs ===" % final_rem)
            n = len(global_a)
            S = global_a.sum()
            a_polish, sc_polish = subgradient_optimize(global_a, n, S, min(final_rem, 120.0),
                                                       rng, target_obj=1.495)
            if sc_polish < global_best:
                global_best = sc_polish
                global_a = a_polish
                print("  Polish improved to: %.6f" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_best = 2.0
        print("Using fallback uniform sequence, score=%.4f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verification: evaluate_sequence = %.6f" % ver)
        if ver < global_best:
            global_best = ver
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver if np.isfinite(ver) else 2.0
        print("Fallback verification: %.6f" % ver)

    print("Final answer: length=%d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
