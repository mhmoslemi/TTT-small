def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

def lse_gradient(a, c, beta):
    """Compute LSE-smoothed gradient of self-convolution max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))
    return 2.0 * cross[:n]

    """Multiplicative (mirror) descent step."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Flatten the power spectrum partially."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()

    # Keep the phase of A

    A_new = mag_new * np.exp(1j * phase)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))
    return 2.0 * cross[:n]


def mirror_step(a, grad, eta, S):
    """Multiplicative (mirror) descent step maintaining sum=S and a>=0."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, S, alpha=0.05):
    """Partially flatten the power spectrum to reduce circular convolution peaks."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    target_power = S * S / n
    phase = np.angle(A)
    mag = np.abs(A)
    mag_new = np.sqrt((1 - alpha) * mag * mag + alpha * target_power)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def compute_objective(a):
    """Compute the objective value 2*n*max(a*a)/sum(a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_sum(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_mirror_lse(a_init, max_time, rng, verbose=False):
    """
    Mirror descent with LSE-smoothed gradient and periodic spectral flattening.
    Returns (best_a, best_obj, final_a, final_c).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S ** 2
    best_a = a.copy()

    # Initial step size: target ~5% relative change per step
    eta = 0.05 * np.sqrt(n / 2.0)
    eta_min = 0.005 * np.sqrt(n / 2.0)
    beta = 20.0
    consecutive_fail = 0
    it = 0
    flatten_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S ** 2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0

        # Compute LSE gradient
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent step
        a_new = mirror_step(a, grad, eta, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S ** 2

        if obj_new < cur_obj:
            a = a_new
            c = c_new
            eta = min(eta * 1.02, 5.0 * np.sqrt(n / 2.0))
        else:
            eta *= 0.75
            consecutive_fail += 1
            if consecutive_fail > 40:
                # Perturb to escape local minimum
                noise_scale = 0.003 * S / np.sqrt(n)
                noise = rng.normal(0, noise_scale, n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
                eta = 0.03 * np.sqrt(n / 2.0)
                consecutive_fail = 0
            if consecutive_fail > 100:
                # Larger perturbation
                noise_scale = 0.01 * S / np.sqrt(n)
                noise = rng.normal(0, noise_scale, n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)
                eta = 0.05 * np.sqrt(n / 2.0)
                consecutive_fail = 0

        # Periodic spectral flattening (helps escape flat directions)
        if it % 500 == 0 and flatten_count < 20:
            for fa in [0.02, 0.05, 0.1]:
                a_flat = spectral_flatten(a, S, fa)
                c_flat = fft_convolve_self(a_flat)
                obj_flat = 2.0 * n * c_flat.max() / S ** 2
                if obj_flat < cur_obj:
                    a = a_flat
                    c = c_flat
                    flatten_count += 1
                    break

        # Gradually increase beta (sharpen the LSE approximation to max)
        beta = 20.0 + 0.05 * it

        if verbose and it % 5000 == 0:
            print("    mirror_lse it=%d, obj=%.6f, beta=%.1f, eta=%.2f" % (
                it, best_obj, beta, eta))

    return best_a, best_obj, a, c


def run_peak_targeting(a_init, max_time, rng, verbose=False):
    """
    Iteratively target the top peaks of the convolution.
    For each peak, compute the exact subgradient and take a step.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S ** 2
    best_a = a.copy()

    it = 0
    consecutive_fail = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / S ** 2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            consecutive_fail = 0

        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        # Try reducing the top peak with a targeted step
        k_star = int(peak_indices[0])
        # Subgradient at peak k_star
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        grad = np.zeros(n)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] = 2.0 * a[k_star - i_vals]
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Adaptive step
        step = S / (n * gnorm) * 0.5
        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / S ** 2

        if obj_try < cur_obj:
            a = a_try
            c = c_try
        else:
            # Try a combined top-k gradient
            grad_k = np.zeros(n)
            for k_idx in peak_indices:
                lo2 = max(0, k_idx - n + 1)
                hi2 = min(k_idx, n - 1)
                i_vals2 = np.arange(lo2, hi2 + 1)
                grad_k[i_vals2] += a[k_idx - i_vals2]
            grad_k *= 2.0 / top_k
            gnorm_k = np.linalg.norm(grad_k)
            if gnorm_k > 1e-15:
                step_k = S / (n * gnorm_k) * 0.3
                a_try2 = project_simplex(a - step_k * grad_k, S)
                c_try2 = fft_convolve_self(a_try2)
                obj_try2 = 2.0 * n * c_try2.max() / S ** 2
                if obj_try2 < cur_obj:
                    a = a_try2
                    c = c_try2
                else:
                    consecutive_fail += 1
            else:
                consecutive_fail += 1

        if consecutive_fail > 60:
            noise_scale = 0.002 * S / np.sqrt(n)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            consecutive_fail = 0

        if verbose and it % 5000 == 0:
            print("    peak_target it=%d, obj=%.6f" % (it, best_obj))

    return best_a, best_obj


def run_hybrid_optimize(a_init, max_time, rng, verbose=False):
    """
    Hybrid optimizer: alternates between mirror descent (LSE) and peak targeting.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / S ** 2
    best_a = a.copy()

    # Phase 1: Mirror LSE (60% of time)
    t1 = max_time * 0.6
    a1, obj1, a, c = run_mirror_lse(a, t1, rng, verbose=verbose)
    best_a = a1.copy()
    best_obj = obj1

    # Phase 2: Peak targeting (40% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a2, obj2 = run_peak_targeting(a, remaining, rng, verbose=verbose)
        if obj2 < best_obj:
            best_a = a2
            best_obj = obj2

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Raised cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Inverted raised cosine (higher at edges)
    inits.append(np.maximum(0.5 * (1 - np.cos(2 * np.pi * tt)), 0.01))
    # Gaussian (centered)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.1))
    # Inverted Gaussian (edges higher)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.3) + 0.5 * np.exp(-20.0 * tt ** 2) + 0.5 * np.exp(-20.0 * (1 - tt) ** 2))
    # Sinusoidal
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt), 0.01))
    # Multi-harmonic
    inits.append(np.maximum(1.0 + 0.2 * np.cos(4 * np.pi * tt) + 0.1 * np.sin(6 * np.pi * tt), 0.01))
    # Random
    inits.append(rng.random(n) + 0.1)
    # U-shape
    inits.append(1.0 + 2.0 * (tt ** 3 + (1 - tt) ** 3))

    # If we have a parent sequence, add it and interpolated versions
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if len(p) == n:
            inits.append(p.copy())
        else:
            # Interpolate to length n
            from numpy import interp
            xp = np.linspace(0, 1, len(p))
            fp = interp(np.linspace(0, 1, n), xp, p)
            inits.append(np.maximum(fp, 0.01))
            # Also add a smoothed version
            from scipy.ndimage import gaussian_filter1d
            sm = gaussian_filter1d(fp, sigma=np.sqrt(n / len(p)) if n > len(p) else 1.0)
            inits.append(np.maximum(sm, 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses mirror descent with LSE gradients, spectral flattening, and peak targeting.
    Explores multiple lengths (8192, 16384).
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization. Budget: %.0fs" % budget_s)
    if given is not None:
        print("Previous best sequence: length %d, obj=%.6f" % (
            len(given), compute_objective(given)))

    t0 = time.monotonic()

    # Define candidate lengths
    candidate_lengths = [8192, 16384]
    if given is not None:
        # Put the previous length first
        if len(given) in candidate_lengths:
            candidate_lengths.remove(len(given))
            candidate_lengths.insert(0, len(given))
        else:
            candidate_lengths = [len(given)] + candidate_lengths

    # Time allocation: give more time to larger lengths
    total_time = deadline - time.monotonic()
    # Weight by length (larger n has more potential improvement)
    weights = [float(l) for l in candidate_lengths]
    total_w = sum(weights)
    time_alloc = [total_time * w / total_w for w in weights]

    # Minimum time per length
    for i in range(len(candidate_lengths)):
        time_alloc[i] = max(60.0, time_alloc[i])
    rescale = total_time / sum(time_alloc)
    time_alloc = [t * rescale for t in time_alloc]

    for li, n in enumerate(candidate_lengths):
        if time.monotonic() > deadline - 30:
            break

        S = np.sqrt(2.0 * n)
        t_alloc = min(time_alloc[li], deadline - time.monotonic() - 20)
        if t_alloc < 30:
            continue

        print("\n=== Length n=%d, allocated %.0fs ===" % (n, t_alloc))
        t_len_start = time.monotonic()

        # Generate initializations
        inits = generate_initializations(n, rng, parent=given)

        # Run optimization from each initialization (time-split)
        n_inits = len(inits)
        per_init_time = t_alloc / n_inits

        for ii, init_arr in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            t_remaining_for_init = min(per_init_time, deadline - time.monotonic() - 20)
            if t_remaining_for_init < 10:
                continue

            init_obj = compute_objective(init_arr)
            if init_obj < global_best:
                global_best = init_obj
                global_a = normalize_sum(init_arr, n).copy()
                print("  Init %d: obj=%.6f (new best!)" % (ii, global_best))

            # Run hybrid optimization
            t_run = min(t_remaining_for_init * 0.8, 600.0)
            a_opt, obj_opt = run_hybrid_optimize(init_arr, t_run, rng)
            a_opt = normalize_sum(a_opt, n)
            obj_opt = compute_objective(a_opt)

            if obj_opt < global_best:
                global_best = obj_opt
                global_a = a_opt.copy()
                print("  Init %d after opt: obj=%.6f (new best!)" % (ii, global_best))

        # After trying all inits for this length, do a final long run on the best
        t_final = deadline - time.monotonic() - 30
        if t_final > 30 and global_a is not None and len(global_a) == n:
            print("  Final run for n=%d (%.0fs)..." % (n, t_final))
            a_fin, obj_fin = run_hybrid_optimize(global_a, t_final, rng)
            a_fin = normalize_sum(a_fin, n)
            obj_fin = compute_objective(a_fin)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  Final: obj=%.6f" % global_best)

    # Final polish with very small steps
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        print("\n=== Final polish (%.0fs) ===" % remaining)
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_polish = global_a.copy()
        polish_start = time.monotonic()

        # Very fine LSE with high beta
        c_p = fft_convolve_self(a_polish)
        best_p = 2.0 * n_best * c_p.max() / S_best ** 2
        eta_p = 0.01 * np.sqrt(n_best / 2.0)

        while time.monotonic() - polish_start < remaining - 5:
            grad = lse_gradient(a_polish, c_p, 200.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            a_new = mirror_step(a_polish, grad, eta_p, S_best)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n_best * c_new.max() / S_best ** 2
            if obj_new < best_p:
                a_polish = a_new
                c_p = c_new
                best_p = obj_new
                eta_p = min(eta_p * 1.01, 0.5 * np.sqrt(n_best / 2.0))
            else:
                eta_p *= 0.8
                if eta_p < 1e-10:
                    # Try perturbation
                    noise = rng.normal(0, 1e-4 * S_best / np.sqrt(n_best), n_best)
                    a_polish = project_simplex(a_polish + noise, S_best)
                    c_p = fft_convolve_self(a_polish)
                    eta_p = 0.01 * np.sqrt(n_best / 2.0)

        a_polish = normalize_sum(a_polish, n_best)
        s_polish = compute_objective(a_polish)
        if s_polish < global_best:
            global_best = s_polish
            global_a = a_polish.copy()
            print("  After polish: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("\nReturning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
