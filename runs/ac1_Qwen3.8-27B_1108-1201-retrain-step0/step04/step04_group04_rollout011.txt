```python
def compute_delta_effect(a, j, m, n):
    """Compute d[k] = change in c_k per unit delta for transfer from m to j."""
    # d[k] = a[j+k] + a[j-k] - a[m+k] - a[m-k] (with bounds)
    k = np.arange(2*n-1)
    jk = j + k
    jmk = j - k
    mk = m + k
    mmk = m - k
    
    val_j = np.zeros(2*n-1)
    mask_j = (jk >= 0) & (jk < n)
    val_j[mask_j] = a[jk[mask_j]]
    mask_jm = (jmk >= 0) & (jmk < n)
    val_j[mask_jm] += a[jmk[mask_jm]]
    
    val_m = np.zeros(2*n-1)
    mask_m = (mk >= 0) & (mk < n)
    val_m[mask_m] = a[mk[mask_m]]
    mask_mm = (mmk >= 0) & (mmk < n)
    val_m[mask_mm] += a[mmk[mask_mm]]
    
    return val_j - val_m
```

```python
"""
Minimize 2*n*max_conv/sum^2 using LP-based active-set iteration,
vectorized coordinate descent, and FISTA polishing.

Key idea: Use the linearized LP to find a compatible direction at each step,
then line-search for the optimal interpolation parameter. This exploits the
local structure of the problem more effectively than gradient-based methods
alone.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def build_active_constraints(a, active_lags, n):
    """Build dense constraint matrix for active lags.
    Row k: A[k, j] = a[k-j] for valid j (so that (a*g)_k = A[k,:] @ g)
    """
    m = len(active_lags)
    A_mat = np.zeros((m, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A_mat[i, js] = a[k - js]
    return A_mat


def solve_lp_for_direction(a, M, active_lags, n, eps=0.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M*(1-eps) for active k, g >= 0.
    Returns g normalized to sum=sqrt(2n), or None.
    """
    m = len(active_lags)
    if m == 0:
        return None
    
    A_mat = build_active_constraints(a, active_lags, n)
    b_vec = np.full(m, M * (1.0 - eps))
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, 
                  bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    
    S = np.sqrt(2.0 * n)
    g = g * (S / g_sum)
    return g


def line_search_step(a, g, n, max_time=0.5):
    """Find optimal t in [0, 0.5] minimizing max(((1-t)*a + t*g)*(same)).
    Uses golden section search.
    """
    S = np.sqrt(2.0 * n)
    
    def objective(t):
        a_new = (1.0 - t) * a + t * g
        c = fft_convolve_self(a_new)
        return c.max()
    
    # Golden section search on [0, 0.5]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 0.5
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = objective(c_pt)
    fd = objective(d_pt)
    f_lo = objective(0.0)
    
    t_start = time.monotonic()
    for _ in range(30):
        if time.monotonic() - t_start > max_time:
            break
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = objective(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = objective(d_pt)
    
    t_opt = (lo + hi) / 2.0
    f_opt = objective(t_opt)
    
    # Also check t=0
    if f_lo < f_opt:
        t_opt = 0.0
        f_opt = f_lo
    
    return t_opt, f_opt

    """LP-based active set iteration.
    Returns best a found.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    best_a = a.copy()
    best_score = np.inf
    t_start = time.monotonic()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best_score:
            best_score = M
            best_a = a.copy()
        
        # Select active lags
        threshold = M * (1.0 - tol)
        active_lags = np.where(c >= threshold)[0]
        # Also always include top K
        top_k = np.argsort(c)[::-1][:K]
        active_lags = np.unique(np.concatenate([active_lags, top_k]))
        active_lags = sorted(active_lags.tolist())
        
        # Try LP with decreasing eps
        g = None
        for eps in [eps_init, 0.0, -eps_init]:
            g = solve_lp_for_direction(a, M, active_lags, n, eps=eps)
            if g is not None:
                break
        
        if g is None:
            # LP infeasible even with relaxed bound
            # Add noise and restart
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            continue
        
        # Check if g is meaningfully different from a
        diff = np.linalg.norm(g - a)
        if diff < 1e-6 * S:
            # Stuck, add perturbation
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            continue
        
        # Line search
        t_opt, f_opt = line_search_step(a, g, n, max_time=0.5)
        
        if f_opt < M - 1e-10:
            a_new = (1.0 - t_opt) * a + t_opt * g
            a = normalize_sum(a_new, n)
        else:
            # Line search didn't help, try smaller step
            for t_try in [0.01, 0.005, 0.001]:
                a_try = (1.0 - t_try) * a + t_try * g
                c_try = fft_convolve_self(a_try)

                    a = normalize_sum(a_try, n)
                    break
            else:
                # No improvement, perturb
                a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_score

def fista_polish(a, n, S, max_time, rng, beta=500.0):
    """FISTA with high beta for final polish."""
    a = normalize_sum(a, n)
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = np.inf
    t_start = time.monotonic()
    it = 0
    
    while time.monotonic() - t_start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        # Gradient: 2 * (w * a)_truncated
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta)
        # Adaptive: reduce if no improvement recently
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best_score
```
