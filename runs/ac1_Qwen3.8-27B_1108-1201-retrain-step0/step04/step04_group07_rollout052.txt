"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Projected subgradient with Nesterov acceleration on the max-convolution objective
2. Spectral power flattening via iterative FFT corrections
3. Multi-start from diverse initializations including the previous best
4. Adaptive step sizing and periodic perturbation for escape from local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) - cssv > 0)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def subgradient_nesterov(a_init, n, S, max_time, rng, top_k=50, 
                         alpha_base=1.0, perturb_every=200, perturb_prob=0.3):
    """
    Nesterov-accelerated projected subgradient method for minimizing max(c).
    We fix sum(a) = S and minimize max(c).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    
    best_score = compute_score(y)
    best_local = y.copy()
    it = 0
    N = 1 << (2 * n - 1).bit_length()
    
    # Precompute FFT size
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        # Compute convolution at y
        c = fft_convolve_self(y)
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_local = y.copy()
        
        # Get top-k lags weighted by value
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx].astype(np.float64)
        # Exponential weighting
        weights = np.exp(-0.5 * (top_vals - top_vals[0]) / max(top_vals[0] - top_vals[-1], 1e-10) * top_k)
        weights /= weights.sum()
        
        # Compute gradient as weighted combination
        # grad[j] = sum_i weights[i] * (y[top_idx[i] - j] + y[j - top_idx[i]])
        # This is a weighted sum of shifted y's
        grad = np.zeros(n, dtype=np.float64)
        for i in range(top_k):
            k = int(top_idx[i])
            w = weights[i]
            # y shifted left by k (contribution from a[k-j])
            # For j in [0, n-1], a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if hi1 >= lo1:
                grad[lo1:hi1+1] += w * y[k - hi1:k - lo1 + 1][::-1]
            # y shifted right by k (contribution from a[j-k])
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if hi2 >= lo2:
                grad[lo2:hi2+1] += w * y[lo2 - k:hi2 - k + 1]
        
        gnorm = np.linalg.norm(grad)

def spectral_flatten(a, max_time, rng, alpha_init=0.5, target=1.50):
    """
    Iterative spectral flattening: push |FFT(a)|^2 toward uniform.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        
        # Target: geometric mean level
        P_target = np.exp(np.mean(np.log(P + 1e-30)))
        
        # Alpha anneals from alpha_init to 0.05
        progress = it / max(1, it)
        alpha = max(0.05, alpha_init * np.exp(-0.01 * it))
        
        # Multiplicative correction
        ratio = P / P_target
        # Compress: new_ratio = ratio^(1-alpha) for ratio > 1, ratio^(1+alpha) for ratio < 1
        P_new = np.zeros_like(P)
        above = ratio >= 1.0
        P_new[above] = P[above] * np.power(ratio[above], -alpha)
        P_new[~above] = P[~above] * np.power(ratio[~above], alpha)
        P_new = np.maximum(P_new, 0.0)
        
        # Preserve phase, reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if best_score <= target:
            break
        
        # Occasional noise
        if it % 100 == 50 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def make_initializations(n, rng):

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = u * np.arange(1, n + 1) - cssv
    valid = np.where(rho_candidates > 0)[0]

        return np.full(n, S / n)

    return np.maximum(v - theta, 0.0)
