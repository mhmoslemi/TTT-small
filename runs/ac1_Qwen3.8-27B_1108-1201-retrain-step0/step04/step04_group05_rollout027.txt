If j = k/2 (i.e., k = 2j), it's a single term a[j]*a[j] = a[j]^2.

So the total contribution of a[j] to conv(a,a)_k is:
- If 0 <= k-j < n AND k-j != j: 2 * a[j] * a[k-j] (linear in a[j])
- If k = 2j (i.e., k-j = j): a[j]^2 (quadratic in a[j])
- Otherwise: 0

So the max convolution as a function of a[j] (with a[j] >= 0) is the max of:
- Some constants (convolution values that don't involve a[j])
- Some linear functions: 2*a[j]*a[k-j] + (constant part of conv_k) for various k
- One quadratic: a[j]^2 + (constant part of conv_{2j}) [if 2j < 2n-1, i.e., j <= n-1, which is always true]

Wait, actually the quadratic term a[j]^2 appears in conv(a,a)_{2j} only. And it's the only quadratic term. All other terms are linear in a[j].

So the objective (max of convolution) as a function of a[j] is:
max( max_k [c_k + 2*a[j]*a[k-j]], a[j]^2 + c_{2j} )

where c_k is the part of conv_k not involving a[j].

This is the max of a set of linear functions and one quadratic function, all convex. So it's convex in a[j]. The minimum is where the derivative changes sign.

The derivative w.r.t. a[j] is:
- 2*a[k-j] for the linear terms (where k-j is valid and k-j != j)
- 2*a[j] for the quadratic term

The subgradient of the max is the max of the derivatives of the active (maximum) terms.

To find the optimal a[j], I need to find where the subgradient contains 0 (or the constraint a[j] >= 0 is active).

This is doable! The approach:
1. For each j, compute the "candidate slopes" (2*a[k-j] for all valid k) and the "quadratic slope" (2*a[j] at the current point).
2. The optimal a[j]* is where the max of the active linear functions equals the quadratic, or where the derivative of the max changes sign.

"""

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_objective_and_grad(a, N, beta):
    """Compute LSE-smoothed max of conv and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    # Numerical stability: shift by c_max
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        # Only the max is active
        w = np.zeros_like(c)
        w[c == c_max] = 1.0 / np.sum(c == c_max)
    else:
        w = exp_bc / total
    lse_val = c_max + np.log(max(total, 1e-300)) / beta
    
    # Gradient via FFT
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max

def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def lbfgs_block(a_init, n, S, N, max_time, rng, beta_start=10.0, beta_end=500.0, m_lbfgs=15):
    """L-BFGS on LSE-smoothed objective."""
    a = project_simplex(a_init.copy(), S)
    beta = beta_start
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, f_val, c, c_max = lse_objective_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    start_time = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / max_time
        # Ramp beta from beta_start to beta_end
        beta = beta_start + (beta_end - beta_start) * min(1.0, frac * 1.5)
        beta = min(beta, 2000.0)
        
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search: backtracking
        step = S / (n * 0.01)
        a_new = a
        cmax_new = c_max
        accepted = False
        
        for ls in range(20):
            candidate = project_simplex(a + step * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1 + 1e-12) or ls == 19:
                a_new = candidate
                cmax_new = cm_ls
                accepted = True
                if cm_ls <= c_max:
                    break
            step *= 0.5
        
        if not accepted:
            break
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update L-BFGS history
        g_new, _, _, _ = lse_objective_and_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[1:]
                y_hist = y_hist[1:]
        
        a = a_new
        g = g_new
        c_max = cmax_new
        
        # Periodic perturbation to escape local minima
        if it % 300 == 0 and rng.random() < 0.15:
            noise_scale = 0.005 * S / np.sqrt(n) * (1 + no_improve * 0.1)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            s_hist = []
            y_hist = []
            g, _, _, c_max = lse_objective_and_grad(a, N, beta)
            no_improve = 0
        
        # Truncation
        if len(s_hist) >= m_lbfgs and it % 100 == 0:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
    
    return best_a, best_cmax

def coordinate_descent_polish(a_init, n, S, N, max_time, rng, max_sweeps=100):
    """Coordinate descent: optimize one coordinate at a time."""
    a = project_simplex(a_init.copy(), S)
    c = compute_conv(a, N)
    c_max = c.max()
    best_a = a.copy()
    best_cmax = c_max
    start_time = time.monotonic()
    
    for sweep in range(max_sweps):
        if time.monotonic() - start_time > max_time:
            break
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        for j in order:
            # Try to optimize a[j]
            # The max conv as a function of a[j] is convex
            # Binary search for optimal a[j]
            if a[j] < 1e-15:
                # Try small positive values
                for trial in [1e-10, 1e-8, 1e-6, 1e-4]:
                    a[j] = trial
                    # Recompute affected convolution values
                    # Only need to check if max decreased
                    c_j = compute_conv(a, N)
                    if c_j.max() < c_max:
                        c = c_j
                        c_max = c.max()
                        improved = True
                        break
                    a[j] = 0
                continue
            
            # Binary search: the optimal a[j] is where the subgradient of max changes sign
            # Approximate: try reducing a[j] and see if max decreases
            for factor in [0.8, 0.5, 0.2, 0.1, 1.2, 1.5, 2.0]:
                a_new_j = a[j] * factor
                if a_new_j < 0 or a_new_j > 1000:
                    continue
                a[j] = a_new_j
                c_j = compute_conv(a, N)
                cm = c_j.max()
                if cm < c_max:
                    c = c_j
                    c_max = cm
                    improved = True
                    break
                a[j] = a[j] / factor  # restore
            
            if a[j] != 0:
                # Renormalize
                a = project_simplex(a, S)
        
        a = project_simplex(a, S)
        c_final = compute_conv(a, N)
        if c_final.max() < best_cmax:
            best_cmax = c_final.max()
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_cmax

def spectral_equalize(a_init, n, S, N, max_time, rng, n_iter=5000):
    """Iteratively equalize the power spectrum to flatten autocorrelation."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > max_time:
            break
        
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Get power spectrum
        A = fft(a, N)
        P = np.abs(A) ** 2
        
        # Compute "target" spectrum: we want the autocorrelation to be flat
        # The autocorrelation is IDFT(P) (circular), so we want P to be flat
        # But we need to maintain a >= 0, so we do a partial equalization
        P_target = P ** 0.5  # Geometric mean with flat
        P_target = 0.5 * P + 0.5 * P.mean()
        
        # Blend
        alpha = 0.05
        P_new = P * (1 - alpha) + P_target * alpha
        P_new = np.maximum(P_new, 0)
        
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
    
    return best_a, best_cmax

def peak_shaving(a_init, n, S, N, max_time, rng, top_k=5, n_iter=10000):
    """Target the top peaks in the convolution directly."""
    a = project_simplex(a_init.copy(), S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > max_time:
            break
        
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Get top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        weights = (top_vals - c.mean()) / max((top_vals - c.mean()).sum(), 1e-10)
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-10:
            break
        weights = weights / weights.sum()
        
        # Compute gradient of top-k weighted sum
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Gradient step with projection
        step = 2.0 * S / (n * np.linalg.norm(grad) + 1e-10)
        a_new = project_simplex(a - step * grad, S)
        a = a_new
        
        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

def make_init_rand(n, rng):
    return rng.random(n) + 0.05

def make_init_tapered(n, rng, taper=0.3):
    """Initialize with a tapered shape (more weight in middle or edges)."""
    t = np.linspace(0, 1, n)
    # Tapered: more in middle
    base = 1.0 - taper * (2 * t - 1) ** 2
    return np.clip(base + 0.05 * rng.standard_normal(n), 0.01, None)

def make_init_oscillating(n, rng, freq=7):
    """Initialize with oscillation to create spectral gaps."""
    t = np.linspace(0, 1, n, endpoint=False)
    base = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t + rng.random() * 2 * np.pi)
    return np.clip(base + 0.05 * rng.standard_normal(n), 0.01, None)

def make_init_fourier_random(n, N, rng):
    """Random Fourier phase initialization."""
    P = rng.exponential(1.0, N)
    P[0] = n
    phase = rng.uniform(0, 2 * np.pi, N)
    # Enforce Hermitian symmetry for real signal
    A = np.sqrt(P) * np.exp(1j * phase)
    A = (A + np.conj(A[np.mod(-np.arange(N), N)])) / 2
    a = np.real(ifft(A))[:n]
    return np.clip(a, 0.01, None)
