def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def solve_lp_cutting_plane(f, n, S, M, max_iters=50, solver='SCS'):
    """Solve the LP using cutting plane (constraint generation)."""
    active_constraints = []  # list of lag indices
    g = None
    
    for it in range(max_iters):
        # Solve LP with current constraints
        if len(active_constraints) == 0:
            # No constraints: solution is g = ones (maximize sum)
            g = np.ones(n)
        else:
            # Build constraint matrix
            rows = np.zeros((len(active_constraints), n))
            for i, k in enumerate(active_constraints):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                js = np.arange(lo, hi + 1)
                rows[i, js] = f[k - js]
            
            g_var = cp.Variable(n)
            constraints = [rows @ g_var <= M, g_var >= 0]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except:
                prob.solve(solver='ECOS', verbose=False)
            
            if g_var.value is None:
                return None
            g = np.maximum(g_var.value, 0)
        
        if g is None or g.sum() < 1e-10:
            return None
        
        # Check for violated constraints
        fg = fft_convolve_self_full(f, g)  # need cross-convolution
        violations = np.where(fg > M * 1.001)[0]
        
        if len(violations) == 0:
            break
        elif len(violations) <= 10:
            # Add all violated constraints
            active_constraints = list(set(active_constraints + list(violations)))
        else:
            # Add worst 10 violations
            worst = violations[np.argsort(fg[violations])[-10:]]
            active_constraints = list(set(active_constraints + list(worst)))
    
    return g

def fft_convolve_self(a):

    fsize = 1 << (max(size, 64) - 1).bit_length()

def fft_convolve(a, b):

    m = len(b)
    size = n + m - 1

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def accelerated_subgradient(f_init, n, S, max_time, rng, top_k=100, temp_init=0.05, 
                            decay=0.999, temp_min=0.005, use_polyak=True, lb=1.0):
    """Accelerated subgradient method (Nesterov) for minimizing max(f*f)."""
    f = normalize_sum(f_init, n)
    y = f.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        temp = max(temp_min, temp_init * (decay ** it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_f = y.copy()
        
        grad, M = compute_subgradient(y, c, top_k=top_k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size
        if use_polyak:
            step = (M - lb) / (gnorm * gnorm + 1e-10)
            step = min(step, 1.0 / (2.0 * gnorm))  # cap the step
        else:
            step = 2.0 / (best * (it + 2))  # Nesterov step
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new
        
        # Periodic noise to escape
        if it % 5000 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_f, best

def lp_iteration(f, n, S, solver='SCS', time_limit=15):
    """One LP iteration: solve LP, find best step size."""
    c = fft_convolve_self(f)
    M = c.max()
    
    g = solve_lp_cutting_plane(f, n, S, M, max_rounds=10, solver=solver, time_limit=time_limit)
    if g is None:
        return f.copy(), M
    
    g = g * (S / g.sum())  # Rescale to same sum
    
    # Find best step size
    def eval_t(t_val):
        f_test = (1 - t_val) * f + t_val * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()
    
    # Try a few values
    best_t = 0.0
    best_val = M
    for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
        val = eval_t(t_val)
        if val < best_val:
            best_val = val
            best_t = t_val
    
    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val
    
    return f.copy(), M

"""
Minimize 2*n*max(f*f)/sum(f)^2 using a combination of:
1. Cutting-plane LP best response (main driver)
2. Accelerated subgradient with softmax smoothing (polishing)
3. Multi-scale exploration (different n values)
4. Periodic mirror perturbations to escape local minima

Key improvements over previous:
- Adaptive cutting-plane LP that builds constraint set from most-violated
- Better step size selection via quadratic interpolation
- Lower-bounded Polyak steps for subgradient
- Interpolation between n values for diverse restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (max(size, 64) - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (max(size, 64) - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient(f, c, top_k=100, temperature=0.02):
    n = len(f)
    M = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    w_top = np.exp((c[idx] - M) / temperature)
    w_top /= w_top.sum()
    w_full = np.zeros(2 * n - 1)
    w_full[idx] = w_top
    g_rev = f[::-1]
    fsize_g = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w_full, fsize_g)
    G = fft(g_rev, fsize_g)
    conv = np.real(ifft(W * G))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, M


def accelerated_subgradient(f_init, n, S, max_time, rng, top_k=100,
                            temp_init=0.08, decay=0.9995, temp_min=0.008,
                            use_polyak=True, lb=1.0, noise_period=3000):
    f = normalize_sum(f_init, n)
    y = f.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        temp = max(temp_min, temp_init * (decay ** it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_f = y.copy()
            last_improve = it

        grad, M = compute_subgradient(y, c, top_k=top_k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        if use_polyak:
            step = (M - lb) / (gnorm * gnorm + 1e-10)
            step = min(step, 5.0 / (gnorm + 1e-10))
        else:
            step = 2.0 / (np.sqrt(2.0 * n) * (it + 2))

        if it - last_improve > noise_period:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t = t_new

    return best_f, best


def solve_lp_cutting_plane(f, n, S, M, max_rounds=30, solver='SCS', time_limit=20,
                           max_constraints=600):
    c = fft_convolve_self(f)
    M = c.max()
    active = list(np.argsort(c)[::-1][:80])
    g = None
    start_time = time.monotonic()

    for round_i in range(max_rounds):
        if time.monotonic() - start_time > time_limit:
            break

        if len(active) == 0:
            g = np.ones(n) * S / n
        else:
            rows = np.zeros((len(active), n))
            for i, k in enumerate(active):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                js = np.arange(lo, hi + 1)
                rows[i, js] = f[k - js]

            g_var = cp.Variable(n)
            constraints = [rows @ g_var <= M, g_var >= 0]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

            solved = False
            for sol in [solver, 'ECOS', 'SCS']:
                if solved:
                    break
                try:
                    prob.solve(solver=sol, verbose=False)
                    solved = True
                except Exception:
                    continue

            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
            else:
                break

        if g is None or g.sum() < 1e-10:
            return None

        fg = fft_convolve(f, g)
        violations = np.where(fg > M * 1.0005)[0]

        if len(violations) == 0:
            break

        if len(violations) > 30:
            worst = violations[np.argsort(fg[violations])[-30:]]
        else:
            worst = violations

        new_active = list(set(active + list(worst)))
        if len(new_active) > max_constraints:
            fg_vals = fg[new_active]
            keep = np.argsort(fg_vals)[::-1][:max_constraints]
            new_active = [new_active[i] for i in keep]
        active = new_active

    return g


def lp_step(f, n, S, solver='SCS', time_limit=15):
    c = fft_convolve_self(f)
    M = c.max()
    g = solve_lp_cutting_plane(f, n, S, M, max_rounds=15, solver=solver,
                                time_limit=time_limit)
    if g is None or g.sum() < 1e-10:
        return f.copy(), M

    g = g * (S / g.sum())
    g = np.maximum(g, 0)
    g = project_simplex(g, S)

    def eval_t(t_val):
        f_test = (1.0 - t_val) * f + t_val * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        return fft_convolve_self(f_test).max()

    best_t = 0.0
    best_val = M
    for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        val = eval_t(t_val)
        if val < best_val:
            best_val = val
            best_t = t_val

    if best_t > 0.01:
        f_new = (1.0 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val

    return f.copy(), M


def interpolate_to_length(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def multiplicative_update(f, n, S, beta=30.0, lr=0.05):
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    g_rev = f[::-1]
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, fsize)
    G = fft(g_rev, fsize)
    rep = np.real(ifft(W * G))[n - 1:2 * n - 1]
    rep_norm = rep / max(rep.max(), 1e-10)
    f_new = f * np.exp(-lr * rep_norm)
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new


def compute_objective(f, n):
    c = fft_convolve_self(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)
