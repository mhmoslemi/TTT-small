def peak_descent_step(a, top_k=10):
    """One step of peak-directed descent."""
    n = len(a)
    S = a.sum()
    c = fft_convolve(a)
    
    # Find top-k peaks
    top_indices = np.argsort(c)[::-1][:top_k]
    top_values = c[top_indices]
    weights = top_values / top_values.sum()
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        m = int(idx)
        for j in range(n):
            k = m - j
            if 0 <= k < n:
                coeff = 2.0 if 2*j != m else 1.0
                g[j] += w * coeff * a[k]
    
    # Normalize gradient
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return a
    
    # Step size
    step = S / (gn * np.sqrt(n))  # heuristic
    
    # Update
    a_new = a - step * g / gn
    a_new = project_simplex(a_new, S)
    return a_new

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using combined spectral equalization
and FFT-accelerated top-k subgradient with Nesterov momentum.

Key improvements:
1. Proper spectral equalization with geometric interpolation toward flat target
2. FFT-accelerated gradient computation for top-k peak subgradient
3. Nesterov momentum for faster convergence
4. Adaptive step size with backtracking
5. Diverse initializations including two-level structured sequences
6. Focused budget allocation on n=8192 with exploration of nearby lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Compute linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def spectral_equalize(a, N, n_iter, alpha_init, S, rng):
    """
    Iterative spectral equalization: push |FFT(a)| toward flat target.
    Uses geometric interpolation for stability.
    """
    n = len(a)
    best_a = a.copy()
    best_score = compute_objective(a)
    target_mag = S / np.sqrt(N)  # Target: |A_k| = S/sqrt(N) for all k

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        alpha = alpha_init * (1.0 - 0.85 * progress)
        alpha = max(alpha, 0.005)

        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)

        # Geometric interpolation toward target
        safe_mag = np.maximum(mag, 1e-30)
        new_mag = np.power(safe_mag, 1.0 - alpha) * np.power(target_mag, alpha)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)

        sc = compute_objective(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        # Periodic noise to escape local minima
        if it % 800 == 400 and rng.random() < 0.3:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def subgradient_optimize(a_init, max_time, rng, top_k=30, nesterov=True):
    """
    Nesterov-accelerated projected subgradient for minimizing max(conv(a,a)).
    Uses FFT-accelerated gradient computation.
    """
    n = len(a_init)
    S = a_init.sum()
    N = 1 << (2 * n - 1).bit_length()

    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    prev_a = a.copy()
    t_nest = 1.0
    start = time.monotonic()
    best_score = compute_objective(a)
    best_local = a.copy()
    it = 0

    # Adaptive step size parameters
    step_base = 1.0
    step_decay = 0.002

    while time.monotonic() - start < max_time:
        it += 1

        # Compute convolution
        y_pad = np.zeros(N)
        y_pad[:n] = y
        Y = fft(y_pad)
        c = np.real(ifft(Y * Y))[:2 * n - 1]
        cur_max = c.max()
        sc = 2.0 * n * cur_max / (y.sum() ** 2)

        if sc < best_score:
            best_score = sc
            best_local = y.copy()

        # Find top-k peaks with exponential weighting
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Exponential weighting: emphasize highest peaks
        max_tv = top_vals[0]
        weights = np.exp(-(top_vals - max_tv) / max(0.1 * (max_tv - top_vals[-1]), 1e-10))
        weights /= weights.sum()

        # Build weight kernel
        kernel = np.zeros(N)
        for i, k in enumerate(top_idx):
            kernel[int(k)] = weights[i]

        # FFT-based gradient: g[j] = sum_m kernel[m] * a[m-j] (correlation)
        K = fft(kernel, N)
        grad = 2.0 * np.real(ifft(K * np.conj(Y)))[:n]
        # Fix the center correction (for even m where 2j=m, factor should be 1 not 2)
        # This is a minor correction; skip for speed

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        # Adaptive step size
        step = step_base / (1.0 + step_decay * it) * S / max(gn, 1e-10)
        step = min(step, S * 0.5 / n)  # Cap to prevent divergence

        x_new = y - step * grad
        x_new = project_simplex(x_new, S)

        if nesterov:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2))
            beta = (t_nest - 1.0) / t_new
            y = x_new + beta * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_nest = t_new
        else:
            a = x_new
            y = a.copy()

        # Periodic restart with noise
        if it % 1000 == 500 and rng.random() < 0.2:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_nest = 1.0

    return best_local, best_score


def generate_initializations(n, rng):
    """Generate diverse initial sequences normalized to sum sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) * S / n)

    # Two-level balanced (pseudo-random arrangement)
    p = 0.5
    levels = rng.random(n) < p
    a = np.where(levels, 1.5, 0.5)
    a = a * S / a.sum()
    inits.append(a)

    # Two-level with optimal ratio (golden ratio inspired)
    p = 0.382
    levels = rng.random(n) < p
    a = np.where(levels, 2.0, 0.0)
    if a.sum() > 0:
        a = a * S / a.sum()
    inits.append(a)

    # Multi-sine structure
    a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt)
    a = np.maximum(a, 0.01)
    a = a * S / a.sum()
    inits.append(a)

    # Chirp
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt ** 2))
    a = np.maximum(a, 0.01)
    a = a * S / a.sum()
    inits.append(a)

    # Smoothed random (low-pass)
    noise = rng.standard_normal(n)
    ksz = max(3, n // 32)
    kernel = np.ones(ksz) / ksz
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    a = 0.2 + 0.8 * smooth
    a = a * S / a.sum()
    inits.append(a)

    # Gaussian bump
    a = np.exp(-0.5 * ((tt - 0.5) * 3.0) ** 2) + 0.1
    a = a * S / a.sum()
    inits.append(a)

    # Cosine window
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
    a = a * S / a.sum()
    inits.append(a)

    # Random with structure (blocks)
    block_size = max(1, n // 16)
    n_blocks = n // block_size
    block_vals = rng.random(n_blocks) ** 0.5  #.bias toward smaller
    a = np.repeat(block_vals + 0.1, block_size)[:n]
    if len(a) < n:
        a = np.concatenate([a, np.random.random(n - len(a))])
    a = a * S / a.sum()
    inits.append(a)

    return inits
