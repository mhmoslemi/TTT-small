def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def grad_single_peak(a, N, k):
    """Gradient of b[k] w.r.t. a, where b = conv(a, a)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a

    freq = np.exp(-2j * np.pi * np.arange(N) * k / N)

    return grad

    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += (top_vals[i] / total) * np.cos(2.0 * np.pi * m_arr * int(k) / N)
    # Reduce the power where it contributes to peaks
    P = np.abs(A) ** 2
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

def peak_descent(a, N, S, max_time, rng, tol=0.01, max_iter=100000):
    """Active-set peak descent with backtracking."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = project_simplex(a, S)
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        # Find active peaks (within tol of max)
        threshold = c_max * (1 - tol)
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]
        active_vals = c[active_idx]
        weights = active_vals / active_vals.sum()
        
        # Compute gradient
        grad = grad_weighted_peaks(a, N, active_idx, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Backtracking line search
        # Target: reduce c_max by at least 0.1%
        target_reduction = 0.001 * c_max
        # Initial step: alpha such that alpha * gnorm^2 ≈ target_reduction
        alpha = min(1.0, target_reduction / max(gnorm * gnorm, 1e-30))
        alpha = max(alpha, 1e-6)
        
        improved = False
        for ls in range(20):
            a_new = project_simplex(a - alpha * grad, S)
            c_new = compute_conv(a_new, N)
            cmax_new = c_new.max()
            if cmax_new < c_max - 1e-15:
                a = a_new
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try a smaller fixed step or random perturbation
            if rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(len(a)), len(a)), S)
            else:
                # Try negative gradient with small step
                small_step = 0.01 * S / np.sqrt(len(a))
                a = project_simplex(a - small_step * grad / gnorm, S)
        
        # Periodic frequency flattening
        if it % 500 == 0:
            a = freq_flatten(a, N, alpha=0.1, top_k=15)
            a = project_simplex(a, S)
        
        # If stuck, try larger perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(len(a)), len(a)), S)
    
    return best_a, best_cmax

def lse_lbfgs_block(a_init, N, S, max_time, rng, beta_init=20.0, beta_max=500.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective."""

    def lse_grad_at(a_curr, beta_curr):
        c = compute_conv(a_curr, N)
        c_max = c.max()
        exp_bc = np.exp(beta_curr * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            return np.zeros(n), c_max, c, c_max
        w = exp_bc / total
        lse_val = c_max + np.log(total) / beta_curr
        fsize = 1 << (3 * n).bit_length()
        W = fft(w, fsize)
        Af = fft(a_curr, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        return grad, lse_val, c, c_max

        m = len(s_h)
        n_curr = len(g_curr)
        gn = np.linalg.norm(g_curr)
        if m == 0 or gn < 1e-18:
            return -g_curr / gn if gn > 1e-18 else np.zeros(n_curr)
        q = g_curr.copy()

            y_i = y_h[i]

        y_last = y_h[-1]

            y_i = y_h[i]

        if np.dot(g_curr, direction) > 0:

        return direction

                best_cmax = c_max
                best_a = a.copy()

        for ls in range(15):

            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 14:
                a_new = candidate
                c_new = c_ls
                cmax_new = cm_ls

        s_k = a_new - a

        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)

            y_hist.append(y_k)

                y_hist = y_hist[-m_lbfgs:]

        g = g_new
        c = c_nw
        c_max = cmax_nw
    
    return best_a, best_cmax

def make_inits(n, N, rng, S):
    """Generate diverse initializations."""
    inits = []
    # Random
    inits.append(("rand", project_simplex(rng.random(n) + 0.01, S)))
    # Chirp
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 2 * tt * tt)) + 0.05 * rng.standard_normal(n)
    inits.append(("chirp", project_simplex(np.clip(a, 0.01, None), S)))
    # Fourier
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    inits.append(("fourier", project_simplex(np.abs(a_full) + 0.01, S)))
    # Triangle (anti-symmetric)
    t = np.arange(n)
    a = (n - 1 - t).astype(float) / n + 0.01
    inits.append(("triangle", project_simplex(a, S)))
    # Bimodal
    a = np.exp(-((tt - 0.25) ** 2) / 0.01) + np.exp(-((tt - 0.75) ** 2) / 0.01)
    inits.append(("bimodal", project_simplex(a + 0.01, S)))
    return inits
