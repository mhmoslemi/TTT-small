"""

4. Multi-size exploration: try n = 4096, 8192, 16384 and keep the best.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_gradient(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a) via FFT."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def peak_clipping_step(a, c, N, top_k=10, transfer_frac=0.05):
    """
    Frank-Wolfe style peak reduction:
    - Identify top-K peaks
    - Transfer mass from high-contribution positions to low-impact positions
    """
    n = len(a)
    S = a.sum()
    top_idx = np.argsort(c)[::-1][:top_k]
    peak_vals = c[top_idx]
    
    # Softmax weights for peaks (sharper = more focus on max)
    temp = max(1e-6, (peak_vals[0] - peak_vals[-1]) * 0.5)
    weights = np.exp(-(peak_vals - peak_vals.max()) / temp)
    weights /= weights.sum()
    
    # Compute combined gradient (weighted sum of peak gradients)
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            jj = k - j
            if 0 <= jj < n:
                grad[j] += weights[i] * a[jj]
                if jj != j:
                    grad[jj] += weights[i] * a[j]
    
    # Find positions to reduce (high gradient) and add (low "cost")
    # Cost of adding to position j: max over k of a[k-j] (approximate by local max)
    # Approximate cost using the convolution itself
    cost = np.zeros(n)
    for j in range(n):
        # The impact of a[j] on all c[k] is proportional to a shifted
        # Approximate: use the convolution values near the center
        lo_c = max(0, j)
        hi_c = min(2 * n - 2, j + n - 1)
        # Use a simplified cost: the value of a at positions that interact with j
        cost[j] = a[j]  # simple cost: reduce high values, add to low values
    
    # Use gradient magnitude as reduction priority
    grad_abs = np.abs(grad)
    
    # Reduce from top contributing positions
    n_reduce = max(1, n // 50)
    reduce_idx = np.argsort(grad_abs)[::-1][:n_reduce]
    # Add to bottom cost positions
    n_add = max(1, n // 50)
    add_idx = np.argsort(cost)[:n_add]
    
    # Ensure no overlap
    reduce_set = set(reduce_idx.tolist())
    add_idx = [x for x in add_idx if x not in reduce_set]
    if len(add_idx) == 0:
        add_idx = [x for x in np.argsort(cost)[n_add:2*n_add] if x not in reduce_set]
    
    if len(add_idx) == 0:
        return a
    
    transfer_idx = np.array(add_idx[:len(reduce_idx)])
    
    # Transfer mass
    total_reduce = 0.0
    for j in reduce_idx:
        amount = min(a[j] * transfer_frac, grad_abs[j] * 0.01 * S / max(1, grad_abs.max()))
        amount = min(amount, a[j] * 0.5)
        a[j] -= amount
        total_reduce += amount
    
    # Distribute to add positions (proportional to inverse cost)
    if total_reduce > 1e-15:
        add_costs = np.maximum(1e-10, cost[transfer_idx])
        add_weights = 1.0 / add_costs
        add_weights /= add_weights.sum()
        for j, w in zip(transfer_idx, add_weights):
            a[j] += total_reduce * w
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, f_new, c_nw, cmax_nw = lse_gradient(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        if it % 500 == 0 and rng.random() < 0.2:
            A = fft(a, N)
            mag = np.abs(A)
            old_phase = np.angle(A)
            n_rand = max(1, int(N * 0.3))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            old_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
            a_new_sp = np.real(ifft(mag * np.exp(1j * old_phase)))[:n]
            a_new_sp = np.maximum(a_new_sp, 0)
            s_sp = a_new_sp.sum()
            if s_sp > 1e-15:
                a = a_new_sp * (S / s_sp)
                s_hist = []
                y_hist = []
                g, f_val, c, c_max = lse_gradient(a, N, beta)
    
    return best_a, best_cmax


def nesterov_polish(a_init, n, S, N, max_time, rng, top_k=50):
    """Nesterov accelerated subgradient for final polishing."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        current_k = min(top_k + it // 20, 200)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argsort(c)[::-1][:current_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            if int(k) < fsize:
                w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        if it % 5 == 0:
            c_x = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        
        if it % 2000 == 0 and rng.random() < 0.1:
            A = fft(y, N)
            mag = np.abs(A)
            new_phase = rng.uniform(0, 2 * np.pi, N)
            n_keep = int(N * 0.5)
            keep_idx = np.argsort(np.abs(A))[:n_keep]
            old_phase = np.angle(A)
            new_phase[keep_idx] = old_phase[keep_idx]
            a_sp = np.real(ifft(mag * np.exp(1j * new_phase)))[:n]
            a_sp = np.maximum(a_sp, 0)
            s_sp = a_sp.sum()
            if s_sp > 1e-15:
                y = a_sp * (S / s_sp)
                t = 1.0
    
    return best_a, best_cmax


def deflation_init(n, S, N, n_iters=3000, rng=None):
    """Deflation-based constructive initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.ones(n) * S / n
    
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        
        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        
        # Find position with minimum impact
        min_impact = np.inf
        i_min = 0
        for j in range(n):
            if j == i_max:
                continue
            # Approximate impact: how much would adding to j increase the peak?
            lo_j = max(0, k_star - n + 1)
            hi_j = min(n - 1, k_star)
            impact = a[max(0, min(n-1, k_star - j))] if lo_j <= j <= hi_j else 0
            if impact < min_impact:
                min_impact = impact
                i_min = j
        
        if i_max == i_min:
            break
        transfer = min(0.01 * a[i_max], 0.2 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (7 * tt + 2 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    i = np.arange(n)
    c = rng.integers(1, n)
    phase = -np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for freq in [3, 7, 13, 23, 41]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_windowed(n, rng):
    """Raised cosine window with noise."""
    tt = np.linspace(0, 1, n)
    a = 0.5 * (1 - np.cos(2 * np.pi * tt))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def solve_size(n, S, N, budget, rng, given_seq=None):
    """Solve for a specific size n within the given budget."""
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = np.maximum(a_candidate, 0).astype(np.float64)
        s_c = a_cand.sum()
        if s_c < 1e-15:
            return
        a_cand = a_cand * (S / s_c)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            print("    [%s] n=%d peak=%.8f score=%.6f" % (label, n, sc, score))
        return sc
    
    t_start = time.monotonic()
    
    # Phase 1: Fast peak reduction from multiple starts
    inits = []
    if given_seq is not None and len(given_seq) == n:
        inits.append(("given", np.clip(given_seq, 0, 1000).astype(np.float64)))
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("multi_freq", make_init_multi_freq(n, rng)))
    inits.append(("windowed", make_init_windowed(n, rng)))
    
    phase1_time = budget * 0.35
    per_init = phase1_time / len(inits)
    
    for name, arr in inits:
        if time.monotonic() - t_start > t_start + phase1_time - 5:
            break
        a_norm = np.maximum(arr, 0)
        s_a = a_norm.sum()
        if s_a < 1e-15:
            a_norm = np.ones(n) * S / n
        else:
            a_norm = a_norm * (S / s_a)
        
        rem = max(1.0, per_init - 0.5)
        a_r, sc_r = fast_peak_reduction(a_norm, n, S, N, min(rem, 80.0), rng,
                                         top_k=8, transfer_frac=0.04)
        update_best(a_r, name + "/peak")
    
    # Also try deflation
    if time.monotonic() - t_start < t_start + phase1_time:
        a_def = deflation_init(n, S, N, 2000, rng)
        update_best(a_def, "deflation")
        rem = max(1.0, phase1_time - (time.monotonic() - t_start))
        a_r, sc_r = fast_peak_reduction(a_def, n, S, N, min(rem, 60.0), rng,
                                         top_k=10, transfer_frac=0.03)
        update_best(a_r, "deflation/peak")
    
    if global_a is None:
        global_a = np.ones(n) * S / n
        c = compute_conv(global_a, N)
        global_best = c.max()
    
    # Phase 2: L-BFGS refinement
    phase2_time = budget * 0.35
    t2_start = time.monotonic()
    a_curr = global_a.copy()
    stall = 0
    
    while time.monotonic() - t2_start < phase2_time:
        block_time = min(120.0, (t2_start + phase2_time) - time.monotonic())
        if block_time < 10:
            break
        beta_start = 10.0 + 5.0 * min(stall, 8)
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=min(500.0, 100 + 50 * stall))
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2 * n * sc_new / (S * S)
            print("    [P2] n=%d score=%.6f (beta=%.0f)" % (n, score, beta_start))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                # Spectral restart
                A = fft(a_curr, N)
                mag = np.abs(A)
                new_phase = rng.uniform(0, 2 * np.pi, N)
                a_sp = np.real(ifft(mag * np.exp(1j * new_phase)))[:n]
                a_sp = np.maximum(a_sp, 0)
                s_sp = a_sp.sum()
                if s_sp > 1e-15:
                    a_curr = a_sp * (S / s_sp)
                stall = 0
            elif stall >= 4:
                a_curr = np.ones(n) * S / n
                # Quick peak reduction
                a_curr, _ = fast_peak_reduction(a_curr, n, S, N, 30.0, rng)
                stall = 0
    
    # Phase 3: Nesterov polishing
    remaining3 = min(budget * 0.2, (budget - (time.monotonic() - t_start)) - 20)
    if remaining3 > 15:
        a_ng, sc_ng = nesterov_polish(global_a, n, S, N, remaining3, rng, top_k=30)
        update_best(a_ng, "nesterov")
    
    # Phase 4: Final L-BFGS with very high beta
    remaining4 = max(0.0, budget - (time.monotonic() - t_start) - 10)
    if remaining4 > 15:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining4, 120.0),
                                         rng, beta_init=50.0, beta_max=800.0)
        update_best(a_fin, "final_LBFGS")
    
    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses iterative peak reduction (Frank-Wolfe style) + L-BFGS + Nesterov.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    # Try to load given sequence
    given_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if gs.size >= 1024 and gs.size <= 32768:
            given_seq = gs
            print("Using given sequence of length %d" % gs.size)
    except Exception:
        pass
    
    global_best_score = np.inf
    global_best_seq = None
    
    # Try multiple sizes
    sizes_to_try = []
    if given_seq is not None:
        sizes_to_try.append(len(given_seq))
    sizes_to_try.extend([4096, 8192, 16384])
    # Remove duplicates, keep order
    seen = set()
    unique_sizes = []
    for s in sizes_to_try:
        if s not in seen and 1024 <= s <= 32768:
            seen.add(s)
            unique_sizes.append(s)
    
    print("Trying sizes: %s" % unique_sizes)
    
    # Allocate time proportionally to sizes (give more time to the given size if present)
    total_weight = 0
    weights = []
    for sz in unique_sizes:
        if given_seq is not None and sz == len(given_seq):
            w = 3.0
        else:
            w = 1.0
        weights.append(w)
        total_weight += w
    
    time_alloc = []
    for w in weights:
        time_alloc.append(budget_s * 0.9 * w / total_weight)
    
    for i, n in enumerate(unique_sizes):
        if time.monotonic() > deadline:
            break
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        alloc = min(time_alloc[i], deadline - time.monotonic() - 10)
        if alloc < 30:
            continue
        
        print("=== Solving n=%d (S=%.2f, N=%d, alloc=%.0fs) ===" % (n, S, N, alloc))
        
        a_best, peak_best = solve_size(n, S, N, alloc, rng, given_seq)
        
        if a_best is not None:
            score = 2.0 * n * peak_best / (S * S)
            if score < global_best_score:
                global_best_score = score
                global_best_seq = a_best.copy()
                print("  New global best: n=%d, score=%.6f" % (n, score))
    
    # If we haven't found anything, use a fallback
    if global_best_seq is None:
        print("Fallback to all-ones n=256")
        global_best_seq = np.ones(256, dtype=np.float64)
        global_best_score = 2.0
    
    # Final verification
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)
    if global_best_seq.sum() < 0.01:
        global_best_seq = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_best_seq]
    ver = evaluate_sequence(final_list)
    print("Final verification: %.6f (length %d)" % (ver, len(final_list)) if np.isfinite(ver) else "Verification: inf")
    
    if not np.isfinite(ver):
        print("  Rejected! Using safe fallback.")
        fallback = np.ones(256, dtype=np.float64)
        ver2 = evaluate_sequence([float(x) for x in fallback])
        if np.isfinite(ver2):
            return [float(x) for x in fallback]
        # Last resort
        return [1.0] * 256
    
    print("Returning sequence of length %d with score %.6f" % (len(final_list), ver))
    return final_list
