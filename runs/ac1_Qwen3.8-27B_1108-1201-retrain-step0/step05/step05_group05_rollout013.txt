   """
   Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional search.
   
   Key insight: At each step, solve an LP to find the best feasible direction
   that keeps the convolution bounded, then take a line-search step.

   import numpy as np
   import time
   from scipy.fft import fft, ifft
   from scipy.optimize import linprog
   
   def fft_convolve_self(a):
       n = len(a)
       size = 2 * n - 1
       fsize = 1 << (size - 1).bit_length()
       A = fft(a, fsize)
       c = np.real(ifft(A * A))
       return c[:size]
   
   def project_simplex(v, S):
       n = len(v)
       v = np.asarray(v, dtype=np.float64).copy()
       w = np.sort(v)[::-1]
       cumsum = np.cumsum(w)
       idx = np.arange(1, n + 1, dtype=np.float64)
       tc = (cumsum - S) / idx
       valid = tc <= w
       if not np.any(valid):
           rho = n
       else:
           rho = int(np.max(np.where(valid)[0]) + 1)
       theta = (cumsum[rho - 1] - S) / rho
       return np.maximum(v - theta, 0.0)
   
   def normalize_to_sum(a, S):
       a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
       s = a.sum()
       if s < 1e-15:
           return np.full(len(a), S / len(a))
       return a * (S / s)
   
   def objective_value(a):
       a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
       n = len(a)
       s = a.sum()
       if s < 0.01:
           return np.inf
       b = fft_convolve_self(a)
       return 2.0 * n * b.max() / (s * s)
   
   def solve_lp_direction(a, M, active_lags):
       """
       Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0
       Returns the optimal b (or None if solver fails).
       """

       m = len(active_lags)
       if m == 0:
           return None
       
       A_ub = np.zeros((m, n))
       for i, k in enumerate(active_lags):
           lo = max(0, k - n + 1)
           hi = min(k, n - 1)
           if lo > hi:

           j_idx = np.arange(lo, hi + 1)
           b_idx = k - j_idx  # = k - lo, ..., k - hi

           return None
       return np.maximum(res.x, 0.0)

       """
       Find step t in [0, 1] that minimizes max(((1-t)*a + t*b) * ((1-t)*a + t*b)).
       Uses golden section search.
       """
       def f(t):
           a_new = (1 - t) * a + t * b
           a_new = np.maximum(a_new, 0)
           c = fft_convolve_self(a_new)
           return c.max()

       c2 = lo + gr * (hi - lo)
       f1 = f(c1)
       f2 = f(c2)

               c2 = c1
               f2 = f1

           else:

               c1 = c2
               f1 = f2
               c2 = lo + gr * (hi - lo)
               f2 = f(c2)
           evals += 1

       return t_best, f(t_best)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional search
with active set management, combined with gradient-based polishing.

Key idea: At each step, solve a linear program to find the best feasible
direction that keeps the convolution bounded, then take a line-search step.
Alternate with LSE-smoothed gradient steps for final precision.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    max_diff = diff.max()
    if max_diff > 50.0 / beta:
        beta = 50.0 / max(max_diff, 1e-300)
    scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=32):
    """Subgradient from top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def solve_lp_direction(a, M, active_lags, time_limit=15):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0.
    Returns optimal b or None.
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return None

    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        j_idx = np.arange(lo, hi + 1)
        b_idx = k - j_idx
        A_ub[i, b_idx] = a[j_idx]

    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search_step(a, b_dir, S, M, max_evals=8):
    """
    Find best step t in [0,1] minimizing max(((1-t)*a + t*b_dir) * ((1-t)*a + t*b_dir)).
    Returns (t_best, val_best, a_best).
    """
    def eval_at(t):
        a_new = (1.0 - t) * a + t * b_dir
        a_new = np.maximum(a_new, 0.0)
        c = fft_convolve_self(a_new)
        return c.max()

    # Try geometric sequence + golden section refinement
    candidates = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]
    best_t = 0.0
    best_val = M  # current max

    evals = 0
    for t in candidates:
        if evals >= max_evals:
            break
        val = eval_at(t)
        evals += 1
        if val < best_val:
            best_val = val
            best_t = t
        if val < M * 0.999:
            break

    return best_t, best_val, (1.0 - best_t) * a + best_t * b_dir


def run_lp_optimization(a_init, S, max_time, active_set_size=100,
                        expand_every=20, max_active=500, rng=None):
    """
    LP-based iterative optimization.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    M = c.max()
    best = M
    best_a = a.copy()
    iteration = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        iteration += 1
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 5:
            break

        # Update active set: top-k lags
        c = fft_convolve_self(a)
        M = c.max()

        if M < best - 1e-10:
            best = M
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Determine active set size (grow over time, reset if stuck)
        if stuck_count > 30:
            active_k = min(active_set_size // 2, max_active)
            stuck_count = 0
        else:
            active_k = min(active_set_size + iteration // expand_every, max_active)
            active_k = max(active_k, 20)

        # Get top-k lags
        top_lags = np.argsort(c)[::-1][:active_k]
        # Filter to only lags above the mean (to focus on "problem" lags)
        mean_c = c.mean()
        problem_lags = top_lags[c[top_lags] > mean_c * 1.01]
        if len(problem_lags) < 10:
            problem_lags = top_lags[:min(50, len(top_lags))]

        # Solve LP
        b_dir = solve_lp_direction(a, M, problem_lags, time_limit=min(10, remaining * 0.3))
        if b_dir is None:
            # Fallback: use peak-shave gradient
            grad = peak_shave_gradient(a, c, k=active_k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (n * gnorm + 1e-10) * 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < M:
                a = a_new
            continue

        # Ensure b_dir has positive sum
        b_sum = b_dir.sum()
        if b_sum < 1e-10:
            break

        # Rescale to match normalization
        b_dir = b_dir * (S / b_sum)

        # Line search
        t_best, val_best, a_new = line_search_step(a, b_dir, S, M, max_evals=6)

        if t_best > 0 and val_best < M * (1 + 1e-10):
            a = np.maximum(a_new, 0.0)
            # Re-normalize to be safe
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)
        else:
            # No progress from LP; try a small gradient step
            grad = peak_shave_gradient(a, c, k=min(50, active_k))
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = S / (n * (gnorm + 1e-10)) * 0.1
                a_new = project_simplex(a - step * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < M:
                    a = a_new

    return best_a, best


def run_fista_polish(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.005,
                     beta_max=10000.0):
    """FISTA with LSE for final polishing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best - 1e-12:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best


def run_combined_optimization(a_init, S, max_time, rng):
    """
    Combined LP + FISTA optimization with periodic mode switching.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    mode = 0  # 0 = LP, 1 = FISTA
    mode_timer = 0
    mode_duration = 60  # seconds per mode

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 10:
            break

        # Switch modes periodically
        mode_timer += 1
        if mode_timer > mode_duration:
            mode = 1 - mode
            mode_timer = 0
            # Adjust mode duration based on remaining time
            mode_duration = max(20, remaining * 0.1)

        if mode == 0:
            # LP-based step
            t_alloc = min(30.0, remaining)
            a_new, val = run_lp_optimization(a, S, t_alloc,
                                             active_set_size=80,
                                             expand_every=10, max_active=300,
                                             rng=rng)
        else:
            # FISTA polish
            t_alloc = min(30.0, remaining)
            t_alloc = max(t_alloc, 10.0)
            a_new, val = run_fista_polish(a, S, t_alloc, rng,
                                          beta_init=20 + 10 * min(mode_timer, 50),
                                          beta_rate=0.003,
                                          beta_max=5000 + 100 * min(mode_timer, 50))

        if val < best - 1e-10:
            best = val
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Keep current a if no improvement, but try slightly perturbed
            a = best_a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a, S)

    return best_a, best


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)
