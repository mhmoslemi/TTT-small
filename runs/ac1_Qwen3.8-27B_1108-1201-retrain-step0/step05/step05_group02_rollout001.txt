"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Verify uniform baseline, then use LSE-smoothed FISTA with
adaptive beta, simplex projection, coordinate descent refinement,
and periodic restarts. Multi-scale approach with upsample.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, beta):
    """Compute LSE value, gradient, and convolution."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Y = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
    return c, w, grad

def coordinate_descent(a_init, max_time, rng):
    """
    Coordinate descent via mass transfer between pairs.
    Moves mass from coordinates that contribute to peak convolution
    to coordinates that don't.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 15000:
            break

        # Find the peak lag
        k_star = int(np.argmax(c))

        # Compute contribution of each coordinate to the top few lags
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        multi_contrib = np.zeros(n)
        for kk in top_idx:
            lo = max(0, int(kk) - n + 1)
            hi = min(n - 1, int(kk))
            if hi >= lo:
                multi_contrib[lo:hi + 1] += a[int(kk) - hi:int(kk) - lo + 1][::-1]

        # Score for source (want high contribution, large value)
        src_scores = multi_contrib * np.maximum(a, 1e-10)
        # Score for target (want low contribution, room to grow)
        tgt_scores = -multi_contrib * (S / max(n, 1) - a + 1e-10)

        n_try = min(8, n)
        src_candidates = np.argsort(src_scores)[::-1][:n_try]
        tgt_candidates = np.argsort(tgt_scores)[::-1][:n_try]

        best_delta_obj = cur_obj
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in src_candidates[:n_try]:
            for l in tgt_candidates[:n_try]:
                if j == l or a[j] < 1e-12:
                    continue
                delta = a[j] * 0.03
                if delta < 1e-14:
                    continue

                # Compute change in convolution
                delta_c = np.zeros(2 * n - 1)
                js = j
                je = min(2 * n - 2, j + n - 1)
                coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                delta_c[js:je + 1] -= delta * coeff_j

                ls = l
                le = min(2 * n - 2, l + n - 1)
                coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                delta_c[ls:le + 1] += delta * coeff_l

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj - 1e-14:
                    best_delta_obj = new_obj
                    best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            # Update c
            js = j
            je = min(2 * n - 2, j + n - 1)
            coeff_j = a[max(0, js - j):min(n, je - j + 1)]
            c[js:je + 1] += delta * coeff_j  # Note: a[j] already reduced

            ls = l
            le = min(2 * n - 2, l + n - 1)
            coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
            c[ls:le + 1] += delta * coeff_l
        else:
            # No improvement; try random perturbation if stuck
            if it - last_improve > 2000 and it % 100 == 0:
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 != idx2 and a[idx1] > 1e-10:
                    delta = a[idx1] * 0.01
                    a[idx1] -= delta
                    a[idx2] += delta
                    c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "uniform":
        a = np.ones(n)
    elif method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.1 ** 2))
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt)
             + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3)
             + 0.15 * np.sin(2 * np.pi * 13 * tt + 2.1))
        a = np.abs(a) + 0.01
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "smoothed_random":
        a = rng.random(n)
        # Smooth with a small kernel
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

        if obj_pol < global_best_obj - 1e-12:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

    # Final verification
    print("\n=== Final Result ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)

        else:

            n_fb = 8192

            final_list = [1.0] * n_fb
    else:
        n_fb = 8192
        final_list = [1.0] * n_fb
