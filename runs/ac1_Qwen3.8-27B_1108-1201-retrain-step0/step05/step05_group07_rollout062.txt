def util_project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n=len(v)
    if n==0: return v
    if not np.isfinite(S) or S <= 0:
       return np.zeros(n)
    v = np.where(np.isfinite(v), v, 0.0)
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    rho_vals = np.arange(1, n+1, dtype=np.float64)
    theta_vals = (css - S) / rho_vals
    cond = theta_vals <= x
    if not np.any(cond):
        rho = n
    else:
        rho = int(np.max(np.where(cond)[0]) + 1)
    theta = (css[rho-1] - S) / rho
    z = np.maximum(v - theta, 0.0)
    zsum = z.sum()
    if zsum <= 1e-300:
       z = np.full(n, S/n)
    else:
       z *= S / zsum
    return z

```python
"""
Search for non-negative sequences minimizing

    2*n*max(conv(a,a)) / (sum(a))^2

by rewriting the problem as minimizing max(conv(a,a)) under the
normalization sum(a)=sqrt(2n).

The program combines several complementary local search methods:
  * projected gradient descent using log-sum-exp smoothed maxima,
  * multiplicative log-space descent,
  * a spectral peak-shaving projection,
  * low-dimensional basis exploration,
  * random perturbations and a final high-temperature polish.

The result is scaled to sum(a)=sqrt(2n), which makes the internal
max(conv(a,a)) equal to the official evaluation score.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def util_next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def util_time_left(deadline):
    return max(0.0, deadline - time.monotonic())


def util_fft_convolve_self(a):
    """Linear self-convolution via zero padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = util_next_pow2(size)
    A = rfft(a, L)
    c = irfft(A * A, L)[:size]
    c = np.maximum(c, 0.0)
    return c


def util_fft_corr_w_a(w, a):
    """
    Gradient of dot(w, conv(a,a)) with respect to a.

    For c_k = sum_{i+j=k} a_i a_j,
    d/d a_i dot_k(w_k c_k) = 2 * sum_k w_k a_{k-i}.
    This is computed as 2 * conv(w, reverse(a)) in the zero padded circle.
    """
    n = len(a)
    size = 2 * n - 1
    L = util_next_pow2(size)
    W = rfft(w, L)
    A = rfft(a, L)
    r = irfft(W * np.conj(A), L)
    g = 2.0 * r[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def util_project_simplex(v, S):
    """Project v onto {x >= 0, sum(x)=S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    if not np.isfinite(S) or S <= 0:
        return np.zeros(n, dtype=np.float64)

    v = np.where(np.isfinite(v), v, 0.0)
    x = np.sort(v)[::-1]
    css = np.cumsum(x)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_vals = (css - S) / idx
    cond = theta_vals <= x
    if not np.any(cond):
        rho = n
    else:
        rho = int(np.max(np.where(cond)[0]) + 1)

    theta = (css[rho - 1] - S) / rho
    z = np.maximum(v - theta, 0.0)
    s = z.sum()
    if s <= 1e-300:
        z = np.full(n, S / float(n), dtype=np.float64)
    else:
        z *= S / s
    return z


def util_candidate_normalize(a):
    """Normalize any candidate to non-negative sum sqrt(2n)."""
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        a = np.ones(8192, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-300:
        a = np.full(n, S / float(n), dtype=np.float64)
    else:
        a *= S / s
        a = util_project_simplex(a, S)
    return a


def util_resample(a, n_new):
    """Log-linear resampling of a probability sequence."""
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == 0:
        return np.full(n_new, 1.0 / float(n_new))
    if n_old == n_new:
        s = a.sum()
        if s > 0:
            return a / s
        return np.full(n_new, 1.0 / float(n_new))

    pos = np.where(np.isfinite(a), a, 0.0)
    pos = np.maximum(pos, 0.0)
    s = pos.sum()
    if s <= 0:
        p = np.full(n_old, 1.0 / float(n_old))
    else:
        p = pos / s

    lp = np.log(np.maximum(p, 1e-300))
    t_old = np.linspace(0.0, 1.0, n_old)
    t_new = np.linspace(0.0, 1.0, n_new)
    q = np.interp(t_new, t_old, lp)
    q -= np.max(q)
    q = np.exp(q)
    s = q.sum()
    if s <= 0:
        return np.full(n_new, 1.0 / float(n_new))
    return q / s


def util_try_symmetry(a, S):
    """Return symmetric version if it does not worsen the score."""
    b = 0.5 * (a + a[::-1])
    b = util_project_simplex(b, S)
    ob = float(util_fft_convolve_self(b).max())
    oa = float(util_fft_convolve_self(a).max())
    if ob <= oa:
        return b
    return a


def util_consolidate(a, best_a, best_obj, label=""):
    """Update best_a, best_obj if a is better."""
    if a is None:
        return best_a, best_obj
    a = util_candidate_normalize(a)
    if a is None:
        return best_a, best_obj
    obj = float(util_fft_convolve_self(a).max())
    if obj < best_obj:
        print(f"  new best [{label}: {len(a)}] obj={obj:.7f}")
        return a.copy(), obj
    return best_a, best_obj


def util_lse_weights(c, beta):
    """Normalized softmax weights for log-sum-exp smoothing of c."""
    c = np.asarray(c, dtype=np.float64)
    m = len(c)
    if m == 0:
        return np.array([], dtype=np.float64)
    if beta <= 0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)

    cmax = float(c.max())
    x = beta * (c - cmax)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    w = e / s
    return w


def util_elastic_weights(c, target, eta=2.0):
    """Weights emphasizing coefficients above a target level."""
    c = np.asarray(c, dtype=np.float64)
    m = len(c)
    if m == 0:
        return np.array([], dtype=np.float64)

    target = float(target)
    excess = np.maximum(c - target, 0.0)
    if excess.sum() <= 1e-14:
        k = min(32, m)
        top = np.argpartition(c, m - k)[-k:]
        w = np.zeros(m, dtype=np.float64)
        w[top] = 1.0 / float(k)
        return w

    eta = float(max(0.5, min(4.0, eta)))
    w = excess ** eta
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / s


def util_smooth_score(c, target):
    """Cheap acceptance score: peak plus total excess above target."""
    c = np.asarray(c, dtype=np.float64)
    target = float(target)
    excess = np.maximum(c - target, 0.0)
    return float(c.max() + 0.1 * float(excess.sum()))


def optimize_lse(a0, max_time, rng, beta_start=100.0, beta_end=50000.0,
                 verbose=False, label=""):
    """
    Projected gradient descent using LSE weights.
    All sequences are maintained with sum sqrt(2n), so score=max(conv).
    """
    a = util_candidate_normalize(a0)
    n = len(a)
    S = np.sqrt(2.0 * float(n))

    c = util_fft_convolve_self(a)
    best_c = float(c.max())
    best_a = a.copy()

    if max_time <= 0:
        return best_a, best_c

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 250
    step = 0.25

    while util_time_left(start + max_time) > 0.1:
        it += 1
        c = util_fft_convolve_self(a)
        cur_c = float(c.max())

        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        if beta_end > beta_start:
            beta = beta_start * (beta_end / beta_start) ** progress
        else:
            beta = beta_start

        avg = float(c.mean())
        step_target = max(avg, best_c - 0.05 * max(1e-6, best_c - avg))
        gap = max(1e-8, cur_c - step_target)

        w = util_lse_weights(c, beta)
        grad = util_fft_corr_w_a(w, a)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        step = min(0.5, max(1e-7, 2.0 * gap / (gnorm * gnorm + 1e-30)))
        score_target = max(avg, best_c - 0.002)
        score_cur = util_smooth_score(c, score_target)
        accepted = False

        for _ in range(8):
            a_try = util_project_simplex(a - step * grad, S)
            c_try = util_fft_convolve_self(a_try)
            m_try = float(c_try.max())
            score_try = util_smooth_score(c_try, score_target)

            if (m_try < cur_c - 1e-11) or (
                score_try < score_cur - 1e-12 and m_try < cur_c + 0.02
            ):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if accepted:
            stuck = 0
        else:
            stuck += 1
            a = util_project_simplex(a - max(1e-8, step) * grad, S)

        if stuck > patience:
            noise_scale = 1e-5 * S / np.sqrt(float(n))
            noise_scale *= (1.0 - 0.5 * progress)
            if noise_scale > 0:
                a = util_project_simplex(a + rng.normal(0.0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.3)

        if it % 2000 == 0:
            if np.min(a) < 1e-8 * S / float(n):
                a = np.maximum(a, 1e-10 * S / float(n))
                a = util_project_simplex(a, S)

        if verbose and it % 500 == 0:
            print(f"    LSE {label} it={it} best={best_c:.7f}")

    a = util_candidate_normalize(best_a)
    c = util_fft_convolve_self(a)
    return a, float(c.max())


def optimize_multiplicative(a0, max_time, rng, mode="lse",
                            beta_start=100.0, beta_end=50000.0,
                            verbose=False, label=""):
    """
    Multiplicative update in log-space:
        a_i <- a_i * exp(-eta * centered_log_gradient_i)
    This keeps the sequence strictly positive and often polishes better
    than projected gradient near boundaries.
    """
    a = util_candidate_normalize(a0)
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    a = np.maximum(a, 1e-12)
    a = util_project_simplex(a, S)

    c = util_fft_convolve_self(a)
    best_c = float(c.max())
    best_a = a.copy()

    if max_time <= 0:
        return best_a, best_c

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 250
    prev_eta = 5.0

    while util_time_left(start + max_time) > 0.1:
        it += 1
        c = util_fft_convolve_self(a)
        cur_c = float(c.max())

        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        avg = float(c.mean())
        target = max(avg, best_c - 0.001)

        if mode == "elastic":
            w = util_elastic_weights(c, target, eta=2.0)
        else:
            if beta_end > beta_start:
                beta = beta_start * (beta_end / beta_start) ** progress
            else:
                beta = beta_start
            w = util_lse_weights(c, beta)

        H = 0.5 * util_fft_corr_w_a(w, a)
        H = np.where(np.isfinite(H), H, 0.0)
        Hc = H - float(np.median(H))

        hscale = float(np.percentile(np.abs(Hc), 90.0))
        if not np.isfinite(hscale) or hscale < 1e-14:
            break
        eta = min(100.0, max(0.01, 0.05 / hscale))
        eta = 0.5 * (prev_eta + eta)

        score_target = max(avg, best_c - 0.001)
        score_cur = util_smooth_score(c, score_target)
        accepted = False
        et = eta

        for _ in range(8):
            arg = np.clip(-et * Hc, -3.0, 3.0)
            a_try = a * np.exp(arg)
            s = float(a_try.sum())
            if not np.isfinite(s) or s <= 1e-300:
                et *= 0.5
                continue
            a_try *= S / s
            c_try = util_fft_convolve_self(a_try)
            m_try = float(c_try.max())
            score_try = util_smooth_score(c_try, score_target)

            if (m_try < cur_c - 1e-11) or (
                score_try < score_cur - 1e-12 and m_try < cur_c + 0.02
            ):
                a = a_try
                accepted = True
                prev_eta = et
                break
            et *= 0.5

        if accepted:
            stuck = 0
        else:
            stuck += 1
            arg = np.clip(-0.01 * Hc, -0.1, 0.1)
            a_try = a * np.exp(arg)
            s = float(a_try.sum())
            if np.isfinite(s) and s > 0:
                a = a_try * (S / s)
            else:
                a = util_project_simplex(a + rng.normal(0.0, 1e-7 * S / float(n), n), S)

        if stuck > patience:
            noise_scale = 1e-5 * S / np.sqrt(float(n))
            noise_scale *= (1.0 - 0.5 * progress)
            if noise_scale > 0:
                a = util_project_simplex(a + rng.normal(0.0, noise_scale, n), S)
                a = np.maximum(a, 1e-12)
                a = util_project_simplex(a, S)
            stuck = 0
            patience = int(patience * 1.3)

        if it % 500 == 0 and np.min(a) < 1e-12:
            a = np.maximum(a, 1e-12)
            a = util_project_simplex(a, S)

        if verbose and it % 500 == 0:
            print(f"    MULT {mode} {label} it={it} best={best_c:.7f}")

    a = util_candidate_normalize(best_a)
    c = util_fft_convolve_self(a)
    return a, float(c.max())


def optimize_spectral(a0, max_time, rng, verbose=False, label=""):
    """
    Spectral peak-shaving.

    Given current convolution c, form a flatter target q by removing some
    of its peak excess. Then multiply the current spectrum A by a square-root
    correction so that A^2 is nudged toward Q(q), inverse FFT, project to a
    non-negative length-n sequence, and accept only on improvement.
    """
    a = util_candidate_normalize(a0)
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    size = 2 * n - 1
    L = util_next_pow2(size)
    avg = (S * S) / float(size)

    c = util_fft_convolve_self(a)
    best_c = float(c.max())
    best_a = a.copy()

    if max_time <= 0:
        return best_a, best_c

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 80
    gamma = 0.30

    while util_time_left(start + max_time) > 0.1:
        it += 1
        A = rfft(a, L)
        c = irfft(A * A, L)[:size]
        c = np.maximum(c, 0.0)
        cur_c = float(c.max())

        if cur_c < best_c - 1e-11:
            best_c = cur_c
            best_a = a.copy()
            stuck = 0
            gamma = max(0.05, gamma * 0.97)
        else:
            stuck += 1
            gamma = min(0.80, gamma * 1.08)

        target = max(avg, best_c * 0.999)
        excess = np.maximum(c - target, 0.0)
        q = c - gamma * excess
        q = np.maximum(q, 1e-12)
        s = float(q.sum())
        if s <= 1e-300:
            q = np.full(size, avg, dtype=np.float64)
        else:
            q *= (S * S) / s

        Qt = rfft(q, L)
        Qc = A * A

        R = np.ones_like(Qt, dtype=np.complex128)
        mask = np.abs(Qc) > 1e-10
        R[mask] = Qt[mask] / Qc[mask]

        A_new = A * np.sqrt(R)
        if (not np.all(np.isfinite(A_new))) or (
            np.max(np.abs(A_new)) > 10.0 * (np.max(np.abs(A)) + 1e-12)
        ):
            A_new = A.copy()
        if np.isfinite(A_new[0]):
            A_new[0] = A[0]

        accepted = False
        for rho in (1.0, 0.5, 0.25, 0.1, 0.05):
            A_try = (1.0 - rho) * A + rho * A_new
            full = irfft(A_try, L).real
            a_try = np.maximum(full[:n], 0.0)
            s_try = float(a_try.sum())
            if not np.isfinite(s_try) or s_try <= 1e-300:
                continue
            a_try *= S / s_try
            c_try = util_fft_convolve_self(a_try)
            m_try = float(c_try.max())
            if m_try < cur_c - 1e-11:
                a = a_try
                accepted = True
                break

        if not accepted:
            A_try = 0.98 * A + 0.02 * A_new
            full = irfft(A_try, L).real
            a_try = np.maximum(full[:n], 0.0)
            s_try = float(a_try.sum())
            if np.isfinite(s_try) and s_try > 1e-300:
                a_try *= S / s_try
                c_try = util_fft_convolve_self(a_try)
                m_try = float(c_try.max())
                if m_try <= cur_c + 0.01:
                    a = a_try

        if stuck > patience:
            noise_scale = 1e-5 * S / np.sqrt(float(n))
            a = util_project_simplex(a + rng.normal(0.0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.3)
            gamma = 0.30

        if verbose and it % 500 == 0:
            print(f"    SPECTRAL {label} it={it} best={best_c:.7f}")

    a = util_candidate_normalize(best_a)
    c = util_fft_convolve_self(a)
    return a, float(c.max())


def util_make_basis(n, dim):
    """Smooth orthonormal DCT-II-like basis for low-dimensional exploration."""
    dim = int(min(dim, max(1, n - 1)))
    i = np.arange(n, dtype=np.float64)
    cols = []
    for r in range(1, dim + 1):
        cols.append(np.cos(np.pi * r * (i + 0.5) / float(n)))
    B = np.column_stack(cols)
    U, _, _ = np.linalg.svd(B, full_matrices=False)
    Q = U[:, :dim]
    return Q


def util_param_softmax(theta, B):
    X = B @ theta
    X = X - float(np.max(X))
    E = np.exp(X)
    s = float(E.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return E / s


def optimize_param_basis(p0, max_time, rng, dim=10,
                         beta_start=50.0, beta_end=20000.0,
                         verbose=False, label=""):
    """
    Low-dimensional exploration in a smooth basis:
        p = softmax(B theta)
    Gradient is computed exactly for LSE-smoothed max convolution.
    Returns a probability sequence (sum=1).
    """
    if p0 is None:
        return None
    p0 = np.asarray(p0, dtype=np.float64)
    n = len(p0)
    p = util_project_simplex(p0, 1.0)
    S2 = 2.0 * float(n)

    B = util_make_basis(n, dim)
    logp = np.log(np.maximum(p, 1e-300))
    logp -= float(np.mean(logp))
    theta = B.T @ logp

    c = util_fft_convolve_self(p)
    best_obj = S2 * float(c.max())
    best_p = p.copy()

    if max_time <= 0:
        return best_p

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 40

    while util_time_left(start + max_time) > 0.1:
        it += 1
        pcur = util_param_softmax(theta, B)
        if pcur is None:
            pcur = p
        c = util_fft_convolve_self(pcur)
        obj = S2 * float(c.max())

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_p = pcur.copy()
            stuck = 0
        else:
            stuck += 1

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        if beta_end > beta_start:
            beta = beta_start * (beta_end / beta_start) ** progress
        else:
            beta = beta_start

        c_a = S2 * c
        w = util_lse_weights(c_a, beta)
        gp = S2 * util_fft_corr_w_a(w, pcur)
        pdot = float(np.dot(pcur, gp))
        gx = pcur * (gp - pdot)
        gt = B.T @ gx
        gt = np.where(np.isfinite(gt), gt, 0.0)
        gnorm = float(np.linalg.norm(gt))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        step = min(0.05, max(1e-5, 0.02 / (gnorm + 1e-12)))
        accepted = False

        for mult in (1.0, 0.5, 0.25, 0.1):
            theta_try = theta - step * mult * gt
            nrm = float(np.linalg.norm(theta_try))
            if nrm > 50.0:
                theta_try = theta_try * (50.0 / nrm)

            p_try = util_param_softmax(theta_try, B)
            if p_try is None:
                continue
            c_try = util_fft_convolve_self(p_try)
            obj_try = S2 * float(c_try.max())

            if obj_try < obj - 1e-12:
                theta = theta_try
                pcur = p_try
                accepted = True
                break
            if obj_try < best_obj and obj_try < obj + 1e-7:
                theta = theta_try
                pcur = p_try
                accepted = True
                break

        if accepted:
            stuck = 0
        else:
            if stuck > patience:
                theta = theta + rng.normal(0.0, 0.05, len(theta))
                stuck = 0
                patience = int(patience * 1.3)

        if verbose and it % 200 == 0:
            print(f"    PARAM {label} it={it} best={best_obj:.7f}")

    return best_p


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for the best non-negative sequence and return it as a list.
    """
    seed = int(seed)
    start = time.monotonic()
    deadline = start + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # Load the previous construction if available.
    init_a = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size >= 16 and arr.sum() > 0:
            init_a = arr
    except Exception:
        init_a = None

    if init_a is None:
        n0 = 8192
        init_a = np.full(n0, np.sqrt(2.0 / float(n0)), dtype=np.float64)
    else:
        if len(init_a) < 512 or len(init_a) > 16384:
            p = util_resample(init_a, 8192)
            init_a = p * np.sqrt(2.0 * 8192)

    init_a = util_candidate_normalize(init_a)
    S0 = np.sqrt(2.0 * float(len(init_a)))
    init_a = util_try_symmetry(init_a, S0)
    best_a, best_obj = util_consolidate(init_a, best_a, best_obj, "init")

    # ------------------------------------------------------------------
    # Phase 1: quick LSE + multiplicative polish on the inherited state.
    # ------------------------------------------------------------------
    t = min(400.0, util_time_left(deadline) * 0.20)
    if t > 25:
        a_opt, _ = optimize_lse(
            best_a, t, rng,
            beta_start=100.0, beta_end=50000.0,
            verbose=False, label="LSE1"
        )
        best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "LSE1")

    t = min(250.0, util_time_left(deadline) * 0.14)
    if t > 25:
        a_opt, _ = optimize_multiplicative(
            best_a, t, rng,
            mode="lse",
            beta_start=100.0, beta_end=50000.0,
            verbose=False, label="MULT1"
        )
        best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "MULT1")

    # ------------------------------------------------------------------
    # Phase 2: low-dimensional basis exploration.
    # This is intentionally different from the full local search: it can
    # move the sequence into a different smooth basin before polishing.
    # ------------------------------------------------------------------
    if util_time_left(deadline) > 220:
        nbest = len(best_a)
        Sbest = np.sqrt(2.0 * float(nbest))
        pbest = best_a / Sbest

        tparam = min(180.0, util_time_left(deadline) * 0.12)
        if tparam > 25:
            p_opt = optimize_param_basis(
                pbest, tparam, rng,
                dim=10,
                beta_start=50.0, beta_end=20000.0,
                verbose=False, label="PARAM_BEST"
            )
            if p_opt is not None:
                a_opt = p_opt * np.sqrt(2.0 * float(len(p_opt)))
                best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "PARAM_BEST")

        if util_time_left(deadline) > 150:
            p_uni = np.full(nbest, 1.0 / float(nbest), dtype=np.float64)
            tparam2 = min(120.0, util_time_left(deadline) * 0.08)
            if tparam2 > 25:
                p_opt = optimize_param_basis(
                    p_uni, tparam2, rng,
                    dim=10,
                    beta_start=50.0, beta_end=20000.0,
                    verbose=False, label="PARAM_UNI"
                )
                if p_opt is not None:
                    a_opt = p_opt * np.sqrt(2.0 * float(len(p_opt)))
                    best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "PARAM_UNI")

    # ------------------------------------------------------------------
    # Phase 3: alternating core local searches.
    # ------------------------------------------------------------------
    block = 0
    while util_time_left(deadline) > 320:
        alloc = min(160.0, util_time_left(deadline) * 0.25)
        if alloc < 30:
            break

        if block % 3 == 0:
            a_opt, _ = optimize_lse(
                best_a, alloc, rng,
                beta_start=100.0, beta_end=100000.0,
                verbose=False, label=f"CORE_LSE{block}"
            )
        elif block % 3 == 1:
            a_opt, _ = optimize_multiplicative(
                best_a, alloc, rng,
                mode="elastic",
                beta_start=100.0, beta_end=50000.0,
                verbose=False, label=f"CORE_MUL{block}"
            )
        else:
            a_opt, _ = optimize_spectral(
                best_a, alloc, rng,
                verbose=False, label=f"CORE_SPEC{block}"
            )

        best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, f"CORE{block}")
        block += 1

    # ------------------------------------------------------------------
    # Phase 4: local restarts from the best solution.
    # ------------------------------------------------------------------
    for scale in (0.001, 0.004, 0.012):
        if util_time_left(deadline) < 95:
            break
        nbest = len(best_a)
        Sbest = np.sqrt(2.0 * float(nbest))
        noise = rng.normal(0.0, scale * Sbest / np.sqrt(float(nbest)), nbest)
        a_pert = np.maximum(best_a + noise, 0.0)

        t = min(80.0, util_time_left(deadline) * 0.30)
        if t < 25:
            break
        a_opt, _ = optimize_lse(
            a_pert, t, rng,
            beta_start=200.0, beta_end=100000.0,
            verbose=False, label=f"RESTART_LSE{scale}"
        )
        best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, f"RESTART_LSE{scale}")

        t2 = min(45.0, util_time_left(deadline) * 0.20)
        if t2 > 20:
            a_opt, _ = optimize_multiplicative(
                a_opt, t2, rng,
                mode="elastic",
                beta_start=200.0, beta_end=100000.0,
                verbose=False, label=f"RESTART_MUL{scale}"
            )
            best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, f"RESTART_MUL{scale}")

    # ------------------------------------------------------------------
    # Phase 5: optional length increase to 16384.
    # Only accepted if the improvement is meaningful, because the final
    # evaluator's direct convolution is more expensive for longer n.
    # ------------------------------------------------------------------
    if len(best_a) < 16384 and util_time_left(deadline) > 320:
        n_new = 16384
        S_old = np.sqrt(2.0 * float(len(best_a)))
        p_old = best_a / S_old
        p_new = util_resample(p_old, n_new)
        S_new = np.sqrt(2.0 * float(n_new))
        a_new = p_new * S_new

        t = min(250.0, util_time_left(deadline) * 0.40)
        a_opt, _ = optimize_lse(
            a_new, t, rng,
            beta_start=100.0, beta_end=50000.0,
            verbose=False, label="L16384_LSE"
        )
        t2 = min(120.0, util_time_left(deadline) * 0.20)
        if t2 > 25:
            a_opt, _ = optimize_multiplicative(
                a_opt, t2, rng,
                mode="elastic",
                beta_start=100.0, beta_end=50000.0,
                verbose=False, label="L16384_MUL"
            )

        a_norm = util_candidate_normalize(a_opt)
        obj = float(util_fft_convolve_self(a_norm).max())
        if obj < best_obj - 0.0005:
            best_a = a_norm.copy()
            best_obj = obj
            print(f"  accepted 16384-length candidate obj={obj:.7f}")

    # ------------------------------------------------------------------
    # Phase 6: final ultra polished refinement.
    # ------------------------------------------------------------------
    rem = util_time_left(deadline)
    if rem > 80 and best_a is not None:
        t1 = min(150.0, rem * 0.5)
        a_opt, _ = optimize_lse(
            best_a, t1, rng,
            beta_start=500.0, beta_end=300000.0,
            verbose=False, label="FINAL_LSE"
        )
        best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "FINAL_LSE")

        rem = util_time_left(deadline)
        if rem > 45:
            t2 = max(0.0, rem - 25.0)
            a_opt, _ = optimize_multiplicative(
                best_a, t2, rng,
                mode="elastic",
                beta_start=500.0, beta_end=300000.0,
                verbose=False, label="FINAL_MUL"
            )
            best_a, best_obj = util_consolidate(a_opt, best_a, best_obj, "FINAL_MUL")

    # ------------------------------------------------------------------
    # Finalize.
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(8192, np.sqrt(2.0 / 8192.0), dtype=np.float64)

    final_a = util_candidate_normalize(best_a)
    final_a = np.where(np.isfinite(final_a), final_a, 0.0)
    final_a = np.clip(final_a, 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(4096, dtype=np.float64)

    final_list = [float(x) for x in final_a]
    try:
        verdict = evaluate_sequence(final_list)
        print("=" * 60)
        print(f"FINAL len={len(final_list)} internal={best_obj:.7f} evaluate={verdict:.7f}")
        print("=" * 60)
    except Exception as exc:
        print("evaluate_sequence call failed:", exc)

    return final_list
```
