def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def lse_gradient(y, c, beta):

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization at n=16384 with:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta, c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient from the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    w_full = np.zeros(len(c))
    w_full[k] = 1.0
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def upsample_linear(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

    a = normalize_seq(a, S)

        it += 1
        c = fft_convolve_self(y)
        obj2 = np.sum(c * c)
        if obj2 < best_obj2:
            best_obj2 = obj2
            best_a = y.copy()

        fsize = 1 << (len(c) + n - 1).bit_length()
        C = fft(c, fsize)
        A = fft(y, fsize)
        grad = 4.0 * np.real(ifft(C * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 0.5 / (4.0 * np.max(a) + 1.0)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a


def run_lse_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.005,
                  beta_max=100000.0):
    """Main LSE FISTA phase with adaptive beta and restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_val - 1e-15:
            best_val = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 500:
            beta = beta_init * (1.0 + beta_rate * it)
        elif stuck < 5000:
            beta = min(beta_max, beta * 1.01)
        else:
            beta = min(beta_max, beta * 1.05)
        beta = min(beta, beta_max)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 0.7 / (beta * np.max(a) + 1.0)
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart based on stuck count
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 15000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            # Also try structured perturbation
            if stuck > 8000 and it % 500 == 0:
                tt = np.linspace(0, 1, n, endpoint=False)
                freq = rng.choice([1, 2, 3, 4])
                w = 1.0 + 0.01 * np.cos(2 * np.pi * freq * tt)
                y = project_simplex(y * w, S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_val:
        best_val = c_final.max()
    return best_a, best_val


def run_subgradient_armijo(a_init, max_time, rng):
    """Exact subgradient method with Armijo backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    c_armijo = 1e-4
    step_init = 1.0 / max(np.max(a), 1e-6)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val - 1e-15:
            best_val = cur_max
            best_a = a.copy()

        grad = exact_subgradient(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Armijo backtracking
        step = step_init
        for _ in range(30):
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            new_max = c_new.max()
            if new_max <= cur_max - c_armijo * step * gnorm * 0.5 + 1e-18:
                a = x_new
                break
            step *= 0.5
        else:
            # If Armijo fails, take a tiny step
            a = project_simplex(a - step * grad, S)

        # Adaptive step init
        if cur_max > best_val:
            step_init = max(step_init * 0.9, 1e-8)
        else:
            step_init = min(step_init * 1.05, 1.0)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_val:
        best_val = c_final.max()
    return best_a, best_val


def run_coordinate_descent(a_init, max_time, rng):
    """Coordinate descent: optimize individual coordinates via golden section."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0

    # Golden section ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val - 1e-15:
            best_val = cur_max
            best_a = a.copy()

        # Find active lag
        k_star = int(np.argmax(c))

        # Determine important coordinates (those contributing to active lag)
        # Coordinate m affects lag k_star through a[k_star - m]
        offsets = np.arange(n)
        km = (k_star - offsets) % (2 * n - 1)
        # Actually, a[k-m] is nonzero only if 0 <= k-m <= n-1
        valid_idx = np.where((k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n))[0]
        if len(valid_idx) == 0:
            valid_idx = np.arange(n)

        # Pick top coordinates by |a[k_star - m]|
        contrib = np.zeros(n)
        for m in valid_idx:
            contrib[m] = a[k_star - m]
        
        # Also include coordinates with large a[m] (they affect lag 0 and 2m)
        top_by_val = np.argsort(a)[::-1][:50]
        top_by Contrib = np.argsort(contrib)[::-1][:50]
        candidates = np.unique(np.concatenate([top_by_val, top_by_contrib]))
        candidates = candidates[rng.permutation(len(candidates))][:30]

        for m in candidates:
            # Optimize coordinate m
            # c[k]_new = c[k] + delta where delta depends on x = a[m]
            old_am = a[m]
            if old_am < 1e-15:
                continue

            # Upper bound for x
            x_hi = min(old_am * 3.0, S * 0.5)
            x_lo = max(0.0, old_am * 0.1)

            if x_hi - x_lo < 1e-12:
                continue

            def obj_1d(x):
                # Compute max autocorrelation if a[m] = x
                # Only need to update lags affected by coordinate m
                # Affected lags: k where k-m in [0,n-1], i.e., k in [m, m+n-1]
                # and also k where m-k in [0,n-1], i.e., k in [m-n+1, m]
                k_lo = max(0, m - n + 1)
                k_hi = min(2 * n - 2, m + n - 1)
                
                new_c = c.copy()
                # For each affected k:
                # old contribution: a[m]*a[k-m] (if valid) + a[k-m]*a[m] (if valid and k-m != m)
                # new contribution: x*a[k-m] + a[k-m]*x (or x^2 if k=2m)
                
                ks = np.arange(k_lo, k_hi + 1)
                km = ks - m  # k - m
                valid = (km >= 0) & (km < n)
                if not np.any(valid):
                    return cur_max
                ks_v = ks[valid]
                km_v = km[valid]
                
                # Old contribution at these lags
                old_contrib = np.zeros(len(ks_v))
                am_m = a[m]
                for i in range(len(ks_v)):
                    k = ks_v[i]
                    km_i = km_v[i]
                    if km_i == m:
                        old_contrib[i] = 2.0 * am_m * a[km_i] if k != 2 * m else am_m * am_m
                    else:
                        old_contrib[i] = 2.0 * am_m * a[km_i]
                
                # New contribution
                new_contrib = np.zeros(len(ks_v))
                for i in range(len(ks_v)):
                    k = ks_v[i]
                    km_i = km_v[i]
                    if k == 2 * m:
                        new_contrib[i] = x * x
                    else:
                        new_contrib[i] = 2.0 * x * a[km_i]
                
                new_c[ks_v] = c[ks_v] - old_contrib + new_contrib
                return new_c.max()

            # Golden section search
            a_gs = x_lo
            b_gs = x_hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = obj_1d(c_gs)
            fd = obj_1d(d_gs)

            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_1d(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_1d(d_gs)

            x_opt = (a_gs + b_gs) / 2.0
            # Only update if improvement
            if obj_1d(x_opt) < cur_max - 1e-14:
                a[m] = x_opt
                # Renormalize
                a = normalize_seq(a, S)
                c = fft_convolve_self(a)
                cur_max = c.max()
                k_star = int(np.argmax(c))
                if time.monotonic() - start > max_time:
                    break

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_val:
        best_val = c_final.max()
    return best_a, best_val


def run_lse_fista_fast(a_init, max_time, rng, beta_fix=500.0):
    """FISTA with fixed high beta for polishing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_fix

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_val - 1e-15:
            best_val = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 1000:
            beta = min(200000.0, beta * 1.03)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 0.5 / (beta * np.max(a) + 1.0)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000:
            t_mom = 1.0
            stuck = 0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_val:
        best_val = c_final.max()
    return best_a, best_val


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: upsample to n=16384, power-2 warmup, LSE-FISTA,
    subgradient polish, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Upsample to n=16384
    target_n = 16384
    if a_start is not None:
        if len(a_start) < target_n:
            factor = target_n // len(a_start)
            a_start = upsample_linear(a_start, factor=max(factor, 2))
            a_start = a_start[:target_n]
        elif len(a_start) > target_n:
            a_start = a_start[:target_n]
    else:
        a_start = np.ones(target_n)

    n = target_n
    S = np.sqrt(2.0 * n)
    a_start = normalize_seq(a_start, S)

    print("Starting n = %d, initial obj = %.6f" % (n, compute_obj(a_start)))
    if compute_obj(a_start) < global_best_obj:
        global_best_obj = compute_obj(a_start)
        global_best_a = a_start.copy()

    # === Phase 1: Power-2 warmup (short) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.05
    if p1_budget > 5:
        print("\n=== Phase 1: Power-2 warmup (%.0fs) ===" % p1_budget)
        t0 = time.monotonic()
        a_p2 = run_power2_fista(a_start, p1_budget, rng)
        obj_p2 = compute_obj(a_p2)
        print("  obj=%.6f" % obj_p2)
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()

    # === Phase 2: Main LSE FISTA ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.30
    if p2_budget > 10:
        print("\n=== Phase 2: LSE FISTA (%.0fs) ===" % p2_budget)
        t0 = time.monotonic()
        a_lse, _ = run_lse_fista(global_best_a, p2_budget, rng,
                                  beta_init=10.0, beta_rate=0.008, beta_max=50000.0)
        obj_lse = compute_obj(a_lse)
        print("  obj=%.6f" % obj_lse)
        if obj_lse < global_best_obj:
            global_best_obj = obj_lse
            global_best_a = a_lse.copy()

    # === Phase 3: High-beta FISTA polish ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20
    if p3_budget > 10:
        print("\n=== Phase 3: High-beta FISTA (%.0fs) ===" % p3_budget)
        t0 = time.monotonic()
        a_hb, _ = run_lse_fista_fast(global_best_a, p3_budget, rng, beta_fix=500.0)
        obj_hb = compute_obj(a_hb)
        print("  obj=%.6f" % obj_hb)
        if obj_hb < global_best_obj:
            global_best_obj = obj_hb
            global_best_a = a_hb.copy()

    # === Phase 4: Subgradient with Armijo ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 10:
        print("\n=== Phase 4: Subgradient Armijo (%.0fs) ===" % p4_budget)
        t0 = time.monotonic()
        a_sg, _ = run_subgradient_armijo(global_best_a, p4_budget, rng)
        obj_sg = compute_obj(a_sg)
        print("  obj=%.6f" % obj_sg)
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()

    # === Phase 5: Coordinate descent ===
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.15
    if p5_budget > 10:
        print("\n=== Phase 5: Coordinate descent (%.0fs) ===" % p5_budget)
        t0 = time.monotonic()
        a_cd, _ = run_coordinate_descent(global_best_a, p5_budget, rng)
        obj_cd = compute_obj(a_cd)
        print("  obj=%.6f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

    # === Phase 6: Final LSE polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final polish (%.0fs) ===" % remaining)
        a_fin, _ = run_lse_fista_fast(global_best_a, remaining * 0.5, rng, beta_fix=2000.0)
        obj_fin = compute_obj(a_fin)
        print("  obj=%.6f" % obj_fin)
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        rem = deadline - time.monotonic()
        if rem > 5:
            a_sg2, _ = run_subgradient_armijo(global_best_a, rem, rng)
            obj_sg2 = compute_obj(a_sg2)
            print("  Subgrad2: obj=%.6f" % obj_sg2)
            if obj_sg2 < global_best_obj:
                global_best_obj = obj_sg2
                global_best_a = a_sg2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f (n=%d) ===" % (ver, len(final_list)))
        else:
            print("WARNING: rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
