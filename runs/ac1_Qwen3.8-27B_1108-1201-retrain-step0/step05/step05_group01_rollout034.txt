"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.
Uses LP-based alternating projection (for moderate n) combined with
FISTA LSE-smoothed gradient descent (for large n) and L-BFGS-B polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import minimize as scipy_minimize
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def conv_gradient(y, w):
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def run_lp_alternating(a_init, n, max_time, rng, step_size=0.3, lp_solver=cp.CLARABEL):
    """LP-based alternating method."""
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 0, 1000).copy()
    a = a * (S / a.sum())
    
    # Precompute matrix structure
    rows_list = []
    cols_list = []
    for i in range(n):
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        rows_list.extend(range(k_min, k_max + 1))
        cols_list.extend([i] * (k_max - k_min + 1))
    rows = np.array(rows_list)
    cols = np.array(cols_list)
    idx_diff = rows - cols  # k - i for each entry
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best_conv - 1e-14:
            best_conv = M
            best_a = a.copy()
            if it % 10 == 1:
                print("    LP iter %d: max_conv=%.8f, obj=%.6f" % (it, M, 2*n*M/(S*S)))
        
        # Build matrix values
        vals = a[idx_diff]
        C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve LP
        try:
            g_var = cp.Variable(n)
            ones_vec = np.ones(2*n-1)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [C @ g_var <= M * ones_vec, g_var >= 0])
            prob.solve(solver=lp_solver)
            g = g_var.value
            if g is None:
                break
            g = np.clip(g, 0, 1000)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g = g * (S / g_sum)
            
            # Update
            t = step_size
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new * (S / a_new.sum())
            a = a_new
            
        except Exception as e:
            print("    LP solve error: %s" % str(e))
            break
    
    return best_a, best_conv

def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with increasing beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    betas = [30.0, 80.0, 200.0, 500.0, 1200.0]
    
    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            # Add small penalty for deviation from target sum
            sum_x = x.sum()
            obj += 10.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 20.0 * (sum_x - S)
            return obj, grad
        
        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            a_new = np.clip(result.x, 0, 1000)
            a_new = a_new * (S / max(a_new.sum(), 1e-15))
            obj_new = compute_obj(a_new)
            elapsed = time.monotonic() - t0
            print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs)" % (beta, obj_new, elapsed))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception as e:
            print("    L-BFGS error: %s" % str(e))
            break
    
    return best_a, best_obj

def resample_sequence(a, n_new):
    """Resample a sequence to a new length using cubic interpolation."""
    from scipy.interpolate import CubicSpline
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    cs = CubicSpline(x_old, a, bc_type='not-a-knot')
    a_new = cs(x_new)
    a_new = np.maximum(a_new, 0)
    return a_new

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
