"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Newton-type optimization with LSE smoothing (CG for Hessian solve),
combined with basin hopping via Fourier-domain mutations and peak-targeted perturbations.
Final polish with very high beta LSE-Newton and KKT-guided adjustments.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def correlation_vectors(a, lags):
    """
    For given lags, compute v_k[j] = a[j+k] + a[j-k] (with zero extension).
    Returns array of shape (m, n) where m = len(lags).
    """
    n = len(a)
    m = len(lags)
    G = np.zeros((m, n))
    for i, k in enumerate(lags):
        # a[j+k] for j = 0..n-1-k, zero elsewhere
        if k >= 0 and k < n:
            G[i, :n-k] = a[k:]
        # a[j-k] for j = k..n-1, zero elsewhere
        if k >= 0 and k < n:
            G[i, k:] += a[:n-k]
        elif k < 0:
            kk = -k
            if kk < n:
                G[i, kk:] += a[:n-kk]
        if k < 0:
            kk = -k
            if kk < n:
                G[i, :n-kk] += a[kk:]
    return G

        # v_k = a[j+k] + a[j-k], compute v_k^T v
        vkTv = 0.0
        # a[j+k] * v[j]
        if 0 <= k < n:
            vkTv += np.dot(a[k:], v[:n-k])
            # a[j-k] * v[j]
            vkTv += np.dot(a[:n-k], v[k:])
        elif k < 0:
            kk = -k
            if kk < n:
                vkTv += np.dot(a[:n-kk], v[kk:])
        
        result += wi * vkTv  # This is wrong, need v_k * (v_k^T v)
        # Actually result += wi * (v_k^T v) * v_k
        # Let me redo this properly
    
    # Let me redo: H*v = [sum_k w_k * v_k * (v_k^T v)] - [g * (g^T v)] + [sum_k w_k * (shift_k(v) + shift_{-k}(v))]
    
    result = np.zeros(n)
    gTv = np.dot(grad_F, v)
    
    for i in range(m):
        wi = weights[i]
        if wi < 1e-12:
            continue
        k = lags[i]
        # Compute v_k vector components and v_k^T v
        # v_k[j] = a[j+k] + a[j-k]
        vkTv = 0.0
        # Contribution to result from v_k * (v_k^T v)
        # v_k[j] = a[j+k] + a[j-k]
        # result[j] += wi * v_k[j] * vkTv
        
        # Compute vkTv = sum_j (a[j+k]+a[j-k]) * v[j]
        if 0 <= k < n:
            vkTv = np.dot(a[k:], v[:n-k]) + np.dot(a[:n-k], v[k:])
        elif k < 0:
            kk = -k
            if kk < n:
                vkTv = np.dot(a[:n-kk], v[kk:])
        
        # Now add wi * vkTv * v_k to result
        if 0 <= k < n:
            result[:n-k] += wi * vkTv * a[k:]
            result[k:] += wi * vkTv * a[:n-k]
        elif k < 0:
            kk = -k
            if kk < n:
                result[kk:] += wi * vkTv * a[:n-kk]
        
        # Banded Hessian part: wi * (v[j+k] + v[j-k])
        if 0 <= k < n:
            result[:n-k] += wi * v[k:]  # v[j+k] for j=0..n-1-k means v[k..n-1]
            result[k:] += wi * v[:n-k]  # v[j-k] for j=k..n-1 means v[0..n-1-k]
        elif k < 0:
            kk = -k
            if kk < n:
                result[kk:] += wi * v[:n-kk]  # v[j+k] = v[j-kk] for j=kk..n-1
                # v[j-k] = v[j+kk] for j=0..n-1-kk-1... wait
                # k = -kk, so j-k = j+kk. For j=0..n-1, j+kk ranges from kk to n-1+kk
                # valid when j+kk < n, i.e., j < n-kk
                result[:n-kk] += wi * v[kk:]  # v[j+kk] for j=0..n-1-kk

    # Subtract g * (g^T v)
    result -= grad_F * gTv
    
    return result


def lse_newton_step(a, S, beta, max_cg_iter=50, tol_cg=1e-6, max_iters=200, time_limit=None):
    """
    Newton optimization of LSE-smoothed max convolution.
    Returns (best_a, best_max_c, best_obj).
    """
    n = len(a)
    a = normalize_to_sum(a, S).copy()
    
    if time_limit is None:
        time_limit = 60.0
    start = time.monotonic()
    
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_max_c = c.max()
    
    obj = 2.0 * n * best_max_c / (a.sum() ** 2)
    best_obj = obj
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        c = fft_convolve_self(a)
        cmax = c.max()
        if cmax < best_max_c - 1e-12:
            best_max_c = cmax
            best_a = a.copy()
            best_obj = 2.0 * n * cmax / (a.sum() ** 2)
        
        # Compute LSE weights
        scaled = beta * (c - cmax)
        # Clip to avoid overflow
        scaled = np.clip(scaled, -500, 500)
        exp_scaled = np.exp(scaled)
        Z = exp_scaled.sum()
        if Z < 1e-300:
            break
        w = exp_scaled / Z
        
        # Find significant lags (top m by weight)
        m_max = min(40, len(c))
        top_idx = np.argsort(w)[::-1][:m_max]
        # Only keep those with w > threshold
        sig_idx = top_idx[w[top_idx] > 1e-6]
        if len(sig_idx) == 0:
            sig_idx = top_idx[:1]
        sig_idx = sig_idx[np.argsort(sig_idx)]  # Sort by lag
        sig_w = w[sig_idx]
        sig_w = sig_w / sig_w.sum()  # Renormalize
        
        # Compute gradient: g[j] = sum_k sig_w[k] * (a[j+k] + a[j-k])
        g = np.zeros(n)
        for i, k in enumerate(sig_idx):
            wi = sig_w[i]
            if wi < 1e-12:
                continue
            if 0 <= k < n:
                g[:n-k] += wi * a[k:]
                g[k:] += wi * a[:n-k]
            elif k < 0:
                kk = -k
                if kk < n:
                    g[kk:] += wi * a[:n-kk]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # CG to solve H * d = -g
        # H*v computed via hessian_matvec
        r = -g.copy()
        p = r.copy()
        rz = np.dot(r, r)
        if rz < 1e-30:
            break
        
        d_newton = np.zeros(n)
        for cg_it in range(max_cg_iter):
            Hp = hessian_matvec(p, a, sig_w, sig_idx, g)
            # Add small regularization for numerical stability
            Hp += 1e-10 * p
            pHp = np.dot(p, Hp)
            if abs(pHp) < 1e-30:
                break
            alpha_cg = rz / pHp
            d_newton += alpha_cg * p
            r_new = r - alpha_cg * Hp
            rz_new = np.dot(r_new, r_new)
            if np.sqrt(rz_new) < tol_cg * np.sqrt(rz) or np.sqrt(rz_new) < 1e-10:
                r = r_new
                break
            beta_cg = rz_new / rz
            p = r_new + beta_cg * p
            r = r_new
            rz = rz_new
        
        # Line search
        dn = np.linalg.norm(d_newton)
        if dn < 1e-15:
            break
        
        # Ensure the Newton direction decreases the Hessian curvature
        # (it should by construction, but check)
        step = 1.0
        best_step_obj = best_obj
        best_step_a = None
        
        for ls_it in range(20):
            a_trial = project_simplex(a + step * d_newton, S)
            c_trial = fft_convolve_self(a_trial)
            max_trial = c_trial.max()
            obj_trial = 2.0 * n * max_trial / (a_trial.sum() ** 2)
            
            if obj_trial < best_obj - 1e-12 or obj_trial < 2.0 * n * cmax / (a.sum()**2) * (1 - 1e-4):
                best_step_obj = obj_trial
                best_step_a = a_trial
                break
            step *= 0.5
            if step < 1e-10:
                break
        
        if best_step_a is not None:
            a = best_step_a
        else:
            # Fallback: small gradient step
            step_fb = 0.01 * S / n / max(gnorm / (S/n), 1e-10)
            a = project_simplex(a - step_fb * g, S)
        
        # Check for stagnation
        c_check = fft_convolve_self(a)
        if c_check.max() < best_max_c:
            best_max_c = c_check.max()
            best_a = a.copy()
            best_obj = 2.0 * n * best_max_c / (a.sum() ** 2)
    
    return best_a, best_max_c, best_obj


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain mutation: perturb phase, IFFT, project to simplex."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = np.abs(A) * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def peak_targeted_perturb(a, c, rng, n_perturb=50):
    """
    Perturb coordinates that contribute most to the current peak.
    This helps escape local optima where the peak is "stuck."
    """
    n = len(a)
    S = a.sum()
    k_star = np.argmax(c)
    
    # Coordinates that contribute most to c[k_star] are those where a[j]*a[j+k_star] is large
    # i.e., where a[j] and a[j+k_star] are both large
    contrib = np.zeros(n)
    if 0 <= k_star < n:
        contrib[:n-k_star] = a[:n-k_star] * a[k_star:]
        contrib[k_star:] += a[k_star:] * a[:n-k_star]
    else:
        kk = -k_star
        if kk < n:
            contrib[kk:] = a[kk:] * a[:n-kk]
            contrib[:n-kk] += a[:n-kk] * a[kk:]
    
    # Select top contributors to perturb
    top = np.argsort(contrib)[::-1][:n_perturb]
    a_new = a.copy()
    # Randomly increase/decrease these coordinates
    deltas = rng.normal(0, 0.05 * S / n, n_perturb)
    a_new[top] += deltas
    return project_simplex(a_new, S)


def block_swap_perturb(a, rng, n_blocks=4):
    """Randomly swap blocks of the sequence."""
    n = len(a)
    S = a.sum()
    a_new = a.copy()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        a_new[i:i+block_size], a_new[j:j+block_size] = a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
    return project_simplex(a_new, S)


def run_newton_optimize(a_init, S, time_limit, rng, beta_start=30, beta_end=500, 
                        n_outer=5):
    """
    Run Newton optimization with increasing beta.
    Each outer iteration uses a higher beta for sharper peak focusing.
    """
    a = normalize_to_sum(a_init, S).copy()
    start = time.monotonic()
    remaining = time_limit
    
    c = fft_convolve_self(a)
    best_max_c = c.max()
    best_a = a.copy()
    best_obj = 2.0 * len(a) * best_max_c / (a.sum() ** 2)
    
    n_outer = max(2, int(time_limit / 30))
    time_per_outer = remaining / n_outer
    
    for outer in range(n_outer):
        if time.monotonic() - start > time_limit - 5:
            break
        progress = outer / max(1, n_outer - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        t_alloc = min(time_per_outer, time_limit - (time.monotonic() - start))
        if t_alloc < 5:
            break
        
        a, max_c, obj = lse_newton_step(a, S, beta, time_limit=t_alloc)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_max_c = max_c
            print("      Newton outer %d: beta=%.1f, obj=%.8f" % (outer, beta, obj))
    
    return best_a, best_obj


def run_basin_hopping(a_init, S, time_limit, rng):
    """
    Basin hopping with diverse mutation types and Newton local optimization.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S).copy()
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (a.sum() ** 2)
    best_a = a.copy()
    
    # Optimize time per iteration
    opt_time = 15.0
    n_iters = max(1, int((time_limit - 30) / (opt_time + 2)))
    
    # Population for diversity
    population = [(best_obj, a.copy())]
    
    stuck_count = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit - opt_time - 5:
            break
        progress = it / max(1, n_iters)
        
        # Adaptive mutation scale
        base_scale = 0.3 * np.exp(-1.5 * progress) + 0.03
        
        # Choose mutation type
        r = rng.random()
        if len(population) >= 2 and r < 0.2:
            # Crossover
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a1, a2 = population[i1][1], population[i2][1]
                fsize = 1 << (2 * n - 1).bit_length()
                A1 = fft(a1, fsize)
                A2 = fft(a2, fsize)
                r2 = rng.random(fsize)
                alpha = 0.3 + 0.4 * r2
                new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
                new_phase = np.where(r2 < 0.5, np.angle(A1), np.angle(A2))
                new_A = new_mag * np.exp(1j * new_phase)
                new_A[0] = A1[0]
                a = np.real(ifft(new_A, fsize))[:n]
                a = project_simplex(a, S)
            else:
                a = fourier_phase_mutate(a, rng, base_scale)
        elif r < 0.5:
            a = fourier_phase_mutate(a, rng, base_scale)
        elif r < 0.7:
            # Peak-targeted perturbation
            c_curr = fft_convolve_self(a)
            a = peak_targeted_perturb(a, c_curr, rng, n_perturb=max(20, n // 100))
        elif r < 0.85:
            a = block_swap_perturb(a, rng, n_blocks=rng.integers(2, 8))
        else:
            # Coordinate-level random perturbation
            n_perturb = rng.integers(n // 10, n // 3)
            idx = rng.choice(n, n_perturb, replace=False)
            a_temp = a.copy()
            a_temp[idx] *= np.exp(rng.normal(0, base_scale, n_perturb))
            a = project_simplex(a_temp, S)
        
        # Local optimization
        t_opt = min(opt_time, time_limit - (time.monotonic() - start) - 5)
        if t_opt < 5:
            break
        a_opt, obj_opt = run_newton_optimize(a, S, t_opt, rng, 
                                              beta_start=20 + 30 * min(it, 10),
                                              beta_end=200 + 100 * min(it, 10),
                                              n_outer=max(2, int(t_opt / 10)))
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            stuck_count = 0
            print("    BH iter %d: new best obj=%.8f" % (it, best_obj))
        else:
            stuck_count += 1
        
        # Maintain population
        population.append((obj_opt, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]
    
    return best_a, best_obj


def kkt_check(a, tol=1e-4):
    """
    Check KKT conditions. Returns (is_optimal, violation).
    At optimum: for j with a_j > 0, sum_k lambda_k * (a[j+k]+a[j-k]) = mu
    """
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    active = np.where(c >= cmax - tol * cmax)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Use uniform weights on active set
    lam = np.ones(len(active)) / len(active)
    
    # Compute g[j] = sum_k lam_k * (a[j+k] + a[j-k])
    g = np.zeros(n)
    for i, k in enumerate(active):
        wi = lam[i]
        if 0 <= k < n:
            g[:n-k] += wi * a[k:]
            g[k:] += wi * a[:n-k]
        elif k < 0:
            kk = -k
            if kk < n:
                g[kk:] += wi * a[:n-kk]
    
    # For a_j > 0: g[j] should equal mu
    # For a_j = 0: g[j] should be >= mu
    positive = a > 1e-10
    if not np.any(positive):
        return True, 0.0
    
    mu = g[positive].mean()
    violation = np.max(np.abs(g[positive] - mu)) if np.any(positive) else 0.0
    
    # Also check that g[j] >= mu for j where a_j = 0
    zero = a <= 1e-10
    if np.any(zero):
        v2 = np.max(np.maximum(0, mu - g[zero]))
        violation = max(violation, v2)
    
    return violation < 1e-3, float(violation)


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    
    if given_seq is not None and len(given_seq) == n:
        inits.append(np.clip(given_seq, 0, 1000).copy())
    
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Various window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25) + 0.01)
    
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)
    
    # Block structures
    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)
    
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search:
    1. Quick Newton screening from diverse inits
    2. Extended Newton optimization from best candidates  
    3. Basin hopping with diverse mutations
    4. Final high-precision Newton polish with KKT checking
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20
    
    print("Starting optimization, budget=%.0fs, seed=%d" % (budget_s, seed))
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass
    
    n = 8192
    S = np.sqrt(2.0 * n)
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, val):
        nonlocal global_best, global_a
        if val < global_best:
            global_best = val
            global_a = a.copy()
            return True
        return False
    
    # Phase 1: Quick screening (120s)
    print("\n=== Phase 1: Quick screening ===")
    p1_budget = min(120.0, (deadline - total_start) * 0.10)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(6, len(inits))
    per_init = p1_budget / n_screen
    
    candidates = []
    for i in range(n_screen):
        t_remain = deadline - time.monotonic() - 60
        if t_remain < 15:
            break
        t_alloc = min(per_init, t_remain)
        if t_alloc < 10:
            break
        a_opt, obj = run_newton_optimize(inits[i], S, t_alloc, rng,
                                          beta_start=20, beta_end=200, n_outer=3)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.8f" % (i, obj))
        update_global(a_opt, obj)
    
    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.8f" % global_best)
    
    # Phase 2: Extended Newton from best candidates (400s)
    print("\n=== Phase 2: Extended Newton optimization ===")
    p2_budget = min(400.0, (deadline - time.monotonic()) * 0.25)
    
    n_ext = min(3, len(candidates))
    time_per_ext = p2_budget / n_ext
    for i in range(n_ext):
        t_remain = deadline - time.monotonic() - 30
        if t_remain < 30:
            break
        t_alloc = min(time_per_ext, t_remain)
        if t_alloc < 20:
            break
        print("  Extended Newton from candidate %d (obj=%.8f), t=%.0fs" % (i, candidates[i][0], t_alloc))
        a_opt, obj = run_newton_optimize(candidates[i][1], S, t_alloc, rng,
                                          beta_start=30, beta_end=500, 
                                          n_outer=max(3, int(t_alloc / 20)))
        if obj < global_best:
            update_global(a_opt, obj)
            print("    New global best: obj=%.8f" % global_best)
        else:
            print("    No improvement, obj=%.8f" % obj)
    
    # Phase 3: Basin hopping (1000s)
    print("\n=== Phase 3: Basin hopping ===")
    p3_budget = min(1000.0, (deadline - time.monotonic()) * 0.50)
    
    if global_a is not None and p3_budget > 60:
        # Run basin hopping from best
        bh_time = p3_budget * 0.6
        print("  BH from global best (obj=%.8f), t=%.0fs" % (global_best, bh_time))
        a_bh, obj_bh = run_basin_hopping(global_a, S, bh_time, rng)
        if obj_bh < global_best:
            update_global(a_bh, obj_bh)
            print("  BH improved: obj=%.8f" % global_best)
        else:
            print("  BH best: obj=%.8f (no improvement)" % obj_bh)
        
        # Additional BH from second-best if time remains
        t_remain = deadline - time.monotonic() - 200
        if t_remain > 120 and len(candidates) >= 2:
            bh_time2 = min(t_remain * 0.5, 300)
            print("  BH from candidate 1 (obj=%.8f), t=%.0fs" % (candidates[1][0], bh_time2))
            a_bh2, obj_bh2 = run_basin_hopping(candidates[1][1], S, bh_time2, rng)
            if obj_bh2 < global_best:
                update_global(a_bh2, obj_bh2)
                print("  BH2 improved: obj=%.8f" % global_best)
    
    # Phase 4: Final high-precision polish (remaining time)
    print("\n=== Phase 4: Final polish ===")
    p4_budget = deadline - time.monotonic() - 10
    
    if global_a is not None and p4_budget > 30:
        # Very high beta Newton
        seg1 = p4_budget * 0.4
        print("  High-beta Newton, t=%.0fs" % seg1)
        a_hb, obj_hb = run_newton_optimize(global_a, S, seg1, rng,
                                            beta_start=200, beta_end=2000,
                                            n_outer=max(2, int(seg1 / 15)))
        if obj_hb < global_best:
            update_global(a_hb, obj_hb)
            print("  High-beta improved: obj=%.8f" % global_best)
        
        # KKT check
        t_remain = deadline - time.monotonic() - 10
        if t_remain > 10:
            is_opt, violation = kkt_check(global_a, tol=1e-4)
            print("  KKT check: optimal=%s, violation=%.2e" % (is_opt, violation))
        
        # If KKT violation is large, do targeted fix
        if not is_opt and t_remain > 30:
            print("  Targeted KKT fix, t=%.0fs" % (t_remain - 5))
            a_fix, obj_fix = run_newton_optimize(global_a, S, t_remain - 5, rng,
                                                  beta_start=500, beta_end=5000,
                                                  n_outer=max(2, int((t_remain-5) / 10)))
            if obj_fix < global_best:
                update_global(a_fix, obj_fix)
                print("  KKT fix improved: obj=%.8f" % global_best)
    
    # Final verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback used: obj=%.8f" % ver)
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
