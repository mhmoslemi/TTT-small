"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def adaptive_topk_lse_gradient(a, c, beta, k_frac_range=(0.02, 0.15)):
    """Compute gradient of adaptive top-k LSE of self-convolution."""
    n = len(a)
    conv_len = len(c)
    
    # Adaptive K: look at the peak structure
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    c_mean = c.mean()
    
    # If the max is much larger than the mean, use small K
    # If the max is close to the mean, use larger K
    ratio = c_max / max(c_mean, 1e-15)
    if ratio > 5.0:
        k_frac = k_frac_range[0]
    elif ratio > 2.0:
        k_frac = (k_frac_range[0] + k_frac_range[1]) / 2
    else:
        k_frac = k_frac_range[1]
    
    k = max(4, int(k_frac * conv_len))
    k = min(k, conv_len)
    
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    
    # Prevent overflow
    scaled_max = scaled.max()
    if scaled_max > 500:
        beta = 500.0 / max(1e-15, (c_top - cm).max())
        if beta <= 0:
            beta = 1e-6
        scaled = beta * (c_top - cm)
    
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(k)
        w[0] = 1.0
    else:
        w = e / tot
    
    w_full = np.zeros(conv_len)
    w_full[idx] = w
    
    fsize = 1 << (conv_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    
    ratio = power / max(mean_p, 1e-20)
    # Damp high-power frequencies, boost low-power
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio * 0.5))
    new_power = np.maximum(new_power, 1e-30)
    
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def random_phase_init(n, rng, S):
    """Generate a sequence with random phases in frequency domain, then clip to non-negative."""
    fsize = 1 << (2 * n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fsize)
    # Enforce conjugate symmetry for real signal
    for i in range(1, fsize // 2 + 1):
        phases[fsize - i] = -phases[i]
    
    A = np.exp(1j * phases)
    a = np.real(ifft(A, fsize))[:n]
    a = np.clip(a, 0, None)
    return normalize(a, S)


def run_fista_adaptive(a_init, S, max_time, rng,
                       beta_init=10.0, beta_rate=0.004, beta_max=5000.0,
                       k_frac_low=0.02, k_frac_high=0.15,
                       use_momentum=True, restart_interval=1000):
    """FISTA with adaptive top-k LSE."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = adaptive_topk_lse_gradient(y, c, beta, (k_frac_low, k_frac_high))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > restart_interval:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Periodic spectral perturbation
        if it % 2000 == 0 and rng.random() < 0.25:
            y = spectral_flatten(y, 0.1, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_grient_descent(a_init, S, max_time, rng,
                       beta_init=50.0, beta_rate=0.01, beta_max=10000.0,
                       lr_init=0.01, lr_decay=0.999):
    """Plain gradient descent with adaptive LSE, no momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    lr = lr_init

    while time.monotonic() - start < max_time:
        beta = min(beta_max, beta_init * np.exp(beta_rate))
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = adaptive_topk_lse_gradient(a, c, beta, (0.01, 0.1))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_new = project_simplex(a - lr * grad / max(gnorm, 1e-10), S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
            # Slightly increase lr
            lr = min(lr * 1.01, lr_init * 2)
        else:
            lr *= 0.7
            a = a_new

        # Decay lr over time
        lr *= lr_decay

    return best_a, best


def run_combinatorial_polish(a_init, S, max_time, rng):
    """Try local modifications: swap, merge, split elements."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # For small n, try combinatorial moves
    if n > 2048:
        # Too large for combinatorial moves, just do gradient-based
        return run_fista_adaptive(a, S, max_time, rng,
                                  beta_init=200.0, beta_rate=0.002,
                                  beta_max=50000.0, use_momentum=False)
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find the position of the max
        k_star = np.argmax(c)
        
        # Try to reduce the max by adjusting elements
        # The elements that contribute to c[k_star] are a[j] * a[k_star - j]
        # We want to reduce products at this position
        # Strategy: find the largest product and reduce one of the factors
        
        if k_star < n:
            # a[j] * a[k_star - j] for j = 0..k_star
            products = a[:k_star+1] * a[k_star::-1]
        elif k_star == n - 1:
            products = a * a[::-1]
        else:
            # k_star >= n: a[k_star - n + 1 : n] * a[n - 1 : k_star - n : -1]
            j_start = k_star - n + 1
            j_end = n
            products = a[j_start:j_end] * a[n-1:(n-1)-(j_end-j_start)+1:-1]
        
        if len(products) > 0 and products.max() > 0:
            # Find which pair contributes most
            max_idx = np.argmax(products)
            # Reduce one of the two elements
            if k_star < n:
                i1 = max_idx
                i2 = k_star - max_idx
            elif k_star == n - 1:
                i1 = max_idx
                i2 = k_star - max_idx
            else:
                i1 = (k_star - n + 1) + max_idx
                i2 = n - 1 - max_idx
            
            if i1 != i2 and 0 <= i1 < n and 0 <= i2 < n:
                # Move mass from the larger to the smaller (or to a random position)
                delta = 0.01 * a[i1]
                a[i1] -= delta
                # Add to a random position far from i2
                target = rng.integers(0, n)
                while abs(target - i2) < n // 4:
                    target = rng.integers(0, n)
                a[target] += delta
                a = project_simplex(np.maximum(a, 0), S)
        
        # Also try random perturbation
        if it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

            step *= 0.4
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Restart when stuck
        if stuck > stuck_restart:
            noise = noise_scale * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_val = 1.0
            stuck = 0

        # Spectral perturbation
        if it % spectral_interval == 0 and rng.random() < 0.3:
            y = spectral_flatten(y, spectral_alpha, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_subgradient_polish(a_init, S, max_time, rng):
    """Polish using exact subgradient of max with small steps."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = exact_max_gradient(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.3 * step_scale
        a_new = project_simplex(a - step * grad / max(gnorm, 1e-10), S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a = a_new

        # Occasional perturbation
        if it % 500 == 0 and step_scale < 0.1:
            noise = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best


def run_spectral_only(a_init, S, max_time, rng):
    """Pure spectral flattening iteration."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Vary alpha
        alpha = 0.05 + 0.15 * np.sin(0.1 * it) ** 2
        a = spectral_flatten(a, alpha, rng)
        cur_max = fft_convolve_self(a).max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        # Occasionally add gradient step
        if it % 50 == 0:
            c = fft_convolve_self(a)
            grad = lse_gradient(a, c, max(4, int(0.02 * len(c))), 100.0)
            a = project_simplex(a - 0.01 * grad / max(np.linalg.norm(grad), 1e-10), S)

    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: diverse init -> FISTA -> spectral -> subgradient polish.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: len=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def compute_obj(a):
        a = np.clip(a, 0.0, 1000.0)
        n = len(a)
        s = a.sum()
        if s < 0.01:
            return np.inf
        b = fft_convolve_self(a)
        return 2.0 * n * b.max() / (s * s)

    def try_sequence(a, desc=""):
        nonlocal global_best, global_a
        a_clipped = np.clip(a, 0.0, 1000.0)
        obj = compute_obj(a_clipped)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_clipped.copy()
            print("  [%.0fs] %s: obj=%.6f (n=%d) *** NEW BEST ***" %
                  (time.monotonic() - start_total, desc, obj, len(a)))
        else:
            print("  [%.0fs] %s: obj=%.6f" % (time.monotonic() - start_total, desc, obj))
        return a_clipped

    # Determine primary length
    n_primary = 8192
    S = 1.0  # Scale-invariant, use S=1 for numerical stability

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse initializations ===")
    inits = {}

    if given is not None:
        if len(given) == n_primary:
            inits['given'] = normalize(given.copy(), S)
        else:
            inits['given'] = normalize(resample(given, n_primary), S)

    inits['uniform'] = np.ones(n_primary) * S / n_primary
    inits['random_phase'] = random_phase_init(n_primary, rng, S)

    t_arr = np.linspace(-1, 1, n_primary)
    inits['gaussian'] = normalize(np.exp(-2.0 * t_arr ** 2), S)
    inits['triangular'] = normalize(np.maximum(0, 1 - np.abs(t_arr)), S)

    # Absolute value of random phase signal (ensures non-negativity)
    fsize_init = 1 << (2 * n_primary - 1).bit_length()
    ph_init = rng.uniform(0, 2 * np.pi, fsize_init)
    for i in range(1, fsize_init // 2 + 1):
        ph_init[fsize_init - i] = -ph_init[i]
    a_abs = np.abs(np.real(ifft(np.exp(1j * ph_init), fsize_init)))[:n_primary]
    inits['abs_phase'] = normalize(a_abs, S)

    for name, a_init in inits.items():
        try_sequence(a_init, "init:" + name)

    # === Phase 2: Quick FISTA on each initialization ===
    print("\n=== Phase 2: Quick FISTA on diverse starts ===")
    quick_time = 80

    for name in ['given', 'random_phase', 'abs_phase', 'gaussian', 'uniform', 'triangular']:
        if name not in inits:
            continue
        remaining = deadline - time.monotonic()
        if remaining < quick_time + 90:
            break
        t_alloc = min(quick_time, remaining - 90)
        if t_alloc < 15:
            break
        print("  FISTA on '%s' for %.0fs..." % (name, t_alloc))
        a_opt, val_opt = run_fista(inits[name], S, t_alloc, rng,
                                   beta_init=5.0, beta_rate=0.006,
                                   beta_max=2000.0,
                                   k_frac_low=0.02, k_frac_high=0.12,
                                   use_momentum=True, stuck_restart=500,
                                   spectral_interval=800, spectral_alpha=0.15)
        try_sequence(a_opt, "FISTA(" + name + ")")

    # === Phase 3: Intensive FISTA on best ===
    print("\n=== Phase 3: Intensive FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        t_alloc = min(600, remaining * 0.35)
        print("  Running %.0fs..." % t_alloc)
        a_opt, val_opt = run_fista(global_a, S, t_alloc, rng,
                                   beta_init=30.0, beta_rate=0.003,
                                   beta_max=8000.0,
                                   k_frac_low=0.01, k_frac_high=0.06,
                                   use_momentum=True, stuck_restart=800,
                                   spectral_interval=1200, spectral_alpha=0.1)
        try_sequence(a_opt, "Intensive FISTA")

    # === Phase 4: Try n=4096 ===
    print("\n=== Phase 4: Try n=4096 ===")
    n_sec = 4096
    S_sec = 1.0
    remaining = deadline - time.monotonic()
    if remaining > 200:
        # From resampled best
        a_sec1 = None
        if global_a is not None:
            a_sec1 = normalize(resample(global_a, n_sec), S_sec)
        a_sec2 = random_phase_init(n_sec, rng, S_sec)

        t_each = min(120, remaining * 0.15)
        if t_each > 20:
            print("  n=4096 from resampled, %.0fs..." % t_each)
            a_opt1, _ = run_fista(a_sec1, S_sec, t_each, rng,
                                  beta_init=5.0, beta_rate=0.006,
                                  beta_max=2000.0,
                                  k_frac_low=0.03, k_frac_high=0.15)
            try_sequence(a_opt1, "n4096_resampled")

            remaining = deadline - time.monotonic()
            t_each2 = min(120, remaining * 0.1)
            if t_each2 > 20:
                print("  n=4096 from random_phase, %.0fs..." % t_each2)
                a_opt2, _ = run_fista(a_sec2, S_sec, t_each2, rng,
                                      beta_init=5.0, beta_rate=0.006,
                                      beta_max=2000.0,
                                      k_frac_low=0.03, k_frac_high=0.15)
                try_sequence(a_opt2, "n4096_random")

    # === Phase 5: Spectral polishing on global best ===
    print("\n=== Phase 5: Spectral polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        t_alloc = min(150, remaining * 0.2)
        print("  Spectral only for %.0fs..." % t_alloc)
        a_opt, _ = run_spectral_only(global_a, global_a.sum(), t_alloc, rng)
        try_sequence(a_opt, "Spectral polish")

    # === Phase 6: Subgradient fine polish ===
    print("\n=== Phase 6: Subgradient fine polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_a is not None:
        t_alloc = min(200, remaining * 0.3)
        print("  Subgradient polish for %.0fs..." % t_alloc)
        a_opt, _ = run_subgradient_polish(global_a, global_a.sum(), t_alloc, rng)
        try_sequence(a_opt, "Subgradient polish")

    # === Phase 7: Final high-beta FISTA ===
    print("\n=== Phase 7: Final high-beta polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        t_alloc = remaining * 0.5
        S_best = global_a.sum()
        a_opt, _ = run_fista(global_a, S_best, t_alloc, rng,
                             beta_init=500.0, beta_rate=0.001,
                             beta_max=50000.0,
                             k_frac_low=0.005, k_frac_high=0.02,
                             use_momentum=False, stuck_restart=3000,
                             spectral_interval=5000, spectral_alpha=0.05,
                             noise_scale=0.001)
        try_sequence(a_opt, "Final high-beta")

    # === Phase 8: One more subgradient pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        t_alloc = remaining * 0.7
        a_opt, _ = run_subgradient_polish(global_a, global_a.sum(), t_alloc, rng)
        try_sequence(a_opt, "Final subgradient")

    # Verification
    print("\n=== Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            print("  Sum too small, fallback")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
    else:
        print("  No valid sequence, fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    print("Final: len=%d, obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
