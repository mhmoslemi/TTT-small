def evaluate_sequence(seq):
    """Evaluate the sequence's objective: 2*n*max(conv(a,a)) / sum(a)^2, with input validation."""
    n = len(seq)
    # Convert to numpy float array for safe handling
    seq = np.array(seq, dtype=np.float64)
    # Reject invalid inputs
    if not seq.flags.writeable:
        return np.inf
    if np.any(seq < 0) or np.any(np.isnan(seq)) or np.isinf(seq):
        return np.inf
    # Clipping to 0
    seq = np.maximum(seq, 0.0)
    # Handle if sum is too low (defensive clip)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # Compute max of linear convolution
    conv = np.correlate(seq, seq, mode='full')
    max_conv = np.max(np.abs(conv))
    return float(2 * n * max_conv / (s ** 2))
