from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (gradient, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def hard_subgradient(a, c, top_k=3):
    """Hard subgradient using top_k lags achieving (near) max."""
    n = len(a)
    cm = c.max()
    # Get top_k lags
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weight by how close to max
    weights = np.exp(100.0 * (c[sorted_idx] - cm))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(sorted_idx, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        partners = k - js
        grad[js] += 2.0 * w * a[partners]
    return grad


def run_mirror_descent(a_init, n, S, max_time, rng,
                       beta_start=200.0, beta_end=80000.0,
                       eta_start=1.0, eta_decay=0.9998,
                       restart_interval=10000):
    """
    Mirror descent (multiplicative) with LSE smoothing.
    More natural for simplex than Euclidean FISTA in late stages.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    eta = eta_start
    last_improve = 0
    stagnation = 0
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best - 1e-14:
            best = cm
            best_a = a.copy()
            last_improve = it
            stagnation = 0
        else:
            stagnation += 1
        
        # Beta schedule: smooth ramp
        frac = min(1.0, it / max(1, restart_interval * 3))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 200000.0)
        
        # LSE gradient
        grad, _ = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Mirror descent: multiplicative update
        # a_new[j] = a[j] * exp(-eta * grad[j]) / Z
        scaled = eta * grad / gnorm * 0.5  # Normalize gradient, scale by eta
        log_w = -scaled
        log_w -= log_w.max()
        weights = np.exp(log_w)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-20)
        a = a_new / a_new.sum() * S
        
        eta *= eta_decay
        
        # Stagnation handling
        if stagnation > restart_interval:
            # Perturb and restart
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stagnation / 50000.0)
            if noise_scale > 1e-9:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            eta = eta_start * 0.5
            stochastic_rate = 0.1
            last_improve = it
            stagnation = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

def run_kkt_refinement(a_init, n, S, max_time, rng,
                       active_threshold_start=0.001,
                       active_threshold_end=0.0,
                       max_lags=10):

    start = time.monotonic()

    best_a = a.copy()

        cm = c.max()

            best = cm
            best_a = a.copy()

        frac = min(1.0, it / 5000.0)
        threshold = active_threshold_start * (1 - frac) + active_threshold_end * frac
        threshold = max(threshold, 1e-10 / (it + 1))

        active_mask = c >= cm - threshold
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > max_lags:
            # Keep only top max_lags
            top_positions = np.argsort(c[active_idx])[::-1][:max_lags]
            active_idx = active_idx[top_positions]
        elif len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])

        m = len(active_idx)
        grad = np.zeros(n)
        for k in active_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            partners = k - js
            grad[js] += 2.0 * a[partners]

            grad_tan = grad.copy()
            grad_tan[support] -= np.mean(grad[support])
        else:
            grad_tan = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / (it * 0.01 + 1.0)

def run_flattening(a_init, n, S, max_time, rng, n_rounds=200):
    """
    Novel "flattening" heuristic: directly target equalizing top and bottom
    convolution lags by transferring mass between positions.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        cm = c.max()

            best = cm
            best_a = a.copy()

        top_k = 5
        top_idx = np.argsort(c)[::-1][:top_k]
        # Bottom lags (to increase - they represent "room" in the distribution)
        bot_idx = np.argsort(c)[:3]
        
        # Compute responsibility for top lags
        resp_reduce = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Weight by excess over target (use mean as target)
            excess = max(0.0, c[k] - np.mean(c))
            resp_reduce[js] += contrib * excess
        
        # Compute "benefit" for bottom lags (what would increase them)
        resp_increase = np.zeros(n)
        for k in bot_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            partners = k - js
            # Increasing a[j] increases c[k] by a[partner] for each valid partner
            resp_increase[js] += a[partners]
        
        # Net direction: reduce where resp_reduce is high, increase where resp_increase is high
        direction = resp_increase * 0.3 - resp_reduce
        # Normalize
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        direction /= dnorm
        
        # Apply small step

        c_new = fft_convolve_self(a_new)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def chebyshev_window(n, main_lobe_width):
    """
    Generate a window inspired by Dolph-Chebyshev design.
    Higher at edges relative to a simple raised cosine, to flatten autocorrelation.
    """
    t = np.linspace(0, 1, n)
    # Use a "super" raised cosine with additional harmonics
    # w(x) = 0.5*(1+cos(pi*x/w))^2 + harmonics
    half = main_lobe_width / 2.0
    x = (t - 0.5) / half
    a = np.ones(n)
    mask = np.abs(x) < 1.0
    # Use a flatter-top window: sum of cosines (Nuttall-like)
    a[mask] = (1 + 0.5 * np.cos(np.pi * x[mask]) + 
               0.15 * np.cos(2 * np.pi * x[mask]) +
               0.03 * np.cos(3 * np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)
    
    # Chebyshev-inspired windows

    # Raised cosine variants
    for cf in [0.3, 0.4, 0.5]:

            mask = np.abs(x) < 1

            a[~mask] = 0.0
            inits.append(np.maximum(a, 1e-10))
    
    # Super-Gaussians
    for p in [1.5, 2.0, 3.0, 4.0]:
        for w in [0.15, 0.25, 0.35]:
            a = np.exp(-np.abs(t - 0.5) ** (2 * p) / (2 * w ** 2))
            inits.append(np.maximum(a, 1e-10))
    
    # Bimodal
    for sep in [0.15, 0.2, 0.3]:
        a = np.exp(-((t - 0.5 + sep) ** 2) / 0.01 +

        a = np.exp(-((t - (0.5 - sep)) ** 2) / 0.01) + np.exp(-((t - (0.5 + sep)) ** 2) / 0.01)
        inits.append(np.maximum(a, 1e-10))
    
    # Periodic modulations of uniform
    for k in [1, 2, 3]:
        for amp in [0.1, 0.3, 0.5]:
            a = 1.0 + amp * np.cos(2 * np.pi * k * t)
            inits.append(np.maximum(a, 1e-10))
    
    # Random smooth
    for _ in range(3):
        a = np.random.default_rng().normal(0, 1, n)

        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=20)
        inits.append(np.maximum(a, 1e-10))
    
    return inits

    Multi-phase: FISTA -> Mirror Descent -> KKT Refinement -> Flattening -> Final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))
    t0 = time.monotonic()
    
    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  ** New best: %.10f (n=%d) [t=%.0fs]" % (val, n, time.monotonic() - t0))
    
    # Determine working length
    if given is not None:
        n_work = len(given)
    else:
        n_work = 8192
    
    S_work = np.sqrt(2.0 * n_work)
    
    # === Phase 1: Quick FISTA on given ===
    if given is not None:
        remaining = deadline - time.monotonic()
        t1 = min(30.0, remaining * 0.08)
        print("=== Phase 1: Quick FISTA (n=%d, %.0fs) ===" % (n_work, t1))
        a1, v1 = run_fista_lse(given, n_work, S_work, t1, rng,
                                beta_start=200.0, beta_end=5000.0, beta_steps=50)
        try_sequence(a1)
    
    # === Phase 2: Mirror Descent (main optimization) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.30
        print("=== Phase 2: Mirror Descent (n=%d, %.0fs) ===" % (n_work, t2))
        a2, v2 = run_mirror_descent(global_a, n_work, S_work, t2, rng,
                                     beta_start=500.0, beta_end=100000.0,
                                     eta_start=0.8, eta_decay=0.99985)
        try_sequence(a2)
    
    # === Phase 3: KKT Active-Set Refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.20
        print("=== Phase 3: KKT Refinement (n=%d, %.0fs) ===" % (n_work, t3))
        a3, v3 = run_kkt_refinement(global_a, n_work, S_work, t3, rng,
                                     active_threshold_start=0.005,
                                     active_threshold_end=0.0,
                                     max_lags=8)
        try_sequence(a3)
    
    # === Phase 4: More FISTA with different params ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.15
        print("=== Phase 4: FISTA round 2 (n=%d, %.0fs) ===" % (n_work, t4))
        a4, v4 = run_fista_lse(global_a, n_work, S_work, t4, rng,
                                beta_start=2000.0, beta_end=80000.0, beta_steps=100,
                                step_scale=0.6, restart_gap=500)
        try_sequence(a4)
    
    # === Phase 5: Flattening heuristic ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t5 = remaining * 0.10
        print("=== Phase 5: Flattening (n=%d, %.0fs) ===" % (n_work, t5))
        a5, v5 = run_flattening(global_a, n_work, S_work, t5, rng)
        try_sequence(a5)
    
    # === Phase 6: Try alternative initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t6 = remaining * 0.20
        print("=== Phase 6: Alternative inits (n=%d, %.0fs) ===" % (n_work, t6))
        inits = generate_inits(n_work, rng)
        
        # Score inits quickly
        init_scores = []
        for i, init in enumerate(inits):
            init = normalize_sum(init, n_work)
            c = fft_convolve_self(init)
            init_scores.append((c.max(), i))
        init_scores.sort()
        
        per_init = max(2.0, t6 / 8.0)
        tested = 0
        for score, idx in init_scores:
            if time.monotonic() > deadline - 30:
                break
            if tested >= 8:
                break
            if score < global_best * 1.02:  # Only try promising inits
                init = normalize_sum(inits[idx], n_work)
                ai, vi = run_mirror_descent(init, n_work, S_work, per_init, rng,
                                             beta_start=200.0, beta_end=50000.0,
                                             eta_start=1.0, eta_decay=0.9998)
                try_sequence(ai)
                tested += 1
    
    # === Phase 7: Final intensive polishing ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            
            # 7a: Tight KKT with very few active lags
            t7a = remaining * 0.30
            print("=== Phase 7a: Tight KKT (n=%d, %.0fs) ===" % (n_f, t7a))
            a7a, v7a = run_kkt_refinement(global_a, n_f, S_f, t7a, rng,
                                           active_threshold_start=0.001,
                                           active_threshold_end=0.0,
                                           max_lags=3)
            try_sequence(a7a)
            
            # 7b: Mirror descent very high beta
            remaining = deadline - time.monotonic()
            if remaining > 10:
                t7b = remaining * 0.40
                print("=== Phase 7b: High-beta Mirror (n=%d, %.0fs) ===" % (n_f, t7b))
                a7b, v7b = run_mirror_descent(global_a, n_f, S_f, t7b, rng,
                                               beta_start=10000.0, beta_end=200000.0,
                                               eta_start=0.3, eta_decay=0.9999)
                try_sequence(a7b)
            
            # 7c: Final FISTA polish
            remaining = deadline - time.monotonic()
            if remaining > 5:
                t7c = remaining * 0.50
                print("=== Phase 7c: Final FISTA (n=%d, %.0fs) ===" % (n_f, t7c))
                a7c, v7c = run_fista_lse(global_a, n_f, S_f, t7c, rng,
                                          beta_start=5000.0, beta_end=150000.0,
                                          beta_steps=50, step_scale=0.7, restart_gap=300)
                try_sequence(a7c)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("RETURNING: length=%d, best_score=%.10f, total_time=%.0fs" %
          (len(global_a), global_best, time.monotonic() - t0))
    return [float(x) for x in global_a]
