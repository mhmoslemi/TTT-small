Let me also add a "spectral initialization" that tries to create a sequence whose convolution is as flat as possible from the start.

**Spectral initialization idea:**
We want a*a ≈ constant. In the frequency domain, this means |A(f)|^2 ≈ constant * |W(f)|^2 where W is the DFT of the all-ones vector.

So |A(f)| ≈ c * |W(f)| = c * |sinc(f)|

The inverse DFT of |sinc(f)| is... well, sinc is the DFT of a rectangle, so |sinc| is not the DFT of a simple function. But we can approximate:

1. Compute W = DFT(ones(2n-1)) (the sinc-like spectrum)
2. Set |A(f)| = sqrt(W(f)/max(W)) (normalize)
3. Set phase of A(f) = 0 (for real non-negative a, the phase should be 0 or pi)
4. a = IDFT(A)
5. Clip a to be non-negative
6. Normalize

This gives a rough starting point. But the clipping might destroy the spectral properties. A better approach is to iterate:
1. Start with a = ones(n)
2. Compute B = |DFT(a)|^2
3. Target: B_target = W / max(W) (normalized)
4. Update: DFT(a_new) = DFT(a) * sqrt(B_target / B)
5. a_new = IDFT(DFT(a_new))
6. Clip to non-negative
7. Repeat

This is similar to the Gerchberg-Saxton / error reduction algorithm used in phase retrieval.

Let me include this as an initialization method.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase with spectral init, stabilized FISTA, 
efficient coordinate descent, and multi-scale refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_loss_grad(a, beta):
    """Compute LSE loss and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: d/d a[j] of LSE = 2 * sum_k w[k] * a[k-j]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

def spectral_init(n, rng, iters=50):
    """Initialize using spectral method to flatten the convolution."""
    a = np.ones(n)
    S = np.sqrt(2.0 * n)
    a = a * (S / n)
    
    # Target: flat convolution
    target = np.ones(2*n-1)
    target[:n-1] = np.arange(1, n)  # triangular for uniform
    # Actually, for flat: target = ones
    target = np.ones(2*n-1)
    
    for it in range(iters):
        c = fft_convolve_self(a)
        # In frequency domain: |A|^2 should be proportional to DFT(target)
        fsize = 1 << (2*n).bit_length()
        A = fft(a, fsize)
        C = fft(c, fsize)
        # Target spectrum
        T = fft(target, fsize)
        # Update: A_new = A * sqrt(T / C)
        ratio = np.abs(T) / (np.abs(C) + 1e-30)
        ratio = np.minimum(ratio, 10.0)  # clip to avoid instability
        A_new = A * np.sqrt(ratio)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-15:
            break
        a = a_new * (S / a_new.sum())
    
    return a

def run_fista_stable(a_init, max_time, rng, beta_init=10.0, beta_rate=0.005,
                     beta_max=5000.0, use_backtrack=True):
    """FISTA with proper step size and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.1 / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_loss_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            # Check Armijo condition
            obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
            bt = 0
            while obj_trial > cur_obj + 1e-10 and step_local > 1e-12 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 0.5 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        else:
            y = a.copy()
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj

def run_adam(a_init, max_time, rng, beta_init=5.0, beta_rate=0.003,
             beta_max=5000.0, lr_init=0.01, lr_end=0.0001):
    """Adam optimizer for the LSE-smoothed objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    total_iters = max(1, int(max_time * 50))  # estimate
    
    while time.monotonic() - start < max_time:
        it += 1
        # Cosine annealing
        progress = min(1.0, it / total_iters)
        lr = lr_end + 0.5 * (lr_init - lr_end) * (1 + np.cos(np.pi * progress))
        # Sometimes reset the schedule
        if it % 50000 == 0:
            progress = 0.0
            lr = lr_init * 0.5
        
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_loss_grad(a, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)
        
        # Occasionally add noise to escape local minima
        stuck = it - last_improve
        if stuck > 5000 and it % 100 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            a_new = project_simplex(a_new + rng.normal(0, noise_scale, n), S)
        
        a = a_new
    
    return best_a, best_obj

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def downsample(a, factor=2):
    n = len(a)
    n_new = n // factor
    idx = np.linspace(0, n-1, n_new).astype(int)
    return a[idx]
