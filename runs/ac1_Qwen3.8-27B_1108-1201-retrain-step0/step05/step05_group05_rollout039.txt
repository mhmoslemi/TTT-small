"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Spectral annealing (GS-like) for global exploration
2. L-BFGS-B local optimization with adaptive LSE smoothing
3. Basin hopping with spectral mutations
4. High-beta polishing for final precision

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_objective_and_grad(x, n, fsize, beta):
    """
    Compute LSE objective and gradient for the normalized problem.
    x is the sequence (length n), we compute LSE of its self-convolution.
    Returns (objective, gradient).
    """
    A = fft(x, fsize)
    c = np.real(ifft(A * A))[:2*n-1]
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    obj = cm + np.log(tot) / beta
    
    # Gradient
    w = e / tot
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return obj, grad


def spectral_shape(a, n_iters=100, alpha_init=0.5, alpha_final=0.95, rng=None):
    """
    GS-like spectral shaping: iteratively modify the power spectrum
    to reduce the peak of the autocorrelation, while maintaining
    non-negativity of the time-domain sequence.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    S = a.sum()
    a = a.copy()
    
    for it in range(n_iters):
        # Compute spectrum
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        
        # Compute autocorrelation
        b = np.real(ifft(P))[:2*n-1]
        
        # Modify power spectrum to reduce peak
        # Use alpha-compression: P_new = P^alpha * P_mean^(1-alpha)
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        P_mean = P.mean()
        # Compress: bring high values down, low values up
        P_new = np.power(P / P_mean, alpha) * P_mean
        
        # Keep DC fixed
        P_new[0] = P[0]
        # Maintain symmetry (real sequence)
        P_new = (P_new + P_new[::-1]) / 2  # For even fsize
        # Actually, for real signals, P[k] = P[fsize-k]
        # Let's just enforce it
        for k in range(1, fsize // 2 + 1):
            P_new[k] = P_new[fsize - k] = (P_new[k] + P_new[fsize - k]) / 2
        
        # Reconstruct with new magnitude, same phase
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
    
    return a


def spectral_shape_v2(a, n_iters=200, rng=None):
    """
    Improved spectral shaping: use peak-directed power reduction.
    At each iteration, identify the peak lag(s) and reduce the power
    at frequency bins that constructively interfere there.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    S = a.sum()
    a = a.copy()
    
    for it in range(n_iters):
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        b = np.real(ifft(P))[:2*n-1]
        
        bmax = b.max()
        bmean = b.mean()
        
        if bmax - bmean < 1e-10:
            break
        
        # Find top peak lags
        n_peaks = max(1, min(8, int(0.01 * len(b))))
        peak_idx = np.argsort(b)[::-1][:n_peaks]
        
        # For each peak, the contribution from bin k is P[k] * exp(2*pi*i*m*k/fsize)
        # We want to reduce P[k] where the phase aligns with the peak
        # Simplification: reduce P proportional to its "coherence" with the peak pattern
        peak_weight = np.zeros(fsize)
        for m in peak_idx:
            phase = np.exp(2j * np.pi * m * np.arange(fsize) / fsize)
            peak_weight += np.real(phase)
        
        # Normalize
        pw_max = np.abs(peak_weight).max()
        if pw_max > 1e-15:
            peak_weight = peak_weight / pw_max
        
        # Reduce power where it constructively interferes with peaks
        reduction = np.maximum(peak_weight, 0) * 0.1
        P_new = P * (1 - reduction)
        
        # Also do mild compression
        P_mean = P_new.mean()
        P_new = 0.9 * P_new + 0.1 * P_mean
        
        # Keep DC
        P_new[0] = P[0]
        # Enforce Hermitian symmetry
        P_new = np.maximum(P_new, 0)
        for k in range(1, fsize // 2 + 1):
            P_new[k] = P_new[fsize - k] = (P_new[k] + P_new[fsize - k]) / 2
        
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
    
    return a


def lbfogs_optimize(a_init, S, beta, max_time, rng):
    """
    Optimize using scipy L-BFGS-B on the LSE-smoothed objective.
    The variable is 'a' (the sequence), and we minimize LSE_beta(conv(a,a))
    subject to a >= 0 (handled by bounds) and sum(a) = S (handled by 
    normalizing after each call, or we can use the fact that the objective
    is scale-invariant).
    
    Actually, since the objective is scale-invariant, we can fix sum(a) = S
    by parameterizing a = S * x / sum(x) where x >= 0. But this is awkward
    for L-BFGS. Instead, let's just minimize the LSE of conv(a,a) / (sum(a))^2
    which is scale-invariant, and let L-BFGS handle the scaling.
    
    Wait, the objective 2*n*max(b)/(sum(a))^2 is scale-invariant. So we can
    just minimize LSE_beta(b) / (sum(a))^2 * 2*n. But the gradient is more
    complex then.
    
    Simpler approach: fix sum(a) = S by normalizing, and minimize LSE_beta(b).
    We'll use a penalty method: minimize LSE_beta(b) + mu * (sum(a) - S)^2.
    Or even simpler: since we normalize anyway, just minimize LSE_beta(b)
    and the scale doesn't matter.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    
    a0 = normalize_to_sum(a_init, S)
    
    def func_and_grad(x):
        # Normalize to maintain sum = S (soft constraint)
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        x_norm = x * (S / s)
        
        A = fft(x_norm, fsize)
        c = np.real(ifft(A * A))[:2*n-1]
        cm = c.max()
        scaled = np.clip(beta * (c - cm), -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        obj = cm + np.log(tot) / beta
        
        # Gradient of LSE w.r.t. x_norm
        w = e / tot
        W = fft(w, fsize)
        grad_norm = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Chain rule for normalization: x_norm = x * S / sum(x)
        # d(x_norm)_j / d(x)_i = S/s * (delta_ji - x_j/s)
        # grad_x = (S/s) * (grad_norm - mean(grad_norm * x / s * S) ... )
        # Actually this gets complicated. Let's just use the gradient of x_norm
        # and note that the normalization is approximately identity near the optimum.
        # For simplicity, just return grad_norm (approximate).
        return obj, grad_norm
    
    # Use L-BFGS-B
    best_result = None
    best_val = func_and_grad(a0)[0]
    
    while time.monotonic() - start < max_time:
        result = minimize(
            func_and_grad, a0,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0, 1000)] * n,
            options={'maxiter': 500, 'maxfun': 1000, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        a_opt = normalize_to_sum(result.x, S)
        val = func_and_grad(a_opt)[0]
        
        if val < best_val:
            best_val = val
            best_result = a_opt.copy()
        
        if not result.success and 'maximum number of iterations' in str(result.message):
            # Try again from the current best
            a0 = best_result.copy()
        else:
            break
        
        if time.monotonic() - start > max_time - 5:
            break
    
    if best_result is None:
        best_result = a0
    
    return best_result


def lbfogs_optimize_direct(a_init, S, beta, max_time, rng):
    """
    Cleaner L-BFGS-B optimization. We parameterize the problem as:
    minimize LSE_beta(conv(a, a)) subject to a >= 0, sum(a) = S.
    
    To handle the equality constraint, we use the trick:
    a = S * softmax(z) / (n * softmax(z).mean())  -- no, this is over-parameterized.
    
    Better: just use the bounds (a >= 0) and add a penalty for sum constraint.
    Or: parameterize a_j = S * exp(z_j) / sum(exp(z_i)) (softmax parameterization).
    This automatically satisfies a >= 0 and sum(a) = S.
    
    The gradient w.r.t. z is:
    dz/dz_j = (delta_jk - a_k/S) * a_j / S  (Jacobian of softmax)
    
    So grad_z = J^T * grad_a where J is the Jacobian of the softmax.
    
    For large n, this is O(n^2) which is too slow. Let's use a different approach.
    
    Actually, for L-BFGS-B, we can just work with the constraint a >= 0 (bounds)
    and not enforce sum(a) = S exactly (the objective is scale-invariant anyway,
    so the optimum will be at some scale, and we normalize at the end).
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    
    # Since objective is scale-invariant, just minimize LSE(conv(a,a))
    # with a >= 0. The scale will be determined by the optimizer.
    # We'll normalize at the end.
    
    a0 = np.maximum(a_init, 0.01)  # Small floor to avoid zero
    
    def func_and_grad(x):
        A = fft(x, fsize)
        c = np.real(ifft(A * A))[:2*n-1]
        cm = c.max()
        scaled = np.clip(beta * (c - cm), -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        obj = cm + np.log(tot) / beta
        
        w = e / tot
        W = fft(w, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        return obj, grad
    
    best_a = a0.copy()
    best_obj = func_and_grad(a0)[0]
    
    while time.monotonic() - start < max_time:
        result = minimize(
            func_and_grad, a0,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxls': 20, 'max_correction': 10}
        )
        
        a_new = np.maximum(result.x, 0.0)
        S_new = a_new.sum()
        if S_new > 0.01:
            a_new = a_new * (S / S_new)
        
        obj_new = func_and_grad(a_new)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        if result.success:
            break
        a0 = best_a.copy()
        
        if time.monotonic() - start > max_time - 10:
            break
    
    return best_a


def run_fourier_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping with Fourier-domain mutations and L-BFGS local optimization.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    
    # Evaluate initial
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    opt_time = max(15.0, max_time / 20.0)
    
    # Population for diversity
    population = [(best, a.copy())]
    
    it = 0
    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = it / max(1, max_time / opt_time)
        
        # Choose mutation
        r = rng.random()
        if len(population) >= 2 and r < 0.25:
            # Crossover
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a = fourier_crossover(population[i1][1], population[i2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, scale=0.1)
        elif r < 0.5:
            # Phase mutation
            a = fourier_phase_mutate(a, rng, scale=0.15 * np.exp(-progress))
        elif r < 0.7:
            # Spectral shaping mutation
            a = spectral_shape(a, n_iters=50, alpha_init=0.3, alpha_final=0.7, rng=rng)
        else:
            # Random perturbation
            noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
        
        # Local optimization
        t_alloc = min(opt_time, max_time - opt_time - (time.monotonic() - start))
        if t_alloc < 5:
            break
        
        beta = 20 + 50 * min(it, 30)
        a_opt = lbfogs_optimize_direct(a, S, beta, t_alloc, rng)
        
        c_opt = fft_convolve_self(a_opt)
        val = c_opt.max()
        
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))
        
        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]
    
    return best_a, best


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def generate_inits(n, rng, fsize, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    S = np.sqrt(2.0 * n)
    
    if given_seq is not None:
        inits.append(normalize_to_sum(given_seq, S))
    
    # Spectrally-flat initializations
    # A sequence whose DFT has flat magnitude (like a chirp)
    for freq_factor in [1.0, 2.0, 3.0, 5.0, 7.0]:
        # Chirp: a_j = exp(i*pi*f*j^2/n) -> real part
        phase = np.pi * freq_factor * (np.arange(n) ** 2) / n
        a = np.abs(np.cos(phase)) + 0.01
        inits.append(normalize_to_sum(a, S))
    
    # Window functions
    inits.append(normalize_to_sum(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01), S))
    inits.append(normalize_to_sum(np.sin(np.pi * tt) ** 2 + 0.01, S))
    inits.append(normalize_to_sum(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25) + 0.01, S))
    
    # Random
    inits.append(normalize_to_sum(rng.random(n) + 0.1, S))
    inits.append(normalize_to_sum(np.abs(rng.normal(0.8, 0.4, n)) + 0.05, S))
    
    # Spectral flatness: start with flat spectrum, inverse FFT, clip
    for trial in range(3):
        phase = rng.uniform(0, 2*np.pi, fsize)
        # Enforce Hermitian symmetry
        phase[0] = 0
        for k in range(1, fsize // 2 + 1):
            phase[fsize - k] = -phase[k]
        A = np.sqrt(np.ones(fsize)) * np.exp(1j * phase)
        A[0] = S  # DC = sum
        a = np.real(ifft(A, fsize))[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 0.01:
            inits.append(a * (S / a.sum()))
    
    # Block structures
    for nb in [4, 8, 16]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(normalize_to_sum(block[:n], S))
    
    return inits


def objective_from_max_b(a):
    """Compute objective from the max of convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Spectral annealing (GS-like iterative amplitude compression) for global exploration
2. L-BFGS-B local optimization with adaptive LSE smoothing (scale-invariant formulation)
3. Basin hopping with hybrid spectral/gradient mutations
4. High-beta polishing with antithetic averaging for final precision

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection-free method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_and_grad(a, beta, fsize):
    """Compute LSE of self-convolution and its gradient."""
    n = len(a)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:2 * n - 1]
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def lse_ratio_func_and_grad(x, beta, fsize):
    """
    Scale-invariant objective: LSE_beta(a*a) / (sum(a))^2
    Returns (value, gradient).
    """
    n = len(x)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    lse_val, grad = compute_lse_and_grad(x, beta, fsize)
    obj = lse_val / (S * S)
    # grad of ratio: grad/S^2 - 2*LSE/S^3 * ones
    np_ones = np.ones(n)
    grad_obj = grad / (S * S) - 2.0 * lse_val / (S * S * S) * np_ones
    return obj, grad_obj


def lse_ratio_func_and_grad_v2(x, beta, fsize, S_target):
    """
    Alternative: normalize x to sum S_target, then minimize LSE.
    This avoids the ratio gradient issues near zero.
    """
    n = len(x)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    x_norm = x * (S_target / S)
    lse_val, grad_norm = compute_lse_and_grad(x_norm, beta, fsize)
    # Chain rule: d(x_norm)/dx = S_target/S * (I - x_norm*x^T/S)
    # grad_x = (S_target/S) * (grad_norm - dot(grad_norm, x_norm)/S * x)
    dot = np.dot(grad_norm, x_norm)
    grad_x = (S_target / S) * (grad_norm - (dot / S) * x)
    return lse_val, grad_x


def spectral_anneal(a, n_iters, fsize, S, rng, compression_start=0.3, compression_end=0.85):
    """
    GS-like spectral annealing: iteratively compress the power spectrum
    to flatten the autocorrelation, then project back to non-negative sequences.
    This is the key novel global exploration method.
    """
    n = len(a)
    a = a.copy()
    best_max_b = np.inf
    best_a = a.copy()

    for it in range(n_iters):
        A = fft(a, fsize)
        P = np.abs(A) ** 2

        # Adaptive compression exponent
        progress = it / max(1, n_iters - 1)
        alpha = compression_start + (compression_end - compression_start) * progress

        # Compress dynamic range of power spectrum
        P_max = P.max()
        P_min = P[P > 0].min() if np.any(P > 0) else 1e-10
        log_P = np.log(np.maximum(P, P_min))
        log_P_max = log_P.max()
        log_P_min = log_P.min()
        log_range = log_P_max - log_P_min
        if log_range < 1e-10:
            break

        # Soft compression in log domain
        log_P_compressed = log_P_min + (log_P - log_P_min) * alpha
        P_new = np.exp(log_P_compressed)

        # Preserve DC component (sum constraint)
        P_new[0] = P[0]
        P_new[fsize // 2] = P[fsize // 2] if fsize % 2 == 0 else P_new[fsize // 2]

        # Enforce Hermitian symmetry for real output
        for k in range(1, fsize // 2):
            P_new[k] = P_new[fsize - k] = 0.5 * (P_new[k] + P_new[fsize - k])

        # Reconstruct
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new * (S / s)

        # Track best
        b = fft_convolve_self(a)
        bmax = b.max()
        if bmax < best_max_b:
            best_max_b = bmax
            best_a = a.copy()

    return best_a


def spectral_peak_shave(a, n_iters, fsize, S, rng):
    """
    Peak-directed spectral modification: identify peak lags in autocorrelation
    and reduce spectral power at bins that constructively interfere there.
    """
    n = len(a)
    a = a.copy()

    for it in range(n_iters):
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        b = np.real(ifft(P))[:2 * n - 1]

        bmax = b.max()
        bmean = b.mean()
        if bmax - bmean < 1e-12 * bmax:
            break

        # Find peak lags
        n_peaks = max(1, min(4, len(b) // 100))
        peak_idx = np.argsort(b)[::-1][:n_peaks]

        # Compute "peak interference" weight for each frequency bin
        # Bin k contributes P[k]*cos(2*pi*m*k/fsize) to b[m]
        interference = np.zeros(fsize)
        for m in peak_idx:
            interference += np.cos(2.0 * np.pi * m * np.arange(fsize) / fsize)
        interference = np.maximum(interference, 0)
        if interference.max() > 0:
            interference /= interference.max()

        # Reduce power at interfering bins
        reduction_rate = 0.05 + 0.1 * (it / max(1, n_iters))
        P_new = P * (1.0 - reduction_rate * interference)

        # Mild overall compression to maintain progress
        P_mean = P_new[P_new > 0].mean() if np.any(P_new > 0) else 1.0
        P_new = 0.95 * P_new + 0.05 * P_mean

        P_new[0] = P[0]  # Preserve DC
        for k in range(1, fsize // 2):
            P_new[k] = P_new[fsize - k] = 0.5 * (P_new[k] + P_new[fsize - k])

        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new * (S / s)

    return a

        mi = max(20, min(maxiter_per_call, int(remaining * 300)))

        result = minimize(

            args=(beta, fsize),

            jac=True,

def fista_local_optimize(a_init, S, max_time, fsize, rng, beta_init=20.0, beta_max=5000.0):
    """
    FISTA with LSE smoothing as fallback local optimizer.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.002 ** it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = c.max()
        scaled = np.clip(beta * (c - cm), -500, 500)
        e = np.exp(scaled)
        w = e / e.sum()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (3.0 * beta)

        if it - last_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a


def fourier_phase_mutate(a, rng, scale=0.1, fsize=None):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def spectral_mutation(a, rng, fsize, S, mutation_type=0):
    """Apply a spectral-domain mutation."""
    n = len(a)
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    phase = np.angle(A)

    if mutation_type == 0:
        # Power compression
        alpha = 0.5 + 0.3 * rng.random()
        P_mean = P[P > 0].mean() if np.any(P > 0) else 1.0
        P_new = np.power(np.maximum(P, 1e-10) / P_mean, alpha) * P_mean
    elif mutation_type == 1:
        # Random power redistribution
        noise = rng.normal(0, 0.2, fsize)
        P_new = P * np.exp(noise)
    else:
        # Antithetic: flip phase of random subset
        mask = rng.random(fsize) < 0.3
        phase_new = phase.copy()
        phase_new[mask] += np.pi
        P_new = P
        return np.maximum(np.real(ifft(np.sqrt(P_new) * np.exp(1j * phase_new), fsize))[:n], 0) * (S / max(np.real(ifft(np.sqrt(P_new) * np.exp(1j * phase_new), fsize))[:n].sum(), 1e-10))

    P_new[0] = P[0]
    for k in range(1, fsize // 2):
        P_new[k] = P_new[fsize - k] = 0.5 * (P_new[k] + P_new[fsize - k])

    A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
    return a_new * (S / s)


def antithetic_improve(a, S, fsize):
    """
    Apply antithetic averaging: average a with its reversal.
    By symmetry, this can reduce asymmetry-related peaks.
    """
    a_rev = a[::-1].copy()
    a_avg = 0.5 * (a + a_rev)
    s = a_avg.sum()
    if s > 0.01:
        a_avg = a_avg * (S / s)
    # Keep the better of original and antithetic average
    b1 = fft_convolve_self(a).max()
    b2 = fft_convolve_self(a_avg).max()
    if b2 < b1:
        return a_avg
    return a


def generate_inits(n, fsize, S, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(normalize_to_sum(given_seq, S))

    # Windows
    inits.append(normalize_to_sum(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01), S))
    inits.append(normalize_to_sum(np.sin(np.pi * tt) ** 2 + 0.01, S))
    inits.append(normalize_to_sum(np.sin(np.pi * tt) ** 4 + 0.01, S))
    inits.append(normalize_to_sum(np.exp(-6.0 * (tt - 0.5) ** 2 / 0.3) + 0.01, S))

    # Chirp-based (good spectral properties)
    for freq in [1.3, 2.7, 5.1]:
        phase = np.pi * freq * (np.arange(n) ** 2) / n
        a = np.abs(np.cos(phase)) + 0.01
        inits.append(normalize_to_sum(a, S))

    # Random
    inits.append(normalize_to_sum(rng.random(n) + 0.1, S))
    inits.append(normalize_to_sum(np.abs(rng.normal(0.8, 0.4, n)) + 0.05, S))

    # Flat-spectrum initializations (random phase, flat magnitude)
    for trial in range(4):
        phase = rng.uniform(0, 2 * np.pi, fsize)
        phase[0] = 0
        for k in range(1, fsize // 2 + 1):
            if k < fsize - k:
                phase[fsize - k] = -phase[k]
            elif k == fsize - k:
                phase[k] = 0.0
        mag = np.ones(fsize) * (S / np.sqrt(fsize))
        mag[0] = S
        A = mag * np.exp(1j * phase)
        a = np.real(ifft(A, fsize))[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 0.01:
            inits.append(a * (S / a.sum()))

    # Block structures
    for nb in [4, 8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        heights = 0.5 * (heights + heights[::-1])  # Make symmetric
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(normalize_to_sum(block[:n], S))

    return inits


def evaluate_obj(a, n):
    """Compute the objective value."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse init + spectral annealing + L-BFGS screening (~250s)
    2. Deeper L-BFGS with increasing beta (~350s)
    3. Basin hopping with spectral mutations (~900s)
    4. High-beta FISTA + antithetic polish (remaining)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 40

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Load given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    n = 8192
    fsize = 1 << (2 * n - 1).bit_length()  # 16384
    S = np.sqrt(2.0 * n)

    global_best = np.inf
    global_a = None

    # Phase 1: Diverse initialization with spectral annealing
    print("\n=== Phase 1: Spectral init + screening (~250s) ===")
    p1_start = time.monotonic()
    p1_budget = 250.0

    inits = generate_inits(n, fsize, S, rng, given_seq=given)
    n_screen = min(10, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.70:
            print("  Time limit for phase 1")
            break
        t_alloc = min(per_init_time, deadline - time.monotonic() - 800)
        if t_alloc < 15:
            break

        # Spectral annealing for global exploration
        t_shape = min(t_alloc * 0.4, 40.0)
        n_shape_iters = max(30, int(t_shape * 50))
        a_shaped = spectral_anneal(inits[i].copy(), n_shape_iters, fsize, S, rng,
                                     compression_start=0.3, compression_end=0.8)

        # Peak shaving
        t_shave = min(t_alloc * 0.15, 15.0)
        n_shave_iters = max(10, int(t_shave * 30))
        a_shaved = spectral_peak_shave(a_shaped, n_shave_iters, fsize, S, rng)

        # L-BFGS local optimization
        t_opt = t_alloc - t_shape - t_shave - 5.0
        if t_opt > 10:
            a_opt = lbfgs_local_optimize(a_shaved, S, beta=15.0,
                                          max_time=t_opt, fsize=fsize, rng=rng)
        else:
            a_opt = a_shaved

        # Antithetic improvement
        a_opt = antithetic_improve(a_opt, S, fsize)

        obj = evaluate_obj(a_opt, n)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()

    candidates.sort(key=lambda x: x[0])
    print("  Best after Phase 1: obj=%.6f" % global_best)

    # Phase 2: Deeper optimization with increasing beta
    print("\n=== Phase 2: Deep L-BFGS (~350s) ===")
    p2_start = time.monotonic()
    p2_budget = 350.0

    n_deep = min(4, len(candidates))
    for i in range(n_deep):
        if time.monotonic() > deadline - 600:
            break
        t_alloc = min(p2_budget / (n_deep - i), deadline - time.monotonic() - 600)
        if t_alloc < 30:
            break

        a_current = candidates[i][1] if i < len(candidates) else global_a
        if a_current is None:
            break

        # Progressive beta increase
        betas = [30, 80, 200, 500]
        per_beta_time = t_alloc / len(betas)
        for beta in betas:
            if time.monotonic() > deadline - 600:
                break
            t_seg = min(per_beta_time, deadline - time.monotonic() - 600)
            if t_seg < 10:
                break
            a_opt = lbfgs_local_optimize(a_current, S, beta=beta,
                                          max_time=t_seg, fsize=fsize, rng=rng)
            a_opt = antithetic_improve(a_opt, S, fsize)
            obj = evaluate_obj(a_opt, n)
            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  Phase 2, cand %d, beta=%d: obj=%.6f" % (i, beta, global_best))
            a_current = a_opt

    # Phase 3: Basin hopping with spectral mutations
    print("\n=== Phase 3: Basin hopping (~900s) ===")
    p3_start = time.monotonic()
    p3_budget = min(900.0, (deadline - p3_start) * 0.65)

    if global_a is not None and p3_budget > 60:
        a_cur = global_a.copy()
        start_bh = time.monotonic()
        population = [(global_best, a_cur.copy())]
        bh_it = 0

        while time.monotonic() - start_bh < p3_budget:
            bh_it += 1
            progress = (time.monotonic() - start_bh) / p3_budget
            t_opt = max(15.0, min(60.0, p3_budget * 0.04))

            if time.monotonic() - start_bh > p3_budget - t_opt - 30:
                break

            # Choose mutation strategy
            r = rng.random()
            if len(population) >= 2 and r < 0.2:
                i1 = rng.integers(len(population))
                i2 = rng.integers(len(population))
                if i1 != i2:
                    a_cur = fourier_crossover(population[i1][1], population[i2][1], rng, fsize)
                else:
                    a_cur = fourier_phase_mutate(a_cur, rng, scale=0.1, fsize=fsize)
            elif r < 0.45:
                # Spectral mutation (novel)
                mtype = rng.integers(3)
                a_cur = spectral_mutation(a_cur, rng, fsize, S, mutation_type=mtype)
            elif r < 0.65:
                # Phase mutation
                scale = 0.2 * np.exp(-2.0 * progress) + 0.03
                a_cur = fourier_phase_mutate(a_cur, rng, scale=scale, fsize=fsize)
            elif r < 0.8:
                # Spectral peak shaving (short burst)
                a_cur = spectral_peak_shave(a_cur, 20, fsize, S, rng)
            else:
                # Random perturbation
                noise = rng.normal(0, 0.03 * S / np.sqrt(n), n)
                a_cur = project_simplex(a_cur + noise, S)

            # Local optimization
            t_opt_actual = min(t_opt, deadline - time.monotonic() - 120)
            if t_opt_actual < 5:
                break

            beta = 30 + 100 * min(bh_it, 20)
            a_opt = lbfgs_local_optimize(a_cur, S, beta=beta,
                                          max_time=t_opt_actual, fsize=fsize, rng=rng)
            a_opt = antithetic_improve(a_opt, S, fsize)
            obj = evaluate_obj(a_opt, n)

            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  BH iter %d: new best obj=%.6f" % (bh_it, global_best))

            population.append((obj, a_opt.copy()))
            if len(population) > 6:
                # Keep top 6
                population = sorted(population, key=lambda x: x[0])[:6]

        print("  After basin hopping: obj=%.6f" % global_best)

    # Phase 4: Final polish
    print("\n=== Phase 4: Final polish ===")
    p4_start = time.monotonic()
    p4_budget = max(0.0, deadline - p4_start - 15)

    if global_a is not None and p4_budget > 20:
        # High-beta L-BFGS
        seg1 = p4_budget * 0.5
        a_hb = lbfgs_local_optimize(global_a, S, beta=500,
                                     max_time=seg1, fsize=fsize, rng=rng)
        a_hb = antithetic_improve(a_hb, S, fsize)
        obj = evaluate_obj(a_hb, n)
        if obj < global_best:
            global_best = obj
            global_a = a_hb.copy()
        print("  After high-beta L-BFGS: obj=%.6f" % global_best)

        # Very high-beta FISTA
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_fista = fista_local_optimize(global_a, S, remaining, fsize, rng,
                                            beta_init=500.0, beta_max=20000.0)
            a_fista = antithetic_improve(a_fista, S, fsize)
            obj = evaluate_obj(a_fista, n)
            if obj < global_best:
                global_best = obj
                global_a = a_fista.copy()
            print("  After high-beta FISTA: obj=%.6f" % global_best)

        # Final antithetic + normalization
        global_a = antithetic_improve(global_a, S, fsize)
        final_obj = evaluate_obj(global_a, n)
        if final_obj < global_best:
            global_best = final_obj

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
