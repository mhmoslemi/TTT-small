```python
def build_constraint_matrix(a, selected_lags):

    K = len(selected_lags)

    vals = []
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LP-guided direction finding combined with FISTA and L-BFGS polishing.
The LP approach solves: max sum(g) s.t. conv(a,g) <= M, g >= 0
to find a direction that simultaneously addresses all active constraints.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def solve_lp_direction(a, M, selected_lags, time_limit=5.0):
    """
    Solve the LP: max sum(g) s.t. (A_sel @ g) <= M, g >= 0.
    Returns the optimal g vector.
    """
    n = len(a)
    K = len(selected_lags)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for ki, k in enumerate(selected_lags):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(ki)
            cols.append(j)
            vals.append(a[k - j])
    
    C = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    # Solve LP using HiGHS
    c_obj = -np.ones(n)  # minimize -sum(g)
    try:
        result = linprog(c_obj, A_ub=C, b_ub=np.full(K, M),
                        bounds=[(0, None)] * n, method='highs',
                        options={'presolve': True, 'time_limit': time_limit})
        if result.status == 0:
            return result.x
        else:
            return None
    except Exception as e:
        print(f"    LP solve failed: {e}")
        return None


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def run_lp_fista_hybrid(a_init, max_time, rng, n_lags=100, fista_iters_per_lp=50,
                        beta_start=20.0):
    """
    Hybrid LP + FISTA optimization.
    Alternates between LP direction steps and FISTA gradient steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        # === LP Step ===
        if it % (fista_iters_per_lp + 1) == 0:
            c = fft_convolve_self(a)
            M = c.max()
            
            # Select top lags
            k_sel = min(n_lags, 2 * n - 1)
            top_idx = np.argsort(c)[::-1][:k_sel]
            
            g = solve_lp_direction(a, M, top_idx, time_limit=3.0)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g_norm = g * (S / g_sum)
                    # Line search for step size
                    for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
                        a_trial = (1 - t) * a + t * g_norm
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial * (S / max(a_trial.sum(), 1e-15))
                        obj_trial = compute_obj(a_trial)
                        if obj_trial < best_obj:
                            a = a_trial
                            best_obj = obj_trial
                            best_a = a.copy()
                            print(f"    LP step (t={t}): obj={best_obj:.8f}")
                            break
        
        # === FISTA Step ===
        c = fft_convolve_self(a)
        w = lse_weights(c, beta_start * (1.0 + 0.01 * it))
        grad = conv_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            step = S / (gnorm * (1.0 + 0.005 * it))
            a_new = project_simplex(a - step * grad, S)
            a = a_new
        
        it += 1
        if it % 200 == 0:
            obj_now = compute_obj(a)
            if obj_now < best_obj:
                best_obj = obj_now
                best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"  Iter {it}: obj={best_obj:.8f} (current={obj_now:.8f}, t={elapsed:.0f}s)")
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_start=10.0):
    """Standard FISTA for polishing."""
    n = len(a_init)
```

```python
import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def build_sparse_conv_matrix(a, selected_lags):
    """Build sparse matrix C where C[k,i] = a[k-i] for valid indices."""
    n = len(a)
    K = len(selected_lags)
    row_list = []
    col_list = []
    val_list = []
    for ki, k in enumerate(selected_lags):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        row_list.append(np.full(len(js), ki, dtype=np.int32))
        col_list.append(js.astype(np.int32))
        val_list.append(a[k - js])
    if not row_list:
        return csr_matrix((K, n))
    rows = np.concatenate(row_list)
    cols = np.concatenate(col_list)
    vals = np.concatenate(val_list)
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def solve_lp_direction(a, M, selected_lags, time_limit=5.0):
    """
    Solve reduced LP: max sum(g) s.t. (A_sel @ g) <= M, g >= 0.
    Returns optimal g or None on failure.
    """
    n = len(a)
    K = len(selected_lags)
    if K == 0 or K > 2 * n - 1:
        return None

    C = build_sparse_conv_matrix(a, selected_lags)
    c_obj = -np.ones(n)

    try:
        result = linprog(c_obj, A_ub=C, b_ub=np.full(K, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': time_limit,
                                  'dual_feasibility_tolerance': 1e-8,
                                  'primal_feasibility_tolerance': 1e-8})
        if result.status == 0:
            g = result.x
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        if stuck > 2000 and it % 500 == 0:
            scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t_mom = 1.0

        if it % 2000 == 0:
            print(f"  [Active] it={it}, best={best_obj:.8f}, stuck={stuck}")

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with increasing beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 80.0, 200.0, 500.0, 1200.0]
    pen = [5.0, 10.0, 20.0, 50.0, 100.0]

    for beta, p in zip(betas, pen):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 15:
            break

        def obj_and_grad(x, beta=beta, p=p, S=S, n=n):
            x_clip = np.maximum(x, 0.0)
            b = fft_convolve_self(x_clip)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            fval = cm_local + np.log(e.sum()) / beta
            fval += p * (x_clip.sum() - S) ** 2
            grad = conv_gradient(x_clip, w)
            grad = grad + 2.0 * p * (x_clip.sum() - S)
            return fval, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B', jac=True,
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 300, 'maxls': 50, 'ftol': 1e-16,
                     'gtol': 1e-12, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        print(f"    L-BFGS beta={beta:.0f}: obj={obj_new:.8f} (t={elapsed:.1f}s, it={result.nit})")

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def generate_structured_init(n, rng, S, family="cosine"):
    """Generate structured initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    if family == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif family == "rectified_cosine":
        a = np.maximum(np.cos(2 * np.pi * t), 0.0) + 0.05
    elif family == "double_cosine":
        a = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 0.01)
    elif family == "kaiser":
        beta_k = 5.0
        x = 2.0 * t - 1.0
        a = np.ones(n)
        from scipy.special import i0
        a = i0(np.pi * beta_k * np.sqrt(1.0 - x**2)) / i0(np.pi * beta_k)
    elif family == "gaussian_center":
        a = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n / 8)) ** 2)
    elif family == "smoothed_random":
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(3, n // 80)
        F[cutoff:-cutoff] *= 0.05
        a = np.real(ifft(F))[:n]
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Strategy: LP-guided direction + FISTA + L-BFGS, multi-init, multi-size.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    n0 = 8192
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        S0 = np.sqrt(2.0 * n0)
        a_start = a_start * (S0 / a_start.sum())
        obj0 = compute_obj(a_start)
        print(f"Starting from height_sequence_1: n={n0}, obj={obj0:.8f}")
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        S0 = np.sqrt(2.0 * n0)
        a_start = generate_structured_init(n0, rng, S0, "cosine")
        obj0 = compute_obj(a_start)
        print(f"Generated init: n={n0}, obj={obj0:.8f}")
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Quick FISTA warmup (8% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.08
    print(f"\n=== Phase 1: FISTA warmup (budget {p1:.0f}s) ===")
    a1, obj1 = run_fista_accel(a_start, p1, rng, beta_start=10.0)
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()
        print(f"  After warmup: obj={obj1:.8f}")

    # === Phase 2: LP-guided optimization (45% time) ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.45
    print(f"\n=== Phase 2: LP+FISTA hybrid (budget {p2:.0f}s, n={n0}) ===")

    # Start with fewer LP constraints, increase over time
    a2, obj2 = run_lp_fista_hybrid(global_best_a, p2, rng,
                                    n_lags=60, lp_every=60, beta_start=15.0)
    if obj2 < global_best_obj:
        global_best_obj = obj2
        global_best_a = a2.copy()
        print(f"  After LP+FISTA: obj={obj2:.8f}")

    # Second LP pass with more constraints
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2b = min(remaining * 0.3, 300.0)
        print(f"\n=== Phase 2b: LP+FISTA more constraints (budget {p2b:.0f}s) ===")
        a2b, obj2b = run_lp_fista_hybrid(global_best_a, p2b, rng,
                                         n_lags=150, lp_every=40, beta_start=25.0)
        if obj2b < global_best_obj:
            global_best_obj = obj2b
            global_best_a = a2b.copy()
            print(f"  After LP+FISTA v2: obj={obj2b:.8f}")

    # === Phase 3: FISTA refinement (20% time) ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.20
    print(f"\n=== Phase 3: FISTA refinement (budget {p3:.0f}s) ===")
    a3, obj3 = run_fista_accel(global_best_a, p3, rng, beta_start=30.0)
    if obj3 < global_best_obj:
        global_best_obj = obj3
        global_best_a = a3.copy()
        print(f"  After FISTA: obj={obj3:.8f}")

    # === Phase 4: L-BFGS polishing (15% time) ===
    remaining = deadline - time.monotonic()
    p4 = remaining * 0.15
    print(f"\n=== Phase 4: L-BFGS polish (budget {p4:.0f}s) ===")
    a4, obj4 = run_lbfgs_polish(global_best_a, p4, rng)
    if obj4 < global_best_obj:
        global_best_obj = obj4
        global_best_a = a4.copy()
        print(f"  After L-BFGS: obj={obj4:.8f}")

    # === Phase 5: Active-set final polish (8% time) ===
    remaining = deadline - time.monotonic()
    p5 = remaining * 0.08
    print(f"\n=== Phase 5: Active-set polish (budget {p5:.0f}s) ===")
    a5, obj5 = run_active_set_polish(global_best_a, p5, rng)
    if obj5 < global_best_obj:
        global_best_obj = obj5
        global_best_a = a5.copy()
        print(f"  After active-set: obj={obj5:.8f}")

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Final L-BFGS squeeze (budget {remaining:.0f}s) ===")
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.8, rng)
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()
            print(f"  Final squeeze: obj={obj6:.8f}")

    # Verify and return
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"\nFinal verified: {ver:.8f} (n={len(final_list)})")
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
    # Fallback
    n_fb = 8192
    S_fb = np.sqrt(2.0 * n_fb)
    a_fb = np.ones(n_fb) * (S_fb / n_fb)
    final_list = [float(x) for x in a_fb]
    print(f"Fallback obj: {evaluate_sequence(final_list):.8f}")
    return final_list
```
