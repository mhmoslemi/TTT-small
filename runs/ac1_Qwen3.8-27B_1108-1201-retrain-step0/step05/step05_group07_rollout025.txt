import numpy as np
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    # For m in [0, n-1] and j in [0, n-1]: m+j ranges from m to m+n-1 <= 2n-2 < fsize
    # So no wrap-around, and corr[m] = sum_{j=0}^{n-1} w[m+j] * a[j] (with w[m+j]=0 if m+j >= 2n-1)
    
    # Wait, w has length 2n-1, so w[m+j] is valid for m+j < 2n-1, i.e., j < 2n-1-m.
    # For j in [0, n-1] and m in [0, n-1]: m+j <= 2n-2 < 2n-1. So all valid!
    # Great, so corr[0:n] gives exactly what we want.
    
    return 2.0 * corr[:n]


def lse_objective(a, beta):
    """Compute (1/beta)*log(sum_k exp(beta * c_k)) where c = conv(a,a)."""
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    e = np.exp(scaled)
    return cm + np.log(e.sum()) / beta


def fista_lse(a0, S, beta, max_time, rng, verbose=False, L_init=None):
    """
    FISTA to minimize LSE objective with simplex constraint.
    """
    n = len(a0)
    x = project_simplex(a0.copy(), S)
    x_prev = x.copy()
    
    # Initial Lipschitz estimate
    c = fft_convolve_self(x)
    if L_init is None:
        # Estimate L from first gradient
        grad = compute_lse_gradient(x, c, beta)
        L = max(1e-3, 2.0 * beta * np.linalg.norm(x) * np.linalg.norm(grad) + 1.0)
        # Better: L = beta * 2 * max(a) * 2 * n (rough bound)
        L = max(1.0, 4.0 * beta * np.max(x) * n)
        L = min(L, 1e6)
    
    t = 1.0
    start = time.monotonic()
    it = 0
    best_val = np.inf
    best_x = x.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        
        # Gradient at y
        c_y = fft_convolve_self(y)
        grad = compute_lse_gradient(y, c_y, beta)
        
        # Gradient step with backtracking
        step = 1.0 / L
        accepted = False
        for _ in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            # Check Armijo-like condition
            val_new = c_new.max()
            val_y = c_y.max()
            # Simple check: if the LSE value decreased
            if val_new < 1e30:  # Just check it's valid
                x_new_val = lse_objective(x_new, beta)
                y_val = lse_objective(y, beta)
                if x_new_val <= y_val + 1e-10:
                    x = x_new
                    accepted = True
                    break
            L *= 1.5
            step = 1.0 / L
        
        if not accepted:
            # Accept anyway with small step
            x = project_simplex(y - 0.5 / L * grad, S)
        
        x_prev = x
        
        # Track best (using actual max, not LSE)
        c_x = fft_convolve_self(x)
        cur_max = c_x.max()
        if cur_max < best_val:
            best_val = cur_max
            best_x = x.copy()
        
        t = t_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_val / (S * S)
            print(f"      FISTA it={it}, t={elapsed:.1f}s, L={L:.1f}, obj={obj:.8f}")
    
    return best_x, best_val

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses multiplicative weights (exponential gradient) with LSE smoothing and
continuation method. The multiplicative update naturally respects a>=0,
and the LSE-continuation progressively sharpens the max approximation.
Combines this with FISTA polishing and multi-start exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    Returns array of length n.
    """
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300 or not np.isfinite(tot):
        return np.zeros(n)
    w = e / tot  # length 2n-1

    # grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # via cross-correlation using FFT
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    med = np.median(c)
    weights = np.maximum(c[top_idx] - med, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def multiplicative_weights(a0, S, beta, max_time, rng, lr_init=0.05, decay=0.9995, verbose=False):
    """
    Multiplicative weights (exponential gradient) with LSE smoothing.
    Natural for non-negative constraints: a_new = a * exp(-lr * grad), then normalize.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    best_max = np.inf
    best_a = a.copy()
    start = time.monotonic()
    lr = lr_init
    it = 0
    patience = 200
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive learning rate: scale by gap to encourage progress
        gap = cur_max - c.mean()
        adaptive_scale = max(0.1, min(2.0, gap / (gnorm * np.max(a) + 1e-15)))
        effective_lr = lr * adaptive_scale

        # Multiplicative update
        log_update = -effective_lr * grad / (gnorm + 1e-15)
        log_update = np.clip(log_update, -5.0, 5.0)
        a_mult = a * np.exp(log_update)

        # Normalize to maintain sum = S
        a_mult_sum = a_mult.sum()
        if a_mult_sum < 1e-30 or not np.isfinite(a_mult_sum):
            a = np.full(n, S / n)
        else:
            a = a_mult * (S / a_mult_sum)

        # Decay learning rate
        lr *= decay

        # Noise injection if stuck
        if stuck > patience:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.1)
            lr = max(lr, lr_init * 0.1)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_max / (S * S)
            print(f"        MW it={it}, t={elapsed:.1f}s, lr={lr:.4f}, obj={obj:.8f}")

    return best_a, best_max


def fista_smooth(a0, S, beta, max_time, rng, verbose=False):
    """
    FISTA for the LSE-smoothed problem with simplex projection.
    Uses backtracking for step size.
    """
    n = len(a0)
    x = project_simplex(a0.copy(), S)
    z = x.copy()
    t_val = 1.0
    L = max(1.0, 4.0 * beta * np.max(x) * n * 0.1)
    start = time.monotonic()
    it = 0
    best_max = np.inf
    best_x = x.copy()
    stuck = 0
    patience = 500

    while time.monotonic() - start < max_time:
        it += 1
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum = (t_val - 1.0) / t_new
        y = x + momentum * (x - z)

        # Compute gradient at y (clip to non-negative for stability)
        y_clip = np.maximum(y, 0.0)
        if y_clip.sum() < 1e-15:
            y_clip = np.full(n, S / n)
        c_y = fft_convolve_self(y_clip)
        grad = compute_lse_gradient(y_clip, c_y, beta)

        # Gradient step with backtracking
        step = 1.0 / L
        accepted = False
        for _ in range(15):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= c_y.max() * (1.0 + 1e-9) + 1e-15:
                x = x_new
                accepted = True
                break
            L *= 1.5
            step = 1.0 / L
        if not accepted:
            x = project_simplex(y - 0.1 / L * grad, S)

        z = x
        t_val = t_new

        c_x = fft_convolve_self(x)
        cur_max = c_x.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > patience:
            noise = 0.005 * S / np.sqrt(n)
            x = project_simplex(x + rng.normal(0, noise, n), S)
            z = x.copy()
            stuck = 0
            patience = int(patience * 1.1)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_max / (S * S)
            print(f"        FISTA it={it}, t={elapsed:.1f}s, L={L:.1f}, obj={obj:.8f}")

    return best_x, best_max


def continuation_optimize(a0, S, max_time, rng, verbose=True):
    """
    Run multiplicative weights + FISTA at increasing beta values.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    start = time.monotonic()

    # Beta schedule: start low, increase geometrically
    betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]

    for i, beta in enumerate(betas):
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 30:
            break

        # Allocate time: more for higher beta
        t_alloc = min(remaining * 0.35, max(10, 30 * (i + 1) ** 0.7))

        if verbose:
            print(f"      beta={beta}, time={t_alloc:.0f}s")

        # Phase A: Multiplicative weights (fast, handles non-negativity)
        t_mw = t_alloc * 0.5
        if t_mw > 5:
            a, best_max = multiplicative_weights(a, S, beta, t_mw, rng,
                                                 lr_init=0.08 / (1 + beta / 100),
                                                 verbose=(i >= 4))

        # Phase B: FISTA polish (more precise)
        rem = t_alloc - (time.monotonic() - start - elapsed + t_mw)
        rem = min(rem, deadline_holder[0] - time.monotonic() - 10)
        if rem > 5:
            a, best_max = fista_smooth(a, S, beta, rem, rng,
                                       verbose=(i >= 5))

        if verbose:
            obj = 2.0 * n * best_max / (S * S)
            print(f"      After beta={beta}: obj={obj:.8f}")

    return a


def generate_init_shape(n, shape='cosine'):
    """Generate various analytic initial shapes."""
    t = np.linspace(0, 1, n, endpoint=False)
    if shape == 'cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05
    elif shape == 'gaussian':
        t_c = t - 0.5
        a = np.exp(-2 * (t_c / 0.25) ** 2) + 0.01
    elif shape == 'bell':
        t_c = t - 0.5
        a = 1.0 / (1.0 + (t_c / 0.3) ** 4) + 0.01
    elif shape == 'uniform':
        a = np.ones(n)
    elif shape == 'bimodal':
        t_c = t - 0.5
        a = np.exp(-4 * (t_c / 0.15) ** 2) + np.exp(-4 * ((t_c - 0.3) / 0.15) ** 2) + \
            np.exp(-4 * ((t_c + 0.3) / 0.15) ** 2) + 0.01
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05
    return a


# Global deadline holder (to avoid closure issues)
deadline_holder = [0.0]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Load provided sequence and normalize.
    2. Run continuation optimization (multiplicative weights + FISTA at increasing LSE beta).
    3. Try diverse initializations (analytic shapes, perturbations).
    4. Try alternative lengths.
    5. Final high-beta polish.
    """
    global deadline_holder
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)
    deadline_holder[0] = deadline

    print(f"=== Optimization starting, budget={budget_s}s, deadline in {budget_s-40:.0f}s ===")

    global_best = np.inf
    global_a = None

    def update_global(a, source=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.8f} (n={len(a)}, {source}) ***")

    # Load provided sequence
    a_provided = None
    try:
        a_prov = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_prov = np.clip(a_prov, 0, 1000)
        if a_prov.sum() > 0.01 and len(a_prov) >= 256:
            a_provided = a_prov
            print(f"Provided: len={len(a_prov)}, sum={a_prov.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        a_provided = None

    # ========== Phase 1: Primary optimization on provided sequence ==========
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())
        a1 = project_simplex(a1, S1)

        t1 = min(900, (deadline - time.monotonic()) * 0.40)
        print(f"\n--- Phase 1: Continuation on provided, n={n1}, time={t1:.0f}s ---")
        a_opt = continuation_optimize(a1, S1, t1, rng, verbose=True)
        update_global(a_opt, "Phase 1 primary")

    # ========== Phase 2: Diverse initializations ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n = len(global_a)
        S = np.sqrt(2.0 * n)

        shapes = ['cosine', 'gaussian', 'bimodal', 'bell']
        for si, shape in enumerate(shapes):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(150, rem * 0.2)
            if t_alloc < 20:
                continue
            print(f"\n--- Phase 2: {shape} init, n={n}, time={t_alloc:.0f}s ---")
            a_init = generate_init_shape(n, shape)
            a_init = project_simplex(a_init * (S / a_init.sum()), S)
            a_opt = continuation_optimize(a_init, S, t_alloc, rng, verbose=False)
            update_global(a_opt, f"Phase 2 {shape}")

    # ========== Phase 3: Perturbation restarts from best ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = 4
        t_per = (remaining - 30) / n_restarts

        print(f"\n--- Phase 3: {n_restarts} perturbation restarts, {t_per:.0f}s each ---")
        for ri in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(t_per, rem - 10)
            if t_alloc < 15:
                break

            # Different perturbation scales
            scales = [0.01, 0.03, 0.05, 0.10]
            noise_scale = scales[ri % len(scales)] * S / np.sqrt(n)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)

            # Use continuation with moderate time
            a_opt = continuation_optimize(a_pert, S, t_alloc, rng, verbose=False)
            update_global(a_opt, f"Phase 3 restart {ri} (scale={scales[ri % len(scales)]})")

    # ========== Phase 4: Try different lengths ==========
    remaining = deadline - time.monotonic()
    for n_try in [4096, 16384]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if global_a is not None and len(global_a) == n_try:
            continue
        S = np.sqrt(2.0 * n_try)
        t_alloc = min(200, rem * 0.25)
        if t_alloc < 30:
            continue

        print(f"\n--- Phase 4: n={n_try}, time={t_alloc:.0f}s ---")
        if global_a is not None and len(global_a) > 1:
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_init = np.interp(t_new, t_old, global_a)
        else:
            a_init = generate_init_shape(n_try, 'cosine')
        a_init = np.maximum(a_init, 0)
        a_init = project_simplex(a_init * (S / a_init.sum()), S)
        a_opt = continuation_optimize(a_init, S, t_alloc, rng, verbose=False)
        update_global(a_opt, f"Phase 4 n={n_try}")

    # ========== Phase 5: Final intensive polish ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a_curr = project_simplex(global_a, S)
        t5 = remaining - 15
        print(f"\n--- Phase 5: Final polish, n={n}, time={t5:.0f}s ---")

        # Very high beta FISTA
        a_opt, _ = fista_smooth(a_curr, S, 50000, t5 * 0.5, rng, verbose=True)
        update_global(a_opt, "Phase 5 FISTA high-beta")

        # Then multiplicative weights with even higher beta
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt2, _ = multiplicative_weights(a_opt, S, 100000, rem, rng,
                                               lr_init=0.01, verbose=True)
            update_global(a_opt2, "Phase 5 MW ultra-high-beta")

    # ========== Final result ==========
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(4096) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(4096) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={len(final_list)}, obj={ver:.8f}")
    print(f"{'='*60}")

    return final_list
