**Spectral shaping approach:**

Goal: Make |A(f)|^2 as flat as possible, where A = FFT(a) and a >= 0.

Algorithm:
1. Compute A = FFT(a)
2. Compute P = |A|^2
3. Target: P_target = mean(P) (flat spectrum)
4. Update: P_new = P^(1-alpha) * P_target^alpha for alpha in (0,1)

5. a_new = real(IFFT(A_new))
6. Clip: a_new = max(a_new, 0)
7. Renormalize: a_new = a_new / sum(a_new) * S

The parameter alpha controls the aggressiveness. Small alpha = gentle flattening, large alpha = aggressive.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_shaping_peak_targeted(a, alpha, beta_flat=0.1, rng=None):
    """
    Peak-targeted spectral shaping.
    Flattens the power spectrum while biasing against the current peak lag.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    # Compute convolution to find peak
    c = np.real(ifft(power))
    peak_idx = np.argmax(c[:2*n-1])
    
    # Compute peak kernel: cos(2*pi*f*peak_idx/fsize)
    freqs = np.arange(fsize)
    peak_kernel = np.cos(2.0 * np.pi * freqs * peak_idx / fsize)
    
    # Combine flat target with peak reduction
    # Target power: mean_p * (1 - alpha * peak_kernel)
    # This reduces power at frequencies that contribute to the peak
    target = mean_p * (1.0 - alpha * peak_kernel)
    target = np.maximum(target, mean_p * 0.1)  # Don't reduce too much
    
    # Interpolate between current power and target
    new_power = power * (1.0 - beta_flat) + target * beta_flat
    new_power = np.maximum(new_power, 1e-30)
    
    # Update amplitude, keep phase
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * (S / s_new)


def spectral_flatten(a, alpha):
    """Standard spectral flattening (no peak targeting)."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    # Geometric mean interpolation: new_power = power^(1-alpha) * mean_p^alpha
    log_p = np.log(np.maximum(power, 1e-30))
    log_target = (1.0 - alpha) * log_p + alpha * np.log(mean_p)
    new_power = np.exp(log_target)
    
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * (S / s_new)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Novel multi-phase approach with peak-targeted spectral shaping as the
primary optimization engine, combined with adaptive LSE gradient polishing.
Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_vec(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    diff = c_top - cm
    scaled = beta * diff
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, diff.max())
        if scale > 0:
            beta = scale
            scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def peak_targeted_spectral_step(a, fsize, alpha=0.15, num_peaks=3):
    """
    Peak-targeted spectral shaping step.
    Uses cosine kernel analysis to identify which frequency bins
    contribute most to convolution peaks, then redistributes power.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    # Get convolution to find peaks
    c = np.real(ifft(power))
    c_lin = c[:2 * n - 1]
    avg_c = c_lin.mean()

    # Find top peaks (significantly above average)
    threshold = avg_c + 0.1 * (c_lin.max() - avg_c)
    peak_mask = c_lin > threshold
    peak_indices = np.where(peak_mask)[0]

    # Limit to top num_peaks
    if len(peak_indices) > num_peaks:
        peak_vals = c_lin[peak_indices]
        top_order = np.argsort(peak_vals)[::-1][:num_peaks]
        peak_indices = peak_indices[top_order]

    if len(peak_indices) == 0:
        return a

    # Compute peak pressure: H(f) = sum over peaks of w_k * cos(2*pi*f*k/fsize)
    freqs = np.arange(fsize)
    H = np.zeros(fsize)
    for k_star in peak_indices:
        # Weight proportional to how much this peak exceeds average
        w_k = (c_lin[k_star] - avg_c) / max(1e-15, c_lin.max() - avg_c)
        H += w_k * np.cos(2.0 * np.pi * freqs * k_star / fsize)

    # Normalize H
    H_max = np.abs(H).max()
    if H_max < 1e-20:
        return a
    H = H / H_max

    # Multiplicative update: reduce power where H > 0, increase where H < 0
    # Also include a flattening component
    flat_component = 1.0  # Target: flat spectrum
    peak_component = 1.0 - alpha * H

    # Combine: interpolate between current and target
    target_power = mean_p * flat_component * peak_component
    target_power = np.maximum(target_power, mean_p * 0.05)

    # Geometric mean interpolation (works better in log domain)
    log_cur = np.log(np.maximum(power, 1e-30))
    log_tgt = np.log(target_power)
    mix = 0.3  # Mixing factor
    log_new = (1.0 - mix) * log_cur + mix * log_tgt
    new_power = np.exp(log_new)

    # Reconstruct with same phase
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)

    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * (S / s_new)


def spectral_flatten_step(a, fsize, alpha=0.1):
    """Standard spectral flattening (geometric mean interpolation)."""
    n = len(a)
    S = a.sum()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    log_p = np.log(np.maximum(power, 1e-30))
    log_mean = np.log(mean_p)
    log_new = (1.0 - alpha) * log_p + alpha * log_mean
    new_power = np.exp(log_new)

    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)

    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * (S / s_new)


def hybrid_spectral(a, fsize, alpha_peak, alpha_flat, num_peaks, do_gradient, grad_beta, grad_k, S):
    """One hybrid step: spectral shaping + optional gradient step."""
    n = len(a)
    # Spectral step
    if np.random.random() < 0.6:
        a = peak_targeted_spectral_step(a, fsize, alpha_peak, num_peaks)
    else:
        a = spectral_flatten_step(a, fsize, alpha_flat)

    # Optional gradient step
    if do_gradient:
        c = fft_convolve_self(a)
        grad = topk_lse_gradient(a, c, grad_k, grad_beta)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            step = 0.5 / max(grad_beta * 0.1, 1e-8)
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c.max():
                a = a_new
    return a

def run_fista_polish(a_init, S, max_time, seed=0):
    """High-beta FISTA polishing for final refinement."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = normalize_vec(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stuck = 0

    k = max(2, int(0.005 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 50.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        step = 1.0 / max(2.0 * beta * 0.05, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        improved = False
        for _ in range(12):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-13:
                improved = True
                break
            step *= 0.5
            if step < 1e-16:
                break
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Restart if stuck
        if stuck > 3000:
            progress = (time.monotonic() - start) / max_time
            noise = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise > 1e-12:
                y = project_simplex(y + rng.normal(0, noise, n), S)
                t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def generate_random_init(n, S, rng):
    """Generate a random non-negative sequence with the given sum."""
    a = rng.random(n)
    a = a / a.sum() * S
    return a


def generate_spectral_init(n, S, rng):
    """Generate initialization from a random-phase flat spectrum."""
    fsize = 1 << (2 * n - 1).bit_length()
    # Random phase, flat amplitude
    amp = np.ones(fsize) * np.sqrt(fsize)
    phase = rng.uniform(0, 2 * np.pi, fsize)
    # Ensure Hermitian symmetry for real output
    phase[0] = 0
    for f in range(1, fsize // 2 + 1):
        if fsize - f < fsize:
            phase[fsize - f] = -phase[f]
    A = amp * np.exp(1j * phase)
    a = np.real(ifft(A, fsize))[:n]
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) * S / n
    return a * (S / s)


def run_peak_reduction_coordinate(a_init, S, max_time, seed=0):
    """
    Coordinate descent: for each coordinate, find the optimal pairing transfer.
    Uses a randomized subset of pairs per sweep for efficiency.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = normalize_vec(a_init.copy(), S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    # Number of random pairs to try per sweep
    num_pairs = min(200, n)

    while time.monotonic() - start < max_time:
        # Pick random pairs
        indices = rng.integers(0, n, size=2 * num_pairs)
        for i in range(num_pairs):
            j = indices[2 * i]
            m = indices[2 * i + 1]
            if j == m:
                continue

            # Current values
            aj = a[j]
            if aj < 1e-15:
                continue

            # Try transferring mass from j to m
            # b_new[k] = b[k] + t * (a[k-m] - a[k-j]) for affected k
            # We want to find t in [0, aj] that minimizes max(b_new)

            # Compute the "transfer kernel"
            # Contribution from j: a[k-j] for k in [j, j+n-1]
            # Contribution from m: a[k-m] for k in [m, m+n-1]

            # Simplified: just try a few candidate t values
            # t = aj * frac for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]
            best_t = 0.0
            best_max = c.max()

            # Quick check: would transferring help?
            # The peak of c is at some k*. Removing a[j] helps if a[k*-j] is large
            # (reduces the peak), adding a[m] hurts if a[k*-m] is large.
            k_star = np.argmax(c)
            remove_factor = a[k_star - j] if 0 <= k_star - j < n else 0.0
            add_factor = a[k_star - m] if 0 <= k_star - m < n else 0.0

            # Only try if removal helps more than addition hurts
            if remove_factor <= add_factor:
                continue

            # Try small transfers
            for frac in [0.02, 0.05, 0.1, 0.15]:
                t = aj * frac
                a[j] -= t
                a[m] += t

                # Recompute convolution (expensive but needed for accuracy)
                # Actually, let's use the incremental update for speed
                # b_new[k] = b[k] - t*a[k-j] + t*a[k-m]
                # But this only works for k in the affected range...
                # For safety, just recompute fully (we can optimize later)
                c_new = fft_convolve_self(a)
                max_new = c_new.max()

                if max_new < best_max:
                    best_max = max_new
                    best_t = t
                    if max_new < best:
                        best = max_new
                        best_a = a.copy()
                else:
                    a[j] += t
                    a[m] -= t

            # Apply best transfer
            if best_t > 0:
                a[j] -= best_t
                a[m] += best_t
                c = fft_convolve_self(a)
            else:
                # Restore (should already be restored)
                pass

    return best_a, best

    if t3_limit > 60 and global_a is not None:
        n_large = 16384
        S_large = np.sqrt(2.0 * n_large)
        a_large = resample_sequence(global_a, n_large)
        a_large = normalize_vec(a_large, S_large)
        print("\n=== Phase 3: Hybrid spectral, n=%d, t=%.0fs ===" % (n_large, t3_limit))
        a_opt, val_opt = run_hybrid_spectral_phase(a_large, S_large, t3_limit, seed=seed + 2)
        try_sequence(a_opt, "Phase3 large-n")

        # Quick FISTA polish on large-n result
        remaining2 = deadline - time.monotonic()
        t_pol = min(100, remaining2 * 0.3)
        if t_pol > 20:
            print("  FISTA polish on n=%d, t=%.0fs" % (n_large, t_pol))
            a_opt2, val_opt2 = run_fista_polish(a_opt, S_large, t_pol, seed=seed + 3)
            try_sequence(a_opt2, "Phase3 polish")

    # ============================================================
    # Phase 4: Alternative initialization and restart
    # ============================================================
    remaining = deadline - time.monotonic()
    t4_limit = min(400, remaining * 0.5)
    if t4_limit > 60:
        # Try a different initialization
        a_alt = generate_spectral_init(n_primary, S_primary, rng)
        print("\n=== Phase 4: Alternative init, n=%d, t=%.0fs ===" % (n_primary, t4_limit))
        a_opt, val_opt = run_hybrid_spectral_phase(a_alt, S_primary, t4_limit, seed=seed + 100)
        try_sequence(a_opt, "Phase4 alt-init")

        # Also try random init
        remaining3 = deadline - time.monotonic()
        t_rand = min(200, remaining3 * 0.4)
        if t_rand > 30:
            a_rand = generate_random_init(n_primary, S_primary, rng)
            print("  Random init, t=%.0fs" % t_rand)
            a_opt, val_opt = run_hybrid_spectral_phase(a_rand, S_primary, t_rand, seed=seed + 200)
            try_sequence(a_opt, "Phase4 random")

    # ============================================================
    # Phase 5: Final intensive polishing
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_final = remaining * 0.7
        print("\n=== Phase 5: Final polish, n=%d, t=%.0fs ===" % (n_best, t_final))

        # High-beta FISTA
        a_fin, val_fin = run_fista_polish(global_a, S_best, t_final, seed=seed + 500)
        try_sequence(a_fin, "Phase5 FISTA")

        # Additional spectral polish
        remaining4 = deadline - time.monotonic() - 10
        if remaining4 > 20 and global_a is not None:
            n_b2 = len(global_a)
            S_b2 = np.sqrt(2.0 * n_b2)
            a_sp, val_sp = run_hybrid_spectral_phase(global_a, S_b2, remaining4, seed=seed + 600)
            try_sequence(a_sp, "Phase5 spectral")

    # ============================================================
    # Verification
    # ============================================================
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
