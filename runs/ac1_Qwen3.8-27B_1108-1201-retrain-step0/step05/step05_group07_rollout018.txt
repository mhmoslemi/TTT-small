import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0)
    # Standard simplex projection
    if v.sum() >= S:
        # Need to reduce
        mu = np.zeros(n)
        r = np.sort(v)[::-1]
        tmp = np.cumsum(r) - S
        cond = r - tmp / np.arange(1, n+1) > 0
        if np.any(cond):
            mu = tmp[np.where(cond)[0][-1]] / np.where(cond)[0][-1]
        result = np.maximum(v - mu, 0)
    else:
        # Need to increase - distribute proportionally or uniformly
        deficit = S - v.sum()
        if v.sum() > 0:
            result = v + deficit * v / v.sum()
        else:
            result = np.full(n, S/n)
    return result

def lse_gradient(a, c, beta):
    """Compute gradient of LSE objective using all lags."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    w = e / e.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    N = 1 << (2*n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    g = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return g

Now let me think about the Adam optimizer. Adam maintains:
- `m = beta1 * m + (1-beta1) * g` (first moment)
- `v = beta2 * v + (1-beta2) * g^2` (second moment)
- `m_hat = m / (1 - beta1^t)`
- `v_hat = v / (1 - beta2^t)`
- `a = a - lr * m_hat / (sqrt(v_hat) + eps)`

After the update, we project onto the simplex.

The issue with Adam + simplex projection: the projection changes the direction. But in practice, this works well.

Let me also think about the "all-lag" vs "top-k" tradeoff. The LSE with high beta approximates the max, so it automatically focuses on the peaks. With low beta, it's more uniform. I'll start with moderate beta and increase it over time.

Let me also add a "flatness" reward. The ideal is a perfectly flat convolution. We can measure flatness as `max(c) / mean(c)` or `max(c) - min(c)`. Minimizing this is equivalent to minimizing the peak (since the sum is fixed).

Actually, since sum(c) = (sum(a))^2 is fixed, minimizing max(c) is equivalent to minimizing max(c) - mean(c) (the excess over average). So the LSE approach is exactly right.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed gradient descent
with Adam optimizer, adaptive beta scheduling, and periodic restarts.
Key innovations over previous approach:
1. All-lag LSE gradient (not just top-k subgradient) for smoother optimization
2. Adam optimizer with per-component adaptive learning rates
3. Periodic exponential backtracking for step size
4. Coordinate descent polishing
5. Multi-initialization and diversity maintenance
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    
    # First, handle the case where sum is already close to S
    current_sum = v.sum()
    if abs(current_sum - S) < 1e-10:
        return np.maximum(v, 0)
    
    if current_sum < S:
        # Need to add mass
        deficit = S - current_sum
        if current_sum > 1e-15:
            # Distribute proportionally
            v = v + deficit * v / current_sum
        else:
            # Distribute evenly
            v = v + deficit / n
        # After proportional scaling, might still not be exact due to clipping
        v = np.maximum(v, 0)
        diff = S - v.sum()
        if diff > 0:
            v += diff / n
        return v
    
    # current_sum > S: standard simplex projection
    v = np.maximum(v, 0)
    r = np.sort(v)[::-1]
    cumsum_r = np.cumsum(r)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tmp = (cumsum_r - S) / idx
    cond = r > tmp
    if np.any(cond):
        rho = int(np.where(cond)[0][-1]) + 1
        theta = (cumsum_r[rho - 1] - S) / rho
    else:
        theta = 0.0
    v = np.maximum(v - theta, 0.0)
    # Correct for floating point
    v += (S - v.sum()) / n
    v = np.maximum(v, 0)
    return v


def compute_lse_grad(a, c, beta):
    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -600, 0)
    e = np.exp(scaled)
    w = e / e.sum()
    
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    g = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return g


def compute_topk_grad(a, c, k):
    """Subgradient from top-k lags (fallback for very high beta)."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    weights = np.ones(k) / k
    
    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def opt_lse_adam(a0, S, max_time, rng, beta_init=20.0, beta_final=5000.0,
                 lr_init=0.1, verbose=True):
    """
    LSE-smoothed gradient descent with Adam optimizer.
    Minimizes max(conv(a,a)) subject to sum(a)=S, a>=0.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    start = time.monotonic()
    stuck = 0
    patience = 2000
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adaptive beta: increase over time for sharper peak focusing
        beta = beta_init * np.exp(np.log(beta_final / beta_init) * progress**0.5)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute gradient
        if beta > 2000:
            grad = compute_topk_grad(a, c, k=int(4 + 64 * progress))
        else:
            grad = compute_lse_grad(a, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.1 * progress)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = project_simplex(a - step, S)
        
        # Line search: accept if it doesn't increase the max too much
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max * (1.0 + 1e-8):
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - 0.5 * step, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max * (1.0 + 1e-8):
                a = a_new
            else:
                # Very small step
                a = project_simplex(a - 0.1 * step, S)
        
        # Noise injection when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.001 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            # Reset Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stuck = 0
            patience = int(patience * 1.3)
        
        if verbose and t % 2000 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    it={t}, t={elapsed:.0f}s, obj={obj:.6f}, "
                  f"max={best_max:.8f}, beta={beta:.0f}")
    
    return best_a, best_max


def coordinate_descent(a0, S, max_time, rng, verbose=False):
    """
    Coordinate descent: for each component, find optimal value via line search.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        # Random order of coordinates
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > max_time:
                break
            
            # Try perturbations
            best_local = a[i]
            best_cmax = c.max()
            
            # Try a few perturbation sizes
            for scale in [0.5, 1.0, 2.0, 5.0]:
                delta = scale * S / n * rng.uniform(-1, 1)
                a_try = a.copy()
                a_try[i] = max(0, a[i] + delta)
                # Re-normalize
                a_try = project_simplex(a_try, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_cmax:
                    best_cmax = c_try.max()
                    best_local = a_try[i]
                    improved = True
            
            # Apply best local improvement
            if a[i] != best_local:
                a[i] = best_local
                a = project_simplex(a, S)
                c = fft_convolve_self(a)
        
        if c.max() < best_max - 1e-14:
            best_max = c.max()
            best_a = a.copy()
        
        if not improved:
            break
        
        if verbose:
            print(f"    CD round complete, max={best_max:.8f}")
    
    return best_a, best_max


def generate_random_init(n, rng, S):
    """Generate a random non-negative sequence with sum S."""
    a = rng.random(n)
    a = a / a.sum() * S
    return a


def generate_spectral_init(n, rng, S):
    """
    Generate initialization with flat-ish spectrum.
    Start with random phase in frequency domain, project to non-negative.
    """
    N = 1 << (n - 1).bit_length()
    # Random magnitudes (close to flat) with random phases
    mags = S / n * (1 + 0.3 * rng.random(N // 2 + 1))
    phases = 2 * np.pi * rng.random(N // 2 + 1)
    
    # Build conjugate-symmetric spectrum
    spec = np.zeros(N, dtype=np.complex128)
    spec[0] = mags[0]
    for k in range(1, N // 2 + 1):
        spec[k] = mags[k] * np.exp(1j * phases[k])
        spec[N - k] = np.conj(spec[k])
    
    a = np.real(ifft(spec))[:n]
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum() * S
    else:
        a = np.full(n, S / n)
    return a
