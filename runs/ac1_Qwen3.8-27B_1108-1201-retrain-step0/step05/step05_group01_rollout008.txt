"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FISTA with LSE-smoothed max, spectral shaping, top-k polishing,
and multi-start with diverse initializations including Fourier-domain
flattening. Key insight: self-convolution flatness relates to
magnitude spectrum flatness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y).
    c[k] = sum_j y[j]*y[k-j], so d(c[k])/d(y[m]) = y[k-m] (if valid).
    Thus grad_m = 2 * sum_k w[k] * y[k-m] = 2 * (w * y_rev)[m + n - 1]"""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient focusing on largest convolution entries."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val < 1e-30:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_flatten_step(a, rng, n_iters=50, step_size=0.3):
    """
    Try to flatten |A(k)| in Fourier domain, then project back to simplex.
    This encourages a flatter convolution.
    """
    n = len(a)
    S = a.sum()
    a_copy = a.copy()
    
    for _ in range(n_iters):
        A = fft(a_copy)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flatten magnitude
        target_mag = np.full(n, S / np.sqrt(n))
        # Blend current and target
        new_mag = (1 - step_size) * mag + step_size * target_mag
        # Ensure positivity
        new_mag = np.maximum(new_mag, 0)
        # Reconstruct (enforce real-valued by conjugate symmetry)
        new_A = new_mag * np.exp(1j * phase)
        # Enforce conjugate symmetry for real output
        new_A_sym = new_A.copy()
        new_A_sym[1:n//2] = new_A[1:n//2]
        new_A_sym[n//2+1:] = np.conj(new_A_sym[1:n//2][::-1])
        if n % 2 == 0:
            new_A_sym[n//2] = np.real(new_A_sym[n//2])
        else:
            new_A_sym[(n-1)//2] = np.real(new_A_sym[(n-1)//2])
        new_a = np.real(ifft(new_A_sym))
        new_a = np.maximum(new_a, 0.0)
        if new_a.sum() > 1e-15:
            new_a = new_a * (S / new_a.sum())
        a_copy = new_a
    
    return a_copy


def structured_perturb(a, c, rng, S, scale=0.01):
    """
    Perturb in directions that specifically try to reduce the top convolution peaks.
    For each top peak at lag k, add noise that redistributes mass away from
    the indices contributing most to that peak.
    """
    n = len(a)
    a_pert = a.copy()
    
    # Find top peaks
    top_k = np.argsort(c)[::-1][:5]
    
    for k in top_k:
        # c[k] = sum_j a[j]*a[k-j]
        # The main contributors are where both a[j] and a[k-j] are large
        valid_j = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(valid_j) == 0:
            continue
        contrib = a[valid_j] * a[k - valid_j]
        # Perturb: reduce mass where contribution is high
        mask = np.zeros(n)
        mask[valid_j] = contrib
        noise = rng.normal(0, scale, n) * mask
        a_pert = a_pert + noise
    
    a_pert = project_simplex(a_pert, S)
    return a_pert

            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        time_budget_this = min(remaining * 0.9, 120.0)
        t0 = time.monotonic()

        maxiter = max(20, int(time_budget_this / 0.5))
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': maxiter, 'maxls': 20, 'ftol': 1e-14,
                     'gtol': 1e-10, 'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (iters=%d)" %
              (beta, pen, obj_new, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def generate_diverse_inits(n, rng, S):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine with various frequencies
    for freq in [1, 2]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 - amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 0.01)
            a = a * (S / a.sum())
            inits.append(a.copy())

    # U-shaped (high at edges)
    for width in [0.2, 0.3, 0.5]:
        center = 0.5
        a = 1.0 / (1.0 + ((t - center) / width) ** 4)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Two-peak
    for frac in [0.25, 0.33]:
        a = np.exp(-0.5 * ((np.arange(n) - n * frac) / (n * 0.05)) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - n * (1 - frac)) / (n * 0.05)) ** 2)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Smooth random (low-pass filtered)
    for _ in range(3):
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(3, n // 200)
        F[cutoff:-cutoff] *= 0.01
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() > 1e-10:
            smooth = smooth * (S / smooth.sum())
            inits.append(smooth.copy())

    # Spectrally shaped: try to make |FFT(a)| flat
    phase = rng.uniform(-np.pi, np.pi, n)
    # Enforce conjugate symmetry
    phase[0] = 0
    if n % 2 == 0:
        phase[n//2] = 0
    for k in range(1, n // 2 + 1):
        nk = n - k
        if nk < n:
            phase[nk] = -phase[k]
    mag = np.ones(n) * (S / np.sqrt(n))
    A = mag * np.exp(1j * phase)
    a_spec = np.real(ifft(A))
    a_spec = np.maximum(a_spec, 0.0)
    if a_spec.sum() > 1e-10:
        a_spec = a_spec * (S / a_spec.sum())
        inits.append(a_spec.copy())

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA with spectral shaping, top-k polish,
    L-BFGS-B, multi-start, and final squeeze.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except (NameError, Exception):
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.6f" % (n0, obj0))
        if obj0 < global_best_obj:
            global_best_obj = obj0
            global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 1.0 - 0.5 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.01)
        S = np.sqrt(2.0 * n0)
        a_start = a_start * (S / a_start.sum())
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    S = np.sqrt(2.0 * n0)

    # === Phase 1: Main FISTA (35% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: FISTA main (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_main(a_start, p1_budget, rng, beta_start=5.0, mode="lse")
    obj1 = compute_obj(a1)
    print("  FISTA main: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: Top-k polish (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: Top-k polish (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_topk_polish(global_best_a, p2_budget, rng)
        obj2 = compute_obj(a2)
        print("  Top-k polish: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: L-BFGS-B polishing (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.15
        print("\n=== Phase 3: L-BFGS-B polishing (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, obj3 = run_lbfgs_polish(global_best_a, p3_budget, rng, n_beta_levels=4)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Multi-start (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Multi-start diverse inits (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        inits = generate_diverse_inits(n0, rng, S)
        inits.append(global_best_a.copy())

        time_per_init = p4_budget / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time_per_init < 3:
                break
            a_m, _ = run_fista_main(init, min(time_per_init, 15.0), rng,
                                    beta_start=10.0 + 5 * idx, mode="lse")
            obj_m = compute_obj(a_m)
            if obj_m < global_best_obj:
                global_best_obj = obj_m
                global_best_a = a_m.copy()
                print("  Multi-start %d: obj=%.8f (IMPROVED!)" % (idx, obj_m))
            else:
                print("  Multi-start %d: obj=%.8f" % (idx, obj_m))

    # === Phase 5: Final FISTA with high beta (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: High-beta FISTA (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, _ = run_fista_main(global_best_a, p5_budget, rng, beta_start=50.0, mode="lse")
        obj5 = compute_obj(a5)
        print("  High-beta FISTA: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final top-k squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final top-k squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, _ = run_topk_polish(global_best_a, remaining * 0.8, rng)
        obj6 = compute_obj(a6)
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj + 1e-10:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
