import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_topk_weights(c, K, beta):
    """Compute exponential-decay weights on top K lags."""
    cmax = c.max()
    range_c = cmax - c.min() + 1e-15
    # Weight: exp(-beta * (cmax - c[k]) / range_c) for top K
    threshold = np.sort(c)[::-1][min(K, len(c)) - 1]
    w = np.zeros(len(c))
    mask = c >= threshold
    w[mask] = np.exp(-beta * (cmax - c[mask]) / range_c)
    wsum = w.sum()
    if wsum < 1e-30:
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    return w


def weighted_convolve_gradient(a, w, fsize=None):
    """Compute 2 * convolve(w, a) truncated to len(a)."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def mirror_step(a, grad, eta, S):
    """Mirror descent (KL-proximal) step."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()

    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return np.maximum(a_new * (S / s), 0.0)
