def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (css - S) / rho > 0
    rho_val = np.max(np.where(cond)[0]) + 1
    theta = (css[rho_val-1] - S) / rho_val
    return np.maximum(v - theta, 0.0)

def lse_grad(a, p):
    """Gradient of LSE(conv(a,a)) w.r.t. a."""
    n = len(a)
    fsize = 1 << (3*n - 1).bit_length()
    A = fft(a, fsize)
    P = fft(p, fsize)
    Pr = fft(p[::-1], fsize)
    c1 = np.real(ifft(P * A))
    c2 = np.real(ifft(Pr * A))
    return 2.0 * (c2[2*n-2:3*n-2] + c1[:n])
