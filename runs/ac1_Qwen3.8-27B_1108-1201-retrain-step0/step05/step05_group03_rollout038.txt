"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: LP-based alternating optimization with gradient polishing.
The key idea is to alternately fix one sequence and solve an LP for the other,
which is a principled approach different from pure gradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import scipy.sparse as sp


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def build_lp_matrix(a):
    """Build the sparse constraint matrix A for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. A@g <= M*1, g >= 0.
    Returns g if successful, None otherwise.
    """
    n = len(a)
    A = build_lp_matrix(a)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 300, 'presolve': True})
        if res.status == 0:
            return res.x
        elif res.status == 9:  # time limit
            print("  LP: time limit reached")
            return None
        else:
            print("  LP failed: status %d, message: %s" % (res.status, res.message))
            return None
    except Exception as e:
        print("  LP exception: %s" % e)
        return None


def run_lp_optimizer(a_init, S, max_time, rng, tol=1e-6, max_iter=200):
    """LP-based optimization: alternately solve LP for g given a.
    Returns (best_a, best_max_conv).
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    M = c.max()
    best_M = M
    best_a = a.copy()
    S_orig = a.sum()
    
    print("  LP start: M=%.6f, obj=%.6f" % (M, 2.0 * n * M / (S_orig ** 2)))
    
    for it in range(max_iter):
        if time.monotonic() - start > max_time:
            break
        
        t0 = time.monotonic()
        g = solve_lp(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            print("  Iter %d: LP failed, switching to gradient." % it)
            break
        
        sum_g = g.sum()
        if sum_g < S * (1.0 + tol):
            print("  Iter %d: LP converged (sum_g=%.6f, S=%.6f)" % (it, sum_g, S))
            break
        
        # Rescale g to have sum S
        g = g * (S / sum_g)
        
        # Check if this reduces max convolution
        c_g = fft_convolve_self(g)
        M_g = c_g.max()
        
        # The new "effective M" after blending
        # If we just replace a with g, the new M is M_g
        # But we need a*g <= M, which means the blended sequence might have lower max
        
        # Let's try: new_a = blend of a and g
        # Actually, let's just use g directly (rescaled)
        if M_g < best_M:
            best_M = M_g
            best_a = g.copy()
            a = g
            M = M_g
            print("  Iter %d: M improved to %.6f (LP time=%.1fs)" % (it, M, lp_time))
        else:
            # Try blending
            for t in [0.5, 0.7, 0.9]:
                a_blend = (1 - t) * a + t * g
                a_blend = normalize(a_blend, S)
                c_blend = fft_convolve_self(a_blend)
                M_blend = c_blend.max()
                if M_blend < best_M:
                    best_M = M_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    M = M_blend
                    print("  Iter %d: blend t=%.1f, M=%.6f" % (it, t, M))
                    break
            else:
                print("  Iter %d: no improvement, M_g=%.6f >= M=%.6f" % (it, M_g, M))
                # Try small perturbation
                a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                c = fft_convolve_self(a)
                M = c.max()
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
    
    return best_a, best_M

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Hybrid LP + gradient optimization.
- LP phase: alternately solve max sum(g) s.t. a*g <= M, g >= 0 (from SoTA theory)
- Gradient phase: FISTA with adaptive LSE smoothing
- Alternate between phases, try multiple n values
- Key insight: non-symmetric sequences can have max convolution below (sum a)^2/n
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def build_lp_structure(n):
    """Precompute row/col indices for LP constraint matrix (2n-1) x n."""
    ks = np.arange(2 * n - 1)
    i_starts = np.maximum(0, ks - n + 1)
    i_ends = np.minimum(n - 1, ks)
    counts = i_ends - i_starts + 1
    rows = np.repeat(ks, counts).astype(np.int32)
    total = int(counts.sum())
    cols = np.empty(total, dtype=np.int32)
    pos = 0
    for k in range(2 * n - 1):
        c = int(counts[k])
        cols[pos:pos + c] = np.arange(i_starts[k], i_starts[k] + c, dtype=np.int32)
        pos += c
    return rows, cols


def solve_lp_for_g(a, M, rows, cols, time_limit=120):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    indices = rows.astype(np.int64) - cols.astype(np.int64)
    vals = a[indices]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return res.x
        else:
            return None
    except Exception:
        return None


def run_lp_optimizer(a_init, S, max_time, rng, n_lp=None, lp_time_limit=60):
    """LP-based optimization loop."""
    if n_lp is None:
        n_lp = len(a_init)
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    
    # If n_lp != n, we work at n_lp resolution
    if n_lp != n:
        # Downsample a to n_lp for LP, but keep a at full resolution
        a_lp = resample_to(a, n_lp)
        a_lp = normalize(a_lp, S)
    else:
        a_lp = a.copy()
    
    S_lp = a_lp.sum()
    n_active = len(a_lp)
    
    rows, cols = build_lp_structure(n_active)
    print("  LP structure built: %d entries" % len(rows))
    
    start = time.monotonic()
    c = fft_convolve_self(a_lp)
    M = c.max()
    best_M = M
    best_a_lp = a_lp.copy()
    
    for it in range(500):
        if time.monotonic() - start > max_time:
            break
        t0 = time.monotonic()
        g = solve_lp_for_g(a_lp, M, rows, cols, time_limit=lp_time_limit)
        lp_time = time.monotonic() - t0
        
        if g is None:
            print("  LP iter %d: failed (%.1fs)" % (it, lp_time))
            break
        
        sum_g = g.sum()
        if sum_g < S_lp * (1.0 + 1e-8):
            print("  LP iter %d: converged (sum_g=%.4f)" % (it, sum_g))
            break
        
        g_norm = g * (S_lp / sum_g)
        c_g = fft_convolve_self(g_norm)
        M_g = c_g.max()
        
        if M_g < best_M:
            best_M = M_g
            best_a_lp = g_norm.copy()
            a_lp = g_norm
            M = M_g
            print("  LP iter %d: M=%.6f (sum_g/S=%.4f, time=%.1fs)" % (it, M, sum_g / S_lp, lp_time))
        else:
            # Try blending
            improved = False
            for t in [0.3, 0.5, 0.7, 0.9]:
                a_blend = (1 - t) * a_lp + t * g_norm
                a_blend = normalize(a_blend, S_lp)
                c_bl = fft_convolve_self(a_blend)
                if c_bl.max() < best_M:
                    best_M = c_bl.max()
                    best_a_lp = a_blend.copy()
                    a_lp = a_blend
                    M = best_M
                    improved = True
                    print("  LP iter %d: blend t=%.1f, M=%.6f" % (it, t, M))
                    break
            if not improved:
                print("  LP iter %d: no improvement (M_g=%.6f >= M=%.6f)" % (it, M_g, M))
                break
    
    # Upsample back to original n if needed
    if len(best_a_lp) != n:
        best_a = resample_to(best_a_lp, n)
        best_a = normalize(best_a, S)
    else:
        best_a = best_a
    
    return best_a, M


def resample_to(a, n_target):
    """Resample array a to length n_target."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.004, beta_max=20000.0, k_frac=0.04):
    """FISTA with top-k LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-13:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new
        
        # Periodic restart
        if stuck > 800:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
        if it % 300 == 0 and stuck > 200 and rng.random() < 0.4:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
    
    return best_a, best


def run_subgradient(a_init, S, max_time, rng):
    """Subgradient method targeting the exact max position."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    step_size = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Subgradient at argmax
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            v1 = a[k_star - i] if 0 <= k_star - i < n else 0.0
            v2 = a[i - k_star] if 0 <= i - k_star < n else 0.0
            grad[i] = v1 + v2
        
        # Decaying step size
        step_size = step_size * 0.9995 + 0.01 / (1.0 + 0.01 * it)
        a_new = project_simplex(a - step_size * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
        else:
            step_size *= 0.7
            a_new = project_simplex(a - step_size * grad, S)
            a = a_new
        
        # Perturbation
        if it % 500 == 0 and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())

                                   beta_init=20.0, beta_rate=0.002, beta_max=10000.0, k_frac=0.04)
        obj6 = 2.0 * n_sec * m_opt6 / (a_opt6.sum() ** 2)
        print("  Phase 6 (n=16384): obj=%.6f" % obj6)
        
        if obj6 < global_best:
            global_best = obj6
            global_a = a_opt6.copy()
            print("  *** New global best: %.6f ***" % global_best)
    
    # Final verification
    print("\n=== Final Result ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
    
    print("Returning sequence of length %d, final obj=%.6f, time=%.0fs" % 
          (len(global_a), global_best, time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
