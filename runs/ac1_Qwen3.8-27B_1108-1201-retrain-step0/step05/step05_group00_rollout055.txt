"""
Minimize 2*n*max_conv/sum^2 using:
1. Projected L-BFGS on LSE-smoothed peak (main optimizer)
2. Nesterov-accelerated peak gradient descent (fast polishing)
3. LP-based direction finding (periodic restarts)
4. Multi-n exploration (4096, 8192, 16384)
5. Spectral phase randomization for escaping local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv of length 2n-1."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def topk_subgrad(a, N, top_k):
    """Subgradient of sum of top-K convolution peaks."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    return grad, c, c_max


def nesterov_peak_block(a_init, n, S, N, max_time, rng, top_k=1):
    """Nesterov-accelerated peak gradient descent."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    y = a.copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    t_schema = 1.0
    g_scale = 2.0 * S / n
    start = time.monotonic()
    it = 0
    step_mult = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c_y = compute_conv(y, N)
        cur_max = c_y.max()
        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = y.copy()

        if top_k <= 1:
            k_star = int(np.argmax(c_y))
            grad = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                grad[j] = 2.0 * y[k_star - j]
        else:
            top_idx = np.argsort(c_y)[::-1][:top_k]
            grad = np.zeros(n)
            for k in top_idx:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += 2.0 * y[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = g_scale * step_mult / (1.0 + 0.002 * it)
        x_new = project_simplex(y - step * grad / gnorm, S)

        if it % 50 == 0:
            c_x = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_schema * t_schema))
        y = x_new + ((t_schema - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t_schema = t_new

        if cur_max > best_cmax * 1.01:
            step_mult = max(0.1, step_mult * 0.7)
        elif (it + 1) % 100 == 0:
            step_mult = min(3.0, step_mult * 1.05)

def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    s_hist = []
    y_hist = []
    g, c, c_max = lse_grad(a, N, beta)
    step_scale = 3.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 3.0 * S / (n * max(beta, 1.0))
            g, c, c_max = lse_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if it % 50 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a.copy()
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1.0 + 1e-10) or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, c_nw, cmax_nw = lse_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-13 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs + 2:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 300 == 0 and rng.random() < 0.3:
            a = spectral_partial_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            g, c, c_max = lse_grad(a, N, beta)

    return best_a, best_cmax


def spectral_partial_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of the FFT phases."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    rand_idx = rng.choice(Nf, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_full_restart(a, N, rng):
    """Full spectral phase randomization."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, Nf)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def lp_direction(f, n, S, N, time_limit=30.0):
    """
    Solve LP: min t s.t. sum(b)=S, b>=0, (f*b)[k] <= t for all k.
    Returns b (the optimal solution) or None if LP fails/is too slow.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    t_start = time.monotonic()
    A_rows = []
    A_cols = []
    A_data = []

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_data.append(f[j])
        A_rows.append(k)
        A_cols.append(n)
        A_data.append(-1.0)

    A = csr_matrix((A_data, (A_rows, A_cols)), shape=(2 * n - 1, n + 1))

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    b_ub = np.full(2 * n - 1, np.inf)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0, 1000.0)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit, 'presolve': True})
        if time.monotonic() - t_start > time_limit * 1.5:
            return None
        if res.success:
            b = res.x[:n]
            s = b.sum()
            if s < 1e-15:
                return None
            b = b * (S / s)
            b = np.maximum(b, 0.0)
            return b
    except Exception:
        pass
    return None


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 47]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    P = rng.exponential(1.0, Nf)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, Nf)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_deflation(n, S, N, n_iters=1000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.03 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def optimize_one_n(n, S, N, budget_s, rng, start_a=None):
    """Full optimization pipeline for a single n."""
    start_time = time.monotonic()
    deadline = start_time + budget_s

    if start_a is not None:
        a = np.asarray(start_a, dtype=np.float64).copy()
        a = np.clip(a, 0, 1000.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
    else:
        a = np.ones(n) * S / n

    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    print("  [n=%d] initial peak=%.6f, score=%.6f" % (n, best_cmax, 2.0 * n * best_cmax / (S * S)))

    def score(cmax):
        return 2.0 * n * cmax / (S * S)

    def try_update(candidate, label=""):
        nonlocal best_a, best_cmax
        c = compute_conv(candidate, N)
        cm = c.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = candidate.copy()
            if label:
                print("    [%s] peak=%.6f score=%.6f" % (label, cm, score(cm)))
        return cm

    # Phase A: L-BFGS on LSE (multi-init if no start)
    if start_a is None:
        inits = [
            ("rand", make_init_rand(n, rng)),
            ("chirp", make_init_chirp(n, rng)),
            ("zadoff", make_init_zadoff(n, rng)),
            ("multifreq", make_init_multi_freq(n, rng)),
            ("deflation", make_init_deflation(n, S, N, 500, rng)),
        ]
        per_init = max(5.0, min(30.0, (budget_s * 0.15) / len(inits)))
        for name, arr in inits:
            if time.monotonic() > deadline - budget_s * 0.5:
                break
            a_cand = project_simplex(np.clip(arr, 0, 1000.0), S)
            a_l, sc_l = lbfgs_lse_block(a_cand, n, S, N, per_init, rng, beta_init=5.0, beta_max=80.0, m_lbfgs=8)
            try_update(a_l, name)

    # Phase B: Main L-BFGS with periodic Nesterov and LP
    print("  [n=%d] Phase B: main optimization" % n)
    stall = 0
    lp_count = 0
    block_idx = 0

    while time.monotonic() < deadline - budget_s * 0.25:
        block_time = min(60.0, (deadline - budget_s * 0.25) - time.monotonic())
        if block_time < 10:
            break
        block_idx += 1

        beta_start = 10.0 + 5.0 * min(stall, 8)
        beta_end = min(500.0, 50.0 * (1 + 0.5 * stall))
        a_new, sc_new = lbfgs_lse_block(best_a, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end, m_lbfgs=12)
        if sc_new < best_cmax - 1e-10:
            best_cmax = sc_new
            best_a = a_new.copy()
            stall = 0
            print("    [B%d] LBFGS peak=%.6f score=%.6f" % (block_idx, sc_new, score(sc_new)))
        else:
            best_a = a_new.copy() if a_new is not None else best_a
            stall += 1

        # Periodic Nesterov polishing
        if time.monotonic() < deadline - budget_s * 0.2:
            nest_time = min(30.0, (deadline - budget_s * 0.2) - time.monotonic())
            if nest_time > 5:
                a_ng, sc_ng = nesterov_peak_block(best_a, n, S, N, nest_time, rng, top_k=1)
                try_update(a_ng, "nesterov")
                if stall >= 2:
                    top_k2 = min(10, 2 + stall)
                    a_ng2, sc_ng2 = nesterov_peak_block(best_a, n, S, N, min(nest_time, 20.0), rng, top_k=top_k2)
                    try_update(a_ng2, "topk%d" % top_k2)

        # Periodic LP restart
        if stall >= 3 and lp_count < 8 and time.monotonic() < deadline - budget_s * 0.3:
            lp_count += 1
            print("    [B%d] LP step (stall=%d)" % (block_idx, stall))
            b_lp = lp_direction(best_a, n, S, N, time_limit=20.0)
            if b_lp is not None:
                a_blend = 0.5 * best_a + 0.5 * b_lp
                a_blend = project_simplex(a_blend, S)
                try_update(a_blend, "LP")
                if b_lp.sum() > 0.01:
                    try_update(b_lp, "LP_pure")
            stall = 0

        # Spectral restart if stuck
        if stall >= 5:
            best_a = spectral_full_restart(best_a, N, rng)
            stall = 0
            best_cmax = compute_conv(best_a, N).max()

    # Phase C: Final polishing with high beta
    remaining = deadline - time.monotonic() - 5
    if remaining > 20:
        print("  [n=%d] Phase C: final polishing (%.0fs)" % (n, remaining))
        a_fin, sc_fin = lbfgs_lse_block(best_a, n, S, N, min(remaining * 0.5, 120.0), rng,
                                         beta_init=50.0, beta_max=800.0, m_lbfgs=15)
        try_update(a_fin, "final_LBFGS")

        remaining2 = deadline - time.monotonic() - 5
        if remaining2 > 15:
            a_ng, sc_ng = nesterov_peak_block(best_a, n, S, N, min(remaining2 * 0.5, 60.0), rng, top_k=3)
            try_update(a_ng, "final_nest")

    print("  [n=%d] Final: peak=%.6f score=%.6f" % (n, best_cmax, score(best_cmax)))
    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-n exploration, L-BFGS on LSE, Nesterov peak gradient,
    and LP-based restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Try to use the given sequence
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024:
            given_seq = given
            print("Found given sequence of length %d" % given.size)
    except Exception:
        pass

    # Determine which n values to try
    n_candidates = []
    if given_seq is not None:
        n_candidates.append(len(given_seq))
    for n_try in [8192, 4096, 16384]:
        if n_try not in n_candidates and n_try <= 16384:
            n_candidates.append(n_try)

    # Limit to at most 3 n values based on budget
    if budget_s < 1200:
        n_candidates = n_candidates[:2]
    else:
        n_candidates = n_candidates[:3]

    print("Will try n values: %s, budget=%.0fs" % (n_candidates, budget_s))

    global_best = np.inf
    global_a = None
    global_n = 0

    for n in n_candidates:
        if time.monotonic() > deadline - 200:
            break

        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()

        # Determine time budget for this n
        if len(n_candidates) == 1:
            n_budget = deadline - time.monotonic()
        elif n == n_candidates[0]:
            n_budget = min(deadline - time.monotonic(), budget_s * 0.5)
        else:
            n_budget = min(deadline - time.monotonic(), budget_s * 0.25)

        if n_budget < 100:
            break

        print("\n=== Optimizing n=%d (budget=%.0fs) ===" % (n, n_budget))

        # Use given sequence if it matches this n
        start_a = None
        if given_seq is not None and len(given_seq) == n:
            start_a = np.clip(given_seq, 0, 1000.0)

        a_res, cm_res = optimize_one_n(n, S, N, n_budget, rng, start_a=start_a)

        if cm_res < global_best:
            global_best = cm_res
            global_a = a_res.copy()
            global_n = n

    # Final verification and return
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("  Rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning length %d, score=%.6f" % (len(global_a), ver if np.isfinite(ver) else float('inf')))
    return final_list
