    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        result += diff / n
        result = np.maximum(result, 0.0)
    return result


def lse_gradient(a, c, beta):
    """Gradient of log-sum-exp(beta * c) with respect to a.
    Returns gradient of the LSE-smoothed max of c."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of sum_k w_k * (a*a)[k] w.r.t. a
    # d/d a_i [(a*a)[k]] = a[k-i] + a[i-k] (when valid)
    # Use FFT: grad = 2 * ifft(fft(w) * conj(fft(a)))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(a, c, k):
    """Standard subgradient using top-k highest convolution values."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    # Weights proportional to excess over median
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def nesterov_lse(a0, S, time_budget, rng, beta_init=20.0, beta_max=20000.0,
                  momentum_init=0.9, verbose=False, tag=""):
    """
    Nesterov accelerated gradient descent with LSE smoothing.
    Minimizes max(conv(a,a)) subject to sum(a)=S, a>=0.
    """
    n = len(a0)
    y = project_simplex(a0, S).copy()
    v = np.zeros(n)
    
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()

    start = time.monotonic()
    it = 0
    stagnation = 0
    patience = 200

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive beta: increase over time
        beta = min(beta_max, beta_init * np.exp(0.005 * it))

        # Nesterov lookahead
        c = fft_convolve_self(y)
        grad_y = lse_gradient(y, c, beta)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-15:
            break

        # Step size: ~ 1/(beta * ||grad||) with Nesterov schedule
        step = 1.0 / (1.5 * beta)
        step = min(step, 3.0 / (gn + 1e-15))

        # Nesterov momentum
        t_it = (1 + np.sqrt(1 + 4 * momentum_init * momentum_init)) / 2
        # Simple momentum (not full Nesterov for stability with projection)
        v = momentum_init * v - step * grad_y
        y_try = project_simplex(y + v, S)

        # Evaluate
        c_try = fft_convolve_self(y_try)
        new_max = c_try.max()

        if new_max < best - 1e-12:
            best = new_max
            best_a = y_try.copy()
            stagnation = 0
        else:
            stagnation += 1

        # If stuck, reduce momentum and add noise
        if stagnation > patience:
            # Simulated annealing: large perturbation
            noise_scale = 0.05 * S / np.sqrt(n) * np.exp(-0.5 * progress)
            y_try = project_simplex(y + rng.normal(0, noise_scale, n), S)
            c_try = fft_convolve_self(y_try)
            if c_try.max() < best:
                best = c_try.max()
                best_a = y_try.copy()
            y = y_try
            v = np.zeros(n)
            stagnation = 0
            patience = int(patience * 1.3)
        elif stagnation > patience // 2:
            # Reduce momentum
            momentum_init = max(0.5, momentum_init * 0.99)
            v = v * 0.5

        y = y_try

        if verbose and it % 500 == 0:
            obj = 2.0 * n * best / (S * S)
            print(f"  [{tag}] it={it}, t={elapsed:.0f}s, beta={beta:.0f}, "
                  f"best_max={best:.6f}, obj={obj:.6f}")

    return best_a, best


def subgradient_optimize(a0, S, time_budget, rng, verbose=False, tag=""):
    """
    Subgradient optimization with top-k weighting and adaptive step size.
    Used as a complementary method to LSE.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stagnation = 0
    patience = 300

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adaptive top-k
        k = int(4 + 60 * progress)
        grad = subgradient_topk(a, c, k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        # Step size with decreasing schedule
        step = 2.0 * (cur_max - c.mean()) / (gn * gn + 1e-15)
        step = min(step, 5.0 / gn)

        # Line search
        accepted = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            step = max(0.001 / gn, 1e-10)
            a = project_simplex(a - step * grad, S)

        # Noise when stuck
        if stagnation > patience:
            noise_scale = 0.03 * S / np.sqrt(n) * (1 - 0.5 * progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stagnation = 0
            patience = int(patience * 1.2)

        if verbose and it % 1000 == 0:
            obj = 2.0 * n * best / (S * S)
            print(f"  [{tag}] it={it}, t={elapsed:.0f}s, best={best:.6f}, obj={obj:.6f}")

    return best_a, best


def geometric_init(n, c_val, rng=None):
    """
    Generate a geometric-like initialization: a[i] = r^i where r = 1 + c/n.
    Optionally add small noise.
    Also returns the reversed version implicitly handled by caller.
    """
    r = 1.0 + c_val / n
    a = np.power(r, np.arange(n, dtype=np.float64))
    if rng is not None:
        noise = rng.normal(0, 0.01, n)
        a = a * (1.0 + noise)
        a = np.maximum(a, 0.0)
    return a


def windowed_geometric_init(n, c_val, window_power=2.0, rng=None):
    """
    Geometric sequence modulated by a raised-cosine window.
    a[i] = r^i * (0.5 + 0.5*cos(pi*(2*i/(n-1) - 1)))^window_power
    """
    r = 1.0 + c_val / n
    i = np.arange(n, dtype=np.float64)
    geo = np.power(r, i)
    # Raised cosine window
    if n > 1:
        t = 2.0 * i / (n - 1) - 1.0
        window = np.power(0.5 + 0.5 * np.cos(np.pi * t), window_power)
    else:
        window = np.ones(1)
    a = geo * window
    if rng is not None:
        a = a * (1.0 + rng.normal(0, 0.01, n))
        a = np.maximum(a, 0.0)
    return a


def parabolic_init(n, rng=None):
    """Parabolic bump: a[i] = (1 - (2i/(n-1) - 1)^2)."""
    if n > 1:
        t = 2.0 * np.arange(n) / (n - 1) - 1.0
        a = np.maximum(1.0 - t * t, 0.0)
    else:
        a = np.ones(1)
    if rng is not None:
        a = a * (1.0 + rng.normal(0, 0.01, n))
        a = np.maximum(a, 0.0)
    return a


def uniform_init(n):
    return np.ones(n)


def interpolate_to_length(a, new_n):
    """Interpolate sequence a to new length new_n."""
    if len(a) == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Load provided sequence if available
    2. Generate diverse initializations (geometric, windowed, parabolic)
    3. For each, run LSE+Nesterov optimization
    4. Track global best, refine with subgradient and more LSE passes
    5. Final polish with high-beta LSE
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s, deadline in ~{budget_s-30}s")

    # Try to load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Loaded provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    # Determine target length
    target_n = 8192
    if a_provided is not None and len(a_provided) >= 2048:
        target_n = len(a_provided)

    S = np.sqrt(2.0 * target_n)
    global_best = np.inf
    global_a = None

    def try_update(a, label=""):
        nonlocal global_best, global_a
        if a is None:
            return
        a = np.maximum(a, 0.0)
        S_local = a.sum()
        if S_local < 0.01:
            return
        obj = 2.0 * len(a) * np.max(fft_convolve_self(a)) / (S_local * S_local)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.6f} (n={len(a)}) [{label}]")

    # === Phase 1: Optimize provided sequence (if available) ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())
        a1 = project_simplex(a1, S1)

        t1 = min(500, (deadline - time.monotonic()) * 0.20)
        if t1 > 30:
            print(f"\nPhase 1a: LSE+Nesterov on provided, n={n1}, t={t1:.0f}s")
            a_opt, _ = nesterov_lse(a1, S1, t1, rng, beta_init=50,
                                    beta_max=10000, tag="P1a")
            try_update(a_opt, "Phase 1a LSE")

            # Follow up with subgradient
            remaining = deadline - time.monotonic()
            t1b = min(200, remaining * 0.10)
            if t1b > 20:
                print(f"Phase 1b: Subgradient on provided, t={t1b:.0f}s")
                a_opt2, _ = subgradient_optimize(a_opt, S1, t1b, rng, tag="P1b")
                try_update(a_opt2, "Phase 1b subgrad")

    # === Phase 2: Geometric initializations (key insight!) ===
    # The asymptotic analysis suggests r = 1 + c/n with c ~ 1 is optimal
    c_values = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    
    if target_n != 8192:
        # If we're using a different length, adjust
        pass

    n2 = target_n
    S2 = np.sqrt(2.0 * n2)
    
    # Allocate time for geometric inits
    n_geo = len(c_values) + 2  # +2 for reversed and windowed
    remaining = deadline - time.monotonic()
    t_per_geo = min(150, remaining * 0.35 / n_geo)
    
    print(f"\nPhase 2: Geometric inits, n={n2}, {n_geo} trials, ~{t_per_geo:.0f}s each")

    geo_results = []
    for c_val in c_values:
        if time.monotonic() > deadline - 60:
            break
        
        a_init = geometric_init(n2, c_val, rng)
        a_init = project_simplex(a_init, S2)
        # Try both orientations
        for orient in range(2):
            if orient == 1:
                a_init_c = a_init[::-1].copy()
            else:
                a_init_c = a_init.copy()
            
            a_opt, _ = nesterov_lse(a_init_c, S2, t_per_geo, rng,
                                    beta_init=10, beta_max=5000,
                                    momentum_init=0.95, tag=f"geo_c{c_val}_o{orient}")
            try_update(a_opt, f"geo c={c_val} orient={orient}")
            geo_results.append(a_opt)

    # Windowed geometric
    if time.monotonic() < deadline - 60:
        a_init = windowed_geometric_init(n2, 1.0, window_power=1.0, rng)
        a_init = project_simplex(a_init, S2)
        a_opt, _ = nesterov_lse(a_init, S2, t_per_geo, rng,
                                beta_init=10, beta_max=5000,
                                momentum_init=0.95, tag="wgeo")
        try_update(a_opt, "windowed geo")
        geo_results.append(a_opt)

    # === Phase 3: Refine best from Phase 2 ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 100:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        
        # Alternate LSE and subgradient
        t3a = min(300, remaining * 0.30)
        t3b = min(200, remaining * 0.20)
        
        print(f"\nPhase 3: Refine best, n={n3}")
        
        a_curr = global_a.copy()
        a_curr, _ = nesterov_lse(a_curr, S3, t3a, rng,
                                 beta_init=100, beta_max=20000,
                                 momentum_init=0.98, tag="P3a")
        try_update(a_curr, "Phase 3a LSE refine")

        if time.monotonic() < deadline - 30:
            a_curr, _ = subgradient_optimize(a_curr, S3, t3b, rng, tag="P3b")
            try_update(a_curr, "Phase 3b subgrad")

    # === Phase 4: Coarse-to-fine (optimize smaller n, interpolate up) ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        n_small = max(512, target_n // 16)
        S_small = np.sqrt(2.0 * n_small)
        
        print(f"\nPhase 4: Coarse-to-fine, n_small={n_small}")
        
        # Optimize for small n with geometric init
        a_small = geometric_init(n_small, 1.0, rng)
        a_small = project_simplex(a_small, S_small)
        t_small = min(100, remaining * 0.15)
        a_small_opt, _ = nesterov_lse(a_small, S_small, t_small, rng,
                                       beta_init=20, beta_max=5000, tag="P4_small")
        
        # Interpolate to target length
        a_interp = interpolate_to_length(a_small_opt, target_n)
        a_interp = project_simplex(a_interp, S)
        
        t_fine = min(200, remaining * 0.25)
        a_fine, _ = nesterov_lse(a_interp, S, t_fine, rng,
                                  beta_init=50, beta_max=10000, tag="P4_fine")
        try_update(a_fine, "Phase 4 coarse-to-fine")

    # === Phase 5: Random restarts around global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        n_restarts = 3
        t_per_restart = (remaining - 20) / n_restarts
        
        print(f"\nPhase 5: {n_restarts} perturbations, ~{t_per_restart:.0f}s each")
        
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Different perturbation scales
            scales = [0.03, 0.05, 0.08]
            noise_scale = scales[ri % len(scales)] * S5 / np.sqrt(n5)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n5), S5)
            a_pert, _ = nesterov_lse(a_pert, S5, t_per_restart, rng,
                                      beta_init=50, beta_max=8000,
                                      tag=f"P5_r{ri}")
            try_update(a_pert, f"Phase 5 restart {ri}")

    # === Phase 6: Final high-beta LSE polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n6 = len(global_a)
        S6 = np.sqrt(2.0 * n6)
        t6 = remaining - 10
        print(f"\nPhase 6: Final LSE polish, t={t6:.0f}s")
        a_final, _ = nesterov_lse(global_a, S6, t6, rng,
                                   beta_init=500, beta_max=50000,
                                   momentum_init=0.99, tag="P6")
        try_update(a_final, "Phase 6 final polish")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No solution found, using uniform fallback")
        global_a = np.ones(1024)
        global_best = compute_obj(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024)

    # Scale to a "nice" normalization (doesn't change objective)
    S_final = global_a.sum()
    # Keep as is - objective is scale-invariant

    final_list = [float(x) for x in global_a]
    
    # Verify with the actual evaluation function
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    
    print(f"\n{'='*60}")
    print(f"FINAL: n={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence = {ver:.6f}")
    print(f"{'='*60}")
    
    # If the evaluation differs from our internal computation, use the valid one
    if np.isfinite(ver) and ver < global_best:
        print(f"Note: evaluation gives better value {ver:.6f}")
    
    return final_list
