import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        result += diff / n
        result = np.maximum(result, 0.0)
    return result


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def exact_subgrad(a, conv, top_k=1):
    """Exact subgradient of max(conv) at point a, averaged over top_k argmax positions."""
    n = len(a)
    k = min(top_k, len(conv))
    idx = np.argsort(conv)[::-1][:k]
    g = np.zeros(n)
    for p in idx:
        p = int(p)
        # g[m] += 2 * a[p-m] for 0 <= p-m < n
        end = min(p, n - 1) + 1
        g[p - end + 1: p + 1] += 2.0 * a[:end][::-1]
    g /= float(k)
    return g


def lse_subgrad(a, conv, k, beta):
    """Log-sum-exp smoothed gradient of top-k of conv."""
    n = len(a)
    k = min(k, len(conv))
    idx = np.argsort(conv)[::-1][:k]
    c_top = conv[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(conv))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_val(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fft_convolve_self(a)
    return float(2.0 * n * conv.max() / (s * s))


def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Random phase perturbation
    phase = rng.normal(0, amount, fsize)
    phase[0] = 0
    A_new = A * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def phase_noise_perturb(a, amount, rng):
    """Perturb by adding phase noise in frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Multiply by random phase
    angles = rng.uniform(-amount, amount, fsize)
    A_new = A * np.exp(1j * angles)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:

    return a_new * (S / s)
