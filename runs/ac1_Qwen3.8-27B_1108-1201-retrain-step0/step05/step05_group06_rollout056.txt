def convolution_matrix(n, f):
    """Build the (2n-1) x n matrix A where (A @ g)_j = (f*g)_j."""
    rows = []
    cols = []
    vals = []
    for j in range(2*n - 1):
        i_min = max(0, j - (n-1))
        i_max = min(n-1, j)
        for i in range(i_min, i_max + 1):
            rows.append(j)
            cols.append(i)
            vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

conv(f_new, f_new) = (1-t)^2 conv(f,f) + 2t(1-t) conv(f,g*) + t^2 conv(g*,g*)

The first term is <= (1-t)^2 * M (since max conv(f,f) = M)
The second term is <= 2t(1-t) * M (since conv(f,g*) <= M)
The third term is t^2 * max conv(g*,g*)

= M(1-t)^2 + 2t(1-t)M + t^2 max conv(g*,g*)
= M(1 - 2t + t^2 + 2t - 2t^2) + t^2 max conv(g*,g*)
= M(1 - t^2) + t^2 max conv(g*,g*)
= M - t^2(M - max conv(g*,g*))

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LP-based direction finding (conditional gradient / cutting plane)
combined with LSE FISTA polishing and active-set refinement.
"""
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time
import sys

def fft_convolve_full(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = ifft(A * A)
    return np.real(c[:size])

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_full(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_subgradient(y, c, k):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def build_conv_matrix(n, f):
    """Build sparse matrix A where (A @ x)_j = (f * x)_j for full convolution."""
    k = np.arange(n * n, dtype=np.int64)
    cols = (k // n).astype(np.int32)
    rows = ((k // n) + (k % n)).astype(np.int32)
    vals = np.repeat(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_solve_direction(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'max_iter': 10000})
    dt = time.monotonic() - t0
    
    if res.success:
        if verbose:
            print("    LP solved in %.2fs, sum(g)=%.4f" % (dt, -res.fun))
        return res.x
    else:
        if verbose:
            print("    LP failed: %s" % res.message)
        return None

def run_lp_phase(f_init, budget_s, rng, n_lp=None, verbose=True):
    """LP-based direction finding phase."""
    if n_lp is None:
        n_lp = len(f_init)
    
    # Downsample to n_lp if needed
    if len(f_init) > n_lp:
        t_old = np.linspace(0, 1, len(f_init), endpoint=False)
        t_new = np.linspace(0, 1, n_lp, endpoint=False)
        f = np.interp(t_new, t_old, f_init)
    else:
        f = f_init.copy()[:n_lp]
    
    S = np.sqrt(2.0 * n_lp)
    f = normalize_seq(f, S)
    
    start = time.monotonic()
    best_obj = compute_obj(f)
    best_f = f.copy()
    it = 0
    
    if verbose:
        print("  LP Phase: n=%d, initial obj=%.6f" % (n_lp, best_obj))
    
    while time.monotonic() - start < budget_s:
        it += 1
        c = fft_convolve_full(f)
        M = c.max()
        
        # Solve LP
        g = lp_solve_direction(f, M, verbose=(it % 10 == 1))
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Normalize g to have sum S
        g = g * (S / g_sum)
        
        # Update: f_new = (1-t)*f + t*g
        # Try a few step sizes and pick the best
        best_t = 0.1
        best_c_max = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            f_try = (1-t) * f + t * g
            f_try = np.maximum(f_try, 0)
            c_try = fft_convolve_full(f_try)
            if c_try.max() < best_c_max:
                best_c_max = c_try.max()
                best_t = t
        
        f = (1-best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = normalize_seq(f, S)
        
        obj = compute_obj(f)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
            if verbose and it % 5 == 0:
                print("    it=%d, obj=%.6f" % (it, best_obj))
    
    if verbose:
        print("  LP Phase done: %d iterations, best obj=%.6f" % (it, best_obj))
    return best_f

def run_fista_lse(f_init, budget_s, rng, beta_start=10, beta_max=50000):
    """FISTA with LSE approximation."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(f_init, 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start
    
    while time.monotonic() - start < budget_s:
        it += 1
        c = fft_convolve_full(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / S**2
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta
        if stuck > 1000:
            beta = min(beta_max, beta * 1.05)
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = 0.5 / (beta * cur_max / n + 1.0)
        step = max(step, 1e-12)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if stuck > 5000:
            t_mom = 1.0
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            last_improve = it
        
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LP-based active-set direction finding (conditional gradient approach),
combined with FISTA LSE polishing and exact subgradient refinement.
Key innovation: reduced LP with adaptive active constraints for fast
global-direction updates that escape local minima of gradient methods.
"""
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def fft_convolve_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = ifft(A * A)
    return np.real(c[:size])


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_full(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k):
    """Top-k subgradient with uniform weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_full_conv_matrix(n, f):
    """Build sparse (2n-1) x n matrix for full convolution with f."""
    k = np.arange(n * n, dtype=np.int64)
    cols = (k // n).astype(np.int32)
    rows = ((k // n) + (k % n)).astype(np.int32)
    vals = np.repeat(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve_full(f, M):
    """Solve full LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_full_conv_matrix(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'max_iter': 50000})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_reduced(f, M, active_lags):
    """Solve reduced LP with only active lag constraints."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return np.ones(n) / n

    rows_list = []
    cols_list = []
    vals_list = []

    for k, j in enumerate(active_lags):
        i_min = max(0, j - (n - 1))
        i_max = min(n - 1, j)
        count = i_max - i_min + 1
        if count > 0:
            rows_list.append(np.full(count, k, dtype=np.int32))
            cols_list.append(np.arange(i_min, i_max + 1, dtype=np.int32))
            vals_list.append(f[i_min:i_max + 1])

    if not rows_list:
        return np.ones(n) / n

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)

    A = csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'max_iter': 50000})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def get_active_lags(c, K):
    """Get indices of top-K convolution values."""
    K = min(K, len(c))
    return np.argsort(c)[::-1][:K].tolist()

def run_fista_lse(f_init, budget_s, rng, beta_start=10.0, beta_max=50000.0):
    """FISTA with LSE approximation."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(f_init, 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start

    while time.monotonic() - start < budget_s:
        it += 1
        c = fft_convolve_full(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / (S * S)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 500:
            beta = min(beta_max, beta * 1.03)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / max(beta * cur_max / n + 1.0, 1.0)
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000:
            t_mom = 1.0
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)

def upsample_seq(a, factor):
    """Upsample using linear interpolation."""
    n = len(a)
    if factor <= 1:
        return a.copy()
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample_seq(a, n_target):
    """Downsample via interpolation."""
    n = len(a)
    if n <= n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_coordinate_descent(f_init, budget_s, rng):
    """Coordinate descent with smart moves."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = normalize_seq(np.clip(f_init, 0, 1000), S)
    start = time.monotonic()
    best_obj = compute_obj(f)
    best_f = f.copy()
    it = 0
    last_improve = 0

    c = fft_convolve_full(f)
    M = c.max()

    while time.monotonic() - start < budget_s:
        it += 1
        # Pick coordinate to update: weighted by contribution to max
        # Find positions that contribute most to the max lag
        max_lag = np.argmax(c)
        # Gradient for max lag: grad_i = 2 * f[max_lag - i] (for valid i)
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        valid = np.arange(i_min, i_max + 1)
        grad_at_valid = 2.0 * f[max_lag - valid]

        # Pick the coordinate with highest reduction potential
        # If we decrease f[i], the max lag decreases by ~f[i] * f[max_lag-i]
        # But we need to maintain the sum, so we increase some other coordinate
        # Simple approach: move mass from high-contributing to low-contributing
        if len(valid) > 0:
            target_idx = valid[np.argmax(grad_at_valid)]
            # Find a safe index to move mass to (one with low contribution)
            all_contrib = np.zeros(n)
            all_contrib[valid] = grad_at_valid
            # Move from target to a position with low contribution
            candidates = np.where(all_contrib < np.median(all_contrib))[0]
            if len(candidates) > 0:
                source_idx = candidates[rng.integers(len(candidates))]
            else:
                source_idx = rng.integers(n)

            if source_idx == target_idx:
                source_idx = (source_idx + 1) % n

            # Try moving a small amount
            delta = f[target_idx] * 0.05
            f_try = f.copy()
            f_try[target_idx] -= delta
            f_try[source_idx] += delta
            f_try = np.maximum(f_try, 0)
            f_try = normalize_seq(f_try, S)

            c_try = fft_convolve_full(f_try)
            obj_try = 2.0 * n * c_try.max() / (f_try.sum() ** 2)

            if obj_try < best_obj:
                f = f_try
                c = c_try
                best_obj = obj_try
                best_f = f.copy()
                last_improve = it
            else:
                # Try random perturbation
                idx1 = rng.integers(n)
                idx2 = rng.integers(n)
                if idx1 != idx2:
                    delta2 = f[idx1] * 0.02
                    f_try2 = f.copy()
                    f_try2[idx1] -= delta2
                    f_try2[idx2] += delta2
                    f_try2 = np.maximum(f_try2, 0)
                    f_try2 = normalize_seq(f_try2, S)
                    c_try2 = fft_convolve_full(f_try2)
                    obj_try2 = 2.0 * n * c_try2.max() / (f_try2.sum() ** 2)
                    if obj_try2 < best_obj:
                        f = f_try2
                        c = c_try2
                        best_obj = obj_try2
                        best_f = f.copy()
                        last_improve = it

        stuck = it - last_improve
        if stuck > 200:
            # Inject noise
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            f = project_simplex(f + noise, S)
            c = fft_convolve_full(f)
            last_improve = it
            stuck = 0

        if it % 500 == 0:
            print("    CD it=%d, obj=%.6f" % (it, best_obj))

    return best_f, best_obj


def propose_candidate(seed=42, max_time=2400):
    """
    Multi-phase optimization combining:
    1. Full LP at small n (global shape search)
    2. Reduced active-set LP at target n (direction finding)
    3. FISTA LSE (smooth polishing)
    4. Active-set exact subgradient (fine refinement)
    5. Coordinate descent (local search)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max_time - 30.0

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    global_best_obj = np.inf
    global_best_a = None
    target_n = 8192

    def try_update(a, label):
        nonlocal global_best_obj, global_best_a
        if a is None:
            return
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  ** NEW BEST (%s): obj=%.6f (n=%d) **" % (label, obj, len(a)))
        return obj

    # === Phase 1: Full LP at n=1024 ===
    print("=== Phase 1: Full LP (n=1024) ===")
    t_phase = time.monotonic()
    p1_time = min(120.0, (deadline - time.monotonic()) * 0.06)

    if p1_time > 10:
        n1 = 1024
        if a_start is not None:
            f1 = downsample_seq(a_start, n1)
        else:
            tt = np.linspace(0, 1, n1, endpoint=False)
            f1 = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
        f1 = normalize_seq(f1, np.sqrt(2.0 * n1))
        f1_result = run_lp_phase_full(f1, p1_time, n1)
        try_update(f1_result, "LP-n1024")

    # === Phase 1b: Full LP at n=2048 ===
    print("\n=== Phase 1b: Full LP (n=2048) ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    p1b_time = min(200.0, remaining * 0.10)

    if p1b_time > 15:
        n1b = 2048
        base = global_best_a if global_best_a is not None else f1_result
        if len(base) >= n1b:
            f1b = downsample_seq(base, n1b)
        else:
            f1b = upsample_seq(base, n1b // max(1, len(base)))
        f1b = normalize_seq(f1b, np.sqrt(2.0 * n1b))
        f1b_result = run_lp_phase_full(f1b, p1b_time, n1b)
        try_update(f1b_result, "LP-n2048")

    # === Phase 2: Reduced LP at target n ===
    print("\n=== Phase 2: Reduced active-set LP (n=%d) ===" % target_n)
    remaining = deadline - time.monotonic()
    p2_time = min(900.0, remaining * 0.40)

    if p2_time > 30:
        if a_start is not None and len(a_start) == target_n:
            f2 = a_start.copy()
        elif global_best_a is not None:
            if len(global_best_a) == target_n:
                f2 = global_best_a.copy()
            elif len(global_best_a) < target_n:
                f2 = upsample_seq(global_best_a, target_n // max(1, len(global_best_a)))
            else:
                f2 = downsample_seq(global_best_a, target_n)
        else:
            tt = np.linspace(0, 1, target_n, endpoint=False)
            f2 = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

        f2 = normalize_seq(f2, np.sqrt(2.0 * target_n))
        print("  Starting reduced LP, initial obj=%.6f" % compute_obj(f2))
        f2_result = run_lp_phase_reduced(f2, p2_time, rng, k_start=100, k_max=1000)
        try_update(f2_result, "Reduced-LP")

    # === Phase 3: FISTA LSE ===
    print("\n=== Phase 3: FISTA LSE ===")
    remaining = deadline - time.monotonic()
    p3_time = min(500.0, remaining * 0.25)

    if p3_time > 20 and global_best_a is not None:
        f3 = global_best_a.copy()
        if len(f3) != target_n:
            if len(f3) < target_n:
                f3 = upsample_seq(f3, target_n // max(1, len(f3)))
            else:
                f3 = downsample_seq(f3, target_n)
        f3 = normalize_seq(f3, np.sqrt(2.0 * target_n))

        # Run with medium beta
        f3a, obj3a = run_fista_lse(f3, p3_time * 0.5, rng,
                                    beta_start=10.0, beta_max=10000.0)
        try_update(f3a, "FISTA-mid")

        remaining = deadline - time.monotonic()
        if remaining > 20 and global_best_a is not None:
            # Run with high beta
            f3b = global_best_a.copy()
            if len(f3b) != target_n:
                if len(f3b) < target_n:
                    f3b = upsample_seq(f3b, target_n // max(1, len(f3b)))
                else:
                    f3b = downsample_seq(f3b, target_n)
            f3b = normalize_seq(f3b, np.sqrt(2.0 * target_n))
            f3b_res, obj3b = run_fista_lse(f3b, min(remaining * 0.8, 200.0), rng,
                                            beta_start=200.0, beta_max=50000.0)
            try_update(f3b_res, "FISTA-high")

    # === Phase 4: Active-set polishing ===
    print("\n=== Phase 4: Active-set polishing ===")
    remaining = deadline - time.monotonic()
    p4_time = min(400.0, remaining * 0.20)

    if p4_time > 20 and global_best_a is not None:
        f4 = global_best_a.copy()
        if len(f4) != target_n:
            if len(f4) < target_n:
                f4 = upsample_seq(f4, target_n // max(1, len(f4)))
            else:
                f4 = downsample_seq(f4, target_n)
        f4 = normalize_seq(f4, np.sqrt(2.0 * target_n))
        f4_result, obj4 = run_active_polish(f4, p4_time, rng)
        try_update(f4_result, "Active-polish")

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate descent ===")
    remaining = deadline - time.monotonic()
    p5_time = min(200.0, remaining * 0.20)

    if p5_time > 15 and global_best_a is not None:
        f5 = global_best_a.copy()
        if len(f5) != target_n:
            if len(f5) < target_n:
                f5 = upsample_seq(f5, target_n // max(1, len(f5)))
            else:
                f5 = downsample_seq(f5, target_n)
        f5 = normalize_seq(f5, np.sqrt(2.0 * target_n))
        f5_result, obj5 = run_coordinate_descent(f5, p5_time, rng)
        try_update(f5_result, "Coord-desc")

    # === Phase 6: Final short reduced LP burst ===
    print("\n=== Phase 6: Final LP burst ===")
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.5

    if p6_time > 20 and global_best_a is not None:
        f6 = global_best_a.copy()
        if len(f6) != target_n:
            if len(f6) < target_n:
                f6 = upsample_seq(f6, target_n // max(1, len(f6)))
            else:
                f6 = downsample_seq(f6, target_n)
        f6 = normalize_seq(f6, np.sqrt(2.0 * target_n))
        f6_result = run_lp_phase_reduced(f6, p6_time, rng, k_start=200, k_max=800)
        try_update(f6_result, "Final-LP")

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        print("\n=== FINAL RESULT ===")
        print("Objective: %.6f (n=%d)" % (global_best_obj, len(final_list)))
    else:
        n_fb = 8192
        final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb
        print("\n=== FALLBACK ===")
        print("Using uniform sequence, obj should be ~2.0")

    return final_list
