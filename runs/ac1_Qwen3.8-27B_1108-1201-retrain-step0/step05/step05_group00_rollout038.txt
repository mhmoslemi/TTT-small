    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_with_fft(a, A_cached, N):
    """Self-convolution using cached FFT of a."""
    n = len(a)
    c = np.real(ifft(A_cached * A_cached))[:2 * n - 1]
    return c


def peak_subgrad_fast(a, c, peak_idx, n):
    """Compute subgradient of c[peak_idx] with respect to a.
    Assumes c is already computed."""
    grad = np.zeros(n)
    k = int(peak_idx)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n:
            grad[i] += a[j]
            if i != j:
                grad[j] += a[i]
    return grad


def peak_subgrad_topk(a, c, n, top_k):
    """Compute weighted subgradient of top-k peaks."""
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * a[j]
                if i != j:
                    grad[j] += w * a[i]
    return grad


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases, project to non-negative, preserve sum."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    if frac >= 1.0:
        phase = rng.uniform(0, 2 * np.pi, N)
    else:
        phase = np.angle(A).copy()
        n_rand = max(1, int(N * frac))
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def nesterov_peak_descent(a_init, n, S, N, max_time, rng,
                           top_k=1, mu=0.9, step_init=None,
                           stall_restarts=300, div_frac=0.3):
    """
    Nesterov-accelerated direct peak subgradient descent.
    """
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    c = compute_conv(a, N)
    best_cmax = c.max()
    
    if step_init is None:
        step = S / (n * 0.5)
    
    y = a.copy()  # Nesterov look-ahead
    t = 1.0
    v = np.zeros(n)  # velocity
    
    start_time = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Compute gradient at y (Nesterov point)
        c_y = compute_conv(y, N)
        cmax_y = c_y.max()
        
        if top_k <= 1:
            peak_idx = int(np.argmax(c_y))
            grad = peak_subgrad_fast(y, c_y, peak_idx, n)
        else:
            grad = peak_subgrad_topk(y, c_y, n, top_k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        
        grad_unit = grad / gnorm
        
        # Nesterov step
        x_new = project_simplex(y - step * grad_unit, S)
        
        # Update momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        
        # Check improvement
        c_new = compute_conv(x_new, N)
        cmax_new = c_new.max()
        
        if cmax_new < best_cmax - 1e-10:
            best_cmax = cmax_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = x_new
        y = y_new
        t = t_new
        
        # Adaptive step size
        if stall > 50:
            step *= 0.95
        elif stall == 0:
            step = min(step * 1.01, S / (n * 0.1))
        
        # Periodic diversification
        if it % stall_restarts == 0:
            y = spectral_restart(y, N, rng, frac=div_frac)
            a = y.copy()
            t = 1.0
            stall = 0
        
    return best_a, best_cmax


def lse_smoothed_descent(a_init, n, S, N, max_time, rng, 
                          beta_init=10.0, beta_max=500.0,
                          m_lbfgs=10, div_every=200):
    """L-BFGS on LSE-smoothed peak."""
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    beta = beta_init
    best_a = a.copy()
    c = compute_conv(a, N)
    best_cmax = c.max()
    
    s_hist = []
    y_hist = []
    
    start_time = time.monotonic()
    it = 0
    
    def lse_grad(a, beta):
        n_local = len(a)
        c_local = compute_conv(a, N)
        cmax_local = c_local.max()
        exp_bc = np.exp(beta * (c_local - cmax_local))
        total = exp_bc.sum()
        weights = exp_bc / total
        fsize = 1 << (3 * n_local - 2).bit_length()
        W = fft(weights, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n_local]
        return grad, cmax_local
    
    def lbfgs_direction(s_h, y_h, g):
        m = len(s_h)
        n_local = len(g)
        gn = np.linalg.norm(g)
        if m == 0 or gn < 1e-18:
            if gn < 1e-18:
                return np.zeros(n_local)
            return -g / gn
        q = g.copy()
        alphas = np.zeros(m)
        for i in range(m - 1, -1, -1):
            s_i = s_h[i]
            y_i = y_h[i]
            yy = np.dot(y_i, y_i)
            if yy < 1e-30:
                alphas[i] = 0.0
            else:
                alphas[i] = np.dot(s_i, q) / yy
                q = q - alphas[i] * y_i
        s_last = s_h[-1]
        y_last = y_h[-1]
        yy_last = np.dot(y_last, y_last)
        if yy_last > 1e-30:
            gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
        else:
            gamma_use = 1.0
        r = gamma_use * q
        for i in range(m):
            s_i = s_h[i]
            y_i = y_h[i]
            yy = np.dot(y_i, y_i)
            if yy < 1e-30:
                continue
            beta_i = np.dot(s_i, y_i) / yy
            r = r + (alphas[i] - beta_i) * y_i
        direction = -r
        if np.dot(g, direction) > 0:
            direction = -g
        return direction
    
    g, c_cur = lse_grad(a, beta)
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.08)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
        
        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_direction(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_cur
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_cur or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_cur:
                    break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax - 1e-10:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, c_nw = lse_grad(a_new, beta)
        
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_cur = c_nw
        
        if it % div_every == 0 and rng.random() < 0.3:
            a = spectral_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            g, c_cur = lse_grad(a, beta)
    
    return best_a, best_cmax
