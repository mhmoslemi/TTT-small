Minimize 2*n*max_conv/sum^2 using variance-regularized FISTA, 
fast coordinate descent, and iterated local search.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj_and_conv(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(2*n-1)
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s), c

def lse_smooth_max(c, beta):
    """Compute LSE approximation to max(c) and the weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / tot
    return cm + np.log(tot) / beta, w

def objective_gradient(a, beta, lam_var):
    """
    Gradient of LSE(c, beta) + lam_var * mean(c^2) w.r.t. a.
    Both terms are differentiable (LSE is smooth approximation to max).
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # LSE gradient
    lse_val, w = lse_smooth_max(c, beta)
    
    # Variance (second moment) gradient
    if lam_var > 0:

        C = fft(c, fsize2)
        A = fft(a, fsize2)

    else:

    total_obj = lse_val + lam_var * np.mean(c**2)
    total_grad = lse_grad + lam_var * second_moment_grad
    
    return total_obj, total_grad, c

import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute LSE value and softmax weights for approximating max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / tot
    return cm + np.log(tot) / beta, w


def obj_and_grad(a, beta, lam_var):
    """
    Compute smooth objective LSE(c,beta) + lam_var*mean(c^2) and its gradient.
    Both terms have O(n log n) gradients via FFT.
    """
    n = len(a)
    c = fft_convolve_self(a)

    # LSE part
    lse_val, w = lse_weights(c, beta)
    # grad_LSE[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation of w and a)
    fsize1 = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize1)
    A1 = fft(a, fsize1)
    lse_grad = 2.0 * np.real(ifft(W * np.conj(A1)))[:n]

    # Variance (second moment) part: grad of mean(c^2)
    if lam_var > 1e-12:
        # grad[j] = 2/(2n-1) * sum_k c[k] * a[k-j]
        fsize2 = 1 << (3 * n - 2).bit_length()
        C = fft(c, fsize2)
        A2 = fft(a, fsize2)
        sm_grad = (2.0 / (2 * n - 1)) * np.real(ifft(C * np.conj(A2)))[:n]
    else:
        sm_grad = 0.0

    total_grad = lse_grad + lam_var * sm_grad
    total_val = lse_val + lam_var * np.mean(c ** 2)
    return total_val, total_grad, c

    return best_a, best_obj


def run_subgrad_polish(a_init, max_time, rng):
    """Polish with exact subgradient (top-1 to top-k lags) and diminishing step."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        # Adaptive k: increase when stuck
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = max(3, stuck // 200)
        else:
            k = min(64, 4 + stuck // 200)

        k = min(k, len(c))
        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k

        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Diminishing step with backtracking
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, 3.0 * S / gnorm)

        for _ in range(5):
            a_new = project_simplex(a - step * grad, S)
            obj_new = compute_obj(a_new)
            if obj_new <= cur_obj + 1e-12:
                a = a_new
                break
            step *= 0.4

        if stuck > 2000 and it % 50 == 0:
            ns = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 6000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def multi_scale_upsample(a, factor=2, rng=None):
    """Upsample with slight randomization to avoid symmetry traps."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_up = np.interp(t_new, t_old, a)
    if rng is not None:
        a_up = a_up * (1.0 + 0.01 * rng.standard_normal(n_new))
        a_up = np.maximum(a_up, 0.0)
    return a_up


def make_init(n, method, rng):
    """Create initial sequences with various profiles."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.12 ** 2))
    elif method == "multi_tone":
        a = (0.25 + 0.2 * np.sin(2 * np.pi * 3 * tt)
             + 0.15 * np.sin(2 * np.pi * 7 * tt + 1.1)
             + 0.1 * np.sin(2 * np.pi * 13 * tt + 2.3))
        a = np.abs(a) + 0.05
    elif method == "chirp":
        a = 0.4 + 0.4 * np.cos(2 * np.pi * (2 * tt + 3 * tt ** 2))
    elif method == "random_smooth":
        a = rng.random(n)
        # Smooth it
        k = max(1, n // 100)
        for _ in range(3):
            a = np.convolve(a, np.ones(k) / k, mode='same')
        a = a + 0.1
    elif method == "bump":
        a = np.maximum(0.05, 1.0 - 1.5 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining variance-regularized FISTA, fast coordinate
    descent, subgradient polish, and iterated local search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from parent
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Quick variance-regularized FISTA (15% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.15
    if p1_time > 30:
        print("\n=== Phase 1: Var-Reg FISTA (budget %.0fs) ===" % p1_time)
        a_opt, obj_opt = run_fista(global_best_a, p1_time, rng,
                                    beta_init=30, beta_rate=0.006,
                                    lam_var_init=0.1, lam_var_decay=1500,
                                    use_backtrack=True)
        print("  Phase 1: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 2: Intensive FISTA with multiple configs (30% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.30
    if p2_time > 60:
        print("\n=== Phase 2: Intensive FISTA (budget %.0fs) ===" % p2_time)
        configs = [
            (50, 0.004, 0.05, 3000, True),
            (200, 0.002, 0.02, 5000, True),
            (500, 0.001, 0.0, 8000, False),
            (1000, 0.0005, 0.0, 10000, True),
        ]
        per_time = p2_time / len(configs)
        for i, (bi, br, lv, ld, bt) in enumerate(configs):
            if time.monotonic() >= deadline:
                break
            rem = min(per_time * 1.2, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        lam_var_init=lv, lam_var_decay=ld,
                                        use_backtrack=bt)
            print("  FISTA config %d (beta0=%d): obj=%.8f" % (i + 1, bi, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Fast Coordinate Descent (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 30:
        print("\n=== Phase 3: Fast CD (budget %.0fs) ===" % p3_time)
        a_cd, obj_cd = fast_cd(global_best_a, p3_time * 0.6, rng, top_k=5, delta_frac=0.03)
        print("  CD: obj=%.8f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # Follow up with FISTA on CD result
        rem3 = deadline - time.monotonic()
        if rem3 > 15:
            a_cd2, obj_cd2 = run_fista(global_best_a, rem3 * 0.5, rng,
                                        beta_init=200, beta_rate=0.002,
                                        lam_var_init=0.0, lam_var_decay=99999,
                                        use_backtrack=True)
            print("  FISTA after CD: obj=%.8f" % obj_cd2)
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()

        # Another CD round
        rem3b = deadline - time.monotonic()
        if rem3b > 20:
            a_cd3, obj_cd3 = fast_cd(global_best_a, rem3b * 0.7, rng, top_k=8, delta_frac=0.015)
            print("  CD round 2: obj=%.8f" % obj_cd3)
            if obj_cd3 < global_best_obj:
                global_best_obj = obj_cd3
                global_best_a = a_cd3.copy()

    # === Phase 4: Subgradient Polish (10% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.10
    if p4_time > 15:
        print("\n=== Phase 4: Subgradient Polish (budget %.0fs) ===" % p4_time)
        a_pol, obj_pol = run_subgrad_polish(global_best_a, p4_time, rng)
        print("  Subgrad: obj=%.8f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

    # === Phase 5: Try larger n (16384) (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.10
    if p5_time > 60 and global_best_obj < 1.51:
        n_large = 16384
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n_large, p5_time))
        a_up = multi_scale_upsample(global_best_a, factor=2, rng=rng)
        S2 = np.sqrt(2.0 * n_large)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled init: obj=%.8f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        # Optimize the upsampled version
        rem5 = min(p5_time * 0.5, deadline - time.monotonic())
        if rem5 > 10:
            a_up_opt, obj_up_opt = run_fista(a_up, rem5, rng,
                                              beta_init=10, beta_rate=0.005,
                                              lam_var_init=0.05, lam_var_decay=2000,
                                              use_backtrack=False)
            print("  FISTA upsampled: obj=%.8f" % obj_up_opt)
            if obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()

    # === Phase 6: Final polish with iterated local search (10% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.90
    if p6_time > 20:
        print("\n=== Phase 6: Final ILS (budget %.0fs) ===" % p6_time)
        n_cur = len(global_best_a)
        S_cur = np.sqrt(2.0 * n_cur)

        # Multiple short perturb-and-reoptimize cycles
        n_cycles = max(3, int(p6_time / 30))
        per_cycle = p6_time / n_cycles

        for cycle in range(n_cycles):
            if time.monotonic() >= deadline:
                break
            # Perturb
            noise_scale = 0.01 * S_cur / np.sqrt(n_cur) * (1.0 + 0.5 * cycle)
            a_pert = project_simplex(global_best_a + rng.normal(0, noise_scale, n_cur), S_cur)
            # Short FISTA
            rem = min(per_cycle, deadline - time.monotonic())
            if rem < 3:
                break
            a_try, obj_try = run_fista(a_pert, rem, rng,
                                        beta_init=500, beta_rate=0.001,
                                        lam_var_init=0.0, lam_var_decay=99999,
                                        use_backtrack=True)
            if obj_try < global_best_obj:
                global_best_obj = obj_try
                global_best_a = a_try.copy()
                print("  ILS cycle %d: obj=%.8f ***NEW BEST***" % (cycle + 1, obj_try))
            else:
                print("  ILS cycle %d: obj=%.8f" % (cycle + 1, obj_try))

    # Final verification and return
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: invalid, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
