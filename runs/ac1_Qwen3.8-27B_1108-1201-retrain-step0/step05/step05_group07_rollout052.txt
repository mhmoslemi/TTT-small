def fft_convolve_self(a):

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp continuation,
Nesterov-accelerated subgradient, spectral initialization, and multi-length exploration.
Key innovations over previous: second-order L-BFGS, adaptive continuation schedule,
diverse initializations (random, spectral, interpolated), and perturbation-refinement loops.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_grad(a, beta):
    """Compute LSE-smoothed max of self-convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(w, fsize)
    A_flipped = fft(a[::-1].copy(), fsize)
    conv_result = np.real(ifft(W * A_flipped))
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    val = cm + np.log(tot) / beta
    return val, grad


def obj_from_conv(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def project_nonneg(a):
    """Project onto non-negative orthant."""
    return np.maximum(a, 0.0)


def nesterov_optimize(a0, max_time, rng, top_k=10, verbose=False):
    """
    Nesterov-accelerated projected subgradient method.
    Minimizes max(conv(a,a)) over a >= 0 (scale-invariant, so no sum constraint needed).
    Uses top-k lag weighting for the subgradient.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a_prev = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    t = 1.0
    stuck = 0
    patience = 200

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 1.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)

        c_y = fft_convolve_self(y)
        cur_max = c_y.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Top-k subgradient
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c_y)[::-1][:k]
        median_c = np.median(c_y)
        weights = np.maximum(c_y[top_idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum

        g = np.zeros(n)
        for ii in range(k):
            kid = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kid - n + 1)
            j_end = min(n - 1, kid)
            if j_start <= j_end:
                indices = kid - np.arange(j_start, j_end + 1)
                g[j_start:j_end + 1] += 2.0 * wi * y[indices]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Nesterov step size
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        step = 2.0 / (t_new * gnorm * t_new)
        step = min(step, 5.0 / gnorm)

        a_prev = a.copy()
        a = np.maximum(a - step * g, 0.0)
        t = t_new

        # Noise injection when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.003 * (stuck / patience))
            a = a + noise_scale * rng.standard_normal(n) / np.sqrt(n)
            a = np.maximum(a, 0.0)
            t = 1.0
            a_prev = a.copy()
            stuck = 0
            patience = int(patience * 1.3)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it={it}, t={elapsed:.0f}s, best_max={best:.8f}")

    return best_a, best


def optimize_lbfgs_continuation(a0, max_time, rng, verbose=True):
    """
    L-BFGS optimization with continuation on beta.
    Minimizes LSE-smoothed max of self-convolution.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()

    # Adaptive continuation schedule
    betas = [3, 5, 10, 20, 40, 80, 150, 300, 600, 1200, 2500, 5000]

    for bi, beta in enumerate(betas):
        if time.monotonic() - start > max_time - 5:
            break

        remaining = max_time - (time.monotonic() - start)
        time_per_beta = remaining / (len(betas) - bi)

        def fun(x):
            val, _ = lse_value_and_grad(x, beta)
            return val

        def grad_fun(x):
            _, g = lse_value_and_grad(x, beta)
            return g

        max_iter = max(20, int(60 * time_per_beta))
        max_iter = min(max_iter, 500)

        try:
            result = minimize(fun, a.copy(), method='L-BFGS-B', jac=grad_fun,
                              bounds=[(0.0, 1000.0)] * n,
                              options={'maxiter': max_iter, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxfun': max_iter * 3})
            if result.x is not None:
                a_new = np.maximum(result.x, 0.0)
                c_new = fft_convolve_self(a_new)
                cur = c_new.max()
                if cur < best:
                    best = cur
                    best_a = a_new.copy()
                a = a_new
        except Exception as e:
            if verbose:
                print(f"      L-BFGS failed at beta={beta}: {e}")
            continue

        if verbose:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (a.sum() ** 2)
            print(f"      beta={beta:5d}, obj={cur_obj:.6f}, max_c={best:.8f}, "
                  f"t={elapsed:.0f}s, status={result.status if result else 'N/A'}")

    # Final Nesterov polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 10:
        a_pol, best_pol = nesterov_optimize(best_a, remaining * 0.7, rng, verbose=False)
        c_pol = fft_convolve_self(a_pol)
        if c_pol.max() < best:
            best = c_pol.max()
            best_a = a_pol.copy()

    return best_a, best


def spectral_init(n, rng, n_iter=40):
    """
    Generate initialization with roughly flat autocorrelation
    using iterative peak reduction.
    """
    a = np.abs(rng.standard_normal(n)) + 0.05
    a = a / a.sum()

    for it in range(n_iter):
        c = fft_convolve_self(a)
        peak_idx = np.argmax(c)
        cur_peak = c[peak_idx]
        mean_c = c.mean()
        if cur_peak < mean_c * 1.05:
            break
        # Reduce contribution to peak lag
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        if j_start <= j_end:
            indices = peak_idx - np.arange(j_start, j_end + 1)
            contrib = a[indices]
            # Dampen the contributing elements
            damping = np.exp(-0.3 * contrib / max(contrib.max(), 1e-10))
            a[j_start:j_end + 1] *= damping
        a = np.maximum(a, 1e-8)
        a = a / a.sum()

    return a


def cosine_init(n):
    """Smooth cosine-based initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    a += 0.01
    return a / a.sum()


def random_smooth_init(n, rng, smoothness=3.0):
    """Random initialization with smoothness control."""
    a = np.abs(rng.standard_normal(n)) + 0.01
    # Apply moving average for smoothness
    k = max(1, int(smoothness))
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def interpolate_to_length(base, n):
    """Interpolate a sequence to a new length."""
    if len(base) == n:
        return base.copy()
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, base)

            a_opt, _ = optimize_lbfgs_continuation(a_pert, t_per * 0.7, rng, verbose=False)
            update_global_state(state, a_opt, f"Phase6 pert {pi} (scale={scale:.4f})")

    # === Final verification ===
    if state['best_a'] is None:
        print("WARNING: No valid solution found, using fallback")
        state['best_a'] = np.ones(256)
        state['best'] = obj_from_conv(state['best_a'])

    final_a = np.clip(state['best_a'], 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    final_list = [float(x) for x in final_a]
    n_final = len(final_a)
    final_obj = 2.0 * n_final * fft_convolve_self(final_a).max() / (final_a.sum() ** 2)

    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: len={n_final}, obj={final_obj:.6f}")
    print(f"Verified via evaluate_sequence: ", end="")
    try:
        ver = evaluate_sequence(final_list)
        print(f"{ver:.6f}")
    except Exception as e:
        print(f"Error: {e}")
    print(f"{'=' * 60}")

    return final_list
