"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Fast FISTA screening of diverse initializations
2. L-BFGS-B on LSE-smoothed objective for fast basin convergence
3. Excess-squared objective optimization with adaptive threshold
4. Exact subgradient method with diminishing steps for final polish
5. Fourier-domain restarts for escaping local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_objective_and_grad(a, beta, S):
    """Compute LSE-smoothed objective and gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    lse_val = (cm + np.log(tot) / beta)
    w = e / tot
    # Gradient: (1/beta) * cross_correlation(w, a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    # The 2n/S^2 factor: include it for correct scaling
    scale = 2.0 * n / (S * S)
    return scale * lse_val, scale * (1.0 / beta) * grad


def excess_objective_and_grad(a, T, p, S):
    """Compute excess-p objective and gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    excess = np.maximum(c - T, 0.0)
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        obj = 2.0 * n * c.max() / (S * S)
        return obj, np.zeros(n)
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = 2.0 * n * c.max() / (S * S)  # Report the true objective
    return obj, grad * (2.0 * n / (S * S))


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = c.max()
        scaled = np.clip(beta * (c - cm), -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)  # Use a, not y, for step
        grad = np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta * max(c.max(), 1e-10) + gnorm * 0.1)

        if it - last_improve > 2000:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a, fft_convolve_self(best_a).max()


def lbfgs_optimize(a_init, S, max_time, rng, beta=100.0, use_excess=False, T_frac=0.8):
    """
    L-BFGS-B optimization on the LSE or excess-smoothed objective.
    Uses the reparameterization a[0] = S - sum(a[1:n]).
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    # Reparameterize: x = a[1:n], a[0] = S - sum(x)
    x0 = a[1:n].copy()

    def func_and_grad(x):
        # Reconstruct a
        a_full = np.empty(n)
        a_full[0] = S - x.sum()
        a_full[1:] = x
        # Clip for numerical safety (a[0] might go slightly negative)
        if a_full[0] < 0:
            a_full = np.maximum(a_full, 0)
            a_full *= S / a_full.sum()
        
        if use_excess:
            c = fft_convolve_self(a_full)
            T = T_frac * c.max() + (1 - T_frac) * c.mean()
            excess = np.maximum(c - T, 0.0)
            w = 2.0 * excess
            wsum = w.sum()
            if wsum < 1e-30:
                return 0.0, np.zeros(n - 1)
            w = w / wsum
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a_full, fsize)
            grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            # df/dx[k] = grad_a[k+1] - grad_a[0]
            grad_x = grad_a[1:] - grad_a[0]
            obj = c.max()
        else:
            c = fft_convolve_self(a_full)
            cm = c.max()
            scaled = np.clip(beta * (c - cm), -500, 500)
            e = np.exp(scaled)
            tot = e.sum()
            if tot < 1e-300:
                return cm, np.zeros(n - 1)
            w = e / tot
            fsize = 1 << (len(w) + n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a_full, fsize)
            grad_a = (1.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]
            grad_x = grad_a[1:] - grad_a[0]
            obj = cm + np.log(tot) / beta

        return obj, grad_x

    # Bounds for x: each x[k] in [0, S]
    bounds = [(0, S)] * (n - 1)

    res = minimize(func_and_grad, x0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 100000, 'maxfun': 50000,
                            'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-8,
                            'iprint': -1})

    # Reconstruct
    x_opt = np.clip(res.x, 0, S)
    a_opt = np.empty(n)
    a_opt[0] = max(0, S - x_opt.sum())
    a_opt[1:] = x_opt
    a_opt = normalize_to_sum(a_opt, S)

    val = fft_convolve_self(a_opt).max()
    if val < best:
        best = val
        best_a = a_opt

    elapsed = time.monotonic() - start
    print("    L-BFGS: %d iters, %.0fs, max_b=%.8f" % (res.nit, elapsed, best))
    return best_a, best


def exact_subgradient_polish(a_init, S, max_time, rng, n_top=8):
    """
    Exact subgradient method: minimize max(b) using the true subgradient
    from the top-k lags, with diminishing step sizes.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    y = a.copy()
    t_mom = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        if it - last_improve > 5000:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 10000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        # Top-k subgradient
        k = min(n_top + it // 10000, n // 8)
        idx = np.argsort(c)[::-1][:k]
        weights = np.ones(k) / k
        w_full = np.zeros(2 * n - 1)
        w_full[idx] = weights
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Diminishing step size
        step = S / (n * (1.0 + 0.001 * it) * max(gnorm / (S / n), 1e-10))
        step = min(step, 0.1 * S / n)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a, fft_convolve_self(best_a).max()


def fourier_perturb(a, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    phase_noise = rng.normal(0, scale, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))

    # Raised cosine variants
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 3 + 0.01)
    
    # Asymmetric inits
    inits.append(np.sin(np.pi * tt * 0.8 + 0.2) ** 2 + 0.01)
    inits.append(np.maximum(0.3 + 0.7 * np.sin(np.pi * tt * 1.3 + 0.5), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)

    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)
