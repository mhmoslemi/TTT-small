"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements over previous approach:
- Coordinate descent with exact 1D minimax subproblem solving
- Aggressive beta ramping in LSE-FISTA phase
- Multi-perturbation escape from local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex_nonneg(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, active_indices):
    """Subgradient from specified active lags."""
    n = len(y)
    w_full = np.zeros(len(c))
    w_full[active_indices] = 1.0 / len(active_indices)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_max=100000.0):
    """LSE FISTA with aggressive adaptive beta ramping."""
    n = len(a_init)
    S = np.sqrt(float(2) * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    patience = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            patience = 0
        else:
            patience += 1

        # Aggressive beta schedule
        if patience < 200:
            beta = beta_init * (1.0 + 0.01 * it)
        elif patience < 1000:
            beta = min(beta_max, beta * 1.02)
        elif patience < 5000:
            beta = min(beta_max, beta * 1.05)
        else:
            beta = min(beta_max, beta * 1.10)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        step = 0.9 / (beta * S + 1.0)
        step = max(step, 1e-12)

        x_new = project_simplex_nonneg(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with noise when stuck
        if patience > 3000:
            if patience % 500 == 0:
                t_mom = 1.0
                noise_scale = max(1e-8, 0.01 * S / np.sqrt(n) * np.exp(-patience / 20000.0))
                y = project_simplex_nonneg(y + rng.normal(0, noise_scale, n), S)
                patience = 0
                last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """Active-set polishing with adaptive top-k focus."""
    n = len(a_init)
    S = np.sqrt(float(2) * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 50:
            tol = 0.0
        elif stuck < 200:
            tol = 1e-6
        elif stuck < 1000:
            tol = 1e-5
        elif stuck < 5000:
            tol = 5e-5
        else:
            tol = 1e-4

        active = np.where(c >= cur_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = exact_subgradient(y, c, active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = min(0.5 / gnorm, S / (gnorm * (1.0 + 0.005 * it)))
        step = max(step, 1e-14)

        x_new = project_simplex_nonneg(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 100 == 0:
            noise_scale = max(1e-8, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex_nonneg(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_conv


def coordinate_descent_step(a, c, i, K=15, max_iters=30):
    """
    Solve the 1D minimax subproblem for coordinate i.
    Only considers the top-K convolution lags for efficiency.
    Returns the optimal value of a[i].
    """
    n = len(a)
    cur_val = a[i]

    # Get top-K lags
    top_k_idx = np.argsort(c)[::-1][:K]

    # Compute base values and weights for affected lags
    # c_k(t) = base_k + 2*t*w_k (for k != 2i) or base_k + t^2 (for k = 2i)
    # Only lags where 0 <= k-i < n are affected (i.e., k in [i, i+n-1])

    affected = []
    for k in top_k_idx:
        k_int = int(k)
        if 0 <= k_int - i < n:
            # This lag is affected by a[i]
            if k_int == 2 * i:
                # Quadratic term: c_k(t) = (c_k - a[i]^2) + t^2
                base = c[k_int] - cur_val * cur_val
                affected.append((k_int, base, 'quad'))
            else:
                w = a[k_int - i]
                base = c[k_int] - 2.0 * cur_val * w
                affected.append((k_int, base, w))

    # Non-affected top-K lags contribute a constant
    non_affected_max = 0.0
    for k in top_k_idx:
        k_int = int(k)
        if not (0 <= k_int - i < n):
            non_affected_max = max(non_affected_max, c[k_int])

    if len(affected) == 0:
        return cur_val  # No change possible

    # Also consider the max of all non-top-K lags that are affected
    # For safety, include the current max as a lower bound
    lower_bound = max(non_affected_max, c.max() - 1e-10)

    # 1D convex minimax: minimize max(lower_bound, max_{affected} c_k(t))
    # Use ternary search on [0, 3*cur_val] (bounded)
    lo = 0.0
    hi = max(3.0 * cur_val, 1.0)

    def obj_1d(t):
        mx = lower_bound
        for (k, base, typ) in affected:
            if typ == 'quad':
                val = base + t * t
            else:
                val = base + 2.0 * t * typ
            if val > mx:
                mx = val
        return mx

    # Ternary search
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_1d(m1) <= obj_1d(m2):
            hi = m2
        else:
            lo = m1

    return 0.5 * (lo + hi)


def run_coordinate_descent(a_init, max_time, rng, top_K=15):
    """
    Coordinate descent with exact 1D minimax subproblem.
    Multiple sweeps, selecting coordinates by gradient magnitude.
    """
    n = len(a_init)
    S = np.sqrt(float(2) * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    sweep = 0

    order = np.arange(n)
    np.random.default_rng(rng.integers(0, 2**31)).shuffle(order)

    while time.monotonic() - start < max_time:
        sweep += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()

        # Compute gradient for coordinate selection
        active = np.where(c >= cur_max - 1e-8)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient
        fsize = 1 << (len(active) + n - 1).bit_length()
        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / len(active)
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Select coordinates: those with largest |grad| and non-zero a
        # Use a subset for speed
        subset_size = min(n, max(100, n // 4))
        idx = np.argsort(np.abs(grad))[::-1][:subset_size]

        # Shuffle within subset for diversity
        subset_order = order[np.argsort(np.searchsorted(order, idx))][:subset_size]

        improved = False
        for i in subset_order:
            t_opt = coordinate_descent_step(a, c, i, K=top_K, max_iters=25)
            if abs(t_opt - a[i]) > 1e-12:
                old_val = a[i]
                a[i] = t_opt
                # Quick check: recompute only if significant change
                if abs(t_opt - old_val) > 1e-6 * S / n:
                    improved = True

        # Renormalize after each sweep
        a = normalize_seq(a, S)

        # Check time
        if time.monotonic() - start > max_time * 0.9:
            break

        # Periodic full recomputation for accuracy
        if sweep % 3 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_conv - 1e-15:
                best_conv = c_check.max()
                best_a = a.copy()

    return best_a, best_conv


def run_perturbation_search(a_init, max_time, rng, num_perturbations=20):
    """Try random perturbations and quick re-optimization from each."""
    n = len(a_init)
    S = np.sqrt(float(2) * n)
    best_a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    best_a = normalize_seq(best_a, S)
    best_conv = fft_convolve_self(best_a).max()
    start = time.monotonic()

    time_per_pert = max_time / num_perturbations

    for p in range(num_perturbations):
        if time.monotonic() - start > max_time:
            break

        # Generate perturbation: mix of random and structured
        style = p % 3
        if style == 0:
            # Pure noise
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            perturbed = normalize_seq(best_a + noise, S)
        elif style == 1:
            # Sinusoidal perturbation
            freq = rng.uniform(1, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n, dtype=np.float64)
            noise = 0.03 * np.sin(2 * np.pi * freq * tt / n + phase)
            noise = noise * (S / n)
            perturbed = normalize_seq(best_a + noise, S)
            perturbed = np.maximum(perturbed, 0)
            perturbed = normalize_seq(perturbed, S)
        else:
            # Block perturbation
            block_size = rng.integers(n // 16, n // 2)
            start_idx = rng.integers(0, n - block_size)
            noise = np.zeros(n)
            noise[start_idx:start_idx + block_size] = rng.uniform(
                -0.05, 0.05, block_size) * S / n
            perturbed = normalize_seq(best_a + noise, S)
            perturbed = np.maximum(perturbed, 0)
            perturbed = normalize_seq(perturbed, S)

        # Quick LSE optimization from perturbed point
        quick_time = time_per_pert * 0.7
        a_quick, _ = run_lse_fista(perturbed, quick_time, rng,
                                    beta_init=50.0, beta_max=20000.0)
        conv_quick = fft_convolve_self(a_quick).max()

        if conv_quick < best_conv:
            best_conv = conv_quick
            best_a = a_quick.copy()
            print("  Perturbation %d improved: %.10f" % (p, best_conv))

    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    # Final verification
    if global_best_a is not None:
