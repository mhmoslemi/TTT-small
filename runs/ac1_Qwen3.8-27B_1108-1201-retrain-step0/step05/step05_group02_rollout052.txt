2. The term where `k-j = m`, i.e., `j = k-m`: contributes `a_{k-m} * a_m`, derivative is `a_{k-m}`

If `m = k-m` (i.e., `k = 2m`), it's the same term `a_m^2`, so `dc_k/da_m = 2*a_m = 2*a_{k-m}`.

So in all cases: `dc_k/da_m = 2*a_{k-m}` (when `0 <= k-m < n`).

Therefore: `grad_m = sum_k w_k * dc_k/da_m = 2 * sum_k w_k * a_{k-m}` (over valid k).

So indeed `grad = 2 * corr(w, a)` (truncated to length n). Good, the factor of 2 is correct.

    it = 0
    last_improve = 0
    best_eta = eta

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            best_eta = eta

        w, _ = lse_weights(c, beta)
        grad = compute_conv_grad(a, w)

        # Exponentiated update
        log_update = -eta * grad
        # Clip to avoid overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = (1.0 - damping) * a + damping * a_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * (S / a_new.sum())
        a = a_new

        eta *= eta_decay
        beta *= (1.0 + beta_rate)
        beta = min(beta, 50000.0)

        # Escape mechanism
        stuck = it - last_improve
        if stuck > escape_interval and it % 200 == 0:
            noise_scale = escape_noise * np.exp(-stuck / (escape_interval * 3))
            noise = noise_scale * rng.standard_normal(n)
            a = a * np.exp(noise)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            eta = best_eta * 0.7
            beta = beta_init

        if stuck > escape_interval * 5:
            break

    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with adaptive LSE for smoothing the max."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        w, _ = lse_weights(c, beta)
        grad = compute_conv_grad(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj * 1.001 + 1e-10 and step_local > 1e-14 and bt < 5:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1

        if bt >= 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.02, 4.0 / max(beta, 1.0))

        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 4000:
            t_mom = 1.0
            ns = max(1e-6, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, ns, n), S)

    return best_a, best_obj


def run_peak_reduction(a_init, max_time, rng):
    """Direct peak reduction: iteratively reduce the max convolution value
    by transferring mass from contributing coordinates to non-contributing ones."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-6, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 5000:
            break

        # Find top peaks
        n_peaks = min(3, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute contribution of each coordinate to the peaks
        multi_contrib = np.zeros(n)
        for kk in peak_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                multi_contrib[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Find best transfer: from high multi_contrib to low multi_contrib
        # Score source by contribution * value, target by lack of contribution
        src_score = multi_contrib * a
        tgt_score = -multi_contrib * (1.0 - a / S + 0.001)

        n_try = min(8, n)
        src_cand = np.argsort(src_score)[::-1][:n_try]
        tgt_cand = np.argsort(tgt_score)[:n_try]

        best_delta_obj = cur_obj
        best_move = None

        for j in src_cand:
            for l in tgt_cand:
                if j == l or a[j] < 1e-10:
                    continue
                # Try several fractions
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    # Estimate new max
                    # c changes: for lag k, delta_c[k] = -delta*a[k-j] + delta*a[k-l]
                    new_max = cur_max
                    affected = [int(kk) for kk in peak_idx]
                    for kk in affected:
                        kk = int(kk)
                        dc = 0.0
                        # Effect of reducing a[j]
                        if 0 <= kk - j < n:
                            dc -= delta * a[kk - j]
                        # Effect of increasing a[l]
                        if 0 <= kk - l < n:
                            dc += delta * a[kk - l]
                        if c[kk] + dc > new_max:
                            new_max = c[kk] + dc
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_delta_obj - 1e-14:
                        best_delta_obj = new_obj
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            # Update c incrementally (approximate)
            # Recompute fully every 50 iterations for accuracy
            if it % 50 == 0:
                c = fft_convolve_self(a)
            else:
                # Incremental update
                js = 0
                je = min(2 * n - 2, j + n - 1)
                for k in range(js, je + 1):
                    m = k - j
                    if 0 <= m < n:
                        c[k] -= delta * a[m]
                ls = 0
                le = min(2 * n - 2, l + n - 1)
                for k in range(ls, le + 1):
                    m = k - l
                    if 0 <= m < n:
                        c[k] += delta * a[m]
        else:
            # No improvement; add small noise
            if it - last_improve > 1000:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    a = np.maximum(a, 1e-15)
                    a = a * (S / a.sum())
                    c = fft_convolve_self(a)

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.4) ** 2 / (0.2 ** 2))
    elif method == "ramp":
        a = tt
    elif method == "inv_ramp":
        a = 1.0 - tt
    elif method == "asym_gauss":
        a = np.exp(-2.0 * (tt - 0.3) ** 2 / (0.15 ** 2))
    elif method == "bimodal":
        a = np.exp(-20 * (tt - 0.2) ** 2) + np.exp(-20 * (tt - 0.8) ** 2)
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt + 0.5)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "log_decay":
        a = np.exp(-3.0 * tt)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging."""
    n = len(a)
    n_new = n // factor
    if n_new < 2:
        return a.copy()
    a_new = a[:n_new * factor].reshape(n_new, factor).mean(axis=1)
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init, exponentiated gradient, FISTA, peak reduction,
    multi-scale upsample, final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from previous best
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Starting from previous: n=%d, obj=%.6f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = make_init(n0, "cosine", rng)
        obj0 = compute_obj(a_start)
        global_best_obj = obj0
        global_best_a = a_start.copy()
        print("No previous, starting fresh: n=%d, obj=%.6f" % (n0, obj0))

    # ============================================================
    # Phase 1: Exponentiated gradient from multiple starts (35% time)
    # ============================================================
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.35
    if p1_time > 60:
        print("\n=== Phase 1: Exponentiated gradient (budget %.0fs) ===" % p1_time)

        # First: run on the previous best
        a_opt, obj_opt = run_exp_gradient(global_best_a, p1_time * 0.35, rng,
                                           beta_init=15.0, beta_rate=0.004,
                                           eta_init=0.6, eta_decay=0.9996,
                                           damping=1.0, escape_interval=3000)
        print("  ExpGrad(prev): obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # Then: diverse initializations
        n0 = len(global_best_a)
        methods = ["asym_gauss", "ramp", "log_decay", "bimodal", "random"]
        per_time = p1_time * 0.65 / len(methods)
        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = run_exp_gradient(a_div, per_time, rng,
                                               beta_init=10.0, beta_rate=0.005,
                                               eta_init=0.5, eta_decay=0.9995,
                                               damping=0.9, escape_interval=2000)
            status = " ***NEW***" if obj_opt < global_best_obj else ""
            print("  ExpGrad(%s): obj=%.6f%s" % (method, obj_opt, status))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # ============================================================
    # Phase 2: FISTA refinement (25% time)
    # ============================================================
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 45:
        print("\n=== Phase 2: FISTA refinement (budget %.0fs) ===" % p2_time)

        variants = [
            (20, 0.006, 20000, 1.0),
            (100, 0.003, 50000, 0.5),
            (500, 0.001, 100000, 0.3),
        ]
        for i, (bi, br, bm, ss) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            a_opt, obj_opt = run_fista(global_best_a, p2_time * 0.33, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss)
            status = " ***NEW***" if obj_opt < global_best_obj else ""
            print("  FISTA%d: obj=%.6f%s" % (i + 1, obj_opt, status))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # ============================================================
    # Phase 3: Peak reduction (20% time)
    # ============================================================
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 30:
        print("\n=== Phase 3: Peak reduction (budget %.0fs) ===" % p3_time)
        a_pr, obj_pr = run_peak_reduction(global_best_a, p3_time, rng)
        status = " ***NEW***" if obj_pr < global_best_obj else ""
        print("  Peak reduction: obj=%.6f%s" % (obj_pr, status))
        if obj_pr < global_best_obj:
            global_best_obj = obj_pr
            global_best_a = a_pr.copy()

        # Quick FISTA after peak reduction
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            a_opt, obj_opt = run_fista(global_best_a, remaining3 * 0.6, rng,
                                        beta_init=50, beta_rate=0.005,
                                        beta_max=30000, step_scale=0.8)
            status = " ***NEW***" if obj_opt < global_best_obj else ""
            print("  FISTA after PR: obj=%.6f%s" % (obj_opt, status))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # ============================================================
    # Phase 4: Multi-scale upsample (15% time)
    # ============================================================
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    if p4_time > 40 and len(global_best_a) <= 16384:
        n2 = min(len(global_best_a) * 2, 32768)
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        # Optimize the upsampled version
        a_opt, obj_opt = run_exp_gradient(a_up, p4_time * 0.5, rng,
                                           beta_init=5.0, beta_rate=0.005,
                                           eta_init=0.4, eta_decay=0.9995,
                                           damping=1.0, escape_interval=3000)
        status = " ***NEW***" if obj_opt < global_best_obj else ""
        print("  ExpGrad upsampled: obj=%.6f%s" % (obj_opt, status))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_opt2, obj_opt2 = run_fista(global_best_a, remaining4 * 0.6, rng,
                                          beta_init=30, beta_rate=0.005,
                                          beta_max=20000, step_scale=0.7)
            status = " ***NEW***" if obj_opt2 < global_best_obj else ""
            print("  FISTA upsampled: obj=%.6f%s" % (obj_opt2, status))
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # ============================================================
    # Phase 5: Final polish (5% time)
    # ============================================================
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.9
    if p5_time > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)

        # Very high-beta FISTA to sharply reduce the max
        a_opt, obj_opt = run_fista(global_best_a, p5_time * 0.5, rng,
                                    beta_init=200, beta_rate=0.002,
                                    beta_max=200000, step_scale=0.3)
        status = " ***NEW***" if obj_opt < global_best_obj else ""
        print("  High-beta FISTA: obj=%.6f%s" % (obj_opt, status))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # Final exponentiated gradient with fine parameters
        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_opt2, obj_opt2 = run_exp_gradient(global_best_a, remaining5, rng,
                                                 beta_init=100.0, beta_rate=0.001,
                                                 eta_init=0.2, eta_decay=0.9998,
                                                 damping=0.8, escape_interval=5000)
            status = " ***NEW***" if obj_opt2 < global_best_obj else ""
            print("  Fine ExpGrad: obj=%.6f%s" % (obj_opt2, status))
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
