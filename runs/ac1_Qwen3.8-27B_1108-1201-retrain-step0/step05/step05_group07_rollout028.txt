    v = np.asarray(v, dtype=np.float64).copy()

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses L-BFGS on log-sum-exp smoothed objective with softmax parameterization,
beta annealing, and multi-start with smooth perturbations.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x_max = np.max(x)
    e = np.exp(x - x_max)
    s = e.sum()
    return e / s


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_obj_grad(t, beta, S, fsize):
    """Compute LSE-smoothed objective and gradient w.r.t. softmax parameters t."""
    n = len(t)
    a = S * softmax(t)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:2 * n - 1]
    
    cm = c.max()
    scaled = beta * (c - cm)
    e = np.exp(scaled)
    e_sum = e.sum()
    w = e / e_sum
    lse_val = cm + np.log(e_sum) / beta
    
    # Gradient of LSE w.r.t. a
    W = fft(w, fsize)
    grad_a = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    
    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_t = a * (grad_a - dot / S)
    
    return lse_val, grad_t


def lse_obj_only(t, beta, S, fsize):
    """Compute just the LSE-smoothed objective (for verification)."""
    n = len(t)
    a = S * softmax(t)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:2 * n - 1]
    cm = c.max()
    scaled = beta * (c - cm)
    e = np.exp(scaled)
    return cm + np.log(e.sum()) / beta


def optimize_lbfgs(t0, beta, S, fsize, max_iter=500, gtol=1e-8):
    """Run L-BFGS to minimize LSE-smoothed objective."""
    
    def fun(t):
        val, _ = lse_obj_grad(t, beta, S, fsize)
        return val
    
    def fun_grad(t):
        return lse_obj_grad(t, beta, S, fsize)
    
    res = minimize(fun_grad, t0, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'gtol': gtol, 'maxls': 50})
    return res.x, res.fun


def t_from_a(a, S):
    """Convert a positive vector a (with sum S) to softmax parameters t."""
    # a = S * softmax(t) => softmax(t) = a/S => t = log(a/S) (up to a constant)
    # We use t = log(a) - log(S) (the constant doesn't matter for softmax)
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe) - np.log(S)


def a_from_t(t, S):
    """Convert softmax parameters t to vector a with sum S."""
    return S * softmax(t)


def subgradient_step(a, c, S, k, step):

    n = len(a)

def smooth_noise(rng, n, bandwidth):
    """Generate smooth random noise with given bandwidth."""
    # Generate white noise in frequency domain, keep only low frequencies
    fsize = 1 << (n - 1).bit_length()
    freqs = np.fft.fftfreq(fsize)
    mask = np.abs(freqs) < bandwidth
    noise_freq = rng.normal(0, 1, fsize)
    noise_freq[~mask] = 0
    noise = np.real(ifft(noise_freq, fsize))[:n]
    noise /= (np.std(noise) + 1e-30)
    return noise
