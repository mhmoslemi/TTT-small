Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: 
1. Diverse initialization with quick screening.
2. Enhanced subgradient descent with exact argmax subgradient, 
   backtracking line search, and simulated annealing.
3. Periodic spectral equalization (Gerchberg-Saxton) to escape local minima.
4. Block perturbation for diversity.
5. Final polishing with high-beta LSE FISTA.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def exact_subgradient(a, c):
    """Subgradient of max(c) w.r.t. a, where c = conv(a,a).
    If c[m] = max(c), then grad = 2 * (a correlated with delta_m).
    Equivalently, grad[j] = a[m-j] + a[j-m] (with 0 padding).
    """

    Reduce the peak of the autocorrelation by modifying the spectral magnitude.
