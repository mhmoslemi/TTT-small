import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    Nf = max(N, 1 << (2 * n - 1).bit_length())
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def gs_spectral_shaping(a, N, n_steps, alpha, S):
    """
    Gerchberg-Saxton-like spectral shaping.
    Iteratively flattens the power spectrum while maintaining non-negativity.
    alpha: interpolation parameter (0 = no change, 1 = full flattening)
    """
    n = len(a)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_score = np.inf
    
    for step in range(n_steps):
        A = fft(a_curr, N)
        P = np.abs(A) ** 2
        # Target flat power
        P_target = P.mean()  # = 2n / N * N = 2n (by Parseval)
        # Geometric interpolation
        P_new = P ** (1 - alpha) * P_target ** alpha
        # Reconstruct with original phase
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        # Non-negativity projection
        a_new = np.maximum(a_new, 0.0)
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        
        # Check improvement
        c, _ = compute_conv(a_new, N)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a_curr = a_new
    
    return best_a, best_score


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    Nf = max(N, 1 << (2 * n - 1).bit_length())
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def toplist_grad(a, N, top_k):

    n = len(a)

        idx = np.argsort(c)[::-1][:top_k]
        vals = c[idx]
        total = vals.sum()
        if total < 1e-30:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = np.zeros(len(c))
            weights[idx] = vals / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def gs_spectral_shaping(a_init, n, S, N, n_steps, alphas):
    """
    GS-like iterative spectral shaping.
    Alternates between flattening |FFT|^2 and projecting to non-negative time domain.
    alphas: list of alpha values for geometric interpolation (cycled through).
    """
    a = a_init.copy()
    best_a = a.copy()
    best_score = np.inf
    current_alpha_idx = 0

    for step in range(n_steps):
        alpha = alphas[current_alpha_idx % len(alphas)]
        current_alpha_idx += 1

        A = fft(a, N)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        if P_mean < 1e-30:
            break

        # Geometric interpolation toward flat
        log_P = np.log(np.maximum(P, 1e-300))
        log_target = np.log(P_mean)
        log_P_new = (1 - alpha) * log_P + alpha * log_target
        P_new = np.exp(log_P_new)

        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)

        c, _ = compute_conv(a_new, N)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def spectral_partial_restart(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

        a_gs, sc_gs = gs_spectral_shaping(a_norm, n, S, N, n_gs_steps, gs_alphas)

    print("Phase 1 best: %.6f" % global_best)

    # --- Phase 2: L-BFGS refinement ---
    print("=== Phase 2: L-BFGS refinement ===")
    t2 = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2) * 0.40)

    a_curr = global_a.copy() if global_a is not None else np.ones(n) * S / n
    stall_count = 0
    use_topk = False
    top_k_curr = 50

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(200.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 20:
            break

        beta_start = 10.0 + 5.0 * min(stall_count, 8)
        a_new, sc_new = lbfgs_block(a_curr, n, S, N, block_time,
                                     beta_init=beta_start, beta_max=500.0,
                                     m_lbfgs=15, use_topk=use_topk, top_k=top_k_curr)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            print("  [P2/tk=%d] max_c=%.6f" % (top_k_curr, sc_new))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count == 2:
                top_k_curr = min(200, top_k_curr * 2)
                use_topk = not use_topk
            elif stall_count >= 4:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.3)
                stall_count = 0

    # --- Phase 3: Peak-targeting descent ---
    print("=== Phase 3: Peak-targeting descent ===")
    t3 = time.monotonic()
    remaining3 = min(400.0, (deadline - t3) * 0.25)
    if remaining3 > 30:
        a_pt, sc_pt = peak_target_descent(global_a, n, S, N, remaining3)
        update_best(a_pt, "peak_target")

    # --- Phase 4: Final high-beta L-BFGS ---
    print("=== Phase 4: Final high-beta L-BFGS ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 20
    if remaining4 > 30:
        a_fin, sc_fin = lbfgs_block(global_a, n, S, N, min(remaining4, 400.0),
                                     beta_init=50.0, beta_max=1000.0,
                                     m_lbfgs=20, use_topk=False)
        update_best(a_fin, "final_LBFGS")

    # --- Phase 5: GS + LBFGS alternation ---
    remaining5 = deadline - time.monotonic() - 15
    if remaining5 > 40:
        half1 = remaining5 * 0.4
        half2 = remaining5 * 0.6
        a_gs2, _ = gs_spectral_shaping(global_a, n, S, N, max(20, int(half1 * 15)),
                                         [0.2, 0.3, 0.25, 0.35])
        update_best(a_gs2, "GS_refine")
        if deadline - time.monotonic() > 25:
            a_fin2, sc_fin2 = lbfgs_block(global_a, n, S, N,
                                           min(deadline - time.monotonic() - 10, half2),
                                           beta_init=100.0, beta_max=2000.0,
                                           m_lbfgs=20, use_topk=True, top_k=30)
            update_best(a_fin2, "final_topk")

    # --- Phase 6: One more spectral restart + polish ---
    remaining6 = deadline - time.monotonic() - 10
    if remaining6 > 50:
        a_restart = spectral_partial_restart(global_a, N, rng, frac=0.15)
        a_r, sc_r = lbfgs_block(a_restart, n, S, N, min(remaining6, 200.0),
                                 beta_init=20.0, beta_max=400.0, m_lbfgs=15)
        update_best(a_r, "restart_polish")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    # Compute actual score
    final_arr = global_a
    fin_sum = final_arr.sum()
    fin_conv = np.convolve(final_arr, final_arr)
    fin_score = 2.0 * len(final_arr) * fin_conv.max() / (fin_sum ** 2)
    print("Final: len=%d, score=%.6f" % (len(final_arr), fin_score))

    return [float(x) for x in global_a]
