"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FISTA with LSE smoothing, backtracking line search, 
basin hopping with Fourier mutations, and exact-subgradient polishing.
Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

# --- Core utilities ---

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

# --- Gradient computations ---

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    diff = c - cm
    maxdiff = diff.max()
    if maxdiff > 500 / beta:
        beta = 500.0 / maxdiff
    scaled = beta * diff
    e = np.exp(scaled)
    e -= e.max()  # for numerical stability
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Compute gradient = 2 * (w *~ a) = 2 * cross-correlation
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def exact_subgradient_topk(a, c, k=32):
    """Exact subgradient: convex combination of top-k active gradients."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above median
    thresh = c[int(0.5 * len(c) * np.random.random() * 0.5)]  # Hmm, this is wrong
    # Better: weight by (c_k - c_median)
    c_median = np.median(c)
    weights = np.maximum(c[idx] - c_median, 0.0)
    if weights.sum() < 1e-30:
        weights = np.ones(k)
    weights = weights / weights.sum()
    
    # Compute combined gradient
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

# --- Optimization routines ---

def run_fista_bt(a_init, S, max_time, rng, 
                 beta_init=10.0, beta_rate=0.003, beta_max=10000.0,
                 use_mirror=False, mirror_period=5000):
    """
    FISTA with backtracking line search on LSE-smoothed objective.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 2.0 / (4.0 * beta)  # Initial step
        for bt in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            # Check Armijo condition (for LSE objective, approximate with max)
            if c_new.max() <= cur_max + 0.01 * step * gnorm * 0.1 + 1e-10:
                break
            step *= 0.5
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Stuck handling
        if it - last_improve > 3000:
            if use_mirror and it % mirror_period == 0:
                g_max = np.max(np.abs(grad))
                if g_max > 1e-30:
                    a_new = a * np.exp(-0.1 * grad / g_max)
                    a = normalize_to_sum(a_new, S)
            else:
                noise = 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
                a = project_simplex(a + rng.normal(0, max(noise, 1e-12), n), S)
            y = a.copy()
            t_mom = 1.0
            last_improve = it
    
    return best_a, best

def run_polish(a_init, S, max_time, rng, k=64):
    """
    Polishing phase: exact subgradient with very small steps.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        # Use top-k subgradient with tight k
        k_eff = max(4, int(k * (1.0 - progress)))
        grad = exact_subgradient_topk(a, c, k=k_eff)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak-like step size
        step = 0.5 * S / (n * (1.0 + 0.001 * it))
        
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < best:
            a = a_new
        else:
            # Reject step, try smaller
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best:
                a = a_new
            else:
                # Add noise and try again
                noise = 1e-4 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return best_a, best

def fourier_phase_mutate(a, rng, scale=0.1):
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)
