Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses LP-based alternating minimization with cutting plane, combined with
subgradient descent and LSE smoothing for fine-tuning.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross(a, b):
    """Cross-correlation/convolution of a and b via FFT.
    Returns the full convolution of length len(a)+len(b)-1."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_for_g(f, S, active_lags=None, max_active=None):
    """
    Solve LP: minimize M subject to (f*g)_k <= M for k in active_lags,
    sum(g) = S, g >= 0.
    
    Returns (g, M) or (None, None) if infeasible.
    """
    n = len(f)
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    if max_active is not None:
        active_lags = active_lags[:max_active]
    
    K = len(active_lags)
    
    # Build constraint matrix: for each lag k, (f*g)_k = sum_j f[k-j]*g_j (with bounds)
    # A_ub: K x (n+1), b_ub: K
    # Plus equality: sum(g) = S
    
    # We build A_ub row by row
    # Row for lag k: [f[k], f[k-1], ..., f[k-(n-1)spaced]..., -1]
    # But we need to be careful about bounds.
    
    # For lag k, (f*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j]*g_j
    # So the coefficients for g_j are f[k-j] if 0<=k-j<=n-1
    
    # Building the full matrix is O(K*n) which for K=16383, n=8192 is ~134M - too large!
    # Need to use sparse representation or a different approach.
    
    # For scipy linprog, we can use sparse matrices.
    from scipy.sparse import csr_matrix, coo_matrix
    
    # Let's build in COO format
    rows = []
    cols = []
    data = []
    
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            for j in range(j_start, j_end + 1):
                rows.append(i)
                cols.append(j)
                data.append(f[k - j])
        # M coefficient
        rows.append(i)
        cols.append(n)  # M is the (n+1)-th variable (0-indexed: n)
        data.append(-1.0)
    
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    data = np.array(data, dtype=np.float64)
    
    A_ub = coo_matrix((data, (rows, cols)), shape=(K, n + 1)).tocsr()
    b_ub = np.zeros(K)
    
    # Equality constraint: sum(g) = S
    A_eq_one = np.ones(n + 1)
    A_eq_one[n] = 0.0  # M coefficient is 0
    A_eq = A_eq_one.reshape(1, -1)
    b_eq = np.array([S])
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, M free
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n]
    M = result.x[n]
    g = np.maximum(g, 0)  # Clean up numerical issues
    return g, M

def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta_eff = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def optimize_subgradient(a0, S, max_time, rng, verbose=True):
    """Subgradient optimization with top-k lags and adaptive step."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(10 + 100 * progress ** 0.7)
        k = min(k, 2 * n - 1)

        grad = compute_subgradient_topk(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = cur_max - c.mean()
        if gap < 1e-12:
            break

        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 5.0 / gnorm)

        # Line search
        accepted = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            step = 0.001 / gnorm
            a = project_simplex(a - step * grad, S)

        if stuck > patience:
            noise_scale = min(0.05, 0.005 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    SG it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best


def optimize_lse_polish(a0, S, max_time, rng, beta_init=50.0, beta_max=10000.0):
    """High-beta LSE polishing."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(beta_max, beta_init * np.exp(0.002 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (1.5 * beta)
        step = min(step, 2.0 / gnorm)

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-10):
            a = a_try
        else:
            a = project_simplex(a - 0.5 * step * grad, S)

    return best_a, best


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses LP-based alternating minimization (solving LP for g given f via cutting plane),
combined with multiplicative-weights dual approximation, subgradient descent,
and LSE smoothing. Key innovation: the LP subproblem gives exact solutions
to the "other half" of the problem, leading to faster convergence than pure
subgradient methods.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(f, y):
    """Compute (f*y)_k for k=0..len(f)+len(y)-2 using FFT.
    This is standard linear convolution."""
    size = len(f) + len(y) - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    Y = fft(y, fsize)
    return np.real(ifft(F * Y))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, active_lags):
    """Build sparse constraint matrix for LP.
    Returns (A_ub, b_ub) where A_ub is (K, n+1) sparse.
    The variables are [g_0,...,g_{n-1}, M].
    Constraint for lag k: sum_j f[k-j]*g_j - M <= 0
    """
    n = len(f)
    K = len(active_lags)
    rows = []
    cols = []
    data = []

    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            fkj = f[k - j]
            if abs(fkj) > 1e-15:
                rows.append(i)
                cols.append(j)
                data.append(fkj)
        # M coefficient (variable index n)
        rows.append(i)
        cols.append(n)
        data.append(-1.0)

    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    data = np.array(data, dtype=np.float64)

    A_ub = coo_matrix((data, (rows, cols)), shape=(K, n + 1)).tocsr()
    b_ub = np.zeros(K)
    return A_ub, b_ub


def solve_lp_for_g(f, S, active_lags):
    """
    Solve LP: minimize M s.t. (f*g)_k <= M for k in active_lags,
    sum(g) = S, g >= 0.
    Returns (g, M) or (None, None).
    """
    n = len(f)
    K = len(active_lags)

    A_ub, b_ub = build_lp_matrix(f, active_lags)

    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimize M (variable index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: g_j >= 0, M free
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status != 0:
            return None, None
        g = np.maximum(result.x[:n], 0)
        M = float(result.x[n])
        return g, M
    except Exception:
        return None, None


def multiplicative_weights_dual(f, eta, n_iters, rng):
    """
    Solve the dual: max min_j (f*y)_j s.t. y>=0, sum(y)=1
    using multiplicative weights.
    Returns approximate optimal y and the value min_j (f*y)_j.
    """
    n = len(f)
    K = 2 * n - 1
    y = np.ones(K) / K

    best_val = -np.inf
    y_bar = np.zeros(K)

    for t in range(n_iters):
        # Compute v_j = sum_k y_k * A_k_j = (f * y)_j (cross-correlation)
        # Here A_k_j = f[k-j] if 0 <= k-j < n
        # This is the cross-correlation of f with y
        # (f*y)_j = sum_k y_k * f[k-j] = sum_k y_k * A_k_j
        # This equals the cross-correlation: v_j = sum_k y[k] * f[k-j]
        # Which is the convolution of y with reversed f... 
        # Actually: (f*y)_k = sum_j f[j]*y[k-j]. We want v_j = sum_k y[k]*f[k-j].
        # Let's define: v_j = sum_{k=j}^{j+n-1} y[k] * f[k-j] (with y out of range = 0)
        # This is the cross-correlation: v = correlate(y, f) at lags 0..n-1
        # correlate(y, f)[j] = sum_k y[k+j] * conjugate(f[k])
        # So v_j = sum_k y[k+j] * f[k] for k=0..n-1 (with y=0 outside [0,K-1])
        
        # Using FFT: v = ifft(fft(y, F) * conj(fft(f, F)))[:n]
        F = 1 << (K + n - 1).bit_length()
        Yf = fft(y, F)
        Ff = fft(f, F)
        v = np.real(ifft(Yf * np.conj(Ff)))[:n]
        
        val_jstar = v.min()
        if val_jstar > best_val:
            best_val = val_jstar
            best_y = y.copy()
        
        y_bar += y / (t + 1)

        jstar = int(np.argmin(v))
        
        # Update: y_k *= exp(-eta * A_k_jstar) = exp(-eta * f[jstar-k])
        # A_k_jstar = f[jstar-k] if 0 <= jstar-k < n, else 0
        # So log-update for y_k: -eta * f[max(0, min(n-1, jstar-k))] (with 0 if out of range)
        # We only update k where jstar-k is in [0, n-1], i.e., k in [jstar-n+1, jstar]
        k_lo = max(0, jstar - n + 1)
        k_hi = min(K - 1, jstar)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            f_vals = f[jstar - ks]
            y[k_lo:k_hi + 1] *= np.exp(-eta * f_vals)
        
        y_sum = y.sum()
        if y_sum < 1e-300:
            y = np.ones(K) / K
        else:
            y /= y_sum

    return y_bar, best_val, best_y


def lp_alternating(f_init, S, max_time, rng, verbose=True):

    """

        it += 1

        c = fft_convolve_self(f)
        cur_M = c.max()

        if cur_M < best_M - 1e-10:
            best_M = cur_M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        # Select active lags: top-K from current f*f
        progress = min(1.0, it / 50)
        K = min(2 * n - 1, K_start + int(100 * progress) + it * 5)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            gap = cur_M - c.mean()
            step = 2.0 * max(gap, 1e-6) / (gnorm * gnorm)
            step = min(step, 5.0 / gnorm)
            f = project_simplex(f - step * grad, S)
            continue

        # Update f with line search on t
        best_t = 0.0
        best_obj = cur_M
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            f_try = project_simplex((1 - t) * f + t * g, S)

            print(f"  LP-alt it={it}, t={elapsed:.0f}s, obj={obj:.6f}, M={c_check.max():.4f}")

    return best_f, best_M


def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def optimize_subgradient(a0, S, max_time, rng, verbose=True):
    """Subgradient optimization with adaptive step and noise."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    stuck = 0

        it += 1
        c = fft_convolve_self(a)

            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        k = int(10 + 150 * progress ** 0.6)
        k = min(k, 2 * n - 1)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = cur_max - c.mean()
        if gap < 1e-12:
            break

        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 5.0 / gnorm)

        accepted = False
        for _ in range(8):

            c_try = fft_convolve_self(a_try)

            patience = int(patience * 1.15)

        if verbose and it % 500 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    SG it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    max_scaled = scaled.max()
    if max_scaled > 350:
        beta_eff = 350.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    stuck = 0

    while time.monotonic() - start < max_time:

        elapsed = time.monotonic() - start

        beta = min(beta_max, beta_init * np.exp(0.003 * it))

            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (1.5 * beta)
        step = min(step, 3.0 / gnorm)

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-10):

        else:

def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

                                                beta_init=1000.0, beta_max=80000.0)
            update_global(a_curr, f"Phase 4 iter {phase_i}")

    # === Phase 5: Random restarts with diverse perturbations ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = max(1, int((remaining - 15) / 45))
        t_per = (remaining - 15) / max(1, n_restarts)
        print(f"\nPhase 5: {n_restarts} restarts, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Different perturbation strategies
            if ri % 3 == 0:
                noise_scale = 0.03 * S / np.sqrt(n)
                a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            elif ri % 3 == 1:
                # Sparse perturbation: zero out random 5% and redistribute
                a_pert = global_a.copy()
                n_zero = max(1, n // 20)
                zero_idx = rng.choice(n, size=n_zero, replace=False)
                a_pert[zero_idx] = 0
                a_pert = project_simplex(a_pert, S)
            else:
                # Smooth perturbation
                a_pert = global_a.copy()
                p2 = np.fft.rfft(a_pert)
                p2 *= (1.0 + 0.05 * rng.standard_normal(len(p2)))
                a_pert = np.fft.irfft(p2, n)
                a_pert = project_simplex(a_pert, S)
            
            a_pert, _ = optimize_subgradient(a_pert, S, t_per, rng, verbose=False)
            update_global(a_pert, f"Phase 5 restart {ri}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
