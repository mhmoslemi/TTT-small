"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral flattening for initialization
2. FISTA on quartic objective sum(c_k^2) for main optimization
3. LSE-smoothed gradient polishing for the max objective
4. Multi-length exploration
"""
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    # standard projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_quartic_grad(a, c):
    """Gradient of sum(c_k^2) w.r.t. a. Result: 4 * d where d_i = sum_m a[m]*c[m+i]."""
    n = len(a)
    N = 1 << (n + len(c) - 1).bit_length()
    A = fft(a, N)
    C = fft(c, N)
    d = np.real(ifft(C * np.conj(A)))
    return 4.0 * d[:n]

def compute_lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 300:
        beta = 300.0 / max(1e-15, (c.max() - cm))
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # grad of LSE w.r.t. a: 2 * sum_k w_k * M_k * a = 2 * convolve(w, a_reversed)[:n]
    # Actually: d/d a_i [sum_k w_k c_k] = 2 * sum_k w_k * a_{k-i} (for valid k-i)
    # = 2 * (w cross-correlated with a)
    # Using FFT: = 2 * ifft(FFT(w, N) * conj(FFT(a, N)))[:n]
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def gs_flatten(a0, S, n_iter=200, rng=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()  # FFT size for zero-padded DFT
    a = a0.copy()
    
    # Target magnitude
    # F[0] = S (fixed), F[m] for m != 0 should be flat
    # Energy: sum |F[m]|^2 = N * sum a_j^2
    # For flat: F[0]^2 + (N-1) * alpha^2 = N * sum a_j^2
    # We'll just use alpha = S / sqrt(N) as a rough target for m != 0
    
    for it in range(n_iter):
        A = fft(a, N)
        # Set target magnitude
        target_mag = np.full(N, S / np.sqrt(N))
        target_mag[0] = S  # DC is fixed
        # Blend: new_mag = sqrt(current_mag^2 * target_mag^2)^(1/2) for hermitian
        # Or just: new_A = target_mag * exp(1j * angle(A))
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Ensure Hermitian symmetry
        A_new[0] = S  # real
        for m in range(1, N // 2 + 1):
            if m < N - m:  # avoid double counting
                A_new[N - m] = np.conj(A_new[m])
            elif m == N - m:  # Nyquist
                A_new[m] = np.abs(A_new[m])
        a = np.real(ifft(A_new))[:n]
        a = project_simplex(a, S)
    
    return a

def fista_quartic(a0, S, max_time, verbose=False):
    """FISTA on quartic objective sum(c_k^2) with simplex constraint sum(a)=S, a>=0."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_obj = np.sum(c * c)
    best_a = a.copy()
    
    x_prev = a.copy()
    t = 1.0
    step = 1.0  # initial step size
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # FISTA update
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a + (t_prev - 1.0) / t * (a - x_prev) if it > 1 else a.copy()
        y = np.maximum(y, 0.0)  # keep non-negative for gradient evaluation
        
        # Compute gradient at y
        c_y = fft_convolve_self(y)
        grad = compute_quartic_grad(y, c_y)
        
        # Backtracking line search
        y_sum = y.sum()
        f_y = np.sum(c_y * c_y)
        accepted = False
        for _ in range(20):
            a_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(a_new)
            f_new = np.sum(c_new * c_new)
            # sufficient decrease
            if f_new <= f_y - 0.5 * step * np.sum(grad * (a_new - y)):
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            a_new = project_simplex(y - step * grad, S)
        
        a = a_new
        x_prev = a.copy()
        
        # Update best
        c = fft_convolve_self(a)
        obj = np.sum(c * c)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start

import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_quartic_grad(a, c):
    """
    Gradient of sum(c_k^2) where c = convolve(a,a).
    Result: 4 * d where d_i = sum_m a[m]*c[m+i].
    Computed via FFT cross-correlation.

    N = 1 << (n + len(c) - 1).bit_length()
    A = fft(a, N)
    C = fft(c, N)
    d = np.real(ifft(C * np.conj(A)))

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        if a is None or len(a) == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 0.01:
            return
        obj = compute_obj(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.6f} (n={len(a)}) [{msg}]")

    # === Phase 1: Primary optimization on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())
        obj_init = compute_obj(a1)
        print(f"Initial obj: {obj_init:.6f} (n={n1})")
        update_global(a1, "initial")

        # 1a: GS spectral flattening
        remaining = deadline - time.monotonic()
        t_gs = min(120, remaining * 0.04)
        n_gs = max(100, int(400 * t_gs / 120))
        print(f"\nPhase 1a: GS flatten, n={n1}, ~{n_gs} iters")
        a_gs = gs_flatten(a1, S1, n_iter=n_gs)
        obj_gs = compute_obj(a_gs)
        print(f"  GS obj: {obj_gs:.6f}")
        update_global(a_gs, "GS")

        # 1b: Gradient descent on quartic
        remaining = deadline - time.monotonic()
        t_q = min(900, remaining * 0.40)
        print(f"\nPhase 1b: Quartic GD, n={n1}, t={t_q:.0f}s")
        a_q = grad_desc_quartic(a_gs, S1, t_q, verbose=True)
        obj_q = compute_obj(a_q)
        print(f"  Quartic GD obj: {obj_q:.6f}")
        update_global(a_q, "quartic GD")

        # 1c: LSE polish
        remaining = deadline - time.monotonic()
        t_lse = min(400, remaining * 0.18)
        if t_lse > 30:
            print(f"\nPhase 1c: LSE polish, n={n1}, t={t_lse:.0f}s")
            a_l, _ = lse_polish(a_q, S1, t_lse, beta_init=50, beta_max=80000)
            update_global(a_l, "LSE polish")

    # === Phase 2: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    print(f"\nPhase 2: Multi-length exploration, remaining={remaining:.0f}s")

    # Determine lengths to try
    lengths_to_try = []
    if global_a is not None:
        base_n = len(global_a)
        candidates = [base_n // 2, base_n, base_n * 2, base_n * 4]
        for cn in candidates:
            if 256 <= cn <= 65536 and cn not in lengths_to_try:
                lengths_to_try.append(cn)
    else:
        lengths_to_try = [4096, 8192, 16384]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 80:
            break
        if global_a is not None and len(global_a) == n:
            continue  # already optimized at this length

        S = np.sqrt(2.0 * n)
        t_alloc = min(500, (deadline - time.monotonic()) * 0.25)
        if t_alloc < 60:
            continue

        print(f"\n  Trying n={n}, t={t_alloc:.0f}s")

        # Initialize
        if global_a is not None:
            init = resample_sequence(global_a, n)
            init = np.maximum(init, 0)
        else:
            init = np.ones(n) * 0.5 + 0.3 * np.abs(rng.standard_normal(n))
        init = project_simplex(init, S)

        # Brief GS
        a_gs2 = gs_flatten(init, S, n_iter=100)
        update_global(a_gs2, f"GS n={n}")

        # Quartic GD
        t_q2 = t_alloc * 0.55
        a_q2 = grad_desc_quartic(a_gs2, S, t_q2, verbose=False)
        update_global(a_q2, f"GD n={n}")

        # LSE
        rem = deadline - time.monotonic()
        if rem > 40:
            a_l2, _ = lse_polish(a_q2, S, min(150, rem * 0.3),
                                 beta_init=200, beta_max=200000)
            update_global(a_l2, f"LSE n={n}")

    # === Phase 3: Intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 80:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t3 = remaining - 30
        print(f"\nPhase 3: Intensive refinement, n={n}, t={t3:.0f}s")

        a_curr = global_a.copy()
        n_rounds = 8
        for i in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            t_phase = min(t3 / n_rounds, deadline - time.monotonic() - 15)
            if t_phase < 10:
                break

            if i % 3 == 0:
                # Quartic GD round
                a_curr = grad_desc_quartic(a_curr, S, t_phase, verbose=False)
            elif i % 3 == 1:
                # LSE polish round
                a_curr, _ = lse_polish(a_curr, S, t_phase,
                                       beta_init=500 + i * 200,
                                       beta_max=500000)
            else:
                # GS + brief GD
                a_curr = gs_flatten(a_curr, S, n_iter=50)
                a_curr = grad_desc_quartic(a_curr, S, min(t_phase, 60),
                                           verbose=False)
            update_global(a_curr, f"Phase3 r{i}")

        # Perturbation-based restarts
        remaining = deadline - time.monotonic()
        if remaining > 40:
            n_restarts = min(4, int(remaining // 40))
            print(f"\n  {n_restarts} perturbation restarts")
            for ri in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                # Different noise scales
                scales = [0.005, 0.01, 0.02, 0.05]
                noise_scale = scales[ri % len(scales)] * S / np.sqrt(n)
                a_p = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
                t_r = min(80, (deadline - time.monotonic() - 15) / (n_restarts - ri))
                if t_r < 10:
                    break
                a_p = grad_desc_quartic(a_p, S, t_r * 0.6, verbose=False)
                a_p, _ = lse_polish(a_p, S, t_r * 0.4, beta_init=1000,
                                   beta_max=300000)
                update_global(a_p, f"restart {ri}")

    # === Phase 4: Final LSE squeeze ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print(f"\nPhase 4: Final LSE squeeze, t={remaining-10:.0f}s")
        a_final, _ = lse_polish(global_a, S, remaining - 10,
                               beta_init=5000, beta_max=1000000)
        update_global(a_final, "final LSE")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_best={global_best:.6f}")
    print(f"evaluate_sequence: {ver:.6f}")
    print(f"{'='*60}")
    return final_list
