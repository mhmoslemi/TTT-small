reduces the peak autocorrelation, while FISTA provides fast local convergence.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def solve_lp_direction(f, M, k_active=50):
    """
    Solve the restricted LP to find a direction g that reduces peak autocorrelation.
    
    max sum(g)
    s.t. (f*g)(k) <= M for k in active_lags
         g >= 0
    
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    # Compute full autocorrelation to find active lags
    c = fft_convolve_self(f)
    
    # Select active lags (top-k closest to max)
    active = np.argsort(c)[::-1][:k_active]
    active = np.sort(active)  # keep sorted for efficiency
    
    # Build constraint matrix A for active lags
    # A[i, j] = f[k_i - j] if 0 <= k_i - j < n, else 0
    # where k_i is the i-th active lag
    A = np.zeros((len(active), n))
    for i, k in enumerate(active):
        # For lag k: sum_j f[j] * g[k-j] <= M
        # j ranges over [max(0, k-n+1), min(n-1, k)]
        # Let i = k-j, so i ranges over [max(0, k-n+1), min(n-1, k)]
        # A[i, j] contribution: f[j] * g[k-j] = f[j] * g[i] where i = k-j
        # So for column j (which corresponds to g[j]):
        #   A[i', j] = f[j] if (active[i'] - j) is in [0, n-1]
        # Wait, let me redo this.
        # c[k] = sum_j f[j] * g[k-j]
        # For column j of A (corresponding to variable g[j]):
        #   The contribution to c[k] is f[j] if 0 <= k-j < n, else 0
        #   i.e., A[row_for_k, j] = f[j] if 0 <= k-j < n
    # This is O(k_active * n) which is fine for k_active=50, n=8192
    
    for i, k in enumerate(active):
        # g[j] contributes f[j] to c[k] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[i, j_min:j_max+1] = f[j_min:j_max+1]
    
    # Solve: max sum(g) s.t. A @ g <= M, g >= 0
    # In linprog form: min -sum(g) s.t. A @ g <= M, -g <= 0
    c_obj = -np.ones(n)
    A_ub = A  # A @ g <= M
    b_ub = np.full(len(active), M)
    # g >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return f (normalized)
        return f.copy()


def run_lp_iteration(f_init, max_time, rng, k_active=20, t_blend=0.3):
    """
    Run LP-based active set iterations.
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())
    
    start = time.monotonic()
    best_f = f.copy()
    best_conv = fft_convolve_self(f).max()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(f)
        M = c.max()
        
        if M < best_conv:
            best_conv = M
            best_f = f.copy()
        
        # Solve LP for direction
        try:
            g = solve_lp_direction(f, M, k_active=k_active)
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                g = f.copy()
                g_sum = f.sum()
            g = g * (S / g_sum)
        except Exception as e:
            print("  LP failed: %s" % str(e))
            break
        
        # Blend with line search
        best_t = t_blend
        best_obj = M
        for t_candidate in [t_blend, t_blend*0.5, t_blend*0.25, t_blend*1.5, t_blend*2.0]:
            f_trial = (1 - t_candidate) * f + t_candidate * g
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial * (S / f_trial.sum())
            c_trial = fft_convolve_self(f_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_t = t_candidate
        
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f = f * (S / f.sum())
        
        if it % 5 == 0:
            print("  LP iter %d: max_conv=%.8f, obj=%.6f" % (it, f_sum_c := fft_convolve_self(f).max(), 2*n*f_sum_c/S**2))
        
        # Reduce k_active over time (focus more precisely)
        k_active = max(5, k_active - it // 20)
    
    return best_f, best_conv


def run_fista_adaptive(a_init, max_time, rng):
    """
    FISTA with adaptive beta and backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    y = a.copy()
    t_mom = 1.0
    beta = 20.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive beta: increase when progressing, decrease when stuck
        if stuck < 100:
            beta = min(beta * 1.01, 5000)
        elif stuck < 500:
            beta = max(beta * 0.99, 10)
        else:
            beta = max(beta * 0.95, 5)
        
        # Gradient
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = 2.0 / beta  # initial step
        
        # Simple backtracking: try a few step sizes
        for attempt in range(5):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max * (1 + 1e-10):
                break
            step *= 0.5
        
        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart momentum periodically or when stuck
        if stuck > 2000 or it % 10000 == 0:
            t_mom = 1.0
        
        # Add structured noise when very stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = 0.002 * S / np.sqrt(n)
            F = fft(rng.normal(0, 1, n))
            cutoff = max(3, n // 20)
            F[cutoff:-cutoff] *= 0.1
            noise = np.real(ifft(F))[:n]
            noise = noise / (np.linalg.norm(noise) + 1e-15) * noise_scale
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            stuck = 0
    
    return best_a, best_conv


def run_final_polish(a_init, max_time, rng):
    """
    Final polishing with very high beta and exact subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Use top-k with increasing k as we get stuck
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 3
        elif stuck < 1000:
            k = min(8, 3 + stuck // 100)
        else:
            k = min(32, 8 + stuck // 200)
        
        grad = topk_gradient(y, c, k, power=1.0)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Conservative step size
        step = 0.5 * S / (gnorm + 1e-15)
        step = min(step, S / (2 * gnorm + 1e-15))
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_conv

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LP-based active set iteration (solving restricted convex subproblems
to find improvement directions) combined with adaptive LSE-FISTA and
spectral initialization for diverse starting points.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def build_lp_matrix(f, active_lags):
    """Build constraint matrix for restricted LP.
    A[row, i] = f[k - i] if 0 <= k-i < n, else 0, where k = active_lags[row].
    """
    n = len(f)
    k_act = len(active_lags)
    A = np.zeros((k_act, n))
    for row, k in enumerate(active_lags):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            j_start = k - i_max
            j_end = k - i_min + 1
            A[row, i_min:i_max + 1] = f[j_start:j_end][::-1]
    return A


def solve_lp_direction(f, M, active_lags):
    """Solve restricted LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, active_lags)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': max(5.0, 30.0)})
    if result.success and result.x is not None:
        return np.maximum(result.x, 0.0)
    return None


def run_lp_iterations(f_init, max_time, rng, initial_k=30, min_k=5):
    """
    LP-based active set iterations.
    At each step: solve restricted LP to find direction g, blend with current f.
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())

    start = time.monotonic()
    best_f = f.copy()
    best_conv = fft_convolve_self(f).max()
    it = 0
    k_active = initial_k
    t_blend = 0.4
    consecutive_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        c = fft_convolve_self(f)
        M = c.max()

        if M < best_conv - 1e-12:
            best_conv = M
            best_f = f.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Adapt k_active: start broad, narrow down
        target_k = max(min_k, initial_k - it // 30)
        k_active = target_k

        # Select active lags
        thresholds = np.sort(c)[::-1][min(k_active, len(c) - 1)]
        active_lags = np.where(c >= thresholds * 0.999)[0]
        if len(active_lags) > k_active:
            active_lags = np.argsort(c)[::-1][:k_active]
        active_lags = np.sort(active_lags)

        # Solve LP
        g = solve_lp_direction(f, M, active_lags)
        if g is None:
            # Fallback: use subgradient direction
            grad = topk_gradient(f, c, k_active, power=1.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (gnorm + 1e-15)
            g = project_simplex(f - step * grad, S)
        else:
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                g = f.copy()
                g_sum = S
            g = g * (S / g_sum)

        # Line search over blend factor
        best_t = 0.1
        best_trial_conv = np.inf
        t_candidates = [t_blend, t_blend * 0.5, t_blend * 0.25,
                        t_blend * 1.5, t_blend * 2.0, 0.1, 0.05]
        for t_cand in t_candidates:
            if t_cand > 1.0 or t_cand < 0.01:
                continue
            f_trial = (1.0 - t_cand) * f + t_cand * g
            f_trial = np.maximum(f_trial, 0.0)
            s_trial = f_trial.sum()
            if s_trial < 1e-10:
                continue
            f_trial = f_trial * (S / s_trial)
            c_trial = fft_convolve_self(f_trial)
            if c_trial.max() < best_trial_conv:
                best_trial_conv = c_trial.max()
                best_t = t_cand

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f * (S / f.sum())

        # Reduce blend factor if stuck
        if consecutive_no_improve > 20:
            t_blend = max(0.05, t_blend * 0.8)
        elif consecutive_no_improve == 0:
            t_blend = min(0.5, t_blend * 1.1)

        if it % 10 == 0:
            obj_now = 2.0 * n * fft_convolve_self(f).max() / (S * S)
            print("    LP iter %d: max_conv=%.10f obj=%.8f k=%d t=%.3f" %
                  (it, fft_convolve_self(f).max(), obj_now, k_active, t_blend))

    return best_f, best_conv


def run_fista_adaptive(a_init, max_time, rng, beta_start=20.0):
    """
    FISTA with adaptive beta, backtracking, and structured restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    beta = beta_start
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta
        if stuck < 50:
            beta = min(beta * 1.02, 5000.0)
        elif stuck < 500:
            beta = max(beta * 0.98, 5.0)
        else:
            beta = max(beta * 0.93, 3.0)

        # Gradient with LSE
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Step with backtracking
        step = 2.0 / max(beta, 1.0)
        x_new = None
        for _ in range(8):
            candidate = project_simplex(y - step * grad, S)
            c_cand = fft_convolve_self(candidate)
            if c_cand.max() <= cur_max + 1e-10:
                x_new = candidate
                break
            step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 3000 or it % 20000 == 0:
            t_mom = 1.0

        # Structured perturbation when very stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                F_noise = fft(rng.normal(0, 1.0, n))
                cutoff = max(3, n // 30)
                F_noise[cutoff:-cutoff] *= 0.05
                noise = np.real(ifft(F_noise))[:n]
                nrm = np.linalg.norm(noise)
                if nrm > 1e-15:
                    y = project_simplex(y + noise / nrm * noise_scale, S)
                    t_mom = 1.0
                    stuck = 0
                    last_improve = it

    return best_a, best_conv

def spectral_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    # Random phases on FFT grid
    fsize = 1 << (2 * n - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fsize)
    # Ensure Hermitian symmetry for real signal
    spectrum = np.exp(1j * phases)
    spectrum[0] = 1.0
    if fsize % 2 == 0:
        spectrum[fsize // 2] = 1.0
    # Make Hermitian
    for i in range(1, fsize // 2):
        spectrum[fsize - i] = np.conj(spectrum[i])
    sig = np.real(ifft(spectrum)[:n])
    sig = np.maximum(sig, 0.0)
    if sig.sum() < 1e-10:
        sig = np.ones(n) / n
    return sig


def cosine_init(n, freq=1.0, asym=0.0):
    """Cosine-based initialization with optional asymmetry."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
    if asym != 0.0:
        a = a * (1.0 + asym * np.sin(2 * np.pi * t))
    a = np.maximum(a, 0.005)
    return a

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = cosine_init(n0, freq=1.0)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Multi-start exploration (10% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.10
    print("\n=== Phase 1: Multi-start (budget %.0fs) ===" % p1_budget)
    t0 = time.monotonic()

    n1 = n0
    S1 = np.sqrt(2.0 * n1)
    inits = [global_best_a.copy()]

    # Add diverse inits
    for freq in [0.5, 1.5, 2.0]:
        a_init = cosine_init(n1, freq=freq)
        a_init = a_init * (S1 / a_init.sum())
        inits.append(a_init)

    a_init = spectral_init(n1, rng)
    a_init = a_init * (S1 / a_init.sum())
    inits.append(a_init)

    # Random smooth
    raw = rng.random(n1)
    F_r = fft(raw)
    cutoff = max(3, n1 // 100)
    F_r[cutoff:-cutoff] *= 0.01
    smooth = np.real(ifft(F_r))[:n1]
    smooth = np.maximum(smooth, 0.005)
    smooth = smooth * (S1 / smooth.sum())
    inits.append(smooth)

    time_per = p1_budget / len(inits)
    for i, init in enumerate(inits):
        if time.monotonic() > deadline - (p1_budget * 0.1):
            break
        a_m, _ = run_fista_adaptive(init, min(time_per, 25.0), rng, beta_start=15.0)
        obj_m = compute_obj(a_m)
        if obj_m < global_best_obj:
            global_best_obj = obj_m
            global_best_a = a_m.copy()
            print("  Multi-start %d: obj=%.8f (improved!)" % (i, obj_m))
        else:
            print("  Multi-start %d: obj=%.8f" % (i, obj_m))

    print("  Phase 1 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # === Phase 2: LP-based active set on n=8192 (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.30
        print("\n=== Phase 2: LP active set n=%d (budget %.0fs) ===" % (n0, p2_budget))
        t0 = time.monotonic()

        a2, _ = run_lp_iterations(global_best_a, p2_budget, rng,
                                  initial_k=40, min_k=5)
        obj2 = compute_obj(a2)
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()
            print("  LP improved: obj=%.8f" % obj2)
        else:
            print("  LP result: obj=%.8f (no improvement)" % obj2)

        print("  Phase 2 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # === Phase 3: FISTA polishing (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.25
        print("\n=== Phase 3: FISTA polish (budget %.0fs) ===" % p3_budget)
        t0 = time.monotonic()

        a3, _ = run_fista_adaptive(global_best_a, p3_budget, rng, beta_start=30.0)
        obj3 = compute_obj(a3)
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        print("  Phase 3 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # === Phase 4: LP round 2 with tighter focus (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: LP round 2 (budget %.0fs) ===" % p4_budget)
        t0 = time.monotonic()

        a4, _ = run_lp_iterations(global_best_a, p4_budget, rng,
                                  initial_k=15, min_k=3)
        obj4 = compute_obj(a4)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()
            print("  LP round 2 improved: obj=%.8f" % obj4)
        else:
            print("  LP round 2: obj=%.8f" % obj4)

        print("  Phase 4 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # === Phase 5: Top-k final polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Top-k polish (budget %.0fs) ===" % p5_budget)
        t0 = time.monotonic()

        a5, _ = run_topk_polish(global_best_a, p5_budget, rng)
        obj5 = compute_obj(a5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        print("  Phase 5 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # === Phase 6: L-BFGS final squeeze (remaining time, min 10s) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t0 = time.monotonic()

        n = len(global_best_a)
        S = np.sqrt(2.0 * n)
        a = global_best_a.copy()
        a = a * (S / a.sum())

        beta_lbfgs = 300.0

        def lbfgs_obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta_lbfgs * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj_val = cm + np.log(e.sum()) / beta_lbfgs
            sum_x = x.sum()
            obj_val += 20.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * 20.0 * (sum_x - S)
            return obj_val, grad

        try:
            result = scipy_minimize(
                lbfgs_obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': 30, 'maxls': 20, 'ftol': 1e-14,
                         'gtol': 1e-10, 'iprint': -1}
            )
            a_new = np.clip(result.x, 0, 1000)
            a_new = a_new * (S / max(a_new.sum(), 1e-15))
            obj_new = compute_obj(a_new)
            if obj_new < global_best_obj:
                global_best_obj = obj_new
                global_best_a = a_new.copy()
                print("  L-BFGS improved: obj=%.8f" % obj_new)
            else:
                print("  L-BFGS: obj=%.8f" % obj_new)
        except Exception as e:
            print("  L-BFGS failed: %s" % str(e))

        print("  Phase 6 done: best obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t0))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using uniform fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
