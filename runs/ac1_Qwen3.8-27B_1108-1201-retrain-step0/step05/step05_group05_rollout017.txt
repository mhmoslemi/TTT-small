"""

import numpy as np
import time

from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(u * idx > (cssv - S))[0]
    if len(rho) == 0:
        theta = (cssv[-1] - S) / n
    else:
        theta = (cssv[rho[-1]] - S) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def normalize_simplex(a, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))

def grad_excess(a, c, target, alpha=2.0, top_k=32):
    """
    Compute gradient of sum((c_k - target)^alpha_+) restricted to top-k.
    Returns (gradient, objective_value_of_excess).
    """
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    excess = np.maximum(c[idx] - target, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if alpha == 1.0:
        w = excess.copy()
    elif alpha == 2.0:
        w = 2.0 * excess
    else:
        w = alpha * excess ** (alpha - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** alpha)
    return grad, obj

def backtrack_line_search(a, S, grad, max_iter=50):
    """
    Backtracking line search for minimizing max(conv) with simplex constraint.
    Returns (new_a, new_max, step_taken).
    """
    n = len(a)
    C0 = fft_convolve_self(a)
    m0 = C0.max()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy(), m0, 0.0

    # Initial step
    step0 = 1.0 / gnorm
    step = step0
    best_a = a.copy()
    best_m = m0
    best_step = 0.0

    for _ in range(max_iter):
        a_try = project_simplex(a - step * grad, S)
        if np.all(a_try == a):
            step *= 0.5
            continue
        C_try = fft_convolve_self(a_try)
        m_try = C_try.max()
        if m_try < m0 - 1e-12 * step * gnorm:
            return a_try, m_try, step
        if m_try < best_m:
            best_m = m_try
            best_a = a_try.copy()
            best_step = step
        step *= 0.5
        if step < 1e-10:
            break

    if best_step > 0:
        return best_a, best_m, best_step
    return a.copy(), m0, 0.0

    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 3.0 * (freqs / fsize)
    noise = rng.normal(0, scale, fsize) * weight
    new_A = mag * np.exp(1j * (phase + noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def main_optimize(a_init, S, n, max_time, rng, use_lp=True):
    """
    Main optimization loop combining excess gradient, LSE, and LP directions
    with backtracking line search.
    """
    a = normalize_simplex(a_init, S)
    C0 = fft_convolve_self(a)
    best_m = C0.max()
    best_a = a.copy()
    start = time.monotonic()

    it = 0
    last_improve = 0
    stuck_count = 0
    target = C0.max() * 0.95  # Target level for excess

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time

        C = fft_convolve_self(a)
        m = C.max()
        if m < best_m - 1e-10:
            best_m = m
            best_a = a.copy()
            last_improve = it
            stuck_count = 0
            # Tighten target
            target = max(C.mean() + 0.1 * (m - C.mean()), best_m * 0.92)

        if it - last_improve > 500:
            stuck_count += 1
            # Stuck: try LP or kick
            if use_lp and stuck_count % 5 == 1 and time.monotonic() - start < max_time - 10:
                delta = lp_direction(a, C, n_active=min(6, n // 100))
                if delta is not None and np.linalg.norm(delta) > 1e-12:
                    # Try LP direction with line search
                    a_try, m_try, _ = backtrack_line_search(a, S, -delta, max_iter=20)
                    if m_try < m:
                        a = a_try
                        continue
            # Spectral kick
            kick_scale = 0.01 + 0.03 * np.exp(-stuck_count * 0.5)
            a = spectral_kick(a, S, rng, scale=kick_scale)
            last_improve = it
            stuck_count = 0
            continue

        # Choose gradient method based on progress
        if progress < 0.3:
            # Early: use top-k for aggressive peak shaving
            k = max(4, int(32 * (1 - progress * 2)))
            grad = grad_topk(a, C, k=k)
        elif progress < 0.7:
            # Mid: excess gradient with adaptive target
            alpha = 2.0 + 3.0 * (progress / 0.4)
            top_k = max(8, int(64 * (1 - progress)))
            grad, _ = grad_excess(a, C, target=target, alpha=alpha, top_k=top_k)
        else:
            # Late: LSE with increasing beta for precision
            beta = 50.0 + 500.0 * ((progress - 0.7) / 0.3)
            grad = grad_lse(a, C, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        a_new, m_new, step = backtrack_line_search(a, S, grad, max_iter=40)
        if m_new < m:
            a = a_new
        else:
            # Tiny random nudge
            a = project_simplex(a + rng.normal(0, 1e-5 * S / n, n), S)

    return best_a, best_m


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None and len(given_seq) > 0:
        inits.append(np.clip(given_seq, 0, 1000).copy())

    # Various windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2 / 0.1) + 0.01)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(6 * np.pi * tt + 0.5)
                            + 0.2 * np.cos(12 * np.pi * tt + 1.2), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(4 * np.pi * tt)
                            + 0.15 * np.cos(10 * np.pi * tt + 0.3), 0.01))

    # Asymmetric (important!)
    inits.append(np.maximum(np.exp(-3.0 * tt) + 0.3 * np.exp(-3.0 * (1 - tt)), 0.01))
    inits.append(np.maximum(tt * np.sin(np.pi * tt) + 0.05, 0.01))
    inits.append(np.maximum((1 - tt) * np.sin(np.pi * tt) + 0.05, 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: diverse init -> adaptive peak-shaving with backtracking
    -> LP-guided refinement -> high-precision LSE polish.
    Tries multiple n values.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" %
                  (len(given), given.sum(), obj_value(given)))
    except Exception:
        pass

    # Try multiple n values
    n_candidates = [8192, 16384]
    if budget_s > 2000:
        n_candidates.append(4096)

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_global(a, val):
        nonlocal global_best, global_a, global_n
        n = len(a)
        s = a.sum()
        if s < 0.01:
            return
        obj = 2.0 * n * val / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = n
            print("  NEW GLOBAL BEST: obj=%.8f (n=%d, max_b=%.6f)" % (obj, n, val))

    # Distribute time among n values
    time_per_n = (budget_s - 60) / len(n_candidates)

    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        t_alloc = min(time_per_n, remaining - 30)
        if t_alloc < 60:
            break

        print("\n=== Optimizing n=%d, time=%.0fs ===" % (n, t_alloc))
        S = np.sqrt(2.0 * n)

        # Generate inits
        inits = generate_inits(n, rng, given_seq=given)
        inits = [normalize_simplex(x, S) for x in inits]

        # Phase 1: Quick screening (top 4 inits, short time each)
        n_screen = min(4, len(inits))
        screen_time = t_alloc * 0.2
        candidates = []
        for i in range(n_screen):
            if time.monotonic() > deadline - 60:
                break
            t_i = screen_time / (n_screen - i) - 5
            if t_i < 5:
                break
            a_opt, val = main_optimize(inits[i], S, n, t_i, rng, use_lp=False)
            candidates.append((val, a_opt.copy()))
            s_opt = a_opt.sum()
            obj_i = 2.0 * n * val / (s_opt * s_opt)
            print("  Screen %d: obj=%.6f" % (i, obj_i))
            update_global(a_opt, val)

        candidates.sort(key=lambda x: x[0])

        # Phase 2: Full optimization from best candidates (with LP)
        n_full = min(3, len(candidates))
        full_time_total = t_alloc * 0.55
        for i in range(n_full):
            if time.monotonic() > deadline - 90:
                break
            t_full = full_time_total / (n_full - i) - 10
            if t_full < 30:
                break
            print("  Full opt from candidate %d (max_b=%.6f), t=%.0fs" %
                  (i, candidates[i][0], t_full))
            a_opt, val = main_optimize(candidates[i][1], S, n, t_full, rng, use_lp=True)
            update_global(a_opt, val)

        # Phase 3: High-precision polish on the best
        if time.monotonic() < deadline - 60 and candidates:
            a_best = global_a if global_n == n else candidates[0][1]
            t_polish = min(t_alloc * 0.2, deadline - time.monotonic() - 20)
            if t_polish > 20:
                print("  Polish phase, t=%.0fs" % t_polish)
                # LSE with very high beta
                a_p, val_p = _highbeta_polish(a_best, S, n, t_polish, rng)
                update_global(a_p, val_p)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nFinal verification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(8192) * (np.sqrt(2.0 * 8192) / 8192)
            final_list = [float(x) for x in global_a]
    else:
        n_fallback = 8192
        global_a = np.ones(n_fallback) * (np.sqrt(2.0 * n_fallback) / n_fallback)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Best obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return final_list


def _highbeta_polish(a_init, S, n, max_time, rng):
    """Very high beta LSE polish with FISTA momentum."""
    a = normalize_simplex(a_init, S)
    C0 = fft_convolve_self(a)
    best_m = C0.max()
    best_a = a.copy()
    start = time.monotonic()

    y = a.copy()
    t_mom = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = 100.0 + 2000.0 * progress

        C = fft_convolve_self(y)
        if C.max() < best_m - 1e-12:
            best_m = C.max()
            best_a = y.copy()

        grad = grad_lse(y, C, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a, best_m
