import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    - Momentum decays near convergence
    - Periodic structured restarts

    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 10.0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (y.sum() ** 2)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:

        # Adaptive beta: increase until LSE is tight

            beta = max(10.0, min(beta * 1.02, 500.0))

        else:

        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        max_backtrack = 10

            break  # Skip full backtracking for speed

        x_new = project_simplex(y - step * grad, S)

        # Momentum with decay near convergence

        if stuck_count > 2000:

            a = y.copy()
            t_mom = 1.0

        if it - last_log >= log_interval:
            elapsed = time.monotonic() - start
            print("  FISTA: it=%d, obj=%.8f, beta=%.0f, stuck=%d (t=%.0fs)" %
                  (it, best_obj, beta, stuck, elapsed))
            last_log = it

    return best_a, best_obj


def run_fista_highbeta(a_init, max_time, rng, beta_start=5000.0, beta_end=50000.0):
    """FISTA with very high beta for final refinement (near-exact subgradient)."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        beta = min(beta * 1.003, beta_end * 3)

        # Use LSE with high beta (close to exact subgradient)
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (2.0 * beta * w.max() + 1e-10)
        if stuck > 1000:
            step *= max(0.3, 1.0 - 0.0003 * stuck)

        x_new = project_simplex(y - step * grad, S)

        if stuck > 0 and cur_obj > best_obj * (1 + 1e-10):
            t_mom = 1.0
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = min(t_new, 20.0)

        if stuck > 4000 and it % 500 == 0:
            y = best_a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(np.maximum(y, 0), S)
            a = y.copy()
            t_mom = 1.0

        if it % 20000 == 0:
            print("  HighBeta FISTA: it=%d, obj=%.8f, beta=%.0f (t=%.0fs)" %
                  (it, best_obj, beta, time.monotonic() - start))

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with multiple beta levels and sum penalty."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    beta_levels = [100, 500, 2000, 10000, 30000]
    for beta in beta_levels:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 15:
            break

        c = fft_convolve_self(a)
        cm = c.max()

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w_local = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += 0.5 * 0.1 * (sum_x - S) ** 2 / S
            grad = conv_gradient(x, w_local)
            grad += 0.1 * (sum_x - S) / S
            return obj, grad

        t0 = time.monotonic()
        max_it = min(500, int(remaining * 20))
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': max_it, 'maxls': 40, 'ftol': 1e-15,
                     'gtol': 1e-12, 'iprint': -1, 'maxfun': max_it * 3}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        print("  L-BFGS beta=%d: obj=%.8f (t=%.1fs, nit=%d)" %
              (beta, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_subgradient_polish(a_init, max_time, rng):
    """Subgradient method with expanding active set for final polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Expanding active set tolerance
        if stuck < 100:
            tol = 0.0
        elif stuck < 500:
            tol = 0.0005
        elif stuck < 2000:
            tol = 0.002
        elif stuck < 5000:
            tol = 0.005
        else:
            tol = min(0.015, 0.001 + 0.000002 * stuck)

        active = np.where(c >= cur_max * (1.0 - tol))[0]
        k = len(active)

        if k == 0:
            active = np.array([np.argmax(c)])
            k = 1

        if k <= 4:
            w = np.zeros(len(c))
            w[active] = 1.0 / k
        else:
            # Weight by proximity to max
            margins = c[active] - cur_max * (1.0 - tol)
            margins = np.maximum(margins, 1e-10)
            w_active = margins / margins.sum()
            w = np.zeros(len(c))
            w[active] = w_active

        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polak-ribiere style step
        step = S / (gnorm * (1.0 + 0.0005 * it))
        step = min(step, 3.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = min(t_new, 15.0)

        if stuck > 5000 and it % 1000 == 0:
            eps = 0.0005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, eps, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it

        if it % 30000 == 0:
            print("  SubGrad: it=%d, obj=%.8f, |active|=%d (t=%.0fs)" %
                  (it, best_obj, k, time.monotonic() - start))

    return best_a, best_obj


def generate_alt_starts(n, rng, S, current_best):
    """Generate alternative starting points for multi-start."""
    starts = []

    # Current best
    starts.append(current_best.copy())

    # Modulated cosines
    t = np.arange(n, dtype=np.float64)
    for freq in [3, 5, 7, 11]:
        for amp in [0.3, 0.5]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
            a = np.maximum(a, 0.01)
            a = a * (S / a.sum())
            starts.append(a.copy())

    # Smoothed random
    for _ in range(2):
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(5, n // 200)
        F[cutoff:-cutoff] *= 0.01
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.1)
        smooth = smooth * (S / smooth.sum())
        starts.append(smooth.copy())

    return starts


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA (monotone beta), L-BFGS-B, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except (NameError, Exception):
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.01)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    S = np.sqrt(2.0 * n0)
    print("Starting optimization: n=%d, initial obj=%.8f" % (n0, global_best_obj))

    # === Phase 1: Main FISTA with monotone beta (55% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.55
    print("\n=== Phase 1: FISTA main (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    a1, obj1 = run_fista_main(global_best_a, p1_budget, rng,
                               beta_start=10.0, beta_end=20000.0)
    print("  FISTA main: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: L-BFGS-B polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: L-BFGS-B polish (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, obj2 = run_lbfgs_polish(global_best_a, p2_budget, rng)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: High-beta FISTA refinement (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.15
        print("\n=== Phase 3: High-beta FISTA (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, obj3 = run_fista_highbeta(global_best_a, p3_budget, rng,
                                       beta_start=5000.0, beta_end=50000.0)
        print("  HighBeta: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Subgradient polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = remaining * 0.10
        print("\n=== Phase 4: Subgradient polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, obj4 = run_subgradient_polish(global_best_a, p4_budget, rng)
        print("  SubGrad: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Multi-start exploration (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 5: Multi-start (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()

        starts = generate_alt_starts(n0, rng, S, global_best_a)
        time_per = remaining / len(starts)

        for idx, st in enumerate(starts):
            if time.monotonic() > deadline:
                break
            if time_per < 5:
                # Very short: just one FISTA burst
                alloc = max(3.0, remaining - (time.monotonic() - t5))
            else:
                alloc = min(time_per, 30.0)

            a_m, obj_m = run_fista_main(st, alloc, rng,
                                         beta_start=50.0, beta_end=10000.0)
            if obj_m < global_best_obj:
                global_best_obj = obj_m
                global_best_a = a_m.copy()
                print("  Multi-start %d: obj=%.8f (IMPROVED!)" % (idx, obj_m))
            else:
                print("  Multi-start %d: obj=%.8f" % (idx, obj_m))

    # === Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Final squeeze (budget %.0fs) ===" % remaining)
        a_f, obj_f = run_lbfgs_polish(global_best_a, remaining * 0.9, rng)
        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_best_a = a_f.copy()
        print("  Final: obj=%.8f" % global_best_obj)

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected input, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
