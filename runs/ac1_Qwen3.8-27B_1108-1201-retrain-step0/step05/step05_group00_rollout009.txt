def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv of length 2n-1."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_val_and_weights(c, beta):
    """Compute LSE value and softmax weights for convolution vector c."""
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        return c_max, weights
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        return c_max, weights
    weights = exp_bc / total
    lse = c_max + np.log(total) / beta
    return lse, weights

def lbfgs_two_loop(s_hist, y_hist, g):
    """Standard L-BFGS two-loop recursion for search direction."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        sigma = np.dot(y_i, y_i)
        beta_i = np.dot(s_i, y_i) / sigma
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize a fraction of FFT phases while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    Nf = max(N, 1 << (2 * n - 1).bit_length())
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    if frac >= 1.0:
        new_phase = rng.uniform(0, 2 * np.pi, Nf)
    else:
        rand_idx = rng.choice(Nf, size=n_rand, replace=False)
        new_phase = old_phase.copy()
        new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

def make_init_fourier(n, rng, N):
    """Generate from controlled magnitude spectrum."""
    Nf = 1 << (2 * n - 1).bit_length()
    freqs = np.arange(Nf)
    # Elliptical magnitude spectrum (flat center, decay at edges)
    norm_f = freqs.astype(float) / Nf
    P = 1.0 / (1.0 + 0.5 * (np.abs(norm_f - 0.5) * 4.0) ** 2)
    P[0] = float(n) * 0.5
    phase = rng.uniform(0, 2 * np.pi, Nf)
    # Enforce conjugate symmetry for real signal
    for k in range(1, Nf // 2 + 1):
        P[k] = P[Nf - k]
        phase[k] = -phase[Nf - k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.05

def deflation_init(n, S, N, n_iters=1000, rng=None):
    """Reduce peak from all-ones by transferring mass from high-contrib to low-contrib indices."""

        rng = np.random.default_rng(42)

        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        if np.max(contrib) < 1e-10:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:

        transfer = min(0.05 * a[i_max], 0.5 * a[i_max])
        if transfer < 1e-10:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def dual_gradient_polish(a_init, n, S, N, max_time, rng, epsilon=0.05):
    """
    Primal-dual polishing exploiting KKT conditions.
    At optimum: sum_{k active} lambda_k * 2*a[k-j] = mu for all j in support.
    We iteratively correct violations of this condition.
    """
    a = safe_normalize(a_init, n, S)
    fsize = 1 << (3 * n - 2).bit_length()
    best_a = a.copy()
    best_peak = np.inf
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_peak:
            best_peak = c_max
            best_a = a.copy()

        # Identify active set
        threshold = c_max - epsilon * c_max
        active = c >= threshold
        if not np.any(active):
            active = c >= c_max - 1e-6
        n_active = active.sum()
        if n_active < 1:
            break

        # Dual weights (proportional to excess)
        excess = np.maximum(c[active] - (c_max - epsilon * c_max), 0)
        if excess.sum() < 1e-30:
            lam = np.ones(n_active) / n_active
        else:
            lam = excess / excess.sum()

        # Build full weight vector
        weights = np.zeros(2 * n - 1)
        weights[active] = lam

        # Compute dual gradient: g[j] = sum_k weights[k] * 2*a[k-j]
        W = fft(weights, fsize)
        A = fft(a, fsize)
        dual_grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]

        # On support, dual_grad should be constant (= mu)
        support = a > 1e-10
        if support.sum() < 2:
            break
        mu = np.mean(dual_grad[support])
        residual = dual_grad[support] - mu

        # Direction: reduce variance of dual_grad on support
        # d[j] = -residual for j in support, 0 otherwise
        direction = np.zeros(n)
        direction[support] = -residual
        # Make it sum-zero on support to preserve sum(a) = S
        direction[support] -= np.mean(direction[support])

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Line search: try a few step sizes
        step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005]
        improved = False
        for ss in step_sizes:
            alpha = ss * S / dir_norm
            candidate = project_simplex(a + alpha * direction, S)
            c_cand = compute_conv(candidate, N)
            if c_cand.max() < c_max:
                a = candidate
                c_max = c_cand.max()
                improved = True
                if c_max < best_peak:
                    best_peak = c_max
                    best_a = a.copy()
                break

        if not improved:
            # Try the LSE gradient instead
            beta = 50.0
            _, lse_g, _, _ = lse_objective_and_grad(a, N, beta)
            gn = np.linalg.norm(lse_g)
            if gn < 1e-18:
                break
            for ss in [0.01, 0.005, 0.002]:
                alpha = ss * S / gn
                candidate = project_simplex(a - alpha * lse_g, S)
                c_cand = compute_conv(candidate, N)
                if c_cand.max() < c_max:
                    a = candidate
                    c_max = c_cand.max()
                    if c_max < best_peak:
                        best_peak = c_max
                        best_a = a.copy()
                    break

        # Occasionally restart
        if it % 200 == 0 and rng.random() < 0.15:
            a = spectral_phase_restart(a, N, rng, frac=0.15)

    return best_a, best_peak


def toppk_nesterov(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=500):
    """Nesterov-accelerated top-k subgradient method."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    fsize = 1 << (3 * n - 2).bit_length()
    start_time = time.monotonic()
    best_a = a.copy()
    best_peak = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_peak:
            best_peak = cur
            best_a = y.copy()

        # Top-k weights
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            weights_full[int(k)] = top_vals[i] / total

        W = fft(weights_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(np.conj(W) * Af))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 1.5 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        if it % 20 == 0:
            c_x = compute_conv(x_new, N)
            if c_x.max() < best_peak:
                best_peak = c_x.max()
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 1000 == 0 and rng.random() < 0.1:
            y = spectral_phase_restart(y, N, rng, frac=0.2)
            t = 1.0

    return best_a, best_peak


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Strategy: multi-size exploration, L-BFGS on LSE, dual-gradient polishing,
    and Nesterov top-k subgradient.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Try to use given sequence as a starting point
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and given.size <= 65536 and np.all(given >= 0):
            given_seq = given
            print("Using given sequence of length %d as starting point" % len(given))
    except Exception:
        pass

    # Decide on sizes to try
    if given_seq is not None:
        n_primary = len(given_seq)
        n_list = [n_primary]
        if n_primary <= 8192:
            n_list.append(16384)
        elif n_primary <= 16384:
            n_list.append(8192)
    else:
        n_primary = 8192
        n_list = [8192, 16384]

    print("Sizes to try: %s" % n_list)

    global_best = np.inf
    global_a = None
    global_n = n_primary

    def score_from_peak(peak, n):
        S = np.sqrt(2.0 * n)
        return float(2 * n * peak / (S * S))

    def update_best(a_candidate, n, label=""):
        nonlocal global_best, global_a, global_n
        S = np.sqrt(2.0 * n)
        a_cand = safe_normalize(a_candidate, n, S)
        N = 1 << (2 * n - 1).bit_length()
        c = compute_conv(a_cand, N)
        peak = c.max()
        sc = score_from_peak(peak, n)
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            global_n = n
            if label:
                print("  [NEW BEST] n=%d peak=%.6f score=%.6f (%s)" % (n, peak, sc, label))
        return sc

    # === Phase 1: Quick multi-size exploration ===
    print("=== Phase 1: Multi-size exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.12)

    for n in n_list:
        if time.monotonic() > deadline - 600:
            break
        S = np.sqrt(2.0 * n)
        N = 1 << (2 * n - 1).bit_length()
        print("  Exploring n=%d (S=%.2f, N=%d)" % (n, S, N))

        t0 = time.monotonic()
        per_n = phase1_budget / len(n_list)
        rem = max(5.0, per_n - (time.monotonic() - t0))

        # Generate initializations
        inits = []
        if given_seq is not None and len(given_seq) == n:
            inits.append(("given", safe_normalize(given_seq, n, S)))
        inits.append(("rand", safe_normalize(make_init_rand(n, rng), n, S)))
        inits.append(("chirp", safe_normalize(make_init_chirp(n, rng), n, S)))
        inits.append(("chirp2", safe_normalize(make_init_chirp2(n, rng), n, S)))
        inits.append(("fourier", safe_normalize(make_init_fourier(n, rng, N), n, S)))
        inits.append(("deflation", deflation_init(n, S, N, min(500, n // 4), rng)))

        per_init = max(2.0, rem / len(inits))
        for name, a_init in inits:
            if time.monotonic() > deadline - 600:
                break
            a_l, sc_l = lbfgs_lse_optimize(a_init, n, S, N, min(per_init, 40.0), rng,
                                            beta_init=8.0, beta_max=100.0, m_lbfgs=10)
            update_best(a_l, n, name)

    if global_a is None:
        print("  No valid result from Phase 1, using all-ones fallback")
        global_n = 8192
        S = np.sqrt(2.0 * global_n)
        global_a = np.ones(global_n) * S / global_n
        N = 1 << (2 * global_n - 1).bit_length()
        c = compute_conv(global_a, N)
        global_best = score_from_peak(c.max(), global_n)

    print("Phase 1 best: score=%.6f (n=%d)" % (global_best, global_n))

    # === Phase 2: Focused optimization on best size ===
    print("=== Phase 2: Focused L-BFGS optimization ===")
    t2 = time.monotonic()
    n = global_n
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()
    phase2_budget = min(1200.0, (deadline - t2) * 0.50)

    a_curr = global_a.copy()
    stall_count = 0
    block_num = 0

    while time.monotonic() < t2 + phase2_budget:
        block_num += 1
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 15:
            break

        # Adaptive parameters based on stall
        beta_start = 10.0 + 5.0 * min(stall_count, 8)
        beta_end = min(800.0, 50.0 * (1.0 + 0.5 * stall_count))
        a_new, sc_new = lbfgs_lse_optimize(a_curr, n, S, N, block_time, rng,
                                            beta_init=beta_start, beta_max=beta_end,
                                            m_lbfgs=12)
        peak_new = sc_new * S * S / (2.0 * n)
        if peak_new < global_best * S * S / (2.0 * n) - 1e-8:
            update_best(a_new, n, "P2_block%d" % block_num)
            a_curr = a_new.copy()
            stall_count = 0
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count == 2:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.4)
                stall_count = 0
            elif stall_count >= 4:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.7)
                # Also try a fresh deflation from current
                if rng.random() < 0.3:
                    a_curr = deflation_init(n, S, N, 500, rng)
                stall_count = 0

    print("Phase 2 best: score=%.6f" % global_best)

    # === Phase 3: Dual-gradient polishing ===
    print("=== Phase 3: Dual-gradient polishing ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.20)
    if phase3_budget > 20:
        a_pg, sc_pg = dual_gradient_polish(global_a, n, S, N, phase3_budget, rng, epsilon=0.03)
        peak_pg = sc_pg
        sc_pg_score = score_from_peak(peak_pg, n)
        if sc_pg_score < global_best:
            global_best = sc_pg_score
            global_a = a_pg.copy()
            print("  Dual polish improved: score=%.6f" % global_best)
        else:
            # Keep the best from polishing even if not global best
            # (it might be useful for further polishing)
            pass

    # === Phase 4: Nesterov top-k polishing ===
    print("=== Phase 4: Nesterov top-k ===")
    t4 = time.monotonic()
    phase4_budget = min(300.0, (deadline - t4) * 0.15)
    if phase4_budget > 20:
        a_ng, sc_ng = toppk_nesterov(global_a, n, S, N, phase4_budget, rng,
                                      top_k_init=50, top_k_max=600)
        peak_ng = sc_ng
        sc_ng_score = score_from_peak(peak_ng, n)
        if sc_ng_score < global_best:
            global_best = sc_ng_score
            global_a = a_ng.copy()
            print("  Nesterov improved: score=%.6f" % global_best)

    # === Phase 5: Final high-beta polish ===
    print("=== Phase 5: Final high-beta L-BFGS ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 10
    if remaining5 > 30:
        a_fin, sc_fin = lbfgs_lse_optimize(global_a, n, S, N, remaining5, rng,
                                            beta_init=50.0, beta_max=2000.0, m_lbfgs=15)
        peak_fin = sc_fin
        sc_fin_score = score_from_peak(peak_fin, n)
        if sc_fin_score < global_best:
            global_best = sc_fin_score
            global_a = a_fin.copy()
            print("  Final L-BFGS improved: score=%.6f" % global_best)

    # === Phase 6: One more dual polish if time remains ===
    t6 = time.monotonic()
    remaining6 = deadline - t6 - 8
    if remaining6 > 20:
        a_p6, sc_p6 = dual_gradient_polish(global_a, n, S, N, remaining6, rng, epsilon=0.02)
        peak_p6 = sc_p6
        sc_p6_score = score_from_peak(peak_p6, n)
        if sc_p6_score < global_best:
            global_best = sc_p6_score
            global_a = a_p6.copy()
            print("  Final dual polish improved: score=%.6f" % global_best)

    # === Verification and return ===
    print("=== Final verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
            global_n = 256
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("  Verified score: %.6f" % ver)
            if ver < global_best:
                global_best = ver
            print("  Final: n=%d, score=%.6f" % (len(global_a), ver))
        else:
            print("  Verification failed (inf), using fallback")
            global_a = np.ones(256, dtype=np.float64)
            global_n = 256
            ver = evaluate_sequence([float(x) for x in global_a])
            print("  Fallback score: %.6f" % ver)
    else:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256
        ver = evaluate_sequence([float(x) for x in global_a])
        global_best = ver
        print("  Fallback used, score=%.6f" % ver)

    return [float(x) for x in global_a]
