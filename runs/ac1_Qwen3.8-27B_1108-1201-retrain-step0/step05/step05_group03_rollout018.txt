import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_topk_gradient(a, c, k, beta):
    """
    Compute gradient of top-k LSE smooth max of self-convolution.
    Uses cross-correlation via FFT for efficiency.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Numerical safety
    max_scaled = scaled.max()
    if max_scaled > 500:
        beta = 500.0 / max(1e-15, max_scaled)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    # Build full weight vector
    w_full = np.zeros(len(c))
    w_full[idx] = w

    # Gradient via cross-correlation: grad[i] = 2 * sum_k w_full[k] * a[k-i]
    # This is 2 * cross_corr(w_full, a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] = ifft(fft(w) * conj(fft(a)))
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def lse_topk_value(c, k, beta):
    """Compute top-k LSE value."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 500:
        beta = 500.0 / max(1e-15, max_scaled)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    return cm + np.log(max(tot, 1e-300)) / beta


def exact_subgradient(a, c, top_k=10):
    """
    Compute exact subgradient of max(c) using uniform weights over top_k peaks.
    Returns gradient vector.
    """
    n = len(a)
    idx = np.argsort(c)[::-1][:top_k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / top_k

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def spectral_flatten(a, alpha, rng):
    """
    Spectral flattening: modify |A|^2 to be flatter, then project back.
    alpha in (0, 1) controls strength.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a.copy()

    # Sharpen: reduce high-power, boost low-power
    ratio = power / max(mean_p, 1e-20)
    # new_power = power * ratio^(-alpha) = power^(1-alpha) * mean_p^(alpha)
    new_power = power * (ratio ** (-alpha))
    new_power = np.maximum(new_power, 1e-30)
    scale_factor = np.sqrt(new_power / np.maximum(power, 1e-30))
    A_new = A * scale_factor
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain (phase perturbation)."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using FFT interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    # Use zero-padding interpolation via FFT
    fsize_old = 1 << (2 * n_src - 1).bit_length()
    fsize_new = 1 << (2 * n_target - 1).bit_length()
    A_old = fft(a, fsize_old)
    # Resample in frequency domain
    freqs_old = np.fft.fftfreq(fsize_old)
    freqs_new = np.fft.fftfreq(fsize_new)
    A_new = np.zeros(fsize_new, dtype=complex)
    for i, f in enumerate(freqs_new):
        # Find corresponding frequency in old
        idx = int(round(f * fsize_old)) % fsize_old
        A_new[i] = A_old[idx]
    a_new = np.real(ifft(A_new, fsize_new))[:n_target]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_target)
    return a_new / s


def run_nesterov_lse(a_init, S, max_time, rng,
                     beta_init=30.0, beta_rate=0.004, beta_max=8000.0,
                     k_frac=0.05, momentum=True, flat_interval=3000,
                     perturb_interval=5000):
    """
    Nesterov-accelerated proximal gradient with top-k LSE smoothing.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(8, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Compute gradient
        grad = lse_topk_gradient(y, c_y, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search on LSE value
        step = 2.0 / (beta * k * 0.05 + 1e-10)
        step = min(step, 5.0)
        x_new = project_simplex(y - step * grad, S)
        lse_current = lse_topk_value(c_y, k, beta)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            lse_new = lse_topk_value(c_new, k, beta)
            if lse_new <= lse_current + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
            t_new = 1.0
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Periodic spectral flattening
        if it % flat_interval == 0 and stuck > 200:
            y = spectral_flatten(y, 0.3, rng)
            t_val = 1.0
            stuck = 0

        # Periodic perturbation to escape local minima
        if it % perturb_interval == 0 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - 0.8 * stuck / 10000)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Hard reset if very stuck
        if stuck > 3000:
            y = spectral_perturb(y, 0.15, rng)
            t_val = 1.0
            stuck = 0

    return best_a, best


def run_exact_subgradient_polish(a_init, S, max_time, rng,
                                  top_k=20, initial_step=0.5):
    """

    step = initial_step
    no_improve = 0

        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

            base_step *= 1.05

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            # Slightly increase step if improving
            initial_step = min(initial_step * 1.01, 5.0)
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.5 * grad, S)
            c_new2 = fft_convolve_self(a_new)
            if c_new2.max() < cur_max:
                a = a_new
            initial_step = max(initial_step * 0.95, 0.01)

        # Rare perturbation
        if it % 5000 == 0 and no_improve > 500 and rng.random() < 0.1:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_support_refinement(a_init, S, max_time, rng):
    """
    Alternating: optimize on current support, then refine support.
    Key idea: the optimal solution likely has a specific support structure.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1

        # Phase 1: Optimize all components (small step)
        c = fft_convolve_self(a)
        grad = exact_subgradient(a, c, top_k=15)
        step = 0.3 / (1.0 + 0.005 * it)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < c.max():
            a = a_new
            if a.max() < best:
                best = a.max()
                best_a = a.copy()

        # Phase 2: Support refinement (every 200 iterations)
        if it % 200 == 0:
            c = fft_convolve_self(a)
            cur_max = c.max()

            # Try reducing the smallest components to zero
            # (this reduces the "waste" in the sequence)
            threshold = 0.01 * a.mean()
            mask = a > threshold
            if np.sum(mask) < n // 4:
                mask = np.ones(n, dtype=bool)

            # Project onto the masked support
            a_masked = np.zeros(n)
            a_masked[mask] = a[mask]
            a = normalize(a_masked, S)

            # Optimize within the support
            for _ in range(50):
                c = fft_convolve_self(a)
                if c.max() < best:
                    best = c.max()
                    best_a = a.copy()
                grad = exact_subgradient(a, c, top_k=10)
                step = 0.2
                a_new = np.zeros(n)
                a_new[mask] = a[mask] - step * grad[mask]
                a = normalize(a_new, S)

            # Check if removing support helped
            c = fft_convolve_self(a)
            if c.max() > best * 1.01:
                a = best_a.copy()

    return best_a, best


def compute_objective(a):
    """Compute the actual objective value."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: Nesterov LSE -> Exact subgradient polish -> Support refinement.
    Tries multiple lengths and returns the best.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.5, budget_s - 30.0)

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_length(n, a_start, time_limit):
        """Run full optimization pipeline for a given length."""
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None:
            if len(a_start) != n:
                a_start = resample_sequence(a_start, n)
                a_start = normalize(a_start, S)
            else:
                a_start = normalize(a_start.copy(), S)
        else:
            a_start = np.ones(n) * S / n

        print("\n--- n=%d, time_limit=%.0fs ---" % (n, time_limit))
        t_start = time.monotonic()

        best_local = fft_convolve_self(a_start).max()
        best_a_local = a_start.copy()
        obj_init = 2.0 * n * best_local / (S * S)
        print("  Initial: obj=%.6f" % obj_init)

        # Phase 1: Nesterov LSE (50% of time)
        t1 = time_limit * 0.45
        remaining = min(t1, deadline - time.monotonic())
        if remaining > 30:
            a_opt, val_opt = run_nesterov_lse(
                best_a_local, S, remaining, rng,
                beta_init=40.0, beta_rate=0.003, beta_max=10000.0,
                k_frac=0.04, momentum=True,
                flat_interval=2000, perturb_interval=4000
            )
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Nesterov LSE: obj=%.6f (t=%.0fs)" % (obj, time.monotonic() - t_start))

        # Phase 2: Second Nesterov with different params (20% of time)
        t2 = time_limit * 0.20
        remaining = min(t2, deadline - time.monotonic())
        if remaining > 20:
            # Slightly perturb to explore
            a_pert = spectral_perturb(best_a_local, 0.05, rng)
            a_opt, val_opt = run_nesterov_lse(
                a_pert, S, remaining, rng,
                beta_init=200.0, beta_rate=0.002, beta_max=20000.0,
                k_frac=0.02, momentum=True,
                flat_interval=3000, perturb_interval=6000
            )
            obj2 = 2.0 * n * val_opt / (a_opt.sum() ** 2)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Nesterov LSE-2: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t_start))

        # Phase 3: Exact subgradient polish (20% of time)
        t3 = time_limit * 0.20
        remaining = min(t3, deadline - time.monotonic())
        if remaining > 15:
            a_opt, val_opt = run_exact_subgradient_polish(
                best_a_local, S, remaining, rng,
                top_k=15, initial_step=0.3
            )
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Exact subgrad: obj=%.6f (t=%.0fs)" % (obj, time.monotonic() - t_start))

        # Phase 4: Support refinement (10% of time)
        t4 = time_limit * 0.10
        remaining = min(t4, deadline - time.monotonic())
        if remaining > 10:
            a_opt, val_opt = run_support_refinement(
                best_a_local, S, remaining, rng
            )
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Support refin: obj=%.6f (t=%.0fs)" % (obj, time.monotonic() - t_start))

        # Final tiny polish
        remaining = min(10.0, deadline - time.monotonic())
        if remaining > 3:
            a_opt, val_opt = run_exact_subgradient_polish(
                best_a_local, S, remaining, rng,
                top_k=5, initial_step=0.1
            )
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  FINAL n=%d: obj=%.6f" % (n, obj_final))

        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_local.copy()
            print("  *** New global best: %.6f ***" % global_best)

        return best_a_local, best_local

    # Try n=8192 first (main)
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_start_8192 = None
        if given is not None:
            a_start_8192 = resample_sequence(given, 8192)
            a_start_8192 = normalize(a_start_8192, np.sqrt(2 * 8192))
        try_length(8192, a_start_8192, remaining * 0.50)

    # Try n=4096 (faster iterations, potentially different structure)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_start_4096 = None
        if global_a is not None:
            a_start_4096 = resample_sequence(global_a, 4096)
            a_start_4096 = normalize(a_start_4096, np.sqrt(2 * 4096))
        elif given is not None:
            a_start_4096 = resample_sequence(given, 4096)
            a_start_4096 = normalize(a_start_4096, np.sqrt(2 * 4096))
        try_length(4096, a_start_4096, remaining * 0.25)

    # Try n=16384 (more elements, potentially better)
    remaining = deadline - time.monotonic()
    if remaining > 80 and global_a is not None:
        a_start_16384 = resample_sequence(global_a, 16384)
        a_start_16384 = normalize(a_start_16384, np.sqrt(2 * 16384))
        try_length(16384, a_start_16384, remaining * 0.20)

    # Final intensive polish on the best found
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 10 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs) ===" % remaining)

        # Very high-beta Nesterov
        t_fin = remaining * 0.4
        a_fin, val_fin = run_nesterov_lse(
            global_a, S_best, t_fin, rng,
            beta_init=2000.0, beta_rate=0.001, beta_max=50000.0,
            k_frac=0.01, momentum=False,
            flat_interval=10000, perturb_interval=10000
        )
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final Nesterov: obj=%.6f" % global_best)

        # Exact subgradient with very small steps
        remaining2 = deadline - time.monotonic() - 5.0
        if remaining2 > 8:
            a_pol, val_pol = run_exact_subgradient_polish(
                global_a, S_best, remaining2, rng,
                top_k=8, initial_step=0.1
            )
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  Final exact subgrad: obj=%.6f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
