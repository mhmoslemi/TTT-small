Minimize 2*n*max(f*f)/sum(f)^2 using exact subgradient methods with adaptive 
step sizes, Adam on log-scale, peak shaving, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

# Helper functions (all top-level)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def exact_subgradient(a, c, n_active=1):
    """Compute exact subgradient of max(c) where c = a*a."""
    n = len(a)
    cm = c.max()
    # Get indices of active lags (within small tolerance of max)
    tol = 1e-10 * cm
    active = np.where(c >= cm - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    g = np.zeros(n)
    for k in active:
        # g[m] += a[k-m] (if 0 <= k-m < n) + a[m-k] (if 0 <= m-k < n)
        # a[k-m] valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            g[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
        
        # a[m-k] valid when 0 <= m-k < n, i.e., k <= m <= k+n-1
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
    
    # Average over active lags
    g /= len(active)
    return g

def run_subgradient(a_init, n, S, max_time, rng, n_active_expand=5, alpha=0.05):
    """Exact subgradient descent with adaptive step and backtracking."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1

        cur_max = c.max()

            best_a = a.copy()

        # Use lags within a fraction of the max
        cm = cur_max
        threshold = cm * (1 - 0.001 * n_active_expand / it)  # Widen active set over time
        active_mask = c >= threshold
        n_act = max(1, np.sum(active_mask))
        
        g = np.zeros(n)
        active_indices = np.where(active_mask)[0]
        for k in active_indices[:min(n_act, 50)]:  # Cap at 50 active lags
            lo1 = max(0, k - n + 1)
            hi1 = min(k, n - 1)
            if lo1 <= hi1:
                g[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
            lo2 = max(k, 0)
            hi2 = min(k + n - 1, n - 1)
            if lo2 <= hi2:
                g[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
        g /= min(n_act, 50)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size

        improved = False

            c_test = fft_convolve_self(a_test)
            if c_test.max() < base * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:

            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Periodic restart

            if noise_scale > 1e-10:

    return best_a, best

def run_adam_logscale(a_init, n, S, max_time, rng, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):

    a = normalize_sum(a_init, n)
    # Initialize log parameters
    log_a = np.log(np.maximum(a, 1e-10))
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Reconstruct a from log_a
        a = np.exp(log_a - np.logaddexp.reduce(log_a)) * S  # Normalize to sum = S
        # Actually, simpler: a = exp(log_a) / sum(exp(log_a)) * S
        # But for numerical stability:
        max_log = log_a.max()
        exp_a = np.exp(log_a - max_log)
        a = exp_a / exp_a.sum() * S

        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # a[j] = S * exp(log_a[j]) / Z where Z = sum(exp(log_a))
        # da[j]/d(log_a[m]) = a[j] * (delta[j,m] - a[m]/S)
        # Gradient of max(c) w.r.t. a is g_a (subgradient)
        # Gradient w.r.t. log_a: g_log[m] = sum_j da[j]/d(log_a[m]) * g_a[j]
        # = a[m] * g_a[m] - (a[m]/S) * sum_j g_a[j] * a[j]
        
        # Compute g_a

        threshold = cm * 0.999
        active_mask = c >= threshold

        g_a = np.zeros(n)
        for k in active_indices:
            lo1 = max(0, k - n + 1)
            hi1 = min(k, n - 1)
            if lo1 <= hi1:
                g_a[lo1:hi1+1] += a[k - np.arange(lo1, hi1+1)]
            lo2 = max(k, 0)
            hi2 = min(k + n - 1, n - 1)
            if lo2 <= hi2:
                g_a[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
        g_a /= max(1, len(active_indices))

        g_log = a * g_a - (a * (a @ g_a / S))

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Keep log_a in a reasonable range

def peak_shaving(a, n, S, max_time, rng):
    """Surgically reduce peaks by modifying the sequence at peak-contributing positions."""
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    a = a.copy()
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Find the top peaks
        k_star = np.argmax(c)
        
        # Compute the "responsibility" vector: how much each a[j] contributes to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # So a[j] * a[k_star-j] is the contribution
        j_valid = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
        partners = k_star - j_valid
        contrib = a[j_valid] * a[partners]
        
        # Reduce a[j] where contrib is large
        # Use a multiplicative update
        reducible = contrib > np.mean(contrib)
        if reducible.any():
            scale = 0.995  # Reduce by 0.5%
            a[j_valid[reducible]] *= (1 - 0.005)

def run_lse_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_true_max = c.max()
        
        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart if stagnant
        if it % 1000 == 0:
            t_mom = 1.0
            y = a.copy()
    
    return best_a, best

shaping (Gerchberg-Saxton style), Adam on log-scale, LSE-FISTA, and diverse restarts.
Key insight: the subgradient of max(a*a) at lag k* is 2*a reversed/shifted by k*,

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient(a, c, n_active=1):
    """

        top_k = sorted_idx[:20]
        active_k = top_k[c[top_k] > c_mean]
        if len(active_k) == 0:
            active_k = top_k[:1]
        
        g = np.zeros(n)
        w_sum = 0.0
        for k in active_k:
            k = int(k)
            w = c[k] - c_mean
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                m_vals = np.arange(lo, hi + 1)
                g[m_vals] += w * 2.0 * a[k - m_vals]
            w_sum += w
        
        if w_sum > 1e-30:
            g /= w_sum
        else:
            break
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step: larger when far from optimal, smaller when close
        gap = cur_max - c_mean  # How much the peak exceeds the mean
        step = alpha * gap / (gnorm * gnorm) * S
        step = min(step, 0.5 * S / gnorm)  # Cap step size
        
        # Backtracking line search
        base = cur_max
        improved = False
        for trial in range(15):
            a_test = project_simplex(a - step * g, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < base * (1 - 1e-8):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Escape: inject noise
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            else:
                # Try a random direction
                rand_dir = rng.normal(0, 1, n)
                rand_dir = rand_dir - rand_dir.mean()  # Keep sum approximately constant
                a = project_simplex(a + 0.001 * S * rand_dir / np.linalg.norm(rand_dir), S)
        
        # Periodic momentum restart (just reset to avoid drift)
        if it % 2000 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best:
                best = c_check.max()
                best_a = a.copy()
    
    return best_a, best


def run_freq_shaping(a_init, n, S, max_time, rng, n_iter_max=2000):
    """
    Gerchberg-Saxton style frequency domain shaping.
    Alternates between:
    - Time domain: enforce a >= 0, sum(a) = S
    - Frequency domain: flatten |A(f)|^2 to reduce time-domain peaks
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # FFT size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    while time.monotonic() - start < max_time and it < n_iter_max:
        it += 1
        # Compute current convolution
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Frequency domain step: modify |A|^2 to be flatter
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        # Raise the floor, lower the ceiling
        p_min = np.percentile(power[power > 0], 10)
        p_max = power.max()
        if p_max < 1e-15:
            break
        
        # Soft clipping: push extreme values toward the mean
        p_mean = power.mean()
        gamma = 0.1  # Strength of flattening
        
        # New power: interpolate between current and a "flattened" version
        # Flattened: replace top 5% with mean, boost bottom 5% toward mean
        modified_power = power.copy()
        top_thresh = np.percentile(power, 95)
        bot_thresh = np.percentile(power[power > 0], 5)
        modified_power[power > top_thresh] = top_thresh + gamma * (p_mean - top_thresh)
        modified_power = np.maximum(modified_power, p_min * 0.5)
        
        # Reconstruct: keep phase, modify amplitude
        phase = np.angle(A)
        new_A = np.sqrt(modified_power) * np.exp(1j * phase)
        
        # Inverse FFT to get time domain
        a_new = np.real(ifft(new_A, fsize))[:n]
        
        # Enforce constraints: a >= 0, sum(a) = S
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new, S)
        
        # Check if this helps
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * 1.001:
            a = a_new
        else:
            # Blend
            a = 0.5 * a + 0.5 * a_new
            a = project_simplex(a, S)
        
        # Periodic noise injection
        if it % 100 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def run_adam_logscale(a_init, n, S, max_time, rng, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on log-scale parameterization (a[j] = exp(f[j]) normalized)."""
    a = normalize_sum(a_init, n)
    # Initialize log parameters with some spread
    log_f = np.log(np.maximum(a, 1e-10))
    # Center it
    log_f -= log_f.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        t += 1
        # Reconstruct a from log_f with normalization
        max_log = log_f.max()
        exp_a = np.exp(log_f - max_log)
        Z = exp_a.sum()
        if Z < 1e-30:
            break
        a = exp_a / Z * S
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Subgradient of max(c) w.r.t. a
        sorted_idx = np.argsort(c)[::-1]
        active_k = sorted_idx[:10]
        active_k = active_k[c[active_k] >= c.max() * 0.999]
        if len(active_k) == 0:
            active_k = sorted_idx[:1]
        
        g_a = np.zeros(n)
        for k in active_k:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                m_vals = np.arange(lo, hi + 1)
                g_a[m_vals] += 2.0 * a[k - m_vals]
        g_a /= len(active_k)
        
        # Chain rule: d(a)/d(log_f[m]) = a[m] * (delta[j,m] - a[m]/S)
        # So g_log[m] = a[m]*g_a[m] - (a[m]*sum_j a[j]*g_a[j])/S
        ag_a = a @ g_a
        g_log = a * g_a - a * (ag_a / S)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_log
        v = beta2 * v + (1 - beta2) * g_log ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr * 0.9995 ** (t // 100)  # Slow decay
        log_f = log_f - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Prevent numerical issues
        log_f = log_f - log_f.mean()
        log_f = np.clip(log_f, -10, 10)
        
        if t % 500 == 0:
            # Reset moments to avoid stuck gradients
            m *= 0.5
            v *= 0.5
    
    return best_a, best


def run_lse_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_true_max = c.max()
        
        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart momentum periodically
        if it % 500 == 0:
            t_mom = 1.0
            y = a.copy()
    
    return best_a, best


def run_peak_shaving(a_init, n, S, max_time, rng):
    """
    Direct peak shaving: identify contributing pairs at peak lags and
    multiplicatively reduce them while compensating elsewhere.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find top 3 peak lags
        top_lags = np.argsort(c)[::-1][:3]
        
        # For each peak lag, identify the "worst" contributing pairs
        reduce_mask = np.zeros(n)
        for k in top_lags:
            k = int(k)
            w = max(0, c[k] - cur_max * 0.99)
            if w < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            m_vals = np.arange(lo, hi + 1)
            partners = k - m_vals
            contrib = a[m_vals] * a[partners]
            # Add to reduce mask weighted by contribution
            reduce_mask[m_vals] += contrib * w
        
        if reduce_mask.max() < 1e-15:
            break
        
        # Multiplicative reduction
        lr = 0.003
        a *= np.exp(-lr * reduce_mask / reduce_mask.max())
        a = project_simplex(a, S)
        
        # Periodic noise
        if it % 200 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def try_and_update(a, global_state):
    """Evaluate sequence and update global best if improved. Returns (global_best, global_a)."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    c = fft_convolve_self(a)
    val = c.max()
    if val < global_state[0]:
        global_state[0] = val
        global_state[1] = a.copy()
        print("  New best: %.8f (n=%d)" % (val, n))
    return global_state


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses weighted subgradient, frequency-domain shaping, Adam log-scale,
    LSE-FISTA, peak shaving, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))
    
    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass
    
    # Global state: [best_value, best_array]
    global_state = [np.inf, None]
    
    # === Phase 1: Quick LSE-FISTA warmup ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(45.0, (deadline - time.monotonic()) * 0.03)
        print("=== Phase 1: LSE-FISTA warmup (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng, beta_start=100, beta_end=5000, beta_steps=40)
        global_state = try_and_update(a1, global_state)
    else:
        n_g = 8192
        S_g = np.sqrt(2.0 * n_g)
        t = np.linspace(0, 1, n_g)
        given = np.exp(-0.5 * ((t - 0.4) / 0.2) ** 2)
        given = normalize_sum(given, n_g)
        global_state = try_and_update(given, global_state)
    
    # === Phase 2: Weighted subgradient (main optimizer) ===
    if global_state[1] is not None:
        n_main = len(global_state[1])
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.3
        print("=== Phase 2: Weighted subgradient (n=%d, %.0fs) ===" % (n_main, t2))
        a2, v2 = run_weighted_subgradient(global_state[1], n_main, S_main, t2, rng, alpha=0.15)
        global_state = try_and_update(a2, global_state)
    
    # === Phase 3: Frequency domain shaping ===
    if global_state[1] is not None:
        n_main = len(global_state[1])
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.15
        print("=== Phase 3: Freq shaping (n=%d, %.0fs) ===" % (n_main, t3))
        a3, v3 = run_freq_shaping(global_state[1], n_main, S_main, t3, rng)
        global_state = try_and_update(a3, global_state)
    
    # === Phase 4: Adam log-scale ===
    if global_state[1] is not None:
        n_main = len(global_state[1])
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.15
        print("=== Phase 4: Adam log-scale (n=%d, %.0fs) ===" % (n_main, t4))
        a4, v4 = run_adam_logscale(global_state[1], n_main, S_main, t4, rng, lr=0.005)
        global_state = try_and_update(a4, global_state)
    
    # === Phase 5: Peak shaving ===
    if global_state[1] is not None:
        n_main = len(global_state[1])
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t5 = remaining * 0.1
        print("=== Phase 5: Peak shaving (n=%d, %.0fs) ===" % (n_main, t5))
        a5, v5 = run_peak_shaving(global_state[1], n_main, S_main, t5, rng)
        global_state = try_and_update(a5, global_state)
    
    # === Phase 6: Diverse restarts ===
    if global_state[1] is not None:
        n_main = len(global_state[1])
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t6 = remaining * 0.2
        print("=== Phase 6: Diverse restarts (n=%d, %.0fs) ===" % (n_main, t6))
        
        n_restarts = 4
        per_restart = max(5.0, t6 / n_restarts)
        noise_scales = [0.01, 0.03, 0.005, 0.02]
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            ns = noise_scales[r]
            perturbed = global_state[1] + rng.normal(0, ns * S_main / np.sqrt(n_main), n_main)
            perturbed = normalize_sum(perturbed, n_main)
            
            if r % 2 == 0:
                ar, vr = run_weighted_subgradient(perturbed, n_main, S_main, per_restart, rng, alpha=0.1)
            else:
                ar, vr = run_lse_fista(perturbed, n_main, S_main, per_restart, rng,
                                        beta_start=200, beta_end=10000, beta_steps=50)
            global_state = try_and_update(ar, global_state)
    
    # === Phase 7: Try different lengths ===
    if global_state[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t7 = min(remaining * 0.4, 100)
            print("=== Phase 7: Different lengths (%.0fs) ===" % t7)
            
            for n_try in [len(global_state[1]) * 2, len(global_state[1]) + 256]:
                if time.monotonic() > deadline - 30:
                    break
                if n_try > 32768 or n_try < 256:
                    continue
                S_try = np.sqrt(2.0 * n_try)
                t_old = np.linspace(0, 1, len(global_state[1]))
                t_new = np.linspace(0, 1, n_try)
                init = np.interp(t_new, t_old, global_state[1])
                init = normalize_sum(init, n_try)
                per_len = max(10.0, t7 / 2)
                a7, v7 = run_lse_fista(init, n_try, S_try, per_len, rng,
                                        beta_start=100, beta_end=5000, beta_steps=40)
                global_state = try_and_update(a7, global_state)
    
    # === Phase 8: Final polish with high-beta LSE ===
    if global_state[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(global_state[1])
            S_f = np.sqrt(2.0 * n_f)
            t8 = remaining * 0.6
            print("=== Phase 8: Final LSE polish (%.0fs) ===" % t8)
            a8, v8 = run_lse_fista(global_state[1], n_f, S_f, t8, rng,
                                    beta_start=5000, beta_end=50000, beta_steps=30)
            global_state = try_and_update(a8, global_state)
            
            # Quick subgradient polish after LSE
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a8b, v8b = run_weighted_subgradient(global_state[1], n_f, S_f, remaining2 * 0.5, rng, alpha=0.05)
                global_state = try_and_update(a8b, global_state)
    
    # Final verification
    if global_state[1] is not None:
        final_list = [float(x) for x in np.clip(global_state[1], 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_state[0], len(global_state[1])))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_state[1] = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_state[1] = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    result = np.clip(global_state[1], 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("RETURNING: length=%d, best_score=%.8f" % (len(result), global_state[0]))
    return [float(x) for x in result]
