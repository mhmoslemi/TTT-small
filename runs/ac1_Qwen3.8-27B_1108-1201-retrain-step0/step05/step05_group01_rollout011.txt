"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-phase optimization: adaptive LSE-FISTA with backtracking,
active-set refinement with frequency-domain perturbations, and L-BFGS polishing.
Key improvements over previous: backtracking line search, adaptive active set,
frequency-domain structured perturbations, and mirror descent hybrid.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def freq_perturb(a, rng, strength=0.01):
    """Apply structured perturbation in frequency domain."""
    n = len(a)
    S = a.sum()
    A = fft(a, 2 * n)
    # Perturb amplitudes at random non-DC frequencies
    mask = rng.random(2 * n) < 0.3
    mask[0] = False  # Don't perturb DC
    dA = strength * S / np.sqrt(n) * (rng.random(2 * n) - 0.5) * mask
    A_new = A + dA
    # Maintain Hermitian symmetry for real signal
    A_new = (A_new + np.conj(A_new[::1][::-1])) / 2.0
    # Simpler: just take real part after IFFT
    a_new = np.real(ifft(A_new, 2 * n))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def run_adaptive_fista(a_init, max_time, rng, beta_start=5.0, beta_end=200.0,
                       use_backtrack=True, mode="lse"):
    """
    FISTA with adaptive LSE beta (monotonic increase), backtracking line search,
    and frequency-domain perturbations when stuck.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    step_lr = 1.0  # Relative step size multiplier

    while time.monotonic() - start < max_time:
        it += 1

        # Monotonically increasing beta
        frac = min(1.0, it / (max_time * 100))  # approximate progress
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Compute gradient
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(2, min(64, n // 16))
            idx = np.argsort(c)[::-1][:k]
            w = np.zeros(len(c))
            w[idx] = 1.0 / k
            grad = conv_gradient(y, w)
        else:  # hybrid: LSE with widening tolerance
            if stuck_count < 500:
                w = lse_weights(c, beta)
            elif stuck_count < 2000:
                tol = 0.001 * stuck_count
                active = np.where(c >= cur_max * (1.0 - tol))[0]
                w = np.zeros(len(c))
                w[active] = 1.0 / len(active)
            else:
                tol = min(0.05, 0.0005 * stuck_count)
                active = np.where(c >= cur_max * (1.0 - tol))[0]
                w = np.zeros(len(c))
                w[active] = 1.0 / len(active)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        if use_backtrack:
            step = step_lr * S / (gnorm * max(beta, 1.0))
            step = max(1e-10, min(step, 10.0 * S / gnorm))
            # Try the step
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cur_max + 1e-13:
                # Accept
                step_lr = min(step_lr * 1.5, 5.0)
            else:
                # Reject, reduce step
                step_lr = max(step_lr * 0.5, 0.01)
                step = step_lr * S / (gnorm * max(beta, 1.0))
                x_trial = project_simplex(y - step * grad, S)
        else:
            step = step_lr / max(beta, 1.0)
            x_trial = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_trial
        y = y_new
        t_mom = t_new

        # Frequency-domain perturbation when stuck
        if stuck_count > 3000 and it % 500 == 0:
            strength = 0.005 * np.exp(-stuck_count / 15000.0)
            if strength > 1e-5:
                y = freq_perturb(y, rng, strength)
                a = y.copy()
                t_mom = 1.0
                stuck_count = 0
                step_lr = 1.0

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_mirror_descent(a_init, max_time, rng, beta=20.0):
    """
    Mirror descent with entropy regularizer (multiplicative weights update).
    Natural for non-negative constraints.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    a = np.maximum(a, 1e-10)  # Keep strictly positive for mirror descent

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = 1.0 / max(beta, 1.0)

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive eta
        eta_cur = eta * np.exp(-stuck / 5000.0)
        eta_cur = max(eta_cur, 1e-6)

        # LSE weights
        w = lse_weights(c, max(beta, 5.0))
        grad = conv_gradient(a, w)

        # Mirror descent update: a_i <- a_i * exp(-eta * grad_i), then normalize
        log_scale = -eta_cur * grad / (a + 1e-15)
        log_scale = np.clip(log_scale, -50, 50)
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * (S / a_new.sum())

        a = a_new

        # Occasionally project to simplex (to handle very small values)
        if it % 1000 == 0:
            a = project_simplex(a, S)
            a = np.maximum(a, 1e-15)

        if stuck > 5000 and it % 1000 == 0:
            a = freq_perturb(a, rng, 0.003)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """
    L-BFGS-B polishing with LSE objective and increasing beta.
    Uses penalty for sum constraint.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [15.0, 40.0, 100.0, 300.0, 800.0, 1500.0][:n_levels]
    time_per_level = max_time / n_levels

    for idx, beta in enumerate(betas):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        pen = max(1.0, beta * 0.5)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        time_budget = min(remaining * 0.9, time_per_level * 1.5)

        try:
            result = scipy_minimize(
                obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': 80, 'maxls': 20, 'ftol': 1e-15,
                         'gtol': 1e-10, 'iprint': -1,
                         'maxfun': 200}
            )
        except Exception:
            break

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-10:
            a_new = a_new * (S / a_new.sum())
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_active_set_fista(a_init, max_time, rng):
    """
    Active-set FISTA: uses exact subgradient from lags near the max.
    Adaptive active set width.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_lr = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set
        if stuck < 200:
            tol_frac = 0.0
        elif stuck < 1000:
            tol_frac = min(0.002, 0.0005 * stuck / 200)
        elif stuck < 5000:
            tol_frac = min(0.01, 0.002 + 0.00001 * stuck)
        else:
            tol_frac = min(0.05, 0.01 + 0.00002 * stuck)

        active = np.where(c >= cur_max * (1.0 - tol_frac))[0]
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_lr * S / (gnorm * (1.0 + 0.01 * stuck))
        step = min(step, 5.0 * S / gnorm)
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Backtrack on non-improvement
        if stuck > 100:
            step_lr = max(0.1, step_lr * 0.98)

        # Frequency perturbation
        if stuck > 4000 and it % 500 == 0:
            strength = 0.003 * np.exp(-stuck / 20000.0)
            if strength > 1e-5:
                y = freq_perturb(y, rng, strength)
                a = y.copy()
                t_mom = 1.0
                stuck = 0
                step_lr = 1.0

        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def generate_raised_cosine(n, S, alpha=0.5):
    """Generate a raised-cosine shaped initial sequence."""
    t = np.arange(n) / n
    a = 1.0 - alpha * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Phases: adaptive FISTA, mirror descent, active-set FISTA, L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        S0 = np.sqrt(2.0 * n0)
        a_start = generate_raised_cosine(n0, S0)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    n0 = len(global_best_a)
    S = np.sqrt(2.0 * n0)

    # === Phase 1: Adaptive LSE-FISTA (25% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.25
    print("\n=== Phase 1: Adaptive LSE-FISTA (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_adaptive_fista(global_best_a, p1_budget * 0.6, rng,
                               beta_start=5.0, beta_end=100.0,
                               use_backtrack=True, mode="lse")
    obj1 = compute_obj(a1)
    print("  FISTA-lse: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_adaptive_fista(global_best_a, rem1, rng,
                                    beta_start=20.0, beta_end=200.0,
                                    use_backtrack=True, mode="hybrid")
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Mirror Descent (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.15
        print("\n=== Phase 2: Mirror Descent (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_mirror_descent(global_best_a, p2_budget, rng, beta=15.0)
        obj2 = compute_obj(a2)
        print("  Mirror descent: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Active-set FISTA (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.25
        print("\n=== Phase 3: Active-set FISTA (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, _ = run_active_set_fista(global_best_a, p3_budget, rng)
        obj3 = compute_obj(a3)
        print("  Active-set: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: L-BFGS polishing (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.25
        print("\n=== Phase 4: L-BFGS polishing (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, obj4 = run_lbfgs_polish(global_best_a, p4_budget, rng, n_levels=5)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final LSE-FISTA + L-BFGS squeeze (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.5
        print("\n=== Phase 5: Final squeeze (budget %.0fs) ===" % (remaining * 0.5))
        t5 = time.monotonic()

        a5, _ = run_adaptive_fista(global_best_a, p5_budget, rng,
                                   beta_start=50.0, beta_end=500.0,
                                   use_backtrack=True, mode="topk")
        obj5 = compute_obj(a5)
        print("  Final FISTA: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = remaining * 0.5 - (time.monotonic() - t5)
        if rem5 > 10:
            a5b, obj5b = run_lbfgs_polish(global_best_a, rem5, rng, n_levels=3)
            print("  Final L-BFGS: obj=%.8f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Retry with different n if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 6: Try n=16384 (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        n_alt = 16384
        S_alt = np.sqrt(2.0 * n_alt)
        a_alt = generate_raised_cosine(n_alt, S_alt)
        obj_alt_init = compute_obj(a_alt)
        print("  n=%d init: obj=%.8f" % (n_alt, obj_alt_init))

        if obj_alt_init < global_best_obj or obj_alt_init < 1.55:
            # Worth optimizing
            time_for_fista = remaining * 0.5
            a_alt_opt, _ = run_adaptive_fista(a_alt, time_for_fista, rng,
                                              beta_start=5.0, beta_end=50.0,
                                              use_backtrack=True, mode="lse")
            obj_alt = compute_obj(a_alt_opt)
            print("  n=%d FISTA: obj=%.8f (t=%.0fs)" % (n_alt, obj_alt, time.monotonic() - t6))
            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_alt_opt.copy()
                print("  IMPROVED! New best: %.8f" % global_best_obj)

            rem6 = remaining * 0.5 - (time.monotonic() - t6)
            if rem6 > 10 and obj_alt < global_best_obj + 1e-6:
                a_alt2, obj_alt2 = run_lbfgs_polish(a_alt_opt, rem6, rng, n_levels=3)
                print("  n=%d L-BFGS: obj=%.8f" % (n_alt, obj_alt2))
                if obj_alt2 < global_best_obj:
                    global_best_obj = obj_alt2
                    global_best_a = a_alt2.copy()
                    print("  IMPROVED! New best: %.8f" % global_best_obj)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
            ver = evaluate_sequence(final_list)
            print("Fallback verified: %.8f" % ver)
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
