`LSE_beta(c) = (1/beta) * log(sum(exp(beta * c_k)))`

This is a weighted average of the subgradients, with weights concentrated on the largest c_k for large beta.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using LSE-FISTA + Active-Set LP direction finding.
Key innovation: Use a linear program at each step to find the optimal direction that
simultaneously reduces all active (near-maximal) convolution peaks while preserving
non-negativity and sum constraints. This gives a much better last-mile refinement than
pure subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_active_set(c, tol_frac=0.001, max_size=50):
    cm = c.max()
    threshold = cm - tol_frac * cm
    active = np.where(c >= threshold)[0]
    if len(active) > max_size:
        idx = np.argsort(c)[::-1][:max_size]
        active = idx
    return list(active)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=200):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta
        stagnant = it - last_improve_it
        if stagnant > 500:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 50000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def lp_direction(a, active_set, t_max=0.5):
    """Solve LP to find descent direction minimizing the max active peak rate of change."""
    n = len(a)
    m = len(active_set)
    if m == 0:
        return None, 0.0

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    for i, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        A_ub[i, js] = 2.0 * a[partners]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    lower_bounds = -a / t_max
    bounds = list(zip(lower_bounds, np.full(n, np.inf))) + [(None, None)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[-1]
    except Exception:
        pass
    return None, 0.0


def run_active_set_lp(a_init, n, S, max_time, rng):
    """Active-set LP refinement: solve LP each step for optimal direction."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    tol_frac = 0.002
    max_active = 30
    t_base = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        active_set = compute_active_set(c, tol_frac, max_active)

        # Solve LP for direction
        d, z = lp_direction(a, active_set, t_max=1.0)

        if d is not None and z < -1e-10:
            # Good descent direction found via LP
            # Line search for step size
            step = t_base
            improved = False
            for _ in range(6):
                a_test = a + step * d
                a_test = np.maximum(a_test, 0.0)
                a_test = project_simplex(a_test, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur_max * (1 - 1e-7):
                    a = a_test
                    improved = True
                    break
                step *= 0.5

            if not improved:
                # Try very small step
                step = 0.01
                a_test = project_simplex(a + step * d, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur_max:
                    a = a_test
        else:
            # LP found no improvement or failed; use subgradient fallback
            cm = c.max()
            grad = np.zeros(n)
            for k in active_set:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                js = np.arange(lo, hi + 1)
                partners = k - js
                grad[js] += 2.0 * a[partners]
            grad /= len(active_set)

            step = 0.01 / (1 + it * 0.0001)
            a = project_simplex(a - step * grad, S)

        # Periodic: expand active set tolerance to escape
        if it % 200 == 0:
            tol_frac = max(0.0005, tol_frac * 0.8)
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

        # Adaptive: if stuck, increase tolerance
        if it % 500 == 0:
            c_check = fft_convolve_self(a)
            if abs(c_check.max() - best) < 1e-10:
                tol_frac = min(0.01, tol_frac * 1.5)

    return best_a, best


def run_subgradient_refine(a_init, n, S, max_time, rng):
    """Pure subgradient with Polyak-like step size for final polish."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Estimate lower bound (average of convolution)
    c = fft_convolve_self(a)
    lower_bound = c.sum() / (2 * n - 1)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Update lower bound
        lower_bound = max(lower_bound, c.sum() / (2 * n - 1))

        # Active set
        active_set = compute_active_set(c, 0.003, 20)

        # Subgradient
        grad = np.zeros(n)
        for k in active_set:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            grad[js] += 2.0 * a[partners]
        grad /= len(active_set)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step size
        gap = cur_max - lower_bound
        if gap > 1e-12:
            step = gap / (gnorm * gnorm)
        else:
            step = 1.0 / (gnorm * np.sqrt(it))
        step = min(step, 0.1)

        a = project_simplex(a - step * grad, S)

        # Occasional noise
        if it % 500 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def coordinate_descent_refine(a_init, n, S, max_iters=200):
    """Try random coordinate perturbations to find small improvements."""
    rng = np.random.default_rng(12345)
    a = normalize_sum(a_init, n)
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()

    for it in range(max_iters):
        # Pick a pair of coordinates to swap mass between
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue

        # Try moving mass from i to j (or vice versa)
        cur_max = best_val
        for direction in [1, -1]:
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                delta = direction * frac * abs(a[i])
                if a[i] + delta < 0:
                    continue
                a_test = a.copy()
                a_test[i] += delta
                a_test[j] -= delta
                a_test = np.maximum(a_test, 0.0)
                s = a_test.sum()
                if s < 1e-15:
                    continue
                a_test = a_test * (S / s)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < best_val * (1 - 1e-8):
                    a = a_test
                    best_val = c_test.max()
                    best_a = a.copy()
                    break

    return best_a, best_val

            a4, v4 = run_lse_fista(init_large, n_large, S_large, t4 * 0.7, rng,
                                    beta_start=100.0, beta_end=10000.0, beta_steps=80)
            try_sequence(a4)

            remaining2 = deadline - time.monotonic()
            if remaining2 > 30 and global_a is not None:
                a4b, v4b = run_active_set_lp(global_a if len(global_a) < n_large else a4,
                                             len(global_a) if len(global_a) < n_large else n_large,
                                             S_main if len(global_a) < n_large else S_large,
                                             remaining2 * 0.5, rng)
                try_sequence(a4b)

    # === Phase 5: Coordinate descent on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_cd = len(global_a)
            S_cd = np.sqrt(2.0 * n_cd)
            n_iters = min(100, int(remaining / 0.5))
            print("=== Phase 5: Coordinate descent (n=%d, ~%d iters) ===" % (n_cd, n_iters))
            a5, v5 = coordinate_descent_refine(global_a, n_cd, S_cd, max_iters=n_iters)
            try_sequence(a5)

    # === Phase 6: Final LSE polish on absolute best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            print("=== Phase 6: Final LSE polish (n=%d, %.0fs) ===" % (n_f, remaining))
            a6, v6 = run_lse_fista(global_a, n_f, S_f, remaining * 0.6, rng,
                                    beta_start=5000.0, beta_end=50000.0, beta_steps=30)
            try_sequence(a6)

            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a6b, v6b = run_active_set_lp(global_a, n_f, S_f, remaining2 * 0.5, rng)
                try_sequence(a6b)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence=%.10f, internal=%.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
