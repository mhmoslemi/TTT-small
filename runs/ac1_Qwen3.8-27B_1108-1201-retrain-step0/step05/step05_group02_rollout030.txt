```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
mirror descent (multiplicative weights) with adaptive LSE smoothing.

Key ideas:
- Scale invariance: normalize sum(a) = sqrt(2n), objective = max(conv(a,a))
- Mirror descent with entropy potential: a_new[i] ∝ a[i] * exp(-eta * grad[i])
  naturally preserves non-negativity without projection.
- Adaptive beta schedule for LSE-smoothed max.
- Periodic coordinate descent for fine-tuning.
- Multi-size search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_lse_and_weights(c, beta):
    """Compute LSE value and softmax weights."""
    cm = c.max()
    if beta < 1e-6:
        # Uniform weights (mean)
        return cm, np.ones(len(c)) / len(c)
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return cm, w
    lse = cm + np.log(tot) / beta
    w = e / tot
    return lse, w


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w[k] * c[k] w.r.t. a,
    where c = conv(a, a).
    Result: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices).
    """
    n = len(a)
    P = 2 * n - 1
    # Cross-correlation: g[j] = sum_k w[k] * a[k-j]
    N = 1 << (2 * P).bit_length()  # >= 2P for linear cross-corr
    a_ext = np.zeros(N)
    a_ext[:n] = a
    W = fft(w, N)
    A = fft(a_ext, N)
    g = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * g


def mirror_descent_step(a, gradient, eta, S):
    """Apply one multiplicative weight update."""
    # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
    # Use log-space for numerical stability
    log_a = np.zeros(len(a))
    mask = a > 1e-300
    log_a[mask] = np.log(a[mask])
    log_a[~mask] = -700  # log(1e-300)
    
    log_a -= eta * gradient
    log_a -= log_a.max()  # Shift for stability
    a_new = np.exp(log_a)
    
    # Renormalize
    s = a_new.sum()
    if s < 1e-300:
        a_new = np.ones(len(a)) * S / len(a)
    else:
        a_new *= S / s
    return a_new


def run_mirror_descent(a_init, max_time, rng, beta_init=20.0, beta_final=5000.0,
                       beta_rate=0.002, eta_base=1.0, use_line_search=False):
    """
    Mirror descent (multiplicative weights) with adaptive LSE.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        beta = min(beta_final, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Compute LSE weights
        _, w = compute_lse_and_weights(c, beta)
        
        # Compute gradient
        grad = compute_gradient(a, w)
        
        # Step size
        gmax = np.max(np.abs(grad))
        if gmax < 1e-300:
            break
        eta = eta_base / gmax
        
        # Apply mirror descent step
        a_new = mirror_descent_step(a, grad, eta, S)
        
        # Optional: line search (try a few step sizes)
        if use_line_search:
            best_new_obj = cur_obj
            best_new_a = a_new
            for eta_frac in [0.5, 0.25, 0.1]:
                a_try = mirror_descent_step(a, grad, eta * eta_frac, S)
                obj_try = compute_obj(a_try)
                if obj_try < best_new_obj:
                    best_new_obj = obj_try
                    best_new_a = a_try
            a_new = best_new_a
        else:
            # Quick check: if objective worsened, use smaller step
            obj_new = compute_obj(a_new)
            if obj_new > cur_obj * 1.01:  # More than 1% worse
                a_new = mirror_descent_step(a, grad, eta * 0.1, S)
        
        a = a_new
        
        # Periodic perturbation when stuck
        if stuck_count > 500 and stuck_count % 100 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck_count / 2000))
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 0)
            a *= S / a.sum()
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.005,
              beta_max=20000.0, step_scale=1.0):
    """FISTA with LSE for comparison and complementarity."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        _, w = compute_lse_and_weights(c, beta)
        grad = compute_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 8:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        
        if bt > 4:
            step = step_local * 0.7
        else:
            step = min(step * 1.05, 3.0 / max(beta, 1.0))
        
        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y *= S / ys
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 5000:
            t_mom = 1.0
            noise = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise, n), S)
    
    return best_a, best_obj


def peak_shaving(a_init, max_time, rng):
    """
    Direct peak shaving: iteratively reduce the max by
    redistributing mass from high-contribution to low-contribution positions.
    More aggressive than coordinate descent.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a *= S / a.sum()
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 50000:
            break
        
        # Find top lags
        k_star = int(np.argmax(c))
        
        # Compute contribution of each position to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[j]
        
        # Sort by contribution
        order = np.argsort(contrib)
        n_transfer = max(1, n // 100)  # Transfer from top 1% to bottom 1%
        
        # Move mass from high-contrib to low-contrib positions
        total_moved = 0
        for frac_idx in range(n_transfer):
            src = order[n - 1 - frac_idx]  # High contribution
            tgt = order[frac_idx]  # Low contribution
            if a[src] < 1e-15:
                continue
            # Move a small fraction
            delta = a[src] * 0.02
            a[src] -= delta
            a[tgt] += delta
            total_moved += delta
        
        if total_moved > 1e-15:
            # Update c incrementally
            c = fft_convolve_self(a)  # Full recompute for simplicity
        else:
            # Perturb
            idx = rng.integers(0, n)
            delta = a[idx] * 0.01
            if delta > 1e-12:
                a[idx] -= delta
                a[rng.integers(0, n)] += delta
                c = fft_convolve_self(a)
        
        if it % 50 == 0:
            c = fft_convolve_self(a)  # Ensure no drift
    
    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.3) ** 2 / (0.12 ** 2))
    elif method == "skewed":
        # Asymmetric: concentrated on one side
        a = np.exp(-4.0 * tt / (1.0 - tt + 0.01))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 5 * tt + 1.0)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1) + 0.1 * np.sin(2 * np.pi * 19 * tt + 0.5)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump_left":
        # Bump concentrated on the left
        a = np.exp(-5.0 * tt / (tt + 0.1))
    elif method == "two_bumps":
        a = np.exp(-8.0 * (tt - 0.25) ** 2 / (0.05 ** 2)) + 0.7 * np.exp(-8.0 * (tt - 0.75) ** 2 / (0.05 ** 2))
    elif method == "linear_ramp":
        a = 0.1 + 0.9 * tt  # Linearly increasing
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
```
