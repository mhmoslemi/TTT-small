import numpy as np
import time

from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k, power=2.0):
    """Top-k subgradient with power weighting on the values."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = np.maximum(c[idx], 1e-30)
    w = np.power(vals, power)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)

def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trimmed = a[:n_new * factor]
    return a_trimmed.reshape(n_new, factor).mean(axis=1)


def run_fista(a_init, max_time, rng, beta_init=5.0, beta_max=50000.0,
              beta_growth=1.002, restart_interval=5000, noise_base=0.003):
    """FISTA with LSE objective and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: grow faster when stuck
        if stuck < 500:
            beta = beta_init * (1.0 + 0.005 * it)
        else:
            growth = beta_growth * (1.0 + stuck / 5000.0)
            beta = min(beta_max, beta * growth)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        step = 1.0 / max(beta * 2.0 * np.max(a) + np.max(a), 1.0)
        step = min(step, 1.0 / gnorm * 0.9)
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with noise when stuck
        if stuck > restart_interval:
            t_mom = 1.0
            noise_scale = noise_base * S / np.sqrt(n) * np.exp(-stuck / (restart_interval * 4))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
            it = 0  # Reset iteration counter for beta

        # Periodic momentum restart
        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def run_lp_direction(a, n_lp=None, max_time_lp=30.0):
    """
    Solve the LP to find a good direction.
    If n_lp is specified, downsample to n_lp before solving.
    Returns the upsampled direction normalized to have the same sum as a.
    """
    n = len(a)
    if n_lp is None or n_lp >= n:
        a_lp = a.copy()
        scale_up = 1
    else:
        a_lp = downsample(a, n // n_lp)
        scale_up = n // n_lp

    n_curr = len(a_lp)
    S = a_lp.sum()
    c = fft_convolve_self(a_lp)
    M = c.max()

    if M <= 0:
        return a.copy()

    # LP: max sum(b) s.t. (a_lp * b)_k <= M for all k, b >= 0
    # Constraint matrix: (2*n_curr - 1) x n_curr
    # Row k, col j: a_lp[j] if max(0, k-n_curr+1) <= j <= min(k, n_curr-1), else 0

    n_constraints = 2 * n_curr - 1

    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    for k in range(n_constraints):
        j_min = max(0, k - n_curr + 1)
        j_max = min(k, n_curr - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(a_lp[j])

    A_lp = csr_matrix((data, (rows, cols)), shape=(n_constraints, n_curr))

    # Objective: minimize -sum(b), i.e., c_obj = -1
    c_obj = -np.ones(n_curr)
    b_vec = np.full(n_constraints, M)

    # Solve
    start_lp = time.monotonic()
    result = linprog(c_obj, A_ub=A_lp, b_ub=b_vec, bounds=(0, None),
                     method='highs', options={'time_limit': max_time_lp})
    lp_time = time.monotonic() - start_lp

    if result.status != 0 or result.x is None:
        return None

    b_lp = np.maximum(result.x, 0)
    b_sum = b_lp.sum()
    if b_sum < 1e-15:
        return None

    # Upsample if needed
    if scale_up > 1:
        direction = upsample(b_lp, scale_up)
    else:
        direction = b_lp

    # Normalize to same sum as a
    direction = normalize_seq(direction, a.sum())
    return direction


def run_coordinate_polish(a_init, max_time, rng, top_k_lags=32):

    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0

    # Pre-compute the "neighbor" pattern for each coordinate
    # For coordinate j, changing a_j by delta changes b_k by delta*(a_{k-j} + a_{j-k})
    # We'll compute on the fly

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        s = a.sum()

        # Select top k lags
        idx_top = np.argsort(c)[::-1][:top_k_lags]

        # Select coordinates to update: those with high a_j or high gradient
        # Simple strategy: random subset
        n_update = min(n, 200 + it * 10)
        if n_update < n:
            coord_idx = rng.choice(n, n_update, replace=False)
        else:
            coord_idx = np.arange(n)

        improved = False
        for j in coord_idx:
            if time.monotonic() - start > max_time - 5:
                break

            a_j = a[j]
            if a_j < 1e-10:
                continue

            # 1D search: find best delta in range [-a_j, a_j * 10]
            # f(delta) = 2*n*max(c_k + delta*(a_{k-j}+a_{j-k}) + delta^2*1[k=2j]) / (s+delta)^2
            # We'll do a coarse-to-fine search

            # Compute the "pattern" p_k = a_{k-j} + a_{j-k}
            # For k from 0 to 2n-2:
            # p_k = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            p = np.zeros(2 * n - 1)
            # a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
            k1_min = j
            k1_max = min(j + n - 1, 2 * n - 2)
            p[k1_min:k1_max + 1] += a[0:min(n, k1_max + 1 - k1_min)]
            # a[j-k] is nonzero when 0 <= j-k < n, i.e., j-n+1 <= k <= j
            k2_min = max(0, j - n + 1)
            k2_max = j
            if k2_max >= k2_min:
                # a[j-k] for k from k2_min to k2_max: indices j-k2_max to j-k2_min
                idx = np.arange(j - k2_max, j - k2_min + 1)
                p[k2_min:k2_max + 1] += a[idx]

            # The change in b_k is delta * p_k + delta^2 * 1[k=2j]
            # (since the delta^2 term only appears when both terms in b_k involve j)
            # Actually: b_k has terms a_j * a_{k-j} and a_{k-j} * a_j, so if k-j=j, i.e., k=2j,
            # then b_k includes a_j^2, which changes by 2*a_j*delta + delta^2.
            # And the other terms change by delta * a_{k-j} (from a_j -> a_j+delta) and
            # delta * a_{j-k} (from a_{j-k} -> a_{j-k} when k-j != j).
            # Wait, I need to be more careful.
            # b_k = sum_i a_i * a_{k-i}
            # If we change a_j -> a_j + delta:
            # b_k_new = b_k + delta * (a_{k-j} + a_{j-k}) + delta^2 * 1[k-j=j]
            #        = b_k + delta * p_k + delta^2 * 1[k=2j]
            # where a_{k-j} means a at index k-j (0 if out of range).
            # The 1[k=2j] term: when k-j = j, i.e., k=2j, the term a_j * a_j becomes (a_j+delta)^2 = a_j^2 + 2*a_j*delta + delta^2.
            # But in our formula, the delta*p_k term already includes 2*a_j*delta when k=2j (since p_{2j} includes a_j from both terms).
            # So the extra delta^2 term is correct.

            is_2j = (2 * j < 2 * n - 1)  # k=2j is in range?

            def obj_1d(delta):
                if delta <= -a_j + 1e-15:
                    return np.inf
                s_new = s + delta
                if s_new < 0.01:
                    return np.inf
                b_new = c + delta * p
                if is_2j:
                    b_new[2 * j] += delta * delta
                return 2.0 * n * b_new.max() / (s_new * s_new)

            # Golden section search in [-a_j*0.5, a_j*3]
            lo = -0.5 * a_j
            hi = 3.0 * a_j

            # Quick check: evaluate at a few points
            candidates = [0.0, -0.5*a_j, 0.5*a_j, -a_j*0.9, a_j]
            best_delta = 0.0
            best_val = obj_1d(0.0)
            for d in candidates:
                if d < -a_j + 1e-12:
                    continue
                v = obj_1d(d)
                if v < best_val:
                    best_val = v
                    best_delta = d

            # Fine golden section
            if best_val < best_obj:
                gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
                a_g = lo + (1 - gr) * (hi - lo)
                b_g = lo + gr * (hi - lo)
                fa = obj_1d(a_g)
                fb = obj_1d(b_g)
                for _ in range(20):
                    if fa < fb:
                        hi = b_g
                        b_g = a_g
                        fb = fa
                        a_g = lo + (1 - gr) * (hi - lo)
                        fa = obj_1d(a_g)
                    else:
                        lo = a_g
                        a_g = b_g
                        fa = fb
                        b_g = lo + gr * (hi - lo)
                        fb = obj_1d(b_g)

                best_pt = a_g if fa < fb else b_g
                best_v = min(fa, fb)
                if best_v < best_val:
                    best_delta = best_pt
                    best_val = best_v

            if best_val < best_obj - 1e-12:
                a[j] += best_delta
                a[j] = max(0.0, a[j])
                best_obj = best_val
                best_a = a.copy()
                improved = True

        if not improved and it > 20:
            # Add small noise to escape
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = normalize_seq(np.maximum(a + noise, 0), S)

    return best_a


def run_active_polish(a_init, max_time, rng):
    """Active-set polishing with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: start with exact max, increase as we get stuck
        if stuck < 50:
            k = 1
            power = 1.0
        elif stuck < 200:
            k = 3
            power = 2.0
        elif stuck < 1000:
            k = min(16, 3 + stuck // 50)
            power = 2.0
        elif stuck < 5000:
            k = min(64, 16 + stuck // 100)
            power = 3.0
        else:
            k = min(256, 64 + stuck // 50)
            power = 4.0

        grad = topk_subgradient(y, c, k, power)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = min(0.9 / gnorm, S / (gnorm * (1.0 + 0.005 * it)))
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 6000 and it % 500 == 0:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

    return best_a
