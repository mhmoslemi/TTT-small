def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    beta_val = float(beta)
    if beta_val < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta_val * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta_val
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_partial_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of the FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_full_restart(a, N, rng):
    """Randomize all FFT phases while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    s_hist = []
    y_hist = []
    m_current = m_lbfgs_init

    g, f_val, c, c_max = lse_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    patience = 200

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta schedule
        if it % 100 == 0:
            # Increase beta, but occasionally anneal
            if rng.random() < 0.1 and beta > 10:
                beta = max(beta_init, beta * 0.5)  # Annealing
            else:
                beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Grow L-BFGS memory
        if it % 300 == 0 and m_current < m_lbfgs_max:
            m_current += 2

        # Trim history
        if len(s_hist) > m_current:
            s_hist = s_hist[-m_current:]
            y_hist = y_hist[-m_current:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Aggressive line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        improved = False
        for ls in range(20):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv_full(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls < c_max * (1.0 + ls * 0.001):
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    improved = True
                    break
            alpha_ls *= 0.6  # More aggressive backtracking

        if not improved:
            a_new = candidate  # Take the last candidate anyway
            cmax_new = cm_ls

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0

        g_new, f_new, c_nw, cmax_nw = lse_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_current:
                s_hist = s_hist[-m_current:]
                y_hist = y_hist[-m_current:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        # Periodic partial restart if stalled
        if stall_count > patience and it % 100 == 0:
            frac = 0.3 + 0.3 * min(stall_count / (patience * 3), 1.0)
            a = spectral_partial_restart(a, N, rng, frac=frac)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_grad(a, N, beta)

    return best_a, best_cmax


def peak_chase_block(a_init, n, S, N, max_time, rng, top_k=5):
    """Direct peak-chasing: identify top peaks and reduce them via targeted mass transfer."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()

    for it in range(50000):
        if time.monotonic() - start_time > max_time:
            break

        c, _ = compute_conv_full(a, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = a.copy()

        # Get top-k peak positions
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        if top_vals[0] <= top_vals[-1] * 1.001:
            # Peaks are nearly flat - stop
            break

        peak_k = int(top_idx[0])
        peak_val = c[peak_k]
        
        # Find all i that contribute to b[peak_k]
        # b[peak_k] = sum_i a[i] * a[peak_k - i] for valid i
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        
        # Compute contribution of each pair
        pairs = []
        for i in range(lo, hi + 1):
            j = peak_k - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                pairs.append((i, j, contrib))
        
        if not pairs:
            break
            
        pairs.sort(key=lambda x: -x[2])
        
        # Transfer mass from the largest contributing pair
        best_transferred = False
        for i, j, contrib in pairs[:10]:
            if i == j:
                # Single term a[i]^2: reduce a[i] and add to a position far from peak
                delta = 0.05 * a[i]
                if delta < 1e-15:
                    continue
                # Find a position that minimally contributes to the top peaks
                # Move to the position least involved in top peaks
                candidate_pos = (i + n // 2) % n
                if a[candidate_pos] < a[i] - delta:
                    a[i] -= delta
                    a[candidate_pos] += delta
                    best_transferred = True
                    break
            else:
                # Two terms: reduce the larger and add to the smaller or a neutral position
                if a[i] >= a[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                delta = 0.03 * min(a[src], a[i] if i != src else a[j])
                delta = 0.03 * a[src]
                if delta < 1e-15:
                    continue
                # Move mass to a position that doesn't contribute to the peak
                # Choose a position far from the peak
                mid = n // 2
                candidate_pos = (src + n // 3) % n
                a[src] -= delta
                a[candidate_pos] += delta
                best_transferred = True
                break
        
        if not best_transferred:
            # Try a random perturbation
            idx = rng.integers(0, n)
            delta = 0.01 * a.mean()
            if a[idx] > delta:
                a[idx] -= delta
                other = rng.integers(0, n)
                a[other] += delta
            else:
                break

    return best_a, best_cmax


def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov top-k subgradient for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        c, A = compute_conv_full(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        if it % 10 == 0:
            c_x, _ = compute_conv_full(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 2000 == 0 and rng.random() < 0.1:
            y = spectral_partial_restart(y, N, rng, frac=0.2)
            t = 1.0
    return best_a, best_cmax


def deflation_init(n, S, N, n_iters=1000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = np.real(ifft(np.abs(fft(a, N)) ** 2))[:2 * n - 1]
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    n_freqs = min(8, n // 100)
    for _ in range(n_freqs):
        freq = rng.integers(1, max(2, n // 10))
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_binary_like(n, rng):
    a = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def make_init_smooth_random(n, rng, N):
    """Generate a smooth random non-negative sequence via filtered random."""
    # Random in frequency domain with low-pass bias
    Nf = 1 << (2 * n - 1).bit_length()
    A = np.zeros(Nf, dtype=complex)
    # Low frequencies get more power
    freqs = np.arange(Nf)
    weights = 1.0 / (1.0 + freqs.astype(np.float64) / (n * 0.1))
    A = rng.standard_normal(Nf) + 1j * rng.standard_normal(Nf)
    A *= weights
    A[0] = float(n)  # Ensure non-zero DC
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.01
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-phase approach: diverse initialization, L-BFGS+LSE optimization,
    peak-chasing coordinate descent, and Nesterov polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Collect initializations
    inits = []

    # Try to use the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    # We'll try multiple n values and keep the best
    n_candidates = [8192, 4096, 16384]
    
    # Add more initializations for n=8192 (primary)
    n_primary = 8192
    inits.append(("rand", make_init_rand(n_primary, rng)))
    inits.append(("chirp", make_init_chirp(n_primary, rng)))
    inits.append(("zadoff", make_init_zadoff(n_primary, rng)))
    inits.append(("multifreq", make_init_multi_freq(n_primary, rng)))
    inits.append(("fourier", make_init_fourier(n_primary, rng, 1 << (2 * n_primary - 1).bit_length())))
    inits.append(("binary", make_init_binary_like(n_primary, rng)))
    inits.append(("smooth", make_init_smooth_random(n_primary, rng, 1 << (2 * n_primary - 1).bit_length())))
    inits.append(("deflation", deflation_init(n_primary, np.sqrt(2.0 * n_primary), 1 << (2 * n_primary - 1).bit_length(), 1500, rng)))

    global_best_score = np.inf
    global_best_seq = None

    def try_update(score, seq_list, label=""):
        nonlocal global_best_score, global_best_seq
        if score < global_best_score:
            global_best_score = score
            global_best_seq = list(seq_list)
            if label:
                print("  [NEW BEST] %s: score=%.6f" % (label, score))

    # --- Phase 1: Quick multi-start exploration at n=8192 ---
    print("=== Phase 1: Multi-start exploration (n=%d) ===" % n_primary)
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.15)

    S_primary = np.sqrt(2.0 * n_primary)
    N_primary = 1 << (2 * n_primary - 1).bit_length()

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    best_phase1_a = None
    best_phase1_score = np.inf

    for name, arr in inits:
        if time.monotonic() > deadline - 400:
            break
        arr_n = np.asarray(arr, dtype=np.float64).reshape(-1)
        # Resize to n_primary if needed
        if arr_n.size != n_primary:
            if arr_n.size > n_primary:
                arr_n = arr_n[:n_primary]
            else:
                # Pad or interpolate
                new_arr = np.zeros(n_primary)
                if arr_n.size > 0:
                    idx_old = np.linspace(0, 1, arr_n.size)
                    idx_new = np.linspace(0, 1, n_primary)
                    new_arr = np.interp(idx_new, idx_old, arr_n)
                arr_n = new_arr
        a_norm = safe_normalize(arr_n, n_primary, S_primary)
        t0 = time.monotonic()
        rem = max(2.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = lbfgs_lse_block(a_norm, n_primary, S_primary, N_primary, min(rem, 40.0), rng,
                                     beta_init=5.0, beta_max=80.0,
                                     m_lbfgs_init=8, m_lbfgs_max=15)
        score = float(2 * n_primary * sc_l / (S_primary * S_primary))
        if score < best_phase1_score:
            best_phase1_score = score
            best_phase1_a = a_l.copy()
        try_update(score, [float(x) for x in a_l], name + "/P1")

    print("Phase 1 done. Best score: %.6f" % best_phase1_score)

    if best_phase1_a is None:

                                         m_lbfgs_init=12, m_lbfgs_max=20)
        score_new = float(2 * n_primary * sc_new / (S_primary * S_primary))
        if score_new < global_best_score - 1e-8:

            stall = 0
            print("  [P2] score=%.6f (beta %.0f-%.0f)" % (score_new, beta_start, beta_end))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_partial_restart(a_curr, N_primary, rng, frac=0.3)

            a_pc = peak_chase_block(np.array(global_best_seq), n_primary, S_primary, N_primary, remaining3, rng, top_k=5)
        score_pc = float(2 * n_primary * sc_pc / (S_primary * S_primary))
        try_update(score_pc, [float(x) for x in a_pc], "peak_chase")

    # --- Phase 4: Nesterov polishing ---
    print("=== Phase 4: Nesterov polishing ===")
    t4 = time.monotonic()
    remaining4 = min(200.0, (deadline - t4) * 0.15)
    if remaining4 > 20:
        start_a = a_curr
        if global_best_seq is not None and len(global_best_seq) == n_primary:

        a_ng, sc_ng = nesterov_toppk_block(start_a, n_primary, S_primary, N_primary, remaining4, rng,

    # --- Phase 5: Try n=4096 if time permits ---
    if deadline - time.monotonic() > 150:

        n_alt = 4096

        N_alt = 1 << (2 * n_alt - 1).bit_length()

        budget5 = min(150.0, deadline - t5 - 50)
        
        # Use a few diverse inits
        alt_inits = [
            make_init_rand(n_alt, rng),
            make_init_multifreq(n_alt, rng),
            make_init_smooth_random(n_alt, rng, N_alt),
        ]
        per_alt = budget5 / len(alt_inits)
        for idx_alt, arr_alt in enumerate(alt_inits):
            if time.monotonic() > deadline - 50:
                break
            a_norm_alt = safe_normalize(arr_alt, n_alt, S_alt)
            a_alt, sc_alt = lbfgs_lse_block(a_norm_alt, n_alt, S_alt, N_alt, min(per_alt, 50.0), rng,
                                             beta_init=5.0, beta_max=100.0,
                                             m_lbfgs_init=10, m_lbfgs_max=18)
            score_alt = float(2 * n_alt * sc_alt / (S_alt * S_alt))
            try_update(score_alt, [float(x) for x in a_alt], "n4096_%d" % idx_alt)
            # Quick polish
            rem5 = max(10.0, per_alt - 50.0 - 5)
            if rem5 > 10:
                a_pol, sc_pol = nesterov_toppk_block(a_alt, n_alt, S_alt, N_alt, rem5, rng,

    print("=== Phase 6: Final polishing ===")
    remaining6 = deadline - time.monotonic() - 15
    if remaining6 > 30 and global_best_seq is not None:

    print("Returning length %d, best score=%.6f" % (len(global_best_seq), global_best_score))
