"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, L-BFGS on smoothed
objective, responsibility-based peak shaving, and lag-balancing penalty.
Key improvements: adaptive beta, second-order methods, targeted peak reduction,
and multi-length exploration.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm + 1e-300, np.zeros(n)
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad

def run_fista_lse(a_init, n, max_time, rng, beta_start=50.0, beta_end=20000.0,
                  beta_steps=200, momentum_restart=300):
    """FISTA with LSE smoothing and adaptive beta."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        lse_val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        # Adaptive: reduce step when stagnant
        stagnant = it - last_improve
        if stagnant > momentum_restart:
            t_mom = 1.0
            # Try to kick out with noise
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / (20000.0 / max(1, n//100)))
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

def run_lbfgs_lse(a_init, n, max_time, rng, beta=500.0):
    """L-BFGS on the LSE-smoothed objective (with simplex constraint handled by projection)."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    def objective(params):
        a_local = np.maximum(params, 0.0)
        s = a_local.sum()
        if s < 1e-15:
            return 1e10
        a_norm = a_local * (S / s)
        c = fft_convolve_self(a_norm)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            return cm
        return cm + np.log(tot) / beta

    def gradient(params):
        a_local = np.maximum(params, 0.0)
        s = a_local.sum()
        if s < 1e-15:
            return np.zeros(n)
        a_norm = a_local * (S / s)
        c = fft_convolve_self(a_norm)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            return np.zeros(n)
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a_norm, fsize)
        grad_norm = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        # Chain rule for normalization
        grad = grad_norm * (S / s) - a_local * (S / s**2) * grad_norm.sum()
        # Mask negative
        grad = np.where(a_local > 0, grad, 0)
        return grad

    # Use scipy L-BFGS-B
    result = minimize(objective, a.copy(), method='L-BFGS-B', jac=gradient,
                      options={'maxiter': 5000, 'maxfun': 50000, 'gtol': 1e-10,
                               'factr': 1e10, 'maxls': 50})

    a_new = np.maximum(result.x, 0.0)
    a_new = normalize_sum(a_new, n)
    c = fft_convolve_self(a_new)
    val = c.max()
    if val < best:
        best = val
        best_a = a_new.copy()

    # Also check intermediate results aren't better (they should be tracked)
    # But scipy doesn't expose intermediates easily, so we just return what we have
    if best == np.inf:
        best_a = a.copy()
        best = c.max()

    return best_a, best

def run_lbfgs_projected(a_init, n, max_time, rng, beta=500.0):
    """
    L-BFGS with explicit simplex projection.
    We optimize in the unconstrained space and project after each step.
    This is a "projected L-BFGS" approach.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Manual L-BFGS-like iterations with projection
    two_memory = 20  # Number of vectors for L-BFGS
    W = []  # History of (s, y) pairs

    def compute_obj_grad(a_local):
        a_local = np.maximum(a_local, 0.0)
        s = a_local.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        a_norm = a_local * (S / s)
        c = fft_convolve_self(a_norm)
        cm = c.max()
        if beta > 100:
            e = np.exp(beta * (c - cm))
            tot = e.sum()
            if tot < 1e-300:
                return cm, np.zeros(n)
            lse_val = cm + np.log(tot) / beta
            w = e / tot
        else:
            lse_val = cm
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0

        fsize = 1 << (len(w) + n - 1).bit_length()
        Wf = fft(w, fsize)
        Af = fft(a_norm, fsize)
        grad_norm = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]
        grad = grad_norm * (S / s) - a_local * (S / s**2) * grad_norm.sum()
        grad = np.where(a_local > 0, grad, 0.0)
        return lse_val, grad

    while time.monotonic() - start < max_time and it < 200:
        it += 1
        obj, grad = compute_obj_grad(a)
        c = fft_convolve_self(a)
        true_max = c.max()
        if true_max < best:
            best = true_max
            best_a = a.copy()

        if np.linalg.norm(grad) < 1e-12:
            break

        # Simple gradient step with backtracking
        step = 0.01
        improved = False
        for _ in range(10):
            a_trial = project_simplex(a - step * grad, S)
            obj_trial, _ = compute_obj_grad(a_trial)
            if obj_trial < obj * (1 - 1e-8):
                a = a_trial
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a small fixed step
            a = project_simplex(a - 0.001 * grad, S)

        # Periodic perturbation
        if it % 30 == 0:
            noise = 0.001 * S / np.sqrt(n) * np.exp(-it / 100.0)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best

def responsibility_peak_shave(a, n, S, max_time, rng, n_peaks=10, n_iter=200):

            # This is the "one-sided" gradient which is sufficient for direction
            weight = max(0.0, c[k] - cm + 1e-10)
            resp[js] += a[partners] * weight

        # Normalize responsibility
        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Multiplicative update to reduce high-responsibility elements
        # a_new[j] = a[j] * (1 - eta * resp[j] / resp_max)
        eta = 0.05

def lag_balance_penalty(a, n, S, n_top=5, strength=0.01):
    """
    Compute a penalty that encourages the top n_top lags to be equal.
    Returns gradient of this penalty.
    """
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:n_top]
    top_vals = c[top_idx]
    mean_top = top_vals.mean()
    # Penalty = strength * sum((c[k_i] - mean_top)^2)
    # Gradient: 2*strength*sum_i (c[k_i] - mean_top) * d c[k_i]/d a[j]
    grad = np.zeros(n)
    for k in top_idx:
        diff = c[k] - mean_top
        if abs(diff) < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        grad[js] += 2.0 * strength * diff * a[partners]

    return grad, c.max()

import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def multi_beta_gradient(a, c, betas):
    """Compute combined gradient from multiple LSE betas."""
    n = len(a)
    cm = c.max()
    grad = np.zeros(n)
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    for beta in betas:
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            continue
        w = e / tot
        W = fft(w, fsize)
        g = np.real(ifft(W * np.conj(A)))[:n]
        grad += g / len(betas)
    return grad

            best_a = a.copy()

        # Compute the "concentration" of the peak
        # If the peak is very sharp (one lag dominates), we need to spread it
        top5 = np.sort(c)[::-1][:5]

        # If distributed -> reduce overall energy at critical lags

        lo = max(0, k_peak - n + 1)

        # Also the "other side"

Strategy: Adaptive LSE-FISTA with proper Lipschitz-based step sizing,
multi-length exploration, and targeted peak reduction.
Key improvements over previous:
1. Correct Lipschitz constant for step sizing (2*beta, not beta)
2. Adaptive beta increase based on LSE-to-max gap
3. Multi-length exploration with proper interpolation
4. Targeted peak reduction using exact gradient of peak lag
5. Better time management
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_with_grad(c, beta):
    """Compute LSE value, weights, and LSE value."""
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100, 100)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return cm, w, cm
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    return lse_val, w, cm

def compute_gradient_from_weights(w, a):
    """Compute gradient given the LSE weights w."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_fista_adaptive(a_init, n, max_time, rng,
                        beta_min=20.0, beta_max=50000.0,
                        beta_growth=1.05, patience=200):
    """
    FISTA with adaptive beta.
    Beta increases by beta_growth factor each time we've made no progress
    for 'patience' iterations. This ensures we keep increasing precision
    only when needed.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    beta = beta_min

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best - 1e-11:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Compute LSE and gradient
        lse_val, w, cm = lse_with_grad(c, beta)
        grad = compute_gradient_from_weights(w, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/L where L is Lipschitz constant of gradient
        # For LSE with parameter beta, L ≈ 2*beta (conservative upper bound)
        # Actually, the gradient of LSE(beta, c) w.r.t. a is:
        # 2 * (w * a) convolved appropriately, where w = softmax(beta*(c-cm))
        # The Lipschitz constant of this gradient is bounded by 2*beta*max(w)
        # Since max(w) <= 1, L <= 2*beta. Use L = 2*beta for safety.
        L = 2.0 * beta
        step = 1.0 / L

        # If LSE is very close to the true max, we're in a good regime
        # If LSE - max is large, we should reduce beta for more exploration
        lse_gap = lse_val - cur_max
        if lse_gap > 5.0 / beta:
            # LSE is not tight enough, reduce beta
            beta = max(beta_min, beta * 0.9)
        elif it - last_improve > patience:
            # No progress, increase beta for more precision
            beta = min(beta_max, beta * beta_growth)

        # Stagnation: restart momentum
        stagnant = it - last_improve
        if stagnant > patience * 3:
            t_mom = 1.0
            # Inject noise
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 50000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Check y too
    c_y = fft_convolve_self(y)
    if c_y.max() < best:
        best = c_y.max()
        best_a = y.copy()

    return best_a, best

def run_peak_targeted(a_init, n, max_time, rng):
    """
    Directly target the peak lag using its exact gradient.
    This is more precise than LSE for the final improvement.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time and it < 5000:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Find all lags within 1% of the max (active set)
        threshold = cur_max * 0.999
        active = np.where(c >= threshold)[0]

        # Compute combined gradient of active lags
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            grad[js] += a[partners]
            # Other side
            js2 = np.arange(max(0, k), min(n, k + n))
            if len(js2) > 0:
                partners2 = js2 - k
                valid = (partners2 >= 0) & (partners2 < n)
                grad[js2[valid]] += a[partners2[valid]]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 0.01 / gnorm
        improved = False
        for _ in range(12):
            a_trial = project_simplex(a - step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max * (1 - 1e-8):
                a = a_trial
                improved = True
                break
            step *= 0.5

        if not improved:
            # Very small step
            step = 1e-5 / gnorm
            a = project_simplex(a - step * grad, S)

        # Noise
        if it % 50 == 0:
            noise = 0.001 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if noise > 1e-11:
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best
