"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
New approach: Progressive L-BFGS with extended iterations, softmax parameterization,
symmetrization, multi-n exploration, and adaptive peak-shaving.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_obj_and_grad(x, beta, S):
    """LSE objective and gradient for L-BFGS, with penalty for sum constraint."""
    x = np.maximum(x, 0.0)
    c = fft_convolve_self(x)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    obj = cm + np.log(e.sum()) / beta
    sum_x = x.sum()
    # Soft penalty for sum constraint
    pen = 100.0
    obj += pen * (sum_x - S) ** 2
    grad = conv_gradient(x, w)
    grad += 2.0 * pen * (sum_x - S)
    return obj, grad


def lse_obj_and_grad_strong(x, beta, S, pen):
    """LSE objective with stronger sum penalty."""
    x = np.maximum(x, 0.0)
    c = fft_convolve_self(x)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    obj = cm + np.log(e.sum()) / beta
    sum_x = x.sum()
    obj += pen * (sum_x - S) ** 2
    grad = conv_gradient(x, w)
    grad += 2.0 * pen * (sum_x - S)
    return obj, grad


def run_lbfgs_extended(a_init, max_time, beta_schedule, maxiter_per_stage=500):
    """
    Extended L-BFGS with progressive beta schedule.
    Much longer runs than before.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    bounds = [(0, 1000)] * n

    for stage, beta in enumerate(beta_schedule):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        # Adaptive penalty: start loose, tighten over stages
        pen = max(1.0, 50.0 * (stage + 1) / len(beta_schedule))

        def obj_fn(x):
            return lse_obj_and_grad_strong(x, beta, S, pen)

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_fn, a.copy(), method='L-BFGS-B',
            jac=True, bounds=bounds,
            options={'maxiter': maxiter_per_stage, 'maxls': 50,
                     'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0

        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        obj_new = compute_obj(a_new)

        print("    L-BFGS stage %d: beta=%.0f pen=%.1f obj=%.8f (t=%.1fs, nit=%d)" %
              (stage, beta, pen, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_fista_refined(a_init, max_time, rng, beta_start=20.0, step_scale=1.0):
    """
    Refined FISTA with adaptive step and periodic restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive beta
        beta = beta_start * (1.0 + 0.01 * it)
        beta = min(beta, 20000.0)

        # Gradient with LSE
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size (Barzilai-Borwein inspired)
        step = step_scale / max(beta, 1.0)
        step = min(step, S / (gnorm + 1e-15))

        # When stuck, use exact subgradient at the peak
        if stuck_count > 500:
            k_峰 = int(np.argmax(c))
            w_exact = np.zeros(len(c))
            w_exact[k_峰] = 1.0
            grad_exact = conv_gradient(y, w_exact)
            # Blend: use exact subgradient more heavily when stuck
            blend = min(1.0, stuck_count / 2000.0)
            grad = (1 - blend) * grad + blend * grad_exact
            gnorm = np.linalg.norm(grad)
            step = min(step, S / (gnorm + 1e-15))

        # DCT-domain structured noise when very stuck
        if stuck_count > 3000 and it % 100 == 0:
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0))
            noise = rng.normal(0, 1.0, n)
            F = fft(noise)
            cutoff = max(3, n // 40)
            F[cutoff:-cutoff] *= 0.05
            noise_smooth = np.real(ifft(F))[:n]
            noise_norm = np.linalg.norm(noise_smooth)
            if noise_norm > 1e-15:
                noise_smooth = noise_smooth / noise_norm * noise_scale
            y = project_simplex(y + noise_smooth, S)
            t_mom = 1.0
            stuck_count = max(0, stuck_count - 1000)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

    return best_a


def run_peak_shaving(a_init, max_time, rng):
    """
    Direct peak shaving: identify the peak lag and explicitly reduce it.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()

        # Find all peaks within 0.1% of max
        thresh = cur_max * 0.999
        peak_lags = np.where(c >= thresh)[0]

        if len(peak_lags) == 0:
            break

        # For each peak lag k, c[k] = sum_j a[j]*a[k-j]
        # To reduce c[k], we want to reduce a[j] where both a[j] and a[k-j] are large.
        # Compute the "blame" for each index j across all peak lags.
        blame = np.zeros(n)
        for k in peak_lags:
            # c[k] = sum_j a[j]*a[k-j] (valid j)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                blame[j] += a[j] * a[k - j]

        # Normalize blame
        blame_sum = blame.sum()
        if blame_sum < 1e-15:
            break
        blame = blame / blame_sum

        # Reduce blamed indices, increase others (maintain sum)
        # Use a smooth interpolation
        alpha = 0.1  # blending factor
        a_new = a * (1 - alpha * blame) + alpha * (S / n) * (1 - blame)
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        else:
            break

        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max - 1e-10:
            a = a_new
        else:
            # Try smaller step
            alpha /= 2
            if alpha < 1e-6:
                break
            a_new = a * (1 - alpha * blame) + alpha * (S / n) * (1 - blame)
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new * (S / a_new.sum())
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max - 1e-12:
                    a = a_new
                else:
                    break
            else:
                break

    return best_a

def symmetrize(a):
    """Symmetrize a: a_sym[j] = (a[j] + a[n-1-j])/2."""

    return a_sym

            a6, obj6 = run_lbfgs_extended(global_best_a, rem6, beta_sched_short, maxiter_per_stage=300)
            if obj6 < global_best_obj:
                global_best_obj = obj6
                global_best_a = a6.copy()

        print("  Phase 6 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t6))

    # === Phase 7: Final L-BFGS squeeze (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p7_budget = remaining * 0.85
        print("\n=== Phase 7: Final L-BFGS squeeze (budget %.0fs) ===" % p7_budget)
        t7 = time.monotonic()

        beta_sched_final = [500.0, 1000.0, 2000.0, 5000.0, 10000.0]
        a7, obj7 = run_lbfgs_extended(global_best_a, p7_budget, beta_sched_final, maxiter_per_stage=300)
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj7, time.monotonic() - t7))

    # === Phase 8: One more FISTA pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 8: Final FISTA (budget %.0fs) ===" % remaining)
        a8 = run_fista_refined(global_best_a, remaining * 0.9, rng, beta_start=500.0, step_scale=0.3)
        obj8 = compute_obj(a8)
        if obj8 < global_best_obj:
            global_best_obj = obj8
            global_best_a = a8.copy()
        print("  Final FISTA: obj=%.8f" % obj8)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
