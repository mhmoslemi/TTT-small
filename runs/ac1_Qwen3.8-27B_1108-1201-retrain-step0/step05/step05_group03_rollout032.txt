Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Hybrid L2-flattening + peak-shaving with adaptive step sizes.
Key differences from previous:
1. Uses L2 flattening (smooth) for global structure, then peak shaving for refinement

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def l2_flatten_grad(a, c):
    """
    Gradient of L2 flattening objective: sum_k (c[k] - mean(c))^2
    with respect to a.
    
    grad[j] = 4 * sum_k (c[k] - mean_c) * a[k-j] for valid k-j
    This is 4 * cross_correlation(c - mean_c, a)[j]
    """
    n = len(a)
    c_size = len(c)
    mean_c = c.mean()
    d = c - mean_c
    
    # cross-correlation: result[j] = sum_k d[k] * a[k-j] for valid k-j
    # This is the convolution of d with reversed a
    a_rev = a[::-1]
    # convolve d (length 2n-1) with a_rev (length n)
    # result has length 2n-1 + n - 1 = 3n-2, we want the first n elements
    # Actually, we want: result[j] = sum_k d[k] * a[k-j]
    # = (d * a_rev)[j + n - 1] ... let me think more carefully
    
    # d has length 2n-1, a has length n
    # We want result[j] = sum_{k: 0<=k<2n-1, 0<=k-j<n} d[k] * a[k-j]
    # Let m = k-j, so k = m+j. We need 0 <= m < n and 0 <= m+j < 2n-1
    # So result[j] = sum_{m: max(0,-j) <= m < min(n, 2n-1-j)} d[m+j] * a[m]
    # This is the cross-correlation of d and a
    
    # Using FFT: result = ifft(fft(d) * conj(fft(a))) (with appropriate padding)
    fsize = 1 << (c_size + n - 1).bit_length()
    D = fft(d, fsize)
    A = fft(a, fsize)
    # cross-correlation = ifft(D * conj(A))
    result = np.real(ifft(D * np.conj(A)))
    # The cross-correlation gives: result[m] = sum_k d[k] * a[k-m+n-1]... 
    # Actually, ifft(FFT(d) * conj(FFT(a))) gives the circular cross-correlation
    # For linear cross-correlation with proper zero-padding, we get:
    # result[j] = sum_k d[k+j] * a[k] (with valid ranges)
    # This is what we want (shifted appropriately)
    
    # Let me verify: the linear cross-correlation (d ⋆ a)[j] = sum_k d[k+j] * a[k]
    # We want result[j] = sum_m d[m+j] * a[m] which is exactly (d ⋆ a)[j]
    
    return 4.0 * result[:n]


def peak_shave_grad(a, c, peak_indices):
    """
    Gradient of sum_{k in peak_indices} c[k] with respect to a.
    grad[j] = 2 * sum_{k in peak_indices} a[k-j] for valid k-j
    """
    n = len(a)
    # Create a sparse representation of the peaks
    peak_vec = np.zeros(len(c))
    peak_vec[peak_indices] = 1.0
    
    # Cross-correlation with a
    A = fft(a, 1 << (len(c) + n - 1).bit_length())
    P = fft(peak_vec, len(A))
    result = np.real(ifft(P * np.conj(A)))[:n]
    
    return 2.0 * result


def top_k_lse_grad(a, c, k, beta):
    """Top-k LSE gradient (same as before but cleaner)."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def chirp_init(n, rng):
    """
    Generate a chirp-based initialization.
    Uses the real part of a linear chirp, rectified and normalized.
    """
    # Linear chirp: a[j] = exp(i * pi * j^2 / n)
    # Real part: cos(pi * j^2 / n)
    # For non-negative, use |cos| or (1 + cos) / 2
    j = np.arange(n)
    phase = np.pi * j * j / n
    a = 0.5 + 0.5 * np.cos(phase)
    a = np.maximum(a, 0.0)
    # Add some randomness
    a = a * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * n)
    return normalize(a, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Random phase perturbation
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def hybrid_optimize(a_init, S, max_time, rng, mode="hybrid"):

def weighted_difference_method(a_init, S, max_time, rng):
    """
    WDMA (Weighted Difference Method) for low-PSL waveform design.
    This is a classical method from radar/sonar literature.
    
    The idea: iteratively apply a correction that reduces the sidelobes
    of the autocorrelation while preserving the main lobe (DC component).
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    mu = 0.1  # Step size
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # WDMA: Apply a "window" in the lag domain to suppress sidelobes
        # The window w[k] = 0 for |k| < main_lobe_width, w[k] = 1 otherwise
        # Then the correction is: delta_a = ifft(FFT(a) * w * conj(FFT(a)))
        # But for our case (non-negative constraint), we need to be more careful.
        
        # Alternative WDMA approach:
        # 1. Compute autocorrelation c
        # 2. Identify sidelobes (all lags except the center)
        # 3. Apply correction: a_new = a - mu * sum of sidelobe contributions
        
        # The sidelobe contribution at lag k is c[k] - c_center (for k != center)
        # We want to reduce the sidelobes while keeping the center lobe
        
        center = n - 1
        # Create a mask: 0 at center, 1 elsewhere (for sidelobes)
        # Actually, we want to reduce ALL peaks, not just sidelobes
        # Let's use: correction = c - mean(c) (flatten towards the mean)
        
        mean_c = c.mean()
        excess = np.maximum(c - mean_c, 0)  # Only positive excess (peaks)
        
        # Gradient of sum of excess with respect to a
        # This is the cross-correlation of excess with a
        fsize = 1 << (len(excess) + n - 1).bit_length()
        E = fft(excess, fsize)
        A = fft(a, fsize)
        grad = np.real(ifft(E * np.conj(A)))[:n]
        
        # Also include the negative part (valleys) with less weight
        deficit = np.maximum(mean_c - c, 0)
        E_def = fft(deficit, fsize)
        grad_def = np.real(ifft(E_def * np.conj(A)))[:n]
        
        # Combined gradient: reduce peaks more than boost valleys
        grad = 2.0 * grad - 0.3 * grad_def
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = mu
        found = False
        for _ in range(8):
            a_trial = project_simplex(a - step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max + 1e-10:
                found = True
                break
            step *= 0.5
        
        if found:
            a = a_trial
            mu = min(mu * 1.05, 1.0)
        else:
            mu = max(mu * 0.7, 0.001)
            a = project_simplex(a - step * grad, S)
        
        # Decay the asymmetry over time (become more symmetric)
        # This helps converge to a flatter autocorrelation
        progress = (time.monotonic() - start) / max_time
        if progress > 0.5:
            # Switch to symmetric flattening
            pass  # Already handled by the 0.3 weight
        
        # Periodic perturbation
        if it % 2000 == 0 and progress < 0.7:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

Strategy: Multi-phase optimization with:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def cross_corr_fft(d, a):
    """
    Compute linear cross-correlation: result[j] = sum_k d[k+j] * a[k]
    using FFT. d has length >= n, a has length n.
    Returns result of length n.
    """
    n_a = len(a)
    n_d = len(d)
    fsize = 1 << (n_d + n_a - 1).bit_length()
    D = fft(d, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(D * np.conj(A)))
    return result[:n_a]


def l2_flatten_gradient(a, c):
    """
    Gradient of 0.5 * sum_k (c[k] - mean(c))^2 with respect to a.
    = 2 * cross_correlation(c - mean, a)
    """
    n = len(a)
    mean_c = c.mean()
    d = c - mean_c
    return 2.0 * cross_corr_fft(d, a)


def peak_shave_gradient(a, c, peak_indices):
    """
    Gradient of sum_{k in peak_indices} c[k] with respect to a.
    = 2 * cross_correlation(peak_vector, a)
    """
    n = len(a)
    peak_vec = np.zeros(len(c))
    peak_vec[peak_indices] = 1.0
    return 2.0 * cross_corr_fft(peak_vec, a)


def topk_lse_gradient(a, c, k, beta):
    """Top-k LogSumExp smoothed max gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w
    return 2.0 * cross_corr_fft(w_full, a)


def wdma_gradient(a, c, asymmetry=1.0):
    """
    WDMA-style gradient: asymmetric weighting of peaks vs valleys.
    Peaks get full weight, valleys get reduced (or zero) weight.
    """
    n = len(a)
    mean_c = c.mean()
    excess = np.maximum(c - mean_c, 0.0)
    deficit = np.maximum(mean_c - c, 0.0)
    # Weight: peaks fully, valleys partially (controlled by asymmetry)
    weight = excess - asymmetry * deficit
    return 2.0 * cross_corr_fft(weight, a)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def chirp_init(n, rng):
    """Generate a chirp-based non-negative initialization."""
    j = np.arange(n, dtype=np.float64)
    # Use multiple chirp rates for diversity
    rate = 1.0 + 0.5 * rng.random()
    phase = np.pi * rate * j * j / n
    a = 0.5 + 0.5 * np.cos(phase)
    # Mix with a second chirp
    phase2 = np.pi * 0.7 * j * j / n
    a = 0.6 * a + 0.4 * (0.5 + 0.5 * np.cos(phase2))
    a = np.maximum(a, 0.0)
    a = a * (1.0 + 0.05 * rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * n)
    return normalize(a, S)


def spectral_flatten(a, alpha, n_iter=10):
    """Apply spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iter):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-20:
            break
        ratio = power / max(mean_p, 1e-20)
        # Damp high-power, boost low-power
        new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio + 1e-10))
        new_power = np.maximum(new_power, 1e-30)
        A_new = A * np.sqrt(new_power / power)
        a = np.real(ifft(A_new, fsize))[:n]
        a = np.clip(a, 0, None)
        a = normalize(a, S)
    return a


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain for diversity."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def optimized_gradient_step(a, c, S, n, it, total_iter, rng):
    """
    Compute a blended gradient combining:
    - Top-k LSE (sharp max approximation)
    - L2 flattening (smooth global)
    - WDMA (asymmetric peak/valley)
    
    Returns the gradient vector.
    """
    progress = it / max(total_iter, 1)
    
    # Phase weights: transition from smooth to sharp
    w_lse = 0.3 + 0.5 * progress       # 0.3 -> 0.8
    w_l2 = 0.7 - 0.5 * progress        # 0.7 -> 0.2  
    w_wdma = 0.2 * (1.0 - progress)    # 0.2 -> 0
    
    # Adaptive k for LSE
    k_frac = max(0.005, 0.08 * (1.0 - progress))
    k = max(4, int(k_frac * len(c)))
    
    # Adaptive beta for LSE
    beta = min(5000.0, 20.0 * (1.0 + 2.0 * progress))
    
    grad = np.zeros(n)
    
    if w_lse > 0.01:
        grad += w_lse * topk_lse_gradient(a, c, k, beta)
    
    if w_l2 > 0.01:
        grad += w_l2 * l2_flatten_gradient(a, c)
    
    if w_wdma > 0.01:
        grad += w_wdma * wdma_gradient(a, c, asymmetry=0.3)
    
    return grad


def nesterov_optimize(a_init, S, max_time, rng, total_iter_est=5000):

    Nesterov accelerated gradient descent on the simplex with blended gradient.
    """

    y = a.copy()
    t_val = 1.0
    start = time.monotonic()

    best = c.max()
    best_a = y.copy()

    step_size = 1.0 / max(2.0 * 20.0 * 0.1, 1e-10)
    last_improve_it = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:

        c = fft_convolve_self(y)

            best_a = y.copy()

        else:
            stuck_count += 1

        c_alpha = 1e-4

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        # Renormalize y to maintain feasibility (momentum can push off simplex)
        y_new = normalize(y_new, S)

        best_a = a.copy()
    
    return best_a, best


def high_beta_polish(a_init, S, max_time, rng):
    """
    Very high beta polishing: focus almost exclusively on the single worst peak.
    Uses small steps with careful backtracking.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    step_size = 0.5
    t_val = 1.0
    y = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = y.copy()
        
        # Very high beta, small k
        k = max(2, int(0.003 * len(c)))
        beta = min(10000.0, 200.0 * (1.0 + it / 200.0))
        
        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = step_size
        x_new = None
        for _ in range(10):
            candidate = project_simplex(y - step * grad, S)
            c_candidate = fft_convolve_self(candidate)
            if c_candidate.max() <= cur_max + 1e-14:
                x_new = candidate
                break
            step *= 0.5
        
        if x_new is None:
            step_size = max(step, 1e-8)
            x_new = project_simplex(y - step * grad, S)
        else:
            step_size = min(step * 1.03, 2.0)
        
        # Momentum (light)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + 0.5 * ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, S)
        
        a = x_new
        y = y_new
        t_val = t_new
        
        # Very rare perturbation
        if it % 10000 == 0 and (time.monotonic() - start) / max_time < 0.4:
            noise_scale = 0.0005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_val = 1.0
            step_size = 0.5
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def wdma_optimize(a_init, S, max_time, rng):
    """
    WDMA-inspired optimization with adaptive asymmetry.
    Early: strong peak suppression (high asymmetry)
    Late: balanced flattening (low asymmetry)
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    step_size = 0.5
    t_val = 1.0
    y = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = y.copy()
        
        progress = (time.monotonic() - start) / max_time
        # Asymmetry: start high (focus on peaks), decrease over time
        asymmetry = max(0.0, 0.8 * (1.0 - progress))
        
        grad = wdma_gradient(y, c, asymmetry=asymmetry)
        # Blend with L2 for stability
        grad += 0.3 * l2_flatten_gradient(y, c)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = step_size
        x_new = None
        for _ in range(10):
            candidate = project_simplex(y - step * grad, S)
            c_candidate = fft_convolve_self(candidate)
            if c_candidate.max() <= cur_max + 1e-12:
                x_new = candidate
                break
            step *= 0.5
        
        if x_new is None:
            step_size = max(step, 1e-8)
            x_new = project_simplex(y - step * grad, S)
        else:
            step_size = min(step * 1.04, 3.0)
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, S)
        
        a = x_new
        y = y_new
        t_val = t_new
        
        # Perturbation
        if it % 3000 == 0 and progress < 0.6:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_val = 1.0
            step_size = 0.5
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def coarse_to_fine(a_coarse, n_fine, S_fine, max_time, rng):
    """
    Take a coarse solution and upscale + refine to fine resolution.
    """
    n_coarse = len(a_coarse)
    # Upsample
    a = resample_sequence(a_coarse, n_fine)
    a = normalize(a, S_fine)
    
    # Apply a few spectral flattening steps at fine resolution
    a = spectral_flatten(a, 0.03, n_iter=5)
    
    # Then run Nesterov optimization
    a_opt, val_opt = nesterov_optimize(a, S_fine, max_time, rng)
    return a_opt, val_opt


def run_pipeline_for_length(n, a_start, time_limit, rng, desc=""):
    """
    Run the full optimization pipeline for a given sequence length.
    Returns (best_a, best_max_conv).
    """
    S = np.sqrt(2.0 * n)
    
    if a_start is not None and len(a_start) != n:
        a_start = resample_sequence(a_start, n)
        a_start = normalize(a_start, S)
    if a_start is None:
        a_start = np.full(n, S / n)
    
    c0 = fft_convolve_self(a_start)
    best = c0.max()
    best_a = a_start.copy()
    start_time = time.monotonic()
    
    print("  [%s] n=%d, initial max_conv=%.6f, obj=%.6f" % (
        desc, n, best, 2.0 * n * best / S**2))
    
    # Phase 1: Nesterov with blended gradient (45% of time)
    t1 = time_limit * 0.40
    if t1 > 30:
        a1, val1 = nesterov_optimize(best_a, S, t1, rng)
        obj1 = 2.0 * n * val1 / (a1.sum()**2)
        print("  [%s] After Nesterov: max_conv=%.6f, obj=%.6f" % (desc, val1, obj1))
        if val1 < best:
            best = val1
            best_a = a1.copy()
    
    # Phase 2: WDMA (20% of time)
    t2 = time_limit * 0.20
    remaining = (start_time + time_limit) - time.monotonic()
    if t2 > 20 and remaining > 40:
        a2, val2 = wdma_optimize(best_a, S, min(t2, remaining - 30), rng)
        obj2 = 2.0 * n * val2 / (a2.sum()**2)
        print("  [%s] After WDMA: max_conv=%.6f, obj=%.6f" % (desc, val2, obj2))
        if val2 < best:
            best = val2
            best_a = a2.copy()
    
    # Phase 3: High-beta polish (25% of time)
    t3 = time_limit * 0.25
    remaining = (start_time + time_limit) - time.monotonic()
    if t3 > 15 and remaining > 30:
        a3, val3 = high_beta_polish(best_a, S, min(t3, remaining - 20), rng)
        obj3 = 2.0 * n * val3 / (a3.sum()**2)
        print("  [%s] After polish: max_conv=%.6f, obj=%.6f" % (desc, val3, obj3))
        if val3 < best:
            best = val3
            best_a = a3.copy()
    
    # Phase 4: Final short Nesterov burst (10% of time)
    t4 = time_limit * 0.10
    remaining = (start_time + time_limit) - time.monotonic()
    if t4 > 10 and remaining > 15:
        a4, val4 = nesterov_optimize(best_a, S, min(t4, remaining - 10), rng)
        obj4 = 2.0 * n * val4 / (a4.sum()**2)
        print("  [%s] After final burst: max_conv=%.6f, obj=%.6f" % (desc, val4, obj4))
        if val4 < best:
            best = val4
            best_a = a4.copy()
    
    final_obj = 2.0 * n * best / (best_a.sum()**2)
    print("  [%s] Best: obj=%.6f" % (desc, final_obj))
    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    
    Multi-phase, multi-length strategy:
    1. Primary: n=8192 from given sequence (Nesterov + WDMA + polish)
    2. Secondary: n=16384 (upsampled + refine)
    3. Alternative: chirp initialization at n=8192
    4. Final: intensive polishing of the global best
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 25.0)
    
    print("=" * 60)
    print("Starting optimization, budget=%.0fs" % budget_s)
    print("=" * 60)
    
    # Load given sequence
    given = None
    given_n = 0
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            given_n = len(given)
            given_obj = objective_value(given)
            print("Given sequence: n=%d, obj=%.6f" % (given_n, given_obj))
    except Exception:
        given = None
        given_n = 8192
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, n):
        nonlocal global_best, global_a
        obj = 2.0 * n * fft_convolve_self(a).max() / (a.sum()**2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  >>> GLOBAL BEST: obj=%.6f (n=%d)" % (obj, n))
        return obj
    
    # === PHASE A: Primary optimization at n=8192 ===
    n_primary = 8192
    remaining = deadline - time.monotonic()
    t_primary = min(remaining * 0.45, 900)
    
    if t_primary > 60:
        if given is not None:
            a_start = given.copy() if len(given) == n_primary else resample_sequence(given, n_primary)
            a_start = normalize(a_start, np.sqrt(2.0 * n_primary))
        else:
            a_start = chirp_init(n_primary, rng)
        
        a_best, val_best = run_pipeline_for_length(n_primary, a_start, t_primary, rng, "Primary")
        update_global(a_best, n_primary)
    
    # === PHASE B: Try n=16384 ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        n_secondary = 16384
        if global_a is not None:
            a_start_sec = resample_sequence(global_a, n_secondary)
        elif given is not None:
            a_start_sec = resample_sequence(given, n_secondary)
        else:
            a_start_sec = chirp_init(n_secondary, rng)
        
        t_sec = min(remaining * 0.30, 400)
        if t_sec > 60:
            a_sec, val_sec = run_pipeline_for_length(n_secondary, a_start_sec, t_sec, rng, "n=16384")
            update_global(a_sec, n_secondary)
    
    # === PHASE C: Chirp-based alternative at n=8192 ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_chirp = chirp_init(n_primary, rng)
        # Apply spectral flattening for better init
        a_chirp = spectral_flatten(a_chirp, 0.05, n_iter=10)
        t_chirp = min(remaining * 0.20, 300)
        if t_chirp > 30:
            a_chirp_opt, val_chirp = run_pipeline_for_length(n_primary, a_chirp, t_chirp, rng, "Chirp")
            update_global(a_chirp_opt, n_primary)
    
    # === PHASE D: Final intensive polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n" + "=" * 60)
        print("Final polishing phase (%.0fs remaining)" % remaining)
        print("=" * 60)
        
        # Multiple short polish rounds with small perturbations between
        n_rounds = max(2, int(remaining / 60))
        for r in range(n_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_round = min(remaining * 0.4, 60)
            
            if r > 0:
                # Small perturbation before each round
                noise_scale = 0.001 * S_best / np.sqrt(n_best) * (1.0 / r)
                a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n_best), S_best)
            else:
                a_pert = global_a.copy()
            
            a_pol, val_pol = high_beta_polish(a_pert, S_best, t_round, rng)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum()**2)
            print("  Polish round %d: obj=%.6f" % (r + 1, obj_pol))
            
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
        
        # One more Nesterov burst
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            a_fin, val_fin = nesterov_optimize(global_a, S_best, remaining, rng)
            obj_fin = 2.0 * n_best * val_fin / (a_fin.sum()**2)
            print("  Final Nesterov: obj=%.6f" % obj_fin)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
    
    # === Verification ===
    print("\n" + "=" * 60)
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)
        
        final_obj = objective_value(global_a)
        print("Final objective (internal): %.6f" % final_obj)
        
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification via evaluate_sequence: %.6f" % ver)
        
        if not np.isfinite(ver):
            print("WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - start_total
    print("Total time: %.0fs" % elapsed)
    print("Returning sequence of length %d" % len(global_a))
    print("=" * 60)
    
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
