"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta):
    """Gradient of LSE_beta(c) w.r.t. a. c = convolution of a with itself."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def power2_grad(a, c):
    """Gradient of sum(c^2) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    C = fft(c, fsize)
    A = fft(a, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]


def max_subgrad(a, c, tol_frac=0.0):
    """Subgradient of max(c) w.r.t. a, using lags within tolerance of max."""
    n = len(a)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgrad(a, c, k, power=1.0):
    """Subgradient using top-k lags with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective: 2*n*max_c/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize to sum = S, keeping non-negative."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fista_lse(a_init, max_time, rng, beta_start=5.0, beta_end=50000.0,
              use_backtrack=True, restart_interval=5000):
    """FISTA for LSE-smoothed objective with backtracking and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max_c = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start
    step = 1.0  # initial step size guess

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(y)
        cur_max = c.max()
        if cur_max < best_max_c - 1e-12:
            best_max_c = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: increase exponentially when stuck
        if stuck < 200:
            beta = beta_start * (1.0 + 0.01 * it)
        elif stuck < 2000:
            beta = min(beta_end, beta * 1.02)
        elif stuck < 10000:
            beta = min(beta_end, beta * 1.05)
        else:
            beta = min(beta_end, beta * 1.1)
        beta = max(beta, beta_start)
        beta = min(beta, beta_end)

        # Compute gradient
        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: use backtracking
        if use_backtrack:
            # Start from previous step (scaled down for larger beta)
            if it > 1:
                step = step * 0.95  # slightly decrease
            else:
                step = 1.0 / (beta + 1.0)
            step = max(step, 1e-10)
            step = min(step, 10.0 / (beta + 1.0))

            # Backtracking: check descent condition
            max_bt = 8
            for _ in range(max_bt):
                x_new = proj_simplex(y - step * grad, S)
                c_try = fft_conv_self(x_new)
                # Descent condition for smooth function:
                # f(x) <= f(y) + <grad, x-y> + (1/(2L))||x-y||^2
                # For LSE, use: LSE(x) <= LSE(y) + <grad(x-y)> + (0.5/step)*||x-y||^2
                diff = x_new - y
                lhs = c_try.max()  # approximate LSE by max for check
                rhs = cur_max + np.dot(grad, diff) + 0.5 * np.dot(diff, diff) / step
                if lhs <= rhs + 1e-10 * max(1.0, cur_max):
                    break
                step *= 0.5
            # If all backtracks failed, use a very small step
            else:
                step = max(step * 0.1, 1e-12)
                x_new = proj_simplex(y - step * grad, S)
        else:
            step = 0.5 / max(beta, 1.0)
            step = max(step, 1e-12)
            x_new = proj_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        a_prev = a.copy()
        a = x_new
        y = a + ((t_mom - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        t_mom = t_new

        # Restart conditions
        if stuck > 3000:
            t_mom = 1.0
            y = a.copy()
            # Structured noise: perturb more at positions that contribute to peaks
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 15000.0)
            if noise_scale > 1e-10:
                # Bias noise toward positions in the support of the gradient
                grad_magnitude = np.abs(grad)
                total_gm = grad_magnitude.sum()
                if total_gm > 1e-15:
                    bias = grad_magnitude / total_gm
                    noise = rng.normal(0, noise_scale, n) * (1.0 + 2.0 * bias * n / max(bias.sum(), 1e-15))
                else:
                    noise = rng.normal(0, noise_scale, n)
                y = proj_simplex(y + noise, S)
            last_improve = it
            stuck = 0

        if it % restart_interval == 0:
            t_mom = 1.0
            y = a.copy()

    c_final = fft_conv_self(best_a)
    if c_final.max() < best_max_c:
        best_max_c = c_final.max()
    return best_a, best_max_c

        t_mom = t_new

def subgradient_polish(a_init, max_time, rng, lower_bound=1.001):
    """Subgradient method with Polyak step for max(c) with simplex constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max_c = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.1
    # Moving average of best for Polyak step
    best-so-far tracking
    polyak_best = best_max_c

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(y)
        cur_max = c.max()

        if cur_max < best_max_c - 1e-12:
            best_max_c = cur_max
            best_a = y.copy()
            last_improve = it
            polyak_best = min(polyak_best, cur_max)

        stuck = it - last_improve

        # Determine which subgradient to use
        if stuck < 50:
            tol_frac = 0.0
            k_eff = 1
        elif stuck < 200:
            tol_frac = 0.0
            k_eff = 2
        elif stuck < 1000:
            tol_frac = 0.005
            k_eff = min(8, 2 + stuck // 100)
        elif stuck < 5000:
            tol_frac = 0.01
            k_eff = min(32, 8 + stuck // 200)
        else:
            tol_frac = 0.02
            k_eff = min(128, 32 + stuck // 500)

        if k_eff <= 1:
            grad = max_subgrad(y, c, tol_frac=0.0)
        else:
            grad = topk_subgrad(y, c, k_eff, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step size: (f(y) - f*) / ||g||^2
        # We use the gap: cur_max - lower_bound
        gap = max(cur_max - lower_bound, 1e-4)
        step_polyak = gap / (gnorm * gnorm)
        # Blend with previous step for stability
        step = 0.7 * min(step_polyak, 0.5) + 0.3 * step
        step = max(step, 1e-8)
        step = min(step, 1.0)

        x_new = proj_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        a_prev = a.copy()
        a = x_new
        y = a + ((t_mom - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        t_mom = t_new

        # Restart when stuck
        if stuck > 3000:
            t_mom = 1.0
            y = a.copy()
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-10:

            last_improve = it
            stuck = 0

        if it % 5000 == 0:
            t_mom = 1.0
            y = a.copy()

    c_final = fft_conv_self(best_a)
    if c_final.max() < best_max_c:
        best_max_c = c_final.max()
    return best_a, best_max_c


def coordinate_descent(a_init, max_time, rng):
    """Coordinate descent: try to improve one coordinate at a time."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_max_c = np.inf
    best_a = a.copy()
    c = fft_conv_self(a)
    best_max_c = c.max()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Random subset of coordinates
        k = min(64, n)
        idx = rng.choice(n, size=k, replace=False)

        improved = False
        for j in idx:
            if a[j] < 1e-15:
                # Try setting a[j] to a small positive value
                for target in [S * 1e-4 / n, S * 1e-3 / n, S * 1e-2 / n]:
                    a_try = a.copy()
                    a_try[j] = target
                    a_try = proj_simplex(a_try, S)
                    c_try = fft_conv_self(a_try)
                    if c_try.max() < best_max_c - 1e-10:
                        a = a_try
                        c = c_try
                        best_max_c = c.try.max() if hasattr(c, 'try') else c_try.max()
                        best_max_c = c_try.max()
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Try reducing a[j]
                for factor in [0.5, 0.8, 0.9]:
                    a_try = a.copy()
                    a_try[j] *= factor
                    a_try = proj_simplex(a_try, S)
                    c_try = fft_conv_self(a_try)
                    if c_try.max() < best_max_c - 1e-10:
                        a = a_try
                        c = c_try
                        best_max_c = c_try.max()
                        best_a = a.copy()
                        improved = True
                        break

        if not improved and it % 10 == 0:
            # Add small noise to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 50.0)
            if noise_scale > 1e-12:
                a = proj_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best_max_c


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)

def make_raised_cos_init(n):

def make_barker_like(n):
    """Approximate Barker-like construction with smooth transitions."""
    # Start with a random ±1 sequence, then smooth
    rng_local = np.random.default_rng(12345)
    signs = rng_local.choice([-1, 1], size=n)
    # Smooth via averaging with a wide kernel
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(signs, kernel, mode='same')
    # Make non-negative
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() < 1e-10:
        return make_raised_cos_init(n)
    return smooth


def make_wavelet_init(n):
    """Initialize based on a prolate-like function (concentrated in time)."""
    tt = np.arange(n, dtype=np.float64) / n
    # Sinc-like shape (concentrated in time, spread in frequency)
    x = (tt - 0.5) * n * 0.3
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.where(np.abs(x) < 1e-10, 1.0, np.sin(np.pi * x) / (np.pi * x))
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return make_raised_cos_init(n)
    return a


def upsample(a, factor=2):
    """Upsample using spline interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a, period=1)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    block = n // factor
    n_out = factor
    result = np.zeros(n_out)
    for i in range(n_out):
        start = i * block
        end = start + block
        result[i] = a[start:end].sum() / block
    return result

        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Candidate n values
    n_candidates = [4096, 8192, 16384]
    if a_start is not None:
        n0 = len(a_start)
        # Include the provided n and nearby powers of 2
        n_candidates = sorted(set(n_candidates + [n0]))
    else:
        n0 = 8192

    # Initializations for each n
    def get_inits_for_n(n):
        inits = []
        names = []
        inits.append(make_raised_cos_init(n))
        names.append("raised_cos")
        inits.append(make_cosine_init(n))
        names.append("cosine")
        inits.append(make_flat_top_init(n))
        names.append("flat_top")
        inits.append(make_wavelet_init(n))
        names.append("wavelet")
        r = rng.random(n)
        r = r ** 1.5
        inits.append(r)
        names.append("random_pow")
        r2 = rng.random(n)
        inits.append(r2)
        names.append("random")
        if a_start is not None and n == len(a_start):
            inits.append(a_start.copy())
            names.append("provided")
        return inits, names

    # === Quick initialization evaluation ===
    print("=== Quick initialization ===")
    best_overall = None
    best_overall_obj = np.inf
    best_overall_n = n0

    for n in n_candidates:
        inits, names = get_inits_for_n(n)
        S = np.sqrt(2.0 * n)
        best_init_for_n = None
        best_init_obj = np.inf
        for a_init, name in zip(inits, names):
            a_norm = normalize_seq(a_init, S)
            obj = compute_obj(a_norm)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init_for_n = a_norm
            if obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a_norm.copy()
        if best_init_for_n is not None:
            print("  n=%d: best init obj=%.6f" % (n, best_init_obj))
            if best_init_obj < best_overall_obj:
                best_overall_obj = best_init_obj
                best_overall = best_init_for_n
                best_overall_n = n

    print("Best initial: n=%d, obj=%.6f" % (best_overall_n, best_overall_obj))
    print("Time elapsed: %.1fs" % (time.monotonic() - t_start))

    # ===== Main optimization on the best n =====
    n_best = best_overall_n
    a_current = best_overall.copy()
    S = np.sqrt(2.0 * n_best)

    # Phase 1: Power-2 warmup (5-10% of time)
    remaining = deadline - time.monotonic()
    p1_time = min(remaining * 0.08, 60.0)
    if p1_time > 5:
        print("\n=== Phase 1: P2 warmup (n=%d, %.0fs) ===" % (n_best, p1_time))
        t_p1 = time.monotonic()
        a_p2 = fista_p2(a_current, p1_time, rng)
        obj_p2 = compute_obj(a_p2)
        print("  P2 done: obj=%.6f (t=%.1fs)" % (obj_p2, time.monotonic() - t_p1))
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()
        a_current = a_p2

    # Phase 2: LSE FISTA with backtracking (40% of time)
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.40
    if p2_time > 10:
        print("\n=== Phase 2: LSE FISTA (n=%d, %.0fs) ===" % (n_best, p2_time))
        t_p2 = time.monotonic()
        a_lse, _ = fista_lse(a_current, p2_time, rng,
                              beta_start=3.0, beta_end=100000.0,
                              use_backtrack=True)
        obj_lse = compute_obj(a_lse)
        print("  LSE done: obj=%.6f (t=%.1fs)" % (obj_lse, time.monotonic() - t_p2))
        if obj_lse < global_best_obj:
            global_best_obj = obj_lse
            global_best_a = a_lse.copy()
        a_current = a_lse

    # Phase 3: Subgradient polish (25% of time)
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 10:
        print("\n=== Phase 3: Subgradient polish (n=%d, %.0fs) ===" % (n_best, p3_time))
        t_p3 = time.monotonic()
        # Estimate lower bound for Polyak step
        # Theoretical: 2n/(2n-1)
        lb = 2.0 * n_best / (2.0 * n_best - 1.0) + 0.01
        a_sg, _ = subgradient_polish(a_current, p3_time, rng, lower_bound=lb)
        obj_sg = compute_obj(a_sg)
        print("  SG done: obj=%.6f (t=%.1fs)" % (obj_sg, time.monotonic() - t_p3))
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()
        a_current = a_sg

    # Phase 4: Another LSE pass with very high beta (15% of time)
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    if p4_time > 10:
        print("\n=== Phase 4: High-beta LSE (n=%d, %.0fs) ===" % (n_best, p4_time))
        t_p4 = time.monotonic()
        a_hd, _ = fista_lse(a_current, p4_time, rng,
                             beta_start=500.0, beta_end=200000.0,
                             use_backtrack=True)
        obj_hd = compute_obj(a_hd)
        print("  High-beta LSE done: obj=%.6f (t=%.1fs)" % (obj_hd, time.monotonic() - t_p4))
        if obj_hd < global_best_obj:
            global_best_obj = obj_hd
            global_best_a = a_hd.copy()
        a_current = a_hd

    # Phase 5: Coordinate descent (remaining time)
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.40
    if p5_time > 5:
        print("\n=== Phase 5: Coordinate descent (n=%d, %.0fs) ===" % (n_best, p5_time))
        t_p5 = time.monotonic()
        a_cd, _ = coordinate_descent(a_current, p5_time, rng)
        obj_cd = compute_obj(a_cd)
        print("  CD done: obj=%.6f (t=%.1fs)" % (obj_cd, time.monotonic() - t_p5))
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()
        a_current = a_cd

    # Phase 6: Try a different n (if time permits)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Try the next best n
        alt_n = None
        for n in n_candidates:
            if n != n_best:
                alt_n = n
                break
        if alt_n is not None:
            p6_time = remaining * 0.60
            print("\n=== Phase 6: Alternative n=%d (%.0fs) ===" % (alt_n, p6_time))
            t_p6 = time.monotonic()
            inits, names = get_inits_for_n(alt_n)
            S_alt = np.sqrt(2.0 * alt_n)
            best_alt_init = None
            best_alt_obj = np.inf
            for a_i in inits:
                a_norm = normalize_seq(a_i, S_alt)
                obj_i = compute_obj(a_norm)
                if obj_i < best_alt_obj:
                    best_alt_obj = obj_i
                    best_alt_init = a_norm

            if best_alt_obj < global_best_obj + 0.1:
                # Worth exploring
                a_alt, _ = fista_lse(best_alt_init, p6_time * 0.6, rng,
                                      beta_start=5.0, beta_end=100000.0)
                obj_alt = compute_obj(a_alt)
                print("  Alt n=%d: obj=%.6f (t=%.1fs)" % (alt_n, obj_alt, time.monotonic() - t_p6))
                if obj_alt < global_best_obj:
                    global_best_obj = obj_alt
                    global_best_a = a_alt.copy()

                rem6 = p6_time - (time.monotonic() - t_p6)
                if rem6 > 10 and obj_alt < global_best_obj + 0.05:
                    a_alt2, _ = subgradient_polish(a_alt, rem6, rng,
                                                    lower_bound=2.0 * alt_n / (2.0 * alt_n - 1.0) + 0.01)
                    obj_alt2 = compute_obj(a_alt2)
                    print("  Alt n=%d polished: obj=%.6f" % (alt_n, obj_alt2))
                    if obj_alt2 < global_best_obj:
                        global_best_obj = obj_alt2
                        global_best_a = a_alt2.copy()

    # Phase 7: Final intensive polish on the best found
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_best_a is not None:
        p7_time = remaining * 0.85
        print("\n=== Phase 7: Final polish (%.0fs) ===" % p7_time)
        t_p7 = time.monotonic()
        n_final = len(global_best_a)
        S_final = np.sqrt(2.0 * n_final)
        lb_final = 2.0 * n_final / (2.0 * n_final - 1.0) + 0.005

        # Alternate between LSE and subgradient
        a_fin = global_best_a.copy()
        half_time = p7_time / 2.0
        a_fin, _ = fista_lse(a_fin, half_time, rng, beta_start=1000.0, beta_end=500000.0)
        obj_fin1 = compute_obj(a_fin)
        if obj_fin1 < global_best_obj:
            global_best_obj = obj_fin1
            global_best_a = a_fin.copy()

        rem7 = p7_time - (time.monotonic() - t_p7)
        if rem7 > 5:
            a_fin2, _ = subgradient_polish(a_fin, rem7, rng, lower_bound=lb_final)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final polish: obj=%.6f (t=%.1fs)" % (obj_fin2, time.monotonic() - t_p7))
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    print("\n=== Summary ===")
    print("Best objective (internal): %.6f" % global_best_obj)

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected! Using fallback.")
            n_fb = 8192
            a_fb = make_raised_cos_init(n_fb)
            a_fb = normalize_seq(a_fb, np.sqrt(2.0 * n_fb))
            final_list = [float(x) for x in a_fb]
            ver = evaluate_sequence(final_list)
            print("Fallback objective: %.6f" % ver)
    else:
        n_fb = 8192
        a_fb = make_raised_cos_init(n_fb)
        a_fb = normalize_seq(a_fb, np.sqrt(2.0 * n_fb))
        final_list = [float(x) for x in a_fb]
        ver = evaluate_sequence(final_list)
        print("Fallback objective: %.6f" % ver)

    print("Total time: %.1fs" % (time.monotonic() - t_start))
    return final_list
