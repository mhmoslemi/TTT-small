**Key algorithm: "Peak Shaving" via iterative redistribution**

The idea:
1. Start with some initial a
2. Compute c = conv(a, a)
3. Identify the "hot spots" (positions where c is above average)
4. For each hot spot k, we want to reduce c[k] = sum_j a_j * a_{k-j}
5. To reduce c[k], we can reduce a_j where a_{k-j} is also large (i.e., where the product a_j * a_{k-j} is large)
6. Move the mass to coordinates that don't contribute much to any hot spot

This is similar to the "heat equation" approach: treat c as a temperature field, and move mass from hot to cold regions.

More concretely:
- Compute the "contribution" of coordinate j to each hot spot k:
  contrib[j, k] = a_j * a_{k-j} (if k-j is in range)
- The total "pressure" on coordinate j is:
  pressure[j] = sum_k w_k * a_{k-j} (where w_k is the weight of peak k, e.g., softmax of c)
- Update: a_j_new = a_j * (1 - eta * (pressure[j] - avg_pressure))
- This is a multiplicative weight update that reduces coordinates with high pressure and increases those with low pressure.

This is essentially the **Multiplicative Weights Update (MWU)** or **exponential weights** algorithm, applied to the "convolution peak" objective.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smooth_and_grad(a, beta):
    """Compute LSE value and gradient of beta*LSE(conv) w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

def peak_shave(a_init, max_time, rng):
    """
    Peak-shaving via multiplicative weight updates targeting top peaks.
    Directly reduces the maximum convolution value by redistributing mass
    away from coordinates that contribute heavily to the top peaks.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    eta = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 5000:
            break

        # Select top peaks with adaptive count
        if stuck < 200:
            K = 1
        elif stuck < 1000:
            K = 3
        elif stuck < 3000:
            K = min(10, 3 + stuck // 300)
        else:
            K = min(50, 10 + stuck // 200)

        K = min(K, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:K]
        c_avg = c.mean()
        c_std = c.std() + 1e-15

        # Compute weights for top peaks
        peak_vals = c[top_idx]
        weights = (peak_vals - c_avg) / (K * c_std)
        weights = np.maximum(weights, 0)
        if weights.sum() > 0:
            weights = weights / weights.sum()

        # Build weight sequence (length 2n-1)
        w_seq = np.zeros(2 * n - 1)
        w_seq[top_idx] = weights

        # Pressure on coordinate j = sum_k w_k * a[k-j]
        # This is a cross-correlation: pressure = ifft(conj(FFT(w_seq)) * FFT(a))
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        W = fft(w_seq, fsize)
        A = fft(a, fsize)
        pressure = np.real(ifft(np.conj(W) * A))[:n]

        # Multiplicative update (exponential tilting)
        p_mean = pressure.mean()
        p_std = pressure.std() + 1e-15
        # Normalize pressure to have moderate range
        p_norm = (pressure - p_mean) / p_std
        a_new = a * np.exp(-eta * p_norm)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * (S / a_new.sum())

        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < cur_obj + 1e-12:
            a = a_new
            c = c_new
            # Decrease eta slowly
            eta *= 0.999
        else:
            # Try smaller eta
            eta *= 0.5
            if eta < 1e-8:
                # Reset and perturb
                a = best_a.copy()
                c = fft_convolve_self(a)
                eta = 0.1
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)

    return best_a, best_obj

def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        hi = min(n - 1, k_star)
        if hi >= lo:
            contrib[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]

        # Also consider second peak
        c_copy = c.copy()
        c_copy[k_star] = -np.inf
        k_star2 = int(np.argmax(c_copy))
        contrib2 = np.zeros(n)
        lo2 = max(0, k_star2 - n + 1)
        hi2 = min(n - 1, k_star2)
        if hi2 >= lo2:
            contrib2[lo2:hi2 + 1] = a[k_star2 - hi2:k_star2 - lo2 + 1][::-1]

        # Source: high contribution AND high value (can give mass)
        # Target: low contribution (receiving mass won't increase peak much)
        src_score = (contrib + 0.5 * contrib2) * np.maximum(a, 0.01)
        tgt_score = -(contrib + 0.5 * contrib2) * (S / n - a + 0.01)

        n_try = min(6, n)
        src_cands = np.argsort(src_score)[::-1][:n_try]
        tgt_cands = np.argsort(tgt_score)[:n_try]

        best_improvement = cur_obj + 1e-15
        found = False

        for j in src_cands[:n_try // 2 + 1]:
            for l in tgt_cands[:n_try // 2 + 1]:
                if j == l or a[j] < 1e-10:
                    continue
                delta = a[j] * 0.03
                if delta < 1e-12:
                    continue

                # Compute change in c
                delta_c = np.zeros(2 * n - 1)
                # Reduce a[j] by delta
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                delta_c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
                # Increase a[l] by delta
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                delta_c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)]

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_improvement:
                    best_improvement = new_obj
                    found = True
                    a[j] -= delta
                    a[l] += delta
                    c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)] + delta * delta
                    c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)] + delta * delta

        if not found and stuck > 500:
            # Random perturbation
            idx = rng.integers(0, n)
            delta = a[idx] * 0.02
            if delta > 1e-12:
                a[idx] -= delta
                a[rng.integers(0, n - 1 - idx) + (1 if idx == 0 else 0)] += delta
                c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA, peak-shaving, Nesterov subgradient, upsample, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from previous best
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01 or len(a_start) < 10:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        a_start = np.ones(n) * S / n

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA with diverse configs (25% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.25
    print("\n=== Phase 1: FISTA adaptive (budget %.0fs) ===" % p1_time)

    configs = [
        {"beta_init": 20, "beta_rate": 0.008, "beta_max": 20000, "step_scale": 1.0},
        {"beta_init": 50, "beta_rate": 0.005, "beta_max": 50000, "step_scale": 0.7},
        {"beta_init": 100, "beta_rate": 0.003, "beta_max": 100000, "step_scale": 0.5},
    ]
    per_time = p1_time / len(configs)
    for i, cfg in enumerate(configs):
        if time.monotonic() >= deadline:
            break
        rem = min(per_time, deadline - time.monotonic())
        if rem < 5:
            break
        a_opt, obj_opt = run_fista_adaptive(global_best_a, rem, rng, **cfg)
        print("  FISTA%d: obj=%.6f" % (i + 1, obj_opt))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 2: Peak shaving (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    print("\n=== Phase 2: Peak shaving (budget %.0fs) ===" % p2_time)
    a_ps, obj_ps = peak_shave(global_best_a, p2_time, rng)
    print("  Peak shave: obj=%.6f" % obj_ps)
    if obj_ps < global_best_obj:
        global_best_obj = obj_ps
        global_best_a = a_ps.copy()

    # Follow up with FISTA
    remaining2 = deadline - time.monotonic()
    if remaining2 > 10:
        a_f, obj_f = run_fista_adaptive(global_best_a, remaining2 * 0.6, rng,
                                         beta_init=50, beta_rate=0.006,
                                         beta_max=50000, step_scale=0.8)
        print("  FISTA after PS: obj=%.6f" % obj_f)
        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_best_a = a_f.copy()

    # === Phase 3: Nesterov subgradient (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    print("\n=== Phase 3: Nesterov subgradient (budget %.0fs) ===" % p3_time)
    a_ns, obj_ns = nesterov_subgrad(global_best_a, p3_time, rng)
    print("  Nesterov: obj=%.6f" % obj_ns)
    if obj_ns < global_best_obj:
        global_best_obj = obj_ns
        global_best_a = a_ns.copy()

    # === Phase 4: Coordinate descent (15% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    print("\n=== Phase 4: Coordinate descent (budget %.0fs) ===" % p4_time)
    a_cd, obj_cd = coordinate_descent_refined(global_best_a, p4_time, rng)
    print("  Coord descent: obj=%.6f" % obj_cd)
    if obj_cd < global_best_obj:
        global_best_obj = obj_cd
        global_best_a = a_cd.copy()

    # === Phase 5: Upsample to larger n (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.10
    if p5_time > 30 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_fista_adaptive(a_up, p5_time * 0.7, rng,
                                                   beta_init=5, beta_rate=0.005,
                                                   beta_max=10000, step_scale=1.0)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

    # === Phase 6: Final intensive polish (5% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.90
    if p6_time > 5:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_time)
        # High-beta FISTA for final precision
        a_final, obj_final = run_fista_adaptive(global_best_a, p6_time * 0.5, rng,
                                                 beta_init=500, beta_rate=0.002,
                                                 beta_max=200000, step_scale=0.3)
        print("  High-beta FISTA: obj=%.6f" % obj_final)
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_best_a = a_final.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 5:
            a_ps2, obj_ps2 = peak_shave(global_best_a, remaining6, rng)
            print("  Peak shave final: obj=%.6f" % obj_ps2)
            if obj_ps2 < global_best_obj:
                global_best_obj = obj_ps2
                global_best_a = a_ps2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
