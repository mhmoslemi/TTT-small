- The "energy" at frequency m is C_m = |A_m|^2
- The autocorrelation at lag k is c_k = (1/N) * sum_m C_m * exp(2πi*mk/N)
- To make c_k small, we want the C_m's to be "anti-aligned" with exp(2πi*mk/N)
- But C_m >= 0 always (since it's |A_m|^2)
- So the best we can do is make C_m concentrated at frequencies where exp(2πi*mk/N) is negative for the bad lags k

For a single bad lag k, we want C_m concentrated where cos(2πmk/N) < 0, i.e., m in (N/4, 3N/4) (roughly).

For multiple bad lags, we want to find frequencies that are "negative" for all of them simultaneously.

def spectral_mask_update(a, M, rng):
    n = len(a)
    S = a.sum()
    N = 1 << (2*n - 1).bit_length()
    A = fft(a, N)
    C = np.abs(A)**2  # Power spectrum
    c = np.real(ifft(C))[:2*n-1]  # Autocorrelation
    
    # Find bad lags
    bad = c > M
    if not np.any(bad):
        return a
    
    # For each frequency m, compute the "alignment" with bad lags
    # alignment_m = sum_{k: bad} exp(2πi*mk/N)
    # We want to reduce C_m where alignment is positive (would increase c_k)
    # and increase C_m where alignment is negative (would decrease c_k)
    
    m = np.arange(N)
    # Create a "target spectrum" that's anti-correlated with bad lags
    bad_indices = np.where(bad)[0]
    # Compute the "bad frequency response"
    bad_resp = np.zeros(N)
    for k in bad_indices[:50]:  # Limit to 50 bad lags for efficiency
        bad_resp += np.cos(2*np.pi*m*k/N)
    
    # Reduce power where bad_resp > 0, increase where bad_resp < 0
    # But we must maintain sum(a) = S, so A[0] is fixed
    A[0] = S  # DC component fixed
    
    mask = 1.0 - 0.1 * np.tanh(0.5 * bad_resp / max(np.abs(bad_resp).max(), 1e-10))
    mask[0] = 1.0
    A_new = A * np.sqrt(np.maximum(mask, 0.1))
    A_new[0] = A[0]
    
    a_new = np.real(ifft(A_new, N))[:n]
    return project_simplex(a_new, S)

1. Multi-length search (8192, 12288, 16384) with interpolation of given sequence
2. Modified objective: max(c) + gamma*std(c) to encourage flatter autocorrelation
3. Fourier-domain spectral masking episodes that suppress peak-lag contributions
4. Pairwise mass-exchange refinement for direct peak reduction
5. Longer convergence with adaptive step sizes
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def level_gradient(a, c, beta, gamma, T):
    """Gradient of LSE_max(c; beta) + gamma * sum((c_k - T)^2 for c_k > T)."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w_max = np.zeros(len(c))
        w_max[np.argmax(c)] = 1.0
    else:
        w_max = e / tot
    if gamma > 0:
        excess = np.maximum(c - T, 0.0)
        w_excess = 2.0 * gamma * excess
    else:
        w_excess = np.zeros(len(c))
    w_total = w_max + w_excess
    wsum = w_total.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w_total = w_total / wsum
    fsize = 1 << (len(w_total) + n - 1).bit_length()
    W = fft(w_total, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient(a, c, k=8):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    cmax = c[idx[0]]
    weights = np.maximum(c[idx] - c.mean(), 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_mask_update(a, M, rng, strength=0.3):
    """Fourier-domain spectral shaping to suppress peak lags."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    C = np.abs(A) ** 2
    c = np.real(ifft(C))[:size]
    bad = c > M
    if not np.any(bad):
        return a
    bad_idx = np.where(bad)[0]
    if len(bad_idx) > 100:
        bad_idx = bad_idx[np.argsort(c[bad_idx])[::-1]][:100]
    m_arr = np.arange(N)
    align = np.zeros(N)
    for k in bad_idx:
        align += np.cos(2.0 * np.pi * m_arr * k / N)
    max_align = np.abs(align).max()
    if max_align < 1e-15:
        return a
    align = align / max_align
    mask = 1.0 - strength * np.tanh(0.5 * align)
    mask = np.maximum(mask, 0.05)
    mask[0] = 1.0
    A_new = A * np.sqrt(mask)
    A_new[0] = A[0]
    a_new = np.real(ifft(A_new, N))[:n]
    result = project_simplex(a_new, S)
    return result


def pairwise_refine(a, S, max_iters=5000, rng=None):
    """Direct pairwise mass exchange to reduce peak convolution."""
    n = len(a)
    a = normalize_to_sum(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    for it in range(max_iters):
        c = fft_convolve_self(a)
        cmax = c.max()
        if cmax < best_obj:
            best_obj = cmax
            best_a = a.copy()
        peak_lag = np.argmax(c)
        if peak_lag < n:
            pairs = [(i, peak_lag - i) for i in range(peak_lag + 1) if i < peak_lag - i]
        else:
            offset = peak_lag - (n - 1)
            pairs = [(i, i + offset) for i in range(n - offset) if i < i + offset]
        if not pairs:
            break
        best_delta = 0.0
        best_pair = None
        best_gain = 1e-15
        sample_size = min(len(pairs), 200)
        if len(pairs) > sample_size and rng is not None:
            chosen = rng.choice(len(pairs), sample_size, replace=False)
            sample_pairs = [pairs[i] for i in chosen]
        else:
            sample_pairs = pairs
        for (i, j) in sample_pairs:
            if i >= n or j >= n:
                continue
            ai, aj = a[i], a[j]
            if ai < 1e-15 and aj < 1e-15:
                continue
            delta_candidates = [
                0.1 * (aj - ai) / 2.0,
                0.3 * (aj - ai) / 2.0,
                0.5 * (aj - ai) / 2.0,
                -0.1 * (aj - ai) / 2.0,
            ]
            for d in delta_candidates:
                if a[i] + d < 0 or a[j] - d < 0:
                    continue
                if abs(d) < 1e-15:
                    continue
                a[i] += d
                a[j] -= d
                c_new = fft_convolve_self(a)
                gain = cmax - c_new.max()
                if gain > best_gain:
                    best_gain = gain
                    best_delta = d
                    best_pair = (i, j)
                a[i] -= d
                a[j] += d
        if best_pair is None:
            break
        a[best_pair[0]] += best_delta
        a[best_pair[1]] -= best_delta
        a = np.maximum(a, 0.0)
        a = a / a.sum() * S
    return best_a, best_obj


def fourier_phase_mutate(a, rng, scale=0.1):
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(N, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * N)
    phase_noise = rng.normal(0, scale * weight, N)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, N))[:n]
    return project_simplex(a_new, S)


def run_fista_level(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.004,
                    beta_max=10000.0, gamma=0.1, use_spectral_every=5000):
    """FISTA with leveling objective and periodic spectral shaping."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        T = c.mean() + 0.3 * (cmax - c.mean() + 1e-10)
        grad = level_gradient(y, c, beta, gamma, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.5 * beta + gamma * 10.0)

        if it - last_improve > 3000:
            if use_spectral_every > 0 and it % use_spectral_every < 5:
                y = spectral_mask_update(y, best * 0.99, rng, strength=0.2)
            else:
                noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
                if noise_scale > 1e-12:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        if use_spectral_every > 0 and it % use_spectral_every == 0:
            y_spec = spectral_mask_update(y, cmax * 0.95, rng, strength=0.15)
            c_spec = fft_convolve_self(y_spec)
            if c_spec.max() < cmax:
                y = y_spec
                t_mom = 1.0
                continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom  # keep for Pepe's formula

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_basin_hopping_v2(a_init, S, max_time, rng):
    """Basin hopping with mixed mutation strategies."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    n_iters = max(1, int(max_time / (opt_time + 3)))
    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = it / max(1, n_iters)
        base_scale = 0.3 * np.exp(-1.5 * progress) + 0.03
        scale = base_scale * (0.8 + 0.4 * rng.random())

        r = rng.random()
        if len(population) >= 2 and r < 0.2:
            idx1 = rng.integers(len(population))
            idx2 = rng.integers(len(population))
            if idx1 != idx2:
                a1, a2 = population[idx1][1], population[idx2][1]
                N = 1 << (2 * n - 1).bit_length()
                A1 = fft(a1, N)
                A2 = fft(a2, N)
                rr = rng.random(N)
                alpha = 0.3 + 0.4 * rr
                new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
                new_phase = np.where(rr < 0.5, np.angle(A1), np.angle(A2))
                new_A = new_mag * np.exp(1j * new_phase)
                new_A[0] = A1[0]
                a = project_simplex(np.real(ifft(new_A, N))[:n], S)
            else:
                a = fourier_phase_mutate(a, rng, scale)
        elif r < 0.5:
            a = fourier_phase_mutate(a, rng, scale)
        else:
            a = spectral_mask_update(a, best * 1.05, rng, strength=0.2 + 0.2 * progress)
            a = project_simplex(a, S)

        beta_start = 15 + 30 * min(it, 10)
        a_opt, val = run_fista_level(a, S, opt_time, rng,
                                      beta_init=beta_start, beta_rate=0.003,
                                      beta_max=5000, gamma=0.05,
                                      use_spectral_every=4000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH v2 iter %d: new best max_b=%.8f" % (it, val))

        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]

    return best_a, best


def run_exchange_polish(a_init, S, max_time, rng):
    """Polish using pairwise exchange + gradient steps alternately."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        if it % 2 == 1:
            a, val = pairwise_refine(a, S, max_iters=200, rng=rng)
            if val < best:
                best = val
                best_a = a.copy()
                last_improve = it
        else:
            c = fft_convolve_self(a)
            beta = 50.0 + 10.0 * it
            grad = peak_shave_gradient(a, c, k=min(4 + it // 500, 16))
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = 0.5 * S / (n * max(gnorm / (S / n), 1e-10) + 1.0)
                a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
                last_improve = it

        if it - last_improve > 500:
            noise = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            last_improve = it

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def interpolate_sequence(a_orig, new_n):
    """Interpolate a sequence to a new length."""
    n_orig = len(a_orig)
    if n_orig == new_n:
        return a_orig.copy()
    x_orig = np.linspace(0, 1, n_orig, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(x_new, x_orig, a_orig)
    return np.maximum(a_new, 0.0)


def generate_inits(n, rng, given_seq=None):
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None and len(given_seq) >= 32:
        inits.append(interpolate_sequence(given_seq, n))

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.15) + 0.01)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.5)
                            + 0.2 * np.cos(12 * np.pi * tt + 1.0), 0.01))
    inits.append(rng.random(n) + 0.05)
    # Asymmetric init (important: optimal is asymmetric)
    inits.append(np.maximum(np.exp(-2.0 * tt / 0.3) + 0.3 * np.exp(-2.0 * (tt - 0.4) ** 2 / 0.1), 0.01))
    inits.append(np.maximum(0.2 + 0.8 * np.sin(np.pi * tt * 0.7) ** 2, 0.01))
    # Block structure
    for nb in [8, 16]:
        blk = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, blk)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: length exploration -> leveling FISTA -> basin hopping
    -> exchange polish -> final micro-polish.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" %
                  (len(given), given.sum(), objective_value(given)))
    except Exception:
        pass

    global_best = np.inf
    global_a = None
    global_n = 0

    def try_update(a, n):
        nonlocal global_best, global_a, global_n
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return False
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (s * s)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = n
            print("  NEW GLOBAL BEST: obj=%.6f (n=%d)" % (obj, n))
            return True
        return False

    # Phase 1: Multi-length quick exploration
    print("\n=== Phase 1: Multi-length exploration (200s) ===")
    p1_budget = min(200.0, (deadline - total_start) * 0.10)
    lengths_to_try = [8192, 12288, 16384]

    if given is not None:
        lengths_to_try = [len(given)] + [l for l in lengths_to_try if l != len(given)]

    phase1_results = []
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - (deadline - total_start) * 0.88:
            break
        S_try = np.sqrt(2.0 * n_try)
        t_alloc = min(p1_budget / len(lengths_to_try) + 20,
                      deadline - time.monotonic() - (deadline - total_start) * 0.85)
        if t_alloc < 30:
            break
        print("  Trying n=%d, alloc=%.0fs" % (n_try, t_alloc))
        inits = generate_inits(n_try, rng, given_seq=given)
        best_val = np.inf
        best_a_local = None
        per_init = max(5.0, t_alloc / (len(inits) + 1))
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - (deadline - total_start) * 0.87:
                break
            t_sub = min(per_init, deadline - time.monotonic() - (deadline - total_start) * 0.86)
            if t_sub < 5:
                break
            a_opt, val = run_fista_level(init, S_try, t_sub, rng,
                                          beta_init=5.0, beta_rate=0.01,
                                          beta_max=5000, gamma=0.05)
            if val < best_val:
                best_val = val
                best_a_local = a_opt.copy()
        if best_a_local is not None:
            obj = 2.0 * n_try * best_val / (best_a_local.sum() ** 2)
            phase1_results.append((obj, n_try, best_a_local))
            try_update(best_a_local, n_try)
        print("  n=%d done: best obj=%.6f" % (n_try, global_best))

    phase1_results.sort(key=lambda x: x[0])

    if global_a is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        a0 = np.ones(n) * S / n
        global_a = a0
        global_n = n
        global_best = objective_value(a0)

    best_n = global_n
    best_a = global_a.copy()
    print("  Phase 1 best: obj=%.6f, n=%d" % (global_best, best_n))

    # Phase 2: Main leveling FISTA on best candidate
    print("\n=== Phase 2: Main leveling optimization (700s) ===")
    p2_budget = min(700.0, (deadline - total_start) * 0.30)
    S_best = best_a.sum()
    n_best = len(best_a)

    # Run multiple sub-phases with different gamma values
    gammas = [0.0, 0.1, 0.3]
    per_gamma = p2_budget / len(gammas)
    for g in gammas:
        if time.monotonic() > deadline - (deadline - total_start) * 0.65:
            break
        t_alloc = min(per_gamma, deadline - time.monotonic() - (deadline - total_start) * 0.64)
        if t_alloc < 30:
            break
        print("  Leveling with gamma=%.1f, t=%.0fs" % (g, t_alloc))
        a_opt, val = run_fista_level(best_a, S_best, t_alloc, rng,
                                      beta_init=10.0, beta_rate=0.003,
                                      beta_max=20000.0, gamma=g,
                                      use_spectral_every=5000)
        try_update(a_opt, n_best)
        if a_opt.sum() > 0:
            obj_local = 2.0 * n_best * val / (a_opt.sum() ** 2)
            if obj_local < 2.0 * n_best * val / (S_best ** 2) + 0.01:
                S_best = a_opt.sum()
                best_a = a_opt.copy()

    # Phase 3: Basin hopping
    print("\n=== Phase 3: Basin hopping (700s) ===")
    p3_budget = min(700.0, (deadline - total_start) * 0.30)
    if time.monotonic() < deadline - 200:
        t_bh = min(p3_budget, deadline - time.monotonic() - 200)
        if t_bh > 60:
            print("  BH alloc=%.0fs" % t_bh)
            a_bh, val_bh = run_basin_hopping_v2(best_a, best_a.sum(), t_bh, rng)
            try_update(a_bh, n_best)

    # Phase 4: Exchange polish
    print("\n=== Phase 4: Exchange polish (400s) ===")
    p4_budget = min(400.0, (deadline - total_start) * 0.20)
    if global_a is not None and time.monotonic() < deadline - 100:
        t_ex = min(p4_budget, deadline - time.monotonic() - 100)
        if t_ex > 30:
            print("  Exchange alloc=%.0fs" % t_ex)
            a_ex, val_ex = run_exchange_polish(global_a, global_a.sum(), t_ex, rng)
            try_update(a_ex, global_n)

    # Phase 5: Final micro-polish with very high beta
    print("\n=== Phase 5: Final micro-polish ===")
    p5_budget = max(20.0, deadline - time.monotonic() - 10)
    if global_a is not None and p5_budget > 20:
        print("  Micro-polish alloc=%.0fs" % p5_budget)
        a_final, val_final = run_fista_level(global_a, global_a.sum(), p5_budget, rng,
                                              beta_init=500.0, beta_rate=0.001,
                                              beta_max=50000.0, gamma=0.0,
                                              use_spectral_every=0)
        try_update(a_final, global_n)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f (n=%d)" % (ver, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
            final_list = [float(x) for x in global_a]
    else:
        n = 8192
        global_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
