"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses KL-projection gradient descent with LSE smoothing, adaptive step sizes,
momentum, and active-set refinement.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, c, beta):
    n = len(a)
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    max_scaled = scaled.max()
    if max_scaled > 500:
        # Rescale
        beta_eff = 500.0 / max(1e-30, (c - c.max()).max())
        scaled = beta_eff * c_shifted
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    
    fsize = 1 << (3 * n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def compute_topk_grad(a, c, k):
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    # Weight by excess over median
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 1e-10)
    wsum = weights.sum()
    weights = weights / wsum
    
    grad = np.zeros(n)
    fsize = 1 << (3 * n - 1).bit_length()
    # Compute weighted cross-correlation
    # grad[i] = 2 * sum_j w_j * a[j+i]
    # = 2 * cross_corr(w_weighted, a)[i]
    W = np.zeros(2 * n - 1)
    W[top_idx] = weights
    A = fft(a, fsize)
    Wf = fft(W, fsize)
    corr = np.real(ifft(Wf * np.conj(A)))[:n]
    return 2.0 * corr


def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_lse(a0, S, max_time, rng, beta_start=10.0, beta_end=500.0,
                 verbose=True):
    """LSE gradient descent with KL projection and adaptive beta."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    prev_grad = None
    prev_a = None
    
    # Momentum
    momentum = np.zeros(n)
    mu = 0.8
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive beta curriculum
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(5.0, min(beta, 5000.0))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-10:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        grad = compute_lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use a simple adaptive rule
        # gap = cur_max - c.min() gives a scale for the step
        gap = cur_max - c.mean()
        if gap < 1e-12:
            break
        
        # BB-like step size
        if prev_grad is not None and prev_a is not None:
            s = a - prev_a
            y = grad - prev_grad
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            if ss > 1e-30 and sy > 1e-30:
                step = sy / ss
                step = min(step, 5.0 / gnorm)
            else:
                step = 2.0 * gap / (gnorm * gnorm)
        else:
            step = 2.0 * gap / (gnorm * gnorm)
        
        step = max(1e-10, min(step, 5.0 / gnorm))

def active_set_qp(a0, S, max_time, rng, n_active=20):
    """
    Local optimization using active set concept.
    Identifies the active lags and tries to equalize them.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
        
        # Get active set (top lags)
        top_idx = np.argsort(c)[::-1][:n_active]
        active_vals = c[top_idx]
        target = active_vals.mean()
        
        # Direction: try to reduce the spread among active lags
        # and reduce the overall level
        weights = np.zeros(2 * n - 1)
        # Weight more toward the higher active values
        w = (active_vals - active_vals.min())
        w_max = w.max()
        if w_max > 1e-10:
            w = w / w_max
        else:
            w = np.ones(len(w)) / len(w)
        weights[top_idx] = w
        weights /= weights.sum()
        
        # Compute gradient direction
        fsize = 1 << (3 * n - 1).bit_length()
        A = fft(a, fsize)
        W = fft(weights, fsize)
        corr = np.real(ifft(W * np.conj(A)))[:n]
        grad = 2.0 * corr
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Also add a component to reduce the overall level
        # (push all active values down)
        level_weight = np.zeros(2 * n - 1)
        level_weight[top_idx] = 1.0 / len(top_idx)
        fsize2 = 1 << (3 * n - 1).bit_length()
        LW = fft(level_weight, fsize2)
        corr2 = np.real(ifft(LW * np.conj(A)))[:n]
        grad_level = 2.0 * corr2
        
        # Combine: primarily reduce the peak, secondarily equalize
        grad_combined = 0.7 * grad + 0.3 * grad_level
        gc_norm = np.linalg.norm(grad_combined)
        if gc_norm < 1e-14:
            break
        
        gap = cur_max - c.mean()
        step = 2.0 * gap / (gc_norm * gc_norm)
        step = min(step, 2.0 / gc_norm)
        
        a_try = project_simplex(a - step * grad_combined, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max + 1e-10:
            a = a_try
        
        if it % 200 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_max / (S * S)
            spread = active_vals.max() - active_vals.min()
            print(f"    QP it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, spread={spread:.6f}")
    
    return best_a, best_max

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses FFT-accelerated LSE gradient descent with adaptive beta curriculum,
momentum, Barzily-Borwein step sizes, targeted perturbations based on

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def cross_corr_fft(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_k w[k] a[k-i] via FFT."""
    n = len(a)
    m = len(w)
    fsize = 1 << (n + m - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def compute_lse_grad(a, c, beta):
    """Gradient of LSE(c; beta) w.r.t. a. Returns 2 * cross_corr(softmax(beta*c), a)."""
    n = len(a)
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    max_s = scaled.max()
    if max_s > 400.0:
        beta = 400.0 / max(1e-30, (c - c.max()).max())
        scaled = beta * c_shifted
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        top = np.where(c == c.max())[0]
        w[top] = 1.0 / len(top)
    else:
        w = e / tot
    return 2.0 * cross_corr_fft(w, a)


def compute_weighted_grad(a, c, k):
    """Weighted gradient from top-k convolution lags (excess-weighted)."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    excess = np.maximum(c[top_idx] - median_c, 1e-10)
    w_vec = np.zeros(2 * n - 1)
    w_vec[top_idx] = excess / excess.sum()
    return 2.0 * cross_corr_fft(w_vec, a)


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_gradient_descent(a0, S, max_time, rng, beta_start=5.0, beta_end=500.0,
                         momentum_rate=0.85, verbose=False):
    """
    LSE-smoothed gradient descent on the simplex with momentum and adaptive step.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 150
    prev_a = a.copy()
    prev_grad = None
    vel = np.zeros(n)

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Curriculum: beta increases over time
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)
        beta = max(3.0, min(beta, 5000.0))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-11:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        grad = compute_lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        gap = cur_max - c.mean()
        if gap < 1e-12:
            break

        # Adaptive step size
        if prev_grad is not None:
            s = a - prev_a
            y = grad - prev_grad
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            if ss > 1e-30 and sy > 0:
                step = sy / ss
            else:
                step = 2.0 * gap / (gnorm * gnorm)
        else:
            step = 2.0 * gap / (gnorm * gnorm)

        step = max(1e-12, min(step, 3.0 / gnorm))

        # Momentum
        vel = momentum_rate * vel + (1.0 - momentum_rate) * grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 1e-15:
            vel = vel * (gnorm / vnorm)
            step_vel = step
        else:
            vel = grad
            step_vel = step

        # Line search (backtracking)
        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - step_vel * vel, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-10:
                a = a_try
                accepted = True
                break
            step_vel *= 0.5

        if not accepted:
            a = project_simplex(a - 0.01 * gap / max(gnorm, 1e-15) * grad, S)

        prev_a = a.copy()
        prev_grad = grad.copy()

        # Structured noise when stuck
        if stuck > patience:
            scale = min(0.03, 0.005 * (stuck / patience)) * S / np.sqrt(n)
            if scale > 1e-10:
                a = project_simplex(a + rng.normal(0, scale, n), S)
            vel = np.zeros(n)
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 200 == 0:
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    LSE it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={cur_obj:.6f}, stuck={stuck}")

    return best_a, best_max


def topk_subgradient_descent(a0, S, max_time, rng, verbose=False):
    """Top-k excess-weighted subgradient descent with line search."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 250

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Grow k over time: from focused (small k) to broad (large k)
        k = int(4 + 100 * progress ** 0.5)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-11:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        grad = compute_weighted_grad(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        gap = cur_max - c.mean()
        if gap < 1e-12:
            break

        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 4.0 / gnorm)

        accepted = False
        for _ in range(15):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-10:
                a = a_try
                accepted = True
                break
            step *= 0.4

        if not accepted:
            a = project_simplex(a - 0.005 * gap / max(gnorm, 1e-15) * grad, S)

        if stuck > patience:
            scale = 0.015 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)
            stuck = 0
            patience = int(patience * 1.15)

        if verbose and it % 200 == 0:
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    TopK it={it}, t={elapsed:.0f}s, k={k}, obj={cur_obj:.6f}, stuck={stuck}")

    return best_a, best_max


def active_set_refinement(a0, S, max_time, rng, n_active=30, verbose=False):
    """
    Refine by targeting the spread among top active lags.
    Tries to equalize the top convolution values while reducing their level.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Active set: top lags
        top_idx = np.argsort(c)[::-1][:n_active]
        active_vals = c[top_idx]
        spread = active_vals.max() - active_vals.min()

        if spread < 1e-10:
            # All active lags are equal; just push them all down
            w_vec = np.zeros(2 * n - 1)
            w_vec[top_idx] = 1.0 / n_active
        else:
            # Weight proportional to excess above the minimum active value
            w_raw = np.zeros(2 * n - 1)
            excess = np.maximum(active_vals - active_vals.min(), 0.0)
            w_raw[top_idx] = excess / max(excess.sum(), 1e-10)
            # Add small uniform component to also reduce the overall level
            w_vec = 0.6 * w_raw + 0.4 / (2 * n - 1)
            w_vec = w_vec / w_vec.sum()

        grad = 2.0 * cross_corr_fft(w_vec, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        gap = cur_max - c.mean()
        step = 1.5 * gap / (gnorm * gnorm)
        step = min(step, 2.0 / gnorm)

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max + 1e-10:
            a = a_try

        if stuck > 400:
            scale = 0.008 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)
            stuck = 0

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    ASR it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, spread={spread:.6f}")

    return best_a, best_max


def perturb_and_refine(a0, S, t_budget, rng, strategy="random"):
    """Apply a targeted perturbation then briefly refine."""
    n = len(a0)

    if strategy == "random":
        scale = 0.025 * S / np.sqrt(n)
        a_pert = project_simplex(a0 + rng.normal(0, scale, n), S)
    elif strategy == "peak_targeted":
        # Reduce elements contributing most to the peak lag
        c = fft_convolve_self(a0)
        peak_lag = int(np.argmax(c))
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] = a0[j]
            else:
                contrib[i] = 0.0
        # Transfer mass from high-contrib to low-contrib
        med = np.median(contrib)
        high_mask = contrib > med + 1e-15
        low_mask = contrib <= med
        if high_mask.any() and low_mask.any():
            total_high = a0[high_mask].sum()
            transfer = 0.03 * total_high
            a_pert = a0.copy()
            a_pert[high_mask] -= transfer * a0[high_mask] / max(total_high, 1e-15)
            a_pert[low_mask] += transfer * a0[low_mask] / max(a0[low_mask].sum(), 1e-15)
        else:
            a_pert = a0.copy()
        a_pert = project_simplex(a_pert, S)
    elif strategy == "multiplicative":
        log_pert = rng.normal(0, 0.04, n)
        a_pert = a0 * np.exp(log_pert)
        a_pert = project_simplex(a_pert, S)
    elif strategy == "low_freq":
        # Perturb with a low-frequency pattern
        t = np.arange(n) / n
        pert = np.cos(2 * np.pi * rng.integers(1, 20) * t)
        scale = 0.02 * S / np.sqrt(n)
        a_pert = project_simplex(a0 + scale * pert, S)
    else:
        scale = 0.02 * S / np.sqrt(n)
        a_pert = project_simplex(a0 + rng.normal(0, scale, n), S)

    # Brief refinement
    a_opt, _ = topk_subgradient_descent(a_pert, S, t_budget, rng, verbose=False)
    return a_opt


def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    k1 = rng.integers(1, 30)
    k2 = rng.integers(30, 100)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * k1 * tt) + 0.2 * np.cos(2 * np.pi * k2 * tt) + 0.1
    return np.maximum(a, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. LSE gradient descent with curriculum on provided sequence.
    2. Top-k subgradient polishing.
    3. Active-set refinement for equalizing top lags.
    4. Targeted perturbations with brief refinement.
    5. Final high-precision LSE polish.
    6. Optional: try alternative lengths.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + (total_budget - 25.0)

    print(f"=== Starting optimization, budget={total_budget}s ===")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0.0, 1000.0)
        if a_provided.sum() > 0.01 and len(a_provided) >= 128:
            print(f"Loaded provided sequence: n={len(a_provided)}, sum={a_provided.sum():.4f}")
        else:
            a_provided = None
    except Exception as e:
        print(f"Warning: could not load provided sequence: {e}")

    global_best = np.inf
    global_a = None

    def update_global(a, tag=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  ** NEW BEST: obj={obj:.8f} [{tag}]")

    # ========== PHASE 1: LSE gradient descent ==========
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(600.0, (deadline - time.monotonic()) * 0.30)
        t1 = max(t1, 30.0)
        print(f"\n--- Phase 1: LSE descent, n={n1}, t={t1:.0f}s ---")
        a_opt, _ = lse_gradient_descent(a1, S1, t1, rng,
                                        beta_start=3.0, beta_end=300.0,
                                        momentum_rate=0.8, verbose=True)
        update_global(a_opt, "P1-LSE")

        # ========== PHASE 2: Top-k subgradient ==========
        remaining = deadline - time.monotonic()
        t2 = min(400.0, remaining * 0.20)
        t2 = max(t2, 20.0)
        if remaining > 30:
            print(f"\n--- Phase 2: TopK subgradient, n={n1}, t={t2:.0f}s ---")
            a_opt2, _ = topk_subgradient_descent(a_opt, S1, t2, rng, verbose=True)
            update_global(a_opt2, "P2-TopK")
            if obj_value(a_opt2) < obj_value(a_opt):
                a_opt = a_opt2

    # ========== PHASE 3: Active set refinement ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        t3 = min(200.0, remaining * 0.12)
        t3 = max(t3, 15.0)
        print(f"\n--- Phase 3: Active-set refinement, n={n3}, t={t3:.0f}s ---")
        a_opt3, _ = active_set_refinement(global_a, S3, t3, rng,
                                          n_active=40, verbose=True)
        update_global(a_opt3, "P3-ActiveSet")

    # ========== PHASE 4: Targeted perturbations ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n4 = len(global_a)
        S4 = np.sqrt(2.0 * n4)
        n_pert = 4
        t_per = max(20.0, (remaining - 30.0) / (n_pert + 1))
        print(f"\n--- Phase 4: Perturbations, {n_pert} attempts, ~{t_per:.0f}s each ---")

        strategies = ["peak_targeted", "multiplicative", "random", "low_freq"]
        for pi in range(n_pert):
            if time.monotonic() > deadline - 15:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_p = perturb_and_refine(global_a, S4, t_alloc * 0.7, rng,
                                     strategy=strategies[pi % len(strategies)])
            update_global(a_p, f"P4-pert-{strategies[pi % len(strategies)]}")

    # ========== PHASE 5: High-precision LSE polish ==========
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        t5 = min(300.0, remaining - 15.0)
        t5 = max(t5, 20.0)
        print(f"\n--- Phase 5: High-beta LSE polish, n={n5}, t={t5:.0f}s ---")
        a_opt5, _ = lse_gradient_descent(global_a, S5, t5, rng,
                                         beta_start=100.0, beta_end=3000.0,
                                         momentum_rate=0.5, verbose=True)
        update_global(a_opt5, "P5-HighBeta")

        # ========== PHASE 6: Final active set polish ==========
        remaining = deadline - time.monotonic()
        if global_a is not None and remaining > 30:
            n6 = len(global_a)
            S6 = np.sqrt(2.0 * n6)
            t6 = min(150.0, remaining - 10.0)
            t6 = max(t6, 10.0)
            print(f"\n--- Phase 6: Final active-set polish, t={t6:.0f}s ---")
            a_opt6, _ = active_set_refinement(global_a, S6, t6, rng,
                                              n_active=60, verbose=True)
            update_global(a_opt6, "P6-FinalAS")

    # ========== PHASE 7: Try alternative length (if time) ==========
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_alt = 16384
        if len(global_a) != n_alt:
            S_alt = np.sqrt(2.0 * n_alt)
            t_alt = min(200.0, remaining - 20.0)
            if t_alt > 50:
                print(f"\n--- Phase 7: Alternative length n={n_alt}, t={t_alt:.0f}s ---")
                init_alt = generate_init(n_alt, rng, base=global_a)
                init_alt = project_simplex(init_alt, S_alt)
                a_alt, _ = lse_gradient_descent(init_alt, S_alt, t_alt * 0.5, rng,
                                                beta_start=5.0, beta_end=200.0,
                                                verbose=False)
                update_global(a_alt, f"P7-n={n_alt}-LSE")
                rem = deadline - time.monotonic()
                if rem > 30:
                    a_alt2, _ = topk_subgradient_descent(a_alt, S_alt,
                                                         min(100, rem - 10), rng,
                                                         verbose=False)
                    update_global(a_alt2, f"P7-n={n_alt}-TopK")

    # ========== FINAL ==========
    if global_a is None:
        print("WARNING: No solution found, using fallback")
        global_a = np.ones(512) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    nf = len(global_a)
    cm = fft_convolve_self(global_a).max()
    sm = global_a.sum()
    print(f"\n{'='*60}")
    print(f"FINAL: n={nf}, internal_obj={global_best:.8f}")
    print(f"       max_conv={cm:.6f}, sum={sm:.6f}")
    print(f"       evaluate_sequence={ver:.8f}")
    print(f"{'='*60}")

    return final_list
