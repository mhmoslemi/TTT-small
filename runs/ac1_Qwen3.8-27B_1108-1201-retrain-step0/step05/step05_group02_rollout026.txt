```python
def pfpe_step(a, P_ind, eta, S):
    """One step of Proportional Fair Peak Equalization."""
    n = len(a)
    N = 1 << (2*n-1).bit_length()
    A = fft(a, N)
    P = fft(P_ind, N)
    
    # h1 = cross-correlation of a with P_ind
    h1 = np.real(ifft(conj(A) * P))[:n]  # (a ⋆ P)[j] for j=0,...,n-1
    # h2 = convolution of P_ind with a  
    h2 = np.real(ifft(A * P))[n-1:n-1+n]  # (P*a)[j] for j=0,...,n-1
    
    g = h1 + h2
    
    # Proportional fair update
    active = g > 1e-15
    if not np.any(active):
        return a
    
    mean_g = g[active].mean()
    mean_a = a[active].mean()
    
    # a_new[j] = a[j] * (mean_a / (g[j] + eps))^eta
    ratio = mean_a / (g + 1e-30)
    a_new = a * np.power(ratio, eta)
    a_new = np.maximum(a_new, 0)
    
    # Renormalize
    total = a_new.sum()
    if total < 1e-30:
        return a
    a_new = a_new * (S / total)
    
    return a_new
```

Actually, instead of a hard peak set, let me use a soft weight:

This is the LSE weight. Then:
g[j] = sum_k w[k] * (a[k-j]*1[0<=k-j<n] + a[j-k]*1[0<=j-k<n])

This is the LSE gradient (up to a factor of 2). So the PFPE with soft weights is actually just a multiplicative version of the LSE gradient descent!

The difference from standard gradient descent is:
- GD: a_new = project(a - step * g, S)
- PFPE: a_new = normalize(a * (mean_a / g)^eta)

The PFPE update is always non-negative (no projection needed) and preserves the "shape" of a more gently. It's like a "relative" gradient step.

This is actually related to the "mirror descent" with the negative entropy (KL) potential, which gives exactly this multiplicative update!

Mirror descent with KL:
a_new = argmin_{a >= 0, sum(a) = S} <g, a> + (1/eta) * KL(a || a_old)

The solution is: a_new[j] = a_old[j] * exp(-eta * g[j]) / Z (where Z is a normalization constant)

This is a "soft" multiplicative update. For small eta, it's approximately:
a_new[j] ≈ a_old[j] * (1 - eta * g[j]) (after normalization)

Which is similar to the proportional fair update but with exp instead of power.

So the PFPE is essentially mirror descent (or multiplicative weight descent) applied to the LSE-smoothed objective!

This is a well-studied method. The convergence rate of mirror descent with LSE is O(1/sqrt(T)) for the LSE objective, and the LSE objective converges to max as beta → ∞.

Key advantage over FISTA: Mirror descent doesn't require the objective to be smooth (it works with subgradients), and it naturally handles the non-negativity constraint (no projection needed). It might be more robust in the non-smooth regime.

"""
Minimize 2*n*max_conv/sum^2 using Mirror Descent with LSE + FISTA hybrid.
Novel contribution: Multiplicative (mirror descent) updates with adaptive
log-sum-exp smoothing, combined with FISTA and coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)
    total = v.sum()
    if total < 1e-300:
        return np.ones(n) * S / n
    # Use the water-filling algorithm
    if total <= S:
        # Need to add S - total to v
        # Simple: just scale
        return v * (S / total)
    # Need to reduce: use proper projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute LSE weights: w[k] proportional to exp(beta*(c[k]-cmax))."""
    cmax = c.max()
    shifted = beta * (c - cmax)
    shifted = shifted - shifted.max()  # for numerical stability
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    return w, cmax


def weighted_gradient(a, w):
    """
    Compute the gradient of sum_k w[k] * c[k] where c = conv(a,a).
    Result: g[j] = 2 * sum_k w[k] * (a[k-j]*1[0<=k-j<n] + a[j-k]*1[0<=j-k<n])
    Using FFT: g = 2 * (cross_correlation(a, w) + convolution(w, a))
    """
    n = len(a)
    m = len(w)
    N = 1 << (n + m - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    
    # h1[j] = sum_k w[k] * a[k-j] = (a ⋆ w)[j] for j=0,...,n-1
    h1 = np.real(ifft(np.conj(A) * W))[:n]
    
    # h2[j] = sum_k w[k] * a[j-k] = (w * a)[j] for j=0,...,n-1
    h2 = np.real(ifft(W * A))[:n]
    
    return 2.0 * (h1 + h2)


def run_mirror_descent(a_init, max_time, rng, beta_init=50.0, beta_max=100000.0,
                       beta_rate=1.002, eta_init=0.1, eta_decay=0.001,
                       restart_interval=5000, restart_noise=0.05):
    """
    Mirror descent (multiplicative weight) with LSE smoothing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (beta_rate ** it))
        eta = eta_init / (1.0 + eta_decay * it)
        
        c = fft_convolve_self(a)
        cmax = c.max()
        cur_obj = 2.0 * n * cmax / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        w, _ = lse_weights(c, beta)
        g = weighted_gradient(a, w)
        
        # Mirror descent update: a_new[j] = a[j] * exp(-eta * g[j])
        # For numerical stability, center the gradient
        g_centered = g - g.mean()
        # Clip to avoid overflow
        g_centered = np.clip(g_centered, -50.0 / max(eta, 1e-10), 50.0 / max(eta, 1e-10))
        
        log_update = -eta * g_centered
        a_new = a * np.exp(log_update)
        
        # Renormalize
        total = a_new.sum()
        if total < 1e-30:
            # Fallback: revert
            a_new = a.copy()
        else:
            a_new = a_new * (S / total)
        
        a = a_new

        if it - last_improve > restart_interval:

            a = project_simplex(a + noise, S)
            last_improve = it  # Reset counter
    
    return best_a, best_obj
