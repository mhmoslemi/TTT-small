def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    Nf = max(N, 1 << (2 * n - 1).bit_length())
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_value_only(a, N, beta):
    """Just compute LSE value (no gradient)."""
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        return c_max
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max
    return c_max + np.log(total) / beta


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def l2_flatten_pass(a_init, n, S, N, max_time, rng):
    """Minimize the L2 deviation of conv from its mean (flattening), then the peak."""
    a = np.clip(a_init, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()

    # mean_c = S^2 / (2n-1)
    mean_c = S * S / (2 * n - 1)
    lr = S / (n * 10.0)
    it = 0
    
    while time.monotonic() - start_time < max_time * 0.5:
        c = compute_conv(a, N)
        c_mean = c.mean()
        dev = c - c_mean
        # Gradient of sum(dev^2) = 2 * sum_k 2*dev[k] * dc[k]/da[i]
        # = 4 * ifft(fft(dev) * conj(A))[:n]  (approximately)
        fsize = 1 << (2 * n + 1).bit_length()
        D = fft(dev, fsize)
        Af = fft(a, fsize)
        grad = 4.0 * np.real(ifft(D * np.conj(Af)))[:n]
        
        a_new = project_simplex(a - lr * grad / (np.linalg.norm(grad) + 1e-15), S)
        lr *= 0.999
        a = a_new
        it += 1
        
        if it % 100 == 0:
            c = compute_conv(a, N)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()

    beta = 20.0
    s_hist = []
    y_hist = []
    g, lse_val, c, c_max = lse_value_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 3.0 * S / (n * beta)
    t2 = time.monotonic()
    
    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 80 == 0:

            y_hist = y_hist[-10:]

        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (fraction `frac` of them)."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    n_rand = max(1, int(N * frac))
    idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = np.angle(A).copy()
    new_phase[idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    return a_new * (S / s_new)


def make_init_sinusoidal(n, rng):
    """Sinusoidal initialization with random frequencies."""
    tt = np.arange(n) / n
    a = np.ones(n)
    for _ in range(rng.integers(2, 6)):
        freq = rng.uniform(1, min(20, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    """Zadoff-Chu inspired."""
    i = np.arange(n)
    c = rng.integers(1, n)
    phase = -np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.5 * np.cos(phase) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_random(n, rng):
    return rng.random(n) + 0.1


def make_init_block(n, rng, block_size=4):
    """Block-structured initialization."""
    n_blocks = n // block_size
    block_vals = rng.random(n_blocks) + 0.1
    a = np.repeat(block_vals, block_size)
    if len(a) < n:
        a = np.concatenate([a, rng.random(n - len(a)) + 0.1])
    return a[:n]


def make_init_coarse_fine(n, rng):
    """Coarse sinusoid + fine noise."""
    tt = np.arange(n) / n
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def topk_value_grad(a, N, K):
    """Top-K weighted gradient of max(conv(a,a))."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    K_actual = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:K_actual]
    top_vals = c[top_idx]
    top_sum = top_vals.sum()
    if top_sum < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = np.zeros(len(c))
        weights[top_idx] = top_vals / top_sum
    fsize = 1 << (2 * n + 1).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def hybrid_grad(a, N, beta, topK):
    """Choose between LSE and top-K gradient based on peak spread."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    sorted_c = np.sort(c)[::-1]
    K_check = min(50, len(c))
    spread = (sorted_c[0] - sorted_c[K_check - 1]) / max(c_max, 1e-30)
    if spread < 0.05:
        return lse_value_grad(a, N, beta)
    else:
        g, c2, cm2 = topk_value_grad(a, N, topK)
        return g, cm2, c2, cm2


def lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    yy_last = np.dot(y_hist[-1], y_hist[-1])
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_hist[-1], y_hist[-1]) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lse_lbfgs(a_init, n, S, N, max_time, rng, beta_init=20.0, beta_max=1000.0,
              m_lbfgs=12, topK_init=30):
    """L-BFGS on LSE/smoothed objective with adaptive gradient selection."""
    a = np.clip(a_init, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    beta = beta_init
    topK = topK_init
    best_a = a.copy()
    best_cm = np.inf
    s_hist = []
    y_hist = []

    c = compute_conv(a, N)
    best_cm = c.max()
    sorted_c = np.sort(c)[::-1]
    K_ck = min(50, len(c))
    if (sorted_c[0] - sorted_c[K_ck - 1]) / max(c.max(), 1e-30) < 0.05:
        g, lse_v, c, cm = lse_value_grad(a, N, beta)
    else:
        g, c, cm = topk_value_grad(a, N, topK)

    arr_scale = S / n
    step = 3.0 * arr_scale / max(beta, 5.0)
    t0 = time.monotonic()
    it = 0
    stall = 0
    stall_best = best_cm

    while time.monotonic() - t0 < max_time:
        it += 1

        if it % 60 == 0:
            beta = min(beta_max, beta * 1.15)
            topK = min(300, topK + 3)
            step = 3.0 * arr_scale / max(beta, 5.0)

        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_direction(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha = step
        a_new = a
        for ls in range(16):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= cm * (1.0 + 1e-12) or ls == 15:
                a_new = candidate
                if cm_ls < cm:
                    stall = 0
                break
            alpha *= 0.5

        cm_new = c_ls.max()
        if cm_new < best_cm:
            best_cm = cm_new
            best_a = a_new.copy()
            stall = 0
            stall_best = best_cm
        else:
            stall += 1

        # Recompute gradient at new point
        c_check = compute_conv(a_new, N)
        sorted_new = np.sort(c_check)[::-1]
        K_ck2 = min(50, len(c_check))
        spread_new = (sorted_new[0] - sorted_new[K_ck2 - 1]) / max(c_check.max(), 1e-30)
        if spread_new < 0.05:
            g_new, _, c_new, cm_new2 = lse_value_grad(a_new, N, beta)
        else:
            g_new, c_new, cm_new2 = topk_value_grad(a_new, N, topK)

        sk = a_new - a
        yk = g_new - g
        sy = np.dot(sk, yk)
        ss = np.dot(sk, sk)
        if sy > 1e-10 * ss + 1e-20 and ss > 1e-20:
            s_hist.append(sk)
            y_hist.append(yk)

        a = a_new
        g = g_new
        cm = cm_new2

        # Spectral restart on prolonged stall
        if stall > 400 and it % 200 == 0:
            A_sp = fft(a, N)
            mag = np.abs(A_sp)
            n_rand = max(1, int(N * 0.3))
            idx_rand = rng.choice(N, size=n_rand, replace=False)
            new_phase = np.angle(A_sp).copy()
            new_phase[idx_rand] = rng.uniform(0, 2 * np.pi, n_rand)
            A_new = mag * np.exp(1j * new_phase)
            a_try = np.real(ifft(A_new))[:n]
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try * (S / s_try)
                c_try = compute_conv(a_try, N)
                if c_try.max() < best_cm:
                    a = a_try
                    s_hist = []
                    y_hist = []
                    stall = 0
                    sorted_try = np.sort(c_try)[::-1]
                    if (sorted_try[0] - sorted_try[min(50, len(c_try)-1)]) / max(c_try.max(), 1e-30) < 0.05:
                        g, _, c, cm = lse_value_grad(a, N, beta)
                    else:
                        g, c, cm = topk_value_grad(a, N, topK)

    return best_a, best_cm


def l2_flatten_optimize(a_init, n, S, N, max_time, rng):
    """Phase 1: L2-flatten the convolution. Phase 2: LSE peak minimization."""
    a = np.clip(a_init, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cm = np.inf
    t0 = time.monotonic()
    half_time = max_time * 0.4

    # Phase 1: L2 flattening
    mean_target = S * S / (2 * n - 1)
    lr = 5.0 * S / (n * n)
    momentum = 0.9
    vel = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < half_time:
        c = compute_conv(a, N)
        c_mean = c.mean()
        dev = c - c_mean
        fsize = 1 << (2 * n + 1).bit_length()
        D = fft(dev, fsize)
        Af = fft(a, fsize)
        grad = 4.0 * np.real(ifft(D * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        vel = momentum * vel - lr * grad / gnorm
        a_new = project_simplex(a + vel, S)
        a = a_new
        lr *= 0.9995
        it += 1

        if it % 50 == 0:
            c_check = compute_conv(a, N)
            if c_check.max() < best_cm:
                best_cm = c_check.max()
                best_a = a.copy()

    # Phase 2: LSE L-BFGS from flattened solution
    if time.monotonic() - t0 < max_time - 5:
        rem = max_time - (time.monotonic() - t0)
        best_a2, best_cm2 = lse_lbfgs(a, n, S, N, rem, rng,
                                       beta_init=30.0, beta_max=800.0,
                                       m_lbfgs=12, topK_init=30)
        if best_cm2 < best_cm:
            best_a = best_a2
            best_cm = best_cm2

    return best_a, best_cm


def spectral_shaping(a_init, n, S, N, max_time, rng):
    """Flatten power spectrum via compression, then phase retrieval + local opt."""
    a = np.clip(a_init, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cm = compute_conv(a, N).max()
    t0 = time.monotonic()

    # Try different compression levels
    for alpha in [0.7, 0.85, 0.95]:
        if time.monotonic() - t0 > max_time - 30:
            break

        A_sp = fft(a, N)
        P = np.abs(A_sp) ** 2
        P_mean = P.mean()
        # Compress: P_new = P^alpha * P_mean^(1-alpha) to preserve total power
        P_new = np.power(P / P_mean, alpha) * P_mean
        target_mag = np.sqrt(P_new)

        # Gerchberg-Saxton phase retrieval
        x = a.copy()
        x_sp = np.zeros(N)
        x_sp[:n] = x
        for gs_it in range(100):
            X = fft(x_sp, N)
            mag_old = np.abs(X)
            phase_old = np.angle(X)
            mag_new = np.where(mag_old > 1e-10, target_mag * mag_old / (mag_old + 1e-10), 0.0)
            X_new = mag_new * np.exp(1j * phase_old)
            x_sp = np.real(ifft(X_new, N))
            x = np.maximum(x_sp[:n], 0.0)

        s_x = x.sum()
        if s_x < 1e-15:
            continue
        x = x * (S / s_x)
        c_x = compute_conv(x, N)
        if c_x.max() < best_cm:
            best_cm = c_x.max()
            best_a = x.copy()

        # Local polish
        if time.monotonic() - t0 < max_time - 20:
            a_pol, cm_pol = lse_lbfgs(x, n, S, N, min(60.0, max_time - (time.monotonic() - t0) - 10),
                                       rng, beta_init=40.0, beta_max=600.0, m_lbfgs=10, topK_init=20)
            if cm_pol < best_cm:
                best_cm = cm_pol
                best_a = a_pol

    return best_a, best_cm


def deflation_construct(n, S, N, n_iters=3000, rng=None):
    """Constructive deflation from all-ones sequence."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.03 * a[i_max], 0.5 * a[i_max])
        a[i_max] -= transfer
        a[i_min] += transfer

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_sinusoidal(n, rng):
    tt = np.arange(n) / n
    a = np.ones(n)
    for _ in range(int(rng.integers(2, 6))):
        freq = rng.uniform(1, min(15, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.35)
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    i = np.arange(n)
    c = int(rng.integers(1, n))
    phase = -np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.4 * np.cos(phase) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_sin(n, rng):
    tt = np.arange(n) / n
    a = np.ones(n)
    for freq in [2, 5, 11, 17]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_binaryish(n, rng):
    """Near-binary with random levels."""
    a = rng.random(n)
    threshold = 0.5
    a = np.where(a < threshold, 0.1 + 0.2 * rng.random(n), 0.7 + 0.3 * rng.random(n))
    return a


def spectral_partial_restart(a, N, rng, frac=0.4):
    n = len(a)
    S = a.sum()
    A_sp = fft(a, N)
    mag = np.abs(A_sp)
    n_rand = max(1, int(N * frac))
    idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = np.angle(A_sp).copy()
    new_phase[idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    return a_new * (S / s_new)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 using:
    1. Quick n-sweep to find best length
    2. Hybrid LSE/top-K L-BFGS from given sequence and diverse inits
    3. L2-flattening pre-pass
    4. Spectral shaping
    5. High-beta final polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    # Try to get the given sequence
    given_seq = None
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(g) >= 512:
            given_seq = g
    except Exception:
        pass

    global_best = np.inf
    global_a = None
    global_n = 8192

    def update_best(a_cand, n_val, label=""):
        nonlocal global_best, global_a, global_n
        a_c = np.clip(a_cand, 0.0, 1000.0)
        s = a_c.sum()
        if s < 0.01:
            return np.inf
        n_val = len(a_c)
        c = compute_conv(a_c, 1 << (2 * n_val).bit_length())
        cm = c.max()
        S = np.sqrt(2.0 * n_val)
        score = 2.0 * n_val * cm / (S * S)
        a_norm = a_c * (S / s)
        if score < global_best:
            global_best = score
            global_n = n_val
            global_a = a_norm.copy()
            print("  [%s] n=%d score=%.6f (t=%.0fs)" % (label, n_val, score,
                  time.monotonic() - (deadline - budget_s + 10)))
        return score

    print("Budget: %.0fs" % budget_s)

    # === Phase 0: Quick n-sweep ===
    print("=== Phase 0: Quick n-sweep ===")
    t0 = time.monotonic()
    sweep_budget = min(120.0, (deadline - t0) * 0.05)

    n_candidates = [4096, 8192, 12288, 16384]
    if given_seq is not None:
        n_candidates.insert(0, len(given_seq))

    for n_test in n_candidates:
        if time.monotonic() - t0 > sweep_budget:
            break
        if n_test < 512 or n_test > 32768:
            continue
        S_test = np.sqrt(2.0 * n_test)
        N_test = 1 << (2 * n_test).bit_length()

        if n_test == len(given_seq) if given_seq is not None else False:
            a_test = np.clip(given_seq, 0, 1000)
        else:
            a_test = make_init_sinusoidal(n_test, rng)

        update_best(a_test, n_test, "sweep_init_%d" % n_test)

    # Choose best n
    best_n = 8192
    if given_seq is not None and len(given_seq) in n_candidates:
        best_n = len(given_seq)

    # If sweep found something better with different n, use it
    if global_a is not None and global_n != best_n:
        # Compare: if the different n gave better score, use it
        pass  # We'll keep both tracks

    n = best_n
    S = np.sqrt(2.0 * n)
    N = 1 << (2 * n).bit_length()
    print("Using n=%d, S=%.2f, N=%d" % (n, S, N))

    # === Phase 1: Quick local opt from given sequence ===
    print("=== Phase 1: Local opt from given sequence ===")
    t1 = time.monotonic()
    p1_budget = min(300.0, (deadline - t1) * 0.12)

    if given_seq is not None and len(given_seq) == n:
        a_given = np.clip(given_seq, 0, 1000)
        s_g = a_given.sum()
        a_given = a_given * (S / s_g)
        a_res, cm_res = lse_lbfgs(a_given, n, S, N, p1_budget, rng,
                                   beta_init=30.0, beta_max=600.0,
                                   m_lbfgs=12, topK_init=30)
        update_best(a_res, n, "given_local")

    # === Phase 2: Multi-start with diverse inits + L2-flatten ===
    print("=== Phase 2: Multi-start + L2-flatten ===")
    t2 = time.monotonic()
    p2_budget = min(700.0, (deadline - t2) * 0.28)

    inits = []
    inits.append(("sinusoidal", make_init_sinusoidal(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("multi_sin", make_init_multi_sin(n, rng)))
    inits.append(("binaryish", make_init_binaryish(n, rng)))
    inits.append(("deflation", deflation_construct(n, S, N, 2000, rng)))

    if global_a is not None and len(global_a) == n:
        inits.append(("best_so_far", global_a))

    n_inits = len(inits)
    per_init = p2_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 500:
            break
        a_norm = np.clip(arr, 0, 1000)
        s_arr = a_norm.sum()
        if s_arr < 1e-15:
            continue
        a_norm = a_norm * (S / s_arr)

        t_init = time.monotonic()
        rem = max(10.0, per_init - (time.monotonic() - t_init) - 2)
        # Alternate between L2-flatten and direct LSE
        if name in ("deflation", "binaryish"):
            a_l, cm_l = l2_flatten_optimize(a_norm, n, S, N, min(rem, 120.0), rng)
        else:
            a_l, cm_l = lse_lbfgs(a_norm, n, S, N, min(rem, 120.0), rng,
                                    beta_init=15.0, beta_max=500.0,
                                    m_lbfgs=10, topK_init=20)
        update_best(a_l, n, name)

        # If this improved things, do a quick follow-up
        if cm_l < (compute_conv(a_norm, N).max()) * 0.95 and time.monotonic() < deadline - 500:
            a_l2, cm_l2 = lse_lbfgs(a_l, n, S, N, min(40.0, deadline - time.monotonic() - 500),
                                     rng, beta_init=40.0, beta_max=700.0,
                                     m_lbfgs=12, topK_init=30)
            update_best(a_l2, n, name + "_followup")

    # === Phase 3: Spectral shaping ===
    print("=== Phase 3: Spectral shaping ===")
    t3 = time.monotonic()
    p3_budget = min(300.0, (deadline - t3) * 0.12)
    if p3_budget > 30 and global_a is not None:
        a_sp, cm_sp = spectral_shaping(global_a, n, S, N, p3_budget, rng)
        update_best(a_sp, n, "spectral")

    # === Phase 4: Aggressive high-beta polish ===
    print("=== Phase 4: High-beta polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4 - 10
    if remaining > 20 and global_a is not None:
        a_pol, cm_pol = lse_lbfgs(global_a, n, S, N, min(remaining, 300.0),
                                    rng, beta_init=60.0, beta_max=1500.0,
                                    m_lbfgs=15, topK_init=50)
        update_best(a_pol, n, "polish_high_beta")

    # === Phase 5: Final attempts with restarts ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 40 and global_a is not None:
        print("=== Phase 5: Final restarts ===")
        for i in range(4):
            if time.monotonic() > deadline - 15:
                break
            a_try = spectral_partial_restart(global_a, N, rng, frac=0.3 + 0.1 * i)
            c_try = compute_conv(a_try, N)
            if c_try.max() < compute_conv(global_a, N).max():
                a_fin, cm_fin = lse_lbfgs(a_try, n, S, N, min(40.0, deadline - time.monotonic() - 15),
                                           rng, beta_init=50.0, beta_max=1000.0,
                                           m_lbfgs=12, topK_init=40)
                update_best(a_fin, n, "restart_%d" % i)
            else:
                # Try direct polish from perturbed
                a_fin, cm_fin = lse_lbfgs(a_try, n, S, N, min(30.0, deadline - time.monotonic() - 15),
                                           rng, beta_init=80.0, beta_max=1200.0,
                                           m_lbfgs=10, topK_init=30)
                update_best(a_fin, n, "restart_polish_%d" % i)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: n=%d, verified score=%.6f" % (len(final_list), ver))

    if not np.isfinite(ver):
        print("  Fallback to ones(256)")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    return final_list
