    """

    # --- Security Checks ---

    # Verify that the input is a list

        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        if isinstance(x, bool) or not isinstance(x, (int, float)):

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)

def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten_step(a, alpha):
    """One step of spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    ratio = power / max(mean_p, 1e-20)
    # Damp high-power, boost low-power frequencies
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_fista_refined(a_init, S, max_time, rng,

                      k_frac=0.08, use_momentum=True):

            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_new = fft_convolve_self(x_new)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))

        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

            t_val = 1.0
            stuck = 0

    if c_final.max() < best:

        best_a = a.copy()

def run_adam(a_init, S, max_time, rng,
             lr_init=0.02, beta_adam=50.0, beta_rate=0.01):
    """Adam optimizer with adaptive LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    k = max(4, int(0.06 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = min(2000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate: decay over time
        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.7 * progress)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        # Periodic restart
        if t % 5000 == 0 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

        # Spectral flattening step
        if t % 3000 == 0 and rng.random() < 0.1:
            a = spectral_flatten_step(a, 0.05)

    return best_a, best


def run_polish(a_init, S, max_time, rng):
    """High-beta polishing with very careful step control."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    k = max(4, int(0.03 * (2 * n - 1)))
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 200.0 * (1.0 + it / 500.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small steps with adaptive scaling
        base_step = 0.5 / (1.0 + 0.001 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(2.0, step_scale * 1.02)  # Slightly increase
        else:
            step_scale = max(0.01, step_scale * 0.5)  # Halve
            a_new = project_simplex(a - step * grad, S)
            a = a_new

        # Occasional perturbation to escape
        if it % 2000 == 0 and (time.monotonic() - start) / max_time < 0.7:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

def run_multiplicative(a_init, S, max_time, rng):
    """Multiplicative (ISRA-style) update, naturally maintains positivity."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    k = max(4, int(0.05 * (2 * n - 1)))
    it = 0
    eta = 1.0 / (2.0 * 50.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(5000.0, 20.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)

        # Multiplicative update: a_j *= exp(-eta * g_j)
        # Use first-order: a_j *= (1 - eta * g_j) but clip to stay positive
        update = 1.0 - eta * grad
        update = np.maximum(update, 0.01)
        a_new = a * update
        a = project_simplex(a_new, S)

        # Decay eta
        eta = 1.0 / (2.0 * (20.0 + it * 0.05))

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)
