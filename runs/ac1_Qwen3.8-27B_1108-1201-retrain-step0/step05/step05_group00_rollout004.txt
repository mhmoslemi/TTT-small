"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE-LBFGS with peak surgery,
spectral phase restarts, and Nesterov top-K polishing.
Key improvements: peak-specific mass redistribution, adaptive multi-beta
gradients, better line search, and simulated annealing on top peaks.

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak. Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        weights = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_partial_restart(a, N, rng, frac=0.4):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    rand_idx = rng.choice(Nf, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def _multi_beta_grad(a, N, betas):
    """Weighted combination of LSE gradients at multiple beta levels."""
    n = len(a)
    total_grad = np.zeros(n)
    c = None
    c_max = 0.0
    for i, beta in enumerate(betas):
        g, f_val, c_i, c_max_i = lse_grad(a, N, beta)
        w = (1.0 + i) / (len(betas) * (len(betas) + 1) / 2.0)
        total_grad += w * g
        if c_i is not None:
            c = c_i
            c_max = c_max_i
    total_grad /= len(betas)
    if c is None:
        c = compute_conv(a, N)
        c_max = c.max()
    return total_grad, c, c_max


def peak_surgery(a, N, rng, n_steps=500, S=None):
    """Reduce peak by mass redistribution from top contributing pair."""
    n = len(a)
    if S is None:
        S = a.sum()
    a = a.copy()
    c = compute_conv(a, N)
    best_cmax = c.max()
    it = 0

    while it < n_steps:
        it += 1
        k_star = int(np.argmax(c))
        cur_max = c[k_star]

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo >= hi:
            break

        # Find top contributing pair
        best_contrib = 0
        best_i = lo
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_i = i

        best_j = k_star - best_i
        if best_i == best_j:
            # Diagonal term
            src = best_i
            delta = 0.02 * a[src]
        else:
            # Off-diagonal: reduce the larger one
            if a[best_i] >= a[best_j]:
                src = best_i
            else:
                src = best_j
            delta = 0.015 * a[src]

        if delta < 1e-15:
            break

        # Try multiple random target positions
        best_new_cmax = cur_max
        best_dst = -1
        for trial in range(8):
            dst = rng.integers(0, n)
            if dst == src:
                continue
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * (S / s_t)
            c_t = compute_conv(a_trial, N)
            cm_t = c_t.max()
            if cm_t < best_new_cmax:
                best_new_cmax = cm_t
                best_dst = dst

        if best_dst >= 0 and best_new_cmax < best_cmax:
            a[src] -= delta
            a[best_dst] += delta
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * (S / s_a)
            c = compute_conv(a, N)
            best_cmax = c.max()
        else:
            # Try smaller delta
            delta *= 0.5
            if delta < 1e-18:
                break

    return a, best_cmax


def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=20, top_k_max=300):
    """Nesterov top-K subgradient for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    fsize = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        a_pad = np.zeros(Nf)
        a_pad[:n] = y
        A = fft(a_pad)
        c = np.real(ifft(A * A))[:2 * n - 1]
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 500 == 0 and rng.random() < 0.1:
            y = spectral_partial_restart(y, N, rng, frac=0.15)
            t = 1.0

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2.0 * np.pi * (5.0 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_tukey(n, rng):
    tt = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(np.pi * tt)) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_gaussian(n, rng):
    tt = np.linspace(0, 1, n)
    a = np.exp(-2.0 * (tt - 0.5) ** 2) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_two_peak(n, rng):
    tt = np.linspace(0, 1, n)
    a = np.exp(-8.0 * tt ** 2) + np.exp(-8.0 * (tt - 1.0) ** 2) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    P = rng.exponential(1.0, Nf)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, Nf)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_flat_perturbed(n, rng, base=None):

def anneal_topk(a, N, rng, n_iter=2000, S=None):
    """Simulated annealing targeting top peaks with mass moves."""
    n = len(a)
    if S is None:
        S = a.sum()
    a = a.copy()
    c = compute_conv(a, N)
    best_cmax = c.max()
    best_a = a.copy()
    temperature = 0.05

    for it in range(n_iter):
        temperature *= 0.999
        k_star = int(np.argmax(c))
        cur_max = c[k_star]
        # Pick a random source from the top contributing region
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        src = rng.integers(lo, hi + 1)
        # Pick a random destination
        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = min(0.01 * a[src], 0.005 * S / n)
        if delta < 1e-15:
            break
        a_trial = a.copy()
        a_trial[src] -= delta
        a_trial[dst] += delta
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t > 1e-15:
            a_trial = a_trial * (S / s_t)
        c_t = compute_conv(a_trial, N)
        new_max = c_t.max()
        improvement = cur_max - new_max
        if improvement > 0 or rng.random() < np.exp(improvement / max(temperature, 1e-10)):
            a = a_trial
            c = c_t
            if new_max < best_cmax:
                best_cmax = new_max
                best_a = a.copy()

    return best_a, best_cmax

    use_mb = True
    best_score_p2 = 2.0 * n * global_best / (S * S)

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 15:
            break
        beta_start = 10.0 + 5.0 * min(stall, 8)
        beta_end = min(600.0, 50.0 * (1.0 + 0.5 * stall))
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         m_lbfgs=14, use_multi_beta=use_mb)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2.0 * n * sc_new / (S * S)
            print("  [P2] score=%.6f (beta %.0f-%.0f, mb=%s)" % (score, beta_start, beta_end, use_mb))
            best_score_p2 = score
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.35)
                use_mb = not use_mb
                stall = 0
            elif stall >= 4:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.6)
                stall = 0

    print("Phase 2 done. Best score: %.6f" % (2.0 * n * global_best / (S * S)))

    # --- Phase 3: Peak surgery ---
    print("=== Phase 3: Peak surgery ===")
    t3 = time.monotonic()
    remaining3 = min(300.0, (deadline - t3) * 0.15)
    if remaining3 > 30:
        a_ps, sc_ps = peak_surgery(global_a, N, rng, n_steps=2000, S=S)
        _update_best(a_ps, "peak_surgery")
        if time.monotonic() < deadline - 100:
            a_ps2, sc_ps2 = peak_surgery(global_a, N, rng, n_steps=1500, S=S)
            _update_best(a_ps2, "peak_surgery2")

    # --- Phase 4: Simulated annealing ---
    print("=== Phase 4: Simulated annealing ===")
    t4 = time.monotonic()
    remaining4 = min(200.0, deadline - t4 - 100)
    if remaining4 > 30:
        a_an, sc_an = anneal_topk(global_a, N, rng, n_iter=3000, S=S)
        _update_best(a_an, "anneal")

    # --- Phase 5: Nesterov polishing ---
    print("=== Phase 5: Nesterov polishing ===")
    t5 = time.monotonic()
    remaining5 = min(300.0, deadline - t5 - 60)
    if remaining5 > 20:
        a_ng, sc_ng = nesterov_toppk_block(global_a, n, S, N, remaining5, rng,
                                            top_k_init=20, top_k_max=350)
        _update_best(a_ng, "nesterov")

    # --- Phase 6: Final high-beta L-BFGS ---
    print("=== Phase 6: Final refinement ===")
    t6 = time.monotonic()
    remaining6 = deadline - t6 - 10
    if remaining6 > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining6, 300.0),
                                         rng, beta_init=40.0, beta_max=800.0,
                                         m_lbfgs=16, use_multi_beta=False)
        _update_best(a_fin, "final")
        if deadline - time.monotonic() > 20:
            a_fin2, sc_fin2 = nesterov_toppk_block(global_a, n, S, N,
                                                    min(deadline - time.monotonic() - 5, 150.0),
                                                    rng, top_k_init=30, top_k_max=200)
            _update_best(a_fin2, "final_nest")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("  Rejected by evaluator, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
