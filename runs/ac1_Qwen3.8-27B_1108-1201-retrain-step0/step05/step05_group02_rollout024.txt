    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    # And grad_j = 2 * sum_k w_k * a_{k-j} = 2 * sum_i w_i * a_{i-j} (reindexing k->i)
    
    # Great, so grad = 2 * ifft(fft(w) * conj(fft(a)))[:n] is correct!
    
    return lse_val, grad, c


def spectral_projection(a, gamma=0.3):
    """
    Spectral projection: compute the 'ideal' sequence that would give
    a more flat convolution, and blend with current sequence.
    """
    n = len(a)
    s = a.sum()
    c = fft_convolve_self(a)
    need = 2 * n - 1
    size = 1 << (need - 1).bit_length()
    if size < need:
        size *= 2
    
    # Blend current convolution with flat target
    c_flat = np.full(need, s * s / need)
    c_target = (1 - gamma) * c + gamma * c_flat
    
    # Compute DFT of target convolution
    C_t = fft(c_target, size)
    
    # Take magnitude (should be real and non-negative for symmetric c, but let's be safe)
    mag = np.abs(C_t)
    
    # Square root to get |A|
    A_mag = np.sqrt(mag)
    
    # Inverse DFT to get candidate sequence
    a_candidate = np.real(ifft(A_mag, size))[:n]
    
    # Project to non-negative simplex
    a_candidate = project_simplex(a_candidate, s)
    
    return a_candidate


def run_fista_v2(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
                 beta_max=50000.0, use_backtrack=True, spectral_every=100,
                 spectral_gamma=0.2):
    """Improved FISTA with adaptive LSE and spectral projection."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c = lse_gradient_full(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
            bt = 0
            while obj_trial > cur_obj + 1e-10 and step_local > 1e-14 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 5.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)

        # Spectral projection every few iterations
        if it % spectral_every == 0:
            a_sp = spectral_projection(x_new, gamma=spectral_gamma)
            # Blend
            x_new = (1 - 0.3) * x_new + 0.3 * a_sp
            x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Restart if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def coordinate_descent_v2(a_init, max_time, rng, max_pairs=8):
    """
    Efficient coordinate descent with O(n) convolution updates.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 50000:
            break

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute contribution of each coordinate to the peaks
        contrib = np.zeros(n)
        for kk in peak_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                contrib[lo:hi+1] += a[kk - hi:kk - lo + 1][::-1]

        # Find good (source, target) pairs
        # Source: high contribution, has mass to give
        # Target: low contribution, has room to receive
        src_score = contrib * a
        tgt_score = -contrib * (S / n - a)
        
        n_cand = min(max_pairs, n // 4)
        src_cand = np.argsort(src_score)[::-1][:max(2, n_cand // 2)]
        tgt_cand = np.argsort(tgt_score)[:max(2, n_cand // 2)]

        best_improvement = cur_obj + 1e-15
        best_move = None

        for j in src_cand:
            for l in tgt_cand:
                if j == l or a[j] < 1e-10:
                    continue
                # Try moving a fraction of a[j] to a[l]
                delta = a[j] * 0.1
                if delta < 1e-12:
                    continue
                
                # Compute new max efficiently
                # delta_c from changing a[j] by -delta
                # and changing a[l] by +delta
                dc = np.zeros(2 * n - 1)
                # Effect of a[j] -= delta
                lo_j = j
                hi_j = min(2*n-2, j + n - 1)
                dc[lo_j:hi_j+1] -= delta * a[lo_j - j:hi_j - j + 1]
                if 2*j <= hi_j and 2*j >= lo_j:
                    dc[2*j] += delta * delta  # correction for the quadratic term
                    # Actually: old term was a[j]^2, new is (a[j]-delta)^2
                    # Change = -2*a[j]*delta + delta^2
                    # We already subtracted delta * a[j] in the linear part
                    # So we need to add delta^2 - delta*a[j]... 
                    # Hmm, let me just be more careful.
                    # The change at k=2j from reducing a[j] by delta:
                    # Old: a[j]^2 (as part of sum_i a[i]*a[2j-i], the i=j term)
                    # New: (a[j]-delta)^2
                    # Difference: (a[j]-delta)^2 - a[j]^2 = -2*a[j]*delta + delta^2
                    # In our formula dc[k] -= delta * a[k-j], at k=2j: dc[2j] -= delta * a[j]
                    # But the actual change is -2*a[j]*delta + delta^2
                    # So we're missing -delta*a[j] + delta^2 at k=2j
                    # Wait, I think the issue is that the linear approximation 
                    # dc[k] = -delta * a[k-j] only accounts for one of the two terms.
                    # Let me redo this properly.
                
                # Let me use a cleaner approach: just compute the change directly
                # For a[j] -> a[j] - delta:
                # c_new[k] - c_old[k] = -delta * a[k-j] (from a[j]*a[k-j] term)
                #                       - delta * a[k-j] (from a[k-j]*a[j] term)
                #                       + (if k=2j) delta^2 (from the cross term)
                # Wait no. c[k] = sum_i a[i]*a[k-i]
                # The term involving a[j] is: a[j]*a[k-j] (when i=j) and a[k-j]*a[j] (when i=k-j)
                # If j != k-j, these are two separate terms, each changing by -delta*a[k-j]
                # If j = k-j (k=2j), it's a single term a[j]^2, changing by -2*a[j]*delta + delta^2
                
                # So: dc[k] = -2*delta*a[k-j] for k != 2j (and k-j valid)
                #           dc[k] = -2*a[j]*delta + delta^2 for k = 2j
                
                # Let me just use the simpler (slightly inaccurate) linear approx for now
                # and verify with full recompute periodically.
                
                # Simpler: just estimate the new max
                # The peak at peak_idx[0] changes by approximately:
                k0 = int(peak_idx[0])
                # Change in c[k0] from moving delta from j to l:
                # -= delta * a[k0-j] (if valid) *= 2 (two terms)
                # += delta * a[k0-l] (if valid) *= 2
                delta_c0 = 0.0
                if 0 <= k0 - j < n:
                    delta_c0 -= 2 * delta * a[k0 - j]
                if 0 <= k0 - l < n:
                    delta_c0 += 2 * delta * a[k0 - l]
                
                # Rough estimate of new max
                new_max_est = max(c + delta_c0, c[1:] + 0)  # very rough
                # This is too rough. Let me just try the move and check.
                
                # Actually, for efficiency, let's just try a few moves and check the full c
                # But that's O(n log n) per check...
                # For now, use the linear approximation at the peak
                new_max_approx = c[k0] + delta_c0
                # Also check other peaks
                for kk in peak_idx[1:3]:
                    kk = int(kk)
                    dc_kk = 0.0
                    if 0 <= kk - j < n:
                        dc_kk -= 2 * delta * a[kk - j]
                    if 0 <= kk - l < n:
                        dc_kk += 2 * delta * a[kk - l]
                    new_max_approx = max(new_max_approx, c[kk] + dc_kk)
                
                new_obj_est = 2.0 * n * new_max_approx / (S * S)
                if new_obj_est < best_improvement:
                    best_improvement = new_obj_est
                    best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            # Update c (full recompute for safety, or O(n) update)
            # O(n) update:
            # Change from a[j] decreasing by delta:
            lo_j = j
            hi_j = min(2*n-2, j + n - 1)
            c[lo_j:hi_j+1] -= 2 * delta * a[lo_j - j:hi_j - j + 1]
            # Correction for k=2j (the quadratic term)
            if 2*j < len(c):
                c[2*j] += 2 * delta * a[j] - delta * delta  # add back the linear, subtract quadratic
                # Actually: the true change at k=2j is -2*a[j]_old*delta + delta^2
                # We applied -2*delta*a[j]_old (linear) above
                # So we need to add delta^2: c[2*j] += delta*delta
                # Wait, we already did c[2*j] -= 2*delta*a[j], which is the linear part
                # The true change is -2*a[j]*delta + delta^2
                # So we need: c[2*j] += delta^2
                c[2*j] += delta * delta
            # Hmm this is getting error-prone. Let me just recompute periodically.
            
            # Change from a[l] increasing by delta:
            lo_l = l
            hi_l = min(2*n-2, l + n - 1)
            c[lo_l:hi_l+1] += 2 * delta * a[lo_l - l:hi_l - l + 1]
            if 2*l < len(c):
                c[2*l] += delta * delta
        
        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)
            # Verify sum
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

    return best_a, best_obj
