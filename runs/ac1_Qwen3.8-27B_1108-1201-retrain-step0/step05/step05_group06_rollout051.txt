Actually, one more important consideration: the previous code used `project_simplex` which is O(n log n) due to sorting. For n=16384, this is called at every iteration. Is there a faster way?

For projecting onto the simplex {x >= 0, sum(x) = S}, the standard algorithm is:
1. Sort v in decreasing order: O(n log n)
2. Find rho (the number of positive elements in the projection): O(n)
3. Compute theta = (sum of top rho elements of v - S) / rho: O(1)
4. Result: max(v - theta, 0): O(n)

The bottleneck is the sort. For n=16384, sorting takes about 16384 * 14 ≈ 230K comparisons. This is comparable to the FFT cost.

Alternative: use a "partial sort" or "selection" algorithm to find just the threshold. But in practice, numpy's sort is highly optimized and fast.

Actually, there's a trick: if we're doing FISTA and the iterate is already close to the boundary, we can use an approximation. But let's not over-optimize and just use the standard approach.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LSE-FISTA with adaptive beta, backtracking, and
exact active-set polishing. Multi-start with diverse initializations.
Attempts both n=8192 and n=16384.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad(y, c, k=1, tol_frac=0.0):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    active = active[:k] if len(active) > k else active
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_random_init(n, rng):
    a = rng.random(n)
    return a ** 1.5


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_growth=1.005,
                  beta_max=50000.0, use_exact_every=200, exact_k=1):
    """
    LSE-FISTA with adaptive beta and periodic exact subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    consecutive_fails = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            consecutive_fails = 0

        stuck = it - last_improve

        # Update beta
        if stuck > 500:
            beta = min(beta_max, beta * beta_growth)
        elif stuck > 2000:
            beta = min(beta_max, beta * 1.01)
        elif stuck > 5000:
            beta = min(beta_max, beta * 1.02)

        # Choose gradient type
        if use_exact_every > 0 and it % use_exact_every == 0 and stuck > 100:
            grad = exact_subgrad(y, c, k=exact_k, tol_frac=0.001)
        else:
            grad = lse_grad(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        # Initial step: based on Lipschitz estimate
        L_est = 2.0 * beta * c.max()
        step = 1.0 / max(L_est, 1.0)

        # Try the step with backtracking
        found_step = False
        for bt in range(5):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            obj_new = c_new.max()
            if obj_new < cur_max or (it % 100 == 0 and obj_new < cur_max * 1.01):
                found_step = True
                break
            step *= 0.5

        if not found_step:
            # Accept anyway if it's close
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max * 1.05:
                found_step = True
            else:
                # Just add small noise

                x_new = project_simplex(y + noise, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart if stuck
        if stuck > 8000 and it % 300 == 0:
            t_mom = 1.0
            noise_scale = max(1e-5, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        # Periodic momentum restart
        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_polish(a_init, max_time, rng):
    """Final polishing with exact subgradient from top lags."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k for exact subgradient
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = min(4, 2 + stuck // 100)
        else:
            k = min(8, 4 + stuck // 200)

        grad = exact_subgrad(y, c, k=k, tol_frac=0.005 if k > 1 else 0.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = min(1.0 / gnorm, 0.1)
        # Backtracking
        for bt in range(4):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max:
                break
            step *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 200 == 0:
            t_mom = 1.0
            y = project_simplex(y + rng.normal(0, 1e-4 * S / np.sqrt(n), n), S)
            stuck = 0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LSE-FISTA with adaptive beta, occasional LP-based direction finding,
and exact active-set polishing. Multi-start with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad(y, c, k=1, tol_frac=0.0):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) > k:
        # Take top k
        idx = np.argsort(c)[::-1][:k]
        active = idx
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_gaussian_init(n, sigma_frac=0.3):
    tt = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((tt - center) / sigma) ** 2)


def make_random_init(n, rng):
    a = rng.random(n)
    return a ** 1.5


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
