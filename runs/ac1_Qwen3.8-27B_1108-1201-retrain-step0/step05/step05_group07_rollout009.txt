"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a hybrid approach combining:
1. Weighted subgradient descent with adaptive top-k lag emphasis
2. Periodic LP-based direction finding to escape local minima
3. LSE-smoothed gradient polishing
4. Multi-length exploration and simulated annealing restarts
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, k, temperature=1.0):
    """
    Compute a weighted subgradient of max(conv(a,a)) using top-k lags.
    The weight for each lag is proportional to exp(c_j / temperature - max(c) / temperature).
    """
    n = len(a)
    k = min(k, 2 * n - 1)
    
    # Softmax-like weights on the top-k lags
    top_idx = np.argsort(c)[::-1][:k]
    c_top = c[top_idx]
    
    # Temperature-scaled softmax weights
    scaled = (c_top - c_top.max()) / max(temperature, 1e-15)
    weights = np.exp(scaled)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    
    # Compute gradient
    grad = np.zeros(n)
    for ii in range(k):
        j = int(top_idx[ii])
        w = weights[ii]
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        if i_start <= i_end:
            indices = j - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w * a[indices]
    
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of the log-sum-exp smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # Gradient of sum_j w_j * (a*a)_j w.r.t. a
    # = 2 * sum_j w_j * (shifted a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lp_direction(a, M, k_constraints=200):
    """
    Solve LP to find direction g >= 0, sum(g)=1, such that (a*g)_j <= M for top-k lags.
    Returns the optimal g or None if infeasible.
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # Select top-k constraints (the lags where a*a is largest)
    top_idx = np.argsort(c)[::-1][:k_constraints]
    
    # Build constraint matrix: (a * g)_j <= M for j in top_idx
    # (a * g)_j = sum_i a_i * g_{j-i}
    rows = []
    cols = []
    vals = []
    
    for jj in top_idx:
        j = int(jj)
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            g_idx = j - i  # index into g
            if 0 <= g_idx < n:
                rows.append(len(rows))
                cols.append(g_idx)
                vals.append(a[i])
    
    if len(rows) == 0:
        return None
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(k_constraints, n))
    b_ub = np.full(k_constraints, M)
    
    # Maximize sum(g) subject to g >= 0 and A_ub @ g <= b_ub
    # Convert to minimization: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    
    return g / g_sum  # normalize to sum = 1

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def obj_val(a):

    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_main(a0, S, max_time, rng, beta_start=20.0, beta_end=2000.0):

    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # momentum velocity

    best_max = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:

            k = int(8 + 40 * progress)
            temp = 0.01 * S / np.sqrt(n)

        elif progress < 0.8:

            frac = (progress - 0.5) / 0.3
            k = int(48 + 32 * frac)
            temp = 0.005 * S / np.sqrt(n)
            beta = beta_start + (beta_end - beta_start) * frac
            beta = min(beta, 400.0 / max(1e-15, c.max() - c.min()))
        else:
            # Late: full LSE
            k = 80
            beta = min(beta_end, 400.0 / max(1e-15, c.max() - c.min()))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute gradient
        if progress < 0.5:
            grad = compute_gradient_topk(a, c, k, temp)
        else:
            grad = compute_lse_grad(a, c, beta)

        if gnorm < 1e-14:
            break

        if progress < 0.5:
            step = 2.0 * gap / (gnorm * gnorm)

        else:
            step = 1.0 / (1.2 * beta)
            step = min(step, 2.0 / gnorm)

        y = project_simplex(a + 0.9 * (y - a) + v, S)

        c_y = fft_convolve_self(y)

            a = y

            a = project_simplex(a - step * grad, S)
            v = np.zeros(n)
            y = a.copy()

            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            v = np.zeros(n)
            stuck = 0
            patience = int(patience * 1.2)

            elapsed = time.monotonic() - start
            print(f"    it={it}, t={elapsed:.0f}s, obj={2.0*n*best_max/(S*S):.6f}")
    
    return best_a, best_max


def lp_direction_find(a, M, k_constraints=300):
    """
    Find direction g >= 0 with sum(g)=1 such that (a*g)_j <= M for top-k lags.
    """
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:k_constraints]
    
    rows_list = []
    cols_list = []
    vals_list = []
    
    for idx in range(len(top_idx)):
        j = int(top_idx[idx])
        i_start = max(0, j - n + 1)
        i_end = min(n - 1, j)
        for i in range(i_start, i_end + 1):
            g_idx = j - i
            rows_list.append(idx)
            cols_list.append(g_idx)
            vals_list.append(a[i])
    
    if not rows_list:
        return None
    
    A_ub = csr_matrix((vals_list, (rows_list, cols_list)), shape=(len(top_idx), n))
    b_ub = np.full(len(top_idx), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60, 'feasibility_tolerance': 1e-7})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    
    return None
