"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-scale optimization with adaptive active-set subgradients:
1. Multi-scale initialization (small n -> upsample -> large n)
2. LSE-FISTA with adaptive beta and backtracking

4. Random restarts to escape local minima
5. Final high-precision polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgrad(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight function: uniform over top-k
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_topk_subgrad(a, c, top_idx, weights):
    """Subgradient with custom weights on selected lags."""
    n = len(a)
    w_full = np.zeros(len(c))
    w_full[top_idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
                  beta_max=50000.0, step_scale=1.0, use_backtrack=True,
                  noise_period=3000, noise_base=0.002):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 3.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Periodic noise injection
        if it - last_improve > noise_period and it % 50 == 0:
            t_mom = 1.0
            decay = (it - last_improve) / noise_period
            ns = max(1e-7, noise_base * S / np.sqrt(n) * np.exp(-decay))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            a = y.copy()

    return best_a, best_obj


def run_active_set(a_init, max_time, rng, k_init=4, k_max=64):
    """Optimize using subgradient of top-K active lags."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    k = k_init
    step = 0.5

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 10000:
            break

        # Adapt k: increase as we get stuck
        k = min(k_max, k_init + stuck // 200)

        top_idx = np.argsort(c)[::-1][:k]
        # Use geometric weights: more weight on higher lags
        vals = c[top_idx]
        vals_shifted = vals - vals.min() + 1e-10
        weights = vals_shifted / vals_shifted.sum()
        
        grad = weighted_topk_subgrad(a, c, top_idx, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step
        step = min(step, 1.5 * S / gnorm)
        step = max(step, 1e-10)
        
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)
        
        if obj_new <= cur_obj + 1e-13:
            a = a_new
            c = c_new
            step = min(step * 1.02, 3.0 * S / max(gnorm, 1e-15))
        else:
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            obj_new2 = compute_obj(a_new)
            if obj_new2 <= cur_obj + 1e-13:
                a = a_new
                c = fft_convolve_self(a)

        # Noise when stuck
        if stuck > 2000 and it % 100 == 0:
            ns = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj

def upsample_linear(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "uniform":
        a = np.ones(n)
    elif method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-4.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt + 0.5)
    elif method == "multi_tone":
        a = (0.3 + 0.2*np.sin(2*np.pi*5*tt) + 
             0.2*np.sin(2*np.pi*13*tt+1.0) + 
             0.15*np.sin(2*np.pi*23*tt+2.0))
        a = np.abs(a) + 0.1
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "barker_like":
        # Generate pseudo-random binary and smooth
        b = (rng.random(n) > 0.5).astype(float)
        # Smooth with a window
        w = np.ones(5) / 5
        a = np.convolve(b, w, mode='same') + 0.1
    elif method == "zadoff_chu":
        # Zadoff-Chu-like: a_j = exp(2*pi*alpha*j*(j+1)/(2n))
        alpha = 3
        a = np.exp(2j * np.pi * alpha * tt * (tt + 1/n) / 2)
        a = np.abs(a) + 0.3  # Make non-negative
    elif method == "triangular":
        a = 1.0 - 2.0 * np.abs(tt - 0.5)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def coordinate_descent_refine(a_init, max_time, rng):

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 30000:
            break

        k_star = int(np.argmax(c))
        
        # Compute contribution of each coordinate to c[k_star]
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            # c[k_star] = sum_{j=lo}^{hi} a[j] * a[k_star - j]

                    c = fft_convolve_self(a)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-scale optimization with adaptive LSE-FISTA, active-set
subgradient refinement, and coordinate descent:
1. Multi-scale warm start (optimize at small n, upsample to target)
2. Intensive LSE-FISTA with multiple parameter variants
3. Active-set subgradient (top-K lags) for focused refinement
4. Coordinate descent for fine-tuning peak lags
5. Ultra-high-beta final polish for precision
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(y, beta):
    """Compute LSE value and gradient of smoothed max conv w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgrad(a, c, k):
    """Uniform subgradient from top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_topk_subgrad(a, c, top_idx, weights):
    """Subgradient with custom weights on selected lags."""
    n = len(a)
    w_full = np.zeros(len(c))
    w_full[top_idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

                  beta_max=50000.0, step_scale=1.0, use_backtrack=True,
                  noise_period=3000, noise_base=0.002):

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:

            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > noise_period and it % 50 == 0:
            t_mom = 1.0
            decay = (it - last_improve) / noise_period
            ns = max(1e-7, noise_base * S / np.sqrt(n) * np.exp(-decay))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            a = y.copy()

    return best_a, best_obj


def run_active_set(a_init, max_time, rng, k_init=4, k_max=64):
    """Optimize using weighted subgradient of top-K active lags."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    k = k_init
    step = 0.5

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 15000:
            break

        k = min(k_max, k_init + stuck // 200)
        top_idx = np.argsort(c)[::-1][:k]
        vals = c[top_idx]
        vals_shifted = np.maximum(vals - vals.min() + 1e-10, 1e-10)
        weights = vals_shifted / vals_shifted.sum()

        grad = weighted_topk_subgrad(a, c, top_idx, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = min(step, 1.5 * S / gnorm)
        step = max(step, 1e-10)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)

        if obj_new <= cur_obj + 1e-13:
            a = a_new
            step = min(step * 1.02, 3.0 * S / max(gnorm, 1e-15))
        else:
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 2000 and it % 100 == 0:
            ns = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def coordinate_descent_refine(a_init, max_time, rng):
    """Coordinate descent: transfer mass to reduce peak convolution."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np_clip_nonneg(a_init, n)
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 40000:
            break

        k_star = int(np.argmax(c))

        # Contribution of each coordinate j to c[k_star]
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            partner = k_star - j
            if 0 <= partner < n:
                contrib[j] += a[partner]

        # Also consider top-2 lags for multi-lag awareness
        top2 = np.argsort(c)[::-1][:2]
        multi_contrib = contrib.copy()
        if len(top2) > 1:
            k2 = int(top2[1])
            lo2 = max(0, k2 - n + 1)
            hi2 = min(n - 1, k2)
            for j in range(lo2, hi2 + 1):
                partner = k2 - j
                if 0 <= partner < n:
                    multi_contrib[j] += 0.5 * a[partner]

        src_score = multi_contrib * np.maximum(a, 1e-10)
        tgt_score = -multi_contrib

        n_try = min(10, n)
        src_cands = np.argsort(src_score)[::-1][:n_try]
        tgt_cands = np.argsort(tgt_score)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_move = None

        for j in src_cands:
            if a[j] < 1e-10:
                continue
            for l in tgt_cands:
                if j == l:
                    continue
                delta = a[j] * 0.03
                if delta < 1e-14:
                    continue

                dc_star = 0.0
                if 0 <= k_star - j < n:
                    dc_star -= delta * a[k_star - j]
                if 0 <= k_star - l < n:
                    dc_star += delta * a[k_star - l]

                if dc_star < -1e-15:
                    new_peak = c[k_star] + dc_star
                    rest_max = 0.0
                    if new_peak > rest_max:
                        rest_max = new_peak
                    # Check second max lag
                    k2 = int(top2[1]) if len(top2) > 1 else -1
                    if k2 >= 0 and k2 != k_star:
                        dc2 = 0.0
                        if 0 <= k2 - j < n:
                            dc2 -= delta * a[k2 - j]
                        if 0 <= k2 - l < n:
                            dc2 += delta * a[k2 - l]
                        if c[k2] + dc2 > rest_max:
                            rest_max = c[k2] + dc2
                    new_max = max(rest_max, new_peak)
                    new_obj_est = 2.0 * n * new_max / (S * S)
                    if new_obj_est < best_delta_obj:
                        best_delta_obj = new_obj_est
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            c = fft_convolve_self(a)
        else:
            if it - last_improve > 500:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    c = fft_convolve_self(a)

    return best_a, best_obj


def np_clip_nonneg(a_init, n):
    """Clip to non-negative and valid range."""
    return np.clip(np.asarray(a_init, dtype=np.float64).flatten()[:n], 0, 1000)


def upsample_linear(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "uniform":
        a = np.ones(n)
    elif method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-4.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt + 0.5)
    elif method == "multi_tone":
        a = (0.3 + 0.2 * np.sin(2 * np.pi * 5 * tt) +
             0.2 * np.sin(2 * np.pi * 13 * tt + 1.0) +
             0.15 * np.sin(2 * np.pi * 23 * tt + 2.0))
        a = np.abs(a) + 0.1
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "barker_like":
        b = (rng.random(n) > 0.5).astype(float)
        w = np.ones(7) / 7
        a = np.convolve(b, w, mode='same') + 0.1
    elif method == "triangular":
        a = 1.0 - 2.0 * np.abs(tt - 0.5)
        a = np.maximum(a, 0.01)
    elif method == "smoothed_random":
        a = rng.random(n)
        w = np.ones(11) / 11
        a = np.convolve(a, w, mode='same') + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.

    Multi-phase strategy:
    1. Multi-scale warm start with diverse initializations
    2. Intensive LSE-FISTA with multiple parameter sweeps
    3. Active-set subgradient refinement
    4. Coordinate descent for peak reduction
    5. Ultra-high-beta final polish
    6. Optional larger-n exploration
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from global
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        target_n = len(a_start)
    else:
        target_n = 8192

    print("Target n = %d, budget = %.0fs" % (target_n, budget_s))

    # === Phase 1: Multi-scale warm start ===
    p1_time = min(100.0, (deadline - time.monotonic()) * 0.12)
    if p1_time > 20:
        print("\n=== Phase 1: Multi-scale warm start (%.0fs) ===" % p1_time)

        small_n = min(512, target_n)
        best_small = None
        best_small_obj = np.inf

        methods = ["uniform", "cosine", "random", "multi_tone", "smoothed_random"]
        time_per = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_s = make_init(small_n, method, rng)
            a_opt, obj_opt = run_lse_fista(a_s, time_per, rng,
                                            beta_init=15, beta_rate=0.025,
                                            beta_max=5000, step_scale=1.0,
                                            use_backtrack=True)
            if obj_opt < best_small_obj:
                best_small_obj = obj_opt
                best_small = a_opt.copy()
            print("  small n=%d %s: obj=%.6f" % (small_n, method, obj_opt))

        if best_small is not None:
            # Progressive upsample with optimization at each level
            a_cur = best_small.copy()
            n_cur = small_n
            while n_cur < target_n:
                n_next = min(n_cur * 2, target_n)
                if n_next > n_cur:
                    t_old = np.linspace(0, 1, n_cur, endpoint=False)
                    t_new = np.linspace(0, 1, n_next, endpoint=False)
                    a_cur = np.interp(t_new, t_old, a_cur)
                    n_cur = n_next
                    S = np.sqrt(2.0 * n_cur)
                    a_cur = a_cur * (S / a_cur.sum())
                    if time.monotonic() < deadline - 30:
                        opt_time = min(40.0, (deadline - time.monotonic()) * 0.08)
                        if opt_time > 10:
                            a_cur, _ = run_lse_fista(
                                a_cur, opt_time, rng,
                                beta_init=5, beta_rate=0.01,
                                beta_max=3000, step_scale=1.0,
                                use_backtrack=False)
            S_t = np.sqrt(2.0 * target_n)
            a_cur = a_cur * (S_t / a_cur.sum())
            obj_up = compute_obj(a_cur)
            print("  Upsampled to n=%d: obj=%.6f" % (target_n, obj_up))
            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_cur.copy()

    # Use provided start if available
    if a_start is not None:
        S = np.sqrt(2.0 * len(a_start))
        a_norm = a_start * (S / a_start.sum())
        obj_start = compute_obj(a_norm)
        print("\n  Provided sequence: obj=%.6f" % obj_start)
        if obj_start < global_best_obj:
            global_best_obj = obj_start
            global_best_a = a_norm.copy()

    if global_best_a is None:
        global_best_a = make_init(target_n, "cosine", rng)
        global_best_obj = compute_obj(global_best_a)
        print("  Fallback init: obj=%.6f" % global_best_obj)

    # === Phase 2: Intensive LSE-FISTA ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.30
    if p2_time > 30:
        print("\n=== Phase 2: LSE-FISTA sweep (%.0fs) ===" % p2_time)

        variants = [
            (20, 0.008, 20000, 1.0, True),
            (50, 0.005, 50000, 0.6, True),
            (100, 0.003, 100000, 0.4, True),
            (10, 0.015, 8000, 1.5, False),
            (30, 0.007, 30000, 0.8, True),
        ]
        time_per = p2_time / len(variants)
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            a_opt, obj_opt = run_lse_fista(global_best_a, time_per, rng,
                                            beta_init=bi, beta_rate=br,
                                            beta_max=bm, step_scale=ss,
                                            use_backtrack=bt)
            tag = "***" if obj_opt < global_best_obj else ""
            print("  V%d beta=%d rate=%.4f: obj=%.6f %s" % (i+1, bi, br, obj_opt, tag))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Active-set subgradient ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.22
    if p3_time > 30:
        print("\n=== Phase 3: Active-set refinement (%.0fs) ===" % p3_time)
        a_as, obj_as = run_active_set(global_best_a, p3_time, rng,
                                       k_init=3, k_max=32)
        tag = "***" if obj_as < global_best_obj else ""
        print("  Active-set: obj=%.6f %s" % (obj_as, tag))
        if obj_as < global_best_obj:
            global_best_obj = obj_as
            global_best_a = a_as.copy()

    # === Phase 4: Coordinate descent ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 30:
        print("\n=== Phase 4: Coordinate descent (%.0fs) ===" % p4_time)
        a_cd, obj_cd = coordinate_descent_refine(global_best_a, p4_time, rng)
        tag = "***" if obj_cd < global_best_obj else ""
        print("  Coord descent: obj=%.6f %s" % (obj_cd, tag))
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # Follow-up FISTA after CD
        remaining4 = deadline - time.monotonic()
        if remaining4 > 20:
            a_cd2, obj_cd2 = run_lse_fista(global_best_a, remaining4 * 0.6, rng,
                                            beta_init=50, beta_rate=0.006,
                                            beta_max=20000, step_scale=0.7,
                                            use_backtrack=True)
            tag = "***" if obj_cd2 < global_best_obj else ""
            print("  FISTA after CD: obj=%.6f %s" % (obj_cd2, tag))
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()

    # === Phase 5: Final high-precision polish ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.90
    if p5_time > 20:
        print("\n=== Phase 5: Final polish (%.0fs) ===" % p5_time)

        # Ultra-high beta FISTA
        t_a = p5_time * 0.4
        a_pol, obj_pol = run_lse_fista(global_best_a, t_a, rng,
                                        beta_init=500, beta_rate=0.001,
                                        beta_max=300000, step_scale=0.25,
                                        use_backtrack=True,
                                        noise_period=5000, noise_base=0.0008)
        tag = "***" if obj_pol < global_best_obj else ""
        print("  High-beta FISTA: obj=%.6f %s" % (obj_pol, tag))
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        # Active-set with small k
        remaining5 = deadline - time.monotonic()
        if remaining5 > 15:
            a_as2, obj_as2 = run_active_set(global_best_a, remaining5 * 0.4, rng,
                                             k_init=2, k_max=12)
            tag = "***" if obj_as2 < global_best_obj else ""
            print("  Active-set final: obj=%.6f %s" % (obj_as2, tag))
            if obj_as2 < global_best_obj:
                global_best_obj = obj_as2
                global_best_a = a_as2.copy()

        # One more FISTA round
        remaining5b = deadline - time.monotonic()
        if remaining5b > 10:
            a_f, obj_f = run_lse_fista(global_best_a, remaining5b, rng,
                                         beta_init=2000, beta_rate=0.0003,
                                         beta_max=800000, step_scale=0.15,
                                         use_backtrack=True,
                                         noise_period=10000, noise_base=0.0003)
            tag = "***" if obj_f < global_best_obj else ""
            print("  Ultra-high-beta: obj=%.6f %s" % (obj_f, tag))
            if obj_f < global_best_obj:
                global_best_obj = obj_f
                global_best_a = a_f.copy()

    # === Phase 6: Try larger n ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and target_n <= 8192:
        n_large = target_n * 2
        print("\n=== Phase 6: Larger n=%d (%.0fs) ===" % (n_large, remaining * 0.85))

        t_old = np.linspace(0, 1, target_n, endpoint=False)
        t_new = np.linspace(0, 1, n_large, endpoint=False)
        a_large = np.interp(t_new, t_old, global_best_a)
        S_l = np.sqrt(2.0 * n_large)
        a_large = a_large * (S_l / a_large.sum())

        a_lo, obj_lo = run_lse_fista(a_large, remaining * 0.4, rng,
                                       beta_init=10, beta_rate=0.006,
                                       beta_max=10000, step_scale=1.0,
                                       use_backtrack=True)
        tag = "***" if obj_lo < global_best_obj else ""
        print("  Large-n FISTA: obj=%.6f %s" % (obj_lo, tag))
        if obj_lo < global_best_obj:
            global_best_obj = obj_lo
            global_best_a = a_lo.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 40:
            a_la2, obj_la2 = run_active_set(global_best_a, remaining6 * 0.6, rng,
                                             k_init=4, k_max=32)
            tag = "***" if obj_la2 < global_best_obj else ""
            print("  Large-n active-set: obj=%.6f %s" % (obj_la2, tag))
            if obj_la2 < global_best_obj:
                global_best_obj = obj_la2
                global_best_a = a_la2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f, n=%d ===" % (ver, len(final_list)))
        else:
            print("WARNING: invalid, using uniform fallback")
            n_fb = 8192
            final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb
    else:
        n_fb = 8192
        final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb

    return final_list
