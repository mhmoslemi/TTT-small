def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def subgradient_topk(a, c, k=5):
    """Compute a weighted subgradient using top-k peaks."""
    n = len(a)
    cm = c.max()
    # Get top k peaks
    top_k = np.argsort(c)[::-1][:k]
    # Weight by excess
    weights = np.maximum(c[top_k] - cm + (c[top_k].max() - c[top_k].min()), 0)
    weights = np.maximum(weights, 0.1)  # minimum weight
    weights /= weights.sum()
    
    # Compute subgradient: 2 * sum w_i * a shifted to position k_i
    grad = np.zeros(n)
    for i, (ki, wi) in enumerate(zip(top_k, weights)):
        # (a * delta_ki)[j] = a[ki - j] if 0 <= ki-j < n
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        grad[lo:hi+1] += wi * a[ki - hi:ki - lo + 1][::-1]
    grad *= 2.0
    return grad

def fista_lse(a_init, n, S, max_time, rng, beta_start=50, beta_end=20000,

    a = normalize_sum(a_init, n)

    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:

            best_a = y.copy()
            last_improve = it
        
        if use_subgrad and beta > 1000:
            grad = subgradient_topk(y, c, topk)
            step = 1.0 / (2.0 * np.max(y) + 1e-10)
        else:

        # The Hessian involves beta * (something), so L ~ 2*beta*||a||_inf.

            break

        stagnant = it - last_improve

            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))

    c_final = fft_convolve_self(a)

        best_a = a.copy()

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length n.
Uses FISTA with LSE smoothing for broad optimization, followed by
coordinate descent polish for fine-tuning. Exploits sequence asymmetry
to reduce the center convolution peak.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    vals = beta * (c - cm)
    # Clip to avoid overflow
    vals = np.clip(vals, -500, 500)
    e = np.exp(vals)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def fista_optimize(a_init, n, S, max_time, rng,
                   beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stagnation_counter = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: gradual increase
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/L where L is Lipschitz constant
        # For LSE with parameter beta, L ≈ 2 * beta * ||a||_inf
        a_max = np.max(y)
        L = 2.0 * beta * (a_max + 1e-10)
        step = 1.0 / L

        # Stagnation handling: reset momentum and add noise
        if stagnation_counter > 300:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnation_counter / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            stagnation_counter = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def coordinate_polish(a, n, max_time, max_cycles=500, n_elements=300):
    """
    Coordinate descent polish: iterate over top-responsibility elements,
    finding optimal value via ternary search. Maintains c incrementally.
    """
    a = np.clip(a.copy(), 0.0, 1000.0)
    S = a.sum()
    c = fft_convolve_self(a)
    best = c.max()
    start = time.monotonic()
    
    conv_len = 2 * n - 1
    
    for cycle in range(max_cycles):
        if time.monotonic() - start > max_time:
            break
        
        cm = c.max()
        
        # Compute responsibility for each element
        # Responsibility: how much each a[j] contributes to the top peaks
        top_k = min(10, conv_len)
        top_peaks = np.argsort(c)[::-1][:top_k]
        
        resp = np.zeros(n)
        for k in top_peaks:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - 0.995 * cm)
            if weight > 0:
                resp[js] += contrib * weight
        
        # Get top elements to modify
        order = np.argsort(resp)[::-1][:n_elements]
        
        cycle_improved = False
        for j in order:
            v = a[j]
            if v < 1e-15:
                continue
            
            # Find optimal x via ternary search
            # f(x) = max_k c_new[k](x)
            # c_new[k](x) = c[k] + 2*(x-v)*a[k-j] for k in [j, j+n-1], k != 2j
            # c_new[2j](x) = c[2j] + (x-v)*(x+v)
            # c_new[k](x) = c[k] for k outside [j, j+n-1]
            
            k_start = j
            k_end = min(j + n, conv_len)  # exclusive
            a_shifted = a[0:k_end - j].copy()  # a[k-j] for k in [k_start, k_end-1]
            
            # Save for restoring (we need a_shifted before updating a[j])
            # a_shifted[j] = a[j] = v (this is the old value, which is what we want)
            
            c_outside_max = 0.0
            if k_start > 0:
                c_outside_max = max(c_outside_max, c[:k_start].max())
            if k_end < conv_len:
                c_outside_max = max(c_outside_max, c[k_end:].max())
            
            c_2j = c[2 * j] if 2 * j < conv_len else 0.0
            k_2j_in_range = (2 * j >= k_start) and (2 * j < k_end)
            
            def f_of_x(x):
                delta = x - v
                # Linear part (for k in range, k != 2j)
                c_aff = c[k_start:k_end].copy() + 2.0 * delta * a_shifted
                if k_2j_in_range:
                    # Override with quadratic
                    c_aff[2 * j - k_start] = c_2j + delta * (x + v)
                return max(c_aff.max(), c_outside_max)
            
            # Ternary search
            x_lo = 0.0
            x_hi = max(2.0 * v, v + 0.1 * S / n)
            
            for _ in range(30):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0
                if f_of_x(m1) < f_of_x(m2):
                    x_hi = m2
                else:
                    x_lo = m1
            
            x_opt = 0.5 * (x_lo + x_hi)
            x_opt = max(0.0, min(x_opt, 1000.0))
            
            if abs(x_opt - v) < 1e-12:
                continue
            
            # Update c and a
            delta = x_opt - v
            c[k_start:k_end] += 2.0 * delta * a_shifted
            if k_2j_in_range:
                c[2 * j] += delta * delta  # Correction: was 2*delta*v, need delta*(x+v), diff = delta^2
            a[j] = x_opt
            
            if c.max() < best:
                best = c.max()
                cycle_improved = True
        
        # Optional: re-normalize to maintain sum
        # (coordinate descent preserves non-negativity but may change the sum slightly)
        # Since the objective is scale-invariant, this doesn't matter for the value,
        # but let's keep it clean.
    
    return a, best


def multi_peak_subgradient(a, c, k=5):
    """Compute a weighted subgradient using top-k peaks."""
    n = len(a)
    cm = c.max()
    top_idx = np.argsort(c)[::-1][:k]
    # Weight: uniform over top k (or weighted by excess)
    weights = np.ones(k) / k
    
    grad = np.zeros(n)
    for ki, wi in zip(top_idx, weights):
        # (a * delta_ki)[j] = a[ki - j] if 0 <= ki-j < n
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += wi * a[ki - hi:ki - lo + 1][::-1]
    grad *= 2.0
    return grad


def subgrad_polish(a, n, S, max_time, rng, k=5):
    """Polish using multi-peak subgradient with backtracking."""
    a = np.clip(a.copy(), 0.0, 1000.0)
    S = a.sum()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        grad = multi_peak_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / (2.0 * np.max(a) + 1e-10)
        for _ in range(15):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-8):
                a = a_test
                break
            step *= 0.5
        else:
            # No improvement, add small noise
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            else:
                break
    
    return best_a, best
