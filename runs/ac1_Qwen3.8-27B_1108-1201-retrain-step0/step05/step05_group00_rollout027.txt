def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)

        return v

    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]

    a = a.copy().astype(np.float64)
    n = len(a)
    beta = beta_init
    s_hist = []
    y_hist = []
    best_a = a.copy()
    best_cmax = np.inf
    stall_count = 0

    c, A = compute_conv(a, N)
    lse_val, weights, c_max = lse_value_and_weights(c, beta)
    g = grad_from_conv_weights(weights, a, N)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = S * 2.0 / (n * max(beta, 1.0) * 0.5)
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta increase
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = S * 2.0 / (n * max(beta, 1.0) * 0.5)

        # Compute direction
        direction = lbfgs_direction(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            direction = -g / (np.linalg.norm(g) + 1e-30)
            dir_norm = np.linalg.norm(direction)
            if dir_norm < 1e-18:
                break

        # Line search
        alpha = step_scale
        a_new = a
        cmax_new = c_max
        best_cmax_ls = c_max
        best_a_ls = a

        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls < best_cmax_ls:
                best_cmax_ls = cm_ls
                best_a_ls = candidate.copy()
            if cm_ls <= c_max * 1.001 or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha *= 0.5

        # Update best
        if best_cmax_ls < best_cmax:
            best_cmax = best_cmax_ls
            best_a = best_a_ls.copy()
            stall_count = 0
        elif cmax_new < best_cmax - 1e-9:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Compute new gradient
        c_new, _ = compute_conv(a_new, N)
        lse_new, weights_new, cmax_new = lse_value_and_weights(c_new, beta)
        g_new = grad_from_conv_weights(weights_new, a_new, N)

        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-12 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_new

        # Periodic restart if stuck
        if stall_count >= 50:
            frac = 0.3 + 0.1 * (stall_count // 10)
            frac = min(frac, 0.8)
            A_full = fft(a, N)
            mag = np.abs(A_full)
            n_rand = max(1, int(N * frac))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            new_phase = np.angle(A_full).copy()
            new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
            A_new = mag * np.exp(1j * new_phase)
            a_restart = np.real(ifft(A_new))[:n]
            a_restart = np.maximum(a_restart, 0.0)
            s_re = a_restart.sum()
            if s_re > 1e-15:
                a = a_restart * (S / s_re)
            else:
                a = np.ones(n) * S / n
            s_hist = []
            y_hist = []
            c, _ = compute_conv(a, N)
            lse_val, weights, c_max = lse_value_and_weights(c, beta)
            g = grad_from_conv_weights(weights, a, N)
            stall_count = 0

    return best_a, best_cmax


def coord_descent_block(a, S, N, max_time, rng, top_k=3):
    """
    Coordinate descent polishing: target top-K peaks, transfer mass
    from peak contributors to low contributors.
    Returns (best_a, best_cmax).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    c, A = compute_conv(a, N)
    best_a = a.copy()
    best_cmax = c.max()
    start_time = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        # Find top peaks
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        # For the dominant peak, find best transfer
        k_star = top_idx[0]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        i_range = np.arange(lo, hi + 1)
        j_range = k_star - i_range
        products = a[i_range] * a[j_range]

        if products.max() < 1e-20:
            break

        i_best = i_range[int(np.argmax(products))]

        # Find transfer target: minimize contribution to ALL top peaks
        contrib_total = np.zeros(n)
        for kp in top_idx:
            lo_p = max(0, kp - n + 1)
            hi_p = min(n - 1, kp)
            ri = np.arange(lo_p, hi_p + 1)
            rj = kp - ri
            contrib_total[ri] += a[ri] * a[rj]

        # Exclude i_best from targets
        contrib_total[i_best] = np.inf
        i_min = int(np.argmin(contrib_total))

        # Transfer amount: adaptive
        transfer = min(0.08 * a[i_best], 0.2 * S / n)
        # Also ensure we don't make a[i_best] negative
        transfer = min(transfer, 0.9 * a[i_best])
        if transfer < 1e-18:
            break

        a[i_best] -= transfer
        a[i_min] += transfer

        # Recompute c (every few steps to save time)
        if it % 3 == 0 or it < 10:
            c, A = compute_conv(a, N)
            cur_max = c.max()
            if cur_max < best_cmax:
                best_cmax = cur_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # If stuck for too long, try a random perturbation
        if no_improve > 200:
            n_perturb = rng.integers(1, max(2, n // 100))
            pert_idx = rng.choice(n, size=n_perturb, replace=False)
            epsilon = S / n * 0.01
            for pidx in pert_idx:
                donor = int(np.argmax(a))
                if donor != pidx:
                    tr = min(epsilon, 0.5 * a[donor])
                    a[donor] -= tr
                    a[pidx] += tr
            no_improve = 0

    return best_a, best_cmax


def spectral_shaping(a, S, N, rng, strength=0.3):
    """
    Shape the spectral magnitude toward flat while preserving non-negativity.
    Returns improved sequence or original.
    """
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A).copy()
    phase = np.angle(A)

    # Target: geometric mean between current magnitude and flat
    target_mag = np.full(N, S / np.sqrt(N))
    # Geometric interpolation
    new_mag = np.power(mag, (1.0 - strength)) * np.power(target_mag, strength)

    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new * (S / s_new)
    return a_new


def spectral_phase_shake(a, S, N, rng, frac=0.4):
    """Randomize a fraction of spectral phases."""
    n = len(a)
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    return a_new * (S / s_new)


def annealed_max_descent(a, S, N, max_time, rng, T_init=0.5, T_min=0.01):
    """
    Simulated annealing on the max-convolution objective.
    Random transfers with acceptance criterion.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    c, A = compute_conv(a, N)
    best_a = a.copy()
    best_cmax = c.max()
    start_time = time.monotonic()
    elapsed = 0.0
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Anneal temperature
        frac_done = (time.monotonic() - start_time) / max_time
        T = T_init * (T_min / T_init) ** frac_done

        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Determine transfer direction and amount
        if a[i] > a[j]:
            donor, rcpt = i, j
        else:
            donor, rcpt = j, i
        transfer = min(a[donor] * 0.05, S / n * 0.1)
        if transfer < 1e-18:
            continue

        a[donor] -= transfer
        a[rcpt] += transfer

        # Compute new c (incremental)
        # c[k] changes by: -transfer * (a_new[Donor_shift] + ...) + transfer * (a_new[Rcpt_shift] + ...)
        # Just recompute for correctness
        c_new, _ = compute_conv(a, N)
        new_max = c_new.max()
        delta = new_max - c.max()

        # Acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c = c_new
            if new_max < best_cmax:
                best_cmax = new_max
                best_a = a.copy()
        else:
            # Reject
            a[donor] += transfer
            a[rcpt] -= transfer

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_flat_spec(n, N, rng):
    """Initialize with flat magnitude spectrum."""
    target_mag = np.ones(N) * (np.sqrt(n) * 0.8)
    target_mag[0] = np.sqrt(n)
    phase = rng.uniform(0, 2 * np.pi, N)
    phase[0] = 0
    A = target_mag * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.01
    return a


def make_init_binary(n, rng, p=0.5):
    a = (rng.random(n) < p).astype(np.float64)
    a[a > 0] = 0.5 + 0.5 * rng.random((a > 0).sum())
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses multi-phase optimization: L-BFGS on LSE, coordinate descent,
    spectral shaping, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # Get starting point from given sequence
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_start = np.clip(given, 0, 1000).astype(np.float64)
            s_given = a_start.sum()
            if s_given > 0.01:
                a_start = a_start * (S / s_given)
        elif given.size >= 1024 and given.size <= 32768:
            # Try to use it by truncating or padding
            if given.size < n:
                a_full = np.zeros(n)
                a_full[:given.size] = given
            else:
                a_full = given[:n]
            a_start = np.clip(a_full, 0, 1000)
            s_given = a_start.sum()
            if s_given > 0.01:
                a_start = a_start * (S / s_given)
    except Exception:
        pass

    if a_start is None:
        a_start = np.ones(n) * S / n

    # Evaluate starting point
    c_start, _ = compute_conv(a_start, N)
    best_cmax = c_start.max()
    best_a = a_start.copy()
    best_score = 2.0 * n * best_cmax / (S * S)
    print("Starting score: %.6f" % best_score)

    global_best_cmax = best_cmax
    global_best_a = best_a.copy()

    def update_global(a_candidate, label=""):
        nonlocal global_best_cmax, global_best_a
        a_c = a_candidate.copy()
        c_c, _ = compute_conv(a_c, N)
        sc = c_c.max()
        if sc < global_best_cmax:
            global_best_cmax = sc
            global_best_a = a_c.copy()
            score = 2.0 * n * sc / (S * S)
            print("  [%s] peak=%.8f score=%.6f" % (label, sc, score))
        return sc

    # ============ Phase 1: L-BFGS on LSE (main workhorse) ============
    print("=== Phase 1: L-BFGS on LSE ===")
    t1 = time.monotonic()
    phase1_budget = min(600.0, (deadline - t1) * 0.30)
    a_curr = best_a.copy()

    # Multiple L-BFGS blocks with increasing beta
    n_blocks = 4
    block_time = phase1_budget / n_blocks
    for bi in range(n_blocks):
        if time.monotonic() > deadline - 300:
            break
        beta_init = 5.0 * (1 + bi * 0.5)
        beta_max = 50.0 * (1 + bi)
        a_new, sc_new = lbfgs_lse_block(a_curr, S, N, block_time, rng,
                                         beta_init=beta_init, beta_max=beta_max,
                                         m_lbfgs=12, use_high_beta=(bi >= 2))
        if sc_new < global_best_cmax:
            global_best_cmax = sc_new
            global_best_a = a_new.copy()
            a_curr = a_new.copy()
            print("  [P1-%d] score=%.6f" % (bi, 2.0 * n * sc_new / (S * S)))
        else:
            a_curr = a_new.copy()

    # ============ Phase 2: Coordinate Descent Polishing ============
    print("=== Phase 2: Coordinate Descent ===")
    t2 = time.monotonic()
    phase2_budget = min(500.0, (deadline - t2) * 0.25)
    a_cd, sc_cd = coord_descent_block(global_best_a, S, N, phase2_budget, rng, top_k=3)
    update_global(a_cd, "coord_descent")

    # ============ Phase 3: Spectral Shaping + L-BFGS ============
    print("=== Phase 3: Spectral Shaping + L-BFGS ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.20)
    n_shape_iters = 3
    shape_budget = phase3_budget / (n_shape_iters * 2)

    for si in range(n_shape_iters):
        if time.monotonic() > deadline - 150:
            break
        # Spectral shaping
        a_shaped = spectral_shaping(global_best_a, S, N, rng,
                                     strength=0.2 + 0.1 * si)
        # Small L-BFGS refinement
        a_ref, sc_ref = lbfgs_lse_block(a_shaped, S, N, shape_budget, rng,
                                         beta_init=10.0, beta_max=100.0,
                                         m_lbfgs=10)
        update_global(a_ref, "shape_%d" % si)

        # Also try phase shake
        if time.monotonic() < deadline - 150:
            a_shake = spectral_phase_shake(global_best_a, S, N, rng, frac=0.3)
            a_shake_ref, sc_shake = lbfgs_lse_block(a_shake, S, N, shape_budget * 0.5,
                                                     rng, beta_init=10.0,
                                                     beta_max=80.0, m_lbfgs=10)
            update_global(a_shake_ref, "shake_%d" % si)

    # ============ Phase 4: Annealed Max Descent ============
    print("=== Phase 4: Annealed Max Descent ===")
    t4 = time.monotonic()
    remaining4 = min(300.0, deadline - t4 - 30)
    if remaining4 > 30:
        a_anneal, sc_anneal = annealed_max_descent(global_best_a, S, N,
                                                    min(remaining4, 250.0), rng,
                                                    T_init=0.3, T_min=0.005)
        update_global(a_anneal, "anneal")

    # ============ Phase 5: Final High-Beta L-BFGS ============
    print("=== Phase 5: Final High-Beta L-BFGS ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 10
    if remaining5 > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_best_a, S, N, remaining5, rng,
                                         beta_init=20.0, beta_max=500.0,
                                         m_lbfgs=15)
        update_global(a_fin, "final_LBFGS")

        # One more coordinate descent pass
        if deadline - time.monotonic() > 30:
            a_cd2, sc_cd2 = coord_descent_block(global_best_a, S, N,
                                                 min(deadline - time.monotonic() - 10, 100.0),
                                                 rng, top_k=2)
            update_global(a_cd2, "final_cd")

    # ============ Phase 6: Multi-restart exploration ============
    remaining6 = deadline - time.monotonic() - 10
    if remaining6 > 60:
        print("=== Phase 6: Multi-restart ===")
        inits = [
            make_init_rand(n, rng),
            make_init_chirp(n, rng),
            make_init_flat_spec(n, N, rng),
            make_init_binary(n, rng),
        ]
        # Also include perturbed versions of current best
        for _ in range(2):
            perm = rng.permutation(n)
            inits.append(global_best_a[perm].copy())

        per_init = remaining6 / len(inits)
        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 20:
                break
            a_norm = project_simplex(init, S)
            a_try, sc_try = lbfgs_lse_block(a_norm, S, N, min(per_init, 60.0),
                                             rng, beta_init=8.0, beta_max=150.0,
                                             m_lbfgs=10)
            if sc_try < global_best_cmax:
                update_global(a_try, "restart_%d" % ii)

    # ============ Final verification and output ============
    print("=== Final Verification ===")
    if global_best_a is not None:
        global_best_a = np.clip(global_best_a, 0.0, 1000.0)
        if global_best_a.sum() < 0.01:
            global_best_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("Verification failed, using fallback")
            global_best_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in global_best_a]

    final_score = 2.0 * len(global_best_a) * global_best_cmax / (S * S)
    print("Returning length %d, best peak=%.8f, approx score=%.6f" %
          (len(global_best_a), global_best_cmax, final_score))

    return [float(x) for x in global_best_a]
