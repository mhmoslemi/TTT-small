def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):

def peak_shaving_direct(a, n, S, n_steps=200, top_k=5):

            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib[js] += a[js] * a[partners]
        
        # Reduce high-contribution coordinates
        contrib_max = contrib.max()
        if contrib_max < 1e-15:

        if c_new.max() >= cm:

            a = project_simplex(a / np.exp(-lr * contrib / contrib_max), S)

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def grad_subgradient(a, lags, n):
    """
    Exact gradient of sum_{k in lags} (f*f)[k] with respect to f.
    Each (f*f)[k] has gradient 2*f[k-j] at position j (where valid).
    Result: grad[j] = 2 * sum_{k in lags, valid} f[k-j]
    """
    w = np.zeros(2 * n - 1)
    for k in lags:
        w[k] = 1.0
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    h = np.real(ifft(W * np.conj(A)))
    return 2.0 * h[:n]


def grad_soft(a, c, n, beta_soft):
    """
    Gradient with soft weights: w[k] proportional to exp(beta*(c[k]-max)).
    """
    cm = c.max()
    shifted = beta_soft * (c - cm)
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    h = np.real(ifft(W * np.conj(A)))
    return 2.0 * h[:n]


def grad_lse(a, c, n, beta_lse):
    """
    Gradient of LSE(beta, c) with respect to f.
    Returns (gradient, lse_value).
    """
    cm = c.max()
    shifted = beta_lse * (c - cm)
    shifted = np.clip(shifted, -50.0, 50.0)
    e = np.exp(shifted)
    tot = e.sum()
    w = e / tot
    lse_val = cm + np.log(tot) / beta_lse
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    h = np.real(ifft(W * np.conj(A)))
    return 2.0 * h[:n], lse_val

def peak_shave_reversible(a, n, S, n_steps=300, top_k=5, lr=0.02):
    """
    Peak shaving with reversibility: only keep a step if it reduces the max.
    """
    a = a.copy()
    best_c = fft_convolve_self(a).max()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        top_lags = np.argsort(c)[::-1][:top_k]
        contrib = np.zeros(n)
        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib[js] += a[js] * a[partners]

        cmax = contrib.max()
        if cmax < 1e-15:
            break

        # Multiplicative reduction proportional to contribution
        reduction = lr * contrib / cmax
        a_new = a * np.exp(-reduction)
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cm - 1e-12:
            a = a_new
            best_c = c_new.max()
        else:
            # Try smaller lr
            a_new2 = a * np.exp(-reduction * 0.3)
            a_new2 = project_simplex(a_new2, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cm - 1e-12:
                a = a_new2
                best_c = c_new2.max()
            else:
                break

        for na in [3, 5, 8, 12]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t2 = min(t2_total / 4.0, remaining * 0.9)
            print("  Active set n_active=%d, time=%.0fs" % (na, t2))
            a2, v2 = run_active_subgrad(global_a, n2, S2, t2, rng,
                                         n_active=na, use_backtrack=True)
            try_seq(a2)

    # === Phase 3: Soft subgradient (10% time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n3 = len(global_a)
            S3 = np.sqrt(2.0 * n3)
            t3 = min(240.0, remaining * 0.10)
            print("=== Phase 3: Soft subgradient (n=%d, %.0fs) ===" % (n3, t3))
            for bs in [50.0, 100.0, 200.0]:
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                t3i = min(80.0, remaining * 0.5)
                a3, v3 = run_soft_subgrad(global_a, n3, S3, t3i, rng, beta_soft=bs)
                try_seq(a3)

    # === Phase 4: Peak shaving (10% time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n4 = len(global_a)
            S4 = np.sqrt(2.0 * n4)
            print("=== Phase 4: Peak shaving ===")
            for tk in [3, 5, 8]:
                for lr in [0.01, 0.03, 0.05]:
                    a4 = peak_shave_reversible(global_a, n4, S4,
                                                n_steps=200, top_k=tk, lr=lr)
                    try_seq(a4)

    # === Phase 5: High-beta LSE polish (10% time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n5 = len(global_a)
            S5 = np.sqrt(2.0 * n5)
            t5 = min(240.0, remaining * 0.10)
            print("=== Phase 5: High-beta polish (n=%d, %.0fs) ===" % (n5, t5))
            a5, v5 = run_short_lse(global_a, n5, S5, t5, rng)
            try_seq(a5)

    # === Phase 6: Active-set after all previous (10% time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n6 = len(global_a)
            S6 = np.sqrt(2.0 * n6)
            t6 = min(240.0, remaining * 0.10)
            print("=== Phase 6: Final active-set (n=%d, %.0fs) ===" % (n6, t6))
            a6, v6 = run_active_subgrad(global_a, n6, S6, t6, rng,
                                         n_active=5, use_backtrack=True, noise_period=100)
            try_seq(a6)

    # === Phase 7: Try longer sequence (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 180 and global_a is not None:
        n_base = len(global_a)
        n7 = n_base * 2
        if n7 <= 32768:
            S7 = np.sqrt(2.0 * n7)
            t_old = np.linspace(0, 1, n_base)
            t_new = np.linspace(0, 1, n7)
            init7 = np.interp(t_new, t_old, global_a)
            init7 = normalize_sum(init7, n7)
            t7 = min(360.0, remaining * 0.5)
            print("=== Phase 7: Longer seq (n=%d, %.0fs) ===" % (n7, t7))
            a7, v7 = run_lse_fista(init7, n7, S7, t7, rng,
                                    beta_start=100.0, beta_end=50000.0, n_lse_steps=200)
            try_seq(a7)

            remaining = deadline - time.monotonic()
            if remaining > 60 and global_a is not None:
                n7b = n7 if len(global_a) == n7 else len(global_a)
                S7b = np.sqrt(2.0 * n7b)
                t7b = min(120.0, remaining * 0.5)
                a7b, v7b = run_active_subgrad(global_a, n7b, S7b, t7b, rng,
                                               n_active=5, use_backtrack=True)
                try_seq(a7b)

    # === Final cleanup ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        nF = len(global_a)
        SF = np.sqrt(2.0 * nF)
        aF, vF = run_active_subgrad(global_a, nF, SF, remaining * 0.6, rng,
                                     n_active=3, use_backtrack=True)
        try_seq(aF)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %s, internal = %.8f (n=%d)" %
              (("%0.8f" % ver) if np.isfinite(ver) else "INVALID",
               global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  Invalid! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best=%.8f, total_time=%.0fs" %
          (len(global_a), global_best, elapsed()))
    return [float(x) for x in global_a]
