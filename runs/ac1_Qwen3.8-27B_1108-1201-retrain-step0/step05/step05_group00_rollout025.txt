import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def gradient_conv(a, weights, N):
    """
    Gradient of weighted sum of convolution values.
    weights: array of length 2n-1, gives weight to each conv value.
    Uses the closed form: d(conv)_k/da_m = 2*a_{k-m} (in bounds).
    Equivalent to: grad = 2 * ifft(FFT(weights) * conj(FFT(a)))[:n]
    """
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def lse_weights(c, beta):
    """Compute LSE weights for the convolution values."""
    c_max = c.max()
    if beta < 1e-6:
        return np.ones(len(c)) / len(c), c_max
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w, c_max
    return exp_bc / total, c_max


def lse_value(c, beta):
    """Compute LSE value."""
    c_max = c.max()
    if beta < 1e-6:
        return c_max
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max
    return c_max + np.log(total) / beta


def toppk_weights(c, K):
    """Top-K weights: uniform over top K peaks."""
    K = min(K, len(c))
    top_idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / K
    return w


def coord_descent_polish(a, n, S, N, max_iters, rng):

    coordinate that most contributes to it, and transfer mass to a coordinate
    that contributes least. Uses 1D line search.
    """

    best_a = a.copy()
    best_max = compute_conv(a, N).max()
    
    for it in range(max_iters):
        c = compute_conv(a, N)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        
        # Find peak position
        p = int(np.argmax(c))
        
        # Gradient of conv[p] w.r.t. a_i: 2*a[p-i] (if in bounds)
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        grad_at_peak = np.zeros(n)
        for i in range(lo, hi + 1):
            j = p - i
            if 0 <= j < n:
                if i == j:
                    grad_at_peak[i] += 2.0 * a[j]
                else:
                    grad_at_peak[i] += a[j]
        
        if np.max(grad_at_peak) < 1e-15:
            break
        
        # Find coordinate to decrease (highest contribution to peak)
        i_dec = int(np.argmax(grad_at_peak))
        a_i_dec = a[i_dec]
        if a_i_dec < 1e-15:
            break
        
        # Find coordinate to increase (lowest contribution to peak, or zero)
        # Prefer coordinates with a[i] close to 0 (they won't create new peaks)
        # and with low grad_at_peak
        # Candidate: the coordinate with minimum grad_at_peak among those with small a
        # Actually, we want to increase a_j where grad_at_peak[j] is small
        # (so it doesn't increase the current peak much)

        # Approximate: pick j with smallest grad_at_peak[j]
        # But also consider: increasing a_j increases conv[j+q] for all q
        # A safe choice: j where a[j] is small (so the "footprint" is small)

                d[k] -= 1.0

            if 0 <= k1 < 2 * n - 1:

            if 0 <= k2 < 2 * n - 1:
                d2[k2] -= a[q]
        
        # We want to find delta in [0, a_i_dec] that minimizes max(c + delta * d2)
        # Binary search / ternary search on delta

        delta_hi = a_i_dec
        
        # Try a few values

def nesterov_toppk(a_init, n, S, N, max_time, rng, K_init=20, K_max=512):
    """Nesterov-accelerated top-K subgradient method."""
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_max = compute_conv(a, N).max()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        K = min(K_max, K_init + it // 30)
        c = compute_conv(y, N)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = y.copy()
        
        w = toppk_weights(c, K)
        grad = gradient_conv(y, w, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 200 == 0 and rng.random() < 0.1:
            # Small perturbation to escape
            y += 0.01 * S / n * rng.standard_normal(n)
            y = np.maximum(y, 0.0)
            s = y.sum()
            if s > 1e-15:
                y = y * (S / s)
            t = 1.0
    
    return best_a, best_max

                if cm_ls <= c_max:
                    break
            alpha *= 0.5
        
        if cmax_new < best_max:
            best_max = cmax_new
            best_a = a_new.copy()
        
        # Update L-BFGS history
        w_new, _ = lse_weights(compute_conv(a_new, N), beta)
        g_new = gradient_conv(a_new, w_new, N)
        c = compute_conv(a_new, N)
        c_max = c.max()
        
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        
        # Occasional partial restart
        if it % 500 == 0 and rng.random() < 0.2:
            _partial_spectral_perturb(a, N, rng, frac=0.2)
            s_hist = []
            y_hist = []
            c = compute_conv(a, N)
            c_max = c.max()
            w, _ = lse_weights(c, beta)
            g = gradient_conv(a, w, N)
    
    return best_a, best_max


def _lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def _partial_spectral_perturb(a, N, rng, frac=0.3):
    """Perturb a fraction of FFT phases in-place."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return
    a[:] = a_new * (S / s)


def _full_spectral_restart(a, N, rng):
    """Full random phase restart, return new a."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def deflation_init(n, S, N, n_iters=3000, rng=None):
    """Constructive initialization: iteratively reduce peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    contrib[i] += 2.0 * a[j]
                else:
                    contrib[i] += a[j]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = 0.02 * a[i_max]
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (7 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.5 * np.cos(phase) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_binary_opt(n, rng):
    """Optimized binary-like: random threshold with smoothing."""
    a = (rng.random(n) > 0.35).astype(np.float64)
    # light smoothing
    kernel = np.array([0.2, 0.3, 0.3, 0.2])
    a = np.convolve(a, kernel, mode='same')
    return np.clip(a, 0.0, None)


def make_init_fourier_guess(n, rng, N):
    """Initialize from random Fourier magnitudes with structured phases."""
    A = np.zeros(N, dtype=complex)
    # DC component
    A[0] = float(n) * 0.5
    # Other components with random magnitudes
    mag = rng.exponential(1.0, N)
    mag[0] = 0
    # Chirp-like phase
    idx = np.arange(N)
    center = N // 2
    phase = -np.pi * (idx - center) ** 2 / N * 0.5 + rng.uniform(-0.5, 0.5, N)
    A = mag * np.exp(1j * phase)
    A[0] = float(n)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def optimize_single_n(n, S, N, budget, rng, start_a=None):
    """Run the full optimization pipeline for a given n."""
    global_best = np.inf
    global_a = None
    
    def try_update(a_candidate, label=""):
        nonlocal global_best, global_a
        a_c = project_simplex(a_candidate, S)
        c = compute_conv(a_c, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_c.copy()
            score = 2.0 * n * sc / (S * S)
            print("    [%s] peak=%.6f score=%.6f" % (label, sc, score))
    
    t0 = time.monotonic()
    
    # Build initializations
    inits = []
    if start_a is not None and len(start_a) == n:
        inits.append(("start", np.clip(start_a, 0, None)))
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("binary", make_init_binary_opt(n, rng)))
    inits.append(("deflation", deflation_init(n, S, N, 2000, rng)))
    inits.append(("fourier", make_init_fourier_guess(n, rng, N)))
    
    n_inits = len(inits)
    per_init = budget / (n_inits * 3)  # rough split
    
    # Phase A: Short L-BFGS on each init
    for name, arr in inits:
        if time.monotonic() - t0 > budget * 0.7:
            break
        a_norm = project_simplex(arr, S)
        rem = max(0.5, budget * 0.08 - (time.monotonic() - t0))
        a_l, sc_l = lbfgs_lse(a_norm, n, S, N, min(rem, 60), rng,
                              beta_init=5.0, beta_max=500, m_lbfgs=8)
        try_update(a_l, name)
    
    # Phase B: Focused optimization from best
    if global_a is not None:
        t_b = time.monotonic()
        rem_b = min(budget * 0.25, budget - (t_b - t0))
        if rem_b > 5:
            a_b, sc_b = lbfgs_lse(global_a, n, S, N, rem_b, rng,
                                  beta_init=20.0, beta_max=3000, m_lbfgs=14)
            try_update(a_b, "focus")
    
    # Phase C: Nesterov top-K
    if global_a is not None:
        t_c = time.monotonic()
        rem_c = min(budget * 0.15, budget - (t_c - t0))
        if rem_c > 5:
            a_ng, sc_ng = nesterov_toppk(global_a, n, S, N, rem_c, rng, K_init=30, K_max=600)
            try_update(a_ng, "nesterov")
    
    # Phase D: Coordinate descent polish
    if global_a is not None:
        t_d = time.monotonic()
        rem_d = min(budget * 0.1, budget - (t_d - t0))
        if rem_d > 2:
            n_cd_iters = int(rem_d * 50)  # roughly 50 CD steps per second
            n_cd_iters = max(100, min(n_cd_iters, 5000))
            a_cd, sc_cd = coord_descent_polish(global_a, n, S, N, n_cd_iters, rng)
            try_update(a_cd, "coord_descent")
    
    # Phase E: Final high-beta L-BFGS
    if global_a is not None:
        t_e = time.monotonic()
        rem_e = max(0, budget - (t_e - t0))
        if rem_e > 5:
            a_fin, sc_fin = lbfgs_lse(global_a, n, S, N, min(rem_e, 120), rng,
                                      beta_init=100.0, beta_max=5000, m_lbfgs=16)
            try_update(a_fin, "final_high_beta")
    
    # Phase F: Another coord descent + L-BFGS
    if global_a is not None:
        t_f = time.monotonic()
        rem_f = max(0, budget - (t_f - t0) - 5)
        if rem_f > 5:
            n_cd2 = max(200, int(rem_f * 0.5 * 30))
            a_cd2, sc_cd2 = coord_descent_polish(global_a, n, S, N, n_cd2, rng)
            try_update(a_cd2, "cd2")
            if time.monotonic() < t0 + budget - 10:
                rem_f2 = max(0, budget - (time.monotonic() - t0) - 5)
                if rem_f2 > 5:
                    a_f2, sc_f2 = lbfgs_lse(global_a, n, S, N, min(rem_f2, 80), rng,
                                            beta_init=200.0, beta_max=5000, m_lbfgs=16)
                    try_update(a_f2, "final2")
    
    return global_a, global_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-n sweep with L-BFGS, Nesterov, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    
    # Try to use previous solution
    start_a_8192 = None
    start_a_4096 = None
    start_a_12288 = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == 8192:
            start_a_8192 = np.clip(given, 0, 1000)
        elif given.size == 4096:
            start_a_4096 = np.clip(given, 0, 1000)
        elif given.size == 12288:
            start_a_12288 = np.clip(given, 0, 1000)
        else:
            # Resample to desired sizes
            if given.size >= 4096:
                n_g = len(given)
                for target_n in [4096, 8192, 12288]:
                    if target_n <= n_g:
                        idx = np.linspace(0, n_g - 1, target_n).astype(int)
                        if target_n == 8192:
                            start_a_8192 = given[idx]
                        elif target_n == 4096:
                            start_a_4096 = given[idx]
                        elif target_n == 12288:
                            start_a_12288 = given[idx]
    except Exception:
        pass
    
    global_best_overall = np.inf
    global_a_overall = None
    
    # Sequence lengths to try (primary: 8192, alternatives for diversity)
    # Score = 2*n*max_b/sum_a^2; with sum_a=sqrt(2n), score = max_b
    # So we're minimizing max_b directly (when normalized)
    
    n_candidates = [8192, 4096, 12288]
    
    # Allocate time: give most time to the primary (8192 or whatever has a start)
    if start_a_8192 is not None:
        time_alloc = {8192: 0.55, 4096: 0.2, 12288: 0.2}
    elif start_a_4096 is not None:
        time_alloc = {4096: 0.55, 8192: 0.25, 12288: 0.2}
    else:
        time_alloc = {8192: 0.5, 4096: 0.25, 12288: 0.25}
    
    total_time = budget_s - 30  # reserve 30s for overhead
    print("Total budget: %.0fs, allocating: %s" % (budget_s, 
          {k: "%.0fs" % (v * total_time) for k, v in time_alloc.items()}))
    
    for n in n_candidates:
        if n not in time_alloc:
            continue
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        alloc_time = time_alloc[n] * total_time
        
        start_a = None
        if n == 8192:
            start_a = start_a_8192
        elif n == 4096:
            start_a = start_a_4096
        elif n == 12288:
            start_a = start_a_12288
        
        print("\n=== Optimizing n=%d (S=%.2f, N=%d, budget=%.0fs) ===" % (n, S, N, alloc_time))
        
        a_best, peak_best = optimize_single_n(n, S, N, alloc_time, rng, start_a)
        
        if a_best is not None:
            score = 2.0 * n * peak_best / (S * S)
            if score < global_best_overall:
                global_best_overall = score
                global_a_overall = a_best.copy()
                print("  New global best: score=%.6f (n=%d)" % (score, n))
    
    # Final verification and fallback
    if global_a_overall is not None:
        global_a_overall = np.clip(global_a_overall, 0.0, 1000.0)
        if global_a_overall.sum() < 0.01:
            global_a_overall = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a_overall]
        ver = evaluate_sequence(final_list)
        print("\nVerification score: %.6f" % ver if np.isfinite(ver) else "\nVerification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a_overall = np.ones(256, dtype=np.float64)
    else:
        global_a_overall = np.ones(256, dtype=np.float64)
    
    n_out = len(global_a_overall)
    S_out = np.sqrt(2.0 * float(n_out))
    c_out = compute_conv(global_a_overall, 1 << (2 * n_out - 1).bit_length())
    print("Returning length %d, peak=%.6f, approx score=%.6f" % 
          (n_out, c_out.max(), 2.0 * n_out * c_out.max() / (S_out * S_out)))
    
    return [float(x) for x in global_a_overall]
