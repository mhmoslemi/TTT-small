        else:
            g = exact_subgradient(a, c)

        i_star = np.argmin(g)
        j_star = np.argmax(g)

            break
        
        # Compute transfer direction
        # r_i[k] = a[k - i] for valid k
        r_i = np.zeros(2 * n - 1)
        r_j = np.zeros(2 * n - 1)

        valid_i = np.arange(max(0, i_star), min(2 * n - 1, i_star + n))
        r_i[valid_i] = a[valid_i - i_star]
        
        valid_j = np.arange(max(0, j_star), min(2 * n - 1, j_star + n))
        r_j[valid_j] = a[valid_j - j_star]

        delta_min = -a[i_star]
        delta_max = a[j_star]

        hi = delta_max

            elif subgrad > 0:

        a[j_star] -= delta

            beta = min(beta_max, beta * 1.08)
        else:
            beta = min(beta_max, beta * 1.2)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking safety
        step = 2.0 / max(beta + 1.0, 1.0)
        step = min(step, 0.5 * S / (gnorm + 1e-10))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with perturbations when stuck
        if stuck > 3000:
            t_mom = 1.0
            # Try Fourier phase perturbation
            fsize = 1 << (n - 1).bit_length()
            A = fft(a, fsize)
            mags = np.abs(A)
            phases = np.angle(A) + rng.normal(0, 0.1, fsize)
            # Make it symmetric (real signal)
            phases[0] = 0
            if fsize % 2 == 0:
                phases[fsize // 2] = 0
            phases_new = 0.5 * (phases[:fsize // 2 + 1] + np.conj(phases[:fsize // 2 + 1][::-1]))
            A_new = mags * np.exp(1j * np.angle(np.exp(1j * phases)))
            # Ensure Hermitian symmetry
            A_new[:n] = np.real(ifft(np.conj(fft(np.conj(A_new[:n])))))
            a_pert = np.real(ifft(A_new))[:n]
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() > 1e-10:
                a_pert = normalize_seq(a_pert, S)
                y = a_pert.copy()
            else:
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                y = project_simplex(a + noise, S)
            last_improve = it
            stuck = 0

        if t_mom > 1000:
            t_mom = 1.0

    return best_a, best_conv


def run_pairwise_descent(a_init, max_time, rng, beta_start=50.0):
    """
    New: Pairwise coordinate descent.
    Transfers mass from high-gradient to low-gradient coordinates.
    Uses 1D convex line search for exact transfer amount.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()

        # Compute gradient (LSE with moderate beta for smooth pair selection)
        grad = lse_gradient(a, c, beta)

        # Find best pair: most negative gradient (increase) and most positive (decrease)
        i_star = int(np.argmin(grad))
        j_star = int(np.argmax(grad))

        grad_diff = grad[j_star] - grad[i_star]
        if grad_diff < 1e-12:
            # Try with exact subgradient
            grad = smooth_subgradient(a, c, tol_frac=0.02)
            i_star = int(np.argmin(grad))
            j_star = int(np.argmax(grad))
            grad_diff = grad[j_star] - grad[i_star]
            if grad_diff < 1e-12:
                break

        # Compute transfer direction in convolution space
        # d[k] = a[k - i_star] - a[k - j_star] (padded)
        d = np.zeros(2 * n - 1)
        # Add contribution from i_star: a[k - i_star] for valid k
        lo_i = max(0, i_star)
        hi_i = min(2 * n - 1, i_star + n)
        d[lo_i:hi_i] += a[lo_i - i_star:hi_i - i_star]
        # Subtract contribution from j_star
        lo_j = max(0, j_star)
        hi_j = min(2 * n - 1, j_star + n)
        d[lo_j:hi_j] -= a[lo_j - j_star:hi_j - j_star]

        # 1D convex minimization: minimize max(c + 2*delta*d) over delta in [-a[i_star], a[j_star]]
        delta_lo = -a[i_star]
        delta_hi = a[j_star]

        # Quick check: if all d <= 0, go to delta_hi; if all d >= 0, go to delta_lo
        if d.max() <= 1e-15:
            delta = delta_hi
        elif d.min() >= -1e-15:
            delta = delta_lo
        else:
            # Bisection on subgradient (12 iterations)
            lo = delta_lo
            hi = delta_hi
            for _ in range(12):
                mid = 0.5 * (lo + hi)
                vals = c + 2.0 * mid * d
                max_val = vals.max()
                # Subgradient: mean of d over near-active set
                tol = max_val * 0.001
                active_mask = vals >= max_val - tol
                if active_mask.sum() == 0:
                    active_mask = vals >= max_val
                subgrad = d[active_mask].mean()
                if subgrad < -1e-15:
                    lo = mid
                elif subgrad > 1e-15:
                    hi = mid
                else:
                    lo = hi = mid
                    break
            delta = 0.5 * (lo + hi)
            delta = np.clip(delta, delta_lo, delta_hi)

        if abs(delta) < 1e-14:
            # Try a random pair instead
            candidates = np.argsort(grad)[:max(1, n // 100)]
            anti_candidates = np.argsort(grad)[::-1][:max(1, n // 100)]
            i_star = int(candidates[rng.integers(len(candidates))])
            j_star = int(anti_candidates[rng.integers(len(anti_candidates))])
            if i_star == j_star:
                beta = min(1e5, beta * 1.01)
                continue
            # Simple step for random pair
            delta = 0.01 * a[j_star] * np.sign(grad[i_star] - grad[j_star])
            delta = np.clip(delta, -a[i_star], a[j_star])

        # Apply update
        a[i_star] += delta
        a[j_star] -= delta
        a = np.maximum(a, 0)

        # Occasionally renormalize (should be approximately preserving sum)
        if it % 100 == 0:
            s = a.sum()
            if abs(s - S) / S > 0.01:
                a = normalize_seq(a, S)

        # Gradually increase beta for sharper gradient
        beta = min(1e5, beta * (1.0 + 0.002))

        # Periodic check for restart
        if it % 5000 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_conv


def run_smooth_subgrad_fista(a_init, max_time, rng, tol_frac=0.05):
    """FISTA with smooth (tolerance-based) subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance: tighter when stuck (more focused)
        tol = max(0.001, tol_frac * np.exp(-stuck / 5000.0))
        grad = smooth_subgradient(y, c, tol_frac=tol)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = min(2.0 * S / (gnorm + 1e-10), 0.5)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000:
            t_mom = 1.0
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            last_improve = it
            stuck = 0

        if t_mom > 500:
            t_mom = 1.0

    return best_a, best_conv


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_chirp_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / n)


def make_random_init(n, rng):
    a = rng.random(n)
    a = a ** 2
    return a


def make_bandlimited_init(n, rng, K=3):
    """Create a band-limited initial sequence."""
    tt = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, K + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4) / k
        a += amp * np.cos(2 * np.pi * k * tt / n + phase)
    return np.maximum(a, 0.0)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: power2 warmup -> LSE FISTA -> pairwise descent -> smooth subgrad -> final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get the provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Determine base n
    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)

    print("Base n = %d" % n0)

    # Generate diverse initializations
    n = n0
    S = np.sqrt(2.0 * n)
    inits = []
    names = []

    if a_start is not None:
        inits.append(normalize_seq(a_start, S))
        names.append("provided")

    inits.append(normalize_seq(make_cosine_init(n), S))
    names.append("cosine")

    inits.append(normalize_seq(make_chirp_init(n), S))
    names.append("chirp")

    inits.append(normalize_seq(make_random_init(n, rng), S))
    names.append("random1")

    inits.append(normalize_seq(make_bandlimited_init(n, rng, K=3), S))
    names.append("bandlimited")

    inits.append(normalize_seq(make_random_init(n, rng), S))
    names.append("random2")

    # Quick evaluation
    print("\n=== Initialization evaluation ===")
    best_init = None
    best_init_obj = np.inf
    for a_init, name in zip(inits, names):
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.6f" % (name, obj))

    # === Phase 1: Power-2 FISTA warmup ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.08
    if p1_budget > 5:
        print("\n=== Phase 1: Power-2 FISTA (budget %.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a_p2 = run_power2_fista(best_init, p1_budget, rng)
        obj_p2 = compute_obj(a_p2)
        print("  Power2: obj=%.6f (t=%.1fs)" % (obj_p2, time.monotonic() - t1))
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()
        best_init = a_p2

    # === Phase 2: LSE FISTA (main optimization) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.35
    if p2_budget > 10:
        print("\n=== Phase 2: LSE FISTA (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a_lse1, _ = run_lse_fista(best_init, p2_budget * 0.45, rng,
                                   beta_init=5.0, beta_max=5000.0)
        obj_lse1 = compute_obj(a_lse1)
        print("  LSE-low: obj=%.6f (t=%.1fs)" % (obj_lse1, time.monotonic() - t2))
        if obj_lse1 < global_best_obj:
            global_best_obj = obj_lse1
            global_best_a = a_lse1.copy()

        rem2 = p2_budget - (time.monotonic() - t2)
        if rem2 > 10:
            a_lse2, _ = run_lse_fista(global_best_a, rem2, rng,
                                       beta_init=50.0, beta_max=50000.0)
            obj_lse2 = compute_obj(a_lse2)
            print("  LSE-high: obj=%.6f (t=%.1fs)" % (obj_lse2, time.monotonic() - t2))
            if obj_lse2 < global_best_obj:
                global_best_obj = obj_lse2
                global_best_a = a_lse2.copy()

    # === Phase 3: Pairwise coordinate descent (NEW) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    if p3_budget > 30:
        print("\n=== Phase 3: Pairwise descent (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a_pw1, _ = run_pairwise_descent(global_best_a, p3_budget * 0.6, rng,
                                         beta_start=20.0)
        obj_pw1 = compute_obj(a_pw1)
        print("  Pairwise1: obj=%.6f (t=%.1fs)" % (obj_pw1, time.monotonic() - t3))
        if obj_pw1 < global_best_obj:
            global_best_obj = obj_pw1
            global_best_a = a_pw1.copy()

        rem3 = p3_budget - (time.monotonic() - t3)
        if rem3 > 10:
            a_pw2, _ = run_pairwise_descent(global_best_a, rem3, rng,
                                             beta_start=200.0)
            obj_pw2 = compute_obj(a_pw2)
            print("  Pairwise2: obj=%.6f (t=%.1fs)" % (obj_pw2, time.monotonic() - t3))
            if obj_pw2 < global_best_obj:
                global_best_obj = obj_pw2
                global_best_a = a_pw2.copy()

    # === Phase 4: Smooth subgradient FISTA ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15
    if p4_budget > 10:
        print("\n=== Phase 4: Smooth subgrad FISTA (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a_sg, _ = run_smooth_subgrad_fista(global_best_a, p4_budget, rng,
                                            tol_frac=0.03)
        obj_sg = compute_obj(a_sg)
        print("  SmoothSubgrad: obj=%.6f (t=%.1fs)" % (obj_sg, time.monotonic() - t4))
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()

    # === Phase 5: Final intensive LSE + pairwise polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p5_budget = remaining * 0.60
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # High-beta LSE
        a_fin1, _ = run_lse_fista(global_best_a, p5_budget * 0.4, rng,
                                   beta_init=500.0, beta_max=100000.0)
        obj_fin1 = compute_obj(a_fin1)
        print("  Final LSE: obj=%.6f (t=%.1fs)" % (obj_fin1, time.monotonic() - t5))
        if obj_fin1 < global_best_obj:
            global_best_obj = obj_fin1
            global_best_a = a_fin1.copy()

        # Final pairwise
        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 10:
            a_fin2, _ = run_pairwise_descent(global_best_a, rem5, rng,
                                              beta_start=500.0)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final PW: obj=%.6f (t=%.1fs)" % (obj_fin2, time.monotonic() - t5))
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # === Phase 6: Multi-scale (upsample to 2n) if beneficial ===
    remaining = deadline - time.monotonic()
    if remaining > 180 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p6_budget = remaining * 0.75
        print("\n=== Phase 6: Upsample to n=%d (budget %.0fs) ===" % (n2, p6_budget))
        t6 = time.monotonic()

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)

        a_up1, _ = run_lse_fista(a_up, p6_budget * 0.4, rng,
                                  beta_init=5.0, beta_max=5000.0)
        obj_up1 = compute_obj(a_up1)
        print("  Upsampled LSE: obj=%.6f (t=%.1fs)" % (obj_up1, time.monotonic() - t6))
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 30:
            a_up2, _ = run_pairwise_descent(global_best_a, rem6, rng,
                                             beta_start=50.0)
            obj_up2 = compute_obj(a_up2)
            print("  Upsampled PW: obj=%.6f (t=%.1fs)" % (obj_up2, time.monotonic() - t6))
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
