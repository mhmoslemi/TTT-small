```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Mirror descent (multiplicative weights) with LSE smoothing,
active-set KKT-based direction, and equalization heuristics.
Key insight: Use the natural geometry of the non-negative simplex via
multiplicative updates rather than Euclidean projection. Combined with
aggressive active-set handling and multi-scale optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, S):
    """Rescale to sum = S, enforcing non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def mirror_descent_step(a, grad, eta, S):
    """Multiplicative update: a[j] *= exp(-eta * grad[j]), then normalize."""
    # For numerical stability, shift grad
    g = grad - grad.mean()
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 0.0)
    return normalize_sum(a_new, S)


def active_set_direction(a, c, n_active=20, beta_act=100.0):
    """
    Compute a direction based on active set KKT conditions.
    Find weights on active lags that make the combined gradient as flat as possible.
    """
    n = len(a)
    cm = c.max()
    
    # Select active set: lags within threshold of max
    threshold = cm - 10.0 / beta_act
    active_mask = c >= threshold
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c)])
    if len(active_idx) > n_active:
        # Take top n_active
        top = np.argsort(c[active_idx])[::-1][:n_active]
        active_idx = active_idx[top]
    
    # Compute gradient contributions from each active lag
    # grad_from_k[j] = dc[k]/da[j] = 2*a[k-j] (approximately, for interior)
    # Use FFT for efficiency
    m = len(active_idx)
    grads = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        # dc[k]/da[j] computed via FFT
        w = np.zeros(2 * n - 1)
        w[k] = 1.0
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grads[i] = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Find weights w (non-negative, sum=1) to minimize ||grads^T w||^2
    # subject to sum(w) = 1, w >= 0
    # This is a QP. Use a simple approach: iterate
    # Minimize (grads^T w)^T (grads^T w) = w^T (grads grads^T) w
    G = grads @ grads.T  # m x m
    # Solve: min w^T G w s.t. sum(w)=1, w>=0
    # Use projected gradient or simple water-filling
    
    # Simple approach: start with uniform, do a few projected gradient steps
    w = np.ones(m) / m
    for _ in range(100):
        gw = G @ w  # m
        w_new = w - 0.01 * gw
        w_new = np.maximum(w_new, 0)
        s = w_new.sum()
        if s < 1e-15:
            w_new = np.ones(m) / m
        else:
            w_new = w_new / s
        if np.allclose(w, w_new):
            break
        w = w_new
    
    # Combined gradient
    combined_grad = w @ grads  # shape (n,)
    return combined_grad, active_idx


def run_mirror_descent(a_init, n, S, max_time, rng,
                       eta_start=0.1, eta_end=0.001, 
                       beta_start=100.0, beta_end=50000.0,
                       beta_steps=200):
    """
    Mirror descent (multiplicative weights) with LSE smoothing.
    """
    a = normalize_sum(a_init.copy(), S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it
        
        # Adaptive beta
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # LSE gradient
        grad, _ = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        frac = min(1.0, it / max(1, beta_steps * 2))
        eta = eta_start * (eta_end / eta_start) ** frac
        
        # Mirror descent step
        a = mirror_descent_step(a, grad, eta, S)

def run_fista_refined(a_init, n, S, max_time, rng,

            x_new = a  # fallback

            x_new = x_new * (S / s)

            t_mom = 1.0

    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def equalization_pass(a, n, S, n_iters=50):
    """
    Direct equalization: try to make top convolution coefficients equal.
    For the top lag k*, reduce contributions by decreasing a[j]*a[k*-j] pairs.
    """
    a = a.copy()
    for _ in range(n_iters):
        c = fft_convolve_self(a)
        cm = c.max()
        
        # Find the top lag
        k_star = np.argmax(c)
        
        # Compute responsibility: for each j, how much does a[j] contribute to c[k_star]?
        # dc[k_star]/da[j] = 2*a[k_star-j] (for interior j)
        k = k_star
        resp = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        resp[js] = 2.0 * a[partners]
        
        # Reduce high-responsibility elements
        resp_max = resp.max()
        if resp_max < 1e-15:
            break
        
        # Multiplicative reduction
        lr = 0.05
        a[js] *= np.exp(-lr * resp / resp_max)
        a = normalize_sum(a, S)
    
    return a
```

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Combined mirror descent (multiplicative weights), FISTA with aggressive
LSE beta schedule, active-set KKT direction finding, and structural equalization.
Key innovations: (1) multiplicative updates exploit natural non-negativity geometry,
(2) adaptive active-set weighting via QP for sharper gradient directions,
(3) anti-concentration heuristic to prevent mass from clustering.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, S):
    """Rescale to sum = S, enforcing non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a. Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    max_shifted = shifted.max()
    e = np.exp(shifted - max_shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + (max_shifted + np.log(tot)) / beta
    return grad, lse_val


def mirror_descent_step(a, grad, eta, S):
    """Multiplicative update: a[j] *= exp(-eta * grad[j]), then normalize."""
    g = grad - grad.mean()
    # Clamp to avoid overflow
    g = np.clip(g, -50.0 / eta, 50.0 / eta) if eta > 0 else g
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-30)
    return normalize_sum(a_new, S)


def compute_active_gradient(a, c, n_active=20, gap=5.0):
    """
    Compute gradient direction from active set with optimal weighting.
    Uses a simple QP to find weights that minimize gradient variance.
    """
    n = len(a)
    cm = c.max()
    
    # Select active lags: those within 'gap' of max
    active_idx = np.where(c >= cm - gap)[0]
    if len(active_idx) > n_active:
        top = np.argsort(c[active_idx])[::-1][:n_active]
        active_idx = active_idx[top]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c)])
    
    m = len(active_idx)
    
    # Compute gradient for each active lag via FFT
    grads = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        w = np.zeros(2 * n - 1)
        w[k] = 1.0
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grads[i] = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    if m == 1:
        return grads[0], active_idx
    
    # Find weights w to minimize ||grads^T w||^2 subject to sum(w)=1, w>=0
    # Equivalent to: min w^T G w where G = grads @ grads.T
    G = grads @ grads.T
    np.fill_diagonal(G, G.diagonal() + 1e-8)  # ensure PSD
    
    # Solve QP via projected gradient with many iterations
    w = np.ones(m) / m
    step_w = 2.0 / (np.abs(G).max() + 1e-10)
    for _ in range(200):
        gw = G @ w
        w_new = w - step_w * gw
        w_new = np.maximum(w_new, 0.0)
        s = w_new.sum()
        if s < 1e-15:
            w_new = np.ones(m) / m
        else:
            w_new = w_new / s
        if np.linalg.norm(w_new - w) < 1e-12:
            break
        w = w_new
    
    combined_grad = w @ grads
    return combined_grad, active_idx


def run_mirror_descent(a_init, n, S, max_time, rng,
                       eta_start=0.05, eta_end=0.0005,
                       beta_start=100.0, beta_end=80000.0,
                       beta_steps=200, active_every=50):
    """Mirror descent with LSE smoothing and periodic active-set refinement."""
    a = normalize_sum(a_init.copy(), S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it
        
        # Adaptive beta (geometric increase)
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)
        
        # LSE gradient
        grad, _ = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        frac_eta = min(1.0, it / max(1, beta_steps * 3))
        eta = eta_start * (eta_end / eta_start) ** frac_eta
        
        # Scale gradient to unit norm for step size interpretation
        grad_unit = grad / (gnorm + 1e-30)
        a = mirror_descent_step(a, grad_unit, eta * gnorm / n, S)
        
        # Periodic active-set refinement
        if it % active_every == 0:
            comb_grad, _ = compute_active_gradient(a, c, n_active=20, gap=3.0)
            cg_norm = np.linalg.norm(comb_grad)
            if cg_norm > 1e-14:
                a = mirror_descent_step(a, comb_grad / cg_norm, eta * 0.5 * cg_norm / n, S)
        
        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 300:
            # Inject structured noise: perturbation concentrated where gradient is high
            noise = np.zeros(n)
            top_grad = np.abs(grad).argsort()[::-1][:max(1, n // 10)]
            noise[top_grad] = rng.normal(0, 0.01 * S / np.sqrt(n), len(top_grad))
            a = normalize_sum(a + noise, S)
            last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_fista_aggressive(a_init, n, S, max_time, rng,
                         beta_start=500.0, beta_end=80000.0, beta_steps=100):
    """FISTA with aggressive beta and careful step size control."""
    a = normalize_sum(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)
        
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: conservative for stability
        step = 1.0 / (beta + gnorm * 0.1)
        step = min(step, 1.0 / (gnorm + 1e-10))
        
        # Euclidean projection onto {x >= 0, sum(x) = S}
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            x_new = a
        else:
            x_new = x_new * (S / s)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x_new.copy()
        else:
            y = y * (S / s_y)
        
        a = x_new
        t_mom = t_new
        
        # Restart on stagnation
        if it - last_improve > 500:
            t_mom = 1.0
            y = a.copy()
            # Small perturbation
            noise_scale = 0.005 * S / np.sqrt(n)
            a = normalize_sum(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_active_refine(a_init, n, S, max_time, rng, n_active=25, gap=2.0):
    """Active-set refinement with backtracking line search."""
    a = normalize_sum(a_init.copy(), S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Get active-set gradient direction
        comb_grad, active_idx = compute_active_gradient(a, c, n_active=n_active, gap=gap)
        gnorm = np.linalg.norm(comb_grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.5
        improved = False
        for _ in range(12):
            a_test = a - step * comb_grad
            a_test = np.maximum(a_test, 0.0)
            s = a_test.sum()
            if s < 1e-15:
                step *= 0.5
                continue
            a_test = a_test * (S / s)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-8):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try mirror descent step as alternative
            if gnorm > 1e-14:
                a = mirror_descent_step(a, comb_grad / gnorm, 0.01 * gnorm / n, S)
        
        # Periodic perturbation
        if it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if noise_scale > 1e-12:
                a = normalize_sum(a + rng.normal(0, noise_scale, n), S)
        
        # Gradually increase gap (widen active set)
        # gap is fixed for this run
        
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def equalization_pass(a, n, S, n_iters=80, gap=3.0):
    """
    Direct equalization: identify lags near max and reduce their contributions
    via targeted multiplicative updates.
    """
    a = a.copy()
    for _ in range(n_iters):
        c = fft_convolve_self(a)
        cm = c.max()
        
        # Get active lags
        active_idx = np.where(c >= cm - gap)[0]
        if len(active_idx) > 30:
            top = np.argsort(c[active_idx])[::-1][:30]
            active_idx = active_idx[top]
        
        # Compute total responsibility for each element
        resp = np.zeros(n)
        for k in active_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            weight = max(0.0, c[k] - cm * 0.95)
            resp[js] += 2.0 * a[partners] * weight
        
        resp_max = resp.max()
        if resp_max < 1e-20:
            break
        
        # Multiplicative reduction
        lr = 0.1
        a = a * np.exp(-lr * resp / resp_max)
        a = normalize_sum(a, S)
    
    return a


def anti_concentration(a, n, S, strength=0.1):
    """Reduce concentration: push mass from high-value elements to low-value ones."""
    a = a.copy()
    # Find elements much larger than average
    mean_val = S / n
    high_mask = a > 3.0 * mean_val
    low_mask = a < 0.5 * mean_val
    
    if not np.any(high_mask) or not np.any(low_mask):
        return a
    
    # Redistribute
    excess = (a[high_mask] - 2.0 * mean_val).sum()
    deficit_count = low_mask.sum()
    if deficit_count > 0:
        a[high_mask] -= excess / high_mask.sum() * 0.5
        a[low_mask] += excess / deficit_count * 0.5
    
    a = np.maximum(a, 0.0)
    return normalize_sum(a, S)


def generate_inits(n, rng):
    """Generate diverse initializations optimized for low max-convolution."""
    inits = []
    t = np.linspace(0, 1, n)
    
    # Asymmetric raised cosines (center shifted left)
    for center in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        for width in [0.25, 0.3, 0.35, 0.4, 0.5]:
            x = (t - center) / width
            a = np.zeros(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            a = np.maximum(a, 1e-10)
            inits.append(a.copy())
    
    # Exponential decay (mass concentrated at left)
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-rate * t)
        a = np.maximum(a, 1e-10)
        inits.append(a.copy())
    
    # Power law decay
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (1 - t + 0.01) ** p
        a = np.maximum(a, 1e-10)
        inits.append(a.copy())
    
    # Two-bump asymmetric
    for mu1, mu2, sig, amp2 in [(0.2, 0.5, 0.08, 0.3), (0.25, 0.6, 0.1, 0.5),
                                  (0.15, 0.4, 0.07, 0.4), (0.3, 0.7, 0.12, 0.3)]:
        a = np.exp(-0.5 * ((t - mu1) / sig)**2) + amp2 * np.exp(-0.5 * ((t - mu2) / sig)**2)
        a = np.maximum(a, 1e-10)
        inits.append(a.copy())
    
    # Sinusoidal with left bias
    for k in [1, 2, 3]:
        for phase in [0, 0.5, 1.0]:
            a = 0.2 + 0.8 * (0.5 + 0.5 * np.sin(np.pi * k * t + phase))
            a = np.maximum(a, 0.01)
            inits.append(a.copy())
    
    # Triangular (asymmetric peak at left)
    for peak_pos in [0.2, 0.3, 0.4]:
        a = np.maximum(0, 1 - np.abs(t - peak_pos) / (1 - peak_pos * 0.5))
        a = np.maximum(a, 1e-10)
        inits.append(a.copy())
    
    return inits


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() < 1e-15:
        return np.ones(n) * S / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses a multi-phase strategy combining mirror descent, FISTA, active-set
    KKT refinement, equalization, and anti-concentration heuristics.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    t0 = time.monotonic()
    
    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - safety))
    
    # Best tracking (no closures)
    best_val = np.inf
    best_a = None
    
    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass
    
    # Helper to check and update best
    def check_and_update(a_candidate):
        nonlocal best_val, best_a
        n_c = len(a_candidate)
        S_c = np.sqrt(2.0 * n_c)
        a_c = normalize_sum(a_candidate, S_c)
        c_c = fft_convolve_self(a_c)
        val_c = c_c.max()
        if val_c < best_val:
            best_val = val_c
            best_a = a_c.copy()
            print("  [t=%+6.0fs] New best: %.8f (n=%d)" % (time.monotonic() - t0, val_c, n_c))
        return a_c, val_c
    
    # === Phase 1: Optimize given sequence (if available) ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        remaining = deadline - time.monotonic()
        t1 = min(120.0, remaining * 0.15)
        print("=== Phase 1: Mirror descent on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_mirror_descent(given, n_g, S_g, t1, rng,
                                     eta_start=0.02, eta_end=0.0005,
                                     beta_start=200.0, beta_end=50000.0,
                                     beta_steps=100, active_every=30)
        check_and_update(a1)
        
        # Follow up with FISTA
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t1b = min(90.0, remaining * 0.15)
            print("=== Phase 1b: FISTA aggressive (n=%d, %.0fs) ===" % (n_g, t1b))
            a1b, v1b = run_fista_aggressive(a1, n_g, S_g, t1b, rng,
                                             beta_start=2000.0, beta_end=80000.0,
                                             beta_steps=80)
            check_and_update(a1b)
    
    # === Phase 2: Main hybrid optimization ===
    if best_a is not None:
        n_main = len(best_a)
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.35
        print("=== Phase 2: Main optimization (n=%d, %.0fs) ===" % (n_main, t2))
        
        # Sub-phase 2a: Mirror descent
        t2a = t2 * 0.4
        a2a, v2a = run_mirror_descent(best_a, n_main, S_main, t2a, rng,
                                       eta_start=0.01, eta_end=0.0003,
                                       beta_start=5000.0, beta_end=80000.0,
                                       beta_steps=80, active_every=20)
        check_and_update(a2a)
        
        # Sub-phase 2b: FISTA
        remaining = deadline - time.monotonic()
        t2b = min(t2 * 0.3, remaining * 0.4)
        if t2b > 5:
            print("  Sub-phase 2b: FISTA (%.0fs)" % t2b)
            a2b, v2b = run_fista_aggressive(a2a, n_main, S_main, t2b, rng,
                                             beta_start=10000.0, beta_end=80000.0,
                                             beta_steps=50)
            check_and_update(a2b)
        
        # Sub-phase 2c: Active refinement
        remaining = deadline - time.monotonic()
        t2c = min(t2 * 0.3, remaining * 0.4)
        if t2c > 5:
            print("  Sub-phase 2c: Active refine (%.0fs)" % t2c)
            a2c, v2c = run_active_refine(a2b if t2b > 5 else a2a, n_main, S_main, t2c, rng,
                                          n_active=25, gap=2.0)
            check_and_update(a2c)

    remaining = deadline - time.monotonic()
    if remaining > 100 and best_a is not None:

        if n_large <= 32768 and remaining > 120:

            t_old = np.linspace(0, 1, len(best_a))
            t_new = np.linspace(0, 1, n_large)

            init_large = normalize_sum(init_large, S_large)
            t3 = min(100.0, remaining * 0.2)
            print("=== Phase 3a: Larger n=%d (%.0fs) ===" % (n_large, t3))
            a3a, v3a = run_mirror_descent(init_large, n_large, S_large, t3, rng,
                                           eta_start=0.05, eta_end=0.001,
                                           beta_start=100.0, beta_end=30000.0,
                                           beta_steps=60)
            check_and_update(a3a)
        
        # Try halving
        remaining = deadline - time.monotonic()
        n_small = max(1024, len(best_a) // 2)
        if n_small != len(best_a) and remaining > 60:
            S_small = np.sqrt(2.0 * n_small)
            t_old = np.linspace(0, 1, len(best_a))
            t_new = np.linspace(0, 1, n_small)
            init_small = np.interp(t_new, t_old, best_a)
            init_small = normalize_sum(init_small, S_small)
            t3b = min(50.0, remaining * 0.15)

            a3b, v3b = run_fista_aggressive(init_small, n_small, S_small, t3b, rng,
                                             beta_start=500.0, beta_end=30000.0,
                                             beta_steps=50)
            check_and_update(a3b)
    
    # === Phase 4: Diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_try = 8192
        S_try = np.sqrt(2.0 * n_try)
        t4 = remaining * 0.25
        print("=== Phase 4: Diverse inits (n=%d, total %.0fs) ===" % (n_try, t4))
        
        inits = generate_inits(n_try, rng)
        
        # Score all initializations
        init_scores = []
        for i, init in enumerate(inits):
            init_n = normalize_sum(init, S_try)
            c = fft_convolve_self(init_n)
            init_scores.append((c.max(), i))
        init_scores.sort()
        
        print("  Top init score: %.4f" % init_scores[0][0])
        
        per_init = max(2.0, t4 / min(10, len(init_scores)))
        
        for rank, (score, idx) in enumerate(init_scores[:10]):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 2:
                break
            init = normalize_sum(inits[idx], S_try)
            print("  [t=%+6.0fs] Init %d (score=%.4f)" % (time.monotonic() - t0, idx, score))
            
            # Use mirror descent for these (faster per-iteration)
            ai, vi = run_mirror_descent(init, n_try, S_try, per_init, rng,
                                         eta_start=0.08, eta_end=0.001,
                                         beta_start=50.0, beta_end=20000.0,
                                         beta_steps=40, active_every=50)
            check_and_update(ai)
    
    # === Phase 5: Equalization and polish ===
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_p = len(best_a)
            S_p = np.sqrt(2.0 * n_p)
            print("=== Phase 5: Equalization + polish (remaining %.0fs) ===" % remaining)
            
            # Equalization pass
            a_eq = equalization_pass(best_a, n_p, S_p, n_iters=60, gap=2.0)
            a_eq, v_eq = check_and_update(a_eq)
            
            # Anti-concentration
            a_anti = anti_concentration(a_eq, n_p, S_p, strength=0.05)
            a_anti, v_anti = check_and_update(a_anti)
            
            # Polish with FISTA
            remaining2 = max(1.0, deadline - time.monotonic())
            if remaining2 > 10:
                t_pol = remaining2 * 0.6
                a_pol, v_pol = run_fista_aggressive(a_anti, n_p, S_p, t_pol, rng,
                                                     beta_start=20000.0, beta_end=100000.0,
                                                     beta_steps=30)
                check_and_update(a_pol)
    
    # === Phase 6: Final active-set refinement ===
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(best_a)
            S_f = np.sqrt(2.0 * n_f)
            t6 = remaining * 0.7
            print("=== Phase 6: Final active refine (remaining %.0fs) ===" % remaining)
            a_f, v_f = run_active_refine(best_a, n_f, S_f, t6, rng, n_active=30, gap=1.5)
            check_and_update(a_f)
    
    # === Phase 7: One more round of mirror descent on best ===
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_m = len(best_a)
            S_m = np.sqrt(2.0 * n_m)
            t7 = remaining * 0.8
            print("=== Phase 7: Final mirror descent (remaining %.0fs) ===" % remaining)

                                           eta_start=0.005, eta_end=0.0001,
                                           beta_start=20000.0, beta_end=100000.0,
                                           beta_steps=30, active_every=10)
            check_and_update(a_m)
    
    # Final verification
    if best_a is not None:
        final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, best_val, len(best_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            best_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    else:

        best_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    # Ensure validity
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    final_score = evaluate_sequence([float(x) for x in best_a])
    print("RETURNING: length=%d, score=%.8f" % (len(best_a), final_score))
    return [float(x) for x in best_a]
```
