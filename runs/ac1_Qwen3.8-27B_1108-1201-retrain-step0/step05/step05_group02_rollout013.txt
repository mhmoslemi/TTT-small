"""

Strategy: Multi-phase with multiplicative weights, high-beta LSE FISTA,
adaptive upscaling, and noise-injection restarts for local minimum escape.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    x = np.maximum(v - theta, 0.0)
    s = x.sum()
    if s > 1e-15:
        x = x * (S / s)
    return x


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, beta):
    """Compute LSE value and gradient. Returns (lse_val, grad, max_c)."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, cm, w


def run_fista_adaptive(a_init, max_time, rng, beta_init=50.0, beta_rate=0.005,
                       beta_max=50000.0, use_backtrack=True, restart_interval=3000):
    """FISTA with adaptive beta, backtracking, and periodic restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0) * 0.5
    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        lse_val, grad, cm, w = lse_grad(y, beta)
        cur_obj = 2.0 * n * cm / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            obj_y = cur_obj
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > obj_y + 1e-12 and step_local > 1e-16 and bt < 10:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = max(step_local * 0.5, 1e-16)
            else:
                step = min(step * 1.05, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = 0.5 / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Periodic restart to escape local minima
        stuck = it - last_improve
        if stuck > restart_interval:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.005 * S / np.sqrt(n) * np.exp(-stuck / (restart_interval * 2.0)))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            step = 1.0 / max(beta, 1.0) * 0.3

        # Very long stagnation: full restart with more noise
        if stuck > restart_interval * 3:
            noise_scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it
            step = 1.0 / max(beta, 1.0) * 0.3

    return best_a, best_obj


def run_mult_weights(a_init, max_time, rng, beta_init=20.0, beta_rate=0.003,
                     beta_max=20000.0):
    """Multiplicative weights (exponentiated gradient) on the simplex."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    # Ensure all entries are positive for multiplicative updates
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = 1.0 / max(beta_init, 1.0) * 0.5

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, cm, w = lse_grad(a, beta)
        cur_obj = 2.0 * n * cm / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Multiplicative update: a * exp(-eta * grad), then normalize
        eta_local = 1.0 / max(beta, 1.0) * 0.3
        log_factor = -eta_local * grad
        # Clip to avoid overflow
        log_factor = np.clip(log_factor, -50, 50)
        a_new = a * np.exp(log_factor)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / sn)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new * (S / a_new.sum())

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-10:
            a = a_new
            # Adapt eta
            if obj_new < cur_obj - 1e-10:
                eta = min(eta * 1.1, 2.0 / max(beta, 1.0))
            else:
                eta = min(eta * 0.95, 2.0 / max(beta, 1.0))
        else:
            eta = max(eta * 0.5, 1e-10)
            # Retry with smaller eta
            log_factor = -eta * grad
            log_factor = np.clip(log_factor, -50, 50)
            a_new = a * np.exp(log_factor)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new * (S / sn)
                a_new = np.maximum(a_new, 1e-12)
                a_new = a_new * (S / a_new.sum())
                if compute_obj(a_new) <= cur_obj + 1e-10:
                    a = a_new

        # Periodic noise injection
        stuck = it - last_improve
        if stuck > 2000 and it % 500 == 0:
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 4000))
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via cubic interpolation for smoother result."""
    n = len(a)
    n_new = n * factor
    t_old = np.arange(n, dtype=np.float64)
    t_new = np.arange(n_new, dtype=np.float64) * (n / n_new)
    # Use linear interpolation (cubic can introduce negative values)
    result = np.interp(t_new, t_old, a)
    result = np.maximum(result, 0.0)
    return result


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) +
             0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) +
             0.15 * np.sin(2 * np.pi * 11 * tt + 2.1))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "triangle":
        a = np.abs(1.0 - 2.0 * tt)
        a = np.maximum(a, 0.01)
    elif method == "asymmetric":
        a = np.exp(-3.0 * tt) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def coordinate_refine(a_init, max_time, rng):
    """
    Coordinate-wise refinement: at each step, find the coordinate that
    most reduces the peak if slightly adjusted, and adjust it.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 5000:
            break

        k_star = int(np.argmax(c))

        # Compute gradient of c[k_star] w.r.t. a
        grad_pk = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            grad_pk[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Also consider top 3 peaks
        top_idx = np.argsort(c)[::-1][:3]
        grad_multi = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo2 = max(0, kk - n + 1)
            hi2 = min(n - 1, kk)
            if hi2 >= lo2:
                grad_multi[lo2:hi2+1] += 2.0 * a[kk - hi2:kk - lo2 + 1][::-1]
        grad_multi /= 3.0

        # Find best coordinate to decrease (high contribution to peak, large value)
        # and best coordinate to increase (low contribution, small value)
        scores_dec = grad_multi * np.maximum(a, 0.001 * S / n)
        scores_inc = -grad_multi * (S / n - a + 0.001 * S / n)

        n_top = min(10, n)
        dec_candidates = np.argsort(scores_dec)[::-1][:n_top]
        inc_candidates = np.argsort(scores_inc)[::-1][:n_top]

        best_obj_trial = cur_obj
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in dec_candidates[:5]:
            if a[j] < 1e-12:
                continue
            for l in inc_candidates[:5]:
                if j == l:
                    continue
                # Try transferring 5% of a[j] to a[l]
                delta = a[j] * 0.05
                if delta < 1e-14:
                    continue

                # Estimate effect on peak
                # c changes by: -delta * (a convolved with delta_j) + delta * (a convolved with delta_l)
                # where delta_j is 1 at position j
                # Effect on c[k]: c[k] -= delta * a[k-j] (if valid)
                #               c[k] += delta * a[k-l] (if valid)
                max_change = 0.0
                for kk in top_idx:
                    kk = int(kk)
                    change = 0.0
                    if 0 <= kk - j < n:
                        change -= delta * a[kk - j]
                    if 0 <= kk - l < n:
                        change += delta * a[kk - l]
                    new_val = c[kk] + change
                    if new_val > max_change:
                        max_change = new_val

                new_max = max(max_change, c[np.argmin(c)] * 0.5)  # rough lower bound
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_obj_trial - 1e-14:
                    best_obj_trial = new_obj
                    best_j, best_l, best_delta = int(j), int(l), delta

        if best_j >= 0:
            a[best_j] -= best_delta
            a[best_l] += best_delta
            a = np.maximum(a, 0.0)
            # Update c incrementally
            j, l, delta = best_j, best_l, best_delta
            # c[k] -= delta * a_old[k-j] for valid k-j
            for k in range(max(0, j), min(2*n-1, j+n)):
                kj = k - j
                if 0 <= kj < n:
                    c[k] -= delta * a[kj]  # approximate (a has changed slightly)
            for k in range(max(0, l), min(2*n-1, l+n)):
                kl = k - l
                if 0 <= kl < n:
                    c[k] += delta * a[kl]
        else:
            # No improvement; small random perturbation
            if it - last_improve > 1000:
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 != idx2:
                    delta = max(1e-8, a[idx1] * 0.02)
                    a[idx1] -= delta
                    a[idx2] += delta
                    a = np.maximum(a, 0)
                    c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj

        print("\n=== Phase 3: Upsample to n=%d (budget %.0fs) ===" % (n2, p3_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        # FISTA on upsampled
        t3a = p3_time * 0.45
        a_up1, obj_up1 = run_fista_adaptive(
            global_best_a, t3a, rng,
            beta_init=20, beta_rate=0.005, beta_max=30000,
            use_backtrack=True, restart_interval=3000)
        print("  FISTA upsampled: obj=%.6f" % obj_up1)
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        # Multiplicative weights on upsampled
        remaining3 = deadline - time.monotonic()
        t3b = remaining3 * 0.4
        if t3b > 15:
            a_up2, obj_up2 = run_mult_weights(
                global_best_a, t3b, rng,
                beta_init=15, beta_rate=0.004, beta_max=20000)
            print("  MW upsampled: obj=%.6f" % obj_up2)
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # === Phase 4: High-beta intensive FISTA (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 30:
        print("\n=== Phase 4: High-beta FISTA (budget %.0fs) ===" % p4_time)
        # Run with very high starting beta
        a_hb, obj_hb = run_fista_adaptive(
            global_best_a, p4_time * 0.6, rng,
            beta_init=500, beta_rate=0.003, beta_max=100000,
            use_backtrack=True, restart_interval=5000)
        print("  High-beta FISTA: obj=%.6f" % obj_hb)
        if obj_hb < global_best_obj:
            global_best_obj = obj_hb
            global_best_a = a_hb.copy()

        # Coordinate refinement
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_cd, obj_cd = coordinate_refine(global_best_a, remaining4 * 0.8, rng)
            print("  Coord refine: obj=%.6f" % obj_cd)
            if obj_cd < global_best_obj:
                global_best_obj = obj_cd
                global_best_a = a_cd.copy()

    # === Phase 5: Final polish - alternate between methods (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.85
    if p5_time > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)
        # Cycle through methods
        n_cycles = max(1, int(p5_time / 60))
        t_per_cycle = p5_time / n_cycles

        for cyc in range(n_cycles):
            if time.monotonic() >= deadline:
                break
            rem = min(t_per_cycle, deadline - time.monotonic())
            if rem < 5:
                break
            # Alternate: FISTA, MW, FISTA with different params
            if cyc % 3 == 0:
                a_r, obj_r = run_fista_adaptive(
                    global_best_a, rem, rng,
                    beta_init=200, beta_rate=0.002, beta_max=80000,
                    use_backtrack=True, restart_interval=2000)
                print("  Polish FISTA: obj=%.6f" % obj_r)
            elif cyc % 3 == 1:
                a_r, obj_r = run_mult_weights(
                    global_best_a, rem, rng,
                    beta_init=50, beta_rate=0.003, beta_max=50000)
                print("  Polish MW: obj=%.6f" % obj_r)
            else:
                # Small noise + FISTA
                noise = 0.002 * np.sqrt(2.0 * len(global_best_a)) / np.sqrt(len(global_best_a))
                a_noisy = global_best_a + rng.normal(0, noise, len(global_best_a))
                a_noisy = np.maximum(a_noisy, 0)
                a_noisy = a_noisy * (np.sqrt(2.0 * len(a_noisy)) / a_noisy.sum())
                a_r, obj_r = run_fista_adaptive(
                    a_noisy, rem, rng,
                    beta_init=100, beta_rate=0.004, beta_max=50000,
                    use_backtrack=True, restart_interval=1500)
                print("  Noisy FISTA: obj=%.6f" % obj_r)

            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_r.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
