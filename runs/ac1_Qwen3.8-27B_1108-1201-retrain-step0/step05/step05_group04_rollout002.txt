`Δc[k*] ≈ 2 * (a[k*-j] - a[k*-i]) * δ`

where δ is the amount transferred. To decrease c[k*], we want `a[k*-j] < a[k*-i]`, so we should take from i and give to j where a[k*-j] < a[k*-i].

This gives a simple coordinate descent rule! Let me develop this:

```python
def coordinate_swap(a, n, S, max_time, rng):
    """Coordinate swap: transfer small amounts between coordinates to reduce the peak."""
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        k_star = c.argmax()
        cur_max = c[k_star]
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # For the peak lag k_star, the gradient is:
        # grad[i] = 2 * a[k_star - i] (for valid indices)
        # To decrease c[k_star], we want to move mass from high-grad to low-grad coordinates.
        
        # Compute the "partner" values
        partners = np.zeros(n)
        valid = (k_star >= 0) & (k_star < 2*n-1)
        if valid:
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            for i in range(lo, hi+1):
                partners[i] = a[k_star - i]
        
        # Simple heuristic: move from coordinates with high partner value to low partner value
        # But this is O(n) per step, which is fine for n=8192
        
        # Find the coordinate with highest partner value (most "harmful" to the peak)
        # and the one with lowest (most "helpful")
        
        # Actually, let's do a batch approach:
        # Sort by partner value, take top batch and bottom batch
        # Move a small amount from top to bottom
        
        sorted_idx = np.argsort(partners)
        n_swap = 100
        src_idx = sorted_idx[-n_swap:]  # High partner values (remove from these)
        dst_idx = sorted_idx[:n_swap]   # Low partner values (add to these)
        
        # Small transfer
        total_src = a[src_idx].sum()
        if total_src < 1e-10:
            continue
        frac = 0.01  # Transfer 1% of the source mass
        amounts = a[src_idx] * frac / max(1, len(src_idx))
        
        for i in range(len(src_idx)):
            a[src_idx[i]] -= amounts[i]
            a[dst_idx[i % len(dst_idx)]] += amounts[i]
        
        a = np.maximum(a, 0)
        a = a * (S / a.sum())  # Renormalize
    
    return best_a, best

Let me also add a key insight: **the "anti-peak" strategy**. Instead of just reducing the peak, we can also try to INCREASE the values at lags that are currently below the peak, which effectively redistributes the energy. This is what the LSE does implicitly (it treats all lags above the "soft minimum" as contributing to the gradient), but we can be more explicit.

Actually, the LSE gradient already does this: it weights each lag by exp(beta * (c[k] - cm)), so lags near the peak have high weight and lags far below have low weight. The gradient points in the direction that reduces the weighted sum, which effectively reduces the peak while (less strongly) increasing the other lags.

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-phase optimization:
1. High-beta LSE-FISTA for broad improvement
2. Adam with LSE for adaptive per-coordinate steps
3. Exact peak gradient with backtracking for the last mile
4. Random perturbations to explore nearby basins
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to avoid overflow/underflow
    shifted = np.clip(shifted, -50, 0)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val

def run_fista_highbeta(a_init, n, S, max_time, rng, beta_start=500, beta_end=50000):
    """FISTA starting from high beta for fine-tuning."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 3000)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Periodic restart
        if it % 200 == 0:
            t_mom = 1.0
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def run_adam_lse(a_init, n, S, max_time, rng, lr=0.05, beta_start=200, beta_end=30000):
    """Adam optimizer with LSE smoothing."""
    a = normalize_sum(a_init, n)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    epsilon = 1e-8
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / 5000)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        grad, _ = lse_gradient(a, c, beta)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr_t = lr * np.exp(-t / 10000.0)
        
        a_new = project_simplex(a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon), S)
        
        # Check for improvement
        if t % 100 == 0:
            c_check = fft_convolve_self(a_new)
            if c_check.max() > cur_max * 1.1:
                # If it got much worse, reduce step
                lr *= 0.5
                a = a  # Don't update
            else:
                a = a_new
        
        if t % 1000 == 0:
            print("  Adam iter %d: max=%.8f, lr=%.6f, beta=%.0f" % (t, best, lr, beta))
    
    return best_a, best

def run_exact_peak_descent(a_init, n, S, max_time, rng):
    """Exact max gradient descent with backtracking line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        k_star = np.argmax(c)
        cur_max = c[k_star]
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Compute exact gradient at the peak
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_star - i]
        
        # Also consider the second peak to avoid oscillation
        c_sorted = np.sort(c)[::-1]
        if c_sorted[1] > cur_max * 0.99:
            # Multiple peaks, use weighted gradient
            top_k = np.argsort(c)[::-1][:5]
            weights = c[top_k] - c.min() + 1e-10
            weights /= weights.sum()
            grad = np.zeros(n)
            for k in top_k:
                lo_k = max(0, k - n + 1)
                hi_k = min(k, n - 1)
                for i in range(lo_k, hi_k + 1):
                    grad[i] += weights[top_k.tolist().index(k)] * 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.01
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try a very small fixed step
            step = 1e-5
            a = project_simplex(a - step * grad, S)
        
        # Periodic noise
        if it % 50 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def run_multi_start(a_init, n, S, max_time, rng, n_starts=5):
