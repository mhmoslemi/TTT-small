"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses annealed-LSE FISTA with backtracking, periodic coordinate polish,
directional perturbations, and L-BFGS-B final squeeze.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient (numerically stable)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient with fractional power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value(c, beta):
    """LSE-smoothed max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def coordinate_polish(a, S, c, max_iters=200, tol=1e-12):
    """
    Greedy coordinate descent on the simplex for the LSE-smoothed objective.
    Cycles through coordinates, finding the best update for each.
    """
    n = len(a)
    a = a.copy()
    beta = 50.0  # Local smoothing for coordinate eval

    for it in range(max_iters):
        improved = False
        # Process coordinates in order of importance (high values first)
        order = np.argsort(-a)[:min(200, n)]

        for i in order:
            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try removing some mass from coordinate i and redistributing
            # Simple: try a[i] * (1 - delta) for a few deltas
            best_obj = lse_value(fft_convolve_self(a), beta)
            best_new = old_val

            for frac in [0.5, 0.7, 0.85, 0.95]:
                a[i] = old_val * frac
                a_new_sum = a.sum()
                if a_new_sum < 1e-10:
                    a[i] = old_val
                    continue
                # Renormalize
                a_norm = a * (S / a_new_sum)
                obj_try = lse_value(fft_convolve_self(a_norm), beta)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_new = old_val * frac
                a[i] = old_val

            if best_new != old_val:
                a[i] = best_new
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                improved = True

        if not improved:
            break

    return a, c


def directional_perturb(a, c, S, rng, strength=0.05):
    """
    Perturb along the direction that most reduces the peak convolution lag.
    The peak is at lag k*; we want to reduce b[k*] = sum_i a[i]*a[i+k*].
    Direction: decrease a[i] where a[i]*a[i+k*] is large, increase elsewhere.
    """
    n = len(a)
    k_star = int(np.argmax(c))

    # Compute the "pressure" at each coordinate from the peak lag
    # b[k*] involves pairs (i, k*-i) for valid i
    pressure = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            p = a[i] * a[j]
            pressure[i] += p
            pressure[j] += p

    # Direction: subtract proportional to pressure, add uniformly
    p_sum = pressure.sum()
    if p_sum < 1e-15:
        return a

    direction = -pressure / p_sum + 1.0 / n
    direction = direction * (S / (n * direction.sum() + 1e-15)) * n

    a_new = a + strength * direction
    a_new = project_simplex(a_new, S)
    return a_new


def run_annealed_fista(a_init, max_time, rng,
                       beta_start=30.0, beta_end=20000.0,
                       n_beta_steps=8, use_backtrack=True):
    """
    FISTA with annealed beta schedule and optional backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (4.0 * beta_start)

    # Beta schedule: geometric from beta_start to beta_end
    beta_schedule = np.geomspace(beta_start, beta_end, n_beta_steps)
    beta_idx = 0
    steps_per_beta = 2000

    while time.monotonic() - start < max_time:
        it += 1

        # Update beta
        if it % steps_per_beta == 0 and beta_idx < len(beta_schedule) - 1:
            beta_idx += 1
            t_mom = 1.0  # Reset momentum on beta change
        beta = beta_schedule[beta_idx]

        c = fft_convolve_self(y)
        cur_lse = lse_value(c, beta)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Compute gradient
        if stuck > 5000:
            # Use top-k gradient when very stuck
            k = min(32, 4 + stuck // 500)
            grad = topk_gradient(y, c, k, power=0.5)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size
        if use_backtrack:
            # Estimated Lipschitz: L ~ 4 * beta * max(w) * 2 (for conv gradient)
            L_est = 8.0 * beta
            step = min(2.0 / L_est, 0.5 / beta)
            # Backtracking
            α = step
            cur_val = lse_value(fft_convolve_self(y), beta)
            for bt in range(20):
                x_trial = project_simplex(y - α * grad, S)
                trial_c = fft_convolve_self(x_trial)
                trial_val = lse_value(trial_c, beta)
                # Quadratic model: f(y) + grad^T(x-y) + L/2 ||x-y||^2
                model = cur_val + np.dot(grad, x_trial - y) + 0.5 * L_est * np.dot(x_trial - y, x_trial - y)
                if trial_val <= model + 1e-15:
                    break
                α *= 0.5
        else:
            α = 1.0 / max(beta, 1.0)

        x_new = project_simplex(y - α * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

        # Directional perturbation when stuck
        if stuck > 8000 and it % 2000 == 0:
            strength = max(0.01, 0.05 * np.exp(-stuck / 20000.0))
            y = directional_perturb(y, c, S, rng, strength)
            t_mom = 1.0
            stuck = 0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=4):
    """L-BFGS-B polishing with increasing beta and penalty."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [50.0, 200.0, 800.0, 3000.0][:n_beta_levels]
    penalties = [10.0, 50.0, 200.0, 800.0][:n_beta_levels]

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        c = fft_convolve_self(a)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        max_iter = max(20, int(remaining * 3))

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': min(max_iter, 200), 'maxls': 50,
                     'ftol': 1e-15, 'gtol': 1e-12, 'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_gradient_flow(a_init, max_time, rng):
    """
    Pure gradient flow with adaptive step size (no momentum).
    Uses exact subgradient of max at the argmax lag.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    stuck = 0
    lr = 0.01

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()  # Since sum = sqrt(2n), obj = max_b

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stuck += 1
            if stuck > 100:
                lr *= 0.9
                if lr < 1e-6:
                    break

        # Find all lags within 1% of max
        max_val = c.max()
        active = np.where(c >= max_val * 0.99)[0]
        k = len(active)

        # Average subgradient from active lags
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / k
        grad = conv_gradient(a, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Normalized gradient step
        step = lr * S / (n * gnorm + 1e-10)
        a_new = a - step * grad
        a = project_simplex(a_new, S)

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: annealed FISTA, gradient flow, L-BFGS-B polish,
    directional perturbations, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        S0 = np.sqrt(2.0 * n0)
        a_start = a_start * (S0 / a_start.sum())
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        S0 = np.sqrt(2.0 * n0)
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_start = a_start * (S0 / a_start.sum())
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Annealed FISTA with backtracking (35% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: Annealed FISTA (n=%d, budget %.0fs) ===" % (len(global_best_a), p1_budget))
    t1 = time.monotonic()

    a1, _ = run_annealed_fista(global_best_a, p1_budget * 0.65, rng,
                               beta_start=30.0, beta_end=50000.0,

    obj1 = compute_obj(a1)
    print("  Annealed FISTA: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 15:
        a1b, _ = run_annealed_fista(global_best_a, rem1, rng,
                                    beta_start=100.0, beta_end=100000.0,
                                    n_beta_steps=6, use_backtrack=False)
        obj1b = compute_obj(a1b)
        print("  High-beta FISTA: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Gradient flow with exact subgradient (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.15
        print("\n=== Phase 2: Gradient flow (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, obj2 = run_gradient_flow(global_best_a, p2_budget, rng)
        print("  Gradient flow: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))

        a4, obj4 = run_lbfgs_polish(global_best_a, p4_budget, rng, n_beta_levels=4)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Try different n (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Multi-n exploration (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        for n_try in [4096, 16384]:
            if time.monotonic() > deadline - 10:
                break
            S_try = np.sqrt(2.0 * n_try)

            # Initialize from global_best via interpolation
            if n_try > len(global_best_a):
                # Upsample
                a_try = np.interp(np.linspace(0, 1, n_try, endpoint=False),
                                  np.linspace(0, 1, len(global_best_a), endpoint=False),
                                  global_best_a)
            else:
                # Downsample (average blocks)
                block = len(global_best_a) // n_try
                if block < 1:
                    continue
                a_try = np.array([global_best_a[i*block:(i+1)*block].mean()
                                  for i in range(n_try)])

            a_try = np.maximum(a_try, 0) * (S_try / max(a_try.sum(), 1e-15))
            obj_try_init = compute_obj(a_try)
            print("  n=%d init: obj=%.8f" % (n_try, obj_try_init))

            # Quick optimization
            time_left = deadline - time.monotonic() - 5
            if time_left > 10:
                a_try_opt, _ = run_annealed_fista(a_try, time_left * 0.6, rng,
                                                  beta_start=20.0, beta_end=5000.0,
                                                  n_beta_steps=4, use_backtrack=True)
                obj_try = compute_obj(a_try_opt)
                print("  n=%d optimized: obj=%.8f (t=%.0fs)" % (n_try, obj_try, time.monotonic() - t5))
                if obj_try < global_best_obj:

    # === Phase 6: Final L-BFGS squeeze ===

    if remaining > 15:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()

            global_best_obj = obj6
            global_best_a = a6.copy()

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
