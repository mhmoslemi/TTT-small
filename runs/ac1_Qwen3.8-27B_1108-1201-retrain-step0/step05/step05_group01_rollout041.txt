from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smoothed_max(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    val = cm + np.log(e.sum()) / beta
    w = e / e.sum()
    # Gradient: 2 * cross-correlation of w and a_reversed
    # (conv(a,a))_k = sum_i a_i a_{k-i}
    # d/dc_k in direction of a: 2 * sum_i w_k * a_{k-i} (for each k)
    # Total gradient w.r.t. a_j: 2 * sum_k w_k * (d(a*a)_k / da_j)
    # d(a*a)_k / da_j = a_{k-j} + a_{j-k} (if in range)
    # So grad_j = 2 * sum_k w_k * (a_{k-j} + a_{j-k} if in range)
    # This is 2 * (w * a_reversed)(j) in the appropriate range
    a_rev = a[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wa_rev = np.real(ifft(fft(w, L) * fft(a_rev, L)))
    grad = 2.0 * conv_wa_rev[n - 1: 2 * n - 1]
    return val, grad


def run_lbfgs_aggressive(a_init, max_time, rng, betas=None):
    """
    Aggressive L-BFGS-B optimization with progressive beta.
    Uses the LSE approximation of max with increasing sharpness.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    if betas is None:
        betas = [10, 25, 50, 100, 200, 400, 800, 1600, 3200, 6400]
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Each beta level gets proportional time (later levels get more)
        time_this = min(remaining * 0.8, max(5, remaining / 3))
        
        c = fft_convolve_self(a)
        cm = c.max()
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            val, grad = lse_smoothed_max(x, beta)
            # Add penalty for sum constraint (soft)
            sum_x = x.sum()
            pen = 50.0
            val += pen * (sum_x - S) ** 2
            grad = grad + 2.0 * pen * (sum_x - S)
            return val, grad
        
        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12,
                     'm': 30, 'iprint': -1}
        )
        
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        
        print("    L-BFGS beta=%d: obj=%.10f (t=%.1fs, iters=%d, status=%s)" %
              (beta, obj_new, elapsed, result.nit, result.message))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def run_projected_subgradient(a_init, max_time, rng, beta_start=50.0):
    """
    Projected subgradient method with adaptive step size.
    Uses LSE smoothing for the gradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    step = S  # initial step size
    beta = beta_start
    
    while time.monotonic() - start < max_time:
        it += 1
        val, grad = lse_smoothed_max(a, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size (decreasing)
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, S)  # cap step size
        
        # Gradient step + projection
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Half the step if no improvement (simple backtracking)
            a_new = a - step * 0.5 * grad
            a_new = project_simplex(a_new, S)
            a = a_new
            step *= 0.9
        
        # Slowly increase beta
        beta *= 1.005
        beta = min(beta, 10000.0)
    
    return best_a, best_obj


def structured_perturb(a, rng, scale=0.01):
    """
    Apply a structured perturbation to escape local minima.
    Uses smooth (low-frequency) random perturbations.
    """
    n = len(a)
    S = a.sum()
    
    # Generate smooth random perturbation
    pert = rng.normal(0, 1.0, n)
    F = fft(pert)
    # Low-pass filter
    cutoff = max(3, n // 20)
    mask = np.ones(n)
    mask[cutoff:-cutoff] = 0.0
    F *= mask
    pert = np.real(ifft(F))[:n]
    
    # Normalize perturbation
    pnorm = np.linalg.norm(pert)
    if pnorm > 1e-15:
        pert = pert / pnorm * scale * np.linalg.norm(a)
    
    a_new = a + pert
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * (S / a_new.sum())
    
    return a_new


def basin_hopping(a_init, max_time, rng, opt_fn, n_restarts=20, pert_scale_start=0.005):
    """
    Basin hopping: optimize, perturb, re-optimize.
    """
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    time_per_restart = max_time / n_restarts
    pert_scale = pert_scale_start
    
    for i in range(n_restarts):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        t_budget = min(time_per_restart, remaining)
        if t_budget < 3:
            break
        
        # Optimize from current point
        a_opt, obj_opt = opt_fn(a, t_budget * 0.7, rng)
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
        
        # Perturb
        a_new = structured_perturb(a_opt, rng, scale=pert_scale)
        
        # Check if perturbation helps (sometimes we want to escape by making it worse)
        obj_pert = compute_obj(a_new)
        
        # Metropolis acceptance: accept if better, or with probability exp(-delta/T)
        if obj_pert < best_obj:
            a = a_new
            print("  Basin %d: perturbed to better point, obj=%.8f" % (i, obj_pert))
        else:
            # Accept with some probability (temperature decreases over time)
            T = max(0.001, 0.01 * (1.0 - i / n_restarts))
            delta = obj_pert - best_obj
            if delta < 1.0 and rng.random() < np.exp(-delta / T):
                a = a_new
                print("  Basin %d: accepted worse point (delta=%.6f) via Metropolis" % (i, delta))
            else:
                a = a_opt
        
        # Decrease perturbation scale
        pert_scale *= 0.8
    
    return best_a, best_obj


def run_fista_refined(a_init, max_time, rng, beta_start=50.0):
    """
    Refined FISTA with momentum and adaptive beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    beta = beta_start
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        val, grad = lse_smoothed_max(y, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Lipschitz estimate (upper bound)
        # For LSE smoothed max, Lipschitz constant is approximately beta * max_conv * 2
        # But we'll use a practical estimate
        L = 2.0 * beta * (val + 1.0)
        step = 1.0 / L
        step = min(step, S / max(gnorm, 1e-15))
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        obj_a = compute_obj(a)
        if obj_a < best_obj - 1e-14:
            best_obj = obj_a
            best_a = a.copy()
            last_improve = it
        
        # Reset momentum periodically
        if it % 2000 == 0:
            t_mom = 1.0
        
        # Increase beta slowly
        beta = min(beta * 1.002, 50000.0)
        
        # If stuck, do a small perturbation
        if it - last_improve > 5000:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            beta = max(beta * 0.5, beta_start)
    
    return best_a, best_obj


def try_different_lengths(a_ref, max_time, rng, lengths=None):
    """
    Try different sequence lengths by up/down sampling.
    """
    if lengths is None:
        lengths = [8192, 12288, 16384]
    
    best_a = None
    best_obj = np.inf
    start = time.monotonic()
    
    for n_target in lengths:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 30:
            break
        
        # Resample from reference
        if len(a_ref) == n_target:
            a = a_ref.copy()
        else:
            # Linear interpolation
            x_old = np.linspace(0, 1, len(a_ref), endpoint=False)
            x_new = np.linspace(0, 1, n_target, endpoint=False)
            a = np.interp(x_new, x_old, a_ref)
        
        S = np.sqrt(2.0 * n_target)
        a = a * (S / max(a.sum(), 1e-15))
        
        # Quick FISTA optimization
        t_budget = min(remaining * 0.8, 120)
        a_opt, _ = run_fista_refined(a, t_budget, rng, beta_start=50.0)
        obj = compute_obj(a_opt)
        
        print("  Length %d: obj=%.10f (t=%.1fs)" % (n_target, obj, time.monotonic() - start))
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Aggressive L-BFGS-B with progressive beta smoothing, LP-based active-set
method with constraint generation, basin hopping with structured perturbations,
and FISTA refinement. Multi-phase approach with adaptive time allocation.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value: 2*n*max(a*a)/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smoothed_max_grad(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    val = cm + np.log(e.sum()) / beta
    w = e / e.sum()
    # Gradient: 2 * (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wa_rev = np.real(ifft(fft(w, L) * fft(a_rev, L)))
    grad = 2.0 * conv_wa_rev[n - 1: 2 * n - 1]
    return val, grad

            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 300, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'm': m, 'iprint': -1, 'maxfun': 5000}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        improved = "YES" if obj_new < best_obj - 1e-12 else "no"
        print("    L-BFGS beta=%6d: obj=%.10f (t=%5.1fs, it=%4d) %s" %
              (beta, obj_new, elapsed, result.nit, improved))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_lp_active_set(a_init, max_time, rng, K_start=30, K_max=200):
    """
    LP-based active set method with constraint generation.
    Iteratively solves LPs with growing constraint sets.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    K = K_start
    stall_count = 0

    for iteration in range(200):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        c = fft_convolve_self(a)
        cur_peak = c.max()
        cur_obj = 2.0 * n * cur_peak / (S * S)

        # Select active constraints: top-K by value + any within tolerance of max
        tol = cur_peak * 0.001
        active_mask = c >= cur_peak - tol
        active_idx_all = np.where(active_mask)[0]

        # Also include top-K by value to ensure coverage
        top_k_idx = np.argsort(c)[::-1][:K]
        active_idx = np.unique(np.concatenate([active_idx_all, top_k_idx]))
        active_idx = active_idx[:K_max]  # Cap total constraints
        K_actual = len(active_idx)

        # Build LP
        # Variables: [a_0, ..., a_{n-1}, t]
        # Minimize: t
        # Subject to: for each active k: sum_i 2*a0[k-i]*a_i - t <= c[k]
        #             sum(a) = S
        #             a >= 0, t >= 0

        A_ub = np.zeros((K_actual, n + 1))
        b_ub = np.zeros(K_actual)

        for idx, k in enumerate(active_idx):
            k_int = int(k)
            # Coefficients: 2*a0[k-i] for i=0,...,n-1 where 0<=k-i<n
            i_min = max(0, k_int - (n - 1))
            i_max = min(n - 1, k_int)
            if i_min <= i_max:
                A_ub[idx, i_min:i_max + 1] = 2.0 * a[k_int - i_max:k_int - i_min + 1][::-1]
            A_ub[idx, n] = -1.0
            b_ub[idx] = c[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * (n + 1)

        t0 = time.monotonic()
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': max(1, remaining - 2)})
        except Exception as e:
            print("    LP error: %s" % str(e))
            break

        lp_time = time.monotonic() - t0

        if not result.success:
            if iteration < 5:
                K = min(K * 2, K_max)
                continue
            break

        a_lp = result.x[:n]
        t_lp = result.x[n]

        # Check actual objective of LP solution
        a_lp = np.maximum(a_lp, 0)
        a_lp = a_lp * (S / max(a_lp.sum(), 1e-15))
        obj_lp = compute_obj(a_lp)
        c_lp = fft_convolve_self(a_lp)
        peak_lp = c_lp.max()

        # Line search: find best step towards LP solution
        best_step_obj = cur_obj
        best_step_a = a.copy()
        for lam in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = (1 - lam) * a + lam * a_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try * (S / max(a_try.sum(), 1e-15))
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()

        if best_step_obj < cur_obj - 1e-14:
            a = best_step_a
            if obj_lp < best_obj or best_step_obj < best_obj:
                best_obj = min(compute_obj(a), best_obj)
                best_a = a.copy()
            stall_count = 0
            # Grow K to include more constraints
            K = min(K + 5, K_max)
            if iteration % 10 == 0:
                print("    LP iter %3d: obj=%.10f, K=%d, t_lp=%.6f, lp_t=%.1fs" %
                      (iteration, best_obj, K_actual, t_lp, lp_time))
        else:
            stall_count += 1
            if stall_count > 15:
                # Try larger K or break
                K = min(K * 2, K_max)
                if K >= K_max:
                    # Try perturbing and restarting
                    pert = rng.normal(0, 0.002, n)
                    F = fft(pert)
                    cutoff = max(3, n // 30)
                    F[cutoff:-cutoff] *= 0.1
                    pert = np.real(ifft(F))[:n]
                    a = a + pert * np.linalg.norm(a) / (np.linalg.norm(pert) + 1e-15) * 0.01
                    a = np.maximum(a, 0)
                    a = a * (S / a.sum())
                    stall_count = 0
                    K = K_start
                continue
            # No improvement, still continue
            if iteration % 20 == 0:
                print("    LP iter %3d: stalled (best=%.10f, lp_time=%.1fs)" %
                      (iteration, best_obj, lp_time))

    return best_a, best_obj


def structured_perturb(a, rng, scale=0.01):
    """Apply a structured (smooth) random perturbation."""
    n = len(a)
    S = a.sum()
    pert = rng.normal(0, 1.0, n)
    F = fft(pert)
    cutoff = max(3, n // 20)
    mask = np.ones(n)
    mask[cutoff:-cutoff] = 0.0
    F *= mask
    pert = np.real(ifft(F))[:n]
    pnorm = np.linalg.norm(pert)
    if pnorm > 1e-15:
        pert = pert / pnorm * scale * np.linalg.norm(a)
    a_new = a + pert
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new * (S / a_new.sum())
    return a_new


def run_fista_polish(a_init, max_time, rng, beta_start=200.0):
    """FISTA with momentum and adaptive beta for final polishing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    beta = beta_start
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        val, grad = lse_smoothed_max_grad(y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 1.0 / max(beta, 1.0)
        step = min(step, S * 0.5 / max(gnorm, 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 50 == 0:
            obj_a = compute_obj(a)
            if obj_a < best_obj - 1e-14:
                best_obj = obj_a
                best_a = a.copy()
                last_improve = it

        if it % 2000 == 0:
            t_mom = 1.0

        beta = min(beta * 1.001, 100000.0)

        if it - last_improve > 5000 and it % 500 == 0:
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            beta = max(beta * 0.5, beta_start)

    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    return best_a, best_obj


def basin_hopping(a_init, max_time, rng, n_restarts=12, scale_start=0.004):
    """Basin hopping with Metropolis acceptance."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    scale = scale_start
    time_per = max_time / n_restarts

    for i in range(n_restarts):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        t_budget = min(time_per, remaining)
        if t_budget < 5:
            break

        # Quick FISTA optimization
        t_opt = t_budget * 0.6
        a_opt, obj_opt = run_fista_polish(a, t_opt, rng, beta_start=100.0)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print("  Basin %2d: improved to %.10f" % (i, best_obj))

        # Structured perturbation
        a_pert = structured_perturb(a_opt, rng, scale=scale)
        obj_pert = compute_obj(a_pert)

        # Metropolis acceptance
        T = max(0.0005, 0.005 * (1.0 - i / max(1, n_restarts)))
        delta = obj_pert - best_obj
        if delta < 0 or (delta < 0.01 and rng.random() < np.exp(-delta / T)):
            a = a_pert
        else:
            a = a_opt

        scale *= 0.85

    return best_a, best_obj

            a_start = None
