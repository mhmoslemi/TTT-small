"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Exact subgradient Nesterov method with adaptive step and restarts
- Balancing subproblem (Woodbury-based) to equalize top lags
- LSE-smoothed FISTA for final refinement

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_at_lag(a, k_star):
    """Subgradient of c_{k_star}(a) = (a*a)[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def lse_gradient(a, c, beta):
    """LSE (log-sum-exp) smoothed gradient."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute objective: 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def balance_delta(a, c, k_top, alpha=1.0):
    """
    Compute balancing perturbation to equalize top-k lags.
    Uses Woodbury identity for efficiency.
    Returns delta with sum(delta)=0 that reduces the gap between top lags.
    """
    n = len(a)
    k_valid = min(k_top, len(c))
    top_idx = np.argsort(c)[::-1][:k_valid]
    M_prime = c[top_idx[0]]
    
    # Check if all top lags are the same (no improvement possible)
    if M_prime - c[top_idx[-1]] < 1e-12:
        return np.zeros(n)
    
    G = np.zeros((k_valid, n))
    b = np.zeros(k_valid)
    for i, ki in enumerate(top_idx):
        G[i] = subgradient_at_lag(a, ki)
        b[i] = M_prime - c[ki]
    
    # δ = (1/alpha) * G^T * (I + (1/alpha)*G*G^T)^{-1} * b
    GGt = G @ G.T  # k x k
    A_k = np.eye(k_valid) + (1.0 / alpha) * GGt
    try:
        tmp = np.linalg.solve(A_k, b)
    except np.linalg.LinAlgError:
        tmp = np.linalg.lstsq(A_k, b, rcond=None)[0]
    delta = (1.0 / alpha) * (G.T @ tmp)
    
    # Project onto sum=0
    delta -= np.mean(delta)
    return delta


def run_subgrad_nesterov(a_init, max_time, rng, step_base=5.0, restart_interval=3000):
    """Exact subgradient Nesterov with adaptive step and restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a_init, S).copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_obj_conv - 1e-14:
            best_obj_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        k_star = int(np.argmax(c))
        g = subgradient_at_lag(y, k_star)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step: Polyak-like with decay
        step = step_base / (gnorm * np.sqrt(1.0 + it / 100.0))
        step = min(step, S / gnorm)
        
        x_new = project_simplex(y - step * g, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart when stuck
        if stuck > restart_interval or it % 10000 == 0:
            t_mom = 1.0
            if stuck > restart_interval:
                # Structured perturbation: push down the peak lag
                delta = balance_delta(y, c, k_top=5, alpha=0.1)
                dnorm = np.linalg.norm(delta)
                if dnorm > 1e-12:
                    scale = 0.01 * S / dnorm
                    y = project_simplex(y + scale * delta, S)
                stuck = 0
                last_improve = it
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_obj_conv:
        best_obj_conv = c_final.max()
    return best_a, best_obj_conv

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_blackman_init(n):
    """Blackman window."""
    tt = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * tt / (n - 1)) + 0.08 * np.cos(4 * np.pi * tt / (n - 1))
    return np.maximum(a, 0.01)


def make_excess_kurtosis_init(n):
    """Sequence designed to flatten the convolution."""
    tt = np.linspace(0, 1, n, endpoint=False)
    return 1.0 + 0.3 * np.cos(2 * np.pi * tt) + 0.1 * np.cos(4 * np.pi * tt)


def perturb_structured(a, rng, scale=0.05):
    """Apply structured perturbation (smooth noise)."""
    n = len(a)
    tt = np.arange(n, dtype=np.float64)
    # Smooth perturbation using random Fourier components
    n_modes = 10
    modes = rng.integers(1, n // 2, size=n_modes)
    amplitudes = rng.normal(0, 1, size=n_modes) * scale
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    noise = np.zeros(n)
    for m, amp, ph in zip(modes, amplitudes, phases):
        noise += amp * np.cos(2 * np.pi * m * tt / n + ph)
    return np.maximum(a + noise, 0.0)


def upsample(a, factor=2):
    """Upsample using cubic interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        print("  Phase 5 total: %.1fs" % (time.monotonic() - t5))
        main_a = global_best_a.copy()
    
    # === Phase 6: Final intensive polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Final polish (%.0fs) ===" % remaining)
        t6 = time.monotonic()
        
        # Alternate between different methods
        sub_budget = remaining * 0.4
        if sub_budget > 20:
            a6a, _ = run_subgrad_nesterov(global_best_a, sub_budget, rng, step_base=2.0)
            obj_6a = update_best(a6a)
            print("  subgrad: obj=%.6f" % obj_6a)
        
        sub_budget = remaining * 0.3
        if time.monotonic() < deadline - 30 and sub_budget > 20:
            a6b, _ = run_lse_fista(global_best_a, sub_budget, rng, beta_init=500.0, beta_max=200000.0)
            obj_6b = update_best(a6b)
            print("  LSE: obj=%.6f" % obj_6b)
        
        sub_budget = remaining * 0.25
        if time.monotonic() < deadline - 20 and sub_budget > 20:
            # Final balancing iterations
            a6c, _ = run_alternating(global_best_a, sub_budget, rng, k_balance=8)
            obj_6c = update_best(a6c)
            print("  balance: obj=%.6f" % obj_6c)
        
        print("  Phase 6 total: %.1fs" % (time.monotonic() - t6))
    
    # Final result
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f, n=%d, time=%.0fs ===" % (
                ver, len(final_list), time.monotonic() - t_total_start))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected best, using fallback")
    
    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    return [float(x) for x in a_fb]
