def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, beta_at=300.0):
    """Gradient using hard active set (top lags within threshold of max)."""
    n = len(a)
    cm = c.max()
    threshold = cm - 5.0 / beta_at
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def raised_cosine_init(n, center_frac=0.3, width_frac=0.5):
    """Generate a raised-cosine initialization centered at center_frac*n."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def generate_diverse_inits(n, rng):
    """Generate diverse initializations with varying asymmetry."""
    inits = []
    t = np.linspace(0, 1, n)

    # Various raised cosine shapes
    for cf in [0.25, 0.3, 0.35, 0.4, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6, 0.7]:
            inits.append(raised_cosine_init(n, cf, wf))

    # Smooth bumps
    for sigma in [0.1, 0.15, 0.2, 0.25]:
        for mu in [0.3, 0.4, 0.5, 0.6]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Power-law shapes
    for p in [0.5, 1.0, 1.5, 2.0]:
        a = t ** p
        inits.append(np.maximum(a, 1e-10))
        a = (1 - t) ** p
        inits.append(np.maximum(a, 1e-10))

    # Sinusoidal
    for k in [1, 2, 3]:
        a = 0.5 + 0.5 * np.sin(np.pi * k * t)
        inits.append(np.maximum(a, 0.01))

    return inits


def peak_shaving(a, n, S, n_peaks=5, n_steps=50, lr_scale=0.01):
    """
    Surgical peak shaving: identify the top lags and directly reduce contributions.
    For each peak lag k, reduce a[j] where a[j]*a[k-j] is large.
    """
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute "responsibility" for each a[j]
        resp = np.zeros(n)
        for k in top_idx:
            # Contribution of a[j] to c[k] is a[j] * a[k-j]
            # For valid j: 0 <= j < n and 0 <= k-j < n
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Add to responsibility (weighted by how much this lag exceeds target)
            weight = max(0, c[k] - cm * 0.99)
            np.add.at(resp, js, contrib * weight)

        # Normalize and apply
        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Multiplicative update: reduce high-responsibility elements
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a
