import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def peak_gradient(a, c, n_peaks=5, temperature=0.01):
    """
    Gradient that specifically targets the worst peaks.
    Uses exponential weighting of peaks above (1-temperature)*max.
    """
    n = len(a)
    cm = c.max()
    # Weight peaks by how much they exceed the threshold
    threshold = cm * (1.0 - temperature)
    w = np.maximum(c - threshold, 0.0)
    tot = w.sum()
    if tot < 1e-300:
        # Just target the single worst lag
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    """

    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Compute initial

    cur = c.max()
    best = cur

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        
        # Adaptive beta

        c = fft_convolve_self(y)
        cur = c.max()
        
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient

        best_a = y.copy()

                break
            step *= 0.5
        else:
            # Even with tiny step, move
            x_new = project_simplex(y - step * grad, S)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Stagnation handling
        stag = it - last_improve
        if stag > 400:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stag / 8000.0)
            if noise_scale > 1e-11:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
            last_improve = it
        
        iter_times.append(time.monotonic() - it0)
        if it % 500 == 0:
            avg_t = np.mean(iter_times[-100:])
            print("    it=%d, obj=%.8f, beta=%.0f, step=%.4f, avg_time=%.3fms" %
                  (it, best, beta, step, avg_t * 1000))
    
    # Final check
    c_final = fft_convolve_self(y)
    if c_final.max() < best:
        best = c_final.max()
        best_a = y.copy()
    
    return best_a, best, it


def peak_chase(a_init, n, max_time, rng, n_peaks=3):
    """
    Direct peak chasing using exact subgradients of the max function.
    Targets the worst few lags specifically.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()
        
        # Exact subgradient
        grad = exact_peak_subgradient(a, c, n_peaks=n_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        
        # Backtracking
        step = 0.1
        for _ in range(15):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur - 1e-13 * cur:
                a = a_new
                break
            step *= 0.5
        else:
            a = project_simplex(a - step * grad, S)
        
        # Periodic noise to escape local minima
        if it % 200 == 0:
            ns = 0.001 * S / np.sqrt(n) * max(0.01, np.exp(-it / 5000.0))
            if ns > 1e-11:
                a = project_simplex(a + rng.normal(0, ns, n), S)
        
        if it % 500 == 0:
            print("    peak_chase it=%d, obj=%.8f" % (it, best))
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = y.copy() if 'y' in dir() else a.copy()
        best_a = a.copy()
    
    return best_a, best, it


def multiplicative_peak_shave(a_init, n, max_time, rng, n_peaks=5, n_inner=200):
    """
    Multiplicative weight update focused on peak reduction.
    For each peak lag k, compute responsibility of each a[j] and
    multiplicatively reduce high-responsibility elements.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in sorted_idx:
            weight = max(0.0, c[k] - cur * 0.995)
            if weight < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            js = np.arange(lo, hi)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += contrib * weight
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative update with adaptive learning rate
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
        
        # Sometimes add noise
        if it % 10 == 0:
            ns = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, ns, n), S)
        
        if it % 200 == 0:
            print("    shave it=%d, obj=%.8f" % (it, best))
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best, it


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses adaptive FISTA with backtracking, peak chasing, and
    multiplicative shaving in a carefully timed multi-phase strategy.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))
    
    # Load starting sequence
    a_init = None
    n_load = 0
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            a_init = np.clip(given, 0, 1000)
            n_load = len(a_init)
            print("Loaded height_sequence_1: n=%d, sum=%.2f" % (n_load, a_init.sum()))
    except Exception as e:
        print("Could not load height_sequence_1: %s" % str(e))
    
    if a_init is None:
        n_load = 8192
        t = np.linspace(0, 1, n_load)
        # Raised cosine initialization
        a_init = 0.5 * (1 + np.cos(2 * np.pi * t))
        a_init = np.maximum(a_init, 1e-10)
        print("Using default raised cosine init, n=%d" % n_load)
    
    n = n_load
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None
    
    def try_seq(a_candidate, label=""):
        nonlocal global_best, global_a
        a_candidate = normalize_sum(a_candidate, n)
        c = fft_convolve_self(a_candidate)
        v = c.max()
        if v < global_best:
            global_best = v
            global_a = a_candidate.copy()
            print("  NEW BEST: %.8f (n=%d) %s" % (v, n, label))
        return v
    
    # Initial evaluation
    try_seq(a_init, "init")
    
    # === Phase 1: Main FISTA with LSE + backtracking (~55% time) ===
    elapsed = time.monotonic() - time.monotonic()
    remaining = deadline - time.monotonic()
    t1 = remaining * 0.55
    print("\n=== Phase 1: FISTA LSE+BT (n=%d, %.0fs) ===" % (n, t1))
    a1, v1, it1 = fista_baktrack(a_init, n, t1, rng,
                                  beta_start=30.0, beta_end=50000.0,
                                  beta_steps=200, use_active=True,
                                  threshold_start=0.1, threshold_end=0.02)
    print("  Phase 1 done: iters=%d, obj=%.8f" % (it1, v1))
    try_seq(a1, "FISTA")
    
    # === Phase 2: Peak chasing (~20% time) ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.35
    if t2 > 20:
        print("\n=== Phase 2: Peak chase (%.0fs) ===" % t2)
        a2, v2, it2 = peak_chase(a1, n, t2, rng, n_peaks=3)
        print("  Phase 2 done: iters=%d, obj=%.8f" % (it2, v2))
        try_seq(a2, "peak-chase")
    
    # === Phase 3: Multiplicative shaving (~10% time) ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.3
    if t3 > 15:
        print("\n=== Phase 3: Multiplicative shave (%.0fs) ===" % t3)
        a3, v3, it3 = multiplicative_peak_shave(global_a, n, t3, rng,
                                                 n_peaks=5, n_inner=200)
        print("  Phase 3 done: iters=%d, obj=%.8f" % (it3, v3))
        try_seq(a3, "shave")
    
    # === Phase 4: High-beta FISTA polish (~10% time) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.5
    if t4 > 15:
        print("\n=== Phase 4: High-beta FISTA polish (%.0fs) ===" % t4)
        a4, v4, it4 = fista_baktrack(global_a, n, t4, rng,
                                      beta_start=10000.0, beta_end=100000.0,
                                      beta_steps=30, use_active=True,
                                      threshold_start=0.02, threshold_end=0.005)
        print("  Phase 4 done: iters=%d, obj=%.8f" % (it4, v4))
        try_seq(a4, "polish")
    
    # === Phase 5: Multi-restart from best (~remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: Multi-restart (%.0fs) ===" % (remaining - 30))
        t_restarts = remaining - 30
        n_restarts = 3
        per_restart = t_restarts / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if per_restart < 20:
                break
            
            # Different perturbation strategies
            if r == 0:
                ns = 0.005 * S / np.sqrt(n)
                init_r = project_simplex(global_a + rng.normal(0, ns, n), S)
            elif r == 1:
                # Frequency-domain perturbation
                fsize = 1 << (2 * n - 1).bit_length()
                A = fft(global_a, fsize)
                phase_noise = rng.normal(0, 0.02, fsize)
                A_modified = A * np.exp(1j * phase_noise)
                init_r = np.real(ifft(A_modified))[:n]
                init_r = project_simplex(np.maximum(init_r, 0), S)
            else:
                # Random coordinate perturbation
                init_r = global_a.copy()
                n_perturb = n // 20
                indices = rng.choice(n, n_perturb, replace=False)
                init_r[indices] *= (1 + rng.normal(0, 0.05, n_perturb))
                init_r = project_simplex(np.maximum(init_r, 0), S)
            
            a_r, v_r, it_r = fista_baktrack(init_r, n, per_restart, rng,
                                             beta_start=500.0, beta_end=50000.0,
                                             beta_steps=80, use_active=True,
                                             threshold_start=0.05, threshold_end=0.01)
            print("  Restart %d: iters=%d, obj=%.8f" % (r, it_r, v_r))
            try_seq(a_r, "restart-%d" % r)
    
    # === Final peak chase on very best ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Final: Peak chase polish (%.0fs) ===" % (remaining - 10))
        a_f, v_f, it_f = peak_chase(global_a, n, remaining - 10, rng, n_peaks=2)
        try_seq(a_f, "final")
    
    # Final verification
    if global_a is None:
        print("WARNING: No valid result! Using uniform fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
    
    result = [float(x) for x in global_a]
    ver = evaluate_sequence(result)
    if not np.isfinite(ver):
        print("CRITICAL: evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        result = [float(x) for x in global_a]
        ver = evaluate_sequence(result)
    
    print("\nFINAL: internal=%.8f, evaluated=%.8f, n=%d" %
          (global_best, ver if np.isfinite(ver) else 999.0, len(global_a)))
    return result
