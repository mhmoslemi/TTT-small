    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total < 1e-15:
        return np.ones(n) * S / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, beta, k_top=3):
    """Compute LSE value and gradient using top-k peaks."""
    n = len(a)
    c = fft_convolve_self(a)
    k = min(k_top, len(c))
    sorted_idx = np.argsort(c)[::-1][:k]
    top_vals = c[sorted_idx]
    
    shifted = beta * (top_vals - top_vals[0])
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(k)
        w[0] = 1.0
        lse_val = top_vals[0]
    else:
        w = e / tot
        lse_val = top_vals[0] + np.log(tot) / beta
    
    w_full = np.zeros(len(c))
    for i, idx in enumerate(sorted_idx):
        w_full[idx] = w[i]
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c


def sigmoid_beta(it, beta_min, beta_max, center, width):
    """Sigmoid-scheduled beta."""
    t = (it - center) / width
    t = np.clip(t, -20, 20)
    return beta_min + (beta_max - beta_min) / (1.0 + np.exp(-t))


def run_fista(a_init, max_time, rng, beta_min=5.0, beta_max=50000.0,
              beta_center=2000, beta_width=1000, k_top=3, step_scale=1.0):
    """FISTA with sigmoid beta schedule and multi-peak LSE."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_min, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = sigmoid_beta(it, beta_min, beta_max, beta_center, beta_width)
        
        lse_val, grad, c = lse_grad(y, beta, k_top)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step with backtracking
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-10 and step_local > 1e-14 and bt < 6:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        if bt > 3:
            step = step_local * 0.6
        else:
            step = min(step * 1.04, 4.0 / max(beta, 1.0))
        x_new = x_trial
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Structured restart when stuck
        stuck = it - last_improve
        if stuck > 2000:
            t_mom = 1.0
            intensity = min(1.0, stuck / 5000.0)
            # Move mass from high-contribution to low-contribution coordinates
            k_star = int(c.argmax())
            contrib = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if hi >= lo:
                contrib[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]
            
            n_move = max(1, n // 50)
            high = np.argsort(contrib * a)[::-1][:n_move]
            low = np.argsort(contrib * (S / n - a + 1e-8))[:n_move]
            for i in range(min(len(high), len(low))):
                delta = a[high[i]] * 0.02 * intensity
                if delta > 1e-10:
                    a[high[i]] -= delta
                    a[low[i]] += delta
            noise = rng.normal(0, 0.002 * S / np.sqrt(n) * intensity, n)
            y = project_simplex(a + noise, S)
    
    return best_a, best_obj


def coord_descent_v2(a_init, max_time, rng, top_k=5):
    """
    Improved coordinate descent: for each top peak lag, find optimal
    pairwise mass transfer that reduces the peak while not increasing others.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 10000:
            break
        
        # Get top peaks
        k_peaks = np.argsort(c)[::-1][:top_k]
        
        # For each peak lag, compute contribution vector
        # c[k] = sum_j a[j] * a[j+k] (with bounds)
        # d(c[k])/d(a[j]) = a[j+k] + a[j-k] (with bounds, 2*a[j] for k=0)
        best_delta_obj = cur_obj + 1e-13
        best_j = -1
        best_l = -1
        best_delta = 0.0
        best_sign = 1  # +1 means a[j] decreases, a[l] increases
        
        for kk in k_peaks[:3]:
            kk = int(kk)
            # Gradient of c[kk] w.r.t. a
            grad_k = np.zeros(n)
            # a[j] contributes to c[kk] as a[j]*a[j+kk] and a[j-kk]*a[j]
            for j in range(n):
                val = 0.0
                if 0 <= j + kk < n:
                    val += a[j + kk]
                if 0 <= j - kk < n:
                    val += a[j - kk]
                if kk == 0:
                    val = 2.0 * a[j]
                grad_k[j] = val
            
            # We want to find j (source) and l (target) such that
            # transferring delta from j to l reduces c[kk] the most.
            # Change in c[kk] ≈ delta * (grad_k[l] - grad_k[j])
            # So we want grad_k[j] > grad_k[l] (move from high grad to low grad)
            # But we also need a[j] > 0 and the change shouldn't hurt other peaks.
            
            # Score: high grad_k * a[j] is good source, low grad_k is good target
            src_score = grad_k * np.maximum(a, 1e-10)
            tgt_score = grad_k
            
            n_candidates = min(20, n)
            src_cand = np.argsort(src_score)[::-1][:n_candidates]
            tgt_cand = np.argsort(tgt_score)[:n_candidates]
            
            for j in src_cand[:8]:
                if a[j] < 1e-10:
                    continue
                for l in tgt_cand[:8]:
                    if j == l:
                        continue
                    delta = min(a[j] * 0.1, S * 0.001)
                    if delta < 1e-12:
                        continue
                    # Quick estimate of new max
                    # c changes by: -delta * a[j] (at lags involving j)
                    #                            + delta * a[l] (at lags involving l)
                    # Approximate: new c[kk] ≈ c[kk] + delta*(grad_k[l] - grad_k[j])
                    # For other peaks, approximate similarly
                    new_c_est = c.copy()
                    for jj in range(n):
                        # Effect of a[j] -= delta
                        for kk2 in [jj, jj + j if jj + j < len(c) else -1, jj - j if jj - j >= 0 else -1]:
                            pass
                    # Simplified: just check the target peak
                    new_kk = c[kk] + delta * (grad_k[l] - grad_k[j])
                    # Check if this would create a new max elsewhere
                    # Approximate by checking a few other peaks
                    ok = True
                    for kk2 in k_peaks:
                        if kk2 == kk:
                            continue
                        grad_k2 = np.zeros(n)
                        for j2 in range(n):
                            val2 = 0.0
                            if 0 <= j2 + int(kk2) < n:
                                val2 += a[j2 + int(kk2)]
                            if 0 <= j2 - int(kk2) < n:
                                val2 += a[j2 - int(kk2)]
                            if int(kk2) == 0:
                                val2 = 2.0 * a[j2]
                            grad_k2[j2] = val2
                        new_kk2 = c[kk2] + delta * (grad_k2[l] - grad_k2[j])
                        if new_kk2 > new_kk + 0.01 * cur_max:
                            ok = False
                            break
                    if ok and new_kk < cur_max - 1e-10:
                        new_obj_est = 2.0 * n * max(new_kk, c[k_peaks[0]] if k_peaks[0] != kk else new_kk) / (S * S)
                        # More conservative: assume max increases slightly
                        new_obj_est = 2.0 * n * max(new_kk + 0.1 * (cur_max - new_kk), new_kk) / (S * S)
                        if new_obj_est < best_delta_obj:
                            best_delta_obj = new_obj_est
                            best_j, best_l, best_delta, best_sign = int(j), int(l), delta, 1
        
        if best_j >= 0:
            a[best_j] -= best_delta
            a[best_l] += best_delta
            # Update c directly
            c = fft_convolve_self(a)  # Full recompute for safety
        else:
            # No improvement from systematic search; try random perturbations
            if it - last_improve > 500:
                n_rnd = min(5, n)
                for _ in range(n_rnd):
                    j = rng.integers(0, n)
                    l = rng.integers(0, n)
                    if j == l or a[j] < 1e-10:
                        continue
                    delta = a[j] * rng.uniform(0.01, 0.05)
                    if delta < 1e-12:
                        continue
                    a[j] -= delta
                    a[l] += delta
                    c_new = fft_convolve_self(a)
                    obj_new = 2.0 * n * c_new.max() / (S * S)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        last_improve = it
                        c = c_new
                        break
                    else:
                        a[j] += delta
                        a[l] -= delta
        
        if it % 50 == 0:
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-4.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) +
             0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) +
             0.15 * np.sin(2 * np.pi * 13 * tt + 2.1))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "plateau":
        a = np.exp(-10.0 * (tt - 0.5) ** 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
