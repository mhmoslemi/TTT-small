"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Multi-start with diverse initializations
- LSE FISTA warmup for rapid initial improvement
- Exact coordinate descent with optimal step computation as main optimizer
- High-beta LSE polishing for final refinement
Key innovation: exact per-coordinate optimal step derived analytically,
combined with periodic full convolution recomputation and adaptive peak tracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, max_time, rng, beta_init=5.0, beta_max=5000.0):
    """LSE FISTA optimization."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it) + (stuck * 0.01 if stuck > 500 else 0))

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / max(beta * 2, 1.0)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def run_coordinate_descent(a_init, max_time, rng, batch_size=200, step_fraction=1.0):
    """
    Exact coordinate descent with optimal per-coordinate steps.
    For each coordinate j, the optimal delta to minimize C[k*]/S^2 is:
      If k* != 2j and 0 <= k*-j < n: delta = S - C[k*]/a[k*-j]
      If k* == 2j: delta = (C[k*] - a[j]*S) / (S - a[j])
    We apply a fraction of this step for stability.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S_target)
    S = a.sum()

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    C = fft_convolve_self(a)
    k_star = int(np.argmax(C))
    c_peak = C[k_star]
    cur_obj = 2.0 * n * c_peak / (S * S)
    best_obj = cur_obj
    best_a = a.copy()

    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Build priority order: coordinates with largest impact on C[k_star]
        # Impact of j on C[k_star] is proportional to a[k*-j] (if valid)
        # We use a mix of priority-based and random ordering
        if it % 10 == 0:
            # Priority order every 10th cycle
            impact = np.zeros(n)
            idx1 = k_star - np.arange(n)
            valid1 = (idx1 >= 0) & (idx1 < n)
            impact[valid1] = a[idx1[valid1]]
            idx2 = np.arange(n) - k_star
            valid2 = (idx2 >= 0) & (idx2 < n)
            impact[valid2] += a[idx2[valid2]]
            # Add small noise to break ties
            impact += rng.random(n) * 0.001
            order = np.argsort(impact)[::-1]
        else:
            order = rng.permutation(n)

        # Process a batch of coordinates
        for j in order[:batch_size]:
            if a[j] < 1e-18:
                continue

            # Determine optimal delta
            if k_star == 2 * j:
                denom = S - a[j]
                if denom < 1e-12:
                    continue
                delta = (c_peak - a[j] * S) / denom
            else:
                idx = k_star - j
                if idx < 0 or idx >= n:
                    continue
                if a[idx] < 1e-18:
                    continue
                delta = S - c_peak / a[idx]

            # Apply step fraction and clamp
            delta *= step_fraction
            max_delta = max(0.01 * S / n, 0.5 * a[j])
            delta = np.clip(delta, -a[j], max_delta)

            if abs(delta) < 1e-16:
                continue

            # Apply update
            a[j] += delta
            S += delta

            # Update c_peak incrementally
            if k_star == 2 * j:
                # c_peak changes by 2*delta*old_a[j] + delta^2
                # But old_a[j] = a[j] - delta
                old_aj = a[j] - delta
                c_peak += 2.0 * delta * old_aj + delta * delta
            else:
                idx = k_star - j
                if 0 <= idx < n:
                    c_peak += 2.0 * delta * a[idx]

            # Safety: c_peak should stay positive
            if c_peak < 0:
                c_peak = 0.0

        # Periodically recompute full convolution
        if it % 5 == 0 or it == 1:
            C = fft_convolve_self(a)
            new_k_star = int(np.argmax(C))
            new_c_peak = C[new_k_star]
            new_obj = 2.0 * n * new_c_peak / (S * S) if S > 1e-15 else np.inf

            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            k_star = new_k_star
            c_peak = new_c_peak

            # If stuck, try a different strategy
            if no_improve > 50:
                # Add small noise to escape
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = np.maximum(a + noise, 0)
                S = a.sum()
                # Renormalize
                a = normalize_seq(a, S_target)
                S = a.sum()
                C = fft_convolve_self(a)
                k_star = int(np.argmax(C))
                c_peak = C[k_star]
                no_improve = 0

    return best_a


def run_hybrid_polish(a_init, max_time, rng):

    best_a = a.copy()

    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Alternate between coordinate descent and LSE
        if it % 3 != 0:
            # Coordinate descent step
            if k_star == 2 * j_check(k_star, n):
                pass  # handled below

            impact = np.zeros(n)
            idx1 = k_star - np.arange(n)
            valid1 = (idx1 >= 0) & (idx1 < n)
            impact[valid1] = a[idx1[valid1]]
            idx2 = np.arange(n) - k_star
            valid2 = (idx2 >= 0) & (idx2 < n)
            impact[valid2] += a[idx2[valid2]]

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 100:
                a = normalize_seq(a + rng.normal(0, 0.002, n), S_target)
                a = np.maximum(a, 0)
                S = a.sum()
                no_improve = 0

    return best_a


def j_check(k, n):
    """Helper to find j such that k = 2j, if it exists in [0, n)."""
    if k % 2 == 0 and k // 2 < n:
        return k // 2
    return -1


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_hann_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1 + np.cos(2 * np.pi * tt / n))


def make_blackman_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.42 + 0.5 * np.cos(2 * np.pi * tt / n) + 0.08 * np.cos(4 * np.pi * tt / n)


def make_exp_decay_init(n):
    tt = np.arange(n, dtype=np.float64)
    return np.exp(-tt / (n * 0.3))


def make_dual_peak_init(n):
    tt = np.arange(n, dtype=np.float64)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.1
    return np.exp(-((tt - center1) / width) ** 2) + np.exp(-((tt - center2) / width) ** 2)


def make_golden_init(n):
    """Sequence based on golden ratio fractional parts."""
    phi = (1 + np.sqrt(5)) / 2
    tt = np.arange(n, dtype=np.float64)
    fracs = (tt * phi) % 1.0
    return 0.3 + 0.7 * fracs


def make_thue_morse_init(n):
    """Thue-Morse sequence mapped to non-negative values."""
    a = np.ones(n)
    for j in range(n):
        # Count bits
        x = j
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        a[j] = 1.0 if cnt % 2 == 0 else 0.4
    return a
