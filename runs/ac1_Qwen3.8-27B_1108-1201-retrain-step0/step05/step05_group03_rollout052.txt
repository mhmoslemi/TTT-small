                best_a = a.copy()
                no_improve = 0
            if no_improve > 5000:
                # Perturb and restart
                noise_scale = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                no_improve = 0

    return best_a, best_max


def run_spectral_flatten(a_init, S, max_time, rng):
    """Spectral power flattening: reduce peaks in the power spectrum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-30:
            break

        # Target: flatten power spectrum (reduce high-power bins)
        # Use soft clipping: power_new = power * (1 + alpha*(1 - power/mean_p))
        alpha = 0.1 + 0.2 * (it / 10000.0)
        ratio = power / max(mean_p, 1e-30)
        damping = 1.0 + alpha * (1.0 - ratio) / (1.0 + 0.1 * ratio)
        damping = np.maximum(damping, 0.1)
        new_power = power * damping
        new_power = np.maximum(new_power, 1e-30)

        A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        a = normalize(a_new, S)

        if it % 50 == 0:
            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best_max - 1e-10:
                best_max = cur_max
                best_a = a.copy()

    c_final = fft_convolve_self(a)
    if c_final.max() < best_max:
        best_max = c_final.max()
        best_a = a.copy()
    return best_a, best_max


def make_initialization(n, S, init_type, rng):
    """Create various initializations."""
    if init_type == 'uniform':
        return np.full(n, S / n)
    elif init_type == 'triangular':
        x = np.linspace(-1, 1, n)
        a = np.abs(x)
        a = a / a.sum() * S
        return a
    elif init_type == 'inverted_triang':
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = a / a.sum() * S
        return a
    elif init_type == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * x * x)
        a = a / a.sum() * S
        return a
    elif init_type == 'random':
        a = rng.uniform(0, 1, n)
        a = a / a.sum() * S
        return a
    elif init_type == 'm_sequence_approx':
        # Approximate m-sequence structure: pseudo-random with balanced bits
        a = rng.choice([0.5, 1.5], size=n)
        a = a / a.sum() * S
        return a
    elif init_type == 'bimodal':
        a = np.zeros(n)
        mid = n // 2
        width = n // 4
        a[mid - width:mid + width] = 1.0
        a = a / a.sum() * S
        return a
    else:
        return np.full(n, S / n)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_full_optimize(a_init, S, max_time, rng):
    """Multi-phase optimization: LSE -> Pairwise -> Spectral -> LSE (high beta)."""
    n = len(a_init)
    start = time.monotonic()
    best_max = fft_convolve_self(a_init).max()
    best_a = normalize(a_init.copy(), S)

    # Phase 1: LSE with moderate beta (40% of time)
    t1 = max_time * 0.40
    if t1 > 10:
        a, val = run_lse_optimize(best_a, S, t1, rng,
                                  beta_init=8.0, beta_max=2000.0,
                                  beta_growth=1.003, k_frac=0.06,
                                  use_momentum=True)
        if val < best_max:
            best_max = val
            best_a = a

    # Phase 2: Pairwise transfer (30% of time)
    t2 = max_time * 0.30
    remaining = max_time - (time.monotonic() - start)
    t2 = min(t2, remaining * 0.5)
    if t2 > 5:
        a, val = run_pairwise_transfer(best_a, S, t2, rng)
        if val < best_max:
            best_max = val
            best_a = a

    # Phase 3: Spectral flattening (15% of time)
    t3 = max_time * 0.15
    remaining = max_time - (time.monotonic() - start)
    t3 = min(t3, remaining * 0.5)
    if t3 > 5:
        a, val = run_spectral_flatten(best_a, S, t3, rng)
        if val < best_max:
            best_max = val
            best_a = a

    # Phase 4: High-beta LSE polishing (15% of time)
    t4 = max_time * 0.15
    remaining = max_time - (time.monotonic() - start)
    t4 = min(t4, remaining * 0.5)
    if t4 > 5:
        a, val = run_lse_optimize(best_a, S, t4, rng,
                                  beta_init=200.0, beta_max=50000.0,
                                  beta_growth=1.001, k_frac=0.02,
                                  use_momentum=False)
        if val < best_max:
            best_max = val
            best_a = a

    return best_a, best_max


def propose_candidate():
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase, multi-length optimization with diverse initializations.
    """
    total_budget = 2400.0
    start_total = time.monotonic()
    deadline = start_total + total_budget - 30.0

    print("Starting optimization, budget=%.0fs" % total_budget)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    rng = np.random.default_rng(12345)
    global_best = np.inf
    global_a = None

    def try_optimization(n, a_start, time_limit, desc):
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
        if a_start is None:
            a_start = np.full(n, S / n)
        a_start = normalize(a_start, S)

        init_obj = 2.0 * n * fft_convolve_self(a_start).max() / (
            a_start.sum() ** 2)
        print("\n=== %s: n=%d, init_obj=%.6f, time=%.0fs ===" %
              (desc, n, init_obj, time_limit))

        a_opt, max_c = run_full_optimize(a_start, S, time_limit, rng)
        obj = 2.0 * n * max_c / (a_opt.sum() ** 2)
        print("  Best: max_c=%.8f, obj=%.6f" % (max_c, obj))

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  *** New global best: %.6f ***" % global_best)

        return a_opt, max_c

    # Strategy: Focus on n=8192 (given length) with multiple initializations,
    # then try other lengths if time permits.

    # Run 1: n=8192 from given sequence (most important)
    if given is not None and len(given) == 8192:
        remaining = deadline - time.monotonic()
        t = min(500, remaining * 0.40)
        if t > 30:
            try_optimization(8192, given, t, "Given-8192")

    # Run 2: n=8192 from diverse initialization
    remaining = deadline - time.monotonic()
    t = min(300, remaining * 0.25)
    if t > 20:
        # Use perturbed version of best so far
        if global_a is not None and len(global_a) == 8192:
            perturb = global_a + rng.normal(0, 0.01 * np.sqrt(2 * 8192) / np.sqrt(8192), 8192)
            try_optimization(8192, perturb, t, "Perturbed-8192")
        else:
            a_init = make_initialization(8192, np.sqrt(2 * 8192), 'random', rng)
            try_optimization(8192, a_init, t, "Random-8192")

    # Run 3: Try n=4096 (smaller, might find different optimum)
    remaining = deadline - time.monotonic()
    t = min(200, remaining * 0.20)
    if t > 20:
        a_init = make_initialization(4096, np.sqrt(2 * 4096), 'inverted_triang', rng)
        try_optimization(4096, a_init, t, "Triang-4096")

    # Run 4: Try n=8192 with different structured init
    remaining = deadline - time.monotonic()
    t = min(250, remaining * 0.25)
    if t > 20:
        a_init = make_initialization(8192, np.sqrt(2 * 8192), 'gaussian', rng)
        try_optimization(8192, a_init, t, "Gauss-8192")

    # Run 5: n=12288 (larger, more freedom)
    remaining = deadline - time.monotonic()
    t = min(200, remaining * 0.20)
    if t > 20 and global_a is not None:
        a_init = resample_sequence(global_a, 12288)
        try_optimization(12288, a_init, t, "Resampled-12288")

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (remaining %.0fs) ===" % remaining)

        # High-beta LSE with very small k
        t_fin = remaining * 0.5
        if t_fin > 10:
            a_fin, val_fin = run_lse_optimize(global_a, S_best, t_fin, rng,
                                              beta_init=500.0, beta_max=100000.0,
                                              beta_growth=1.0005, k_frac=0.01,
                                              use_momentum=False)
            obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin
                print("  Final LSE: obj=%.6f" % global_best)

        # Pairwise transfer polish
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 10:
            a_pol, val_pol = run_pairwise_transfer(global_a, S_best, remaining2, rng)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol
                print("  Final Pairwise: obj=%.6f" % global_best)

    # Final verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

            if not np.isfinite(ver):
                print("  evaluate_sequence rejected! Using fallback.")
                n_fb = 256
                S_fb = np.sqrt(2.0 * n_fb)
                final_list = [float(S_fb / n_fb)] * n_fb
        except Exception as e:
            print("  Verification error: %s" % str(e))

        print("Returning sequence of length %d, best obj=%.6f" %
              (len(global_a), global_best))
        return final_list
    else:
        n_fb = 256
        S_fb = np.sqrt(2.0 * n_fb)
        final_list = [float(S_fb / n_fb)] * n_fb
        print("Fallback: uniform sequence of length %d" % n_fb)
        return final_list
