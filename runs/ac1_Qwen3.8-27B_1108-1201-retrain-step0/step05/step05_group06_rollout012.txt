"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimizer with:
- LSE-FISTA with adaptive beta and backtracking
- Exact subgradient polishing with line search
- Multi-n exploration (8192, 12288, 16384, 24576)
- Structured perturbation escaping
- Coordinate descent on active lags
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE value: (1/beta) * log(sum(exp(beta * c)))"""
    cm = c.max()
    if beta * (c.max() - c.min()) > 700:
        # Avoid overflow: scale
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        log_sum = np.log(np.exp(shifted).sum())
        return cm + log_sum / beta
    else:
        return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_subgradient(y, c, top_k=1):
    """Exact subgradient from top-k active lags."""
    n = len(y)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_max_conv(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_fista_lse(a_init, max_time, rng, beta_init=5.0, beta_max=50000.0,
                   beta_growth=1.002, use_backtrack=True):
    """
    FISTA with LSE surrogate and adaptive beta.
    Uses backtracking line search when available.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = lse_value(fft_convolve_self(a), beta_init)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    step = 1.0 / max(beta, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_lse = lse_value(c, beta)
        
        if cur_lse < best_val - 1e-12:
            best_val = cur_lse
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta
        if stuck > 100:
            beta = min(beta_max, beta * beta_growth)
            step = 1.0 / max(beta, 1.0)
        
        # Compute LSE gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        
        if use_backtrack and stuck > 500:
            # Check if we're making progress
            c_check = fft_convolve_self(x_new)
            lse_check = lse_value(c_check, beta)
            if lse_check > cur_lse + 1e-10 and it % 10 == 0:
                # Try smaller step
                x_new = project_simplex(y - step * 0.5 * grad, S)
                t_mom = 1.0  # Restart momentum
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart when stuck
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
        
        if it % 5000 == 0:
            t_mom = 1.0
    
    return best_a

def run_nesterov_backtrack(a_init, max_time, rng, beta_init=10.0):
    """
    Nesterov accelerated gradient with backtracking line search.
    Uses LSE surrogate with slowly increasing beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    start = time.monotonic()
    best_conv = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    L = beta  # Lipschitz constant estimate
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_lse = lse_value(c, beta)
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        if stuck > 200:
            beta = min(50000.0, beta * 1.01)
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking: find step size alpha such that f(y - alpha*grad) < f(y)
        alpha = 1.0 / L
        for _ in range(20):
            x_trial = project_simplex(y - alpha * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if lse_value(c_trial, beta) < cur_lse - 1e-14 * max(1.0, abs(cur_lse)):
                break
            alpha *= 0.5
            L *= 2.0
        
        # Simple gradient step (no momentum for stability with backtracking)
        a = project_simplex(y - alpha * grad, S)
        
        # Nesterov update for y
        t_k = 1.5  # Fixed momentum parameter for simplicity with backtracking
        y = a + (t_k - 1) / (t_k + 1) * (a - y)
        y = np.maximum(y, 0.0)
        
        if stuck > 5000 and it % 500 == 0:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def upsample(a, factor=2):
    """Upsample using sinc interpolation (zero-stuffing in frequency domain)."""
    n = len(a)
    n_new = n * factor
    # Zero-stuff in frequency domain
    A = fft(a, n_new)
    return np.real(ifft(A))

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE: (1/beta)*log(sum(exp(beta*c)))"""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    log_sum = np.log(e.sum() + 1e-300)
    return cm + log_sum / beta


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_subgradient(y, c, top_k=1):
    """Exact subgradient from top-k active lags."""
    n = len(y)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_max_conv(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

            t_mom = 1.0
            stuck = 0
            last_improve = it

            t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:

    return best_a, best_conv


def run_active_polish(a_init, max_time, rng, top_k_start=1, top_k_max=128):

        # Adaptive top-k
        if stuck < 20:
            k = top_k_start

            k = min(top_k_max, 4 + stuck // 50)

def run_nesterov_bt(a_init, max_time, rng, beta_init=10.0):
    """
    Nesterov with backtracking line search (no momentum for stability).
    """
    n = len(a_init)

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 300:
            beta = min(50000.0, beta * 1.005)
        
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking
        alpha = 1.0 / L
        for _ in range(15):
            x_trial = project_simplex(a - alpha * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if lse_value(c_trial, beta) < cur_lse - 1e-14:
                break
            alpha *= 0.5
            L *= 1.5
        
        a = project_simplex(a - alpha * grad, S)
        L = max(L * 0.95, 1.0)  # Slowly decrease L estimate
        
        if stuck > 3000 and it % 200 == 0:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_hann(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1 + np.cos(2 * np.pi * tt / (n - 1)))


def zero_stuff_upsample(a, n_new):
    """Upsample by zero-stuffing in frequency domain."""
    A = fft(a, n_new)
    return np.real(ifft(A))
