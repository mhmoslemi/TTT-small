import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, k):
    """Subgradient of conv[k] w.r.t. a. conv[k] = sum_j a[j]*a[j+k] (with bounds).
    d(conv[k])/da[j] = a[j-k] + a[j+k] (with out-of-bounds = 0)."""
    n = len(a)
    grad = np.zeros(n)
    # a[j + k] contributes to conv[k] as a[j] * a[j+k], derivative w.r.t. a[j] is a[j+k]
    # a[j - k] contributes to conv[k] as a[j-k] * a[j], derivative w.r.t. a[j] is a[j-k]
    if k > 0:
        # a[j+k] for j from 0 to n-1-k
        grad[:n-k] += a[k:]
        # a[j-k] for j from k to n-1
        grad[k:] += a[:n-k]
    # k=0: grad = 2*a (but this is the gradient of sum(a^2))
    if k == 0:

    return grad

def bb_step(a_prev, a_curr, g_prev, g_curr):

    s = a_curr - a_prev
    y = g_curr - g_prev
    ss = np.dot(s, s)
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    if ss < 1e-30:
        return 0.01
    if yy < 1e-30:
        return 0.01
    # BB_plus
    return ss / max(sy, 1e-10) if sy > 0 else ss / max(-sy, 1e-10)

def direct_center_reduction(a, n, S, n_iters=1000, lr=0.01):
    """Directly reduce the center peak by anti-aligning pairs (i, n-1-i)."""
    a = a.copy()
    for _ in range(n_iters):
        c = fft_convolve_self(a)
        # Focus on center lags
        center = n - 1
        # For each pair (i, n-1-i), try to anti-align
        # But do it in a coordinated way
        
        # Compute the "pressure" on each element from the center peak
        # The center peak is sum_i a[i]*a[n-1-i]
        # To reduce it, for each i, we want a[i]*a[n-1-i] to be small
        # This means: if a[i] > a[n-1-i], reduce a[i] and increase a[n-1-i]
        # But this might increase other peaks
        
        # Simple approach: gradient of center peak w.r.t. a[i] is a[n-1-i]
        # So we subtract a small multiple of a[n-1-i] from a[i]
        # And add it to a[n-1-i] (to keep sum constant)
        
        a_rev = a[::-1]
        # Gradient of center peak: a_rev[i] = a[n-1-i]
        # We want to move in -gradient direction for the center peak
        # But also consider lag 0 peak (gradient is 2*a)
        
        # Combined gradient: weight * a_rev + (1-weight) * 2*a
        # Choose weight to balance
        center_val = c[n-1]
        lag0_val = c[0]
        
        if center_val > lag0_val:
            w = 0.8  # Focus more on center
        else:
            w = 0.3  # More balanced
        
        grad_combined = w * a_rev + (1-w) * 2.0 * a
        grad_combined -= np.mean(grad_combined)  # Zero mean to preserve sum
        
        a = a - lr * grad_combined
        a = np.maximum(a, 0)
        # Re-project to simplex
        a = project_simplex(a, S)
    
    return a

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Uses a multi-phase strategy: aggressive LSE-FISTA, exact subgradient refinement,
diverse restarts, and multi-length exploration. Key improvements over baseline:
faster beta schedule, better stagnation handling, exact subgradient with BB steps,
and targeted local search on top elements.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = c - cm
    e = np.exp(beta * shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def exact_subgradient_topk(a, c, top_k=5, weight_temp=50.0):
    """Compute weighted subgradient from top-k convolution lags."""
    n = len(a)
    max_val = c.max()
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    weights = np.exp(-weight_temp * (max_val - top_vals))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(top_k)
        weights[0] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    for ii, k in enumerate(top_indices):
        wk = weights[ii]
        if k == 0:
            grad += wk * 2.0 * a
        else:
            grad[:n - k] += wk * a[k:]
            grad[k:] += wk * a[:n - k]
    return grad


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=30.0, beta_end=80000.0, beta_power=0.6):
    """FISTA with LSE smoothing and adaptive beta."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** (frac ** beta_power)
        beta = min(beta, 200000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve
        if stagnant > 200:
            t_mom = 1.0
            y = a.copy()
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 15000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.4

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_subgrad_refine(a_init, n, S, max_time, rng, top_k=5, weight_temp=100.0):
    """Refine using exact subgradient with backtracking and momentum."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    lr = 0.5 / S

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad = exact_subgradient_topk(y, c, top_k, weight_temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = lr
        base_obj = cur_max
        x_new = None
        for _ in range(6):
            candidate = project_simplex(y - step * grad, S)
            c_test = fft_convolve_self(candidate)
            test_obj = c_test.max()
            if test_obj < base_obj * (1.0 - 1e-7):
                x_new = candidate
                break
            step *= 0.5

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        stagnant = it - last_improve
        if stagnant > 500:
            t_mom = 1.0
            y = a.copy()
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 8000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            lr = 0.3 / S

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def local_search_top(a, n, S, max_time, rng, n_top=50, n_trials=200):
    """Local search: perturb top elements and accept if improvement."""
    a = normalize_sum(a_init=a, n=n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()

    trial = 0
    while time.monotonic() - start < max_time and trial < n_trials:
        trial += 1
        # Pick a random pair of top elements
        top_idx = np.argsort(a)[::-1][:n_top]
        i = int(rng.choice(top_idx))
        j = int(rng.randint(n))
        if i == j:
            continue

        # Try moving some mass from larger to smaller (or vice versa)
        delta = (a[i] - a[j]) * rng.uniform(0.05, 0.3) * rng.choice([-1, 1])
        a_test = a.copy()
        a_test[i] = max(0.0, a[i] - delta)
        a_test[j] = max(0.0, a[j] + delta)
        # Renormalize
        s = a_test.sum()
        if s < 1e-15:
            continue
        a_test = a_test * (S / s)

        c_test = fft_convolve_self(a_test)
        if c_test.max() < best - 1e-10:
            a = a_test
            best = c_test.max()
            best_a = a.copy()

    return best_a, best


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    # Asymmetric bumps
    for mu in [0.15, 0.25, 0.35, 0.5, 0.65, 0.75, 0.85]:
        for sig in [0.06, 0.1, 0.15, 0.2]:
            a = np.exp(-0.5 * ((t - mu) / sig) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Double bumps
    for mu1, mu2 in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7), (0.15, 0.85)]:
        for sig in [0.06, 0.1, 0.15]:
            a = np.exp(-0.5 * ((t - mu1) / sig) ** 2) + np.exp(-0.5 * ((t - mu2) / sig) ** 2)
            inits.append(a)

    # Power law
    for p in [0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1 - t) ** p, 1e-10))

    # Sine-based
    for k in [1, 2, 3, 4]:
        inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * k * t + rng.uniform(0, np.pi)), 0.01))

    # U-shaped
    inits.append(np.maximum((t - 0.5) ** 2 + 0.01, 0.001))
    inits.append(np.maximum(1.0 - 4.0 * (t - 0.3) ** 2, 0.01))
    inits.append(np.maximum(1.0 - 4.0 * (t - 0.7) ** 2, 0.01))

    # Smooth random
    for _ in range(3):
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=n * 0.05)
        inits.append(np.maximum(a, 1e-10))

    return inits
