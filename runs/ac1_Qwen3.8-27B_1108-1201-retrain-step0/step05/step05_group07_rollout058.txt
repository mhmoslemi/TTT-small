"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses symmetry enforcement, FISTA on LSE-smoothed objective, and multi-phase 
optimization with adaptive parameters.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    # Already feasible?
    if v.sum() == S and np.all(v >= 0):
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w) & (w > 0)
    if not np.any(valid):
        rho = n
        theta = (v.sum() - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def make_symmetric_seq(half, n):
    """Create full symmetric sequence of length n from first ceil(n/2) elements."""
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])


def lse_gradient(a, beta):
    """Compute gradient of LSE-smoothed max of conv(a,a)."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Avoid overflow
    if scaled.max() > 500:
        beta = 500.0 / (c - c_max).max()
        scaled = beta * (c - c_max)
    e = np.exp(scaled)
    w = e / e.sum()
    # Gradient = 2 * conv(w, a) truncated to n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad, c, c_max


def fista_lse(a0, S, max_time, beta, verbose=True):
    """
    FISTA (accelerated projected gradient) on LSE-smoothed objective.
    Minimizes (1/beta)*log(sum exp(beta*conv(a,a))) subject to sum(a)=S, a>=0.
    """
    n = len(a0)
    x = project_simplex(a0.copy(), S)
    y = x.copy()
    t = 1.0
    
    start = time.monotonic()
    it = 0
    best_obj = np.inf
    best_x = x.copy()
    
    # Estimate Lipschitz constant
    # L ≈ 2 + 4*beta*max(a)^2 (rough bound)
    L = 2.0 + 4.0 * beta * x.max()**2
    L = max(L, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute gradient at y
        grad, c, c_max = lse_gradient(y, beta)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        # Step size
        step = 1.0 / L
        
        # Gradient step + projection
        x_new = project_simplex(y - step * grad, S)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)  # Keep non-negative for stability
        t = t_new
        
        x = x_new
        
        # Track best
        c_x = fft_convolve_self(x)
        obj = c_x.max()
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
        
        # Adaptive L estimation (backtracking)
        # Check if the step was too large
        if it % 100 == 0:
            L = max(1.0, L * 0.9)  # Gradually decrease L (increase step)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA LSE it={it}, t={elapsed:.1f}s, max_conv={best_obj:.6f}, L={L:.2f}")
    
    return best_x, best_obj


def subgradient_optimize(a0, S, max_time, rng, verbose=True):
    """
    Projected subgradient with adaptive step and top-k lag weighting.
    """
    n = len(a0)
    x = project_simplex(a0.copy(), S)
    start = time.monotonic()
    it = 0
    best = np.inf
    best_x = x.copy()
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(x)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive k
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(4 + 64 * progress)
        
        # Top-k weighted subgradient
        top_idx = np.argsort(c)[::-1][:k]
        # Weights proportional to excess over median
        threshold = c[int(len(c) * 0.75)]  # 75th percentile
        weights = np.maximum(c[top_idx] - threshold, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
        
        grad = np.zeros(n)
        for ii in range(k):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start <= j_end:
                indices = kidx - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * x[indices]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use gap-based heuristic
        gap = cur_max - c.mean()
        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 3.0 / gnorm)
        
        # Backtracking line search
        accepted = False
        for _ in range(8):
            x_try = project_simplex(x - step * grad, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                x = x_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted and stuck > 50:
            # Noise kick
            noise_scale = 0.01 * S / np.sqrt(n) * min(1.0, stuck / 200.0)
            x = project_simplex(x + rng.normal(0, noise_scale, n), S)
            stuck = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      Subgrad it={it}, t={elapsed:.1f}s, max_conv={best:.6f}")
    
    return best_x, best

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses symmetry enforcement, subgradient descent with adaptive active-set tracking,
KKT alternating polish, and FISTA on LSE-smoothed objective for final convergence.
Key improvements: symmetry halving, informed noise, KKT system solving,
and multi-scale initialization with diverse window functions.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_pair(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
        theta = (v.sum() - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def make_symmetric(half, n):
    """Create full symmetric sequence of length n from first n//2 elements (even n)."""
    return np.concatenate([half, half[::-1]])


def lse_grad_and_vals(a, beta):
    """Compute LSE-smoothed gradient and convolution values."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    max_scaled = scaled.max()
    if max_scaled > 400:
        beta = 400.0 / max(1e-15, (c - c_max).max())
        scaled = beta * (c - c_max)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), c, c_max
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad, c, c_max

        gnorm = np.linalg.norm(grad_half)
        if gnorm < 1e-14:
            break
        
        # Newton-like step: assume H ≈ mu * I (isotropic approximation)
        # Estimate mu from the gap
        gap = cur_max - c[top_idx[-1]] + 1e-10  # spread among active lags
        mu = max(gap, 1e-6)
        
        step = 1.0 / (mu + 1e-6)
        step = min(step, 3.0 / gnorm)
        
        # Line search
        accepted = False
        for trial in range(8):
            x_try = project_simplex(x - step * grad_half, S_half)
            full_try = make_symmetric(x_try, n)
            c_try = fft_convolve_self(full_try)
            if c_try.max() <= cur_max * (1.0 + 1e-8):
                x = x_try
                accepted = True
                break
            step *= 0.6
        
        if not accepted and it % 50 == 0:
            noise_scale = 0.003 * S_half / np.sqrt(n_half)
            x = project_simplex(x + rng.normal(0, noise_scale, n_half), S_half)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"        kkt it={it}, t={elapsed:.0f}s, max={best:.7f}, m={m}")
    
    return best_x, best


def fista_lse_half(half0, n, S, max_time, beta, verbose=True):
    """
    FISTA on LSE-smoothed objective in half-space.
    """
    n_half = n // 2
    S_half = S / 2.0
    x = project_simplex(half0.copy(), S_half)
    y = x.copy()
    t_mom = 1.0
    
    start = time.monotonic()
    best = np.inf
    best_x = x.copy()
    it = 0
    
    # Adaptive L estimation via backtracking
    L = 2.0 + 4.0 * beta * x.max() ** 2
    L = max(L, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        full_y = make_symmetric(y, n)
        grad_full, c_full, c_max = lse_grad_and_vals(full_y, beta)
        
        # Project gradient to half-space
        grad_half = (grad_full[:n_half] + grad_full[n_half - 1::-1])
        gnorm = np.linalg.norm(grad_half)
        
        if gnorm < 1e-15:
            break
        
        # FISTA step
        step = 1.0 / L
        x_new = project_simplex(y - step * grad_half, S_half)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        t_mom = t_new
        x = x_new
        
        # Evaluate
        full_x = make_symmetric(x, n)
        c_x = fft_convolve_self(full_x)
        if c_x.max() < best - 1e-12:
            best = c_x.max()
            best_x = x.copy()
        
        # Adaptive L (slowly decrease to take larger steps)
        if it % 200 == 0:
            L = max(1.0, L * 0.92)
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"        fista it={it}, t={elapsed:.0f}s, max={best:.7f}, L={L:.1f}, beta={beta}")
    
    return best_x, best


def generate_init_scattered(n, shape, rng):
    """Generate initialization with given shape."""
    tt = np.linspace(0, 1, n)
    if shape == 'cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * tt))
    elif shape == 'gaussian':
        a = np.exp(-3.0 * (2 * tt - 1) ** 2)
    elif shape == 'parabola':
        a = np.maximum(1.0 - (2 * tt - 1) ** 2, 0.0)
    elif shape == 'quartic':
        a = np.maximum(1.0 - (2 * tt - 1) ** 4, 0.0)
    elif shape == 'flattop':
        # Non-negative flat-top: modified raised cosine
        c = 2 * tt - 1
        a = 0.42 + 0.5 * np.cos(np.pi * c) + 0.08 * np.cos(2 * np.pi * c)
        a = np.maximum(a, 0.0)
    elif shape == 'sinc':
        x = (tt - 0.5) * 6
        a = np.sinc(x)  # sin(pi*x)/(pi*x)
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    # Symmetrize
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 1e-10)
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Phases: diverse inits -> subgradient -> KKT polish -> FISTA LSE -> final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)
    
    print(f"Starting, budget={budget_s}s, deadline in {budget_s-60:.0f}s")
    
    # Get provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0.01 and len(a_provided) >= 512:
            print(f"Provided: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass
    
    n = 8192
    S = np.sqrt(2.0 * n)
    n_half = n // 2
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, msg):
        nonlocal global_best, global_a
        S_val = a.sum()
        if S_val < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S_val * S_val)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: {obj:.7f} ({msg}) ***")
    
    # === Build initializations ===
    inits = []
    
    if a_provided is not None:
        if len(a_provided) != n:
            t_old = np.linspace(0, 1, len(a_provided))
            t_new = np.linspace(0, 1, n)
            a_res = np.interp(t_new, t_old, a_provided)
        else:
            a_res = a_provided.copy()
        a_res = np.maximum(a_res, 0)
        a_res = 0.5 * (a_res + a_res[::-1])
        if a_res.sum() > 0:
            a_res = a_res * (S / a_res.sum())
            inits.append(a_res[:n_half])
    
    for shape in ['cosine', 'parabola', 'gaussian', 'flattop']:
        a = generate_init_scattered(n, shape, rng)
        a = a * (S / a.sum())
        inits.append(a[:n_half].copy())
    
    print(f"Total inits: {len(inits)}")
    
    # === Phase 1: Subgradient on each init ===
    # Allocate time: more time for provided init
    n_inits = len(inits)
    base_time = (deadline - time.monotonic()) * 0.45 / n_inits
    base_time = min(base_time, 400.0)
    
    half_solutions = []
    
    for i, half_init in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(base_time * (1.5 if i == 0 else 1.0), deadline - time.monotonic() - 30)
        if t_alloc < 20:
            break
        
        label = f"init{i}"
        print(f"\n  Subgradient {label}, t={t_alloc:.0f}s")
        half_opt, best_max = subgradient_half(half_init, n, S, t_alloc, rng,
                                              verbose=(i == 0))
        half_solutions.append(half_opt)
        
        full = make_symmetric(half_opt, n)
        full = np.maximum(full, 0)
        full = full * (S / full.sum())
        update_global(full, label)
    
    # === Phase 2: KKT polish on best solutions ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # Take top 2 solutions from phase 1
        candidates = []
        seen = set()
        for h in half_solutions:
            key = tuple(np.round(h[:10], 4))
            if key not in seen:
                seen.add(key)
                candidates.append(h)
            if len(candidates) >= 2:
                break
        
        t_kkt = min(300.0, remaining * 0.35)
        for ci, half_c in enumerate(candidates):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = t_kkt / len(candidates)
            print(f"\n  KKT polish {ci}, t={t_alloc:.0f}s")
            half_opt, _ = kkt_polish(half_c, n, S, t_alloc, rng, verbose=False)
            full = make_symmetric(half_opt, n)
            full = np.maximum(full, 0)
            full = full * (S / full.sum())
            update_global(full, f"kkt{ci}")
            half_solutions.append(half_opt)
    
    # === Phase 3: FISTA LSE polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        half_best = global_a[:n_half].copy()
        half_best = 0.5 * (half_best + global_a[n_half - 1::-1])
        
        betas = [100, 500, 2000, 8000, 30000]
        t_per = (remaining - 20) / len(betas)
        t_per = min(t_per, 120.0)
        
        for beta in betas:
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            print(f"\n  FISTA LSE beta={beta}, t={t_alloc:.0f}s")
            half_opt, _ = fista_lse_half(half_best, n, S, t_alloc, beta, verbose=False)
            full = make_symmetric(half_opt, n)
            full = np.maximum(full, 0)
            full = full * (S / full.sum())
            update_global(full, f"fista_b{beta}")
            if global_a is not None:
                half_best = 0.5 * (global_a[:n_half] + global_a[n_half - 1::-1])
    
    # === Phase 4: Final subgradient refinement ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        half_best = 0.5 * (global_a[:n_half] + global_a[n_half - 1::-1])
        half_best = project_simplex(half_best, S / 2.0)
        
        t_final = remaining - 15
        print(f"\n  Final subgradient, t={t_final:.0f}s")
        half_opt, _ = subgradient_half(half_best, n, S, t_final, rng, verbose=False)
        full = make_symmetric(half_opt, n)
        full = np.maximum(full, 0)
        full = full * (S / full.sum())
        update_global(full, "final_subgrad")
        
        # Quick FISTA polish
        remaining2 = deadline - time.monotonic()
        if global_a is not None and remaining2 > 15:
            half_best2 = 0.5 * (global_a[:n_half] + global_a[n_half - 1::-1])
            half_best2 = project_simplex(half_best2, S / 2.0)
            half_opt2, _ = fista_lse_half(half_best2, n, S, remaining2 - 5, 50000, verbose=False)
            full2 = make_symmetric(half_opt2, n)
            full2 = np.maximum(full2, 0)
            full2 = full2 * (S / full2.sum())
            update_global(full2, "final_fista")
    
    # === Phase 5: Try n=16384 if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        n_half2 = n2 // 2
        t2 = remaining - 20
        print(f"\n  Trying n={n2}, t={t2:.0f}s")
        
        t_old = np.linspace(0, 1, len(global_a))
        t_new = np.linspace(0, 1, n2)
        a_res2 = np.interp(t_new, t_old, global_a)
        a_res2 = np.maximum(a_res2, 0)
        a_res2 = 0.5 * (a_res2 + a_res2[::-1])
        a_res2 = a_res2 * (S2 / a_res2.sum())
        
        half2 = a_res2[:n_half2]
        # Quick subgradient
        t_sub = t2 * 0.6
        half_opt2, _ = subgradient_half(half2, n2, S2, t_sub, rng, verbose=False)
        full2 = make_symmetric(half_opt2, n2)
        full2 = np.maximum(full2, 0)
        full2 = full2 * (S2 / full2.sum())
        update_global(full2, "n=16384")
        
        # KKT polish
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            half_opt3, _ = kkt_polish(half_opt2, n2, S2, rem2 - 5, rng, verbose=False)
            full3 = make_symmetric(half_opt3, n2)
            full3 = np.maximum(full3, 0)
            full3 = full3 * (S2 / full3.sum())
            update_global(full3, "n=16384_kkt")
    
    # === Final output ===
    if global_a is None:
        print("WARNING: No valid solution found!")
        global_a = np.ones(512) * 1.0
        global_best = 2.0
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(512) * 1.0
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"FINAL: len={len(global_a)}, internal_obj={global_best:.7f}")
    print(f"evaluate_sequence = {ver:.7f}")
    print(f"{'='*60}")
    
    return final_list
