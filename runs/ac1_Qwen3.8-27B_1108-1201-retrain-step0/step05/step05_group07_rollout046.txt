```python
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c[k])))."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -700, 700)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), 0.0
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w

def lse_value(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -700, 700)
    return cm + np.log(np.exp(scaled).sum()) / beta
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses Nesterov-accelerated LSE gradient descent, mirror descent, coordinate descent,
and multi-phase search with adaptive beta scheduling and length exploration.
Key improvements over baseline: true gradient (not subgradient), Nesterov momentum,
multiplicative updates for better boundary handling, and targeted peak chasing.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_and_weights(a, c, beta):
    """Gradient of LSE-smoothed max and softmax weights."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -700, 700)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), np.ones(len(c)) / len(c)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w


def lse_value(c, beta):
    """LSE-smoothed max value."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -700, 700)
    return cm + np.log(np.exp(scaled).sum()) / beta


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_nesterov(a0, S, max_time, rng, beta_init=20.0, beta_max=5000.0,
                      mom_init=0.85, mom_decay=0.0002):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed max convolution.
    Returns (best_a, best_max_val).
    """
    n = len(a0)
    x = project_simplex(a0, S)
    y = x.copy()
    c = fft_convolve_self(x)
    best_max = c.max()
    best_x = x.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 400

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        t_frac = min(1.0, elapsed / max_time)

        # Adaptive beta: exponential growth
        beta = beta_init * np.exp(np.log(beta_max / beta_init) * t_frac)
        # Momentum: gradual decay
        mom = max(0.3, mom_init - mom_decay * it * 50)

        it += 1

        # Nesterov lookahead
        y_prev = y.copy()
        y = project_simplex(x + mom * (x - y_prev), S)

        c_y = fft_convolve_self(y)
        grad, w = lse_gradient_and_weights(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: based on smoothness estimate
        step = 1.0 / (beta * gnorm + 1e-10)
        step = min(step, 3.0 / gnorm)
        f_y = lse_value(c_y, beta)

        # Backtracking line search
        accepted = False
        for _ in range(25):
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            f_try = lse_value(c_try, beta)
            if f_try <= f_y - 1e-8:
                x = x_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            x = project_simplex(y - step * grad, S)

        c_x = fft_convolve_self(x)
        if c_x.max() < best_max - 1e-12:
            best_max = c_x.max()
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1

        # Perturb when stuck
        if stuck > patience:
            noise_scale = 0.015 * S / np.sqrt(n) * (1.0 - 0.5 * t_frac)
            x = project_simplex(x + rng.normal(0, noise_scale, n), S)
            y = x.copy()
            stuck = 0
            patience = int(patience * 1.3)

        if it % 2000 == 0:
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    Nesterov it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")

    return best_x, best_max


def optimize_mirror(a0, S, max_time, rng, beta_init=50.0, beta_max=10000.0):
    """
    Mirror descent (multiplicative updates) for LSE-smoothed max.
    Better handling of boundary (zeros) than Euclidean projection.
    Returns (best_a, best_max_val).
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    # Ensure strict positivity for multiplicative updates
    a = np.maximum(a, S / n * 1e-8)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        t_frac = min(1.0, elapsed / max_time)
        beta = beta_init * np.exp(np.log(beta_max / beta_init) * t_frac)

        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            noise_scale = 0.01 * S / np.sqrt(n) * max(0.1, 1.0 - t_frac)
            a = np.maximum(a + rng.normal(0, noise_scale, n), S / n * 1e-8)
            a = a * (S / a.sum())
            stuck = 0
            continue

        grad, w = lse_gradient_and_weights(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent: a_new = a * exp(-eta * grad), then normalize
        eta = 1.0 / (1.5 * beta)
        update = np.exp(-eta * grad)
        a_try = a * update
        a_try = np.maximum(a_try, 0.0)
        if a_try.sum() < 1e-10:
            break
        a_try = a_try * (S / a_try.sum())

        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-6):
            a = a_try
        else:
            eta *= 0.5
            a_try = a * np.exp(-eta * grad)
            a_try = np.maximum(a_try, 0.0)
            if a_try.sum() > 1e-10:
                a = a_try * (S / a_try.sum())

        if it % 2000 == 0:
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    Mirror it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")

    return best_a, best_max


def optimize_coord_descent(a0, S, max_time, rng, top_k=30):
    """
    Coordinate descent: tries to reduce peak by adjusting individual coordinates.
    Focuses on top peak lags and their contributing coordinates.
    Returns (best_a, best_max_val).
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < max_time:
        if stall > 5000:
            noise = 0.008 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            stall = 0
            continue

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(c)[::-1][:top_k]
        improved = False

        for kidx in top_idx[:8]:
            kidx = int(kidx)
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start > j_end:
                continue
            indices = kidx - np.arange(j_start, j_end + 1)
            contrib = a[j_start:j_end + 1] * a[indices]

            # Find coordinate with largest individual contribution
            max_local = np.argmax(contrib)
            i = j_start + max_local
            if a[i] < 1e-12:
                continue

            # Try several reduction fractions
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = a[i] * frac
                a_try = a.copy()
                a_try[i] -= delta
                # Redistribute: add to coordinate with least peak contribution
                min_idx_j = rng.choice(range(n))
                while min_idx_j == i:
                    min_idx_j = rng.choice(range(n))
                a_try[min_idx_j] += delta
                a_try = project_simplex(a_try, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max - 1e-10:
                    a = a_try
                    cur_max = c_try.max()
                    improved = True
                    break

        # Try random pairwise swaps
        for _ in range(200):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue
            delta = min(a[i], S / n) * rng.uniform(0.02, 0.1)
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = project_simplex(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max - 1e-10:
                a = a_try
                cur_max = c_try.max()
                improved = True
                break

        if not improved:
            pass  # stall counter already incremented

    return best_a, best_max


def generate_init(n, rng, base=None):
    """Generate initial sequence, optionally interpolating from base."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.cos(2 * np.pi * tt) + 0.15 * np.cos(4 * np.pi * tt + 0.7)
    a = np.maximum(a, 0.01)
    return a


def do_refinement(a_curr, S, n, rng, time_budget, phase_seed):
    """
    Run a mixed-refinement cycle: Nesterov -> Mirror -> CoordDescent.
    Returns best (a, max_val) found.
    """
    local_best_a = a_curr.copy()
    local_best_max = fft_convolve_self(a_curr).max()

    remaining = time_budget
    t_nest = min(remaining * 0.45, max(10, remaining - 60))
    t_mir = min(remaining * 0.30, max(10, remaining - t_nest - 30))
    t_cd = max(5, remaining - t_nest - t_mir)

    # Nesterov
    a1, m1 = optimize_nesterov(local_best_a, S, t_nest, rng,
                               beta_init=50.0, beta_max=5000.0)
    if m1 < local_best_max:
        local_best_a, local_best_max = a1, m1

    remaining = time_budget - (time.monotonic() - (time.monotonic() - time_budget))
    # Re-check time
    elapsed_total = time.monotonic() - (time.monotonic() - time_budget)

    # Mirror
    mir_time = min(t_mir, time_budget - elapsed_total - t_cd - 5)
    if mir_time > 10:
        a2, m2 = optimize_mirror(local_best_a, S, mir_time, rng,
                                 beta_init=200.0, beta_max=15000.0)
        if m2 < local_best_max:
            local_best_a, local_best_max = a2, m2

    # Coord descent
    cd_time = max(5, time_budget - (time.monotonic() - (time.monotonic() - time_budget)) - 5)
    if cd_time > 10:
        a3, m3 = optimize_coord_descent(local_best_a, S, cd_time, rng, top_k=30)
        if m3 < local_best_max:
            local_best_a, local_best_max = a3, m3

    return local_best_a, local_best_max


def track_and_update(state, a, msg=""):
    """Update global best tracking in a state dict."""
    S = a.sum()
    if S < 0.01:
        return
    c = fft_convolve_self(a)
    obj = 2.0 * len(a) * c.max() / (S * S)
    if obj < state['best']:
        state['best'] = obj
        state['a'] = a.copy()
        print(f"  NEW BEST: obj={obj:.6f} ({msg})")


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LSE, mirror descent, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Track global best
    state = {'best': np.inf, 'a': None}

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.4f}")
            track_and_update(state, a_provided, "initial")
        else:
            a_provided = None
    except Exception:
        pass

    # === Phase 1: Primary Nesterov optimization on provided ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(700, (deadline - time.monotonic()) * 0.35)
        print(f"\nPhase 1: Nesterov LSE, n={n1}, time={t1:.0f}s")
        a_opt, _ = optimize_nesterov(a1, S1, t1, rng,
                                     beta_init=20.0, beta_max=4000.0,
                                     mom_init=0.85, mom_decay=0.00015)
        track_and_update(state, a_opt, "P1 Nesterov")

    # === Phase 2: Mirror descent refinement ===
    if state['a'] is not None:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        t2 = min(400, remaining * 0.25)
        if t2 > 30:
            print(f"\nPhase 2: Mirror descent, n={n}, time={t2:.0f}s")
            a_m, _ = optimize_mirror(state['a'], S, t2, rng,
                                     beta_init=100.0, beta_max=12000.0)
            track_and_update(state, a_m, "P2 Mirror")

    # === Phase 3: Coordinate descent polish ===
    if state['a'] is not None:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        t3 = min(250, remaining * 0.2)
        if t3 > 20:
            print(f"\nPhase 3: Coord descent, n={n}, time={t3:.0f}s")
            a_cd, _ = optimize_coord_descent(state['a'], S, t3, rng, top_k=40)
            track_and_update(state, a_cd, "P3 CoordDesc")

    # === Phase 4: Alternative lengths ===
    if state['a'] is not None:
        tried_lengths = {len(state['a'])}
        candidate_lengths = [8192, 4096, 16384, 32768, 12288, 1024]
        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 120:
                break
            if n_try in tried_lengths:
                continue
            if n_try < 512 or n_try > 65536:
                continue
            S_try = np.sqrt(2.0 * n_try)
            t4 = min(350, (deadline - time.monotonic()) * 0.2)
            if t4 < 30:
                continue
            tried_lengths.add(n_try)
            print(f"\nPhase 4: n={n_try}, time={t4:.0f}s")
            init = generate_init(n_try, rng, base=state['a'])
            init = project_simplex(init, S_try)
            a_new, _ = optimize_nesterov(init, S_try, t4 * 0.65, rng,
                                         beta_init=10.0, beta_max=3000.0)
            track_and_update(state, a_new, f"P4 n={n_try} nest")
            if time.monotonic() < deadline - 60:
                a_cd, _ = optimize_coord_descent(a_new, S_try,
                                                 min(120, deadline - time.monotonic() - 30),
                                                 rng, top_k=20)
                track_and_update(state, a_cd, f"P4 n={n_try} cd")

    # === Phase 5: Intensive alternating refinement on best ===
    if state['a'] is not None:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\nPhase 5: Alternating refinement, n={n}, time={remaining-20:.0f}s")
            a_curr = state['a'].copy()
            n_rounds = 4
            for ri in range(n_rounds):
                if time.monotonic() > deadline - 15:
                    break
                t_round = min((remaining - 30) / (n_rounds - ri),
                              deadline - time.monotonic() - 15)
                if t_round < 10:
                    break
                if ri % 2 == 0:
                    a_curr, _ = optimize_nesterov(a_curr, S, t_round, rng,
                                                  beta_init=300.0, beta_max=20000.0,
                                                  mom_init=0.7, mom_decay=0.0001)
                else:
                    a_curr, _ = optimize_mirror(a_curr, S, t_round, rng,
                                                beta_init=500.0, beta_max=25000.0)
                track_and_update(state, a_curr, f"P5 round {ri}")

    # === Phase 6: Multi-perturbation restarts ===
    if state['a'] is not None:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        if remaining > 40:
            n_restarts = 5
            t_per = (remaining - 30) / n_restarts
            print(f"\nPhase 6: {n_restarts} restarts, ~{t_per:.0f}s each")
            scales = [0.005, 0.015, 0.03, 0.06, 0.1]
            for ri in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = scales[ri % len(scales)] * S / np.sqrt(n)
                a_pert = project_simplex(state['a'] + rng.normal(0, noise_scale, n), S)
                a_pert, _ = optimize_nesterov(a_pert, S, t_per * 0.7, rng,
                                              beta_init=200.0, beta_max=8000.0)
                track_and_update(state, a_pert, f"P6 restart {ri}")
                # Quick coord descent on result
                if time.monotonic() < deadline - 10:
                    a_cd, _ = optimize_coord_descent(a_pert, S,
                                                     min(30, deadline - time.monotonic() - 10),
                                                     rng, top_k=20)
                    track_and_update(state, a_cd, f"P6 restart {ri} cd")

    # === Phase 7: Final squeeze with high-beta Nesterov ===
    if state['a'] is not None:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\nPhase 7: Final squeeze, time={remaining-10:.0f}s")
            a_final, _ = optimize_nesterov(state['a'], S, remaining - 15, rng,
                                           beta_init=2000.0, beta_max=50000.0,
                                           mom_init=0.5, mom_decay=0.0005)
            track_and_update(state, a_final, "P7 final")
            a_cd_final, _ = optimize_coord_descent(state['a'], S,
                                                   max(5, remaining - 15),
                                                   rng, top_k=50)
            track_and_update(state, a_cd_final, "P7 final cd")

    # Final verification
    if state['a'] is None:
        print("WARNING: No valid solution, using fallback")
        state['a'] = np.ones(256)
        state['best'] = obj_value(state['a'])

    final_a = np.clip(state['a'], 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    n_final = len(final_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, internal_obj={state['best']:.6f}, evaluated={ver:.6f}")
    print(f"{'='*60}")

    return final_list
```
