Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a multi-method approach:
1. Subgradient Frank-Wolfe (conditional gradient) on the probability simplex
2. Multiplicative weights (mirror descent) as an alternative geometry
3. LSE-smoothed polishing with adaptive temperature
4. Multi-length exploration and structured restarts

The key insight is that the problem is equivalent to minimizing max(conv(p,p))
for p on the probability simplex, which is a non-convex problem amenable to
first-order methods with appropriate care.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_topk_subgradient(p, c, k):
    """
    Compute subgradient of max(conv) using top-k lags.
    p is on probability simplex (sum=1, p>=0).
    c is the full convolution of p with p (length 2n-1).
    Returns gradient g (length n) of max(conv(p,p)).
    """
    n = len(p)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    
    # Weight by how much each lag exceeds the minimum (or use uniform)
    max_c = c[top_idx[0]]
    min_top = c[top_idx[-1]]
    if max_c - min_top > 1e-15:
        weights = (c[top_idx] - min_top) / (max_c - min_top)
        weights = weights / weights.sum()
    else:
        weights = np.ones(k) / k
    
    # For each top lag l, the gradient contribution is 2*a[l-i] for valid i
    grad = np.zeros(n)
    for ii in range(k):
        l = int(top_idx[ii])
        w = weights[ii] * 2.0
        # i ranges from max(0, l-n+1) to min(n-1, l)
        i_start = max(0, l - n + 1)
        i_end = min(n - 1, l)
        if i_start <= i_end:
            # grad[i] += w * p[l-i]
            # l-i ranges from l-i_end to l-i_start
            indices = l - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += w * p[indices]
    return grad


def compute_lse_subgradient(p, c, beta):

    n = len(p)
    cm = c.max()

    # Clip for numerical stability

    e = np.exp(scaled)

    w = e / tot  # probability distribution over lags

    W = fft(w, fsize)

    # We want sum_l w[l] * p[l-i] for i = 0,...,n-1
    # This is the cross-correlation of w and p at lag -i
    # = conj(FFT(w)) * FFT(p) ... let me think
    # cross_corr(w, p)[d] = sum_l w[l] * p[l-d]
    # FFT(cross_corr) = conj(FFT(w)) * FFT(p)
    # So cross_corr = ifft(conj(W) * P)
    result = np.real(ifft(np.conj(W) * P))
    # We need cross_corr at lags 0 to n-1
    # cross_corr[d] for d >= 0: sum_l w[l] * p[l-d]
    # The ifft gives us the circular cross-correlation, which for the linear case
    # (with zero padding) gives us what we want at the right indices.
    # Actually, let me just do it directly for clarity, or use the full convolution approach.
    
    # Alternative: grad[i] = 2 * sum_l w[l] * p[l-i]
    # = 2 * (full_convolve(flip(w), p))[n-1-i] ... this is getting confusing.

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a multi-method approach combining:
1. Frank-Wolfe (conditional gradient) with top-k subgradients
2. Multiplicative weights (mirror descent) 
3. LSE-smoothed gradient polishing with adaptive temperature
4. Multi-length exploration and structured restarts

The problem is reformulated on the probability simplex (sum(p)=1, p>=0)
where the objective becomes 2n*max(conv(p,p)), a scale-invariant formulation.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_topk_subgrad(p, c, k):
    """Subgradient of max(conv(p,p)) using top-k lags."""
    n = len(p)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    
    c_top = c[top_idx]
    spread = c_top[0] - c_top[-1]
    if spread > 1e-15:
        weights = (c_top - c_top[-1]) / spread
        weights = weights / weights.sum()
    else:
        weights = np.ones(k) / k
    
    grad = np.zeros(n)
    for ii in range(k):
        l = int(top_idx[ii])
        w = weights[ii] * 2.0
        i_start = max(0, l - n + 1)
        i_end = min(n - 1, l)
        if i_start <= i_end:
            idx = l - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += w * p[idx]
    return grad


def compute_lse_grad(p, c, beta):
    """Gradient of LSE-smoothed max via FFT cross-correlation."""
    n = len(p)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # grad[i] = 2 * sum_l w[l] * p[l-i]
    # = 2 * full_conv(w, p[::-1])[n-1 : 2n-1]
    q = p[::-1]
    conv_size = len(w) + n - 1  # (2n-1) + n - 1 = 3n-2
    fsize = 1 << (conv_size - 1).bit_length()
    W = fft(w, fsize)
    Q = fft(q, fsize)
    conv = np.real(ifft(W * Q))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad[:n]


def frank_wolfe_step(p, grad, step):
    """Take a Frank-Wolfe step: move toward the minimizer vertex."""
    min_idx = np.argmin(grad)
    v = np.zeros(len(p))
    v[min_idx] = 1.0
    return (1.0 - step) * p + step * v


def multiplicative_weights_step(p, grad, step):
    """Mirror descent step on the simplex (multiplicative weights)."""
    # p_new[i] = p[i] * exp(-step * grad[i]) / Z
    logits = np.log(p + 1e-300) - step * grad
    logits -= logits.max()  # for stability
    exp_logits = np.exp(logits)
    Z = exp_logits.sum()
    if Z < 1e-300:
        return p.copy()
    return exp_logits / Z


def optimize_fw(p0, max_time, rng, k_init=8, k_max=200, verbose=True):
    """
    Frank-Wolfe with top-k subgradient and line search.
    p0 is on the probability simplex.
    """
    n = len(p0)
    p = p0.copy()
    c = fft_convolve_self(p)
    best_val = c.max()
    best_p = p.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(p)
        cur = c.max()
        
        if cur < best_val - 1e-14:
            best_val = cur
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(k_init + (k_max - k_init) * progress)
        
        grad = compute_topk_subgrad(p, c, k)
        
        # Frank-Wolfe with line search
        min_idx = int(np.argmin(grad))
        v = np.zeros(n)
        v[min_idx] = 1.0
        
        # Line search: try geometric steps
        best_t = 0.0
        best_obj = cur
        t = 1.0
        for _ in range(8):
            p_try = (1.0 - t) * p + t * v
            c_try = fft_convolve_self(p_try)
            obj_try = c_try.max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t
            t *= 0.5
            if t < 0.01:
                break
        
        if best_t > 0:
            p = (1.0 - best_t) * p + best_t * v
        else:
            # Default FW step
            t_default = 2.0 / (it + 2.0)
            p = (1.0 - t_default) * p + t_default * v
        
        # Noise when stuck
        if stuck > 200:
            noise_scale = min(0.05, 0.01 * (stuck / 200.0))
            p = p + rng.normal(0, noise_scale / np.sqrt(n), n)
            p = np.maximum(p, 0)
            p = p / p.sum()
            stuck = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it={it}, t={elapsed:.0f}s, max_conv={best_val:.8f}")
    
    return best_p, best_val


def optimize_mw(p0, max_time, rng, k_init=16, k_max=300, verbose=True):
    """
    Multiplicative weights (mirror descent) with top-k subgradient.
    """
    n = len(p0)
    p = p0.copy()
    c = fft_convolve_self(p)
    best_val = c.max()
    best_p = p.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(p)
        cur = c.max()
        
        if cur < best_val - 1e-14:
            best_val = cur
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(k_init + (k_max - k_init) * progress ** 0.5)
        
        grad = compute_topk_subgrad(p, c, k)
        
        # Adaptive step size for mirror descent
        gnorm = np.linalg.norm(grad)
        step = 1.0 / np.sqrt(it + 1.0)
        step = min(step, 2.0 / max(gnorm, 1e-10))
        
        p_new = multiplicative_weights_step(p, grad, step)
        
        # Line search
        c_try = fft_convolve_self(p_new)
        if c_try.max() > cur * 1.01:
            # Try smaller step
            p_new = multiplicative_weights_step(p, grad, step * 0.5)
            c_try = fft_convolve_self(p_new)
            if c_try.max() > cur * 1.01:
                p_new = p  # Don't move
        
        p = p_new
        
        # Noise when stuck
        if stuck > 300:
            noise_scale = min(0.03, 0.005 * (stuck / 300.0))
            p = p + rng.normal(0, noise_scale / np.sqrt(n), n)
            p = np.maximum(p, 0)
            p = p / p.sum()
            stuck = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    MW it={it}, t={elapsed:.0f}s, max_conv={best_val:.8f}")
    
    return best_p, best_val


def optimize_lse(p0, max_time, rng, beta_init=100.0, beta_max=50000.0, verbose=True):
    """
    LSE-smoothed gradient descent for final polishing.
    """
    n = len(p0)
    p = p0.copy()
    c = fft_convolve_self(p)
    best_val = c.max()
    best_p = p.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(beta_max, beta_init * np.exp(0.005 * it))
        
        c = fft_convolve_self(p)
        cur = c.max()
        
        if cur < best_val - 1e-14:
            best_val = cur
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 500:
            noise_scale = 0.002 * (1.0 - progress) / np.sqrt(n)
            if noise_scale > 1e-10:
                p = p + rng.normal(0, noise_scale, n)
                p = np.maximum(p, 0)
                p = p / p.sum()
            stuck = 0
            continue
        
        grad = compute_lse_grad(p, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: for LSE, the function is (1/beta)-smooth
        step = 1.0 / (2.0 * beta)
        step = min(step, 1.0 / gnorm)
        
        # Try the step
        p_try = p - step * grad
        p_try = np.maximum(p_try, 0)
        p_try = p_try / p_try.sum()
        
        c_try = fft_convolve_self(p_try)
        if c_try.max() <= c.max() * (1.0 + 1e-10):
            p = p_try
        else:
            p_try = p - 0.5 * step * grad
            p_try = np.maximum(p_try, 0)
            p_try = p_try / p_try.sum()
            p = p_try
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    LSE it={it}, t={elapsed:.0f}s, beta={beta:.0f}, max_conv={best_val:.8f}")
    
    return best_p, best_val


def objective_value(p):
    """Compute 2n*max(conv(p,p)) for p on probability simplex."""
    n = len(p)
    c = fft_convolve_self(p)
    return 2.0 * n * c.max()
