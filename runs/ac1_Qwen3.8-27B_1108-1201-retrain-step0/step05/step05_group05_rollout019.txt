"""

Improvements over previous:
1. Backtracking line search in FISTA for better step sizes
2. Multi-active-lag subgradient (sharper than pure LSE)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Compute LSE-smoothed max."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    return cm + np.log(np.exp(scaled).sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution w.r.t. a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def multi_peak_gradient(a, c, k=8):
    """
    Gradient using top-k lags with weight proportional to excess above
    the (k+1)-th largest value. This gives a sharper gradient than LSE
    when the top few lags dominate.
    """
    n = len(a)
    k = min(k, len(c))
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[min(k, len(c)) - 1] if k < len(c) else sorted_c[-1]
    excess = np.maximum(c - threshold, 0.0)
    wsum = excess.sum()
    if wsum < 1e-30:
        # Fall back to single peak
        idx = np.argmax(c)
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0
    else:
        w_full = excess / wsum
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def mirror_descent_step(a, grad, eta, S):
    """KL-proximal (mirror descent) step."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled = -np.clip(eta * grad / g_max, -30, 30)
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_fista_bt(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
                 beta_max=50000.0, use_multi_peak=False, mp_k=16,
                 restart_threshold=3000, perturbation_period=4000):
    """
    FISTA with backtracking line search and adaptive LSE/multi-peak gradient.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    lse_prev = lse_value(c0, beta_init)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if use_multi_peak:
            grad = multi_peak_gradient(y, c, k=mp_k)
            L_bound = 4.0 * beta  # rough Lipschitz constant
        else:
            grad = lse_gradient(y, c, beta)
            L_bound = 4.0 * beta

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L_bound
        x_new = project_simplex(y - step * grad, S)
        c_new = fft_convolve_self(x_new)
        lse_new = lse_value(c_new, beta)

        # Backtrack if no improvement in LSE
        bt_count = 0
        while lse_new > lse_prev * (1 + 1e-10) and bt_count < 5 and step > 1e-10:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, beta)
            bt_count += 1

        lse_prev = lse_new

        # Stuck detection
        if it - last_improve > restart_threshold:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 15000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            else:
                y = fourier_phase_mutate(y, rng, 0.05)
            t_mom = 1.0
            last_improve = it
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (10.0 * perturbation_period))
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_peak_equalization(a_init, S, max_time, rng):
    """
    Peak equalization: iteratively target the current maximum lag,
    trying to flatten the top of the auto-correlation.
    Uses a "water-filling" target level that decreases over time.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    y = a.copy()
    it = 0
    target = c0.max() * 0.999
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
            # Lower the target when we improve
            target = min(target, c.max() * (1.0 - 0.001 * (1 - progress)))

        # Stuck handling
        if it - last_improve > 2000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            target = c.max() * 1.001
            continue

        # Find lags above target
        above = c > target
        if not np.any(above):
            # All below target: lower target
            target = c.max() * 0.998
            continue

        # Use multi-peak gradient on the above-target lags
        k = min(32, int(above.sum()) + 1)
        grad = multi_peak_gradient(y, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Adaptive step size
        step = 0.5 * S / (n * gnorm / (S / n) + n * 0.01)
        step = min(step, S / (n * 0.1))

        x_new = project_simplex(y - step * grad, S)
        c_check = fft_convolve_self(x_new)
        if c_check.max() > c.max() * 1.01:
            step *= 0.1
            x_new = project_simplex(y - step * grad, S)

        y = x_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000)
        if len(g) == n and g.sum() > 0:
            inits.append(g.copy())
        elif g.size >= 64 and g.sum() > 0:
            # Resample to length n
            from numpy import interp
            x_old = np.linspace(0, 1, len(g))
            x_new = np.linspace(0, 1, n)
            resampled = interp(x_new, x_old, g)
            inits.append(np.clip(resampled, 0, 1000))

    # Windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4 * np.pi * tt + 0.5), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)

    # Asymmetric windows (important for self-convolution!)
    inits.append(np.maximum(0.3 + 0.7 * tt, 0.01))  # linear ramp up
    inits.append(np.maximum(0.3 + 0.7 * (1 - tt), 0.01))  # linear ramp down
    inits.append(np.maximum(0.2 + 0.8 * tt ** 2, 0.01))  # quadratic ramp
    inits.append(np.maximum(0.2 + 0.8 * (1 - tt) ** 2, 0.01))

    # Bimodal structures
    inits.append(np.maximum(0.5 * np.exp(-20 * (tt - 0.25) ** 2) + 0.5 * np.exp(-20 * (tt - 0.75) ** 2) + 0.05, 0.01))
    inits.append(np.maximum(0.3 * np.exp(-10 * (tt - 0.3) ** 2) + 0.3 * np.exp(-10 * (tt - 0.6) ** 2) + 0.3 * np.exp(-10 * (tt - 0.9) ** 2) + 0.02, 0.01))

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

    # Block structures
    for nb in [4, 8, 16]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def objective_value(a, n):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def optimize_for_n(n, S, time_budget, rng, given_seq=None):
    """Full optimization pipeline for a given n."""
    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            return True
        return False

    t_start = time.monotonic()
    print("  n=%d, budget=%.0fs" % (n, time_budget))

    # Phase 1: Quick screening
    p1_budget = min(60.0, time_budget * 0.1)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(6, len(inits))
    per_init = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        if time.monotonic() - t_start > time_budget * 0.85:
            break
        t_alloc = min(per_init, time_budget * 0.85 - (time.monotonic() - t_start))
        if t_alloc < 5:
            break
        a_opt, val = run_fista_bt(inits[i], S, t_alloc, rng,
                                   beta_init=5.0, beta_rate=0.01, beta_max=5000)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("    Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping
    p2_budget = (time_budget - (time.monotonic() - t_start)) * 0.35
    if time.monotonic() - t_start < time_budget * 0.7:
        bh_start = candidates[0][1] if candidates else inits[0]
        a_bh, val_bh = run_basin_hopping_v2(bh_start, S, p2_budget, rng)
        update_global(a_bh, val_bh)
        print("    Best after BH: obj=%.6f" % global_best)

    # Phase 3: Extended polish
    remaining = time_budget - (time.monotonic() - t_start) - 5
    if remaining > 30 and global_a is not None:
        a_pol, val_pol = run_extended_polish(global_a, S, remaining, rng)
        update_global(a_pol, val_pol)
        print("    Final obj for n=%d: %.6f" % (n, global_best))

    return global_a, global_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Try multiple sequence lengths (4096, 8192, 16384)
    - For each: screen -> basin hop -> extended polish
    - Return the best overall
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    # Try multiple n values, allocating time proportionally
    # Prioritize n=8192 (where we know we're close to target)
    # but also try n=16384 for potential improvement
    n_options = [8192, 16384, 4096]
    # Time allocation: 50% for n=8192, 35% for n=16384, 15% for n=4096
    time_fracs = [0.50, 0.35, 0.15]

    overall_best = np.inf
    overall_a = None

    for i, n in enumerate(n_options):
        t_alloc = time_fracs[i] * (budget_s - 60)
        # Check remaining time
        elapsed = time.monotonic() - total_start
        if elapsed > budget_s - 60:
            break
        t_alloc = min(t_alloc, budget_s - 60 - elapsed)
        if t_alloc < 30:
            break

        S = np.sqrt(2.0 * n)
        # Use a separate RNG stream for each n
        rng_n = np.random.default_rng(seed + i * 1000)

        print("\n=== Optimizing with n=%d (budget=%.0fs) ===" % (n, t_alloc))
        a_best, obj_best = optimize_for_n(n, S, t_alloc, rng_n, given_seq=given)

        if a_best is not None and obj_best < overall_best:
            overall_best = obj_best
            overall_a = a_best.copy()
            print("  New overall best: obj=%.6f (n=%d)" % (overall_best, n))

    # Final verification and return
    if overall_a is not None:
        overall_a = np.clip(overall_a, 0.0, 1000.0)
        if overall_a.sum() < 0.01:
            n = len(overall_a)
            overall_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in overall_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n = 8192
            overall_a = np.ones(n) * np.sqrt(2.0 * n) / n
            final_list = [float(x) for x in overall_a]
    else:
        n = 8192
        overall_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in overall_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, overall_best, len(overall_a)))
    return [float(x) for x in np.clip(overall_a, 0.0, 1000.0)]
