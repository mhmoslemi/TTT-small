By minimax theorem (if we can interchange min and max, which requires convexity in a and concavity in lambda - the function is linear in lambda but quadratic in a, so we can't directly apply Sion's theorem).

But if we fix lambda, the inner problem is: min_a sum_k lambda[k] * (a*a)[k] s.t. a >= 0, sum(a) = S.

This is a Quadratic Program (QP)! The objective is `a^T (sum_k lambda[k] Q_k) a` where Q_k is the lag-k matrix. The matrix `Q = sum_k lambda[k] Q_k` is PSD (as a positive combination of... wait, Q_k is not necessarily PSD as we showed earlier).

Hmm, so the QP might not be convex. Let me check: `Q_k[i,j] = 1` if `i+j=k`. For k=0: Q_0 = [[1,0,...],[0,0,...],...] which has eigenvalues 1 and -1 (for n>1... actually for n>1, Q_0 = I_1 ⊕ 0, so it's PSD). Wait, Q_0[0,0] = 1 and all other entries are 0. So Q_0 is PSD (it's a rank-1 PSD matrix).

For k=1 (n≥2): Q_1[0,1] = Q_1[1,0] = 1, all other entries 0. Eigenvalues: 1, -1, 0, ..., 0. NOT PSD.

So the QP is not convex in general. This means the dual approach doesn't directly work.

However, if we restrict to symmetric sequences (a[i] = a[n-1-i]), then Q_k becomes a different structure. For symmetric a, (a*a)[k] has a specific form...

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-phase optimization with:
- LSE surrogate with adaptive beta and backtracking
- Active-set subgradient polishing
- Nesterov acceleration with momentum restarts
- Chirp-based alternative initialization
- Multi-scale exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute LSE of c with temperature beta."""
    cm = c.max()
    if beta * cm > 700:  # Numerical stability
        shifted = beta * (c - cm)
    else:
        shifted = beta * c
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    return cm + np.log(e.sum()) / beta  # Approximate: (1/beta) * log(sum(exp(beta*c)))


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lag(y, lag):
    """Exact subgradient of (a*a)[lag] w.r.t. a."""
    n = len(y)
    w = np.zeros(len(y) + n - 1)
    w[lag] = 1.0
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k):
    """Subgradient of sum of top-k elements of c."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value 2n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_chirp_init(n, freq=1):
    """Create a chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    # Quadratic chirp, made non-negative
    a = 1.0 + 0.8 * np.cos(2 * np.pi * freq * t * t / n)
    a = np.maximum(a, 0.0)
    return a


def make_cosine_init(n):
    """Cosine window initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * t)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_lse_fista(a_init, max_time, rng, beta_start=5.0, beta_end=500.0, 
                  use_backtrack=True, momentum_restart=5000):
    """LSE-optimized FISTA with adaptive beta and optional backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Precompute FFT of y (will update as y changes)
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta: ramp up over time, accelerate when stuck
        progress = min(1.0, it / max(it, 1000))
        if stuck < 200:
            beta = beta_start * (beta_end / beta_start) ** progress
        elif stuck < 2000:
            beta = min(beta_end * 5, beta * 1.005)
        else:
            beta = min(beta_end * 10, beta * 1.01)
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: start with 1/beta, use backtracking if needed
        step = 1.0 / max(beta, 1.0)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        
        if use_backtrack:
            # Try the step, check if LSE decreased
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            lse_trial = lse_value(c_trial, beta)
            lse_cur = lse_value(c, beta)
            
            # Backtrack if needed
            attempts = 0
            while lse_trial > lse_cur + 1e-12 and attempts < 10 and step > 1e-12:
                step *= 0.5
                x_trial = project_simplex(y - step * grad, S)
                c_trial = fft_convolve_self(x_trial)
                lse_trial = lse_value(c_trial, beta)
                attempts += 1
            
            x_new = x_trial
        else:
            # Fixed step (faster but less robust)
            L_bound = 2.0 * beta * np.max(y) + 4.0 * beta * c.max() / n
            step = min(step, 0.5 / max(L_bound, 1.0))
            step = max(step, 1e-12)
            x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Momentum restart
        if stuck > momentum_restart or it % (momentum_restart * 2) == 0:
            t_mom = 1.0
        
        # Perturbation when very stuck
        if stuck > 10000 and it % 500 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            stuck = 0
            last_improve = it
    
    return best_a


def run_active_polish(a_init, max_time, rng, start_k=1, end_k=20):
    """Active-set polishing with increasing k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    current_k = start_k
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Increase k as we get stuck (more active constraints)
        if stuck > 200:
            current_k = min(end_k, start_k + (stuck - 200) // 100)
        
        # Compute subgradient from active set
        if current_k == 1:
            grad = subgradient_at_lag(y, int(np.argmax(c)))
        else:
            grad = topk_subgradient(y, c, current_k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = min(2.0 / gnorm, S / (gnorm + 1e-10))
        
        # Try the step
        x_trial = project_simplex(y - step * grad, S)
        c_trial = fft_convolve_self(x_trial)
        
        if c_trial.max() <= cur_max + 1e-12:
            x_new = x_trial
        else:
            # Backtrack
            for _ in range(15):
                step *= 0.5
                x_trial = project_simplex(y - step * grad, S)
                c_trial = fft_convolve_self(x_trial)
                if c_trial.max() <= cur_max + 1e-12:
                    break
            x_new = x_trial
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart
        if stuck > 5000:
            t_mom = 1.0
            if stuck > 20000:
                noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                stuck = 0
                last_improve = it
    
    return best_a


def run_direct_max_min(a_init, max_time, rng):
    """Direct optimization targeting the max element specifically.
    Uses a nested approach: for the current argmax lag, do a QP-like update."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
        
        # Find the argmax lag
        lag = int(np.argmax(c))
        
        # Compute subgradient
        grad = subgradient_at_lag(a, lag)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: we want to reduce (a*a)[lag] by a certain amount
        # (a*a)[lag] is quadratic, so a step of size s reduces it by ~s*gnorm
        # Target: reduce max by 1%
        target_reduction = 0.01 * cur_max
        step = min(target_reduction / (gnorm + 1e-10), S / (gnorm + 1e-10) * 0.1)
        step = max(step, 1e-10)
        
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1, 0.01]:
                a_try = project_simplex(a - step * factor * grad, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max:
                    a = a_try
                    break
            else:
                # No improvement, add small perturbation
                if it % 1000 == 0:
                    noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
    
    return best_a

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c))) with numerical stability."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    return cm + np.log(e.sum()) / beta


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence y."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_lags(y, lags):
    """Subgradient of average of (a*a)[k] for k in lags."""
    n = len(y)

    fsize = 1 << (len(w) + n - 1).bit_length()

        else:
            w = w / wsum

    w_full[idx] = w

    A = fft(y, fsize)

    print("Best init: %s (obj=%.6f)" % (best_init_name, best_init_obj), flush=True)
    a_current = inits[best_init_name].copy()

    # === Phase 1: LSE FISTA warmup (20% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20
    if p1_budget > 15:
        print("\n=== Phase 1: LSE warmup (%.0fs) ===" % p1_budget, flush=True)
        t1 = time.monotonic()
        a_p1 = run_lse_fista(a_current, p1_budget, rng,
                             beta_start=3.0, beta_end=100.0,
                             use_backtrack=True, momentum_restart=3000)
        obj_p1 = compute_obj(a_p1)
        print("  Warmup: obj=%.6f (%.1fs)" % (obj_p1, time.monotonic() - t1), flush=True)
        if obj_p1 < global_best_obj:
            global_best_obj = obj_p1
            global_best_a = a_p1.copy()
        a_current = global_best_a.copy()

    # === Phase 2: LSE refinement (25% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25
    if p2_budget > 15:
        print("\n=== Phase 2: LSE refinement (%.0fs) ===" % p2_budget, flush=True)
        t2 = time.monotonic()
        a_p2 = run_lse_fista(a_current, p2_budget, rng,
                             beta_start=50.0, beta_end=2000.0,
                             use_backtrack=True, momentum_restart=5000,
                             interleave_exact=True, exact_every=30)
        obj_p2 = compute_obj(a_p2)
        print("  Refine: obj=%.6f (%.1fs)" % (obj_p2, time.monotonic() - t2), flush=True)
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()
        a_current = global_best_a.copy()

    # === Phase 3: Active-set polish (25% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    if p3_budget > 15:
        print("\n=== Phase 3: Active polish (%.0fs) ===" % p3_budget, flush=True)
        t3 = time.monotonic()
        a_p3 = run_active_polish(a_current, p3_budget, rng,
                                 start_k=1, end_k=32, epsilon_ramp=True)
        obj_p3 = compute_obj(a_p3)
        print("  Polish: obj=%.6f (%.1fs)" % (obj_p3, time.monotonic() - t3), flush=True)
        if obj_p3 < global_best_obj:
            global_best_obj = obj_p3
            global_best_a = a_p3.copy()
        a_current = global_best_a.copy()

    # === Phase 4: Interleaved local search (15% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15
    if p4_budget > 10:
        print("\n=== Phase 4: Interleaved (%.0fs) ===" % p4_budget, flush=True)
        t4 = time.monotonic()
        a_p4 = run_interleaved(a_current, p4_budget, rng, beta=200.0)
        obj_p4 = compute_obj(a_p4)
        print("  Interleave: obj=%.6f (%.1fs)" % (obj_p4, time.monotonic() - t4), flush=True)
        if obj_p4 < global_best_obj:
            global_best_obj = obj_p4
            global_best_a = a_p4.copy()
        a_current = global_best_a.copy()

    # === Phase 5: Final high-beta LSE (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final LSE (%.0fs) ===" % remaining, flush=True)
        t5 = time.monotonic()
        a_p5 = run_lse_fista(a_current, remaining * 0.6, rng,
                             beta_start=500.0, beta_end=20000.0,
                             use_backtrack=True, momentum_restart=2000)
        obj_p5 = compute_obj(a_p5)
        print("  Final LSE: obj=%.6f (%.1fs)" % (obj_p5, time.monotonic() - t5), flush=True)
        if obj_p5 < global_best_obj:
            global_best_obj = obj_p5
            global_best_a = a_p5.copy()

        rem5 = remaining - (time.monotonic() - t5)
        if rem5 > 15:
            a_p5b = run_active_polish(global_best_a, rem5 * 0.8, rng,
                                       start_k=1, end_k=8, epsilon_ramp=False)
            obj_p5b = compute_obj(a_p5b)
            print("  Final polish: obj=%.6f" % obj_p5b, flush=True)
            if obj_p5b < global_best_obj:
                global_best_obj = obj_p5b
                global_best_a = a_p5b.copy()

    # === Phase 6: Try alternative initialization (if time) ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        # Try from the second-best initialization
        alt_names = [nm for nm in inits if nm != best_init_name]
        if alt_names:
            # Pick the second-best
            alt_objs = [(compute_obj(inits[nm]), nm) for nm in alt_names]
            alt_objs.sort()
            alt_name = alt_objs[0][1]
            print("\n=== Phase 6: Alt init '%s' (%.0fs) ===" % (alt_name, remaining), flush=True)
            t6 = time.monotonic()
            a_alt = run_lse_fista(inits[alt_name], remaining * 0.4, rng,
                                  beta_start=5.0, beta_end=500.0,
                                  use_backtrack=True, momentum_restart=3000)
            obj_alt = compute_obj(a_alt)
            print("  Alt: obj=%.6f (%.1fs)" % (obj_alt, time.monotonic() - t6), flush=True)
            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_alt.copy()

            rem6 = remaining - (time.monotonic() - t6)
            if rem6 > 30:
                a_alt2 = run_active_polish(global_best_a, rem6 * 0.7, rng,
                                           start_k=1, end_k=16)
                obj_alt2 = compute_obj(a_alt2)
                if obj_alt2 < global_best_obj:
                    global_best_obj = obj_alt2
                    global_best_a = a_alt2.copy()

    # === Phase 7: Multi-scale upsample (if time and n is small enough) ===
    remaining = deadline - time.monotonic()
    if remaining > 250 and global_best_a is not None and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 7: Upsample to n=%d (%.0fs) ===" % (n2, remaining), flush=True)
        t7 = time.monotonic()
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)

        a_up1 = run_lse_fista(a_up, remaining * 0.4, rng,
                               beta_start=5.0, beta_end=200.0,
                               use_backtrack=True, momentum_restart=3000)
        obj_up1 = compute_obj(a_up1)
        print("  Upsample LSE: obj=%.6f (%.1fs)" % (obj_up1, time.monotonic() - t7), flush=True)
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        rem7 = remaining - (time.monotonic() - t7)
        if rem7 > 40:
            a_up2 = run_active_polish(global_best_a, rem7 * 0.7, rng,
                                       start_k=1, end_k=16)
            obj_up2 = compute_obj(a_up2)
            print("  Upsample polish: obj=%.6f" % obj_up2, flush=True)
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # Final output
    print("\n=== SUMMARY ===", flush=True)
    print("Best objective: %.6f, n=%d" % (global_best_obj, len(global_best_a) if global_best_a is not None else 0), flush=True)
    print("Total time: %.1fs" % (time.monotonic() - total_start), flush=True)

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver, flush=True)
        else:
            print("WARNING: evaluator rejected, using fallback", flush=True)
            n_fb = 8192
            final_list = [float(x) for x in np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)]
    else:
        n_fb = 8192
        final_list = [float(x) for x in np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)]

    return final_list
