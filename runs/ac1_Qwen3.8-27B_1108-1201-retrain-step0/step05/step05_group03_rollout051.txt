Strategy: Multi-n optimization with adaptive FISTA, subgradient polishing,
and spectral diversification. Tries n=4096, 8192, 16384 with multiple

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Prevent overflow
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c_top - cm).max())
        if beta > 0:
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c_top - cm).max())
        if beta > 0:
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def subgradient_max(a, c):
    """Subgradient of max(c) = gradient of c[argmax(c)]."""
    n = len(a)
    k_max = np.argmax(c)
    w_full = np.zeros(len(c))
    w_full[k_max] = 1.0

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, k):
    """Average subgradient of top-k max elements."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src + 1)
    a_padded = np.append(a, 0)  # pad with 0 at the end
    t_new = np.linspace(0, 1, n_target + 1)
    a_resampled = np.interp(t_new, t_old, a_padded)[:-1]
    return a_resampled
