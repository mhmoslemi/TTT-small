Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-phase optimization: LSE-FISTA with backtracking, 
active-set subgradient refinement, and symmetry exploitation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value_and_gradient(a, c, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm + np.log(n) / beta, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    val = cm + np.log(tot) / beta
    return val, grad


def subgradient_from_active_set(a, c, k_active):
    """Compute subgradient from active lags."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    w_full = np.zeros(len(c))
    for k in k_active:
        w_full[k] = 1.0
    w_full /= len(k_active)
    W = fft(w_full, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, S, max_time, rng, 
                  beta_init=5.0, beta_rate=0.01, beta_max=50000.0,
                  use_backtracking=True, stall_patience=2000):
    """
    FISTA on LSE-smoothed objective with adaptive beta and backtracking.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time
        
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it
            if best < 1.5027:
                break
        
        val, grad = lse_value_and_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        if use_backtracking:
            # Start with a reasonable step
            step = 1.0 / (beta * 2.0 * y.max() * 1.5 + 1e-10)
            # Backtracking: ensure function value decreases
            ftol = 1e-4
            for _ in range(10):
                x_try = project_simplex(y - step * grad, S)
                c_try = fft_convolve_self(x_try)
                val_try, _ = lse_value_and_gradient(x_try, c_try, beta)
                if val_try <= val - ftol * step * gnorm ** 2:
                    break
                step *= 0.5
        else:
            step = 1.0 / (2.0 * beta * max(y.max(), 0.1) + 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Stuck detection
        if it - last_improve > stall_patience:
            # Perturb
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
        
        # Periodic small perturbation
        if it % 5000 == 0:
            decay = np.exp(-progress * 3.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

    t_mom = 1.0
    start = time.monotonic()

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Multi-phase approach: LSE-FISTA with backtracking line search,
active-set subgradient with Nesterov momentum, symmetry exploitation,
and peak-breaking refinements. Different from parent by using
backtracking, multi-n search, and targeted active-set methods.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value_and_gradient(a, c, beta):
    """Compute LSE-smoothed value and gradient of max(c)."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm + np.log(len(c)) / beta, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    val = cm + np.log(tot) / beta
    return val, grad


def subgradient_active(a, c, active_idx):
    """Subgradient from active set of lags."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    w_full = np.zeros(len(c))
    for k in active_idx:
        w_full[k] = 1.0
    w_full /= len(active_idx)
    W = fft(w_full, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista_bt(a_init, S, max_time, rng,
                     beta_init=5.0, beta_rate=0.012, beta_max=80000.0,
                     stall_patience=1500, perturbation_period=4000):
    """
    FISTA with LSE smoothing and backtracking line search.
    Returns (best_a, best_max_b).
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * beta_init * max(a.max(), 0.1))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        val, grad = lse_value_and_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        ftol = 1e-4
        accepted = False
        for _ in range(8):
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            val_try, _ = lse_value_and_gradient(x_try, c_try, beta)
            if val_try <= val - ftol * step * gnorm * gnorm:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Use the last try anyway
            x_new = project_simplex(y - step * grad, S)
        else:
            x_new = x_try

        # Restore step for next iteration (don't keep shrinking)
        step = min(step * 1.5, 1.0 / (2.0 * beta * max(y.max(), 0.1) + 1e-10))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Stuck detection
        if it - last_improve > stall_patience:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, max(noise_scale, 1e-12), n), S)
            t_mom = 1.0
            last_improve = it

        # Periodic perturbation
        if perturbation_period > 0 and it % perturbation_period == 0:
            elapsed = time.monotonic() - start
            decay = np.exp(-2.0 * elapsed / max_time)
            noise_scale = 0.004 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_active_set_refine(a_init, S, max_time, rng,
                          tol_frac=0.005, max_active=40,
                          momentum_beta=0.7, step_scale=0.5):
    """
    Active-set subgradient with Nesterov momentum and adaptive step.
    Directly targets reducing max_b.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    momentum = np.zeros(n)
    step = step_scale * S / (n * 5.0)
    it = 0
    last_improve = 0
    y = a.copy()  # Nesterov lookahead

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        # Find active set
        tol = tol_frac * cmax + 1e-12
        active = np.where(c >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        active = active[:max_active]

        # Subgradient
        grad = subgradient_active(y, c, active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Nesterov momentum
        momentum = momentum_beta * momentum + grad
        dir_norm = np.linalg.norm(momentum)
        if dir_norm < 1e-15:
            break

        # Adaptive step: target a reduction of ~1% of cmax
        step_eff = step * gnorm / (dir_norm * (0.01 * cmax + 1e-12))
        step_eff = min(step_eff, S / (n * 0.5))
        step_eff = max(step_eff, 1e-10)

        x_new = project_simplex(y - step_eff * momentum / dir_norm * dir_norm, S)
        # Simplify: x_new = project_simplex(y - step_eff * momentum, S)
        x_new = project_simplex(y - step_eff * momentum, S)

        # Check improvement with line search
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cmax * 1.001:
            y = x_new
        else:
            # Half step
            x_half = project_simplex(y - 0.5 * step_eff * momentum, S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < cmax:
                y = x_half
            else:
                step *= 0.7
                momentum = np.zeros(n)

        # Slow step decay
        step *= 0.9998

        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            momentum = np.zeros(n)
            last_improve = it
            tol_frac = max(0.001, tol_frac * 0.5)

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_symmetric_opt(a_init, S, max_time, rng,
                      beta_init=10.0, beta_rate=0.01, beta_max=50000.0):
    """
    Optimize with symmetry constraint a[j] = a[n-1-j].
    """
    n = len(a_init)
    if n % 2 != 0:
        n = n - 1
        a_init = a_init[:n]

    half = n // 2

    # Extract symmetric parameterization
    x = 0.5 * (a_init[:half] + a_init[half:][::-1])

    def x_to_a(xv):
        return np.concatenate([xv, xv[::-1]])

    def x_from_a(av):
        return 0.5 * (av[:half] + av[half:][::-1])

    def project_x(vx, S_total):
        ax = x_to_a(vx)
        ap = project_simplex(ax, S_total)
        return x_from_a(ap)

    a0 = x_to_a(x)
    s = a0.sum()
    if s < 1e-15:
        x = np.full(half, S / n)
    else:
        x = x * (S / s)

    y = x.copy()
    t_mom = 1.0
    start = time.monotonic()

    a_y = x_to_a(y)
    c = fft_convolve_self(a_y)
    best = c.max()
    best_a = a_y.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * beta_init * max(a_y.max(), 0.1))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        a_y = x_to_a(y)
        c = fft_convolve_self(a_y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = a_y.copy()
            last_improve = it

        val, grad_full = lse_value_and_gradient(a_y, c, beta)
        # Symmetrize gradient
        grad = 0.5 * (grad_full[:half] + grad_full[half:][::-1])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking
        ftol = 1e-4
        for _ in range(6):
            x_try = project_x(y - step * grad, S)
            a_try = x_to_a(x_try)
            c_try = fft_convolve_self(a_try)
            val_try, _ = lse_value_and_gradient(a_try, c_try, beta)
            if val_try <= val - ftol * step * gnorm * gnorm:
                break
            step *= 0.5

        x_new = project_x(y - step * grad, S)
        step = min(step * 1.4, 1.0 / (2.0 * beta * max(a_y.max(), 0.1) + 1e-10))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new
        t_mom = t_new

        if it - last_improve > 2000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_x(y + rng.normal(0, noise_scale, half), S)
            t_mom = 1.0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def peak_break_step(a, S, n_steps=200):
    """
    Try to reduce peak by redistributing mass away from peak-contributing positions.
    """
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    peak_lag = np.argmax(c)

    # Contributions to peak: c[peak_lag] = sum_i a[i]*a[peak_lag-i]
    # Find pairs with large contributions
    contribs = np.zeros(n)
    for i in range(n):
        j = peak_lag - i
        if 0 <= j < n:
            contribs[i] = a[i] * a[j]

    # Sort by contribution
    order = np.argsort(contribs)[::-1]

    # Try moving mass from high-contributing positions to low-contributing ones
    a_new = a.copy()
    for _ in range(n_steps):
        # Pick a high-contributing position
        hi = order[rng.integers(len(order))]
        # Pick a low-contributing position
        lo = order[rng.integers(len(order) // 2, len(order))]
        if hi == lo:
            continue

        # Move small amount
        delta = 0.01 * a[hi]
        if delta < 1e-15:
            continue
        a_new[hi] -= delta
        a_new[lo] += delta
        if a_new[hi] < 0:
            a_new[lo] -= a_new[hi]
            a_new[hi] = 0

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)

    return a_new


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000).astype(np.float64)
        if len(g) >= n:
            g = g[:n]
        elif len(g) > 0 and len(g) >= 64:
            old_t = np.linspace(0, 1, len(g))
            new_t = np.linspace(0, 1, n)
            g = np.interp(new_t, old_t, g)
        if len(g) == n and g.sum() > 0:
            inits.append(g.copy())

    # Various smooth windows
    inits.append(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2.0 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * tt) * 0.5 + 0.01, 0.01))
    inits.append(np.exp(-6.0 * (tt - 0.5) ** 2) + 0.01)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(6.0 * np.pi * tt + 0.7)
                            + 0.15 * np.cos(14.0 * np.pi * tt + 0.3), 0.01))
    inits.append(np.maximum(1.0 + 0.2 * np.sin(10.0 * np.pi * tt)
                            + 0.1 * np.cos(20.0 * np.pi * tt + 1.0), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.5, 0.3, n)) + 0.05)

    # Chirp-inspired
    inits.append(np.abs(np.cos(2.0 * np.pi * 3.0 * tt + np.pi * 5.0 * tt * tt)) + 0.01)

    return inits


def objective_value(a):
    """Compute objective: 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Screen diverse inits with LSE-FISTA (backtracking)
    2. Refined high-beta FISTA from best
    3. Symmetric optimization
    4. Active-set subgradient refinement
    5. Peak-breaking micro-optimization
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Phase 1: Quick screening
    print("\n=== Phase 1: LSE-FISTA screening ===")
    p1_start = time.monotonic()
    p1_budget = (deadline - p1_start) * 0.25

    n_values = [8192, 12288, 6144]
    candidates = []

    for n in n_values:
        if time.monotonic() > deadline - (deadline - total_start) * 0.55:
            break
        S = np.sqrt(2.0 * n)
        remaining = min(p1_budget * (n_values.index(n) + 1),
                        deadline - time.monotonic() - 150)
        if remaining < 40:
            break

        inits = generate_inits(n, rng, given_seq=given)
        n_screen = min(5, len(inits))
        per_init = remaining / (n_screen + 2)

        for i in range(n_screen):
            if time.monotonic() > deadline - (deadline - total_start) * 0.5:
                break
            t_alloc = min(per_init, deadline - time.monotonic() - 120)
            if t_alloc < 20:
                break

            a_opt, val = run_lse_fista_bt(inits[i], S, t_alloc, rng,
                                          beta_init=3.0, beta_rate=0.02,
                                          beta_max=30000.0,
                                          stall_patience=1000)
            obj = 2.0 * n * val / (a_opt.sum() ** 2)
            candidates.append((obj, a_opt.copy(), n))
            print("  n=%d init %d: obj=%.8f" % (n, i, obj))

            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("    *** NEW GLOBAL BEST: %.8f ***" % obj)

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: %.8f" % global_best)

    # Phase 2: Refined high-beta FISTA
    print("\n=== Phase 2: High-beta refinement ===")
    p2_start = time.monotonic()
    p2_budget = (deadline - p2_start) * 0.35

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # 2a: Standard high-beta FISTA
        t_alloc = min(p2_budget * 0.6, deadline - time.monotonic() - 120)
        print("  2a: High-beta FISTA (n=%d), t=%.0fs" % (n_best, t_alloc))
        if t_alloc > 30:
            a_hb, val_hb = run_lse_fista_bt(global_a, S_best, t_alloc, rng,
                                            beta_init=50.0, beta_rate=0.004,
                                            beta_max=200000.0,
                                            stall_patience=2000)
            obj_hb = 2.0 * n_best * val_hb / (a_hb.sum() ** 2)
            print("    obj=%.8f" % obj_hb)
            if obj_hb < global_best:
                global_best = obj_hb
                global_a = a_hb.copy()
                print("    *** NEW GLOBAL BEST: %.8f ***" % obj_hb)

        # 2b: Symmetric optimization
        t_alloc = min(p2_budget * 0.4, deadline - time.monotonic() - 60)
        print("  2b: Symmetric optimization, t=%.0fs" % t_alloc)
        if t_alloc > 20 and n_best % 2 == 0:
            a_sym, val_sym = run_symmetric_opt(global_a, S_best, t_alloc, rng,
                                               beta_init=20.0, beta_rate=0.006,
                                               beta_max=80000.0)
            obj_sym = 2.0 * n_best * val_sym / (a_sym.sum() ** 2)
            print("    obj=%.8f" % obj_sym)
            if obj_sym < global_best:
                global_best = obj_sym
                global_a = a_sym.copy()
                print("    *** NEW GLOBAL BEST: %.8f ***" % obj_sym)

    # Phase 3: Active-set subgradient refinement
    print("\n=== Phase 3: Active-set refinement ===")
    p3_start = time.monotonic()
    p3_budget = deadline - p3_start - 60

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_alloc = min(p3_budget, deadline - time.monotonic() - 40)
        print("  Active-set subgrad (n=%d), t=%.0fs" % (n_best, t_alloc))
        if t_alloc > 20:
            a_as, val_as = run_active_set_refine(global_a, S_best, t_alloc, rng,
                                                 tol_frac=0.003, max_active=30,
                                                 momentum_beta=0.5, step_scale=0.3)
            obj_as = 2.0 * n_best * val_as / (a_as.sum() ** 2)
            print("    obj=%.8f" % obj_as)
            if obj_as < global_best:
                global_best = obj_as
                global_a = a_as.copy()
                print("    *** NEW GLOBAL BEST: %.8f ***" % obj_as)

    # Phase 4: Peak-breaking micro-optimization
    print("\n=== Phase 4: Peak-breaking ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            c = fft_convolve_self(global_a)
            cmax = c.max()
            peak_lag = int(np.argmax(c))
            print("  Peak at lag %d, value=%.8f" % (peak_lag, cmax))

            # Try multiple peak-breaking attempts
            n_attempts = max(3, int(remaining / 5))
            best_pb = global_a.copy()
            best_pb_val = cmax

            for att in range(n_attempts):
                if time.monotonic() > deadline - 15:
                    break
                a_pb = peak_break_step(global_a, S_best, n_steps=100)
                c_pb = fft_convolve_self(a_pb)
                if c_pb.max() < best_pb_val:
                    best_pb_val = c_pb.max()
                    best_pb = a_pb.copy()
                    obj_pb = 2.0 * n_best * best_pb_val / (S_best ** 2)
                    print("    Attempt %d: max_b=%.8f (obj=%.8f)" %
                          (att + 1, best_pb_val, obj_pb))

            obj_pb = 2.0 * n_best * best_pb_val / (best_pb.sum() ** 2)
            if obj_pb < global_best:
                global_best = obj_pb
                global_a = best_pb.copy()
                print("    *** NEW GLOBAL BEST: %.8f ***" % obj_pb)

    # Final: quick FISTA polish
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print("\n=== Final polish (%.0fs) ===" % remaining)
            a_final, val_final = run_lse_fista_bt(global_a, S_best, remaining, rng,
                                                  beta_init=500.0, beta_rate=0.002,
                                                  beta_max=500000.0,
                                                  stall_patience=3000)
            obj_final = 2.0 * n_best * val_final / (a_final.sum() ** 2)
            print("  Final obj=%.8f" % obj_final)
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_final.copy()

    # Verification and output
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            n = len(global_a)
            global_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if np.isfinite(ver):
            if ver < global_best:
                global_best = ver
        else:
            print("  evaluate_sequence rejected! Fallback.")
            n = 8192
            global_a = np.ones(n) * np.sqrt(2.0 * n) / n
            final_list = [float(x) for x in global_a]
    else:
        n = 8192
        global_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" %
          (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
