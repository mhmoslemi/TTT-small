"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Hybrid spectral-flattening + LSE-FISTA + basin hopping.
The key insight is to alternate between:
1. Spectral domain operations that flatten |A(f)|^2 (reducing autocorrelation peak)
2. Time-domain gradient steps that directly minimize the peak
This hybrid approach combines the global flattening effect of spectral methods
with the local precision of gradient-based optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_flatten_step(a, gamma, S):
    """
    One step of spectral flattening:
    1. FFT
    2. Geometrically interpolate magnitude toward flat
    3. IFFT
    4. Project to non-negative, renormalize
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target magnitude (flat spectrum)
    m_target = np.sqrt(np.mean(mag ** 2))
    
    # Geometric interpolation
    # mag_new = mag^(1-gamma) * m_target^gamma
    log_mag = np.log(mag + 1e-30)
    log_target = np.log(m_target + 1e-30)
    log_new = (1 - gamma) * log_mag + gamma * log_target
    new_mag = np.exp(log_new)
    
    # Reconstruct
    new_A = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(new_A, fsize))[:n]
    
    # Project to non-negative and normalize
    a_new = np.maximum(a_new, 0.0)
    return normalize_to_sum(a_new, S)


def run_spectral_flatten(a_init, S, max_time, rng, gamma_init=0.3, gamma_min=0.01):
    """
    Run spectral flattening iterations with adaptive gamma.
    Start with large gamma (aggressive flattening) and decrease over time.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive gamma: large early, small late
        gamma = gamma_init * np.exp(-3.0 * progress) + gamma_min
        # Add some randomness
        gamma *= (0.8 + 0.4 * rng.random())
        gamma = min(gamma, 0.8)
        
        a_new = spectral_flatten_step(a, gamma, S)
        
        # Occasionally mix with a direct step toward the "ideal" (flat spectrum)
        if it % 10 == 0:
            # Try a more aggressive step
            a_agg = spectral_flatten_step(a, min(0.7, gamma * 2), S)
            c_agg = fft_convolve_self(a_agg)
            c_cur = fft_convolve_self(a_new)
            if c_agg.max() < c_cur.max():
                a_new = a_agg
        
        c = fft_convolve_self(a_new)
        if c.max() < best * (1 - 1e-10):
            best = c.max()
            best_a = a_new.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                # Perturb to escape
                noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stall_count / 200.0)
                a_new = project_simplex(a_new + rng.normal(0, max(noise_scale, 1e-10), n), S)
                stall_count = 0
        
        a = a_new
        
    return best_a, best


def run_fista_lse(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
                  beta_max=10000.0):
    """
    FISTA with LSE smoothing. Improved version with better step size control.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        if c.max() < best * (1 - 1e-12):
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking-like adaptation
        step = 1.0 / (2.5 * beta)
        
        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic perturbation
        if it % 10000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Frequency-dependent noise
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 0.5 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve DC
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def amplitude_mutate(a, rng, scale=0.1):
    """Fourier-domain amplitude mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A) + 1e-30
    phase = np.angle(A)
    
    # Multiplicative noise on amplitude
    log_noise = rng.normal(0, scale, fsize)
    new_A = mag * np.exp(log_noise) * np.exp(1j * phase)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Hybrid algorithm combining:
1. Spectral domain flattening (geometric interpolation toward flat power spectrum)
2. LSE-FISTA gradient optimization (smoothed max of autocorrelation)
3. Basin hopping with Fourier-domain mutations
4. High-beta precision polishing
5. Multi-length exploration (n=8192 primary, n=16384 secondary)

Key insight: alternating between spectral flattening (large moves that reduce
the overall dynamic range of |A(f)|^2) and time-domain subgradient steps
(precise peak shaving) escapes local minima that pure gradient methods fall into.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta_eff = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_flatten_step(a, gamma, S):
    """
    One step of spectral flattening via geometric interpolation
    between current magnitude and flat (target) magnitude.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

    m_target = np.sqrt(np.mean(mag ** 2))

    log_mag = np.log(mag + 1e-30)
    log_target = np.log(m_target + 1e-30)
    log_new = (1.0 - gamma) * log_mag + gamma * log_target
    new_mag = np.exp(log_new)

    new_A = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(new_A, fsize))[:n]

    a_new = np.maximum(a_new, 0.0)
    return normalize_to_sum(a_new, S)


def spectral_step_line_search(a, S, max_gamma=0.5):
    """Try multiple gamma values, pick the one that reduces max autocorrelation."""
    c_cur = fft_convolve_self(a)
    val_cur = c_cur.max()
    best_a = a.copy()
    best_val = val_cur

    gammas = [max_gamma, max_gamma * 0.7, max_gamma * 0.5, max_gamma * 0.3, max_gamma * 0.15]
    for g in gammas:
        a_try = spectral_flatten_step(a, g, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() < best_val:
            best_val = c_try.max()
            best_a = a_try.copy()

    return best_a, best_val


def run_spectral_optimize(a_init, S, max_time, rng, gamma_init=0.4, gamma_min=0.02):
    """
    Spectral domain optimization with adaptive gamma and line search.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        gamma = (gamma_init * np.exp(-2.5 * progress) + gamma_min)
        gamma *= (0.7 + 0.6 * rng.random())
        gamma = min(gamma, 0.7)

        a_new, val_new = spectral_step_line_search(a, S, gamma)

        if val_new < best * (1.0 - 1e-10):
            best = val_new
            best_a = a_new.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1
            if stall > 200:
                noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stall / 300.0)
                a_new = project_simplex(a_new + rng.normal(0, max(noise_scale, 1e-10), n), S)
                stall = 0

        a = a_new

    return best_a, best


def run_fista_lse(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.004,
                  beta_max=10000.0):
    """FISTA with LSE smoothing for the autocorrelation peak."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best * (1.0 - 1e-12):
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.5 * beta)

        if it - last_improve > 4000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 8000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 15000 == 0:
            decay = np.exp(-it / 60000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 0.5 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_amp_mutate(a, rng, scale=0.1):
    """Fourier-domain amplitude perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A) + 1e-30
    phase = np.angle(A)
    log_noise = rng.normal(0, scale, fsize)
    new_A = mag * np.exp(log_noise) * np.exp(1j * phase)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with diverse mutation strategies."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(12.0, max_time / 25.0)

    it = 0
    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        base_scale = 0.25 * np.exp(-1.5 * progress) + 0.03
        scale = base_scale * (0.5 + 1.0 * rng.random())

        r = rng.random()
        if r < 0.35:
            a_mut = fourier_phase_mutate(a, rng, scale)
        elif r < 0.65:
            a_mut = fourier_amp_mutate(a, rng, scale)
        else:
            noise_scale = scale * S / np.sqrt(n)
            a_mut = project_simplex(a + rng.normal(0, noise_scale, n), S)

        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 10)
        if t_alloc < 5:
            break

        # Alternate between spectral and FISTA for local optimization
        if it % 3 == 0:
            a_opt, val = run_spectral_optimize(a_mut, S, t_alloc, rng,
                                                gamma_init=0.2, gamma_min=0.01)
        else:
            a_opt, val = run_fista_lse(a_mut, S, t_alloc, rng,
                                         beta_init=30 + 100 * progress,
                                         beta_rate=0.002,
                                         beta_max=3000 + 5000 * progress)

        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))

        a = a_opt

    return best_a, best


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)
