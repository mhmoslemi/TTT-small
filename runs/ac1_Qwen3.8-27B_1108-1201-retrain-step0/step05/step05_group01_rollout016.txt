import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y (the relevant part)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def pnorm_gradient(y, c, p):
    """Gradient of (sum c_k^p)^(1/p) w.r.t. y."""
    n = len(y)
    cp = np.maximum(c, 1e-30)
    w = cp ** (p - 1)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / wsum
    return conv_gradient(y, w)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_to(a, S):
    """Normalize a to have sum S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def run_lbfgs_polish(a_init, max_time, rng, n_levels=6):

    n = len(a_init)
    S = np.sqrt(2.0 * n)

    for idx, (beta, pen) in enumerate(levels):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

            b = fft_convolve_self(x)
            cm = b.max()

            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            objective = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            objective += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return objective, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = normalize_to(a_new, S)
        obj_new = compute_obj(a_new)

        print("    L-BFGS level %d (beta=%d, pen=%.1f): obj=%.8f (t=%.1fs, nit=%d)" %

        if obj_new < best_obj:

            best_a = a_new.copy()

        c = fft_convolve_self(a)
        cmax = c.max()
        cur_obj = 2.0 * n * cmax / S**2

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print("    Active-Newton it %d: obj=%.8f" % (it, cur_obj))

        # Find active set: lags within 0.1% of max
        tol = cmax * 0.001
        active = np.where(c >= cmax - tol)[0]
        active = active[active < 2 * n - 1]  # valid range

        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Use at most 32 active lags (most important ones)
        if len(active) > 32:
            # Keep the ones closest to max
            active_sorted = active[np.argsort(cmax - c[active])[:32]]
            active = active_sorted

        # Subgradient from active set
        w_full = np.zeros(2 * n - 1)
        w_full[active] = 1.0 / len(active)
        grad = conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Newton-like step: solve (approximate) Hessian system
        # Use a damped step
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, S / (0.5 * gnorm))
        
        # Try the step
        a_trial = project_simplex(a - step * grad, S)
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() <= cmax * (1 + 1e-8):
            a = a_trial
        else:
            # Backtrack
            for _ in range(4):
                step *= 0.5
                a_trial = project_simplex(a - step * grad, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() <= cmax * (1 + 1e-8):
                    a = a_trial
                    break
            else:
                # No improvement, add small noise
                noise = rng.normal(0, 1e-5 * S / n, n)
                a = project_simplex(a + noise, S)

    return best_a, best_obj


def run_multi_n_search(a_best_8192, max_time, rng, verbose=True):
    """Try a different n value to see if it gives better objective."""
    start = time.monotonic()
    best_obj = np.inf
    best_a = None

    for n_try in [16384, 4096]:
        if time.monotonic() - start > max_time:
            break
        time_left = max_time - (time.monotonic() - start)
        if time_left < 30:
            break
        time_per_n = time_left * 0.6

        S_try = np.sqrt(2.0 * n_try)
        
        # Resample the best sequence to new length
        if a_best_8192 is not None and len(a_best_8192) > 0:
            old_n = len(a_best_8192)
            new_idx = (np.arange(n_try) * old_n / n_try).astype(int)
            new_idx = np.clip(new_idx, 0, old_n - 1)
            a_try = a_best_8192[new_idx]
            # Smooth the resampled sequence
            kernel = np.ones(5) / 5.0
            a_try = np.convolve(a_try, kernel, mode='same')
            a_try = np.maximum(a_try, 0)
        else:
            t_arr = np.linspace(0, 1, n_try, endpoint=False)
            a_try = 0.5 + 0.5 * np.cos(2 * np.pi * t_arr)
        
        a_try = normalize_to(a_try, S_try)
        obj_try = compute_obj(a_try)
        if verbose:
            print("    n=%d initial: obj=%.6f" % (n_try, obj_try))

        # Quick FISTA optimization
        a_opt, _ = run_fista_lse(a_try, min(time_per_n, 120), rng,
                                  beta_start=20.0, beta_end=2000.0, verbose=False)
        obj_opt = compute_obj(a_opt)
        if verbose:
            print("    n=%d after FISTA: obj=%.8f" % (n_try, obj_opt))

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses Adam+LSE with curriculum, FISTA, L-BFGS, and active-set Newton.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t_arr = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t_arr)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    S0 = np.sqrt(2.0 * n0)

    # === Phase 1: Adam+LSE with p-norm curriculum (40% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.40
    print("\n=== Phase 1: Adam+LSE curriculum (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_adam_lse(global_best_a, p1_budget, rng,
                         beta_start=2.0, beta_end=8000.0,
                         lr=0.005, beta1=0.9, beta2=0.999,
                         p_curriculum=True, verbose=True)
    obj1 = compute_obj(a1)
    print("  Adam+LSE done: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: FISTA refinement (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: FISTA refinement (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_fista_lse(global_best_a, p2_budget, rng,
                               beta_start=100.0, beta_end=10000.0, verbose=True)
        obj2 = compute_obj(a2)
        print("  FISTA done: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: L-BFGS polish (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: L-BFGS polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, obj3 = run_lbfgs_polish(global_best_a, p3_budget, rng, n_levels=6)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Active-set Newton (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Active-set Newton (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, obj4 = run_active_newton(global_best_a, p4_budget, rng, verbose=True)
        print("  Active-Newton done: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Try different n (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.05
        print("\n=== Phase 5: Multi-n search (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, obj5 = run_multi_n_search(global_best_a, p5_budget, rng, verbose=True)
        if a5 is not None and obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()
            print("  Multi-n improved: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        else:
            print("  Multi-n: no improvement (t=%.0fs)" % (time.monotonic() - t5))

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: Final squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng, n_levels=3)
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
