import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    Nf = N
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a). Returns (grad, lse_val, c, c_max)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    beta_val = float(beta)
    if beta_val < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta_val * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta_val
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = weights
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val, c, c_max

def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize a fraction of FFT phases, project to non-negative, renormalize."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_envelope_optimize(a, N, rng, n_iters=50):
    """Optimize the spectral envelope |A| while keeping phase, to flatten |A|^2."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    phase = np.angle(A)
    mag = np.abs(A).copy()
    for _ in range(n_iters):
        a_curr = np.real(ifft(mag * np.exp(1j * phase)))[:n]
        a_curr = np.maximum(a_curr, 0.0)
        s = a_curr.sum()
        if s < 1e-15:
            break
        a_curr = a_curr * S / s
        c = np.real(ifft(fft(a_curr, N) * fft(a_curr, N)))[:2 * n - 1]
        c_max = c.max()
        # Update envelope: push energy away from frequencies that create peaks
        # Use a simple heuristic: flatten |A|^2 toward the mean
        target = np.sqrt(np.mean(mag**2))
        mag = 0.9 * mag + 0.1 * target
        # Add small random perturbation to escape

        mag = np.maximum(mag, 1e-10)

    a_final = np.maximum(a_final, 0.0)

    if s < 1e-15:

def mass_transfer_polish(a, N, max_time, rng):
    """Hot-to-cold mass transfer: move mass from high-gradient to low-gradient coords."""
    n = len(a)
    S = a.sum()
    a_curr = a.copy()
    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = np.real(ifft(fft(a_curr, N) * fft(a_curr, N)))[:2 * n - 1]
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a_curr.copy()
        
        # Compute gradient of max (using top-5 peaks)
        top_k = 5
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = top_vals / total
        w_pad = np.zeros(N)
        w_pad[:2 * n - 1] = weights
        W = fft(w_pad)
        A = fft(a_curr, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        grad_sorted = np.sort(grad)
        n_hot = max(1, n // 20)
        n_cold = max(1, n // 20)
        hot_threshold =(grad_sorted[-n_hot * 2])
        cold_threshold = grad_sorted[n_cold]
        
        hot_mask = grad > hot_threshold
        cold_mask = grad < cold_threshold
        
        if not np.any(hot_mask) or not np.any(cold_mask):
            break
        
        # Pick a random hot and cold pair
        hot_idx = np.where(hot_mask)[0]
        cold_idx = np.where(cold_mask)[0]
        
        # Try multiple random transfers
        n_tries = 20
        for _ in range(n_tries):
            if time.monotonic() - start > max_time:
                break
            i = hot_idx[rng.integers(len(hot_idx))]
            j = cold_idx[rng.integers(len(cold_idx))]
            if i == j:
                continue
            max_delta = min(a_curr[i], S * 0.01)
            if max_delta < 1e-10:
                continue

            for delta_frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.5]:
                delta = max_delta * delta_frac
                a_try = a_curr.copy()
                a_try[i] -= delta
                a_try[j] += delta
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try * (S / a_try.sum())
                c_try = np.real(ifft(fft(a_try, N) * fft(a_try, N)))[:2 * n - 1]
                if c_try.max() < cmax:
                    a_curr = a_try
                    cmax = c_try.max()
                    break
    return best_a, best_cmax

        improved = False

                # Add to a random cold coordinate
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
                a_try[j] += transfer
                a_try = project_simplex(a_try, S)
                c_try = np.real(ifft(fft(a_try, N) * fft(a_try, N)))[:2 * n - 1]
                if c_try.max() < cmax * 0.999:
                    a_curr = a_try
                    c = c_try
                    cmax = cmax * 0.999  # conservative update
                    improved = True
                    break
    return best_a, best_cmax


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def optimize_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0,
                   m_lbfgs=12, use_active=False, n_active=5):
    """L-BFGS optimization block with LSE or active-set gradient."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    if use_active:
        g, c, c_max = active_grad(a, N, top_k=n_active)
    else:
        g, f_val, c, c_max = lse_grad(a, N, beta)
    
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stagnant = 0
    prev_best = best_cmax
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adapt beta
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            if use_active:
                n_active = min(50, n_active + 2)
            if best_cmax < prev_best - 1e-8:
                stagnant = 0
            else:
                stagnant += 1
            prev_best = best_cmax
        
        # Trim history
        if it % 60 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search with Armijo condition
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        accepted = False
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = np.real(ifft(fft(candidate, N) * fft(candidate, N)))[:2 * n - 1]
            cm_ls = c_ls.max()
            if cm_ls <= c_max + 1e-12 or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max + 1e-12:
                    accepted = True
                break
            alpha_ls *= 0.6
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        # Compute new gradient
        if use_active:
            g_new, c_nw, cmax_nw = active_grad(a_new, N, top_k=n_active)
        else:
            g_new, f_new, c_nw, cmax_nw = lse_grad(a_new, N, beta)
        
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        # Random restart within block
        if it % 300 == 0 and stagnant > 2 and rng.random() < 0.3:
            a = spectral_phase_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            beta = max(beta_init, beta * 0.5)
            if use_active:
                g, c, c_max = active_grad(a, N, top_k=n_active)
            else:
                g, f_val, c, c_max = lse_grad(a, N, beta)
            stagnant = 0
    
    return best_a, best_cmax


def make_init_hann(n, rng):
    """Hann-window inspired initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 47]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_binary_like(n, rng):
    a = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-phase: diverse init -> L-BFGS LSE/active-set -> mass transfer -> polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= given.size <= 32768:
            n = given.size
    except Exception:
        pass
    
    S = np.sqrt(2.0 * float(n))
    N = 1 << (max(2 * n - 1, 1024).bit_length())
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = np.real(ifft(fft(a_cand, N) * fft(a_cand, N)))[:2 * n - 1]
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc
    
    # === Phase 1: Multi-start exploration (400s) ===
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(400.0, (deadline - t1) * 0.16)
    
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
            inits.append(("given+noise", np.clip(given * (1 + 0.1 * rng.standard_normal(n)), 0, None)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("hann", make_init_hann(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("binary", make_init_binary_like(n, rng)))
    
    # Also try spectral randomization of the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            g_norm = safe_normalize(given, n, S)
            inits.append(("given_spectral1", spectral_phase_restart(g_norm, N, rng, frac=0.5)))
            inits.append(("given_spectral2", spectral_phase_restart(g_norm, N, rng, frac=0.8)))
    except Exception:
        pass
    
    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))
    
    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = optimize_block(a_norm, n, S, N, min(rem, 80.0), rng,
                                   beta_init=8.0, beta_max=100.0, m_lbfgs=10)
        update_best(a_l, name + "/LBFGS")
    
    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))
    
    # === Phase 2: Focused optimization with alternating methods (1200s) ===
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0
    
    print("=== Phase 2: Focused optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.5)
    
    a_curr = global_a.copy()
    stall = 0
    method_toggle = 0
    
    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 15:
            break
        
        # Alternate between LSE and active-set methods
        use_active = (method_toggle % 3 == 2)
        method_toggle += 1
        
        beta_start = 10.0 + 5.0 * min(stall, 8)
        beta_end = min(600.0, 50.0 * (1 + 0.5 * stall))
        
        a_new, sc_new = optimize_block(a_curr, n, S, N, block_time, rng,
                                        beta_init=beta_start, beta_max=beta_end,
                                        m_lbfgs=14, use_active=use_active)
        
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2 * n * sc_new / (S * S)
            print("  [P2] score=%.6f (beta %.0f-%.0f, active=%s)" % (score, beta_start, beta_end, use_active))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.4)
                stall = 0
            elif stall == 4:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.7)
                # Also try envelope optimization
                a_env = spectral_envelope_optimize(a_curr, N, rng, n_iters=30)
                c_env = np.real(ifft(fft(a_env, N) * fft(a_env, N)))[:2 * n - 1]
                if c_env.max() < a_curr.max():
                    pass  # Keep the better one
                a_curr = safe_normalize(a_curr, n, S)
                stall = 0
            elif stall >= 6:
                # Full restart from spectral randomization
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=1.0)
                stall = 0
    
    # === Phase 3: Mass transfer polishing (500s) ===
    print("=== Phase 3: Mass transfer polishing ===")
    t3 = time.monotonic()
    remaining3 = min(500.0, (deadline - t3) * 0.2)
    if remaining3 > 30:
        a_mt, sc_mt = mass_transfer_polish(global_a, N, remaining3, rng)
        update_best(a_mt, "mass_transfer")
    
    # === Phase 4: High-beta final L-BFGS (300s) ===
    print("=== Phase 4: Final high-beta L-BFGS ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 20
    if remaining4 > 30:
        a_fin, sc_fin = optimize_block(global_a, n, S, N, min(remaining4, 350.0),
                                        rng, beta_init=40.0, beta_max=800.0,
                                        m_lbfgs=16, use_active=True, n_active=10)
        update_best(a_fin, "final_LBFGS")
    
    # === Phase 5: One more mass transfer round ===
    remaining5 = deadline - time.monotonic() - 15
    if remaining5 > 20:
        a_mt2, sc_mt2 = mass_transfer_polish(global_a, N, min(remaining5, 200.0), rng)
        update_best(a_mt2, "mass_transfer2")
    
    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
