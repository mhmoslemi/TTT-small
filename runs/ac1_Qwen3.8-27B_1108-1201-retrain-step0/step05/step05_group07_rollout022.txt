import numpy as np
from scipy.fft import fft, ifft
import time

def compute_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient_z(a, z, beta):
    """
    Gradient of LSE-smoothed objective with respect to z, where a = exp(z).
    Chain rule: ∂g/∂z_i = ∂g/∂a_i * a_i
    """
    n = len(a)
    c = compute_conv(a)
    S = a.sum()
    
    # LSE
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot  # softmax weights over convolution lags
    F = (c.max() + np.log(tot) / beta)  # LSE value
    
    # Cross-correlation: (w ⊛ a)_i = sum_j w[i+j] * a[j]
    # This is the cross-correlation of w and a
    # Using FFT: ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    
    # ∂g/∂a_i = (2*S*cross_i - 2*F*a_i) / S^3
    grad_a = (2.0 * S * cross - 2.0 * F * a) / (S * S * S)
    
    # ∂g/∂z_i = grad_a_i * a_i
    grad_z = grad_a * a
    
    return grad_z, F / (S * S)  # return gradient and (unscaled) objective

def fft_conv_self(a):
    """Full linear self-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(w, a):
    """
    Compute h[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i = 0, ..., n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    w_rev = w[::-1]
    fsize = 1 << (n + len(w_rev) - 1).bit_length()

    W = fft(w_rev, fsize)

    """
    Solve the 1D subproblem for coordinate i.
    Finds t >= 0 minimizing max_k(D[k]+2t*h[k]+t^2*delta) / (S_b+t)^2.
    Returns optimal t.
    """
    old_val = a[i]
    b = a.copy()
    b[i] = 0.0
    S_b = b.sum()

    if S_b < 1e-30:
        return old_val

    D = fft_conv_self(b)
    h = np.zeros(2 * n - 1)
    vs = i
    ve = min(2 * n - 1, i + n)
    h[vs:ve] = b[:ve - vs]

    has_quad = (2 * i < 2 * n - 1)

    # Determine search range
    t_upper = max(S_b * 0.5, old_val * 2 + S_b / n)
    t_grid = np.linspace(0, t_upper, 30)

    best_t = t_grid[0]
    best_val = np.inf
    for t in t_grid:
        C = D + 2.0 * t * h
        if has_quad:
            C[2 * i] += t * t
        obj = C.max() / (S_b + t) ** 2
        if obj < best_val:
            best_val = obj
            best_t = t

    # Golden section refinement
    idx = int(np.argmin(np.abs(t_grid - best_t)))
    lo = t_grid[max(0, idx - 1)]
    hi = t_grid[min(len(t_grid) - 1, idx + 1)]
    if hi - lo < 1e-10:
        return best_t

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    for _ in range(25):
        C1 = D + 2.0 * c1 * h
        if has_quad:
            C1[2 * i] += c1 * c1
        obj1 = C1.max() / (S_b + c1) ** 2

        C2 = D + 2.0 * c2 * h
        if has_quad:
            C2[2 * i] += c2 * c2
        obj2 = C2.max() / (S_b + c2) ** 2

        if obj1 < obj2:
            hi = c2
            c2 = c1
            c1 = hi - gr * (hi - lo)
        else:
            lo = c1
            c1 = c2
            c2 = lo + gr * (hi - lo)

    return (lo + hi) / 2.0


def optimize_coordinate_descent(a_init, max_time, rng):
    """Randomized coordinate descent with 1D line search."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()

    c = fft_conv_self(a)
    S = a.sum()
    cur_obj = c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()

    order = rng.permutation(n)
    pos = 0
    stuck = 0
    check_interval = max(1, n // 50)

    while time.monotonic() - start < max_time:
        i = int(order[pos % n])
        pos += 1

        # Small probability of forcing update on near-zero coords
        threshold = 1e-12 * S
        if a[i] < threshold and rng.random() < 0.85:
            continue

        new_t = cd_1d_solve(a, i, n)
        a[i] = new_t

        if pos % check_interval == 0:
            c = fft_conv_self(a)
            S = a.sum()
            cur_obj = c.max() / (S * S)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 50:
                    # Random restart perturbation
                    idx = rng.integers(0, n, size=max(1, n // 10))
                    a[idx] += rng.normal(0, 0.01 * S / np.sqrt(n), len(idx))
                    a = np.maximum(a, 0.0)
                    stuck = 0

    return best_a, best_obj


def generate_structured_init(n, rng, style):
    """Generate diverse initial sequences."""
    t = np.linspace(0, 1, n)
    if style == 0:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.01
    elif style == 1:
        a = np.exp(-2.0 * np.abs(t - 0.5) * 4.0) + 0.01
    elif style == 2:
        a = 1.0 / (1.0 + np.exp(-10.0 * (t - 0.5))) + 0.01
    elif style == 3:
        a = rng.random(n) + 0.01
    elif style == 4:
        a = 1.0 + 0.3 * np.sin(4.0 * np.pi * t) + 0.01
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) * 0.1
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LSE Adam + coordinate descent hybrid with multiple restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s")

    global_best = np.inf
    global_a = None

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            c_p = fft_conv_self(a_provided)
            S_p = a_provided.sum()
            obj_p = 2.0 * len(a_provided) * c_p.max() / (S_p * S_p)
            print(f"Provided: n={len(a_provided)}, obj={obj_p:.6f}")
            global_best = obj_p
            global_a = a_provided.copy()
    except Exception:
        pass

    # Determine main length
    if a_provided is not None:
        main_n = len(a_provided)
    else:
        main_n = 8192

    # === Phase 1: LSE Adam (multiple restarts) ===
    remaining = deadline - time.monotonic()
    n_restarts = 2
    t_per = min(300, remaining * 0.25 / n_restarts)
    print(f"\nPhase 1: LSE Adam, n={main_n}, {n_restarts} restarts, {t_per:.0f}s each")

    for ri in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break

        if ri == 0 and a_provided is not None:
            a_init = a_provided.copy()
        elif ri == 0:
            a_init = generate_structured_init(main_n, rng, 0)
        else:
            a_init = generate_structured_init(main_n, rng, 3)

        a_opt, _ = optimize_lse_adam(a_init, t_per, rng, beta_start=5.0, beta_end=3000.0, lr=0.05)
        c = fft_conv_self(a_opt)
        S = a_opt.sum()
        obj = 2.0 * main_n * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            eval_obj = evaluate_sequence([float(x) for x in np.clip(a_opt, 0, 1000)])
            print(f"  Phase1 restart {ri}: eval_obj={eval_obj:.6f}")

    # === Phase 2: Coordinate descent (main refinement) ===
    remaining = deadline - time.monotonic()
    t2 = min(900, remaining * 0.45)
    if t2 > 30 and global_a is not None:
        n2 = len(global_a)
        print(f"\nPhase 2: Coordinate descent, n={n2}, time={t2:.0f}s")
        a_cd, _ = optimize_coordinate_descent(global_a, t2, rng)
        c = fft_conv_self(a_cd)
        S = a_cd.sum()
        obj = 2.0 * n2 * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_cd.copy()
            eval_obj = evaluate_sequence([float(x) for x in np.clip(a_cd, 0, 1000)])
            print(f"  Phase2 CD: eval_obj={eval_obj:.6f}")

    # === Phase 3: LSE polish with high beta ===
    remaining = deadline - time.monotonic()
    t3 = min(400, remaining * 0.35)
    if t3 > 20 and global_a is not None:
        print(f"\nPhase 3: LSE polish (high beta), time={t3:.0f}s")
        a_pol, _ = optimize_lse_adam(global_a, t3, rng, beta_start=200.0, beta_end=30000.0, lr=0.005)
        c = fft_conv_self(a_pol)
        S = a_pol.sum()
        obj = 2.0 * len(a_pol) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_pol.copy()
            eval_obj = evaluate_sequence([float(x) for x in np.clip(a_pol, 0, 1000)])
            print(f"  Phase3 polish: eval_obj={eval_obj:.6f}")

    # === Phase 4: Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        t4 = remaining - 10
        print(f"\nPhase 4: Final CD, time={t4:.0f}s")
        a_cd2, _ = optimize_coordinate_descent(global_a, t4, rng)
        c = fft_conv_self(a_cd2)
        S = a_cd2.sum()
        obj = 2.0 * len(a_cd2) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_cd2.copy()
            eval_obj = evaluate_sequence([float(x) for x in np.clip(a_cd2, 0, 1000)])
            print(f"  Phase4 CD: eval_obj={eval_obj:.6f}")

    # === Phase 5: Try different length ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alt_n = main_n * 2 if main_n < 16384 else main_n // 2
        if alt_n >= 1024:
            t5 = remaining - 10
            print(f"\nPhase 5: Alt length n={alt_n}, time={t5:.0f}s")
            a_init = generate_structured_init(alt_n, rng, 0)
            if global_a is not None:
                a_interp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(global_a)), global_a)
                a_init = a_interp + 0.01

            a_opt, _ = optimize_lse_adam(a_init, t5 * 0.4, rng, beta_start=10.0, beta_end=5000.0, lr=0.03)
            a_opt, _ = optimize_coordinate_descent(a_opt, t5 * 0.5, rng)
            c = fft_conv_self(a_opt)
            S = a_opt.sum()
            obj = 2.0 * alt_n * c.max() / (S * S)
            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                eval_obj = evaluate_sequence([float(x) for x in np.clip(a_opt, 0, 1000)])
                print(f"  Phase5 alt_n: eval_obj={eval_obj:.6f}")

    # Final verification
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    final_obj = evaluate_sequence(final_list)
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(final_list)}, obj={final_obj:.6f}")
    print(f"{'='*50}")

    return final_list
