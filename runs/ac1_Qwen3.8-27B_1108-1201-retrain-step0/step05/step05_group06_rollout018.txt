from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, top_k=1, tol=0.0):
    n = len(y)
    max_val = c.max()
    if tol > 0:
        active = np.where(c >= max_val * (1.0 - tol))[0]
    else:
        # Use top_k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        active = top_idx
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    # Weight by how close to max
    weights = c[active] / max_val if max_val > 0 else np.ones(len(active))
    weights = weights ** 10  # Sharp weighting
    weights = weights / weights.sum()
    w_full[active] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def normalize_seq(a, S):

def exact_subgrad_backtrack(a_init, max_time, rng, top_k=3):

    c_alpha = 1e-4

        k_use = min(top_k + stuck // 1000, 20)
        grad = exact_subgradient(y, c, top_k=k_use, tol=0.0)

        if gnorm < 1e-15:
            break

        alpha_try = alpha
        improved = False
        for _ in range(15):
            x_trial = y - alpha_try * grad
            x_trial = project_simplex(x_trial, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cur_max:
                improved = True
                break
            alpha_try *= 0.5
        
        if improved:
            y = x_trial
            alpha = min(alpha * 1.5, 10.0)  # Increase step if it worked

Minimize 2*n*max_conv/sum^2 using hybrid smooth-exact optimization.
Strategy:
- Aggressive LSE-FISTA with fast beta ramp for global progress
- Exact subgradient with backtracking for precise peak targeting
- Multiple n values (8192, 16384) with competitive selection
- Diverse spectral initializations
- Momentum-aware restarts with structured perturbation
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, top_k=1, tol=0.0):
    """Subgradient from lags within tolerance of max, with power weighting."""
    n = len(y)
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    if tol > 0:
        active = np.where(c >= max_val * (1.0 - tol))[0]
    else:
        active = np.argsort(c)[::-1][:top_k]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    # Power-weight by closeness to max
    ratios = c[active] / max_val
    weights = ratios ** 20
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[active] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_max=100000.0,
                  ramp_rate=0.015, ramp_accel=0.005):
    """
    LSE-FISTA with aggressive adaptive beta schedule.
    Key: beta ramps up quickly to get accurate max approximation,
    then accelerates when stuck.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Aggressive beta ramp: faster when stuck
        if stuck < 100:
            beta = min(beta_max, beta_init * (1.0 + ramp_rate * it))
        elif stuck < 1000:
            beta = min(beta_max, beta * (1.0 + ramp_rate * 0.5))
        elif stuck < 5000:
            beta = min(beta_max, beta * (1.0 + ramp_rate))
        else:
            beta = min(beta_max, beta * (1.0 + ramp_accel * (1 + stuck // 1000)))

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: balance between LSE smoothness and progress
        # L ~ 2*beta*max(a) is a rough Lipschitz bound
        L_est = 2.0 * beta * np.max(a) + 2.0 * np.max(a) + 1e-6
        step = 0.95 / L_est
        # Cap to avoid overshooting
        step = min(step, 0.5 * S / (gnorm + 1e-10))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Momentum restart with structured noise when stuck
        if stuck > 3000:
            t_mom = 1.0
            # Structured noise: modulate with a low-frequency pattern
            tt = np.arange(n, dtype=np.float64)
            pattern = np.cos(2 * np.pi * tt / n * rng.integers(1, 5))
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 15000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + noise_scale * (pattern + rng.normal(0, 0.5, n)), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_exact_subgrad_bt(a_init, max_time, rng, top_k_start=2, top_k_max=15):
    """
    Exact subgradient method with backtracking line search.
    Directly targets the actual peak(s) of the autocorrelation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Adaptive step
    alpha = 1.0
    alpha_max = 5.0
    c_armijo = 1e-4

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Increase top_k as we get more stuck (spread the subgradient)
        k_use = min(top_k_start + stuck // 500, top_k_max)
        # Also use tolerance-based selection for more robust subgradient
        tol = min(0.001 * (1 + stuck // 1000), 0.05)
        grad = exact_subgradient(y, c, top_k=k_use, tol=tol)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search (max 12 evaluations)
        alpha_try = min(alpha, alpha_max)
        improved = False
        best_trial = None
        best_trial_max = cur_max
        for _ in range(12):
            x_trial = y - alpha_try * grad
            x_trial = project_simplex(x_trial, S)
            c_trial = fft_convolve_self(x_trial)
            trial_max = c_trial.max()
            if trial_max < best_trial_max:
                best_trial_max = trial_max
                best_trial = x_trial.copy()
            if trial_max < cur_max:
                improved = True
                break
            alpha_try *= 0.5

        if improved:
            y = best_trial if best_trial is not None else x_trial
            alpha = min(alpha * 1.3, alpha_max)
        elif best_trial is not None:
            # Even if not strictly improved over cur_max, take the best trial
            y = best_trial
            alpha = max(alpha * 0.7, 1e-10)
        else:
            # Very small step
            alpha_try = 0.01 / (gnorm + 1.0)
            y = project_simplex(y - alpha_try * grad, S)
            alpha = max(alpha * 0.5, 1e-10)

        # Periodic noise injection
        if stuck > 2000 and it % 1000 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            stuck = 0
            alpha = 1.0

    return best_a, best_obj


def run_hybrid(a_init, max_time, rng, beta_init=30.0, beta_max=200000.0):
    """
    Hybrid: alternate between LSE gradient (smooth) and exact subgradient (sharp).
    Uses a "confidence" measure based on how close LSE is to the true max.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Beta schedule
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it) ** (1.0 + min(stuck / 5000.0, 2.0)))

        # Compute LSE value to gauge smoothness
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        lse_val = cm + np.log(np.exp(shifted).sum()) / beta
        # Gap between LSE and true max (should be small for good approximation)
        lse_gap = lse_val - cm  # This is always positive

        # Choose gradient: if LSE is a good approximation, use it; otherwise use exact
        # The gap is approximately log(K)/beta where K = number of near-max peaks
        if lse_gap < 0.1 / beta or stuck < 200:
            grad = lse_gradient(y, c, beta)
        elif stuck < 5000:
            # Blend LSE and exact
            grad_lse = lse_gradient(y, c, beta)
            grad_exact = exact_subgradient(y, c, top_k=3, tol=0.002)
            mix = min(1.0, (lse_gap * beta) / 5.0)
            grad = (1.0 - mix) * grad_lse + mix * grad_exact
        else:
            grad = exact_subgradient(y, c, top_k=min(10, 3 + stuck // 1000), tol=0.01)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with simple backtracking (max 5 steps for speed)
        L_est = 2.0 * max(beta, 10.0) * np.max(a) + 2.0 * np.max(a) + 1e-6
        step = 0.9 / L_est
        step = min(step, 0.3 * S / (gnorm + 1e-10))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)

        # Quick check: if no improvement, try smaller step
        if it % 10 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() >= cur_max * 1.001:
                x_new = project_simplex(y - 0.3 * step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 4000:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 15000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def make_spectral_init(n, rng):
    """
    Create initialization using spectral design:
    Start with a nearly flat |A|^2 spectrum, take IFFT, clip to non-negative.
    """
    # Target: flat spectrum means flat autocorrelation
    # Start with a raised-cosine spectrum (smooth transition)
    fsize = 1 << (n - 1).bit_length()
    # Design spectrum
    freqs = np.arange(fsize)
    # Raised cosine window in frequency
    band = n * 0.8
    spectrum = np.ones(fsize)
    # Smooth rolloff
    for i in range(fsize):
        d = abs(i - fsize // 2)
        if d > band and d < band * 1.5:
            t = (d - band) / (band * 0.5)
            spectrum[i] = 0.5 * (1 + np.cos(np.pi * t))
        elif d >= band * 1.5:
            spectrum[i] = 0.0
    spectrum[fsize // 2] = 1.0
    # Make it Hermitian-symmetric
    spectrum = np.real(np.fft.irfft(spectrum, n)) if False else spectrum
    # Actually, just use the spectrum to modulate a base signal
    tt = np.linspace(0, 1, n, endpoint=False)
    base = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    # Add spectral structure
    phase = rng.uniform(0, 2 * np.pi, n)
    a = base * (1 + 0.3 * np.cos(2 * np.pi * rng.uniform(2, 8, 1) * tt + phase[0]))
    a = np.maximum(a, 0)
    return a
