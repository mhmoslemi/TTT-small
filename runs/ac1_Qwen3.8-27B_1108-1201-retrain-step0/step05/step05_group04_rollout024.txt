def active_set_refine(a, n, S, max_time, rng):
    """
    Refine using exact subgradient from active set with backtracking line search.
    Uses adaptive active set size and periodic strategy changes.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    K = 10  # active set size
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Find active set
        threshold = cm - 0.01 * cm  # within 1% of max
        active = np.where(c >= threshold)[0]
        if len(active) < K:
            # Take top K lags
            active = np.argsort(c)[::-1][:K]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            # grad_k[j] = 2*a[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * a[k - js]
        
        # Average over active set
        grad /= len(active)
        
        # Project onto simplex tangent: subtract the mean (to maintain sum)
        # Actually, the tangent space of {x: sum(x)=S, x>=0} at a is
        # {d: sum(d)=0, d_j >= 0 where a_j = 0}
        # For simplicity, project -grad onto sum=0
        grad_tangent = grad - np.mean(grad)
        
        # Backtracking line search
        step = 0.1
        for _ in range(20):
            a_new = a - step * grad_tangent
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < S * 0.9:
                step *= 0.5
                continue
            a_new = a_new * (S / s)  # renormalize
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cm * (1 - 1e-8):
                a = a_new
                break

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Hybrid of LSE-FISTA (smooth phase) and exact active-set subgradient 
refinement (sharp phase) with FFT-accelerated gradient computation, 
adaptive active set, backtracking line search, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def active_gradient(a, active_lags):
    """Compute subgradient of max(conv) using FFT."""
    n = len(a)
    size = 2 * n - 1
    b = a[::-1].astype(np.float64)
    I = np.zeros(size)
    I[active_lags] = 1.0
    
    fsize = 1 << (size + n - 1).bit_length()
    I_f = fft(I, fsize)
    b_f = fft(b, fsize)
    conv = np.real(ifft(I_f * b_f))
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    size = 2 * n - 1
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def run_subgradient(a_init, n, max_time, rng, K_start=10, line_search_max=15):
    """
    Exact subgradient method with backtracking line search.
    Uses FFT to compute the active-set gradient efficiently.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    K = K_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set: lags within adaptive threshold of max
        # Threshold adapts: start wide, narrow over time
        threshold_frac = max(0.001, 0.05 * np.exp(-it / 500.0))
        threshold = cm * (1 - threshold_frac)
        active = np.where(c >= threshold)[0]
        
        # Limit active set size
        if len(active) > K:
            top_k = np.argsort(c)[::-1][:K]
            active = top_k
        elif len(active) < 2:
            active = np.argsort(c)[::-1][:2]
        
        # Compute active-set subgradient
        grad = active_gradient(a, active)
        
        # Project onto simplex tangent (sum = 0)
        grad_tan = grad - np.mean(grad)
        
        # Backtracking line search
        step = 0.5 / (1.0 + 0.01 * it)
        found = False
        for ls in range(line_search_max):
            a_new = a - step * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_new = a_new * (S / s)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cm * (1.0 - 1e-10):
                a = a_new
                found = True
                break
            step *= 0.5
        
        if not found:
            # Take a tiny step anyway to make progress
            a_new = a - (step / 100.0) * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new * (S / s)
        
        # Adapt K: grow when stuck, shrink when making progress
        if no_improve > 100:
            K = min(K + 2, 50)
        elif no_improve == 0 and K > K_start:
            K = max(K - 1, K_start)
        
        # Periodic noise injection to escape local minima
        if no_improve > 200 and it % 50 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-no_improve / 1000.0)
            if noise_scale > 1e-12:
                a = np.maximum(a + rng.normal(0, noise_scale, n), 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
                no_improve = 0
    
    return best_a, best


def run_lse_fista_v2(a_init, n, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=100):
    """
    Improved LSE-FISTA with better step size and restart strategy.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)
        
        c = fft_convolve_self(y)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it
        
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size: use 1/beta but cap it
        step = 1.0 / beta
        
        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 200:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def peak_shaving_v2(a, n, max_time, rng):
    """
    Peak shaving: directly reduce the top peaks via multiplicative updates.
    """
    S = np.sqrt(2.0 * n)
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Get top peaks
        n_peaks = 8
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.98)
            resp[js] += contrib * weight
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative update
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    return best_a, best

def diversity_inits(n, rng):

**Alternative: Coordinate descent on the peaks**
Instead of updating all a_j simultaneously, update one at a time:

The max is a convex piecewise-linear function of a_j. Its minimum can be found by binary search or by finding the intersection of the top two pieces.
