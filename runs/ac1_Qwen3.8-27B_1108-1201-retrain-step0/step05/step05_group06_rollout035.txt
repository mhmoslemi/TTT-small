"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_loss_and_grad(a, c, beta):
    """Compute LSE loss and gradient w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    loss = cm + np.log(tot) / beta
    return loss, grad

def topk_grad(a, c, k, power=1.0):
    """Top-k power gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    c = fft_convolve_self(a)

    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 using multi-resolution FISTA with:
- Power-2 warmup for spectral flattening
- Adaptive LSE FISTA (global convergence)
- Exact subgradient polishing (active set)
- Larger n for more optimization freedom
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta):
    """LSE gradient w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_grad(a, c, k, power=1.0):
    """Top-k gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:

        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power2_grad(a, c):
    """Gradient of sum(c^2) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(a, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]

    """Compute objective."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Multi-resolution search (n=16384 primary, n=32768 if time permits)
- Power-2 warmup for spectral flattening
- FISTA with adaptive LSE smoothing (global convergence)
- Exact subgradient polishing with growing active set
- Mirror descent for diversity
- Inner-product safety checks for FISTA stability

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta):
    """LSE gradient w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_grad(a, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power2_grad(a, c):
    """Gradient of sum(c^2) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(a, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_power2_warmup(a_init, max_time, rng):
    """Minimize sum(c^2) as spectral flattening warmup."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()

        grad = power2_grad(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (4.0 * c.max() / max(n, 1) + 4.0 * np.max(a) + 1.0)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        inner = np.dot(y, a)
        if inner < 0:
            y = x_new
        a = x_new
        t_mom = t_new
        if t_mom > 200:
            t_mom = 1.0

    return best_a


def run_fista_lse(a_init, max_time, rng, beta_init=5.0, beta_max=100000.0):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta schedule
        if stuck < 500:
            beta = beta_init + 0.02 * it
        elif stuck < 3000:
            beta = min(beta_max, beta * 1.003)
        elif stuck < 10000:
            beta = min(beta_max, beta * 0.015)
        else:
            beta = min(beta_max, beta * 0.08)

        beta_eff = min(beta, 5000.0)

        # Hybrid gradient: switch to top-k when very stuck
        if stuck > 5000 and it % 3 == 0:
            k = min(64, 2 + stuck // 500)
            grad = topk_grad(y, c, k, power=2.0)
        else:
            grad = lse_grad(y, c, beta_eff)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: balanced between Lipschitz and norm-based
        L = 2.0 * (1.0 + beta_eff * max(cur_max / max(S, 1e-10), 0.5))
        step = 0.8 / L
        step = min(step, S / (gnorm * 3.0 + 1e-30))
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        inner = np.dot(y, a)
        if inner < 0:
            y = x_new
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 10000:
            t_mom = 1.0
            noise = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 50000.0))
            y = project_simplex(y + rng.normal(0, noise, n), S)
            last_improve = it

        if it % 20000 == 0:
            t_mom = 1.0

    return best_a


def run_subgradient(a_init, max_time, rng):
    """FISTA-style with exact subgradient and growing active set."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k for active set size
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = min(4, 1 + stuck // 150)
        elif stuck < 2000:
            k = min(16, 4 + (stuck - 500) // 200)
        else:
            k = min(128, 16 + (stuck - 2000) // 300)

        grad = topk_grad(y, c, k, power=1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / max(gnorm, 1e-30)
        step = min(step, S / (gnorm * 2.0 + 1e-30))
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        inner = np.dot(y, a)
        if inner < 0:
            y = x_new
        a = x_new
        t_mom = t_mom

        if stuck > 5000:
            t_mom = 1.0
            noise = max(1e-9, 0.001 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise, n), S)
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def run_mirror_descent(a_init, max_time, rng, beta_init=10.0):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 300:
            beta = min(50000.0, beta * 1.01)

        grad = lse_grad(y, c, min(beta, 10000.0))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (max(beta, 1.0) * gnorm + 1e-30)
        step = min(step, 0.05)

        y_new = y * np.exp(-step * grad)
        s_new = y_new.sum()
        if s_new < 1e-15:
            break
        y_new = y_new * (S / s_new)

        c_check = fft_convolve_self(y_new)
        if c_check.max() < c.max():
            y = y_new
        elif it % 10 == 0:
            y = y_new

    return best_a


def make_inits(n, rng):
    """Create diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []

    # Cosine
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(normalize_seq(0.5 + 0.5 * np.cos(2 * np.pi * t), S))

    # Gaussian bump
    t = np.arange(n) / n - 0.5
    inits.append(normalize_seq(np.exp(-4.0 * t * t), S))

    # Random
    inits.append(normalize_seq(rng.random(n), S))

    # Random power spectrum
    f = rng.standard_normal(n)
    A = fft(f, n)
    a_spec = np.abs(A) ** 0.5 + 0.1
    inits.append(normalize_seq(a_spec, S))

    # Multi-harmonic
    t = np.arange(n) / n
    a = np.ones(n)
    for freq in [2, 3, 5, 7, 11]:
        a = a + 0.15 * np.cos(2.0 * np.pi * freq * t + rng.random() * 2.0 * np.pi)
    a = np.maximum(a, 0.01)
    inits.append(normalize_seq(a, S))

    # Tukey/Hann window
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    inits.append(normalize_seq(a, S))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-resolution FISTA search minimizing 2*n*max_conv/sum^2.

    Phases:
    1. Power-2 warmup (spectral flattening)
    2. FISTA LSE (main global optimization)
    3. Subgradient polish (active set refinement)
    4. Mirror descent (diversity)
    5. Scale up to n=32768 if time permits
    6. Final intensive polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get provided sequence as warm start
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Primary dimension
    n_primary = 16384
    S_primary = np.sqrt(2.0 * n_primary)

    print("=== Primary: n=%d ===" % n_primary)

    # Build initializations
    a_inits = []
    if a_start is not None:
        t_new = np.linspace(0, 1, n_primary, endpoint=False)
        t_old = np.linspace(0, 1, len(a_start), endpoint=False)
        a_interp = np.interp(t_new, t_old, a_start)
        a_inits.append(normalize_seq(a_interp, S_primary))

    for a_i in make_inits(n_primary, rng):
        a_inits.append(a_i)

    # Evaluate all initializations
    best_init = None
    best_init_obj = np.inf
    for i, a_i in enumerate(a_inits):
        obj = compute_obj(a_i)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_i
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_i.copy()
        print("  init %d: obj=%.6f" % (i, obj))

    print("Best init: obj=%.6f" % best_init_obj)

    # Phase 1: Power-2 warmup
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.08
    print("\n=== Phase 1: Power-2 warmup (%.0fs) ===" % p1)
    t0 = time.monotonic()
    a1 = run_power2_warmup(best_init, p1, rng)
    obj1 = compute_obj(a1)
    print("  obj=%.6f (t=%.1fs)" % (obj1, time.monotonic() - t0))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # Phase 2: FISTA LSE (main optimization)
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.35
    print("\n=== Phase 2: FISTA LSE (%.0fs) ===" % p2)
    t0 = time.monotonic()
    a2 = run_fista_lse(global_best_a, p2, rng, beta_init=3.0, beta_max=50000.0)
    obj2 = compute_obj(a2)
    print("  obj=%.6f (t=%.1fs)" % (obj2, time.monotonic() - t0))
    if obj2 < global_best_obj:
        global_best_obj = obj2
        global_best_a = a2.copy()

    # Phase 3: Subgradient polish
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.22
    print("\n=== Phase 3: Subgradient polish (%.0fs) ===" % p3)
    t0 = time.monotonic()
    a3 = run_subgradient(global_best_a, p3, rng)
    obj3 = compute_obj(a3)
    print("  obj=%.6f (t=%.1fs)" % (obj3, time.monotonic() - t0))
    if obj3 < global_best_obj:
        global_best_obj = obj3
        global_best_a = a3.copy()

    # Phase 4: Mirror descent for exploration
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p4 = remaining * 0.13
        print("\n=== Phase 4: Mirror descent (%.0fs) ===" % p4)
        t0 = time.monotonic()
        a4 = run_mirror_descent(global_best_a, p4, rng, beta_init=20.0)
        obj4 = compute_obj(a4)
        print("  obj=%.6f (t=%.1fs)" % (obj4, time.monotonic() - t0))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # Phase 5: Upscale to n=32768
    remaining = deadline - time.monotonic()
    if remaining > 400:
        n_large = 32768
        S_large = np.sqrt(2.0 * n_large)
        p5 = remaining * 0.65
        print("\n=== Phase 5: Upscale to n=%d (%.0fs) ===" % (n_large, p5))
        t0 = time.monotonic()

        t_new = np.linspace(0, 1, n_large, endpoint=False)
        t_old = np.linspace(0, 1, len(global_best_a), endpoint=False)
        a5_init = np.interp(t_new, t_old, global_best_a)
        a5_init = normalize_seq(a5_init, S_large)

        a5 = run_fista_lse(a5_init, p5 * 0.55, rng, beta_init=3.0, beta_max=30000.0)
        obj5 = compute_obj(a5)
        print("  LSE: obj=%.6f (t=%.1fs)" % (obj5, time.monotonic() - t0))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5 - (time.monotonic() - t0)
        if rem5 > 30:
            a5b = run_subgradient(global_best_a, rem5, rng)
            obj5b = compute_obj(a5b)
            print("  Polish: obj=%.6f (t=%.1fs)" % (obj5b, time.monotonic() - t0))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # Phase 6: Final intensive polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p6 = remaining * 0.85
        print("\n=== Phase 6: Final polish (%.0fs) ===" % p6)
        t0 = time.monotonic()

        a6 = run_subgradient(global_best_a, p6 * 0.5, rng)
        obj6 = compute_obj(a6)
        print("  Subgrad: obj=%.6f (t=%.1fs)" % (obj6, time.monotonic() - t0))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

        rem6 = p6 - (time.monotonic() - t0)
        if rem6 > 10:
            a6b = run_fista_lse(global_best_a, rem6, rng, beta_init=100.0, beta_max=100000.0)
            obj6b = compute_obj(a6b)
            print("  LSE-high: obj=%.6f (t=%.1fs)" % (obj6b, time.monotonic() - t0))
            if obj6b < global_best_obj:
                global_best_obj = obj6b
                global_best_a = a6b.copy()

    # Final verification and output
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f, n=%d ===" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 16384
            final_list = [float(x) for x in np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)]
    else:
        n_fb = 16384
        final_list = [float(x) for x in np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)]

    return final_list
