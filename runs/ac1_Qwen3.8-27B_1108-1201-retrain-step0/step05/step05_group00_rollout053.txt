def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv of length 2n-1."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_fft_cached(a, N, cache=None):
    """Self-convolution, returns (conv, A_fft) for reuse."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A, Nf

    top_vals = c[top_idx]
    top_max = top_vals.max()

    exp_bc = np.exp(beta * (top_vals - top_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights_top = np.zeros(K)
        weights_top[int(np.argmax(top_vals))] = 1.0
        lse_val = top_max
    else:
        weights_top = exp_bc / total
        lse_val = top_max + np.log(total) / beta

    # Build full weight vector
    weights = np.zeros(2 * n - 1)
    weights[top_idx] = weights_top

    # Gradient: 2 * ifft(fft(weights) * conj(fft(a)))[ :n ]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max, top_idx


def exact_subgrad(a, N, k_star):
    """Exact subgradient of max_k conv[k] at peak position k_star.
    grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def topk_exact_subgrad(a, N, top_indices, top_vals):
    """Weighted combination of exact subgradients for top peaks."""
    n = len(a)
    # Softmax weights on top peaks
    mv = top_vals.max()
    w = np.exp(5.0 * (top_vals - mv))
    w /= w.sum()
    grad = np.zeros(n)
    for kk, ww in zip(top_indices, w):
        int_kk = int(kk)
        lo = max(0, int_kk - n + 1)
        hi = min(n - 1, int_kk)
        for i in range(lo, hi + 1):
            j = int_kk - i
            if 0 <= j < n:
                grad[i] += ww * 2.0 * a[j]
    return grad


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (fraction frac) while preserving magnitudes, project to non-negative."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    rand_idx = rng.choice(Nf, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    # Enforce conjugate symmetry for real output
    new_phase[Nf // 2] = 0.0
    if Nf > 1:
        for i in range(1, Nf // 2):
            new_phase[Nf - i] = -new_phase[i]
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=20.0, beta_max=3000.0,
                    m_lbfgs=12, top_k=1, top_k_max=4, beta_ramp_rate=1.02):
    """
    L-BFGS on top-K LSE-smoothed objective with adaptive beta ramp-up.
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    k = min(top_k, 2 * n - 1)
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c, c_max, _ = topk_lse_grad(a, N, beta, k)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 4.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    last_improvement = c_max

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta ramp
        if it % 50 == 0:
            beta = min(beta_max, beta * beta_ramp_rate)
            k = min(top_k_max, top_k + it // 200)
            step_scale = 4.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max, _ = topk_lse_grad(a, N, beta, k)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        # History management
        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # Get LBFGS direction
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Line search (backtracking)
        alpha = step_scale
        a_new = a.copy()
        cmax_new = c_max
        improved = False
        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max + 1e-12:
                a_new = candidate
                cmax_new = cm_ls
                improved = True
                break
            alpha *= 0.5
        if not improved:
            a_new = candidate
            cmax_new = cm_ls

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Compute new gradient
        g_new, f_new, c_nw, cmax_nw, _ = topk_lse_grad(a_new, N, beta, k)

        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        # Periodic spectral restart if stuck
        if stall_count > 80 and it % 100 == 0:
            a = spectral_phase_restart(a, N, rng, frac=0.2)
            s_hist = []
            y_hist = []
            stall_count = 0
            beta = max(20.0, beta * 0.5)  # Reset beta on restart
            g, f_val, c, c_max, _ = topk_lse_grad(a, N, beta, k)

    return best_a, best_cmax


def exact_subgrad_block(a_init, n, S, N, max_time, rng):
    """
    Exact subgradient method on max(conv) with adaptive step size.
    Uses top-1 peak with occasional top-3.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    step = S / n  # Initial step size
    use_top3 = False

    while time.monotonic() - start_time < max_time:
        it += 1

        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        if use_top3 or it % 5 == 0:
            K = min(3, 2 * n - 1)
            top_idx = np.argpartition(c, -K)[-K:]
            top_vals = c[top_idx]
            g = topk_exact_subgrad(a, N, top_idx, top_vals)
        else:
            k_star = int(np.argmax(c))
            g = exact_subgrad(a, N, k_star)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break

        # Adaptive step: proportional to the "excess" of the peak over the mean
        mean_c = c.sum() / len(c)
        excess = max(0.0, c_max - mean_c)
        step = max(1e-6, min(S / (2 * n), excess / (gnorm * gnorm + 1e-30) * 2.0 * S))

        a_new = project_simplex(a - step * g / gnorm, S)
        c_new = compute_conv(a_new, N)
        cm_new = c_new.max()

        if cm_new <= c_max:
            a = a_new
            # Increase step slightly if we made good progress
            step = min(S / n, step * 1.05)
        else:
            # Step too large, reduce
            step *= 0.5
            a = project_simplex(a - step * g / gnorm, S)

        # Occasional small random perturbation
        if it % 500 == 0:
            a += rng.standard_normal(n) * 1e-5 * S / n
            a = project_simplex(a, S)

    return best_a, best_cmax


def deflation_init(n, S, N, n_iters=3000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        imax = int(np.argmax(contrib))
        imin = int(np.argmin(contrib))
        if imax == imin:
            break
        transfer = min(0.01 * a[imax], 0.2 * a[imax])
        if transfer < 1e-15:
            break
        a[imax] -= transfer
        a[imin] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (7 * tt + 3 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_flat_fft(n, N, rng):
    """Try to make the DFT magnitude flat (low PAPR in circular sense)."""
    Nf = int(N)
    if Nf < n:
        Nf = 1 << n.bit_length()
    target_mag = np.ones(Nf) * (n / Nf) ** 0.5
    phase = rng.uniform(0, 2 * np.pi, Nf)
    phase[0] = 0
    phase[Nf // 2] = 0
    for i in range(1, Nf // 2):
        phase[Nf - i] = -phase[i]
    A = target_mag * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses warm start + adaptive top-K LSE L-BFGS + exact subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    # Try to use given sequence length if reasonable
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 2048 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.8f score=%.6f" % (label, sc, score))
        return sc

    # --- Get initial sequence ---
    a_init = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_init = np.clip(given, 0, 1000).astype(np.float64)
            print("Using given sequence as warm start, current peak:", end=" ")
            a_norm = safe_normalize(a_init, n, S)
            c0 = compute_conv(a_norm, N)
            print("%.8f (score=%.6f)" % (c0.max(), 2 * n * c0.max() / (S * S)))
    except Exception:
        pass

    if a_init is None:
        a_init = np.ones(n) * S / n
        print("No given sequence, using all-ones init.")

    a_init = safe_normalize(a_init, n, S)
    update_best(a_init, "init")

    # --- Phase 1: Quick LSE L-BFGS (moderate beta, 120s) ---
    print("=== Phase 1: Quick LSE L-BFGS (moderate beta) ===")
    t1 = time.monotonic()
    p1_budget = min(120.0, (deadline - t1) * 0.08)
    if p1_budget > 10:
        a_p1, sc_p1 = lbfgs_lse_block(a_init, n, S, N, p1_budget, rng,
                                       beta_init=10.0, beta_max=200.0,
                                       m_lbfgs=10, top_k=1, top_k_max=3,
                                       beta_ramp_rate=1.03)
        update_best(a_p1, "P1_LSE")

    # --- Phase 2: Main LSE L-BFGS (high beta, ~60% of remaining) ---
    print("=== Phase 2: Main LSE L-BFGS (high beta) ===")
    t2 = time.monotonic()
    p2_budget = min(1200.0, (deadline - t2) * 0.55)

    a_curr = global_a.copy() if global_a is not None else a_init
    stall = 0

    while time.monotonic() < t2 + p2_budget:
        block_time = min(200.0, (t2 + p2_budget) - time.monotonic())
        if block_time < 15:
            break
        beta_start = 30.0 + 10.0 * min(stall, 15)
        beta_end = min(5000.0, 200.0 * (1 + 0.5 * stall))
        tk = 1 if stall < 3 else 2
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         m_lbfgs=15, top_k=tk, top_k_max=4,
                                         beta_ramp_rate=1.015)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2 * n * sc_new / (S * S)
            print("  [P2] score=%.6f (beta %.0f-%.0f, tk=%d)" % (score, beta_start, beta_end, tk))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 3:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.3)
                stall = 0
            elif stall >= 5:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=1.0)
                stall = 0
            elif stall >= 2:
                # Small perturbation
                a_curr = a_curr + rng.standard_normal(n) * 1e-4 * S / n
                a_curr = project_simplex(a_curr, S)

    # --- Phase 3: Exact subgradient polishing ---
    print("=== Phase 3: Exact subgradient polishing ===")
    t3 = time.monotonic()
    p3_budget = min(400.0, (deadline - t3) * 0.2)
    if p3_budget > 20:
        a_p3, sc_p3 = exact_subgrad_block(global_a, n, S, N, p3_budget, rng)
        update_best(a_p3, "P3_subgrad")

    # --- Phase 4: Final high-beta LSE ---
    print("=== Phase 4: Final high-beta LSE ===")
    t4 = time.monotonic()
    p4_budget = min(300.0, deadline - t4 - 20.0)
    if p4_budget > 20:
        a_p4, sc_p4 = lbfgs_lse_block(global_a, n, S, N, p4_budget, rng,
                                       beta_init=100.0, beta_max=10000.0,
                                       m_lbfgs=20, top_k=1, top_k_max=1,
                                       beta_ramp_rate=1.01)
        update_best(a_p4, "P4_finalLSE")

    # --- Phase 5: Final subgradient squeeze ---
    t5 = time.monotonic()
    p5_budget = deadline - t5 - 10.0
    if p5_budget > 15:
        a_p5, sc_p5 = exact_subgrad_block(global_a, n, S, N, min(p5_budget, 150.0), rng)
        update_best(a_p5, "P5_finalSubgrad")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.8f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
