def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value_and_grad(c, beta):
    """Compute LSE value and subgradient weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    lse = cm + np.log(tot) / beta
    w = e / tot
    return lse, w


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def run_fista_improved(a_init, max_time, rng, beta_start=20.0, mode="adaptive"):
    """
    Improved FISTA with backtracking line search and adaptive beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    beta = beta_start
    step = 1.0 / beta

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: slowly increase, with periodic drops
        if it % 500 == 0:
            beta *= 1.2
        if it % 3000 == 0:
            beta = max(beta_start, beta * 0.3)
        beta = min(beta, 100000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Get gradient
        _, w = lse_value_and_grad(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search (simplified)
        step = 1.0 / (beta * 0.8)
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            # Check sufficient decrease in LSE
            lse_new, _ = lse_value_and_grad(c_new, beta)
            lse_cur, _ = lse_value_and_grad(c, beta)
            if lse_new <= lse_cur - 1e-8 * step * gnorm * gnorm:
                break
            step *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_next = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_next = np.maximum(y_next, 0.0)
        a = x_new
        y = y_next
        t_mom = t_new

        # Periodic momentum reset and noise when stuck
        if stuck_count > 1500:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, noise_scale, n)
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = 0
                a = y.copy()

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lp_frank_wolfe(a_init, max_time, rng, n_constraints=200):
    """
    LP-based Frank-Wolfe: solve LP to find best direction maintaining peak constraint.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())

    start = time.monotonic()
    best_f = f.copy()
    best_conv = fft_convolve_self(best_f).max()
    it = 0
    step_t = 0.1
    consecutive_fail = 0

    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(f)
        M = b.max()

        if M < best_conv - 1e-14:
            best_conv = M
            best_f = f.copy()
            consecutive_fail = 0

        # Find active constraints (top-K lags near max)
        active_k = min(n_constraints, len(b))
        top_idx = np.argsort(b)[::-1][:active_k]
        active_vals = b[top_idx]
        # Use a slightly relaxed threshold
        threshold = active_vals.min() * 1.001

        # Build constraint matrix: for each active lag k, the coefficients of g
        # (f*g)_k = sum_i f_{k-i} * g_i (where indices are valid)
        A_rows = []
        for k in top_idx:
            row = np.zeros(n)
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                fi = k - i  # index into f
                if 0 <= fi < n:
                    row[i] = f[fi]
            A_rows.append(row)
        A_mat = np.array(A_rows)
        b_vec = np.full(len(top_idx), M)

        # Solve LP: max sum(g) s.t. A_mat @ g <= b_vec, g >= 0
        try:
            c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs')
            if res.success:
                g = res.x
                g_sum = g.sum()
                if g_sum > S * 1.001:
                    # Normalize g to have sum S
                    g = g * (S / g_sum)
                    # Take a step toward g
                    f_new = (1 - step_t) * f + step_t * g
                    f_new = np.maximum(f_new, 0)
                    f_new = f_new * (S / f_new.sum())
                    # Check if improvement
                    b_new = fft_convolve_self(f_new)
                    if b_new.max() <= M * (1 + 1e-6):
                        f = f_new
                        consecutive_fail = 0
                        step_t = min(step_t * 1.1, 0.5)
                    else:
                        # Peak increased, reduce step
                        step_t *= 0.7
                        consecutive_fail += 1
                else:
                    # No improvement possible from LP
                    consecutive_fail += 1
            else:
                consecutive_fail += 1
        except Exception as e:
            consecutive_fail += 10

        # If stuck, try a different approach: perturb and re-solve
        if consecutive_fail > 20:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            f = project_simplex(f + noise, S)
            step_t = 0.05
            consecutive_fail = 0

        # Periodic step size reset
        if it % 50 == 0:
            step_t = max(0.01, step_t * 0.95)

    c_final = fft_convolve_self(best_f)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_f, best_conv


def run_slsqp_polish(a_init, max_time, rng, n_beta_levels=4):
    """SLSQP polishing with LSE objective and direct sum constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [50.0, 200.0, 500.0, 1500.0][:n_beta_levels]

    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            c = fft_convolve_self(x)
            lse, w = lse_value_and_grad(c, beta)
            grad = conv_gradient(x, w)
            return lse, grad

        # Constraint: sum(x) = S
        cons = [{'type': 'eq',
                 'fun': lambda x: x.sum() - S,
                 'jac': lambda x: np.ones(len(x))}]

        time_budget = remaining * 0.85
        t0 = time.monotonic()

        try:
            result = scipy_minimize(
                obj_and_grad, a.copy(), method='SLSQP',
                jac=True, bounds=[(0, 1000)] * n,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0}
            )
            a_new = np.clip(result.x, 0, 1000)
            if a_new.sum() > 1e-10:
                a_new = a_new * (S / a_new.sum())
                obj_new = compute_obj(a_new)
                elapsed = time.monotonic() - t0
                print("    SLSQP beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
                      (beta, obj_new, elapsed, result.nit))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
        except Exception as e:
            print("    SLSQP beta=%.0f failed: %s" % (beta, str(e)))

    return best_a, best_obj


def run_gradient_descent_exact(a_init, max_time, rng):
    """
    Gradient descent using exact subgradient of max (not LSE).
    Uses Barzilai-Borwein step sizes and periodic restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    prev_a = a.copy()
    prev_grad = None
    y_mom = a.copy()
    t_mom = 1.0

    # BB step size parameters
    bb_s = None
    bb_y = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Find active set
        if stuck < 500:
            active = np.array([int(np.argmax(c))])
        elif stuck < 2000:
            tol = 0.002
            active = np.where(c >= cur_max * (1 - tol))[0][:20]
        else:
            tol = min(0.05, 0.005 * stuck / 1000)
            active = np.where(c >= cur_max * (1 - tol))[0][:50]

        # Subgradient: average of active shifts
        n_active = len(active)
        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / n_active
        grad = conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: BB if available, else fixed
        if bb_s is not None and bb_y is not None:
            sy = np.dot(bb_s, bb_y)
            yy = np.dot(bb_y, bb_y)
            if yy > 1e-20 and sy > 1e-20:
                step_bb = sy / yy
                step = np.clip(step_bb, 0.01 * S / gnorm, 2.0 * S / gnorm)
            else:
                step = S / gnorm
        else:
            step = S / (gnorm * (1.0 + 0.01 * it % 100))

        x_new = project_simplex(a - step * grad, S)

        # Update BB parameters
        if prev_grad is not None:
            bb_s = x_new - prev_a
            bb_y = grad - prev_grad
            # Check BB condition
            if np.dot(bb_s, bb_y) <= 1e-10:
                bb_s = None
                bb_y = None

        prev_a = a.copy()
        prev_grad = grad.copy()

        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_mom = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_mom = np.maximum(y_mom, 0.0)
        a = x_new
        t_mom = t_new

        # Restarts
        if stuck > 1000 or it % 5000 == 0:
            t_mom = 1.0
            bb_s = None
            bb_y = None
            if stuck > 2000:
                noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
                if noise_scale > 1e-9:
                    y_mom = project_simplex(y_mom + rng.normal(0, noise_scale, n), S)
                    t_mom = 1.0
                    stuck = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def generate_initial(n, rng, S):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine
    for freq in [1, 2]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.001)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Gaussian-like
    center = n // 2
    a = np.exp(-0.5 * ((np.arange(n) - center) / (n / 6)) ** 2)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Smooth random
    raw = rng.random(n)
    F = fft(raw)
    cutoff = max(3, n // 50)
    F[cutoff:-cutoff] *= 0.01
    smooth = np.real(ifft(F))[:n]
    smooth = np.maximum(smooth, 0.001)
    smooth = smooth * (S / smooth.sum())
    inits.append(smooth.copy())

    # Uniform
    inits.append(np.ones(n) * (S / n))

    return inits

    return final_list
