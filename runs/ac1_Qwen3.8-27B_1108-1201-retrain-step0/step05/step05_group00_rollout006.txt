"""
Minimize 2*n*max_conv/sum^2 using adaptive top-K gradient descent with Nesterov
momentum, coordinate descent polishing, basin-hopping multistart, and structured
escape moves. Key innovations: (1) direct top-K peak gradient avoiding LSE,
(2) two-timescale optimization (low-freq shape + high-freq detail), (3) 
impact-ordered coordinate descent, (4) simulated-annealing basin hopping.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (2n-1)-length array."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def topk_gradient(a, c, top_k, F):
    """
    Compute gradient of weighted top-K peaks of conv(a,a) w.r.t. a.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sorted_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    W = np.zeros(F)
    W[sorted_idx] = weights
    A = np.zeros(F)
    A[:n] = a
    grad = np.real(ifft(np.conj(fft(W)) * fft(A)))[:n]
    return grad


def safe_normalize(a, n, S):
    """Clip to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def nesterov_topk(a_init, n, S, N, F, max_time, rng, K_init=1, K_max=120,
                  step_base=None, noise_frac=0.0):
    """
    Nesterov-accelerated top-K gradient descent with adaptive K schedule.
    Returns (best_a, best_cmax).
    """
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t_mom = 1.0
    if step_base is None:
        step_base = 0.5 * S / n
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    stall = 0
    stall_threshold = 200

    while time.monotonic() - start_time < max_time:
        it += 1
        K = min(K_max, K_init + it // 80)
        beta_anneal = 1.0 + 0.5 * min(it / 500.0, 3.0)

        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax - 1e-12:
            best_cmax = cur
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Periodic escape if stalled
        if stall > stall_threshold and stall % 50 == 0:
            n_perturb = max(3, n // 64)
            idx_pert = rng.choice(n, size=n_perturb, replace=False)
            perturbation = rng.standard_normal(n_perturb) * step_base * 2.0
            y[idx_pert] += perturbation
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * (S / s_y)
            stall = 0
            t_mom = 1.0
            continue

        grad = topk_gradient(y, c, K, F)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step size with Nesterov
        step = step_base / (1.0 + 0.002 * it) * beta_anneal
        if noise_frac > 0:
            noise = rng.standard_normal(n) * noise_frac * step
            grad = grad + gnorm * noise

        x_new = project_simplex(y - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Final check
    c = compute_conv(y, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = y.copy()

    return best_a, best_cmax


def coordinate_descent(a_init, n, S, N, max_time, rng, n_passes_max=50):
    """
    Impact-ordered coordinate descent with mini line search.
    For each coordinate, try adjusting it to reduce the peak.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    n_passes = 0

    while time.monotonic() - start_time < max_time and n_passes < n_passes_max:
        n_passes += 1
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Determine impact order: coordinates that most affect the peak
        # db_k*/da_i = a_{k*-i} (if valid)
        impact = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                impact[i] = a[j]

        # Process in order of impact (highest first)
        order = np.argsort(impact)[::-1]
        # Mix in some randomness for exploration
        if rng.random() < 0.3:
            n_mix = n // 4
            order[:n_mix] = rng.choice(n, size=n_mix, replace=False)

        improved = False
        for i in order:
            if time.monotonic() - start_time > max_time - 5:
                break
            a_old = a[i]
            if a_old < 1e-15:
                # Try adding to this coordinate
                delta = 0.01 * S / n
            else:
                # Try reducing
                delta = -0.1 * a_old

            # Try a few candidate values
            candidates = [a_old + delta, a_old * 0.5, a_old + 0.05 * S / n, max(0, a_old - 0.3 * a_old)]
            best_delta = 0
            best_local = peak_val

            for cand in candidates:
                if cand < 0:
                    continue
                a[i] = cand
                # Adjust to maintain sum (change coordinate j that's far from peak)
                delta_sum = S - a.sum()
                if abs(delta_sum) > 1e-15:
                    # Add/remove from a low-impact coordinate
                    j = int(order[-1])
                    if j != i:
                        a[j] = max(0, a[j] + delta_sum)
                    else:
                        a[i] = a_old
                        continue
                c_new = compute_conv(a, N)
                new_peak = c_new.max()
                if new_peak < best_local - 1e-12:
                    best_local = new_peak
                    best_delta = cand
                a[i] = a_old

            a[i] = best_delta
            if best_delta != a_old:
                improved = True

        c = compute_conv(a, N)
        if c.max() < best_cmax:
            best_cmax = c.max()
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_cmax


def lowfreq_perturb(a, n, rng, n_modes=3, amplitude=0.1):
    """Perturb only the low-frequency components of the sequence."""
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    # Perturb low-frequency bins
    idx = list(range(1, n_modes + 1)) + list(range(N - n_modes, N))
    for k in idx:
        if k < N:
            A[k] += amplitude * (rng.standard_normal() + 1j * rng.standard_normal()) * np.abs(A[0]) / n
            A[N - k] = np.conj(A[k]) if N - k < N else A[k]
    a_new = np.real(ifft(A))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)


def basin_hopping(begin_a, n, S, N, F, max_time, rng, n_tries=20):
    """
    Basin hopping: try random perturbations, optimize each briefly, keep best.
    """
    best_a = begin_a.copy()
    best_cmax = compute_conv(begin_a, N).max()
    start_time = time.monotonic()
    per_try = max_time / max(1, n_tries)

    for trial in range(n_tries):
        if time.monotonic() - start_time > max_time - per_try:
            break
        # Generate perturbation of decreasing "temperature"
        frac = 1.0 - trial / max(1, n_tries)
        temp = frac * 0.15 + 0.01

        # Types of perturbations
        r = rng.random()
        if r < 0.3:
            # Random noise
            a_pert = begin_a + rng.standard_normal(n) * temp * S / n
        elif r < 0.6:
            # Low-frequency perturbation
            a_pert = lowfreq_perturb(begin_a, n, rng, n_modes=2 + trial % 5,
                                      amplitude=temp)
        else:
            # Block perturbation: change a contiguous block
            block_len = max(4, n // (8 + trial // 3))
            start_pos = rng.integers(0, n - block_len)
            a_pert = begin_a.copy()
            a_pert[start_pos:start_pos + block_len] *= (1.0 + rng.standard_normal(block_len) * temp)

        a_pert = safe_normalize(a_pert, n, S)

        # Quick Nesterov optimization
        t_rem = max(1.0, per_try - (time.monotonic() - start_time) - 0.5)
        a_opt, cmax = nesterov_topk(a_pert, n, S, N, F, min(t_rem, per_try * 0.8),
                                     rng, K_init=2, K_max=40,
                                     step_base=0.3 * S / n)
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a_opt.copy()

    return best_a, best_cmax


def two_timescale_block(a_init, n, S, N, F, max_time, rng):
    """
    Two-timescale optimization: alternate between low-frequency adjustment
    (slow) and full-resolution Nesterov (fast).
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    Nf = N  # full resolution FFT size

    # Low-frequency FFT size (much smaller)
    n_low = 64
    N_low = 256

    it = 0
    while time.monotonic() - start_time < max_time:
        it += 1

        # Fast timescale: full-resolution Nesterov (short burst)
        t_fast = min(30.0, (max_time - (time.monotonic() - start_time)) * 0.7)
        if t_fast < 5:
            break
        K = min(80, 2 + it * 2)
        a, cmax = nesterov_topk(a, n, S, N, F, t_fast, rng,
                                 K_init=K, K_max=K + 20,
                                 step_base=0.4 * S / n)
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        # Slow timescale: low-frequency perturbation + re-optimize
        if it % 3 == 0 and time.monotonic() - start_time < max_time - 60:
            a_pert = lowfreq_perturb(a, n, rng, n_modes=3, amplitude=0.05)
            a_slow, cmax_slow = nesterov_topk(a_pert, n, S, N, F,
                                               min(40.0, max_time - (time.monotonic() - start_time)),
                                               rng, K_init=5, K_max=30,
                                               step_base=0.5 * S / n)
            if cmax_slow < best_cmax:
                best_cmax = cmax_slow
                best_a = a_slow.copy()
                a = a_slow.copy()

    return best_a, best_cmax

def make_init_fourier(n, rng, N):
    """Random Fourier magnitude with random phases, take abs."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_zadoff(n, rng):

    a = 1.0 + 0.6 * np.cos(phase) + 0.08 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def proposal_lse_block(a_init, n, S, N, F, max_time, rng, beta_init=8.0, beta_max=300.0):
    """
    LSE-based L-BFGS style optimization (simpler than original, kept as backup).
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    t_mom = 1.0

    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)

        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        # LSE gradient
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total > 1e-300:
            weights = exp_bc / total
        else:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0

        # Compute gradient via FFT
        W = np.zeros(F)
        W[:len(c)] = weights
        A = np.zeros(F)
        A[:n] = a
        grad = np.real(ifft(np.conj(fft(W)) * fft(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = 0.5 * S / (n * max(beta, 1.0))
        x_new = project_simplex(a - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive top-K Nesterov, two-timescale optimization, coordinate
    descent, and basin hopping.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    F = 1 << (3 * n - 2).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, F(FFT)=%d, budget=%.0fs" % (n, S, N, F, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f (it=%.0fs)" % (label, sc, score, time.monotonic() - (budget_s - (deadline - time.monotonic()))))
            return True
        return False

    # Get initial sequence
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_start = np.clip(given, 0, 1000).astype(np.float64)
            a_start = safe_normalize(a_start, n, S)
            update_best(a_start, "given")
    except Exception:
        pass

    if a_start is None:
        a_start = safe_normalize(make_init_rand(n, rng), n, S)

    # === Phase 1: Multi-start exploration (15% of time) ===
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(300.0, (deadline - t1) * 0.15)
    t1_end = t1 + phase1_budget

    inits = [("given", a_start)]
    inits.append(("rand", safe_normalize(make_init_rand(n, rng), n, S)))
    inits.append(("chirp", safe_normalize(make_init_chirp(n, rng), n, S)))
    inits.append(("multifreq", safe_normalize(make_init_multi_freq(n, rng), n, S)))
    inits.append(("fourier", safe_normalize(make_init_fourier(n, rng, N), n, S)))
    inits.append(("zadoff", safe_normalize(make_init_zadoff(n, rng), n, S)))

    per_init = phase1_budget / max(1, len(inits))
    for name, arr in inits:
        if time.monotonic() > t1_end - 10:
            break
        # Quick LSE optimization
        trem = max(2.0, per_init - 1.0)
        a_l, sc_l = proposal_lse_block(arr, n, S, N, F, min(trem, 50.0), rng,
                                        beta_init=5.0, beta_max=100.0)
        update_best(a_l, name + "/LSE")
        # Quick Nesterov
        if time.monotonic() < t1_end - 5:
            a_n, sc_n = nesterov_topk(a_l, n, S, N, F, min(20.0, t1_end - time.monotonic()),
                                       rng, K_init=1, K_max=30, step_base=0.4 * S / n)
            update_best(a_n, name + "/Nest")

    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    # === Phase 2: Main Nesterov with adaptive K (45% of time) ===
    print("=== Phase 2: Main Nesterov top-K ===")
    t2 = time.monotonic()
    phase2_budget = min(1100.0, (deadline - t2) * 0.45)
    t2_end = t2 + phase2_budget

    a_curr = global_a.copy() if global_a is not None else a_start.copy()
    step_base = 0.4 * S / n
    stall_count = 0
    best_so_far_p2 = global_best

    while time.monotonic() < t2_end - 10:
        block_time = min(120.0, t2_end - time.monotonic())
        if block_time < 5:
            break

        # Adaptive K: start focused, expand
        K_init = 1 + stall_count * 3
        K_max = 30 + stall_count * 20

        a_new, sc_new = nesterov_topk(a_curr, n, S, N, F, block_time, rng,
                                       K_init=min(K_init, 20), K_max=min(K_max, 150),
                                       step_base=step_base)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            step_base = min(0.8 * S / n, step_base * 0.9)
            score = 2 * n * sc_new / (S * S)
            print("  [P2] score=%.6f K=%d-%d stall_reset" % (score, K_init, K_max))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            step_base = step_base * 1.1  # increase step to escape

            if stall_count >= 2:
                # Escape: low-freq perturbation
                a_curr = lowfreq_perturb(a_curr, n, rng, n_modes=4, amplitude=0.08)
                stall_count = 0
            elif stall_count >= 4:
                # Bigger escape
                a_curr = lowfreq_perturb(a_curr, n, rng, n_modes=8, amplitude=0.15)
                a_curr += rng.standard_normal(n) * 0.02 * S / n
                a_curr = safe_normalize(a_curr, n, S)
                stall_count = 0

    # === Phase 3: Two-timescale optimization (20% of time) ===
    print("=== Phase 3: Two-timescale ===")
    t3 = time.monotonic()
    phase3_budget = min(500.0, (deadline - t3) * 0.20)
    if phase3_budget > 30:
        a_2t, sc_2t = two_timescale_block(global_a, n, S, N, F, phase3_budget, rng)
        update_best(a_2t, "two_timescale")

    # === Phase 4: Basin hopping (10% of time) ===
    print("=== Phase 4: Basin hopping ===")
    t4 = time.monotonic()
    phase4_budget = min(250.0, (deadline - t4) * 0.12)
    if phase4_budget > 20:
        a_bh, sc_bh = basin_hopping(global_a, n, S, N, F, phase4_budget, rng,
                                     n_tries=15)
        update_best(a_bh, "basin_hopping")

    # === Phase 5: Coordinate descent polishing ===
    print("=== Phase 5: Coordinate descent polishing ===")
    t5 = time.monotonic()
    phase5_budget = min(200.0, deadline - t5 - 15)
    if phase5_budget > 10:
        a_cd, sc_cd = coordinate_descent(global_a, n, S, N, phase5_budget, rng,
                                          n_passes_max=30)
        update_best(a_cd, "coord_descent")

    # === Phase 6: Final sprint - high-K Nesterov ===
    print("=== Phase 6: Final sprint ===")
    t6 = time.monotonic()
    remaining6 = deadline - t6 - 10
    if remaining6 > 10:
        a_fin, sc_fin = nesterov_topk(global_a, n, S, N, F, remaining6, rng,
                                       K_init=10, K_max=200,
                                       step_base=0.2 * S / n,
                                       noise_frac=0.01)
        update_best(a_fin, "final_sprint")

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf (rejected)")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
