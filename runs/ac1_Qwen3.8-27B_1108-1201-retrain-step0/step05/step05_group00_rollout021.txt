```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient methods with
LSE smoothing, direct argmax polishing, and structural restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Full linear convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_grad_topk(a, weights, N):
    """
    Gradient of sum_k weights[k]*c[k] where c=conv(a,a).
    Uses symmetry: grad[j] = conv(w,a)[j] + conv(w,a)[2n-2-j]
    """
    n = len(a)
    w_len = len(weights)
    # Need FFT size >= (2n-1) + n - 1 = 3n-2
    L = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:w_len] = weights
    A = fft(a_pad)
    W = fft(w_pad)
    conv = np.real(ifft(W * A))
    # grad[j] = conv[j] + conv[2n-2-j] for j=0,...,n-1
    grad = conv[:n] + conv[n-1:2*n-1][::-1]
    return grad


def lse_objective_and_grad(a, N, beta):
    """Compute LSE-smoothed peak and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        # Uniform weighting
        w = np.ones(2*n-1) / (2*n-1)
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w = exp_bc / total
    lse_val = c_max + np.log(w.sum() * np.exp(0)) / beta  # This is just c_max + log(sum(exp))/beta
    # Actually: LSE = c_max + log(sum(exp(beta*(c-c_max))))/beta = c_max + log(total)/beta
    lse_val = c_max + np.log(total) / beta if beta >= 1e-6 else c_max
    grad = compute_grad_topk(a, w, N)
    return lse_val, c_max, grad, c, w


def direct_subgrad(a, N, top_k=1):
    """Direct subgradient at the top-k lags."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    # Get top-k lags
    if top_k >= 2*n-1:
        top_idx = np.arange(2*n-1)
        w = np.ones(2*n-1) / (2*n-1)
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(2*n-1)
        # Weight proportional to (c[top_idx] - c_max + epsilon) or just uniform
        w[top_idx] = 1.0 / top_k
    grad = compute_grad_topk(a, w, N)
    return c_max, grad, c


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_two_loop(s_hist, y_hist, g, m):
    """L-BFGS two-loop recursion to compute search direction."""
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy_i = np.dot(y_i, y_i)
        if yy_i < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy_i
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_lse_phase(a_init, n, S, N, max_time, rng, beta_init, beta_max, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    f_val, c_max, g, c, w = lse_objective_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    stall_best = best_cmax
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive beta increase
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            f_val, c_max, g, c, w = lse_objective_and_grad(a, N, beta)
        
        # Truncate history
        if it % 30 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        # Compute direction
        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search (backtracking on LSE value, but track actual c_max)
        alpha = step_scale
        a_new = a
        cmax_new = c_max
        f_new = f_val
        
        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            # Check sufficient decrease on LSE
            f_ls, _, _, _, _ = lse_objective_and_grad(candidate, N, beta)
            if f_ls <= f_val + 1e-4 * alpha * np.dot(direction, g) or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                f_new = f_ls
                break
            alpha *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
    
    return best_a, best_cmax
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive LSE-smoothed L-BFGS with
direct subgradient polishing, active-set identification, and structured restarts.
Key improvements: (1) symmetry-exploiting gradient, (2) exact-objective line search,
(3) adaptive top-K subgradient polishing, (4) structured geometric restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_gradient_weighted(a, weights, N_grad):
    """
    Gradient of sum_k weights[k] * c[k] where c = conv(a,a).
    Exploits symmetry: grad[j] = conv(w,a)[j] + conv(w,a)[2n-2-j].
    """
    n = len(a)
    w_len = len(weights)
    a_pad = np.zeros(N_grad)
    a_pad[:n] = a
    w_pad = np.zeros(N_grad)
    w_pad[:w_len] = weights
    A = fft(a_pad)
    W = fft(w_pad)
    conv = np.real(ifft(W * A))
    # grad[j] = conv[j] + conv[2n-2-j], j=0..n-1
    grad = conv[:n] + conv[n-1:2*n-1][::-1]
    return grad


def lse_grad(a, N, N_grad, beta):
    """LSE-smoothed peak of conv(a,a) and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        lse_val = c.max()
    else:
        shifted = beta * (c - c_max)
        # Numerically stable logsumexp
        shifted_max = shifted.max()
        exp_shifted = np.exp(shifted - shifted_max)
        total = exp_shifted.sum()
        if total < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            w = exp_shifted / total
            lse_val = c_max + (shifted_max + np.log(total)) / beta
    grad = compute_gradient_weighted(a, w, N_grad)
    return lse_val, c_max, grad, c, w


def activ

e_set_grad(a, N, N_grad, top_k, margin=0.0):
    """
    Subgradient at top-k lags (uniformly weighted).
    If margin > 0, include all lags within margin of the max.
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if margin > 0:
        active = c >= (c_max - margin)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        k_active = int(active.sum())
    else:
        k_active = min(top_k, 2 * n - 1)
        active = np.zeros(2 * n - 1, dtype=bool)
        top_idx = np.argpartition(c, -k_active)[-k_active:]
        active[top_idx] = True
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / active.sum()
    grad = compute_gradient_weighted(a, w, N_grad)
    return c_max, grad, c, w


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy_i = np.dot(y_i, y_i)
        if yy_i < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy_i
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def normalize(a, n, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_of(c_max, n, S):
    """Compute the objective value from peak autocorrelation."""
    return 2.0 * n * c_max / (S * S)


def lse_lbfgs_phase(a_init, n, S, N, N_grad, max_time, rng,
                    beta_init=10.0, beta_max=500.0, m_lbfgs=12):
    """L-BFGS on LSE with adaptive beta and exact-objective line search."""
    a = normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    f_val, c_max, g, c, w = lse_grad(a, N, N_grad, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta
        if it % 80 == 0:
            beta = min(beta_max, beta * 1.12)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            f_val, c_max, g, c, w = lse_grad(a, N, N_grad, beta)

        # Trim history
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # Direction
        direction = lbfgs_dir(s_hist, y_hist, g)
        dn = np.linalg.norm(direction)
        if dn < 1e-18:
            break

        # Line search: accept if LSE decreases OR if actual c_max improves
        alpha = step_scale
        a_new = a
        cmax_new = c_max
        f_new = f_val
        g_new = g

        accepted = False
        for ls in range(14):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls < best_cmax:
                # Great improvement in actual objective
                a_new = candidate
                cmax_new = cm_ls
                f_new, _, g_candidate, _, _ = lse_grad(candidate, N, N_grad, beta)
                g_new = g_candidate
                accepted = True
                break
            f_ls, _, g_candidate, _, _ = lse_grad(candidate, N, N_grad, beta)
            if f_ls <= f_val + 1e-4 * alpha * np.dot(direction, g) or ls == 13:
                a_new = candidate
                cmax_new = cm_ls
                f_new = f_ls
                g_new = g_candidate
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            a_new = project_simplex(a - 0.1 * step_scale * g / max(dn, 1e-18), S)
            cmax_new = compute_conv(a_new, N).max()
            f_new, _, g_new, _, _ = lse_grad(a_new, N, N_grad, beta)

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        ss = np.dot(s_k, s_k)
        sy = np.dot(s_k, y_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)

        a = a_new
        g = g_new
        f_val = f_new
        c_max = cmax_new

        # Restart if stuck
        if no_improve > 500:
            # Gentle perturbation in Fourier domain
            A = fft(a, N)
            mag = np.abs(A)
            phase = np.angle(A) + rng.normal(0, 0.05, N)
            a = np.real(ifft(mag * np.exp(1j * phase)))[:n]
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a * (S / s)
            s_hist = []
            y_hist = []
            f_val, c_max, g, c, w = lse_grad(a, N, N_grad, beta)
            no_improve = 0

    return best_a, best_cmax


def subgrad_polish_phase(a_init, n, S, N, N_grad, max_time, rng,
                         top_k_init=10, top_k_max=100):
    """
    Direct subgradient polishing with momentum and adaptive top-K.
    Uses Polyak step size and periodic reshuffling.
    """
    a = normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()

    # Momentum state
    v = np.zeros(n)
    mu = 0.9  # momentum
    t = 1.0
    step_base = 1.0 / max(top_k_init, 1)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)
        margin = 0.01 * best_cmax if it > 500 else 0.0

        c_max, g, c, w = active_set_grad(a, N, N_grad, top_k, margin)

        if c_max < best_cmax - 1e-12:
            best_cmax = c_max
            best_a = a.copy()

        # Adaptive step size (decreasing)
        step = step_base / (1.0 + 0.001 * it)
        # Scale by the norm of the gradient for consistency
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-18:
            g = g / gnorm
            step *= max(1.0, S / n)

        # Momentum update
        v = mu * v - step * g
        a_new = project_simplex(a + v, S)

        # Check improvement
        c_check = compute_conv(a_new, N)
        cm_check = c_check.max()
        if cm_check > c_max * 1.05:
            # Overshot: reduce step
            a_new = project_simplex(a - 0.5 * step * g, S)
            v *= 0.5

        a = a_new

        # Periodic restart with larger perturbation
        if it % 2000 == 0 and rng.random() < 0.3:
            A = fft(a, N)
            mag = np.abs(A)
            phase = np.angle(A) + rng.normal(0, 0.1, N)
            a = np.real(ifft(mag * np.exp(1j * phase)))[:n]
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a * (S / s)
            v = np.zeros(n)

    return best_a, best_cmax


def structured_restart(a, n, S, N, rng):
    """Geometric restart: modify the autocorrelation profile."""
    c = compute_conv(a, N)
    c_max = c.max()
    # Find the "flatness" - how many lags are near the max
    n_active = int((c > 0.95 * c_max).sum())
    if n_active < 5:
        # Very peaked: try to spread energy
        # Reduce the peak lag's contribution by modifying a
        k_star = int(np.argmax(c))
        # Compute which a[i] contribute most to c[k_star]
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        # Transfer from high contributors to low ones
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        transfer = 0.05 * a[i_max]
        a[i_max] -= transfer
        a[i_min] += transfer
    else:
        # Already somewhat flat: try a spectral perturbation
        A = fft(a, N)
        mag = np.abs(A)
        # Perturb middle frequencies more (where sidelobes are controlled)
        perturbation = np.abs(rng.normal(0, 0.02, N))
        mid_start = N // 4
        mid_end = 3 * N // 4
        perturbation[:mid_start] *= 0.3
        perturbation[mid_end:] *= 0.3
        A = A * (1 + perturbation * np.exp(1j * rng.uniform(0, 2 * np.pi, N)))
        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def make_cosine_init(n, rng):
    """Cosine bump initialization."""
    k = np.arange(n)
    center = n / 2.0
    a = 0.3 + 0.7 * np.cos(np.pi * (k - center) / n) ** 2
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_random_smooth_init(n, rng):
    """Random smooth initialization via low-pass filtering."""
    a = rng.random(n)
    # Apply a few passes of averaging
    for _ in range(3):
        a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    return np.clip(a, 0.01, None)
```
