Minimize 2*n*max_conv/sum^2 using hybrid FISTA + multiplicative weights +

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    Full self-convolution via FFT.
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    Project v onto {x >= 0, sum(x) = S}.
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    Compute softmax weights for LSE gradient.
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    Compute the actual objective value.
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_pressure(a, c, active_frac=0.05):
    Compute pressure at each position from active lags.
    pressure[i] = sum_{k active} a[k-i] (normalized).
    n = len(a)
    max_val = c.max()
    threshold = max_val * (1.0 - active_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active_indicator = np.zeros(len(c))
    active_indicator[active] = 1.0 / len(active)
    f_padded = np.zeros(len(c))
    f_padded[:n] = a
    L = 1 << (len(c) - 1).bit_length()
    AI = fft(active_indicator, L)
    FP = fft(f_padded, L)
    pressure = np.real(ifft(np.conj(AI) * FP))[:n]
    return np.maximum(pressure, 0.0)


def multiplicative_weights_step(a, c, eta, active_frac=0.05):
    Perform a multiplicative weights update based on pressure.
    n = len(a)
    pressure = compute_pressure(a, c, active_frac)
    mean_p = pressure.mean() + 1e-15
    scale = 1.0 / (1.0 + eta * (pressure - mean_p) / mean_p)
    scale = np.maximum(scale, 0.01)
    new_a = a * scale
    new_a_sum = new_a.sum()
    if new_a_sum > 1e-15:
        new_a = new_a * (a.sum() / new_a_sum)
    return new_a


def lp_inspired_step(a, c, alpha, active_frac=0.03):
    LP-inspired update: redistribute mass from high-pressure to low-pressure positions.
    n = len(a)
    pressure = compute_pressure(a, c, active_frac)
    # Target: uniform pressure -> ideal allocation
    # g[i] ∝ 1/(1 + pressure[i]/mean_p)
    mean_p = pressure.mean() + 1e-15
    target = 1.0 / (1.0 + pressure / mean_p)
    target_sum = target.sum()
    if target_sum > 1e-15:
        target = target * (a.sum() / target_sum)
    new_a = (1.0 - alpha) * a + alpha * target
    new_a = np.maximum(new_a, 0.0)
    new_a = new_a * (a.sum() / new_a.sum())
    return new_a

            if time.monotonic() - start > max_time:
                break
            i_start = b * block_size
            i_end = min(i_start + block_size, n)
            if i_start >= n:
                break

            # Try several perturbations of this block
            best_block_val = a[i_start:i_end].copy()
            best_block_obj = compute_obj(a)
            base_sum = a[i_start:i_end].sum()

            for trial in range(3):
                if trial == 0:
                    # Gradient-based: reduce values that contribute to peak
                    c = fft_convolve_self(a)
                    # For each element in block, compute its contribution to max lag
                    max_lag = int(np.argmax(c))
                    contrib = np.zeros(block_size)
                    for j in range(i_start, i_end):
                        offset = max_lag - j
                        if 0 <= offset < n:
                            contrib[j - i_start] = a[offset]
                        else:
                            contrib[j - i_start] = 0.0
                    # Reduce high-contribution elements

                    new_block = best_block_val + 0.01 * S / np.sqrt(n) * pert
                    new_block = np.maximum(new_block, 0.0)

                    new_block = np.maximum(new_block, 0.0)

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.01)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    S = np.sqrt(2.0 * n0)
    global_best_a = global_best_a * (S / global_best_a.sum())

    # === Phase 1: Main FISTA (30% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.30
    print("\n=== Phase 1: FISTA v2 (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_v2(global_best_a, p1_budget * 0.5, rng,
                         beta_start=10.0, beta_growth=0.003,
                         mw_interval=1000, lp_interval=2000, noise_interval=4000)
    obj1 = compute_obj(a1)
    print("  FISTA-1: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 20:
        a1b, _ = run_fista_v2(global_best_a, rem1, rng,
                              beta_start=30.0, beta_growth=0.002,
                              mw_interval=1500, lp_interval=3000, noise_interval=5000)
        obj1b = compute_obj(a1b)
        print("  FISTA-2: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: High-beta sweep (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: High-beta sweep (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a2, obj2 = run_high_beta_sweep(global_best_a, p2_budget, rng)
        print("  Phase 2 done: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Multi-start with perturbations (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Multi-start perturbations (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        inits = generate_perturbed_inits(global_best_a, n0, rng, S, n_inits=5)
        inits.append(global_best_a.copy())

        time_per = p3_budget / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time_per < 10:
                break
            a_m, _ = run_fista_v2(init, min(time_per * 0.9, 60.0), rng,
                                  beta_start=20.0, beta_growth=0.004,
                                  mw_interval=800, lp_interval=1500, noise_interval=3000)
            obj_m = compute_obj(a_m)
            tag = "IMPROVED" if obj_m < global_best_obj else ""
            print("  Multi-start %d: obj=%.8f %s" % (idx, obj_m, tag))
            if obj_m < global_best_obj:
                global_best_obj = obj_m
                global_best_a = a_m.copy()

    # === Phase 4: Coordinate descent polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Coordinate descent polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, obj4 = run_coord_descent(global_best_a, p4_budget, rng, block_size=128)
        print("  CoordDesc: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final FISTA squeeze (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Final FISTA squeeze (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()
        a5, _ = run_fista_v2(global_best_a, p5_budget * 0.6, rng,
                             beta_start=200.0, beta_growth=0.001,
                             mw_interval=3000, lp_interval=5000, noise_interval=8000)
        obj5 = compute_obj(a5)
        print("  Final FISTA: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 10:
            a5b, _ = run_fista_v2(global_best_a, rem5, rng,
                                  beta_start=500.0, beta_growth=0.0005,
                                  mw_interval=5000, lp_interval=8000, noise_interval=10000)
            obj5b = compute_obj(a5b)
            print("  Final FISTA 2: obj=%.8f (t=%.0fs)" % (obj5b, time.monotonic() - t5))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: L-BFGS final polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p6_budget = remaining * 0.05
        print("\n=== Phase 6: L-BFGS polish (budget %.0fs) ===" % p6_budget)
        t6 = time.monotonic()
        n = len(global_best_a)
        S6 = np.sqrt(2.0 * n)
        a6 = global_best_a * (S6 / global_best_a.sum())
        beta = 100.0

        def lse_obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            grad = conv_gradient(x, w)
            return obj, grad

        result = scipy_minimize(
            lse_obj_and_grad, a6.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 50, 'maxls': 20, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        )
        a6_new = np.clip(result.x, 0, 1000)
        a6_new = a6_new * (S6 / max(a6_new.sum(), 1e-15))
        obj6 = compute_obj(a6_new)
        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6_new.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected input, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
