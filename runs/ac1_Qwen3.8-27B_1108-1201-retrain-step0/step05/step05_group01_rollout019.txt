"""
Multi-phase optimizer for minimizing 2*n*max_conv/sum^2 over non-negative sequences.
Uses L-BFGS-B with LSE smoothing (ramping beta), active-set subgradient refinement,
pressure-based redistribution perturbations, and final high-beta polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient for near-active lags."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def run_lbfgs_lse(a_init, beta, time_budget, S):
    """Run L-BFGS-B on LSE-smoothed objective with sum constraint via penalty."""
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    pen = 50.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        b = fft_convolve_self(x)
        cm = b.max()
        shifted = beta * (b - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()
        obj = cm + np.log(e.sum()) / beta
        sum_x = x.sum()
        obj += pen * (sum_x - S) ** 2
        grad = conv_gradient(x, w)
        grad += 2.0 * pen * (sum_x - S)
        return obj, grad

    result = scipy_minimize(
        obj_and_grad, a.copy(), method='L-BFGS-B',
        jac=True, bounds=[(0, 1000)] * n,
        options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                 'iprint': -1}
    )

    a_new = np.clip(result.x, 0, 1000)
    a_new = a_new * (S / max(a_new.sum(), 1e-15))
    elapsed = time.monotonic() - start
    return a_new, elapsed, result.nit


def run_fista_subgrad(a_init, max_time, rng, mode="adaptive"):
    """FISTA with adaptive top-k subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k based on stuckness
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 500:
            k = 4
        elif stuck < 1500:
            k = min(16, 8 + (stuck - 500) // 100)
        else:
            k = min(64, 16 + (stuck - 1500) // 200)

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size with backtracking
        step = S / (gnorm * np.sqrt(1.0 + 0.01 * it))
        step = min(step, 2.0 * S / (gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

        # Stuck: apply small noise
        if stuck > 5000 and it % 200 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 15000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def pressure_redistribute(a, rng, n_rounds=20):
    """
    Pressure-based redistribution: move mass from high-pressure to low-pressure
    positions relative to the peak lag.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_conv = compute_obj(a)

    for rnd in range(n_rounds):
        c = fft_convolve_self(a)
        k_star = int(np.argmax(c))
        max_val = c[k_star]

        # Compute pressure: contribution of each position to the peak
        # b[k*] = sum_i a[i] * a[k*-i]
        # Pressure at i is a[k*-i] (the paired value)
        paired = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                paired[i] = a[j]

        # We want to reduce a[i] where paired[i] is large
        # Direction: d[i] = -paired[i] (reduce where paired is large)
        pressure = paired.copy()

        # Find high and low pressure positions
        thresh_high = np.percentile(pressure, 75)
        thresh_low = np.percentile(pressure, 25)

        high_mask = (pressure > thresh_high) & (a > 1e-10)
        low_mask = (pressure < thresh_low)

        if not np.any(high_mask) or not np.any(low_mask):
            break

        # Compute how much to move
        total_high = a[high_mask].sum()
        total_low_slots = len(low_mask)
        # Move a fraction of high mass to low positions
        frac = 0.1
        move_amount = frac * total_high / max(total_low_slots, 1)

        # Spread the move
        n_high = np.sum(high_mask)
        per_high = move_amount / max(n_high, 1)

        a_new = a.copy()
        high_idx = np.where(high_mask)[0]
        low_idx = np.where(low_mask)[0]

        # Random pairing for diversity
        rng.shuffle(high_idx)
        rng.shuffle(low_idx)
        n_moves = min(len(high_idx), len(low_idx))

        for i in range(n_moves):
            take = min(per_high, a_new[high_idx[i]] * 0.5)
            a_new[high_idx[i]] -= take
            a_new[low_idx[i]] += take

        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))

        new_conv = compute_obj(a_new)
        if new_conv < best_conv:
            best_conv = new_conv
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step
            frac = frac * 0.5
            if frac < 0.005:
                break

    return best_a, best_conv

    n = len(a_init)

    best_a = a.copy()

    n_betas = min(len(beta_schedule), max(2, int(time_budget / 20)))

    for idx in range(n_betas):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        beta = beta_schedule[min(idx, len(beta_schedule) - 1)]
        if idx >= len(beta_schedule):
            beta = beta_schedule[-1] * (idx - len(beta_schedule) + 2)

        a_new, t_elapsed, niters = run_lbfgs_lse(a, beta, remaining * 0.9, S)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        print("    L-BFGS beta=%d: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, obj_new, t_elapsed, niters))

    return best_a, best_obj


def try_different_length(a_ref, rng, budget):
    """Try optimizing with a different sequence length."""
    best_obj = np.inf
    best_a = None

    for n_try in [4096, 16384]:
        if budget < 60:
            break
        S = np.sqrt(2.0 * n_try)
        # Initialize by interpolating from reference
        t = np.linspace(0, 1, n_try, endpoint=False)
        a_try = np.interp(t, np.linspace(0, 1, len(a_ref), endpoint=False), a_ref)
        a_try = np.maximum(a_try, 0.01)
        a_try = a_try * (S / a_try.sum())

        obj_init = compute_obj(a_try)
        print("    n=%d init obj=%.6f" % (n_try, obj_init))

        # Quick FISTA
        t_budget = min(budget * 0.4, 120)
        a_opt, _ = run_fista_subgrad(a_try, t_budget, rng)
        obj_opt = compute_obj(a_opt)
        print("    n=%d FISTA obj=%.8f" % (n_try, obj_opt))

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            budget -= t_budget

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Phases: L-BFGS-B LSE, FISTA subgradient, pressure redistribution,
            multi-beta polishing, and optional length exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.6f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: L-BFGS-B with ramping beta (35% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.35
    print("\n=== Phase 1: L-BFGS-B LSE ramping (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    a1, obj1 = run_multibeta_lbfgs(global_best_a, p1_budget, rng)
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()
        print("  Phase 1 improved: %.8f" % global_best_obj)
    else:
        print("  Phase 1: no improvement, best=%.8f" % global_best_obj)

    # === Phase 2: FISTA with adaptive subgradient (25% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25
    if p2_budget > 20:
        print("\n=== Phase 2: FISTA adaptive subgradient (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a2, _ = run_fista_subgrad(global_best_a, p2_budget, rng)
        obj2 = compute_obj(a2)
        print("  FISTA: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Pressure redistribution + re-optimization (20% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20
    if p3_budget > 20:
        print("\n=== Phase 3: Pressure redistribution (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        # Multiple rounds of redistribution + short re-optimization
        n_redist_rounds = 3
        for rnd in range(n_redist_rounds):
            if time.monotonic() > deadline:
                break
            sub_budget = p3_budget / (n_redist_rounds - rnd)
            if sub_budget < 15:
                break

            # Redistribution
            a3, obj3 = pressure_redistribute(global_best_a, rng,
                                             n_rounds=max(5, int(sub_budget / 2)))
            if obj3 < global_best_obj:
                global_best_obj = obj3
                global_best_a = a3.copy()
                print("  Redistribution round %d: obj=%.8f (improved!)" %
                      (rnd, global_best_obj))
            else:
                print("  Redistribution round %d: obj=%.8f" % (rnd, obj3))

            # Short re-optimization after redistribution
            rem3 = deadline - time.monotonic()
            if rem3 > 30:
                a3b, obj3b = run_multibeta_lbfgs(global_best_a, min(rem3 * 0.4, 60), rng)
                if obj3b < global_best_obj:
                    global_best_obj = obj3b
                    global_best_a = a3b.copy()
                    print("  Re-opt after redist %d: obj=%.8f" % (rnd, global_best_obj))

    # === Phase 4: Multi-restart with perturbations (10% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10
    if p4_budget > 20:
        print("\n=== Phase 4: Multi-restart perturbations (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        n_restarts = 3
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)
        time_per_restart = p4_budget / n_restarts

        for rst in range(n_restarts):
            if time.monotonic() > deadline:
                break
            if time_per_restart < 10:
                break

            # Different perturbation strategies
            if rst == 0:
                # Gaussian noise
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a_pert = project_simplex(global_best_a + noise, S)
            elif rst == 1:
                # Frequency-domain smooth perturbation
                noise_time = rng.normal(0, 1.0, n)
                F = fft(noise_time)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.01
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15)
                a_pert = project_simplex(global_best_a + 0.02 * S * noise_smooth, S)
            else:
                # Targeted: increase where paired value is small
                c = fft_convolve_self(global_best_a)
                k_star = int(np.argmax(c))
                paired = np.zeros(n)
                for i in range(n):
                    j = k_star - i
                    if 0 <= j < n:
                        paired[i] = global_best_a[j]
                # Boost positions with low paired value
                boost = (1.0 / (paired + 0.01)) - 1.0
                boost = np.maximum(boost, 0)
                boost = boost * 0.01 * S / (boost.sum() + 1e-15)
                a_pert = project_simplex(global_best_a + boost, S)

            # Optimize from perturbed point
            a_r, obj_r = run_multibeta_lbfgs(a_pert, time_per_restart * 0.7, rng)
            print("  Restart %d: obj=%.8f" % (rst, obj_r))
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_r.copy()
                print("  Restart %d IMPROVED: %.8f" % (rst, obj_r))

    # === Phase 5: Final high-beta L-BFGS polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 5: Final high-beta polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)

        # Very high beta L-BFGS
        a5, _ = run_lbfgs_lse(global_best_a, 5000, remaining * 0.5, S)
        obj5 = compute_obj(a5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        # Final FISTA with exact subgradient
        rem5 = deadline - time.monotonic()
        if rem5 > 20:
            a5b, _ = run_fista_subgrad(global_best_a, rem5 * 0.8, rng)
            obj5b = compute_obj(a5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

        print("  Final polish: obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t5))

    # === Phase 6: Try different length if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 6: Try different lengths (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = try_different_length(global_best_a, rng, remaining * 0.6)
        if a6 is not None and obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()
            print("  Different length IMPROVED: obj=%.8f (n=%d)" % (obj6, len(a6)))
        else:
            print("  Different length: no improvement")

    # Final verification
    print("\n=== Final Verification ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
