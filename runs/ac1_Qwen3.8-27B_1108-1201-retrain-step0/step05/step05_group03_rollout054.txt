    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum_k exp(beta*c[k])).
    Uses FFT for efficiency.
    
    grad[j] = 2 * sum_k w[k] * a[k-j] (for valid indices)
    where w = softmax(beta * c)
    """
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot  # softmax weights, length 2n-1
    
    # grad[j] = 2 * sum_{k: 0<=k-j<2n-1} w[k] * a[j - (k-j)]
    # = 2 * sum_k w[k] * a[k-j] (for 0 <= k-j < n)
    # This is a cross-correlation: grad = 2 * (w ⊛ a)
    # where (w ⊛ a)[j] = sum_k w[k] * a[k-j]
    
    # Compute via FFT:
    # (w ⊛ a)[j] = sum_k w[k] * a[k-j]
    # Let's define b = flip(w) so that (w ⊛ a)[j] = sum_k b[2n-2-k] * a[k-j]
    # Hmm, let me think more carefully.
    
    # (w ⊛ a)[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] (where a[k-j] = 0 if not 0 <= k-j < n)
    # = sum_{l=0}^{n-1} w[j+l] * a[l] (where l = k-j, so k = j+l)
    
    # This is the cross-correlation of w (padded to length >= 3n-2) with a.
    # Alternatively: let's define v = w (length 2n-1, zero-padded to 3n-2).
    # Then (w ⊛ a)[j] = sum_l v[j+l] * a[l] = (v * flip(a))[j + n - 1] ... 
    
    # Actually, the simplest way: use the fact that
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[j+l]
    # This is a "sliding dot product" of a with w.
    # We can compute it via FFT:
    # Let A = fft(a, N) and W = fft(w, N) with N >= 3n-2.
    # Then the cross-correlation is ifft(conj(A) * W) ... 
    # Hmm, let me just use a direct approach or a careful FFT.
    
    # Actually, the cleanest way:
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[j+l]
    # Define u = w (length 2n-1), pad to length N >= 3n-2.
    # Define v = flip(a) (length n), pad to length N.
    # Then (u * v)[j + n - 1] = sum_k u[k] * v[j+n-1-k] = sum_k u[k] * a[n-1-(j+n-1-k)] = sum_k u[k] * a[k-j]
    # = sum_k w[k] * a[k-j] = grad[j] / 2
    
    # So grad = 2 * (convolve(w, flip(a)))[n-1 : n-1+n]
    
    fsize = 1 << (3 * n - 2 - 1).bit_length()  # enough for conv of length (2n-1) + n - 1 = 3n-2
    W = fft(w, fsize)
    A_flip = fft(a[::-1], fsize)
    corr = np.real(ifft(W * A_flip))
    grad = 2.0 * corr[n-1 : 2*n-1]
    
    return grad


def lse_gradient_topk(a, c, k, beta):
    """Top-k LSE gradient (faster when k << 2n-1)."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w_full, fsize)
    A_flip = fft(a[::-1], fsize)
    corr = np.real(ifft(W * A_flip))
    grad = 2.0 * corr[n-1 : 2*n-1]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def run_lse_descent(a_init, S, max_time, rng,
                    beta_init=10.0, beta_final=5000.0,
                    use_topk=False, k_frac=0.1):
    """LSE-smoothed gradient descent with adaptive beta and backtracking."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    lr = 1.0 / max(beta_init, 1.0)  # initial learning rate
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adaptive beta: increase over time
        beta = beta_init * (beta_final / beta_init) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute gradient
        if use_topk:
            k = max(4, int(k_frac * (2 * n - 1)))
            grad = lse_gradient_topk(a, c, k, beta)
        else:
            grad = lse_gradient(a, c, min(beta, 200.0))  # cap beta for numerical stability in gradient
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = lr
        g_normalized = grad / gnorm
        success = False
        for _ in range(10):
            a_new = project_simplex(a - step * g_normalized, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max + 1e-12:
                a = a_new
                success = True
                break
            step *= 0.5
        
        if not success:
            # Try a smaller base lr
            lr *= 0.5
            continue
        
        it += 1
        
        # Periodic restart
        if stuck > 500 and progress < 0.8:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_polish(a_init, S, max_time, rng):
    """High-beta polishing with careful line search, targeting the true max."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
        
        # Find the peak(s)
        peak_threshold = cur_max * 0.999
        peak_idx = np.where(c >= peak_threshold)[0]
        
        # Average gradient over peaks
        # grad[j] = 2 * mean over peaks of a[k-j]
        # = 2 * (w_avg * a)[j] where w_avg is uniform over peaks
        w = np.zeros(len(c))
        w[peak_idx] = 1.0 / len(peak_idx)
        
        n = len(a)
        fsize = 1 << (3 * n - 2 - 1).bit_length()
        W = fft(w, fsize)
        A_flip = fft(a[::-1], fsize)
        corr = np.real(ifft(W * A_flip))
        grad = 2.0 * corr[n-1 : 2*n-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        g_norm = grad / gnorm
        
        # Line search
        base_step = 0.1 / (1.0 + 0.01 * it)
        step = base_step * step_scale
        success = False
        for _ in range(8):
            a_new = project_simplex(a - step * g_norm, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
                step_scale = min(3.0, step_scale * 1.05)
                success = True
                break
            step *= 0.5
        
        if not success:
            step_scale = max(0.01, step_scale * 0.5)
        
        # Occasional perturbation
        if it % 500 == 0 and (time.monotonic() - start) / max_time < 0.6:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_spectral(a_init, S, max_time, rng):
    """Spectral domain optimization: flatten the power spectrum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        
        if mean_p < 1e-30:
            break
        
        # Flattening: reduce dynamic range of power spectrum
        # new_power = power^(alpha) * mean_p^(1-alpha) (geometric interpolation)
        alpha = 0.3 / (1.0 + 0.01 * it)  # decrease aggressiveness over time
        new_power = power ** alpha * mean_p ** (1 - alpha)
        # Ensure non-negative
        new_power = np.maximum(new_power, 1e-30)
        
        # Update magnitude, keep phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        A_new = phase * np.sqrt(new_power)
        
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        
        # Check if this improves
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best:
            a = normalize(a_new, S)
            best = c_new.max()
            best_a = a.copy()
        else:
            # Try smaller alpha
            alpha2 = alpha * 0.3
            new_power2 = power ** alpha2 * mean_p ** (1 - alpha2)
            new_power2 = np.maximum(new_power2, 1e-30)
            A_new2 = phase * np.sqrt(new_power2)
            a_new2 = np.real(ifft(A_new2, fsize))[:n]
            a_new2 = np.clip(a_new2, 0, None)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < best:
                a = normalize(a_new2, S)
                best = c_new2.max()
                best_a = a.copy()
    
    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)
