def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def deflation_step(a, S, N, top_k=1, eta=None, c=None):
    """
    One deflation step: reduce elements that contribute most to the top-k peaks.
    Returns (new_a, c, c_max, grad_used).

    if c is None:
        c = compute_conv(a, N)
    c_max = c.max()
    
    # Get top-k peak positions
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:top_k]
    
    # Weight peaks by their excess over the mean
    c_mean = c.mean()
    excess = np.maximum(c[top_idx] - c_mean, 0)
    total_excess = excess.sum()
    if total_excess < 1e-30:
        return a.copy(), c, c_max, np.zeros(n)
    weights = excess / total_excess
    
    # Compute gradient
    grad = topk_grad(a, N, c * 0 + np.array([0.0]) * 0)  # placeholder, will use direct
    # Actually let's just compute directly
    full_w = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        full_w[int(k)] = weights[i]
    grad = topk_grad(a, N, full_w)
    
    # Determine step size
    if eta is None:
        # Line search: try decreasing eta
        g_max = grad.max()
        if g_max < 1e-30:
            return a.copy(), c, c_max, grad
        eta = min(0.1, S * 0.01 / g_max)
    
    # Try several step sizes
    best_a = a.copy()
    best_cmax = c_max
    for ls in range(8):

        c_try = compute_conv(a_try, N)

            best_cmax = cm_try
            best_a = a_try.copy()
        if cm_try <= c_max:
            break
        eta *= 0.6
    return best_a, c, c_max, grad


def chirp_spectral_restart(a, N, rng, frac=0.3, chirp_order=1):
    """
    Structured spectral restart: perturb phases with a chirp-like pattern.
    This is more structured than pure random and can find better escape directions.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    
    # Create a chirp-like phase perturbation
    t = np.arange(N) / N
    if chirp_order == 1:
        perturbation = 2 * np.pi * t  # linear chirp
    elif chirp_order == 2:
        perturbation = 2 * np.pi * t * t  # quadratic chirp
    else:
        perturbation = 2 * np.pi * t ** chirp_order
    
    n_rand = max(1, int(N * frac))
    rand_idx = np.random.default_rng(rng.integers(0, 2**31)).choice(N, size=n_rand, replace=False)
    
    new_phase = old_phase.copy()
    # Add scaled chirp perturbation to selected frequencies
    scale = rng.uniform(0.3, 1.0)
    new_phase[rand_idx] = old_phase[rand_idx] + scale * perturbation[rand_idx] + rng.uniform(-0.5, 0.5, n_rand)
    
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

def lse_gradient(a, N, beta):
    """Compute LSE-smoothed gradient and value."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        weights = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max

        # Reduce elements that contribute to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        # For c[k_star] = sum_i a[i]*a[k_star-i], element a[j] contributes a[k_star-j]
        for j in range(lo, hi + 1):
            partner = k_star - j
            if 0 <= partner < n:
                contrib[j] += a[partner]
        if contrib.max() > contrib.min() + 1e-15:
            i_max = int(np.argmax(contrib))
            i_min = int(np.argmin(contrib))
            transfer = 0.05 * a[i_max]
            a[i_max] -= transfer
            a[i_min] += transfer
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
    return a

def lse_lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=300.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = a * (S / max(a.sum(), 1e-15))
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, f_val, c, c_max = lse_gradient(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 3.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 3.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_gradient(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        # L-BFGS direction
        direction = _lbfgs_direction(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, _, c_nw, cmax_nw = lse_gradient(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        if it % 300 == 0 and rng.random() < 0.2:
            a = random_spectral_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_gradient(a, N, beta)
    
    return best_a, best_cmax


def _lbfgs_direction(s_hist, y_hist, g):
    """Two-loop L-BFGS recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return np.zeros(n) if gn < 1e-18 else -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        yy = np.dot(y_hist[i], y_hist[i])
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_hist[i], q) / yy
            q = q - alphas[i] * y_hist[i]
    yy_last = np.dot(y_hist[-1], y_hist[-1])
    gamma = max(1e-8, np.dot(s_hist[-1], y_hist[-1]) / yy_last) if yy_last > 1e-30 else 1.0
    r = gamma * q
    for i in range(m):
        yy = np.dot(y_hist[i], y_hist[i])
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_hist[i], y_hist[i]) / yy
        r = r + (alphas[i] - beta_i) * y_hist[i]
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def nesterov_toppk(a_init, n, S, N, max_time, rng, top_k_init=10, top_k_max=300):
    """Nesterov-accelerated top-K subgradient method."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    stall_count = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax - 1e-12:
            best_cmax = cur
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 500 and rng.random() < 0.1:
            y = random_spectral_restart(y, N, rng, frac=0.3)
            t = 1.0
            stall_count = 0
        
        # Top-K weighted gradient
        sorted_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[sorted_idx]
        # Exponential weighting within top-K
        w = np.exp(0.5 * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        full_w = np.zeros(2 * n - 1)
        for i, k in enumerate(sorted_idx):
            full_w[int(k)] = w[i]
        
        grad = topk_grad(y, N, full_w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mu = (t - 1.0) / t_new
        y = x_new + mu * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best_cmax


def adaptive_deflation_loop(a_init, n, S, N, max_time, rng):
    """
    Main optimization loop: adaptive deflation with spectral restarts.
    This is the core new algorithm.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = a * (S / max(a.sum(), 1e-15))
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    stall = 0
    top_k = 1
    c = compute_conv(a, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = a.copy()
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive top-K: grow slowly
        top_k = min(200, 1 + it // 20)
        
        # Compute gradient
        sorted_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[sorted_idx]
        c_min = c.min()
        excess = np.maximum(top_vals - c_min, 1e-10)
        weights = excess / excess.sum()
        
        full_w = np.zeros(2 * n - 1)
        for i, k in enumerate(sorted_idx):
            full_w[int(k)] = weights[i]
        
        grad = topk_grad(a, N, full_w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size with line search
        base_step = S * 0.05 / (gnorm * np.sqrt(n))
        alpha = base_step
        a_new = a
        cmax_new = c.max()
        
        for ls in range(6):
            candidate = project_simplex(a - alpha * grad, S)
            c_try = compute_conv(candidate, N)
            cm_try = c_try.max()
            if cm_try < cmax_new:
                cmax_new = cm_try
                a_new = candidate
            if cm_try <= c.max() * (1 - 1e-8):
                break
            alpha *= 0.5
        
        # Update
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        c = compute_conv(a, N)
        
        # Stagnation handling
        if stall > 100:
            # Try chirp spectral restart
            if stall % 50 == 0:
                a = chirp_spectral_restart(a, N, rng, frac=0.3, chirp_order=rng.integers(1, 4))
                c = compute_conv(a, N)
                if c.max() < best_cmax:
                    best_cmax = c.max()
                    best_a = a.copy()
                    stall = 0
            if stall > 300:
                a = random_spectral_restart(a, N, rng, frac=0.5)
                c = compute_conv(a, N)
                stall = 0
        
        # Periodic LSE polishing step
        if it % 500 == 0 and time.monotonic() - start_time > 5:
            g_lse, _, c_lse, cm_lse = lse_gradient(a, N, beta=20.0)
            a_lse = project_simplex(a - 0.01 * S / n * g_lse / max(np.linalg.norm(g_lse), 1e-10), S)
            c_lse2 = compute_conv(a_lse, N)
            if c_lse2.max() < best_cmax:
                best_cmax = c_lse2.max()
                best_a = a_lse.copy()
                a = a_lse
                c = c_lse2
                stall = 0
    
    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive deflation as the primary optimizer with spectral hybrid restarts,
    L-BFGS polishing, and Nesterov acceleration.
    """
    global N_global
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)
    
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass
    
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    N_global = N
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = np.clip(a_candidate, 0, 1000).astype(np.float64)
        s_cand = a_cand.sum()
        if s_cand < 1e-10:
            return global_best
        a_cand = a_cand * (S / s_cand)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [{}] peak={:.6f} score={:.6f}" .format(label, sc, score))
        return sc
    
    # === Phase 1: Diverse initialization + quick deflation ===
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(360.0, (deadline - t1) * 0.15)
    
    inits = []
    inits.append(("uniform", make_init_uniform(n, S)))
    inits.append(("random", make_init_random(n, S, rng)))
    inits.append(("chirp", make_init_chirp(n, S, rng)))
    inits.append(("zadoff", make_init_zadoff(n, S, rng)))
    inits.append(("multifreq", make_init_multifreq(n, S, rng)))
    inits.append(("fourier", make_init_fourier_random(n, S, N, rng)))
    
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("seeded", make_init_seeded(n, S, N, rng, given)))
            # Also try the given sequence directly (normalized)
            g_norm = np.clip(given, 0, 1000)
            g_norm = g_norm * (S / g_norm.sum()) if g_norm.sum() > 0.01 else g_norm
            inits.append(("given_norm", g_norm))
    except Exception:
        pass
    
    inits.append(("anti_auto", make_init_anti_autocorr(n, S, rng)))
    
    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  {} inits, ~{:.1f}s each".format(n_inits, per_init))
    
    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        a_norm = np.clip(arr, 0, 1000).astype(np.float64)
        s_arr = a_norm.sum()
        if s_arr < 0.01:
            continue
        a_norm = a_norm * (S / s_arr)
        
        t0 = time.monotonic()
        rem = max(2.0, per_init - (time.monotonic() - t0) - 0.5)
        # Run quick deflation
        a_l, sc_l = adaptive_deflation_loop(a_norm, n, S, N, min(rem, 30.0), rng)
        update_best(a_l, name + "/defl")
    
    print("Phase 1 done. Best score: {:.6f}".format(2 * n * global_best / (S * S) if global_best < np.inf else np.inf))
    
    if global_a is None:
        global_a = make_init_uniform(n, S)
        global_best = 2.0
    
    # === Phase 2: Extended adaptive deflation on best ===
    print("=== Phase 2: Extended adaptive deflation ===")
    t2 = time.monotonic()
    phase2_budget = min(900.0, (deadline - t2) * 0.45)
    
    a_curr = global_a.copy()
    best_p2 = global_best
    stall_p2 = 0
    
    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 5:
            break
        a_new, sc_new = adaptive_deflation_loop(a_curr, n, S, N, block_time, rng)
        if sc_new < best_p2 - 1e-10:
            best_p2 = sc_new
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_p2 = 0
            score = 2 * n * sc_new / (S * S)
            print("  [P2] score={:.6f}".format(score))
        else:
            a_curr = a_new.copy()
            stall_p2 += 1
            if stall_p2 == 2:
                a_curr = chirp_spectral_restart(a_curr, N, rng, frac=0.4, chirp_order=rng.integers(1, 5))
                stall_p2 = 0
            elif stall_p2 >= 4:
                a_curr = random_spectral_restart(a_curr, N, rng, frac=0.6)
                stall_p2 = 0
    
    # === Phase 3: Nesterov polishing ===
    print("=== Phase 3: Nesterov polishing ===")
    t3 = time.monotonic()
    remaining3 = min(400.0, (deadline - t3) * 0.2)
    if remaining3 > 20:
        a_ng, sc_ng = nesterov_toppk(global_a, n, S, N, remaining3, rng,
                                       top_k_init=20, top_k_max=300)
        update_best(a_ng, "nesterov")
    
    # === Phase 4: L-BFGS on LSE (high beta) ===
    print("=== Phase 4: L-BFGS polishing ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 15
    if remaining4 > 20:
        a_fin, sc_fin = lse_lbfgs_block(global_a, n, S, N, min(remaining4, 300.0),
                                         rng, beta_init=30.0, beta_max=500.0, m_lbfgs=12)
        update_best(a_fin, "lbfgs")
    
    # === Phase 5: Final short deflation + Nesterov alternation ===
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 40:
        half = remaining5 / 2.0
        a_d, sc_d = adaptive_deflation_loop(global_a, n, S, N, min(half, 150.0), rng)
        update_best(a_d, "final_defl")
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 20:
            a_ng2, sc_ng2 = nesterov_toppk(global_a, n, S, N, min(rem2, 100.0), rng,
                                             top_k_init=30, top_k_max=400)
            update_best(a_ng2, "final_nest")
    
    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: {:.6f}".format(ver) if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length {}, internal peak={:.6f}, approx score={:.6f}".format(
        len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
