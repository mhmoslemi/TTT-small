import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    if v.sum() <= S:
        return v + (S - v.sum()) / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    smin, smax = scaled.min(), scaled.max()
    if smax - smin > 500:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c, n_active=1):
    """Exact subgradient at the max (average over top n_active peaks)."""
    n = len(a)
    idx = np.argsort(c)[::-1][:n_active]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / n_active
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, perturbation_period=4000):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (3.0 * beta)

        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (10.0 * perturbation_period))
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    return best_a, best


def run_subgradient_polish(a_init, S, max_time, rng):
    """
    Subgradient method with line search for final polish.
    Uses exact subgradient at the max with adaptive step sizes.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    step = S / n * 0.5
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Exact subgradient at top few peaks
        n_act = min(1 + stuck // 50, 5)
        grad = exact_subgradient(a, c, n_active=n_act)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: decrease when stuck
        decay = np.exp(-stuck / 200.0)
        step = (S / n) * 0.3 * decay
        if step < 1e-15:
            # Perturb to escape
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            step = S / n * 0.1
            stuck = 0
            continue

        a_new = project_simplex(a - step * grad, S)

        # Line search: try a few step sizes
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cm and stuck > 20:
            # Try smaller step
            a_new = project_simplex(a - step * 0.1 * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() > cm and stuck > 100:
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
                stuck = 0
                continue

        a = a_new

    return best_a, best


def run_coordinate_polish(a_init, S, max_time, rng):
    """
    Coordinate descent: iteratively update the coordinates involved
    in the worst peaks to reduce the maximum.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()

        # Get the worst peaks
        k_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for k in k_peaks:
            if time.monotonic() - start > max_time - 5:
                break
            # Find indices involved in peak k
            # (a*a)[k] = sum_i a[i]*a[k-i] for valid i
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            indices = np.arange(i_lo, i_hi + 1)
            partners = k - indices

            # Try to reduce this peak by moving mass from high-value
            # positions in this peak to positions not in any top peak
            if len(indices) < 2:
                continue

            peak_vals = c[k]
            # Identify the "heavy" contributors
            contrib = a[indices] * a[partners]
            if contrib.max() < 1e-10:
                continue

            heavy_idx = indices[contrib > 0.5 * contrib.max()]
            if len(heavy_idx) == 0:
                continue

            # Find a "light" position not involved in top peaks
            all_peak_indices = set()
            for kp in k_peaks:
                lo2 = max(0, kp - (n - 1))
                hi2 = min(n - 1, kp)
                all_peak_indices.update(range(lo2, hi2 + 1))
                all_peak_indices.update(set(kp - np.arange(lo2, hi2 + 1)))

            # Pick a coordinate to donate to
            pool = [i for i in range(n) if i not in all_peak_indices]
            if not pool:
                pool = list(range(n))
            donor = int(rng.integers(len(pool)))
            target = pool[donor]

            # Move a small amount
            amount = 0.01 * a[heavy_idx[0]]
            if amount < 1e-12:
                continue
            amount = min(amount, a[heavy_idx[0]] * 0.5)
            a[heavy_idx[0]] -= amount
            a[target] += amount

            improved = True

        if not improved:
            # Small perturbation
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best


def run_basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with Fourier mutations and diverse crossover."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    n_iters = max(1, int(max_time / (opt_time + 5)))

    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time - 10:
            break
        progress = it / max(1, n_iters)

        base_scale = 0.3 * np.exp(-1.5 * progress) + 0.01
        scale = base_scale * (1.0 + 0.5 * rng.random())

        r = rng.random()
        if len(population) >= 2 and r < 0.35:
            idx1 = int(rng.integers(len(population)))
            idx2 = int(rng.integers(len(population)))
            if idx1 != idx2:
                a = fourier_crossover(population[idx1][1], population[idx2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, scale)
        elif r < 0.7:
            a = fourier_phase_mutate(a, rng, scale)
        else:
            # Structural mutation: random block swap
            a_list = a.copy()
            blen = rng.integers(max(1, n // 64), max(2, n // 8))
            i1 = rng.integers(0, n - blen)
            i2 = rng.integers(0, n - blen)
            if i1 != i2:
                a_list[i1:i1+blen], a_list[i2:i2+blen] = a_list[i2:i2+blen].copy(), a_list[i1:i1+blen].copy()
            a = project_simplex(a_list, S)

        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 10)
        if t_alloc < 5:
            break

        a_opt, val = run_fista(a, S, t_alloc, rng,
                                beta_init=20 + 30 * it,
                                beta_rate=0.004,
                                beta_max=5000 + 2000 * min(it, 30),
                                perturbation_period=6000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("      BH iter %d: new best max_b=%.8f" % (it, val))

        population.append((val, a_opt.copy()))
        if len(population) > 10:
            population.sort(key=lambda x: x[0])
            population = population[:10]

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000).astype(np.float64)
        inits.append(g)

    # Windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.001))
    inits.append(np.maximum(0.55 + 0.45 * np.cos(2 * np.pi * tt), 0.001))
    inits.append(np.maximum(0.4 - 0.5 * np.cos(2 * np.pi * tt) + 0.1 * np.cos(4 * np.pi * tt), 0.001))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.001)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.001)
    inits.append(np.exp(-6.0 * (tt - 0.3) ** 2) + np.exp(-6.0 * (tt - 0.7) ** 2) + 0.001)

    # Asymmetric
    inits.append(np.maximum(np.cos(np.pi * (tt - 0.3)), 0.0) + 0.001)
    inits.append(np.maximum(np.cos(np.pi * (tt - 0.6)), 0.0) + 0.001)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(6 * np.pi * tt + 0.5) + 0.2 * np.cos(12 * np.pi * tt + 1.0), 0.001))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(8 * np.pi * tt) + 0.1 * np.cos(20 * np.pi * tt + 0.7), 0.001))

    # Random
    inits.append(rng.random(n) + 0.05)
    inits.append(np.abs(rng.normal(1.0, 0.5, n)) + 0.02)

    # Block structures
    for nb in [8, 16, 32, 64]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.1, 2.0, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.001)

    return inits


def objective_from_maxb(a, maxb):
    """Compute objective from the sequence and its max convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: try multiple n values, use diverse init + screening,
    basin hopping, subgradient polish, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Try to get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    def try_update(a, maxb):
        nonlocal global_best_obj, global_a, global_n
        obj = objective_from_maxb(a, maxb)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            global_n = len(a)
            print("    NEW GLOBAL BEST: obj=%.8f (n=%d, maxb=%.6f)" % (obj, global_n, maxb))

    # === Phase 1: Try the given sequence + quick improvement ===
    if given is not None and len(given) >= 256:
        print("\n=== Phase 1: Improve given sequence (n=%d) ===" % len(given))
        p1_start = time.monotonic()
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        a_g = normalize_to_sum(given, S_g)
        c_g = fft_convolve_self(a_g)
        obj_g = objective_from_maxb(a_g, c_g.max())
        print("  Given obj=%.8f" % obj_g)
        try_update(a_g, c_g.max())

        # Quick FISTA to improve
        t_alloc = min(120.0, (deadline - p1_start) * 0.08)
        if t_alloc > 20:
            a_p, mb = run_fista(a_g, S_g, t_alloc, rng, beta_init=100.0, beta_rate=0.002, beta_max=20000)
            try_update(a_p, mb)

        # Subgradient polish
        remaining = (deadline - total_start) * 0.1 - (time.monotonic() - p1_start)
        if remaining > 20:
            a_sp, mb_sp = run_subgradient_polish(global_a, S_g, min(remaining, 60), rng)
            try_update(a_sp, mb_sp)

    # === Phase 2: Try n=16384 from scratch ===
    n_main = 16384
    S_main = np.sqrt(2.0 * n_main)
    print("\n=== Phase 2: Main search at n=%d ===" % n_main)
    p2_start = time.monotonic()

    # Budget allocation for this phase
    p2_budget = min(1200.0, (deadline - p2_start) * 0.65)

    # Screening
    inits = generate_inits(n_main, rng)
    n_screen = min(10, len(inits))
    per_init_time = p2_budget * 0.15 / n_screen

    candidates = []
    for i in range(n_screen):
        remaining = deadline - time.monotonic() - 200
        t_alloc = min(per_init_time, remaining)
        if t_alloc < 10:
            break
        a_opt, val = run_fista(inits[i], S_main, t_alloc, rng,
                                beta_init=5.0, beta_rate=0.01, beta_max=3000)
        obj = objective_from_maxb(a_opt, val)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        try_update(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.8f" % global_best_obj)

    # Basin hopping from top candidates
    p2_remain = deadline - time.monotonic() - 300
    bh_budget = min(p2_remain * 0.6, 700.0)
    bh_starts = min(3, len(candidates))
    for i in range(bh_starts):
        remaining = deadline - time.monotonic() - 300
        t_alloc = bh_budget / (bh_starts - i)
        t_alloc = min(t_alloc, remaining)
        if t_alloc < 30:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = run_basin_hopping(candidates[i][1], S_main, t_alloc, rng)
        try_update(a_bh, val_bh)

    # === Phase 3: Try n=32768 if time allows ===
    remaining_for_p3 = deadline - time.monotonic() - 200
    if remaining_for_p3 > 300:
        n_big = 32768
        S_big = np.sqrt(2.0 * n_big)
        print("\n=== Phase 3: Try n=%d ===" % n_big)
        p3_budget = min(remaining_for_p3 * 0.5, 400.0)

        inits_big = generate_inits(n_big, rng)
        screen_n = min(5, len(inits_big))
        per_t = p3_budget * 0.3 / screen_n

        for i in range(screen_n):
            t_alloc = min(per_t, deadline - time.monotonic() - 200)
            if t_alloc < 10:
                break
            a_opt, val = run_fista(inits_big[i], S_big, t_alloc, rng,
                                     beta_init=5.0, beta_rate=0.01, beta_max=3000)
            obj = objective_from_maxb(a_opt, val)
            print("  Big screen %d: obj=%.6f" % (i, obj))
            try_update(a_opt, val)

        # BH on best
        if deadline - time.monotonic() > 100:
            t_alloc = min(200.0, deadline - time.monotonic() - 100)
            a_bh, val_bh = run_basin_hopping(global_a, S_big, t_alloc, rng)
            try_update(a_bh, val_bh)

    # === Phase 4: Final polish on global best ===
    print("\n=== Phase 4: Final polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 15

    if global_a is not None and p4_budget > 30:
        S_best = np.sqrt(2.0 * len(global_a))

        # Subgradient polish
        seg1 = min(p4_budget * 0.4, 300.0)
        print("  Subgradient polish (%.0fs)..." % seg1)
        a_sp, mb_sp = run_subgradient_polish(global_a, S_best, seg1, rng)
        try_update(a_sp, mb_sp)

        # High-beta FISTA
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            seg2 = min(remaining * 0.4, 200.0)
            print("  High-beta FISTA (%.0fs)..." % seg2)
            a_hb, mb_hb = run_fista(global_a, S_best, seg2, rng,
                                     beta_init=500.0, beta_rate=0.001, beta_max=100000.0)
            try_update(a_hb, mb_hb)

        # Coordinate polish
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            seg3 = min(remaining * 0.5, 100.0)
            print("  Coordinate polish (%.0fs)..." % seg3)
            a_cp, mb_cp = run_coordinate_polish(global_a, S_best, seg3, rng)
            try_update(a_cp, mb_cp)

        # Final subgradient
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            print("  Final subgradient (%.0fs)..." % remaining)
            a_fs, mb_fs = run_subgradient_polish(global_a, S_best, remaining, rng)
            try_update(a_fs, mb_fs)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
            final_list = [float(x) for x in global_a]
    else:
        global_n = 8192
        global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best_obj, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
