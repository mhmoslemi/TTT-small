"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Spectral Equalization with Non-negativity (SENA) as the primary
optimizer, combined with FISTA polishing and basin hopping.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def sena_step(a, S, mu, N):
    """
    One SENA step: blend power spectrum toward flat, project to non-negative.
    """
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_flat = (S ** 2) / N
    P_new = (1 - mu) * P + mu * P_flat
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    # Preserve DC component (sum)
    A_new[0] = S  # DC should be S (sum of a)
    a_new = np.real(ifft(A_new, N))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    return a_new

def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, perturbation_period=5000):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (3.0 * beta)

        if it - last_improve > 2500:
            if it % 100 == 0:
                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (10.0 * perturbation_period))
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_amplitude_mutate(a, rng, scale=0.1):
    """Fourier-domain amplitude mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    # Multiplicative amplitude noise
    amp_noise = rng.normal(1.0, scale, fsize)
    new_mag = mag * np.exp(np.log(np.maximum(amp_noise, 0.1)))
    new_A = new_mag * np.exp(1j * phase)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Spectral Equalization with Non-negativity (SENA) as a novel optimizer,
combined with FISTA (LSE-smoothed), basin hopping with spectral mutations,
and high-beta final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def sena_step(a, S, mu, N):
    """
    One SENA step: blend power spectrum toward flat, project to non-negative.
    This is analogous to Gerchberg-Saxton for phase retrieval.
    """
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_flat = (S ** 2) / N
    P_new = (1.0 - mu) * P + mu * P_flat
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, N))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def sena_step_deflation(a, S, alpha, N, top_k=8):
    """
    SENA with peak deflation: instead of flat target, reduce power
    at frequencies that contribute to the top-k peak lags.
    """
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    c = np.real(ifft(P, N))[:2 * n - 1]
    cmean = c.mean()
    idx = np.argsort(c)[::-1][:top_k]
    excess = np.maximum(c[idx] - cmean, 0.0)
    wsum = excess.sum()
    if wsum < 1e-30:
        return sena_step(a, S, 0.1, N)
    excess = excess / wsum
    # Deflation: reduce P where it contributes to peak lags
    # c[k] = (1/N) sum_f P(f) exp(2j*pi*f*k/N)
    # To reduce c[k*], reduce P(f) weighted by exp(-2j*pi*f*k*/N)
    # Real part: P_new(f) = P(f) - alpha * sum_k excess[k] * cos(2*pi*f*k/N) / N
    freqs = np.arange(N, dtype=np.float64)
    correction = np.zeros(N)
    for i, k in enumerate(idx):
        correction += excess[i] * np.cos(2.0 * np.pi * freqs * k / N)
    P_new = P - alpha * correction
    P_new = np.maximum(P_new, 0.0)
    # Also blend slightly toward flat
    P_flat = (S ** 2) / N
    P_new = 0.95 * P_new + 0.05 * P_flat
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, N))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def run_sena(a_init, S, max_time, mu_init=0.4, mu_decay=0.999,
             use_deflation=False, rng=None):
    """Run SENA algorithm with adaptive mixing parameter."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    mu = mu_init
    alpha = 0.3
    it = 0
    last_improve = 0
    while time.monotonic() - start < max_time:
        it += 1
        if use_deflation:
            a = sena_step_deflation(a, S, alpha, N)
            alpha *= 0.9995
        else:
            a = sena_step(a, S, mu, N)
            mu = max(0.01, mu * mu_decay)
        if it % 25 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
                last_improve = it
        if it % 500 == 0 and it - last_improve > 1000:
            if rng is not None:
                noise = 0.02 * S / np.sqrt(n) * rng.normal(0, 1, n)
                a = project_simplex(a + noise, S)
            last_improve = it
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_sena_adaptive(a_init, S, max_time, rng=None):
    """
    SENA with adaptive strategy: SENA phase, then FISTA polish, then more SENA.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    # Phase 1: SENA flat blending
    phase1_time = max_time * 0.5
    mu = 0.5
    while time.monotonic() - start < phase1_time:
        it += 1
        a = sena_step(a, S, mu, N)
        mu = max(0.02, mu * 0.997)
        if it % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    # Phase 2: Deflation SENA
    phase2_time = max_time * 0.3
    it2 = 0
    alpha = 0.4
    while time.monotonic() - start < phase2_time:
        it2 += 1
        a = sena_step_deflation(a, S, alpha, N)
        alpha = max(0.05, alpha * 0.998)
        if it2 % 50 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()
    # Phase 3: FISTA polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 10:
        a_pol, val_pol = run_fista(best_a, S, remaining, rng,
                                    beta_init=20.0, beta_rate=0.008, beta_max=8000.0)
        if val_pol < best:
            best = val_pol
            best_a = a_pol.copy()
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, perturbation_period=5000, momentum_type='fista'):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (3.0 * beta)
        if it - last_improve > 3000:
            if it % 100 == 0:
                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        x_new = project_simplex(y - step * grad, S)
        if momentum_type == 'linear':
            t_mom = min(t_mom + 1.0, 100.0)
            t_new = t_mom
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_new - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (10.0 * perturbation_period))
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 3.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_amplitude_mutate(a, rng, scale=0.1):
    """Fourier-domain amplitude mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    amp_log = rng.normal(0, scale, fsize)
    new_mag = mag * np.exp(amp_log)
    new_A = new_mag * np.exp(1j * phase)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1.0 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping with SENA+FISTA as local optimizer and spectral mutations.
    Maintains a population for diversity.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    population = [(best, a.copy())]
    opt_time = max(8.0, max_time / 30.0)
    it = 0
    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        scale = 0.35 * np.exp(-1.5 * progress) + 0.02
        r = rng.random()
        if len(population) >= 2 and r < 0.25:
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a = fourier_crossover(population[i1][1], population[i2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, scale)
        elif r < 0.5:
            a = fourier_phase_mutate(a, rng, scale)
        elif r < 0.8:
            a = fourier_amplitude_mutate(a, rng, scale)
        else:
            a = project_simplex(a + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
        # Local optimization: SENA then FISTA
        t1 = opt_time * 0.5
        t2 = opt_time * 0.5
        a_opt, val = run_sena(a, S, t1, mu_init=0.3, mu_decay=0.999, rng=rng)
        if t2 > 3:
            a_opt2, val2 = run_fista(a_opt, S, t2, rng,
                                       beta_init=15.0 + 10.0 * it,
                                       beta_rate=0.008, beta_max=5000.0)
            if val2 < val:
                a_opt, val = a_opt2, val2
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))
        population.append((val, a_opt.copy()))
        if len(population) > 10:
            population.sort(key=sort_candidates)
            population = population[:8]
    return best_a, best


def run_final_polish(a_init, S, max_time, rng):
    """
    Final polishing: alternating high-beta FISTA and tiny SENA steps.
    Uses linear momentum for stability at high beta.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        # Switch between FISTA and SENA every 2000 iterations
        if it % 2000 < 1000:
            # FISTA with very high beta
            beta = 200.0 + 80000.0 * progress ** 2
            c = fft_convolve_self(y)
            if c.max() < best:
                best = c.max()
                best_a = y.copy()
                last_improve = it
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (3.0 * beta)
            if it - last_improve > 8000:
                noise_scale = 0.001 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it
                continue
            x_new = project_simplex(y - step * grad, S)
            t_mom = min(t_mom + 1.0, 80.0)
            t_new = t_mom
            y = x_new + ((t_new - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
        else:
            # Tiny SENA step
            mu = 0.01 + 0.02 * (1.0 - progress)
            y = sena_step(y, S, mu, N)
            c = fft_convolve_self(y)
            if c.max() < best:
                best = c.max()
                best_a = y.copy()
                last_improve = it
            a = y
            t_mom = 1.0
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))
    inits.append(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2.0 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2.0 * np.pi * tt) +
                            0.08 * np.cos(4.0 * np.pi * tt), 0.01))
    # Flat-spectrum random phase inits
    N = 1 << (2 * n - 1).bit_length()
    for _ in range(4):
        phase = rng.uniform(0, 2.0 * np.pi, N)
        # Slightly non-flat: more energy at low frequencies
        freqs = np.arange(N, dtype=np.float64)
        mag = 1.0 + 0.3 * np.exp(-freqs / (0.1 * N))
        A = mag * np.exp(1j * phase)
        a = np.real(ifft(A, N))[:n]
        a = np.maximum(a, 0.01)
        inits.append(a)
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4.0 * np.pi * tt + 0.7) +
                            0.2 * np.cos(10.0 * np.pi * tt + 0.3), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6.0 * np.pi * tt) +
                            0.15 * np.cos(14.0 * np.pi * tt), 0.01))
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)
    # Block structures
    for nb in [4, 8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)
    return inits
