Combines FISTA with adaptive LSE smoothing, pair coordinate descent,
and L-BFGS-B polishing for multi-stage optimization.
"""

import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    n = len(a)
    c = a[j] + a[k]
    
    # Affected m indices
    m_j = np.arange(j, min(j + n, 2*n-1))
    m_k = np.arange(k, min(k + n, 2*n-1))
    m_affected = np.unique(np.concatenate([m_j, m_k]))
    m_affected = m_affected[(m_affected >= 0) & (m_affected <= 2*n-2)]
    
    # For each m, compute A, B, C
    A = np.zeros(len(m_affected))
    B = np.zeros(len(m_affected))
    C = np.zeros(len(m_affected))
    
    # base[m] = b[m] - old_from_j(m) - old_from_k(m)
    base = np.zeros(len(m_affected))
    
    for idx, m in enumerate(m_affected):
        # Old contribution of a[j]
        old_j = 0.0
        if 0 <= m - j < n:
            if m - j == j:  # m = 2j
                old_j = a[j] ** 2
            else:
                old_j = 2 * a[j] * a[m-j]
        
        # Old contribution of a[k]
        old_k = 0.0
        if 0 <= m - k < n:
            if m - k == k:  # m = 2k
                old_k = a[k] ** 2
            else:
                old_k = 2 * a[k] * a[m-k]
        
        base[idx] = b[m] - old_j - old_k
        
        # New contribution from a[j] = u
        if 0 <= m - j < n:
            if m - j == j:  # m = 2j
                A[idx] += 1
                B[idx] += 0
                C[idx] += 0
            elif m - j == k:  # m = j+k
                A[idx] += -2
                B[idx] += 2 * c
                C[idx] += 0
            else:
                A[idx] += 0
                B[idx] += 2 * a[m-j]
                C[idx] += 0
        
        # New contribution from a[k] = c-u
        if 0 <= m - k < n:
            if m - k == k:  # m = 2k
                A[idx] += 1
                B[idx] += -2 * c
                C[idx] += c ** 2
            elif m - k == j:  # m = j+k
                A[idx] += -2
                B[idx] += 2 * c
                C[idx] += 0
            else:
                A[idx] += 0
                B[idx] += -2 * a[m-k]
                C[idx] += 2 * c * a[m-k]
    
    return m_affected, A, B, C, base, c

        f2 = eval_f(m2)

        else:
