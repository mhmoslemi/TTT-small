"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_value_and_weights(c, beta):
    """Compute LSE value and softmax weights for convolution c."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    return lse_val, w


def gradient_from_weights(a, w):
    """Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    return gradient_from_weights(a, w_full)


def run_adaptive_fista(a_init, max_time, rng, beta_init=30.0, beta_rate=0.008,
                       beta_max=50000.0, step_scale=1.0, anneal_temp=0.002,
                       restart_interval=3000):
    """
    Adaptive FISTA with:
    - Backtracking line search
    - Momentum restarts on stall
    - Simulated annealing noise injection
    - Adaptive step size
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    temp = anneal_temp

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, w = lse_value_and_weights(c, beta)
        grad = gradient_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step_local = step
        cur_obj = 2.0 * n * c.max() / (S * S)
        x_trial = project_simplex(y - step_local * grad, S)
        c_trial = fft_convolve_self(x_trial)
        obj_trial = 2.0 * n * c_trial.max() / (S * S)
        bt = 0
        while obj_trial > cur_obj + 1e-13 and step_local > 1e-16 and bt < 8:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            bt += 1
        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.05, 3.0 / max(beta, 1.0))
        x_new = x_trial
        c_new = c_trial

        # Track best
        obj_new = 2.0 * n * c_new.max() / (S * S)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            last_improve = it
            temp = max(anneal_temp * 0.5, temp * 0.95)
        else:
            temp = min(anneal_temp * 3.0, temp * 1.02)

        # Nesterov momentum with restart
        stuck = it - last_improve
        if stuck > restart_interval:
            t_mom = 1.0
            noise_scale = temp * S / np.sqrt(n) * (1.0 + stuck / restart_interval)
            y = project_simplex(x_new + rng.normal(0, noise_scale, n), S)
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y * (S / ys)
            a = x_new
            t_mom = t_new
            c = c_new

        if stuck > restart_interval * 3:
            break

    return best_a, best_obj


def peak_shaving(a_init, max_time, rng, n_peaks=5):
    """
    Direct peak-shaving: identify top convolution peaks and transfer mass
    from high-contributing coordinates to low-contributing ones.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    transfers_per_iter = 4

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 15000:
            break

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute contribution of each coordinate to the top peaks
        contrib = np.zeros(n)
        for kk in top_idx:
            kk_int = int(kk)
            lo = max(0, kk_int - n + 1)
            hi = min(n - 1, kk_int)
            if hi >= lo:
                contrib[lo:hi + 1] += a[kk_int - hi:kk_int - lo + 1][::-1]

        # Score coordinates: high contrib + high value = good source
        # Low contrib + low value = good target
        src_score = contrib * (a + 1e-10)
        tgt_score = -contrib * (S / n - a + 1e-10)

        for _ in range(transfers_per_iter):
            # Pick source
            src_pool = np.argsort(src_score)[::-1][:min(20, n)]
            # Pick target
            tgt_pool = np.argsort(tgt_score)[:min(20, n)]

            best_improvement = 0.0
            best_move = None
            for j in src_pool[:8]:
                if a[j] < 1e-10:
                    continue
                for l in tgt_pool[:8]:
                    if j == l:
                        continue
                    # Try multiple delta values
                    for frac in [0.03, 0.07, 0.15]:
                        delta = a[j] * frac
                        if delta < 1e-12:
                            continue
                        # Quick estimate: effect on top peaks
                        improvement = 0.0
                        for kk in top_idx[:3]:
                            kk_int = int(kk)
                            # Contribution change at j
                            jj = max(0, min(n - 1, kk_int - j))
                            contrib_j = a[jj] if 0 <= jj < n else 0.0
                            # Contribution change at l
                            ll = max(0, min(n - 1, kk_int - l))
                            contrib_l = a[ll] if 0 <= ll < n else 0.0
                            # Net change in c[kk]
                            dc = delta * (contrib_l - contrib_j)
                            if c[kk] >= cur_max * 0.95 and dc < 0:
                                improvement += -dc
                            elif dc > 0 and c[kk] + dc > cur_max:
                                improvement -= (c[kk] + dc - cur_max)
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_move = (int(j), int(l), delta)

            if best_move is not None:
                j, l, delta = best_move
                a[j] -= delta
                a[l] += delta
                # Update c
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
                c[js:je + 1] += delta * delta  # second order (small)
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)]
                c[ls:le + 1] -= delta * delta  # second order

        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def polynomial_optimize(a_init, max_time, rng, degree=12):
    """
    Fit a polynomial to the sequence and optimize polynomial coefficients.
    This reduces dimensionality from n to degree+1.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    t = np.linspace(0, 1, n, endpoint=False)
    # Fit polynomial
    coeffs = np.polyfit(t, a, degree)
    poly = np.poly1d(coeffs)

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    lr = 0.5 / (degree + 1)

    while time.monotonic() - start < max_time:
        it += 1
        a_poly = np.maximum(poly(t), 0.0)
        s = a_poly.sum()
        if s < 1e-15:
            break
        a_norm = a_poly * (S / s)
        obj = compute_obj(a_norm)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_norm.copy()

        # Compute gradient w.r.t. coefficients
        c = fft_convolve_self(a_norm)
        cm = c.max()
        k_star = int(np.argmax(c))
        # Gradient of max_conv w.r.t. a
        grad_a = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad_a[lo:hi + 1] = 2.0 * a_norm[k_star - hi:k_star - lo + 1][::-1]

        # Backprop to coefficients
        # a_norm[j] = max(poly(t_j), 0) * S / s
        # For simplicity, ignore the max and normalization for gradient
        dt = np.polyder(poly, 1)
        dt_coeffs = dt.c
        # grad w.r.t. coeffs: sum_j grad_a[j] * t_j^k / (S/s)
        base = np.maximum(poly(t), 1e-10)
        weights = grad_a * (s / S) / base  # chain rule through normalization
        for k in range(degree + 1):
            grad_coeff = np.sum(weights * t ** k)
            coeffs[k - degree] -= lr * grad_coeff  # polyfit reverses order
        # Scale learning rate
        if it % 100 == 0:
            lr *= 0.99

        if it % 5000 == 0:
            break

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    t = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif method == "gaussian":
        a = np.exp(-4.0 * (t - 0.5) ** 2 / (0.1 ** 2))
    elif method == "raised":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    elif method = "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * t + 0.5)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 2 * t) + 0.2 * np.sin(2 * np.pi * 5 * t + 1.1) + 0.15 * np.sin(2 * np.pi * 9 * t + 2.3)
        a = np.abs(a)
    elif method == "random_smooth":
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(16) / 16
        a = np.convolve(a, kernel, mode='same')
        a = np.abs(a) + 0.05
    elif method == "broad_gauss":
        a = np.exp(-2.0 * (t - 0.5) ** 2 / (0.3 ** 2))
    elif method == "double_hump":
        a = np.exp(-50 * (t - 0.3) ** 2) + np.exp(-50 * (t - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

                t_alloc = t_left / 3.0
                if t_alloc < 5:
                    continue
                a_div = make_init(nv, method, rng)
                a_opt, obj_opt = run_adaptive_fista(a_div, t_alloc, rng,
                                                     beta_init=20, beta_rate=0.01,
                                                     beta_max=20000, step_scale=0.8)
                if obj_opt < global_best_obj:
                    global_best_obj = obj_opt
                    global_best_a = a_opt.copy()
                    print("  n=%d %s: obj=%.6f ***NEW BEST***" % (nv, method, obj_opt))
                else:
                    print("  n=%d %s: obj=%.6f" % (nv, method, obj_opt))

    # === Phase 2: Intensive adaptive FISTA (35% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.35
    if p2_time > 60:
        print("\n=== Phase 2: Intensive FISTA (budget %.0fs) ===" % p2_time)
        n_best = len(global_best_a)
        variants = [
            # (beta_init, beta_rate, beta_max, step_scale, anneal_temp)
            (50, 0.006, 50000, 1.0, 0.003),
            (200, 0.003, 100000, 0.5, 0.002),
            (500, 0.002, 200000, 0.3, 0.001),
            (2000, 0.001, 500000, 0.15, 0.0005),
        ]
        per_variant = p2_time / len(variants)
        for i, (bi, br, bm, ss, at) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            t_alloc = min(per_variant, deadline - time.monotonic())
            if t_alloc < 10:
                break
            a_opt, obj_opt = run_adaptive_fista(global_best_a, t_alloc, rng,
                                                 beta_init=bi, beta_rate=br,
                                                 beta_max=bm, step_scale=ss,
                                                 anneal_temp=at)
            print("  FISTA v%d (b=%d): obj=%.6f" % (i + 1, bi, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Peak shaving (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 30:
        print("\n=== Phase 3: Peak shaving (budget %.0fs) ===" % p3_time)
        a_ps, obj_ps = peak_shaving(global_best_a, p3_time, rng, n_peaks=5)
        print("  Peak shaving: obj=%.6f" % obj_ps)
        if obj_ps < global_best_obj:
            global_best_obj = obj_ps
            global_best_a = a_ps.copy()

        # Follow up with FISTA
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_ps2, obj_ps2 = run_adaptive_fista(global_best_a, remaining3 * 0.6, rng,
                                                 beta_init=100, beta_rate=0.005,
                                                 beta_max=50000, step_scale=0.5,
                                                 anneal_temp=0.001)
            print("  FISTA after PS: obj=%.6f" % obj_ps2)
            if obj_ps2 < global_best_obj:
                global_best_obj = obj_ps2
                global_best_a = a_ps2.copy()

    # === Phase 4: Polynomial optimization (10% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.10
    if p4_time > 30:
        print("\n=== Phase 4: Polynomial optimization (budget %.0fs) ===" % p4_time)
        for deg in [8, 16, 24]:
            if time.monotonic() >= deadline:
                break
            t_alloc = min(p4_time / 3.0, deadline - time.monotonic())
            if t_alloc < 10:
                break
            a_poly, obj_poly = polynomial_optimize(global_best_a, t_alloc, rng, degree=deg)
            print("  Poly deg=%d: obj=%.6f" % (deg, obj_poly))
            if obj_poly < global_best_obj:
                global_best_obj = obj_poly
                global_best_a = a_poly.copy()
                # Follow up with FISTA on the polynomial result
                remaining4 = deadline - time.monotonic()
                if remaining4 > 20:
                    a_pf, obj_pf = run_adaptive_fista(global_best_a, remaining4 * 0.5, rng,
                                                       beta_init=100, beta_rate=0.005,
                                                       beta_max=50000, step_scale=0.5)
                    if obj_pf < global_best_obj:
                        global_best_obj = obj_pf
                        global_best_a = a_pf.copy()

    # === Phase 5: Upsample to larger n (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.15
    if p5_time > 45 and len(global_best_a) <= 16384:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        t_fista = min(p5_time * 0.5, deadline - time.monotonic())
        if t_fista > 20:
            a_up_opt, obj_up_opt = run_adaptive_fista(a_up, t_fista, rng,
                                                       beta_init=10, beta_rate=0.005,
                                                       beta_max=20000, step_scale=1.0,
                                                       anneal_temp=0.003)
            print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
            if obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()

        # Peak shaving on upsampled
        remaining5 = deadline - time.monotonic()
        if remaining5 > 30:
            a_ps_up, obj_ps_up = peak_shaving(global_best_a, remaining5 * 0.7, rng, n_peaks=3)
            print("  PS upsampled: obj=%.6f" % obj_ps_up)
            if obj_ps_up < global_best_obj:
                global_best_obj = obj_ps_up
                global_best_a = a_ps_up.copy()

    # === Phase 6: Final high-precision polish (8% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.85
    if p6_time > 30:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_time)
        # Very high beta FISTA (essentially exact max)
        a_pol, obj_pol = run_adaptive_fista(global_best_a, p6_time * 0.5, rng,
                                             beta_init=5000, beta_rate=0.001,
                                             beta_max=500000, step_scale=0.1,
                                             anneal_temp=0.0005)
        print("  High-beta polish: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        # Final peak shaving
        remaining6 = deadline - time.monotonic()
        if remaining6 > 15:
            a_final, obj_final = peak_shaving(global_best_a, remaining6, rng, n_peaks=3)
            print("  Final PS: obj=%.6f" % obj_final)
            if obj_final < global_best_obj:
                global_best_obj = obj_final
                global_best_a = a_final.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
