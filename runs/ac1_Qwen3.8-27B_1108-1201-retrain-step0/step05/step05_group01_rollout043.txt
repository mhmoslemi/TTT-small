"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses continuation FISTA, frequency-domain spectral shaping,
simplified LP dual polishing, and multi-n exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def fista_continuation(a_init, max_time, rng, betas=(5, 20, 50, 100, 300, 1000, 3000, 10000)):

    FISTA with continuation: solve at increasing beta levels.
    At each level, run FISTA until convergence or time budget.
    """

    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_conv = np.inf
    best_a = a.copy()
    start = time.monotonic()
    
    total_budget = max_time
    # Allocate time: more time for higher beta levels

    weights = np.array(betas, dtype=float)
    weights = weights / weights.sum()
    # Bias towards later (higher beta) levels
    weights = np.power(weights, 0.7)

    for level, beta in enumerate(betas):
        if time.monotonic() - start > max_time:
            break

        a_work = a.copy()
        y = a_work.copy()

        last_improve_it = 0
        prev_obj = np.inf
        no_improve = 0

            # Simpler time check
            pass

        level_start = time.monotonic()
        while time.monotonic() - level_start < level_budget and (time.monotonic() - start) < max_time:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-14:
                best_conv = cur_max
                best_a = y.copy()
                last_improve_it = it

                no_improve += 1
            elif cur_max >= prev_obj - 1e-12:
                no_improve += 1
            else:
                no_improve = 0

            grad = conv_gradient(y, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            step = 1.0 / beta

            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a_work)

            a_work = x_new
            t_mom = t_new

        c_final = fft_convolve_self(a)
        if c_final.max() < best_conv:

            best_a = a.copy()
        
        elapsed_total = time.monotonic() - start
        print("  Continuation beta=%d: conv_max=%.10f (it=%d, t=%.1fs)" %

def spectral_shaping_perturb(a, rng, S, strength=0.01):
    """
    Perturb a in the frequency domain to encourage flatter spectrum.
    Compress the dynamic range of |FFT(a)|.
    """
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, L)
    
    # Compress magnitude
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Power compression: mag_new = mag^alpha / ||mag^alpha||_2 * ||mag||_2
    alpha = 0.7  # < 1 compresses
    mag_new = np.power(np.maximum(mag, 1e-30), alpha)
    # Preserve energy
    scale = np.linalg.norm(mag) / (np.linalg.norm(mag_new) + 1e-30)
    mag_new = mag_new * scale
    
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Blend and project
    a_pert = (1 - strength) * a + strength * np.maximum(a_new, 0)
    a_pert = project_simplex(a_pert, S)
    return a_pert

def lp_dual_weights(c, f, beta_approx=50.0, n_top=None):
    """
    Approximate LP dual weights using a soft version.
    Returns weights w[k] for the active constraints.
    The LP dual: min M*||y||_1 s.t. (f_rev * y)[j] >= 1, y >= 0.
    Approximate by: y[k] = exp(-beta_approx * c[k]) / sum (inverse of c).
    Actually, the dual weights should be proportional to how "binding" each constraint is.
    More binding = larger c[k] (closer to M).
    So weights should be concentrated on large c[k].
    This is the same as LSE weights! So LSE is already approximating the LP dual.
    Let me instead try a different approach: solve a smaller LP.
    
    Alternative: use the KKT conditions. At optimum, for active constraints k:
    sum_j f[j] * y[k-j] = 1 (dual feasibility)
    and y[k] = 0 for inactive constraints.
    
    If we know the active set A, we can solve:
    sum_{k in A} f[k-j] * y[k] = 1 for all j in support(f)
    y[k] = 0 for k not in A
    
    This is a system of linear equations!
    """
    n = len(f)
    M = c.max()
    # Find active set (constraints where c[k] is close to M)
    tol = 0.01
    active = np.where(c >= M * (1 - tol))[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # The dual constraint is: (f_reversed * y)[j] >= 1 for j = 0..n-1
    # f_reversed * y = sum_k f[k-j] * y[k] ... 
    # Actually, (f*g)[k] = sum_j f[j]*g[k-j], so the constraint matrix A[k,j] = f[k-j]
    # The dual constraint for variable j: sum_k A[k,j]*y[k] >= 1, i.e., sum_k f[k-j]*y[k] >= 1
    # Let m = k-j, so k = j+m: sum_m f[m]*y[j+m] >= 1
    # This is (f * y)[j] >= 1 (correlation of f and y)!
    
    # For active set A, we want: sum_{k in A} f[k-j]*y[k] = 1 for all j
    # This is a linear system. But it might be underdetermined or overdetermined.
    # Use least squares: minimize ||(fA * y) - 1||^2 where fA is the submatrix for active rows.
    
    # Actually, this is getting complicated. Let me just return LSE weights with a note.
    w = lse_weights(c, beta_approx)
    return w

            a = project_simplex(0.7 * a + 0.3 * smooth, S)

**Key observation**: The problem is to minimize `2n * max(f*f) / (sum f)^2` over non-negative f. By scale invariance, set `sum f = sqrt(2n)` and minimize `max(f*f)`.

The convolution `f*f` has the property that `sum_k (f*f)[k] = (sum f)^2 = 2n`. So the "average" of the convolution is `2n/(2n-1) ≈ 1`. We want the max to be as close to 1 as possible.

**Connection to flat-top windows**: In signal processing, a "flat-top" window is designed to have a very flat main lobe in the frequency domain. Similarly, we want a time-domain sequence whose autocorrelation (or in this case, convolution) is as flat as possible.

**New idea: Use the explicit structure of the optimal solution**

At the optimum (by KKT conditions), for the active constraints k (where (f*f)[k] = M):
The gradient should be constant on the support of f.

The gradient at coordinate j is: `grad[j] = 2 * sum_{k in A} w[k] * f[k-j]` (with appropriate bounds).

At the optimum: `grad[j] = lambda` for all j in support(f).

This means: `sum_{k in A} w[k] * f[k-j] = lambda/2` for all j in support(f).

This is a system of linear equations in f (given the active set A and weights w). If we know A and w, we can solve for f!

But A and w are what we're trying to find. This is a fixed-point problem.

**Iterative approach**:
1. Start with an estimate of A and w.
2. Solve the linear system to get f.
3. Update A based on the new f*f.
4. Repeat.

This is like the " Sinkhorn-Knopp" algorithm or "iterative proportional fitting."

Let me formalize:
- Given w (weights on the full convolution), the optimal f satisfies:
  (T_w f)[j] = lambda for all j where f[j] > 0
  (T_w f)[j] >= lambda for j where f[j] = 0
  where (T_w f)[j] = sum_k w[k] * f[k-j] (with f out of range = 0).

This is a linear complementarity problem! It can be solved by:
1. Assume a support S.
2. Solve (T_w f)[j] = lambda for j in S, with sum_{j in S} f[j] = S_total, f[j] >= 0.
3. Check: (T_w f)[j] >= lambda for j not in S.
4. If not, update S and repeat.

This is the "active set" method for the LP-type problem!

But this requires knowing w, which depends on the active set of the primal, which depends on f... It's a bilevel problem.

However, we can iterate:
1. Start with f.
2. Compute c = f*f, find active set A = {k : c[k] ≈ max(c)}.
3. Set w[k] = 1/|A| for k in A, 0 otherwise.
4. Solve the linear complementarity problem to get new f.
5. Repeat.

Step 4 is the key. How to solve it?

Given w, T_w is a linear operator: (T_w f)[j] = sum_k w[k] * f[k-j]. This is a convolution operator, so it can be applied via FFT.

The problem is: find f >= 0, sum f = S_total, such that:
- (T_w f)[j] = lambda for j in support(f)
- (T_w f)[j] >= lambda for j not in support(f)

This is equivalent to: minimize (or maximize) some objective subject to the constraint.

Actually, this is exactly the dual of the LP I mentioned earlier! Let me reconsider.

The primal LP (given f, find g):
maximize sum(g)
subject to (f*g)[k] <= M for all k, g >= 0

The dual LP:
minimize M * sum(y)
subject to (f_rev * y)[j] >= 1 for all j, y >= 0

Where f_rev[j] = f[n-1-j].

Primal: max 1^T g
s.t. A g <= M * 1 (where A[k,j] = f[k-j] if valid, else 0)
     g >= 0

Dual: min M * 1^T y
s.t. A^T y >= 1 (where (A^T y)[j] = sum_k A[k,j] * y[k] = sum_k f[k-j] * y[k])
     y >= 0

So the dual constraint is: sum_k f[k-j] * y[k] >= 1 for all j.

Let me define (f * y)[j] = sum_k f[k] * y[j-k] (standard convolution).
Then sum_k f[k-j] * y[k] = ... let m = k-j, so k = j+m:
sum_m f[m] * y[j+m] = (f * y_rev)[j] where y_rev[m] = y[-m]... 

This is getting confusing with the indexing. Let me just use the matrix form.

The key insight is: at the optimal dual solution y*, the constraints that are tight (sum_k A[k,j] * y*[k] = 1) correspond to the support of g*. And y* gives us the "weights" for the subgradient in the primal.

So the algorithm is:
1. Given f, solve the LP (or approximate it) to get y*.
2. The subgradient of max(f*f) in the direction of y* is: grad[j] = 2 * sum_k y*[k] * f[k-j].
3. Update f in the direction that reduces this subgradient.

But solving the LP at each step is expensive. The question is: can we get a good approximation cheaply?

**Approximation via power iteration on the dual:**

The dual LP is: min M * ||y||_1 s.t. A^T y >= 1, y >= 0.

This is a "covering" LP. We can approximate it by:
1. Start with y = 1 (uniform).
2. Check which constraints are violated: (A^T y)[j] < 1.
3. Increase y to satisfy the violated constraints.
4. Iterate until convergence.

This is the "multiplicative weights" or "Hemminger" algorithm for covering LPs. It converges in O(1/epsilon * log(n)) iterations, each taking O(nnz(A)) time.

For our problem, each iteration of the dual approximation takes O(n^2) time (since A has n^2 non-zeros). With n=8192, that's ~67M operations per iteration. With ~100 iterations for convergence, that's ~6.7G operations, which might take a few seconds. This is feasible!

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses continuation FISTA with spectral shaping, multi-n exploration,
adaptive active-set polishing, and diverse multi-start strategies.
Key insight: the convolution peak can be reduced below the triangular
shape by asymmetric/spectral sequences; we use continuation in the LSE
parameter to progressively sharpen the max approximation, with periodic
frequency-domain perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def spectral_shaping_perturb(a, rng, S, strength=0.01, alpha=0.6):
    """
    Perturb a in frequency domain to encourage flatter convolution spectrum.
    Compresses the dynamic range of |FFT(a)|.
    """
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, L)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = np.power(np.maximum(mag, 1e-30), alpha)
    scale = np.linalg.norm(mag) / (np.linalg.norm(mag_new) + 1e-30)
    mag_new = mag_new * scale
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_pert = (1 - strength) * a + strength * np.maximum(a_new, 0)
    a_pert = project_simplex(a_pert, S)
    return a_pert


def random_smooth_init(n, rng, S):
    """Generate a random smooth non-negative sequence."""
    mode = rng.integers(0, 4)
    t = np.linspace(0, 1, n, endpoint=False)
    if mode == 0:
        freq = rng.integers(1, 6)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        freq1 = rng.integers(1, 4)
        freq2 = rng.integers(1, 4)
        a = 0.3 + 0.35 * np.cos(2 * np.pi * freq1 * t) + 0.35 * np.cos(2 * np.pi * freq2 * t + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(3, n // 80)
        F[cutoff:-cutoff] *= 0.05
        a = np.real(ifft(F))[:n]
    else:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    return a


def fista_continuation(a_init, max_time, rng, betas=(5, 10, 30, 100, 300, 1000, 3000)):
    """
    FISTA with continuation: solve at increasing LSE parameter levels.
    At each level, run FISTA until convergence or time budget exhausted.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_conv = np.inf
    best_a = a.copy()
    start = time.monotonic()

    # Time allocation: geometric bias towards higher beta
    n_levels = len(betas)
    weights = np.linspace(0.5, 1.0, n_levels) ** 2
    weights = weights / weights.sum()

    for level, beta in enumerate(betas):
        if time.monotonic() - start > max_time:
            break
        level_budget = weights[level] * max_time * 0.85
        level_start = time.monotonic()

        a_work = a.copy()
        y = a_work.copy()
        t_mom = 1.0
        it = 0
        last_improve_it = 0
        prev_best = np.inf
        stall = 0

        while time.monotonic() - level_start < level_budget and (time.monotonic() - start) < max_time:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()

            if cur_max < best_conv - 1e-14:
                best_conv = cur_max
                best_a = y.copy()
                last_improve_it = it
                stall = 0
            else:
                stall += 1

            if stall > 3000:
                break

            # Adaptive step: use 1/beta as Lipschitz constant for LSE gradient
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            step = 1.0 / beta
            # Safety cap
            step = min(step, 0.5 * S / (gnorm + 1e-15))

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a_work)
            y = np.maximum(y, 0.0)
            a_work = x_new
            t_mom = t_new

            if it % 3000 == 0:
                t_mom = 1.0

        a = a_work.copy()
        c_final = fft_convolve_self(a)
        if c_final.max() < best_conv:
            best_conv = c_final.max()
            best_a = a.copy()

        elapsed = time.monotonic() - start
        print("    beta=%5d: conv_max=%.10f, obj=%.8f (it=%d, t=%.0fs)" %
              (beta, best_conv, 2.0 * n * best_conv / S**2, it, elapsed))

    return best_a, best_conv


def fista_perturb(a_init, max_time, rng, beta=80.0, perturb_interval=2000,
                  perturb_strength=0.02, alpha=0.5):
    """
    FISTA with periodic spectral shaping perturbations to escape local minima.
    Also uses adaptive beta reduction when stalled.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: reduce when stuck to explore more broadly
        eff_beta = beta
        if stuck > 2000:
            eff_beta = max(5.0, beta * np.exp(-stuck / 8000.0))

        w = lse_weights(c, eff_beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / max(eff_beta, 5.0)
        step = min(step, 0.4 * S / (gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Spectral perturbation when stuck
        if stuck > perturb_interval and it % 500 == 0:
            a = spectral_shaping_perturb(a, rng, S,
                                         strength=perturb_strength * min(1.0, stuck / (2 * perturb_interval)),
                                         alpha=alpha)
            y = a.copy()
            t_mom = 1.0

        # Hard restart when very stuck
        if stuck > 12000 and it % 2000 == 0:
            smooth = random_smooth_init(n, rng, S)
            a = project_simplex(0.6 * a + 0.4 * smooth, S)
            y = a.copy()
            t_mom = 1.0

        # Momentum reset
        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def active_set_polish(a_init, max_time, rng):
    """
    polish using exact active-set subgradients with adaptive tolerance.
    Uses increasingly broader active sets when stuck.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance: widen active set when stuck
        if stuck < 50:
            tol_frac = 0.0
        elif stuck < 200:
            tol_frac = 0.001
        elif stuck < 800:
            tol_frac = 0.005
        elif stuck < 3000:
            tol_frac = 0.015
        else:
            tol_frac = min(0.08, 0.015 + 0.005 * (stuck - 3000) // 1000)

        active = np.where(c >= cur_max * (1 - tol_frac))[0]
        w_full = np.zeros(len(c))
        # Weight by proximity to max (sharper for more active)
        active_vals = c[active]
        weights_a = np.exp(-((active_vals - cur_max) / (cur_max * tol_frac + 1e-30)) * 2.0)
        weights_a = weights_a / weights_a.sum()
        w_full[active] = weights_a

        grad = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking-lite
        step = S / (gnorm * 1.5)
        step = min(step, S / (0.5 * gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

        # Small perturbation when very stuck
        if stuck > 8000 and it % 2000 == 0:
            a = spectral_shaping_perturb(a, rng, S, strength=0.003, alpha=0.7)
            y = a.copy()
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def topk_gradient_method(a_init, max_time, rng, k_start=4, k_max=128):
    """
    Gradient method using top-k weighted subgradient.
    Adapts k based on progress.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adapt k
        if stuck < 200:
            k = k_start
        elif stuck < 1000:
            k = min(k_max, k_start + stuck // 100)
        else:
            k = min(k_max, int(k_start + 0.1 * stuck))

        idx = np.argsort(c)[::-1][:k]
        vals = c[idx]
        w = np.maximum(vals - vals[-1] + 1e-30, 1e-30)
        w = w / w.sum()
        w_full = np.zeros(len(c))
        w_full[idx] = w

        grad = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * 2.0)
        step *= (1.0 + 0.001 * it)  # slightly increasing step

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0

        if stuck > 6000 and it % 1500 == 0:
            a = spectral_shaping_perturb(a, rng, S, strength=0.005, alpha=0.6)
            y = a.copy()
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase strategy:
    1. Continuation FISTA from best known
    2. FISTA with spectral perturbations
    3. Multi-n exploration (4096, 16384)
    4. Top-k and active-set polishing
    5. Final refinement
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.8f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = np.ones(n0) * np.sqrt(2.0 * n0) / n0
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
