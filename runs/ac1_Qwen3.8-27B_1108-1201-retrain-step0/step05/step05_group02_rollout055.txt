"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. Multi-scale approach: optimize at multiple sizes
2. FISTA with LSE for smooth optimization
3. LP-based direction finding (from state-of-the-art)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,

def solve_direction_lp(f, M, n):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k.
    Returns g or None if infeasible/too slow.
    """
    # Build constraint matrix: A[k, j] = f[j] if 0 <= k-j < n
    # Shape: (2n-1) x n
    rows = []
    cols = []
    vals = []
    
    # For efficiency, build row by row
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[js].tolist())
    
    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_vec = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=A_sp, b_ub=b_vec, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
        else:
            return None
    except Exception:
        return None

def lp_direction_step(a, max_time_lp=30.0, t_blend=0.1):
    """
    One step of the LP-based direction finding.
    Returns improved a or None.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    M = c.max()
    
    g = solve_direction_lp(a, M * 1.0, n)  # Use slight relaxation
    if g is None or g.sum() < 1e-10:
        return None
    
    # Scale g to match normalization
    g = g * (S / g.sum())
    
    # Blend
    a_new = (1 - t_blend) * a + t_blend * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new * (S / a_new.sum())
    
    return a_new

                le = min(2 * n - 2, l + n - 1)

            a[l] += delta

def run_subgradient_polish(a_init, max_time, rng):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 2
        elif stuck < 5000:
            k = min(8, 2 + stuck // 500)
        else:
            k = min(32, 8 + stuck // 1000)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.002 * it))
        step = min(step, 2.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 3000 and it % 100 == 0:
            ns = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
1. Simplified FISTA (clip+normalize instead of full simplex projection) for speed
2. LP-based direction finding from state-of-the-art for escaping local minima
3. Exact subgradient with Polyak step as alternative when LSE stalls
4. Multi-scale approach with upsampled refinement
5. Adaptive noise and restart strategies

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def clip_normalize(v, S):
    """Clip at 0 and rescale to sum S (cheaper than simplex projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.full(len(v), S / len(v))
    return v * (S / s)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0, use_backtrack=True, use_simplex=False):
    """FISTA with adaptive LSE, using clip+normalize or full simplex projection."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            if use_simplex:
                x_trial = project_simplex(y - step_local * grad, S)
            else:
                x_trial = clip_normalize(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 6:
                step_local *= 0.5
                if use_simplex:
                    x_trial = project_simplex(y - step_local * grad, S)
                else:
                    x_trial = clip_normalize(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.7
            else:
                step = min(step * 1.02, 1.5 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            if use_simplex:
                x_new = project_simplex(y - step * grad, S)
            else:
                x_new = clip_normalize(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        if use_simplex:
            y = project_simplex(y, S)
        else:
            y = clip_normalize(y, S)
        a = x_new
        t_mom = t_new

        # Momentum restart with structured noise
        stuck = it - last_improve
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 3000.0))
            y = clip_normalize(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()

    return best_a, best_obj


def run_exact_subgradient(a_init, max_time, rng):
    """Exact subgradient with Polyak step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    f_star = 1.0  # theoretical lower bound (normalized: max_conv = 1)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj_val = cur_max  # normalized objective (since sum = sqrt(2n))

        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = 5
        else:
            k = min(20, 5 + (stuck - 2000) // 200)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step: (f(x) - f*) / ||g||^2
        gap = max(cur_max - f_star, 1e-6)
        step = gap / (gnorm * gnorm)
        step = min(step, 2.0 * gap / gnorm)

        a_new = clip_normalize(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new * (S * S) / (2.0 * n) <= cur_max + 1e-10:
            a = a_new
        else:
            step *= 0.1
            a_new = clip_normalize(a - step * grad, S)
            if compute_obj(a_new) * (S * S) / (2.0 * n) <= cur_max + 1e-8:
                a = a_new

        if stuck > 2000 and it % 50 == 0:
            ns = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = clip_normalize(a + rng.normal(0, ns, n), S)

    return best_a, 2.0 * n * best_obj / (S * S)


def solve_direction_lp(f, M, n, time_limit=20.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k.
    Returns g or None.
    """
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[js].tolist())

    A_sp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_vec = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_vec, A_ub=A_sp, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_step(a, t_blend=0.1, time_limit=20.0):
    """One step of LP-based direction finding."""
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    M = c.max() * 1.001  # slight relaxation for feasibility

    g = solve_direction_lp(a, M, n, time_limit=time_limit)
    if g is None or g.sum() < 1e-10:
        return None

    g = g * (S / g.sum())
    a_new = (1 - t_blend) * a + t_blend * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new * (S / a_new.sum())
    return a_new


def coordinate_descent(a_init, max_time, rng, max_pairs_per_iter=8):
    """Direct coordinate descent via mass transfer."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 15000:
            break

        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            lo2 = max(0, int(kk) - n + 1)
            hi2 = min(n - 1, int(kk))
            if hi2 >= lo2:
                multi_contrib[lo2:hi2 + 1] += a[int(kk) - hi2:int(kk) - lo2 + 1][::-1]

        src_scores = multi_contrib * np.maximum(a, 0.001)
        tgt_scores = -multi_contrib * (S / n - a + 0.001)

        n_try = min(max_pairs_per_iter, n)
        src_candidates = np.argsort(src_scores)[::-1][:n_try]
        tgt_candidates = np.argsort(-tgt_scores)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in src_candidates[:n_try // 2]:
            for l in tgt_candidates[:n_try // 2]:
                if j == l or a[j] < 1e-10:
                    continue
                delta = a[j] * 0.05
                if delta < 1e-12:
                    continue
                delta_c = np.zeros(2 * n - 1)
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                delta_c[js:je + 1] -= delta * coeff_j
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                delta_c[ls:le + 1] += delta * coeff_l

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj:
                    best_delta_obj = new_obj
                    best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            js = max(0, j)
            je = min(2 * n - 2, j + n - 1)
            c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
            ls = max(0, l)
            le = min(2 * n - 2, l + n - 1)
            c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)]
        else:
            if stuck > 500:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.02
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt)
        a += 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3)
        a += 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "spectral_flat":
        # Start with flat spectrum, give it some time-domain structure
        a = np.ones(n)
        for i in range(1, min(5, n // 2)):
            a += 0.5 / i * np.cos(2 * np.pi * i * tt + rng.random() * 2 * np.pi)
        a = np.abs(a)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

        if n2 <= 32768:
            print("\n=== Phase 6: Upsample to n=%d (budget %.0fs) ===" % (n2, p6_time))
            a_up = upsample(global_best_a, factor=2)
            S2 = np.sqrt(2.0 * n2)
            a_up = a_up * (S2 / a_up.sum())
            obj_up = compute_obj(a_up)
            print("  Upsampled initial: obj=%.6f" % obj_up)
            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_up.copy()

            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt, obj_opt = run_fista(a_up, min(rem * 0.7, p6_time * 0.7), rng,
                                            beta_init=5, beta_rate=0.005,
                                            beta_max=10000, step_scale=1.2,
                                            use_backtrack=False, use_simplex=False)
                print("  FISTA upsampled: obj=%.6f" % obj_opt)
                if obj_opt < global_best_obj:
                    global_best_obj = obj_opt
                    global_best_a = a_opt.copy()

    # === Phase 7: Final high-precision polish (10% time) ===
    remaining = deadline - time.monotonic()
    p7_time = remaining * 0.90
    if p7_time > 10:
        print("\n=== Phase 7: Final polish (budget %.0fs) ===" % p7_time)
        # Very high beta FISTA for precise max reduction
        a_opt, obj_opt = run_fista(global_best_a, p7_time, rng,
                                    beta_init=1000, beta_rate=0.002,
                                    beta_max=500000, step_scale=0.3,
                                    use_backtrack=True, use_simplex=True)
        print("  High-beta final: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # Final verification
    print("\n=== Final Result ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verified: obj=%.6f, n=%d" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
