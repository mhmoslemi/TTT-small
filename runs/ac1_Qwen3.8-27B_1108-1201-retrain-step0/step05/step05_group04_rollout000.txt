"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length 8192.
Strategy: Multi-phase optimization combining adaptive LSE-FISTA, weighted active-set
gradient descent with careful line search, coordinate-level polishing, and
stochastic restarts to escape local minima. The final phase uses a subgradient
method on the true max with extremely small steps for the "last mile."

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def weighted_active_gradient(a, c, top_k=50, excess_power=2.0):
    """
    Gradient using weighted active set. Weights are based on how much
    each peak exceeds the median of the top peaks.
    """
    n = len(a)
    c_sorted = np.sort(c)[::-1]
    cm = c_sorted[0]
    # Use top_k lags
    k = min(top_k, len(c))
    top_vals = c_sorted[:k]
    median_top = np.median(top_vals)
    # Weights: (excess above median)^excess_power
    excess = np.maximum(top_vals - median_top, 0.0) + 1e-15
    weights = excess ** excess_power
    # Also include all lags that are within a small threshold of max
    threshold = cm - 0.01
    mask = c >= threshold
    if mask.sum() > k:
        # Use only top_k
        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_idx] = weights
    else:
        w = np.zeros(len(c))
        w[mask] = 1.0
        w[mask] = (c[mask] - median_top + 1e-15) ** excess_power
    tot = w.sum()
    if tot < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def true_max_subgradient(a, c, n_peaks=20):
    """
    True subgradient of max(c): average over top n_peaks lags.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:n_peaks]
    # Average the contributions
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / n_peaks
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

            #   c[k] - a[j]*a[k-j] + new_val*a[k-j]
            # This is: c_base[k] + (new_val - old_val)*a[k-j]
            # where c_base[k] = c[k] - old_val*a[k-j]

            # Affected lags: k from j to j+n-1, but limited to [0, 2n-2]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            partners = ks - j  # These are the indices of a that multiply a[j]
            # partners range from 0 to n-1 (valid)
            partner_vals = a[partners]

            # Current c at these lags
            # We need to recompute: c[k] = sum_i a[i]*a[k-i]
            # The term involving a[j] is a[j]*a[k-j] = a[j]*partner_vals
            # So c_base[k] = c[k] - old_val * partner_vals

            # We don't have c stored efficiently for arbitrary k, so let's compute
            # Actually, let's just evaluate the objective for a few candidate values

            # Try: reduce a[j] slightly (if it's contributing to peaks)
            # and increase others proportionally
            # But that's expensive. Instead, just try a few values.

            # Quick heuristic: if a[j] is large and contributes to the peak, reduce it
            # Compute peak contribution
            max_partner = partner_vals.max() if len(partner_vals) > 0 else 0
            if max_partner * old_val > best * 0.8:
                # Try reducing
                for frac in [0.9, 0.8, 0.7, 0.5]:
                    new_val = old_val * frac
                    a[j] = new_val
                    # Renormalize: add the difference spread over others
                    diff = old_val - new_val
                    if diff > 0:
                        others = np.ones(n)
                        others[j] = 0
                        # Add proportionally to existing values (or uniformly)
                        a += diff * (a / max(a.sum() - old_val, 1e-30))
                        a[j] = new_val
                        a = project_simplex(a, S)

                    c_test = fft_convolve_self(a)
                    if c_test.max() < best:
                        best = c_test.max()
                        best_a = a.copy()
                        changed_any = True
                        break
                    else:
                        a[j] = old_val
                        # Undo the normalization change
                        a = project_simplex(a, S)

                if not changed_any:
                    a[j] = old_val

        c_final = fft_convolve_self(a)
        if c_final.max() < best:
            best = c_final.max()
            best_a = a.copy()

    return best_a, best


def run_lp_direction(a, n, S, M, top_k=50, solver_timeout=30):
    """
    Solve a small LP to find the direction that maximizes sum(g)
    subject to (a*g)[k] <= M for the top_k lags.
    Returns the direction g (normalized to sum S).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, np.inf

    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:top_k]

    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in top_idx:
        # (a*g)[k] = sum_j a[j]*g[k-j] for valid j, k-j
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        # constraint: sum_{j in js} a[j]*g[partners[j]] <= M
        # Since g is indexed 0..n-1 and partners are 0..n-1:
        # a[j] * g[k-j]
        expr = a[lo:hi+1] @ g[partners]
        constraints.append(expr <= M)

    # Also add a constraint that sum(g) >= some minimum to avoid trivial solution
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    try:
        prob.solve(solver=cp.CLARABEL, max_time=solver_timeout)
        if g.value is not None and np.isfinite(g.value).all():
            gv = np.maximum(g.value, 0)
            sv = gv.sum()
            if sv > 1e-10:
                return gv * (S / sv), np.inf
    except Exception:
        pass

    return None, np.inf


def raised_cosine_init(n, center_frac=0.3, width_frac=0.5):
    """Generate a raised-cosine initialization."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-12)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    for cf in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
            inits.append(raised_cosine_init(n, cf, wf))

    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.3]:
        for mu in [0.25, 0.35, 0.45, 0.5, 0.55, 0.65]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-12))

    # Anti-symmetric-ish (shifted)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = t ** p
        inits.append(np.maximum(a, 1e-12))
        a = (1 - t) ** p
        inits.append(np.maximum(a, 1e-12))

    # Two bumps
    for d in [0.1, 0.15, 0.2, 0.25]:
        a = np.exp(-0.5 * ((t - 0.35) / 0.1) ** 2) + np.exp(-0.5 * ((t - 0.65) / 0.1) ** 2)
        inits.append(np.maximum(a, 1e-12))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence of length 8192 minimizing
    2*n*max(f*f)/sum(f)^2.
    Multi-phase: LSE-FISTA -> Weighted Active Descent -> Subgradient Polish -> Coordinate Polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    n = 8192
    S = np.sqrt(2.0 * n)

    print("Budget: %.0fs, n=%d, S=%.4f" % (budget_s, n, S))

    # Try to load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d, sum=%.4f" % (len(given), given.sum()))
            # If different length, interpolate to n
            if len(given) != n:
                t_old = np.linspace(0, 1, len(given))
                t_new = np.linspace(0, 1, n)
                given = np.interp(t_new, t_old, given)
            given = normalize_sum(given, n)
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  [%.1fs remaining] New best: %.10f" % (deadline - time.monotonic(), val))

    # === Phase 0: If we have a given sequence, quick evaluate ===
    if given is not None:
        try_sequence(given)
        print("Phase 0 initial: %.10f" % global_best)

    # === Phase 1: LSE-FISTA (40% of budget) ===
    remaining = deadline - time.monotonic()
    t1 = remaining * 0.40
    print("=== Phase 1: LSE-FISTA (%.0fs) ===" % t1)
    if given is not None:
        a1, v1 = run_lse_fista(given, n, S, t1, rng,
                                beta_start=50.0, beta_end=15000.0, beta_steps=150)
    else:
        a1, v1 = run_lse_fista(np.ones(n) * 0.5 + 0.5 * np.linspace(0, 1, n), n, S, t1, rng,
                                beta_start=50.0, beta_end=15000.0, beta_steps=150)
    try_sequence(a1)

    # === Phase 2: Weighted Active Descent (30% of remaining) ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.30
    print("=== Phase 2: Weighted Active Descent (%.0fs) ===" % t2)
    if global_a is not None:
        a2, v2 = run_weighted_active_descent(global_a, n, S, t2, rng, top_k=30)
        try_sequence(a2)

    # === Phase 3: Subgradient Polish (15% of remaining) ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.15
    print("=== Phase 3: Subgradient Polish (%.0fs) ===" % t3)
    if global_a is not None:
        a3, v3 = run_subgradient_polish(global_a, n, S, t3, rng, n_peaks=50)
        try_sequence(a3)

    # === Phase 4: Another round of LSE with high beta (10%) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.15
    print("=== Phase 4: High-beta LSE (%.0fs) ===" % t4)
    if global_a is not None:
        a4, v4 = run_lse_fista(global_a, n, S, t4, rng,
                                beta_start=5000.0, beta_end=50000.0, beta_steps=50)
        try_sequence(a4)

    # === Phase 5: Weighted Active Descent again with more peaks ===
    remaining = deadline - time.monotonic()
    t5 = remaining * 0.30
    print("=== Phase 5: Active Descent refined (%.0fs) ===" % t5)
    if global_a is not None:
        a5, v5 = run_weighted_active_descent(global_a, n, S, t5, rng, top_k=50)
        try_sequence(a5)

    # === Phase 6: Final subgradient polish ===
    remaining = deadline - time.monotonic()
    t6 = remaining * 0.5
    print("=== Phase 6: Final polish (%.0fs) ===" % t6)
    if global_a is not None and t6 > 10:
        a6, v6 = run_subgradient_polish(global_a, n, S, t6, rng, n_peaks=100)
        try_sequence(a6)

    # === Phase 7: Try a random restart from the best ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Phase 7: Random restarts (%.0fs) ===" % remaining)
        n_restarts = min(3, int(remaining // 30))
        per_restart = remaining / max(1, n_restarts)
        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Perturb the best
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a_pert = project_simplex(global_a + noise, S)
            # Quick LSE
            a_r, v_r = run_lse_fista(a_pert, n, S, per_restart * 0.6, rng,
                                      beta_start=200.0, beta_end=5000.0, beta_steps=30)
            try_sequence(a_r)
            # Quick active descent
            remaining_r = per_restart - (time.monotonic() - (deadline - remaining)) 
            if time.monotonic() < deadline - 15:
                t_ar = max(5.0, (deadline - time.monotonic()) * 0.3)
                a_ar, v_ar = run_weighted_active_descent(global_a, n, S, t_ar, rng, top_k=40)
                try_sequence(a_ar)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n, dtype=np.float64) * (S / n)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
