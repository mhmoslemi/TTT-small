import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_fista_lse(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0):
    """FISTA with LSE surrogate and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: increase slowly, accelerate when stuck
        if stuck < 200:
            beta = beta_init * (1.0 + 0.01 * it)
        elif stuck < 2000:
            beta = min(beta_max, beta * 1.005)
        else:
            beta = min(beta_max, beta * 1.02)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / (beta * np.max(a) + 1.0)
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_conv

    best_a = a.copy()

        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            step_size = min(step_size * 1.05, 0.5)

        stuck = it - last_improve
        if stuck > 1000:
            step_size = max(0.05, step_size * 0.9)

        # Find top-k peak positions
        k = min(top_k, len(c))
        peak_idx = np.argsort(c)[::-1][:k]
        
        # Compute combined gradient from top-k peaks
        # Weight peaks by their excess over the mean
        mean_c = (S * S) / len(c)
        weights = np.maximum(c[peak_idx] - mean_c, 0)
        if weights.sum() < 1e-30:
            weights = np.ones(k)
        weights = weights / weights.sum()
        
        # Gradient = sum of weighted subgradients
        grad = np.zeros(n)
        for w_i, k_i in zip(weights, peak_idx):
            gi = exact_subgradient(a, c, tol_frac=0.0)
            # Actually, for a specific peak at position k_i, the gradient is:
            # grad[j] = 2 * a[k_i - j] (for valid k_i - j)
            lo = max(0, k_i - n + 1)
            hi = min(n - 1, k_i)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * w_i * a[k_i - j]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Take step
        direction = -grad / gnorm
        # Project: a_new = project(a + step * direction, S)
        a_new = project_simplex(a + step_size * n * direction, S)
        
        # Check if it actually helps
        c_check = fft_convolve_self(a_new)
        if c_check.max() < cur_max:
            a = a_new

            a_new = project_simplex(a + step_size * 0.5 * n * direction, S)
            c_check = fft_convolve_self(a_new)
            if c_check.max() < cur_max:
                a = a_new
            else:
                # Try a random perturbation
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

    return best_a, best_obj

The objective is `2*n*max(conv(a,a))/sum(a)^2`. 

Key observation: if I scale `a` by a constant `c`, the objective is unchanged (both numerator and denominator scale by `c^2`). So the objective only depends on the "shape" of `a`, not its scale.

The constraint is `a >= 0` and `sum(a) > 0`.

So I'm really looking for the shape that minimizes `max(conv(a,a))/sum(a)^2`.

Let me define `r = max(conv(a,a))/sum(a)^2`. I want to minimize `2*n*r`.

For a uniform sequence (all a[j] = 1), `conv(a,a)[k] = min(k+1, 2n-1-k, n)` (a triangular shape). The max is `n` (at the center). `sum(a) = n`. So `r = n/n^2 = 1/n`. Objective = `2*n*(1/n) = 2`.

For the optimal sequence, we're getting about 1.5, which is a 25% improvement over uniform.

What kind of sequence gives a flatter convolution? 

Think about it in the frequency domain. `conv(a,a)` is the IDFT of `|DFT(a)|^2`. To make the convolution flat, we want `|DFT(a)|^2` to be as "spread out" as possible... wait, no. A flat convolution (delta function) corresponds to a flat `|DFT(a)|^2`. But we can't have a delta function convolution with a non-negative sequence.

Actually, for a non-negative sequence, `DFT(a)` at frequency 0 is `sum(a) > 0`, and by the triangle inequality, `|DFT(a)[k]| <= |DFT(a)[0]| = sum(a)` for all k. The autocorrelation is the IDFT of `|DFT(a)|^2`, and the "flat" part corresponds to the AC components of `|DFT(a)|^2`.

For the autocorrelation to be flat (all entries equal to `sum(a)^2/(2n-1)`), we'd need `|DFT(a)|^2` to be a constant, which means `a` is a "flat" sequence in the frequency domain, i.e., a Dirac delta in the time domain. But a Dirac delta isn't non-negative (well, it is, but it has only one non-zero entry).

So the question is: what non-negative sequence has the flattest autocorrelation?

This is related to the "flat finite Fourier transform" problem. The known constructions use:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimization with:
- Diverse initializations (including anti-symmetric components)
- LSE FISTA with adaptive beta and momentum restart
- Perturbation-based local search (simulated annealing) to escape local minima
- Coordinate-descent-style polish
- Multi-n exploration (4096, 8192, 16384)
- Active-set gradient for final polishing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, k_pos):
    """Exact subgradient at a specific lag position."""
    n = len(y)
    lo = max(0, k_pos - n + 1)
    hi = min(n - 1, k_pos)
    grad = np.zeros(n)
    grad[lo:hi+1] = 2.0 * y[k_pos - hi:k_pos - lo + 1][::-1]
    return grad


def topk_combined_gradient(y, c, k, power=1.0):
    """Combined gradient from top-k peak positions."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        weights = np.ones(k) / k
    else:
        weights = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
    grad = np.zeros(n)
    for w_i, k_i in zip(weights, idx):
        lo = max(0, int(k_i) - n + 1)
        hi = min(n - 1, int(k_i))
        grad[lo:hi+1] += w_i * 2.0 * y[k_i - hi:k_i - lo + 1][::-1]
    return grad


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_fista_lse(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0):
    """FISTA with LSE surrogate and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 300:
            beta = beta_init * (1.0 + 0.008 * it)
        elif stuck < 3000:
            beta = min(beta_max, beta * 1.003)
        else:
            beta = min(beta_max, beta * 1.015)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        max_a = np.max(a)
        step = 0.8 / (beta * max_a + 2.0 * max_a + 1.0)
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart with noise when stuck
        if stuck > 4000:
            t_mom = 1.0
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_perturbation_search(a_init, max_time, rng):
    """Simulated annealing style perturbation search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    # Temperature for perturbation size
    scale = 0.05 * S / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        # Anneal: reduce perturbation size over time
        frac = elapsed / max_time
        current_scale = scale * (1.0 - 0.7 * frac)
        current_scale = max(current_scale, 0.001 * S / np.sqrt(n))

        # Try different perturbation types
        r = rng.random()
        if r < 0.3:
            # Gaussian noise
            pert = rng.normal(0, current_scale, n)
        elif r < 0.5:
            # Sparse large perturbation (random subset)
            num_sparse = max(1, n // 100)
            pert = np.zeros(n)
            idx = rng.choice(n, size=num_sparse, replace=False)
            pert[idx] = rng.normal(0, current_scale * 5, num_sparse)
        elif r < 0.7:
            # Multiplicative perturbation
            log_pert = rng.normal(0, 0.1 * (1 - frac), n)
            pert = a * (np.exp(log_pert) - 1)
        else:
            # Swap-based perturbation (move mass between random pairs)
            pert = np.zeros(n)
            for _ in range(max(1, n // 500)):
                i, j = rng.choice(n, size=2, replace=False)
                amt = current_scale * 0.5 * rng.random()
                pert[i] -= amt
                pert[j] += amt

        a_try = project_simplex(a + pert, S)
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            last_improve = it
        elif obj_try < best_obj:
            # Accept with some probability (annealing)
            if rng.random() < 0.5:
                a = a_try.copy()
        elif it - last_improve > 500 and rng.random() < 0.1:
            # Occasionally accept worse solutions to escape
            a = a_try.copy()

    return best_a, best_obj


def run_active_set_polish(a_init, max_time, rng):
    """Active-set polishing with adaptive top-k and line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_size = 0.5 * S / np.sqrt(n)

    y = a.copy()
    t_mom = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k for top-k gradient
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 3 + stuck // 200)
        elif stuck < 8000:
            k = min(64, 16 + stuck // 400)
        else:
            k = min(256, 64 + stuck // 200)

        grad = topk_combined_gradient(y, c, k, power=1.5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with line search
        base_step = step_size
        accepted = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            x_new = project_simplex(y - base_step * shrink * grad / gnorm, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max - 1e-15:
                y = x_new
                accepted = True
                if c_new.max() < cur_max * 0.999:
                    step_size = min(step_size * 1.1, 2.0 * S / np.sqrt(n))
                break

        if not accepted:
            # Try momentum step
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            x_mom = x_new + ((t_mom - 1.0) / t_new) * (x_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            c_mom = fft_convolve_self(x_mom)
            if c_mom.max() < cur_max:
                y = x_mom
            else:
                y = project_simplex(y - base_step * 0.1 * grad / gnorm, S)
            t_mom = t_new
        else:
            t_mom = t_mom * 0.95  # slight momentum decay on success

        # Restart when stuck
        if stuck > 6000 and it % 100 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - stuck / 20000.0)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            step_size = 0.5 * S / np.sqrt(n)

    return best_a, best_obj


def make_init_uniform(n):
    return np.ones(n) * 0.5


def make_init_cosine(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.3 * np.cos(2 * np.pi * tt)


def make_init_antisymmetric(n):
    """Sequence with antisymmetric component to reduce center peak."""
    tt = np.arange(n, dtype=np.float64)
    # Anti-symmetric: high on left, low on right (and vice versa modulation)
    base = 0.5
    asym = 0.3 * np.sin(2 * np.pi * (tt + 0.25) / n)
    return np.maximum(base + asym, 0.01)


def make_init_two_bump(n):
    """Two concentrated bumps to reduce center autocorrelation."""
    tt = np.arange(n, dtype=np.float64)
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * 0.1
    b1 = np.exp(-0.5 * ((tt - center1) / width) ** 2)
    b2 = np.exp(-0.5 * ((tt - center2) / width) ** 2)
    return 0.2 + 0.8 * (b1 + b2)


def make_init_ramp(n):
    """Linear ramp - creates directional bias."""
    tt = np.linspace(0, 1, n)
    return 0.3 + 0.7 * tt


def make_init_random(n, rng):
    a = rng.random(n)
    a = a ** 0.5  # Less peaked than a^2
    return a


def make_init_chirp(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.3 * np.cos(2 * np.pi * tt * tt / (2.0 * n))


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trimmed = a[:n_new * factor]
    return a_trimmed.reshape(n_new, factor).mean(axis=1)

            if obj_fin < global_best_obj:
                global_best_obj = obj_fin
                global_best_a = a_fin.copy()
            print("    Final FISTA: obj=%.6f" % obj_fin)

    # Try a few more perturbation rounds on the global best if time remains
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_best_a is not None:
        n_gb = len(global_best_a)
        S_gb = np.sqrt(2.0 * n_gb)
        print("\n=== Final perturbation polish on n=%d (%.0fs) ===" % (n_gb, remaining * 0.5))
        a_final_pert, obj_fp = run_perturbation_search(global_best_a, remaining * 0.4, rng)
        if obj_fp < global_best_obj:
            global_best_obj = obj_fp
            global_best_a = a_final_pert.copy()
        print("  Perturbation: obj=%.6f" % obj_fp)

    # One more active-set polish
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_best_a is not None:
        a_final_as, obj_fa = run_active_set_polish(global_best_a, remaining * 0.7, rng)
        if obj_fa < global_best_obj:
            global_best_obj = obj_fa
            global_best_a = a_final_as.copy()
        print("  Final active-set: obj=%.6f" % obj_fa)

    # Final verification
    print("\n=== Final Result ===")
    if global_best_a is not None and compute_obj(global_best_a) < global_best_obj:
        global_best_obj = compute_obj(global_best_a)

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Computed obj: %.6f" % global_best_obj)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = make_init_uniform(n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = make_init_uniform(n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
