"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences.
Strategy: LSE-FISTA with backtracking, spectral shaping, adaptive beta,
multi-n exploration (8192, 16384), and targeted anti-sticking perturbations.
Key improvements over prior work: backtracking line search, spectral flattening
steps, larger dimensionality, and smarter noise injection targeting max lags.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, k_count=1, tol_frac=0.0):
    """Subgradient from top-k lags or lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.argsort(c)[::-1][:k_count]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def spectral_shaping(a, strength=0.3, n_harmonics=None):
    """Apply spectral flattening: reduce spectral peaks to flatten autocorrelation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A)
    # Apply power-law compression to flatten spectrum
    if n_harmonics is None:
        n_harmonics = max(4, n // 64)
    # Only modify the first n_harmonics bins (low freq)
    for m in range(1, min(n_harmonics, fsize // 2)):
        if power[m] > 0:
            # Reduce peak: A_new = A * (mean_power / power)^alpha
            mean_power = power[:n_harmonics].mean()
            ratio = mean_power / power[m]
            factor = ratio ** strength
            factor = min(factor, 3.0)  # Cap the adjustment
            A[m] *= factor
            A[fsize - m] *= factor
    a_new = np.real(ifft(A))[:n]
    a_new = np.clip(a_new, 0, None)
    a_new = normalize_seq(a_new, S)
    return a_new

                    t_mom = 1.0
                continue

    if obj_final < best_obj:
        best_a = best_a.copy()
        best_obj = obj_final

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)

def upsample(a, factor=2):
    """Upsample using FFT-based interpolation."""
    n = len(a)
    n_new = n * factor
    fsize_old = 1 << (2 * n - 1).bit_length()
    fsize_new = 1 << (2 * n_new - 1).bit_length()
    A = fft(a, fsize_old)
    # Zero-pad in frequency domain
    A_new = np.zeros(fsize_new, dtype=np.complex128)
    half_old = fsize_old // 2
    half_new = fsize_new // 2
    A_new[:half_old] = A[:half_old]
    A_new[-(half_old - 1):] = A[-(half_old - 1):]
    a_new = np.real(ifft(A_new))[:n_new]
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-15:
        a_new = np.full(n_new, 1.0)
    return a_new

        inits.append(normalize_seq(make_random_smooth_init(n, rng), S))
        names.append("rand_smooth")

        inits.append(normalize_seq(make_dual_peak_init(n), S))
        names.append("dual_peak")

        # If we have a result from a smaller n, upsample it
        if global_best_a is not None and len(global_best_a) < n:
            ratio = n // len(global_best_a)
            if ratio >= 2 and ratio % 1 == 0:
                a_up = upsample(global_best_a, ratio)
                if len(a_up) == n:
                    inits.append(normalize_seq(a_up, S))
                    names.append("upsampled")

        # Evaluate inits
        print("\n=== n=%d (budget %.0fs) ===" % (n, n_budget))
        best_init = None
        best_init_obj = np.inf
        for a_init, name in zip(inits, names):
            obj = compute_obj(a_init)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = a_init
            if obj < global_best_obj:
                global_best_obj = obj
                global_best_a = a_init.copy()
            if obj < 2.0:
                print("  %s: obj=%.6f" % (name, obj))

        print("  Best init: %s obj=%.6f" % (
            names[np.argmin([compute_obj(a) for a in inits])] if best_init is not None else "?",
            best_init_obj))

        # Run optimization strategies, each getting a fraction of the budget
        strategies = [
            ("LSE_FISTA", 0.45),
            ("LSE_high_beta", 0.25),
            ("two_lag", 0.15),
            ("spectral_polish", 0.15),
        ]

        current_best = best_init.copy() if best_init is not None else np.full(n, S / n)
        current_best_obj = best_init_obj

        for strat_name, frac in strategies:
            if time.monotonic() > deadline - 60:
                break
            strat_budget = n_budget * frac
            if strat_budget < 10:
                continue

            t0 = time.monotonic()
            print("  [%s] budget=%.0fs" % (strat_name, strat_budget))

            if strat_name == "LSE_FISTA":
                a_out, obj_out = run_optimized_lse(
                    current_best, strat_budget, rng,
                    beta_init=5.0, beta_max=50000.0,
                    use_spectral=True, spectral_interval=3000)

            elif strat_name == "LSE_high_beta":
                a_out, obj_out = run_optimized_lse(
                    current_best, strat_budget, rng,
                    beta_init=100.0, beta_max=200000.0,
                    use_spectral=False, spectral_interval=10**9)

            elif strat_name == "two_lag":
                a_out, obj_out = run_two_lag_balance(
                    current_best, strat_budget, rng)

            elif strat_name == "spectral_polish":
                # Iterative spectral shaping with FISTA between
                a_out = current_best.copy()
                S_val = np.sqrt(2.0 * n)
                it_sp = 0
                last_sp_improve = 0
                while time.monotonic() - t0 < strat_budget:
                    it_sp += 1
                    a_shaped = spectral_shaping(a_out, strength=0.15)
                    obj_sh = compute_obj(a_shaped)
                    if obj_sh < compute_obj(a_out) - 1e-10:
                        a_out = a_shaped
                        last_sp_improve = it_sp
                    else:
                        # Short FISTA burst
                        a_burst, obj_burst = run_optimized_lse(
                            a_out, 2.0, rng,
                            beta_init=50.0, beta_max=1000.0,
                            use_spectral=False)
                        if obj_burst < compute_obj(a_out):
                            a_out = a_burst
                            last_sp_improve = it_sp
                    if it_sp - last_sp_improve > 50:
                        break
                obj_out = compute_obj(a_out)

            elapsed = time.monotonic() - t0
            print("    %s: obj=%.6f (t=%.1fs)" % (strat_name, obj_out, elapsed))

            if obj_out < current_best_obj:
                current_best_obj = obj_out
                current_best = a_out.copy()
                if obj_out < global_best_obj:
                    global_best_obj = obj_out
                    global_best_a = a_out.copy()
                    print("    ** New global best: %.6f (n=%d) **" % (global_best_obj, n))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Target: 1.5027, Previous: 1.506856")
            if ver == 0:
                # evaluate_sequence returned 0? That's inf, use fallback
                print("WARNING: invalid result, using fallback")
                n_fb = 8192
                global_best_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
                final_list = [float(x) for x in global_best_a]
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        final_list = [float(x) for x in global_best_a]

    return final_list
