def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    A = fft(a, fsize)
    W = fft(w_full, fsize)
    # cross_correlation at lag i = sum_j a[j] * w_full[j+i]
    # = real(ifft(conj(A) * W))[i]  ... let me verify
    # ifft(conj(A)*W)[i] = (1/N) sum_k conj(A_k) W_k exp(2pi i k i/N)
    # A_k = sum_j a[j] exp(-2pi i j k/N)
    # conj(A_k) = sum_j a[j] exp(2pi i j k/N)
    # conj(A_k) * W_k = (sum_j a[j] exp(2pi i j k/N)) * (sum_m w[m] exp(-2pi i m k/N))
    # ifft of that at position i: (1/N) sum_k (sum_j a[j] exp(2pi i j k/N))(sum_m w[m] exp(-2pi i m k/N)) exp(2pi i k i/N)
    # = (1/N) sum_j a[j] sum_m w[m] sum_k exp(2pi i k (j-i)/N) exp(-2pi i m k/N)
    # = (1/N) sum_j a[j] sum_m w[m] N * delta(j-i-m)  [using sum_k exp(2pi i k p/N) = N delta(p mod N)]
    # Wait, I need to be more careful.
    # sum_k exp(2pi i k (j - i + m)/N)... no.
    # Let me redo: exp(2pi i j k/N) * exp(-2pi i m k/N) * exp(2pi i i k/N) = exp(2pi i k (j - m + i)/N)
    # Hmm that's not right either. Let me use a different convention.
    
    # You know what, let me just compute it directly for correctness:
    # result[i] = 2 * sum_{j=0}^{n-1} a[j] * w_full[j+i] (where j+i < len(w_full))
    # This is O(n * n_active) if we do it naively, but we can use the fact that w_full is sparse.
    
    # Since w_full is sparse (only n_active non-zero entries), we can compute:
    # result[i] = 2 * sum_{k in active} w_k * a[k-i] (for valid i)
    # = 2 * sum_{k in active} w_k * (a shifted by k, reversed)
    
    # For efficiency with FFT, let's build the full gradient:
    g = np.zeros(n)
    for w, k in zip(weights, idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * w * a[k - np.arange(lo, hi + 1)]
    
    g = g - g.sum() / n
    return g


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    smax = scaled.max()
    if smax > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled - smax)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_maxconv(a):
    """Compute max of conv(a,a), the key quantity to minimize."""
    c = fft_convolve_self(a)
    return c.max()


def backtracking_step(a, grad, S, c_current, max_conv, max_iters=20):
    """
    Backtracking line search for step size.
    Returns (new_a, new_max_conv, step_size).
    """
    n = len(a)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy(), max_conv, 0.0
    
    # Armijo condition: f(a - eta*g) <= f(a) - c1 * eta * ||g||^2
    # But since we're minimizing max_conv and using a subgradient, we use:
    # f(a_new) <= f(a) - c1 * eta * (grad^T * (a - a_new))
    # For step a_new = a - eta*grad (projected), this becomes:
    # f(a_new) <= f(a) - c1 * eta * ||grad_proj||^2
    
    c1 = 1e-4
    rho = 0.5
    eta = 1.0
    best_a = a.copy()
    best_val = max_conv
    
    for _ in range(max_iters):
        a_trial = project_simplex(a - eta * grad, S)
        val = objective_maxconv(a_trial)
        if val < best_val:
            best_val = val
            best_a = a_trial.copy()
        # Check Armijo
        improvement = max_conv - val
        required = c1 * eta * gnorm * gnorm
        if improvement >= required or val <= max_conv * (1 - 1e-8):
            return a_trial, val, eta
        eta *= rho
        if eta < 1e-12:
            break
    
    return best_a, best_val, eta


def run_active_set_opt(a_init, S, max_time, rng, 
                       initial_n_active=16, max_n_active=64,
                       use_backtrack=True, perturb_every=5000):
    """
    Active-set subgradient optimization with backtracking.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        # Adaptive number of active lags
        n_active = min(initial_n_active + int((max_n_active - initial_n_active) * progress), n // 8)
        
        c = fft_convolve_self(a)
        curr_max = c.max()
        if curr_max < best:
            best = curr_max
            best_a = a.copy()
            last_improve = it
        
        # Stuck handling
        if it - last_improve > perturb_every:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            eta = 1.0
            last_improve = it
            continue
        
        # Compute subgradient
        grad = compute_subgradient(a, c, n_active, weighted=True)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        if use_backtrack:
            a, curr_max, eta_used = backtracking_step(a, grad, S, c, curr_max, max_iters=10)
            # Decay eta slightly
            eta = max(eta_used * 0.9, 1e-6)
        else:
            # Fixed step with decay
            eta = S / (n * (1.0 + 0.1 * it) * max(gnorm / (S / n), 1e-10))
            a = project_simplex(a - eta * grad, S)
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_high_beta_fista(a_init, S, max_time, rng, beta_init=100.0, beta_max=100000.0):
    """High-beta LSE FISTA for final polishing."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.002 * it))
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        if it - last_improve > 5000:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.5 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with active-set optimization."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    opt_time = max(15.0, max_time / 20.0)
    
    while time.monotonic() - start < max_time - opt_time:
        progress = (time.monotonic() - start) / max_time
        base_scale = 0.3 * np.exp(-3.0 * progress) + 0.03
        scale = base_scale * (1.0 + 0.3 * rng.random())
        
        a = fourier_mutate(a, rng, scale)
        
        t_alloc = min(opt_time, max_time - 60 - (time.monotonic() - start))
        if t_alloc < 10:
            break
        
        a_opt, val = run_active_set_opt(a, S, t_alloc, rng,
                                         initial_n_active=8, max_n_active=48,
                                         use_backtrack=True, perturb_every=3000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH: new best max_b=%.8f" % val)
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))
    
    # Various windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25) + 0.01)
    
    # Asymmetric (important: we want non-symmetric for good results)
    inits.append(np.maximum(1.0 + 0.5 * np.sin(4 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(6 * np.pi * tt + 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.5 * tt + 0.2 * np.sin(8 * np.pi * tt), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(1.0, 0.5, n)) + 0.05)
    
    # Block structures
    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)
    
    # Smooth random (low-pass filtered)
    noise = rng.normal(1.0, 0.3, n)
    # Simple low-pass: moving average
    kernel_len = max(3, n // 128)
    kernel = np.ones(kernel_len) / kernel_len
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    
    return inits


def run_lp_direction(a, c, n_active=10):
    """
    Try to find a better direction using a small LP.
    Minimize max_{k in A} grad(b_k)^T d subject to sum(d)=0, d >= -a.
    Approximate with a simplex-like approach.
    """
    n = len(a)
    n_active = min(n_active, len(c))
    idx = np.argsort(c)[::-1][:n_active]
    
    # Compute gradients for active lags
    grads = np.zeros((n_active, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grads[i, lo:hi+1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    
    # For each grad, project to sum-zero
    for i in range(n_active):
        grads[i] -= grads[i].mean()
    
    # We want d that minimizes max_i grads[i] @ d, subject to sum(d)=0, d >= -a
    # This is a LP. For simplicity, use a heuristic:
    # Find the point in the convex hull of {-grads[i]} that is most "negative"
    # in the direction of a (i.e., where a is supported).
    
    # Simple approach: use the grad that gives the most negative dot product with a
    scores = np.array([np.dot(grads[i], a) for i in range(n_active)])
    best_idx = np.argmin(scores)
    
    # Use this as the direction, but blend with the average for robustness
    avg_grad = grads.mean(axis=0)
    direction = 0.7 * grads[best_idx] + 0.3 * avg_grad
    direction -= direction.mean()
    
    return direction


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    global_n = 0
    
    def update_global(a, val):
        nonlocal global_best, global_a, global_n
        n = len(a)
        s = a.sum()
        if s < 0.01:
            return False
        obj = 2.0 * n * val / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = n
            return True
        return False
    
    # Try multiple sequence lengths
    lengths = [8192, 16384]
    
    # Phase 1: Quick screening (~200s)
    print("\n=== Phase 1: Quick screening ===")
    p1_budget = 200.0
    p1_start = time.monotonic()
    
    for n in lengths:
        if time.monotonic() - p1_start > p1_budget:
            break
        S = np.sqrt(2.0 * n)
        
        # Get initializations
        if n == 8192 and given is not None:
            inits = generate_inits(n, rng, given_seq=given)
        else:
            inits = generate_inits(n, rng)
        
        # Quick screen top 4 initializations
        n_screen = min(4, len(inits))
        per_time = (p1_budget - (time.monotonic() - p1_start)) / max(1, (2 - lengths.index(n)) * n_screen)
        per_time = max(20.0, min(60.0, per_time))
        
        for i in range(n_screen):
            if time.monotonic() > deadline - 1200:
                break
            t_alloc = min(per_time, deadline - time.monotonic() - 600)
            if t_alloc < 15:
                break
            
            a_opt, val = run_active_set_opt(inits[i], S, t_alloc, rng,
                                             initial_n_active=8, max_n_active=32,
                                             use_backtrack=True, perturb_every=2000)
            obj = 2.0 * n * val / (a_opt.sum() ** 2)
            print("  Screen n=%d init=%d: obj=%.6f" % (n, i, obj))
            update_global(a_opt, val)
    
    print("  Best after screening: obj=%.6f (n=%d)" % (global_best, global_n))
    
    # Phase 2: Basin hopping (~600s)
    print("\n=== Phase 2: Basin hopping ===")
    p2_budget = 600.0
    p2_start = time.monotonic()
    
    if global_a is not None:
        S = global_a.sum()
        n = len(global_a)
        
        # Run basin hopping
        n_bh = 3
        per_bh = p2_budget / n_bh
        for i in range(n_bh):
            if time.monotonic() > deadline - 600:
                break
            t_alloc = min(per_bh, deadline - time.monotonic() - 300)
            if t_alloc < 60:
                break
            print("  BH run %d/%d, alloc=%.0fs" % (i+1, n_bh, t_alloc))
            a_bh, val_bh = run_basin_hopping(global_a, S, t_alloc, rng)
            update_global(a_bh, val_bh)
            print("  After BH %d: obj=%.6f" % (i+1, global_best))
    
    # Phase 3: High-beta FISTA polish (~500s)
    print("\n=== Phase 3: High-beta FISTA polish ===")
    p3_budget = 500.0
    p3_start = time.monotonic()
    
    if global_a is not None:
        S = global_a.sum()
        n_sub = 3
        sub_budget = p3_budget / n_sub
        for sub in range(n_sub):
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(sub_budget, deadline - time.monotonic() - 100)
            if t_alloc < 30:
                break
            beta_init = 200.0 * (sub + 1)
            print("  Sub-phase %d/%d, beta_init=%.0f, t=%.0fs" % (sub+1, n_sub, beta_init, t_alloc))
            a_p, val_p = run_high_beta_fista(global_a, S, t_alloc, rng,
                                              beta_init=beta_init,
                                              beta_max=200000.0)
            update_global(a_p, val_p)
            print("  After sub-phase %d: obj=%.6f" % (sub+1, global_best))
    
    # Phase 4: Active-set fine polish with backtracking (~300s)
    print("\n=== Phase 4: Fine polish ===")
    p4_budget = 350.0
    p4_start = time.monotonic()
    
    if global_a is not None:
        S = global_a.sum()
        n_runs = 4
        per_run = p4_budget / n_runs
        for i in range(n_runs):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(per_run, deadline - time.monotonic() - 10)
            if t_alloc < 20:
                break
            # Use different parameters for each run
            n_active_start = 4 + 4 * i
            n_active_end = 16 + 8 * i
            print("  Fine polish %d/%d, n_active=%d..%d, t=%.0fs" % (i+1, n_runs, n_active_start, n_active_end, t_alloc))
            a_f, val_f = run_active_set_opt(global_a, S, t_alloc, rng,
                                             initial_n_active=n_active_start,
                                             max_n_active=n_active_end,
                                             use_backtrack=True,
                                             perturb_every=10000)
            update_global(a_f, val_f)
            print("  After fine polish %d: obj=%.6f" % (i+1, global_best))
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(len(global_a)) * np.sqrt(2.0 * len(global_a)) / len(global_a)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n_fb = 8192
            global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback obj=%.6f" % ver)
    else:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
