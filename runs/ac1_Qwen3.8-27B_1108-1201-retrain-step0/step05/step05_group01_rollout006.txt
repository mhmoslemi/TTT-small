"""
Minimize 2*n*max_conv/sum^2 using LP-based iterative refinement (solving
max L1 given convolution bound) combined with FISTA and L-BFGS-B polishing.
The LP provides strong global descent directions that complement local gradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize as scipy_minimize
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def run_fista(a_init, max_time, rng, beta_start=20.0, step_scale=1.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        beta = min(beta_start + 0.01 * it, 5000.0)
        step = step_scale / max(beta, 1.0)

        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

        # Noise when stuck
        if it - last_improve > 3000 and it % 500 == 0:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a


def build_conv_matrix_sparse(f, n):
    """Build sparse convolution matrix A where (A@g)(k) = (f*g)(k)."""
    # A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # For diagonal offset d = k-j (d in [0, n-1]):
    #   (k,j) = (d+j, j) for j in [0, min(n-1, 2n-2-d)]
    
    all_rows = []
    all_cols = []
    all_vals = []
    
    for d in range(n):
        j_max = min(n - 1, 2 * n - 2 - d)
        j = np.arange(j_max + 1)
        all_rows.append(d + j)
        all_cols.append(j)
        all_vals.append(np.full(j_max + 1, f[d]))
    
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    vals = np.concatenate(all_vals)
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(f, M, max_iter_lp=5000):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)(k) <= M for all k.
    Returns the optimal g, or None if infeasible/failed.
    """
    n = len(f)
    
    A = build_conv_matrix_sparse(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'maxiter': max_iter_lp, 'presolve': True})
    
    if result.success and result.x is not None:
        return np.maximum(result.x, 0)
    return None


def lp_iteration(f, max_lp_time, rng):
    """
    One LP-based iteration: solve LP, update f.
    Returns updated f and the new M.
    """
    n = len(f)
    S = np.sqrt(2.0 * n)
    f = f * (S / f.sum())
    
    c = fft_convolve_self(f)
    M = c.max()
    
    # Solve LP
    t0 = time.monotonic()
    g = lp_step(f, M)
    lp_time = time.monotonic() - t0
    
    if g is None or g.sum() < 1e-10:
        return f, M
    
    # Normalize g
    g = g * (S / g.sum())
    
    # Check the self-convolution of g
    c_g = fft_convolve_self(g)
    M_g = c_g.max()
    
    # Use convex combination with line search
    # Try t=1 first (full replacement)
    best_t = 0.0
    best_M = M
    for t_val in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        f_new = (1 - t_val) * f + t_val * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new * (S / s)
        c_new = fft_convolve_self(f_new)
        M_new = c_new.max()
        if M_new < best_M:
            best_M = M_new
            best_t = t_val
            if M_new <= M * (1 - 1e-6):
                break
    
    if best_t == 0.0:
        # Try even smaller steps
        for t_val in [0.02, 0.01, 0.005]:
            f_new = (1 - t_val) * f + t_val * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new * (S / s)
            c_new = fft_convolve_self(f_new)
            M_new = c_new.max()
            if M_new < best_M:
                best_M = M_new
                best_t = t_val
    
    if best_t > 0:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new * (S / f_new.sum())
    else:
        f_new = f
    
    return f_new, best_M


def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    betas = [50.0, 200.0, 500.0]
    
    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break
        
        c = fft_convolve_self(a)
        cm = c.max()
        pen = 10.0
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(max(e.sum(), 1e-300)) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad
        
        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 50, 'maxls': 20, 'ftol': 1e-14,
                     'gtol': 1e-10, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj
