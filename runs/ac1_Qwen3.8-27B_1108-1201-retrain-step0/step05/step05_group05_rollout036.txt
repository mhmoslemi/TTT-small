5. Multi-n search (8192, 16384) with cross-scale transfer
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    max_s = scaled.max()
    if max_s > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient_active(a, c, tol_frac=1e-6, max_active=50):
    """
    Exact subgradient using active set (lags within tol of max).
    Returns average subgradient over active lags.
    """
    n = len(a)
    cmax = c.max()
    tol = tol_frac * cmax
    active = np.where(c >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    active = active[:max_active]
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    grad = np.zeros(n)
    for k in active:
        w_full = np.zeros(len(c))
        w_full[k] = 1.0
        W = fft(w_full, fsize)
        grad += np.real(ifft(W * np.conj(A)))[:n]
    grad *= 2.0 / len(active)
    return grad


def active_set_polish(a_init, S, max_time, rng, tol_frac=1e-7, max_active=30):
    """
    Polish using exact active-set subgradient with adaptive step size.
    Uses backtracking line search on the true objective.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    step = 0.5 * S / n
    stalled = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled > 5000:
            # Perturb to escape
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            step = 0.5 * S / n
            stalled = 0
            continue

        grad = exact_subgradient_active(a, c, tol_frac=tol_frac, max_active=max_active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step: decrease if stuck, use projected gradient
        trial_step = step
        improvement = False
        for _ in range(5):
            a_trial = project_simplex(a - trial_step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c.max() * (1 - 1e-10):
                a = a_trial
                step = min(step * 1.05, 2.0 * S / n)
                improvement = True
                break
            trial_step *= 0.5

        if not improvement:
            step *= 0.7
            if step < S / (n * 1e6):
                # Try mirror-like update
                g_max = np.max(np.abs(grad))
                if g_max > 1e-30:
                    a_new = a * np.exp(-0.01 * grad / g_max)
                    a = project_simplex(a_new, S)
                    step = 0.5 * S / n

    return best_a, best


def run_fista_v2(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.003,
                 beta_max=100000.0, perturbation_period=4000):
    """
    Improved FISTA with faster beta ramp and better stuck detection.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.5 * beta)

        if it - last_improve > 3000:
            if it % 500 == 0:
                # Fourier phase perturbation
                fsize = 1 << (2 * n - 1).bit_length()
                A = fft(y, fsize)
                mag = np.abs(A)
                phase = np.angle(A)
                scale = 0.05 * np.exp(-(it - last_improve) / 50000.0)
                phase_noise = rng.normal(0, scale, fsize)
                new_A = mag * np.exp(1j * (phase + phase_noise))
                new_A[0] = A[0]
                y = project_simplex(np.real(ifft(new_A, fsize))[:n], S)
            else:
                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (20.0 * perturbation_period))
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    return best_a, best


def basin_hopping_v2(a_init, S, max_time, rng):
    """
    Basin hopping with adaptive mutation and population diversity.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    n_iters = max(1, int((max_time - 30) / (opt_time + 5)))

    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time - 10:
            break
        progress = it / max(1, n_iters)

        # Adaptive mutation: large early, very small late
        base_scale = 0.3 * np.exp(-3.0 * progress) + 0.01
        scale = base_scale * (0.5 + rng.random())

        # Choose mutation source
        r = rng.random()
        if len(population) >= 3 and r < 0.3:
            idx = rng.integers(len(population))
            a = population[idx][1].copy()
        elif len(population) >= 2 and r < 0.5:
            idx1 = rng.integers(len(population))
            idx2 = rng.integers(len(population))
            if idx1 != idx2:
                # Fourier crossover
                fsize = 1 << (2 * n - 1).bit_length()
                A1 = fft(population[idx1][1], fsize)
                A2 = fft(population[idx2][1], fsize)
                alpha = 0.4 + 0.2 * rng.random(fsize)
                new_A = alpha * A1 + (1 - alpha) * A2
                new_A[0] = A1[0]
                a = project_simplex(np.real(ifft(new_A, fsize))[:n], S)
            else:
                a = fourier_mutate(a, rng, scale)
        else:
            a = fourier_mutate(a, rng, scale)

        # Local optimization with increasing beta
        beta_init = 20 + 100 * min(it, 10)
        a_opt, val = run_fista_v2(a, S, opt_time, rng,
                                   beta_init=beta_init,
                                   beta_rate=0.003,
                                   beta_max=20000 + 10000 * min(it, 10),
                                   perturbation_period=6000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.10f" % (it, val))

        population.append((val, a_opt.copy()))
        if len(population) > 10:
            population.sort(key=lambda x: x[0])
            population = population[:10]

    return best_a, best


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 2.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def generate_bathtub_inits(n, rng):
    """Generate diverse bathtub-shaped initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Pure cosines (bathtub shapes)
    for alpha in [0.5, 0.7, 0.9, 1.0, 1.2, 1.5]:
        p = np.maximum(alpha * (1 + np.cos(2 * np.pi * tt)), 0.01)
        inits.append(p)

    # Power-law edge concentration
    for gamma in [0.5, 1.0, 2.0, 3.0]:
        edge_dist = np.minimum(tt, 1.0 - tt)  # distance to nearest edge
        p = np.maximum(edge_dist ** gamma, 0.01)
        inits.append(p)

    # Gaussian at edges
    for sigma in [0.1, 0.2, 0.3]:
        p1 = np.exp(-0.5 * (tt / sigma) ** 2)
        p2 = np.exp(-0.5 * ((tt - 1.0) / sigma) ** 2)
        inits.append(np.maximum(p1 + p2, 0.01))

    # Multi-harmonic bathtubs
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * tt) + 0.2 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.8 * np.cos(2 * np.pi * tt) - 0.3 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.5 * np.cos(2 * np.pi * tt) + 0.5 * np.cos(6 * np.pi * tt) + 1.0, 0.01))

    # Sine-based (high at edges)
    inits.append(np.sin(np.pi * tt) ** 0.5 + 0.1)  # This is actually high in middle... 
    # Correct bathtub: 
    inits.append(1.0 / (np.sin(np.pi * tt) + 0.3))  # High at edges due to 1/sin

    # Random smooth (via low-pass filtered noise)
    for _ in range(3):
        freq = rng.integers(1, 5)
        phase_offset = rng.uniform(0, 2 * np.pi)
        p = 1.0 + 0.5 * np.cos(2 * np.pi * freq * tt + phase_offset)
        inits.append(np.maximum(p, 0.01))

    # Uniform (baseline)
    inits.append(np.ones(n))

    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def upsample_sequence(a, factor=2):
    """Upsample by inserting zeros between elements."""
    n = len(a)
    new_n = n * factor
    new_a = np.zeros(new_n)
    new_a[::factor] = a
    return new_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    
    Key insight: optimal solutions have bathtub shape (high at edges, low in middle).
    Uses multi-scale approach with precise active-set polishing.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" % (
                len(given), given.sum(), objective_value(given)))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_update(a, n):
        nonlocal global_best, global_a
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        obj = 2.0 * n * fft_convolve_self(a).max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  NEW GLOBAL BEST: obj=%.8f (n=%d)" % (obj, n))

    # Phase 1: Multi-n initialization and quick screening
    print("\n=== Phase 1: Screening multiple n values ===")
    p1_budget = min(120.0, (deadline - total_start) * 0.10)
    p1_deadline = total_start + p1_budget

    n_values = [8192]
    if given is not None and len(given) == 8192:
        n_values.append(16384)  # Upsample

    screening_results = {}  # n -> (best_a, best_val)

    for n in n_values:
        S = np.sqrt(2.0 * n)
        inits = generate_bathtub_inits(n, rng)

        if given is not None and len(given) == n:
            inits.insert(0, given.copy())
        elif given is not None and len(given) == n // 2:
            inits.insert(0, upsample_sequence(given, 2))

        # Quick screen top initializations
        n_screen = min(6, len(inits))
        per_init_time = p1_budget / (len(n_values) * n_screen)
        per_init_time = max(5.0, per_init_time)

        best_n_init = None
        best_n_val = np.inf

        for i in range(n_screen):
            if time.monotonic() > p1_deadline - 10:
                break
            t_alloc = min(per_init_time, p1_deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, val = run_fista_v2(inits[i], S, t_alloc, rng,
                                       beta_init=10.0, beta_rate=0.01, beta_max=5000)
            if val < best_n_val:
                best_n_val = val
                best_n_init = a_opt.copy()
            try_update(a_opt, n)

        screening_results[n] = (best_n_init, best_n_val)
        print("  n=%d: best max_b=%.8f, obj=%.6f" % (n, best_n_val, 2.0 * n * best_n_val / S**2))

    # Phase 2: Basin hopping from best candidates
    print("\n=== Phase 2: Basin hopping ===")
    p2_deadline = deadline - 200  # Reserve 200s for final polishing
    p2_budget = min(600.0, p2_deadline - time.monotonic())

    # Run basin hopping on the best candidate
    best_n = max(screening_results.keys(), key=lambda x: screening_results[x][1])
    # Actually pick the one with best objective
    best_n = min(screening_results.keys(),
                 key=lambda x: 2.0 * x * screening_results[x][1] / (np.sqrt(2.0 * x))**2)
    best_a_for_bh = screening_results[best_n][0]
    best_n_val = screening_results[best_n][1]
    n_bh = best_n
    S_bh = np.sqrt(2.0 * n_bh)

    n_bh_runs = 2
    bh_time = p2_budget / n_bh_runs
    for bh_i in range(n_bh_runs):
        if time.monotonic() > p2_deadline - 30:
            break
        t_alloc = min(bh_time, p2_deadline - time.monotonic() - 10)
        if t_alloc < 30:
            break
        print("  BH run %d/%d: n=%d, alloc=%.0fs" % (bh_i + 1, n_bh_runs, n_bh, t_alloc))
        a_bh, val_bh = basin_hopping_v2(best_a_for_bh, S_bh, t_alloc, rng)
        try_update(a_bh, n_bh)
        if val_bh < best_n_val:
            best_n_val = val_bh
            best_a_for_bh = a_bh.copy()

    # Phase 3: High-beta FISTA polish
    print("\n=== Phase 3: High-beta FISTA polish ===")
    p3_deadline = deadline - 80
    p3_budget = min(400.0, p3_deadline - time.monotonic())

    if global_a is not None and p3_budget > 30:
        n_pol = len(global_a)
        S_pol = np.sqrt(2.0 * n_pol)
        n_sub = 3
        sub_time = p3_budget / n_sub
        for sub in range(n_sub):
            if time.monotonic() > p3_deadline - 5:
                break
            t_alloc = min(sub_time, p3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            beta_init = 100.0 * (sub + 1)

    print("\n=== Phase 4: Active-set exact polish ===")

        tol_values = [1e-5, 1e-6, 1e-7, 1e-8]
        per_tol_time = p4_budget / len(tol_values)
        for tol in tol_values:
            if time.monotonic() > deadline - 3:
                break
            t_alloc = min(per_tol_time, deadline - time.monotonic() - 3)
            if t_alloc < 5:
                break
            a_fin, val_fin = active_set_polish(global_a, S_fin, t_alloc, rng,
                                                tol_frac=tol, max_active=20)
            try_update(a_fin, n_fin)
            print("  Active-set tol=%.1e: obj=%.8f" % (tol, global_best))

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            fallback_n = 8192

            fallback_n = 8192
            global_a = np.ones(fallback_n) * np.sqrt(2.0 * fallback_n) / fallback_n
            final_list = [float(x) for x in global_a]
    else:
        fallback_n = 8192
        global_a = np.ones(fallback_n) * np.sqrt(2.0 * fallback_n) / fallback_n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
