"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:

2. Multi-method optimization: FISTA, Mirror Descent, Subgradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient(a, c):
    """Compute the exact subgradient at the argmax of the convolution."""
    n = len(a)
    k_star = np.argmax(c)
    # Gradient of c_k w.r.t. a_j is a_{k-j} + a_{j-k} (with 0 padding)
    # This is equivalent to 2 * cross_correlate(delta_k, a)
    # Using FFT: grad = 2 * ifft(fft(w) * conj(fft(a))) where w has 1 at position k_star
    w_full = np.zeros(2 * n - 1)
    w_full[k_star] = 1.0
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.003,
              beta_max=5000.0, k_frac=0.05, use_momentum=True):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    k = max(2, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        if stuck > 1000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        if it % 2000 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.05, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_mirror_descent(a_init, S, max_time, rng, eta_init=None, k_frac=0.05):
    """Mirror descent with entropy geometry (multiplicative weight updates)."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    if eta_init is None:
        eta_init = 1.0  # Will be adapted
    
    k = max(2, int(k_frac * (2 * n - 1)))
    it = 0
    beta = 50.0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Adaptive beta
        beta = min(5000.0, 50.0 * (1.0 + it / 500.0))
        
        grad = topk_lse_gradient(a, c, k, beta)
        
        # Multiplicative update: a_j *= exp(-eta * g_j)
        # Scale gradient to have mean 0 (for stability)
        g_mean = grad.mean()
        g_centered = grad - g_mean
        g_range = g_centered.max() - g_centered.min()
        if g_range < 1e-15:
            break
        
        # Adaptive eta: scale so that the effective step is reasonable
        eta = eta_init if it < 10 else eta_init * max(0.1, 100.0 / (100.0 + it))

def run_adam(a_init, S, max_time, rng, lr_init=0.01, beta_adam=50.0, beta_rate=0.005):
    """Adam optimizer with LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    k = max(2, int(0.04 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        beta = min(5000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.8 * progress)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)
        
        if t % 5000 == 0 and rng.random() < 0.1:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

    return best_a, best

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Multi-length exploration starting from parent sequence (4096, 8192, 16384)
2. Multi-method optimization: FISTA, Mirror Descent, Adam, Subgradient
3. Adaptive LSE smoothing with multiple scales
4. Extensive final polishing with high-beta methods
5. Random restarts with spectral perturbation to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient(a, c):
    """Compute the exact subgradient at the argmax of the convolution."""
    n = len(a)
    k_star = int(np.argmax(c))
    w_full = np.zeros(2 * n - 1)
    w_full[k_star] = 1.0
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten(a, alpha, rng):
    """One step of spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length via interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

            best_a = y.copy()
            stuck = 0
        else:

def run_mirror_descent(a_init, S, max_time, rng, eta_init=1.0, k_frac=0.04):

        g_centered = grad - g_mean
        g_range = g_centered.max() - g_centered.min()
        if g_range < 1e-15:
            break

        # Adaptive eta that decays over time
        progress = min(1.0, (time.monotonic() - start) / max_time)
        eta = eta_init * max(0.1, 1.0 - 0.7 * progress)

        log_update = -eta * g_centered / (g_range + 1e-15)
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a = normalize(a_new, S)

        # Occasional spectral perturbation to escape
        if it % 5000 == 0 and rng.random() < 0.08 and progress < 0.6:
            a = spectral_perturb(a, 0.03, rng)

        # Occasional spectral flattening
        if it % 10000 == 0 and rng.random() < 0.05 and progress < 0.5:
            a = spectral_flatten(a, 0.03, rng)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_adam(a_init, S, max_time, rng, lr_init=0.01, beta_adam=50.0, beta_rate=0.005):
    """Adam optimizer with LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    k = max(2, int(0.04 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.8 * progress)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        if t % 5000 == 0 and rng.random() < 0.08:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

    return best_a, best


def run_polish_high_beta(a_init, S, max_time, rng):
    """Very high beta FISTA for final polishing with careful step control."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    k = max(1, int(0.01 * (2 * n - 1)))
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(100000.0, 1000.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        base_step = 0.3 / (1.0 + 0.0005 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.01, step_scale * 0.3)
            a = project_simplex(a - step * grad, S)

        progress = (time.monotonic() - start) / max_time
        if it % 5000 == 0 and progress < 0.4:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_subgradient(a_init, S, max_time, rng):
    """Subgradient method using exact max with adaptive step size."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    base_step = S / np.sqrt(n)
    improve_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            improve_count += 1

        grad = exact_subgradient(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decaying step size
        step = base_step * 10.0 / (10.0 + 0.01 * it)
        a_new = project_simplex(a - step * grad, S)

        # Check if step helped
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            step = step * 0.1
            a_new = project_simplex(a - step * grad, S)
            a = a_new

        # Random restart when stuck
        if it % 3000 == 0 and improve_count < 3:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            improve_count = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-length, multi-method optimization with extensive polishing.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    global_best = np.inf
    global_a = None

    # Get starting sequence from parent
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            given_sum = given.sum()
            print("Using given sequence: len=%d, sum=%.2f" % (len(given), given_sum))
    except Exception:
        pass

    def update_global(a_cand, obj_cand, method_name, n):
        nonlocal global_best, global_a
        if obj_cand < global_best:
            global_best = obj_cand
            global_a = a_cand.copy()
            print("  [%.1fs] %s (n=%d): obj=%.6f *** NEW GLOBAL BEST ***" %
                  (time.monotonic() - start_total, method_name, n, obj_cand))

    def try_length(n, a_start, time_limit, phase_desc):
        """Run full optimization pipeline for a given length."""
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
            a_start = normalize(a_start, S)
        elif a_start is None:
            a_start = np.ones(n) * S / n
        else:
            a_start = normalize(a_start.copy(), S)

        print("\n  --- %s: n=%d, budget=%.0fs ---" % (phase_desc, n, time_limit))

        best_local = fft_convolve_self(a_start).max()
        best_a_local = a_start.copy()
        obj_local = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Initial: obj=%.6f" % obj_local)

        def evaluate_update(a_cand, val_cand, method_name):
            nonlocal best_local, best_a_local
            obj = 2.0 * n * val_cand / (a_cand.sum() ** 2)
            if val_cand < best_local:
                best_local = val_cand
                best_a_local = a_cand.copy()
                print("    %s: obj=%.6f" % (method_name, obj))
                update_global(a_cand, obj, method_name, n)

        evaluate_update(best_a_local, best_local, "init")

        # Phase A: FISTA exploration (35% of time)
        remain = deadline - time.monotonic()
        t_a = min(time_limit * 0.35, remain * 0.4)
        if t_a > 30:
            a_opt, val_opt = run_fista(best_a_local, S, t_a, rng,
                                       beta_init=15.0, beta_rate=0.004,
                                       beta_max=3000.0, k_frac=0.06,
                                       use_momentum=True)
            evaluate_update(a_opt, val_opt, "FISTA-A")

        # Phase B: Adam optimizer (25% of time)
        remain = deadline - time.monotonic()
        t_b = min(time_limit * 0.25, remain * 0.3)
        if t_b > 20:
            a_opt, val_opt = run_adam(best_a_local, S, t_b, rng,
                                      lr_init=0.01, beta_adam=40.0,
                                      beta_rate=0.006)
            evaluate_update(a_opt, val_opt, "Adam")

        # Phase C: Mirror descent (20% of time)
        remain = deadline - time.monotonic()
        t_c = min(time_limit * 0.20, remain * 0.3)
        if t_c > 20:
            a_opt, val_opt = run_mirror_descent(best_a_local, S, t_c, rng,
                                                eta_init=1.0, k_frac=0.04)
            evaluate_update(a_opt, val_opt, "MirrorDescent")

        # Phase D: High-beta polish (15% of time)
        remain = deadline - time.monotonic()
        t_d = min(time_limit * 0.15, remain * 0.3)
        if t_d > 15:
            a_opt, val_opt = run_polish_high_beta(best_a_local, S, t_d, rng)
            evaluate_update(a_opt, val_opt, "PolishHighBeta")

        # Phase E: Subgradient exact (5% of time)
        remain = deadline - time.monotonic()
        t_e = min(time_limit * 0.05, remain * 0.3)
        if t_e > 10:
            a_opt, val_opt = run_subgradient(best_a_local, S, t_e, rng)
            evaluate_update(a_opt, val_opt, "Subgradient")

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Best for n=%d: obj=%.6f" % (n, obj_final))
        return best_a_local, best_local

    # === Phase 1: Multi-length exploration ===
    # Try key lengths, starting from parent sequence
    lengths_to_try = [4096, 8192, 16384]

    for i, n in enumerate(lengths_to_try):
        remain = deadline - time.monotonic()
        if remain < 90:
            print("  Not enough time remaining for n=%d" % n)
            break

        # Allocate more time to the primary length (8192)
        if n == 8192:
            t_alloc = min(remain * 0.35, 600.0)
        else:
            t_alloc = min(remain * 0.20, 300.0)

        a_start = resample_sequence(given, n) if given is not None else None
        try_length(n, a_start, t_alloc, "Explore")

    # === Phase 2: Try random initialization for diversity ===
    remain = deadline - time.monotonic()
    if remain > 200 and global_a is not None:
        n_rand = 8192
        S_rand = np.sqrt(2.0 * n_rand)
        # Create a random initialization with some structure
        a_rand = np.ones(n_rand) * S_rand / n_rand
        # Add structured noise: sinusoidal modulation
        t = np.linspace(0, 2 * np.pi, n_rand)
        a_rand = a_rand * (1.0 + 0.3 * np.sin(3 * t + rng.uniform(0, 2*np.pi)))
        a_rand = normalize(a_rand, S_rand)
        t_alloc = min(remain * 0.20, 300.0)
        try_length(n_rand, a_rand, t_alloc, "RandomInit")

    # === Phase 3: Focused optimization on best found ===
    remain = deadline - time.monotonic()
    if remain > 100 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t_focus = min(remain * 0.35, 600.0)
        print("\n=== Focused optimization on n=%d, budget=%.0fs ===" % (n_best, t_focus))

        # Multiple rounds of different methods
        a_cur = global_a.copy()
        S_cur = S_best

        # Round 1: FISTA with high beta
        r1 = t_focus * 0.30
        if r1 > 20:
            a_opt, val_opt = run_fista(a_cur, S_cur, r1, rng,
                                       beta_init=100.0, beta_rate=0.002,
                                       beta_max=10000.0, k_frac=0.03,
                                       use_momentum=True)
            obj_opt = 2.0 * n_best * val_opt / (a_opt.sum() ** 2)
            if obj_opt < global_best:
                a_cur = a_opt.copy()
                global_best = obj_opt
                global_a = a_opt.copy()
                print("  Focus FISTA: obj=%.6f" % obj_opt)

        # Round 2: Mirror descent
        remain = deadline - time.monotonic()
        r2 = min(t_focus * 0.25, remain * 0.3)
        if r2 > 15:
            a_opt, val_opt = run_mirror_descent(a_cur, S_cur, r2, rng,
                                                eta_init=0.8, k_frac=0.03)
            obj_opt = 2.0 * n_best * val_opt / (a_opt.sum() ** 2)
            if obj_opt < global_best:
                a_cur = a_opt.copy()
                global_best = obj_opt
                global_a = a_opt.copy()
                print("  Focus Mirror: obj=%.6f" % obj_opt)

        # Round 3: High-beta polish
        remain = deadline - time.monotonic()
        r3 = min(t_focus * 0.30, remain * 0.3)
        if r3 > 15:
            a_opt, val_opt = run_polish_high_beta(a_cur, S_cur, r3, rng)
            obj_opt = 2.0 * n_best * val_opt / (a_opt.sum() ** 2)
            if obj_opt < global_best:
                a_cur = a_opt.copy()
                global_best = obj_opt
                global_a = a_opt.copy()
                print("  Focus Polish: obj=%.6f" % obj_opt)

        # Round 4: Adam fine-tuning
        remain = deadline - time.monotonic()
        r4 = min(t_focus * 0.15, remain * 0.3)
        if r4 > 10:
            a_opt, val_opt = run_adam(a_cur, S_cur, r4, rng,
                                      lr_init=0.005, beta_adam=200.0,
                                      beta_rate=0.003)
            obj_opt = 2.0 * n_best * val_opt / (a_opt.sum() ** 2)
            if obj_opt < global_best:
                a_cur = a_opt.copy()
                global_best = obj_opt
                global_a = a_opt.copy()
                print("  Focus Adam: obj=%.6f" % obj_opt)

    # === Phase 4: Final intensive polishing ===
    remain = deadline - time.monotonic()
    if remain > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs remaining) ===" % remain)

        # Specialized high-beta FISTA
        t_fin = remain * 0.35
        if t_fin > 20:
            a_fin, val_fin = run_polish_high_beta(global_a, S_best, t_fin, rng)
            obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  Final HighBeta: obj=%.6f" % global_best)

        # Very high beta FISTA (nearly exact max)
        remain = deadline - time.monotonic() - 10
        t_fin2 = min(remain * 0.35, remain * 0.4)
        if t_fin2 > 15:
            a_fin2, val_fin2 = run_fista(global_a, S_best, t_fin2, rng,
                                         beta_init=5000.0, beta_rate=0.001,
                                         beta_max=200000.0, k_frac=0.005,
                                         use_momentum=False)
            obj_fin2 = 2.0 * n_best * val_fin2 / (a_fin2.sum() ** 2)
            if obj_fin2 < global_best:
                global_best = obj_fin2
                global_a = a_fin2.copy()
                print("  Final UltraBeta: obj=%.6f" % global_best)

        # One last mirror descent pass
        remain = deadline - time.monotonic() - 10
        t_fin3 = min(remain * 0.30, remain * 0.5)
        if t_fin3 > 10:
            a_fin3, val_fin3 = run_mirror_descent(global_a, S_best, t_fin3, rng,
                                                  eta_init=0.3, k_frac=0.02)
            obj_fin3 = 2.0 * n_best * val_fin3 / (a_fin3.sum() ** 2)
            if obj_fin3 < global_best:
                global_best = obj_fin3
                global_a = a_fin3.copy()
                print("  Final Mirror: obj=%.6f" % global_best)

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            print("  Sum too small, using fallback!")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        print("  No valid sequence found, using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    total_time = time.monotonic() - start_total
    print("\nDone in %.1fs. Final: len=%d, obj=%.6f" % (total_time, len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
