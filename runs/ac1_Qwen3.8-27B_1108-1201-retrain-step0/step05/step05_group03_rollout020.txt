3. Take a step: a_new = (1-t) * a + t * v
4. Choose t (e.g., t = 2/(iter+2) or line search)

This is attractive because:
- Each iteration is O(n) to compute the gradient (using FFT) and O(n) to find the argmin
- It naturally maintains feasibility (a >= 0, sum = S)
- It's provably convergent

The downside is that it converges slowly (O(1/iter) rate). But combined with other methods, it could help escape local minima.

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Combined approach using:
1. Top-k LSE smoothed gradient with adaptive beta
2. Frank-Wolfe (conditional gradient) for exploration
3. Frequency-domain spectral equalization
4. Simulated annealing for escaping local minima
5. Multi-phase: coarse -> fine -> polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, idx, w


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def frank_wolfe_step(a, grad, S, t):
    """One Frank-Wolfe step: move towards the vertex that minimizes grad^T v."""
    n = len(a)
    j_star = np.argmin(grad)
    v = np.zeros(n)
    v[j_star] = S
    return (1 - t) * a + t * v


def spectral_equalization(a, alpha, rng):
    """Equalize the spectrum to reduce peaks in the convolution."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    
    if mean_p < 1e-30:
        return a
    
    # Compress the dynamic range
    ratio = power / mean_p
    new_power = power * (1.0 - alpha) + mean_p * alpha
    
    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)

    """Combined optimization: gradient + Frank-Wolfe + spectral + SA."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    # Parameters
    k_frac = 0.05
    k = max(4, int(k_frac * (2 * n - 1)))
    beta = 50.0
    beta_growth = 1.002
    lr = 0.05
    sa_temperature = 0.01
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        # Adapt parameters based on progress
        beta = min(10000.0, 50.0 * (1.0 + 0.005 * it))
        lr = 0.05 * max(0.01, 1.0 - progress)
        sa_temperature = 0.01 * max(0.01, 1.0 - progress)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        grad, _, _ = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Main gradient step
        step = lr * S / gnorm
        a_grad = project_simplex(a - step * grad, S)
        
        # Frank-Wolfe step (occasionally)
        if it % 10 == 0:
            t_fw = 2.0 / (it + 2.0)
            a_fw = frank_wolfe_step(a, grad / gnorm, S, t_fw)
            c_fw = fft_convolve_self(a_fw)
            if c_fw.max() < cur_max:
                a = a_fw
            else:
                a = a_grad
        else:
            a = a_grad
        
        # Check new value
        c_new = fft_convolve_self(a)
        new_max = c_new.max()
        
        if new_max < cur_max:
            # Good step, maybe increase step size
            pass
        elif new_max > cur_max + sa_temperature * best:
            # Bad step, revert with some probability (simulated annealing)
            if rng.random() > np.exp(-(new_max - cur_max) / max(sa_temperature * best, 1e-10)):
                a = best_a.copy()  # Or revert to previous
        # else: accept (SA)
        
        # Periodic spectral equalization
        if it % 500 == 0 and rng.random() < 0.3:
            a = spectral_equalization(a, 0.1 + 0.2 * progress, rng)

        # Periodic random perturbation (SA-like)

            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

            a = project_simplex(best_a + rng.normal(0, noise_scale, n), S)

    return best_a, best

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-resolution optimization with:
1. Top-k LSE smoothed gradient descent with adaptive beta
2. Frank-Wolfe (conditional gradient) steps for exploration
3. Frequency-domain spectral equalization
4. Simulated annealing for escaping local minima
5. Multi-phase: explore (moderate beta) -> exploit (high beta) -> polish (very high beta, tiny steps)
6. Primary focus on n=16384 for better scaling
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def frank_wolfe_direction(a, grad, S):
    """Compute Frank-Wolfe step: move towards argmin of linearized objective."""
    n = len(a)
    j_star = np.argmin(grad)
    v = np.zeros(n)
    v[j_star] = S
    return v


def spectral_equality_step(a, alpha, rng):
    """Reduce spectral dynamic range to flatten the convolution."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()

    if mean_p < 1e-30:
        return a

    # Compress dynamic range: blend power with its mean
    new_power = (1.0 - alpha) * power + alpha * mean_p
    new_power = np.maximum(new_power, 1e-30)

    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def make_gaussian_init(n, S, center_frac=0.5, width_frac=0.3):
    """Create a Gaussian bump initialization."""
    j = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((j - center) / width) ** 2)
    return normalize(a, S)


def make_cosine_init(n, S):
    """Create a cosine window initialization."""
    j = np.arange(n)
    a = (1.0 + np.cos(2.0 * np.pi * j / n)) / 2.0
    a = a ** 2
    return normalize(a, S)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_optimization(a_init, S, time_limit, rng, n, phase_name=""):
    """
    Main optimization routine with adaptive parameters.
    Uses top-k LSE gradient with progressive beta increase,
    Frank-Wolfe exploration, and spectral equalization.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    it = 0
    no_improve = 0
    beta = 30.0

    print("  [%s] Start: max_conv=%.6f" % (phase_name, best))

    while time.monotonic() - start < time_limit:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_limit)

        # Adaptive parameters
        if progress < 0.3:
            # Exploration phase
            beta = min(200.0, 30.0 * (1.0 + 0.01 * it))
            k = max(4, int(0.08 * (2 * n - 1)))
            lr_scale = 1.0
        elif progress < 0.7:
            # Exploitation phase
            beta = min(5000.0, 200.0 * (1.0 + 0.005 * it))
            k = max(4, int(0.04 * (2 * n - 1)))
            lr_scale = 0.5
        else:
            # Polish phase
            beta = min(50000.0, 5000.0 * (1.0 + 0.002 * it))
            k = max(2, int(0.02 * (2 * n - 1)))
            lr_scale = 0.2

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute smoothed gradient
        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: normalized so that ||step * grad|| ~ S * lr_scale
        step = lr_scale * S / (gnorm * np.sqrt(n)) * 0.1

        # Main gradient step
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()

        if new_max <= cur_max + 1e-14:
            a = a_new
        elif no_improve > 50 and rng.random() < 0.3:
            # Frank-Wolfe step (accept worse solutions during exploration)
            t_fw = 2.0 / (it + 2.0)
            v_fw = frank_wolfe_direction(a, grad / gnorm, S)
            a_fw = (1.0 - t_fw) * a + t_fw * v_fw
            c_fw = fft_convolve_self(a_fw)
            if c_fw.max() < cur_max:
                a = a_fw
            else:
                a = a_new
        else:
            # Try smaller step
            step *= 0.3
            a_new = a - step * grad
            a_new = project_simplex(a_new, S)
            a = a_new

        # Spectral equalization (periodically)
        if it % 300 == 0 and rng.random() < 0.25 and progress < 0.8:
            alpha_spec = 0.05 + 0.15 * progress
            a = spectral_equality_step(a, alpha_spec, rng)

        # Random perturbation when stuck (simulated annealing style)
        if no_improve > 200:
            temp = 0.005 * (1.0 - progress) * S / np.sqrt(n)
            if temp > 1e-15 and rng.random() < 0.05:
                a = project_simplex(a + rng.normal(0, temp, n), S)
                no_improve = 0

        # Hard reset when very stuck
        if no_improve > 2000:
            temp = 0.002 * S / np.sqrt(n)
            a = project_simplex(best_a + rng.normal(0, temp, n), S)
            no_improve = 0

        # Periodic report
        if it % 5000 == 0:
            obj = 2.0 * n * best / (S * S)
            print("  [%s] it=%d, best=%.6f, obj=%.6f, progress=%.2f" % (phase_name, it, best, obj, progress))

    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_polish(a_init, S, time_limit, rng):
    """Very careful polishing with high beta and tiny steps."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    it = 0
    k = max(2, int(0.015 * (2 * n - 1)))
    step_scale = 1.0

    while time.monotonic() - start < time_limit:
        it += 1
        beta = min(80000.0, 1000.0 * (1.0 + 0.001 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.01 * step_scale * S / (gnorm * np.sqrt(n))
        a_new = a - step * grad
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max - 1e-15:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.05, step_scale * 0.7)
            # Don't revert, just accept (helps with non-convexity)
            a = a_new

        # Tiny perturbation to escape exact local minima
        if it % 1000 == 0:
            temp = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, temp, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best
