import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def power_gradient(a, p):
    """
    Compute gradient of (sum c^p)^(1/p) / S^2 * 2n w.r.t. a.
    where c = conv(a, a).
    """
    n = len(a)

    c = fft_convolve_self(a)

    # df/da[i] = sum_m (df/dc[m]) * dc[m]/da[i]

    S = a.sum()
    c = fft_convolve_self(a)

    shifted = shifted - shifted.max()
    e = np.exp(shifted)

    if tot < 1e-300:

    """
    Reduce the peak at c[k_peak] by transferring mass from hot to cold elements.
    """
    n = len(a)
    S = a.sum()
    
    # Find hot elements (those contributing to c[k_peak])
    # c[k_peak] = sum_j a[j] * a[k_peak - j] for valid j
    lo = max(0, k_peak - n + 1)
    hi = min(n - 1, k_peak)
    hot_idx = range(lo, hi + 1)
    hot_contrib = np.array([a[j] * a[k_peak - j] for j in hot_idx])
    
    # Find the top contributors
    top_contributors = hot_idx[np.argsort(hot_contrib)[::-1][:n_transfer]]
    
    # Find cold elements (low contribution to c[k_peak])
    # Score each element by its total contribution to the top few peaks
    top_peaks = np.argsort(c)[::-1][:5]
    total_contrib = np.zeros(n)
    for kp in top_peaks:
        lo2 = max(0, int(kp) - n + 1)
        hi2 = min(n - 1, int(kp))
        for j in range(lo2, hi2 + 1):
            total_contrib[j] += a[j] * a[int(kp) - j]
    
    # Cold elements: low total contribution, not in hot set
    hot_set = set(top_contributors)
    cold_scores = -total_contrib.copy()
    for j in hot_set:
        cold_scores[j] = -np.inf
    
    cold_elements = np.argsort(cold_scores)[::-1][:n_transfer]
    
    # Transfer mass
    total_hot = sum(a[j] for j in top_contributors)
    delta = total_hot * 0.1  # Transfer 10% of hot mass
    
    for j in top_contributors:
        frac = a[j] / total_hot if total_hot > 0 else 1.0/n_transfer
        a[j] -= delta * frac
    
    for l in cold_elements:
        frac = 1.0 / len(cold_elements)
        a[l] += delta * frac
    
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    
    return a
