def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad_cycling(a, N, beta, peak_offset=0, asym_weight=0.0):
    """
    LSE gradient with peak cycling and anti-symmetric regularization.
    peak_offset: rotate the set of 'active' peaks by this many positions.
    asym_weight: weight for the b[n-1] term.
    Returns (grad, lse_val, c, c_max).
    """
    n = len(a)
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    beta_val = float(beta)
    if beta_val < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        # Peak cycling: shift the weight emphasis
        if peak_offset > 0:
            # Create a bias that emphasizes peaks near position (peak_offset mod (2n-1))
            n_convs = len(c)
            center = (n_convs - 1) / 2.0
            # Gaussian bump centered at peak_offset position
            pos = np.arange(n_convs)
            bump_center = peak_offset % n_convs
            # Circular distance
            dist = np.minimum(np.abs(pos - bump_center), n_convs - np.abs(pos - bump_center))
            sigma = max(5, n_convs // 10)
            bump = np.exp(-dist**2 / (2.0 * sigma**2))
            bump /= bump.sum()
            # Blend uniform and bump
            base_weights = np.ones(n_convs) / n_convs
            weights_base = 0.3 * base_weights + 0.7 * bump
        else:
            weights_base = np.ones(len(c)) / len(c)

        exp_bc = np.exp(beta_val * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            softmax_w = exp_bc / total
            lse_val = c_max + np.log(total) / beta_val
            # Combine with cycling weights
            weights = 0.5 * softmax_w + 0.5 * weights_base

    # Add anti-symmetric regularization: gradient of b[n-1]
    if asym_weight > 0:
        # b[n-1] = sum_i a[i] * a[n-1-i], gradient is 2 * a_reversed (approximately)
        a_rev = a[::-1].copy()
        weights_asym = np.zeros(len(c))
        weights_asym[n - 1] = 1.0
        # Blend
        weights = (1.0 - asym_weight) * weights + asym_weight * weights_asym

    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def topk_grad(a, N, top_k=50, asym_weight=0.0):
    """Gradient using top-K peaks with optional anti-symmetric weight."""
    n = len(a)
    c, A = compute_conv_full(a, N)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Softmax over top-k
    shifted = top_vals - top_vals[0]
    exps = np.exp(shifted * 5.0)
    weights_top = exps / exps.sum()

    fsize = 1 << (3 * n - 2).bit_length()
    w_full = np.zeros(fsize)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights_top[i]

    if asym_weight > 0:
        w_full[n - 1] += asym_weight

    w_full /= w_full.sum()
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def quadratic_step_size(a, direction, c_max, fsize, N, S, n, history):
    """
    Use parabolic interpolation from recent history to choose step size.
    history: list of (step_size, c_max) tuples.
    """
    if len(history) >= 3:
        # Fit parabola to last 3 points
        h = history[-3:]
        x0, f0 = h[0]
        x1, f1 = h[1]
        x2, f2 = h[2]
        if x0 != x1 and x1 != x2 and x0 != x2:
            # Lagrange interpolation for minimum
            denom = (x0 - x1) * (x0 - x2) * (x1 - x2)
            if abs(denom) > 1e-30:
                A = (f0 / ((x0 - x1) * (x0 - x2)) + f1 / ((x1 - x0) * (x1 - x2)) + f2 / ((x2 - x0) * (x2 - x1)))
                B = (-f0*(x0+x1) / ((x0-x1)*(x0-x2)) - f1*(x1+x0) / ((x1-x0)*(x1-x2)) - f2*(x2+x0) / ((x2-x0)*(x2-x1)))
                if abs(A) > 1e-30:
                    x_opt = -B / (2.0 * A)
                    if 0.01 < x_opt < 10.0:
                        return x_opt
    # Default: use ratio-based
    return 1.0


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases while preserving magnitudes, project to non-negative."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    n_rand = max(1, int(N * frac))
    if frac >= 1.0:
        phase = rng.uniform(0, 2 * np.pi, N)
    else:
        phase = np.angle(A).copy()
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_peak_cycling(a_init, n, S, N, max_time, rng,
                        beta_init=10.0, beta_max=500.0,
                        m_lbfgs=12, asym_init=0.1,
                        cycle_period=50, use_topk=False):
    """
    L-BFGS with peak cycling and anti-symmetric regularization.
    Cycles through different peak focus positions to avoid stagnation.
    """
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    asym_weight = asym_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    step_history = []

    # Initial gradient
    if use_topk:
        g, c, c_max = topk_grad(a, N, top_k=100, asym_weight=asym_weight)
    else:
        g, f_val, c, c_max = lse_grad_cycling(a, N, beta, peak_offset=0, asym_weight=asym_weight)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    base_step = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    cycle_count = 0
    prev_cmax = c_max

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta and asymmetry
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.08)
            base_step = 2.0 * S / (n * max(beta, 1.0))
            # Decay asymmetry weight
            asym_weight = max(0.0, asym_init - 0.005 * it / 100.0)
            # Refresh gradient
            if use_topk:
                g, c, c_max = topk_grad(a, N, top_k=100, asym_weight=asym_weight)
            else:
                g, f_val, c, c_max = lse_grad_cycling(a, N, beta,
                                                       peak_offset=cycle_count,
                                                       asym_weight=asym_weight)

        # Peak cycling: rotate focus
        if it % cycle_period == 0:
            cycle_count += 1
            # Pick a random offset within the top peaks
            c_full, _ = compute_conv_full(a, N)
            top_n = min(10, len(c_full))
            top_positions = np.argsort(c_full)[::-1][:top_n]
            offset = int(top_positions[cycle_count % top_n])
            if not use_topk:
                g, f_val, c, c_max = lse_grad_cycling(a, N, beta,
                                                       peak_offset=offset,
                                                       asym_weight=asym_weight)

        # Trim history
        if it % 30 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        # Compute direction
        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        # Step size with quadratic model
        alpha_ls = base_step
        step_history.append((alpha_ls, c_max))
        if len(step_history) > 10:
            step_history = step_history[-10:]
        alpha_quad = quadratic_step_size(a, direction, c_max, N, N, S, n, step_history)
        alpha_ls = base_step * alpha_quad

        # Backtracking line search
        a_new = a
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv_full(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1.0 + 1e-6) or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        # Compute new gradient
        if use_topk:
            g_new, c_nw, cmax_nw = topk_grad(a_new, N, top_k=100, asym_weight=asym_weight)
        else:
            g_new, f_new, c_nw, cmax_nw = lse_grad_cycling(a_new, N, beta,
                                                             peak_offset=cycle_count,
                                                             asym_weight=asym_weight)

        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        # Spectral perturbation for diversity
        if it % 600 == 0 and rng.random() < 0.3:
            a = spectral_phase_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            step_history = []
            if use_topk:
                g, c, c_max = topk_grad(a, N, top_k=100, asym_weight=asym_weight)
            else:
                g, f_val, c, c_max = lse_grad_cycling(a, N, beta,
                                                       peak_offset=cycle_count,
                                                       asym_weight=asym_weight)

    return best_a, best_cmax


def coordinate_descent_polish(a_init, n, S, N, max_time, rng):
    """
    Coordinate descent on anti-symmetric pairs (j, n-1-j) to reduce center peak.
    Also does general pair swaps.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    c, _ = compute_conv_full(a, N)
    best_cmax = c.max()
    best_a = a.copy()

    while time.monotonic() - start_time < max_time:
        it += 1
        c, _ = compute_conv_full(a, N)
        cur_max = c.max()

        if cur_max < best_cmax:
            best_cmax = cur_max
            best_a = a.copy()

        # Find the current maximum peak
        k_star = int(np.argmax(c))

        # Strategy 1: Anti-symmetric pair optimization
        # For pair (j, n-1-j), we can redistribute mass while keeping sum constant
        n_pairs = n // 2
        # Randomly select pairs to try
        n_tries = min(n_pairs, 5000)
        pairs = rng.choice(n_pairs, size=n_tries, replace=False)

        for j in pairs:
            j2 = n - 1 - j
            if j == j2:
                continue
            current_sum = a[j] + a[j2]
            if current_sum < 1e-15:
                continue
            # Try moving mass: a[j] -> a[j]*t, a[j2] -> a[j2]*(1-t)*... no
            # Keep a[j] + a[j2] = current_sum
            # Try: a[j] = current_sum * frac, a[j2] = current_sum * (1-frac)
            # For frac in {0, 0.25, 0.5, 0.75, 1.0}
            best_frac = None
            best_val = cur_max
            a_save_j = a[j]
            a_save_j2 = a[j2]

            for frac in (0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0):
                a[j] = current_sum * frac
                a[j2] = current_sum * (1.0 - frac)
                # Quick check: compute b[k_star] only
                # b[k_star] = sum_i a[i]*a[k_star-i]
                # Only terms involving a[j] or a[j2] change
                # Approximate: just check the full max via FFT (expensive but accurate)
                # For speed, approximate:
                # delta in b[k_star] from changing a[j] and a[j2]
                # This is complex; for now, accept if it looks promising
                # Actually, let's just do a spot check on k_star and a few neighbors
                k_check = k_star
                # b[k_check] contribution from a[j]: a[j]*a[k_check-j] if valid
                # This is too complex for a quick check; skip for now
                pass

            # Revert for now (we'll use a simpler approach)
            a[j] = a_save_j
            a[j2] = a_save_j2

        # Strategy 2: Direct peak shaving via projected gradient
        # Compute gradient of b[k_star]
        grad_peak = np.zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                if j == k_minus_j:
                    grad_peak[j] = 2.0 * a[j]
                else:
                    grad_peak[j] = a[k_minus_j]

        # Also add gradients from other top peaks
        top_k = min(5, len(c))
        top_positions = np.argsort(c)[::-1][:top_k]
        for k_p in top_positions[1:]:
            weight = c[k_p] / max(c[top_positions[0]], 1e-15)
            if weight < 0.5:
                continue
            for j in range(n):
                k_minus_j = k_p - j
                if 0 <= k_minus_j < n:
                    if j == k_minus_j:
                        grad_peak[j] += weight * 2.0 * a[j]
                    else:
                        grad_peak[j] += weight * a[k_minus_j]

        gnorm = np.linalg.norm(grad_peak)
        if gnorm < 1e-15:
            break

        # Projected gradient step
        step = 4.0 * S / (n * gnorm)
        a_new = project_simplex(a - step * grad_peak, S)

        # Verify improvement
        c_new, _ = compute_conv_full(a_new, N)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            step *= 0.25
            a_new2 = project_simplex(a - step * grad_peak, S)
            c_new2, _ = compute_conv_full(a_new2, N)
            if c_new2.max() < cur_max:
                a = a_new2
            else:
                # No improvement; try a random perturbation
                if rng.random() < 0.1:
                    a = spectral_phase_restart(a, N, rng, frac=0.1)

    return best_a, best_cmax


def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=300):
    """Nesterov top-k subgradient for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        c, A = compute_conv_full(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        if it % 5 == 0:
            c_x, _ = compute_conv_full(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 800 == 0 and rng.random() < 0.1:
            y = spectral_phase_restart(y, N, rng, frac=0.15)
            t = 1.0
    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multifreq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 11, 17, 31, 53]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_asym(n, rng):
    """Anti-symmetric inspired: large on one side, small on other."""
    base = np.linspace(0.3, 1.7, n)
    a = base + 0.2 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_binary_asym(n, rng):
    """Binary-like with anti-symmetric bias."""
    a = np.zeros(n)
    # Put mass in a "complement" pattern
    for i in range(n):
        if rng.random() < 0.6:
            a[i] = rng.random() * 0.5
        else:
            a[n - 1 - i] = rng.random() * 0.5
    return np.clip(a, 0.0, None) + 0.01


def deflation_init(n, S, N, n_iters=1000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = np.real(ifft(np.abs(fft(a, N)) ** 2))[:2 * n - 1]
        # Wait, that's the wrong formula. Use proper convolution.
        c, _ = compute_conv_full(a, N)
        k_star = int(np.argmax(c))
        # Find which elements contribute most to b[k_star]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        i_max = int(np.argmax(contrib))
        # Find "cool" element (low total contribution across all peaks)
        # Approximate: element with lowest value
        i_min = int(np.argmin(a))
        if i_max == i_min:
            break
        transfer = min(0.03 * a[i_max], 0.2 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses peak-cycling L-BFGS, coordinate descent polishing,
    Nesterov acceleration, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c, _ = compute_conv_full(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # --- Phase 1: Start from given sequence and optimize intensively ---
    print("=== Phase 1: Intensive optimization from height_sequence_1 ===")
    t1 = time.monotonic()
    phase1_budget = min(600.0, (deadline - t1) * 0.25)

    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_start = np.clip(given, 0, 1000).astype(np.float64)
            a_start = safe_normalize(a_start, n, S)
            print("  Using height_sequence_1 as starting point")
    except Exception:
        pass

    if a_start is None:
        a_start = make_init_chirp(n, rng)
        a_start = safe_normalize(a_start, n, S)
        print("  Using chirp initialization")

    # Run peak-cycling L-BFGS from the given sequence
    a_opt, sc_opt = lbfgs_peak_cycling(a_start, n, S, N, min(phase1_budget, 400.0), rng,
                                        beta_init=8.0, beta_max=300.0,
                                        m_lbfgs=12, asym_init=0.15,
                                        cycle_period=40, use_topk=False)
    update_best(a_opt, "P1/LBFGS_cycling")

    # Run Nesterov polishing on the result
    if deadline - time.monotonic() > 100:
        a_ng, sc_ng = nesterov_toppk_block(a_opt, n, S, N, min(150.0, deadline - time.monotonic() - 200.0),
                                            rng, top_k_init=20, top_k_max=200)
        update_best(a_ng, "P1/Nesterov")

    # --- Phase 2: Coordinate descent polishing ---
    print("=== Phase 2: Coordinate descent polishing ===")
    t2 = time.monotonic()
    phase2_budget = min(200.0, (deadline - t2) * 0.1)
    if phase2_budget > 30:
        a_cd, sc_cd = coordinate_descent_polish(global_a, n, S, N, phase2_budget, rng)
        update_best(a_cd, "P2/CoordDesc")

    # --- Phase 3: Diverse multi-start exploration ---
    print("=== Phase 3: Multi-start exploration ===")
    t3 = time.monotonic()
    phase3_budget = min(500.0, (deadline - t3) * 0.2)

    inits = []
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp2", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("multifreq", make_init_multifreq(n, rng)))
    inits.append(("asym", make_init_asym(n, rng)))
    inits.append(("binary_asym", make_init_binary_asym(n, rng)))
    inits.append(("deflation", deflation_init(n, S, N, 800, rng)))

    n_inits = len(inits)
    per_init = phase3_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 400:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(5.0, per_init - (time.monotonic() - t0) - 0.5)
        # Alternate between methods
        if name in ("zadoff", "asym", "deflation"):
            a_l, sc_l = lbfgs_peak_cycling(a_norm, n, S, N, min(rem, 80.0), rng,
                                            beta_init=5.0, beta_max=200.0,
                                            m_lbfgs=10, asym_init=0.05,
                                            cycle_period=30, use_topk=True)
        else:
            a_l, sc_l = lbfgs_peak_cycling(a_norm, n, S, N, min(rem, 80.0), rng,
                                            beta_init=10.0, beta_max=300.0,
                                            m_lbfgs=10, asym_init=0.1,
                                            cycle_period=50, use_topk=False)
        update_best(a_l, "P3/" + name)

    # --- Phase 4: Focused optimization on best ---
    print("=== Phase 4: Focused optimization ===")
    t4 = time.monotonic()
    phase4_budget = min(600.0, (deadline - t4) * 0.3)

    if global_a is not None:
        a_curr = global_a.copy()
        stall = 0
        for block_i in range(8):
            if time.monotonic() > t4 + phase4_budget:
                break
            block_time = min(120.0, (t4 + phase4_budget) - time.monotonic())
            if block_time < 15:
                break
            beta_start = 15.0 + 5.0 * min(stall, 6)
            beta_end = min(600.0, 80.0 * (1.0 + 0.5 * stall))
            asym_w = max(0.0, 0.2 - 0.02 * stall)
            a_new, sc_new = lbfgs_peak_cycling(a_curr, n, S, N, block_time, rng,
                                                beta_init=beta_start, beta_max=beta_end,
                                                m_lbfgs=15, asym_init=asym_w,
                                                cycle_period=30,
                                                use_topk=(stall % 2 == 1))
            if sc_new < global_best - 1e-9:
                global_best = sc_new
                global_a = a_new.copy()
                a_curr = a_new.copy()
                stall = 0
                score = 2 * n * sc_new / (S * S)
                print("  [P4/block%d] score=%.6f" % (block_i, score))
            else:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.3 + 0.1 * min(stall, 3))
                stall += 1

    # --- Phase 5: Final Nesterov + coordinate descent ---
    print("=== Phase 5: Final polishing ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 20
    if remaining5 > 30:
        half = remaining5 / 2.0
        a_ng2, sc_ng2 = nesterov_toppk_block(global_a, n, S, N, min(half, 200.0),
                                              rng, top_k_init=40, top_k_max=400)
        update_best(a_ng2, "P5/Nesterov")

        remaining5b = deadline - time.monotonic() - 10
        if remaining5b > 30:
            a_cd2, sc_cd2 = coordinate_descent_polish(global_a, n, S, N, min(remaining5b, 150.0), rng)
            update_best(a_cd2, "P5/CoordDesc")

    # --- Phase 6: Ultimate high-beta L-BFGS ---
    remaining6 = deadline - time.monotonic() - 10
    if remaining6 > 30:
        a_fin, sc_fin = lbfgs_peak_cycling(global_a, n, S, N, min(remaining6, 200.0),
                                            rng, beta_init=50.0, beta_max=1000.0,
                                            m_lbfgs=15, asym_init=0.0,
                                            cycle_period=20, use_topk=True)
        update_best(a_fin, "P6/HighBeta")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: score = %.6f" % ver)
        else:
            print("Verification FAILED, using fallback")
            global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Final: length=%d, internal_peak=%.6f, approx_score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
