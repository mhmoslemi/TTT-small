import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_objective(a, beta):
    """LSE-smoothed max of convolution."""
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.exp(shifted).sum().__float__() / beta

    """

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        c = fft_convolve_self(a)
        cm = c.max()

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            # LSE
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            # Penalty
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            # Gradient
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)  # penalty gradient (same for all)
            return obj, grad

        time_budget_this = remaining * 0.85
        t0 = time.monotonic()

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 100, 'maxls': 30, 'ftol': 1e-14, 'gtol': 1e-10,
                     'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))

def generate_diverse_inits(n, rng, S):
    """Generate diverse initial sequences for multi-start."""
    inits = []

    # Cosine with different frequencies
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Smooth random (Gaussian kernel)
    for _ in range(3):
        raw = rng.random(n)
        # Smooth via FFT low-pass
        F = fft(raw)
        cutoff = max(3, n // 100)
        F[cutoff:-cutoff] *= 0.05
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() > 1e-10:
            smooth = smooth * (S / smooth.sum())
            inits.append(smooth.copy())

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    if tri.sum() > 0:
        tri = tri * (S / tri.sum())
        inits.append(tri.copy())

    # Two-peak
    peak1 = n // 4
    peak2 = 3 * n // 4
    sigma = n // 20
    a = np.exp(-0.5 * ((np.arange(n) - peak1) / sigma) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - peak2) / sigma) ** 2)
    a = a * (S / a.sum())
    inits.append(a.copy())

    return inits
