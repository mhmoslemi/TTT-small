import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    # Avoid overflow
    c_shifted = c - cm
    e = np.exp(beta * c_shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def spectral_smooth(a, alpha):
    """Apply spectral smoothing: smooth the power spectrum and transform back."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    
    # Smooth the power spectrum with a Gaussian kernel in frequency
    freqs = np.fft.fftfreq(fsize)
    # Gaussian smoothing in frequency domain
    sigma = alpha * fsize / (2 * np.pi * np.sqrt(n))
    kernel = np.exp(-freqs ** 2 / (2 * sigma ** 2))
    from scipy.ndimage import gaussian_filter1d
    power_smooth = gaussian_filter1d(power, alpha)
    
    # Take square root (magnitude) and reconstruct with original phase
    mag = np.sqrt(power_smooth)
    phase = np.angle(A)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new)[:n])
    return np.maximum(a_new, 0)

def smooth_noise(rng, n, scale):
    """Generate smooth (low-frequency) noise."""
    fsize = 1 << (n - 1).bit_length()
    noise_freq = np.zeros(fsize, dtype=np.complex128)

        noise_freq[fsize - k] = np.conj(noise_freq[k])
    noise = np.real(ifft(noise_freq)[:n])
    return noise

import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    return 2.0 * n * c.max() / (S * S)

def smooth_noise(rng, n, scale, n_freq=8):
    """Generate smooth low-frequency noise."""
    fsize = 1 << (n - 1).bit_length()
    noise_freq = np.zeros(fsize, dtype=np.complex128)
    for k in range(1, n_freq + 1):
        re = rng.normal(0, 1)
        im = rng.normal(0, 1)
        noise_freq[k] = re + 1j * im
        noise_freq[fsize - k] = re - 1j * im
    noise = np.real(ifft(noise_freq)[:n])
    noise -= noise.mean()
    return noise * scale
