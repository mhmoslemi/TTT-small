        scaled = beta * (c - cm)
    e = np.exp(scaled - scaled.max())  # stability
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), beta
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, beta

def active_set_gradient(a, c, tol=1e-3):
    """Subgradient using all lags within tol of max."""
    n = len(a)
    cmax = c.max()
    active = np.where(c >= cmax - tol * cmax)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Use weighted average: weights proportional to (c_k - cmax + tol*cmax)
    weights = c[active] - cmax + tol * cmax
    weights = np.maximum(weights, 1e-10)
    weights = weights / weights.sum()
    
    # Compute combined gradient
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    w_full = np.zeros(len(c))
    w_full[active] = weights
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def best_subgradient(a, c, n_active=5):
    """Find the subgradient (from active lags) that is most anti-aligned with a."""
    n = len(a)
    cmax = c.max()
    # Get top n_active lags
    idx = np.argsort(c)[::-1][:min(n_active, len(c))]
    
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    
    best_grad = None
    best_score = np.inf  # minimize score = dot(grad, a) (want it negative)
    
    for k in idx:
        w_full = np.zeros(len(c))
        w_full[k] = 1.0
        W = fft(w_full, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        score = np.dot(grad, a)
        if score < best_score:
            best_score = score
            best_grad = grad
    
    if best_grad is None:
        return np.zeros(n)
    return best_grad

def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_max=10000.0,
              use_active_set=False, active_tol=1e-3, n_active=5):
    """FISTA with LSE or active-set gradient."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        if use_active_set:
            if n_active == 1:
                grad = best_subgradient(y, c, n_active=1)
            else:
                grad = active_set_gradient(y, c, tol=active_tol)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for subgradient method
            step = S / (n * gnorm) * 1.0
        else:
            beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
            grad, beta_eff = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta_eff)

        # Stuck detection
        if it - last_improve > 5000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation
        if it % 10000 == 0:
            decay = np.exp(-it / (20.0 * 10000))
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    return best_a, best

    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()

    best = c0.max()

    it = 0
    while time.monotonic() - start < max_time - opt_time:

        if r < 0.4:

            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n)
            a_new = a + amp * a.sum() * np.sin(2 * np.pi * freq * tt / n + phase)

        elif r < 0.7:
            # Fourier phase mutation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            scale = 0.1 * np.exp(-progress) + 0.01
            phase_noise = rng.normal(0, scale, fsize)
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a_new = np.real(ifft(new_A, fsize))[:n]
            a_new = project_simplex(a_new, S)
        elif r < 0.85:
            # Block perturbation
            block_size = rng.integers(max(2, n // 100), max(3, n // 10))
            pos = rng.integers(0, max(1, n - block_size))
            shift = rng.normal(0, 0.05 * np.exp(-progress))
            a_new = a.copy()
            a_new[pos:pos+block_size] += shift * S / n
            a_new = project_simplex(a_new, S)
        else:
            # Gaussian noise
            scale = 0.03 * np.exp(-progress)
            a_new = project_simplex(a + rng.normal(0, scale * S / np.sqrt(n), n), S)

        # Local optimization
        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 5)
        if t_alloc < 5:

        a_opt, val = run_fista(a_new, S, t_alloc, rng,
                                beta_init=20 + 30 * min(it, 20),
                                beta_max=5000 + 1000 * min(it, 30))
        if val < best:

            best_a = a_opt.copy()
            a = a_opt.copy()

    return best_a, best

def run_simulated_annealing(a_init, S, max_time, rng):
    """Simulated annealing with structural moves."""
    n = len(a_init)
    current = normalize_to_sum(a_init, S)
    c_cur = fft_convolve_self(current)
    best = c_cur.max()
    best_a = current.copy()
    start = time.monotonic()

    T0 = 0.5
    T_end = 0.001
    n_total = int(max_time / 0.05)  # estimate iterations

    for it in range(n_total):
        elapsed = time.monotonic() - start
        if elapsed > max_time:
            break
        progress = min(1.0, elapsed / max_time)
        T = T0 * (T_end / T0) ** progress

        # Generate perturbation
        r = rng.random()
        if r < 0.3:
            # Sinusoidal perturbation
            freq = rng.integers(1, n // 2)
            amp = T * S / n
            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n)
            candidate = current + amp * np.sin(2 * np.pi * freq * tt / n + phase)
        elif r < 0.5:
            # Random coordinate perturbation
            n_pert = rng.integers(max(1, n // 1000), max(2, n // 100))
            idx = rng.integers(0, n, n_pert)
            candidate = current.copy()
            candidate[idx] += rng.normal(0, T * S / np.sqrt(n), n_pert)
        elif r < 0.7:
            # Two-point swap (move mass from one position to another)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            amount = T * S / n
            candidate = current.copy()
            candidate[i] -= amount
            candidate[j] += amount
        else:
            # Full noise
            candidate = current + rng.normal(0, T * S / np.sqrt(n), n)

        candidate = project_simplex(candidate, S)
        c_cand = fft_convolve_self(candidate)
        obj_cand = c_cand.max()
        obj_cur = c_cur.max()

        if obj_cand < obj_cur or rng.random() < np.exp(-(obj_cand - obj_cur) / max(T * obj_cur, 1e-10)):
            current = candidate
            c_cur = c_cand
            if obj_cand < best:
                best = obj_cand
                best_a = candidate.copy()

    return best_a, best

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

    n = len(a)
    cm = c.max()
    spread = c - cm
    if spread.min() < -300.0 / max(beta, 1e-15):
        beta = 300.0 / max(1e-15, -spread.min())
    scaled = beta * spread
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), beta
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, beta


def active_set_gradient(a, c, tol_frac=0.01, max_active=10):
    """
    Subgradient using weighted combination of active lag gradients.
    Weights proportional to excess above (max - tol*max).
    """
    n = len(a)
    cmax = c.max()
    threshold = cmax - tol_frac * cmax
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > max_active:
        # Keep the top max_active
        order = np.argsort(c[active])[::-1][:max_active]
        active = active[order]

    # Weights: proportional to (c_k - threshold + epsilon)
    excess = c[active] - threshold
    weights = np.maximum(excess, 1e-10)
    weights = weights / weights.sum()

    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    w_full = np.zeros(len(c))
    w_full[active] = weights
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def stochastic_subgradient(a, c, n_tries=3, n_active_consider=5):

            a_new = a.copy()
            a_new[pos_src:pos_src + bs] -= amount
            a_new[pos_dst:pos_dst + bs] += amount
            a_new = project_simplex(a_new, S)
        else:
            # Gaussian noise
            scale = base_scale * S / np.sqrt(n)
            a_new = project_simplex(a + rng.normal(0, scale, n), S)

        # Local optimization
        t_rem = max_time - (time.monotonic() - start)
        t_alloc = min(opt_time, t_rem - 2)
        if t_alloc < 5:
            break

        # Alternate between LSE and active-set modes
        if it % 3 == 0:
            a_opt, val = run_fista(a_new, S, t_alloc, rng,
                                    beta_init=15 + 20 * min(it, 30),
                                    beta_max=3000 + 500 * min(it, 40),
                                    mode="active", active_tol=0.005)
        elif it % 3 == 1:
            a_opt, val = run_fista(a_new, S, t_alloc, rng,
                                    beta_init=10 + 15 * min(it, 30),
                                    beta_max=5000 + 1000 * min(it, 30),
                                    mode="lse")
        else:
            a_opt, val = run_fista(a_new, S, t_alloc, rng,
                                    beta_init=20 + 10 * min(it, 20),
                                    beta_max=8000,
                                    mode="stochastic")

        if val < best:
            best = val
            best_a = a_opt.copy()
            a = a_opt.copy()
            print("      BH iter %d: max_b=%.8f, obj=%.6f" % (it, val, 2.0 * n * val / S**2))

    return best_a, best


def run_simulated_annealing(a_init, S, max_time, rng):
    """
    Simulated annealing with structural moves.
    Uses the exact evaluation (max of convolution) with Metropolis acceptance.
    """
    n = len(a_init)
    current = normalize_to_sum(a_init, S)
    c_cur = fft_convolve_self(current)
    best = c_cur.max()
    best_a = current.copy()
    start = time.monotonic()

    # Temperature schedule
    T0 = best * 0.1  # Initial temperature (in objective units)
    T_min = best * 0.001

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        T = T0 * (T_min / T0) ** progress

        if T < T_min * 0.5:
            break

        # Generate perturbation
        r = rng.random()
        scale = T / best  # Relative perturbation size

        if r < 0.30:
            # Sinusoidal move
            freq = rng.integers(1, max(2, n // 4))
            amp = scale * 0.5 * S / n
            phase = rng.uniform(0, 2.0 * np.pi)
            tt = np.arange(n, dtype=np.float64)
            candidate = current + amp * np.sin(2.0 * np.pi * freq * tt / n + phase)
        elif r < 0.50:
            # Coordinate perturbation (sparse)
            n_pert = max(1, min(n, int(rng.integers(n // 5000, n // 200))))
            idx = rng.integers(0, n, n_pert)
            candidate = current.copy()
            candidate[idx] += rng.normal(0, scale * 0.3 * S / np.sqrt(n), n_pert)
        elif r < 0.70:
            # Two-point mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            amount = scale * 0.3 * S / n * rng.uniform(0.5, 2.0)
            candidate = current.copy()
            candidate[i] -= amount
            candidate[j] += amount
        elif r < 0.85:
            # Block perturbation
            bs = rng.integers(max(2, n // 128), max(3, n // 16))
            pos = rng.integers(0, max(1, n - bs))
            shift = scale * 0.3 * S / n
            candidate = current.copy()
            candidate[pos:pos + bs] += rng.normal(0, shift, bs)
        else:
            # Full noise (rare, for exploration)
            candidate = current + rng.normal(0, scale * 0.1 * S / np.sqrt(n), n)

        candidate = project_simplex(candidate, S)
        c_cand = fft_convolve_self(candidate)
        obj_cand = c_cand.max()
        obj_cur = c_cur.max()

        delta = obj_cand - obj_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current = candidate
            c_cur = c_cand
            if obj_cand < best:
                best = obj_cand
                best_a = candidate.copy()
                if it % 1000 == 0:
                    print("      SA iter %d: max_b=%.8f, obj=%.6f" % (it, best, 2.0 * n * best / S**2))

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations including oscillatory patterns."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        # Resample given sequence to length n
        if len(given_seq) != n:
            indices = np.linspace(0, len(given_seq) - 1, n)
            idx_int = np.floor(indices).astype(int)
            idx_frac = indices - idx_int
            idx_next = np.minimum(idx_int + 1, len(given_seq) - 1)
            resampled = given_seq[idx_int] * (1 - idx_frac) + given_seq[idx_next] * idx_frac
        else:
            resampled = given_seq.copy()
        inits.append(np.clip(resampled, 0, 1000))

    # Smooth windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 8 + 0.01)

    # Oscillatory (key for reducing peak)
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(6 * np.pi * tt + 0.5), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(8 * np.pi * tt) + 0.2 * np.cos(12 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.8 + 0.5 * np.cos(2 * np.pi * tt) * np.sin(4 * np.pi * tt), 0.01))

    # Gaussian-like
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2) + 0.01)
    inits.append(1.0 / (1.0 + 2.0 * (tt - 0.5) ** 2) + 0.01)

    # Two-level (inspired by Barker)
    levels = np.ones(n)
    levels[n // 3:2 * n // 3] = 0.3
    inits.append(levels + 0.05)

    # Random
    inits.append(rng.random(n) + 0.1)

    return inits


def run_polish(a_init, S, max_time, rng):
    """
    Final polishing: alternate between very high-beta LSE and active-set.
    Very careful, small steps.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        # Alternate modes
        if it % 2000 < 1000:
            # High-beta LSE
            beta = 500.0 * (1.0 + 50.0 * progress)
            grad, beta_eff = lse_gradient(y, c, beta)
        else:
            # Active set with tight tolerance
            grad = active_set_gradient(y, c, tol_frac=0.001, max_active=8)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Small adaptive step
        if it % 2000 < 1000:
            step = 1.0 / (3.0 * beta_eff)
        else:
            step = 0.3 * S / (n * gnorm)

        # Stuck handling
        if it - last_improve > 6000:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 8000.0)
            if noise_scale < 1e-15:
                break
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_a, best


def resample_sequence(seq, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(seq)
    if n_src == n_target:
        return seq.copy()
    indices = np.linspace(0, n_src - 1, n_target)
    idx_int = np.floor(indices).astype(int)
    idx_frac = indices - idx_int
    idx_next = np.minimum(idx_int + 1, n_src - 1)
    return seq[idx_int] * (1 - idx_frac) + seq[idx_next] * idx_frac


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse initialization + quick FISTA screening
    2. Basin hopping with structural mutations
    3. Simulated annealing
    4. Final KKT/active-set polish
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20

    print("Starting optimization, budget=%.0fs, seed=%d" % (budget_s, seed))

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" % (
                len(given), given.sum(), compute_objective(given)))
    except Exception:
        pass

    # Try multiple n values
    n_candidates = [4096, 8192, 16384]
    if given is not None:
        # Also try the original length if it's reasonable
        n_given = len(given)
        if 2048 <= n_given <= 32768 and n_given not in n_candidates:
            n_candidates.append(n_given)
            n_candidates.sort()

    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        s = a.sum()
        if s < 0.01:
            return
        obj = 2.0 * len(a) * val / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    *** NEW GLOBAL BEST: obj=%.6f (n=%d) ***" % (obj, len(a)))
            return True
        return False

    # === Phase 1: Quick screening ===
    print("\n=== Phase 1: Quick screening ===")
    p1_start = time.monotonic()
    p1_budget = min(120.0, (deadline - p1_start) * 0.08)

    best_per_n = {}
    for n in n_candidates:
        if time.monotonic() > deadline - (deadline - total_start) * 0.80:
            break
        S = np.sqrt(2.0 * n)
        inits = generate_inits(n, rng, given_seq=given)
        n_screen = min(6, len(inits))
        per_init_time = p1_budget / (n_screen * len(n_candidates))

        n_best_val = np.inf
        n_best_a = None
        for i in range(n_screen):
            t_rem = deadline - time.monotonic() - (deadline - total_start) * 0.75
            t_alloc = min(per_init_time, t_rem)
            if t_alloc < 3:
                break
            a_opt, val = run_fista(inits[i], S, t_alloc, rng,
                                    beta_init=5.0, beta_max=2000, mode="lse")
            if val < n_best_val:
                n_best_val = val
                n_best_a = a_opt.copy()
            update_global(a_opt, val)

        best_per_n[n] = (n_best_val, n_best_a)
        obj_n = 2.0 * n * n_best_val / S**2 if n_best_val < np.inf else np.inf
        print("  n=%d: best max_b=%.6f, obj=%.6f" % (n, n_best_val, obj_n))

    if time.monotonic() > deadline - (deadline - total_start) * 0.75:
        print("  WARNING: Phase 1 exceeded time budget")

    # === Phase 2: Basin hopping ===
    print("\n=== Phase 2: Basin hopping ===")
    p2_start = time.monotonic()
    p2_budget = (deadline - p2_start) * 0.35

    # Focus on the best n from screening
    best_n = max(best_per_n.keys(), key=lambda nn: -best_per_n[nn][0]) if best_per_n else 8192
    # Actually pick the n with best objective
    if best_per_n:
        best_n = min(best_per_n.keys(), key=lambda nn: 2.0 * nn * best_per_n[nn][0] / (np.sqrt(2.0*nn))**2)

    S_best = np.sqrt(2.0 * best_n)
    print("  Focusing on n=%d (S=%.2f)" % (best_n, S_best))

    start_a = best_per_n.get(best_n, (np.inf, np.ones(best_n)))[1]
    if start_a is None:
        start_a = np.ones(best_n) * S_best / best_n

    bh_time = min(p2_budget, 400.0)
    a_bh, val_bh = run_basin_hopping(start_a, S_best, bh_time, rng)
    update_global(a_bh, val_bh)
    print("  After BH: global best obj=%.6f" % global_best)

    # === Phase 3: Simulated annealing ===
    print("\n=== Phase 3: Simulated annealing ===")
    p3_start = time.monotonic()
    p3_budget = min(200.0, (deadline - p3_start) * 0.15)

    if p3_budget > 30:
        a_sa, val_sa = run_simulated_annealing(a_bh if val_bh < np.inf else start_a,
                                                S_best, p3_budget, rng)
        update_global(a_sa, val_sa)
        print("  After SA: global best obj=%.6f" % global_best)

    # === Phase 4: Second round of basin hopping from best ===
    print("\n=== Phase 4: Additional BH from best ===")
    p4_start = time.monotonic()
    p4_budget = min(200.0, (deadline - p4_start) * 0.20)

    if p4_budget > 30 and global_a is not None:
        current_best_for_n = global_a
        S_cur = np.sqrt(2.0 * len(current_best_for_n))
        a_bh2, val_bh2 = run_basin_hopping(current_best_for_n, S_cur, p4_budget, rng)
        update_global(a_bh2, val_bh2)
        print("  After BH2: global best obj=%.6f" % global_best)

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    p5_start = time.monotonic()
    p5_budget = deadline - p5_start - 10

    if global_a is not None and p5_budget > 30:
        S_pol = np.sqrt(2.0 * len(global_a))
        n_pol = len(global_a)

        # Split polish time: half LSE, half active-set
        seg1 = p5_budget * 0.4
        seg2 = p5_budget * 0.3
        seg3 = p5_budget * 0.3

        # High-beta LSE polish
        if seg1 > 10:
            a_p1, val_p1 = run_fista(global_a, S_pol, seg1, rng,
                                      beta_init=200.0, beta_max=50000.0, mode="lse")
            update_global(a_p1, val_p1)
            print("  After high-beta LSE: obj=%.6f" % global_best)

        # Active-set polish
        if seg2 > 10:
            a_p2, val_p2 = run_fista(a_p1, S_pol, seg2, rng,
                                      mode="active", active_tol=0.002)
            update_global(a_p2, val_p2)
            print("  After active-set: obj=%.6f" % global_best)

        # Final fine polish
        if seg3 > 10:
            a_p3, val_p3 = run_polish(a_p2, S_pol, seg3, rng)
            update_global(a_p3, val_p3)
            print("  After final polish: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(len(global_a)) * np.sqrt(2.0 * len(global_a)) / len(global_a)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n_fb = 8192
            global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback obj=%.6f" % ver)
    else:
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("\nDone in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best if np.isfinite(global_best) else ver, len(global_a)))

    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
