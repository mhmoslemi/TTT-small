"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. FISTA with LSE for bulk optimization
2. Fast cyclic MM (Majorization Minimization) using direct subgradients
3. Barzilai-Borwein step-size subgradient method
4. Multi-length search (8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def subgrad_lag(a, k):
    """Direct subgradient of (a*a)[k] w.r.t. a. O(n/2) time."""
    n = len(a)
    g = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    idx = np.arange(m_start, m_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

def lse_smooth(a, beta):
    """Compute LSE-smoothed objective and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, subgrad_lag(a, int(np.argmax(c))), c
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    # Gradient: 2 * ifft(fft(w) * conj(fft(a)))[:n]
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c

def run_fista_lse(a_init, max_time, rng, beta_init=20.0, beta_rate=0.005,
                  beta_max=5000.0, step_scale=1.0):
    """FISTA with LSE smoothing and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c = lse_smooth(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 10:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.02, 2.0 / max(beta, 1.0))
        x_new = x_trial
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart condition
        if it - last_improve > 2000:
            t_mom = 1.0
            noise = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000))
            y = project_simplex(y + rng.normal(0, noise, n), S)
    
    return best_a, best_obj

def run_cyclic_mm(a_init, max_time, rng, K=5, n_inner=10, lr=0.1):
    """
    Cyclic Majorization Minimization.
    For each of the top-K lags, do n_inner MM steps targeting that lag.
    Uses direct subgradients (no FFT for the gradient computation).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 5000:
            break
        
        # Find top-K lags
        K_eff = min(K, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:K_eff]
        
        # Cyclic MM: for each top lag, do a few gradient steps
        for k_star in top_lags:
            k_star = int(k_star)
            g = subgrad_lag(a, k_star)
            # Adaptive step size: normalize gradient
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                continue
            alpha = lr * S / gnorm
            a_new = project_simplex(a - alpha * g, S)
            # Check if this improves the targeted lag
            # (approximate: just accept if the overall max doesn't increase too much)
            a = a_new
        
        # Recompute c periodically (or after each cycle)
        if it % 1 == 0:  # recompute every cycle
            c = fft_convolve_self(a)
    
    return best_a, best_obj

def run_bb_subgradient(a_init, max_time, rng, K_init=1, K_max=32):
    """
    Subgradient method with Barzilai-Borwein step size.
    K (number of top lags used for subgradient) increases over time.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    a_prev = a.copy()
    g_prev = None
    alpha = S  # initial step size
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 5000:
            # Try random perturbation
            idx = rng.integers(0, n, size=10)
            for i in range(10):
                j = idx[i]
                delta = a[j] * 0.1
                if delta > 1e-12:
                    a[j] -= delta
                    a[rng.integers(0, n)] += delta
            c = fft_convolve_self(a)
            continue
        
        # Adaptive K
        if stuck < 100:
            K = K_init
        elif stuck < 500:
            K = min(4, K_init + stuck // 50)
        elif stuck < 2000:
            K = min(16, 4 + stuck // 100)
        else:
            K = min(K_max, 16 + stuck // 200)
        
        # Compute subgradient from top-K lags
        top_idx = np.argsort(c)[::-1][:K]
        g = np.zeros(n)
        for kk in top_idx:
            g += subgrad_lag(a, int(kk))
        g /= K
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # BB step size
        if g_prev is not None:
            s_vec = a - a_prev
            y_vec = g - g_prev
            num = np.dot(s_vec, y_vec)
            den = np.dot(y_vec, y_vec)
            if den > 1e-30:
                alpha_bb = num / den
                if 1e-10 < alpha_bb < 10 * S:
                    alpha = 0.5 * alpha + 0.5 * alpha_bb  # damping
        
        a_prev = a.copy()
        a_new = project_simplex(a - alpha * g, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= c.max() + S * S / (2 * n) * 0.01:
            a = a_new
            c = c_new
        else:
            # Reject and reduce step
            alpha *= 0.5
            a_new = project_simplex(a - alpha * g, S)
            c = fft_convolve_self(a_new)
            a = a_new
        
        g_prev = g
    
    return best_a, best_obj

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "uniform":
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
