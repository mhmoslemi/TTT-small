"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length n.
Uses a combination of:
1. LP-based Successive Convex Approximation (SCA) for the main optimization
2. LSE-FISTA as a fallback/supplementary method
3. Adaptive active set management
4. Peak shaving for final refinement
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_active_set(c, c_max, eps=0.01):
    """Get indices where c is within eps of c_max."""
    return np.where(c >= c_max - eps)[0]


def lscap_step(f, S, active_indices, c, max_iters_lp=100):
    """
    One SCA-LP step.
    f: current sequence (sum = S)
    S: target sum
    active_indices: indices of active lags in c
    c: current convolution values
    Returns: new f (or None if LP failed)
    """
    n = len(f)
    
    # Build LP
    # Variables: x[0..n-1], t  -> total n+1 variables
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: 2*g_m[i]*x[i] - t <= c[m] for each active m
    # g_m[i] = f[m-i] if 0 <= m-i < n, else 0
    A_ub_rows = []
    b_ub = []
    for m in active_indices:
        # g_m[i] = f[m-i] if 0 <= m-i < n
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        g = np.zeros(n)
        g[lo:hi+1] = f[m - np.arange(lo, hi+1)]
        # 2*g[i]*x[i] - t <= c[m]
        row = np.zeros(n + 1)
        row[:n] = 2.0 * g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub.append(c[m])
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    # Equality constraint: sum(x) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: x[i] >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    x_new = result.x[:n]
    return x_new


def lscap_optimize(f_init, n, S, max_time, rng, 
                   active_eps_start=0.1, active_eps_end=0.001,
                   n_active_max=200, damping=0.5):
    """
    Optimize using LP-based SCA.
    """
    f = normalize_sum(f_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(f)
        c_max = c.max()
        
        if c_max < best:
            best = c_max
            best_f = f.copy()
            last_improve = it
        
        # Adaptive active set size
        frac = min(1.0, it / 500.0)
        eps = active_eps_start + (active_eps_end - active_eps_start) * frac
        active = get_active_set(c, c_max, eps)
        if len(active) > n_active_max:
            # Keep only the top n_active_max
            top = np.argsort(c[active])[::-1][:n_active_max]
            active = active[top]
        
        # Solve LP
        x_new = lscap_step(f, S, active, c)
        
        if x_new is None:
            # LP failed, inject noise
            noise_scale = 0.001 * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            continue
        
        # Damped update
        f_candidate = (1 - damping) * f + damping * x_new
        f_candidate = np.maximum(f_candidate, 0)
        f_candidate = f_candidate * (S / f_candidate.sum())
        
        # Verify improvement
        c_new = fft_convolve_self(f_candidate)
        c_new_max = c_new.max()
        
        if c_new_max < c_max * (1 + 1e-10):  # Allow tiny numerical increase
            f = f_candidate
        else:
            # Backtrack
            damping_local = 0.1
            f_candidate = (1 - damping_local) * f + damping_local * x_new
            f_candidate = np.maximum(f_candidate, 0)
            f_candidate = f_candidate * (S / f_candidate.sum())
            c_new = fft_convolve_self(f_candidate)
            if c_new.max() < c_max:
                f = f_candidate
            else:
                # Give up this step
                f = project_simplex(f + rng.normal(0, 1e-5 * S / np.sqrt(n), n), S)
        
        # Stagnation handling
        if it - last_improve > 200:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 500.0)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            last_improve = it
        
        if it % 50 == 0:
            print("  SCA it=%d, best=%.8f, c_max=%.8f, |A|=%d" % (it, best, c[f.argmax()], len(active)))
    
    return best_f, best


def lse_fista(a_init, n, S, max_time, rng,
              beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_true_max = c.max()
        
        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Stagnation
        stagnant = it - last_improve_it
        if stagnant > 500:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 1000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.3
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def peak_shaving_refined(a, n, S, n_steps=200, rng=None):
    """
    Refined peak shaving: identify peaks and redistribute mass.
    """
    a = a.copy()
    for step in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        
        # Top peaks
        n_peaks = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute gradient of max (approximated by weighted sum of peak gradients)
        grad = np.zeros(n)
        for k in top_idx:
            # Gradient of c[k] w.r.t. a[i] is a[k-i] (for valid k-i)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1)
            partners = k - i_vals
            grad[i_vals] += 2.0 * a[partners]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        lr = 0.1 * S / (n * gnorm)
        
        # Update
        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)
        
        # Verify
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cm:
            a = a_new
        else:
            # Try smaller step
            a_new = a - 0.01 * lr * grad
            a_new = project_simplex(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cm:
                a = a_new
            else:
                if rng is not None:
                    noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
                break
    
    return a

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length n=8192.
Uses a novel LP-based Successive Convex Approximation (SCA) as the primary engine,
combined with LSE-FISTA for smooth phases and adaptive peak shaving for refinement.
The SCA method linearizes the active peak constraints at each step and solves a
linear program to find the optimal direction, providing superior convergence
near the solution compared to pure gradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import scipy.sparse as sp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_active_set(c, c_max, eps):
    """Get indices where c is within eps of c_max, limited in size."""
    mask = c >= c_max - eps
    return np.where(mask)[0]


def lscap_single_step(f, S, active_indices, c, max_constraints=150):
    """
    One LP-based SCA step. Linearizes active peak constraints and solves LP.
    Returns new x (length-n array) or None on failure.
    """
    n = len(f)
    m_active = len(active_indices)
    if m_active == 0:
        return None
    if m_active > max_constraints:
        order = np.argsort(c[active_indices])[::-1]
        active_indices = active_indices[order[:max_constraints]]
        m_active = max_constraints

    # Build sparse constraint matrix
    # For each active lag m: 2*sum_i f[m-i]*x[i] - t <= c[m]
    # Variables: x[0..n-1], t  (total n+1)
    rows = []
    cols = []
    vals = []
    b_ub = []

    for idx_m in range(m_active):
        m = int(active_indices[idx_m])
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        i_vals = np.arange(lo, hi + 1)
        g_vals = f[m - i_vals]  # f[m-i] for i in [lo, hi]
        # Non-zero entries for x[i]
        nz_mask = np.abs(g_vals) > 1e-15
        if not np.any(nz_mask):
            continue
        i_nz = i_vals[nz_mask]
        g_nz = g_vals[nz_mask]
        for j in range(len(i_nz)):
            rows.append(idx_m)
            cols.append(int(i_nz[j]))
            vals.append(2.0 * g_nz[j])
        # -1 for t (variable at index n)
        rows.append(idx_m)
        cols.append(n)
        vals.append(-1.0)
        b_ub.append(c[m])

    if len(rows) == 0:
        return None

    A_ub = sp.coo_array((vals, (rows, cols)), shape=(len(np.unique(rows)), n + 1)).tocsr()
    b_ub_arr = np.array(b_ub)

    # Equality: sum(x) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Bounds
    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None

    if not result.success:
        return None

    return result.x[:n]


def lscap_optimize(f_init, n, S, max_time, rng,
                   eps_start=0.05, eps_end=0.0005,
                   damping_start=0.5, damping_end=0.1,
                   max_constraints=100):
    """LP-based SCA optimizer with adaptive parameters."""
    f = normalize_sum(f_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0
    stagnant_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed_frac = min(1.0, (time.monotonic() - start) / max_time)

        c = fft_convolve_self(f)
        c_max = c.max()

        if c_max < best:
            best = c_max
            best_f = f.copy()
            last_improve = it
            stagnant_count = 0
        else:
            stagnant_count += 1

        # Adaptive parameters
        eps = eps_start + (eps_end - eps_start) * elapsed_frac
        damping = damping_start + (damping_end - damping_start) * elapsed_frac

        # Active set
        active = get_active_set(c, c_max, eps)
        if len(active) > max_constraints:
            order = np.argsort(c[active])[::-1]
            active = active[order[:max_constraints]]

        # Solve LP
        x_lp = lscap_single_step(f, S, active, c, max_constraints)

        if x_lp is not None:
            # Damped update
            x_candidate = (1.0 - damping) * f + damping * x_lp
            x_candidate = np.maximum(x_candidate, 0.0)
            s = x_candidate.sum()
            if s > 1e-15:
                x_candidate = x_candidate * (S / s)

            c_new = fft_convolve_self(x_candidate)
            c_new_max = c_new.max()

            if c_new_max <= c_max * (1.0 + 1e-9):
                f = x_candidate
            else:
                # Try smaller damping
                for alpha in [0.3, 0.1, 0.05]:
                    x_try = (1.0 - alpha) * f + alpha * x_lp
                    x_try = np.maximum(x_try, 0.0)
                    if x_try.sum() > 1e-15:
                        x_try = x_try * (S / x_try.sum())
                    c_try = fft_convolve_self(x_try)
                    if c_try.max() < c_max:
                        f = x_try
                        break
                else:
                    # No improvement from LP direction; small noise
                    noise_scale = 1e-5 * S / np.sqrt(n)
                    f = project_simplex(f + rng.normal(0, noise_scale, n), S)
        else:
            # LP failed; use gradient step
            grad = np.zeros(n)
            for m in active[:20]:
                lo = max(0, m - n + 1)
                hi = min(m, n - 1)
                i_vals = np.arange(lo, hi + 1)
                grad[i_vals] += 2.0 * f[m - i_vals]
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                lr = 0.01 * S / (n * gnorm)
                f = project_simplex(f - lr * grad, S)
            else:
                noise_scale = 1e-4 * S / np.sqrt(n)
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)

        # Stagnation handling: periodic perturbation
        if stagnant_count > 100:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-stagnant_count / 200.0)
            if noise_scale > 1e-10:
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            stagnant_count = 0

        if it % 100 == 1:
            elapsed = time.monotonic() - start
            print("  SCA it=%d, best=%.8f, cmax=%.8f, |A|=%d, elapsed=%.0fs" %
                  (it, best, c[f.argmax()], len(active), elapsed))

    return best_f, best


def lse_fista(a_init, n, S, max_time, rng,
              beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it

        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta
        stagnant = it - last_improve_it
        if stagnant > 400:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 800.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.4

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def active_set_line_search(f, n, S, max_time, rng, n_active=30):
    """
    Active-set method with exact line search on the max of active lags.
    At each step, compute the steepest descent direction for the active lags,
    then do an exact (golden-section) line search.
    """
    f = normalize_sum(f, n).copy()
    start = time.monotonic()
    best = np.inf
    best_f = f.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        c_max = c.max()

        if c_max < best:
            best = c_max
            best_f = f.copy()
            last_improve = it

        # Active set: top n_active lags
        top_idx = np.argsort(c)[::-1][:n_active]
        active_c = c[top_idx]
        spread = active_c[-1] - active_c[0]

        # Weighted gradient (equal weight or proportional to how close to max)
        weights = np.ones(len(top_idx))
        grad = np.zeros(n)
        for i, m in enumerate(top_idx):
            lo = max(0, m - n + 1)
            hi = min(m, n - 1)
            i_vals = np.arange(lo, hi + 1)
            grad[i_vals] += weights[i] * 2.0 * f[m - i_vals]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Direction: negative gradient projected onto feasibility
        direction = -grad / gnorm
        # Project direction to maintain sum = S
        direction = direction - direction.mean()
        # Project to maintain non-negativity (simple: zero out where f + alpha*dir < 0)

        # Line search
        best_alpha = 0.0
        best_val = c_max
        alphas = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0]
        for alpha in alphas:
            f_try = f + alpha * direction
            f_try = np.maximum(f_try, 0)
            if f_try.sum() < 1e-15:
                continue
            f_try = f_try * (S / f_try.sum())
            c_try = fft_convolve_self(f_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_alpha = alpha
                f = f_try
                break

        if best_alpha == 0.0:
            # No improvement; add noise
            noise = rng.normal(0, 5e-5 * S / np.sqrt(n), n)
            f = project_simplex(f + noise, S)

        if it - last_improve > 150:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 300.0)
            if noise_scale > 1e-10:
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            last_improve = it

        if it % 200 == 1:
            print("  AS-it %d: best=%.8f, cmax=%.8f, spread=%.6f" % (it, best, c_max, spread))

    return best_f, best


def surgical_redistribute(f, n, S, n_rounds=50, rng=None):
    """
    Direct peak reduction: for the top peak, identify the "bottleneck" pairs
    and redistribute mass from high-product pairs to low-product positions.
    """
    f = f.copy()
    for rnd in range(n_rounds):
        c = fft_convolve_self(f)
        cm = c.max()
        m_star = int(c.argmax())

        # Get contributing pairs for the peak
        lo = max(0, m_star - n + 1)
        hi = min(m_star, n - 1)
        i_vals = np.arange(lo, hi + 1)
        products = np.abs(f[i_vals] * f[m_star - i_vals])
        total_contrib = products.sum()
        if total_contrib < 1e-30:
            break

        # "Responsibility" of each position
        resp = np.zeros(n)
        resp[i_vals] = products / total_contrib

        # Find "safe" positions to receive mass (low contribution to any top peak)
        top_lags = np.argsort(c)[::-1][:10]
        danger = np.zeros(n)
        for m in top_lags:
            lo2 = max(0, m - n + 1)
            hi2 = min(m, n - 1)
            i2 = np.arange(lo2, hi2 + 1)
            danger[i2] += np.abs(f[m - i2])

        danger /= danger.max() + 1e-30

        # Move mass: reduce high-resp positions, increase low-danger positions
        source = i_vals.copy()  # positions to reduce
        # Target: positions with low danger and low current value
        target_scores = danger + 0.5 * (f / (S / n + 1e-30))
        # Pick targets from low-score positions
        n_targets = min(len(source), n // 4)
        target_candidates = np.argsort(target_scores)[:n * 3 // 4]
        # Exclude source positions
        target_mask = np.ones(n, dtype=bool)
        target_mask[source] = False
        target_candidates = target_candidates[target_mask[target_candidates]]
        targets = target_candidates[:n_targets]

        if len(targets) == 0:
            break

        # Transfer amount
        transfer = 0.001 * S * np.exp(-rnd / 20.0)
        # Reduce sources proportionally to their responsibility
        reduce_amounts = resp[source] * transfer
        reduce_amounts = np.minimum(reduce_amounts, f[source] * 0.5)
        # Increase targets uniformly
        increase_amount = reduce_amounts.sum() / len(targets)

        f[source] -= reduce_amounts
        f[targets] += increase_amount
        f = np.maximum(f, 0)
        f = f * (S / (f.sum() + 1e-30))

        # Verify improvement
        c_new = fft_convolve_self(f)
        if c_new.max() > cm * 1.01:
            # Revert this round
            c_old = fft_convolve_self(normalize_sum(f, n))
            # Just continue with slight noise
            if rng is not None:
                f = project_simplex(f + rng.normal(0, 1e-5 * S / np.sqrt(n), n), S)
            break

    return f

    print("Budget: %.0fs, n=%d" % (budget_s, n))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 100 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n_local = len(a)
        S_local = np.sqrt(2.0 * n_local)
        a = normalize_sum(a, n_local)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  *** New best: %.8f (n=%d) ***" % (val, n_local))

    # Initialize
    if given is not None:
        if len(given) == n:
            global_a = normalize_sum(given, n)
        else:
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, n)
            global_a = np.interp(t_new, t_old, given)
            global_a = normalize_sum(global_a, n)
        global_best = fft_convolve_self(global_a).max()
        print("Starting value: %.8f" % global_best)
    else:
        # Asymmetric raised cosine init
        t = np.linspace(0, 1, n)
        global_a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.35))
        global_a = np.maximum(global_a, 0.0)
        global_a = normalize_sum(global_a, n)
        global_best = fft_convolve_self(global_a).max()
        print("Default init: %.8f" % global_best)

    # === Phase 1: Quick LSE warmup ===
    remaining = deadline - time.monotonic()
    t1 = min(100.0, remaining * 0.1)
    print("\n=== Phase 1: LSE warmup (%.0fs) ===" % t1)
    if t1 > 10:
        a1, v1 = lse_fista(global_a, n, S, t1, rng,
                            beta_start=100.0, beta_end=5000.0, beta_steps=80)
        try_sequence(a1)

    # === Phase 2: LP-SCA main optimization ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.35
    print("\n=== Phase 2: LP-SCA (%.0fs) ===" % t2)
    if t2 > 30:
        a2, v2 = lscap_optimize(global_a, n, S, t2, rng,
                                 eps_start=0.03, eps_end=0.0003,
                                 damping_start=0.4, damping_end=0.1,
                                 max_constraints=80)
        try_sequence(a2)

    # === Phase 3: Active-set line search ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.25
    print("\n=== Phase 3: Active-set line search (%.0fs) ===" % t3)
    if t3 > 20:
        a3, v3 = active_set_line_search(global_a, n, S, t3, rng, n_active=20)
        try_sequence(a3)

    # === Phase 4: LP-SCA refinement (tighter) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.25
    print("\n=== Phase 4: LP-SCA tight (%.0fs) ===" % t4)
    if t4 > 15:
        a4, v4 = lscap_optimize(global_a, n, S, t4, rng,
                                 eps_start=0.005, eps_end=0.0001,
                                 damping_start=0.2, damping_end=0.05,
                                 max_constraints=40)
        try_sequence(a4)

    # === Phase 5: Surgical redistribution ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 5: Surgical redistribution (%.0fs) ===" % remaining)
    if remaining > 10:
        a5 = surgical_redistribute(global_a, n, S, n_rounds=30, rng=rng)
        try_sequence(a5)
        # Quick LSE polish
        remaining2 = max(1.0, deadline - time.monotonic())
        if remaining2 > 15:
            a5b, v5b = lse_fista(a5, n, S, remaining2 * 0.4, rng,
                                  beta_start=5000.0, beta_end=30000.0, beta_steps=30)
            try_sequence(a5b)

    # === Phase 6: Final LSE polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: Final LSE (%.0fs) ===" % remaining)
        a6, v6 = lse_fista(global_a, n, S, remaining, rng,
                            beta_start=2000.0, beta_end=50000.0, beta_steps=50)
        try_sequence(a6)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nFinal: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n, dtype=np.float64) * (S / n)
    else:
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
