    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 80)))
        beta = beta_init * (beta_end / beta_init) ** progress
        k = max(2, int(k_start - (k_start - k_end) * progress))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        grad, w, idx = lse_grad_and_weights(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Line search
        success = False
        for _ in range(12):
            x_new = project_simplex(y - step * grad / (gnorm + 1e-30), S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-9:
                success = True
                break
            step *= 0.5

        if not success:
            # Fallback: small random step
            a = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            y = a.copy()
            t_mom = 1.0
            step = 0.5
            continue

        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_mom = t_new
        step = min(step * 1.05, 3.0)

        # Restart if stuck
        if stuck_count > 800:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            step = 1.0
            stuck_count = 0

    return best_a, best


def run_peak_polish(a_init, S, max_time, rng,
                    beta_init=500.0, beta_end=100000.0,
                    k_start=20, k_end=1):
    """Phase 2: High-beta peak targeting with no momentum for precision."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step = 0.5

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 60)))
        beta = beta_init * (beta_end / beta_init) ** progress
        k = max(1, int(k_start - (k_start - k_end) * progress))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad, w, idx = lse_grad_and_weights(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Careful line search with Armijo-like condition
        improved = False
        for ls in range(15):
            a_new = project_simplex(a - step * grad / (gnorm + 1e-30), S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max - 1e-12:
                a = a_new
                improved = True
                break
            elif c_new.max() < cur_max + 1e-10:
                a = a_new
                improved = True
                break
            step *= 0.5

        if improved:
            step = min(step * 1.03, 2.0)
        else:
            step = max(step * 0.3, 0.01)
            # Try multiplicative update as alternative
            mult_update = np.exp(-0.01 * grad / (gnorm + 1e-30))
            mult_update = np.clip(mult_update, 0.5, 2.0)
            a_alt = normalize(a * mult_update, S)
            c_alt = fft_convolve_self(a_alt)
            if c_alt.max() < cur_max:
                a = a_alt

        # Small perturbation to escape plateaus
        if it % 300 == 0 and progress < 0.85:
            scale = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            if scale > 1e-12:
                a_pert = project_simplex(a + rng.normal(0, scale, n), S)
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < best * 1.0005:
                    a = a_pert

    return best_a, best


def run_log_space_opt(a_init, S, max_time, rng,
                      beta_init=200.0, beta_end=20000.0,
                      k_start=50, k_end=3):
    """Optimization in log-space (multiplicative updates) for better conditioning."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    lr = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 50)))
        beta = beta_init * (beta_end / beta_init) ** progress
        k = max(1, int(k_start - (k_start - k_end) * progress))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad, w, idx = lse_grad_and_weights(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Log-space gradient: d(LSE)/dx_k = 2 * a_k * ((w*a)_k - C_w)
        C_w = np.dot(w, c[idx]) if k < len(c) else np.dot(w, c)
        # Actually compute properly
        c_weighted = np.sum(w_full_arr := np.zeros(len(c)))  # placeholder
        # The log-space gradient is: 2 * a_k * conv(w, a)_k - 2 * a_k * sum(w_i * c_i)
        # Let's compute conv(w, a) via FFT
        fsize = 1 << (len(c) + n - 1).bit_length()
        W = fft(np.zeros(len(c)))  # We need w_full
        w_full = np.zeros(len(c))
        w_full[idx] = w
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        wa = np.real(ifft(W * A, fsize))[:n]  # This is sum_i w_i * a_{i-k} approximately
        # Actually conv(w_full, a) gives sum_j w_full[j] * a[k-j]
        # What we need is sum_i w_i * a_{i-k} which is cross-correlation
        # = conv(w_full, a)[k] if we handle indices right
        # Let's just use the a-space gradient for the multiplicative update
        # a_j *= exp(-lr * grad_a_j / |grad|)
        
        lr_eff = lr * (1.0 - 0.5 * progress)
        log_update = -lr_eff * grad / (gnorm + 1e-30)
        log_update = np.clip(log_update, -2.0, 2.0)
        a_new = a * np.exp(log_update)
        a_new = normalize(a_new, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr = max(lr * 0.5, 0.001)
            # Try a smaller step
            log_update_small = -lr_eff * 0.1 * grad / (gnorm + 1e-30)
            a_try = normalize(a * np.exp(log_update_small), S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max:
                a = a_try

        # Perturbation
        if it % 400 == 0 and progress < 0.8:
            scale = 0.002 * S / np.sqrt(n)
            a_pert = project_simplex(a + rng.normal(0, scale, n), S)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best * 1.001:
                a = a_pert

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def spectral_flatten(a, alpha, rng):
    """Reduce high-power frequencies to flatten the power spectrum."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power[1:].mean()
    if mean_p < 1e-20:
        return a
    ratio = power[1:] / max(mean_p, 1e-20)
    # Damp peaks in power spectrum
    new_power = np.ones_like(power)
    new_power[0] = power[0]
    new_power[1:] = power[1:] * (1.0 - alpha * (ratio - 1.0) / (1.0 + ratio * 0.1))
    new_power = np.maximum(new_power, 1e-30)
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def propose_candidate():

    n1 = 8192
