import cvxpy as cp

def lp_direction(f, M, n):
    """
    Solve the LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns b*.
    """
    b = cp.Variable(n)
    
    # Build the convolution matrix A where A[k,j] = f[|k-j|] (or appropriate)
    # (f*b)[k] = sum_j f[j]*b[k-j] for 0<=j<n, 0<=k-j<n
    # = sum_j f[j]*b[k-j] where 0<=k-j means j<=k, and k-j<n means j>k-n
    
    # For efficiency, note that (f*b)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j]*b[k-j]
    # Let m = k-j: (f*b)[k] = sum_{m=max(0,k-n+1)}^{min(n-1,k)} f[k-m]*b[m]
    
    # This is a banded matrix structure.
    # A[k, m] = f[k-m] if 0<=k-m<n, else 0, where m=0,...,n-1
    
    # Build the matrix (sparse)
    from scipy.sparse import diags
    
    # Actually, for n=8192, the dense matrix would be 16383 x 8192 = ~134M entries.
    # That's too large for dense. Use sparse.
    # Each row k has at most min(k+1, n, 2n-1-k) non-zero entries.
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(k, n-1)
        for j in range(lo, hi+1):
            m = k - j  # b index
            # (f*b)[k] includes term f[j]*b[k-j] = f[j]*b[m]
            rows.append(k)
            cols.append(m)
            vals.append(f[j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)
    
    return b.value

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, mirror descent (multiplicative updates)
with adaptive active sets, peak shaving, and LP-based direction finding.
Key insight: At the KKT optimum, the 'blurred reflection' of a (using active set as kernel)
must be constant on the support. We iterate toward this condition using exponential-family
updates and adaptive active set management.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (gradient, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_subgradient(a, c, active_idx):
    """
    Compute subgradient using hard active set.
    g[j] = (1/|A|) * sum_{k in A} a[k-j] * 1{0 <= k-j < n}
    The factor of 2 from d(a*a)/da is included.
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k - js
        g[js] += a[partners]
    g *= 2.0 / len(active_idx)
    return g


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200,
                  momentum_restart_gap=500):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_true_max = c.max()

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / (20000.0 / max(1, n // 1024)))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_mirror_descent(a_init, n, S, max_time, rng,
                       init_eta=0.3, eta_decay=0.9995,
                       active_frac=0.03, min_active=3, max_active=200,
                       noise_interval=500, noise_base=0.005):
    """
    Mirror descent (multiplicative updates) with adaptive active set.
    a_new[j] ∝ a[j] * exp(-eta * g[j])
    where g is the subgradient of max autocorrelation using the active set.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    eta = init_eta
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()
            last_improve = it

        # Adaptive active set
        threshold = cm * (1.0 - active_frac)
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) < min_active:
            active_idx = np.argsort(c)[::-1][:min_active]
        if len(active_idx) > max_active:
            active_idx = np.argsort(c)[::-1][:max_active]

        # Compute subgradient
        g = active_set_subgradient(a, c, active_idx)

        # Multiplicative update
        g_max = g.max()
        if g_max < 1e-15:
            break

        # Scale gradient for numerical stability
        scaled = g / g_max
        a_new = a * np.exp(-eta * scaled)

        # Ensure minimum values to avoid underflow
        a_new = np.maximum(a_new, 1e-15)

        # Rescale
        s_new = a_new.sum()
        a = a_new * (S / s_new)

        eta = max(0.005, eta * eta_decay)

        # Periodic noise
        if it % noise_interval == 0:
            noise = rng.normal(0, noise_base * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = min(last_improve, it)  # reset stagnation timer

        # Stagnation: if no improvement for a while, increase noise
        stagnant = it - last_improve
        if stagnant > 2000:
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = it
            eta = min(eta * 2, init_eta)

    return best_a, best


def run_peak_shaving(a_init, n, S, max_time, rng):
    """
    Iterative peak shaving: identify peaks, compute responsibility,
    multiplicatively reduce high-responsibility elements.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        # Get top peaks (adaptive number)
        n_peaks = min(20, max(3, int(0.005 * n)))
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute responsibility for each a[j]
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.95)
            if weight < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-20:
            break

        # Multiplicative reduction
        lr = 0.02 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = np.maximum(a, 1e-15)
        a = project_simplex(a, S)

        # Every 100 iterations, do a small LSE-FISTA polish
        if it % 100 == 0 and time.monotonic() - start < max_time - 10:
            a, _ = run_lse_fista(a, n, S, min(10.0, (max_time - (time.monotonic() - start)) * 0.1),
                                   rng, beta_start=500.0, beta_end=2000.0, beta_steps=20)

    return best_a, best


def run_kkt_refinement(a_init, n, S, max_time, rng):
    """
    KKT-based refinement: iterate toward the condition that the blurred
    reflection is constant on the support of a.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Determine support (elements above threshold)
    support_thresh = 1e-6 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        # Active set: lags within 1% of max
        active_idx = np.where(c >= cm * 0.995)[0]
        if len(active_idx) < 3:
            active_idx = np.argsort(c)[::-1][:5]
        if len(active_idx) > 300:
            active_idx = np.argsort(c)[::-1][:300]

        # Compute blurred reflection: h[j] = (1/|A|) * sum_{k in A} a[k-j]
        h = np.zeros(n)
        for k in active_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            h[js] += a[partners]
        h /= len(active_idx)

        # On the support, h should be constant (lambda)
        support = a > support_thresh
        if support.sum() < 2:
            break

        # Compute lambda as median of h on support (robust)
        lambda_val = np.median(h[support])

        if lambda_val < 1e-15:
            break

        # Measure deviation
        deviation = h[support] / lambda_val - 1.0
        max_dev = np.abs(deviation).max()

        if max_dev < 1e-6:
            # Very close to KKT; do a small perturbation
            break

        # Multiplicative correction: reduce where h is high, increase where h is low
        alpha = min(0.1, 0.01 + 0.1 * max_dev)
        correction = np.ones(n)
        correction[support] = (lambda_val / h[support]) ** alpha
        # Clamp correction to avoid extreme changes
        correction = np.clip(correction, 0.5, 2.0)

        a = a * correction
        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)

    return best_a, best


def run_lp_direction(f, M, n, solver='GLOP'):
    """
    Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns b solution.
    """
    try:
        import cvxpy as cp
        from scipy.sparse import csr_matrix

        b = cp.Variable(n)

        # Build sparse convolution matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = range(lo, hi + 1)
            ms = k - list(js)
            for j, m in zip(js, ms):
                rows.append(k)
                cols.append(m)
                vals.append(f[j])

        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        constraints = [A_mat @ b <= M, b >= 0]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=solver)
        except Exception:
            prob.solve(solver=cp.PDLP)

        if b.value is None:
            return None
        return np.asarray(b.value, dtype=np.float64).flatten()

    except Exception:
        return None


def interpolate_sequence(a_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a_old)
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    a_new = np.interp(t_new, t_old, a_old)
    return a_new

            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
