"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
1. Diverse initialization (analytical, eigenvector-based, random)
2. Basin hopping for global search
3. FISTA with adaptive LSE smoothing and backtracking
4. KKT-optimized subgradient with dual LP weights
5. L-BFGS-B polish on softmax-parameterized LSE
6. Fine-tuning with diminishing steps

import numpy as np
import time
import math
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smoothed_max(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    wsum = e.sum()
    obj = cmax + np.log(wsum) / beta
    w = e / wsum
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad, c, cmax


def fista_lse(a_init, S, max_time, rng, beta_init=10.0, beta_max=5000.0,
              use_backtrack=True, perturb_period=3000):
    """FISTA with adaptive LSE smoothing and backtracking line search."""
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.03 * it))

        obj, grad, c, cmax = lse_smoothed_max(y, beta)
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with backtracking
        step = 1.0 / (2.0 * beta)
        if use_backtrack:
            x_try = project_simplex(y - step * grad, S)
            for bt in range(15):
                c_try = fft_convolve_self(x_try)
                if c_try.max() <= cmax + 1e-8 * max(cmax, 1.0):
                    break
                step *= 0.5
                x_try = project_simplex(y - step * grad, S)
        else:
            x_try = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_try + ((t_mom - 1.0) / t_new) * (x_try - a)
        y = np.maximum(y, 0.0)
        a = x_try
        t_mom = t_new

        # Stuck detection and perturbation
        if it - last_improve > perturb_period:
            decay = np.exp(-(it - last_improve) / (perturb_period * 2))
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
            t_mom = 1.0
            last_improve = it

    return best_a, best


def compute_weighted_subgradient(a, active_lags, weights):
    """Compute weighted subgradient of max(conv(a,a)) at given lags."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    size = 2 * n - 1
    w_full = np.zeros(size)
    for i, k in enumerate(active_lags):
        w_full[k] = weights[i]
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_kkt_weights(a, active_lags, max_iters=50):
    """
    Solve the KKT dual approximately using multiplicative weights.
    Find mu on active_lags that maximizes min_j (C_mu a)_j.
    Returns optimal weights.
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return np.array([0.0])
    if m == 1:
        return np.array([1.0])

    # Build interaction matrix C[j, k] = a[active_lags[k] - j] if valid
    C = np.zeros((n, m))
    for idx_k, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            C[lo:hi, idx_k] = a[k - np.arange(lo, hi)]

    # Initialize uniformly
    mu = np.ones(m) / m

    # Multiplicative weights / exponential weights
    lr = 1.0
    for it in range(max_iters):
        Cm = C @ mu  # shape (n,)
        # Find most violated constraint
        j_min = np.argmin(Cm)
        # Move mu towards the column that best helps j_min
        row = np.maximum(C[j_min, :], 0.0)
        row_sum = row.sum()
        if row_sum > 1e-15:
            mu_target = row / row_sum
        else:
            mu_target = mu
        # Exponential update
        alpha = lr / (1.0 + it * 0.1)
        mu = mu * (1.0 - alpha) + mu_target * alpha
        mu = np.maximum(mu, 1e-10)
        mu = mu / mu.sum()

    return mu


def kkt_subgradient_opt(a_init, S, max_time, rng, max_active=30, mu_update_every=15):
    """
    KKT-optimized subgradient with dual weight solving.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_base = S / n
    mu_cached = None
    active_cached = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = a.copy()
            last_improve = it

        # Determine active lags
        progress = (time.monotonic() - start) / max_time
        tol_rel = max(0.01, 0.1 * np.exp(-2.0 * progress))
        threshold = cmax * (1.0 - tol_rel)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > max_active:
            idx = np.argsort(c)[::-1][:max_active]
            active = np.sort(idx)

        # Solve dual weights periodically
        if mu_cached is None or it % mu_update_every == 0 or len(active) != len(active_cached):
            mu = solve_kkt_weights(a, active, max_iters=40)
            mu_cached = mu
            active_cached = active.copy()
        else:
            # Reuse cached weights if active set hasn't changed much
            mu = mu_cached

        # Compute weighted subgradient
        grad = compute_weighted_subgradient(a, active, mu)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step with geometric decay and backtracking
        step = step_base * np.exp(-0.003 * it)
        improved = False
        for bt in range(10):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cmax - 1e-12:
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # Perturb to escape
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-0.01 * (it - last_improve))
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            else:
                break

        # Stuck detection
        if it - last_improve > 5000:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            mu_cached = None
            last_improve = it

    return best_a, best


def lbfgs_polish(a_init, S, max_time, beta=200.0):
    """
    L-BFGS-B polish using softmax parameterization a = S * softmax(z).
    """
    n = len(a_init)

    def obj_grad_z(z):
        # a = S * softmax(z)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()

        # Compute LSE objective and gradient w.r.t. a
        c = fft_convolve_self(a)
        cmax = c.max()
        scaled = beta * (c - cmax)
        scaled = np.clip(scaled, -500, 0)
        e = np.exp(scaled)
        wsum = e.sum()
        obj = cmax + np.log(wsum) / beta
        w = e / wsum

        # Gradient w.r.t. a
        fsize = 1 << (2 * n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        g_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Chain rule: dz -> a = S * softmax(z)
        # da_j/dz_k = a_j * (delta_jk - a_k / S)
        # gz_k = sum_j g_a_j * da_j/dz_k = a_k * (g_a_k - (1/S) * sum_j g_a_j * a_j)
        sa = np.sum(g_a * a) / S
        gz = a * (g_a - sa)

        return obj, gz

    a = np.clip(a_init, 1e-10, 1000.0)
    a = a / a.sum() * S
    z0 = np.log(a) - np.log(S)  # Inverse softmax parameterization

    start = time.monotonic()
    max_iter = min(300, int(max_time / 0.5))

    try:
        result = minimize(
            lambda z: obj_grad_z(z)[0],
            z0,
            jac=lambda z: obj_grad_z(z)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10,
                     'maxfun': max_iter * 2}
        )
        if result.x is not None:
            z_sol = result.x
            exp_z = np.exp(z_sol - z_sol.max())
            a_sol = S * exp_z / exp_z.sum()
            c_sol = fft_convolve_self(a_sol)
            return a_sol, c_sol.max()
    except Exception:
        pass

    return a_init, fft_convolve_self(a_init).max()


def eigenvector_init(n, S, mu_type='triangular', mu_len=30, rng=None):
    """
    Generate initialization by solving (mu * a) = lambda * 1.
    At KKT optimum, a satisfies this for some mu and lambda.
    """
    # Define mu profile centered at lag n-1
    center = n - 1
    mu = np.zeros(2 * n - 1)
    half = mu_len // 2
    lo = max(0, center - half)
    hi = min(2 * n - 1, center + half)

    for k in range(lo, hi + 1):
        d = k - center
        if mu_type == 'uniform':
            mu[k] = 1.0
        elif mu_type == 'triangular':
            mu[k] = max(0.1, 1.0 - abs(d) / (half + 1))
        elif mu_type == 'gaussian':
            sigma = half * 0.6
            mu[k] = np.exp(-d * d / (2 * sigma * sigma)) if sigma > 0 else 1.0
        else:
            mu[k] = 1.0

    mu_sum = mu.sum()
    if mu_sum < 1e-10:
        return np.full(n, S / n)
    mu = mu / mu_sum

    # Build the banded matrix M: M[j, i] = mu[j - i] if 0 <= j-i <= 2n-2
    # We need (M @ a) = lambda * 1,  1^T @ a = S,  a >= 0
    # Solve [M, -1; 1^T, 0] @ [a; lambda] = [0; S]
    # This is an (n+1) x (n+1) system.

    # For efficiency, build M as a banded matrix
    # M has bandwidth mu_len (non-zero only |j-i| < mu_len)
    # Use numpy to build it (sparse would be better but let's keep it simple for moderate sizes)

    # Actually for n=8192, building a dense n x n matrix is 8192^2 * 8 = ~500MB. Too much!
    # Use iterative solver instead.

    # Use conjugate gradient or GMRES
    # Define matrix-vector product: (M @ x)_j = sum_k mu[k] * x[j-k] (valid range)
    # This is a convolution! Can be done via FFT.

    def matvec(x):
        """Compute M @ x = mu * x (linear convolution, take first n)."""
        fsize = 1 << (2 * n - 1).bit_length()
        MU = fft(mu, fsize)
        X = fft(x, fsize)
        result = np.real(ifft(MU * X))[:n]
        return result

    # Solve M @ a = lambda * 1, sum(a) = S using iterative method
    # Use the constraint to eliminate lambda: lambda = (M @ a)_j for all j
    # This means (M @ a)_j should be constant, equal to lambda.
    # The system is: (M - lambda * I) @ a = 0, 1^T @ a = S
    # Or equivalently: find a in null(M - lambda*I) with sum S.
    # This is an eigenvalue problem! Find eigenvalue lambda and eigenvector a of M
    # such that a >= 0 and sum(a) = S.

    # Use power iteration / Lanczos to find the eigenpair with a >= 0.
    # Actually, let's just use a simple iterative approach:
    # 1. Start with a random a >= 0, sum = S.
    # 2. Iterate: a_new = project_simplex(M @ a, S)
    # 3. This is like power iteration projected to the simplex.

    if rng is None:
        rng = np.random.default_rng(0)

    a = np.ones(n) * S / n
    for it in range(200):
        a_new = matvec(a)
        a = project_simplex(a_new, S)
        # Check convergence
        if it > 10:
            diff = np.linalg.norm(a - a_prev) / max(np.linalg.norm(a), 1e-10)
            if diff < 1e-8:
                break
        a_prev = a.copy()

    # Check if non-negative (should be by construction)
    if a.min() < -1e-10:
        a = np.maximum(a, 0)
        a = a / a.sum() * S

    return a


def basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with adaptive mutations."""
    n = len(a_init)
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(30.0, min(120.0, max_time / 8.0))
    count = 0

    while time.monotonic() - start < max_time - opt_time - 10:
        count += 1
        progress = (time.monotonic() - start) / max_time

        # Adaptive mutation
        scale = 0.4 * np.exp(-2.0 * progress) + 0.02
        r = rng.random()

        if r < 0.5:
            # Gaussian perturbation on random subset
            n_perturb = max(1, int(n * (0.05 + 0.3 * scale)))
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] += rng.normal(0, S / np.sqrt(n_perturb) * scale, n_perturb)
        elif r < 0.8:
            # Fourier phase mutation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            freqs = np.arange(fsize, dtype=np.float64)
            weight = 1.0 + freqs / (0.1 * fsize)
            phase_noise = rng.normal(0, scale * 0.5 * weight, fsize)
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a = np.real(ifft(new_A, fsize))[:n]
        else:
            # Random coordinate reshuffling
            n_swap = max(1, int(n * 0.05))
            for _ in range(n_swap):
                i, j = rng.integers(0, n, 2)
                delta = rng.normal(0, S / np.sqrt(n))
                a[i] += delta
                a[j] -= delta
        a = project_simplex(a, S)

        # Local optimization
        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 10)
        if t_alloc < 15:
            break
        a_opt, val = fista_lse(a, S, t_alloc, rng,
                                beta_init=20 + 20 * count,
                                beta_max=5000,
                                use_backtrack=True)
        if val < best:
            best = val
            best_a = a_opt.copy()
            obj = 2.0 * n * val / (S ** 2)
            print("    BH iter %d: new best obj=%.6f" % (count, obj))

    return best_a, best
