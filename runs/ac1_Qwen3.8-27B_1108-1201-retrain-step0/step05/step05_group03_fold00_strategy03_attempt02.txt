def project_simplex(v, S):
    """
    Project v onto the set {x >= 0, sum(x) = S}.
    If sum(v) is too small or negative, return a uniform sequence with sum=S.
    """
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, len(v)+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = len(v)
    else:
        rho = np.max(np.where(valid)[0]) + 1
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    """
    Clip and rescale a to have sum=S and elements in [0, 1000].
    """
    a = np.asarray(a, dtype=np.float64)

    s = a_clipped.sum()
    if s < 1e-15:
        return np.full(len(a), S/len(a))
    return a_clipped * (S/s)

def fft_convolve_self(a):
    """
    Full linear self-convolution via FFT.
    Returns a convolution array of size 2*n-1.
    """
    n = len(a)
    size = 2*n-1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = ifft(A * A)
    return np.real(c)[:size]
