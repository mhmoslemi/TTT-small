"""
Minimize 2*n*max_conv/sum^2 using Adam optimizer with LSE smoothing,
antipalindromic initializations, adaptive top-K gradient focusing,
and structured spectral restarts. Key innovations:
1. Antipalindromic initializations exploit <a,rev(a)> < ||a||^2 to start below the all-ones barrier.
2. Adam with decoupled weight decay for more stable convergence than L-BFGS.
3. Adaptive top-K gradient that shifts from LSE (many peaks) to hard top-K (few peaks) as optimization progresses.
4. "Peak-directed" restarts that perturb specifically to reduce the current maximum, unlike random restarts.
5. Multi-resolution: brief optimization on truncated sequences to find good coarse structure, then refine.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    if v.sum() == S:
        return v
    dim = np.arange(1, n + 1, dtype=np.float64)
    w = np.sort(v)[::-1]
    cw = np.cumsum(w)
    cond = w > (cw - S) / dim
    rho = int(np.max(np.where(cond)[0])) + 1 if np.any(cond) else n
    theta = (cw[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns conv of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_and_grad(a, N, beta, fsize):
    """Compute LSE value, LSE weights, and gradient. Returns (lse_val, weights, grad, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        weights = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    w_pad = np.zeros(fsize)
    w_pad[:len(weights)] = weights
    W = fft(w_pad)
    a_pad = np.zeros(fsize)
    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, weights, grad, c, c_max


def toppk_grad(a, N, K, fsize):
    """Gradient focusing on top-K peaks. Returns (grad, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    total = top_vals.sum()
    weights = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        weights[int(k)] = top_vals[i] / total
    w_pad = np.zeros(fsize)
    w_pad[:len(weights)] = weights
    W = fft(w_pad)
    a_pad = np.zeros(fsize)
    a_pad[:n] = a
    Af = fft(a_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def safe_normalize(a, n, S):
    """Clamp and normalize to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_antipal(n, rng):
    """Antipalindromic perturbation of uniform: reduces <a,rev(a)> below ||a||^2."""
    t = np.arange(n)
    a = 1.0 + 0.8 * np.sin(2.0 * np.pi * t / n)
    a = np.maximum(a, 0.01)
    return a + 0.05 * rng.standard_normal(n)


def make_init_antipal2(n, rng):
    """Stronger antipalindromic with harmonic."""
    t = np.arange(n)
    a = 1.0 + 0.7 * np.sin(2.0 * np.pi * t / n) + 0.3 * np.sin(4.0 * np.pi * t / n + rng.random() * np.pi)
    a = np.maximum(a, 0.0)
    return a + 0.05 * rng.standard_normal(n)


def make_init_chirp(n, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * t + t * t)) + 0.2 * np.sin(2 * np.pi * 7 * t)
    a = np.maximum(a, 0.0)
    return a + 0.03 * rng.standard_normal(n)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_fourier(n, rng, N):
    """Random magnitude spectrum with flat-ish profile."""
    P = 1.0 + 0.5 * rng.random(N)
    P[0] = float(n)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a = np.real(ifft(A))[:n]
    return np.abs(a) + 0.01


def make_init_ramp(n, rng):
    """Monotone ramp - strongly antipalindromic."""
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * t + 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)


def make_init_dual_ramp(n, rng):
    """Two peaks at 1/4 and 3/4 positions."""
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.25)) * 0.3 + 0.5 * np.cos(2 * np.pi * (t - 0.75)) * 0.3
    a = np.maximum(a, 0.0)
    return a + 0.03 * rng.standard_normal(n)


def adam_lse_block(a_init, n, S, N, fsize, max_time, rng,
                   beta_start=10.0, beta_end=300.0, lr=0.05,
                   beta1=0.9, beta2=0.999, eps=1e-8,
                   switch_to_toppk_at=150.0, topk_final=20):
    """Adam optimizer with LSE gradient, switching to top-K at end."""
    a = safe_normalize(a_init, n, S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    beta = beta_start

    while time.monotonic() - start_time < max_time:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max_time)
        beta = beta_start + (beta_end - beta_start) * progress * progress
        lr_t = lr / (1.0 + 0.001 * progress)

        if elapsed > switch_to_toppk_at:
            K = min(2 * n - 1, max(5, topk_final + int(30 * (1.0 - progress * 2))))
            K = max(3, K)
            grad, c, c_max = toppk_grad(a, N, K, fsize)
        else:
            lse_val, weights, grad, c, c_max = lse_and_grad(a, N, beta, fsize)

        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = project_simplex(a_new, S)

    return best_a, best_cmax


def adam_lse_restart_block(a_init, n, S, N, fsize, max_time, rng,
                           beta_start=10.0, beta_end=500.0, lr=0.05,
                           restart_interval=60.0, beta1=0.9, beta2=0.999):
    """Adam with periodic structured restarts on stalls."""
    a = safe_normalize(a_init, n, S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    beta = beta_start
    stall_count = 0
    last_improvement = start_time
    last_best_cmax = np.inf

    while time.monotonic() - start_time < max_time:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max_time)
        beta = beta_start + (beta_end - beta_start) * progress
        lr_t = lr * (1.0 + 0.5 * (1.0 - progress))

        # Batch of 50 Adam steps
        batch_size = 50
        for _ in range(batch_size):
            if time.monotonic() - last_start > restart_interval:
                break
            lse_val, weights, grad, c, c_max = lse_and_grad(a, N, beta, fsize)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                last_improvement = time.monotonic()
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a = project_simplex(a_new, S)

        # Check for stall
        if time.monotonic() - last_improvement > restart_interval:
            stall_count += 1
            if stall_count <= 2:
                # Partial spectral restart
                a = _spectral_partial_restart(a, N, rng, frac=0.3 + 0.1 * stall_count)
            elif stall_count <= 4:
                # Full spectral restart
                a = _spectral_full_restart(a, N, rng)
            else:
                # Antipalindromic nudge
                a = _antipalindromic_nudge(a, n, S, rng, strength=0.1 * stall_count)
                stall_count = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            last_improvement = time.monotonic()

        last_start = time.monotonic()

    # Final check
    c = compute_conv(a, N)
    if c.max() < best_cmax:
        best_cmax = c.max()
        best_a = a.copy()
    return best_a, best_cmax


def _spectral_partial_restart(a, N, rng, frac=0.4):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def _spectral_full_restart(a, N, rng):
    """Full spectral phase randomization."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def _antipalindromic_nudge(a, n, S, rng, strength=0.1):
    """Perturb towards antipalindromic structure."""
    t = np.arange(n)
    anti = np.sin(2.0 * np.pi * t / n)
    anti = anti - anti.mean()
    anti_norm = anti / (np.max(np.abs(anti)) + 1e-10)
    a_new = a + strength * anti_norm * (a.mean() + 1e-10)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def peak_directed_restart(a, N, n, S, rng, strength=0.05):
    """Perturb specifically to reduce the current peak lag."""
    c = compute_conv(a, N)
    k_star = int(np.argmax(c))
    # Gradient of conv[k_star] w.r.t. a
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contrib = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
    # Move mass from high-contribution to low-contribution positions
    i_max = int(np.argmax(contrib))
    candidates = np.where(contrib < 0.1 * contrib.max())[0]
    if len(candidates) == 0:
        candidates = np.where(contrib == contrib.min())[0]
    i_min = candidates[rng.integers(len(candidates))]
    if i_max == i_min:
        return a.copy()
    transfer = strength * a[i_max]
    a_new = a.copy()
    a_new[i_max] -= transfer
    a_new[i_min] += transfer
    return project_simplex(a_new, S)


def nesterov_toppk(a_init, n, S, N, fsize, max_time, rng, top_k_init=10, top_k_max=100):
    """Nesterov-accelerated top-K subgradient for final polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            weights[int(k)] = top_vals[i] / total

        w_pad = np.zeros(fsize)
        w_pad[:len(weights)] = weights
        W = fft(w_pad)
        a_pad = np.zeros(fsize)
        a_pad[:n] = y
        Af = fft(a_pad)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 500 == 0 and rng.random() < 0.1:
            y = _spectral_partial_restart(y, N, rng, frac=0.15)
            t = 1.0

    return best_a, best_cmax


def lbfgs_lse_block(a_init, n, S, N, fsize, max_time, rng, beta_start=20.0, beta_end=400.0):
    """L-BFGS on LSE for final convergence."""
    m_lbfgs = 10
    a = safe_normalize(a_init, n, S)
    beta = beta_start
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    lse_val, weights, g, c, c_max = lse_and_grad(a, N, beta, fsize)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = S / (n * 0.5)
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 100 == 0:
            beta = min(beta_end, beta * 1.15)
            lse_val, weights, g, c, c_max = lse_and_grad(a, N, beta, fsize)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        direction = _lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        lse_new, w_new, g_new, c_nw, cmax_nw = lse_and_grad(a_new, N, beta, fsize)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 300 == 0 and rng.random() < 0.2:
            a = _spectral_partial_restart(a, N, rng, frac=0.2)
            s_hist = []
            y_hist = []
            lse_val, weights, g, c, c_max = lse_and_grad(a, N, beta, fsize)

    return best_a, best_cmax


def _lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        return -g / max(gn, 1e-18)
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Adam optimizer with LSE/top-K gradients, antipalindromic initializations,
    structured restarts, and multi-phase refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    fsize = 1 << (3 * n - 2).bit_length()

    print("n=%d, S=%.4f, N=%d, fsize=%d, budget=%.0fs" % (n, S, N, fsize, budget_s))

    global_best = np.inf
    global_a = None

    def check_and_update(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = 2.0 * n * sc / (S * S)
            if label:
                print("  [%s] peak=%.8f score=%.8f" % (label, sc, score))
        return sc

    # === Phase 1: Multi-start exploration (250s) ===
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(250.0, (deadline - t1) * 0.12)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass
    inits.append(("antipal", make_init_antipal(n, rng)))
    inits.append(("antipal2", make_init_antipal2(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("ramp", make_init_ramp(n, rng)))
    inits.append(("dual_ramp", make_init_dual_ramp(n, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1fs each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(2.0, per_init - (time.monotonic() - t0) - 0.5)
        a_res, sc_res = adam_lse_block(a_norm, n, S, N, fsize, min(rem, 50.0), rng,
                                       beta_start=5.0, beta_end=80.0, lr=0.04,
                                       switch_to_toppk_at=999.0)
        check_and_update(a_res, name)

    print("Phase 1 best score: %.8f" % (2.0 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    # === Phase 2: Extended Adam with restarts (1200s) ===
    print("=== Phase 2: Extended Adam with restarts ===")
    t2 = time.monotonic()
    if global_a is None:
        global_a = make_init_antipal(n, rng)
        global_a = safe_normalize(global_a, n, S)
        global_best = compute_conv(global_a, N).max()

    phase2_budget = min(1200.0, (deadline - t2) * 0.50)
    a_curr = global_a.copy()
    stall = 0
    block_results = []

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(200.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 30:
            break
        beta_s = 8.0 + 4.0 * min(stall, 8)
        beta_e = min(600.0, 50.0 * (1.0 + 0.5 * stall))
        a_new, sc_new = adam_lse_restart_block(a_curr, n, S, N, fsize, block_time, rng,
                                                beta_start=beta_s, beta_end=beta_e, lr=0.05,
                                                restart_interval=50.0)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2.0 * n * sc_new / (S * S)
            print("  [P2] score=%.8f (beta %d-%d, stall=%d)" % (score, int(beta_s), int(beta_e), stall))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 3:
                a_curr = peak_directed_restart(a_curr, N, n, S, rng, strength=0.03)
                stall = 0
            elif stall >= 5:
                a_curr = _spectral_partial_restart(a_curr, N, rng, frac=0.5)
                stall = 0

    # === Phase 3: Nesterov top-K polishing (400s) ===
    print("=== Phase 3: Nesterov top-K polishing ===")
    t3 = time.monotonic()
    remaining3 = min(400.0, (deadline - t3) * 0.20)
    if remaining3 > 20:
        a_ng, sc_ng = nesterov_toppk(global_a, n, S, N, fsize, remaining3, rng,
                                     top_k_init=10, top_k_max=80)
        check_and_update(a_ng, "nesterov")

    # === Phase 4: L-BFGS final polish (350s) ===
    print("=== Phase 4: L-BFGS final polish ===")
    t4 = time.monotonic()
    remaining4 = min(350.0, (deadline - t4) * 0.20)
    if remaining4 > 20:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, fsize, remaining4, rng,
                                        beta_start=30.0, beta_end=500.0)
        check_and_update(a_fin, "LBFGS_final")

    # === Phase 5: Short Nesterov + check ===
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 30:
        a_ng2 = global_a.copy()
        a_ng2, sc_ng2 = nesterov_toppk(a_ng2, n, S, N, fsize, min(remaining5, 200.0), rng,
                                       top_k_init=5, top_k_max=30)
        check_and_update(a_ng2, "nesterov2")
        # Quick L-BFGS
        remaining5b = deadline - time.monotonic() - 5
        if remaining5b > 20:
            a_fin2, sc_fin2 = lbfgs_lse_block(global_a, n, S, N, fsize, remaining5b, rng,
                                              beta_start=50.0, beta_end=800.0)
            check_and_update(a_fin2, "final2")

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.8f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * compute_conv(global_a, 1 << (2 * len(global_a) - 1).bit_length()).max() / (safe_normalize(global_a, len(global_a), np.sqrt(2.0 * len(global_a))).sum() ** 2)
    print("Returning length %d, score=%.8f" % (len(global_a), final_score))
    return [float(x) for x in global_a]
