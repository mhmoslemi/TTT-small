=> d[m] = -(c[m] + mu) / lambda

Constraint sum d[m] = 0:
sum_m -(c[m] + mu) / lambda = 0
=> -sum_m c[m] - n*mu = 0
=> mu = -mean(c)

So d[m] = (mean(c) - c[m]) / lambda

And ||d|| = delta determines lambda:
delta^2 = sum_m ((mean(c) - c[m]) / lambda)^2 = sum_m (mean(c) - c[m])^2 / lambda^2
=> lambda = sqrt(sum_m (mean(c) - c[m])^2) / delta = ||c - mean(c)|| / delta

So the optimal step is:
d[m] = delta * (mean(c) - c[m]) / ||c - mean(c)||_2

This is just projecting c onto the hyperplane sum=0 and scaling to have norm delta! Very clean.

Now, we need to check if the non-negativity constraint d[m] >= -a_0[m] is violated. If it is, we need to solve the constrained version.

The constrained version:
minimize sum_m c[m] * d[m]
subject to: sum_m d[m] = 0, ||d||_2 <= delta, d[m] >= -a_0[m]

This is a QP that can be solved by active set method. The unconstrained solution is d_uncon[m] = delta * (mean(c) - c[m]) / ||c - mean(c)||.

If d_uncon[m] >= -a_0[m] for all m, we're done.

Otherwise, we fix the violated coordinates at d[m] = -a_0[m] and re.solve on the remaining coordinates.

This is essentially a projected gradient step with the constraint handled by projection.

Let me implement this efficiently.

Actually, for the implementation, I think a simpler approach works well:
1. Compute c[m] as above
2. Compute d_uncon[m] = (mean(c) - c[m]) / ||c - mean(c)||
3. Scale: d[m] = delta * d_uncon[m]
4. Check non-negativity: if a_0[m] + d[m] < 0 for some m, reduce delta until the constraint is satisfied (or use a projection)

For step 4, the maximum allowable delta is:
delta_max = min over m where d_uncon[m] < 0 of (-a_0[m] / d_uncon[m])

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using QP-based peak reduction.

Key innovation over previous approach: Instead of FISTA with LSE smoothing,
use exact quadratic subproblem (QP) steps that project the subgradient
onto the feasible set (simplex + non-negativity) for maximally efficient
peak reduction. Combined with spectral domain perturbations for global
exploration and adaptive multi-peak targeting for convergence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Simple approach: iterative water-filling
    for _ in range(50):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if abs(s - S) < 1e-12 * max(S, 1.0):
            break
        v = v + (S - s) / n
    return np.maximum(v, 0.0)


def project_simplex_fast(v, S):
    """Fast projection onto {x >= 0, sum(x) = S} using sorting."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / rho
    cond = mu > tau
    if not np.any(cond):
        return np.full(n, S / n)
    rho_max = np.max(np.where(cond)[0]) + 1
    theta = tau[rho_max - 1]
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def qp_peak_step(a, r, k_peaks, delta, weights=None):
    """
    Compute QP-based step to reduce peaks at given lags.
    
    Parameters:
    - a: current sequence (non-negative, sum = S)
    - r: autocorrelation of a
    - k_peaks: array of peak lag indices
    - delta: step size (fraction, will be scaled)
    - weights: optional weights for each peak
    
    Returns: new sequence
    """
    n = len(a)
    S = a.sum()
    
    if weights is None:
        # Weight proportional to excess above mean
        r_mean = r.mean()
        weights = np.maximum(r[k_peaks] - r_mean, 1e-10)
        weights = weights / weights.sum()
    
    # Compute c[m] = sum_i w_i * a[k_i - m] (with a[idx]=0 if out of range)
    c = np.zeros(n)
    for i, k in enumerate(k_peaks):
        w = weights[i]
        # a[k - m] for m = 0, ..., n-1
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)  # m goes from lo to hi-1
        if lo < hi:
            # c[m] += w * a[k - m] for m in [lo, hi)
            # a[k-m] for m in [lo, hi) gives a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            c[lo:hi] += w * a[k - np.arange(lo, hi)]
    
    # Unconstrained QP solution: d[m] = (mean(c) - c[m]) / ||c - mean(c)||
    c_centered = c - c.mean()
    cnorm = np.linalg.norm(c_centered)
    if cnorm < 1e-15:
        return a.copy()
    
    d = c_centered / cnorm  # unit vector in the descent direction
    d = -d  # We want to minimize c.dot(d), so go in -c_centered direction
    # Wait: we're minimizing sum c[m]*d[m], so the direction is -c_centered/||c_centered||
    # But we also need sum(d) = 0, which is satisfied since c_centered has mean 0.
    
    # Actually let me redo this. We want to minimize:
    #   sum_m c[m] * d[m]
    # subject to sum(d) = 0, ||d|| = delta
    # 
    # By Cauchy-Schwarz, the minimum is achieved when d is parallel to (c - mean(c)):
    # d[m] = -delta * (c[m] - mean(c)) / ||c - mean(c)||
    
    d = -delta * c_centered / cnorm
    
    # Check non-negativity: a + d >= 0
    # Find the maximum delta that satisfies this
    neg_mask = d < 0
    if np.any(neg_mask):
        # Need: a[m] + d[m] >= 0 => d[m] >= -a[m]
        # d[m] = -delta * c_centered[m] / cnorm
        # -delta * c_centered[m] / cnorm >= -a[m]
        # delta <= a[m] * cnorm / (-c_centered[m])  (when c_centered[m] > 0, i.e., d[m] < 0)
        ratios = a[neg_mask] * cnorm / (-c_centered[neg_mask])
        delta_max = np.min(ratios) if len(ratios) > 0 else delta
        if delta_max < delta:
            delta = max(delta_max * 0.95, 1e-15)
            d = -delta * c_centered / cnorm
    
    a_new = a + d
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S)
    return a_new


def adaptive_delta(a, r, r_best, n):
    """Adaptive step size based on current improvement potential."""
    S = a.sum()
    excess = r.max() - r.mean()
    # Scale step with the excess and sequence scale
    base = S / n
    return base * 0.1 * max(excess / (r.max() + 1e-10), 0.01)


def fista_lse(a_init, S, max_time, rng, beta_start=10.0, beta_rate=0.003, beta_max=10000.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_start * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        if scaled.max() > 500:
            beta_eff = 500.0 / max(1e-15, (c - cm).max())
            scaled = beta_eff * (c - cm)
            beta = beta_eff
        e = np.exp(scaled)
        tot = e.sum()
        w = e / tot
        
        # Gradient via FFT
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.5 * beta)
        
        # Stuck detection
        if it - last_improve > 2000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex_fast(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        x_new = project_simplex_fast(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best


def qp_optimize(a_init, S, max_time, rng):
    """
    QP-based peak reduction optimization.
    """
    n = len(a_init)
    a = normalize(a_init, S)
    start = time.monotonic()
    r = fft_convolve_self(a)
    best = r.max()
    best_a = a.copy()
    it = 0
    stagnation = 0
    
    # Adaptive parameters
    k_peaks = 4  # Number of peaks to target
    base_delta = 0.05  # Base step size fraction
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        if it % 100 == 0:
            r = fft_convolve_self(a)
            if r.max() < best:
                best = r.max()
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 100
            
            if it % 1000 == 0:
                print("    QP iter %d: best=%.8f, cur=%.8f, stag=%d" % (it, best, r.max(), stagnation))
            
            # Adaptive k and delta
            if progress < 0.3:
                k_peaks = 4
                base_delta = 0.05
            elif progress < 0.6:
                k_peaks = 8
                base_delta = 0.02
            else:
                k_peaks = 16
                base_delta = 0.005
        
        # Find top-k peaks
        top_k = min(k_peaks, len(r))
        peak_ratios = np.argsort(r)[::-1][:top_k]
        
        # Compute QP step
        S_val = a.sum()
        delta = base_delta * S_val / n
        
        a_new = qp_peak_step(a, r, peak_ratios, delta)
        
        # Evaluate and accept/reject
        r_new = fft_convolve_self(a_new)
        if r_new.max() < r.max():
            a = a_new
            r = r_new
            if r.max() < best:
                best = r.max()
                best_a = a.copy()
                stagnation = 0
        else:
            # Reject: try smaller step
            a_new = qp_peak_step(a, r, peak_ratios, delta * 0.5)
            r_new = fft_convolve_self(a_new)
            if r_new.max() < r.max():
                a = a_new
                r = r_new
            else:
                stagnation += 1
        
        # Stagnation handling: spectral perturbation
        if stagnation > 2000:
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            scale = 0.05 * (1.0 + progress)
            phase_noise = rng.normal(0, scale, fsize)
            # Weight more at high frequencies
            freqs = np.arange(fsize)
            phase_noise *= (1.0 + freqs / (0.1 * fsize))
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a = project_simplex_fast(np.real(ifft(new_A, fsize))[:n], S)
            r = fft_convolve_self(a)
            stagnation = 0
            t_mom = 1.0
    
    return best_a, best


def spectral_perturb_optimize(a_init, S, max_time, rng):
    """
    Alternate between spectral perturbation and QP optimization.
    """
    n = len(a_init)
    a = normalize(a_init, S)
    start = time.monotonic()
    r = fft_convolve_self(a)
    best = r.max()
    best_a = a.copy()
    
    # Population of candidates
    population = [(best, a.copy())]
    
    opt_time = max(10.0, max_time / 20.0)
    
    while time.monotonic() - start < max_time - opt_time:
        # Choose mutation strategy
        rng_choice = rng.random()
        
        if len(population) >= 3 and rng_choice < 0.3:
            # Crossover
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a1, a2 = population[i1][1], population[i2][1]
                n_mix = rng.integers(n // 4, 3 * n // 4)
                a = np.concatenate([a1[:n_mix], a2[n_mix:]])
            else:
                a = a_init.copy()
        elif rng_choice < 0.6:
            # Spectral perturbation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            scale = 0.2 * rng.random() + 0.05
            phase_noise = rng.normal(0, scale, fsize)
            freqs = np.arange(fsize)
            phase_noise *= (1.0 + freqs / (0.05 * fsize))
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a = project_simplex_fast(np.real(ifft(new_A, fsize))[:n], S)
        else:
            # Time-domain perturbation
            noise = rng.normal(0, 0.1 * S / np.sqrt(n), n)
            a = project_simplex_fast(a + noise, S)
        
        # QP optimize from this starting point
        a_opt, val = qp_optimize(a, S, opt_time, rng)
        
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    SP iter: new best max_b=%.8f" % val)
        
        population.append((val, a_opt.copy()))
        if len(population) > 10:
            population.sort(key=lambda x: x[0])
            population = population[:10]
    
    return best_a, best

"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex_fast(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / rho
    cond = mu > tau
    if not np.any(cond):
        return np.full(n, S / n)
    rho_max = np.max(np.where(cond)[0]) + 1
    theta = tau[rho_max - 1]
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_cost_vector(a, peak_indices, weights):
    """Compute c[m] = sum_i w_i * a[k_i - m] for the QP step."""
    n = len(a)
    c = np.zeros(n)
    for i, k in enumerate(peak_indices):
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            idx = np.arange(lo, hi)
            c[lo:hi] += w * a[k - idx]
    return c


def qp_step(a, cost_vec, delta):
    """
    Compute the QP-optimal step: minimize cost_vec.dot(d)
    subject to sum(d)=0, ||d||<=delta, a+d>=0.
    Returns (new_a, actual_delta_used).
    """
    n = len(a)
    S = a.sum()
    c_centered = cost_vec - cost_vec.mean()
    cnorm = np.linalg.norm(c_centered)
    if cnorm < 1e-15:
        return a.copy(), 0.0
    direction = -c_centered / cnorm
    # Check non-negativity constraint
    neg_mask = direction < -1e-15
    if np.any(neg_mask):
        ratios = a[neg_mask] / (-direction[neg_mask])
        delta_max = np.min(ratios)
        eff_delta = min(delta, delta_max * 0.99)
    else:
        eff_delta = delta
    if eff_delta < 1e-15:
        return a.copy(), 0.0
    d = eff_delta * direction
    a_new = a + d
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new, eff_delta

    a = normalize_seq(a_init, S)

    stagnation = 0

    while time.monotonic() - start < max_time:

        progress = elapsed / max_time
        
        if it % 200 == 0:

                best_a = a.copy()
                stagnation = 0

                stagnation += 200

            base_delta = 0.005

        S_val = a.sum()
        delta = base_delta * S_val / n
        
        best_r = r.max()

                    r = r_new
                    best_r = r.max()
                    accepted = True
                    break

def fista_lse(a_init, S, max_time, rng, beta_start=5.0, beta_rate=0.004, beta_max=10000.0):
    """FISTA with LSE smoothing for broad optimization."""
    n = len(a_init)
    a = normalize_seq(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    r0 = fft_convolve_self(a)
    best = r0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_start * np.exp(beta_rate * it))
        
        r = fft_convolve_self(y)
        if r.max() < best:
            best = r.max()
            best_a = y.copy()
            last_improve = it
        
        cm = r.max()
        scaled = beta * (r - cm)
        if scaled.max() > 400:
            beta = 400.0 / max(1e-15, (r - cm).max())
            scaled = beta * (r - cm)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / tot
        
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.5 * beta)
        
        if it - last_improve > 2000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex_fast(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        x_new = project_simplex_fast(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best


def spectral_basin_hopping(a_init, S, max_time, rng, verbose=True):

def plateau_polish(a_init, S, max_time, rng, verbose=True):
    """
    Plateau polishing: target ALL excess values above a moving threshold
    with very small QP steps.
    """
    n = len(a_init)
    a = normalize_seq(a_init, S)
    start = time.monotonic()
    r = fft_convolve_self(a)
    best = r.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        if it % 100 == 0:
            r = fft_convolve_self(a)
            if r.max() < best:
                best = r.max()
                best_a = a.copy()
            if verbose and it % 1000 == 0:
                print("    Plateau it=%d best=%.8f cur=%.8f" % (it, best, r.max()))
        
        # Moving threshold: start at max, gradually lower
        threshold = r.max() - 0.3 * (r.max() - r.mean()) * (0.5 + 0.5 * progress)
        
        # Find all excess lags
        above = np.where(r > threshold)[0]
        if len(above) == 0:
            above = np.array([np.argmax(r)])
        # Limit to at most 64 lags for efficiency
        if len(above) > 64:
            top_vals = r[above]
            top_order = np.argsort(top_vals)[::-1][:64]

        excess = np.maximum(r[above] - threshold, 1e-10)
        weights = excess / excess.sum()
        
        cost = compute_cost_vector(a, above, weights)
        
        S_val = a.sum()
        delta = 0.003 * S_val / n * (1.0 - 0.5 * progress)
        
        a_new, _ = qp_step(a, cost, delta)
        if not np.allclose(a_new, a):
            r_new = fft_convolve_self(a_new)
            if r_new.max() <= r.max() + 1e-12:
                a = a_new
                r = r_new
    
    return best_a, best

def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:

    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass
    
    n = 16384
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None
    
    def try_update(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  *** NEW BEST: obj=%.6f ***" % obj)
    
    # Phase 1: FISTA screening
    print("\n=== Phase 1: FISTA screening ===")
    p1_budget = min(200.0, (deadline - total_start) * 0.10)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(5, len(inits))
    per_time = p1_budget / n_screen
    
    candidates = []
    for i in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.88:
            break
        t_alloc = min(per_time, deadline - time.monotonic() - 180)
        if t_alloc < 20:
            break
        print("  Screening init %d/%d..." % (i, n_screen - 1))
        a_opt, val = fista_lse(inits[i], S, t_alloc, rng,
                                beta_start=5.0, beta_rate=0.004, beta_max=5000)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Init %d: obj=%.6f" % (i, obj))
        try_update(a_opt, val)
    
    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)
    
    # Phase 2: QP optimization
    print("\n=== Phase 2: QP optimization ===")
    p2_budget = min(500.0, (deadline - time.monotonic()) * 0.30)
    if global_a is not None and p2_budget > 30:
        a_qp, val_qp = qp_optimize(global_a, S, p2_budget, rng, verbose=True)
        try_update(a_qp, val_qp)
        print("  After QP: obj=%.6f" % global_best)
    
    # Phase 3: Spectral basin hopping
    print("\n=== Phase 3: Spectral basin hopping ===")
    p3_budget = min(500.0, (deadline - time.monotonic()) * 0.30)
    if global_a is not None and p3_budget > 30:
        a_bh, val_bh = spectral_basin_hopping(global_a, S, p3_budget, rng, verbose=True)
        try_update(a_bh, val_bh)
        print("  After BH: obj=%.6f" % global_best)
    
    # Phase 4: Plateau polish
    print("\n=== Phase 4: Plateau polish ===")
    p4_budget = min(300.0, (deadline - time.monotonic()) * 0.20)
    if global_a is not None and p4_budget > 20:
        a_pl, val_pl = plateau_polish(global_a, S, p4_budget, rng, verbose=True)
        try_update(a_pl, val_pl)
        print("  After plateau: obj=%.6f" % global_best)
    
    # Phase 5: Final QP
    print("\n=== Phase 5: Final QP polish ===")
    p5_budget = max(deadline - time.monotonic() - 10, 0)
    if global_a is not None and p5_budget > 20:
        a_fin, val_fin = qp_optimize(global_a, S, p5_budget, rng, verbose=True)
        try_update(a_fin, val_fin)
        print("  After final QP: obj=%.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
