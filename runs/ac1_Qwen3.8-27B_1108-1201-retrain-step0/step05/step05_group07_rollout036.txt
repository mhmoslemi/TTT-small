"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided descent with
subgradient polishing, LSE refinement, and multi-length exploration.
Key innovation: use a linear program at each iteration to find the best
descent direction that keeps cross-convolution below current peak.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, active_lags, n):
    """Build the K x n constraint matrix for LP direction finding."""
    K = len(active_lags)
    row_list = []
    col_list = []
    val_list = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        if length <= 0:
            continue
        row_list.append(np.full(length, i, dtype=np.int32))
        col_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))
        val_list.append(f[j_start:j_end + 1].copy())
    if not row_list:
        return csr_matrix((K, n), dtype=np.float64)
    rows = np.concatenate(row_list)
    cols = np.concatenate(col_list)
    vals = np.concatenate(val_list)
    A = csr_matrix((vals, (rows, cols)), shape=(K, n), dtype=np.float64)
    return A


def solve_lp_direction(f, M, active_lags, n):
    """Solve LP to find direction g maximizing sum(g) s.t. f*g <= M at active lags, g >= 0."""
    K = len(active_lags)
    if K == 0:
        return None
    A = build_lp_matrix(f, active_lags, n)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    ub_val = 2.0 * np.max(f)
    bounds = [(0.0, ub_val)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lp_guided_descent(a0, S, max_time, rng, verbose=True):
    """
    LP-guided descent: at each iteration, solve an LP to find the best
    direction that keeps cross-convolution below current peak.
    """
    n = len(a0)
    f = project_simplex(a0.copy(), S)
    c_ff = fft_convolve_self(f)
    M = c_ff.max()
    best_M = M
    best_f = f.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 100
    K_start = 10
    K_max = 80

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        K = int(K_start + (K_max - K_start) * progress)

        # Get active lags
        top_idx = np.argsort(c_ff)[::-1][:K]
        active_lags = top_idx.tolist()

        # Solve LP for direction
        g = solve_lp_direction(f, M, active_lags, n)

        accepted = False
        if g is not None and g.sum() > 1e-10:
            g = np.maximum(g, 0.0)
            g = g * (S / g.sum())

            # Precompute convolutions for fast line search
            c_fg = fft_convolve(f, g)
            c_gg = fft_convolve_self(g)

            # Line search using fast formula
            for t in [0.2, 0.1, 0.05, 0.02, 0.01]:
                c_approx = (1 - t) ** 2 * c_ff + 2 * t * (1 - t) * c_fg + t ** 2 * c_gg
                if c_approx.max() < M * (1.0 - 1e-7):
                    f_try = np.maximum((1 - t) * f + t * g, 0.0)
                    c_exact = fft_convolve_self(f_try)
                    if c_exact.max() < M:
                        f = f_try
                        c_ff = c_exact
                        M = c_ff.max()
                        accepted = True
                        break

        if not accepted:
            # Fallback: subgradient step
            k_sub = min(K, 20)
            grad = compute_subgradient_topk(f, c_ff, k_sub)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            gap = M - c_ff.mean()
            if gap < 1e-12:
                break
            step = 2.0 * gap / (gnorm * gnorm)
            step = min(step, 3.0 / gnorm)
            for _ in range(8):
                f_try = project_simplex(f - step * grad, S)
                c_try = fft_convolve_self(f_try)
                if c_try.max() < M * (1.0 + 1e-10):
                    f = f_try
                    c_ff = c_try
                    if c_ff.max() < M:
                        M = c_ff.max()
                        accepted = True
                    break
                step *= 0.5
            if not accepted:
                step = 0.001 / gnorm
                f = project_simplex(f - step * grad, S)
                c_ff = fft_convolve_self(f)
                M = c_ff.max()

        # Track best
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        # Noise injection when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.001 * (stuck / patience)) * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            c_ff = fft_convolve_self(f)
            M = c_ff.max()
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_M / (S * S)
            print(f"    LP-descent it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, M={best_M:.6f}")

    return best_f, best_M

def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    return a
