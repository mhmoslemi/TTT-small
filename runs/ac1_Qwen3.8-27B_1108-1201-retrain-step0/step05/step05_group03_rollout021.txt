```python
    """
    Compute gradient of sum_k w[k] * (conv(a,a))[k] with respect to a.
    w has length 2n-1, a has length n.
    Returns gradient of length n.

    #         = 2 * sum_m w[i+m] * a[m]  (substituting m = k-i)
    # This is 2 * cross_correlation(a, w) at lags 0, 1, ..., n-1

    # (a ⋆ w)[k] = sum_m a[m] * w[m+k]
    # FFT(a ⋆ w) = conj(FFT(a)) * FFT(w)  [for circular correlation with padding]

    # But we only need lags 0 to n-1, so padding of 2n-1 should suffice?
    # Actually for circular correlation to give correct linear correlation at lags 0..n-1,
    # we need the wrap-around to not affect those lags.
    # a has support [0, n-1], w has support [0, 2n-2].
    # (a ⋆ w)[k] = sum_m a[m] * w[m+k], nonzero for m in [0,n-1] and m+k in [0, 2n-2]

    # For this to not overlap with [0, n-1], need P-k > n-1, i.e., P > n-1+k
    # For k up to n-1: P > 2n-2. So P >= 2n-1 suffices!
```

```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-method optimization combining:
1. Gerchberg-Saxton spectral shaping (iterative spectrum flattening)
2. Exact subgradient method with Polyak step size
3. LSE-smoothed proximal gradient with Nesterov acceleration
4. Coordinate descent local refinement
5. Multi-n exploration and multi-restart
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective value."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def subgrad_max_conv(a, c, top_frac=0.0, beta=0.0):
    """
    Compute subgradient of max(c) where c = conv(a,a).
    If beta > 0, use LSE smoothing with top-k selection.
    Otherwise use exact subgradient at the argmax.
    """
    n = len(a)
    c_len = len(c)

    if beta > 0:
        # LSE smoothing
        k = max(4, int(top_frac * c_len)) if top_frac > 0 else c_len
        k = min(k, c_len)
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
        cm = c_top.max()
        scaled = beta * (c_top - cm)
        if scaled.max() > 400:
            beta = 400.0 / max(1e-15, (c_top - cm).max())
            scaled = beta * (c_top - cm)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / max(tot, 1e-300)
        w_full = np.zeros(c_len)
        w_full[idx] = w
    else:
        # Exact subgradient: uniform weight on all argmax positions
        cmax = c.max()
        mask = (c >= cmax - 1e-10 * max(1.0, cmax))
        w_full = np.zeros(c_len)
        w_full[mask] = 1.0 / mask.sum()

    fsize = 1 << (c_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def run_spectral_shaping(a_init, S, max_time, rng, relax=0.5):
    """
    Gerchberg-Saxton style spectral shaping.
    Iteratively: flatten spectrum -> IFFT -> clip non-negative -> normalize.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        mag_mean = mag.mean()
        if mag_mean < 1e-20:
            break

        # Target: flatten the magnitude spectrum
        # Use partial relaxation to avoid oscillation
        mag_target = np.full_like(mag, mag_mean)
        mag_new = (1.0 - relax) * mag + relax * mag_target
        mag_new = np.maximum(mag_new, 1e-10)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new, S)

        c = fft_convolve_self(a_new)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try less aggressive relaxation or add noise
            if no_improve > 50:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a_new = project_simplex(a_new + noise, S)
                no_improve = 0

        # Adaptive relaxation
        if no_improve > 20:
            relax = max(0.1, relax * 0.9)
        a = a_new

    return best_a, best

def run_subgradient_polyak(a_init, S, max_time, rng, lower_bound=1.0):
    """
    Subgradient method with Polyak step size.
    Uses exact subgradient of the max function.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    # Polyak step: (f(x) - f*) / ||g||^2
    # f* is lower bound on max(c) with sum constraint = S^2/(2n-1)
    f_star = S * S / (2 * n - 1)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        # Exact subgradient
        grad = subgrad_max_conv(a, c, beta=0.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step size
        gap = cur_max - f_star
        if gap < 1e-10:
            break
        step = gap / (gnorm * gnorm)
        # Cap the step to avoid instability
        step = min(step, 1.0 / gnorm)

        a_new = project_simplex(a - step * grad, S)

        # Check if improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Halve the step
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            a = a_new

        # Periodic perturbation
        if it % 2000 == 0:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best


def run_nesterov_lse(a_init, S, max_time, rng,
                     beta_init=30.0, beta_rate=0.002, beta_max=10000.0,
                     k_frac=0.05):
    """
    Nesterov (FISTA) with LSE-smoothed max, backtracking line search.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    k = max(8, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = subgrad_max_conv(y, c, top_frac=k / len(c), beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        L = 2.0 * beta * 0.1
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + step * step * L / 2.0 + 1e-12:
                break
            step *= 0.5
            L *= 2.0
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Restart when stuck
        if stuck > 800:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_coordinate_descent(a_init, S, max_time, rng, group_size=3):
    """
    Coordinate descent: try modifying small groups of coordinates.
    For each group, find the optimal values that reduce the peak.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        # Pick a random group of coordinates to modify
        group = rng.choice(n, size=min(group_size, n), replace=False)
        group_sorted = np.sort(group)

        # Current values
        old_vals = a[group_sorted].copy()
        old_sum = a.sum()

        # Try different allocations of the sum
        # Keep total sum the same
        total = old_vals.sum()
        if total < 1e-15:
            continue

        # Try: equal, concentrated, random splits
        candidates = []
        # Equal split
        candidates.append(np.full(group_sorted.size, total / group_sorted.size))
        # Concentrated on first element
        conc = np.zeros(group_sorted.size)
        conc[0] = total
        candidates.append(conc)
        # Random split
        w = rng.dirichlet(np.ones(group_sorted.size))
        candidates.append(w * total)
        # Tilted: decrease the one that contributes most to peak
        peak_pos = np.argmax(c)
        contributions = np.array([a[peak_pos - gi] if 0 <= peak_pos - gi < n else 0
                                  + a[gi - (peak_pos - (n-1))] if 0 <= gi - (peak_pos - (n-1)) < n else 0
                                  for gi in group_sorted])
        # Simplified: just try decreasing the largest element
        max_idx_in_group = np.argmax(old_vals)
        dec = old_vals.copy()
        dec[max_idx_in_group] *= 0.5
        dec = dec / dec.sum() * total
        candidates.append(dec)

        best_candidate = None
        best_candidate_max = cur_max

        for cand in candidates:
            test_a = a.copy()
            test_a[group_sorted] = cand
            test_a = normalize(test_a, S)
            c_test = fft_convolve_self(test_a)
            if c_test.max() < best_candidate_max:
                best_candidate_max = c_test.max()
                best_candidate = test_a

        if best_candidate is not None:
            a = best_candidate

        # Also try small random perturbations
        if it % 50 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            pert_a = project_simplex(a + noise, S)
            c_pert = fft_convolve_self(pert_a)
            if c_pert.max() < best:
                best = c_pert.max()
                best_a = pert_a.copy()

    return best_a, best


def run_adaptive_lse(a_init, S, max_time, rng):
    """
    LSE-based optimization with adaptive parameters and momentum.
    Different from previous: uses larger k fraction and slower beta growth.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    # Start with larger k for smoother gradient
    k = max(16, int(0.15 * (2 * n - 1)))
    beta = 10.0
    beta_growth = 1.001

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(20000.0, beta * beta_growth)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = subgrad_max_conv(y, c, top_frac=k / len(c), beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta * 0.1)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Adaptive restart
        if stuck > 500:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 + np.sin(0.01 * stuck))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
            # Also reset beta to explore
            if rng.random() < 0.3:
                beta = max(10.0, beta * 0.5)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final
        best_a = a.copy()

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_high_beta_polish(a_init, S, max_time, rng):
    """
    Very high beta polishing - essentially exact subgradient with tiny steps.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(100000.0, 100.0 * (1.0 + it / 100.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        k = max(4, int(0.01 * (2 * n - 1)))
        grad = subgrad_max_conv(a, c, top_frac=k / len(c), beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_scale / (1.0 + 0.0005 * it)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(1.0, step_scale * 1.01)
        else:
            step_scale = max(0.001, step_scale * 0.5)
            a = a_new  # Still take the step (it might help escape)

        # Occasional tiny perturbation
        if it % 1000 == 0:
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: spectral shaping -> Nesterov LSE -> coordinate descent -> polish.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_optimization(n, a_start, time_limit, phase_desc):
        """Run optimization pipeline for given length."""
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
        if a_start is not None:
            a_start = normalize(a_start, S)

        if a_start is None:
            a_start = np.ones(n) * S / n

        print("\n=== %s: n=%d ===" % (phase_desc, n))
        print("  Time limit: %.0fs" % time_limit)

        best_local = fft_convolve_self(a_start).max()
        best_a_local = a_start.copy()
        obj_start = 2.0 * n * best_local / S**2
        print("  Initial: max_conv=%.6f, obj=%.6f" % (best_local, obj_start))

        # Phase A: Spectral shaping (unique to this approach)
        t_a = time_limit * 0.25
        if t_a > 20 and time.monotonic() < deadline - 30:
            a_opt, val_opt = run_spectral_shaping(a_start, S, min(t_a, 120), rng, relax=0.5)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Spectral shaping 1: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

            # Spectral shaping v2
            remaining_a = deadline - time.monotonic() - 30
            t_a2 = min(t_a * 0.8, remaining_a)
            if t_a2 > 20:
                a_opt, val_opt = run_spectral_shaping_v2(best_a_local, S, t_a2, rng)
                if val_opt < best_local:
                    best_local = val_opt
                    best_a_local = a_opt.copy()
                    print("  Spectral shaping 2: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        # Phase B: Nesterov LSE
        remaining = deadline - time.monotonic() - 30
        t_b = min(time_limit * 0.30, remaining)
        if t_b > 20:
            a_opt, val_opt = run_nesterov_lse(best_a_local, S, t_b, rng,
                                               beta_init=20.0, beta_rate=0.003,
                                               beta_max=15000.0, k_frac=0.08)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Nesterov LSE: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        # Phase C: Subgradient with Polyak step
        remaining = deadline - time.monotonic() - 30
        t_c = min(time_limit * 0.15, remaining)
        if t_c > 15:
            a_opt, val_opt = run_subgradient_polyak(best_a_local, S, t_c, rng)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Subgradient Polyak: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        # Phase D: Coordinate descent
        remaining = deadline - time.monotonic() - 20
        t_d = min(time_limit * 0.15, remaining)
        if t_d > 10:
            a_opt, val_opt = run_coordinate_descent(best_a_local, S, t_d, rng, group_size=2)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Coord descent: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        # Phase E: High-beta polish
        remaining = deadline - time.monotonic() - 15
        t_e = min(time_limit * 0.15, remaining)
        if t_e > 10:
            a_opt, val_opt = run_high_beta_polish(best_a_local, S, t_e, rng)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  High-beta polish: max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        # Phase F: Another round of spectral shaping on the best
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_opt, val_opt = run_spectral_shaping(best_a_local, S, remaining, rng, relax=0.3)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                print("  Spectral shaping (round 2): max_conv=%.6f, obj=%.6f" % (val_opt, 2.0*n*val_opt/S**2))

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Best for n=%d: obj=%.6f" % (n, obj_final))

        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_local.copy()
            print("  *** New global best: %.6f ***" % global_best)

        return best_a_local, best_local

    # Main run: n = 8192
    n_primary = 8192
    remaining = deadline - time.monotonic()
    if remaining > 60:
        try_optimization(n_primary, given, remaining * 0.50, "Main")

    # Try n = 16384
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_sec = 16384
        a_sec = resample_sequence(global_a, n_sec)
        S_sec = np.sqrt(2.0 * n_sec)
        a_sec = normalize(a_sec, S_sec)
        try_optimization(n_sec, a_sec, min(remaining * 0.25, 300), "Secondary n=16384")

    # Try n = 4096 (might find different local minimum)
    remaining = deadline - time.monotonic()
    if remaining > 80 and global_a is not None:
        n_ter = 4096
        a_ter = resample_sequence(global_a, n_ter)
        S_ter = np.sqrt(2.0 * n_ter)
        a_ter = normalize(a_ter, S_ter)
        try_optimization(n_ter, a_ter, min(remaining * 0.15, 200), "Tertiary n=4096")

    # Final intensive polishing on the global best
    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs remaining) ===" % remaining)

        # High-beta Nesterov
        t_fin = remaining * 0.4
        a_fin, val_fin = run_nesterov_lse(global_a, S_best, t_fin, rng,
                                          beta_init=200.0, beta_rate=0.001,
                                          beta_max=50000.0, k_frac=0.03)
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final Nesterov: obj=%.6f" % global_best)

        # High-beta polish
        remaining2 = deadline - time.monotonic() - 15
        if remaining2 > 20:
            a_pol, val_pol = run_high_beta_polish(global_a, S_best, remaining2, rng)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  Final Polish: obj=%.6f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        # Fallback: uniform sequence
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.1fs" % (time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
```
