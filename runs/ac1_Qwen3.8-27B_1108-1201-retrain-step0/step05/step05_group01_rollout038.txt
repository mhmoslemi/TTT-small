import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient focusing on largest convolution values."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def generate_freq_init(n, rng, S, n_harmonics=5):
    """
    Generate initialization using structured frequency-domain construction.
    Creates a sequence with controlled spectral content.
    """
    # Start with flat spectrum (white noise) and add structure
    F = np.zeros(2 * n - 1, dtype=np.complex128)
    # DC component
    F[0] = S
    # Add harmonics with random phases
    for k in range(1, min(n_harmonics + 1, n // 2)):
        amplitude = S / n * rng.uniform(0.3, 1.0) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        F[k] = amplitude * np.exp(1j * phase)
        F[2 * n - 1 - k] = np.conj(F[k])

    a = np.real(ifft(F))[:n]
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 1e-10:
        a = a * (S / a_sum)
    else:
        a = np.ones(n) * S / n
    return a

        # Try n = n_main * 2 if time allows
        n_big = n_main * 2
        rem4b = p4_budget - (time.monotonic() - t4)
        if rem4b > 30 and n_big <= 32768:
            S_big = np.sqrt(2.0 * n_big)
            # Upsample: interpolate
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(global_best_a), endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = interp1d(x_old, global_best_a, kind='cubic', fill_value='extrapolate')(x_new)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big * (S_big / a_big.sum())

            a4c, _ = run_fista_adaptive(a_big, rem4b * 0.5, rng,
                                         beta_start=15.0, beta_max=15000.0,
                                         step_scale=1.0, mode="lse",
                                         restart_interval=3000, noise_type="spectral")
            obj4c = compute_obj(a4c)
            print("  n=%d FISTA: obj=%.8f (t=%.0fs)" % (n_big, obj4c, time.monotonic() - t4))
            if obj4c < global_best_obj:
                global_best_obj = obj4c
                global_best_a = a4c.copy()

    # === Phase 5: Active-set final polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Active-set polish v2 (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, _ = run_active_polish_v2(global_best_a, p5_budget, rng)
        obj5 = compute_obj(a5)
        print("  Active polish: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish_v2(global_best_a, remaining * 0.85, rng, n_levels=4)
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: One more FISTA pass if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Final FISTA pass (budget %.0fs) ===" % remaining)
        t7 = time.monotonic()
        a7, _ = run_fista_adaptive(global_best_a, remaining * 0.9, rng,
                                    beta_start=100.0, beta_max=50000.0,
                                    step_scale=0.3, mode="hybrid",
                                    restart_interval=4000, noise_type="spectral")
        obj7 = compute_obj(a7)
        print("  Final FISTA: obj=%.8f (t=%.0fs)" % (obj7, time.monotonic() - t7))
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

    # Final verification and return
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Ensure validity
        if len(final_list) < 1 or sum(final_list) < 0.01:
            n_fb = 8192
            final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            fb_val = np.sqrt(2 * n_fb) / n_fb
            final_list = [float(fb_val)] * n_fb
    else:
        n_fb = 8192
        fb_val = np.sqrt(2 * n_fb) / n_fb
        final_list = [float(fb_val)] * n_fb

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
