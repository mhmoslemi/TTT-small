"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a hybrid approach:
1. Multiplicative weight (mirror descent) updates for fast convergence
2. LP-guided direction finding via CVXPY for principled steps
3. LSE-smoothed gradient polishing for final convergence
4. Multi-length exploration and coordinate descent refinement
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_topk_gradient(a, c, k):
    """Compute weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_mean = c.mean()
    weights = np.maximum(c[top_idx] - c_mean, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def mirror_descent_optimize(a0, S, max_time, rng, verbose=True, topk=15):
    """
    Mirror descent (multiplicative weight) optimization.
    Uses exp(-eta * grad) updates normalized to maintain sum.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-12)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    eta = 1.0 / (5.0 * np.sqrt(n))
    stuck = 0
    patience = 300

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        k = min(topk + it // 200, 60)
        grad = compute_topk_gradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Multiplicative update with line search
        accepted = False
        for backtrack in range(8):
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new * (S / a_new.sum())
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max * (1.0 + 1e-9):
                a = a_new
                accepted = True
                break
            eta *= 0.5

        if accepted:
            eta = min(eta * 1.03, 2.0)
        else:
            eta = max(eta * 0.1, 1e-8)
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-15)
            a = a_new * (S / a_new.sum())

        # Noise when stuck
        if stuck > patience:
            noise_scale = min(0.05, 0.005 * (stuck / patience))
            a = a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"  MD it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, eta={eta:.4e}")

    return best_a, best_max


def lp_direction_step(a, S, max_time_per_lp=30.0):
    """
    Solve LP to find best direction g:
    max sum(b) s.t. b >= 0, (a*b)[k] <= M for top-k lags.
    Returns rescaled direction or None if LP fails.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    top_k_idx = np.argsort(c)[::-1][:25]

    b = cp.Variable(n, nonneg=True)
    constraints = []

    for kidx in top_k_idx:
        kidx = int(kidx)
        i_start = max(0, kidx - n + 1)
        i_end = min(n - 1, kidx)
        if i_start > i_end:
            continue
        b_indices = kidx - np.arange(i_start, i_end + 1)
        a_vals = a[i_start:i_end + 1]
        constraints.append(a_vals @ b[b_indices] <= M)

    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    start_lp = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, max_time=max_time_per_lp)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_time=max_time_per_lp)
        except Exception:
            return None

    if time.monotonic() - start_lp > max_time_per_lp + 5:
        return None

    if b.value is None:
        return None

    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-10:
        return None

    direction = b_val * (S / b_val.sum())
    return direction


def lp_guided_optimize(a0, S, max_time, rng, verbose=True):
    """
    Optimization using LP-guided direction finding with multiplicative updates between.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    lp_interval = 50  # Solve LP every 50 sub-steps
    sub_step = 0
    t_mix = 0.1  # Mixing parameter for LP direction

    while time.monotonic() - start < max_time:
        it += 1

        # Periodically solve LP for direction
        if sub_step % lp_interval == 0 and it % 50 == 0:
            direction = lp_direction_step(a, S, max_time_per_lp=20.0)
            if direction is not None:
                # Mix current a with LP direction
                a_new = (1.0 - t_mix) * a + t_mix * direction
                a_new = project_simplex(a_new, S)
                a_new = np.maximum(a_new, 1e-12)
                a_new = a_new * (S / a_new.sum())
                c_new = fft_convolve_self(a_new)
                if c_new.max() < best_max:
                    a = a_new
                    best_max = c_new.max()
                    best_a = a.copy()
                    print(f"  LP step {it}: obj={2.0*n*best_max/(S*S):.6f}")
                # Animate t_mix
                t_mix = min(0.3, t_mix * 1.1)
                sub_step = 0
                continue

        # Normal mirror descent step
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()

        k = min(15 + it // 100, 50)
        grad = compute_topk_gradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        eta = 1.0 / (8.0 * np.sqrt(n))
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new * (S / a_new.sum())
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max * (1.0 + 1e-9):
            a = a_new
        else:
            a_new = a * np.exp(-0.3 * eta * grad)
            a_new = np.maximum(a_new, 1e-15)
            a = a_new * (S / a_new.sum())

        sub_step += 1

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"  LP-G it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best_max


def lse_polish(a0, S, max_time, rng, beta_init=80.0, beta_max=15000.0):
    """LSE-smoothed gradient polish."""
    n = len(a0)
    a = project_simplex(a0, S)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(0.005 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 400:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - (time.monotonic() - start) / max_time)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        step = min(step, 3.0 / gnorm)

        # Line search
        accepted = False
        for _ in range(6):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - 0.01 * step * grad, S)

    return best_a, best_max


def coordinate_descent_polish(a0, S, max_time, rng, verbose=True):
    """
    Coordinate descent: for each coordinate j, find optimal a[j]
    that minimizes the max convolution value.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    idx_order = rng.permutation(n)
    pos_in_order = 0

    while time.monotonic() - start < max_time:
        j = idx_order[pos_in_order % n]
        pos_in_order += 1

        # Try to find better a[j] by ternary search on [0, 2*a[j]+S/n]
        aj = a[j]
        lo = 0.0
        hi = 2.0 * aj + S / n

        if hi - lo < 1e-12:
            continue

        # Quick evaluation: compute effect on max convolution
        def eval_aj(val):
            a[j] = val
            # Renormalize: subtract (val - aj) from others proportionally
            delta = val - aj
            if abs(delta) < 1e-15:
                c_tmp = fft_convolve_self(a)
                a[j] = aj
                return c_tmp.max()
            # Instead of renormalizing, just check the raw max
            # (we'll renormalize at the end)
            c_tmp = fft_convolve_self(a)
            a[j] = aj
            return c_tmp.max()

        # Simpler: just try a few candidate values
        candidates = [0.0, aj * 0.5, aj * 1.5, aj * 0.8, aj * 1.2, S / n]
        best_val = aj
        best_c = eval_aj(aj)

        for cv in candidates:
            cv = max(0.0, min(cv, hi))
            cval = eval_aj(cv)
            if cval < best_c:
                best_c = cval
                best_val = cv

        if best_val != aj:
            a[j] = best_val
            # Renormalize
            S_cur = a.sum()
            if S_cur > 1e-10:
                a = a * (S / S_cur)
            c_new = fft_convolve_self(a)
            if c_new.max() < best_max:
                best_max = c_new.max()
                best_a = a.copy()

        if verbose and pos_in_order % (n * 5) == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"  CD it={pos_in_order}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best_max


def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    # Use a shape that's known to work well: raised cosine with perturbation
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    a += rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    return a

            if time.monotonic() > deadline - 10:
                break
            # Different perturbation strategies
            if ri % 3 == 0:
                # Small Gaussian noise
                noise_scale = 0.01 * S5 / np.sqrt(n5)
                a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n5), S5)
            elif ri % 3 == 1:
                # Multiplicative noise
                a_pert = global_a * np.exp(rng.normal(0, 0.02, n5))
                a_pert = np.maximum(a_pert, 0)
                a_pert = project_simplex(a_pert, S5)
            else:
                # Swap mass between random pairs
                a_pert = global_a.copy()
                for _ in range(n5 // 10):
                    i, j = rng.integers(0, n5, 2)
                    amt = a_pert[i] * 0.1
                    a_pert[i] -= amt
                    a_pert[j] += amt
                a_pert = np.maximum(a_pert, 0)
                a_pert = project_simplex(a_pert, S5)

            a_res, _ = mirror_descent_optimize(a_pert, S5, t_per, rng, topk=20, verbose=False)
            update_global(a_res, f"Phase 5 restart {ri}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, internal_obj={global_best:.8f}")
    print(f"evaluate_sequence: {ver:.8f}")
    print(f"{'='*60}")

    return final_list
