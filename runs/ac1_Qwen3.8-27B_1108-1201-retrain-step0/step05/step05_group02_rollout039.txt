Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-strategy optimization: LSE-FISTA with slow beta annealing,
exact subgradient steps, and targeted peak shaving.
"""
import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_grad(c, beta):
    """Compute LSE value and weights for log-sum-exp of c with temperature beta."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:

        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    return lse_val, w

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LSE-FISTA with slow beta annealing, exact subgradient steps,
and multi-restart strategy to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute LSE weights (softmax of beta*(c - max(c)))."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    return w


def conv_gradient(a, w):
    """
    Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a).
    a has length n, w has length 2n-1.
    Result: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    """
    n = len(a)
    L = len(w)  # should be 2n-1
    # Use reversal trick: g[j] = conv(w_rev, a)[L-1-j]
    w_rev = w[::-1]
    N = 1 << (L + n - 2).bit_length()
    W = fft(w_rev, N)
    A = fft(a, N)
    conv_full = np.real(ifft(W * A))[:L + n - 1]
    # g[j] = conv_full[L-1-j] for j=0,...,n-1
    # That's conv_full[L-1], conv_full[L-2], ..., conv_full[L-n]
    g = conv_full[L - n:L][::-1]
    return 2.0 * g


def run_fista(a_init, max_time, rng, beta_start=30.0, beta_end=50000.0,
              total_iters=5000, step_init=None, use_backtrack=True):
    """FISTA with annealed beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()

    y = a.copy()
    t_mom = 1.0
    if step_init is None:
        step = 1.0 / max(beta_start, 1.0)
    else:
        step = step_init

    last_improve_it = 0
    it = 0

    while time.monotonic() - start < max_time and it < total_iters:
        it += 1
        # Compute beta (exponential schedule)
        frac = it / max(1, total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)

        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve_it = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            # Use LSE as proxy for objective in backtracking (cheaper)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            bt = 0
            while obj_trial > cur_obj + 1e-10 and step_local > 1e-16 and bt < 6:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                c_trial = fft_convolve_self(x_trial)
                obj_trial = 2.0 * n * c_trial.max() / (S * S)
                bt += 1
            step = step_local
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)
            # Slow step increase
            step = min(step * 1.05, 3.0 / max(beta * 0.01, 1.0))

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 200 == 199 or (it - last_improve_it) > 500:
            t_mom = 1.0
            y = a.copy()
            # Add small noise if stuck
            if it - last_improve_it > 300:
                noise_scale = max(1e-6, 0.002 * S / np.sqrt(n))
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def run_subgradient(a_init, max_time, rng):
    """Exact subgradient method with adaptive k and step decay."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    step_base = S / (np.sqrt(n) * 0.5)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 8000:
            break

        # Adaptive k: start small (exact), grow when stuck
        if stuck < 20:
            k = 1
        elif stuck < 100:
            k = 3
        elif stuck < 500:
            k = min(20, 3 + (stuck - 100) // 20)
        elif stuck < 2000:
            k = min(100, 20 + (stuck - 500) // 20)
        else:
            k = min(500, 100 + (stuck - 2000) // 50)

        k = min(k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k
        grad = conv_gradient(a, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: classical subgradient with decay
        step = step_base / (1.0 + 0.005 * it) ** 0.5
        step = min(step, 3.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)

        if obj_new <= cur_obj + 1e-12:
            a = a_new
        else:
            # Try smaller steps
            accepted = False
            for shrink in [0.5, 0.1, 0.01]:
                a_try = project_simplex(a - step * shrink * grad, S)
                if compute_obj(a_try) <= cur_obj + 1e-12:
                    a = a_try
                    accepted = True
                    break
            if not accepted and stuck > 500:
                # Add noise to escape
                ns = max(1e-5, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
                a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def run_hybrid(a_init, max_time, rng):
    """
    Hybrid: alternate between LSE-FISTA and exact subgradient steps.
    Every 100 FISTA iterations, do 5 subgradient steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    it = 0

    y = a.copy()
    t_mom = 1.0
    step = 0.8 / 30.0
    beta = 30.0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Every 50 iterations, do a subgradient step
        if it % 50 == 0:
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / (S * S)
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

            stuck = it - last_improve
            k = min(5 + stuck // 100, 50)
            top_idx = np.argsort(c)[::-1][:k]
            w = np.zeros(len(c))
            w[top_idx] = 1.0 / k
            grad = conv_gradient(a, w)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                sg_step = min(S / (gnorm * (1.0 + 0.01 * it)), 2.0 * S / gnorm)
                a_new = project_simplex(a - sg_step * grad, S)
                if compute_obj(a_new) <= cur_obj + 1e-12:
                    a = a_new
                    y = a.copy()
                    t_mom = 1.0
            continue

        # FISTA step with LSE
        frac = it / 5000.0
        beta = 30.0 * 1000.0 ** frac

        c = fft_convolve_self(y)
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)

        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        c_trial = fft_convolve_self(x_trial)
        obj_trial = 2.0 * n * c_trial.max() / (S * S)
        bt = 0
        while obj_trial > cur_obj + 1e-10 and step_local > 1e-16 and bt < 5:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            bt += 1
        step = step_local
        x_new = x_trial

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it % 100 == 99:
            t_mom = 1.0
            y = a.copy()

    return best_a, best_obj


def coordinate_polish(a_init, max_time, rng):
    """
    Coordinate-wise polish: for each coordinate, find the optimal value
    via 1D search, considering the effect on the max convolution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0

    # Precompute the "influence" of each coordinate on each lag
    # For coordinate j, it affects c[k] for k in [j, j+n-1] with coefficient a[k-j]
    # and for k in [j-n+1, j] with coefficient a[j-k] (if valid)

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 10000:
            break

        # Process coordinates in order of "importance" (contribution to max)
        k_star = int(np.argmax(c))
        # Find which coordinates contribute most to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        if hi >= lo:
            contrib[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]

        # Process top contributors first
        order = np.argsort(-contrib)
        if np.all(contrib < 1e-10):
            order = np.arange(n)

        for j in order[:min(n, 200)]:
            if time.monotonic() - start >= max_time:
                break
            if a[j] < 1e-12:
                continue

            # Try small perturbations of a[j]
            # The effect on c: c[k] changes by delta * (a[k-j] + a[j-k]) for affected k
            # We want to reduce the max, so try reducing a[j] if it contributes to the peak
            # and adding to a coordinate that doesn't

            delta = a[j] * 0.01
            if delta < 1e-12:
                continue

            # Try moving mass to a few candidates
            # Pick target: coordinate with low contribution to all top peaks
            top_k = min(5, 2 * n - 1)
            top_idx = np.argsort(c)[::-1][:top_k]

            best_target = -1
            best_obj_trial = cur_obj + 1e-15

            # Try a few random targets and the coordinate far from peak
            candidates = list(rng.choice(n, size=min(10, n), replace=False))
            # Also try the coordinate that minimizes contribution to all top peaks
            target_scores = np.zeros(n)
            for kk in top_idx:
                lo2 = max(0, int(kk) - n + 1)
                hi2 = min(n - 1, int(kk))
                if hi2 >= lo2:
                    target_scores[lo2:hi2 + 1] += a[int(kk) - hi2:int(kk) - lo2 + 1][::-1]
            low_contrib = np.argsort(target_scores)[:5]
            candidates.extend(low_contrib.tolist())

            for l in set(candidates):
                if l == j:
                    continue
                trial = a.copy()
                trial[j] -= delta
                trial[l] += delta
                if trial[j] < 0:
                    continue
                # Quick estimate of new max
                # c changes: for k affected by j: c[k] -= delta * (a[k-j] + a[j-k])
                # For k affected by l: c[k] += delta * (a[k-l] + a[j-k])... 
                # Just recompute c for the trial (expensive but accurate)
                trial_c = fft_convolve_self(trial)
                trial_obj = 2.0 * n * trial_c.max() / (S * S)
                if trial_obj < best_obj_trial:
                    best_obj_trial = trial_obj
                    best_target = l

            if best_target >= 0 and best_obj_trial < cur_obj - 1e-13:
                a[j] -= delta
                a[best_target] += delta
                c = fft_convolve_self(a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-strategy optimization for minimizing 2*n*max_conv/sum^2.
    Combines LSE-FISTA with beta annealing, hybrid subgradient-FISTA,
    coordinate polish, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    # Load starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = np.ones(n) * np.sqrt(2.0 / n)

    n = len(a_start)
    S = np.sqrt(2.0 * n)
    a_start = a_start * (S / a_start.sum())

    global_best_obj = compute_obj(a_start)
    global_best_a = a_start.copy()
    print("Initial: n=%d, obj=%.8f" % (n, global_best_obj))

    # === Phase 1: LSE-FISTA with moderate-to-high beta (35% time) ===
    t1 = min(budget_s * 0.35, deadline - time.monotonic())
    if t1 > 60:
        print("\n=== Phase 1: LSE-FISTA annealing (%.0fs) ===" % t1)
        a_opt, obj_opt = run_fista(
            global_best_a, t1, rng,
            beta_start=30, beta_end=30000,
            total_iters=10000, step_init=0.05
        )
        print("  Result: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")

    # === Phase 2: Hybrid FISTA+subgradient (25% time) ===
    t2 = min(budget_s * 0.25, deadline - time.monotonic())
    if t2 > 30:
        print("\n=== Phase 2: Hybrid (%.0fs) ===" % t2)
        a_opt, obj_opt = run_hybrid(global_best_a, t2, rng)
        print("  Result: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")

    # === Phase 3: Exact subgradient (15% time) ===
    t3 = min(budget_s * 0.15, deadline - time.monotonic())
    if t3 > 20:
        print("\n=== Phase 3: Subgradient (%.0fs) ===" % t3)
        a_opt, obj_opt = run_subgradient(global_best_a, t3, rng)
        print("  Result: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")

    # === Phase 4: Coordinate polish (10% time) ===
    t4 = min(budget_s * 0.10, deadline - time.monotonic())
    if t4 > 15:
        print("\n=== Phase 4: Coordinate polish (%.0fs) ===" % t4)
        a_opt, obj_opt = coordinate_polish(global_best_a, t4, rng)
        print("  Result: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")

    # === Phase 5: High-beta FISTA for fine tuning (10% time) ===
    t5 = min(budget_s * 0.10, deadline - time.monotonic())
    if t5 > 15:
        print("\n=== Phase 5: High-beta FISTA (%.0fs) ===" % t5)
        a_opt, obj_opt = run_fista(
            global_best_a, t5, rng,
            beta_start=5000, beta_end=200000,
            total_iters=5000, step_init=0.01
        )
        print("  Result: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")

    # === Phase 6: Multi-restart (5% time) ===
    t6 = min(budget_s * 0.05, deadline - time.monotonic())
    if t6 > 10:
        print("\n=== Phase 6: Multi-restart (%.0fs) ===" % t6)
        n_restarts = 3
        per_restart = t6 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() >= deadline:
                break
            # Perturb the best solution
            pert_scale = [0.001, 0.005, 0.01][r] if r < 3 else 0.01
            perturbed = global_best_a + rng.normal(0, pert_scale * S / np.sqrt(n), n)
            perturbed = project_simplex(perturbed, S)
            a_opt, obj_opt = run_fista(
                perturbed, per_restart, rng,
                beta_start=100, beta_end=50000,
                total_iters=2000, step_init=0.03
            )
            print("  Restart %d (scale=%.4f): obj=%.8f" % (r + 1, pert_scale, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")

    # Final verification
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
    else:
        print("WARNING: invalid result, using fallback")
        n_fb = 8192
        final_list = [float(np.sqrt(2.0 / n_fb))] * n_fb

    return final_list
