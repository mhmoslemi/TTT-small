"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Multigrid-style optimization with spectral flattening and peak shaving.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def topk_lse_value_and_weights(c, k, beta):
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if np.any(scaled > 500):
        max_s = scaled.max()
        if max_s > 500:
            beta = 500.0 / max_s
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    return w, idx

def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    w, idx = topk_lse_value_and_weights(c, k, beta)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

def spectral_flatten_step(a, alpha):
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    # Compress the power spectrum toward the mean
    new_power = (1.0 - alpha) * power + alpha * mean_p
    new_power = np.maximum(new_power, 1e-30)
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)

def spectral_perturb(a, amount, rng):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())

def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def run_fista(a_init, S, max_time, rng, beta_init, beta_rate, beta_max, k_frac, use_momentum):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_new = fft_convolve_self(x_new)

def run_multi_scale(a_init, S, max_time, rng):
    """Multigrid-style optimization: downsample, optimize, upsample, refine."""
    n = len(a_init)
    start = time.monotonic()

    best_a = a_init.copy()
    
    # Create a hierarchy of scales
    scales = []
    n_cur = n
    while n_cur > 128:
        scales.append(n_cur)
        n_cur = max(128, n_cur // 2)
    scales.append(n_cur)
    scales = sorted(set(scales))  # from smallest to largest
    
    # Downsample to the coarsest level
    coarse = resample_sequence(a_init, scales[0])
    S_coarse = coarse.sum()
    
    for i, n_level in enumerate(scales):
        time_remaining = max_time - (time.monotonic() - start)
        if time_remaining < 10:
            break
        
        # Allocate time proportionally
        if i == len(scales) - 1:
            time_level = time_remaining * 0.5  # Give more time to the finest level
        else:
            time_level = time_remaining * 0.3 / (len(scales) - i)
        
        if i == 0:
            a_cur = coarse
            S_cur = S_coarse
        else:
            a_cur = resample_sequence(best_a, n_level)
            S_cur = a_cur.sum()
            if S_cur < 1e-15:
                a_cur = np.ones(n_level) * S / n_level
                S_cur = S
        
        a_cur = normalize(a_cur, S_cur)
        
        # Run FISTA at this level
        a_opt, val_opt = run_fista(a_cur, S_cur, time_level, rng,
                                    beta_init=20.0, beta_rate=0.003,
                                    beta_max=3000.0, k_frac=0.1, use_momentum=True)

                if val < fft_convolve_self(a_opt).max():
                    pass

                best_a = a_opt.copy()
        else:

            best_a = a_opt  # Will be resampled to next level
    
    return best_a, best

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Multi-scale optimization with diverse initializations, adaptive LSE,
spectral flattening, and peak shaving. Primary focus on n=16384 for better
scaling, with n=8192 refinement from parent.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_weights(c, k, beta):
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_s = scaled.max()
    if max_s > 500:
        beta = 500.0 / max(max_s, 1e-15)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    return w, idx


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    w, idx = topk_lse_weights(c, k, beta)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_flatten_step(a, alpha):
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    new_power = (1.0 - alpha) * power + alpha * mean_p
    new_power = np.maximum(new_power, 1e-30)
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng, beta_init, beta_rate, beta_max, k_frac, use_momentum, k_decrease):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        k = max(4, int(k_frac * (2 * n - 1) * k_decrease))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(6):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        if it % 1500 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.06, rng)
            t_val = 1.0
            stuck = 0

        if it % 1000 == 0:
            k_frac_eff = max(0.01, k_frac * (1.0 - 0.5 * it / max(1, int(max_time / 10))))
            k_decrease = max(0.2, 1.0 - 0.6 * it / max(1, int(max_time / 5)))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_adam(a_init, S, max_time, rng, lr_init=0.01, beta_adam=40.0, beta_rate=0.008):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    k = max(4, int(0.06 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = min(3000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.6 * progress)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        if t % 4000 == 0 and rng.random() < 0.12:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

        if t % 5000 == 0 and rng.random() < 0.08:
            a = spectral_flatten_step(a, 0.04)

    return best_a, best


def run_polish(a_init, S, max_time, rng):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    k = max(4, int(0.02 * (2 * n - 1)))
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(80000.0, 300.0 * (1.0 + it / 300.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        base_step = 0.3 / (1.0 + 0.0005 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.03)
        else:
            step_scale = max(0.01, step_scale * 0.4)
            a_new = project_simplex(a - step * grad, S)
            a = a_new

        if it % 3000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_multiplicative(a_init, S, max_time, rng):
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    k = max(4, int(0.05 * (2 * n - 1)))
    it = 0
    eta = 1.0 / (2.0 * 30.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(8000.0, 25.0 * (1.0 + it / 150.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        update = 1.0 - eta * grad
        update = np.maximum(update, 0.005)
        a_new = a * update
        a = project_simplex(a_new, S)

        eta = 1.0 / (2.0 * (25.0 + it * 0.03))

        if it % 2000 == 0 and rng.random() < 0.1:
            a = spectral_flatten_step(a, 0.03)

    return best_a, best


def run_peak_direct(a_init, S, max_time, rng, n_peaks=8):
    """Directly target the top peaks with focused gradient steps."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = min(20000.0, 100.0 * (1.0 + it / 100.0))
        k = n_peaks
        grad = topk_lse_gradient(a, c, k, beta)

        # Multiple step attempts with varying step sizes
        best_step_val = cur_max
        best_a_try = a
        for step_frac in [1.0, 0.5, 0.25, 0.1]:
            step = step_frac / max(1e-10, np.linalg.norm(grad))
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_step_val:
                best_step_val = c_try.max()
                best_a_try = a_try

        a = best_a_try

        if it % 500 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.0008 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best


def multigrid_optimize(a_init, S, max_time, rng):
    """Optimize from coarse to fine scales."""
    n = len(a_init)
    start = time.monotonic()
    best = fft_convolve_self(a_init).max()
    best_a = a_init.copy()

    # Build scale hierarchy
    scales = []
    n_cur = n
    while n_cur > 256:
        scales.append(n_cur)
        n_cur = n_cur // 2
    scales.append(n_cur)
    scales = sorted(set(scales))

    # Start from coarsest
    a_cur = resample_sequence(a_init, scales[0])
    S_cur = a_cur.sum()

    for i, n_level in enumerate(scales):
        time_remaining = max_time - (time.monotonic() - start)
        if time_remaining < 15:
            break

        if i == 0:
            a_level = a_cur
        else:
            a_level = resample_sequence(best_a, n_level)
            S_level = a_level.sum()
            if S_level < 1e-15:
                a_level = np.ones(n_level) * S / n_level
                S_level = S
            a_level = normalize(a_level, S_level)

        # Allocate
        if i == len(scales) - 1:
            t_alloc = time_remaining * 0.55
        else:
            t_alloc = time_remaining * 0.25 / (len(scales) - i)

        t_alloc = max(5.0, t_alloc)

        # FISTA at this level
        a_opt, val_opt = run_fista(a_level, S_level, t_alloc, rng,
                                   beta_init=15.0, beta_rate=0.004,
                                   beta_max=5000.0, k_frac=0.1,
                                   use_momentum=True, k_decrease=1.0)

        # Quick spectral refinement
        if time.monotonic() < start + max_time - 10:
            for alpha in [0.04, 0.06]:
                a_flat = spectral_flatten_step(a_opt, alpha)
                v_flat = fft_convolve_self(a_flat).max()
                if v_flat < val_opt:
                    a_opt = a_flat
                    val_opt = v_flat

        if n_level == n:
            if val_opt < best:
                best = val_opt
                best_a = a_opt.copy()
        else:
            best_a = a_opt

    return best_a, best


def make_initialization(n, kind, S, rng):
    x = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n) * S / n
    elif kind == "sine_env":
        a = 0.5 + 0.5 * np.sin(np.pi * x)
        a = normalize(a, S)
    elif kind == "tukey":
        r = 0.4
        a = np.ones(n)
        m = int(r * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            a[:m] = taper
            a[-m:] = taper[::-1]
        a = normalize(a, S)
    elif kind == "gaussian":
        xc = x - 0.5
        a = np.exp(-3.0 * xc * xc)
        a = normalize(a, S)
    elif kind == "two_peak":
        a = np.exp(-6.0 * (x - 0.25) ** 2) + np.exp(-6.0 * (x - 0.75) ** 2)
        a = normalize(a, S)
    elif kind == "cosine":
        a = np.cos(np.pi * x) ** 2 + 0.1
        a = normalize(a, S)
    elif kind == "random_smooth":
        a_raw = rng.random(n)
        # Smooth it
        fsize = 1 << (n - 1).bit_length()
        A = fft(a_raw, fsize)
        # Low-pass filter
        freqs = np.fft.fftfreq(fsize)
        cutoff = 3.0 / n
        A[np.abs(freqs) > cutoff] *= 0.1
        a = np.real(ifft(A, fsize))[:n]
        a = np.clip(a, 0, None)
        a = normalize(a, S)
    else:
        a = np.ones(n) * S / n
    return a

                global_a = a_opt.copy()
                print("  Phase2a FISTA: obj=%.6f" % obj)

    else:

    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
