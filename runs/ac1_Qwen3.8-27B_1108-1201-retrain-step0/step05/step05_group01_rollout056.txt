"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses spectral flattening (Gerchberg-Saxton), FISTA with LSE, and L-BFGS-B polishing.
Key innovation: frequency-domain spectral flattening to find near-flat autocorrelation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_flatten(a_init, max_iter=10000, N=None, alpha=1.0, rng=None,
                     check_interval=200):
    """
    Gerchberg-Saxton style spectral flattening.
    Alternates between:
      1. Flattening the power spectrum in frequency domain
      2. Enforcing non-negativity in time domain
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a_init)
    if N is None:
        N = 1 << (2 * n - 1 - 1).bit_length()
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    flat_mag = S / np.sqrt(N)
    
    for it in range(max_iter):
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Blend current magnitude with flat target
        new_mag = alpha * flat_mag + (1.0 - alpha) * mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        # Convergence check
        if it > 10 and np.max(np.abs(a_new - a)) < 1e-13 * S:
            a = a_new
            break
        
        a = a_new
        
        # Periodic objective check
        if it % check_interval == 0:
            c = fft_convolve_self(a)
            obj = 2.0 * n * c.max() / S**2
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    # Final check
    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / S**2
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def spectral_flatten_weighted(a_init, max_iter=10000, N=None, alpha=1.0,
                              weight_power=0.5, rng=None, check_interval=200):
    """
    Weighted spectral flattening: instead of perfectly flat, use a weight
    that emphasizes frequencies where the current spectrum is below target.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a_init)
    if N is None:
        N = 1 << (2 * n - 1 - 1).bit_length()
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    flat_mag = S / np.sqrt(N)
    
    for it in range(max_iter):
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Weighted target: boost frequencies where mag < flat_mag
        ratio = flat_mag / (mag + 1e-30)
        weight = np.power(ratio, weight_power)
        weight = weight / weight.sum() * N  # normalize
        
        new_mag = alpha * weight + (1.0 - alpha) * mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        if it > 10 and np.max(np.abs(a_new - a)) < 1e-13 * S:
            a = a_new
            break
        
        a = a_new
        
        if it % check_interval == 0:
            c = fft_convolve_self(a)
            obj = 2.0 * n * c.max() / S**2
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / S**2
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_start=10.0, beta_end=None,
              mode="lse", step_scale=1.0):
    """FISTA on the LSE-smoothed max of convolution."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    if beta_end is None:
        beta_end = beta_start * 20
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Linearly ramp beta
        frac = min(1.0, it / (2 * max_iter_estimate(max_time)))
        beta = beta_start + frac * (beta_end - beta_start)
        beta = max(5.0, min(beta, 100000.0))
        step = step_scale / max(beta, 1.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Choose gradient
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 500:
                k = max(2, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:  # hybrid
            if stuck_count > 500 and (it % 5 == 0 or rng.random() < 0.5):
                tol = min(0.05, 0.001 * stuck_count)
                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # DCT noise when stuck
        if stuck_count > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                noise_time = rng.normal(0, 1.0, n)
                F = fft(noise_time)
                cutoff = max(5, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0
                stuck_count = 0
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if it % 10000 == 0:
            t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def max_iter_estimate(max_time):
    """Estimate max iterations based on time budget."""
    # Rough estimate: ~500 iterations per second for n=8192
    return max(100, int(max_time * 500))


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=4):
    """L-BFGS-B polishing with penalty for sum constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    betas = [30.0, 80.0, 200.0, 500.0, 1000.0][:n_beta_levels]
    penalties = [10.0, 20.0, 50.0, 100.0, 200.0][:n_beta_levels]
    
    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad
        
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )
        
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def active_set_polish(a_init, max_time, rng):
    """Active-set style polishing with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        if stuck < 200:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        elif stuck < 1000:
            grad = topk_gradient(y, c, k=4, power=1.0)
        elif stuck < 5000:
            k = min(32, 4 + stuck // 200)
            grad = topk_gradient(y, c, k=k, power=1.0)
        else:
            k = min(128, 32 + stuck // 500)
            grad = topk_gradient(y, c, k=k, power=0.8)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 2.0 * S / (gnorm + 1e-15))
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: Gerchberg-Saxton spectral flattening combined with FISTA and L-BFGS-B.
The spectral flattening exploits the Wiener-Khinchin theorem: minimizing peak
autocorrelation equals flattening the power spectrum subject to non-negativity.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Softmax weights for LSE gradient of max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient focusing on largest convolution values."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_flatten(a_init, max_iter, N, alpha, rng, check_interval=500):
    """
    Gerchberg-Saxton spectral flattening.
    Alternates: flatten power spectrum (freq domain) + enforce non-negativity (time domain).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_obj = compute_obj(a)
    flat_mag = S / np.sqrt(N)

    for it in range(max_iter):
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        new_mag = alpha * flat_mag + (1.0 - alpha) * mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)

        if it > 20 and np.max(np.abs(a_new - a)) < 1e-14 * S:
            a = a_new
            break

        a = a_new

        if it % check_interval == 0:
            c = fft_convolve_self(a)
            obj = 2.0 * n * c.max() / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / (S * S)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj

        # Beta ramping
        frac = min(1.0, it / max(1, int(max_time * 300)))
        beta = beta_start + frac * (beta_end - beta_start)
        beta = max(5.0, min(beta, 100000.0))
        step = 1.0 / max(beta, 1.0)

        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)

        # When stuck, use top-k gradient
        if stuck_count > 3000:
            k = max(4, min(128, stuck_count // 50))
            grad = topk_gradient(y, c, k, power=1.2)
            step = S / (np.linalg.norm(grad) + 1e-15)
            step = min(step, 2.0 * S / (np.linalg.norm(grad) + 1e-15))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

        # DCT noise when very stuck
        if stuck_count > 8000:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 15000.0)
            if noise_scale > 1e-9:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 200)
                F[cutoff:-cutoff] *= 0.05
                noise_smooth = np.real(ifft(F))[:n]
                nn = np.linalg.norm(noise_smooth)
                if nn > 1e-15:
                    noise_smooth = noise_smooth / nn * noise_scale
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0
                stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_fista(a_init, max_time, rng, beta_start=10.0, beta_end=None,
              mode="lse", step_scale=1.0):
    """Standard FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    if beta_end is None:
        beta_end = beta_start * 30.0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, int(max_time * 400)))
        beta = beta_start + frac * (beta_end - beta_start)
        beta = max(5.0, min(beta, 100000.0))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 2000:
                k = max(2, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            if stuck_count > 1000 and (it % 5 == 0 or rng.random() < 0.5):
                tol = min(0.05, 0.001 * stuck_count)
                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if stuck_count > 5000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 12000.0)
            if noise_scale > 1e-9:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(5, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_s = np.real(ifft(F))[:n]
                nn = np.linalg.norm(noise_s)
                if nn > 1e-15:
                    noise_s = noise_s / nn * noise_scale
                y = project_simplex(y + noise_s, S)
                t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=4):
    """L-BFGS-B polishing with increasing beta levels."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [40.0, 100.0, 300.0, 800.0, 2000.0][:n_beta_levels]
    penalties = [20.0, 50.0, 100.0, 200.0, 500.0][:n_beta_levels]

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        local_a = a.copy()
        local_beta = beta
        local_pen = pen
        local_S = S
        local_n = n

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = local_beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / local_beta
            sum_x = x.sum()
            obj += local_pen * (sum_x - local_S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * local_pen * (sum_x - local_S)
            return obj, grad

        result = scipy_minimize(
            obj_and_grad, local_a, method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 300, 'maxls': 50, 'ftol': 1e-16,
                     'gtol': 1e-12, 'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (nit=%d)" %
              (beta, pen, obj_new, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def active_set_polish(a_init, max_time, rng):
    """Aggressive active-set polishing with adaptive k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 100:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        elif stuck < 500:
            grad = topk_gradient(y, c, k=min(8, max(2, stuck // 50)), power=1.0)
        elif stuck < 3000:
            k = min(64, 8 + stuck // 100)
            grad = topk_gradient(y, c, k=k, power=0.8)
        else:
            k = min(256, 64 + stuck // 100)
            grad = topk_gradient(y, c, k=k, power=0.5)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.0005 * it))
        step = min(step, 3.0 * S / (gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 8000 and it % 1000 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Combines spectral flattening (GS), hybrid spectral-FISTA, standard FISTA,
    L-BFGS-B polishing, and active-set refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = np.ones(n0) / n0 * np.sqrt(2 * n0)
        global_best_obj = compute_obj(a_start)
        global_best_a = a_start.copy()
        print("Fallback init: n=%d, obj=%.8f" % (n0, global_best_obj))

    N = 1 << (2 * n0 - 1 - 1).bit_length()
