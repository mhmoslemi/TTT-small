import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_value(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    m = shifted.max()
    e = np.exp(shifted - m)
    return cm + (1.0/beta) * np.log(e.sum() + 1e-300)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_lse_gradient(y, c, k, beta):
    """LSE over only the top-k values, for a more focused gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    shifted = beta * (c_top - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_top = e / tot
    w_full = np.zeros(len(c))
    w_full[idx] = w_top
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        accepted = False
        for lr in range(20):

            c_new = fft_convolve_self(x_new)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses power-annealing (L_p norm with increasing p) for warmup,
then LSE-FISTA with adaptive beta, top-k active-set polishing,
and multi-size exploration. Key improvements over baseline:
- Power-p annealing schedule for smoother optimization landscape
- Backtracking line search for robust step size selection
- Adaptive top-k LSE gradient (coordinated k and beta)
- Multi-initialization with competitive selection
- Spectral preconditioning via power-spectrum analysis
- Aggressive final polish with alternating exact/subgradient methods
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_p_gradient(y, c, p):
    """Gradient of L_p norm of c w.r.t. y."""
    n = len(y)
    c_p = np.sum(np.maximum(c, 0.0) ** p)
    if c_p < 1e-300:
        return np.zeros(n)
    coeff = c_p ** (1.0 / p - 1.0)
    w = (np.maximum(c, 0.0) ** (p - 1.0)) * coeff
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_lse_gradient(y, c, k, beta):
    """LSE gradient over only top-k values."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    shifted = beta * (c_top - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_top = e / tot
    w_full = np.zeros(len(c))
    w_full[idx] = w_top
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_power_p_phase(a_init, max_time, rng, p_start=2, p_end=4, p_steps=4):
    """Power-p warmup: gradually increase p from p_start to p_end."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0

    p_values = np.exp(np.linspace(np.log(p_start), np.log(p_end), p_steps))

    for pi in range(p_steps):
        p = p_values[pi]
        p_end_time = start + max_time * ((pi + 1) / p_steps)

        while time.monotonic() < p_end_time and time.monotonic() - start < max_time:
            it += 1
            c = fft_convolve_self(y)
            cur_max = c.max()
            obj = 2.0 * n * cur_max / (a.sum() ** 2)
            if obj < best_obj:
                best_obj = obj
                best_a = y.copy()

            grad = power_p_gradient(y, c, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            step = 1.0 / (2.0 * np.max(a) * n / S + 1.0)
            step = min(step, S / (gnorm * 2.0 + 1.0))
            step = max(step, 1e-10)

            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.5:
                step *= 0.25
                x_new = project_simplex(y - step * grad, S)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            if it % 3000 == 0:
                t_mom = 1.0

    return best_a


def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0,
                  beta_growth=1.01, use_backtrack=True):
    """LSE FISTA with adaptive beta and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = it
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        if use_backtrack:
            step = S / (gnorm + 1.0)
            step = max(step, 1e-12)
            step = min(step, 1.0 / max(beta * 0.5, 1.0))
        else:
            step = 0.5 / max(beta, 1.0)

        x_new = project_simplex(y - step * grad, S)
        c_new = fft_convolve_self(x_new)

        # Backtracking: reduce step if objective worsens significantly
        if c_new.max() > cur_max * 1.1:
            for _ in range(5):
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= cur_max * 1.01:
                    break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Adaptive beta increase
        if stuck < 200:
            beta = min(beta_max, beta * 1.001)
        elif stuck < 2000:
            beta = min(beta_max, beta * beta_growth)
        else:
            beta = min(beta_max, beta * (beta_growth * 3))

        # Restart with noise when stuck
        if stuck > 4000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_conv

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_chirp_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / n)


def make_gaussian_init(n):
    tt = np.arange(n, dtype=np.float64) / n
    return np.exp(-4.0 * (tt - 0.5) ** 2) + 0.1


def make_random_smooth_init(n, rng):
    """Generate a smooth random non-negative sequence."""
    freqs = np.fft.rfftfreq(n)
    spec = rng.normal(0, 1, len(freqs))
    spec *= np.exp(-0.5 * (freqs / (n * 0.1)) ** 2)  # Low-pass
    spec[0] = 0
    a = np.real(np.fft.irfft(spec * 2, n))
    a = np.abs(a) + 0.1
    return a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
