```python
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix C where C[k,j] = coefficient of b[j] in conv(a,b)[k]
    # conv(a,b)[k] = sum_j a[k-j]*b[j] for valid j
    # So C[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k-j])
    
    C = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T b s.t. C b <= M, b >= 0
    # scipy linprog minimizes, so: min -1^T b s.t. C b <= M, b >= 0
    c_obj = -np.ones(n)
    result = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        return result.x
    return None
```

```python
# Helper functions at top level

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    if v.sum() == S:
        return np.maximum(v, 0)
    # Standard simplex projection
    if np.all(v <= 0):
        return np.full(n, S / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c_k)) with respect to a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot  # softmax weights
    
    # Gradient: 2 * sum_k w_k * R_k(a)
    # R_k(a)[i] = a[k-i] if 0 <= k-i < n
    # This is equivalent to: for each k, add 2*w_k * (reflected and shifted a)
    # Using convolution: grad = 2 * conv(w, a_reflected)
    # Actually: grad[i] = 2 * sum_k w_k * a[k-i] (if valid)
    # = 2 * sum_j w[i+j] * a[j] (reindexing)
    # This is 2 * conv(w, a) truncated!
    
    # Let's verify: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * sum_j w[i+j] * a[j] for valid i+j
    # This is the cross-correlation of w and a, which equals conv(w, flip(a))
    
    a_rev = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    grad = 2.0 * np.real(ifft(W * AR))[:n]
    
    return grad
```

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses FISTA (accelerated proximal gradient) with adaptive LSE smoothing,
with periodic restarts, line search, and optional LP-based direction finding.
Multi-phase approach: coarse-to-fine beta annealing with FISTA,
followed by final gradient polish and multi-length exploration.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if abs(s - S) < 1e-15:
        return v
    # Standard simplex projection (Chen & Donoho)
    if s == 0:
        return np.full(n, S / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - S)
    if not np.any(cond):
        rho = n
    else:
        rho = int(np.max(rho_candidates[cond]))
    theta = (cssv[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def lse_gradient(a, c, beta):
    """
    Gradient of LSE_beta(a) = (1/beta)*log(sum_k exp(beta*c_k))
    where c_k = (a*a)[k].
    Returns gradient with respect to a.
    """
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -600, 600)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        # All weights effectively zero, use uniform
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = e / tot
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * conv(w, a[::-1])[i + n - 1]
    a_rev = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * AR))
    grad = 2.0 * conv_result[n - 1: 2 * n - 2]
    
    return grad


def lse_value(c, beta):
    """Compute LSE_beta(c) = (1/beta)*log(sum exp(beta*c_k))."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -600, 600)
    return cm + np.log(np.exp(scaled).sum()) / beta


def fista_lse(a_init, S, beta_schedule, time_limit, rng, verbose=True):
    """
    FISTA optimization of LSE-smoothed max convolution.
    
    Parameters:
    - a_init: initial point (will be projected to simplex)
    - S: target sum (sqrt(2n))
    - beta_schedule: list of (beta, n_iters) pairs
    - time_limit: seconds
    - rng: random number generator
    - verbose: whether to print progress
    
    Returns: (best_a, best_lse_value)
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    
    best_a = a.copy()
    best_obj = None  # We'll compute true max_conv for tracking
    
    # Track best true objective
    c = fft_convolve_self(a)
    true_max = c.max()
    best_true_max = true_max
    best_true_a = a.copy()
    
    start = time.monotonic()
    global_it = 0
    
    for beta, n_iters in beta_schedule:
        if time.monotonic() - start > time_limit:
            break
        
        # For FISTA within this beta phase
        y = a.copy()
        t = 1.0
        a_prev = a.copy()
        a_prev_prev = a.copy()
        
        # Estimate initial step size
        # Conservative: L ~ 2 + 2*beta*||a||_2^2
        a_norm2 = np.dot(a, a)
        L_est = 2.0 + 2.0 * beta * a_norm2
        step = 1.0 / L_est
        
        for it in range(n_iters):
            if time.monotonic() - start > time_limit:
                break
            global_it += 1
            
            # Compute gradient at y
            c_y = fft_convolve_self(y)
            grad = lse_gradient(y, c_y, beta)
            
            # Gradient step + projection
            a_new = project_simplex(y - step * grad, S)
            
            # Greedy restart check
            diff_curr = np.linalg.norm(a_new - a)
            diff_prev = np.linalg.norm(a - a_prev)
            
            # Update y for next iteration
            if t == 1.0 or diff_curr < diff_prev:
                t_next = 1.0  # Restart
            else:
                t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            
            y_next = a_new + ((t - 1.0) / t_next) * (a_new - a)
            
            # Update
            a_prev_prev = a_prev
            a_prev = a
            a = a_new
            t = t_next
            
            # Periodic check of true objective
            if global_it % 50 == 0:
                c_check = fft_convolve_self(a)
                true_max_check = c_check.max()
                if true_max_check < best_true_max:
                    best_true_max = true_max_check
                    best_true_a = a.copy()
                    if verbose:
                        elapsed = time.monotonic() - start
                        obj = 2.0 * n * best_true_max / (S * S)
                        print(f"      it={global_it}, beta={beta:.0f}, obj={obj:.6f}, max={best_true_max:.6f}, t={elapsed:.1f}s")
        
        # After each beta phase, set y = a for next phase
        a = a.copy()
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best_true_max:
        best_true_max = c_final.max()
        best_true_a = a.copy()
    
    return best_true_a, best_true_max


def lp_based_direction(a, M, max_time=30.0):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b or None if infeasible/too slow.
    """
    n = len(a)
    
    # Build sparse constraint matrix C where C[k,j] = a[k-j] if 0 <= k-j < n
    # Shape: (2n-1) x n
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    C = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max 1^T b s.t. C b <= M, b >= 0
    # In min form: min -1^T b s.t. C b <= M*1, 0 <= b
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        from scipy.optimize import linprog
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, 
                        bounds=[(0, None)] * n, method='highs')
        if result.success:
            return result.x
    except Exception as e:
        print(f"    LP solve failed: {e}")
    
    return None


def lp_iteration(a, S, max_time=60.0, verbose=True):
    """
    One LP-based iteration: find best direction, line search, update.
    Returns updated a.
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    b_star = lp_based_direction(a, M, max_time)
    if b_star is None:
        return a
    
    b_sum = b_star.sum()
    if b_sum < 1e-15:
        return a
    
    # Rescale to match S
    b_scaled = b_star * (S / b_sum)
    
    # Line search: find best t in (0,1]
    best_t = 0.0
    best_val = M
    
    for t_try in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
        a_try = (1 - t_try) * a + t_try * b_scaled
        a_try = np.maximum(a_try, 0)
        a_try = a_try * (S / a_try.sum())
        c_try = fft_convolve_self(a_try)
        if c_try.max() < best_val:
            best_val = c_try.max()
            best_t = t_try
    
    if best_t > 0:
        a_new = (1 - best_t) * a + best_t * b_scaled
        a_new = np.maximum(a_new, 0)
        a_new *= (S / a_new.sum())
        if verbose:
            print(f"    LP iter: t={best_t}, max: {M:.6f} -> {best_val:.6f}")
        return a_new
    
    return a


def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            return a
    # Fallback: smooth bump
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
    return a
