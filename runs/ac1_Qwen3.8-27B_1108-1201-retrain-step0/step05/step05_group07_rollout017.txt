    #             = a[k-j] + a[j]  (the first term: i=j, gives a[k-j]; the second: k-i=j, i=k-j, gives a[k-j])
    # 
    # Wait: sum_i delta(k-i, j) * a[i] = a[k-j] (substituting i = k-j)
    # And sum_i delta(i,j) * a[k-i] = a[k-j]

    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]?
    # 
    # No wait, I think I'm double-counting. Let me be more careful.
    # c[k] = sum_i a[i] * a[k-i]
    # 
    # This is a product of two a's. The derivative with respect to a[j]:
    # dc[k]/da[j] = sum_i [ (da[i]/da[j]) * a[k-i] + a[i] * (a[k-i]/da[j]) ]
    #             = sum_i [ delta(i,j) * a[k-i] + a[i] * delta(k-i, j) ]
    #             = a[k-j] + a[k-j]  ...

    # 
    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]

    # 
    # If j ≠ k/2: c[k] contains a[j]*a[k-j] (from i=j) and a[k-j]*a[j] (from i=k-j)
    # dc[k]/da[j] = a[k-j] (from the first term) + a[k-j] (from the second term) = 2*a[k-j]

    # Wait, but a[j]*a[k-j] and a[k-j]*a[j] are the SAME term (multiplication is commutative)!
    # In the sum c[k] = sum_i a[i]*a[k-i], when i=j we get a[j]*a[k-j], and when i=k-j we get a[k-j]*a[j].
    # These are two separate terms in the sum (unless j = k-j), so the total contribution is 2*a[j]*a[k-j].
    # dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]. ✓
    
    # So the formula is: dc[k]/da[j] = 2*a[k-j] (with a[m] = 0 for m not in [0, n-1])
    
    # Great! So grad_a[j] = 2 * sum_k w[k] * a[k-j]
    
    # This is: grad_a = 2 * cross_correlation(w, a) = 2 * conv(w, flip(a))
    # where cross_correlation(w, a)[j] = sum_k w[k] * a[k-j]
    # 
    # In FFT terms: cross_correlation(w, a) = ifft(conj(fft(w)) * fft(a))
    # (with appropriate zero-padding)

grad_a = 2 * ifft(conj(fft(w, L)) * fft(a, L))[:n]

where L >= 3n-2 (to avoid circular convolution effects).

Actually, for the correlation: ifft(conj(fft(w, L)) * fft(a, L)) gives the cross-correlation (w, a)[j] = sum_k w[k] * a[k-j] (with zero-padding).

We need L >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses subgradient descent with LSE smoothing, followed by L-BFGS polishing
on a softmax-parameterized version. Includes multi-length exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    if v.sum() == S and np.all(v >= 0):
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value_and_grad(c, beta):
    """Compute LSE value and gradient weights."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    val = cm + np.log(tot) / beta
    return val, w


def lse_grad_a(a, w):
    """Gradient of LSE(conv(a,a)) with respect to a."""
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, L)
    A = fft(a, L)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def subgradient_step(a, c, S, k, step_scale=1.0):
    """One subgradient step focusing on top-k lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    
    # Compute gradient using FFT
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = weights
    grad = lse_grad_a(a, w_full)
    return grad


def lse_subgradient_step(a, c, S, beta):
    """One LSE-smoothed subgradient step."""
    n = len(a)
    val, w = lse_value_and_grad(c, beta)
    grad = lse_grad_a(a, w)
    return val, grad


def optimize_subgradient(a0, S, max_time, rng, verbose=True):
    """Subgradient optimization with adaptive parameters."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive k
        k = int(5 + 100 * progress ** 0.7)
        
        # Compute subgradient
        grad = subgradient_step(a, c, S, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        gap = cur_max - c.mean()
        if gap < 1e-10:
            break
        
        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 3.0 / gnorm)
        
        # Line search
        accepted = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9) - 1e-12:
                a = a_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            a = project_simplex(a - (0.5 * gap / (gnorm * gnorm)) * grad, S)

            noise_scale = min(0.05, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    subgrad it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")
    
    return best_a, best


def optimize_lse_lbfgs(a0, S, max_time, beta_init, beta_max, rng):
    """LSE-smoothed L-BFGS optimization using softmax parameterization."""
    n = len(a0)
    
    # Initialize b such that softmax(b) * S ≈ a0
    a_init = project_simplex(a0, S)
    # b = log(a_init + eps)
    eps = np.finfo(np.float64).eps
    b_init = np.log(a_init + eps * S / n)
    
    def softmax(b):
        b_shifted = b - b.max()
        e = np.exp(b_shifted)
        return e / e.sum()
    
    def objective_and_grad(b):
        p = softmax(b)
        a = S * p
        c = fft_convolve_self(a)
        val, w = lse_value_and_grad(c, beta)
        grad_a = lse_grad_a(a, w)
        # Convert to grad_b: grad_b = S * p * (grad_a - dot(grad_a, p))
        grad_b = S * p * (grad_a - np.dot(grad_a, p))
        return val, grad_b
    
    best_obj = np.inf
    best_b = b_init.copy()
    start = time.monotonic()
    
    # Use multiple L-BFGS calls with increasing beta
    beta = beta_init
    while time.monotonic() - start < max_time and beta < beta_max:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Run L-BFGS for a while
        t_phase = min(remaining * 0.5, 120.0)
        
        def local_obj(b):
            nonlocal beta
            val, grad = objective_and_grad(b)
            p = softmax(b)
            a = S * p
            c = fft_convolve_self(a)
            if c.max() < best_obj / (2.0 * n / (S * S)):
                best_obj = 2.0 * n * c.max() / (S * S)
                best_b = b.copy()
            return val, grad
        
        x0 = best_b.copy()
        result = minimize(local_obj, x0, method='L-BFGS-B',
                         options={'maxiter': 500, 'maxfun': 2000,
                                  'ftol': 1e-12, 'gtol': 1e-10,
                                  'maxls': 50, 'factr': 1e10})

        p = softmax(result.x)
        a = S * p
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_b = result.x.copy()
        
        # Increase beta
        beta = min(beta_max, beta * 2.0)
    
    # Final result
    p = softmax(best_b)
    a_final = S * p
    return a_final, best_obj
