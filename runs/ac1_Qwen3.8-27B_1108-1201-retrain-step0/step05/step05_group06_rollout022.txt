Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization with:
1. Diverse smooth initializations (cosine, parabola, raised cosine)
2. LSE-FISTA with backtracking line search and adaptive beta annealing
3. Direct subgradient polishing with all active lags
4. Multi-scale exploration (n and 2n)
5. Final intensive local search

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value(c, beta):
    cm = c.max()
    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    a = normalize_seq(a, S)

            beta = beta_init * (beta_final / beta_init) ** frac

            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)

        if gnorm < 1e-15:

            # Try the step

        y = np.maximum(y, 0.0)
        a = x_new

        if stuck > 5000:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining:
1. Diverse smooth initializations (cosine, parabola, high-power shapes)
2. LSE-FISTA with adaptive beta annealing and momentum restart
3. Direct subgradient polishing using the full active set
4. Multi-scale exploration (n=8192 and n=16384)
5. Final intensive local search with coordinate perturbations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Log-sum-exp approximation of max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(y, c, beta):
    """Gradient of LSE approximation w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_gradient(y, c, tol_frac):
    """Subgradient using all lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_cosine_init(n, power=1.0):
    """Cosine bump to the given power."""
    t = np.linspace(0, 1, n, endpoint=False)
    base = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    return base ** power


def make_parabola_init(n, power=2.0):
    """Parabolic/bump shape raised to power."""
    t = np.linspace(-1, 1, n, endpoint=False)
    return np.maximum(1.0 - t ** 2, 0.0) ** power


def make_super_gaussian(n, sigma=0.4):
    """Super-Gaussian: exp(-|t/sigma|^2)."""
    t = np.linspace(-1, 1, n, endpoint=False)
    return np.exp(-(t / sigma) ** 2)


def make_optimal_p_init(n, p=3.0):
    """Shape (1 - |t|^p)^q for optimal p."""
    t = np.linspace(-1, 1, n, endpoint=False)
    return np.maximum(1.0 - np.abs(t), 0.0) ** p


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_lse_fista(a_init, max_time, rng, beta_init=5.0, beta_max=10000.0,
                  anneal_fraction=0.4, hybrid_threshold=3000):
    """LSE-FISTA with adaptive beta annealing and hybrid gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    n_anneal_steps = max(1, int(anneal_fraction * 1000))

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta scheduling
        if it < n_anneal_steps * 5:
            frac = min(1.0, it / (n_anneal_steps * 5))
            beta = beta_init * (beta_max / beta_init) ** frac
        else:
            stuck = it - last_improve
            if stuck < 1000:
                beta = min(beta_max * 0.5, beta * 1.002)
            elif stuck < 5000:
                beta = min(beta_max, beta * 1.008)
            else:
                beta = min(beta_max * 2, beta * 1.02)
            beta = min(beta, beta_max * 2)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Choose gradient type
        if stuck > hybrid_threshold and (it % 4 == 0):
            tol = min(0.05, 0.001 * stuck / 1000)
            grad = active_gradient(y, c, tol)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size: balance between LSE smoothness and convergence
        L_est = 2.0 * beta * np.max(a) + 4.0 * c.max() / n
        step = 0.9 / max(L_est, 1e-10)
        step = max(step, 1e-14)
        step = min(step, S / (gnorm * np.sqrt(n)))

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 8000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 15000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
            beta = max(beta_init, beta * 0.5)

        # Periodic restart
        if it % 15000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_subgradient_polish(a_init, max_time, rng):
    """Direct subgradient method with adaptive step size for final polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_base = 0.1 * S / n

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance for active set
        if stuck < 100:
            tol = 0.0
        elif stuck < 500:
            tol = 0.001
        elif stuck < 2000:
            tol = 0.005
        elif stuck < 10000:
            tol = 0.01
        else:
            tol = 0.02

        grad = active_gradient(a, c, tol)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Diminishing step size with adaptive scaling
        step = step_base / (1.0 + 0.01 * it)
        step = min(step, 0.5 * S / gnorm)
        step = max(step, 1e-16)

        a_new = project_simplex(a - step * grad, S)

        # Accept if it doesn't worsen the max too much
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max + 1e-12 or it % 50 == 0:
            a = a_new
        else:
            # Try half step
            a_half = project_simplex(a - 0.5 * step * grad, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() <= cur_max + 1e-12:
                a = a_half
            else:
                # Shrink step
                step_base *= 0.95

        if stuck > 10000 and it % 500 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            step_base = 0.1 * S / n
            stuck = 0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

    inits['super_g2'] = normalize_seq(make_super_gaussian(n_base, 0.25), S_base)
    inits['opt_p3'] = normalize_seq(make_optimal_p_init(n_base, 3.0), S_base)
    inits['opt_p4'] = normalize_seq(make_optimal_p_init(n_base, 4.0), S_base)

    # Evaluate all initializations
    best_init = None
    best_init_obj = np.inf
    for name, a_init in inits.items():
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = name
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.6f" % (name, obj))

    print("Best init: %s with obj=%.6f" % (best_init, best_init_obj))
    print("Phase 1 time: %.1fs" % (time.monotonic() - t_start))

    # === Phase 2: LSE-FISTA optimization on best initializations ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 2: LSE-FISTA (remaining %.0fs) ===" % remaining)
    t2 = time.monotonic()

    # Optimize the top 3 initializations
    sorted_inits = sorted(inits.items(), key=lambda x: compute_obj(x[1]))
    top3 = sorted_inits[:3]

    for name, a_init in top3:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        alloc = rem * 0.25
        print("  Optimizing '%s' (alloc %.0fs)..." % (name, alloc))
        a_opt, _ = run_lse_fista(a_init, alloc, rng,
                                  beta_init=5.0, beta_max=5000.0,
                                  anneal_fraction=0.4)
        obj = update_best(a_opt)
        print("    -> obj=%.6f" % obj)
        # Keep the best result for further polish
        if obj <= global_best_obj + 1e-6:
            global_best_a = a_opt.copy()

    print("Phase 2 time: %.1fs" % (time.monotonic() - t2))

    # === Phase 3: Continue LSE-FISTA with higher beta ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 3: High-beta LSE-FISTA (remaining %.0fs) ===" % remaining)
        t3 = time.monotonic()
        alloc = remaining * 0.35
        a_opt, _ = run_lse_fista(global_best_a, alloc, rng,
                                  beta_init=100.0, beta_max=50000.0,
                                  anneal_fraction=0.3)
        obj = update_best(a_opt)
        print("  obj=%.6f" % obj)
        print("Phase 3 time: %.1fs" % (time.monotonic() - t3))

    # === Phase 4: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 4: Subgradient polish (remaining %.0fs) ===" % remaining)
        t4 = time.monotonic()
        alloc = remaining * 0.30
        a_opt, _ = run_subgradient_polish(global_best_a, alloc, rng)
        obj = update_best(a_opt)
        print("  obj=%.6f" % obj)
        print("Phase 4 time: %.1fs" % (time.monotonic() - t4))

    # === Phase 5: Local search ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 5: Local search (remaining %.0fs) ===" % remaining)
        t5 = time.monotonic()
        alloc = remaining * 0.25
        a_opt = run_local_search(global_best_a, alloc, rng)
        obj = update_best(a_opt)
        print("  obj=%.6f" % obj)
        print("Phase 5 time: %.1fs" % (time.monotonic() - t5))

    # === Phase 6: Multi-scale (upsample to 2n) ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 6: Upsample to n=%d (remaining %.0fs) ===" % (n2, remaining))
        t6 = time.monotonic()
        alloc = remaining * 0.60

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)

        # Quick LSE-FISTA on upsampled
        a_opt, _ = run_lse_fista(a_up, alloc * 0.6, rng,
                                  beta_init=5.0, beta_max=5000.0,
                                  anneal_fraction=0.4)
        obj = update_best(a_opt)
        print("  Upsampled LSE: obj=%.6f" % obj)

        rem6 = alloc - (time.monotonic() - t6)
        if rem6 > 20:
            a_opt2, _ = run_subgradient_polish(global_best_a, rem6, rng)
            obj = update_best(a_opt2)
            print("  Upsampled polish: obj=%.6f" % obj)

        print("Phase 6 time: %.1fs" % (time.monotonic() - t6))

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Final polish (remaining %.0fs) ===" % remaining)
        t7 = time.monotonic()

        # Subgradient with tiny tolerance
        alloc7 = remaining * 0.5
        a_opt, _ = run_subgradient_polish(global_best_a, alloc7, rng)
        obj = update_best(a_opt)
        print("  Subgrad: obj=%.6f" % obj)

        rem7 = remaining - (time.monotonic() - t7)
        if rem7 > 20:
            # High-beta LSE for final push
            a_opt2, _ = run_lse_fista(global_best_a, rem7, rng,
                                       beta_init=500.0, beta_max=100000.0,
                                       anneal_fraction=0.2)
            obj = update_best(a_opt2)
            print("  LSE high: obj=%.6f" % obj)

        print("Phase 7 time: %.1fs" % (time.monotonic() - t7))

    # === Final verification and output ===
    print("\n=== FINAL RESULT ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - t_start))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
