"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Multi-phase FISTA with geometric beta annealing, top-k LSE smoothing,
and efficient partial sort for gradient computation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sort-based method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_grad(a, c, k, beta):
    """Compute LSE-smoothed gradient of self-convolution max.
    Returns (smoothed_value, gradient)."""
    n = len(a)
    n_c = len(c)
    k = min(k, n_c)
    
    # Find top-k using partition (O(n) instead of O(n log n))
    # We want the k largest elements
    thresholds = np.partition(c, n_c - k)[n_c - k]
    mask = c >= thresholds
    # If more than k elements are >= threshold (due to ties), trim
    if mask.sum() > k:
        # Get indices of elements equal to threshold
        eq_idx = np.where((c == thresholds) & mask)[0]
        # Keep only enough to reach k
        extra = mask.sum() - k
        for i in eq_idx[:extra]:
            mask[i] = False
    
    c_top = c[mask]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Numerical safety
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        beta_eff = scale
        scaled = beta_eff * (c_top - cm)
    else:
        beta_eff = beta
    
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    val = cm + np.log(max(tot, 1e-300)) / beta_eff
    
    # Build full weight vector
    w_full = np.zeros(n_c)
    w_full[mask] = w
    
    # Compute gradient: grad[i] = 2 * sum_k w_full[k] * (a[i+k] + a[i-k]) / 2
    # = sum_k w_full[k] * (a[i+k] + a[i-k])
    # This is equivalent to: 2 * real(ifft(fft(w_full) * conj(fft(a))))[:n]
    fsize = 1 << (n_c + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return val, grad


def run_fista_phase(a_init, S, beta, k, max_time, rng, 
                    max_iter=200000, min_improve=1e-10):
    """Run FISTA for a fixed beta until convergence or time limit."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    last_best = best
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time and it < max_iter:
        it += 1
        
        # Compute gradient at y
        c_y = fft_convolve_self(y)
        _, grad = topk_lse_grad(y, c_y, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: 1/L where L ~ 2*beta (Lipschitz constant of gradient)
        step = 1.0 / (2.0 * beta + 1e-10)
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking (a few steps)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= c_y.max() + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        
        a = x_new
        y = y_new
        t_val = t_new
        
        # Check improvement
        c_a = fft_convolve_self(a)
        cur = c_a.max()
        if cur < best - min_improve:
            best = cur
            best_a = a.copy()
            last_best = cur
            no_improve = 0
        else:
            no_improve += 1
        
        # Convergence check
        if no_improve > 5000:
            # Try a small perturbation
            if no_improve > 10000:
                break
            noise = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_val = 1.0
            no_improve = 0
        
        # Periodic momentum reset
        if it % 5000 == 0:
            t_val = 1.0
    
    return best_a, best, it

def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)

    """Polish using exponentiated gradient (multiplicative update)."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 100.0 * (1.0 + it * 0.01))
        k = max(2, int(0.02 * (2 * n - 1)))
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()
        
        _, grad = topk_lse_grad(a, c, k, beta)
        
        # Exponentiated gradient: a_i *= exp(-eta * grad_i)

            noise = 0.0005 * S / np.sqrt(n)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Multi-phase optimization with geometric beta annealing FISTA,
efficient top-k LSE via np.partition, exponentiated gradient polishing,
and LP-based direction finding for final refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_grad(a, c, k, beta):
    """Compute LSE-smoothed value and gradient of self-convolution max.
    Uses np.partition for O(n) top-k selection."""
    n = len(a)
    n_c = len(c)
    k = min(k, n_c)

    if k >= n_c:
        # Use all elements
        mask = np.ones(n_c, dtype=bool)
        c_top = c.copy()
    else:
        # Partition to find k-th largest
        part = np.partition(c, n_c - k)
        threshold = part[n_c - k]
        mask = c >= threshold
        # Handle ties: if too many, randomly select subset
        if mask.sum() > k:
            excess = mask.sum() - k
            eq_indices = np.where(mask & (c == threshold))[0]
            if len(eq_indices) > 0:
                remove = rng_select_top_equivalent(eq_indices, excess, threshold)
                mask[remove] = False
        c_top = c[mask]

    cm = c_top.max()
    diff = c_top - cm
    scaled = beta * diff
    # Numerical safety
    max_scaled = scaled.max()
    if max_scaled > 400:
        beta_eff = 400.0 / max(1e-15, max_scaled)
        scaled = beta_eff * diff
    else:
        beta_eff = beta

    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    val = cm + np.log(max(tot, 1e-300)) / beta_eff

    # Build full weight vector
    w_full = np.zeros(n_c)
    w_full[mask] = w

    # Gradient via FFT: grad = 2 * real(ifft(FFT(w_full) * conj(FFT(a))))[:n]
    fsize = 1 << (n_c + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    return val, grad


def rng_select_top_equivalent(indices, excess, threshold):
    """Select 'excess' indices to remove from a tie set (deterministic)."""
    # Just take the first 'excess' ones (deterministic for reproducibility)
    return indices[:excess]


def run_fista_phase(a_init, S, beta, k, max_time, rng,
                    max_iter=500000, patience=8000):
    """Run FISTA for a fixed beta until convergence or time limit."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    last_best_time = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time and it < max_iter:
        it += 1

        # Compute gradient at y
        c_y = fft_convolve_self(y)
        _, grad = topk_lse_value_and_grad(y, c_y, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (2.0 * beta + 1e-10)

        # FISTA step
        x_new = project_simplex(y - step * grad, S)

        # Backtracking line search (limited)
        c_y_max = c_y.max()
        for _ in range(4):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= c_y_max + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)

        a = x_new
        y = y_new
        t_val = t_new

        # Track best
        if it % 3 == 0:  # Check every 3 iterations for efficiency
            c_a = fft_convolve_self(a)
            cur = c_a.max()
            if cur < best - 1e-11:
                best = cur
                best_a = a.copy()
                last_best_time = time.monotonic()
                no_improve = 0
            else:
                no_improve += 3
        else:
            no_improve += 1

        # Convergence / stagnation handling
        if no_improve > patience:
            if no_improve > patience * 2:
                break
            # Perturb to escape
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            no_improve = 0

        # Periodic momentum reset to avoid overshooting
        if it % 3000 == 0:
            t_val = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best, it


def run_with_annealing(a_init, S, n, time_budget, rng):
    """Run FISTA with geometric beta annealing."""
    start = time.monotonic()
    a = a_init.copy()
    c = fft_convolve_self(a)
    global_best = c.max()
    global_a = a.copy()

    # Geometric beta schedule
    betas = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240]

    # Time allocation: proportional to (1 + log2(beta))
    weights = [1.0 + 0.3 * i for i in range(len(betas))]
    total_w = sum(weights)

    cumulative_time = 0.0
    for i, beta in enumerate(betas):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed - 5.0
        if remaining < 3.0:
            break

        # This phase gets its share of remaining time
        frac = weights[i] / (sum(weights[j] for j in range(i, len(betas))))
        phase_time = max(2.0, remaining * frac)

        # Adaptive k: smaller for larger beta
        n_c = 2 * n - 1
        k = max(4, int(n_c / (1.0 + beta * 0.5)))
        k = min(k, n_c)

        a_opt, val_opt, iters = run_fista_phase(a, S, beta, k, phase_time, rng)

            obj = 2.0 * n * val_opt / S**2
            print("    beta=%5d: obj=%.8f (iters=%d)" % (beta, obj, iters))

def exponentiated_gradient_polish(a_init, S, max_time, rng):
    """Polish using exponentiated gradient with adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()

        c = fft_convolve_self(a)

            best = cur
            best_a = a.copy()
            last_best = cur

        _, grad = topk_lse_value_and_grad(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Multiplicative update
        eta = 1.0 / max(gnorm, 1e-10)
        # Scale eta to be conservative
        eta = min(eta, 0.1)
        log_update = -eta * grad
        # Clip for numerical safety
        log_update = np.clip(log_update, -5.0, 5.0)

        # Stagnation check
        if it % 5000 == 0 and (best - last_best) < 1e-13:
            break

    return best_a, best


def lp_direction_step(a, M, max_time=30.0):
    """Try to solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns the direction g or None if too slow."""
    import scipy.sparse as sp
    from scipy.optimize import linprog

    n = len(a)
    n_c = 2 * n - 1

    # Build constraint matrix A_c[k, j] = a[k-j] if 0 <= k-j < n
    # This is a convolution (Toeplitz) matrix
    # Row k: a[k-j] for j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(n_c):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            a_idx = k - idx
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[a_idx].tolist())

    A_c = sp.csr_matrix((vals, (rows, cols)), shape=(n_c, n))

    # LP: max 1^T g s.t. A_c @ g <= M, g >= 0
    # scipy linprog minimizes, so: min -1^T g s.t. A_c @ g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_c, b_ub=np.full(n_c, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max_time, 'presolve': True})

    if res.status == 0 and res.x is not None:
        g = res.x
        if g.sum() > 0:
            return g
    return None


def lp_polish(a_init, S, max_time, rng):
    """Polish using LP-based direction finding."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        M = c.max()

        g = lp_direction_step(a, M, max_time=15.0)
        if g is None:
            break

        # Rescale g to match sum
        g = normalize(g, S)

        # Move toward g with various step sizes
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best:
                a = a_try
                best = c_try.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)
