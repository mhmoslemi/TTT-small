"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses: (1) Power-2 FISTA warmup, (2) Adaptive active-set LSE with 
Nesterov acceleration and backtracking, (3) Multi-scale upsample,
(4) Targeted coordinate refinement on top lags.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sorted algorithm."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence using FFT."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        # Fallback to uniform
        w = np.ones(len(c)) / len(c)
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_set_gradient(y, c, tol=0.01):
    """
    Subgradient from lags within tolerance of max.
    tol: fraction, i.e., include lags where c_k >= max * (1 - tol)
    """
    n = len(y)
    max_val = c.max()
    if max_val < 1e-15:
        return np.zeros(n)
    active = np.where(c >= max_val * (1.0 - tol))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def blended_gradient(y, c, beta, tol_lse, tol_as):
    """Blend LSE gradient with active-set gradient."""
    grad_lse = lse_gradient(y, c, beta)
    grad_as = active_set_gradient(y, c, tol=tol_as)
    # Blend: weight by how "sharp" the LSE is
    # If LSE is already concentrated, use more active set
    return 0.5 * grad_lse + 0.5 * grad_as


def compute_obj(a):
    """Compute the actual objective value (without normalization change)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def power2_gradient(y, c):
    """Gradient of sum(c^2) w.r.t. a. = 4*(c*a)[0:n]"""
    n = len(y)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(y, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]


def run_power2_fista(a_init, max_time, rng):
    """Warm-up: minimize sum(c^2) using FISTA with backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj2 = np.inf
    best_a = a.copy()
    it = 0
    step = 0.5 / (4.0 * np.max(a) + 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        obj2 = np.sum(c * c)
        if obj2 < best_obj2:
            best_obj2 = obj2
            best_a = y.copy()

        grad = power2_gradient(y, c)
        gnorm = np.linalg.norm(grad)

def run_coordinate_refine(a_init, max_time, rng):
    """
    Coordinate descent refinement: pick random coordinates, 
    optimize them given the current convolution, targeting the max lags.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 50000:
            break

        # Pick a random coordinate to update
        j = rng.integers(0, n)
        
        # The effect of a_j on c_k is: c_k gets a_j * a_{k-j} added (for valid k-j)
        # We want to choose a_j to minimize max(c) while keeping sum(a) = S
        # Simple approach: try a few candidate values for a_j
        
        best_j_val = a[j]
        best_j_obj = cur_max
        
        # Candidate: reduce a_j if it contributes to the max
        # a_j contributes a_j * a_{k-j} to c_k
        # The max contribution is a_j * max(a_{k-j} for k where c_k is near max)
        
        # Simple: try a few multiplicative changes
        for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            trial = a.copy()
            trial[j] = a[j] * factor
            # Renormalize
            trial_sum = trial.sum()
            if trial_sum < S * 0.5:
                continue
            trial = trial * (S / trial_sum)
            c_trial = fft_convolve_self(trial)
            if c_trial.max() < best_j_obj:
                best_j_obj = c_trial.max()
                best_j_val = trial[j]
        
        if best_j_val != a[j]:
            a[j] = best_j_val
            a = normalize_seq(a, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a


def run_momentum_pgd(a_init, max_time, rng, beta=50.0):

    """Initialize by flattening the FFT magnitude of a non-negative sequence."""
    if rng is None:
        rng = np.random.default_rng()

    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())

    target_mag = np.full(fsize, 1.0)  # Flat magnitude target
    
    for _ in range(n_iter):

        A_new = target_mag * np.exp(1j * phase)
        a = np.real(ifft(A_new, fsize))
        a = np.clip(a, 0, None)
        if a.sum() < 1e-10:
            a = rng.random(n) + 0.1
        a = a * (S / a.sum())
    
    return a

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements: FFT-flattened initialization, dual-guided active set,
adaptive step size, multi-scale upsample, stationarity-based convergence.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sorted algorithm."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence using FFT."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def active_set_gradient(y, c, threshold):
    """Subgradient from lags at or above threshold."""
    n = len(y)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power2_gradient(y, c):
    """Gradient of sum(c^2) w.r.t. a = 4*(c*a)[0:n]."""
    n = len(y)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(y, fsize)
    return 4.0 * np.real(ifft(C * np.conj(A)))[:n]

    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    """
    Main optimization: minimize max(c) using adaptive LSE + active set.
    Tracks a dual threshold T for robust active set selection.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    # Initial step size calibration
    c0 = fft_convolve_self(a)
    g0 = lse_gradient(a, c0, beta)
    step = 0.5 / (np.linalg.norm(g0) + 1e-10)
    step = min(step, 0.5 / (2.0 * beta * np.max(a) + 1.0))
    step = max(step, 1e-10)

    # Dual variable (threshold) tracking
    T_dual = c0.max()
    T_smooth = c0.max()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Update dual threshold (exponential moving average)
        T_smooth = 0.99 * T_smooth + 0.01 * cur_max

        # Adaptive beta
        if stuck < 100:
            beta = max(beta_init, beta * 0.999)
        elif stuck < 500:
            beta = min(50000.0, beta * 1.01)
        elif stuck < 2000:
            beta = min(50000.0, beta * 1.03)
        else:
            beta = min(50000.0, beta * 1.05)

        # Choose gradient based on phase
        if stuck < 200:
            # Use LSE
            grad = lse_gradient(y, c, beta)
        elif stuck < 2000:
            # Blend LSE with active set at dual threshold
            tol = max(0.001, (T_smooth - 0.95 * best_conv) / (best_conv + 1e-10))
            threshold = best_conv * (1.0 - min(tol, 0.05))
            grad_as = active_set_gradient(y, c, threshold)
            blend = min(1.0, stuck / 2000.0)
            grad = (1.0 - blend) * lse_gradient(y, c, beta) + blend * grad_as
        else:
            # Primarily active set with adaptive tolerance
            spread = c[np.argsort(c)[::-1][:min(50, len(c))]].std()
            tol = min(0.1, max(0.005, spread / (cur_max + 1e-10)))
            threshold = cur_max * (1.0 - tol)
            grad_as = active_set_gradient(y, c, threshold)
            grad = 0.2 * lse_gradient(y, c, beta) + 0.8 * grad_as

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step size
        # Scale step inversely with gradient norm for stability
        step = max(1e-12, min(0.5 / (beta + 1.0), 2.0 / (gnorm + 1e-10)))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Momentum restart
        if stuck > 5000:
            t_mom = 1.0
            step = 0.5 / (beta + 1.0)

        # Noise injection when very stuck
        if stuck > 15000 and it % 300 == 0:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a


def run_nesterov_pgd(a_init, max_time, rng, beta=100.0):
    """
    Nesterov accelerated projected gradient with LSE.
    Different dynamics from run_adaptive_lse for diversity.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Step size
    c0 = fft_convolve_self(a)
    step = 1.0 / (beta * np.max(a) * 2.0 + 1.0)
    step = max(step, 1e-10)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta increase
        if stuck > 500:
            beta = min(100000.0, beta * 1.02)
            step = 1.0 / (beta * np.max(a) * 2.0 + 1.0)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000:
            t_mom = 1.0
        if stuck > 10000 and it % 500 == 0:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a


def run_subgradient_loopy(a_init, max_time, rng):
    """
    Loopy subgradient method: cycle through active lags,
    apply subgradient from each, with averaging.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.1 / np.sqrt(n)  # Standard subgradient step

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 20000:
            break

        # Find top lags
        top_k = min(10, len(c))
        idx = np.argsort(c)[::-1][:top_k]

        # Average subgradient over top lags
        grad = np.zeros(n)
        fsize = 1 << (len(c) + n - 1).bit_length()
        A = fft(a, fsize)
        for k in idx:
            # Gradient of c_k w.r.t. a is 2*a reversed and shifted by k
            # In FFT: ifft of (shifted omega^k) * conj(A) * 2
            omega = np.exp(2j * np.pi * k / fsize)
            grad_k = 2.0 * np.real(ifft(omega * np.conj(A)))[:n]
            grad += grad_k
        grad /= len(idx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Mirror descent step (multiplicative)
        lr = step / (gnorm + 1e-10)
        lr = min(lr, 0.5)
        a_new = a * np.exp(-lr * grad / (np.max(a) + 1e-10))
        a_new = normalize_seq(a_new, S)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(-0.5 * lr * grad / (np.max(a) + 1e-10))
            a_half = normalize_seq(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur_max:
                a = a_half

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a


def fft_flatten_init(n, n_iter=200, rng=None):
    """
    Initialize by iteratively flattening the FFT magnitude
    while maintaining non-negativity (phase retrieval / error reduction).
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    a = rng.random(n) + 0.1
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())

    fsize = 1 << (2 * n - 1).bit_length()
    target_mag = np.full(fsize, 1.0)

    for _ in range(n_iter):
        A = fft(a, fsize)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a = np.real(ifft(A_new, fsize))
        a = np.clip(a, 0, None)
        if a.sum() < 1e-10:
            a = rng.random(n) + 0.1
        a = a * (S / a.sum())

    return a


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_flat_topped_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    val = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * tt)) ** 0.2
    return val


def make_random_smooth_init(n, rng):
    """Random init passed through a Gaussian kernel for smoothness."""
    a = rng.random(n)
    # Simple smoothing: circular average with small window
    k = max(1, n // 100)
    kernel = np.ones(k * 2 + 1)
    for i in range(len(kernel)):
        kernel[i] = np.exp(-0.5 * ((i - k) / (k * 0.5)) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: (1) diverse inits + FFT-flatten, (2) Power-2 FISTA warmup,
    (3) Adaptive LSE+AS main phase, (4) Nesterov PGD, (5) Subgradient loopy,
    (6) Multi-scale upsample, (7) Final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get the provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01 or len(a_start) < 64:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)

    print("Base n = %d" % n0)

    # Generate diverse initializations
    n = n0
    S = np.sqrt(2.0 * n)
    inits = []
    names = []

    if a_start is not None:
        inits.append(normalize_seq(a_start, S))
        names.append("provided")

    inits.append(normalize_seq(make_cosine_init(n), S))
    names.append("cosine")
    inits.append(normalize_seq(make_flat_topped_init(n), S))
    names.append("flat_top")
    inits.append(normalize_seq(fft_flatten_init(n, n_iter=100, rng=rng), S))
    names.append("fft_flat1")
    inits.append(normalize_seq(fft_flatten_init(n, n_iter=100, rng=rng), S))
    names.append("fft_flat2")
    inits.append(normalize_seq(make_random_smooth_init(n, rng), S))
    names.append("rand_smooth")

    # Quick evaluation
    print("\n=== Init evaluation ===")
    best_init = None
    best_init_obj = np.inf
    for a_init, name in zip(inits, names):
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.6f" % (name, obj))

    print("Best init: obj=%.6f" % best_init_obj)

    # === Phase 1: Power-2 FISTA warmup ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.07
    if p1_budget > 5:
        print("\n=== Phase 1: Power-2 FISTA (%.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a_p2 = run_power2_fista(best_init, p1_budget, rng)
        obj_p2 = compute_obj(a_p2)
        print("  P2: obj=%.6f (%.1fs)" % (obj_p2, time.monotonic() - t1))
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()
        best_init = a_p2

    # === Phase 2: Main adaptive LSE + active set ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.38
    if p2_budget > 10:
        print("\n=== Phase 2: Adaptive LSE+AS (%.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a_main = run_adaptive_lse(best_init, p2_budget, rng, beta_init=20.0)
        obj_main = compute_obj(a_main)
        print("  Main: obj=%.6f (%.1fs)" % (obj_main, time.monotonic() - t2))
        if obj_main < global_best_obj:
            global_best_obj = obj_main
            global_best_a = a_main.copy()

    # === Phase 3: Nesterov PGD with high beta ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.12
    if p3_budget > 10:
        print("\n=== Phase 3: Nesterov PGD (%.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        for beta_try in [50.0, 500.0]:
            a_nest = run_nesterov_pgd(global_best_a, p3_budget * 0.45, rng, beta=beta_try)
            obj_nest = compute_obj(a_nest)
            print("  Nest(beta=%.0f): obj=%.6f" % (beta_try, obj_nest))
            if obj_nest < global_best_obj:
                global_best_obj = obj_nest
                global_best_a = a_nest.copy()

    # === Phase 4: Subgradient loopy ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10
    if p4_budget > 10:
        print("\n=== Phase 4: Subgradient Loopy (%.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a_sg = run_subgradient_loopy(global_best_a, p4_budget, rng)
        obj_sg = compute_obj(a_sg)
        print("  SubGrad: obj=%.6f (%.1fs)" % (obj_sg, time.monotonic() - t4))
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()

    # === Phase 5: Multi-scale upsample ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p5_budget = remaining * 0.55
        print("\n=== Phase 5: Upsample to n=%d (%.0fs) ===" % (n2, p5_budget))
        t5 = time.monotonic()

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)
        obj_up_init = compute_obj(a_up)
        print("  Upsampled init: obj=%.6f" % obj_up_init)

        # Optimize the upsampled version
        a_up1 = run_adaptive_lse(a_up, p5_budget * 0.5, rng, beta_init=10.0)
        obj_up1 = compute_obj(a_up1)
        print("  Up LSE: obj=%.6f (%.1fs)" % (obj_up1, time.monotonic() - t5))
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 10:
            a_up2 = run_nesterov_pgd(global_best_a, rem5, rng, beta=200.0)
            obj_up2 = compute_obj(a_up2)
            print("  Up PGD: obj=%.6f" % obj_up2)
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = remaining * 0.85
        print("\n=== Phase 6: Final polish (%.0fs) ===" % p6_budget)
        t6 = time.monotonic()

        a_fin = run_adaptive_lse(global_best_a, p6_budget * 0.5, rng, beta_init=200.0)
        obj_fin = compute_obj(a_fin)
        print("  Final LSE: obj=%.6f (%.1fs)" % (obj_fin, time.monotonic() - t6))
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 10:
            a_fin2 = run_nesterov_pgd(global_best_a, rem6, rng, beta=1000.0)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final PGD: obj=%.6f" % obj_fin2)
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL === obj=%.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
