"""
Minimize 2*n*max(f*f)/sum(f)^2 using adaptive multi-scale LSE optimization
with Newton active-set refinement and strategic restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient_and_value(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(np.conj(W) * A))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def run_fista_adaptive(a_init, n, S, max_time, rng,
                       beta_start=30.0, beta_end=50000.0,
                       beta_steps=300, restart_every=500,
                       noise_scale_base=0.005):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta_curve = np.linspace(np.log(beta_start), np.log(beta_end), beta_steps)
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        idx = int(frac * (len(beta_curve) - 1))
        beta = np.exp(beta_curve[idx])
        
        # Occasionally decrease beta to explore
        if it % restart_every == 0 and beta > 100:
            beta = max(10.0, beta * 0.1)
            t_mom = 1.0
            y = a.copy()
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad, lse_val = lse_gradient_and_value(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size with backtracking
        step = 2.0 / (beta * gnorm + 1e-10)
        step = min(step, 1.0 / beta * 5)
        
        stagnant = it - last_improve
        if stagnant > restart_every // 2:
            t_mom = 1.0
            noise = noise_scale_base * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
            last_improve = it
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def newton_active_set(a_init, n, S, max_time, rng, max_active=20):
    """
    Newton's method on the active set of lags.
    At the optimum, for active lags k: c_k = M, and for inactive: c_k < M.
    We solve for f and M such that c_k = M for active k, sum(f) = S, f >= 0.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Identify active set: lags within epsilon of max
        eps = 0.1 / (it + 1)  # Tightening threshold
        active_mask = c >= cur_max - eps
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])
        
        if len(active_idx) > max_active:
            # Keep only the top max_active
            top_vals = np.sort(c)[::-1][:max_active]
            threshold = top_vals[-1]
            active_idx = np.where(c >= threshold)[0]
        
        # For the active set, we want to reduce c_k for k in active_idx
        # Gradient of c_k w.r.t. f_j is f_{k-j}
        # So the combined gradient for the active set is:
        # grad_j = sum_{k in active} f_{k-j}
        
        # This is a cross-correlation
        w = np.zeros(2 * n - 1)
        w[active_idx] = 1.0 / len(active_idx)
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = np.real(ifft(np.conj(W) * A))[:n]
        
        # Newton step: we want to find delta such that
        # c_{k+delta} = M - epsilon for active k
        # Linearizing: c_k + grad_k . delta = M - eps
        # With constraint: sum(delta) = 0 (maintain sum)
        
        # Simple gradient descent with line search on the active set gradient
        step = 0.01
        base = cur_max
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            test_max = c_test.max()
            if test_max < base * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try a different direction: increase elements that don't contribute to peaks
            # Find elements with low "responsibility"
            resp = np.zeros(n)
            for k in active_idx[:10]:  # Limit computation
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                js = np.arange(lo, hi + 1)
                resp[js] += a[js] * a[k - js]
            
            # Reduce high responsibility, increase low responsibility
            resp -= resp.mean()
            direction = -resp / (np.linalg.norm(resp) + 1e-10)
            step = 0.01
            for _ in range(10):
                a_test = project_simplex(a + step * direction, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < base * (1 - 1e-7):
                    a = a_test
                    improved = True
                    break
                step *= 0.5
        
        if not improved:
            # Inject noise
            noise = 0.01 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)
            else:
                break
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def generate_exponential_init(n, decay=0.7, center=0.3):
    t = np.arange(n) / n
    a = np.exp(-decay * np.abs(t - center) * n / 3)
    return a


def generate_ramp_init(n, slope=1.0, offset=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * (t - offset)
    return np.maximum(a, 0.01)


def generate_multi_peak_init(n, rng, n_peaks=5):
    a = np.ones(n) * 0.1
    for _ in range(n_peaks):
        center = rng.integers(0, n)
        width = rng.integers(n // 20, n // 5)
        height = rng.uniform(0.5, 2.0)
        idx = np.arange(n)
        a += height * np.exp(-0.5 * ((idx - center) / width) ** 2)
    return a

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        # Upscale
        t_old = np.linspace(0, 1, small_n)
        t_new = np.linspace(0, 1, n)
        a_up = np.interp(t_new, t_old, a_small_opt)
        a_candidate = normalize_sum(a_up, n)

        c = fft_convolve_self(a_candidate)
        if c.max() < best:
            best = c.max()
            best_a = a_candidate.copy()

    return best_a, best


def surgical_peak_reduction(a_init, n, S, max_time, rng, n_top=5):
    """Directly reduce the top peaks by adjusting responsible coordinates."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Get top lags
        top_lags = np.argsort(c)[::-1][:n_top]

        # For each top lag, find the most responsible pair
        # and try to reduce it
        direction = np.zeros(n)
        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Responsibility: reduce f_j where contrib is large
            # But we need to maintain sum, so we shift mass from high to low contrib
            direction[js] -= contrib

        # Also, for lags that are below the peak, we can increase there
        # (this helps maintain sum while reducing peaks)
        # Add a small positive component for low-contrib positions
        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Positions with low contribution can be increased
            direction[partners] += contrib * 0.5

        direction -= direction.mean()  # Zero mean to help maintain sum
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break

        direction /= dnorm
        step = 0.05 * S / n
        improved = False
        for ls in range(10):
            a_test = project_simplex(a + step * direction, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1.0 - 1e-8):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            noise = 0.01 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if noise > 1e-12:
                a = project_simplex(a + rng.normal(0, noise, n), S)
            else:
                break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(f*f)/sum(f)^2 using:
    1. Adaptive LSE-FISTA with cooling restarts
    2. Newton active-set KKT solver
    3. Block coordinate descent
    4. Multi-scale refinement
    5. Surgical peak reduction
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    n = 8192
    S = np.sqrt(2.0 * n)

    print("Budget: %.0fs, n=%d, deadline margin: %.0fs" % (budget_s, n, safety))

    # Load given sequence
    a_init = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
            a_init = np.clip(given, 0, 1000)
            print("Loaded given sequence: n=%d, sum=%.2f" % (len(a_init), a_init.sum()))
    except Exception:
        pass

    if a_init is None:
        a_init = generate_ramp_init(n, slope=2.5, offset=0.0)
        print("Using ramp initialization")

    # Track best
    a_init = normalize_sum(a_init, n)
    c_init = fft_convolve_self(a_init)
    global_best = c_init.max()
    global_a = a_init.copy()
    print("Initial: %.8f" % global_best)

    def check_and_update(a):
        pass  # We'll handle this inline

    # === Phase 1: Main LSE-FISTA (45% of time) ===
    t1 = (deadline - time.monotonic()) * 0.45
    t1 = max(30.0, t1)
    print("=== Phase 1: LSE-FISTA (%.0fs) ===" % t1)
    a1, v1 = run_fista_adaptive(a_init, n, S, t1, rng,
                                 beta_start=30.0, beta_end=60000.0,
                                 beta_steps=350, restart_every=350,
                                 noise_scale_base=0.006)
    c1 = fft_convolve_self(a1)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
        print("  Phase 1 best: %.8f" % global_best)
    else:
        print("  Phase 1: no improvement (%.8f)" % c1.max())

    # === Phase 2: Newton active-set (20% of time) ===
    remaining = deadline - time.monotonic()
    t2 = max(20.0, remaining * 0.20)
    print("=== Phase 2: Newton active-set (%.0fs) ===" % t2)
    a2, v2 = newton_active_set_opt(global_a, n, S, t2, rng, max_active=12)
    if v2 < global_best:
        global_best = v2
        global_a = a2.copy()
        print("  Phase 2 best: %.8f" % global_best)
    else:
        print("  Phase 2: no improvement (%.8f)" % v2)

    # === Phase 3: Surgical peak reduction (15% of time) ===
    remaining = deadline - time.monotonic()
    t3 = max(15.0, remaining * 0.15)
    print("=== Phase 3: Surgical (%.0fs) ===" % t3)
    a3, v3 = surgical_peak_reduction(global_a, n, S, t3, rng, n_top=6)
    if v3 < global_best:
        global_best = v3
        global_a = a3.copy()
        print("  Phase 3 best: %.8f" % global_best)
    else:
        print("  Phase 3: no improvement (%.8f)" % v3)

    # === Phase 4: Block coordinate descent (10% of time) ===
    remaining = deadline - time.monotonic()
    t4 = max(10.0, remaining * 0.10)
    print("=== Phase 4: Block CD (%.0fs) ===" % t4)
    a4, v4 = block_coordinate_descent(global_a, n, S, t4, rng, block_size=64)
    if v4 < global_best:
        global_best = v4
        global_a = a4.copy()
        print("  Phase 4 best: %.8f" % global_best)
    else:
        print("  Phase 4: no improvement (%.8f)" % v4)

    # === Phase 5: High-beta polish (7% of time) ===
    remaining = deadline - time.monotonic()
    t5 = max(10.0, remaining * 0.10)
    print("=== Phase 5: High-beta polish (%.0fs) ===" % t5)
    a5, v5 = run_fista_adaptive(global_a, n, S, t5, rng,
                                 beta_start=10000.0, beta_end=200000.0,
                                 beta_steps=50, restart_every=300,
                                 noise_scale_base=0.001)
    if v5 < global_best:
        global_best = v5
        global_a = a5.copy()
        print("  Phase 5 best: %.8f" % global_best)
    else:
        print("  Phase 5: no improvement (%.8f)" % v5)

    # === Phase 6: Multi-scale if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t6 = remaining * 0.4
        print("=== Phase 6: Multi-scale (%.0fs) ===" % t6)
        a6, v6 = multi_scale_refine(global_a, n, S, t6, rng)
        if v6 < global_best:
            global_best = v6
            global_a = a6.copy()
            print("  Phase 6 best: %.8f" % global_best)
        else:
            print("  Phase 6: no improvement (%.8f)" % v6)
        # Quick FISTA after multi-scale
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a6b, v6b = run_fista_adaptive(global_a, n, S, remaining * 0.5, rng,
                                           beta_start=5000.0, beta_end=100000.0,
                                           beta_steps=30, restart_every=200)
            if v6b < global_best:
                global_best = v6b
                global_a = a6b.copy()
                print("  Phase 6b best: %.8f" % global_best)

    # === Phase 7: Alternative initializations if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print("=== Phase 7: Alternative inits (%.0fs) ===" % remaining)
        inits = [
            generate_exp_decay_init(n, decay=1.5, center=0.25),
            generate_ramp_init(n, slope=4.0, offset=0.1),
            generate_two_bump_init(n, rng, c1=0.2, c2=0.8, w=0.12),
        ]
        per_init = max(5.0, remaining / (len(inits) + 1))
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            init = normalize_sum(init, n)
            c_i = fft_convolve_self(init)
            print("  Init %d initial: %.4f" % (i, c_i.max()))
            ai, vi = run_fista_adaptive(init, n, S, per_init, rng,
                                         beta_start=20.0, beta_end=30000.0,
                                         beta_steps=100)
            if vi < global_best:
                global_best = vi
                global_a = ai.copy()
                print("  Alt init %d best: %.8f" % (i, global_best))

    # Final Newton polish
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        a_fin, v_fin = newton_active_set_opt(global_a, n, S, min(remaining * 0.5, 30.0), rng, max_active=10)
        if v_fin < global_best:
            global_best = v_fin
            global_a = a_fin.copy()

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate=%.10f, internal=%.10f" %
              (ver if np.isfinite(ver) else 999.0, global_best))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(n) * (S / n)
    else:
        print("WARNING: No valid result, using uniform fallback")
        global_a = np.ones(n) * (S / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n) * (S / n)

    print("RETURNING: length=%d, best_internal=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
