```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LSE-FISTA for bulk optimization, LP-direction descent for refinement,
and peak shaving for final polish.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and scale to have sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * (S / s)


def lse_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with adaptive LSE smoothing for the max of self-convolution."""
    a = normalize_to_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta schedule
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve_it = it
        
        # LSE gradient: d/dam [(1/beta)*log(sum_k exp(beta*c_k))]
        # = (2/beta) * sum_k c_k*exp(beta*(c_k - cm)) * a[k-m] / sum_k exp(beta*(c_k-cm))
        # Simplified (dropping the c_k factor which is approximately constant near max):
        e = np.exp(np.clip(beta * (c - cm), -100, 100))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        cross = np.real(ifft(W * np.conj(A)))[:n]
        grad = 2.0 * cross
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size
        step = 1.0 / beta
        
        # Stagnation handling
        stagnant = it - last_improve_it
        if stagnant > 300:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / (20000.0 / max(n, 1)))
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.2
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def build_lp_matrix(a, active_lags):
    """Build sparse constraint matrix C where C[i,j] = a[lag_i - j] if valid."""
    n = len(a)
    K = len(active_lags)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(a[k - js].tolist())
    
    if not rows:
        return csr_matrix((K, n))
    return csr_matrix((vals, (rows, cols)), shape=(K, n))


def lp_direction(a, M, K=200, max_iters_check=50):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for top K lags, g >= 0.
    Returns g or None if infeasible.
    """
    n = len(a)
    c = fft_convolve_self(a)
    active_lags = np.argsort(c)[::-1][:K]
    
    C = build_lp_matrix(a, active_lags)
    
    # LP: maximize sum(g) subject to C@g <= M*ones, g >= 0
    # Convert to: minimize -sum(g) subject to C@g <= M*ones, g >= 0
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g
    except Exception as e:
        print("  LP error: %s" % str(e))
    
    # If infeasible with K constraints, try fewer
    for K_try in [K//2, K//4, K//8]:
        if K_try < 5:
            break
        active_lags = np.argsort(c)[::-1][:K_try]
        C = build_lp_matrix(a, active_lags)
        b_ub = M * np.ones(K_try)
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
            if result.success and result.x is not None:
                g = result.x
                if g.sum() > 1e-10:

        except Exception:

    return None


def peak_shaving(a, n, S, n_peaks=5, n_steps=100, lr_scale=0.01):
    """Surgical peak shaving: reduce contributions to top peak lags."""
    a = a.copy()
    for step in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility for each a[j]
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.99)
            if weight < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += contrib * weight
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    return a
