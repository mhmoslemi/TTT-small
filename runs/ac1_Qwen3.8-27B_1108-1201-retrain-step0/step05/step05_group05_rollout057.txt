"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale precision optimization.

Strategy:
1. Start from the given best sequence (height_sequence_1)
2. Phase 1: High-beta LSE FISTA with adaptive step size (main workhorse)
3. Phase 2: Exact subgradient peak-shaving with multi-peak weighting
4. Phase 3: Coordinate descent / pair-swap polishing with exact line search
5. Phase 4: Ultra-high-beta final refinement with very small steps
6. Multi-restart with small perturbations to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution with adaptive beta."""
    n = len(a)
    cm = c.max()
    spread = c.max() - c.min()
    if spread < 1e-15:
        return np.zeros(n)
    beta_eff = min(beta, 400.0 / spread)
    scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def multi_peak_subgradient(a, c, n_peaks=5, temperature=1e-4):
    """
    Compute subgradient weighted by softmax over top peaks.
    This gives a smoother direction than single-peak, but more focused than LSE.
    """
    n = len(a)
    cmax = c.max()
    # Weight peaks by exp(-(cmax - c_k) / (temperature * cmax))
    weights = np.exp(-(cmax - c) / (temperature * cmax + 1e-30))
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    else:
        weights = weights / wsum
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_peak_subgradient(a, c, k_peaks=3):
    """
    Compute exact subgradient as uniform average over top-k peak lags.
    """
    n = len(a)
    cmax = c.max()
    # Find lags within a small tolerance of the max
    tol = 1e-6 * cmax
    active = np.where(c >= cmax - tol)[0]
    if len(active) > k_peaks:
        # Take the top k_peaks
        active = active[np.argsort(c[active])[::-1][:k_peaks]]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    grad = np.zeros(n)
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    for k in active:
        w_full = np.zeros(len(c))
        w_full[k] = 1.0
        W = fft(w_full, fsize)
        grad += 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    grad /= len(active)
    return grad


def fista_high_beta(a_init, S, max_time, rng, beta_init=100.0, beta_final=200000.0,
                    use_line_search=True, exact_every=5000):

    t_mom = 1.0

    best_a = a.copy()

    last_improve = 0
    c = c0

        it += 1
        progress = (time.monotonic() - start) / max_time

        beta = beta_init * (beta_final / beta_init) ** progress

        grad = lse_gradient(y, c, beta)

        if gnorm < 1e-15:
            break

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

        t_mom = t_new

                last_improve = it
        
        # Periodic exact subgradient step (peak shaving)
        if exact_every > 0 and it % exact_every == 0:
            c = fft_convolve_self(a)
            cur_val = 2.0 * n * c.max() / (a.sum() ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                last_improve = it

def coordinate_descent_polish(a_init, S, max_time, rng, n_pairs_per_cycle=4096):
    """
    Coordinate descent via pair swaps with exact line search.
    At each step, pick a pair (i,j) and find the optimal mass transfer.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_val = 2.0 * n * c.max() / (a.sum() ** 2)
    best_a = a.copy()
    
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Precompute the convolution matrix structure for fast updates
        # When we change a_i by +d and a_j by -d:
        # conv(a,a)_k changes by:
        #   d * (a_{k-i} + a_{k-j}) - d * (a_{k-j} + a_{k-i}) + ... 
        # Actually: new_a = a with a_i += d, a_j -= d
        # new_conv_k = conv_k + d*(a_{k-i} - a_{k-j}) + d*(a_{i+k} - a_{j+k}) - 2*d*(a_{i+k-j} - a_{j+k-i})... 
        # This is getting complicated. Let's just do full FFT updates.
        
        # Pick random pairs, biased towards positions that contribute to peaks
        cmax = c.max()
        peak_lags = np.where(c >= cmax - 0.01 * cmax)[0]
        
        for _ in range(n_pairs_per_cycle):
            if time.monotonic() - start > max_time - 2:
                break
            
            # Choose pair: one position that "feeds" a peak, one that doesn't
            if len(peak_lags) > 0:
                k_peak = peak_lags[rng.integers(len(peak_lags))]
                # Positions i such that a_i * a_{k_peak - i} contributes to the peak
                valid_i = np.arange(max(0, k_peak - n + 1), min(k_peak + 1, n))
                if len(valid_i) > 0:
                    i = valid_i[rng.integers(len(valid_i))]
                else:
                    i = rng.integers(n)
            else:
                i = rng.integers(n)
            
            j = rng.integers(n)
            if i == j:
                continue
            
            if a[i] < 1e-15:
                continue
            
            # Try transferring mass from i to j (or j to i)
            # Use a small step and check if it helps
            base = 0.01 * min(a[i], a[j] + S / n * 0.1)
            for d in [base, base * 0.5, base * 2, base * 0.1]:
                if d >= a[i]:
                    continue
                a_copy = a.copy()
                a_copy[i] -= d
                a_copy[j] += d
                a_copy = np.maximum(a_copy, 0.0)
                # Rescale to maintain sum
                s = a_copy.sum()
                if s < 1e-15:
                    continue
                a_copy = a_copy * (S / s)
                
                c_copy = fft_convolve_self(a_copy)
                val_copy = 2.0 * n * c_copy.max() / (a_copy.sum() ** 2)
                
                if val_copy < best_val:
                    best_val = val_copy
                    best_a = a_copy.copy()
                    a = a_copy
                    c = c_copy
                    stagnant = 0
                    break
            else:
                stagnant += 1
        
        if it % 10 == 0:
            print("    CD iter %d: obj=%.8f, stagnant=%d" % (it, best_val, stagnant))
    
    return best_a, best_val

def fourier_phase_mutate(a, rng, scale=0.05):
    """Fourier-domain phase mutation for diversity."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.02 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

        n_local = len(a)
        obj = 2.0 * n_local * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  ** New best: obj=%.8f" % obj)
            return True
        return False
    
    # Get initial point
    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if a0.size == n and np.sum(np.clip(a0, 0, 1000)) > 0:
            a0 = np.clip(a0, 0, 1000)
            a0 = project_simplex(a0, S)
            print("Using given sequence: length=%d" % len(a0))
        else:
            a0 = np.ones(n)
            a0 = project_simplex(a0, S)
            print("Given sequence invalid, using uniform init")
    except Exception:
        a0 = np.ones(n) * S / n
        print("No given sequence, using uniform init")
    
    c0 = fft_convolve_self(a0)
    val0 = 2.0 * n * c0.max() / (a0.sum() ** 2)
    print("Initial objective: %.8f" % val0)
    global_best = val0
    global_a = a0.copy()
    
    # === Phase 1: High-beta FISTA (main workhorse) ===
    print("\n=== Phase 1: High-beta FISTA ===")
    p1_start = time.monotonic()
    p1_budget = min(600.0, (deadline - p1_start) * 0.25)
    
    a_opt, val_opt = fista_high_beta(a0, S, p1_budget, rng,
                                      beta_init=50.0, beta_final=100000.0,
                                      exact_every=3000)
    update_global(a_opt, val_opt)
    print("  Phase 1 done: obj=%.8f" % global_best)
    
    # === Phase 2: Multi-restart with small perturbations ===
    print("\n=== Phase 2: Multi-restart polish ===")
    p2_start = time.monotonic()
    p2_budget = min(500.0, (deadline - p2_start) * 0.25)
    
    n_restarts = 3
    per_restart = p2_budget / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() > deadline - 400:
            break
        t_alloc = min(per_restart, deadline - time.monotonic() - 400)
        if t_alloc < 30:
            break
        
        # Small perturbation
        scale = 0.02 / (r + 1)
        a_pert = fourier_phase_mutate(global_a, rng, scale)
        a_pert = project_simplex(a_pert, S)
        
        print("  Restart %d: scale=%.4f" % (r, scale))
        a_opt, val_opt = fista_high_beta(a_pert, S, t_alloc, rng,
                                          beta_init=100.0, beta_final=150000.0,
                                          exact_every=2000)
        update_global(a_opt, val_opt)
    
    print("  Phase 2 done: obj=%.8f" % global_best)
    
    # === Phase 3: Coordinate descent polishing ===
    print("\n=== Phase 3: Coordinate descent ===")
    p3_start = time.monotonic()
    p3_budget = min(500.0, (deadline - p3_start) * 0.25)
    
    if p3_budget > 30:
        a_opt, val_opt = coordinate_descent_polish(global_a, S, p3_budget, rng,
                                                     n_pairs_per_cycle=2048)
        update_global(a_opt, val_opt)
        print("  Phase 3 done: obj=%.8f" % global_best)
    
    # === Phase 4: Ultra-fine final polish ===
    print("\n=== Phase 4: Ultra-fine polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 15
    
    if p4_budget > 30:
        # Sub-phase 4a: Ultra-fine with multi-peak
        seg = p4_budget * 0.6
        a_opt, val_opt = ultra_fine_polish(global_a, S, seg, rng)
        update_global(a_opt, val_opt)
        print("  After 4a: obj=%.8f" % global_best)
        
        # Sub-phase 4b: Continue with very high beta FISTA
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_opt, val_opt = fista_high_beta(global_a, S, remaining, rng,
                                              beta_init=20000.0, beta_final=300000.0,
                                              exact_every=1000)
            update_global(a_opt, val_opt)
            print("  After 4b: obj=%.8f" % global_best)
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
