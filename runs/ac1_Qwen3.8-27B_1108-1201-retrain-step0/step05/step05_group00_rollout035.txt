from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    A = fft(a, N)

"""
Minimize 2*n*max_conv/sum^2 using Nesterov-accelerated subgradient with
averaged peak tracking, periodic deflation, and LP-inspired direction finding.

Key innovations over previous approach:
1. Averaged multi-peak subgradient: uses weighted combination of gradients at
   all near-maximal convolution positions for smoother, more effective descent.
2. Nesterov acceleration with Polyak-style step size for O(1/t^2) convergence.
3. Deflation (peak elimination) coordinate steps interleaved with gradient steps.
4. LP-direction approximation: greedily repair constraint violations (faster than full LP).
5. Adaptive top-K: dynamically adjusts how many peaks to track.
6. Multi-start with spectral and structured initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_peak_gradient(a, k, n):
    """Gradient of conv(a,a)[k] with respect to a.
    Returns g where g[j] = 2*a[k-j] if 0<=k-j<n, else 0."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        g[j] = 2.0 * a[k - j]
    return g


def compute_peak_gradient_fft(a, k, N, n):
    """FFT-based computation of gradient of conv(a,a)[k]."""
    e = np.zeros(N)
    e[k] = 1.0
    E = fft(e)
    A = fft(a, N)
    g = 2.0 * np.real(ifft(E * np.conj(A)))[:n]
    return g


def compute_averaged_peak_gradient(a, N, n, c, top_k, fraction=0.05):
    """Compute weighted average of gradients at near-maximal peaks.
    
    Uses weights proportional to (c[k] - c_max + epsilon) for peaks within
    a fraction of the maximum.
    """
    c_max = c.max()
    threshold = c_max - fraction * c_max
    active_mask = c >= threshold
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c)])
    
    # Cap number of active peaks for efficiency
    if len(active_idx) > top_k:
        top_vals = c[active_idx]
        order = np.argsort(top_vals)[::-1][:top_k]
        active_idx = active_idx[order]
    
    # Weights: proportional to how close to max
    vals = c[active_idx]
    weights = np.maximum(vals - c_max * (1 - fraction), 0.0)
    if weights.sum() < 1e-30:
        weights = np.ones(len(active_idx))
    weights = weights / weights.sum()
    
    # Compute gradient as weighted sum of shifted a's
    g = np.zeros(n)
    for w, k in zip(weights, active_idx):
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        g[lo:hi+1] += w * 2.0 * a[int(k) - np.arange(lo, hi + 1)]
    
    return g, active_idx


def deflation_step(a, c, n, N, n_transfers=50, transfer_frac=0.02):
    """Peak elimination: move mass from high-contributing to low-conv positions.
    
    For the current peak, find the dominant contributing positions and
    transfer small amounts of mass to positions with low local convolution.
    """
    a = a.copy()
    S = a.sum()
    c_max = c.max()
    k_star = int(np.argmax(c))
    
    for _ in range(n_transfers):
        # Recompute contribution to peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        
        i_max = int(np.argmax(contrib))
        if a[i_max] < 1e-15:
            break
        
        # Find best target: position with lowest "cost"
        # Cost of adding mass at position p: increases c[k] by ~a[k-p] for peak k
        # We want to minimize the increase at the current peak
        cost = np.zeros(n)
        for p in range(n):
            # Contribution to peak k_star if we increase a[p]
            other = k_star - p
            if 0 <= other < n:
                cost[p] = a[other]
            cost[p] += 0.01 * a[p]  # small penalty for already-large values
        
        # Exclude the source position
        cost[i_max] = np.inf
        i_min = int(np.argmin(cost))
        
        transfer = transfer_frac * a[i_max]
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a *= S / s
    return a


def lp_direction_approx(a, c, N, n, M, n_iters=200):
    """Approximate LP direction: greedily find g >= 0 maximizing sum(g)
    subject to conv(a, g)[k] <= M for all k.
    
    Uses iterative constraint repair (similar to projected subgradient on the LP).
    """
    g = a.copy()
    S = a.sum()
    
    for _ in range(n_iters):
        # Compute conv(a, g)
        Ag = fft(a, N)
        G = fft(g, N)
        conv_ag = np.real(ifft(Ag * G))[:2 * n - 1]
        
        # Check violations
        violations = np.maximum(conv_ag - M, 0)
        if violations.max() < 1e-10:
            break
        
        # Find the most violated constraint
        k_worst = int(np.argmax(violations))
        
        # Find which g[j] contributes most to conv_ag[k_worst]
        # conv_ag[k] = sum_j a[k-j]*g[j]
        lo = max(0, k_worst - n + 1)
        hi = min(n - 1, k_worst)
        contrib_g = np.zeros(n)
        for j in range(lo, hi + 1):
            contrib_g[j] = a[k_worst - j]
        
        # Reduce the most contributing g[j]
        j_reduce = int(np.argmax(contrib_g * g))
        if g[j_reduce] < 1e-15:
            break
        
        # How much to reduce
        excess = violations[k_worst]
        if contrib_g[j_reduce] > 1e-30:
            reduction = min(g[j_reduce], excess / contrib_g[j_reduce] * 0.5)
        else:
            reduction = 0.01 * g[j_reduce]
        
        g[j_reduce] -= reduction
        
        # Redistribute to maintain sum
        # Add to position with least "damage"
        damage = np.zeros(n)
        for p in range(n):
            other = k_worst - p
            if 0 <= other < n:
                damage[p] = a[other]
            damage[p] += 0.01 * g[p]
        damage[j_reduce] = np.inf
        j_add = int(np.argmin(damage))
        g[j_add] += reduction
    
    g = np.maximum(g, 0)
    s = g.sum()
    if s > 0:
        g *= S / s
    return g


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (preserving magnitudes) to escape local minima."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    if frac < 1.0:
        phase = np.angle(A)
        n_rand = max(1, int(N * frac))
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    else:
        phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def nesterov_subgrad_block(a_init, n, S, N, max_time, rng, 
                           top_k=50, use_deflation=True, lp_interval=2000):

    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s
    
    y = a.copy()  # Nesterov momentum
    t = 1.0
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    H = 2.0  # upper bound on radius of feasible set
    
    c = compute_conv(y, N)
    best_cmax = c.max()
    
    stall_count = 0
    lp_counter = 0
    defl_counter = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        lp_counter += 1
        defl_counter += 1
        
        # Compute gradient at current y
        c_y = compute_conv(y, N)
        cmax_y = c_y.max()
        
        # Adaptive top-k: increase as we get closer to convergence
        dynamic_top_k = min(top_k, 10 + it // 100)
        g, active_peaks = compute_averaged_peak_gradient(y, N, n, c_y, dynamic_top_k, fraction=0.1)
        
        # Nesterov step
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:
            break
        
        eta = H / (t + 1.0)
        x_new = project_simplex(y - eta * g / gnorm * gnorm, S)
        # Simplified: x_new = project_simplex(y - eta * g, S)
        x_new = project_simplex(y - eta * g, S)
        
        # Update Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum (for gradient computation)
        sy = y.sum()
        if sy > 0:
            y *= S / sy
        a = x_new
        t = t_new
        
        # Check improvement
        if it % 20 == 0:
            c_a = compute_conv(a, N)
            cmax_a = c_a.max()
            if cmax_a < best_cmax - 1e-10:
                best_cmax = cmax_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        # Periodic deflation
        if use_deflation and defl_counter >= 100:
            defl_counter = 0
            c_a = compute_conv(a, N)
            a_defl = deflation_step(a, c_a, n, N, n_transfers=30, transfer_frac=0.01)
            c_defl = compute_conv(a_defl, N)
            if c_defl.max() < best_cmax:
                best_cmax = c_defl.max()
                best_a = a_defl.copy()
            a = a_defl
            y = a.copy()
            t = 1.0
        
        # Periodic LP direction
        if lp_counter >= lp_interval:
            lp_counter = 0
            c_a = compute_conv(a, N)
            M = c_a.max()
            g_dir = lp_direction_approx(a, c_a, N, n, M, n_iters=100)
            # Blend with current
            for blend in [0.1, 0.3, 0.5]:
                a_test = project_simplex((1 - blend) * a + blend * g_dir, S)
                c_test = compute_conv(a_test, N)
                if c_test.max() < best_cmax:
                    best_cmax = c_test.max()
                    best_a = a_test.copy()
                    a = a_test
                    break
            y = a.copy()
            t = 1.0
        
        # Stall detection and restart
        if stall_count >= 5:
            a = spectral_phase_restart(a, N, rng, frac=0.3)
            y = a.copy()
            t = 1.0
            stall_count = 0
            defl_counter = 0
            lp_counter = 0
    
    # Final check
    c_final = compute_conv(a, N)
    if c_final.max() < best_cmax:
        best_cmax = c_final.max()
        best_a = a.copy()
    
    return best_a, best_cmax

def _lbfgs_direction(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        return -g / max(gn, 1e-18)
    
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-10, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 47, 89]:
        a += 0.1 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov subgradient with averaged peaks, deflation, and LSE polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass
    
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = np.clip(np.asarray(a_candidate, dtype=np.float64), 0, 1000)
        s = a_cand.sum()
        if s < 0.01:
            return np.inf
        c = compute_conv(a_cand, N)
        sc = c.max()
        score = 2.0 * n * sc / (s * s)
        if sc / (s * s) < global_best:
            global_best = sc / (s * s)
            global_a = a_cand.copy()
            if label:
                print("  [%s] ratio=%.8f score=%.6f" % (label, global_best, score))
        return score if np.isfinite(score) else np.inf
    
    # === Phase 1: Multi-start Nesterov ===
    print("=== Phase 1: Multi-start Nesterov subgradient ===")
    t1 = time.monotonic()
    phase1_budget = min(600.0, (deadline - t1) * 0.25)
    
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given.copy(), 0, 1000)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    
    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))
    
    for name, arr in inits:
        if time.monotonic() > deadline - 400:
            break
        a_norm = project_simplex(np.asarray(arr, dtype=np.float64), S)
        t0 = time.monotonic()
        rem = max(2.0, per_init - (time.monotonic() - t0) - 1.0)
        a_l, sc_l = nesterov_subgrad_block(a_norm, n, S, N, min(rem, 80.0), rng,
                                            top_k=50, use_deflation=True, lp_interval=3000)
        update_best(a_l, name + "/nesterov")
    
    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best if global_best < np.inf else np.inf))
    
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 1.0 / n  # all-ones ratio
    
    # === Phase 2: LSE L-BFGS polishing ===
    print("=== Phase 2: LSE L-BFGS polishing ===")
    t2 = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2) * 0.35)
    
    a_curr = global_a.copy()
    stall = 0
    beta_starts = [10, 20, 40, 80, 150, 200, 300]
    idx_bs = 0
    
    while time.monotonic() < t2 + phase2_budget and idx_bs < len(beta_starts):
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 20:
            break
        beta_start = beta_starts[idx_bs]
        a_new, sc_new = lse_block(a_curr, n, S, N, block_time, rng,
                                   beta_init=beta_start, beta_max=beta_start * 10)
        ratio_new = sc_new / (a_new.sum() ** 2)
        if ratio_new < global_best - 1e-12:
            global_best = ratio_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2] score=%.6f (beta=%d)" % (2 * n * ratio_new, beta_start))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.4)
                stall = 0
        idx_bs += 1
    
    # === Phase 3: Nesterov polishing with higher top-k ===
    print("=== Phase 3: High-precision Nesterov ===")
    t3 = time.monotonic()
    remaining3 = min(500.0, (deadline - t3) * 0.2)
    if remaining3 > 50:
        a_ng, sc_ng = nesterov_subgrad_block(global_a, n, S, N, remaining3, rng,
                                              top_k=200, use_deflation=True, lp_interval=5000)
        update_best(a_ng, "nesterov_high_k")
    
    # === Phase 4: Final LSE with very high beta ===
    print("=== Phase 4: Final high-beta LSE ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 20
    if remaining4 > 30:
        a_fin, sc_fin = lse_block(global_a, n, S, N, min(remaining4, 300.0),
                                   rng, beta_init=200.0, beta_max=2000.0)
        update_best(a_fin, "final_LSE")
    
    # === Phase 5: Deflation polishing ===
    print("=== Phase 5: Deflation polishing ===")
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 20 and global_a is not None:
        a_polish = global_a.copy()
        S_curr = a_polish.sum()
        for _ in range(200):
            if time.monotonic() > deadline - 5:
                break
            c_p = compute_conv(a_polish, N)
            a_polish = deflation_step(a_polish, c_p, n, N, n_transfers=100, transfer_frac=0.005)
            if _ % 50 == 49:
                c_check = compute_conv(a_polish, N)
                update_best(a_polish, "deflation")
        update_best(a_polish, "deflation_final")
    
    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
            if not np.isfinite(ver):
                print("  Rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("Verification failed: %s" % str(e))
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    final_score = evaluate_sequence([float(x) for x in global_a])
    print("Final: length=%d, score=%.6f" % (len(global_a), final_score))
    return [float(x) for x in global_a]
