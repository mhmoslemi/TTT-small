import numpy as np
def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates the sequence of coefficients.
    Returns np.inf if the input is invalid.
    """
    # ---Security Checks ---
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        # Reject boolean types (they are subclasses of int) and any other non-integer/non-float types
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    # Convert to float and clip negatives
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    n = len(sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    # Clip to reasonable size
    sequence = [min(1000.0, x) for x in sequence]
    b_sequence = np.convolve(sequence, sequence, mode='full')
    max_b = np.max(b_sequence)
    return float(2*n*max_b/(sum_a**2))

def normalize_vector(strategy, vector, S_val):
    """
    Normalization methods vary. 'clip' uses clip-to-sum, 'soft' uses soft-normalization allowing zero means.
    Parameters:
        strategy: 'clip' or 'soft'
        vector: must be in [0, 1000] and have sum>=0.01
        S_val: target sum for clipping normalization.
    """
    vec = np.asarray(vector, dtype=np.float64)
    total = np.sum(vec * np.clip((vec>550)*2, 1, 4))  # Maybe a better factor? 
    # Using a two-stage normalization: first clip then scale, or soft normalization if needed.
    if strategy == 'clip':
        vec = np.clip(vec, 0., 1000.)
        # If still sum is very small (which may only be at construction time)
        total_clip = np.sum(vec)
        if total_clip < 0.01:
            return vec * (S_val / total_clip)
        else:
            alpha = (S_val / total_clip) * (vec**2).mean() / (vec.mean()**2)  # This might under- or oversample
            vec = vec * (S_val / total_clip)
            return vec
    elif strategy == 'soft':
        if total < 0.01:
            # Not enough sum, introduce variations until condition met.
            while total < 0.01:
                vec = np.clip(vec, 0., None)
                total = np.sum(vec)
                if total < 0.01:
                    vec = np.ones(len(vec)) * (total / len(vec) * 10.)  # Enforce sum
            return vec
    return vec  # Default return

def evaluate_convolve(vector, n_val=65536):
    vec = np.asarray(vector, dtype=np.float64)
    s_clr = vec.sum()
    # If s_clr is not S_val, recalculate the convolution.

    return obj_val
