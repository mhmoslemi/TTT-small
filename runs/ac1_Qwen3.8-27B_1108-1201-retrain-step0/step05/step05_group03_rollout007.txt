import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value_and_grad(a, c, beta, k=None):
    """Compute LSE value and gradient of smoothed max of c = a*a."""
    n = len(a)
    size = len(c)
    if k is not None:
        k = min(k, size)
        idx = np.argsort(c)[::-1][:k]
        c_use = c[idx]
    else:
        idx = np.arange(size)
        c_use = c
    
    cm = c_use.max()
    scaled = beta * (c_use - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    
    w_full = np.zeros(size)
    w_full[idx] = w
    
    # Gradient: d/d a[j] of sum_k w_k * c[k] where c[k] = sum_i a[i]*a[i+k]
    # = sum_k w_k * (2*a if k=0, else a shifted)
    # Using FFT: grad = 2 * IFFT(FFT(w_full) * conj(FFT(a)))[:n]
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    val = cm + np.log(tot) / beta
    return val, grad

def spectral_equalize(a, alpha, iters=10):
    """Iteratively flatten the power spectrum."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(iters):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-30:
            break
        # Compress dynamic range
        new_power = power ** alpha if alpha < 1 else power
        # Or: move toward uniform
        new_power = (1 - alpha) * power + alpha * mean_p
        new_power = np.maximum(new_power, 1e-30)
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
    
    return a
