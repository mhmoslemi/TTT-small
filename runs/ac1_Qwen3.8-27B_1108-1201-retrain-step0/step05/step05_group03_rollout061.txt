```
def resample_sequence(a,n):
    a=np.asarray(a,dtype=float).reshape(-1)
    m=a.size
    if m==n: return a.copy()
    if m==0 or n==0: return np.zeros(n)
    if m==1: return np.full(n, float(a[0]))
    xo=np.linspace(0.0,1.0,m)
    xn=np.linspace(0.0,1.0,n)
    b=np.interp(xn, xo, a)
    b=np.maximum(b,0.0)
    return b
```

```python
- Fast FFT-based self-convolution.
- Log-sum-exp smoothing of the maximum convolution coefficient.
- Projection onto the simplex {x >= 0, sum(x) = sqrt(2n)}.
- Nesterov-style projected gradient with adaptive beta annealing,
  line search, restarts, and multiple diverse initializers.
- Low-resolution exploration followed by high-resolution polishing.

The final sequence is returned as a Python list of non-negative floats.
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def next_power_of_two(x):
    """Smallest power of two >= x (for integer-like x)."""
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize(a, S):
    """Clip to [0, 1000] and rescale so that sum(a) == S."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.size, S / float(a.size), dtype=np.float64)
    a = a * (S / s)
    return a


def resample_sequence(a, n):
    """Linearly resample a non-negative sequence to length n."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    m = a.size
    if m == 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    if m == n:
        return a.copy()
    if m == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    xo = np.linspace(0.0, 1.0, m)
    xn = np.linspace(0.0, 1.0, n)
    b = np.interp(xn, xo, a)
    b = np.maximum(b, 0.0)
    return b


def fft_convolve_self(a, fsize=None):
    """Full linear self-convolution by FFT. Returns length 2n-1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if fsize is None:
        fsize = next_power_of_two(2 * n - 1)
    A = fft(a, fsize)
    c = np.real(ifft(A * A, fsize))
    c = c[:2 * n - 1]
    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def convolve_from_fft(A, n):
    """Recover self-convolution from FFT A of a sequence length n."""
    c = np.real(ifft(A * A))
    c = c[:2 * n - 1]
    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def convolve_max(a):
    """max(convolve(a,a))."""
    c = fft_convolve_self(a)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([S], dtype=np.float64)

    # Fast path: already very close to feasible.
    if np.all(v >= -1e-12):
        pos = np.maximum(v, 0.0)
        s = float(pos.sum())
        if s > 0.0 and abs(s - S) <= 1e-9 * max(1.0, S):
            return pos * (S / s)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (css - S) / idx
    valid = theta <= u
    if not np.any(valid):
        return np.full(n, S / float(n), dtype=np.float64)

    r = int(np.max(np.where(valid)[0])) + 1
    theta_r = (css[r - 1] - S) / float(r)
    out = np.maximum(v - theta_r, 0.0)

    s = float(out.sum())
    if s <= 0.0:
        return np.full(n, S / float(n), dtype=np.float64)

    # Tiny numerical correction.
    if abs(s - S) > 1e-10 * max(1.0, S):
        out = out * (S / s)
    return out


def lse_value_weights(c, beta):
    """
    Compute log-sum-exp soft maximum of c and softmax weights.
    beta is the inverse temperature.
    """
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    scaled = beta * (c - cmax)
    np.clip(scaled, -745.0, 700.0, out=scaled)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return cmax, w
    w = e / s
    val = cmax + float(np.log(s)) / beta
    return val, w


def lse_value_only(c, beta):
    """Log-sum-exp soft maximum only."""
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    scaled = beta * (c - cmax)
    np.clip(scaled, -745.0, 700.0, out=scaled)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 0.0:
        return cmax
    return cmax + float(np.log(s)) / beta


def gradient_from_weights(y, weights, A, fsize):
    """
    Gradient of log-sum-exp(c(y)) with respect to y.
    c_k = sum_i y_i y_{k-i}; weights is on c_k.
    """
    W = fft(weights, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:len(y)]
    return grad


def record_candidate(a, best_a, best_obj):
    """
    Evaluate a candidate with our internal objective and update best.
    We use sum(a)=sqrt(2n), so objective = 2*n*max(convolve(a,a)).
    """
    if a is None:
        return best_a, best_obj
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return best_a, best_obj
    a = normalize(a, np.sqrt(2.0 * n))
    B = convolve_max(a)
    obj = 2.0 * n * B
    if np.isfinite(obj) and obj < best_obj:
        return a.copy(), float(obj)
    return best_a, best_obj


def analytic_start_arrays(n, top_k=5):
    """
    Generate a diverse set of analytically shaped initializers.
    They are scored at moderate resolution, then resampled to n.
    """
    ne = min(max(int(n), 32), 2048)
    x = np.linspace(0.0, 1.0, ne)
    S_ne = np.sqrt(2.0 * ne)
    eps = 1e-8

    cands = []
    scores = []

    # Uniform.
    a = np.ones(ne, dtype=np.float64)
    cands.append(a.copy())
    scores.append(convolve_max(normalize(a, S_ne)))

    # One-parameter exponential tilts.
    for lam in (-4.0, -3.0, -2.0, -1.0, 0.0, 1.0, 2.0, 3.0, 4.0):
        a = np.exp(lam * x)
        if a.sum() > 0:
            cands.append(a.copy())
            scores.append(convolve_max(normalize(a, S_ne)))

    # Beta-like powers at the two ends.
    powers = (0.0, 0.5, 1.0, 1.5, 2.0, 3.0)
    for p in powers:
        for q in powers:
            if p == 0.0 and q == 0.0:
                continue
            a = (x + eps) ** p * (1.0 - x + eps) ** q
            if a.sum() > 0:
                cands.append(a.copy())
                scores.append(convolve_max(normalize(a, S_ne)))

    # Symmetric sine powers.
    for r in (0.5, 1.0, 2.0):
        a = np.sin(np.pi * x) ** r
        if a.sum() > 0:
            cands.append(a.copy())
            scores.append(convolve_max(normalize(a, S_ne)))

    # Mild cosine modulations of the uniform density.
    for m in (1, 2, 3):
        a = np.maximum(0.0, 1.0 + 0.7 * np.cos(2.0 * np.pi * m * x))
        if a.sum() > 0:
            cands.append(a.copy())
            scores.append(convolve_max(normalize(a, S_ne)))

    # One-sided tents to provide additional asymmetry.
    for center in (0.25, 0.50, 0.75):
        a = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - center))
        if a.sum() > 0:
            cands.append(a.copy())
            scores.append(convolve_max(normalize(a, S_ne)))

    if not cands:
        cands = [np.ones(n, dtype=np.float64)]

    order = np.argsort(np.array(scores, dtype=np.float64))
    out = []
    for idx in order[:max(1, int(top_k))]:
        a = cands[int(idx)]
        a = resample_sequence(a, n)
        a = normalize(a, np.sqrt(2.0 * n))
        out.append(a)
    return out


def random_smooth_sequence(n, rng):
    """Smooth positive random initializer in the frequency domain."""
    fsize = next_power_of_two(n)
    z = rng.normal(0.0, 1.0, fsize) + 1j * rng.normal(0.0, 1.0, fsize)
    idx = np.arange(fsize)
    d = np.minimum(idx, fsize - idx).astype(np.float64)
    sigma = max(4.0, float(fsize) // 64.0)
    z *= np.exp(-(d * d) / (2.0 * sigma * sigma))
    z[0] = 0.0
    a = np.real(ifft(z, fsize))[:n]
    a = np.abs(a) + 1e-6
    a = a * a
    return a


def optimize_sequence(a_init, max_time, rng,
                      beta_early=10.0, beta_mid=150.0,
                      beta_late=1500.0, beta_final=5000.0,
                      step_init=0.15, use_momentum=True):
    """
    Main local optimizer.
    Minimizes B = max(convolve(a,a)) under sum(a)=sqrt(2n), a>=0.
    Returns (best_a, best_B).
    """
    start = time.monotonic()
    max_time = float(max_time)
    if max_time <= 0.0:
        return np.asarray(a_init, dtype=np.float64).copy(), np.inf

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    fsize = next_power_of_two(2 * n - 1)

    x = normalize(a_init, S)
    y = x.copy()
    last = x.copy()
    t = 1.0

    A = fft(y, fsize)
    c = convolve_from_fft(A, n)
    B_y = float(np.max(c))
    B_x = B_y
    best = B_y
    best_a = y.copy()

    step = float(step_init)
    it = 0
    stuck = 0
    fails = 0
    bad = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / max_time)

        # Piecewise beta annealing: smooth -> sharp.
        if prog < 0.10:
            beta = beta_early
        elif prog < 0.45:
            beta = beta_early * (beta_mid / beta_early) ** ((prog - 0.10) / 0.35)
        elif prog < 0.75:
            beta = beta_mid * (beta_late / beta_mid) ** ((prog - 0.45) / 0.30)
        else:
            beta = beta_late * (beta_final / beta_late) ** ((prog - 0.75) / 0.25)

        if not np.isfinite(beta) or beta < 1e-6:
            beta = 1e-6

        # Evaluate extrapolated point y.
        A = fft(y, fsize)
        c = convolve_from_fft(A, n)
        B_y = float(np.max(c))

        # Prevent egregious overshoot from momentum.
        if B_y > B_x + 0.05:
            y = x.copy()
            t = 1.0
            A = fft(y, fsize)
            c = convolve_from_fft(A, n)
            B_y = float(np.max(c))

        if B_y < best:
            best = B_y
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        val, w = lse_value_weights(c, beta)
        grad = gradient_from_weights(y, w, A, fsize)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        hard_tol = 1e-10 + 5e-6 * max(1.0, B_y)
        smooth_tol = 0.02 * (1.0 - prog) + 1e-5

        if step < 1e-7:
            step = max(1e-5, step_init * 0.05)

        accepted = False
        B_new = B_y
        x_new = x.copy()

        # Backtracking / accept with slight smooth-improvement allowance early.
        for _ in range(4):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new, fsize)
            B_new = float(np.max(c_new))

            if B_new <= B_y + hard_tol:
                accepted = True
                break

            if prog < 0.75:
                val_new = lse_value_only(c_new, beta)
                if val_new < val - 1e-9 and B_new <= B_y + smooth_tol:
                    accepted = True
                    break

            step *= 0.5
            if step < 1e-8:
                break

        if not accepted:
            fails += 1
            if fails > 30:
                y = best_a.copy()
                x = y.copy()
                last = y.copy()
                t = 1.0
                step = max(step_init * 0.5, 1e-4)
                B_x = best
                fails = 0
                bad = 0
                stuck = 0
            else:
                noise = (0.0005 + 0.0001 * min(20, fails)) * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0.0, noise, n), S)
                x = y.copy()
                last = x.copy()
                t = 1.0
                B_x = best
                fails = 0
            continue

        fails = 0

        if B_new < best:
            best = B_new
            best_a = x_new.copy()
            stuck = 0
            bad = 0
            step = min(1.0, step * 1.06)
        else:
            if B_new > best + 0.01:
                bad += 1
            else:
                bad = 0
            step = max(1e-5, step * 0.97)

        B_x = B_new

        if use_momentum and B_new <= B_y + hard_tol:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_new = x_new + ((t - 1.0) / t_new) * (x_new - last)
            y = project_simplex(y_new, S)
            last = x_new
            x = x_new
            t = t_new
        else:
            last = x_new
            x = x_new
            y = x_new.copy()
            t = 1.0

        # Too many steps away from best: reset to best.
        if bad > 40:
            y = best_a.copy()
            x = y.copy()
            last = y.copy()
            t = 1.0
            B_x = best
            bad = 0
            stuck = 0
            continue

        # Long stagnation: perturb around best.
        if stuck > 800:
            noise = (0.001 + 0.003 * min(1.0, stuck / 4000.0)) * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0.0, noise, n), S)
            x = y.copy()
            last = y.copy()
            t = 1.0
            B_x = best
            step = step_init * 0.5
            stuck = 0

    # Final checks.
    c = fft_convolve_self(x, fsize)
    Bx = float(np.max(c))
    if Bx < best:
        best = Bx
        best_a = x.copy()

    c = fft_convolve_self(y, fsize)
    By = float(np.max(c))
    if By < best:
        best = By
        best_a = y.copy()

    return best_a, best


def run_starts_by_fractions(starts, fractions, n, time_budget, deadline, rng,
                            beta_early=10.0, beta_mid=150.0,
                            beta_late=1500.0, beta_final=5000.0,
                            step_init=0.15, use_momentum=True):
    """
    Run optimize_sequence on several initializers, allocating time fractions.
    Returns (best_a, best_objective) for this group.
    """
    best_a = None
    best_obj = np.inf
    if not starts or time_budget <= 0.0:
        return best_a, best_obj

    for i, s in enumerate(starts):
        now = time.monotonic()
        remaining = deadline - now
        if remaining <= 60.0:
            break

        if i < len(fractions):
            frac = float(fractions[i])
        else:
            frac = 0.10

        tmax = frac * float(time_budget)
        cap = remaining - 30.0
        if tmax > cap:
            tmax = cap
        if tmax < 15.0:
            if cap >= 15.0 and i == 0:
                tmax = 15.0
            else:
                break

        a0 = resample_sequence(s, n)
        a0 = normalize(a0, np.sqrt(2.0 * n))

        t0 = time.monotonic()
        a_opt, B_opt = optimize_sequence(
            a0,
            tmax,
            rng,
            beta_early=beta_early,
            beta_mid=beta_mid,
            beta_late=beta_late,
            beta_final=beta_final,
            step_init=step_init,
            use_momentum=use_momentum
        )
        if a_opt is not None:
            a_opt = normalize(a_opt, np.sqrt(2.0 * n))
            obj = 2.0 * n * B_opt
        else:
            obj = np.inf

        if np.isfinite(obj) and obj < best_obj:
            best_obj = float(obj)
            best_a = a_opt.copy()

        print(
            "start %d/%d time=%.0fs obj=%.6f best=%.6f"
            % (i + 1, len(starts), time.monotonic() - t0,
               obj if np.isfinite(obj) else -1.0,
               best_obj if np.isfinite(best_obj) else -1.0)
        )

    return best_a, best_obj


def propose_candidate(seed=20261, budget_s=2400.0):
    """
    Search for a non-negative sequence minimizing the task objective.
    """
    t_start = time.monotonic()

    # Reserve time for final validation and evaluator safety.
    total_time = min(float(budget_s), 2200.0)
    deadline = t_start + total_time - 120.0

    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # Try to import the previously found sequence, if available.
    prev_a = None
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            arr = np.asarray(hs, dtype=np.float64).reshape(-1)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and float(np.sum(arr)) > 0.0:
                prev_a = arr
                prev_a = normalize(prev_a, np.sqrt(2.0 * prev_a.size))
    except Exception:
        prev_a = None

    print(
        "propose_candidate start: budget=%.0fs prev_len=%s"
        % (budget_s, prev_a.size if prev_a is not None else 0)
    )

    # -------------------------
    # Phase 1: low-resolution exploration.
    # -------------------------
    quick_best_a = None
    quick_starts = []

    if time.monotonic() < deadline - 900.0:
        n_quick = 1024
        quick_starts = analytic_start_arrays(n_quick, top_k=4)
        if prev_a is not None:
            quick_starts.append(resample_sequence(prev_a, n_quick))
        quick_starts.append(random_smooth_sequence(n_quick, rng))
        quick_starts = quick_starts[:5]

        fractions_q = [0.35, 0.20, 0.15, 0.15, 0.15]
        quick_budget = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.08))

        q_a, q_obj = run_starts_by_fractions(
            quick_starts, fractions_q, n_quick, quick_budget, deadline, rng,
            beta_early=6.0, beta_mid=80.0, beta_late=800.0, beta_final=3000.0,
            step_init=0.20, use_momentum=True
        )

        if q_a is not None:
            best_a, best_obj = record_candidate(q_a, best_a, best_obj)
            quick_best_a = q_a.copy()
            print("quick phase best obj=%.6f" % best_obj)

    # -------------------------
    # Phase 2: main high-resolution search.
    # -------------------------
    main_best_a = None
    remaining = deadline - time.monotonic()
    if remaining > 300.0:
        n_main = 16384
        main_starts = []

        if quick_best_a is not None:
            main_starts.append(quick_best_a)

        if prev_a is not None:
            main_starts.append(prev_a)

        if len(quick_starts) > 0:
            main_starts.append(quick_starts[0])
        if len(quick_starts) > 1:
            main_starts.append(quick_starts[1])

        # Explicit asymmetric ramp starts; these are often close to the optimum.
        main_starts.append(np.linspace(0.0, 1.0, n_main, dtype=np.float64))
        main_starts.append(np.linspace(1.0, 0.0, n_main, dtype=np.float64))

        main_starts = main_starts[:5]
        fractions_m = [0.45, 0.25, 0.15, 0.10, 0.05]

        main_budget = min(1300.0, max(300.0, remaining - 250.0))

        m_a, m_obj = run_starts_by_fractions(
            main_starts, fractions_m, n_main, main_budget, deadline, rng,
            beta_early=10.0, beta_mid=150.0, beta_late=1500.0, beta_final=5000.0,
            step_init=0.15, use_momentum=True
        )

        if m_a is not None:
            best_a, best_obj = record_candidate(m_a, best_a, best_obj)
            main_best_a = m_a.copy()
            print("main phase best obj=%.6f len=%d" % (best_obj, len(main_starts[0])))

    # -------------------------
    # Phase 3: optional 8192 refinement.
    # This both preserves the good old length and can find a local improvement
    # that a length-16384 start missed.
    # -------------------------
    remaining = deadline - time.monotonic()
    if remaining > 400.0 and (not np.isfinite(best_obj) or best_obj > 1.5027):
        n_alt = 8192
        alt_starts = []

        if main_best_a is not None:
            alt_starts.append(main_best_a)
        if prev_a is not None:
            alt_starts.append(prev_a)
        if quick_best_a is not None:
            alt_starts.append(quick_best_a)
        alt_starts.append(np.linspace(0.0, 1.0, n_alt, dtype=np.float64))

        alt_starts = alt_starts[:4]
        fractions_a = [0.45, 0.30, 0.15, 0.10]
        alt_budget = min(350.0, max(120.0, remaining * 0.45))

        a_alt, obj_alt = run_starts_by_fractions(
            alt_starts, fractions_a, n_alt, alt_budget, deadline, rng,
            beta_early=20.0, beta_mid=300.0, beta_late=2000.0, beta_final=5000.0,
            step_init=0.10, use_momentum=True
        )

        if a_alt is not None:
            best_a, best_obj = record_candidate(a_alt, best_a, best_obj)
            print("alt 8192 phase best obj=%.6f" % best_obj)

    # -------------------------
    # Phase 4: final high-beta polish on the best known sequence.
    # -------------------------
    remaining = deadline - time.monotonic() - 15.0
    if remaining > 60.0 and best_a is not None:
        n_best = len(best_a)
        polish_budget = min(300.0, remaining)
        a_best = normalize(best_a, np.sqrt(2.0 * n_best))

        a_pol, B_pol = optimize_sequence(
            a_best,
            polish_budget,
            rng,
            beta_early=800.0,
            beta_mid=2000.0,
            beta_late=3500.0,
            beta_final=5000.0,
            step_init=0.03,
            use_momentum=False
        )

        if a_pol is not None:
            best_a, best_obj = record_candidate(a_pol, best_a, best_obj)
            print("final polish best obj=%.6f len=%d" % (best_obj, len(best_a)))

    # -------------------------
    # Fallback if no result was produced.
    # -------------------------
    if best_a is None:
        n_fb = 1024
        best_a = np.linspace(0.0, 1.0, n_fb, dtype=np.float64)

    # Final sanity normalization and clipping.
    final_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    final_a = np.clip(final_a, 0.0, 1000.0)
    n_final = final_a.size
    if n_final == 0:
        n_final = 1024
        final_a = np.linspace(0.0, 1.0, n_final, dtype=np.float64)

    final_a = normalize(final_a, np.sqrt(2.0 * n_final))

    if final_a.size and float(np.max(final_a)) > 1000.0:
        final_a = final_a * (1000.0 / float(np.max(final_a)))

    if float(final_a.sum()) < 0.01:
        final_a = np.ones(1024, dtype=np.float64)
        final_a = normalize(final_a, np.sqrt(2.0 * 1024))

    # Validate with the official evaluator if available.
    final_list = [float(v) for v in final_a]
    try:
        val = evaluate_sequence(final_list)
    except Exception:
        val = np.inf

    if not np.isfinite(val):
        fallback = np.linspace(0.0, 1.0, 1024, dtype=np.float64)
        fallback = normalize(fallback, np.sqrt(2.0 * 1024))
        final_list = [float(v) for v in fallback]
        try:
            val = evaluate_sequence(final_list)
        except Exception:
            val = np.inf

        if not np.isfinite(val):
            fallback = np.ones(1024, dtype=np.float64)
            fallback = normalize(fallback, np.sqrt(2.0 * 1024))
            final_list = [float(v) for v in fallback]

    print(
        "final: len=%d internal_obj=%.6f evaluator=%s"
        % (len(final_list), best_obj if np.isfinite(best_obj) else -1.0,
           val if np.isfinite(val) else "inf")
    )

    return final_list
```
