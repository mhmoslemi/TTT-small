"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses spectral flattening + subgradient descent + LSE polishing.

import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result

def spectral_flatten(a, n_iter, alpha=0.5, rng=None):
    """
    Iteratively flatten the power spectrum of a.
    a: non-negative sequence
    Returns: refined sequence with flatter autocorrelation
    """
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    a = a.copy()
    S = a.sum()
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = np.inf
    
    for it in range(n_iter):
        A = fft(a, size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        
        # Target: flat spectrum
        # Current: |A|^2
        # Geometric interpolation: |A_new| = |A|^alpha * (1/size)^(1-alpha)/2
        # i.e., |A_new|^2 = |A|^(2*alpha) * (1/size)^(1-alpha)
        
        # Better: use the actual average power as target
        avg_power = np.mean(np.abs(A)**2)
        # |A_new|^2 = |A|^(2*alpha) * avg_power^(1-alpha)
        # |A_new| = |A|^alpha * sqrt(avg_power)^(1-alpha)
        
        target_mag = np.power(np.maximum(mag, 1e-30), alpha) * np.sqrt(avg_power)**(1-alpha)
        
        A_new = phase * target_mag
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Project to non-negativity
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:

        a_new = a_new / a_new.sum()

        c = fft_convolve_self(a_new)
        obj = 2.0 * n * c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a * S  # restore original scale

        # Geometric interpolation toward flat spectrum
        avg_power = np.mean(mag ** 2)
        target_mag = np.power(np.maximum(mag, 1e-30), alpha) * np.sqrt(avg_power) ** (1.0 - alpha)

        A_new = phase * target_mag
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()

        c = fft_convolve_self(a_new)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()

        a_norm = a_new

    return best_a * S


def compute_topk_subgradient(a, c, k):
    """Subgradient from top-k convolution lags with weights proportional to excess."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_top = c[top_idx]
    c_mean = c.mean()
    weights = np.maximum(c_top - c_mean, 1e-10)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def structured_noise(n, scale, rng, n_modes=5):
    """Generate low-frequency structured noise."""
    t = np.arange(n)
    noise = np.zeros(n)
    for m in range(n_modes):
        freq = (m + 1) * 2 * np.pi * (m + 1) / n
        noise += rng.normal() * np.sin(t * freq + rng.uniform(0, 2 * np.pi))
    noise += rng.normal() * 0.3 * np.ones(n)
    return noise * scale


def optimize_momentum_subgrad(a0, S, max_time, rng):
    """Subgradient descent with Nesterov momentum and adaptive top-k."""
    n = len(a0)
    a = project_simplex(a0, S)
    v = np.zeros(n)  # momentum
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    momentum = 0.7
    momentum_target = 0.9

    while time.monotonic() - start < max_time:
        it += 1

        # Nesterov: evaluate at lookahead point
        a_look = project_simplex(a + momentum * v, S)
        c_look = fft_convolve_self(a_look)
        cur_max = c_look.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a_look.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Adaptive k: start small, grow
        k = int(2 + 60 * progress ** 0.7)

        grad = compute_topk_subgradient(a_look, c_look, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = cur_max - c_look.mean()
        if gap < 1e-12:
            break

        # Adaptive step size
        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 3.0 / gnorm)

        # Line search
        accepted = False
        for _ in range(8):
            v_new = momentum * v - step * grad
            a_try = project_simplex(a + v_new, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                v = v_new
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            v = momentum * v - 0.01 * step * grad
            a = project_simplex(a + v, S)

        # Increase momentum over time
        momentum = min(momentum_target, momentum + 0.001)

        # Structured noise when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.001 * (stuck - patience) / 100.0) * S / np.sqrt(n)
            noise = structured_noise(n, noise_scale, rng)
            a = project_simplex(a + noise, S)
            v = np.zeros(n)
            stuck = 0
            patience = int(patience * 1.2)

        if it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      mgm it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best


def optimize_lse(a0, S, max_time, rng, beta_start=50.0, beta_end=20000.0):
    """LSE polishing with exponentially increasing beta."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (1.5 * beta)
        step = min(step, 1.5 / gnorm)

        # Line search
        accepted = False
        for _ in range(6):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - 0.1 * step * grad, S)

    return best_a, best


def optimize_balance_top(a0, S, max_time, rng, m_lags=5):
    """Focus on balancing the top m lags to equal levels."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()

        # Get top m lags
        top_idx = np.argsort(c)[::-1][:m_lags]
        c_top = c[top_idx]
        c_top_max = c_top.max()
        c_top_min = c_top.min()

        # If top lags are well-balanced, use LSE on all
        if c_top_max - c_top_min < 1e-6 * c_top_max:
            break

        # Subgradient weighted toward the highest among top m
        weights = np.zeros(len(top_idx))
        for i in range(len(top_idx)):
            weights[i] = max(0.0, c_top[i] - c_top_min)
        if weights.sum() < 1e-10:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start <= j_end:
                indices = kidx - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * a[indices]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * (c_top_max - c.mean()) / (gnorm * gnorm)
        step = min(step, 2.0 / gnorm)

        accepted = False
        for _ in range(6):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - 0.05 * step * grad, S)

        if it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      bal it={it}, t={elapsed:.0f}s, obj={2*n*best/(S*S):.6f}")

    return best_a, best


def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: spectral flatten, momentum subgradient, LSE polish, balance, restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 50.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW BEST: obj={obj:.6f} ({msg})")
        return obj

    # === Phase 1: Spectral flattening ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(180, (deadline - time.monotonic()) * 0.08)
        print(f"\nPhase 1: Spectral flatten, n={n1}, time={t1:.0f}s")
        n_spec_iter = max(5, int(t1 * 30))  # ~30 iterations per second
        a_spec = spectral_flatten_iter(a1, n_spec_iter, alpha=0.6)
        update_global(a_spec, "Phase 1 spectral")

        # === Phase 2: Momentum subgradient ===
        remaining = deadline - time.monotonic()
        t2 = min(600, remaining * 0.35)
        print(f"\nPhase 2: Momentum subgradient, n={n1}, time={t2:.0f}s")
        a_opt, _ = optimize_momentum_subgrad(a_spec, S1, t2, rng)
        update_global(a_opt, "Phase 2 subgrad")

        # === Phase 3: LSE polish ===
        remaining = deadline - time.monotonic()
        t3 = min(400, remaining * 0.30)
        print(f"\nPhase 3: LSE polish, time={t3:.0f}s")
        a_lse, _ = optimize_lse(a_opt, S1, t3, rng, beta_start=100.0, beta_end=30000.0)
        update_global(a_lse, "Phase 3 LSE")

        # === Phase 4: Balance top lags ===
        remaining = deadline - time.monotonic()
        t4 = min(200, remaining * 0.15)
        print(f"\nPhase 4: Balance top lags, time={t4:.0f}s")
        a_bal, _ = optimize_balance_top(a_lse, S1, t4, rng, m_lags=5)
        update_global(a_bal, "Phase 4 balance")

        # Second LSE pass after balancing
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t4b = min(150, remaining * 0.3)
            print(f"\nPhase 4b: LSE after balance, time={t4b:.0f}s")
            a_lse2, _ = optimize_lse(a_bal, S1, t4b, rng, beta_start=500.0, beta_end=50000.0)
            update_global(a_lse2, "Phase 4b LSE2")

    # === Phase 5: Try alternative lengths ===
    remaining = deadline - time.monotonic()
    if global_a is not None:
        n_best = len(global_a)
        alt_lengths = []
        if n_best < 4096:
            alt_lengths = [4096, 8192]
        elif n_best < 8192:
            alt_lengths = [8192, 16384]
        else:
            alt_lengths = [4096]

        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 80:
                break
            if n_alt == len(global_a):
                continue

            S_alt = np.sqrt(2.0 * n_alt)
            t_alt = min(300, (deadline - time.monotonic()) * 0.2)
            if t_alt < 30:
                continue

            print(f"\nPhase 5: Alternative n={n_alt}, time={t_alt:.0f}s")
            init_alt = generate_init(n_alt, rng, base=global_a)
            init_alt = project_simplex(init_alt, S_alt)

            a_alt1 = spectral_flatten_iter(init_alt, 20, alpha=0.6)
            a_alt2, _ = optimize_momentum_subgrad(a_alt1, S_alt, t_alt * 0.5, rng)
            update_global(a_alt2, f"Phase 5 n={n_alt}")

            rem = deadline - time.monotonic()
            if rem > 20:
                a_alt3, _ = optimize_lse(a_alt2, S_alt, min(100, rem * 0.6), rng,
                                         beta_start=200.0, beta_end=20000.0)
                update_global(a_alt3, f"Phase 5 n={n_alt} LSE")

    # === Phase 6: Intensive final refinement ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t6 = remaining - 20

        print(f"\nPhase 6: Final refinement, n={n}, time={t6:.0f}s")

        # Multiple short LSE passes with increasing beta
        a_curr = global_a.copy()
        n_passes = min(6, int(t6 / 20))
        for pi in range(n_passes):
            if time.monotonic() > deadline - 10:
                break
            t_pass = min(t_pass_base := t6 / n_passes, deadline - time.monotonic() - 10)
            if t_pass < 5:
                break
            beta_s = 500.0 * (10 ** pi)
            beta_e = 5000.0 * (10 ** pi)
            a_curr, _ = optimize_lse(a_curr, S, t_pass, rng,
                                     beta_start=min(beta_s, 50000.0),
                                     beta_end=min(beta_e, 100000.0))
            update_global(a_curr, f"Phase 6 pass {pi}")

    # === Phase 7: Random restarts ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = min(4, int(remaining / 40))
        t_per = (remaining - 20) / max(1, n_restarts)

        print(f"\nPhase 7: {n_restarts} restarts, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            # Different perturbation strategies
            if ri % 3 == 0:
                noise = structured_noise(n, 0.01 * S / np.sqrt(n), rng)
            elif ri % 3 == 1:
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            else:
                # Low-amplitude high-frequency noise
                noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
                noise -= noise.mean()

            a_pert = project_simplex(global_a + noise, S)
            a_pert, _ = optimize_lse(a_pert, S, t_per, rng,
                                     beta_start=200.0, beta_end=30000.0)
            update_global(a_pert, f"Phase 7 restart {ri}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = obj_compute(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, internal={global_best:.6f}, evaluate={ver:.6f}")
    print(f"{'='*60}")

    return final_list
