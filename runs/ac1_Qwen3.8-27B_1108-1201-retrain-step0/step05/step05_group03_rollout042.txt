    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    # sum_{k in active} db[k]/da[j] = sum_{k in active} (a[k-j] + a_reflected)
    # This is 2 * (indicator of active set) * a, evaluated at the right shifts.
    # 
    # In FFT domain: if we let W(f) = DFT of the indicator of {k in active} (as a signal of length fsize),
    # then the gradient is 2 * ifft(A * W)[:n]
    
    W_signal = np.zeros(fsize)
    W_signal[active] = 1.0
    W = fft(W_signal)
    grad = (2.0 / n_active) * np.real(ifft(A * W))[:n]
    
    return grad, c_max

def lse_smooth_gradient(a, c, k, beta):
    """
    Compute gradient of top-k LSE of c w.r.t. a.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    fsize = 1 << (2 * n - 1).bit_length()
    W_signal = np.zeros(fsize)
    W_signal[idx] = w
    W = fft(W_signal)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(A * W))[:n]
    return grad

def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Gaussian perturbation on the phase
    perturb = (rng.standard_normal(fsize) + 1j * rng.standard_normal(fsize)) * amount
    A_new = A + perturb * np.abs(A)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a
    return a_new * (s / a_new.sum()) if a_new.sum() > 1e-15 else a

def spectral_shape(a, alpha, rng):
    """Shape the spectrum to reduce peaks."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-30:
        return a
    # Compress dynamic range of power spectrum
    gamma = 1.0 - alpha  # alpha in (0,1): 0=no change, 1=perfectly flat
    new_power = (power / mean_p) ** gamma * mean_p
    phase = A / np.maximum(np.abs(A), 1e-30)
    A_new = phase * np.sqrt(new_power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a.sum()
    if a_new.sum() < 1e-15:
        return a
    return a_new * (s / a_new.sum())

def run_subgradient(a_init, S, max_time, rng, 
                    start_beta=50.0, max_beta=100000.0,
                    k_start=3, k_max=50, beta_growth=0.002,
                    use_exact=False):
    """
    Proximal gradient method with LSE or exact subgradient.
    """
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    step_scale = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if use_exact:
            grad, _ = compute_peak_gradient(a, c, top_k=None, tol=1e-4)
        else:
            k = min(k_max, k_start + it // 100)
            beta = min(max_beta, start_beta * np.exp(beta_growth * it))
            grad = lse_smooth_gradient(a, c, k, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking
        step = step_scale / (1.0 + 0.001 * it)
        a_new = project_simplex(a - step * grad, S)
        
        # Backtracking
        for _ in range(5):
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + step * gnorm * 0.1:
                break
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
        
        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.01)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a = a_new
        
        # Periodic perturbation
        if stuck > 1000 and it % 200 == 0:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            step_scale = 1.0
        
        # Spectral shaping
        if it % 500 == 0 and (time.monotonic() - start) / max_time < 0.6:
            a = spectral_shape(a, 0.1 + 0.1 * rng.random(), rng)
            step_scale = 1.0
            stuck = 0
    
    return best_a, best

def run_multi_perturb(a_init, S, max_time, rng):
    """

def resample_and_normalize(a, n_target, S):
    """Resample sequence to new length and normalize."""
    n_src = len(a)

    t_new = np.linspace(0, 1, n_target)
    a_new = np.interp(t_new, t_old, a)
    return normalize_seq(a_new, S)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase adaptive subgradient optimization with:
1. Exact active-set peak tracking (no LSE smoothing for final phases)
2. Gap-adaptive step sizing
3. Frequency-domain spectral shaping for exploration
4. Multi-restart with diverse perturbations
5. Progressive tightening of active set tolerance
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def compute_gradient_exact(a, c, active_indices):
    """
    Compute exact subgradient of max(c) w.r.t. a.
    Uses the active set (peaks at or near the max).
    grad[j] = 2 * sum_{k in active} a[k-j] * 1[0 <= k-j < n]
    Computed via FFT as: 2 * IFFT(conj(A) * W)[:n]
    """
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    w = np.zeros(fsize)
    w[active_indices] = 1.0 / len(active_indices)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def compute_gradient_lse(a, c, k, beta):
    """
    Compute gradient of top-k LSE of c w.r.t. a.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    fsize = 1 << (3 * n - 2).bit_length()
    W_signal = np.zeros(fsize)
    W_signal[idx] = w
    W = fft(W_signal, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain (additive noise proportional to magnitude)."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    perturb = (rng.standard_normal(fsize) + 1j * rng.standard_normal(fsize)) * amount
    A_new = A + perturb * np.abs(A)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a.sum()
    ns = a_new.sum()
    if ns < 1e-15:
        return a.copy()
    return a_new * (s / ns)


def spectral_shape(a, alpha, rng):
    """Compress dynamic range of power spectrum."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-30:
        return a.copy()
    gamma = max(0.05, 1.0 - alpha)
    new_power = (power / max(mean_p, 1e-30)) ** gamma * mean_p
    phase = np.ones(fsize, dtype=complex)
    mask = np.abs(A) > 1e-30
    phase[mask] = A[mask] / np.abs(A[mask])
    A_new = phase * np.sqrt(new_power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a.sum()
    ns = a_new.sum()
    if ns < 1e-15:
        return a.copy()
    return a_new * (s / ns)


def run_subgradient_phase(a_init, S, max_time, rng,
                          mode="lse", k_init=100, k_max=300,
                          beta_init=50.0, beta_max=5000.0,
                          beta_growth=0.002, tol_init=0.05, tol_final=0.001,
                          perturbation_rate=0.3):
    """
    Generic proximal subgradient phase.
    mode: "lse" uses top-k LSE, "exact" uses exact active set.
    """
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Compute gradient based on mode
        if mode == "exact":
            # Adaptive tolerance: tightens over time
            tol = tol_init * (1 - progress) + tol_final * progress
            threshold = cur_max * (1.0 - tol)
            active = np.where(c >= threshold)[0]
            max_active = max(2, int(20 * (1 - progress) + 2))
            if len(active) > max_active:
                c_act = c[active]
                order = np.argsort(c_act)[::-1][:max_active]
                active = active[order]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            grad = compute_gradient_exact(a, c, active)
        else:
            # LSE mode
            k = int(k_init + (k_max - k_init) * min(1.0, progress * 2))
            k = max(3, min(k, len(c)))
            beta = min(beta_max, beta_init * np.exp(beta_growth * it))
            grad = compute_gradient_lse(a, c, k, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Gap-adaptive step size
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[min(3, len(sorted_c) - 1)]
        gap_factor = 1.0 + min(3.0, gap / max(cur_max, 1e-10)) * 0.5

        step = step_scale * gap_factor / (1.0 + 0.0003 * it)
        a_new = project_simplex(a - step * grad / gnorm, S)

        # Backtracking line search
        accepted = False
        for ls in range(6):
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + 1e-12:
                accepted = True
                break
            step *= 0.4
            a_new = project_simplex(a - step * grad / gnorm, S)

        if accepted:
            a = a_new
            step_scale = min(8.0, step_scale * 1.015)
        else:
            step_scale = max(0.02, step_scale * 0.4)
            a = a_new

        # Periodic perturbation (only in first half)
        if progress < perturbation_rate and it % 300 == 0:
            noise_scale = 0.004 * S / np.sqrt(n) * (1.0 - progress)
            method = rng.random()
            if method < 0.4:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            elif method < 0.7:
                a = spectral_perturb(a, 0.08, rng)
            else:
                a = spectral_shape(a, 0.15 + 0.1 * rng.random(), rng)
            step_scale = 1.0
            stuck = 0

        # Stuck detection: aggressive restart
        if stuck > 2000 and progress < 0.8:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            step_scale = 1.0
            stuck = 0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_coordinate_descent(a_init, S, max_time, rng, n_sweeps=50):
    """
    Coordinate descent: for each coordinate, find best value via 1D search.
    Focus on coordinates that appear in peak convolution terms.
    """
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    # Precompute which coordinates contribute to top peaks
    fsize = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        if c.max() < best - 1e-14:
            best = c.max()
            best_a = a.copy()

        # Get top peaks
        top_k = min(20, n)
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For each peak, the contributing coordinates are a range
        # Try optimizing random subset of coordinates
        n_trial = max(100, n // 10)
        trial_indices = rng.choice(n, size=n_trial, replace=False)

        for j in trial_indices:
            if time.monotonic() - start > max_time:
                break

            a_old_j = a[j]
            # Try a few candidate values
            candidates = [0.0, a_old_j * 0.5, a_old_j * 0.8, a_old_j * 1.2,
                          a_old_j * 1.5, S / n]
            # Add the current value
            best_val = a_old_j
            best_obj = c.max()

            for cv in candidates:
                a[j] = cv
                # Rescale to maintain sum (approximate: just check the peak)
                # For speed, only compute the affected convolution entries
                # But for correctness, compute full convolution
                c_new = fft_convolve_self(a)

def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

        S_sec = np.sqrt(2.0 * n_sec)
        a_sec = resample_sequence(global_a, n_sec)
        a_sec = normalize_seq(a_sec, S_sec)
        t5 = remaining * 0.25
        print("\n=== Phase 5: n=%d optimization (%.0fs) ===" % (n_sec, t5))
        a_opt, v_opt = run_subgradient_phase(
            a_sec, S_sec, t5, rng,
            mode="lse", k_init=15, k_max=60,
            beta_init=200.0, beta_max=50000.0,
            beta_growth=0.001, tol_init=0.02, tol_final=0.002,
            perturbation_rate=0.2)
        try_update(a_opt, n_sec)

        # Quick exact polish on n=16384 result
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t5b = remaining * 0.3
            a_opt2, v_opt2 = run_subgradient_phase(
                a_opt, S_sec, t5b, rng,
                mode="exact", k_init=3, k_max=3,
                beta_init=1.0, beta_max=1.0,
                beta_growth=0.0, tol_init=0.005, tol_final=0.0005,
                perturbation_rate=0.0)
            try_update(a_opt2, n_sec)

    # === Phase 6: Final intensive polish on best (10% of budget) ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)
        t6 = remaining * 0.4
        print("\n=== Phase 6: Final polish (%.0fs, n=%d) ===" % (t6, n_fin))
        a_opt, v_opt = run_subgradient_phase(
            global_a, S_fin, t6, rng,
            mode="exact", k_init=2, k_max=2,
            beta_init=1.0, beta_max=1.0,
            beta_growth=0.0, tol_init=0.003, tol_final=0.0003,
            perturbation_rate=0.0)
        try_update(a_opt, n_fin)

    # === Phase 7: Quick try at n=4096 (sometimes smaller is better) ===
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_a is not None:
        n_sm = 4096
        S_sm = np.sqrt(2.0 * n_sm)
        a_sm = resample_sequence(global_a, n_sm)
        a_sm = normalize_seq(a_sm, S_sm)
        t7 = min(remaining * 0.3, 100)
        print("\n=== Phase 7: n=%d quick try (%.0fs) ===" % (n_sm, t7))
        a_opt, v_opt = run_subgradient_phase(
            a_sm, S_sm, t7, rng,
            mode="lse", k_init=5, k_max=20,
            beta_init=500.0, beta_max=50000.0,
            beta_growth=0.001, tol_init=0.01, tol_final=0.001,
            perturbation_rate=0.1)
        try_update(a_opt, n_sm)

    # Final verification and cleanup
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Done in %.0fs. Final: length=%d, obj=%.6f" % (elapsed, len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
