For the "peak shaving" step, here's the idea:

- For each such lag k, the contribution is sum_i a[i]*a[k-i]
- We can reduce c[k] by reducing a[i] or a[k-i] for the pairs (i, k-i) that contribute most
- But reducing a[i] also affects other lags
- So we need to be careful: reduce a[i] only if the net effect on all top lags is positive

This is essentially a linear programming problem: minimize the max of (c + delta * G) subject to sum(delta) = 0, a + delta >= 0, where G is the "gradient" matrix showing how each coordinate affects each lag.

```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_and_grad(a, beta):
    """Compute beta*LSE(conv) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()  # numerical stability
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        w = e / tot
        lse_val = cm + np.log(tot) / beta
    
    # Gradient of 2*n*beta*LSE / sum^2 ... 
    # Actually, since we fix sum = S, the objective is 2*n*max(c)/S^2
    # And 2*n*beta*LSE(c)/S^2 approximates 2*n*max(c)/S^2
    # d/dc of beta*LSE(c) = w (the softmax weights)
    # d/d(a) of c[k] = 2*a[k'] (the appropriate cross-correlation)
    # So grad = 2 * (w * a reversed... )
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return lse_val, grad, c, w

def fista_optimize(a_init, max_time, rng, beta_init=100.0, beta_rate=0.02,
                   beta_max=100000.0, step_scale=1.0):
    """FISTA with LSE approximation."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_and_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size with backtracking
        step = step_scale / max(beta, 1.0)
        x_trial = project_simplex(y - step * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step > 1e-16 and bt < 8:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        
        x_new = x_trial
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            noise = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            if noise < 1e-10:
                break
    
    return best_a, best_obj
```

However, if we fix all but one variable, it becomes a quadratic in that variable (convex if the coefficient is positive). This suggests coordinate descent might work.

For coordinate i: c[k] as a function of a[i] = a[i]^2 * [k=0 or k=2n-2 and i=0 or n-1] + a[i] * (sum of other terms)
Specifically, d(c[k])/d(a[i]) = 2*a[k-i] (as derived earlier), and d^2(c[k])/d(a[i]^2) = [k-i==i] (i.e., 1 if k=2i, else 0).

So for a fixed k, c[k] is quadratic in a[i] (convex if k=2i, linear otherwise). The max of these is convex in a[i]. So coordinate descent (minimizing over one coordinate at a time) should converge!

For coordinate i, the objective as a function of a[i] = x is:
f(x) = max_k [c_k(x)] where c_k(x) = c[k]_base + x * B_k + x^2 * [k==2i]

where c[k]_base is c[k] with the terms involving a[i] removed, and B_k = 2*a[k-i] (if valid).

This is a 1-D convex minimization problem (max of convex functions). We can solve it with:
- Bisection on the derivative
- Or: the optimum is where the "active" lags (those achieving the max) have zero subgradient

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization with:
1. Quick FISTA warmup with diverse initializations
2. Progressive LSE-FISTA (increasing beta for sharper max tracking)
3. Stochastic local search for escape from local minima
4. Very high-beta final polish
5. Multi-scale upsample exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(y, beta):
    """Compute beta*LSE(c) value, gradient w.r.t. y, and the convolution c."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        w = e / tot
        lse_val = cm + np.log(tot) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def fista_run(a_init, max_time, rng, beta_init=50.0, beta_rate=0.015,
              beta_max=50000.0, step_scale=1.0, use_bt=True, noise_base=0.003):
    """FISTA with adaptive LSE and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_and_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        if use_bt:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 5.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        stuck = it - last_improve
        if stuck > 2000:
            t_mom = 1.0
            ns = max(1e-8, noise_base * S / np.sqrt(n) * np.exp(-stuck / 4000.0))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            if ns < 1e-12:
                break

    return best_a, best_obj


def stochastic_search(a_init, max_time, rng, sigma_init=None):
    """Stochastic local search: random perturbations with greedy acceptance."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    cur_obj = 2.0 * n * c.max() / (S * S)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0

    if sigma_init is None:
        sigma_init = 0.01 * S / np.sqrt(n)

    sigma = sigma_init

    while time.monotonic() - start < max_time:
        it += 1
        # Generate perturbation
        delta = rng.normal(0, sigma, n)
        a_new = project_simplex(a + delta, S)

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < cur_obj - 1e-15:
            a = a_new
            c = c_new
            cur_obj = obj_new
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
                # Slightly increase sigma on success (exploration)
                sigma = min(sigma * 1.02, sigma_init * 2)
        else:
            # Decrease sigma on failure
            sigma *= 0.98

        stuck = it - last_improve
        if stuck > 20000 and sigma < sigma_init * 0.01:
            break

        # Periodic restart from best
        if stuck > 5000 and it % 500 == 0:
            a = best_a.copy()
            c = fft_convolve_self(a)
            cur_obj = best_obj
            sigma = sigma_init * 0.5

    return best_a, best_obj


def topk_subgradient_step(a, max_time, rng, K_init=1, K_max=50):
    """Polish using top-K subgradient with adaptive K."""
    n = len(a_init = a)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        # Adaptive K: start small, grow when stuck
        if stuck < 100:
            k = K_init
        elif stuck < 500:
            k = min(K_max, K_init + stuck // 100)
        elif stuck < 2000:
            k = min(K_max, 5 + stuck // 200)
        else:
            k = min(K_max, 20 + stuck // 500)

        # Compute top-K subgradient
        top_idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / k
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step with backtracking
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 4.0 * S / gnorm)
        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
            c = fft_convolve_self(a)
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new
                c = fft_convolve_self(a)

        if stuck > 3000 and it % 200 == 0:
            ns = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 6000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fft_convolve_self(a)
            if ns < 1e-12:
                break

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt)
             + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3)
             + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "double_bump":
        a = np.exp(-4 * (tt - 0.25) ** 2 / (0.08 ** 2)) + np.exp(-4 * (tt - 0.75) ** 2 / (0.08 ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init, progressive FISTA, stochastic search, polish, upsample.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from previous result if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    # Normalize
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Quick diverse screening (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.10
    if p1_time > 30:
        print("\n=== Phase 1: Diverse starts (budget %.0fs) ===" % p1_time)
        methods = ["cosine", "gaussian", "chirp", "multi_tone", "random", "double_bump"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = fista_run(a_div, per_time * 0.7, rng,
                                        beta_init=30, beta_rate=0.02,
                                        beta_max=5000, step_scale=0.8,
                                        use_bt=False, noise_base=0.005)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.8f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.8f" % (method, obj_opt))

    # === Phase 2: FISTA medium beta (20% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.20
    if p2_time > 45:
        print("\n=== Phase 2: FISTA medium beta (budget %.0fs) ===" % p2_time)
        a_opt, obj_opt = fista_run(global_best_a, p2_time * 0.5, rng,
                                    beta_init=50, beta_rate=0.01,
                                    beta_max=2000, step_scale=1.0,
                                    use_bt=True, noise_base=0.003)
        print("  FISTA med1: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_opt2, obj_opt2 = fista_run(global_best_a, remaining2, rng,
                                          beta_init=200, beta_rate=0.008,
                                          beta_max=5000, step_scale=0.7,
                                          use_bt=True, noise_base=0.002)
            print("  FISTA med2: obj=%.8f" % obj_opt2)
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # === Phase 3: Stochastic local search (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 45:
        print("\n=== Phase 3: Stochastic search (budget %.0fs) ===" % p3_time)
        a_st, obj_st = stochastic_search(global_best_a, p3_time * 0.6, rng)
        print("  Stochastic: obj=%.8f" % obj_st)
        if obj_st < global_best_obj:
            global_best_obj = obj_st
            global_best_a = a_st.copy()

        # Follow up with FISTA
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            a_f, obj_f = fista_run(global_best_a, remaining3, rng,
                                    beta_init=500, beta_rate=0.005,
                                    beta_max=20000, step_scale=0.5,
                                    use_bt=True, noise_base=0.002)
            print("  FISTA after stochastic: obj=%.8f" % obj_f)
            if obj_f < global_best_obj:
                global_best_obj = obj_f
                global_best_a = a_f.copy()

    # === Phase 4: High-beta FISTA + topk subgradient (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 30:
        print("\n=== Phase 4: High-beta polish (budget %.0fs) ===" % p4_time)

        # Very high beta FISTA
        a_hb, obj_hb = fista_run(global_best_a, p4_time * 0.4, rng,
                                  beta_init=2000, beta_rate=0.003,
                                  beta_max=100000, step_scale=0.3,
                                  use_bt=True, noise_base=0.001)
        print("  High-beta FISTA: obj=%.8f" % obj_hb)
        if obj_hb < global_best_obj:
            global_best_obj = obj_hb
            global_best_a = a_hb.copy()

        # Top-K subgradient
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_tk, obj_tk = topk_subgradient_step(global_best_a, remaining4, rng,
                                                  K_init=1, K_max=30)
            print("  TopK subgrad: obj=%.8f" % obj_tk)
            if obj_tk < global_best_obj:
                global_best_obj = obj_tk
                global_best_a = a_tk.copy()

    # === Phase 5: Multi-scale upsample (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.15
    if p5_time > 30 and len(global_best_a) <= 12000:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.8f" % obj_up)

        # Quick FISTA on upsampled
        a_up_opt, obj_up_opt = fista_run(a_up, min(p5_time * 0.5, deadline - time.monotonic() - 10), rng,
                                          beta_init=10, beta_rate=0.01,
                                          beta_max=3000, step_scale=1.0,
                                          use_bt=False, noise_base=0.003)
        print("  FISTA upsampled: obj=%.8f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()
            print("  *** New best from upsample! ***")

    # === Phase 6: Final polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % remaining)

        # Final high-beta FISTA
        a_fin, obj_fin = fista_run(global_best_a, remaining * 0.4, rng,
                                    beta_init=5000, beta_rate=0.002,
                                    beta_max=200000, step_scale=0.2,
                                    use_bt=True, noise_base=0.0005)
        print("  Final FISTA: obj=%.8f" % obj_fin)
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        # Final stochastic
        remaining6 = deadline - time.monotonic()
        if remaining6 > 10:
            a_fin2, obj_fin2 = stochastic_search(global_best_a, remaining6, rng,
                                                  sigma_init=0.003 * np.sqrt(len(global_best_a) * 2) / np.sqrt(len(global_best_a)))
            print("  Final stochastic: obj=%.8f" % obj_fin2)
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    print("\n=== Final Result ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verified: obj=%.8f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
```
