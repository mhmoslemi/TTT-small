    """
    Compute subgradient of max of top-k peaks of self-convolution.
    
    Parameters:
    a: input sequence, length n
    c: self-convolution, length 2n-1
    k: number of top peaks
    topk_weight_decay: geometric decay factor for weights
    
    Returns:
    g: subgradient, length n
    """
    n = len(a)
    
    if k == 1:
        k_pos = int(np.argmax(c))
        g = np.zeros(n)
        j = k_pos - np.arange(n)
        valid = (j >= 0) & (j < n)
        g[valid] = 2.0 * a[j[valid]]

    else:

        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        weights = topk_weight_decay ** np.arange(k)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)

    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def exact_subgradient(a, c, k_top=1, decay=0.85):
    """
    Compute (sub)gradient of max of top-k peaks of self-convolution.
    
    Parameters:
    a: input sequence, length n
    c: self-convolution, length 2n-1
    k_top: number of top peaks to consider
    decay: geometric decay factor for weights
    
    Returns:
    g: subgradient vector, length n
    """
    n = len(a)
    
    if k_top == 1:
        k_pos = int(np.argmax(c))
        g = np.zeros(n)
        j = k_pos - np.arange(n)
        valid = (j >= 0) & (j < n)
        g[valid] = 2.0 * a[j[valid]]
        return g
    else:
        k_top = min(k_top, len(c))
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        weights = decay ** np.arange(k_top)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, kp in zip(weights, top_idx):
            j = kp - np.arange(n)
            valid = (j >= 0) & (j < n)
            g[valid] += w * 2.0 * a[j[valid]]
        return g


def lse_gradient(a, c, k, beta):
    """Compute gradient of top-k LogSumExp smoothing of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 500:
        beta = 500.0 / max(1e-15, (c_top - cm).max())
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten(a, S, alpha=0.1, n_iter=3):
    """
    Apply spectral flattening to reduce autocorrelation peaks.
    Iteratively modifies the amplitude spectrum to be flatter.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-20:
            break
        
        # Target: flatten power spectrum
        # new_power = power^gamma where gamma < 1 compresses dynamic range
        ratio = power / max(mean_p, 1e-20)
        # Damp peaks, boost valleys
        new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + 0.1 * ratio))
        new_power = np.maximum(new_power, 1e-30)
        
        phase = np.angle(A)
        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        a = normalize(a_new, S)
    
    return a


def spectral_perturb(a, S, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def fista_exact(a_init, S, max_time, rng,
                k_top_init=1, k_top_max=8, k_top_growth=0.001,
                decay=0.85, momentum=True):
    """
    FISTA with exact multi-peak subgradient and backtracking.
    
    Key idea: Use exact subgradient (not LSE) for the direction,
    with adaptive number of peaks and backtracking line search.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0
    
    step = 1.0
    min_step = 1e-10
    
    while time.monotonic() - start < max_time:
        it += 1
        k_top = min(k_top_max, k_top_init + int(k_top_growth * it))
        
        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1
        
        # Compute subgradient
        g = exact_subgradient(y, c_y, k_top, decay)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        g = g / gnorm  # Normalize for stable step size
        
        # Backtracking line search
        x_new = project_simplex(y - step * gnorm * g, S)
        backtracks = 0
        while backtracks < 12:
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.5
            if step < min_step:
                break
            x_new = project_simplex(y - step * gnorm * g, S)
            backtracks += 1
        
        if step > min_step * 2:
            step = min(step * 1.05, 10.0 / gnorm) if gnorm > 1e-10 else step
        
        # Update with momentum
        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
            t_val = t_new
        else:
            y_new = x_new.copy()
            t_val = 1.0
        
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        
        # Restart when stuck
        if stuck > 200:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            step = 1.0
            stuck = 0
            if (time.monotonic() - start) / max_time < 0.6:
                y = spectral_flatten(y, S, alpha=0.15, n_iter=2)
        
        # Periodic spectral step
        if it % 500 == 0 and rng.random() < 0.3:
            y = spectral_flatten(y, S, alpha=0.1, n_iter=1)
            t_val = 1.0
            stuck = 0
        
        # Periodic spectral perturbation
        if it % 1000 == 0 and rng.random() < 0.2:
            y = spectral_perturb(y, S, 0.05, rng)
            t_val = 1.0
            stuck = 0
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def fista_lse_hybrid(a_init, S, max_time, rng,
                     beta_init=10.0, beta_rate=0.005, beta_max=20000.0,
                     k_frac=0.05, momentum=True):
    """
    FISTA with LSE smoothing that transitions to exact subgradient.
    When beta is very large, LSE approximates the max, and we switch
    to exact subgradient for the final direction.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stuck = 0
    
    step = 1.0 / max(2.0 * beta_init * 0.1, 1e-10)
    min_step = 1e-12
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Choose gradient: exact if beta is large, else LSE
        if beta > 5000:
            k_top = min(5, int(1 + beta / 5000))
            g = exact_subgradient(y, c_y, k_top, decay=0.8)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            g_dir = g / gnorm
            effective_lip = gnorm * 2.0
        else:
            g = lse_gradient(y, c_y, k, beta)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            g_dir = g / gnorm
            effective_lip = 2.0 * beta
        
        # Backtracking
        cur_step = min(step, 1.0 / max(effective_lip * 0.5, 1e-10))
        x_new = project_simplex(y - cur_step * gnorm * g_dir, S)
        backtracks = 0
        while backtracks < 10:
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            cur_step *= 0.5
            x_new = project_simplex(y - cur_step * gnorm * g_dir, S)
            backtracks += 1
        
        if backtracks < 5:
            step = min(step * 1.03, 1.0 / max(effective_lip * 0.3, 1e-10))
        else:
            step = cur_step
        
        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
            t_val = t_new
        else:
            y_new = x_new.copy()
            t_val = 1.0
        
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        
        # Restart
        if stuck > 300:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            step = 1.0 / max(2.0 * beta_init * 0.1, 1e-10)
            stuck = 0
            if (time.monotonic() - start) / max_time < 0.7:
                y = spectral_flatten(y, S, alpha=0.12, n_iter=2)
        
        # Spectral step
        if it % 800 == 0 and rng.random() < 0.25:
            y = spectral_flatten(y, S, alpha=0.1, n_iter=1)
            t_val = 1.0
            stuck = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def local_search(a_init, S, max_time, rng, group_size=64):
    """
    Local search: randomly perturb groups of coordinates and keep improvements.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    n_iter = 0
    
    while time.monotonic() - start < max_time:
        n_iter += 1
        # Pick a random group
        gs = min(group_size, n)
        start_idx = rng.integers(0, n - gs + 1)
        
        # Perturb the group
        perturbation = rng.normal(0, 0.01 * S / n, gs)
        a_new = a.copy()
        a_new[start_idx:start_idx + gs] += perturbation
        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best - 1e-12:
            best = c_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller perturbation
            a_new2 = a.copy()
            a_new2[start_idx:start_idx + gs] += perturbation * 0.3
            a_new2 = np.clip(a_new2, 0, None)
            a_new2 = normalize(a_new2, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < best - 1e-12:
                best = c_new2.max()
                best_a = a_new2.copy()
                a = a_new2
    
    return best_a, best


def peak_targeting(a_init, S, max_time, rng):
    """
    Directly target the top peaks by shifting mass away from positions
    that contribute to high convolution values.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
        
        # Find top peaks
        top_k = min(5, len(c))
        top_positions = np.argpartition(c, -top_k)[-top_k:]
        top_positions = top_positions[np.argsort(c[top_positions])[::-1]]
        
        # For the top peak, identify which a[i] values contribute most
        k_star = top_positions[0]
        # Contribution of a[i] to (a*a)(k_star) is 2*a[k_star - i]
        contrib = np.zeros(n)
        j = k_star - np.arange(n)
        valid = (j >= 0) & (j < n)
        contrib[valid] = 2.0 * a[j[valid]] * a[valid]
        
        # Identify high contributors and reduce them
        n_reduce = max(1, n // 50)
        high_idx = np.argsort(contrib)[::-1][:n_reduce]
        
        # Shift mass from high contributors to low contributors
        total_shift = 0.02 * contrib[high_idx].sum()
        total_shift = min(total_shift, 0.01 * S)
        
        low_contrib = contrib.copy()
        low_contrib[high_idx] = 0
        n_boost = max(1, n // 20)
        low_idx = np.argsort(low_contrib)[:n_boost]
        
        a_new = a.copy()
        a_new[high_idx] -= total_shift / len(high_idx)
        a_new[low_idx] += total_shift / len(low_idx)
        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try with random perturbation added
            a_new2 = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
            # Otherwise, do a spectral step
            if it % 100 == 0:
                a = spectral_flatten(a, S, alpha=0.08, n_iter=1)
    
    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def get_initialization(n, rng, given_seq=None):
    """Generate different types of initializations."""
    if given_seq is not None and len(given_seq) >= 32:
        if len(given_seq) == n:
            return given_seq.copy()
        return resample_sequence(given_seq, n)
    
    # Random non-negative
    return rng.exponential(1.0, n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    
    Multi-phase approach:
    1. Coarse-to-fine optimization
    2. FISTA with exact subgradient
    3. LSE-hybrid FISTA  
    4. Peak targeting
    5. Local search
    6. Multi-restart
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)
    
    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, deadline - start_total))
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, obj):
        nonlocal global_best, global_a
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  *** New global best: %.8f (n=%d) ***" % (obj, len(a)))
    
    def eval_obj(a):
        n = len(a)
        S = a.sum()
        if S < 0.01:
            return np.inf
        c = fft_convolve_self(a)
        return 2.0 * n * c.max() / (S * S)
    
    # Primary length
    n_primary = 8192
    S_primary = np.sqrt(2.0 * n_primary)
    
    # === Phase 1: Quick FISTA on given sequence ===
    print("\n=== Phase 1: Quick FISTA on given sequence ===")
    a_start = get_initialization(n_primary, rng, given)
    a_start = normalize(a_start, S_primary)
    
    obj_start = eval_obj(a_start)
    print("  Initial obj: %.8f" % obj_start)
    update_global(a_start, obj_start)
    
    t1 = min(120.0, (deadline - time.monotonic()) * 0.2)
    if t1 > 30:
        a_opt, val_opt = fista_exact(a_start, S_primary, t1, rng,
                                      k_top_init=1, k_top_max=5,
                                      k_top_growth=0.002, decay=0.85)
        obj_opt = eval_obj(a_opt)
        print("  FISTA exact: obj=%.8f (peak=%.6f)" % (obj_opt, val_opt))
        update_global(a_opt, obj_opt)
    
    # === Phase 2: LSE Hybrid FISTA ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 2: LSE Hybrid FISTA (time=%.0fs) ===" % remaining)
    if remaining > 60:
        t2 = min(remaining * 0.30, 300.0)
        if global_a is not None:
            a_cur = normalize(global_a.copy(), S_primary)
        else:
            a_cur = normalize(a_start, S_primary)
        
        a_opt, val_opt = fista_lse_hybrid(a_cur, S_primary, t2, rng,
                                           beta_init=10.0, beta_rate=0.004,
                                           beta_max=20000.0, k_frac=0.05)
        obj_opt = eval_obj(a_opt)
        print("  LSE Hybrid: obj=%.8f (peak=%.6f)" % (obj_opt, val_opt))
        update_global(a_opt, obj_opt)
    
    # === Phase 3: Peak Targeting ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 3: Peak Targeting (time=%.0fs) ===" % remaining)
    if remaining > 60 and global_a is not None:
        t3 = min(remaining * 0.15, 180.0)
        a_opt, val_opt = peak_targeting(global_a, S_primary, t3, rng)
        obj_opt = eval_obj(a_opt)
        print("  Peak targeting: obj=%.8f" % obj_opt)
        update_global(a_opt, obj_opt)
    
    # === Phase 4: Multi-restart FISTA with different initializations ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 4: Multi-restart (time=%.0fs) ===" % remaining)
    
    n_restarts = 3
    time_per_restart = min(120.0, remaining * 0.5 / n_restarts)
    
    for r in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break
        
        # Different initialization for each restart
        if r == 0 and global_a is not None:
            # Perturb best
            a_init = normalize(global_a + rng.normal(0, 0.02 * S_primary / np.sqrt(n_primary), n_primary), S_primary)
        elif r == 1:
            # Spectral perturbation of best
            if global_a is not None:
                a_init = spectral_perturb(global_a, S_primary, 0.3, rng)
            else:
                a_init = rng.exponential(1.0, n_primary)
                a_init = normalize(a_init, S_primary)
        else:
            # Random
            a_init = rng.exponential(1.0, n_primary)
            a_init = normalize(a_init, S_primary)
        
        print("  Restart %d: starting obj=%.8f" % (r + 1, eval_obj(a_init)))
        
        a_opt, val_opt = fista_exact(a_init, S_primary, time_per_restart, rng,
                                      k_top_init=1, k_top_max=8,
                                      k_top_growth=0.003, decay=0.8)
        obj_opt = eval_obj(a_opt)
        print("  Restart %d: obj=%.8f" % (r + 1, obj_opt))
        update_global(a_opt, obj_opt)
    
    # === Phase 5: Intensive polishing on best ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 5: Intensive Polishing (time=%.0fs) ===" % remaining)
    
    if remaining > 30 and global_a is not None:
        # High-beta LSE for fine polishing
        t5a = min(remaining * 0.4, 200.0)
        if t5a > 20:
            a_opt, val_opt = fista_lse_hybrid(global_a, S_primary, t5a, rng,
                                               beta_init=200.0, beta_rate=0.001,
                                               beta_max=50000.0, k_frac=0.02,
                                               momentum=True)
            obj_opt = eval_obj(a_opt)
            print("  High-beta polish: obj=%.8f" % obj_opt)
            update_global(a_opt, obj_opt)
        
        # Exact subgradient with very careful steps
        remaining2 = deadline - time.monotonic()
        t5b = min(remaining2 * 0.4, 150.0)
        if t5b > 15:
            a_opt, val_opt = fista_exact(global_a, S_primary, t5b, rng,
                                          k_top_init=1, k_top_max=3,
                                          k_top_growth=0.0, decay=0.9,
                                          momentum=False)
            obj_opt = eval_obj(a_opt)
            print("  Exact subgradient polish: obj=%.8f" % obj_opt)
            update_global(a_opt, obj_opt)
        
        # Local search
        remaining3 = deadline - time.monotonic()
        t5c = min(remaining3 * 0.5, 100.0)
        if t5c > 10:
            a_opt, val_opt = local_search(global_a, S_primary, t5c, rng, group_size=32)
            obj_opt = eval_obj(a_opt)
            print("  Local search: obj=%.8f" % obj_opt)
            update_global(a_opt, obj_opt)
    
    # === Phase 6: Try different length ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 6: Try n=12288 (time=%.0fs) ===" % remaining)
    
    if remaining > 120 and global_a is not None:
        n_alt = 12288
        S_alt = np.sqrt(2.0 * n_alt)
        a_alt = resample_sequence(global_a, n_alt)
        a_alt = normalize(a_alt, S_alt)
        
        obj_alt = eval_obj(a_alt)
        print("  Resampled to n=%d: obj=%.8f" % (n_alt, obj_alt))
        update_global(a_alt, obj_alt)
        
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 30:
            a_opt, val_opt = fista_exact(a_alt, S_alt, t6, rng,
                                          k_top_init=1, k_top_max=5,
