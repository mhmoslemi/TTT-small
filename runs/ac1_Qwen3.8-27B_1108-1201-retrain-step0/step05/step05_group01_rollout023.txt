"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE-FISTA with backtracking, block coordinate descent,
L-BFGS-B polishing, and multi-scale refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_adaptive(a_init, max_time, rng, beta_start=5.0, beta_end=2000.0,
                       n_phases=8, use_backtrack=True):
    """
    FISTA with adaptive beta (geometric schedule) and backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    # Geometric beta schedule
    beta_schedule = np.geomspace(beta_start, beta_end, n_phases)
    phase = 0
    phase_iter = 0
    max_phase_iter = max(200, int(max_time / (n_phases * 0.05)))

    while time.monotonic() - start < max_time:
        # Check time to next phase
        if phase < n_phases - 1 and phase_iter >= max_phase_iter:
            phase += 1
            phase_iter = 0
            t_mom = 1.0  # Reset momentum at phase change

        beta = beta_schedule[min(phase, n_phases - 1)]
        it += 1

def run_lbfgs_polish(a_init, max_time, rng, beta_levels=(100, 300, 1000)):
    """L-BFGS-B polishing with LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for beta in beta_levels:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum() + 1e-300) / beta
            sum_x = x.sum()
            obj += 100.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 200.0 * (sum_x - S)
            return obj, grad

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 50, 'maxls': 20, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_block_cd(a_init, max_time, rng, block_size=64, n_sweeps=10):
    """
    Block coordinate descent: optimize blocks of consecutive elements.
    For each block, use L-BFGS-B to minimize LSE of convolution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    beta = 50.0

    n_blocks = n // block_size
    remaining_elems = n - n_blocks * block_size

    for sweep in range(n_sweeps):
        if time.monotonic() - start > max_time:
            break

        # Shuffle block order
        order = rng.permutation(n_blocks)

        for bi in order:
            if time.monotonic() - start > max_time:
                break

            lo = bi * block_size
            hi = lo + block_size

            # Fix all elements except the block
            a_fixed = a.copy()
            block_sum = a[lo:hi].sum()

            def block_obj_grad(x):
                x = np.maximum(x, 0.0)
                # Reconstruct full sequence
                full = a_fixed.copy()
                full[lo:hi] = x
                b = fft_convolve_self(full)
                cm = b.max()
                shifted = beta * (b - cm)
                shifted = shifted - shifted.max()
                e = np.exp(shifted)
                w = e / e.sum()
                obj = cm + np.log(e.sum() + 1e-300) / beta
                # Penalty for sum constraint on the block
                obj += 10.0 * (x.sum() - block_sum) ** 2
                # Gradient
                full_grad = conv_gradient(full, w)
                grad = full_grad[lo:hi].copy()
                grad += 20.0 * (x.sum() - block_sum)
                return obj, grad

            result = scipy_minimize(
                block_obj_grad, a[lo:hi].copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * block_size,
                options={'maxiter': 20, 'maxls': 10, 'ftol': 1e-14, 'gtol': 1e-8}
            )

            a_new_block = np.clip(result.x, 0, 1000)
            a[lo:hi] = a_new_block

        # Renormalize and check improvement
        a = a * (S / a.sum())
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses: adaptive LSE-FISTA with Lipschitz estimation, exact subgradient
with expanding active set, block coordinate descent via L-BFGS-B,
and multi-scale gradient polishing.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_lipschitz(a_init, max_time, rng, beta_start=5.0, beta_end=5000.0,
                        n_phases=10):
    """
    FISTA with adaptive Lipschitz estimation and geometric beta schedule.
    Uses backtracking when the Lipschitz estimate is violated.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    beta_schedule = np.geomspace(beta_start, beta_end, n_phases)
    phase = 0
    phase_iter = 0
    max_phase_iter = max(300, int(max_time / (n_phases * 0.04)))
    L_est = beta_start  # Lipschitz estimate

    while time.monotonic() - start < max_time:
        if phase < n_phases - 1 and phase_iter >= max_phase_iter:
            phase += 1
            phase_iter = 0
            t_mom = 1.0
            L_est = beta_schedule[phase]

        beta = beta_schedule[min(phase, n_phases - 1)]
        it += 1
        phase_iter += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size from Lipschitz estimate
        step = 1.0 / max(L_est, 1.0)

        # Backtracking
        x_cand = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_cand = fft_convolve_self(x_cand)
            # Check descent in LSE objective
            cm = c.max()
            shifted_old = beta * (c - cm)
            shifted_old = shifted_old - shifted_old.max()
            lse_old = cm + np.log(np.sum(np.exp(shifted_old)) + 1e-300) / beta

            cm_cand = c_cand.max()
            shifted_new = beta * (c_cand - cm_cand)
            shifted_new = shifted_new - shifted_new.max()
            lse_new = cm_cand + np.log(np.sum(np.exp(shifted_new)) + 1e-300) / beta

            # Descent condition: f(x_cand) <= f(y) + <grad, x_cand - y> + (L/2)||x_cand - y||^2
            diff = x_cand - y
            rhs = lse_old + np.dot(grad, diff) + 0.5 * L_est * np.dot(diff, diff)
            if lse_new <= rhs + 1e-12:
                break
            step *= 0.5
            x_cand = project_simplex(y - step * grad, S)
            L_est /= 0.5  # Increase L estimate

        # FISTA update
        x_new = x_cand
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0

        # Adaptive L update (increase if no backtracking needed recently)
        if it % 100 == 0:
            L_est = min(L_est * 1.05, 50000.0)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_subgrad_momentum(a_init, max_time, rng):
    """
    FISTA with exact subgradient and expanding active set.
    Uses subgradient averaging for stability.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    grad_avg = np.zeros(n)
    alpha_avg = 0.7  # EMA coefficient for gradient averaging

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

        # Expanding active set: start with top-1, grow to top-K
        frac = min(1.0, it / 5000.0)
        if it < 100:
            tol_frac = 0.0
        elif it < 1000:
            tol_frac = 0.002 * frac
        elif it < 5000:
            tol_frac = 0.005 + 0.01 * frac
        else:
            tol_frac = 0.05

        grad_exact = exact_subgradient(y, c, tol_frac=tol_frac)
        gnorm = np.linalg.norm(grad_exact)
        if gnorm < 1e-15:
            break

        # Gradient averaging
        grad_avg = alpha_avg * grad_avg + (1 - alpha_avg) * grad_exact
        gnorm_avg = np.linalg.norm(grad_avg)
        if gnorm_avg < 1e-15:
            grad_avg = grad_exact
            gnorm_avg = gnorm

        # Step size: Polyak-like with diminishing factor
        step = S / (gnorm_avg * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / (0.1 * gnorm_avg + 1e-10))
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad_avg, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 4000 == 0:
            t_mom = 1.0
            grad_avg = np.zeros(n)

        # Periodic noise injection to escape stagnation
        if it > 3000 and it % 5000 == 0:
            noise_scale = 0.003 * S / np.sqrt(n)
            noise = rng.normal(0, noise_scale, n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_block_cd(a_init, max_time, rng, block_size=128):
    """
    Block coordinate descent: optimize blocks of consecutive elements
    using L-BFGS-B on the LSE objective.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    beta = 80.0

    n_blocks = n // block_size
    if n_blocks < 1:
        return a, best_obj

    sweep = 0
    while time.monotonic() - start < max_time:
        sweep += 1
        order = rng.permutation(n_blocks)

        # Increase beta over sweeps
        beta = 80.0 * (1.0 + 0.5 * sweep)
        beta = min(beta, 2000.0)

        for bi in order:
            if time.monotonic() - start > max_time:
                break

            lo = bi * block_size
            hi = lo + block_size
            block_sum = a[lo:hi].sum()

            if block_sum < 1e-15:
                continue

            def block_obj_grad(x):
                x = np.maximum(x, 0.0)
                full = a.copy()
                full[lo:hi] = x
                b = fft_convolve_self(full)
                cm = b.max()
                shifted = beta * (b - cm)
                shifted = shifted - shifted.max()
                e = np.exp(shifted)
                w = e / e.sum()
                obj = cm + np.log(e.sum() + 1e-300) / beta
                sum_x = x.sum()
                obj += 50.0 * (sum_x - block_sum) ** 2
                full_grad = conv_gradient(full, w)
                grad = full_grad[lo:hi].copy()
                grad += 100.0 * (sum_x - block_sum)
                return obj, grad

            try:
                result = scipy_minimize(
                    block_obj_grad, a[lo:hi].copy(), method='L-BFGS-B',
                    jac=True, bounds=[(0, 1000)] * block_size,
                    options={'maxiter': 30, 'maxls': 15, 'ftol': 1e-14, 'gtol': 1e-9}
                )
                a[lo:hi] = np.clip(result.x, 0, 1000)
            except Exception:
                pass

        # Renormalize after each sweep
        a = a * (S / max(a.sum(), 1e-15))
        obj = compute_obj(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj

def run_random_restart_local(a_init, max_time, rng, n_restarts=5):

        a1b, _ = run_fista_lipschitz(global_best_a, rem1, rng,
                                      beta_start=500.0, beta_end=10000.0, n_phases=4)
        obj1b = compute_obj(a1b)
        print("  FISTA-LSE (high beta): obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Hybrid gradient (LSE + exact subgradient) (20%) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: Hybrid gradient (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

    # === Phase 3: Block coordinate descent (15%) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.15
        print("\n=== Phase 3: Block CD (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

    # === Phase 4: Subgradient momentum (15%) ===
    remaining = deadline - time.monotonic()

        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Subgradient momentum (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, _ = run_subgrad_momentum(global_best_a, p4_budget, rng)

        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Random restarts with local opt (10%) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Random restarts (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS-B polish (10%) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: Final L-BFGS-B (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng,
                                     beta_levels=(300, 1000, 3000))
        print("  Final polish: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: Ultra-fine subgradient (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Ultra-fine subgradient (budget %.0fs) ===" % remaining)
        t7 = time.monotonic()
        a7, _ = run_subgrad_momentum(global_best_a, remaining * 0.8, rng)
        obj7 = compute_obj(a7)
        print("  Ultra-fine: obj=%.8f (t=%.0fs)" % (obj7, time.monotonic() - t7))
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]

    print("\nBest objective found: %.8f" % global_best_obj)
