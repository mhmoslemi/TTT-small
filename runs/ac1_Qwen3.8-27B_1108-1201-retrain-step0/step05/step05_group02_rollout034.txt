```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses active-lag subgradient descent with LSE smoothing,
adaptive step sizes, and multi-phase optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def active_gradient(a, c, K=10):
    """Compute subgradient from top-K lags."""
    n = len(a)
    K = min(K, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:K]
```

**Another idea: "Deflation" approach**

The peak of the convolution is at some lag k*. If we could make c[k*] equal to c[k*+1] = c[k*-1] = ... (all equal to the average 1/(2n-1)), that would be ideal.

One approach: iteratively "flatten" the convolution. At each step:
1. Find the peak lag k*.
2. Compute the "excess" at k*: excess = c[k*] - avg_c where avg_c = 1/(2n-1) (approximately).
3. Reduce the contribution to c[k*] by adjusting a.

The gradient tells us which positions to adjust. Positions m where a[k*-m] is large are the "offenders." We should reduce a at those positions (or at their "partners" k*-m).

This is essentially what the subgradient method does, but we can make it more aggressive:
- Instead of a small step, find the optimal amount to transfer.
- Use a "water-filling" approach: determine how much mass to move from each "offender" position to "deficit" positions.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: LSE-FISTA initialization, active-lag subgradient descent,
and direct peak reduction with adaptive step sizes.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smoothed_obj_and_grad(a, beta):
    """LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        w = e / tot
        lse_val = cm + np.log(tot) / beta
    # Gradient of LSE w.r.t. a: convolve w with a (and its reverse)
    # d(LSE)/da[m] = sum_k w[k] * 2*a[k-m] (with bounds)
    # This is like a "blurred" version of a weighted by w
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def active_gradient(a, c, K):
    """Compute subgradient from top-K lags."""
    n = len(a)
    K = min(K, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for kk in top_idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            g[lo:hi+1] += a[kk-hi:kk-lo+1][::-1]
    g *= 2.0 / K
    return g


def run_active_subgrad(a_init, max_time, rng, K_init=5, K_max=200,
                       K_incr_period=500, step_init=0.1):
    """
    Active-lag subgradient descent with adaptive K and step size.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    K = K_init
    step = step_init * S / np.linalg.norm(active_gradient(a, c, K) + 1e-30)
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Adaptive K: increase as we get closer to convergence
        stuck = it - last_improve
        K = min(K_max, K_init + stuck // K_incr_period)
        
        g = active_gradient(a, c, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Step size: decrease over time
        step = step_init * S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, 5.0 * S / gnorm)
        
        a_new = project_simplex(a - step * g, S)
        
        # Check if step was good
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)
        
        if obj_new <= cur_obj + 1e-12:
            a = a_new
            c = c_new
        else:
            # Backtrack
            step *= 0.5
            a_new = project_simplex(a - step * g, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new <= cur_obj + 1e-12:
                a = a_new
                c = c_new
            else:
                # Accept anyway (to avoid getting completely stuck)
                a = a_new
                c = c_new
        
        # Periodic noise to escape local minima
        if stuck > 500 and it % 100 == 0:
            noise_scale = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 3000))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.01,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / beta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        lse_val, grad, c_y, w = lse_smoothed_obj_and_grad(y, beta)
        
        # Evaluate at x (not y) for best tracking
        c_x = fft_convolve_self(a)
        cur_obj = 2.0 * n * c_x.max() / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = step_scale / beta
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj


def run_peak_reduction(a_init, max_time, rng):
    """
    Direct peak reduction: identify the peak lag and find the optimal
    mass redistribution to reduce it.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 50000:
            break
        
        # Find top lags
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # For each top lag, compute the "contribution profile"
        # and find the best pair to swap
        best_improvement = 0.0
        best_j, best_l, best_delta = -1, -1, 0.0
        
        for kk in top_idx[:3]:
            kk = int(kk)
            # Contribution to c[kk] from position j: a[j] * a[kk-j]
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi < lo:
                continue
            contrib = a[kk-hi:kk-lo+1][::-1]  # a[kk-j] for j in [lo, hi]
            
            # We want to reduce c[kk]. Moving mass from j to j' changes c[kk] by:
            # -delta * a[kk-j] + delta * a[kk-j']
            # (approximately, ignoring second-order effects)
            # So we want j with large a[kk-j] (high contrib) and j' with small a[kk-j'].
            
            # For the full convolution effect, we need to consider ALL lags.
            # But as a first approximation, focus on the peak lag.
            
            # Find source: large contribution to c[kk]
            src_score = contrib * np.maximum(a[lo:hi+1], 1e-10)
            # Find target: small contribution to c[kk], but we also need
            # to make sure it doesn't increase other c[k] values too much.
            tgt_score = -contrib * (S / n - a[lo:hi+1] + 1e-10)
            
            top_src = np.argsort(src_score)[::-1][:5]
            top_tgt = np.argsort(tgt_score)[:5]
            
            for si in top_src[:3]:
                for ti in top_tgt[:3]:
                    j = lo + int(si)
                    l = lo + int(ti)
                    if j == l or a[j] < 1e-10:
                        continue
                    delta = a[j] * 0.03
                    if delta < 1e-12:
                        continue
                    
                    # Estimate new c for affected lags
                    # Change in a: -delta at j, +delta at l
                    # Change in c[k] = -delta * (contribution of j to c[k]) + delta * (contribution of l to c[k])
                    # Contribution of position p to c[k] is a[k-p] (if valid)
                    
                    # For the peak lag kk:
                    new_ckk = c[kk] - delta * a[kk-j] - delta * a[j-kk] + delta * a[kk-l] + delta * a[l-kk]
                    # (The factor of 2 appears because both a[j] and a[kk-j] change)
                    # Actually: c[kk] = sum_i a[i]*a[kk-i]
                    # After change: a'[j] = a[j]-delta, a'[l] = a[l]+delta
                    # The change is: -delta*a[kk-j] - a[j]*delta... 
                    
                    # This is getting complicated. Let me just compute the new c directly
                    # for a few trial swaps and pick the best.
                    
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[l] += delta
                    c_trial = fft_convolve_self(a_trial)
                    obj_trial = 2.0 * n * c_trial.max() / (S * S)
                    
                    if obj_trial < cur_obj - best_improvement:
                        best_improvement = cur_obj - obj_trial
                        best_j, best_l, best_delta = j, l, delta
                        best_c_trial = c_trial.copy()
                        best_a_trial = a_trial.copy()
        
        if best_j >= 0:
            a = best_a_trial
            c = best_c_trial
            # Adjust delta slightly to ensure sum is preserved (it should be)
        else:
            # No improvement found via swap; try a small random perturbation
            if stuck > 1000:
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 != idx2 and a[idx1] > 1e-10:
                    delta = a[idx1] * 0.01
                    a[idx1] -= delta
                    a[idx2] += delta
                    c = fft_convolve_self(a)
        
        # Periodic full recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def make_init(n, method, rng):
    """Create initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "ramp":
        a = np.linspace(0.1, 1.0, n)
    elif method == "asym_cos":
        a = 0.3 + 0.7 * np.cos(2 * np.pi * 0.7 * tt + 0.5)
        a = np.abs(a) + 0.05
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
