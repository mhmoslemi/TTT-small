```
def project_simplex(v, S):
    v = np.asarray(v, dtype=float)
    if v.size == 0: return v
    # if S == 0
    # We want x = max(v - theta, 0), sum S
```

```python
"""
Multi-stage search for non-negative step heights minimizing
    2*n*max(self-convolution) / (sum)^2.

Main ideas:
  * Work in the scale-invariant normalization sum(a)=1.
  * Use FFT-based self-convolutions for fast objective and gradients.
  * Use several complementary optimizers:
      - spectral alternating projection (project the convolution to a lower
        box, take a phase-aligned square root, and project back to the
        non-negative length-n simplex),
      - Nesterov-style projected gradient descent on a smooth LSE maximization,
      - exact/subgradient top-lag polishing with cheap shift gradients.
  * Initialize from the supplied previous construction when available, plus
    diverse asymmetric windows.
  * Return the best sequence found before the deadline.
"""
import numpy as np
import time

try:
    from scipy.fft import rfft, irfft, next_fast_len
except Exception:
    from numpy.fft import rfft, irfft

    def next_fast_len(n):
        m = 1
        while int(m) < int(n):
            m <<= 1
        return m


def normalize_positive(a, s=1.0):
    """Clip to nonnegative, clip magnitude, and rescale to sum s."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.minimum(np.maximum(a, 0.0), 1000.0)
    ss = float(a.sum())
    if ss <= 1e-300:
        return np.full(a.shape, float(s) / float(a.size))
    return a * (float(s) / ss)


def resize_sequence(a, m):
    """Resize a sequence to length m by linear endpoint-free interpolation."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 0:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    if n == 1:
        return np.full(m, float(a[0]) / float(m), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(x_new, x_old, a).astype(np.float64)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x)=s} in Euclidean distance."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    s = float(s)
    if s <= 0.0:
        return np.zeros(n, dtype=np.float64)

    w = np.sort(v)[::-1]
    cum = np.cumsum(w, dtype=np.float64)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cum - s) / idx
    valid = theta <= w
    if np.any(valid):
        valid_idx = np.nonzero(valid)[0]
        rho = int(valid_idx.max()) + 1
    else:
        rho = n
    theta = float((cum[rho - 1] - s) / rho)
    out = np.maximum(v - theta, 0.0)
    ss = float(out.sum())
    if ss > 0.0 and abs(ss - s) > 1e-12 * max(1.0, abs(s)):
        out *= s / ss
    return out


def compute_objective_general(a):
    """Exact objective for arbitrary nonnegative length-n sequence."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    fsize = next_fast_len(size)
    F = rfft(a, fsize)
    q = irfft(F * F, fsize)
    q = q[:size]
    if q.size:
        q = np.maximum(q, 0.0)
        return float(2.0 * n * float(q.max()) / (s * s))
    return np.inf


def max_conv_unit(a, fsize, size):
    """max coefficient of full self-convolution, assuming sum(a)=1."""
    F = rfft(a, fsize)
    q = irfft(F * F, fsize)
    q = q[:size]
    if q.size:
        q = np.maximum(q, 0.0)
        return float(q.max())
    return 0.0


def conv_and_specs(a, fsize, size):
    """
    Return (q_normalized_sum1, rfft(a)).
    q is the full self-convolution normalized to sum 1, with negative roundoff removed.
    """
    F = rfft(a, fsize)
    q = irfft(F * F, fsize)
    q = q[:size].astype(np.float64, copy=False)
    q[np.maximum(q, 0.0) < 0.0] = 0.0
    q = np.maximum(q, 0.0)
    ss = float(q.sum())
    if ss <= 1e-300:
        q = np.full(q.size, 1.0 / float(q.size), dtype=np.float64)
    else:
        q *= 1.0 / ss
    return q, F


def lse_and_weights(c, beta):
    """Stable LSE value and softmax weights for beta*c."""
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    v = beta * (c - m)
    if v.size:
        v = v - float(v.max())
        e = np.exp(v)
        ss = float(e.sum())
        if ss <= 1e-300:
            w = np.full(c.size, 1.0 / c.size, dtype=np.float64)
            return m, w
        w = e / ss
        return float(m + float(np.log(ss)) / beta), w
    return 0.0, np.array([], dtype=np.float64)


def active_topk_gradient(a, q, k, fsize=None, size=None):
    """
    Gradient for a weighted sum of the top k convolution lags.
    For small k this uses direct shifts; otherwise FFT.
    """
    a = np.asarray(a, dtype=np.float64)
    q = np.asarray(q, dtype=np.float64)
    n = int(a.size)
    k = int(max(1, min(k, q.size)))

    idx = np.argsort(q)[::-1][:k]
    idx = idx.astype(np.int64, copy=False)
    vals = q[idx].astype(np.float64, copy=False)

    if k == 1:
        weights = np.ones(1, dtype=np.float64)
    elif k <= 4:
        weights = np.full(k, 1.0 / float(k), dtype=np.float64)
    else:
        ss = float(vals.sum())
        if ss <= 1e-300:
            weights = np.full(k, 1.0 / float(k), dtype=np.float64)
        else:
            weights = vals / ss

    if k <= 32:
        g = np.zeros(n, dtype=np.float64)
        for ki, wk in zip(idx, weights):
            ki = int(ki)
            if wk == 0.0:
                continue
            jmin = max(0, ki - n + 1)
            jmax = min(n - 1, ki)
            if jmin > jmax:
                continue
            start = ki - jmax
            length = jmax - jmin + 1
            # Need a[k-j] for j=jmin..jmax; this is a reversed slice.
            g[jmin:jmax + 1] += 2.0 * wk * a[start:start + length][::-1]
        return g

    if size is None:
        size = a.size * 2 - 1
    if fsize is None:
        fsize = next_fast_len(size)

    w_full = np.zeros(size, dtype=np.float64)
    w_full[idx] = weights
    W = rfft(w_full, fsize)
    F = rfft(a, fsize)
    return 2.0 * irfft(np.conj(F) * W, fsize)[:n].astype(np.float64, copy=False)


def project_box_sum(q, tmax, total=1.0):
    """
    Project q onto {0 <= x <= tmax, sum(x)=total} in L2.
    q is assumed nonnegative and approximately sum 1.
    """
    m = q.size
    if m == 0:
        return q
    q = np.maximum(np.asarray(q, dtype=np.float64), 0.0)
    s = float(q.sum())
    total = float(total)
    if s <= 1e-300:
        if tmax * m >= total:
            return np.full(m, total / float(m), dtype=np.float64)
        return np.full(m, total / float(m), dtype=np.float64)

    q = q * (total / s)

    if tmax * m < total:
        return np.full(m, total / float(m), dtype=np.float64)
    if tmax <= q.max() + 1e-15:
        pass

    q_sorted = np.sort(q)
    q_cum = np.cumsum(q_sorted, dtype=np.float64)

    lo = 0.0
    hi = max(1e-15, float(tmax - q_sorted[0]))

    # monotone: sum clamp(q + lam, 0, tmax)
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        u = tmax - mid
        pos = int(np.searchsorted(q_sorted, u, side="right"))
        if pos >= m:
            val = float(q_cum[-1] + m * mid)
        else:
            val = float((m - pos) * tmax + q_cum[pos - 1] + pos * mid)
        if val < total:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    p = np.minimum(q + lam, tmax)
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 0.0:
        p *= total / ss
        if p.size and float(p.max()) > tmax:
            p = np.minimum(p, tmax)
            ss = float(p.sum())
            if ss > 0.0:
                p *= total / ss
    return p


def spectral_sqrt_align(Q_half, F_half, fsize):
    """
    Principal square root of Q_half, with sign at each frequency chosen
    to remain close to F_half.  The result is valid for irfft.
    """
    r = np.sqrt(Q_half.astype(np.complex128, copy=False))

    dot = np.real(F_half * np.conj(r))
    r = np.where(dot < 0.0, -r, r)

    if r.size > 0:
        r[0] = abs(np.real(r[0]))

    if int(fsize) % 2 == 0:
        nyq = int(fsize) // 2
        if nyq < r.size:
            val = float(np.real(r[nyq]))
            if float(np.real(F_half[nyq])) < 0.0 and val > 0.0:
                val = -val
            r[nyq] = val

    return r


def make_step_pair(n, v1, v2):
    """Two-block nonnegative window."""
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = v1
    a[half:] = v2
    return a


def make_exp_sequence(n, lam):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.exp(lam * t)


def make_two_cos_sequence(n):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * t)


def make_pi_cos_sequence(n):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(np.pi * t)


def make_sin2_sequence(n):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    return (np.sin(np.pi * t) ** 2).astype(np.float64) + 1e-8


def make_square_sequence(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.sign(np.sin(2.0 * np.pi * t / max(1, n))).astype(np.float64)


def run_alp(a_init, budget_s, rng):
    """
    Spectral alternating projection:
      a -> q=a*a -> box-project q -> frequency square root -> time projection.
    """
    a = normalize_positive(a_init, 1.0)
    budget_s = float(budget_s)
    n = int(a.size)
    if n == 0 or budget_s <= 0.2:
        return a, compute_objective_general(a)

    size = 2 * n - 1
    fsize = next_fast_len(size)

    q, F = conv_and_specs(a, fsize, size)
    best_obj = float(2.0 * n * q.max())
    best_a = a.copy()
    cur_obj = best_obj

    start = time.monotonic()
    deadline = start + max(1.0, budget_s - 0.2)

    delta = 0.010
    alpha = 0.50
    stall = 0
    last_best_it = 0
    it = 0
    tmin = 1.0 / float(size)

    while time.monotonic() < deadline:
        it += 1
        q, F = conv_and_specs(a, fsize, size)
        cur_obj = float(2.0 * n * q.max())
        cur_max = float(q.max())

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            last_best_it = it
        else:
            stall += 1

        if stall > 80:
            delta = max(0.0015, delta * 0.82)
            alpha = max(0.08, alpha * 0.85)
        else:
            delta = min(0.045, delta * 1.05)
            alpha = min(0.85, alpha * 1.04)

        use_delta = delta
        if stall > 180 and (it % 25 == 0):
            use_delta = max(use_delta, 0.035)

        ttarget = max(tmin, cur_max * (1.0 - use_delta))
        p = project_box_sum(q, ttarget, 1.0)

        Q = rfft(p, fsize)
        Fnew = spectral_sqrt_align(Q, F, fsize)
        fnew = irfft(Fnew, fsize)[:n].astype(np.float64, copy=False)

        alphas = [alpha, min(1.0, 1.7 * alpha), max(0.05, 0.45 * alpha)]
        chosen = None
        chosen_obj = cur_obj - 1e-16

        for al in alphas:
            if al <= 0.0:
                continue
            mix = (1.0 - al) * a + al * fnew
            cand = project_simplex(mix, 1.0)
            mc = max_conv_unit(cand, fsize, size)
            obj = float(2.0 * n * mc)
            if obj < chosen_obj:
                chosen_obj = obj
                chosen = cand

        if chosen is not None and chosen_obj < cur_obj - 1e-13:
            a = chosen
            if chosen_obj < best_obj - 1e-13:
                best_obj = chosen_obj
                best_a = a.copy()
                stall = 0
                last_best_it = it
            delta = min(0.06, delta * 1.08)
            alpha = min(0.9, alpha * 1.05)
        else:
            delta = max(0.001, delta * 0.90)
            alpha = max(0.05, alpha * 0.90)

            if stall > 350 and rng is not None:
                noise_scale = max(1e-6, 1e-4 / np.sqrt(float(n)))
                a = project_simplex(a + rng.normal(0.0, noise_scale, n), 1.0)

        if it % 1000 == 0:
            print("  ALP iter %d: obj=%.7f best=%.7f T=%.3e" % (it, cur_obj, best_obj, ttarget))

    q, _ = conv_and_specs(best_a, fsize, size)
    obj = float(2.0 * n * q.max())
    return best_a, min(best_obj, obj)


def run_lse_stage(a_init, budget_s, beta_start, beta_end):
    """
    Nesterov accelerated projected gradient descent for a smooth LSE
    approximation to the maximum convolution coefficient.
    """
    a = normalize_positive(a_init, 1.0)
    budget_s = float(budget_s)
    n = int(a.size)
    if n == 0 or budget_s <= 0.2:
        return a, compute_objective_general(a)

    size = 2 * n - 1
    fsize = next_fast_len(size)

    y = a.copy()
    x = a.copy()
    t = 1.0

    q, F = conv_and_specs(y, fsize, size)
    best_obj = float(2.0 * n * q.max())
    best_a = y.copy()

    beta = float(beta_start)
    L = 4.0 * beta + 1.0
    step = 0.8 / L
    last_lse = None
    start = time.monotonic()
    deadline = start + max(1.0, budget_s - 0.2)
    last_best_it = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1.0, budget_s))
        beta = float(beta_start) + (float(beta_end) - float(beta_start)) * frac
        if beta < 1e-9:
            beta = 1e-9

        q, F = conv_and_specs(y, fsize, size)
        cur_obj = float(2.0 * n * q.max())
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_best_it = it

        lse, w = lse_and_weights(q, beta)
        W = rfft(w, fsize)
        grad = 2.0 * irfft(np.conj(F) * W, fsize)[:n].astype(np.float64, copy=False)

        L = 4.0 * beta + 1.0
        bad = False
        if last_lse is not None and lse > last_lse + 1e-12:
            step *= 0.5
            bad = True
        else:
            step = min(1.8 / L, step * 1.05)

        step = max(1e-12, step)
        x = project_simplex(y - step * grad, 1.0)

        if bad:
            t = 1.0
            y = x.copy()
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_new = x + ((t - 1.0) / t_new) * (x - y).astype(np.float64, copy=False)
            ss = float(y_new.sum())
            if ss <= 1e-300:
                y = x.copy()
            elif abs(ss - 1.0) > 1e-7:
                y = project_simplex(y_new, 1.0)
            else:
                y = y_new
            t = float(t_new)

        if it % 25 == 0:
            mc = max_conv_unit(x, fsize, size)
            xo = float(2.0 * n * mc)
            if xo < best_obj - 1e-13:
                best_obj = xo
                best_a = x.copy()
                last_best_it = it

        if it - last_best_it > 5000:
            t = 1.0
            y = x.copy()
            step *= 0.7

        last_lse = lse

    for cand in (y, x, best_a):
        cand = normalize_positive(cand, 1.0)
        obj = float(2.0 * n * max_conv_unit(cand, fsize, size))
        if obj < best_obj:
            best_obj = obj
            best_a = cand.copy()

    return best_a, best_obj


def run_active_stage(a_init, budget_s, rng):
    """
    Nonsmooth polishing using top-lag subgradients.  For small active sets the
    gradient is computed by explicit shifts, which is very cheap.
    """
    a = normalize_positive(a_init, 1.0)
    budget_s = float(budget_s)
    n = int(a.size)
    if n == 0 or budget_s <= 0.2:
        return a, compute_objective_general(a)

    size = 2 * n - 1
    fsize = next_fast_len(size)

    q, _ = conv_and_specs(a, fsize, size)
    best_obj = float(2.0 * n * q.max())
    best_a = a.copy()

    start = time.monotonic()
    deadline = start + max(1.0, budget_s - 0.2)

    k = 1
    global_scale = 1.0
    stall = 0
    last_best_it = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        q, _ = conv_and_specs(a, fsize, size)
        cur_obj = float(2.0 * n * q.max())

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            last_best_it = it
            global_scale = min(1.8, global_scale * 1.08)
            if k > 1 and it % 100 == 0:
                k = max(1, int(k * 0.7))
        else:
            stall += 1
            if stall > 25:
                k = min(32, k + 1)
            if stall > 120:
                k = min(64, k * 2)
            if stall > 80:
                global_scale = max(0.03, global_scale * 0.85)

        grad = active_topk_gradient(a, q, k, fsize=fsize, size=size)
        ma = float(np.max(np.abs(grad)))
        if ma <= 1e-300:
            break
        base = 1.0 / (float(n) * ma)
        base = min(3.0, max(1e-4, base))

        factors = [global_scale * base, 0.55 * global_scale * base, 0.25 * global_scale * base]
        chosen = None
        chosen_obj = cur_obj - 1e-16

        for fac in factors:
            if fac <= 0.0:
                continue
            cand = project_simplex(a - fac * grad, 1.0)
            obj = float(2.0 * n * max_conv_unit(cand, fsize, size))
            if obj < chosen_obj:
                chosen_obj = obj
                chosen = cand

        if chosen is not None and chosen_obj < cur_obj - 1e-13:
            a = chosen
            if chosen_obj < best_obj - 1e-13:
                best_obj = chosen_obj
                best_a = a.copy()
                stall = 0
                last_best_it = it
            k = max(1, int(k * 0.8))
        else:
            global_scale = max(0.02, global_scale * 0.72)

            if stall > 400 and rng is not None:
                noise_scale = max(1e-7, 5e-5 / np.sqrt(float(n)))
                a = project_simplex(a + rng.normal(0.0, noise_scale, n), 1.0)

        if it % 200 == 0:
            print("  Active iter %d: obj=%.7f best=%.7f k=%d" % (it, cur_obj, best_obj, k))

    q, _ = conv_and_specs(best_a, fsize, size)
    obj = float(2.0 * n * q.max())
    return best_a, min(best_obj, obj)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a good nonnegative sequence under the 2400 s budget and return it.
    """
    rng = np.random.default_rng(seed)
    start_all = time.monotonic()
    budget_s = float(max(30.0, budget_s))
    deadline = start_all + max(30.0, budget_s - 35.0)

    global_best_arr = None
    global_best_obj = np.inf

    # Optional warm start.
    prev = None
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        prev = None

    if prev is not None and prev.size > 0:
        prev = np.minimum(np.maximum(prev, 0.0), 1000.0)
        if float(prev.sum()) > 0.01:
            pob = compute_objective_general(prev)
            if np.isfinite(pob):
                global_best_arr = prev.copy()
                global_best_obj = float(pob)
                print("warm-start objective = %.7f (n=%d)" % (global_best_obj, prev.size))

    # Choose the working length. 32768 still permits a single final direct
    # convolution by the evaluator while giving more surface than 8192.
    if budget_s >= 1500:
        work_n = 32768
    elif budget_s >= 700:
        work_n = 16384
    else:
        work_n = 8192

    work_n = int(work_n)
    size = 2 * work_n - 1
    fsize = next_fast_len(size)

    inits = []
    labels = []

    if prev is not None and prev.size > 0:
        inits.append(normalize_positive(resize_sequence(prev, work_n), 1.0))
        labels.append("prev")

    inits.append(normalize_positive(make_step_pair(work_n, 4.0 / 3.0, 2.0 / 3.0), 1.0))
    labels.append("stepHL")
    inits.append(normalize_positive(make_step_pair(work_n, 2.0 / 3.0, 4.0 / 3.0), 1.0))
    labels.append("stepLH")
    inits.append(normalize_positive(make_exp_sequence(work_n, -4.0), 1.0))
    labels.append("exp-4")
    inits.append(normalize_positive(make_exp_sequence(work_n, -1.5), 1.0))
    labels.append("exp-1.5")
    inits.append(normalize_positive(make_exp_sequence(work_n, 2.0), 1.0))
    labels.append("exp+2")
    inits.append(normalize_positive(make_two_cos_sequence(work_n), 1.0))
    labels.append("cos2")
    inits.append(normalize_positive(make_pi_cos_sequence(work_n), 1.0))
    labels.append("cospi")
    inits.append(normalize_positive(make_sin2_sequence(work_n), 1.0))
    labels.append("sin2")
    inits.append(normalize_positive(make_square_sequence(work_n), 1.0))
    labels.append("square")
    inits.append(normalize_positive(np.exp(rng.normal(0.0, 0.8, work_n)), 1.0))
    labels.append("lognormal")
    inits.append(normalize_positive(rng.random(work_n) ** 2 + 1e-12, 1.0))
    labels.append("uniform2")

    work_arr = None
    work_obj = np.inf
    second_work_arr = None
    second_work_obj = np.inf

    print("initializations at n=%d, fsize=%d" % (work_n, fsize))
    for name, a in zip(labels, inits):
        obj = float(2.0 * work_n * max_conv_unit(a, fsize, size))
        if obj < work_obj:
            if work_arr is not None:
                second_work_arr = work_arr.copy()
                second_work_obj = work_obj
            work_arr = a.copy()
            work_obj = obj
        else:
            if obj < second_work_obj:
                second_work_arr = a.copy()
                second_work_obj = obj
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_arr = a.copy()
        print("  init %-10s obj=%.7f" % (name, obj))

    if work_arr is None:
        work_arr = normalize_positive(np.ones(work_n), 1.0)
        work_obj = compute_objective_general(work_arr)

    def maybe_update_global(arr, obj):
        nonlocal_global_best_arr = global_best_arr
        nonlocal_global_best_obj = global_best_obj
        if obj < nonlocal_global_best_obj:
            global_best_obj = obj
            global_best_arr = np.asarray(arr, dtype=np.float64).copy()

    # Python closure-free update helper is not possible without lambdas, but
    # nested function still uses closure; replace by inline logic below.
    # The previous helper would not be used.

    def update_global(arr, obj):
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_arr = np.asarray(arr, dtype=np.float64).copy()

    def update_work(arr, obj):
        nonlocal_work_arr = work_arr
        nonlocal_work_obj = work_obj
        if obj < nonlocal_work_obj:
            work_obj = obj
            work_arr = np.asarray(arr, dtype=np.float64).copy()

    # Phase A: spectral alternating projection, a different global shaper.
    rem = deadline - time.monotonic()
    if rem > 220:
        alp_budget = min(420.0, rem * 0.16)
        print("ALP budget=%.0fs" % alp_budget)
        a_alp, obj_alp = run_alp(work_arr, alp_budget, rng)
        if obj_alp < work_obj:
            work_arr = np.asarray(a_alp, dtype=np.float64).copy()
            work_obj = float(obj_alp)
        if obj_alp < global_best_obj:
            global_best_obj = float(obj_alp)
            global_best_arr = np.asarray(a_alp, dtype=np.float64).copy()
        print("after ALP: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Phase B: smooth LSE FISTA, low beta.
    rem = deadline - time.monotonic()
    if rem > 220:
        b1 = max(20.0, rem * 0.22)
        print("LSE low budget=%.0fs" % b1)
        a1, o1 = run_lse_stage(work_arr, b1, 1.0, 12.0)
        if o1 < work_obj:
            work_arr = np.asarray(a1, dtype=np.float64).copy()
            work_obj = float(o1)
        if o1 < global_best_obj:
            global_best_obj = float(o1)
            global_best_arr = np.asarray(a1, dtype=np.float64).copy()
        print("after LSE low: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Phase C: smooth LSE FISTA, medium beta.
    rem = deadline - time.monotonic()
    if rem > 180:
        b2 = max(20.0, rem * 0.34)
        print("LSE mid budget=%.0fs" % b2)
        a2, o2 = run_lse_stage(work_arr, b2, 10.0, 220.0)
        if o2 < work_obj:
            work_arr = np.asarray(a2, dtype=np.float64).copy()
            work_obj = float(o2)
        if o2 < global_best_obj:
            global_best_obj = float(o2)
            global_best_arr = np.asarray(a2, dtype=np.float64).copy()
        print("after LSE mid: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # If the working sequence is still markedly worse than the warm start and
    # a good second initialization exists, spend some time on that basin.
    rem = deadline - time.monotonic()
    if rem > 400 and second_work_obj is not None and second_work_obj < work_obj - 1e-5:
        try_budget = min(280.0, rem * 0.30)
        print("try second basin budget=%.0fs" % try_budget)
        a_try, o_try = run_lse_stage(second_work_arr, try_budget, 10.0, 300.0)
        if o_try < work_obj:
            work_arr = np.asarray(a_try, dtype=np.float64).copy()
            work_obj = float(o_try)
        if o_try < global_best_obj:
            global_best_obj = float(o_try)
            global_best_arr = np.asarray(a_try, dtype=np.float64).copy()
        print("after second basin: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Phase D: fast exact top-lag polishing.
    rem = deadline - time.monotonic()
    if rem > 180:
        bd = max(20.0, rem * 0.28)
        print("active budget=%.0fs" % bd)
        a_act, o_act = run_active_stage(work_arr, bd, rng)
        if o_act < work_obj:
            work_arr = np.asarray(a_act, dtype=np.float64).copy()
            work_obj = float(o_act)
        if o_act < global_best_obj:
            global_best_obj = float(o_act)
            global_best_arr = np.asarray(a_act, dtype=np.float64).copy()
        print("after active: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Phase E: high-smoothness LSE finalization.
    rem = deadline - time.monotonic()
    if rem > 90:
        be = max(15.0, rem * 0.45)
        print("LSE high budget=%.0fs" % be)
        a_fin, o_fin = run_lse_stage(work_arr, be, 180.0, 4000.0)
        if o_fin < work_obj:
            work_arr = np.asarray(a_fin, dtype=np.float64).copy()
            work_obj = float(o_fin)
        if o_fin < global_best_obj:
            global_best_obj = float(o_fin)
            global_best_arr = np.asarray(a_fin, dtype=np.float64).copy()
        print("after LSE high: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Phase F: one last exact-active polish if enough time remains.
    rem = deadline - time.monotonic()
    if rem > 70:
        bf = max(10.0, rem * 0.75)
        print("final active budget=%.0fs" % bf)
        a_last, o_last = run_active_stage(work_arr, bf, rng)
        if o_last < work_obj:
            work_arr = np.asarray(a_last, dtype=np.float64).copy()
            work_obj = float(o_last)
        if o_last < global_best_obj:
            global_best_obj = float(o_last)
            global_best_arr = np.asarray(a_last, dtype=np.float64).copy()
        print("after final active: work_obj=%.7f global_obj=%.7f" % (work_obj, global_best_obj))

    # Final sanity select: if the working length sequence is not the global
    # best and global best exists, return global best.  Otherwise return work.
    result = global_best_arr
    final_obj = global_best_obj

    if result is None or result.size == 0:
        result = work_arr if work_arr is not None else normalize_positive(np.ones(8192), 1.0)
        final_obj = compute_objective_general(result)

    # Optional: if the global best is from the smaller warm-start and the
    # optimized larger sequence is only slightly worse, still return the one
    # with lower exact FFT objective.
    if work_arr is not None and work_obj < final_obj + 0.01:
        # Prefer the larger optimized array if it is competitive; this can
        # improve finite-size effects on the evaluator while staying safe.
        if np.isfinite(work_obj) and (work_obj - 1e-7 < final_obj) and work_arr.size >= 8192:
            result = work_arr
            final_obj = work_obj

    result = np.asarray(result, dtype=np.float64).reshape(-1)
    result = np.minimum(np.maximum(result, 0.0), 1000.0)
    rs = float(result.sum())
    if rs < 0.01:
        result = np.ones(8192, dtype=np.float64)
        result /= result.size

    elapsed = time.monotonic() - start_all
    print("final returned n=%d objective_estimate=%.7f elapsed=%.1fs" % (result.size, compute_objective_general(result), elapsed))

    out = []
    for val in result:
        x = float(val)
        if x < 0.0:
            x = 0.0
        elif x > 1000.0:
            x = 1000.0
        out.append(x)
    return out
```
