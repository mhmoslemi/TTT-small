"""

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def conv_peak(a, N):
    """Compute convolution and return (conv, peak_value, peak_index)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    idx = int(np.argmax(c))
    return c, c[idx], idx

def lse_gradient(a, N, beta, top_k=0):
    """
    Compute gradient of either LSE-smoothed max or top-k sum of convolution.
    Returns (gradient, lse_value_or_toppk_value).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    
    if top_k > 0:
        # Top-K weighted gradient
        k = min(top_k, len(c))
        threshold = np.partition(c, len(c) - k)[len(c) - k]
        mask = c >= threshold
        if mask.sum() > k:
            # Tie-breaking: keep only k
            idx = np.where(mask)[0]
            vals = c[idx]
            order = np.argsort(vals)[::-1][:k]
            mask = np.zeros(len(c), dtype=bool)
            mask[idx[order]] = True
        weights = np.where(mask, 1.0 / mask.sum(), 0.0)
        obj_val = c[mask].sum()
    else:
        # LSE
        if beta < 1e-6:
            weights = np.ones(len(c)) / len(c)
            obj_val = c_max
        else:
            exp_bc = np.exp(beta * (c - c_max))
            total = exp_bc.sum()
            if total < 1e-300:
                weights = np.zeros(len(c))
                weights[int(np.argmax(c))] = 1.0
                obj_val = c_max
            else:
                weights = exp_bc / total
                obj_val = c_max + np.log(total) / beta
    
    # Gradient via FFT
    fsize = 1 << (3 * n - 2).bit_length()
    w_full = np.zeros(fsize)
    w_full[:len(weights)] = weights
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, obj_val, c, c_max

def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def optimize_block(a_init, n, S, N, max_time, rng, 
                   beta_init=5.0, beta_max=500.0, m_lbfgs=12,
                   use_toppk=False, top_k=20):
    """
    Main optimization block using L-BFGS with LSE or top-K gradient.
    """
    a = project_simplex(a_init, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, f_val, c, c_max = lse_gradient(a, N, beta, top_k if use_toppk else 0)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    start_time = time.monotonic()
    it = 0
    patience = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive beta
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.05)
            
        # Compute direction
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search with exact objective
        alpha = 2.0 * S / (n * max(beta, 1.0))
        a_new = a
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls, cm_ls, _ = conv_peak(candidate, N)
            if cm_ls < c_max or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls < c_max:
                    break
            alpha *= 0.5
        
        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Update L-BFGS history
        g_new, f_new, c_new, cmax_new2 = lse_gradient(a_new, N, beta, top_k if use_toppk else 0)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_new2
        
        # Periodic small perturbation
        if it % 500 == 0 and patience > 100:
            perturbation_size = 0.01 * S / n
            a = project_simplex(a + perturbation_size * rng.standard_normal(n), S)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_gradient(a, N, beta, top_k if use_toppk else 0)
            patience = 0
        
        # Truncate history periodically
        if it % 100 == 0 and len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
    
    return best_a, best_cmax

def nesterov_toppk(a_init, n, S, N, max_time, rng, top_k_init=10, top_k_max=200):
    """Nesterov-accelerated top-K subgradient method."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    best_topk_val = np.inf
    
    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c, cur, _ = conv_peak(y, N)
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = np.zeros(len(c))
        weights[top_idx] = top_vals / total
        
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        w_full[:len(weights)] = weights
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 10 == 0:
            c_x, cm_x, _ = conv_peak(x_new, N)
            if cm_x < best_cmax:
                best_cmax = cm_x
                best_a = x_new.copy()
    
    return best_a, best_cmax

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses hybrid LSE/top-K optimization with adaptive strategies.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))
    
    # Start from previous best
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = project_simplex(a_candidate, S)
        c, sc, _ = conv_peak(a_cand, N)
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            print("  [%s] peak=%.8f score=%.8f" % (label, sc, score))
        return sc
    
    # Initialize from previous best or random
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_init = np.clip(given, 0, 1000).astype(np.float64)
            a_init = project_simplex(a_init, S)
            print("Using previous best as initial point")
        else:
            raise ValueError
    except Exception:
        a_init = np.ones(n) * S / n
        print("Using uniform initial point")
    
    update_best(a_init, "init")
    
    # --- Phase 1: LSE optimization (beta 5-50) ---

    t0 = time.monotonic()

                                    beta_init=5.0 + 5*i, beta_max=50.0 + 50*i, m_lbfgs=12)

        a_curr = global_a.copy() if global_a is not None else a_curr
    
    # --- Phase 2: Top-K optimization ---
    print("=== Phase 2: Top-K optimization ===")
    t1 = time.monotonic()
    phase2_time = min(300.0, (deadline - t1) * 0.12)
