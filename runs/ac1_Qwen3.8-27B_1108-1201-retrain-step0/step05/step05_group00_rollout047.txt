    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]

    """Build the constraint matrix for the LP.

"""
Minimize 2*n*max_conv/sum^2 using truncated LP-based direction finding combined 
with LSE gradient descent and spectral restarts.

Key innovations:
1. LP-based direction: Solve max sum(g) s.t. (a*g)_k <= M for top-K lags, g>=0.
   This gives a principled descent direction that respects peak constraints.
2. Adaptive K: Start with small K (few constraints) and increase as optimization progresses.
3. Hybrid stepping: Alternate between LP directions and LSE gradient directions.
4. Peak equalization: At convergence, focus on equalizing top peaks rather than just the max.
5. Multiple sequence lengths explored (8192 primary, 4096 as alternative).
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def next_pow2(m):
    return 1 << (m - 1).bit_length()


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if len(a) != n:
        if len(a) > n:
            a = a[:n]
        else:
            a = np.concatenate([a, np.full(n - len(a), S / (n * 10))])
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def conv_peak(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, c.max()


def compute_convolution(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= v_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, N, beta):
    n = len(a)
    c = compute_convolution(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        scaled = beta * (c - c_max)
        max_scaled = scaled.max()
        exp_sc = np.exp(scaled - max_scaled)
        total = exp_sc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_sc / total
            lse_val = c_max + (max_scaled + np.log(total)) / beta
    fsize = next_pow2(3 * n - 1)
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def multi_beta_lse_gradient(a, N, betas):
    n = len(a)
    total_grad = np.zeros(n)
    total_weight = 0.0
    for i, beta in enumerate(betas):
        g, fv, c, cm = lse_gradient(a, N, beta)
        w = 1.0 / (1.0 + i * 0.5)
        total_grad += w * g
        total_weight += w
    total_grad /= total_weight
    c = compute_convolution(a, N)
    return total_grad, c, c.max()


def lbfgs_two_loop(s_hist, y_hist, g):
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(y_i, s_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac=1.0):
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    if frac >= 1.0:
        phase = rng.uniform(0, 2 * np.pi, N)
    else:
        old_phase = np.angle(A)
        phase = old_phase.copy()
        n_rand = max(1, int(N * frac))
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def solve_lp_direction(a, M, top_k_indices, n, epsilon=0.001):
    """Solve max sum(g) s.t. A@g <= M*(1+eps), g >= 0.
    A[k,j] = a[k-j] if 0 <= k-j < n.
    Returns optimal g or None if failed.
    """
    M_bound = M * (1.0 + epsilon)
    K = len(top_k_indices)
    if K == 0:
        return None

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_k_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return None

    A_mat = coo_matrix((vals, (rows, cols)), shape=(K, n))
    A_csr = A_mat.tocsr()

    obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(K, M_bound)

    result = linprog(
        c=obj,
        A_ub=A_csr,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'max_time': 5.0, 'presolve': True}
    )

    if result.status == 0 and result.x is not None:
        g = np.maximum(result.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_step(a, N, K=50, epsilon=0.001, max_time=8.0):
    """One LP-based descent step. Returns (new_a, improved)."""
    n = len(a)
    S = a.sum()
    c, M = conv_peak(a, N)
    top_k = np.argsort(c)[::-1][:K]
    g = solve_lp_direction(a, M, top_k, n, epsilon)
    if g is None or g.sum() < 1e-10:
        return a, False
    g = g * (S / g.sum())
    # Line search
    best_t = 0.0
    best_peak = M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        candidate = (1 - t) * a + t * g
        candidate = np.maximum(candidate, 0.0)
        cs = a.sum()
        ss = candidate.sum()
        if ss < 1e-15:
            continue
        candidate = candidate * (cs / ss)
        _, pk = conv_peak(candidate, N)
        if pk < best_peak:
            best_peak = pk
            best_t = t
    if best_t > 0:
        new_a = (1 - best_t) * a + best_t * g
        new_a = np.maximum(new_a, 0.0)
        ss = new_a.sum()
        if ss > 1e-15:
            new_a = new_a * (S / ss)
        return new_a, True
    return a, False


def deflation_step(a, N, transfer_frac=0.02):
    """Fast O(n) deflation step: move mass from peak contributors to non-contributors."""
    n = len(a)
    S = a.sum()
    c, M = conv_peak(a, N)
    k_star = int(np.argmax(c))
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contrib = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
    if contrib.max() < 1e-15:
        return a, False
    i_max = int(np.argmax(contrib))
    i_min = int(np.argmin(contrib))
    if i_max == i_min:
        return a, False
    transfer = min(transfer_frac * a[i_max], 0.3 * a[i_max])
    if transfer < 1e-15:
        return a, False
    a = a.copy()
    a[i_max] -= transfer
    a[i_min] += transfer
    return a, True

def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=30, top_k_max=500):
    """Nesterov accelerated top-K subgradient."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        c, A = conv_peak(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = next_pow2(3 * n - 1)
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 50 == 0:
            _, pk = conv_peak(a, N)
            if pk < best_cmax:
                best_cmax = pk
                best_a = a.copy()

        if it % 1000 == 0 and rng.random() < 0.2:
            y = spectral_restart(y, N, rng, frac=0.2)
            a = y.copy()
            t = 1.0

    return best_a, best_cmax

            a_init = make_init_rand(n, rng)
    except Exception:
        a_init = make_init_rand(n, rng)

    a_init = safe_normalize(a_init, n, S)
    update_best(a_init, "init")

    # --- Phase 1: LP-based exploration with multiple K values ---
    print("=== Phase 1: LP-based exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(600.0, (deadline - t1) * 0.30)

    a_curr = global_a.copy() if global_a is not None else a_init.copy()
    stall = 0
    K_values = [20, 40, 60, 80, 120]
    K_idx = 0

    while time.monotonic() < t1 + phase1_budget:
        rem = t1 + phase1_budget - time.monotonic()
        if rem < 5:
            break

        # LP step
        t_lp = time.monotonic()
        a_new, improved = lp_step(a_curr, N, K=K_values[K_idx], epsilon=0.001,
                                  max_time=min(6.0, rem * 0.3))
        lp_time = time.monotonic() - t_lp

        if improved:
            _, pk_new = conv_peak(a_new, N)
            if pk_new < global_best:
                global_best = pk_new
                global_a = a_new.copy()
                a_curr = a_new.copy()
                stall = 0
                score = 2.0 * n * pk_new / (S * S)
                print("  [LP K=%d] score=%.7f (%.1fs)" % (K_values[K_idx], score, lp_time))
            else:
                a_curr = a_new.copy()
                stall += 1
        else:
            stall += 1

        # Every few iterations, do an LSE gradient step
        if it % 3 == 0 or stall >= 2:
            rem2 = t1 + phase1_budget - time.monotonic()
            if rem2 > 15:
                a_lse, pk_lse = lbfgs_lse_block(a_curr, n, S, N, min(20.0, rem2 * 0.4),
                                                 rng, beta_init=10.0, beta_max=100.0,
                                                 m_lbfgs=8, use_multi_beta=False)
                if pk_lse < global_best:
                    global_best = pk_lse
                    global_a = a_lse.copy()
                    update_best(a_lse, "P1/LSE")
                a_curr = a_lse.copy()
                stall = 0

        # Adapt K
        if stall >= 3:
            K_idx = (K_idx + 1) % len(K_values)
            stall = 0
        if stall >= 5:
            a_curr = spectral_restart(a_curr, N, rng, frac=0.3)
            stall = 0

    print("Phase 1 done. Best score: %.7f" % (2.0 * n * global_best / (S * S)))

    # --- Phase 2: Focused L-BFGS with higher beta ---
    print("=== Phase 2: Focused L-BFGS ===")
    t2 = time.monotonic()
    phase2_budget = min(700.0, (deadline - t2) * 0.35)

    a_curr = global_a.copy() if global_a is not None else a_init.copy()
    stall = 0

    while time.monotonic() < t2 + phase2_budget:
        rem = t2 + phase2_budget - time.monotonic()
        if rem < 15:
            break
        block_time = min(120.0, rem)
        beta_start = 10.0 + 5.0 * min(stall, 8)
        beta_end = min(500.0, 50.0 * (1.0 + 0.3 * stall))
        use_mb = (stall % 2 == 0)

        a_new, pk_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         m_lbfgs=12, use_multi_beta=use_mb)
        if pk_new < global_best - 1e-9:
            global_best = pk_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2.0 * n * pk_new / (S * S)
            print("  [P2] score=%.7f (beta %.0f-%.0f, mb=%s)" % (score, beta_start, beta_end, use_mb))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                a_curr = spectral_restart(a_curr, N, rng, frac=0.4)
                stall = 0
            elif stall >= 4:
                a_curr = spectral_restart(a_curr, N, rng, frac=1.0)
                stall = 0

    print("Phase 2 done. Best score: %.7f" % (2.0 * n * global_best / (S * S)))

    # --- Phase 3: LP + LSE hybrid with peak equalization ---
    print("=== Phase 3: LP + LSE hybrid ===")
    t3 = time.monotonic()
    phase3_budget = min(500.0, (deadline - t3) * 0.25)

    a_curr = global_a.copy() if global_a is not None else a_init.copy()
    stall = 0

    while time.monotonic() < t3 + phase3_budget:
        rem = t3 + phase3_budget - time.monotonic()
        if rem < 10:
            break

        # Alternate: LP step then LSE step
        a_new, improved = lp_step(a_curr, N, K=60, epsilon=0.0005,
                                  max_time=min(5.0, rem * 0.2))
        if improved:
            _, pk_new = conv_peak(a_new, N)
            if pk_new < global_best:
                global_best = pk_new
                global_a = a_new.copy()
                a_curr = a_new.copy()
                stall = 0
                print("  [P3/LP] score=%.7f" % (2.0 * n * pk_new / (S * S)))

        # LSE gradient step
        rem2 = t3 + phase3_budget - time.monotonic()
        if rem2 > 20:
            a_lse, pk_lse = lbfgs_lse_block(a_curr, n, S, N, min(40.0, rem2 * 0.3),
                                             rng, beta_init=30.0, beta_max=400.0,
                                             m_lbfgs=10, use_multi_beta=True)
            if pk_lse < global_best:
                global_best = pk_lse
                global_a = a_lse.copy()
                a_curr = a_lse.copy()
                stall = 0
                print("  [P3/LSE] score=%.7f" % (2.0 * n * pk_lse / (S * S)))
            else:
                a_curr = a_lse.copy()
                stall += 1

        if stall >= 3:
            a_curr = spectral_restart(a_curr, N, rng, frac=0.25)
            stall = 0

    print("Phase 3 done. Best score: %.7f" % (2.0 * n * global_best / (S * S)))

    # --- Phase 4: Nesterov polishing ---
    print("=== Phase 4: Nesterov polishing ===")
    t4 = time.monotonic()
    remaining4 = min(300.0, (deadline - t4) * 0.20)
    if remaining4 > 20:
        a_ng, pk_ng = nesterov_toppk_block(global_a, n, S, N, remaining4, rng,
                                            top_k_init=30, top_k_max=400)
        update_best(a_ng, "nesterov")

    # --- Phase 5: Final high-beta L-BFGS ---
    print("=== Phase 5: Final L-BFGS ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 10
    if remaining5 > 20:
        a_fin, pk_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining5, 300.0),
                                         rng, beta_init=50.0, beta_max=800.0,
                                         m_lbfgs=15, use_multi_beta=False)
        update_best(a_fin, "final")

    # --- Phase 6: Short LP polishing on best ---
    rem6 = deadline - time.monotonic() - 5
    if rem6 > 10 and global_a is not None:
        a_curr = global_a.copy()
        for _ in range(min(20, int(rem6 // 5))):
            if time.monotonic() > deadline - 3:
                break
            a_new, improved = lp_step(a_curr, N, K=30, epsilon=0.0003, max_time=4.0)
            if improved:
                _, pk_new = conv_peak(a_new, N)
                if pk_new < global_best:
                    global_best = pk_new
                    global_a = a_new.copy()
                    a_curr = a_new.copy()
                    print("  [P6/LP] score=%.7f" % (2.0 * n * pk_new / (S * S)))
                else:
                    a_curr = a_new.copy()
            else:
                break

    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.7f" % ver)
        else:
            print("Verification: inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, peak=%.7f, approx score=%.7f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
