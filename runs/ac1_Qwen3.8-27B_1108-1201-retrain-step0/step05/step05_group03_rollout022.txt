def fft_convolve_self(a):

def project_simplex(v, S):

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def binomial_init(n, S):
    """Create binomial coefficient sequence normalized to sum S."""
    j = np.arange(n, dtype=np.float64)
    # C(2j, j) / 4^j computed log-stably
    from math import lgamma
    log_c = np.array([lgamma(2*int(ji)+1) - 2*lgamma(int(ji)+1) - 4*int(ji)*np.log(2) for ji in j], dtype=np.float64)
    a = np.exp(log_c)
    return normalize(a, S)


def lse_value_and_grad(a, beta, k):
    """Compute LSE value and gradient of top-k LSE of self-convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad, c, idx, w


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def run_gradient_opt(a_init, S, max_time, rng, 
                     beta_init=20.0, beta_rate=0.005, beta_max=10000.0,
                     k_init_frac=0.1, k_min_frac=0.01,
                     lr_init=1.0, lr_decay=0.999):
    """Gradient descent with adaptive LSE, BB step, and backtracking."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    k = max(8, int(k_init_frac * (2 * n - 1)))
    it = 0
    lr = lr_init
    prev_a = a.copy()
    prev_grad = None
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        # Shrink k over time
        progress = min(1.0, it / 50000.0)
        k = max(max(4, int(k_min_frac * (2*n-1))), 
                int((k_init_frac - (k_init_frac - k_min_frac) * progress) * (2*n-1)))
        
        val, grad, c, idx, w = lse_value_and_grad(a, beta, k)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            if it % 10000 == 0:
                print("    it=%d, max=%.8f, obj=%.6f" % (it, cur_max, 2*n*cur_max/S**2))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if s_dot_s > 1e-20 and abs(s_dot_y) > 1e-20:
                bb_step = s_dot_s / abs(s_dot_y)
                lr = np.clip(builtins_max(lr * 0.7 + 0.3 * bb_step, 1e-6), 1e3)
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        
        # Gradient step with backtracking
        step = lr / max(beta * 0.5, 1.0)
        a_new = project_simplex(a - step * grad, S)
        
        for _ in range(6):
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
        
        a = a_new
        lr *= lr_decay
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_fista(a_init, S, max_time, rng,
              beta_init=10.0, beta_rate=0.004, beta_max=5000.0,
              k_init_frac=0.08, k_min_frac=0.005):
    """FISTA with top-k LSE and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        progress = min(1.0, it / 50000.0)
        k = max(4, int((k_init_frac - (k_init_frac - k_min_frac) * progress) * (2*n-1)))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        k_use = min(k, len(c))
        idx = np.argsort(c)[::-1][:k_use]
        c_top = c[idx]
        cm = c_top.max()
        scaled = np.clip(beta * (c_top - cm), -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / max(tot, 1e-300)
        w_full = np.zeros(len(c))
        w_full[idx] = w
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new
        
        # Restart if stuck
        if stuck > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_subgradient_polish(a_init, S, max_time, rng, 
                           beta_start=1000.0, beta_end=50000.0):
    """Polishing with very high beta (near-exact max) and subgradient averaging."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 10)))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Use top-k with very small k (effectively subgradient at max)
        k = max(2, int(0.005 * (2*n-1)))
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
        cm = c_top.max()
        scaled = np.clip(beta * (c_top - cm), -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / max(tot, 1e-300)
        
        w_full = np.zeros(len(c))
        w_full[idx] = w
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Tiny steps
        step = 0.1 / (1.0 + 0.01 * it)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() <= cur_max:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def builtins_max(a, b):
    return a if a > b else b

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
1. Binomial coefficient initialization a[j]=C(2j,j)/4^j - starts near the theoretical optimum
   (gives flat autocorrelation for first n lags, objective ~ pi/2)
2. Adaptive top-k LSE with shrinking k for convergence
3. Barzilai-Borwein adaptive step sizing
4. Multi-phase: FISTA -> gradient with BB -> subgradient polish
5. Spectral domain perturbation for escaping local minima
6. Multiple n values and restarts

import numpy as np
import time
from math import lgamma
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def binomial_init(n, S):
    """Create binomial coefficient sequence C(2j,j)/4^j normalized to sum S."""
    j = np.arange(n, dtype=np.int64)
    log_c = np.array([lgamma(2 * int(ji) + 1) - 2 * lgamma(int(ji) + 1) - 4 * int(ji) * np.log(2.0)
                      for ji in j], dtype=np.float64)
    a = np.exp(log_c)
    return normalize(a, S)


def compute_lse_grad(a, c, k, beta):
    """Compute top-k LSE gradient of self-convolution values c."""
    n = len(a)
    k_use = min(k, len(c))
    idx = np.argsort(c)[::-1][:k_use]
    c_top = c[idx]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)

def lse_value(c, k, beta):
    """Compute top-k LSE value."""
    k_use = min(k, len(c))
    idx = np.argsort(c)[::-1][:k_use]
    c_top = c[idx]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    return cm + np.log(max(tot, 1e-300)) / beta


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng,
              beta_init=10.0, beta_rate=0.004, beta_max=5000.0,
              k_init_frac=0.08, k_min_frac=0.005, label=""):
    """FISTA with top-k LSE and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        progress = min(1.0, it / 30000.0)
        k = max(4, int((k_init_frac + (k_min_frac - k_init_frac) * progress) * (2 * n - 1)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
            if it % 5000 == 0:
                print("    FISTA%s it=%d max=%.8f obj=%.6f" % (label, it, cur_max, 2.0 * n * cur_max / (S * S)))
        else:
            stuck += 1

        grad, idx, w, _ = compute_lse_grad(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        if stuck > 1500:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        if it % 3000 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.05, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_gradient_bb(a_init, S, max_time, rng,
                    beta_init=30.0, beta_rate=0.003, beta_max=10000.0,
                    k_init_frac=0.06, k_min_frac=0.003,
                    lr_init=1.0, label=""):
    """Gradient descent with BB step and backtracking."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    lr = lr_init
    prev_a = a.copy()
    prev_grad = None
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        progress = min(1.0, it / 30000.0)
        k = max(4, int((k_init_frac + (k_min_frac - k_init_frac) * progress) * (2 * n - 1)))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stuck = 0
            if it % 5000 == 0:
                print("    GradBB%s it=%d max=%.8f obj=%.6f" % (label, it, cur_max, 2.0 * n * cur_max / (S * S)))
        else:
            stuck += 1

        grad, idx, w, _ = compute_lse_grad(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if s_dot_s > 1e-20 and abs(s_dot_y) > 1e-20:
                bb_step = s_dot_s / abs(s_dot_y)
                lr = max(1e-6, min(1e4, lr * 0.6 + 0.4 * bb_step))

        prev_a = a.copy()
        prev_grad = grad.copy()

        step = lr / max(beta * 0.5, 1.0)
        a_new = project_simplex(a - step * grad, S)
        improved = False
        for _ in range(6):
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + 1e-10:
                improved = c_new.max() < cur_max
                break
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
        a = a_new
        lr *= 0.9995

        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            prev_a = a.copy()
            prev_grad = None
            stuck = 0

        if it % 4000 == 0 and rng.random() < 0.1:
            a = spectral_perturb(a, 0.04, rng)
            prev_a = a.copy()
            prev_grad = None
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_polish(a_init, S, max_time, rng,
               beta_start=500.0, beta_end=100000.0, label=""):
    """High-beta subgradient polishing with tiny steps."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 5)))
        beta = beta_start * (beta_end / beta_start) ** progress
        k = max(2, int(0.005 * (2 * n - 1)))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            step_scale = min(3.0, step_scale * 1.01)
            if it % 3000 == 0:
                print("    Polish%s it=%d max=%.10f obj=%.8f" % (label, it, cur_max, 2.0 * n * cur_max / (S * S)))

        grad, idx, w, _ = compute_lse_grad(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.2 * step_scale / (1.0 + 0.005 * it)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() <= cur_max + 1e-14:
            a = a_new
            if c_new.max() < cur_max:
                step_scale = min(3.0, step_scale * 1.005)
            else:
                step_scale = max(0.1, step_scale * 0.7)
        else:
            step_scale = max(0.05, step_scale * 0.4)
            step2 = 0.05 * step_scale
            a_new = project_simplex(a - step2 * grad, S)
            a = a_new

        if it % 5000 == 0 and progress < 0.85:
            noise_scale = 0.0003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

                global_a = a_fin3.copy()
                print("  Final GradBB: obj=%.6f" % global_best)

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback verification: %.6f" % ver)
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("FINAL: Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Time used: %.1fs" % (time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
