    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng,
              use_multi=False, lp_interval=0, lp_K=20):
    """FISTA with adaptive LSE, optional multi-scale, and LP-guided updates."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0

    base_k = max(8, int(0.06 * (2 * n - 1)))
    beta_cur = 10.0

    while time.monotonic() - start < max_time:
        it += 1
        # Cycling beta: oscillate between low and high
        beta_phase = 0.5 * (1.0 + np.sin(0.005 * it))
        beta = 10.0 + beta_phase * (500.0 + 0.05 * it)
        beta = min(beta, 10000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Compute gradient
        if use_multi:
            spec_list = [
                (max(4, base_k // 4), beta * 3.0, 0.3),
                (base_k, beta, 0.5),
                (max(3, base_k // 8), beta * 5.0, 0.2),
            ]
            _, grad = multi_scale_grad(y, c, spec_list)
        else:
            _, grad, _ = lse_grad_val(y, c, base_k, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Lipschitz estimate for step size
        L = np.linalg.norm(grad) / max(np.linalg.norm(y), 1e-10) + 1.0
        step = 1.0 / max(L, 2.0 * beta * 0.1, 1e-10)

        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking (limited)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_mom = t_new

        # LP-guided update periodically
        if lp_interval > 0 and it % lp_interval == 0 and it > lp_interval:
            c_cur = fft_convolve_self(y)
            M_cur = c_cur.max()
            g_dir = lp_direction(y, M_cur, lp_K, 1.5 * S, rng)
            if g_dir is not None and g_dir.sum() > 0:
                g_norm = g_dir * (S / g_dir.sum())
                # Blend: try small step toward LP direction
                for t_lp in [0.1, 0.05, 0.02]:
                    y-test = (1.0 - t_lp) * y + t_lp * g_norm
                    c_test = fft_convolve_self(y_test)
                    if c_test.max() < M_cur * (1.0 + 0.01):
                        y = normalize(y_test, S)
                        t_mom = 1.0
                        break

        # Stuck detection: restart
        if stuck > 800:
            progress = (time.monotonic() - start) / max_time
            if progress < 0.8:
                # Spectral perturbation
                y = spectral_perturb(y, 0.05 + 0.1 * rng.random(), rng)
                t_mom = 1.0
                stuck = 0
            else:
                # Small noise
                noise_scale = 0.002 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck = 0

        # Periodic spectral flattening
        if it % 3000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            y = spectral_flatten(y, 0.03, rng)
            t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_polish(a_init, S, max_time, rng):
    """Very high-beta polishing with careful line search."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0

    k = max(3, int(0.02 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 100.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        _, grad, _ = lse_grad_val(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = (0.3 / (1.0 + 0.002 * it)) * step_scale
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.005, step_scale * 0.4)
            # Try even smaller step
            a_new2 = project_simplex(a - step * 0.1 * grad, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2

        # Occasional tiny perturbation
        if it % 3000 == 0 and (time.monotonic() - start) / max_time < 0.5:
            noise = rng.normal(0, 1e-5 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_lp_heavy(a_init, S, max_time, rng):
    """LP-guided optimization: repeatedly solve LP and blend."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0

    K = min(30, 2 * n - 1)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        # Solve LP
        g = lp_direction(a, cur_max, K, 1.2 * S, rng)
        if g is None or g.sum() < 1e-10:
            # Fallback to gradient step
            _, grad, _ = lse_grad_val(a, c, max(5, int(0.05 * (2*n-1))), 50.0)
            a_new = project_simplex(a - 0.5 * grad / max(np.linalg.norm(grad), 1e-10), S)
            a = a_new
            continue

        g_norm = g * (S / g.sum())
        
        # Line search on blend factor
        best_t = 0.0
        best_val = cur_max
        for t_try in [0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = normalize((1.0 - t_try) * a + t_try * g_norm, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_t = t_try
        
        if best_t > 0:
            a = normalize((1.0 - best_t) * a + best_t * g_norm, S)
        else:
            # Small FISTA step as fallback
            _, grad, _ = lse_grad_val(a, c, max(5, int(0.05 * (2*n-1))), 30.0)
            step = 1.0 / max(np.linalg.norm(grad) / max(np.linalg.norm(a), 1e-10) + 1.0, 10.0)
            a = project_simplex(a - step * grad, S)

        # Periodic perturbation
        if it % 50 == 0 and (time.monotonic() - start) / max_time < 0.7:
            y = spectral_perturb(a, 0.03, rng)
            c_y = fft_convolve_self(y)
            if c_y.max() < best * 1.05:
                a = y

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def propose_candidate():
    """
    Search for non-negative sequence minimizing 2n*max(convolve(a,a))/(sum(a))^2.
    Multi-phase: FISTA multi-scale -> LP-guided -> Polish -> Multi-length.
    """
    rng = np.random.default_rng(12345)
    start_total = time.monotonic()
    deadline = start_total + 2370.0  # 30s buffer

    print("Starting optimization, budget=%.0fs" % (deadline - start_total))

    # Get initial sequence from previous best
    a_init = None
    try:
        a_init = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_init = np.clip(a_init, 0, 1000)
        if a_init.size >= 64 and a_init.sum() > 0:
            print("Using parent sequence of length %d, sum=%.2f" % (len(a_init), a_init.sum()))
        else:
            a_init = None
    except Exception as e:
        print("No parent sequence available: %s" % str(e))

    global_best_obj = np.inf
    global_best_a = None

    def update_global(a, n):
        nonlocal global_best_obj, global_best_a
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        obj = 2.0 * n * fft_convolve_self(a).max() / (s * s)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))

    # === Phase 1: Multi-start exploration (300s) ===
    print("\n=== Phase 1: Multi-start exploration ===")
    t_start = time.monotonic()
    phase1_budget = min(300.0, (deadline - t_start) * 0.15)

    S_main = np.sqrt(2.0 * 8192)
    
    # Candidate initializations
    inits = []
    if a_init is not None:
        inits.append(("parent", normalize(resample_sequence(a_init, 8192), S_main)))
    inits.append(("uniform", np.ones(8192) * (S_main / 8192)))
    
    # Random spectral init
    for seed_offset in range(2):
        rng_sub = np.random.default_rng(seed_offset + 999)
        a_rand = np.abs(rng_sub.standard_normal(8192)) + 0.1
        # Apply random frequency modulation
        fsize = 16384
        A = fft(a_rand, fsize)
        A *= np.exp(1j * rng_sub.normal(0, 0.3, fsize))
        a_rand = np.real(ifft(A, fsize))[:8192]
        a_rand = np.clip(a_rand, 0, None)
        inits.append(("spec_rand_%d" % seed_offset, normalize(a_rand, S_main)))

    phase1_per = phase1_budget / len(inits)
    for name, a_start in inits:
        if time.monotonic() > deadline - 60:
            break
        t_budget = min(phase1_per, deadline - time.monotonic() - 30)
        if t_budget < 10:
            break
        print("  Phase1 [%s]: budget=%.0fs" % (name, t_budget))
        a_opt, val_opt = run_fista(a_start, S_main, t_budget, rng, use_multi=True)
        obj = 2.0 * 8192 * val_opt / (a_opt.sum() ** 2)
        print("    Result: obj=%.6f, max_conv=%.6f" % (obj, val_opt))
        update_global(a_opt, 8192)

    # === Phase 2: Main FISTA optimization (60% of remaining) ===
    print("\n=== Phase 2: Main optimization ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    phase2_budget = remaining * 0.45

    if global_best_a is None:
        global_best_a = np.ones(8192) * (S_main / 8192)

    n_main = len(global_best_a)
    S_main = global_best_a.sum()
    
    print("  Starting from n=%d, current best=%.6f" % (n_main, global_best_obj))
    
    # Run FISTA with multi-scale
    t_fista = phase2_budget * 0.6
    a_opt, val_opt = run_fista(global_best_a, S_main, t_fista, rng,
                                use_multi=True, lp_interval=500, lp_K=20)
    update_global(a_opt, n_main)

    # Run LP-heavy
    t_lp = phase2_budget * 0.25
    if time.monotonic() < deadline - 60:
        a_opt, val_opt = run_lp_heavy(global_best_a, S_main, t_lp, rng)
        update_global(a_opt, n_main)

    # Final FISTA pass
    t_fista2 = min(phase2_budget * 0.15, deadline - time.monotonic() - 30)
    if t_fista2 > 10:
        a_opt, val_opt = run_fista(global_best_a, S_main, t_fista2, rng,
                                    use_multi=False, lp_interval=300, lp_K=15)
        update_global(a_opt, n_main)

    # === Phase 3: Intensive polishing (20% of remaining) ===
    print("\n=== Phase 3: Polishing ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    phase3_budget = min(remaining * 0.25, 500.0)

    if global_best_a is not None and phase3_budget > 20:
        n_pol = len(global_best_a)
        S_pol = global_best_a.sum()
        a_opt, val_opt = run_polish(global_best_a, S_pol, phase3_budget, rng)
        update_global(a_opt, n_pol)

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-length ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 60 and global_best_a is not None:
        # Try n=4096
        n_alt = 4096
        S_alt = np.sqrt(2.0 * n_alt)
        a_alt = normalize(resample_sequence(global_best_a, n_alt), S_alt)
        t_alt = min(remaining * 0.4, 300.0)
        if t_alt > 20:
            print("  Trying n=%d, budget=%.0fs" % (n_alt, t_alt))
            a_opt, val_opt = run_fista(a_alt, S_alt, t_alt * 0.6, rng, use_multi=True)
            update_global(a_opt, n_alt)
            if time.monotonic() < deadline - 30:
                a_opt, val_opt = run_polish(a_opt, S_alt, t_alt * 0.3, rng)
                update_global(a_opt, n_alt)

        # Try n=16384
        remaining = deadline - time.monotonic() - 20
        if remaining > 40:
            n_alt2 = 16384
            S_alt2 = np.sqrt(2.0 * n_alt2)
            a_alt2 = normalize(resample_sequence(global_best_a, n_alt2), S_alt2)
            t_alt2 = min(remaining * 0.5, 200.0)
            if t_alt2 > 15:
                print("  Trying n=%d, budget=%.0fs" % (n_alt2, t_alt2))
                a_opt, val_opt = run_fista(a_alt2, S_alt2, t_alt2 * 0.7, rng,
                                            use_multi=False, lp_interval=400, lp_K=15)
                update_global(a_opt, n_alt2)

    # === Phase 5: Final final polish ===
    print("\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic() - 15
    if remaining > 15 and global_best_a is not None:
        n_fin = len(global_best_a)
        S_fin = global_best_a.sum()
        t_fin = remaining * 0.8
        a_opt, val_opt = run_polish(global_best_a, S_fin, t_fin, rng)
        update_global(a_opt, n_fin)

    # === Verification ===
    print("\n=== Verification ===")
    if global_best_a is not None:
        global_best_a = np.clip(global_best_a, 0.0, 1000.0)
        if global_best_a.sum() < 0.01:
            global_best_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("evaluate_sequence result: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_best_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_best_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback result: %.6f" % ver)
    else:
        global_best_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_best_a]

    print("Final sequence length: %d" % len(global_best_a))
    print("Total time: %.1fs" % (time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
