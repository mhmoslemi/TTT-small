"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LSE FISTA with aggressive beta + active-set polish + diversified restart.
Key insight: use provided sequence as warm start, then refine with
progressively sharper LSE approximation and active-set methods.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_and_gradient(y, c, beta):
    """Compute LSE value and gradient."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    logZ = np.log(tot)
    lse_val = cm + logZ / beta
    
    # Gradient via FFT
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, w


def exact_subgradient(c, tol_frac=0.0):
    """Get weights for exact subgradient from active lags."""
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return w_full, active


def run_fista_lse(a_init, max_time, rng, beta_start, beta_end, 
                  max_iter=None, verbose=True):
    """
    FISTA with LSE objective and adaptive beta.
    Progressive beta increase from beta_start to beta_end.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    cur_sum = a.sum()
    if cur_sum < 1e-15:
        a = np.ones(n) * (S / n)
    else:
        a = a * (S / cur_sum)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    if max_iter is None:
        max_iter = 1000000
    
    while time.monotonic() - start < max_time and it < max_iter:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adaptive beta: exponential ramp from beta_start to beta_end
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(beta_start, min(beta, beta_end))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_sum = y.sum()
        obj = 2.0 * n * cur_max / (cur_sum * cur_sum)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Compute LSE gradient
        lse_val, grad, w = lse_and_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 1/L where L is Lipschitz constant of gradient
        # L ~ 2*beta*max(|w|) * 2*max(a) + ... 
        # Conservative: L <= 4 * max(a) * (2*beta + 1)
        L = 4.0 * np.max(a) * (2.0 * beta + 1.0) + 1.0
        step = 1.0 / max(L, 1.0)
        
        # Backtracking line search (simplified)
        # For FISTA, we use the standard step
        step = min(step, 1.0 / max(beta, 1.0) * 0.5)
        step = max(step, 1e-12)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        
        # Ensure y stays on the simplex (re-project if drift)
        if abs(y.sum() - S) > 1e-10:
            y = project_simplex(y, S)
        
        a = x_new
        t_mom = t_new
        
        # Momentum restart conditions
        if stuck > 3000:
            t_mom = 1.0
            # Add small noise
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
        
        if it % 10000 == 0:
            t_mom = 1.0
        
        if verbose and it % 2000 == 0:
            print("  FISTA it=%d beta=%.1f obj=%.6f stuck=%d step=%.2e" % 
                  (it, beta, obj, stuck, step))
    
    return best_a, best_obj

    #   where d is the change in a[p], and we only consider valid indices.

    #   Let me just compute: if a[p] changes by d, then
    #   C_new[k] = sum_i a'_i a'_{k-i} where a'_p = a[p]+d
    #   = C_old[k] + d*a[k-p] + d*a[p] - d^2 (if k-p = p)

def compute_obj_fast(a):
    """Compute objective quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def make_asymmetric_init(n, rng, mode='skew'):
    """Create asymmetric initializations."""
    if mode == 'skew':
        # Skewed distribution (exponential-like)
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-t / (n * 0.3))  # Decays from left
        return a
    elif mode == 'bump':
        # Off-center bump
        center = n * rng.uniform(0.3, 0.7)
        t = np.arange(n, dtype=np.float64)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        return a
    elif mode == 'ramp':
        # Linear ramp
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + t / n
        return a
    elif mode == 'triangle':
        # Asymmetric triangle
        t = np.arange(n, dtype=np.float64) / n
        peak = rng.uniform(0.3, 0.7)
        if t < peak:
            a = t / peak
        else:
            a = (1.0 - t) / (1.0 - peak)
        a = np.maximum(a, 0.01)
        return a
    else:
        return np.ones(n)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements:
1. Subgradient-based polishing with correct Lipschitz constant (=2, not 2*beta).
2. Adaptive active set with hysteresis to avoid oscillation.
3. Adam-like adaptive step sizes for subgradient phase.
4. Focused optimization from provided sequence with multiple refinement stages.
5. Diversified perturbation-based restarts for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj_fast(a):
    """Compute objective value quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient(y, c, tol_frac):
    """Compute subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n], active


def run_lse_fista(a_init, max_time, rng, beta_start, beta_end, verbose=True):
    """
    FISTA with LSE objective. Beta ramps from beta_start to beta_end.
    Step size uses correct Lipschitz bound for LSE: 4*beta*||a||_2^2 + 4*||a||_2^2.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    cur_sum = a.sum()
    if cur_sum < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / cur_sum)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(max_time, 1e-10))

        # Exponential ramp
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(beta_start, min(beta, beta_end))

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_sum = y.sum()
        obj = 2.0 * n * cur_max / (cur_sum * cur_sum)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Lipschitz constant of LSE gradient: 4*beta*||a||_2^2 + 4*||a||_2^2
        a_norm2 = np.dot(a, a)
        L = 4.0 * beta * max(a_norm2, 1.0) + 4.0 * max(a_norm2, 1.0)
        step = 1.0 / max(L, 1.0)
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        if abs(y.sum() - S) > 1e-10:
            y = project_simplex(y, S)
        a = x_new
        t_mom = t_new

        # Restarts
        if stuck > 2000:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
        if it % 8000 == 0:
            t_mom = 1.0

        if verbose and it % 3000 == 0:
            print("    LSE it=%d beta=%.1f obj=%.6f stuck=%d" % (it, beta, obj, stuck))

    return best_a, best_obj


def run_subgrad_polish(a_init, max_time, rng, verbose=True):
    """
    Subgradient FISTA with adaptive active set.
    Key: Lipschitz constant of subgradient is 2 (independent of beta!),
    allowing much larger step sizes than LSE.
    Uses Adam-like adaptive learning rates.
    """

    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    else:

    it = 0
    last_improve = 0

    v_adam = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    adam_t = 0

    active_tol_in = 0.001
    active_tol_out = 0.005
    current_active = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_sum = y.sum()
        obj = 2.0 * n * cur_max / (cur_sum * cur_sum)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve

        # Adaptive active set tolerance
        if stuck < 100:
            tol_in = 0.0
        elif stuck < 500:
            tol_in = 0.001
        elif stuck < 2000:
            tol_in = 0.005
        elif stuck < 10000:
            tol_in = 0.01
        else:
            tol_in = 0.02

        # Compute subgradient

        if gnorm < 1e-16:
            break

        # Step size: for subgradient of max of C[k], Lipschitz = 2
        # Use step = 1/2 with safety factor 0.5
        step = 0.25  # Conservative: 1/L * 0.5 = 0.5 * 0.5 = 0.25
        # Scale by S to handle normalization
        step = min(step, 0.5 * S / max(gnorm, 1e-10))

        # Adam adaptive step
        adam_t += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
        m_hat = m_adam / (1 - beta1 ** adam_t)
        v_hat = v_adam / (1 - beta2 ** adam_t)
        adam_step = 0.01 / (np.sqrt(v_hat) + 1e-8)
        # Blend: use adam_step for direction, but cap overall step

        effective_step = np.maximum(effective_step, 1e-10)

        x_new = project_simplex(y + effective_step * direction, S)

        # Check if this helps; if not, revert to plain subgradient step
        c_check = fft_convolve_self(x_new)

            # Plain subgradient step
            x_new = project_simplex(y - step * grad, S)
            c_check = fft_convolve_self(x_new)
            if c_check.max() > cur_max:

                x_new = project_simplex(y - 0.5 * step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        if abs(y.sum() - S) > 1e-10:
            y = project_simplex(y, S)
        a = x_new

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

                    a[p2] += delta
            y = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t_mom = 1.0
            m_adam[:] = 0
            v_adam[:] = 0

        if it % 10000 == 0:
            t_mom = 1.0

        if verbose and it % 2000 == 0:
            print("    SubGrad it=%d obj=%.6f stuck=%d n_active=%d" % (it, obj, stuck, len(active)))

    return best_a, best_obj


def run_hybrid_optimize(a_init, max_time, rng, verbose=True):
    """
    Two-stage: LSE (smooth) then subgradient (large steps).
    The LSE stage converges to a smooth basin, then subgradient exploits
    the known Lipschitz constant of 2 for fast local optimization.
    """
    t1 = min(max_time * 0.4, 600)
    t2 = max_time - t1 - 10

    # Stage 1: LSE with moderate beta
    a1, obj1 = run_lse_fista(a_init, t1, rng, 5.0, 200.0, verbose=verbose)

    # Stage 2: Subgradient polishing (large steps!)
    a2, obj2 = run_subgrad_polish(a1, t2, rng, verbose=verbose)

    if obj2 < obj1:
        return a2, obj2
    return a1, obj1


def make_asymmetric_inits(n, rng):
    """Create diverse asymmetric initializations."""
    inits = []
    # Exponential decay
    t = np.arange(n, dtype=np.float64)
    inits.append(np.exp(-t / (n * 0.4)))
    # Off-center Gaussian
    center = n * 0.35
    width = n * 0.2
    inits.append(np.exp(-0.5 * ((t - center) / width) ** 2))
    # Ramp
    inits.append(1.0 + 2.0 * t / n)
    # Asymmetric triangle
    peak = n * 0.4
    a_tri = np.zeros(n)
    for i in range(n):
        if i <= peak:
            a_tri[i] = i / peak
        else:
            a_tri[i] = (n - 1 - i) / (n - 1 - peak)
    inits.append(np.maximum(a_tri, 0.01))
    # Power law
    inits.append((t + 1) ** (-0.3))
    # Bimodal
    c1, c2 = n * 0.3, n * 0.7
    w = n * 0.1
    inits.append(np.exp(-0.5 * ((t - c1) / w) ** 2) + 0.7 * np.exp(-0.5 * ((t - c2) / w) ** 2))
    return inits


def run_diversified_search(a_start, max_time, rng, n):
    """Try multiple perturbed versions and keep the best."""
    S = np.sqrt(2.0 * n)
    a0 = a_start * (S / a_start.sum())
    best_obj = compute_obj_fast(a0)
    best_a = a0.copy()

    n_trials = 5
    t_per = max_time / n_trials

    for trial in range(n_trials):
        if time.monotonic() > time.monotonic() + max_time:
            break
        trial_start = time.monotonic()
        # Perturb
        scale = 0.005 + 0.01 * trial
        a_pert = a0 + rng.normal(0, scale * S / np.sqrt(n), n)
        a_pert = project_simplex(a_pert, S)

        # Quick hybrid optimize
        t_avail = min(t_per, max_time - (time.monotonic() - (trial_start - max_time / n_trials)))
        if t_avail < 10:
            break
        a_r, obj_r = run_hybrid_optimize(a_pert, t_avail, rng, verbose=False)
        if obj_r < best_obj:
            best_obj = obj_r
            best_a = a_r
            print("    Trial %d: obj=%.6f (improved!)" % (trial, obj_r))
        else:
            print("    Trial %d: obj=%.6f" % (trial, obj_r))

    return best_a, best_obj
