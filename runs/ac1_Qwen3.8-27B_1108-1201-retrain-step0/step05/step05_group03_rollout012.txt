"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization with:
1. Efficient FFT-based gradient via top-k log-sum-exp smoothing
2. Adaptive backtracking line search with momentum
3. Nesterov-accelerated averaging for non-convex setting
4. Peak-shaving coordinate refinement
5. Multi-length exploration (n=4096, 8192, 16384)
6. Final high-precision polish on best candidate

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_smooth_gradient(a, weights, fsize):
    """
    Compute the smoothed gradient of the max-convolution objective.
    
    The gradient of b_k with respect to a is: 2 * flip_shift(a, k)
    In FFT domain: DFT(grad) = 2 * conj(A) * DFT(weights_zero_padded)
    """
    A = fft(a, fsize)
    W = fft(weights, fsize)  # weights is zero-padded
    grad_fft = 2.0 * np.conj(A) * W
    grad = np.real(ifft(grad_fft))[:len(a)]
    return grad


def get_topk_weights(c, k, beta):
    """Get top-k indices and LSE weights for smoothing."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.full(k, 1.0 / k)
    else:
        w = e / tot
    return idx, w


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

    best_val = c.max()
    best_a = a.copy()
    
    # Nesterov averaging
    avg_a = a.copy()
    sum_t = 1.0
    sum_ta = 1.0 * a.copy()
    
    eta = eta_init
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        # Nesterov average
        t_next = it + 1
        sum_t += t_next
        sum_ta += t_next * y
        avg_candidate = sum_ta / sum_t
        
        c_avg = fft_convolve_self(avg_candidate)
        if c_avg.max() < best_val:
            best_val = c_avg.max()
            best_a = avg_candidate.copy()
            stuck = 0
        else:
            stuck += 1
        
        if cur_max < best_val:
            best_val = cur_max
            best_a = y.copy()
            stuck = 0
        
        if stuck > 2000:
            break
        
        # Get subgradient (high beta ≈ exact subgradient at max)
        idx, w = get_topk_weights(c, k, beta)
        w_full = np.zeros(fsize)
        w_full[idx] = w
        
        A = fft(y, fsize)
        W = fft(w_full, fsize)
        grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Proximal step (projected gradient)
        # For subgradient method: x_{t+1} = P(y - eta * g)
        # Nesterov: y_{t+1} = x_{t+1} + (t/(t+2))(x_{t+1} - x_t)
        x_new = project_simplex(y - eta * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        
        a = x_new
        y = y_new
        t_val = t_new
        eta = eta_decay * eta
        
        # Periodic restart
        if it % 5000 == 0 and stuck > 500:
            y = best_a.copy()
            t_val = 1.0
            eta = eta_init
            stuck = 0
            sum_t = 1.0
            sum_ta = 1.0 * y.copy()
    
    return best_a, best_val


def run_peak_shaving(a_init, S, max_time, rng):
    """
    Coordinate descent peak-shaving: iteratively find the peak b_k,
    then reduce the elements that contribute most to it.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
        
        # Find the peak location
        peak_k = int(np.argmax(c))
        
        # The peak is b_{peak_k} = sum_i a_i * a_{peak_k - i}
        # Find the pair (i, peak_k - i) with the largest product
        valid_i = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        if len(valid_i) == 0:
            break
        partners = peak_k - valid_i
        products = a[valid_i] * a[partners]
        
        if products.max() < 1e-15:
            break
        
        # Reduce the larger of the pair
        max_idx = int(np.argmax(products))
        i_val = valid_i[max_idx]
        partner_val = partners[max_idx]
        
        # Try reducing a[i_val] slightly
        delta = 0.01 * a[i_val]
        a_trial = a.copy()
        a_trial[i_val] -= delta
        if a_trial[i_val] < 0:
            a_trial[i_val] = 0
        a_trial = normalize(a_trial, S)
        
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < cur_max:
            a = a_trial
        
        # Also try reducing a[partner_val]
        a_trial2 = a.copy()
        delta2 = 0.01 * a[partner_val]
        a_trial2[partner_val] -= delta2
        if a_trial2[partner_val] < 0:
            a_trial2[partner_val] = 0
        a_trial2 = normalize(a_trial2, S)
        
        c_trial2 = fft_convolve_self(a_trial2)
        if c_trial2.max() < c_trial.max() if c_trial.max() < cur_max else True:
            if c_trial2.max() < cur_max or (c_trial.max() < cur_max and c_trial2.max() < c_trial.max()):
                a = a_trial2
        
        # Early exit if no improvement in a while
        if it % 500 == 0:
            # Random perturbation to escape local minima
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a_pert = a + noise
            a_pert = project_simplex(a_pert, S)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_val:
                a = a_pert
                best_val = c_pert.max()
                best_a = a.copy()
    
    return best_a, best_val


def run_momentum_polish(a_init, S, max_time, rng):
    """
    High-beta polishing with very careful step control and multiple trial directions.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    
    k = max(2, int(0.02 * (2 * n - 1)))
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 200.0 * (1.0 + it / 200.0))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
        
        # Get near-exact subgradient
        idx, w = get_topk_weights(c, k, beta)
        w_full = np.zeros(fsize)
        w_full[idx] = w
        
        A = fft(a, fsize)
        W = fft(w_full, fsize)
        grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        grad_norm = grad / gnorm
        
        # Try multiple step sizes (geometric)
        best_step_val = None
        best_step_a = None
        
        for factor in [0.1, 0.3, 0.7, 1.5, 3.0]:
            step = factor * S / (gnorm * np.sqrt(n)) * 0.1
            a_trial = project_simplex(a - step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max:
                if best_step_val is None or c_trial.max() < best_step_val:
                    best_step_val = c_trial.max()
                    best_step_a = a_trial.copy()
        
        if best_step_a is not None:
            a = best_step_a
        else:
            # No improvement - try tiny random perturbation
            if it % 100 == 0:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a_pert = project_simplex(a + noise, S)
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < cur_max:
                    a = a_pert
    
    return best_a, best_val
