Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on LSE-smoothed objective
with adaptive beta, coordinate descent polishing, and multi-start exploration.
Key improvement over previous approach: uses second-order (L-BFGS) optimization
on the LSE-smoothed objective for much faster convergence near the optimum.

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    if v.sum() == S and np.all(v >= -1e-15):
        return np.maximum(v, 0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_stable(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        # Fallback to uniform
        return np.ones(len(x)) / len(x)
    return e / s


def lse_obj_and_grad(x, beta):
    """
    Compute LSE-smoothed objective and gradient w.r.t. x (log-parameters).
    a = softmax(x) is the actual sequence.
    Objective: (1/beta) * log(sum_k exp(beta * (conv(a,a))[k]))
    """
    n = len(x)
    a = softmax_stable(x)
    
    # Convolution
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    
    # LSE
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    f = (np.log(tot) + beta * c_max) / beta
    
    # Weights
    w = e / tot
    
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * a[k-i] (with appropriate bounds)

import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_stable(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def lse_value_and_weights(c, beta):
    """Compute LSE value and weights from convolution vector."""
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    f = (np.log(tot) + beta * c_max) / beta
    w = e / tot
    return f, w, c_max


def obj_grad_x(x, beta):
    """
    LSE-smoothed objective and gradient in x-space where a = softmax(x).
    Returns (f, grad_x).
    """
    n = len(x)
    a = softmax_stable(x)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    f, w, _ = lse_value_and_weights(c, beta)
    W = fft(w, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return f, grad_x

    print(f"Starting optimization, budget={total_budget}s, deadline in {total_budget-45:.0f}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 128:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW BEST: obj={obj:.8f} ({msg})")

    # === Phase 1: L-BFGS on provided sequence with increasing beta ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = 1.0
        a1 = a_provided / a_provided.sum() * S1

        # Convert to log-space for L-BFGS
        a1_safe = np.maximum(a1, 1e-25)
        x1 = np.log(a1_safe)

        betas_schedule = [5, 10, 20, 50, 100, 200, 500]
        time_per_beta = min(120, (deadline - time.monotonic()) * 0.35 / len(betas_schedule))

        print(f"\nPhase 1: L-BFGS on provided (n={n1}), {len(betas_schedule)} beta levels")
        x_curr = x1
        for bi, beta in enumerate(betas_schedule):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(time_per_beta, deadline - time.monotonic() - 30)
            if t_alloc < 15:
                break
            print(f"  beta={beta}, alloc={t_alloc:.0f}s")
            x_curr, f_lse, nfev = lbfgs_run(x_curr, beta, t_alloc)
            a_try = softmax_stable(x_curr) * S1
            update_global(a_try, f"L-BFGS beta={beta}")
            x1 = x_curr

        # Also run subgradient on the best found
        if global_a is not None and time.monotonic() < deadline - 60:
            t_sg = min(200, (deadline - time.monotonic()) * 0.3)
            if t_sg > 30:
                print(f"  Subgradient refinement, t={t_sg:.0f}s")
                a_sub, _ = subgradient_optimize(global_a, S1, t_sg, rng)
                update_global(a_sub, "Phase 1 subgrad")

        # LSE polish
        if global_a is not None and time.monotonic() < deadline - 40:
            t_pol = min(150, (deadline - time.monotonic()) * 0.2)
            if t_pol > 20:
                print(f"  LSE polish, t={t_pol:.0f}s")
                a_pol, _ = lse_polish_direct(global_a, S1, t_pol, rng,
                                             beta_start=100, beta_end=5000)
                update_global(a_pol, "Phase 1 LSE polish")

    # === Phase 2: Multi-start L-BFGS exploration ===
    if global_a is not None:
        n_ref = len(global_a)
        S_ref = 1.0

        print(f"\nPhase 2: Multi-start exploration (n={n_ref})")
        n_starts = 5
        t_per_start = min(120, (deadline - time.monotonic()) * 0.25 / n_starts)
        if t_per_start < 20:
            n_starts = 2
            t_per_start = min(100, (deadline - time.monotonic()) * 0.2)

        for ri in range(n_starts):
            if time.monotonic() > deadline - 60:
                break
            # Create perturbed start
            scales = [0.02, 0.05, 0.1, 0.03, 0.08]
            a_pert = perturbed_restart(global_a, S_ref, rng, scales[ri % len(scales)])
            
            # Convert to log-space
            a_pert_safe = np.maximum(a_pert, 1e-25)
            x_pert = np.log(a_pert_safe)
            
            # Quick L-BFGS with moderate beta
            beta_use = 30 + 20 * ri
            x_best, f_best, _ = lbfgs_run(x_pert, beta_use, t_per_start)
            a_try = softmax_stable(x_best) * S_ref
            update_global(a_try, f"Phase 2 restart {ri}")
            
            # Quick subgradient follow-up
            rem = deadline - time.monotonic() - 60
            if rem > 20:
                a_sg, _ = subgradient_optimize(a_try, S_ref, min(60, rem * 0.4), rng)
                update_global(a_sg, f"Phase 2 restart {ri} subgrad")

    # === Phase 3: Try different sequence lengths ===
    print(f"\nPhase 3: Length exploration")
    lengths_to_try = [4096, 16384]
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break
        S_try = 1.0
        t_alloc = min(180, (deadline - time.monotonic()) * 0.2)
        if t_alloc < 30:
            continue
        
        print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
        if global_a is not None:
            # Interpolate from best
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_init = np.interp(t_new, t_old, global_a)
            a_init = np.maximum(a_init, 1e-10)
            a_init = project_simplex(a_init, S_try)
        else:
            a_init = generate_random_init(n_try, rng, S_try)
        
        # L-BFGS
        a_safe = np.maximum(a_init, 1e-25)
        x_init = np.log(a_safe)
        x_best, f_best, _ = lbfgs_run(x_init, 20, t_alloc * 0.6)
        a_try = softmax_stable(x_best) * S_try
        update_global(a_try, f"Phase 3 n={n_try} L-BFGS")
        
        # Subgradient
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            a_sg, _ = subgradient_optimize(a_try, S_try, min(100, rem * 0.5), rng)
            update_global(a_sg, f"Phase 3 n={n_try} subgrad")
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 20:
                a_pol, _ = lse_polish_direct(a_sg, S_try, min(60, rem2 * 0.5), rng,
                                             beta_start=100, beta_end=3000)
                update_global(a_pol, f"Phase 3 n={n_try} polish")

    # === Phase 4: Intensive refinement on global best ===
    if global_a is not None and time.monotonic() < deadline - 30:
        n_g = len(global_a)
        S_g = 1.0
        remaining = deadline - time.monotonic() - 20
        print(f"\nPhase 4: Final refinement (n={n_g}, remaining={remaining:.0f}s)")
        
        # Alternate L-BFGS (high beta) and subgradient
        a_curr = global_a.copy()
        a_safe = np.maximum(a_curr, 1e-25)
        x_curr = np.log(a_safe)
        
        for round_i in range(6):
            if time.monotonic() > deadline - 20:
                break
            t_round = min(80, (deadline - time.monotonic() - 20) / max(1, 6 - round_i))
            if t_round < 10:
                break
            
            if round_i % 2 == 0:
                # L-BFGS with high beta
                beta_h = 100 * (round_i + 1)
                x_curr, f_b, _ = lbfgs_run(x_curr, beta_h, t_round)
                a_curr = softmax_stable(x_curr) * S_g
                update_global(a_curr, f"Phase 4 L-BFGS r{round_i}")
            else:
                # Subgradient
                a_curr, _ = subgradient_optimize(a_curr, S_g, t_round, rng,
                                                 topk_init=16, topk_max=200)
                update_global(a_curr, f"Phase 4 subgrad r{round_i}")
                a_safe = np.maximum(a_curr, 1e-25)
                x_curr = np.log(a_safe)

    # === Phase 5: Final LSE polish ===
    if global_a is not None and time.monotonic() < deadline - 15:
        n_g = len(global_a)
        S_g = 1.0
        t_final = min(60, deadline - time.monotonic() - 15)
        if t_final > 15:
            print(f"\nPhase 5: Final LSE polish, t={t_final:.0f}s")
            a_pol, _ = lse_polish_direct(global_a, S_g, t_final, rng,
                                         beta_start=200, beta_end=10000)
            update_global(a_pol, "Phase 5 final polish")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256)
        global_best = compute_obj(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, obj={global_best:.8f}")
    
    # Verify with evaluate_sequence
    ver = evaluate_sequence(final_list)
    print(f"evaluate_sequence: {ver:.8f}")
    print(f"{'='*60}")

    return final_list
