"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses a multi-phase approach: LSE-FISTA with adaptive beta, power-sum smoothing,
exact subgradient with Nesterov acceleration, active-set QP refinement,

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def power_weights(c, p):
    """Compute power-sum weights: w_k = c_k^p / sum(c_j^p)."""
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** p
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y (appropriately reversed)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def active_set_gradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)

        if len(active) > n_active_max:

        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Line search

        step = min(step, 0.5 * S / gnorm)

        alpha = step

    a_new = np.clip(result.x, 0, 1000)
    if a_new.sum() > 1e-15:
        a_new = a_new * (S / a_new.sum())
    obj_new = compute_obj(a_new)

    if obj_new < best_obj:
        best_obj = obj_new
        best_a = a_new.copy()

    print("    L-BFGS-beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
          (beta, best_obj, time.monotonic() - t0, result.nit))

    return best_a, best_obj


def try_different_length(a_best, obj_best, n_target, rng, max_time):
    """Try optimizing at a different sequence length."""
    n = n_target
    S = np.sqrt(2.0 * n)

    # Interpolate or resize the best sequence to new length
    if len(a_best) == n:
        a_init = a_best.copy()
    else:
        # Resample
        x_old = np.linspace(0, 1, len(a_best))
        x_new = np.linspace(0, 1, n)
        a_init = np.interp(x_new, x_old, a_best)
        a_init = np.maximum(a_init, 0.0)

    a_init = a_init * (S / a_init.sum())
    obj_init = compute_obj(a_init)

    # Quick optimization
    a_opt, _ = run_lse_fista(a_init, max_time * 0.5, rng,
                              beta_start=5.0, beta_end=2000.0,
                              beta_ramp='exp', step_scale=1.0)
    obj_opt = compute_obj(a_opt)

    if obj_opt < obj_best - 0.001:
        print("    Different length n=%d: obj=%.8f (improved!)" % (n, obj_opt))
        return a_opt, obj_opt
    return None, obj_best


def generate_structured_init(n, rng):
    """Generate a structured initial sequence: constant + alternating + smooth noise."""
    S = np.sqrt(2.0 * n)
    k = np.arange(n)

    # Base: constant
    a = np.ones(n)

    # Add alternating component (helps for even n)
    if n % 2 == 0:
        a += 0.3 * (-1.0) ** k
    else:
        a += 0.3 * np.cos(2.0 * np.pi * k / n)

    # Add smooth random perturbation
    noise = rng.normal(0, 1.0, n)
    F = fft(noise)
    cutoff = max(3, n // 50)
    F[cutoff:-cutoff] *= 0.1
    noise_smooth = np.real(ifft(F))[:n]
    noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15)
    a += 0.1 * noise_smooth

    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimization:
    1. LSE-FISTA with fast beta ramp
    2. Power-sum FISTA for diversity
    3. Exact subgradient with Nesterov
    4. Active-set QP refinement
    5. L-BFGS high-beta polish
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = generate_structured_init(n0, rng)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: LSE-FISTA fast ramp (20% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20
    print("\n=== Phase 1: LSE-FISTA fast ramp (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_lse_fista(global_best_a, p1_budget, rng,
                           beta_start=5.0, beta_end=1000.0,
                           beta_ramp='exp', step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  LSE-FISTA: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: Power-sum FISTA (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.15
        print("\n=== Phase 2: Power-sum FISTA (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        # Try a few different power values
        for p_val in [30.0, 80.0]:
            if time.monotonic() > deadline:
                break
            a2, _ = run_lse_fista(global_best_a, p2_budget / 3.0, rng,
                                   beta_start=50.0, beta_end=2000.0,
                                   beta_ramp='linear', step_scale=0.8,
                                   use_power=True, power_p=p_val)
            obj2 = compute_obj(a2)
            print("  Power-p=%.0f: obj=%.8f" % (p_val, obj2))
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_best_a = a2.copy()

    # === Phase 3: Exact subgradient Nesterov (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.25
        print("\n=== Phase 3: Exact subgradient FISTA (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, _ = run_exact_subgrad_fista(global_best_a, p3_budget, rng,
                                         tol_ramp=(0.0, 0.03))
        obj3 = compute_obj(a3)
        print("  Exact-subgrad: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Active-set QP refinement (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.20
        print("\n=== Phase 4: Active-set QP (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, _ = run_qp_refine(global_best_a, p4_budget, rng, n_active_max=5)
        obj4 = compute_obj(a4)
        print("  QP-refine: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: L-BFGS high-beta polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: L-BFGS polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, obj5 = run_lbfgs_high(global_best_a, p5_budget, rng, beta=5000.0)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Second round of exact subgradient (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p6_budget = remaining * 0.10
        print("\n=== Phase 6: Second exact-subgrad round (budget %.0fs) ===" % p6_budget)
        t6 = time.monotonic()

        a6, _ = run_exact_subgrad_fista(global_best_a, p6_budget, rng,
                                         tol_ramp=(0.0, 0.01))
        obj6 = compute_obj(a6)
        print("  2nd exact-subgrad: obj=%.8f" % obj6)
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 7: Final squeeze (budget %.0fs) ===" % remaining)
        a7, obj7 = run_lbfgs_high(global_best_a, remaining * 0.9, rng, beta=10000.0)
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj + 1e-10:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
