Key innovation: Alternating LP method at moderate n for strong directions,
combined with multi-scale approach and FISTA polishing at full scale.

from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def build_conv_dense(a):
    """Build dense convolution matrix C where (C @ x) = convolve(a, x)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1) + 1
        idx = np.arange(i_start, i_end)
        C[k, i_start:i_end] = a[k - idx]
    return C


def solve_minmax_lp(a_fixed):
    """
    Solve: min t s.t. (a_fixed * b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b_opt, t_opt).
    """
    n = len(a_fixed)
    C = build_conv_dense(a_fixed)  # (2n-1) x n

    # Variables: [b_0, ..., b_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: C @ b - t <= 0
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={"presolve": True, "time_limit": 30})

    if not res.success:
        return a_fixed.copy(), np.inf

    b_opt = np.maximum(res.x[:n], 0.0)
    b_opt = b_opt / b_opt.sum()
    t_opt = res.x[-1]
    return b_opt, t_opt

        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient computation
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            if stuck_count > 500:
                k = max(2, k // 2)
            grad = topk_gradient(y, c, k, power=1.3)
        elif mode == "hybrid_lse":
            # Blend LSE and top-k
            w_lse = lse_weights(c, beta)
            k_top = max(4, min(64, n // 32))
            w_top = np.zeros(len(c))
            idx_top = np.argsort(c)[::-1][:k_top]
            w_top[idx_top] = 1.0 / k_top
            blend = min(1.0, stuck_count / 1000.0)
            w = (1 - blend) * w_lse + blend * w_top
            grad = conv_gradient(y, w)
        elif mode == "exact_top":
            # Use only top few lags with decreasing tolerance
            if stuck_count < 200:
                k = 1
            elif stuck_count < 1000:
                k = 3
            elif stuck_count < 5000:
                k = 8
            else:
                k = min(32, 8 + stuck_count // 1000)
            grad = topk_gradient(y, c, k, power=1.0)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Structured noise when stuck
        if noise_period > 0 and stuck_count > noise_period:
            ns = noise_scale_base * S / np.sqrt(n) * np.exp(-stuck_count / (5 * noise_period))
            if ns > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.05
                noise = np.real(ifft(F))[:n]
                noise = noise / (np.linalg.norm(noise) + 1e-15) * ns
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = 0

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """
    Active-set polishing: use exact subgradient from near-max lags,
    with adaptive step size and momentum.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k and tolerance
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = 4
        elif stuck < 5000:
            k = 8
        else:
            k = min(32, 8 + stuck // 500)

        # Find near-max lags
        max_val = c.max()
        tol = max(1e-10, 1e-6 * stuck)
        active = np.where(c >= max_val - tol)[0]
        k_use = min(k, len(active))
        # Take top k from active
        active_sorted = active[np.argsort(c[active])[::-1]][:k_use]

        w = np.zeros(len(c))
        w[active_sorted] = 1.0 / k_use
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: decrease with iterations for convergence
        base_step = S / gnorm
        step = base_step / (1.0 + 0.01 * it)
        step = min(step, 2.0 * S / (gnorm + 1e-15))
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation when very stuck
        if stuck > 3000 and it % 200 == 0:
            ns = max(1e-8, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with LSE-smoothed max-conv."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [50.0, 200.0, 1000.0, 5000.0]
    for idx, beta in enumerate(betas):
        if time.monotonic() - start > max_time * 0.9:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            # Penalty for sum constraint
            sum_x = x.sum()
            pen = 10.0
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w) + 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 50, 'maxls': 20, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_fista_multi_rate(a_init, max_time, rng):
    """
    FISTA with multiple rate phases: start with low beta (explore),
    then increase beta (exploit).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    phases = [
        {"time_frac": 0.3, "beta_start": 5.0, "beta_rate": 0.005, "step_scale": 1.0, "mode": "lse"},
        {"time_frac": 0.3, "beta_start": 50.0, "beta_rate": 0.05, "step_scale": 0.5, "mode": "hybrid_lse"},
        {"time_frac": 0.2, "beta_start": 200.0, "beta_rate": 0.1, "step_scale": 0.3, "mode": "topk"},
        {"time_frac": 0.2, "beta_start": 1000.0, "beta_rate": 0.5, "step_scale": 0.1, "mode": "exact_top"},
    ]

    a = a_init.copy()
    start = time.monotonic()
    total_elapsed = 0.0

    for phase in phases:
        phase_time = max_time * phase["time_frac"] - total_elapsed
        if phase_time < 1.0:
            break
        a, _ = run_fista(a, phase_time, rng,
                         beta_start=phase["beta_start"],
                         beta_rate=phase["beta_rate"],
                         step_scale=phase["step_scale"],
                         mode=phase["mode"])
        total_elapsed = time.monotonic() - start

    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses alternating LP, multi-scale FISTA, and active-set polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None
    global_best_n = 8192

    def update_best(a_new):
        nonlocal global_best_obj, global_best_a, global_best_n
        obj_new = compute_obj(a_new)
        if obj_new < global_best_obj:
            global_best_obj = obj_new
            global_best_a = a_new.copy()
            global_best_n = len(a_new)
            print("  NEW BEST: obj=%.8f (n=%d)" % (obj_new, len(a_new)))
        return obj_new

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0

    # === Phase 1: FISTA warm-up at n=8192 (15% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.15
    print("\n=== Phase 1: FISTA warm-up n=%d (%.0fs) ===" % (n0, p1_time))
    t1 = time.monotonic()
    a1 = run_fista_multi_rate(global_best_a, p1_time, rng)[0]
    update_best(a1)
    print("  Phase 1 done: t=%.0fs" % (time.monotonic() - t1))

    # === Phase 2: Alternating LP at n=2048 (20% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.20
    print("\n=== Phase 2: Alternating LP n=2048 (%.0fs) ===" % p2_time)
    t2 = time.monotonic()

    n_lp = 2048
    # Downsample current best to n_lp
    a_lp_init = global_best_a[:min(n0, n_lp * 2)]
    if len(a_lp_init) > n_lp:
        # Average blocks
        n_blocks = len(a_lp_init) // n_lp
        a_lp_init = a_lp_init[:n_blocks * n_lp].reshape(n_blocks, n_lp).mean(axis=0)
    else:
        a_lp_init = np.pad(a_lp_init, (0, n_lp - len(a_lp_init)))

    a_lp = alternating_lp(a_lp_init, max_iters=3000, time_limit=p2_time * 0.7, verbose=True)

    # Upsample to n0
    a_up1 = upsample_sequence(a_lp, n0)
    S_n0 = np.sqrt(2.0 * n0)
    a_up1 = a_up1 * S_n0
    update_best(a_up1)
    print("  Phase 2 LP done: t=%.0fs" % (time.monotonic() - t2))

    # === Phase 3: FISTA at n=8192 with LP warm start (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    print("\n=== Phase 3: FISTA n=%d LP-initialized (%.0fs) ===" % (n0, p3_time))
    t3 = time.monotonic()

    # Start from best of (current best, LP-upsampled)
    a3_start = global_best_a.copy()
    a3 = run_fista_multi_rate(a3_start, p3_time, rng)[0]
    update_best(a3)

    # Also try from LP upsampled
    a3b = run_fista_multi_rate(a_up1, (deadline - time.monotonic()) * 0.3, rng)[0]
    update_best(a3b)
    print("  Phase 3 done: t=%.0fs" % (time.monotonic() - t3))

    # === Phase 4: Try n=16384 (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p4_time = remaining * 0.15
        print("\n=== Phase 4: FISTA n=16384 (%.0fs) ===" % p4_time)
        t4 = time.monotonic()

        n4 = 16384
        # Upsample from current best
        a4_init = upsample_sequence(global_best_a / global_best_a.sum(), n4)
        S4 = np.sqrt(2.0 * n4)
        a4_init = a4_init * S4

        a4 = run_fista_multi_rate(a4_init, p4_time, rng)[0]
        obj4 = compute_obj(a4)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()
            global_best_n = n4
            print("  NEW BEST at n=16384: obj=%.8f" % obj4)
        else:
            print("  n=16384: obj=%.8f (no improvement)" % obj4)
        print("  Phase 4 done: t=%.0fs" % (time.monotonic() - t4))

    # === Phase 5: Active-set polishing (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.15
    print("\n=== Phase 5: Active-set polish (%.0fs) ===" % p5_time)
    t5 = time.monotonic()
    a5 = run_active_polish(global_best_a, p5_time, rng)[0]
    update_best(a5)
    print("  Phase 5 done: t=%.0fs" % (time.monotonic() - t5))

    # === Phase 6: Final L-BFGS squeeze (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_time = remaining * 0.10
        print("\n=== Phase 6: L-BFGS final squeeze (%.0fs) ===" % p6_time)
        t6 = time.monotonic()
        a6 = run_lbfgs_polish(global_best_a, p6_time, rng)[0]
        update_best(a6)
        print("  Phase 6 done: t=%.0fs" % (time.monotonic() - t6))

    # === Phase 7: Another round of active polish on best (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p7_time = remaining * 0.5
        print("\n=== Phase 7: Final active polish (%.0fs) ===" % p7_time)
        t7 = time.monotonic()
        a7 = run_active_polish(global_best_a, p7_time, rng)[0]
        update_best(a7)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a7b = run_lbfgs_polish(global_best_a, remaining2 * 0.8, rng)[0]
            update_best(a7b)
        print("  Phase 7 done: t=%.0fs" % (time.monotonic() - t7))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f (n=%d)" % (global_best_obj, len(final_list)))
    return final_list
