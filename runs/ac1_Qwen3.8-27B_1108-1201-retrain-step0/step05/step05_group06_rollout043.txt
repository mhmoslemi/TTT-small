"""
Minimize 2*n*max_conv/sum^2 using alternating Hankel-solve and LSE smoothing.
Key idea: alternate between (1) computing softmax weights from convolution, 
and (2) solving the KKT linear system for the current weights.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def hankel_matvec(lambda_vec, v, n):
    """Compute M*v where M[j,k] = lambda[j+k]."""
    # (M*v)[j] = sum_k lambda[j+k] v[k]
    # = conv(reverse(v), lambda)[j + n - 1]
    v_rev = v[::-1]
    fsize = 1 << (n + len(lambda_vec) - 1).bit_length()
    V = fft(v_rev, fsize)
    L = fft(lambda_vec, fsize)
    result = np.real(ifft(V * L))
    return result[n-1 : 2*n-1]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def hankel_direct_solve(lambda_vec, n, S):
    """Direct solve of M*x = 1, project to simplex."""
    j = np.arange(n)
    M = lambda_vec[j[:,None] + j[None,:]]
    # Make symmetric (should already be, but numerical)
    M = 0.5 * (M + M.T)
    try:
        x = np.linalg.solve(M, np.ones(n))
    except np.linalg.LinAlgError:
        x = np.linalg.lstsq(M, np.ones(n), rcond=None)[0]
    x = np.maximum(x, 0)
    x = normalize_seq(x, S)
    return x

def hankel_iterative_solve(lambda_vec, n, S, max_iter=100, tol=1e-8):
    """Iterative solve of M*x = 1 using CG or GMRES."""
    b = np.ones(n)
    x = np.ones(n) / n  # Initial guess
    
    # Use simple Richardson with optimal step
    # Or use a few CG iterations
    # First, compute the optimal step for Richardson:
    # alpha = (x^T r) / (r^T M r)
    
    # Let's use a preconditioned Richardson / steepest descent
    for it in range(max_iter):
        r = b - hankel_matvec(lambda_vec, x, n)
        res = np.linalg.norm(r) / np.linalg.norm(b)
        if res < tol:
            break
        # Precondition: assume M is roughly (1/n) * I in the "average" sense
        # Optimal step: alpha = (r^T r) / (r^T M r)
        Mr = hankel_matvec(lambda_vec, r, n)
        denom = r @ Mr
        if abs(denom) < 1e-30:
            break
        alpha = (r @ r) / denom
        # Limit step to avoid divergence
        alpha = min(alpha, n)  # Conservative limit
        x_new = x + alpha * r
        # Check non-negativity
        if np.min(x_new) < -0.1 * np.max(x_new):
            alpha *= 0.1
            x_new = x + alpha * r
        x = x_new
    
    x = np.maximum(x, 0)
    x = normalize_seq(x, S)
    return x

def hankel_alternating(a_init, n, S, max_iter=200, beta_init=5.0, beta_max=500.0, use_direct=True, max_direct_n=4096):
    """
    Alternating: update lambda from conv, solve M*a = 1.
    """
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    best_obj = compute_obj(a)
    best_a = a.copy()
    beta = beta_init
    
    for it in range(max_iter):
        # Step 1: Compute convolution and update weights
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Softmax weights
        shifted = beta * (c - c.max())
        e = np.exp(shifted)
        total = e.sum()
        if total < 1e-300:
            lambda_vec = np.ones(2*n-1) / (2*n-1)
        else:
            lambda_vec = e / total
        
        # Step 2: Solve M*a = 1

"""
Minimize 2*n*max_conv/sum^2 using alternating Hankel-KKT solve with LSE FISTA polish.
Key innovation: Instead of iterative FISTA on the LSE surrogate, we directly solve
the KKT linear system (M*a = mu*1) for the current active-set weights at each iteration,
giving exact "Newton" steps for the smoothed subproblem. This converges much faster
in the critical final phase where the active set is well-defined.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def hankel_matvec(lambda_vec, v, n):
    """Compute M*v where M[j,k] = lambda[j+k]. Uses FFT for O(n log n)."""
    v_rev = v[::-1]
    fsize = 1 << (n + len(lambda_vec) - 1).bit_length()
    V = fft(v_rev, fsize)
    L = fft(lambda_vec, fsize)
    result = np.real(ifft(V * L))
    return result[n - 1: 2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def hankel_direct_solve(lambda_vec, n, S):
    """Direct solve of M*x = 1 where M[j,k] = lambda[j+k]. Project to simplex."""
    j = np.arange(n)
    M = lambda_vec[j[:, None] + j[None, :]]
    M = 0.5 * (M + M.T)
    # Add small regularization for stability
    reg = 1e-10 * np.trace(M) / n
    M += reg * np.eye(n)
    try:
        x = np.linalg.solve(M, np.ones(n))
    except np.linalg.LinAlgError:
        x = np.linalg.lstsq(M, np.ones(n), rcond=None)[0]
    x = np.maximum(x, 0.0)
    if x.sum() < 1e-15:
        return np.full(n, S / n)
    return normalize_seq(x, S)


def hankel_cg_solve(lambda_vec, n, S, max_iter=200, tol=1e-10):
    """Solve M*x = 1 using preconditioned CG with FFT matvec."""
    b = np.ones(n)
    x = np.ones(n) / n
    r = b - hankel_matvec(lambda_vec, x, n)
    p = r.copy()
    rs = r @ r
    bnorm = np.linalg.norm(b)
    for _ in range(max_iter):
        if np.sqrt(rs) / bnorm < tol:
            break
        Ap = hankel_matvec(lambda_vec, p, n)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        if alpha < 0:
            # Not SPD, use steepest descent
            alpha = rs / max(pAp, 1e-10)
            alpha = min(alpha, 2.0 / max(np.max(np.abs(hankel_matvec(lambda_vec, np.ones(n), n))), 1e-10))
        x_new = x + alpha * p
        r_new = r - alpha * Ap
        rs_new = r_new @ r_new
        if rs_new >= rs:
            # Line search
            for trial in range(10):
                alpha *= 0.5
                x_new = x + alpha * p
                r_new = r - alpha * Ap
                rs_new = r_new @ r_new
                if rs_new < rs:
                    break
        x = x_new
        r = r_new
        rs = rs_new
        if pAp > 0:
            beta_cg = rs_new / rs
            p = r_new + beta_cg * p
        else:
            p = r_new.copy()
    x = np.maximum(x, 0.0)
    if x.sum() < 1e-15:
        return np.full(n, S / n)
    return normalize_seq(x, S)


def hankel_alternating(a_init, n, S, max_time, beta_init=3.0, beta_max=200.0,
                       use_direct=True, max_direct_n=4096, damp=0.6):
    """
    Alternating: (1) compute softmax weights from conv, (2) solve M*a = 1.
    This solves the KKT conditions exactly at each step for the smoothed problem.
    """
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    best_obj = compute_obj(a)
    best_a = a.copy()
    beta = beta_init
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S ** 2
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Softmax weights over convolution lags
        shifted = beta * (c - cur_max)
        e = np.exp(shifted)
        total = e.sum()
        if total < 1e-300:
            lambda_vec = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            lambda_vec = e / total

        # Solve M * x = 1
        if use_direct and n <= max_direct_n:
            x = hankel_direct_solve(lambda_vec, n, S)
        else:
            x = hankel_cg_solve(lambda_vec, n, S, max_iter=100, tol=1e-8)

        # Damped update
        a = (1.0 - damp) * a + damp * x
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, S)

        # Increase beta
        beta = min(beta_max, beta * (1.03 + 0.02 * it / max(it, 10)))

    return best_a, best_obj


def lse_fista(a_init, max_time, rng, beta_init=5.0, beta_max=50000.0):
    """LSE FISTA with adaptive parameters and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S ** 2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: increase slowly, faster when stuck
        if stuck < 200:
            beta = beta_init * (1.0 + 0.005 * it)
        elif stuck < 2000:
            beta = min(beta_max, beta * 1.01)
        else:
            beta = min(beta_max, beta * 1.03)
        beta = max(beta, beta_init)

        # Compute softmax weights (LSE gradient)
        shifted = beta * (c - cur_max)
        e = np.exp(shifted - shifted.max())
        total = e.sum()
        w = e / total

        # Gradient: 2 * M_w * a where M_w[j,k] = w[j+k]
        grad = 2.0 * hankel_matvec(w, a, n)

        # Backtracking line search on the actual objective
        step = 1.0
        base_obj = c.max()
        for _ in range(15):
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() <= base_obj + step * 0.01 * grad.copy().dot(y - step * grad) + 1e-12:
                break
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Momentum restart
        if stuck > 4000:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def exact_subgrad_polish(a_init, max_time, rng, k_start=1, k_max=64):
    """Polish using exact subgradient with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S ** 2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: start with few active lags, increase when stuck
        if stuck < 50:
            k = k_start
        elif stuck < 200:
            k = min(k_max, 2 + stuck // 50)
        elif stuck < 2000:
            k = min(k_max, 8 + stuck // 100)
        else:
            k = min(k_max, 32 + stuck // 200)

        # Get top-k lags
        k = min(k, 2 * n - 1)
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k

        # Gradient
        grad = 2.0 * hankel_matvec(w, a, n)

        # Step size: use diminishing step
        base_step = S / (np.linalg.norm(grad) + 1e-15)
        step = min(base_step, S / (it * 0.1 + 1.0))
        step = max(step, 1e-15)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 6000 and it % 500 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stuck = 0
            last_improve = it

        if it % 4000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def make_initializations(n, rng, provided=None):
    """Create diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    names = []

    if provided is not None:
        a = np.interp(np.linspace(0, 1, n, endpoint=False),
                      np.linspace(0, 1, len(provided), endpoint=False), provided)
        inits.append(normalize_seq(np.clip(a, 0, 1000), S))
        names.append("provided")

    # Cosine
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(normalize_seq(0.5 + 0.5 * np.cos(2 * np.pi * tt), S))
    names.append("cosine")

    # Asymmetric ramp
    t = np.arange(n, dtype=np.float64) / n
    inits.append(normalize_seq(np.maximum(1.0 - 2.0 * np.abs(t - 0.3), 0.1), S))
    names.append("ramp")

    # Random
    a = rng.random(n) ** 2
    inits.append(normalize_seq(a, S))
    names.append("rand1")

    # Structured random
    a = rng.random(n)
    # Apply low-pass filter
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    freqs = np.fft.fftfreq(fsize)
    mask = 1.0 / (1.0 + 10.0 * np.abs(freqs) * n)
    A *= mask
    a = np.maximum(np.real(ifft(A))[:n], 0)
    inits.append(normalize_seq(a, S))
    names.append("filtered")

    return inits, names


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses alternating Hankel-KKT solve (exact inner solve) + LSE FISTA + subgradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get provided sequence
    a_prov = None
    try:
        a_prov = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_prov = np.clip(a_prov, 0, 1000)
        if a_prov.sum() < 0.01:
            a_prov = None
    except Exception:
        pass

    if a_prov is not None:
        obj_prov = compute_obj(a_prov)
        print("Provided: n=%d, obj=%.6f" % (len(a_prov), obj_prov))
        global_best_obj = obj_prov
        global_best_a = a_prov.copy()

    # Determine which n values to try
    n_candidates = []
    if a_prov is not None:
        n_candidates.append(len(a_prov))
    if 4096 not in n_candidates:
        n_candidates.append(4096)
    if 8192 not in n_candidates:
        n_candidates.append(8192)
    n_candidates.append(16384)
    n_candidates = sorted(set(n_candidates))

    print("Target n values: %s" % n_candidates)

    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break

        print("\n" + "=" * 60)
        print("n = %d, remaining budget: %.0fs" % (n, remaining))
        print("=" * 60)

        S = np.sqrt(2.0 * n)

        # Generate initializations
        inits, names = make_initializations(n, rng, provided=a_prov)

        # Quick eval
        best_init = None
        best_init_obj = np.inf
        for a_i, name in zip(inits, names):
            obj_i = compute_obj(a_i)
            if obj_i < best_init_obj:
                best_init_obj = obj_i
                best_init = a_i
            if obj_i < global_best_obj:
                global_best_obj = obj_i
                global_best_a = a_i.copy()
            print("  init %-8s: obj=%.6f" % (name, obj_i))

        # Phase A: Hankel alternating (exact KKT solve)
        if n <= 4096:
            hankel_budget = min(remaining * 0.35, 400)
        else:
            hankel_budget = min(remaining * 0.25, 300)

        if hankel_budget > 20:
            print("\n  [Phase A] Hankel-KKT alternating (budget %.0fs)" % hankel_budget)
            t0 = time.monotonic()
            a_h, obj_h = hankel_alternating(best_init, n, S, hankel_budget,
                                            beta_init=2.0, beta_max=100.0,
                                            use_direct=(n <= 4096),
                                            max_direct_n=4096, damp=0.7)
            print("  Hankel: obj=%.6f, time=%.1fs" % (obj_h, time.monotonic() - t0))
            if obj_h < global_best_obj:
                global_best_obj = obj_h
                global_best_a = a_h.copy()
            best_init = a_h

            # Second Hankel pass with higher beta
            rem_h = hankel_budget - (time.monotonic() - t0)
            if rem_h > 15:
                a_h2, obj_h2 = hankel_alternating(best_init, n, S, rem_h,
                                                   beta_init=20.0, beta_max=500.0,
                                                   use_direct=(n <= 4096),
                                                   max_direct_n=4096, damp=0.8)
                print("  Hankel2: obj=%.6f" % obj_h2)
                if obj_h2 < global_best_obj:
                    global_best_obj = obj_h2
                    global_best_a = a_h2.copy()
                best_init = a_h2

        # Phase B: LSE FISTA
        remaining = deadline - time.monotonic()
        lse_budget = remaining * 0.35
        if lse_budget > 20:
            print("\n  [Phase B] LSE FISTA (budget %.0fs)" % lse_budget)
            t0 = time.monotonic()
            a_l, obj_l = lse_fista(best_init, lse_budget, rng,
                                    beta_init=5.0, beta_max=20000.0)
            print("  LSE: obj=%.6f, time=%.1fs" % (obj_l, time.monotonic() - t0))
            if obj_l < global_best_obj:
                global_best_obj = obj_l
                global_best_a = a_l.copy()

        # Phase C: Exact subgradient polish
        remaining = deadline - time.monotonic()
        sub_budget = remaining * 0.30
        if sub_budget > 15:
            print("\n  [Phase C] Subgradient polish (budget %.0fs)" % sub_budget)
            t0 = time.monotonic()
            a_s, obj_s = exact_subgrad_polish(global_best_a, sub_budget, rng,
                                               k_start=1, k_max=32)
            print("  Subgrad: obj=%.6f, time=%.1fs" % (obj_s, time.monotonic() - t0))
            if obj_s < global_best_obj:
                global_best_obj = obj_s
                global_best_a = a_s.copy()

        # Phase D: Final Hankel pass on best
        remaining = deadline - time.monotonic()
        if remaining > 20 and len(global_best_a) == n:
            final_h_budget = min(remaining * 0.5, 80)
            if final_h_budget > 10:
                print("\n  [Phase D] Final Hankel (budget %.0fs)" % final_h_budget)
                t0 = time.monotonic()
                a_f, obj_f = hankel_alternating(global_best_a, n, np.sqrt(2.0 * n),
                                                 final_h_budget,
                                                 beta_init=50.0, beta_max=1000.0,
                                                 use_direct=(n <= 4096),
                                                 max_direct_n=4096, damp=0.85)
                print("  Final Hankel: obj=%.6f, time=%.1fs" % (obj_f, time.monotonic() - t0))
                if obj_f < global_best_obj:
                    global_best_obj = obj_f
                    global_best_a = a_f.copy()

    # Final check
    print("\n" + "=" * 60)
    print("Best found: obj=%.6f, n=%d" % (global_best_obj, len(global_best_a) if global_best_a is not None else 0))
    print("=" * 60)

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    else:
        n_fb = 4096
        final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: obj=%.6f" % ver)
    else:
        print("WARNING: evaluate_sequence rejected!")
        final_list = [1.0] * 1024

    return final_list
