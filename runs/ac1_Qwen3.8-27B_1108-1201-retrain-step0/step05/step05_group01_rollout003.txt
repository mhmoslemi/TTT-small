"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses mirror descent (multiplicative updates) with adaptive LSE smoothing,
periodic restarts, L-BFGS-B polishing, and multi-start with diverse
initializations including different sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def mirror_descent(a_init, max_time, rng, beta_start=10.0, beta_end=500.0,
                   beta_period=2000, step_scale=1.0, restart_every=5000):
    """
    Mirror descent (multiplicative updates) with LSE smoothing.
    a_i <- a_i * exp(-eta * g_i), then normalize to sum S.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    last_improve = 0
    prev_x = a.copy()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: slowly increases, periodically resets
        progress = min(1.0, it / beta_period)
        beta = beta_start * (beta_end / beta_start) ** progress
        if it % restart_every == 0:
            beta = beta_start  # Reset beta (annealing restart)
            # Small random perturbation
            noise = rng.normal(0, 0.01 * S / n, n)
            a = np.maximum(a + noise, 1e-15)
            a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Compute gradient
        w = lse_weights(c, beta)
        g = conv_gradient(a, w)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Step size for multiplicative update
        # eta controls the maximum relative change: exp(-eta*max|g|)
        eta = step_scale / np.maximum(np.abs(g).max(), 1e-15)
        eta = min(eta, 1.0)  # Cap to avoid extreme updates

        # When stuck, use smaller steps and more smoothing
        if stuck > 1000:
            eta *= 0.5
            beta = max(beta, 50.0)

        # Multiplicative update
        # a_i <- a_i * exp(-eta * g_i)
        log_update = -eta * g
        # Clip to avoid underflow/overflow
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)

        # Normalize
        a_new = a_new * (S / a_new.sum())

        # Ensure positivity
        a = a_new

        # Periodic check: if multiplicative update is making things worse,
        # fall back to a projected gradient step
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            obj_check = 2.0 * n * c_check.max() / (S * S)
            if obj_check > best_obj * 1.05:
                # Degrading - restore best and use projected step
                a = best_a.copy()
                a = a + rng.normal(0, 0.005 * S / n, n)
                a = np.maximum(a, 0)
                a = a * (S / a.sum())

    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_start=20.0, mode="lse"):
    """Standard FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start * (1.0 + 0.005 * it)
        beta = min(beta, 10000.0)

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        w = lse_weights(c, beta)
        g = conv_gradient(y, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        step = 1.0 / beta
        if stuck > 2000:
            # Add noise to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it

        x_new = project_simplex(y - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """L-BFGS-B polishing with increasing LSE beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 100.0, 300.0, 800.0, 2000.0][:n_levels]

    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining FISTA with adaptive LSE,
Riemannian (mirror) descent, L-BFGS-B polishing, and multi-start.
Key innovations: Barzilai-Borwein step size, adaptive beta with
annealing restarts, and frequency-domain structured perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def bb_step_size(x_prev, x_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    s = x_curr - x_prev
    y_vec = g_curr - g_prev
    ss = s @ s
    yy = y_vec @ y_vec
    sy = s @ y_vec
    if yy < 1e-30:
        return 1.0
    if sy > 1e-30:
        return ss / sy  # BB1
    elif sy < -1e-30:
        return yy / (-sy)  # BB2
    return 1.0


def optimize_fista_bb(a_init, max_time, rng, beta_start=15.0, 
                      beta_growth=0.003, beta_max=5000.0,
                      noise_threshold=1500, noise_scale_base=0.003):
    """
    FISTA with Barzilai-Borwein step size and adaptive noise.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    last_improve = 0
    prev_y = y.copy()
    prev_g = None

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_start * np.exp(beta_growth * it), beta_max)

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Gradient
        w = lse_weights(c, beta)
        g = conv_gradient(y, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Step size: BB if possible, else 1/beta
        if prev_g is not None:
            step_bb = bb_step_size(prev_y, y, prev_g, g)
            step = np.clip(step_bb, 0.1 / beta, 5.0 / beta)
        else:
            step = 1.0 / beta

        # Noise for escape
        if stuck > noise_threshold:
            scale = noise_scale_base * S / np.sqrt(n) * np.exp(-stuck / (3 * noise_threshold))
            if scale > 1e-9:
                # Structured noise in frequency domain
                F = fft(rng.normal(0, 1, n))
                F[max(3, n//100):-max(3, n//100)] *= 0.1
                noise = np.real(ifft(F))[:n]
                noise = noise / (np.linalg.norm(noise) + 1e-15) * scale
                y = project_simplex(y + noise, S)
                t_mom = 1.0

        x_new = project_simplex(y - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        prev_y = y.copy()
        prev_g = g.copy()

        if it % 20000 == 0:
            t_mom = 1.0

    # Final: use best_a
    obj_final = compute_obj(best_a)
    if obj_final < best_obj:
        best_obj = obj_final
    return best_a, best_obj
