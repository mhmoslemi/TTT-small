def lp_refine(a, M, K=50, solver='highs'):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for top-K lags k.
    """
    n = len(a)
    c = fft_convolve_self(a)
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix A_ub such that A_ub @ g <= M * ones(K)
    # Row k: sum_j a[|k-j|] * g[j] <= M  (where a[|k-j|] is valid index)
    # Actually: (a*g)[k] = sum_j a[j] * g[k-j] for valid j and k-j
    # = sum_j a[k-j] * g[j] for valid k-j and j
    
    # For row corresponding to lag k:
    # (a*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * g[j]
    # So A_ub[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0
    
    A_ub = np.zeros((K, n))
    for i, k in enumerate(top_k):
        k = int(k)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            A_ub[i, j] = a[k - j]
    
    c_obj = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    b_ub = np.full(K, M)
    
    result = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                                      bounds=[(0, None)]*n, method='highs')
    return result

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. LSE-FISTA for global optimization
2. LP-based iterative refinement (key new element)
3. Pair-based coordinate descent for local polish
4. Multi-scale approach with upsample
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.optimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0, use_backtrack=True):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.8
            else:
                step = min(step * 1.03, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 5000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def lp_refine(a_init, max_time, rng, K=30, m_reduction=0.001):
    """
    LP-based iterative refinement.
    At each step: solve max sum(g) s.t. g>=0, (a*g)[k] <= M for top-K lags.
    Update a towards the LP solution and reduce M.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    M = c.max()
    start = time.monotonic()
    best_obj = 2.0 * n * M / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    t_blend = 0.1  # blending parameter
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Find top K lags
        top_k_idx = np.argsort(c)[::-1][:K]
        
        # Build LP constraint matrix
        # For lag k: (a*g)[k] = sum_j a[k-j] * g[j] for valid indices
        A_ub = np.zeros((K, n))
        for i, k in enumerate(top_k_idx):
            k = int(k)
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmax >= jmin:
                A_ub[i, jmin:jmax+1] = a[k - jmax:k - jmin + 1][::-1]
        
        # Solve LP: max sum(g) s.t. A_ub @ g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        
        try:
            result = scipy.optimize.linprog(
                c_obj, A_ub=A_ub, b_ub=b_ub,
                bounds=[(0, None)] * n, method='highs',
                options={'presolve': True, 'time_limit': 5}
            )
        except Exception:
            break
        
        if not result.success:
            # Try with fewer constraints
            if K > 5:
                K = max(5, K // 2)
                continue
            break
        
        g = result.x
        S_g = g.sum()
        
        if S_g < S * 0.5:
            # LP solution is much smaller than target; increase M
            M *= 1.01
            continue
        
        # Rescale g to have sum S
        g_rescaled = g * (S / S_g)
        
        # Compute the implied new M
        # (a * g_rescaled)[k] = (S/S_g) * (a*g)[k] <= (S/S_g) * M
        M_implied = M * (S / S_g)
        
        # Blend update
        a_new = (1 - t_blend) * a + t_blend * g_rescaled
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        
        # Compute new objective
        c_new = fft_convolve_self(a_new)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new / (S * S)
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            a = a_new
            c = c_new
            M = M_new
            # Slightly reduce M for next iteration
            # M *= (1 - m_reduction)  # Don't force reduction; let LP handle it
            # Increase blending as we get closer
            t_blend = min(0.5, t_blend * 1.02)
        else:
            # No improvement; try different blending
            t_blend = max(0.01, t_blend * 0.7)
            if it - last_improve > 100:
                # Add small noise to escape
                a = a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                M = c.max()
        
        if it - last_improve > 500:
            break
        
        # Periodically increase K
        if it % 50 == 0:
            K = min(200, K + 10)
    
    return best_a, best_obj

            best_a = a.copy()
            last_improve = it

            break

        k_star = int(np.argmax(c))

        # c[k_star] = sum_j a[j] * a[k_star - j] (valid indices)
        # d(c[k_star])/d(a[j]) = a[k_star - j] (if valid) + a[j - k_star] (if valid, for symmetric part)
        # Actually for self-conv: d/dc[k] a[j] = a[k-j] + a[j-k] (if both valid)
        # But since c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = a[k-j] + a[j-k] (counting both terms if j != k-j)
        # For k = k_star: dc[ks]/da[j] = a[ks-j] (if 0<=ks-j<n) + a[j-ks] (if 0<=j-ks<n)
        
        grad_k = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += a[k_star - j]
            if 0 <= j - k_star < n:
                val += a[j - k_star]
            grad_k[j] = val
        
        # We want to reduce c[k_star]: move mass from high-grad to low-grad positions
        # But we also need to consider the effect on other lags

        # and l with lowest grad_k (least contribution), transfer mass
        src = int(np.argmax(grad_k * np.maximum(a, 1e-10)))

        tgt_score = grad_k * (1.0 + a / (S / n))  # prefer low grad and low value
        tgt = int(np.argmin(tgt_score[src == int(np.argmin(tgt_score))]))
        # Actually, let's just pick the min of tgt_score excluding src
        tgt_score[src] = np.inf
        tgt = int(np.argmin(tgt_score))
        
        if src == tgt or a[src] < 1e-10:
            # Random perturbation
            j = rng.integers(0, n)
            l = rng.integers(0, n)
            if j != l and a[j] > 1e-10:
                delta = a[j] * 0.02
                a[j] -= delta
                a[l] += delta
                c = fft_convolve_self(a)
            continue
        
        # Binary search for optimal transfer amount
        lo, hi = 0.0, a[src] * 0.5
        for _ in range(10):
            mid = (lo + hi) / 2
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[tgt] += mid
            c_trial_k = c[k_star] - mid * grad_k[src] + mid * grad_k[tgt]
            if c_trial_k < c[k_star]:
                lo = mid
            else:
                hi = mid

        if delta > 1e-12:
            a[src] -= delta
            a[tgt] += delta
            # Update c approximately (will recompute periodically)
            # dc[ks] due to a[src] -= delta: -delta * grad_k[src]
            # dc[ks] due to a[tgt] += delta: +delta * grad_k[tgt]
            # But this only updates c[k_star]; other lags also change
            # Let's just recompute c periodically
            if it % 50 == 0:
                c = fft_convolve_self(a)
    
    return best_a, best_obj

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
1. Spectral-motivated initialization (flat-top window design)

3. LP-based iterative refinement (key new element using scipy HiGHS)

import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.optimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        lse_val, grad, c, w = lse_gradient(y, beta)

        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:

    last_improve = 0
    it = 0
    t_blend = 0.15
    K = K_init

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start

        # Find top K lags
        top_k_idx = np.argsort(c)[::-1][:K]

        # Build LP constraint matrix (K x n)
        # For lag k: (a*g)[k] = sum_j a[k-j] * g[j] for valid indices
        A_ub = np.zeros((K, n))
        for i in range(K):
            k = int(top_k_idx[i])
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmax >= jmin:
                A_ub[i, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]

        # Solve LP: max sum(g) s.t. A_ub @ g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)

        try:
            result = scipy.optimize.linprog(
                c_obj, A_ub=A_ub, b_ub=b_ub,
                bounds=[(0, None)] * n, method='highs',
                options={'presolve': True, 'time_limit': 10,
                         'dual_feasibility_tolerance': 1e-7,
                         'primal_feasibility_tolerance': 1e-7}
            )
        except Exception:
            if K > 10:
                K = max(10, K // 2)
                continue
            break

        if not result.success:
            if K > 10:
                K = max(10, K // 2)
                continue
            break

        g = result.x
        S_g = g.sum()

        if S_g < S * 0.1:
            M *= 1.02
            continue

        # Rescale g to have sum S
        g_rescaled = g * (S / S_g)

        # Compute implied M reduction
        M_implied = M * (S / S_g)

        # Blend update
        a_new = (1.0 - t_blend) * a + t_blend * g_rescaled
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            t_blend *= 0.5
            continue
        a_new = a_new * (S / s_new)

        # Evaluate
        c_new = fft_convolve_self(a_new)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new / (S * S)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            a = a_new
            c = c_new
            M = M_new
            t_blend = min(0.6, t_blend * 1.03)
            # Gradually increase K to handle more constraints
            if it % 30 == 0:
                K = min(K_max, K + 5)
        else:
            t_blend = max(0.02, t_blend * 0.6)
            stuck = it - last_improve
            if stuck > 50:
                # Perturb to escape
                a = a + rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                M = c.max()
            if stuck > 300:
                # Try more aggressive M reduction
                M *= 0.995
                K = min(K_max, K + 10)

        if it - last_improve > 500:
            break

    return best_a, best_obj


def multiplicative_weight_refine(a_init, max_time, rng, K=50, eta=0.1):
    """
    Fast multiplicative weight updates for the LP dual.
    Instead of solving LP exactly, use FW (Follow-the-Winner) heuristic.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convove_self(a) if False else fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    # Initialize dual weights uniformly over top K lags
    top_k_idx = np.argsort(c)[::-1][:K]
    y = np.ones(K) / K

    while time.monotonic() - start < max_time:
        it += 1

        # Compute (a * g) for current a... actually we need to find best g
        # MW approach: for each coordinate j, compute "violation" = sum_k y_k * A_k[j]
        # where A_k[j] = a[k-j] if valid
        # The coordinate with minimum violation is the most "constrained"
        # Update: y_k *= exp(-eta * A_k[j*])

        # Compute violations for all j
        # violation[j] = sum_k y[k] * a[top_k_idx[k] - j] (if valid)
        # This is: sum_k y[k] * (shifted a)[j]
        # = FFT-based convolution of y (padded) with a

        # Actually, let's compute it directly for K lags
        # violation[j] = sum_{i=0}^{K-1} y[i] * a[top_k_idx[i] - j] if 0 <= top_k_idx[i]-j < n
        viol = np.zeros(n)
        for i in range(K):
            k = int(top_k_idx[i])
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmax >= jmin:
                viol[jmin:jmax + 1] += y[i] * a[k - jmax:k - jmin + 1][::-1]

        # Find most violated coordinate (minimum violation)
        j_star = int(np.argmin(viol))

        # If violation at j_star is very low, the LP dual is "satisfied"
        # Update y
        for i in range(K):
            k = int(top_k_idx[i])
            if 0 <= k - j_star < n:
                y[i] *= np.exp(-eta * a[k - j_star])

        y_sum = y.sum()
        if y_sum < 1e-300:
            y = np.ones(K) / K
        else:
            y = y / y_sum

        # Construct g from dual: g[j] proportional to exp(-eta * sum_k y[k]*A_k[j])
        # But this is the same as viol, so g[j] = exp(-eta * viol[j])
        g = np.exp(-eta * viol)
        g = g * (S / g.sum())

        # Blend
        t_blend = 0.1
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            a = a_new
            c = c_new
            # Update top K
            top_k_idx = np.argsort(c)[::-1][:K]
            # Reset y periodically
            if it % 100 == 0:
                y = np.ones(K) / K
        else:
            if it - last_improve > 200:
                eta = max(0.01, eta * 0.95)
                a = a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                top_k_idx = np.argsort(c)[::-1][:K]
                y = np.ones(K) / K

        if it - last_improve > 500:
            break

    return best_a, best_obj


def subgradient_polish(a_init, max_time, rng):
    """Polish using exact subgradient with decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(10, 2 + stuck // 200)
        elif stuck < 5000:
            k = min(30, 10 + stuck // 300)
        else:
            k = min(64, 30 + stuck // 200)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, 3.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-12:
            a = a_new
        else:
            for trial in range(3):
                step *= 0.3
                a_new = project_simplex(a - step * grad, S)
                if compute_obj(a_new) <= cur_obj + 1e-12:
                    a = a_new
                    break

        if stuck > 2000 and it % 50 == 0:
            ns = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via cubic interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "flat_top":
        # Blackman-Harris style: designed for flat spectral response
        a = (0.35875 - 0.48829 * np.cos(2 * np.pi * tt)
             + 0.14128 * np.cos(4 * np.pi * tt)
             - 0.01168 * np.cos(6 * np.pi * tt))
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.12 ** 2))
    elif method == "asym_skew":
        # Asymmetric: mass concentrated on one side to exploit asymmetry
        a = np.exp(-4.0 * (tt - 0.3) ** 2 / (0.2 ** 2))
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "power_law":
        a = (1.0 - tt) ** 1.5
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 3.0 * np.abs(tt - 0.5))
    elif method == "two_peak":
        a = np.exp(-50 * (tt - 0.3)**2) + np.exp(-50 * (tt - 0.7)**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
