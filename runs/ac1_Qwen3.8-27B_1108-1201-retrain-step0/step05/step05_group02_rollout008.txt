"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses continuation FISTA + Frank-Wolfe + local search + upsample.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    c = np.maximum(c, 0)  # numerical noise
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-15 * S:
        # Distribute the difference among positive entries
        pos = result > 0
        result[pos] += diff / pos.sum()
        result = np.maximum(result, 0)
    return result


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smoothed_grad(a, beta, fsize):
    """Compute LSE-smoothed objective and gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: 2 * ifft(fft(w, fsize) * conj(fft(a, fsize)))[:n]
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def continuous_fista(a_init, max_time, rng, beta_start=5.0, beta_end=50000.0,
                     beta_iters=200, use_momentum=True, verbose=True):
    """
    FISTA with continuation on beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)  # normalization
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * S
    else:
        a = a * (S / s)
    
    fsize = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    cur_beta = beta_start
    step = 1.0 / beta_start
    
    while time.monotonic() - start < max_time:
        it += 1
        # Update beta
        if it % beta_iters == 0:
            cur_beta = min(beta_end, cur_beta * 1.5)
            step = 1.0 / cur_beta
            t_mom = 1.0  # Reset momentum on beta change
        
        lse_val, grad, c, w = lse_smoothed_grad(y, cur_beta, fsize)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        # Evaluate smoothed objective at trial point
        trial_lse, _, _, _ = lse_smoothed_grad(x_trial, cur_beta, fsize)
        bt = 0
        while trial_lse > lse_val + 1e-15 and step_local > 1e-20 and bt < 10:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            trial_lse, _, _, _ = lse_smoothed_grad(x_trial, cur_beta, fsize)
            bt += 1
        
        # Adaptive step
        if bt < 3:
            step = min(step * 1.05, 5.0 / cur_beta)
        else:
            step = step_local * 0.7
        
        x_new = x_trial
        
        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            t_mom = t_new
        else:
            y = x_new.copy()
        
        a = x_new
        
        # Restart if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            y = a.copy()
            noise = rng.normal(0, max(1e-6, 0.005 * S / np.sqrt(n)), n)
            a = project_simplex(a + noise, S)
            y = a.copy()
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print("  FISTA it=%d, beta=%.0f, obj=%.8f, elapsed=%.0fs" % (it, cur_beta, best_obj, elapsed))
    
    return best_a, best_obj


def frank_wolfe_away(a_init, max_time, rng, verbose=True):
    """Frank-Wolfe with away steps."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    a = a * (S / a.sum())
    
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 100.0
    
    # Track support
    support = np.where(a > S * 1e-6)[0]
    
    while time.monotonic() - start < max_time:
        it += 1
        
        _, grad, c, _ = lse_smoothed_grad(a, beta, fsize)
        
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Find s (min gradient) and w (max gradient in support)
        s_idx = int(np.argmin(grad))
        if len(support) > 0:
            w_idx = support[int(np.argmax(grad[support]))]
        else:
            w_idx = -1
        
        # Determine direction: d = (1-gamma)*e_s - gamma*e_w
        # gamma chosen to minimize ||d||^2
        if w_idx >= 0 and a[w_idx] > 0:
            d = -grad[s_idx]  # simplified: just move toward s
            # Actually, let's use the standard FW direction
            # d = (I - gamma*P_w) * (e_s - a) where P_w is projection onto e_w
            # Simplified: move toward e_s with away from e_w
            gamma = a[w_idx] / (a[w_idx] + 1.0)  # heuristic
            if gamma > 0.5:
                gamma = 0.5
        
        # Standard FW step: move toward e_s
        direction = np.zeros(n)
        direction[s_idx] = 1.0
        direction -= a  # direction = e_s - a
        
        if w_idx >= 0 and a[w_idx] > 0:
            # Away step component
            away = np.zeros(n)
            away[w_idx] = -1.0
            away += a  # away = a - e_w
            # Combine: d = direction + lambda * away
            # Optimal lambda: project onto {sum(d + lambda*away) = 0}
            # Since sum(direction) = 0 and sum(away) = 0, any lambda works for sum constraint
            # Choose lambda to minimize ||d + lambda*away||^2
            # = ||d||^2 + 2*lambda*(d.away) + lambda^2*||away||^2
            da = np.dot(direction, away)
            aa = np.dot(away, away)
            if aa > 1e-30:
                lam = max(0, -da / aa)
                lam = min(lam, 1.0)  # Don't go too far
                direction = direction + lam * away
        
        # Line search
        # a_new = a + t * direction, need sum(a_new) = S and a_new >= 0
        # sum(direction) = 0 (since sum(e_s) = 1 = sum(a))
        # So sum(a + t*direction) = S for all t. Good.
        # Need a + t*direction >= 0, so t <= a_i / (-direction_i) for direction_i < 0
        neg_dir = direction < -1e-15
        if np.any(neg_dir):
            t_max = np.min(a[neg_dir] / (-direction[neg_dir]))
        else:
            t_max = 1.0
        t_max = min(t_max, 1.0)
        
        # Try a few step sizes
        best_t = 0.0
        best_trial_obj = cur_obj
        for t_frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t = t_frac * t_max
            if t < 1e-15:
                break
            a_trial = a + t * direction
            if np.any(a_trial < -1e-15):
                t *= 0.5
                a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * (S / a_trial.sum())
            trial_obj = compute_obj(a_trial)
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_t = t
        
        if best_t > 0:
            a = a + best_t * direction
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
        
        # Update support
        if it % 100 == 0:
            support = np.where(a > S * 1e-5)[0]
        
        if it - last_improve > 5000:
            # Restart with noise
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            support = np.where(a > S * 1e-5)[0]
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print("  FW it=%d, obj=%.8f, elapsed=%.0fs" % (it, best_obj, elapsed))
    
    return best_a, best_obj


def local_search(a_init, max_time, rng, verbose=True):
    """Pairwise local search: transfer mass between pairs."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, None).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    # Precompute the "contribution" of each coordinate to each lag
    # For a transfer of delta from j to l:
    # c_new[k] = c[k] - delta*a[k-j] + delta*a[k-l] (approximately, for the affected range)
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 50000:
            break
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Contribution of coordinate j to c[k_star]: a[k_star - j] (if valid)
        # We want to reduce c[k_star], so we want to reduce a[j]*a[k_star-j]
        # Transfer from j (where a[j] is large and contributes to peak) to l (where it doesn't)
        
        # Compute contribution vector for the top few lags
        top_k_count = min(5, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:top_k_count]
        
        # For each coordinate, compute its "peak contribution"
        peak_contrib = np.zeros(n)
        for kk in top_lags:
            # c[kk] = sum_j a[j]*a[kk-j]
            # coord j contributes a[j]*a[kk-j] to c[kk]
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                peak_contrib[lo:hi+1] += a[kk - hi:kk - lo + 1][::-1]
        
        # Source: high peak_contrib AND large a value
        # Target: low peak_contrib AND small a value
        src_score = peak_contrib * a
        tgt_score = -peak_contrib * (1.0 - a / S)  # prefer small a
        
        # Sample candidates
        n_candidates = min(50, n)
        src_pool = np.argsort(src_score)[::-1][:n_candidates]
        tgt_pool = np.argsort(tgt_score)[:n_candidates]
        
        improved = False
        for _ in range(10):
            j = src_pool[rng.integers(0, len(src_pool))]
            l = tgt_pool[rng.integers(0, len(tgt_pool))]
            if j == l or a[j] < 1e-10:
                continue
            
            # Try transferring a fraction of a[j]
            delta = a[j] * rng.uniform(0.01, 0.1)
            if delta < 1e-12:
                continue
            
            # Compute change in c
            # Reducing a[j] by delta: c[k] changes by -delta * a[k-j] for valid k
            # Increasing a[l] by delta: c[k] changes by +delta * a[k-l] for valid k
            delta_c = np.zeros(2 * n - 1)
            
            j_start = j
            j_end = min(2 * n - 2, j + n - 1)
            if j_end >= j_start:
                # c[j: j+n-1] -= delta * a[0:n] (reversed appropriately)
                # c[k] = sum_j' a[j']*a[k-j'], the term with j'=j is a[j]*a[k-j]
                # After reducing a[j] by delta: c[k] -= delta * a[k-j] for k where 0<=k-j<n
                # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
                delta_c[j:j+n] -= delta * a
            
            l_start = l
            l_end = min(2 * n - 2, l + n - 1)
            if l_end >= l_start:
                delta_c[l:l+n] += delta * a
            
            new_max = np.max(c + delta_c)
            new_obj = 2.0 * n * new_max / (S * S)
            
            if new_obj < cur_obj * (1 - 1e-10):
                # Accept
                a[j] -= delta
                a[l] += delta
                c += delta_c
                improved = True
                break
        
        if not improved and it % 1000 == 0:
            # Occasionally do a random perturbation
            idx = rng.integers(0, n)
            delta = a[idx] * 0.01
            if delta > 1e-12:
                a[idx] -= delta
                a[rng.integers(0, n)] += delta
                c = fft_convolve_self(a)  # Full recompute
        
        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - start
            print("  LS it=%d, obj=%.8f, elapsed=%.0fs" % (it, best_obj, elapsed))
        
        # Periodic full recompute to avoid drift
        if it % 5000 == 0:
            c = fft_convolve_self(a)
    
    return best_a, best_obj
