def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)

Key innovation: Use linear programming to find the optimal first-order

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_active_set(b, M, tol=1e-6, max_k=50):
    """Find lags within tol of the maximum."""
    threshold = M - tol * M
    active = np.where(b >= threshold)[0]
    if len(active) > max_k:
        # Keep only top max_k
        order = np.argsort(b[active])[::-1][:max_k]
        active = active[order]
    return active

def solve_lp_direction(a, b, active, max_time=5.0):
    """
    Solve LP to find direction d that minimizes predicted new max.
    
    Minimize t
    Subject to:
      For k in active: 2 * sum_i d_i * a_{k-i} + b_k <= t
      sum(d) = 0
      d_i >= -a_i (so a + d >= 0)
    
    Returns direction d and predicted t.
    """
    n = len(a)
    K = len(active)
    
    # Build constraint matrix for active lags
    # Row k: 2 * (d conv a)_k <= t - b_k
    # i.e., 2 * sum_i d_i * a_{k-i} - t <= -b_k
    
    # Variables: [d_0, ..., d_{n-1}, t]
    # A_ub * x <= b_ub
    # For each active lag k: 2*sum_i d_i*a_{k-i} - t <= -b_k
    # Plus: -2*sum_i d_i*a_{k-i} + t >= b_k (same thing)
    
    # Let's use: 2*sum_i d_i*a_{k-i} - t <= -b_k
    # A_ub has K rows, n+1 columns
    # Row k: [2*a_{k-0}, 2*a_{k-1}, ..., 2*a_{k-(n-1)}, -1]
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for row, k in enumerate(active):
        # (d conv a)_k = sum_i d_i * a[k-i] where 0 <= k-i < n
        # So coefficient of d_i is a[k-i] if 0 <= k-i < n, else 0
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A_ub[row, i] = 2.0 * a[k - i]
        A_ub[row, n] = -1.0  # coefficient of t
        b_ub[row] = -b[k]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: d_i >= -a_i, t is free (but we can bound it)
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        t = result.x[n]
        return d, t
    else:
        return None, None

"""
Hybrid LP-guided + FISTA optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations:
1. Linear programming to find optimal first-order descent direction over the active set
2. Exact active set tracking with cutting-plane refinement
3. Adaptive line search with actual objective evaluation
4. High-beta FISTA polishing phase
5. Basin hopping with structured Fourier mutations for final improvements
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    max_scaled = scaled.max()
    if max_scaled > 400:
        beta_eff = 400.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_value(a, n):
    """Compute the objective 2*n*max_b/(sum_a)^2."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def solve_lp_direction(a, b, active_lags):
    """
    Solve LP to find direction d minimizing predicted new max over active lags.
    
    Minimize t
    Subject to: 2*(d conv a)_k - t <= -b_k  for k in active_lags
                sum(d) = 0
                d_i >= -a_i
    """
    n = len(a)
    K = len(active_lags)
    
    if K == 0:
        return None
    
    # Build constraint matrix: K rows, n+1 columns
    # Variables: [d_0, ..., d_{n-1}, t]
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for row, k in enumerate(active_lags):
        # Coefficient of d_i is 2*a[k-i] if 0 <= k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            A_ub[row, i_min:i_max+1] = 2.0 * a[k - i_max:i_min - 1:-1]
        A_ub[row, n] = -1.0
        b_ub[row] = -b[k]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5.0})
        if result.success:
            d = result.x[:n]
            if np.linalg.norm(d) < 1e-15:
                return None
            return d
    except Exception:
        pass
    return None


def line_search(a, d, S, current_max, n, max_iterations=30):
    """
    Find best step size eps for a + eps*d.
    Uses bisection to find largest eps that reduces the max.
    """
    # Max feasible step (non-negativity)
    negative_mask = d < -1e-15
    if np.any(negative_mask):
        max_eps = np.min(a[negative_mask] / (-d[negative_mask])) * 0.99
    else:
        max_eps = 1.0
    max_eps = min(max_eps, 1.0)
    
    if max_eps < 1e-12:
        return 0.0, current_max
    
    # Bisection: find largest eps in [0, max_eps] where max(conv) < current_max
    lo = 0.0
    hi = max_eps
    best_eps = 0.0
    best_max = current_max
    
    for _ in range(max_iterations):
        mid = 0.5 * (lo + hi)
        if mid < 1e-15:
            break
        a_try = a + mid * d
        a_try = np.maximum(a_try, 0.0)
        a_try_sum = a_try.sum()
        if a_try_sum < 0.01:
            hi = mid
            continue
        b_try = fft_convolve_self(a_try)
        max_try = b_try.max()
        obj_try = 2.0 * n * max_try / (a_try_sum * a_try_sum)
        if obj_try < best_max:
            best_max = obj_try
            best_eps = mid
            lo = mid
        else:
            hi = mid
    
    return best_eps, best_max


def run_lp_phase(a, S, n, max_time, rng):
    """
    LP-guided optimization phase.
    """
    start = time.monotonic()
    a = np.clip(a, 0, 1000).copy()
    a = a * (S / a.sum())
    
    b = fft_convolve_self(a)
    current_obj = 2.0 * n * b.max() / (a.sum() ** 2)
    best_obj = current_obj
    best_a = a.copy()
    
    stall_count = 0
    max_stall = 80
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(a)
        M = b.max()
        current_obj = 2.0 * n * M / (a.sum() ** 2)
        
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            # Perturb to escape
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            stall_count = 0
            continue
        
        if time.monotonic() - start < max_time - 10:
            # Find active lags
            tol = 1e-4 * M
            active = np.where(b >= M - tol)[0]
            # Limit active set size
            if len(active) > 25:
                order = np.argsort(b[active])[::-1][:25]
                active = active[order]
            # Ensure at least the top lag
            if np.argmax(b) not in active:
                active = np.append(active, np.argmax(b))
            
            # Solve LP
            d = solve_lp_direction(a, b, active)
            
            if d is None:
                # Fallback: use negative subgradient of peak
                k_star = np.argmax(b)
                grad = np.zeros(n)
                i_min = max(0, k_star - (n - 1))
                i_max = min(k_star, n - 1)
                grad[i_min:i_max+1] = 2.0 * a[k_star - i_max:i_min - 1:-1]
                d = grad - np.mean(grad)
                d = np.maximum(d, -a)  # Ensure feasibility direction
            
            # Line search
            eps, new_obj = line_search(a, d, S, current_obj, n)
            
            if eps > 1e-15 and new_obj < current_obj:
                a = a + eps * d
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0.01:
                    a = a * (S / a_sum)
                else:
                    a = best_a.copy()
            else:
                # LP direction didn't help; try subgradient step
                k_star = np.argmax(b)
                grad = np.zeros(n)
                i_min = max(0, k_star - (n - 1))
                i_max = min(k_star, n - 1)
                grad[i_min:i_max+1] = 2.0 * a[k_star - i_max:i_min - 1:-1]
                step = 0.1 * S / (n * max(np.linalg.norm(grad), 1e-10))
                a_new = project_simplex(a - step * grad, S)
                b_new = fft_convolve_self(a_new)
                obj_new = 2.0 * n * b_new.max() / (a_new.sum() ** 2)
                if obj_new < current_obj:
                    a = a_new
                else:
                    stall_count += 10
    
    return best_a, best_obj


def run_fista_phase(a, S, n, max_time, rng, beta_start=100.0, beta_end=50000.0):
    """FISTA with exponentially increasing beta for final polish."""
    start = time.monotonic()
    a = np.clip(a, 0, 1000).copy()
    a = a * (S / a.sum())
    
    y = a.copy()
    t_mom = 1.0
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        obj_y = 2.0 * n * c.max() / (y.sum() ** 2)
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.5 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best_obj


def fourier_mutate(a, rng, scale):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 2.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_basin_hopping(a, S, n, max_time, rng):

def generate_alt_inits(n, rng):
    """Generate alternative initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Various window functions
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.maximum(1.0 - 2.0 * tt, 0.0) + np.maximum(2.0 * tt - 1.0, 0.0) + 0.05)
    
    # Parabolic
    x = 2.0 * tt - 1.0
    inits.append(1.0 - x * x + 0.05)
    
    # Multi-lobe
    inits.append(np.maximum(0.5 + 0.4 * np.cos(4 * np.pi * tt) + 0.1 * np.cos(8 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.6 + 0.3 * np.cos(6 * np.pi * tt) - 0.1 * np.cos(12 * np.pi * tt), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    
    return inits
