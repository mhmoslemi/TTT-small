Minimize 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LSE FISTA with exact-max polishing.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    W = fft(w_full, fsize)
    A = fft(a, fsize)

    """Exact subgradient of max conv at the lag k_star."""
    n = len(a)

    i_start = max(0, k_star - (n - 1))
    i_end = min(n - 1, k_star)
    for i in range(i_start, i_end + 1):
        g[i] = 2.0 * a[k_star - i]
    return g

            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Restart when stuck
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_exact_polish(a_init, S, time_limit, rng):
    """Polish using exact subgradient of the max with backtracking line search."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    step = 1.0
    it = 0
    stuck = 0

    while time.monotonic() - start < time_limit:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Find the max lag(s)
        max_val = cur_max
        # Get all lags within 1e-6 of max (for subdifferential)
        active = np.where(c >= max_val - 1e-8 * max_val)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient: average of gradients at active lags
        grad = np.zeros(n)
        for k_star in active[:5]:  # Limit to top 5 for efficiency
            grad += exact_subgradient_fast(a, int(k_star))
        grad /= min(len(active), 5)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Normalize gradient for consistent step size
        grad = grad / gnorm
        # Adaptive step: decay over time
        base_step = 0.1 * S / np.sqrt(n) * (1.0 / (1.0 + 0.01 * stuck))
        step_use = base_step

        a_new = project_simplex(a - step_use * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max - 1e-12:
            a = a_new
            step = step_use
            if it % 5000 == 0:
                print("    Exact polish it=%d, max=%.8f, obj=%.6f" % (
                    it, a_new.sum() ** 2 / (2 * n) * 2 * n / a_new.sum() ** 2 * c_new.max(),
                    2.0 * n * c_new.max() / (a_new.sum() ** 2)))
        else:
            # Try smaller step
            for _ in range(4):
                step_use *= 0.5
                a_new = project_simplex(a - step_use * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max - 1e-12:
                    a = a_new
                    break
            else:
                # No improvement, perturb
                stuck += 50

        if stuck > 500:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            t_reset = True

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_local_search(a_init, S, time_limit, rng):
    """Local random search: try random directions in the simplex tangent space."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    c_best = fft_convolve_self(a).copy()

    # Compute direction vectors: random, zero-mean (to preserve sum)
    accept_count = 0
    reject_count = 0
    temp = 1.0  # cooling

    while time.monotonic() - start < time_limit:
        # Generate random direction in null space of sum constraint
        d = rng.normal(0, 1, n)
        d -= d.mean()  # zero mean
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            continue
        d = d / d_norm

        # Try different step sizes
        progress = (time.monotonic() - start) / time_limit
        base_size = 0.01 * (1.0 - 0.5 * progress)
        sizes = [base_size, base_size * 0.3, base_size * 0.1, base_size * 3.0]

        improved = False
        for sz in sizes:
            for sign in [1.0, -1.0]:
                a_try = a + sign * sz * S / np.sqrt(n) * d
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try * (S / s_try)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best - 1e-10:
                    a = a_try
                    best = c_try.max()
                    best_a = a_try.copy()
                    improved = True
                    accept_count += 1
                    break
            if improved:
                break
        if not improved:
            reject_count += 1
            # Cool down
            temp *= 0.999

        # Periodic larger perturbation
        if (accept_count + reject_count) % 2000 == 0:
            d_large = rng.normal(0, 1, n)
            d_large -= d_large.mean()
            a_try = a + 0.05 * S / np.sqrt(n) * d_large
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try * (S / s_try)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best:
                    a = a_try
                    best = c_try.max()
                    best_a = a_try.copy()

        if it_check_stuck(accept_count, reject_count, 5000):
            # Restart from best
            a = best_a.copy()

    return best_a, best


def it_check_stuck(accept_count, reject_count, threshold):
    """Check if we've been stuck too long."""
    return reject_count > threshold and accept_count == 0


def run_spectral_flatten(a, S, alpha, iters=10):
    """Apply iterative spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()

    for _ in range(iters):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-20:
            break
        # Flatten: reduce peaks in power spectrum
        target = np.full_like(power, mean_p)
        new_power = power * (1.0 - alpha) + target * alpha
        new_power = np.maximum(new_power, 1e-30)
        # Preserve phase
        A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new * (S / s)
    return a


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses adaptive FISTA + exact-max polishing + local search.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, deadline - start_total))

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best_obj = np.inf
    global_a = None

    n_primary = 8192
    S = np.sqrt(2.0 * n_primary)

    # Prepare initial sequence
    if given is not None and len(given) == n_primary:
        a_start = normalize(given.copy(), S)
    elif given is not None:
        a_start = normalize(resample_sequence(given, n_primary), S)
    else:
        a_start = np.ones(n_primary) * S / n_primary

    # Evaluate starting point
    obj_start = compute_objective(a_start)
    print("Starting objective: %.6f" % obj_start)
    global_best_obj = obj_start
    global_a = a_start.copy()

    # Phase 1: Adaptive FISTA (50% of budget)
    t_p1 = (deadline - time.monotonic()) * 0.50
    if t_p1 > 30:
        print("\n=== Phase 1: Adaptive FISTA (%.0fs) ===" % t_p1)
        a_opt, max_val = run_adaptive_fista(
            a_start, S, t_p1, rng,
            k_frac_max=0.10, k_frac_min=0.003,
            beta_start=20.0, beta_end=10000.0)
        obj_p1 = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
        print("  FISTA result: obj=%.6f" % obj_p1)
        if obj_p1 < global_best_obj:
            global_best_obj = obj_p1
            global_a = a_opt.copy()

    # Phase 2: Exact-max polishing (30% of remaining)
    remaining = deadline - time.monotonic()
    t_p2 = remaining * 0.40
    if t_p2 > 20:
        print("\n=== Phase 2: Exact Polish (%.0fs) ===" % t_p2)
        a_opt, max_val = run_exact_polish(global_a, S, t_p2, rng)
        obj_p2 = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
        print("  Polish result: obj=%.6f" % obj_p2)
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_a = a_opt.copy()

    # Phase 3: Local search (25% of remaining)
    remaining = deadline - time.monotonic()
    t_p3 = remaining * 0.40
    if t_p3 > 15:
        print("\n=== Phase 3: Local Search (%.0fs) ===" % t_p3)
        a_opt, max_val = run_local_search(global_a, S, t_p3, rng)
        obj_p3 = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
        print("  Local search result: obj=%.6f" % obj_p3)
        if obj_p3 < global_best_obj:
            global_best_obj = obj_p3
            global_a = a_opt.copy()

    # Phase 4: Second FISTA pass with different params (if time)
    remaining = deadline - time.monotonic()
    t_p4 = remaining * 0.50
    if t_p4 > 30:
        print("\n=== Phase 4: Second FISTA (%.0fs) ===" % t_p4)
        # Slightly perturb the best
        a_perturb = global_a + rng.normal(0, 0.002 * S / np.sqrt(n_primary), n_primary)
        a_perturb = normalize(a_perturb, S)
        a_opt, max_val = run_adaptive_fista(
            a_perturb, S, t_p4, rng,
            k_frac_max=0.05, k_frac_min=0.002,
            beta_start=100.0, beta_end=20000.0)
        obj_p4 = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
        print("  Second FISTA result: obj=%.6f" % obj_p4)
        if obj_p4 < global_best_obj:
            global_best_obj = obj_p4
            global_a = a_opt.copy()

        # Polish the new best
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_opt, max_val = run_exact_polish(global_a, S, remaining * 0.6, rng)
            obj_pol = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
            print("  Final polish: obj=%.6f" % obj_pol)
            if obj_pol < global_best_obj:
                global_best_obj = obj_pol
                global_a = a_opt.copy()

    # Phase 5: Final intensive local search
    remaining = deadline - time.monotonic() - 10
    if remaining > 15 and global_a is not None:
        print("\n=== Phase 5: Final Local Search (%.0fs) ===" % remaining)
        a_opt, max_val = run_local_search(global_a, S, remaining, rng)
        obj_p5 = 2.0 * n_primary * max_val / (a_opt.sum() ** 2)
        print("  Final local search: obj=%.6f" % obj_p5)
        if obj_p5 < global_best_obj:
            global_best_obj = obj_p5
            global_a = a_opt.copy()

    # Final verification
    print("\n=== Final ===")
    print("Best objective (internal): %.6f" % global_best_obj)

    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Verification via evaluate_sequence: %.6f" % ver)
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected! Using fallback.")
                global_a = np.ones(256) * (np.sqrt(512.0) / 256)
                final_list = [float(x) for x in global_a]
        except Exception as e:
            print("  Verification error: %s" % str(e))

    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Total time: %.1fs" % elapsed)
    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
