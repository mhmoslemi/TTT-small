1. FISTA with adaptive top-k LSE smoothing and backtracking
2. Pair-swap coordinate descent (convex 1D subproblem per step)
3. Multi-n exploration (4096, 8192, 16384)
4. Spectral perturbation for escaping local minima
5. Extended high-beta polishing phase

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    S = a.sum()
    return normalize(a_new, S)


def run_fista(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.004,
              beta_max=10000.0, k_frac=0.1, use_momentum=True):
    """Refined FISTA with top-k LSE, backtracking, and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    sticky = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            sticky = 0
        else:
            sticky += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(2.0 * beta * 0.5, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(6):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - y)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        if sticky > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            sticky = 0

        if it % 3000 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.1, rng)
            t_val = 1.0
            sticky = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_pair_swap(a_init, S, max_time, rng):
    """Pair-swap coordinate descent: pick (j, j2), find optimal transfer t."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best = c.max()
    best_a = a.copy()
    it = 0

    # Precompute convolution length
    c_len = 2 * n - 1

    while time.monotonic() - start < max_time:
        it += 1

        # Pick a random pair
        j = rng.integers(0, n)
        j2 = rng.integers(0, n)
        if j == j2:
            j2 = (j + 1) % n

        aj = a[j]
        aj2 = a[j2]

        if aj < 1e-15 and aj2 < 1e-15:
            continue

        # t in [-aj, aj2] to maintain non-negativity
        t_min = -aj
        t_max = aj2

        if t_max - t_min < 1e-15:
            continue

        # Compute the linear coefficient vector v[k] = 2*(a[k-j] - a[k-j2])
        # and quadratic coefficient q[k] = delta_{k,2j} - 2*delta_{k,j+j2} + delta_{k,2*j2}
        v = np.zeros(c_len)
        # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k < j+n
        idx1_start = j
        idx1_end = min(j + n, c_len)
        if idx1_end > idx1_start:
            v[idx1_start:idx1_end] += 2.0 * a[:idx1_end - idx1_start]

        idx2_start = j2
        idx2_end = min(j2 + n, c_len)
        if idx2_end > idx2_start:
            v[idx2_start:idx2_end] -= 2.0 * a[:idx2_end - idx2_start]

        # Quadratic term
        q = np.zeros(c_len)
        if 2 * j < c_len:
            q[2 * j] += 1.0
        if j + j2 < c_len:
            q[j + j2] -= 2.0
        if 2 * j2 < c_len:
            q[2 * j2] += 1.0

        # Golden section search for t in [t_min, t_max]
        # f(t) = max_k (c[k] + t*v[k] + t^2*q[k])
        gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ~0.618

        def eval_f(t):
            return np.max(c + t * v + t * t * q)

        t_a = t_min
        t_d = t_max
        t_b = t_d - gr * (t_d - t_a)
        t_c = t_a + gr * (t_d - t_a)
        f_b = eval_f(t_b)
        f_c = eval_f(t_c)

        for _ in range(40):
            if f_b < f_c:
                t_d = t_c
                t_c = t_b
                f_c = f_b
                t_b = t_d - gr * (t_d - t_a)
                f_b = eval_f(t_b)
            else:
                t_a = t_b
                t_b = t_c
                f_b = f_c
                t_c = t_a + gr * (t_d - t_a)
                f_c = eval_f(t_c)

        t_opt = (t_a + t_d) / 2.0
        f_opt = eval_f(t_opt)

        # Check if this is an improvement
        if f_opt < best - 1e-15 or f_opt < c.max() - 1e-15:
            a[j] += t_opt
            a[j2] -= t_opt
            c = c + t_opt * v + t_opt * t_opt * q
            # Ensure non-negativity (numerical safety)
            a = np.maximum(a, 0.0)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        # Periodic full recompute to avoid drift
        if it % 5000 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        # Periodic perturbation
        if it % 20000 == 0 and (time.monotonic() - start) / max_time < 0.8:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)

    final_c = fft_convolve_self(a)
    if final_c.max() < best:
        best = final_c.max()
        best_a = a.copy()

    return best_a, best


def run_high_beta_polish(a_init, S, max_time, rng):
    """High-beta FISTA polishing with no momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    sticky = 0

    k = max(2, int(0.02 * (2 * n - 1)))
    beta_init = 500.0
    beta_rate = 0.002
    beta_max = 50000.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            sticky = 0
        else:
            sticky += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / max(2.0 * beta * 0.5, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        y = x_new
        a = x_new
        t_val = 1.0

        if sticky > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            sticky = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2380):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA -> Pair-swap CD -> High-beta polish -> Multi-n.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a_candidate, n):
        nonlocal global_best, global_a
        S = a_candidate.sum()
        if S < 0.01:
            return
        obj = 2.0 * n * fft_convolve_self(a_candidate).max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a_candidate.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))

    def try_pipeline(n, a_start, time_limit, label):
        """Run a full optimization pipeline for a given length."""
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
            a_start = normalize(a_start, S)
        elif a_start is None:
            a_start = np.ones(n) * S / n

        print("\n=== %s: n=%d, time=%.0fs ===" % (label, n, time_limit))
        a_cur = a_start.copy()

        # Phase 1: FISTA (45% of time)
        t1 = time_limit * 0.45
        if t1 > 20:
            a_opt, val_opt = run_fista(a_cur, S, t1, rng,
                                       beta_init=15.0, beta_rate=0.004,
                                       beta_max=5000.0, k_frac=0.1,
                                       use_momentum=True)
            a_cur = a_opt
            obj = 2.0 * n * val_opt / (a_cur.sum() ** 2)
            print("  FISTA: obj=%.6f" % obj)
            update_global(a_cur, n)

        # Phase 2: Pair-swap coordinate descent (30% of time)
        remaining = deadline - time.monotonic()
        t2 = min(time_limit * 0.30, max(0, remaining - 30))
        if t2 > 15:
            a_opt, val_opt = run_pair_swap(a_cur, S, t2, rng)
            obj = 2.0 * n * val_opt / (a_opt.sum() ** 2)
            print("  Pair-swap: obj=%.6f" % obj)
            if obj < 2.0 * n * fft_convolve_self(a_cur).max() / (a_cur.sum() ** 2):
                a_cur = a_opt
            update_global(a_cur, n)
            update_global(a_opt, n)

        # Phase 3: High-beta polish (20% of time)
        remaining = deadline - time.monotonic()
        t3 = min(time_limit * 0.20, max(0, remaining - 20))
        if t3 > 10:
            a_opt, val_opt = run_high_beta_polish(a_cur, S, t3, rng)
            obj = 2.0 * n * val_opt / (a_opt.sum() ** 2)
            print("  Polish: obj=%.6f" % obj)
            if obj < 2.0 * n * fft_convolve_self(a_cur).max() / (a_cur.sum() ** 2):
                a_cur = a_opt
            update_global(a_cur, n)
            update_global(a_opt, n)

        # Phase 4: Another pair-swap round on best (5% of time)
        remaining = deadline - time.monotonic()
        t4 = min(time_limit * 0.05, max(0, remaining - 15))
        if t4 > 10:
            a_opt, val_opt = run_pair_swap(a_cur, S, t4, rng)
            obj = 2.0 * n * val_opt / (a_opt.sum() ** 2)
            print("  Pair-swap 2: obj=%.6f" % obj)
            update_global(a_opt, n)

        return a_cur

    # Main run: n = 8192
    remaining = deadline - time.monotonic()
    if remaining > 120:
        try_pipeline(8192, given, remaining * 0.55, "Main n=8192")

    # Secondary: n = 16384
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        a_sec = resample_sequence(global_a, 16384)
        try_pipeline(16384, a_sec, remaining * 0.25, "Secondary n=16384")

    # Tertiary: n = 4096
    remaining = deadline - time.monotonic()
    if remaining > 150 and global_a is not None:
        a_ter = resample_sequence(global_a, 4096)
        try_pipeline(4096, a_ter, remaining * 0.20, "Tertiary n=4096")

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs, n=%d) ===" % (remaining, n_best))

        t_fin = remaining * 0.5
        a_fin, val_fin = run_high_beta_polish(global_a, S_best, t_fin, rng)
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        print("  Final high-beta: obj=%.6f" % obj_fin)
        update_global(a_fin, n_best)

        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 20 and global_a is not None:
            a_pol, val_pol = run_pair_swap(global_a, S_best, remaining2, rng)
            update_global(a_pol, len(a_pol))

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver) or ver > global_best + 0.01:
            print("  Verification failed! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Done in %.1fs. Returning sequence of length %d, best obj=%.6f" %
          (elapsed, len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
