import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    # Standard simplex projection with non-negativity
    n = len(v)
    v = np.asarray(v, float)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=float)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    s = a.sum()
    if s < 1e-15:

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-n exploration with refined subgradient optimization,
surgical coordinate polishing, and adaptive restarts.
Key improvements over previous:
- Direct exact subgradient with Polyak-style step sizing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_to_sum(a, S):
    """Scale a to have sum S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_at_y(y, active_lags):
    """
    Compute subgradient of max_k (y*y)_k at the active lags.
    For full conv b_k = sum_i y_i * y_{k-i}:
    db_k/dy_j = y_{k-j} if 0 <= k-j < n, else 0.
    """
    n = len(y)
    g = np.zeros(n)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            g[j_start:j_end + 1] += y[idx]
    g /= len(active_lags)
    return g


def lse_subgradient(y, c, beta):
    """LSE-smoothed subgradient."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def surgical_peak_reduction(a, max_rounds=100):
    """
    Directly reduce coordinates that contribute to peak lags.
    Returns modified a.
    """
    n = len(a)
    a = a.copy()
    S_target = a.sum()
    
    for _ in range(max_rounds):
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_mean = c.mean()
        
        # If max is close to mean, we're near optimal
        if cur_max < cur_mean * 1.05:
            break
        
        # Find top lags
        k_peak = int(np.argmax(c))
        
        # Find contributing pairs
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        if j_start > j_end:
            break
        
        # Get the paired values
        idx = k_peak - np.arange(j_start, j_end + 1)
        paired_vals = a[idx]
        positions = np.arange(j_start, j_end + 1)

        total_products = products.sum()

        target_reduction = 0.05 * cur_max

        order = np.argsort(paired_vals)[::-1]

                remaining -= reduce_amt * pv

    print("  FISTA-LSE: obj=%.6f (t=%.1fs)" % (obj_fista, time.monotonic() - t2))
    if obj_fista < global_best_obj:
        global_best_obj = obj_fista
        global_best_a = a_fista.copy()
    
    # Continue with exact subgradient
    rem2 = phase2_budget - (time.monotonic() - t2)
    if rem2 > 10:
        t2b = time.monotonic()
        a_sg, obj_sg = optimize_subgradient_only(global_best_a, rem2, rng)
        print("  Subgradient: obj=%.6f (t=%.1fs)" % (obj_sg, time.monotonic() - t2b))
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()
    
    # ============================
    # Phase 3: FISTA at second-best n
    # ============================
    remaining = deadline - time.monotonic()
    phase3_budget = remaining * 0.20
    second_n = None
    for n in n_candidates:
        if n != best_init_n:
            second_n = n
            break
    
    if second_n is not None and phase3_budget > 20:
        print("\n=== Phase 3: FISTA at n=%d (budget %.0fs) ===" % (second_n, phase3_budget))
        t3 = time.monotonic()
        S3 = np.sqrt(2.0 * second_n)
        a_resamp = normalize_to_sum(resample_sequence(global_best_a, second_n), S3)
        a_f3, obj_f3 = optimize_fista(
            a_resamp, phase3_budget * 0.7, rng,
            use_lse=True, beta_start=3.0, beta_end=3000.0, active_tol=0.004
        )
        print("  FISTA n=%d: obj=%.6f (t=%.1fs)" % (second_n, obj_f3, time.monotonic() - t3))
        if obj_f3 < global_best_obj:
            global_best_obj = obj_f3
            global_best_a = a_f3.copy()
        
        rem3 = phase3_budget - (time.monotonic() - t3)
        if rem3 > 10:
            a_sg3, obj_sg3 = optimize_subgradient_only(global_best_a, rem3, rng)
            print("  Subgrad n=%d: obj=%.6f (t=%.1fs)" % (second_n, obj_sg3, time.monotonic() - t3))
            if obj_sg3 < global_best_obj:
                global_best_obj = obj_sg3
                global_best_a = a_sg3.copy()
    
    # ============================
    # Phase 4: Surgical peak reduction
    # ============================
    remaining = deadline - time.monotonic()
    phase4_budget = remaining * 0.20
    if phase4_budget > 10:
        print("\n=== Phase 4: Surgical polish (budget %.0fs) ===" % phase4_budget)
        t4 = time.monotonic()
        
        a_surg = surgical_peak_reduction(global_best_a, max_rounds=200)
        obj_surg = compute_obj(a_surg)
        print("  Surgical: obj=%.6f (t=%.1fs)" % (obj_surg, time.monotonic() - t4))
        if obj_surg < global_best_obj:
            global_best_obj = obj_surg
            global_best_a = a_surg.copy()
        
        # After surgical, do a brief FISTA to clean up
        rem4 = phase4_budget - (time.monotonic() - t4)
        if rem4 > 10:
            a_f4, obj_f4 = optimize_fista(
                global_best_a, rem4, rng,
                use_lse=True, beta_start=50.0, beta_end=5000.0, active_tol=0.002
            )
            print("  Post-surgical FISTA: obj=%.6f (t=%.1fs)" % (obj_f4, time.monotonic() - t4))
            if obj_f4 < global_best_obj:
                global_best_obj = obj_f4
                global_best_a = a_f4.copy()
    
    # ============================
    # Phase 5: Final intensive polishing
    # ============================
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()
        
        # Multiple rounds of subgradient + surgical
        for round_i in range(3):
            rem_round = deadline - time.monotonic()
            if rem_round < 5:
                break
            
            a_sg5, obj_sg5 = optimize_subgradient_only(global_best_a, rem_round * 0.4, rng)
            if obj_sg5 < global_best_obj:
                global_best_obj = obj_sg5
                global_best_a = a_sg5.copy()
            
            a_surg5 = surgical_peak_reduction(global_best_a, max_rounds=50)
            obj_surg5 = compute_obj(a_surg5)
            if obj_surg5 < global_best_obj:
                global_best_obj = obj_surg5
                global_best_a = a_surg5.copy()
            
            rem2_round = deadline - time.monotonic()
            if rem2_round < 5:
                break
            
            a_f5, obj_f5 = optimize_fista(
                global_best_a, rem2_round, rng,
                use_lse=True, beta_start=100.0, beta_end=20000.0, active_tol=0.001
            )
            if obj_f5 < global_best_obj:
                global_best_obj = obj_f5
                global_best_a = a_f5.copy()
        
        print("  Final: obj=%.6f (t=%.1fs)" % (global_best_obj, time.monotonic() - t5))
    
    # ============================
    # Final verification and return
    # ============================
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected!")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]
    
    return final_list
