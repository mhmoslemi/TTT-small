My improvements over the previous approach:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Improved LSE-FISTA with peak-targeted smoothing, adaptive

upsample, and spectral-informed initialization.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smooth_max(c, beta):
    """Compute LSE approximation of max(c) and its gradient weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    return lse_val, w, cm


def topk_weighted_lse(c, beta, k):
    """LSE with extra weight on top-k elements (peak-targeted smoothing)."""
    cm = c.max()
    shifted = beta * (c - cm)
    # Extra penalty on top-k
    idx = np.argsort(c)[::-1][:k]
    shifted[idx] += 2.0  # Additional 2/beta boost to top-k
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    return lse_val, w, cm


def gradient_from_weights(w, a):
    """Compute gradient of sum_k w_k * C_k w.r.t. a, where C = a*a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_improved(a_init, max_time, rng, beta_init=25.0, beta_rate=0.015,
                       beta_max=50000.0, use_backtrack=True, topk_frac=0.05,
                       momentum_restart=3000):
    """
    FISTA with adaptive LSE, peak-targeted smoothing, and backtracking.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)
    prev_grad = None
    prev_x = None
    # Barzilai-Borwein
    bb_step = step
    k_top = max(3, int(topk_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        # Use peak-targeted LSE
        c = fft_convolve_self(y)
        lse_val, w, cur_max = topk_weighted_lse(c, beta, k_top)
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        grad = gradient_from_weights(w, y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_x is not None:
            sy = y - prev_x
            yg = grad - prev_grad
            sy_norm = np.dot(sy, sy)
            yg_norm = np.dot(yg, yg)
            if sy_norm > 1e-30 and np.dot(sy, yg) > 1e-30:
                bb_step = sy_norm / np.dot(sy, yg)
                bb_step = np.clip(bb_step, 0.1 * step, 5.0 * step)

        if use_backtrack:
            step_local = max(step, bb_step * 0.5)
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 6:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 3.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - max(step, bb_step * 0.5) * grad, S)

        # Momentum with periodic restart
        if it - last_improve > momentum_restart:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.005 * S / np.sqrt(n) *
                             np.exp(-(it - last_improve) / momentum_restart))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        prev_grad = grad.copy()
        prev_x = y.copy()

        # Timeout check every 100 iters
        if it % 100 == 0 and time.monotonic() - start >= max_time:
            break

    return best_a, best_obj


def peak_shaving_cd(a_init, max_time, rng, n_top_peaks=5):
    """
    Greedy peak-shaving: identify top peaks in autocorrelation,
    compute sensitivity vectors, transfer mass to reduce the max.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 50000:
            break

        # Get top peaks
        n_peaks = min(n_top_peaks, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute multi-peak sensitivity for each coordinate
        # Sensitivity of coordinate j to peak at lag k: a[k-j] if valid
        sens = np.zeros(n)
        for kk in peak_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                # a[j] contributes a[kk-j] to c[kk] for j in [lo, hi]
                sens[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Score: high sensitivity + high value = good source
        # Low sensitivity + low value = good target
        src_score = sens * (a + 1e-10)
        tgt_score = -sens * (S / n - a + 1e-10)

        # Try top candidates
        n_src = min(8, n)
        n_tgt = min(8, n)
        src_cand = np.argsort(src_score)[::-1][:n_src]
        tgt_cand = np.argsort(tgt_score)[:n_tgt]

        best_imp = cur_obj + 1e-15
        best_j, best_l, best_frac = -1, -1, 0.0

        for j in src_cand:
            if a[j] < 1e-12:
                continue
            for l in tgt_cand:
                if j == l:
                    continue
                # Try transferring fraction f of a[j] to a[l]
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[j] * frac
                    if delta < 1e-14:
                        continue
                    # Compute effect on c
                    # c changes by: -delta * (shifted a) + delta * (shifted a)
                    # Effect of -delta at position j: c[k] -= delta * a[k-j] for valid k
                    # Effect of +delta at position l: c[k] += delta * a[k-l] for valid k
                    delta_c = np.zeros(2 * n - 1)
                    # Position j subtraction
                    lo_j = max(0, j)
                    hi_j = min(2 * n - 2, j + n - 1)
                    # c[k] gets -delta * a[k-j] for k in [j, j+n-1]
                    # a[k-j] for k-j in [0, n-1]
                    delta_c[j:j + n] -= delta * a[:min(n, 2 * n - 1 - j)]
                    # Position l addition
                    lo_l = max(0, l)
                    hi_l = min(2 * n - 2, l + n - 1)
                    delta_c[l:l + n] += delta * a[:min(n, 2 * n - 1 - l)]

                    new_max = np.max(c + delta_c)
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_imp:
                        best_imp = new_obj
                        best_j, best_l, best_frac = int(j), int(l), frac

        if best_j >= 0:
            j, l, frac = best_j, best_l, best_frac
            delta = a[j] * frac
            a[j] -= delta
            a[l] += delta
            # Update c
            delta_c = np.zeros(2 * n - 1)
            delta_c[j:j + n] -= delta * a[:min(n, 2 * n - 1 - j)]
            delta_c[l:l + n] += delta * a[:min(n, 2 * n - 1 - l)]
            c += delta_c
        else:
            # No improvement; add small noise to escape
            if stuck > 500:
                n_noise = min(10, n)
                idx_out = rng.choice(n, size=n_noise, replace=False)
                idx_in = rng.choice(n, size=n_noise, replace=False)
                for i in range(n_noise):
                    if a[idx_out[i]] > 1e-10:
                        d = a[idx_out[i]] * 0.01
                        a[idx_out[i]] -= d
                        a[idx_in[i]] += d
                c = fft_convolve_self(a)

        # Periodic recompute
        if it % 50 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def subgradient_polish(a_init, max_time, rng):
    """Polish using exact subgradient with adaptive step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 5000:
            break

        # Adaptive k for top-k subgradient
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(32, 8 + stuck // 500)

        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k
        grad = gradient_from_weights(w, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 2.0 * S / (gnorm + 1e-15))

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 2000 and it % 200 == 0:
            ns = max(1e-7, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.3 + 0.7 * np.abs(np.sin(np.pi * n * tt ** 2))
    elif method == "multi_tone":
        a = 0.2
        for f, ph in [(3, 0.0), (7, 1.3), (11, 2.1), (17, 3.5), (23, 0.7)]:
            a += 0.15 * np.sin(2 * np.pi * f * tt + ph)
        a = np.abs(a) + 0.05
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "zadoff_chu":
        # Zadoff-Chu inspired: exponential phase
        root = 7  # Chosen root
        a = np.ones(n)
        for j_local in range(n):
            a[j_local] = 1.0 + 0.5 * np.cos(2 * np.pi * root * j_local * (j_local + 1) / (2 * n))
        a = np.maximum(a, 0.0)
    elif method == "sinc_like":
        a = np.sinc(3.0 * (tt - 0.5))
        a = np.abs(a) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample_linear(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

    # Final short peak shaving if time remains
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_final_ps, obj_final_ps = peak_shaving_cd(
            global_best_a, remaining * 0.8, rng, n_top_peaks=3)
        if obj_final_ps < global_best_obj:
            global_best_obj = obj_final_ps
            global_best_a = a_final_ps.copy()
            print("  Final peak shaving: obj=%.8f" % obj_final_ps)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
