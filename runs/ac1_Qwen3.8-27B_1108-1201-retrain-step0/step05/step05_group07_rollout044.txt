"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a multi-phase approach combining:
1. Subgradient descent with Nesterov momentum and adaptive top-k lag weighting
2. LP-based direction finding via constraint generation (solving small LPs on active constraints)
3. LSE (log-sum-exp) smoothed polishing with multiple beta schedules
4. Final L-BFGS-B optimization on the LSE-smoothed objective using softmax parameterization

Key improvements over baseline:
- Nesterov momentum for faster subgradient convergence
- LP directions that optimally balance peak reduction across active constraints
- Softmax-parameterized L-BFGS for final superlinear convergence
- Adaptive k selection based on effective peak support
- Cyclical LSE schedules to escape local minima
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize as scipy_minimize
from scipy import sparse
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_topk_subgrad(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_max = c.max()
    c_min_top = c[top_idx].min()
    span = c_max - c_min_top
    if span > 1e-15:
        weights = (c[top_idx] - c_min_top) / span
        weights += 1.0
    else:
        weights = np.ones(k)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_weights(c, beta):
    """Compute softmax weights for LSE."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled_max = scaled.max()
    if scaled_max > 500:
        scaled = scaled - scaled_max
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return e / tot


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    w = compute_lse_weights(c, beta)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta):
    """Compute LSE value (smooth approximation of max)."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled_max = scaled.max()
    if scaled_max > 500:
        scaled = scaled - scaled_max
    return cm + np.log(np.exp(scaled).sum()) / beta


def effective_peak_support(c, rel_tol=0.01):
    """Number of lags within rel_tol fraction of the max."""
    c_max = c.max()
    threshold = c_max * (1.0 - rel_tol)
    return int(np.sum(c >= threshold))


def solve_lp_direction(f, M, active_lags):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in active_lags, g >= 0.
    Returns the optimal g vector.
    """
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return np.ones(n)

    # Build sparse constraint matrix
    rows = []
    cols = []
    data = []
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = k - np.arange(j_start, j_end + 1)
            rows.extend([i] * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
            data.extend(f[indices].tolist())

    if len(data) == 0:
        return np.ones(n)

    A_ub = sparse.csc_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)

    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 10})

    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def subgradient_momentum_optimize(a0, S, max_time, rng, verbose=True):
    """
    Subgradient optimization with Nesterov momentum, adaptive k,
    and LP-based direction updates.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    # Momentum state
    v = np.zeros(n)
    mu = 0.7  # initial momentum
    mu_max = 0.9
    mu_decay = 1.0005

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    lp_interval = 15  # solve LP every this many iterations
    last_lp_time = 0.0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Adaptive k: grow with progress, but also based on peak support
        k_base = int(5 + 60 * progress ** 0.4)
        peak_support = effective_peak_support(c, rel_tol=0.05)
        k = max(k_base, min(peak_support + 5, 100))

        gap = cur_max - c.mean()
        if gap < 1e-12:
            break

        # --- Get direction ---
        use_lp = (it % lp_interval == 0 and
                  elapsed - last_lp_time > 2.0 and
                  time.monotonic() - start < max_time * 0.7)

        if use_lp:
            top_idx = np.argsort(c)[::-1][:min(50, 2 * n - 1)]
            g_dir = solve_lp_direction(a, cur_max, top_idx)
            if g_dir is not None and g_dir.sum() > 1e-10:
                # Rescale g_dir to have same sum as a
                g_dir = g_dir * (S / g_dir.sum())
                # Use LP direction
                grad = a - g_dir  # direction toward LP solution
                source = "lp"
            else:
                grad = compute_topk_subgrad(a, c, k)
                source = "subgrad"
            last_lp_time = elapsed
        else:
            grad = compute_topk_subgrad(a, c, k)
            source = "subgrad"

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 10.0 / gnorm)

        # Nesterov momentum update
        v = mu * v - step * grad
        a_try = project_simplex(a + v, S)

        # Line search with backtracking
        accepted = False
        for ls in range(8):
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a = a_try
                accepted = True
                break
            v = mu * v + step * grad  # undo
            step *= 0.5
            v = mu * v - step * grad
            a_try = project_simplex(a + v, S)
            # Recompute from scratch if needed
            if ls == 7:
                a_try = project_simplex(a - 0.001 * grad / gnorm, S)
                a = a_try
                accepted = True
                break

        if not accepted:
            a = project_simplex(a - 0.001 * grad / gnorm, S)

        # Update momentum
        mu = min(mu_max, mu * mu_decay)

        # Noise injection when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.001 * (stuck / patience)) * S / np.sqrt(n)
            # Use low-frequency noise (smoother perturbations)
            noise_freq = rng.integers(1, 5)
            t = np.arange(n)
            noise = np.zeros(n)
            for h in range(noise_freq):
                noise += rng.normal() * np.cos(2 * np.pi * (h + 1) * t / n) + \
                         rng.normal() * np.sin(2 * np.pi * (h + 1) * t / n)
            noise = noise - noise.mean()
            a = project_simplex(a + noise_scale * noise, S)
            v = np.zeros(n)
            mu = 0.5
            stuck = 0
            patience = int(patience * 1.12)

        if verbose and it % 500 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, max={best:.4f}, k={k}, src={source}")

    return best_a, best


def lse_polish_optimize(a0, S, max_time, rng, beta_init=20.0, beta_end=500.0,
                        schedule='exp', verbose=True):
    """
    LSE-smoothed gradient polishing with adaptive step size.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Beta schedule
        if schedule == 'exp':
            beta = beta_init * np.exp(np.log(beta_end / beta_init) * progress)
        elif schedule == 'linear':
            beta = beta_init + (beta_end - beta_init) * progress
        else:  # cyclical
            period = max_time / 5.0
            phase = (elapsed % period) / period
            beta = beta_init + (beta_end - beta_init) * phase

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (beta * gnorm + 1e-10)
        step = min(step, S / (beta * gnorm + 1e-10))

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)

        if c_try.max() <= cur_max + 1e-12:
            a = a_try
        else:
            # Halve step
            step *= 0.5
            a_try = project_simplex(a - step * grad, S)
            a = a_try

        if verbose and it % 200 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, beta={beta:.1f}, obj={cur_obj:.6f}")

    return best_a, best


def softmax_lbfgs_polish(a0, S, max_time, beta=50.0, verbose=True):
    """
    Final polishing using L-BFGS-B on LSE-smoothed objective
    with softmax parameterization to handle simplex constraint.
    """
    n = len(a0)
    a_init = project_simplex(a0, S)

    # Initialize z from a via log
    a_safe = np.maximum(a_init, 1e-10)
    z = np.log(a_safe / a_safe.sum() * n)  # log of normalized values
    z = z - z.mean()  # center for numerical stability

    def objective(z_vec):
        """Compute LSE objective from z (log-parameter)."""
        z_shifted = z_vec - z_vec.max()
        p = np.exp(z_shifted)
        p = p / p.sum()
        a = S * p
        c = fft_convolve_self(a)
        return lse_value(c, beta)

    def gradient(z_vec):
        """Compute gradient of LSE objective w.r.t. z."""
        z_shifted = z_vec - z_vec.max()
        p = np.exp(z_shifted)
        p = p / p.sum()
        a = S * p
        c = fft_convolve_self(a)
        grad_a = compute_lse_gradient(a, c, beta)
        # Chain rule: df/dz_i = S * p_i * (grad_a_i - sum(p * grad_a))
        dot_pg = np.dot(p, grad_a)
        return S * p * (grad_a - dot_pg)

    start = time.monotonic()
    best_val = objective(z)
    best_z = z.copy()

    try:
        result = scipy_minimize(
            objective,
            z,
            jac=gradient,
            method='L-BFGS-B',
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 50,
            }
        )
        if result.success or result.fun < best_val:
            best_val = result.fun
            best_z = result.x
    except Exception as e:
        if verbose:
            print(f"      L-BFGS error: {e}")

    elapsed = time.monotonic() - start
    a_opt = project_simplex(S * np.exp(best_z - best_z.max()), S)
    c = fft_convolve_self(a_opt)

    if verbose:
        print(f"      L-BFGS done: {elapsed:.1f}s, lse={best_val:.6f}, max_conv={c.max():.6f}")

    return a_opt, c.max()


def try_different_lengths(a_best, S_best, n_best, deadline, rng, verbose=True):

        if n_try not in [c[0] for c in candidates]:
            candidates.append((n_try, factor))

    for n_try, factor in candidates:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n_best:
            # Already have the best at this length
            S_try = np.sqrt(2.0 * n_try)
            c = fft_convolve_self(a_best)
            obj = 2.0 * n_try * c.max() / (a_best.sum() ** 2)
            results.append((n_try, obj, a_best.copy()))
            continue

        S_try = np.sqrt(2.0 * n_try)
        # Interpolate from best
        if n_best != n_try:
            t_old = np.linspace(0, 1, n_best)
            t_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(t_new, t_old, a_best)
            a_interp = np.maximum(a_interp, 0.0)
        else:
            a_interp = a_best.copy()

        a_interp = project_simplex(a_interp, S_try)

        print(f"\nPhase 3: L-BFGS-B polish, n={n_best}, time={t_phase3:.0f}s")

        # Try multiple beta values
        betas = [30.0, 80.0, 200.0]
        t_per_beta = t_phase3 / len(betas)

        for b_val in betas:
            if time.monotonic() > deadline - 30:
                break
            a_lbfgs, mx = softmax_lbfgs_polish(
                global_a, S_best, t_per_beta, beta=b_val, verbose=True)
            update_global(a_lbfgs, f"Phase 3 L-BFGS beta={b_val}")

        # One more LSE pass after L-BFGS
        if global_a is not None:
            rem2 = deadline - time.monotonic()
            if rem2 > 40:
                a_final, _ = lse_polish_optimize(
                    global_a, S_best, min(100, rem2 * 0.6), rng,
                    beta_init=100.0, beta_end=800.0, schedule='exp', verbose=True)
                update_global(a_final, "Phase 3 final LSE")

    # === Phase 4: Length exploration ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 100:
        print(f"\nPhase 4: Length exploration, remaining={remaining:.0f}s")
        results = try_different_lengths(
            global_a, np.sqrt(2.0 * len(global_a)), len(global_a),
            deadline - 30, rng, verbose=True)

        for n_try, obj, a_try in results:
            update_global(a_try, f"Phase 4 n={n_try}")

    # === Phase 5: Final intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 50:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        t_phase5 = min(200, remaining - 20)

        print(f"\nPhase 5: Final refinement, n={n_final}, time={t_phase5:.0f}s")

        # Alternate subgradient and LSE
        a_curr = global_a.copy()
        n_iters = max(2, int(t_phase5 / 60))
        t_per_iter = t_phase5 / n_iters

        for i in range(n_iters):
            if time.monotonic() > deadline - 10:
                break
            if i % 2 == 0:
                a_curr, _ = subgradient_momentum_optimize(
                    a_curr, S_final, min(t_per_iter, deadline - time.monotonic() - 10),
                    rng, verbose=False)
            else:
                a_curr, _ = lse_polish_optimize(
                    a_curr, S_final, min(t_per_iter, deadline - time.monotonic() - 10),
                    rng, beta_init=50.0, beta_end=500.0, schedule='exp', verbose=False)
            update_global(a_curr, f"Phase 5 iter {i}")

        # Final L-BFGS
        rem_final = deadline - time.monotonic()
        if rem_final > 20:
            a_ff, _ = softmax_lbfgs_polish(
                global_a, S_final, min(60, rem_final - 5),
                beta=100.0, verbose=True)
            update_global(a_ff, "Phase 5 final L-BFGS")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
