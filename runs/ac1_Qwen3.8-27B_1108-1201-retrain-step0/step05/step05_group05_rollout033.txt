def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def peak_shave_gradient_multi(a, c, k=8):

    W = fft(w_full, fsize)
    A = fft(a, fsize)

def lp_direction(a, S, max_time=5.0):
    """
    LP-based direction finding: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= max(a*a) for all k.
    Uses a simplified approach: iteratively find the binding constraints
    and solve a smaller LP.
    """
    n = len(a)
    b = fft_convolve_self(a)
    M = b.max()

    # For efficiency, use a projected gradient approach as approximation
    # to the LP: iteratively project onto the constraint set
    # The LP optimum is the projection of 1 (the objective direction)
    # onto the polytope {g >= 0 : (a*g)[k] <= M for all k}

    # Use a simplified iterative method:
    # Start with g = a (normalized to sum S)
    g = a.copy()
    g = g / g.sum() * S

    for it in range(100):
        if time.monotonic() - max_time > time.monotonic():
            break
        # Compute (a*g)
        ag = fft_convolve_self_p(a, g, n)
        # Find violated constraints
        violated = np.where(ag > M + 1e-10)[0]
        if len(violated) == 0:
            break
        # Reduce g at coordinates that contribute most to violated constraints
        # For each violated k, the contributing coords are j where 0 <= k-j < n
        # with weight a[k-j]
        penalty = np.zeros(n)
        for k in violated[:20]:  # Limit to 20 most violated
            excess = ag[k] - M
            for j in range(n):
                k2 = k - j
                if 0 <= k2 < n:
                    penalty[j] += excess * a[k2]
        # Reduce g proportionally to penalty
        g = g - 0.01 * penalty * S / max(penalty.max(), 1e-10)
        g = np.maximum(g, 0)
        gsum = g.sum()
        if gsum < 1e-15:
            break
        g = g * S / gsum

    return g


def fft_convolve_p(a, b, n):
    """Cross-convolution of a and b (both length n) via FFT."""
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic small perturbation
        if it % 5000 == 0:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_peak_guided_hopping(a_init, S, max_time, rng):
    """
    Basin hopping with peak-guided mutations:
    - Identify peak-contributing coordinates
    - Mutate specifically in those regions
    - Use smaller mutations that target the problem areas
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    n_iters = max(1, int(max_time / (opt_time + 10)))

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = it / max(1, n_iters)

        # Peak-guided mutation
        c = fft_convolve_self(a)
        k_star = int(np.argmax(c))

        # Identify the "problem region": coordinates with high contribution to peak
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib[j] = 2.0 * a[k2]
        
        # Mutation: reduce high-contribution coords, increase low-contribution
        mask_high = contrib > np.percentile(contrib, 70)
        mask_low = contrib < np.percentile(contrib, 30)
        
        scale = 0.15 * (1.0 + np.exp(-3.0 * progress))
        
        a_new = a.copy()
        n_high = mask_high.sum()
        n_low = mask_low.sum()
        if n_high > 0 and n_low > 0:
            # Move mass from high to low
            total_move = scale * a[mask_high].sum()
            move_per_high = total_move / n_high
            move_per_low = total_move / n_low
            a_new[mask_high] = np.maximum(a[mask_high] - move_per_high, 0)
            a_new[mask_low] = a[mask_low] + move_per_low
            a_new = project_simplex(a_new, S)
        
        # Add some random Fourier noise for diversity
        if rng.random() < 0.3:
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a_new, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            phase_noise = rng.normal(0, 0.05 * (1 + np.arange(fsize) / fsize), fsize)
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a_new = np.real(ifft(new_A, fsize))[:n]
            a_new = project_simplex(a_new, S)

        # Short FISTA
        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - opt_time)
        if t_alloc < 5:
            t_alloc = 5
        a_opt, val = run_fista_improved(a_new, S, t_alloc, rng,
                                         beta_init=20 + 20 * it,
                                         beta_rate=0.004,
                                         beta_max=10000 + 1000 * min(it, 20),
                                         use_multi_peak=(it % 2 == 0))
        if val < best:
            best = val
            best_a = a_opt.copy()
            a = a_opt
            print("    Hopping iter %d: new best max_b=%.8f" % (it, val))
        else:
            # Keep a mixture for exploration
            a = 0.7 * a_opt + 0.3 * a
            a = project_simplex(a, S)

    return best_a, best


def run_kkt_polish_final(a_init, S, max_time, rng):
    """
    Final KKT-guided polishing: use very high beta LSE to precisely
    identify the active set and polish.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    y = a.copy()
    t_mom = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()

        # Very high beta for precise subgradient
        beta = 500.0 * (1.0 + 10.0 * progress)
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small steps for precision
        step = 0.05 * S / (n * (1.0 + 0.005 * it))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits_v2(n, rng, given_seq=None):
    """Generate diverse initializations with more variety."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000))

    # Various windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 3 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.2) + 0.01)
    inits.append(np.exp(-5.0 * (tt - 0.5) ** 2 / 0.15) + 0.01)

    # Asymmetric windows
    inits.append(np.maximum(np.sin(np.pi * tt ** 0.7), 0.01))
    inits.append(np.maximum(np.sin(np.pi * tt ** 1.4), 0.01))
    inits.append(np.maximum(tt * (1 - tt) * 4 + 0.01, 0.01))

    # Multi-lobe
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * tt) + 0.1 * np.cos(2 * np.pi * 5 * tt), 0.01))
    inits.append(np.maximum(1.0 - 0.4 * np.cos(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt), 0.01))

    # Uniform (worst case baseline)
    inits.append(np.ones(n) * 0.5)

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(1.0, 0.5, n)) + 0.05)

    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Multi-n diverse screening (~20% time)
    2. Intensive FISTA on top candidates (~30% time)
    3. Peak-guided basin hopping (~25% time)
    4. Coordinate descent refinement (~10% time)
    5. KKT final polish (~15% time)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    total_budget = budget_s - 30
    deadline = total_start + total_budget

    print("Starting optimization, budget=%.0fs" % total_budget)

    # Try to get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" % (
                len(given), given.sum(), objective_value(given)))
    except Exception:
        pass

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_global(a, val):
        nonlocal global_best, global_a, global_n
        s = a.sum()
        if s < 0.01:
            return False
        obj = 2.0 * len(a) * val / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = len(a)
            return True
        return False

    # === Phase 1: Multi-n screening ===
    print("\n=== Phase 1: Multi-n screening ===")
    p1_start = time.monotonic()
    p1_budget = total_budget * 0.18

    # Try different n values
    n_list = [4096, 8192, 16384]
    # If we have a given sequence, also try its length
    if given is not None and given.size not in n_list:
        n_list.insert(1, given.size)

    candidates = []

    for n in n_list:
        if time.monotonic() > deadline - total_budget * 0.75:
            break
        S = np.sqrt(2.0 * n)
        t_alloc = min(p1_budget / len(n_list), deadline - time.monotonic() - total_budget * 0.7)
        if t_alloc < 20:
            continue

        print("  n=%d, alloc=%.0fs" % (n, t_alloc))
        inits = generate_inits_v2(n, rng, given_seq=given if (given is not None and len(given) == n) else None)

        # Quick screen top 4 inits
        n_screen = min(4, len(inits))
        per_init = t_alloc / n_screen

        for i in range(n_screen):
            if time.monotonic() > deadline - total_budget * 0.7:
                break
            tt = min(per_init, deadline - time.monotonic() - total_budget * 0.65)
            if tt < 5:
                break
            a_opt, val = run_fista_improved(inits[i], S, tt, rng,
                                             beta_init=5.0, beta_rate=0.01, beta_max=5000)
            obj = 2.0 * n * val / (a_opt.sum() ** 2)
            candidates.append((obj, n, a_opt.copy()))
            print("    n=%d, init %d: obj=%.6f" % (n, i, obj))
            update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f (n=%d)" % (global_best, global_n))

    # === Phase 2: Intensive FISTA ===
    print("\n=== Phase 2: Intensive FISTA ===")
    p2_start = time.monotonic()
    p2_budget = total_budget * 0.28

    if global_a is not None:
        # Run multiple FISTA variants with different parameters
        n_variants = 3
        S = np.sqrt(2.0 * global_n)
        per_variant = p2_budget / n_variants

        for v in range(n_variants):
            if time.monotonic() > deadline - total_budget * 0.4:
                break
            t_alloc = min(per_variant, deadline - time.monotonic() - total_budget * 0.35)
            if t_alloc < 20:
                break

            if v == 0:
                # Standard LSE FISTA with high beta
                a_opt, val = run_fista_improved(global_a, S, t_alloc, rng,
                                                 beta_init=50.0, beta_rate=0.003, beta_max=50000)
            elif v == 1:
                # Multi-peak subgradient FISTA
                a_opt, val = run_fista_improved(global_a, S, t_alloc, rng,
                                                 beta_init=20.0, beta_rate=0.003,
                                                 beta_max=20000, use_multi_peak=True)
            else:
                # Start from a slightly perturbed version
                perturbed = project_simplex(global_a * (0.9 + 0.2 * rng.random(global_n)), S)
                a_opt, val = run_fista_improved(perturbed, S, t_alloc, rng,
                                                 beta_init=30.0, beta_rate=0.004, beta_max=30000)

            update_global(a_opt, val)
            print("  FISTA variant %d: obj=%.6f" % (v, global_best))

    # === Phase 3: Peak-guided basin hopping ===
    print("\n=== Phase 3: Peak-guided basin hopping ===")
    p3_start = time.monotonic()
    p3_budget = total_budget * 0.25

    if global_a is not None:
        S = np.sqrt(2.0 * global_n)
        a_h, val_h = run_peak_guided_hopping(global_a, S, p3_budget, rng)
        update_global(a_h, val_h)
        print("  After basin hopping: obj=%.6f" % global_best)

    # === Phase 4: Coordinate descent refinement ===
    print("\n=== Phase 4: Coordinate descent refinement ===")
    p4_start = time.monotonic()
    p4_budget = total_budget * 0.10

    if global_a is not None:
        S = np.sqrt(2.0 * global_n)
        max_cd_iters = int(p4_budget * 50)  # Rough estimate of iterations
        a_cd, val_cd = coordinate_descent_peak_shave(global_a, S, max_cd_iters, rng)
        update_global(a_cd, val_cd)

        # Follow up with FISTA from the CD result
        remaining = deadline - time.monotonic() - total_budget * 0.12
        if remaining > 20:
            a_opt, val = run_fista_improved(global_a, S, remaining, rng,
                                             beta_init=100.0, beta_rate=0.002, beta_max=100000)
            update_global(a_opt, val)
        print("  After CD + FISTA: obj=%.6f" % global_best)

    # === Phase 5: KKT final polish ===
    print("\n=== Phase 5: KKT final polish ===")
    p5_start = time.monotonic()
    p5_budget = deadline - p5_start - 10

    if global_a is not None and p5_budget > 20:
        S = np.sqrt(2.0 * global_n)
        a_kkt, val_kkt = run_kkt_polish_final(global_a, S, p5_budget, rng)
        update_global(a_kkt, val_kkt)
        print("  After KKT polish: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(global_n) * np.sqrt(2.0 * global_n) / global_n
            final_list = [float(x) for x in global_a]
    else:
        n = 8192
        global_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
