def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. f*g <= M (componentwise), g >= 0."""
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, 
                  bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):

    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a, S):
    """Normalize a to have sum S, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    size = 2 * n - 1
    # Vectorized: for column j, rows j+j+j+... with values f[0], f[1], ...
    j = np.arange(n, dtype=np.int32)
    d = np.arange(n, dtype=np.int32)
    rows = (j[:, None] + d[None, :]).ravel()
    cols = np.repeat(j, n)
    vals = np.tile(f, n)
    A = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return A


def solve_lp_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. A_f * g <= M (componentwise), g >= 0.
    Returns g* or None if LP fails.
    """
    size = 2 * n - 1
    A = build_conv_matrix(f, n)
    
    c_obj = -np.ones(n)
    b_vec = np.full(size, M)
    
    # Use HiGHS with time limit
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 120, 'presolve': True})
    
    if res.status == 0:
        return res.x
    elif res.status == 3:  # time limit
        # Return best primal solution found
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            return np.maximum(res.x, 0)
    return None


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross
