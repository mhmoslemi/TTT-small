import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return np.array([])
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

            print(f"    sym_nest: it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={obj:.6f}")

    return best_b, best_max


def optimize_asymmetric_nesterov(a0, S, max_time, rng, verbose=True):
    """
    Nesterov-accelerated LSE optimization for general (asymmetric) sequences.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    it = 0
    stuck_count = 0
    patience = 300

    beta_start = 30.0
    beta_end = 15000.0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)

        # Nesterov
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        y = a + momentum * (a - a_prev)
        t_mom = t_new

        a_y = project_simplex(y, S)
        c_y = fft_convolve_self(a_y)
        w, cm = compute_lse_weights(c_y, beta)
        grad = lse_gradient_from_weights(w, a_y)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 1.5)
        step = min(step, 3.0 / (gnorm + 1e-10))

        c_curr = fft_convolve_self(a)
        cur_max = c_curr.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)

        accepted = False
        for _ in range(8):
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a_prev = a
                a = a_try
                accepted = True
                break
            step *= 0.5
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)

        if not accepted:
            a_prev = a
            a = project_simplex(a - (0.001 / gnorm) * grad, S)

        # Noise when stuck
        if stuck_count > patience:
            noise_scale = 0.015 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            a_prev = a.copy()
            t_mom = 1.0
            stuck_count = 0
            patience = int(patience * 1.15)

        if verbose and it % 500 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    asym_nest: it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={obj:.6f}")

    return best_a, best_max


def optimize_mixed_phase(a0, S, max_time, rng, verbose=True):
    """
    Mixed optimization: alternate between top-k subgradient and LSE.
    This helps escape from smooth-approximation Plateaus.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Choose mode based on progress
        if progress < 0.4:
            mode = 'subgrad'
            k = max(4, int(20 * progress))
        elif progress < 0.7:
            mode = 'lse'
            beta = 50.0 + 200.0 * (progress - 0.4) / 0.3
        else:
            mode = 'lse_high'
            beta = 200.0 + 4000.0 * (progress - 0.7) / 0.3

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if mode == 'subgrad':
            # Top-k subgradient
            top_idx = np.argsort(c)[::-1][:k]
            median_c = np.median(c)
            weights = np.maximum(c[top_idx] - median_c, 1e-10)
            weights /= weights.sum()

            grad = np.zeros(n)
            for ii in range(k):
                kidx = int(top_idx[ii])
                wi = weights[ii]
                j_start = max(0, kidx - n + 1)
                j_end = min(n - 1, kidx)
                if j_start <= j_end:
                    indices = kidx - np.arange(j_start, j_end + 1)
                    grad[j_start:j_end + 1] += 2.0 * wi * a[indices]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 2.0 * (cur_max - c.mean()) / (gnorm * gnorm)
            step = min(step, 5.0 / gnorm)

        else:
            w, cm = compute_lse_weights(c, beta)
            grad = lse_gradient_from_weights(w, a)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (beta * 1.5)
            step = min(step, 3.0 / gnorm)

        # Line search
        accepted = False
        for _ in range(10):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - (0.001 / max(gnorm, 1e-10)) * grad, S)

        # Noise
        if stuck_count > 400:
            noise_scale = 0.02 * S / np.sqrt(n) * (1.0 - progress * 0.5)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck_count = 0

        if verbose and it % 1000 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    mixed: it={it}, t={elapsed:.0f}s, mode={mode}, obj={obj:.6f}")

    return best_a, best_max


def generate_init_symmetric(n, rng, base=None):
    """Generate initial symmetric params."""
    half = n // 2
    if base is not None and len(base) >= 4:
        # Interpolate and symmetrize
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a_full = np.interp(t_new, t_old, base)
        a_full = np.maximum(a_full, 0.0)
        b = a_full[:half].copy()
        # Enforce symmetry
        a_sym = expand_symmetric(b, n)
        # Re-extract (should be same)
        return b
    # Default: raised cosine
    tt = np.linspace(0, 1, half, endpoint=False)
    b = 0.5 * (1 + np.cos(2 * np.pi * tt / 2.0)) + 0.05
    return b


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: symmetric Nesterov LSE, asymmetric refinement,
    multi-length exploration, final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0.01 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.6f} ({msg}) ***")

    # === PHASE 1: Symmetric Nesterov on provided length ===
    if a_provided is not None:
        n1 = len(a_provided)
        # Make even
        if n1 % 2 != 0:
            n1 += 1
            a_provided = np.concatenate([a_provided, [a_provided[0]]])

        S1 = np.sqrt(2.0 * n1)
        b1 = extract_symmetric_params(make_symmetric(a_provided * (S1 / a_provided.sum())))
        b1 = project_simplex(b1, S1 / 2.0)

        t1 = min(600, (deadline - time.monotonic()) * 0.30)
        print(f"\nPhase 1: Symmetric Nesterov, n={n1}, time={t1:.0f}s")
        b_opt, best_max = optimize_symmetric_nesterov(b1, n1, S1, t1, rng, verbose=True)
        a_opt = expand_symmetric(b_opt, n1)
        update_global(a_opt, "Phase 1 symmetric")

        # === PHASE 2: Asymmetric refinement ===
        remaining = deadline - time.monotonic()
        t2 = min(400, remaining * 0.25)
        if t2 > 30:
            print(f"\nPhase 2: Asymmetric Nesterov, n={n1}, time={t2:.0f}s")
            a_opt2, _ = optimize_asymmetric_nesterov(a_opt, S1, t2, rng, verbose=True)
            update_global(a_opt2, "Phase 2 asymmetric")

    # === PHASE 3: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    lengths_to_try = [8192, 16384, 32768]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        if global_a is not None and len(global_a) == n_try:
            continue

        S = np.sqrt(2.0 * n_try)
        t_alloc = min(300, (deadline - time.monotonic()) * 0.20)
        if t_alloc < 40:
            continue

        print(f"\nPhase 3: New length n={n_try}, time={t_alloc:.0f}s")

        # Initialize from best if available
        if global_a is not None:
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_init = np.interp(t_new, t_old, global_a)
            a_init = np.maximum(a_init, 0.0)
        else:
            tt = np.linspace(0, 1, n_try, endpoint=False)
            a_init = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1

        a_init = a_init * (S / a_init.sum())

        # Symmetric optimization
        b_init = extract_symmetric_params(make_symmetric(a_init))
        b_init = project_simplex(b_init, S / 2.0)

        t_sym = t_alloc * 0.5
        b_res, _ = optimize_symmetric_nesterov(b_init, n_try, S, t_sym, rng, verbose=False)
        a_res = expand_symmetric(b_res, n_try)
        update_global(a_res, f"Phase 3 n={n_try} sym")

        # Quick asymmetric polish
        rem = deadline - time.monotonic()
        t_asym = min(100, rem * 0.3)
        if t_asym > 20:
            a_res2, _ = optimize_asymmetric_nesterov(a_res, S, t_asym, rng, verbose=False)
            update_global(a_res2, f"Phase 3 n={n_try} asym")

    # === PHASE 4: Intensive final polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_curr = global_a * (S_best / global_a.sum())

        print(f"\nPhase 4: Final polish, n={n_best}, remaining={remaining:.0f}s")

        # Alternate between methods
        n_rounds = 3
        t_per = (remaining - 20) / n_rounds

        for ri in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            t_phase = min(t_per, deadline - time.monotonic() - 10)
            if t_phase < 10:
                break

            if ri % 2 == 0:
                # LSE polish
                a_curr, _ = optimize_asymmetric_nesterov(a_curr, S_best, t_phase, rng, verbose=False)
            else:
                # Mixed
                a_curr, _ = optimize_mixed_phase(a_curr, S_best, t_phase, rng, verbose=False)
            update_global(a_curr, f"Phase 4 round {ri}")

    # === PHASE 5: Random restarts from best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        n_restarts = max(1, int(remaining / 60))
        t_per = (remaining - 15) / max(1, n_restarts)

        print(f"\nPhase 5: {n_restarts} perturbations, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Structured perturbation
            noise_scale = 0.015 * S_best / np.sqrt(n_best)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n_best), S_best)

            t_alloc = min(t_per, deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break

            a_res, _ = optimize_asymmetric_nesterov(a_pert, S_best, t_alloc, rng, verbose=False)
            update_global(a_res, f"Phase 5 restart {ri}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
