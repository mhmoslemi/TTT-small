def project_simplex(v, S):
    v=np.asarray(v,dtype=np.float64)
    n=v.size
    if n==0: return v
    s=v.sum()
    if s <= 1e-15:
       return np.full(n, S/n, dtype=np.float64)
    # fast path
    vclip = np.maximum(v,0.0)
    if np.min(v) >= -1e-10 and abs(s-S) <= 1e-8*max(1.0,S):
       out=vclip
       ss=out.sum()
       if ss>0:
          out=out*(S/ss)
          return out
    # full projection
    w=np.sort(v)[::-1]
    cs=np.cumsum(w)
    idx=np.arange(1,n+1,dtype=np.float64)
    theta=(cs-S)/idx
    valid=theta <= w
    if np.any(valid):
       rho=int(np.max(np.where(valid)[0]))+1
    else:
       rho=n
    theta=(cs[rho-1]-S)/rho
    out=np.maximum(v-theta,0.0)
    ss=out.sum()
    if ss < 1e-15:
       return np.full(n,S/n)
    return out*(S/ss)
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / sum(a)^2

The program works by normalizing sum(a)=sqrt(2n), so the score is just
max(self-convolution).  It combines:
  1) a smooth LSE-FISTA bulk optimizer,
  2) a true active-set subgradient line-search polish,
  3) a linearized active-set LP direction (minimax over active lags),
  4) occasional L-BFGS refinement in log-softmax parameters,
  5) optional low-resolution-to-high-resolution upsampling.

No filesystem/network is used.  The required entry point is propose_candidate().
"""

import math
import time

import numpy as np

from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize


def full_fft_size(n):
    """Next power of two at least 2*n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def normalize_sum(a, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return a
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(a.shape, S / max(1, a.size), dtype=np.float64)
    return a * (S / s)


def convolve_self(a, fsize=None):
    """Full self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if fsize is None:
        fsize = full_fft_size(n)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def evaluate_internal(a):
    """Return (score, normalized_a) for the internal objective."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf, a
    S = math.sqrt(2.0 * n)
    an = normalize_sum(a, S)
    c = convolve_self(an)
    val = float(c.max()) if c.size else np.inf
    return val, an


def consider_candidate(state, a, name):
    """Update state if a is a better internal candidate."""
    if a is None:
        return
    val, an = evaluate_internal(np.asarray(a, dtype=np.float64))
    if np.isfinite(val) and val < state["value"] - 1e-12:
        state["value"] = float(val)
        state["a"] = an.copy()
        print("new best %.10f n=%d source=%s" % (val, an.size, name), flush=True)


def project_simplex(v, S):
    """
    Euclidean projection onto {x >= 0, sum x = S}.
    Fast path is used when v is already almost feasible.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    s = float(v.sum())
    if s <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)

    # Fast path: already non-negative (up to tiny error) and sum close.
    if np.min(v) >= -1e-11 and abs(s - S) <= max(1e-9 * max(1.0, S), 1e-9):
        out = np.maximum(v, 0.0)
        ss = float(out.sum())
        if ss > 1e-15:
            out *= S / ss
            return out

    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cs - S) / idx
    valid = theta <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cs[rho - 1] - S) / rho
    out = np.maximum(v - theta, 0.0)
    ss = float(out.sum())
    if ss <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    return out * (S / ss)


def correlate_weights(a, lam, fsize):
    """
    r_j = sum_k lam_k a_{k-j} (valid indices only), using FFT.
    This is one half of the gradient of sum_k lam_k * c_k.
    """
    n = a.size
    A = fft(a, fsize)
    W = fft(lam, fsize)
    r = np.real(ifft(W * np.conj(A)))[:n]
    return r


def make_active_weights(c, M, active_frac, top_count=16):
    """
    Build probability weights over self-convolution lags.
    It emphasizes both the hardest top lags and lags close to the maximum.
    """
    nconv = c.size
    delta = max(1e-10, active_frac * M)
    lam = np.zeros(nconv, dtype=np.float64)

    thr_idx = np.where(c >= M - delta)[0]
    if thr_idx.size:
        baseline = M - delta
        lam[thr_idx] = np.maximum(0.0, c[thr_idx] - baseline)

    top_count = int(min(max(1, top_count), nconv))
    kth = top_count - 1
    if kth > 0:
        top_idx = np.argpartition(c, kth)[-top_count:]
    else:
        top_idx = np.asarray([int(np.argmax(c))], dtype=np.int64)
    lam[top_idx] += 1.0

    tot = float(lam.sum())
    if not np.isfinite(tot) or tot <= 1e-15:
        lam.fill(0.0)
        lam[int(np.argmax(c))] = 1.0
        tot = 1.0
    return lam / tot


def lse_weights(c, beta):
    """Softmax weights for log-sum-exp smoothing and its value."""
    M = float(c.max())
    if beta <= 0.0:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w, M

    z = beta * (c - M)
    #严防 overflow underflow: z <= 0, exp is safe for underflow.
    e = np.exp(z)
    tot = float(e.sum())
    if not np.isfinite(tot) or tot <= 1e-300:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w, M
    w = e / tot
    val = M + math.log(tot) / beta
    return w, val


def try_gradient_steps(x, grad, S, fsize, alphas, current_M):
    """
    Try several projected gradient steps and return the best feasible point.
    Returns (best_a, best_M, improved).
    """
    n = x.size
    best_a = x
    best_M = current_M

    for alpha in alphas:
        if not np.isfinite(alpha) or alpha <= 0.0:
            continue
        trial = x - alpha * grad
        if not np.all(np.isfinite(trial)):
            continue
        trial = project_simplex(trial, S)
        c = convolve_self(trial, fsize)
        Mnew = float(c.max())
        if not np.isfinite(Mnew):
            continue
        if Mnew < best_M:
            best_M = Mnew
            best_a = trial

    improved = (np.isfinite(best_M) and best_M < current_M - 1e-13)
    return best_a, best_M, improved


def active_gradient(a, lam, fsize):
    """
    Gradient of active-weighted self-convolution, centered onto the tangent
    plane sum(a)=constant.
    """
    r = correlate_weights(a, lam, fsize)
    g = 2.0 * r
    g -= float(g.mean())
    return g


def apply_direction(x, d, S, fsize, current_M):
    """
    Apply a feasible centered direction d with backtracking on true max.
    """
    d = np.asarray(d, dtype=np.float64)
    d = d - float(d.mean())
    n = x.size

    neg = d < -1e-15
    if np.any(neg):
        max_alpha = float(np.min(-x[neg] / d[neg]))
    else:
        max_alpha = 1.0
    if not np.isfinite(max_alpha) or max_alpha <= 0.0:
        max_alpha = 1e-12

    best_a = x
    best_M = current_M

    for fac in (1.0, 0.75, 0.55, 0.40, 0.28, 0.18, 0.10, 0.05):
        alpha = fac * max_alpha
        if alpha > 1.0:
            alpha = 1.0
        if alpha <= 0.0 or not np.isfinite(alpha):
            continue
        y = x + alpha * d
        if np.any(y < -1e-10):
            continue
        y = np.clip(y, 0.0, 1000.0)
        s = float(y.sum())
        if s <= 1e-15:
            continue
        y *= S / s
        c = convolve_self(y, fsize)
        Mnew = float(c.max())
        if np.isfinite(Mnew) and Mnew < best_M:
            best_M = Mnew
            best_a = y

    return best_a, best_M


def lp_active_direction(a, c, M, S, max_active, step_limit, time_limit=5.0):
    """
    Linearize selected active self-convolution lags around a and solve

        min  t
        s.t. 2*L_k b - t <= c_k      for active k
             sum b = S, 0 <= b <= S

    where L_k b = sum_j a_{k-j} b_j.  Return a feasible descent direction b-a.
    """
    n = a.size
    conv_len = c.size
    if n < 2 or conv_len < 2:
        return None

    max_active = int(min(max(8, max_active), 256, conv_len))
    delta_frac = 0.06
    delta = max(1e-10, delta_frac * M)

    active_bool = np.zeros(conv_len, dtype=bool)
    thr = np.where(c >= M - delta)[0]
    if thr.size:
        active_bool[thr] = True

    top_count = min(32, conv_len)
    kth = top_count - 1
    if kth > 0:
        top_idx = np.argpartition(c, kth)[-top_count:]
    else:
        top_idx = np.asarray([int(np.argmax(c))], dtype=np.int64)
    active_bool[top_idx] = True

    active_idx = np.where(active_bool)[0]
    if active_idx.size == 0:
        active_idx = top_idx
    if active_idx.size > max_active:
        order = np.argsort(c[active_idx])[::-1]
        active_idx = active_idx[order[:max_active]]

    K = active_idx.size
    A = np.zeros((K, n + 1), dtype=np.float64)
    bub = np.zeros(K, dtype=np.float64)

    for i, k in enumerate(active_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            # row[j] = 2*a[k-j], for j=lo..hi
            A[i, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        A[i, n] = -1.0
        bub[i] = c[k]

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([S], dtype=np.float64)

    U = max(S, 1.0)
    bounds = np.empty((n + 1, 2), dtype=np.float64)
    bounds[:n, 0] = 0.0
    bounds[:n, 1] = U
    bounds[n, 0] = -100.0
    bounds[n, 1] = 1.0e6

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=bub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.2, float(time_limit))}
        )
    except Exception:
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=bub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="revised simplex"
            )
        except Exception:
            return None

    if res is None or not getattr(res, "success", False):
        return None
    if not hasattr(res, "x") or res.x is None:
        return None

    b = np.asarray(res.x[:n], dtype=np.float64).copy()
    if not np.all(np.isfinite(b)):
        return None
    b = np.clip(b, 0.0, 1000.0)
    s = float(b.sum())
    if s <= 1e-15:
        return None
    b *= S / s

    d = b - a
    d -= float(d.mean())
    norm = float(np.linalg.norm(d))
    if not np.isfinite(norm) or norm <= 1e-13:
        return None

    lim = float(step_limit) if step_limit is not None else 0.0
    if lim > 0.0 and norm > lim:
        d *= lim / norm
    return d


def lse_fista_opt(a, S, fsize, max_time, rng,
                  beta_start=50.0, beta_end=30000.0, beta_steps=180):
    """
    Smooth LSE-FISTA bulk optimizer with a true-objective check.
    """
    start = time.monotonic()
    n = a.size
    x = normalize_sum(a, S)
    y = x.copy()
    t_mom = 1.0
    base_alpha = 0.08

    best_x = x.copy()
    c0 = convolve_self(x, fsize)
    best_M = float(c0.max())

    print("  LSE-FISTA start %.10f (n=%d)" % (best_M, n), flush=True)

    it = 0
    while time.monotonic() - start < max_time:
        rem = max(0.0, max_time - (time.monotonic() - start))
        if rem < 2.0:
            break
        it += 1

        y = normalize_sum(y, S)
        c = convolve_self(y, fsize)
        M = float(c.max())
        if not np.isfinite(M):
            x = project_simplex(x + rng.normal(0.0, 0.002 * S / math.sqrt(n), n), S)
            y = x.copy()
            t_mom = 1.0
            continue

        if M < best_M - 1e-12:
            best_M = M
            best_x = y.copy()

        frac = min(1.0, float(it) / float(max(1, beta_steps)))
        beta = beta_start * (beta_end / beta_start) ** frac
        w, _ = lse_weights(c, beta)

        g = active_gradient(y, w, fsize)
        gnorm = float(np.linalg.norm(g))
        if not np.isfinite(gnorm) or gnorm <= 1e-14:
            x = project_simplex(x + rng.normal(0.0, 0.002 * S / math.sqrt(n), n), S)
            y = x.copy()
            t_mom = 1.0
            continue

        max_step_norm = 0.18 * S
        a0 = base_alpha
        if a0 * gnorm > max_step_norm:
            a0 = max_step_norm / gnorm

        alphas = [a0 * f for f in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.075)]
        new_a, new_M, ok = try_gradient_steps(y, g, S, fsize, alphas, M)

        if ok and new_M < M:
            accepted = new_a
            base_alpha = min(1.0, base_alpha * 1.12)
        else:
            accepted = x
            base_alpha = max(0.01, base_alpha * 0.75)

        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = accepted + ((t_mom - 1.0) / t_new) * (accepted - x)
        y = normalize_sum(y, S)
        x = accepted
        t_mom = t_new

    # Final projection / evaluation
    cfin = convolve_self(x, fsize)
    if float(cfin.max()) < best_M:
        best_M = float(cfin.max())
        best_x = x.copy()

    print("  LSE-FISTA end   %.10f" % best_M, flush=True)
    return best_x, best_M


def optimize_active(a, S, fsize, max_time, rng, alpha_init=0.05, lp_enabled=True):
    """
    True active-set subgradient polish, with occasional LP directions.
    """
    start = time.monotonic()
    n = a.size
    x = normalize_sum(a, S)
    c0 = convolve_self(x, fsize)
    best_M = float(c0.max())
    best_x = x.copy()

    print("  active start %.10f (n=%d)" % (best_M, n), flush=True)

    alpha = alpha_init
    active_frac = 0.02
    stag = 0
    itr = 0

    K_lp = 64
    max_K_lp = 128 if n > 16384 else 256
    lp_interval = 40
    next_lp = 70
    lp_fails = 0

    while time.monotonic() - start < max_time:
        rem = max(0.0, max_time - (time.monotonic() - start))
        if rem < 2.0:
            break
        itr += 1

        c = convolve_self(x, fsize)
        if not np.all(np.isfinite(c)):
            x = project_simplex(x + rng.normal(0.0, 0.002 * S / math.sqrt(n), n), S)
            continue
        M = float(c.max())
        if not np.isfinite(M):
            x = project_simplex(x + rng.normal(0.0, 0.002 * S / math.sqrt(n), n), S)
            continue

        improved = False
        if M < best_M - 1e-12:
            best_M = M
            best_x = x.copy()
            stag = 0
        else:
            stag += 1
            if stag > 70:
                active_frac = min(0.18, active_frac * 1.25)

        lam = make_active_weights(c, M, active_frac, 16)
        g = active_gradient(x, lam, fsize)
        gnorm = float(np.linalg.norm(g))

        if np.isfinite(gnorm) and gnorm > 1e-14:
            max_step_norm = 0.22 * S
            a0 = alpha
            if a0 * gnorm > max_step_norm:
                a0 = max_step_norm / gnorm
            alphas = [a0 * f for f in (1.0, 0.72, 0.52, 0.37, 0.25, 0.16, 0.085)]

            new_a, new_M, ok = try_gradient_steps(x, g, S, fsize, alphas, M)
            if ok and new_M < M:
                x = new_a
                if new_M < M * 0.999:
                    alpha = min(0.85, alpha * 1.18)
                else:
                    alpha = min(0.85, alpha * 1.02)
                active_frac = max(0.008, active_frac * 0.85)
                improved = True

        if not improved:
            alpha = max(0.004, alpha * 0.72)

            can_lp = (
                lp_enabled and
                stag >= 10 and
                itr >= next_lp and
                rem > 8.0
            )

            if can_lp:
                lt = min(5.0, rem * 0.10)
                kt = int(min(K_lp, max_K_lp))
                d = lp_active_direction(x, c, M, S, kt, 0.16 * S, lt)

                lp_ok = False
                if d is not None:
                    lp_a, lp_M = apply_direction(x, d, S, fsize, M)
                    if np.isfinite(lp_M) and lp_M < M - 1e-12:
                        x = lp_a
                        lp_ok = True
                        alpha = 0.12
                        active_frac = 0.015
                        K_lp = max(48, int(K_lp * 1.45))
                        lp_fails = 0
                        lp_interval = 22
                        improved = True
                        print("    LP improving M=%.10f at iter=%d K=%d" % (lp_M, itr, kt), flush=True)

                if not lp_ok:
                    lp_fails += 1
                    if lp_fails >= 4:
                        K_lp = max(32, int(K_lp * 0.75))
                        lp_fails = 0
                    lp_interval = min(140, int(lp_interval * 1.55))

                next_lp = itr + lp_interval

            if not improved:
                scale = 0.0018 * S / math.sqrt(n)
                if stag > 90:
                    scale *= min(3.0, 1.0 + stag / 250.0)
                x = project_simplex(x + rng.normal(0.0, scale, n), S)
                active_frac = min(0.22, active_frac * 1.12)
                if stag > 180:
                    alpha = max(0.01, alpha * 0.88)

    cfin = convolve_self(x, fsize)
    if float(cfin.max()) < best_M:
        best_M = float(cfin.max())
        best_x = x.copy()

    print("  active end   %.10f" % best_M, flush=True)
    return best_x, best_M


def objective_lse_score_z(u, n, fsize, beta):
    """LSE objective for log-softmax variables, scaled to the score."""
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    if u.size != n - 1:
        u2 = np.zeros(n - 1, dtype=np.float64)
        m = min(u.size, n - 1)
        u2[:m] = u[:m]
        u = u2

    z = np.empty(n, dtype=np.float64)
    z[0] = 0.0
    z[1:] = u
    if not np.all(np.isfinite(z)):
        return 1.0e100

    z -= float(z.max())
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return 1.0e100

    p = ez / s
    c = convolve_self(p, fsize)
    if not np.all(np.isfinite(c)):
        return 1.0e100
    cm = float(c.max())

    z2 = beta * (c - cm)
    e = np.exp(z2)
    tot = float(e.sum())
    if not np.isfinite(tot) or tot <= 1e-300:
        val = 2.0 * n * cm
    else:
        val = 2.0 * n * (cm + math.log(tot) / beta)

    if not np.isfinite(val):
        return 1.0e100
    return val


def jac_lse_score_z(u, n, fsize, beta):
    """Gradient of LSE objective for log-softmax variables."""
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    if u.size != n - 1:
        u2 = np.zeros(n - 1, dtype=np.float64)
        m = min(u.size, n - 1)
        u2[:m] = u[:m]
        u = u2

    z = np.empty(n, dtype=np.float64)
    z[0] = 0.0
    z[1:] = u
    if not np.all(np.isfinite(z)):
        return np.zeros(max(0, n - 1), dtype=np.float64)

    z -= float(z.max())
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.zeros(max(0, n - 1), dtype=np.float64)

    p = ez / s
    c = convolve_self(p, fsize)
    if not np.all(np.isfinite(c)):
        return np.zeros(max(0, n - 1), dtype=np.float64)
    cm = float(c.max())

    if beta <= 0.0:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        e = np.exp(beta * (c - cm))
        tot = float(e.sum())
        if not np.isfinite(tot) or tot <= 1e-300:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / tot

    r = correlate_weights(p, w, fsize)
    rbar = float(np.dot(p, r))
    grad = np.zeros(max(0, n - 1), dtype=np.float64)
    if n > 1:
        grad[:] = 4.0 * n * p[1:] * (r[1:] - rbar)
    grad[~np.isfinite(grad)] = 0.0
    return grad


def lbfgs_refine(a, max_time, beta=150000.0, maxiter=70, maxfun=260):
    """Short L-BFGS-B refinement in log-softmax coordinates."""
    start = time.monotonic()
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = math.sqrt(2.0 * n)
    a = normalize_sum(a, S)

    if n < 2 or max_time < 5.0:
        return a

    p = a / S
    p = p + 1e-12
    p /= float(p.sum())
    p0 = max(float(p[0]), 1e-30)

    x0 = np.log(np.maximum(p[1:] / p0, 1e-12))
    x0 = np.clip(x0, -60.0, 60.0)
    x0[~np.isfinite(x0)] = 0.0

    fsize = full_fft_size(n)

    try:
        res = minimize(
            objective_lse_score_z,
            x0,
            args=(n, fsize, beta),
            method="L-BFGS-B",
            jac=jac_lse_score_z,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 10,
                "disp": False
            }
        )
    except Exception:
        return a

    if hasattr(res, "x") and res.x is not None:
        z = np.empty(n, dtype=np.float64)
        z[0] = 0.0
        z[1:] = np.asarray(res.x, dtype=np.float64)
        if np.all(np.isfinite(z)):
            z -= float(z.max())
            ez = np.exp(z)
            s = float(ez.sum())
            if s > 1e-300:
                p2 = ez / s
                b = p2 * S
                b = np.clip(b, 0.0, 1000.0)
                return normalize_sum(b, S)

    return a


def make_fallback_inits(n, rng):
    """Family of diverse non-negative initial shapes."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    cands = []

    cands.append(np.ones(n, dtype=np.float64))

    for cf in (0.2, 0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.8):
        for wf in (0.25, 0.35, 0.5, 0.65, 0.8):
            x = (t - cf) / (0.5 * wf)
            a = np.zeros(n, dtype=np.float64)
            mask = np.abs(x) < 1.0
            a[mask] = 0.5 * (1.0 + np.cos(np.pi * x[mask]))
            a = np.maximum(a, 0.0)
            if a.sum() > 0:
                cands.append(a)

    for pp in (0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        cands.append(t ** pp)
        cands.append((1.0 - t) ** pp)
        cands.append(np.abs(t - 0.5) ** pp)
        cands.append(np.sin(np.pi * t) ** pp)

    for k in (1, 2, 3, 4):
        cands.append(0.5 + 0.5 * np.sin(np.pi * k * t))
        cands.append(0.5 + 0.5 * np.cos(np.pi * k * t))
        cands.append(np.abs(np.sin(np.pi * k * t)))

    for ctr in (0.35, 0.5, 0.65):
        for slope in (2.0, 4.0, 8.0, 16.0):
            a = 1.0 / (1.0 + np.exp(-slope * (t - ctr)))
            cands.append(a)
            cands.append(1.0 / (1.0 + np.exp(slope * (t - ctr))))

    for _ in range(16):
        mu1 = rng.uniform(0.1, 0.9)
        s1 = rng.uniform(0.05, 0.28)
        a1 = np.exp(-0.5 * ((t - mu1) / s1) ** 2)
        cands.append(a1)

        mu2 = rng.uniform(0.1, 0.9)
        s2 = rng.uniform(0.05, 0.25)
        a2 = a1 + 0.8 * np.exp(-0.5 * ((t - mu2) / s2) ** 2)
        cands.append(a2)

    return cands


def best_fallback_init(n, rng):
    """Pick the best cheap initialization from the family."""
    S = math.sqrt(2.0 * n)
    best = None
    best_score = np.inf

    for cand in make_fallback_inits(n, rng):
        a = np.asarray(cand, dtype=np.float64)
        if a.size != n:
            continue
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s <= 1e-15:
            continue
        a = a * (S / s)
        c = convolve_self(a)
        sc = float(c.max())
        if sc < best_score:
            best_score = sc
            best = a.copy()

    if best is None:
        best = np.full(n, S / n, dtype=np.float64)
    return best


def upsample_init(a, n_new):
    """Interpolate candidate to a new length and renormalize."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.full(n_new, math.sqrt(2.0 * n_new) / max(1, n_new), dtype=np.float64)
    if n_new <= 0:
        return a
    if n_new == n:
        return a.copy()
    t_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    t_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    b = np.interp(t_new, t_old, a)
    S = math.sqrt(2.0 * n_new)
    return normalize_sum(b, S)


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Required entry point.  Searches for a non-negative sequence with low score.
    """
    global_start = time.monotonic()
    total_budget = min(float(budget_s), 2400.0)
    safety = 60.0
    deadline = global_start + max(10.0, total_budget - safety)
    rng = np.random.default_rng(seed)

    state = {"value": np.inf, "a": None}

    # Load previous construction if present.
    raw = globals().get("height_sequence_1", None)
    arr = None
    if raw is not None:
        try:
            arr = np.asarray(raw, dtype=np.float64).reshape(-1)
            if arr.size >= 16:
                arr = np.clip(arr, 0.0, 1000.0)
        except Exception:
            arr = None

    if arr is not None and float(arr.sum()) > 1e-8:
        S0 = math.sqrt(2.0 * arr.size)
        a0 = normalize_sum(arr, S0)
        consider_candidate(state, a0, "loaded")
    else:
        n0 = 8192
        a0 = best_fallback_init(n0, rng)
        consider_candidate(state, a0, "fallback")

    cur = state["a"]
    if cur is None:
        n0 = 8192
        cur = best_fallback_init(n0, rng)
        consider_candidate(state, cur, "fallback2")

    ncur = len(cur)
    if ncur > 32768:
        cur = upsample_init(cur, 16384)
        consider_candidate(state, cur, "downsample-in")
    elif ncur < 512:
        target = min(8192, max(1024, 4 * ncur))
        cur = upsample_init(cur, target)
        consider_candidate(state, cur, "upsample-small")

    # Ensure at least one candidate is in state.
    if state["a"] is None:
        cur = normalize_sum(cur, math.sqrt(2.0 * len(cur)))
        consider_candidate(state, cur, "force")

    # Phase 1: smooth bulk refinement on the current length.
    cur = state["a"]
    n = cur.size
    if n <= 32768:
        S = math.sqrt(2.0 * n)
        fsize = full_fft_size(n)
        rem = max(0.0, deadline - time.monotonic())
        if rem > 120:
            t1 = min(0.25 * rem, 600.0)
            a1, _ = lse_fista_opt(cur, S, fsize, t1, rng,
                                  beta_start=80.0,
                                  beta_end=25000.0,
                                  beta_steps=160)
            consider_candidate(state, a1, "lse-fista")

    # Phase 2: main true active-set / LP polish.
    cur = state["a"]
    n = cur.size
    S = math.sqrt(2.0 * n)
    fsize = full_fft_size(n)
    rem = max(0.0, deadline - time.monotonic())
    if rem > 80:
        t2 = min(0.45 * rem, 1400.0)
        a2, _ = optimize_active(cur, S, fsize, t2, rng, alpha_init=0.05, lp_enabled=True)
        consider_candidate(state, a2, "active-main")

    # Phase 3: sharper active polish with a smaller initial step.
    cur = state["a"]
    n = cur.size
    S = math.sqrt(2.0 * n)
    fsize = full_fft_size(n)
    rem = max(0.0, deadline - time.monotonic())
    if rem > 120:
        t3 = min(0.30 * rem, 800.0)
        a3, _ = optimize_active(cur, S, fsize, t3, rng, alpha_init=0.018, lp_enabled=True)
        consider_candidate(state, a3, "active-sharp")

    # Phase 4: L-BFGS in log-softmax coordinates at a couple of beta values.
    cur = state["a"]
    n = cur.size
    if n <= 16384:
        rem = max(0.0, deadline - time.monotonic())
        if rem > 80:
            for beta in (120000.0, 300000.0):
                rem = max(0.0, deadline - time.monotonic())
                if rem < 45:
                    break
                tt = min(90.0, 0.4 * rem)
                a4 = lbfgs_refine(cur, tt, beta=beta, maxiter=60,
                                  maxfun=(260 if n <= 8192 else 170))
                consider_candidate(state, a4, "lbfgs-beta-%.0f" % beta)
                cur = state["a"] if state["a"] is not None else cur

    # Phase 5: If time remains and length is moderate, try one resolution doubling.
    cur = state["a"]
    n = cur.size
    rem = max(0.0, deadline - time.monotonic())
    if rem > 300 and n <= 16384:
        n_new = min(2 * n, 16384)
        if n_new > n:
            init = upsample_init(cur, n_new)
            consider_candidate(state, init, "upsample-init")

            cur = state["a"]
            n = cur.size
            if n == n_new:
                S = math.sqrt(2.0 * n)
                fsize = full_fft_size(n)
                rem = max(0.0, deadline - time.monotonic())
                if rem > 60:
                    tt = min(0.55 * rem, 700.0)
                    a5, _ = optimize_active(cur, S, fsize, tt, rng, alpha_init=0.045, lp_enabled=True)
                    consider_candidate(state, a5, "upsample-opt")

            # If the upsampling really helped, give the new length one extra polish.
            cur = state["a"]
            rem = max(0.0, deadline - time.monotonic())
            if cur is not None and len(cur) == n_new and rem > 80:
                S = math.sqrt(2.0 * len(cur))
                fsize = full_fft_size(len(cur))
                a6, _ = optimize_active(cur, S, fsize, min(rem * 0.5, 300.0),
                                        rng, alpha_init=0.02, lp_enabled=True)
                consider_candidate(state, a6, "upsample-polish")

    # Phase 6: final local polish on the globally best state.
    cur = state["a"]
    rem = max(0.0, deadline - time.monotonic())
    if cur is not None and rem > 25:
        n = cur.size
        S = math.sqrt(2.0 * n)
        fsize = full_fft_size(n)
        a7, _ = optimize_active(cur, S, fsize, min(rem * 0.7, 350.0),
                                rng, alpha_init=0.015, lp_enabled=True)
        consider_candidate(state, a7, "final-polish")

    final = state["a"]
    if final is None:
        nfb = 1024
        final = np.full(nfb, math.sqrt(2.0 * nfb) / nfb, dtype=np.float64)

    final = np.asarray(final, dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or float(final.sum()) < 0.01:
        nfb = 1024
        final = np.full(nfb, math.sqrt(2.0 * nfb) / nfb, dtype=np.float64)

    # Optional exact evaluator check against the problem's own function.
    try:
        test_list = [float(x) for x in final]
        ver = evaluate_sequence(test_list)
        if np.isfinite(ver):
            print("verify evaluate_sequence=%.10f internal=%.10f n=%d" %
                  (ver, state["value"], final.size), flush=True)
            if ver < state["value"] - 1e-12:
                pass  # internal already mirrors FFT exactly; nothing to do.
        else:
            print("verify FAILED, using safer uniform fallback", flush=True)
            nfb = 1024
            final = np.full(nfb, math.sqrt(2.0 * nfb) / nfb, dtype=np.float64)
    except Exception:
        pass

    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or float(final.sum()) < 0.01:
        nfb = 1024
        final = np.full(nfb, math.sqrt(2.0 * nfb) / nfb, dtype=np.float64)

    print("returning length=%d best_internal=%.10f elapsed=%.1fs" %
          (final.size, state["value"], time.monotonic() - global_start), flush=True)
    return [float(x) for x in final]
