"""
Minimize 2*n*max_conv/sum^2 using exact coordinate descent + gradient polishing.

Key innovations:
1. Exact 1D coordinate minimization: exploits the convex structure of f(x) = max_p c_p(x)
   for each coordinate, using ternary search with O(n) per evaluation.
2. Multi-peak gradient method: uses weighted subgradients from top-K peaks for
   coordinated multi-coordinate moves that escape coordinate-wise local minima.
3. Adaptive switching between coordinate descent and gradient steps based on
   improvement rate.
4. Spectral diversity: generates new starting points via phase randomization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def self_conv(a, N=None):
    """Compute self-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def coord_descent_pass(a, c, n, S, order=None, ternary_iters=15):

    One pass of coordinate descent. Updates a and c in place.
    Returns the peak after the pass.
    """
    if order is None:
        order = np.arange(n)
    
    for k in order:
        v = a[k]
        
        # Determine range of p where k participates
        p_lo = k  # since p-k >= 0 means p >= k
        p_hi = min(2 * n - 2, k + n - 1)  # p-k <= n-1 means p <= k+n-1
        
        if p_hi < p_lo:
            continue
        
        # Precompute for the affected range
        p_range = np.arange(p_lo, p_hi + 1)
        a_vals = a[p_range - k]  # a[p-k]
        c_vals = c[p_range].copy()
        
        # For ternary search, f(x) = max(c_other, max(base_i + slope_i*x for affected), quad_val)
        # base_i = c_p - 2*v*a[p-k], slope_i = 2*a[p-k]
        base = c_vals - 2.0 * v * a_vals
        slope = 2.0 * a_vals
        
        # Quadratic term: c_{2k}(x) = c_{2k} - v^2 + x^2
        has_quad = (2 * k >= 0) and (2 * k <= 2 * n - 2) and (p_lo <= 2 * k <= p_hi)
        if has_quad:
            quad_base = c[2 * k] - v * v

        if p_lo > 0:

        else:
            c_other = -np.inf
        if p_hi + 1 < 2 * n - 1:
            c_other2 = c[p_hi + 1:].max()
            if c_other2 > c_other:
                c_other = c_other2

        lo_x = 0.0
        hi_x = 3.0 * v if v > 0 else 1.0
        
        for _ in range(ternary_iters):
            m1 = lo_x + (hi
