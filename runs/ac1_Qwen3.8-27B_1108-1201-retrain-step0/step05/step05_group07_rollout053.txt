"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Approach: FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with 
peak-focused LSE smoothing, simulated annealing polishing, and structural 
perturbation exploration. Builds on the provided initial sequence.

import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_lse_grad_topk(a, c, top_idx, beta):
    """
    Compute gradient of LSE-smoothed max of convolution,
    focusing on top-k lags.
    """
    n = len(a)
    M = c.max()
    # Weight the top lags by their excess
    k = len(top_idx)
    excess = c[top_idx] - M  # all <= 0
    # Use softmax weighting
    scaled = beta * (c[top_idx] - M)  # all <= 0
    # Clip to avoid underflow
    scaled = np.maximum(scaled, -500)
    e = np.exp(scaled)
    w = e / e.sum()
    
    # Gradient: 2 * sum_k w_k * (a convolved with delta at lag k, restricted to [0,n-1])
    # The derivative of c[k] w.r.t. a[j] is a[k-j] (if k-j is in [0,n-1])
    # So grad[j] = 2 * sum_k w_k * a[k-j] (where k-j is valid)
    
    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = 2.0 * w[ii]
        # For lag kidx, the contribution to a[j] is a[kidx - j] where 0 <= kidx-j <= n-1
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += wi * a[indices]
    return grad

def compute_full_lse_grad(a, c, beta):
    """Compute gradient of LSE-smoothed max of full convolution."""
    n = len(a)
    M = c.max()
    scaled = beta * (c - M)
    scaled = np.maximum(scaled, -500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    # Use FFT to compute the weighted convolution
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fista_lse(a0, S, max_time, rng, beta_init=100.0, beta_rate=1.05, 
               topk=50, verbose=False):
    """
    FISTA optimization with LSE-smoothed max of convolution.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    z = a.copy()  # FISTA momentum variable
    theta = 1.0  # FISTA momentum parameter
    
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Adaptive beta: increase over time
        beta = min(50000.0, beta_init * (beta_rate ** it))
        
        # Compute gradient
        c = fft_convolve_self(z)
        cur_max = c.max()
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Top-k for gradient
        k = min(topk, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k]
        
        grad = compute_lse_grad_topk(z, c, top_idx, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size for FISTA: 1/L where L is Lipschitz constant
        # L for LSE max is approximately 2*beta (for the smoothed version)
        # But we also have the combinatorial structure
        step = 1.0 / (2.0 * beta * 1.0 + 1e-10)
        step = min(step, 5.0 / gnorm)
        
        # FISTA update
        z_new = project_simplex(z - step * grad, S)
        
        # Update theta
        theta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * theta * theta))
        z_next = z_new + ((theta - 1.0) / theta_new) * (z_new - a)
        z_next = project_simplex(z_next, S)
        
        a = z_new
        z = z_next
        theta = theta_new
        
        # Noise injection when stuck
        if stuck > 200:
            noise_scale = min(0.01, 0.001 * (stuck / 200.0)) * S / np.sqrt(n)
            z = project_simplex(z + rng.normal(0, noise_scale, n), S)
            a = z.copy()
            theta = 1.0
            stuck = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it={it}, t={elapsed:.0f}s, max_c={best_max:.8f}")
    
    return best_a, best_max

def simulated_annealing(a0, S, max_time, rng, verbose=False):
    """
    Simulated annealing with smart perturbations.
    """
    n = len(a0)
    a = a0.copy()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    T = 0.01  # Initial "temperature"
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        T = 0.01 * (1.0 - progress) ** 2 + 1e-6
        
        # Generate perturbation
        # Mix of strategies:
        # 1. Random sparse perturbation
        # 2. Gradient-informed perturbation
        # 3. Structural perturbation (ramp)
        strategy = rng.integers(0, 3)
        
        a_try = a.copy()
        
        if strategy == 0:
            # Random sparse perturbation
            n_perturb = rng.integers(1, max(2, n // 100))
            indices = rng.choice(n, size=n_perturb, replace=False)
            noise = rng.normal(0, 0.05 * S / np.sqrt(n), n_perturb)
            a_try[indices] += noise
        elif strategy == 1:
            # Gradient-informed: reduce at peak-contributing positions
            c = fft_convolve_self(a)
            peak_lag = int(np.argmax(c))
            j_start = max(0, peak_lag - n + 1)
            j_end = min(n - 1, peak_lag)
            if j_start <= j_end:
                # Reduce a at positions contributing to the peak
                reduction = rng.uniform(0.001, 0.01) * c[peak_lag] / max(j_end - j_start + 1, 1)
                a_try[j_start:j_end+1] -= reduction
        else:
            # Structural: add/remove a small ramp
            ramp = np.linspace(-1, 1, n) * rng.uniform(0.001, 0.005) * S / np.sqrt(n)
            a_try += ramp
        
        a_try = project_simplex(a_try, S)
        c_try = fft_convolve_self(a_try)
        cur_max = c_try.max()
        
        # Accept if better or with probability exp(-delta/T)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_try.copy()
        if cur_max < c.max() or rng.random() < np.exp(-(cur_max - c.max()) / max(T, 1e-10)):
            a = a_try
            c = c_try
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      SA it={it}, t={elapsed:.0f}s, best_max={best_max:.8f}, T={T:.6f}")
    
    return best_a, best_max
