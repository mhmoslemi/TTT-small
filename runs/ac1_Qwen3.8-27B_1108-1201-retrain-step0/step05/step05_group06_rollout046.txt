Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    if tot < 1e-300:
        return lse_val, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def run_fista_lse(a_init, max_time, rng, beta_init=10.0, beta_max=10000.0,
                  symmetric=False):
    """LSE FISTA with backtracking and optional symmetry."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    if symmetric:
        n_half = (n + 1) // 2
        a_half = (a[:n_half] + a[n-1-n_half:n-1-n_half+n_half][::-1]) / 2.0
        # Build full from half
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, a_half[-2::-1]])
        a = normalize_seq(a, S)
    a = normalize_seq(a, S)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        # Gradient

            break

        step = 1.0 / max(beta, 1.0)
        c_lr = 1e-4
        max_bt = 20
        bt_count = 0
        for bt in range(max_bt):
            x_new = y - step * grad
            if symmetric:
                if n % 2 == 0:
                    xh = (x_new[:n//2] + x_new[n-1-n//2:n-1-n//2+n//2][::-1]) / 2.0

            lse_new, _ = lse_value_and_grad(x_new, c_new, beta)
            lse_old, _ = lse_value_and_grad(y, c, beta)
            
            if lse_new <= lse_old - c_lr * step * np.dot(grad, x_new - y) + 1e-15:
                break
            step *= 0.5
            bt_count += 1
        else:
            # If backtracking fails, just use a small step
            x_new = project_simplex(y - step * grad, S)

        # Momentum update

                y = np.concatenate([yh, yh[-2::-1]])
        
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                if symmetric:
                    if n % 2 == 0:
                        yh = (y[:n//2] + y[n-1-n//2:n-1-n//2+n//2][::-1]) / 2.0

            stuck = 0
            last_improve = it

        if it % 5000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def coordinate_descent(a_init, max_time, rng, max_sweeps=200):

    Coordinate descent for minimizing max(conv(a)) with sum(a) fixed.
    Uses 1D golden section search for each coordinate.
    """

    S = a_init.sum()
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_conv = c.max()
    best_a = a.copy()
    sweep = 0
    
    # Precompute: for coordinate i, which convolution terms are affected
    # b_j = sum_k a_k * a_{j-k}
    # When a_i changes by delta:
    #   b_j changes by 2*delta*a_{j-i} for j != 2i (if 0 <= j-i < n)
    #   b_{2i} changes by (a_i+delta)^2 - a_i^2 = 2*a_i*delta + delta^2
    
    while time.monotonic() - start < max_time and sweep < max_sweeps:
        sweep += 1
        improved = False
        
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:

            # keeping sum(a) = S (so we adjust a_i and rescale others)
            # Actually, to keep it simple, just minimize max(b) with a_i free,
            # then renormalize at end of sweep.
            
            # Current b with a_i = 0:
            # b_j(0) = c_j - contribution from a_i
            # contribution from a_i to b_j:
            #   if j == 2i: a_i^2
            #   if 0 <= j-i < n and j != 2i: 2*a_i*a_{j-i}
            
            ai = a[i]
            if ai < 1e-15:
                # Try setting it to a small value
                a[i] = S / (n * 10)
                c_new = fft_convolve_self(a)
                if c_new.max() < best_conv:
                    best_conv = c_new.max()
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.0
                continue
            
            # For golden section, we need to evaluate max(b(a_i)) for various a_i
            # b_j(a_i') = c_j - 2*ai*a_{j-i} + 2*a_i'*a_{j-i} for j != 2i (if 0<=j-i<n)
            #                       - ai^2 + a_i'^2 for j = 2i
            # = c_j + 2*(a_i'-ai)*a_{j-i} for j != 2i
            # = c_{2i} + (a_i'^2 - ai^2) for j = 2i
            
            # Let's define the "baseline" convolution (with a_i = 0):
            # b_j_base = c_j - (contribution of a_i)
            # For j = 2i: b_{2i}_base = c_{2i} - ai^2
            # For j where 0<=j-i<n, j!=2i: b_j_base = c_j - 2*ai*a_{j-i}
            
            # Then b_j(a_i') = b_j_base + (contribution of a_i')
            # For j = 2i: b_j_base + a_i'^2
            # For j where 0<=j-i<n, j!=2i: b_j_base + 2*a_i'*a_{j-i}
            
            # The range for a_i' is [0, S] (since sum = S and all others >= 0)
            # Actually a_i' can be at most S - sum_{k!=i} a_k = S - (S - ai) = ai
            # So a_i' in [0, ai] if we're keeping others fixed... 
            # But wait, we want to allow a_i' to change freely and adjust others.
            # Simplest: just optimize a_i in [0, S] and renormalize.
            
            # Upper bound: S (all mass in one coordinate)
            # But that's extreme. Let's use [0, max(ai*2, S/n*4)] as range.
            lo = 0.0
            hi = max(ai * 2.0, 4.0 * S / n)
            
            def eval_1d(val):
                # Compute max of b when a_i = val
                # b_j = c_j - 2*ai*a_{j-i} (for j!=2i, 0<=j-i<n) 
                #          - ai^2 (for j=2i)

                if abs(delta) < 1e-15:
                    return c.max()

                j_end = min(2*n-2, i + n - 1)

                        b_new[j] = c[j] + delta * (val + ai)
                    else:
                        ji = j - i

            # Golden section search

            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_1d(c1)
            f2 = eval_1d(c2)

                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_1d(c2)
                
                if hi - lo < 1e-10 * S:
                    break
            
            opt_val = (lo + hi) / 2
            opt_f = eval_1d(opt_val)
            
            if opt_f < best_conv - 1e-12:
                # Apply update
                delta = opt_val - ai
                a[i] = opt_val
                # Update c
                j_start = max(0, i)
                j_end = min(2*n-2, i + n - 1)
                for j in range(j_start, j_end + 1):

                best_conv = max(c)
                best_a = a.copy()
                improved = True

        c = fft_convolve_self(a)

        if actual_max < best_conv:
            best_conv = actual_max
            best_a = a.copy()
        
        # Check time more frequently

            break
            
        # Convergence check

    return best_a


def make_arcsine_init(n):
    """Arcsine-like distribution: high at ends, low in middle."""
    tt = np.linspace(0, 1, n, endpoint=False)
    eps = 1e-6
    return 1.0 / np.sqrt(tt * (1 - tt) + eps)


def make_fourier_init(n, K=5, rng=None):
    """Initialize with a few Fourier modes."""
    if rng is None:
        rng = np.random.default_rng(42)
    tt = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, K + 1):
        c_k = rng.normal(0, 0.3 / k)
        phi_k = rng.uniform(0, 2 * np.pi)
        a += c_k * np.cos(2 * np.pi * k * tt / n + phi_k)
    a = np.maximum(a, 0.1)
    return a


def make_symmetric_init(n, shape='arcsine'):
    """Create a symmetric initialization."""
    if shape == 'arcsine':
        tt = np.linspace(0, 1, n, endpoint=False)
        eps = 1e-6
        a = 1.0 / np.sqrt(np.minimum(tt, 1-tt) * np.minimum(tt, 1-tt) + eps)
        # Make symmetric
        a = (a + a[::-1]) / 2
    elif shape == 'gaussian':
        center = (n - 1) / 2.0
        tt = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((tt - center) / sigma) ** 2)
        a = (a + a[::-1]) / 2
    elif shape == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    return a


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
