def solve_lp_direction(f, M):
    n = len(f)
    g = cp.Variable(n)
    
    # Convolution constraints: (f*g)[k] <= M for k=0,...,2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] where g[j]=0 for j<0 or j>=n
    # For a given k: sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[i] * g[k-i] <= M

    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # So (C @ g)[k] = (f*g)[k]
    
    # This is a Toeplitz-like matrix

    # Alternative: use cp.conv or build the matrix as sparse
    import scipy.sparse as sp
    
    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(k, n-1) + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
