from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LSE FISTA warmup + extensive coordinate descent with exact 1D optimization
+ high-beta LSE polishing. Coordinate descent uses ternary search per coordinate,
exploiting the structure that convolution changes are rank-1 (plus one quadratic term).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, max_time, rng, beta_init=5.0, beta_rate=0.005,
                  beta_max=10000.0):
    """LSE FISTA with adaptive beta and momentum restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 300:
            beta = beta_init * (1.0 + beta_rate * it)
        elif stuck < 2000:
            beta = min(beta_max, beta * 1.008)
        else:
            beta = min(beta_max, beta * 1.02)
        beta = min(beta, beta_max)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / max(beta, 1.0)
        L_bound = 2.0 * beta * np.max(a) + 2.0 * np.max(a)
        step = min(step, 0.7 / max(L_bound, 1.0))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 8000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def coord_descent_pass(a, b_full, S, n, rng, time_limit=None):
    """
    One or more passes of coordinate descent.
    a: current sequence (modified in place)
    b_full: current full convolution (modified in place)
    S: current sum (returns updated S)
    Returns updated (a, b_full, S, best_obj).
    """
    best_obj = 2.0 * n * np.max(b_full) / (S * S)
    start_time = time.monotonic()
    passes_done = 0

    while True:
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            break
        passes_done += 1

        perm = rng.permutation(n)
        for j in perm:
            if time_limit is not None and (j % 512 == 0):
                if time.monotonic() - start_time > time_limit:
                    break

            aj = a[j]

            cur_max_b = np.max(b_full)
            cur_obj_val = 2.0 * n * cur_max_b / (S * S)

                lo = 0.0
                hi = S * 0.02
            else:
                lo = -aj * 0.8
                hi = aj * 0.8

            best_g = cur_obj_val

            # Sample points logarithmically spaced

            if lo < 0:
                test_points.extend([lo * f for f in [0.25, 0.5, 0.75, 1.0]])

            for t in test_points:
                if t == 0.0:
                    g_val = cur_obj_val
                else:
                    S_new = S + t
                    if S_new < 1e-10:
                        continue
                    # Compute max of b_full + t*a_slice (placed at [k_start:k_end]) + t^2*E
                    # The changed region is [k_start, k_end)
                    changed = b_full[k_start:k_end] + t * a_slice
                    if has_E and k_start <= e_pos < k_end:
                        changed[e_pos - k_start] += t * t
                    max_changed = np.max(changed)
                    # Max of unchanged region
                    if k_start > 0:
                        max_unchanged_left = np.max(b_full[:k_start])
                    else:
                        max_unchanged_left = -np.inf
                    if k_end < 2 * n - 1:
                        max_unchanged_right = np.max(b_full[k_end:])
                    else:
                        max_unchanged_right = -np.inf
                    max_unchanged = max(max_unchanged_left, max_unchanged_right)
                    b_new_max = max(max_changed, max_unchanged)
                    g_val = 2.0 * n * b_new_max / (S_new * S_new)

                if g_val < best_g - 1e-13:
                    best_g = g_val
                    best_t = t

            if best_t != 0.0:
                spread = max(abs(best_t) * 0.6, aj * 0.15, S * 5e-5)
                lo2 = max(lo, best_t - spread)
                hi2 = min(hi, best_t + spread)

                # Precompute unchanged max
                if k_start > 0:
                    max_unch_left = np.max(b_full[:k_start])
                else:
                    max_unch_left = -np.inf
                if k_end < 2 * n - 1:
                    max_unch_right = np.max(b_full[k_end:])
                else:
                    max_unch_right = -np.inf
                max_unchanged = max(max_unch_left, max_unch_right)

                    if S1 > 1e-10:

                    else:

def run_coord_descent(a_init, max_time, rng):
    """Run coordinate descent for a given time budget."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    b_full = fft_convolve_self(a)
    S = a.sum()

    start = time.monotonic()
    best_obj = 2.0 * n * np.max(b_full) / (S * S)
    best_a = a.copy()

    # Run multiple passes with time budget per pass
    pass_budget = min(60.0, max_time * 0.05)
    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        bt = min(pass_budget, remaining)
        a, b_full, S, obj = coord_descent_pass(a, b_full, S, n, rng, time_limit=bt)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        # Gradually reduce pass budget (diminishing returns)
        pass_budget = max(10.0, pass_budget * 0.9)

    return best_a, best_obj


def make_raised_cosine_init(n):
    """Raised cosine (Hann window) initialization."""
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * tt / (n - 1))


def make_blackman_init(n):
    """Blackman window initialization."""
    tt = np.arange(n, dtype=np.float64)
    N = n - 1
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = np.ones(n)
    for i in range(n):
        t = i / N if N > 0 else 0
        w[i] = a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t)
    w = np.maximum(w, 0.0)
    return w


def make_symmetric_osc_init(n, freq=3.0):
    """Symmetric oscillatory non-negative initialization."""
    tt = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = np.ones(n)
    for i in range(n):
        x = (tt[i] - mid) / mid if mid > 0 else 0
        w[i] = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x) * np.cos(np.pi * x)
    return np.maximum(w, 0.01)

    inits.append(normalize_seq(make_symmetric_osc_init(n, 4.0), S))
    inits.append(normalize_seq(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n), S))

    # Quick evaluation
    print("=== Initialization ===")
    best_init = None
    best_init_obj = np.inf
    for i, a_init in enumerate(inits):
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  Init %d: obj=%.6f" % (i, obj))

    print("Best init: obj=%.6f" % best_init_obj)

    # === Phase 1: LSE FISTA (moderate beta) ===
    remaining = deadline - time.monotonic()
    p1_time = min(120.0, remaining * 0.15)
    if p1_time > 10:
        print("\n=== Phase 1: LSE FISTA (%.0fs) ===" % p1_time)
        t1 = time.monotonic()
        a_lse, _ = run_lse_fista(best_init, p1_time, rng,
                                  beta_init=3.0, beta_rate=0.01, beta_max=2000.0)
        obj_lse = compute_obj(a_lse)
        print("  LSE: obj=%.6f (t=%.1fs)" % (obj_lse, time.monotonic() - t1))
        if obj_lse < global_best_obj:
            global_best_obj = obj_lse
            global_best_a = a_lse.copy()

    # === Phase 2: Coordinate Descent (main) ===
    remaining = deadline - time.monotonic()
    p2_time = min(1200.0, remaining * 0.55)
    if p2_time > 30:
        print("\n=== Phase 2: Coordinate Descent (%.0fs) ===" % p2_time)
        t2 = time.monotonic()
        a_cd, obj_cd = run_coord_descent(global_best_a, p2_time, rng)
        print("  CD: obj=%.6f (t=%.1fs)" % (obj_cd, time.monotonic() - t2))
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

    # === Phase 3: High-beta LSE polish ===
    remaining = deadline - time.monotonic()
    p3_time = min(500.0, remaining * 0.30)
    if p3_time > 20:
        print("\n=== Phase 3: High-beta LSE (%.0fs) ===" % p3_time)
        t3 = time.monotonic()
        a_lse2, _ = run_lse_fista(global_best_a, p3_time, rng,
                                   beta_init=50.0, beta_rate=0.003, beta_max=50000.0)
        obj_lse2 = compute_obj(a_lse2)
        print("  LSE-high: obj=%.6f (t=%.1fs)" % (obj_lse2, time.monotonic() - t3))
        if obj_lse2 < global_best_obj:
            global_best_obj = obj_lse2
            global_best_a = a_lse2.copy()

    # === Phase 4: Final Coordinate Descent ===
    remaining = deadline - time.monotonic()
    p4_time = min(300.0, remaining * 0.60)
    if p4_time > 20:
        print("\n=== Phase 4: Final CD (%.0fs) ===" % p4_time)
        t4 = time.monotonic()
        a_cd2, obj_cd2 = run_coord_descent(global_best_a, p4_time, rng)
        print("  CD-final: obj=%.6f (t=%.1fs)" % (obj_cd2, time.monotonic() - t4))
        if obj_cd2 < global_best_obj:
            global_best_obj = obj_cd2
            global_best_a = a_cd2.copy()

    # === Phase 5: Very high beta LSE ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_time = remaining * 0.7
        print("\n=== Phase 5: Very high beta LSE (%.0fs) ===" % p5_time)
        t5 = time.monotonic()
        a_lse3, _ = run_lse_fista(global_best_a, p5_time, rng,
                                   beta_init=200.0, beta_rate=0.001, beta_max=100000.0)
        obj_lse3 = compute_obj(a_lse3)
        print("  LSE-vhigh: obj=%.6f (t=%.1fs)" % (obj_lse3, time.monotonic() - t5))
        if obj_lse3 < global_best_obj:
            global_best_obj = obj_lse3
            global_best_a = a_lse3.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
