def fft_convolve_self(a):

    size = 2 * n - 1

    A = fft(a, fsize)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)

    else:

Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. FISTA with LSE for coarse optimization
2. Active-set Newton method (CG-based) for KKT refinement
3. L-BFGS-B polish
4. Multi-active-set exploration

import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def conv_gradient(y, w):
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def solve_cg_ones(w, n, max_iters=80, tol=1e-12):
    """Solve H*z = 1 where H*v = 2*conv_gradient(v, w)."""
    one = np.ones(n)
    z = np.zeros(n)
    r = one.copy()
    p = r.copy()
    r0 = r @ r
    if r0 < 1e-30:
        return one / (2.0 * n)  # Fallback: uniform
    for _ in range(max_iters):
        Hp = conv_gradient(p, w)  # This is 2*P_sum*p already
        pH = p @ Hp
        if pH < 1e-30:
            # Fallback
            alpha = 1.0 / (2.0 * n)
            z += alpha * p
            break
        alpha = r0 / pH
        z += alpha * p
        r -= alpha * Hp
        r1 = r @ r
        if np.sqrt(r1) < tol:
            break
        beta = r1 / r0
        p = r + beta * p
        r0 = r1
    return z
