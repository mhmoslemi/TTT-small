import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def exact_subgradient(a, c, tol=1e-10):
    """Exact subgradient of max(c) - uniform over argmax lags."""
    n = len(a)
    cm = c.max()
    # Find lags within tol of max
    active = np.where(c >= cm - tol)[0]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def soft_subgradient(a, c, beta=100.0):
    """Soft subgradient using top-k lags with softmax weighting."""
    n = len(a)
    cm = c.max()
    # Use top lags within a threshold
    threshold = cm - 1.0 / beta
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = [np.argmax(c)]
    # Weight by how close to max
    weights = np.exp(beta * (c[active] - cm))
    weights /= weights.sum()
    w = np.zeros(len(c))
    w[active] = weights
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_fista_lse(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200,
                  momentum_restart_gap=300, noise_decay=50000.0):
    """FISTA with LSE smoothing and adaptive beta."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it

        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / (noise_decay / max(1, n // 1000)))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

        # Adaptive step size
        stagnant = it - last_improve_it
        if stagnant < 50:

            step = 0.001

        # Backtracking
        base_obj = cur_max

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        a = x_new
        t_mom = t_new

        # Restart on stagnation

    c_final = fft_convolve_self(a)

        best = c_final.max()

def run_active_set_lp(a_current, n, S, max_time, rng, n_active_candidates=50):
    """
    Active set method: identify active lags, solve for optimal weights,
    then use those weights to guide gradient descent.
    """
    a = normalize_sum(a_current, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Identify active lags (top n_active_candidates)
        top_lags = np.argsort(c)[::-1][:n_active_candidates]
        
        # Build the constraint: sum of a[i]*a[k-i] over active lags
        # We want to minimize the maximum, so we equalize them
        # Use a "water-filling" approach: increase lags below max, decrease those above
        
        # Compute gradient as average over top lags (active set approximation)
        w = np.zeros(len(c))
        # Weight inversely proportional to how far below max
        margins = cur_max - c[top_lags]
        w[top_lags] = 1.0 / (margins + 0.01)
        w /= w.sum()
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)

                break

        else:
            a = project_simplex(a - step * grad, S)

def peak_shaving(a, n, S, n_peaks=10, n_steps=200, lr_scale=0.005):
    """
    Direct peak shaving: for the top lag, reduce the products a[i]*a[k-i]
    that contribute most.
    """
    a = a.copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.999)
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a

def generate_inits(n, rng):

    inits = []

            x = (t - cf) / width
            a = np.ones(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            a[~mask] = 0.0
            inits.append(np.maximum(a, 1e-10))

    # Gaussian bumps
    for sigma in [0.08, 0.12, 0.15, 0.2]:
        for mu in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Power law (asymmetric)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = t ** p
        inits.append(np.maximum(a, 1e-10))
        a = (1 - t) ** p
        inits.append(np.maximum(a, 1e-10))

    # Sine-based
    for k in [1, 2, 3]:
        a = np.maximum(0, np.sin(np.pi * k * t) * 0.5 + 0.5)
        inits.append(a)

    # Triangle (asymmetric)
    for peak in [0.3, 0.4, 0.5, 0.6]:
        a = np.where(t < peak, t / peak, (1 - t) / (1 - peak))
        inits.append(np.maximum(a, 1e-10))

    return inits

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization with:
1) LSE-FISTA with careful numerical handling
2) Exact subgradient with Nesterov acceleration  
3) Active-set equalization (dual-guided)
4) Coordinate descent for final refinement
5) Multi-n exploration
Key: avoid closures, all helpers top-level.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

# Global time tracking
_g_deadline = 0.0
_g_best_val = np.inf
_g_best_a = None


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def soft_subgradient(a, c, beta=200.0):
    """Soft subgradient: weighted average over near-max lags."""
    n = len(a)
    cm = c.max()
    # Include lags within 1/beta of max
    threshold = cm - 2.0 / beta
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Softmax weighting over active
    wvals = np.exp(beta * (c[active] - cm))
    wvals = wvals / wvals.sum()
    w = np.zeros(len(c))
    w[active] = wvals
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_gradient_fft(a, w):
    """Compute gradient given weight vector w over lags."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_lse(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200,
                  momentum_restart_gap=300, noise_decay=50000.0):
    """FISTA with LSE smoothing and adaptive beta schedule."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it

        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / max(1, noise_decay / n))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.4

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_subgrad_nesterov(a_init, n, S, max_time, rng, beta_soft=500.0):

def run_equalize_lags(a_init, n, S, max_time, rng, n_lags=10):
    """
    Active-set equalization: try to make the top n_lags autocorrelation values equal.
    Uses the dual variables (weights) from the KKT conditions.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Get top lags
        top = np.argsort(c)[::-1][:n_lags]
        top_vals = c[top]
        top_max = top_vals[0]

        # Weights: inverse margin (larger weight for lags closer to max)
        margins = top_max - top_vals + 1e-6
        weights = 1.0 / margins
        weights /= weights.sum()

        w = np.zeros(len(c))
        w[top] = weights

        grad = compute_gradient_fft(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Line search
        step = 0.01
        best_step = step
        best_obj = cur_max
        for trial in range(8):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            obj = c_test.max()
            if obj < best_obj:
                best_obj = obj
                best_step = step
            step *= 0.7

        a = project_simplex(a - best_step * grad, S)

        # Periodic noise

            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 4000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_coordinate_descent(a_init, n, S, max_time, rng, n_coords=500):
    """
    Coordinate descent: optimize individual coordinates to reduce max autocorrelation.
    For each coordinate a[i], find the value that minimizes the max of the affected lags.
    """
    a = normalize_sum(a_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Pre-compute which lags each coordinate affects
    # a[i] affects c[k] for k in [i, i+n-1] (as a[i]*a[k-i]) and symmetrically

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Select coordinates to modify: those with high "responsibility"
        # Responsibility of a[i] = sum over active lags k of a[i]*a[k-i] / active_count
        top_lags = np.argsort(c)[::-1][:20]
        resp = np.zeros(n)
        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            np.add.at(resp, js, a[js] * a[partners])

        # Pick top n_coords to modify
        order = np.argsort(resp)[::-1][:n_coords]

        improved = False
        for i in order:
            if time.monotonic() - start > max_time:
                break
            if a[i] < 1e-12:
                continue

            # Try reducing a[i] slightly

                a_test = a.copy()
                a_test[i] = a[i] * delta_frac
                # Renormalize (just scale back to maintain sum)
                s = a_test.sum()
                if s > 0:
                    a_test = a_test * (S / s)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur_max * (1 - 1e-9):
                    a = a_test
                    cur_max = c_test.max()
                    improved = True
                    break

        if not improved and it > 10:
            # Try adding small amounts to low-responsibility coordinates

def peak_shaving_v2(a, n, S, n_peaks=15, n_steps=300, lr_scale=0.003):
    """
    Enhanced peak shaving with multiplicative updates.
    """
    a = a.copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cm * 0.998)
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a


def generate_inits(n, rng):
    """Generate diverse initializations for optimization."""
    inits = []
    t = np.linspace(0, 1, n)

    # Asymmetric raised cosine
    for cf in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45]:
        for wf in [0.3, 0.4, 0.5, 0.6, 0.7]:
            width = wf / 2.0
            x = (t - cf) / width
            a = np.ones(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1.0 + np.cos(np.pi * x[mask]))
            a[~mask] = 0.0
            inits.append(np.maximum(a, 1e-10))

    # Gaussian bumps
    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25]:
        for mu in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = t ** p
        inits.append(np.maximum(a, 1e-10))
        a = (1.0 - t) ** p
        inits.append(np.maximum(a, 1e-10))

    # Triangle
    for peak in [0.25, 0.3, 0.35, 0.4, 0.5, 0.6]:
        a = np.where(t < peak, t / peak, (1.0 - t) / (1.0 - peak))
        inits.append(np.maximum(a, 1e-10))

    # Sine
    for k in [1, 2, 3, 4]:
        a = np.maximum(0.0, 0.5 + 0.5 * np.sin(np.pi * k * t - 0.5))
        inits.append(a)

    return inits

        if verbose:
            elapsed = time.monotonic() - (_g_deadline - 1740.0)  # approximate
            print("  * NEW BEST: %.10f (n=%d)" % (val, n))
    return val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase strategy: LSE-FISTA -> subgradient -> equalize -> coordinate descent
    -> peak shaving -> multi-n -> final polish.
    """
    global _g_deadline, _g_best_val, _g_best_a
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    _g_deadline = time.monotonic() + max(1.0, budget_s - safety)
    _g_best_val = np.inf
    _g_best_a = None

    t0 = time.monotonic()
    print("Budget: %.0fs, safety: %.0fs, deadline in %.0fs" % (budget_s, safety, budget_s - safety))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous: n=%d" % len(given))
    except Exception:
        pass

    # If no given, use a reasonable default
    if given is None:
        n_default = 8192
        t = np.linspace(0, 1, n_default)
        given = np.exp(-0.5 * ((t - 0.35) / 0.15) ** 2)
        given = np.maximum(given, 1e-10)
        given = normalize_sum(given, n_default)
        print("Using default Gaussian init, n=%d" % n_default)

    # === Phase 1: Quick LSE-FISTA on given ===
    n1 = len(given)
    S1 = np.sqrt(2.0 * n1)
    t_alloc = min(180.0, (_g_deadline - time.monotonic()) * 0.12)
    print("\n=== Phase 1: LSE-FISTA on given (n=%d, %.0fs) ===" % (n1, t_alloc))
    a1, v1 = run_fista_lse(given, n1, S1, t_alloc, rng,
                            beta_start=100.0, beta_end=30000.0, beta_steps=200,
                            momentum_restart_gap=200, noise_decay=40000.0)
    try_update_best(a1)
    print("  Phase 1 done, best=%.10f" % _g_best_val)

    # === Phase 2: Continue LSE-FISTA with different params ===
    if _g_best_a is not None:
        n2 = len(_g_best_a)
        S2 = np.sqrt(2.0 * n2)
        rem = _g_deadline - time.monotonic()
        t_alloc = min(rem * 0.25, 400.0)
        print("\n=== Phase 2: LSE-FISTA continued (n=%d, %.0fs) ===" % (n2, t_alloc))
        a2, v2 = run_fista_lse(_g_best_a, n2, S2, t_alloc, rng,
                                beta_start=500.0, beta_end=60000.0, beta_steps=300,
                                momentum_restart_gap=150, noise_decay=20000.0)
        try_update_best(a2)
        print("  Phase 2 done, best=%.10f" % _g_best_val)

    # === Phase 3: Subgradient Nesterov ===
    if _g_best_a is not None:
        n3 = len(_g_best_a)
        S3 = np.sqrt(2.0 * n3)
        rem = _g_deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 300.0)
        print("\n=== Phase 3: Subgradient Nesterov (n=%d, %.0fs) ===" % (n3, t_alloc))
        a3, v3 = run_subgrad_nesterov(_g_best_a, n3, S3, t_alloc, rng, beta_soft=300.0)
        try_update_best(a3)
        print("  Phase 3 done, best=%.10f" % _g_best_val)

    # === Phase 4: Equalize lags ===
    if _g_best_a is not None:
        n4 = len(_g_best_a)
        S4 = np.sqrt(2.0 * n4)
        rem = _g_deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 250.0)
        print("\n=== Phase 4: Equalize lags (n=%d, %.0fs) ===" % (n4, t_alloc))
        a4, v4 = run_equalize_lags(_g_best_a, n4, S4, t_alloc, rng, n_lags=15)
        try_update_best(a4)
        print("  Phase 4 done, best=%.10f" % _g_best_val)

    # === Phase 5: Peak shaving + LSE polish ===
    if _g_best_a is not None:
        n5 = len(_g_best_a)
        S5 = np.sqrt(2.0 * n5)
        print("\n=== Phase 5: Peak shaving ===")
        a5 = peak_shaving_v2(_g_best_a, n5, S5, n_peaks=20, n_steps=300, lr_scale=0.002)
        try_update_best(a5)
        
        rem = _g_deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.3, 150.0)
            print("  LSE polish after shaving (%.0fs)" % t_alloc)
            a5b, v5b = run_fista_lse(a5, n5, S5, t_alloc, rng,
                                      beta_start=2000.0, beta_end=60000.0, beta_steps=50,
                                      momentum_restart_gap=100, noise_decay=10000.0)
            try_update_best(a5b)
        print("  Phase 5 done, best=%.10f" % _g_best_val)

    # === Phase 6: Coordinate descent ===
    if _g_best_a is not None:
        n6 = len(_g_best_a)
        S6 = np.sqrt(2.0 * n6)
        rem = _g_deadline - time.monotonic()
        t_alloc = min(rem * 0.12, 120.0)
        print("\n=== Phase 6: Coordinate descent (n=%d, %.0fs) ===" % (n6, t_alloc))
        a6, v6 = run_coordinate_descent(_g_best_a, n6, S6, t_alloc, rng, n_coords=300)
        try_update_best(a6)
        print("  Phase 6 done, best=%.10f" % _g_best_val)

    # === Phase 7: Try different n values ===
    if _g_best_a is not None:
        rem = _g_deadline - time.monotonic()
        if rem > 120:
            print("\n=== Phase 7: Multi-n exploration (remaining %.0fs) ===" % rem)
            n_curr = len(_g_best_a)
            
            # Try 2x and 0.5x
            for n_try in [n_curr * 2, n_curr // 2, n_curr * 3 // 2]:
                if n_try < 1024 or n_try > 131072:
                    continue
                rem = _g_deadline - time.monotonic()
                if rem < 60:
                    break
                S_try = np.sqrt(2.0 * n_try)
                t_old = np.linspace(0, 1, n_curr)
                t_new = np.linspace(0, 1, n_try)
                init = np.interp(t_new, t_old, _g_best_a)
                init = normalize_sum(init, n_try)
                
                t_alloc = min(rem * 0.3, 180.0)
                print("  Trying n=%d (%.0fs)..." % (n_try, t_alloc))
                a_try, v_try = run_fista_lse(init, n_try, S_try, t_alloc, rng,
                                              beta_start=100.0, beta_end=30000.0, beta_steps=150)
                val = try_update_best(a_try)
                print("  n=%d result: %.10f" % (n_try, val))

    # === Phase 8: Diverse inits at best n ===
    if _g_best_a is not None:
        rem = _g_deadline - time.monotonic()
        if rem > 60:
            n8 = len(_g_best_a)
            S8 = np.sqrt(2.0 * n8)
            t_alloc = min(rem * 0.2, 200.0)
            print("\n=== Phase 8: Diverse inits (n=%d, %.0fs) ===" % (n8, t_alloc))
            
            inits = generate_inits(n8, rng)
            # Score inits
            init_scores = []
            for i, init in enumerate(inits):
                init_n = normalize_sum(init, n8)
                c = fft_convolve_self(init_n)
                init_scores.append((c.max(), i))
            init_scores.sort()
            
            per_init = max(5.0, t_alloc / 6)
            for score, idx in init_scores[:5]:
                rem = _g_deadline - time.monotonic()
                if rem < 30:
                    break
                print("  Init %d (initial max=%.6f)..." % (idx, score))
                ai, vi = run_fista_lse(inits[idx], n8, S8, min(per_init, rem * 0.4), rng,
                                        beta_start=50.0, beta_end=20000.0, beta_steps=100)
                try_update_best(ai)
            print("  Phase 8 done, best=%.10f" % _g_best_val)

    # === Phase 9: Final intensive refinement ===
    if _g_best_a is not None:
        rem = _g_deadline - time.monotonic()
        if rem > 10:
            n9 = len(_g_best_a)
            S9 = np.sqrt(2.0 * n9)
            print("\n=== Phase 9: Final refinement (remaining %.0fs) ===" % rem)
            
            # Equalize with more lags
            t_a = rem * 0.3
            a9a, _ = run_equalize_lags(_g_best_a, n9, S9, t_a, rng, n_lags=25)
            try_update_best(a9a)
            
            # Subgradient
            rem = _g_deadline - time.monotonic()
            if rem > 10:
                a9b, _ = run_subgrad_nesterov(_g_best_a, n9, S9, rem * 0.3, rng, beta_soft=1000.0)
                try_update_best(a9b)
            
            # Peak shave
            rem = _g_deadline - time.monotonic()
            if rem > 10:
                a9c = peak_shaving_v2(_g_best_a, n9, S9, n_peaks=30, n_steps=200, lr_scale=0.001)
                try_update_best(a9c)
                
                # Quick LSE after
                rem = _g_deadline - time.monotonic()
                if rem > 15:
                    a9d, _ = run_fista_lse(a9c, n9, S9, rem * 0.6, rng,
                                            beta_start=5000.0, beta_end=80000.0, beta_steps=30,
                                            momentum_restart_gap=80, noise_decay=5000.0)
                    try_update_best(a9d)
            
            print("  Phase 9 done, best=%.10f" % _g_best_val)

    # Final verification
    if _g_best_a is None:
        print("WARNING: No valid result, using fallback.")
        _g_best_a = np.ones(1024, dtype=np.float64)
        _g_best_a = normalize_sum(_g_best_a, 1024)

    _g_best_a = np.clip(_g_best_a, 0.0, 1000.0)
    if _g_best_a.sum() < 0.01:
        _g_best_a = np.ones(1024, dtype=np.float64)
        _g_best_a = normalize_sum(_g_best_a, 1024)

    final_list = [float(x) for x in _g_best_a]
    ver = evaluate_sequence(final_list)
    total_time = time.monotonic() - t0
    print("\n=== FINAL ===")
    print("evaluate_sequence = %.10f" % (ver if np.isfinite(ver) else 999.0))
    print("internal best = %.10f (n=%d)" % (_g_best_val, len(_g_best_a)))
    print("Total time: %.1fs" % total_time)

    return final_list
