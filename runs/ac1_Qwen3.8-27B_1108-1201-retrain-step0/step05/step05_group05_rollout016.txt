    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def objective(a):

    s = a.sum()
    if s < 0.01:

    b = fft_convolve_self(a)
    return 2.0 * len(a) * b.max() / (s * s)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1)

    theta = (cumsum[rho - 1] - S) / rho

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FISTA with adaptive LSE, backtracking line search, spectral reshaping,
and multi-restart strategy.
"""
import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value_and_weights(c, beta):
    """Compute LSE value and softmax weights."""
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -700, 0)
    e = np.exp(scaled)
    w = e / e.sum()
    lse_val = cm + np.log(e.sum()) / beta
    return lse_val, w

def lse_subgradient(a, w):
    """Compute 2 * (w * a) restricted to [0, n-1] from weights w on convolution lags."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

                 beta_init=10.0, beta_growth=1.05,
                 beta_max=10000.0, n_restarts=20):
    """
    FISTA with backtracking line search and adaptive LSE.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    step = 1.0 / (4.0 * beta)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Update beta adaptively
        beta = min(beta_max, beta * beta_growth)
        
        c_y = fft_convolve_self(y)
        if c_y.max() < best:
            best = c_y.max()
            best_a = y.copy()
            last_improve = it
        
        # Compute LSE weights and subgradient
        lse_val, w = lse_value_and_weights(c_y, beta)
        grad = lse_subgradient(y, w)
        
        # Backtracking line search
        step = max(step * 0.5, 1e-12)
        L = 4.0 * beta  # Lipschitz constant estimate
        
        improved = False
        for bt_iter in range(20):
            x_new = project_simplex(y - grad / L, S)
            # Check descent condition (approximate, since LSE is expensive to evaluate)
            # Just accept the step (FISTA with fixed L should work)
            improved = True
            break
        
        if not improved:
            L *= 2
            continue
        
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize to maintain sum = S (projection already does this)
        a = x_new

def spectral_reshape(a, S, rng, alpha=0.3):
    """
    Apply a frequency-domain filter that boosts low frequencies.
    This tends to make the signal smoother, which can help with
    certain types of autocorrelation flatness.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    freqs = np.arange(fsize)
    # Gaussian low-pass filter
    center = fsize // 2
    # Create a filter that's 1 at DC and decays
    dist = np.minimum(freqs, fsize - freqs).astype(float)  # distance from DC
    bandwidth = fsize * 0.1  # 10% of frequency range
    filt = np.exp(-0.5 * (dist / bandwidth) ** 2)
    # Mix: keep some high frequency content
    new_A = A * (1 - alpha + alpha * filt)
    new_A[0] = A[0]  # Preserve DC
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

def peak_shave(a, S, rng, top_k=32):
    """
    Targeted peak shaving: identify top lags and apply a perturbation
    that specifically reduces them.
    """

    c = fft_convolve_self(a)
    c_sorted_idx = np.argsort(c)[::-1]
    top_lags = c_sorted_idx[:top_k]

    w = np.zeros(len(c))
    excess = np.maximum(c[top_lags] - c.mean(), 0)
    if excess.sum() < 1e-30:
        return a
    w[top_lags] = excess / excess.sum()
    
    grad = lse_subgradient(a, w)
    
    # Small step in the negative gradient direction
    step = S / (n * 10)
    a_new = project_simplex(a - step * grad, S)
    return a_new

So: a[n-2-j] = lambda for j=0,...,n-2, meaning a[0], a[1], ..., a[n-2] are all = lambda.
And a[1] = lambda (already included).
a[n-1] is free (doesn't appear in the equation for j where a_j > 0, assuming a[n-1] > 0).

Wait, for j = n-1: the condition is a[k*-j] + a[j-k*] = a[n-2-(n-1)] + a[(n-1)-(n-2)] = a[-1] + a[1] = 0 + a[1] = a[1] = lambda.
So a[1] = lambda. And a[n-1] doesn't appear, so it's free.

So the KKT solution with max at lag n-2 (and full support) is:
a[0] = a[1] = ... = a[n-2] = lambda, a[n-1] = free (determined by sum constraint)

sum = (n-1)*lambda + a[n-1] = S
So a[n-1] = S - (n-1)*lambda.

For a[n-1] >= 0: lambda <= S/(n-1).

Now let's compute the convolution:
conv[n-2] = sum_i a[i]*a[n-2-i]
= a[0]*a[n-2] + a[1]*a[n-3] + ... + a[n-2]*a[0]
= n-1 terms of lambda*lambda = (n-1)*lambda^2

conv[n-1] = sum_i a[i]*a[n-1-i]
= a[0]*a[n-1] + a[1]*a[n-2] + ... + a[n-2]*a[1] + a[n-1]*a[0]
= 2*a[0]*a[n-1] + (n-2)*lambda^2
= 2*lambda*(S-(n-1)*lambda) + (n-2)*lambda^2
= 2*lambda*S - 2*(n-1)*lambda^2 + (n-2)*lambda^2
= 2*lambda*S - n*lambda^2

conv[n-3] = sum_i a[i]*a[n-3-i]
= a[0]*a[n-3] + ... + a[n-3]*a[0]
= (n-2)*lambda^2

Let's check: conv[n-2] should equal conv[n-1] at the optimum (since both are at the max).
(n-1)*lambda^2 = 2*lambda*S - n*lambda^2
(n-1+n)*lambda^2 = 2*lambda*S
(2n-1)*lambda = 2S
lambda = 2S/(2n-1)

Then: a[n-1] = S - (n-1)*2S/(2n-1) = S*(2n-1-2n+2)/(2n-1) = S/(2n-1)

conv[n-2] = (n-1)*(2S/(2n-1))^2 = (n-1)*4S^2/(2n-1)^2
conv[n-1] = 2*(2S/(2n-1))*S/(2n-1) + (n-2)*(2S/(2n-1))^2
= 4S^2/(2n-1)^2 + (n-2)*4S^2/(2n-1)^2
= 4S^2*n/(2n-1)^2

Check: (n-1)*4S^2/(2n-1)^2 vs 4S^2*n/(2n-1)^2. These are NOT equal!
(n-1) vs n. So there's an inconsistency.

This means there's no exact KKT solution with max at lag n-2 and full support. The KKT conditions can't be satisfied exactly, which means the optimum either:
- Has the max at multiple lags (so we need a convex combination of subgradients)
- Or has some a_j = 0 (reduced support)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Multi-phase optimization combining adaptive LSE-FISTA with
two-level subgradients, normalized steps, peak-matching, and spectral
reshaping. Uses progressive sharpening of the subgradient focus.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights from convolution with given beta."""
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -700, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return e / tot


def topk_weights(c, k, sharpness=1.0):
    """Compute weights concentrated on top-k lags."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    vmax = vals.max()
    vmin = vals.min()
    if vmax - vmin < 1e-30:
        w = np.zeros(len(c))
        w[idx] = 1.0 / k
        return w
    scaled = -sharpness * (vmax - vals)
    scaled -= scaled.max()
    e = np.exp(scaled)
    w = np.zeros(len(c))
    w[idx] = e / e.sum()
    return w


def blend_weights(c, beta, k, mix, sharpness=1.0):
    """Blend LSE weights with top-k weights."""
    w_lse = lse_weights(c, beta)
    w_top = topk_weights(c, k, sharpness)
    return mix * w_top + (1.0 - mix) * w_lse


def subgradient_from_weights(a, w):
    """Compute 2*conv(w, a) restricted to first n entries."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_adaptive(a_init, S, max_time, rng,
                       beta_start=5.0, beta_end=5000.0,
                       k_start=8, k_end=32,
                       mix_start=0.0, mix_end=0.7,
                       restart_period=3000, noise_scale_base=0.003):
    """
    FISTA with adaptive two-level subgradient.
    Beta and k increase over time; mixing ratio shifts toward top-k.
    Uses normalized gradient steps for stability.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive parameters
        beta = beta_start * (beta_end / beta_start) ** progress
        k = int(k_start + (k_end - k_start) * progress)
        k = max(1, min(k, n // 4))
        mix = mix_start + (mix_end - mix_start) * progress
        sharpness = 1.0 + 5.0 * progress

        c_y = fft_convolve_self(y)
        if c_y.max() < best:
            best = c_y.max()
            best_a = y.copy()
            last_improve = it

        # Compute blended subgradient
        w = blend_weights(c_y, beta, k, mix, sharpness)
        grad = subgradient_from_weights(y, w)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Normalized step with scaling
        # Use a step that's proportional to S/n but scaled by progress
        base_step = S / (n * 2.0)
        step_scale = 1.0 / (1.0 + 0.001 * progress * progress * (beta / beta_end))
        step = base_step * step_scale

        # FISTA update
        x_new = project_simplex(y - step * grad / (gnorm / (S / n) + 1e-10), S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum constraint (projection should handle this, but be safe)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new

        # Restart if stuck
        if it - last_improve > restart_period:
            noise_scale = noise_scale_base * S / np.sqrt(n) * (1.0 + progress)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best


def run_fista_single_peak(a_init, S, max_time, rng):
    """
    FISTA focusing only on the single maximum lag.
    Very aggressive peak reduction.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c_y = fft_convolve_self(y)
        if c_y.max() < best:
            best = c_y.max()
            best_a = y.copy()
            last_improve = it

        # Single peak subgradient
        peak_idx = int(np.argmax(c_y))
        w = np.zeros(len(c_y))
        w[peak_idx] = 1.0
        grad = subgradient_from_weights(y, w)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Decreasing step size
        step = S / (n * (2.0 + 50.0 * progress))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new

        if it - last_improve > 5000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best


def spectral_reshape(a, S, alpha=0.5, bandwidth_frac=0.15):
    """
    Apply a Gaussian low-pass filter in frequency domain.
    Makes the signal smoother, potentially helping autocorrelation.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    freqs = np.arange(fsize)
    dist = np.minimum(freqs, fsize - freqs).astype(np.float64)
    bandwidth = fsize * bandwidth_frac
    filt = np.exp(-0.5 * (dist / bandwidth) ** 2)
    # Preserve DC, blend with filter
    new_A = A * ((1.0 - alpha) + alpha * filt)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_phase_perturb(a, S, rng, scale=0.1):
    """Perturb the phase of the DFT for diversity."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = np.abs(A) * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def peak_matching_step(a, S, n_matches=4):
    """
    Try to equalize the top n_matches peaks by computing a
    correction direction. This is a heuristic: we compute the
    gradient of sum of (c_k - target)^2 for top peaks and take
    a small step.
    """
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:n_matches]
    target = c[top_idx].mean()
    excess = np.zeros(len(c))
    for i in top_idx:
        excess[i] = c[i] - target
    if np.abs(excess).sum() < 1e-30:
        return a
    w = np.abs(excess) / np.abs(excess).sum()
    # Sign: we want to REDUCE peaks, so use negative of the weighted gradient
    grad = subgradient_from_weights(a, w)
    step = S / (n * 5.0)
    return project_simplex(a - step * grad, S)
