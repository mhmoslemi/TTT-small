```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses Adam optimizer with adaptive top-k peak chasing,
softmax reparameterization option, and L-BFGS final polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def conv_gradient(y, w):
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross
```

```python
def run_adam_peak_chase(a_init, max_time, rng):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = a_init * (S / a_init.sum())
    
    # Adam hyperparameters
    lr = 1.0  # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_conv = np.inf
    best_a = a.copy()
    last_improve = 0
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = t
        
        stuck = t - last_improve
        
        # Adaptive top-k
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 3
        elif stuck < 5000:
            k = 8
        else:
            k = min(32, 8 + (stuck - 5000) // 500)
        
        # Compute gradient from top-k lags
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(2*n-1)
        w[idx] = 1.0 / k
        grad = conv_gradient(a, w)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adaptive learning rate (decrease as we get stuck)
        effective_lr = lr / (1 + 0.001 * stuck)
        
        a_new = a - effective_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, S)
        
        # Occasional noise to escape local minima
        if stuck > 2000 and rng.random() < 0.1:
            noise_scale = 0.002 * S / np.sqrt(n)
            a_new = project_simplex(a_new + rng.normal(0, noise_scale, n), S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stuck = 0
        
        a = a_new
    
    return best_a, best_conv
```

FISTA uses a single global step size (inversely proportional to the Lipschitz constant of the smoothed objective). Adam uses per-coordinate step sizes. The advantage of Adam is when different coordinates have very different scales of gradient.

In our problem, the gradient is `2 * (shifted version of a)`, so the scale of the gradient at each coordinate is roughly proportional to the local "energy" of `a`. If `a` is relatively uniform (which it should be for a good solution), then all coordinates have similar gradient scales, and Adam's advantage is minimal.

**Approach: Iterative Proximal Minimization (IPM)**

The problem is: minimize `f(a) = max_k (conv(a,a)_k` s.t. `a >= 0, sum(a) = S`

IPM works as follows:
1. Start with a feasible point `a_0`
2. At each iteration, solve: `a_{t+1} = argmin f(a_t) + <∇f(a_t), a - a_t> + (1/(2*eta_t)) * ||a - a_t||^2` s.t. `a >= 0, sum(a) = S`
3. This is a quadratic program (QP) that can be solved exactly

The QP in step 2 is:
`minimize (1/(2*eta)) * ||a - a_t||^2 + <∇f(a_t), a>`
`s.t. a >= 0, sum(a) = S`

This is equivalent to projecting `(a_t - eta * ∇f(a_t))` onto the simplex `{a >= 0, sum(a) = S}`, which is exactly what FISTA does (without the momentum).

So IPM without momentum = projected gradient descent. IPM with momentum ≈ FISTA. So this is the same as what we already have.

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses Adam peak-chasing with adaptive top-k, aggressive L-BFGS polishing,
n-scaling exploration, and equiripple-inspired equalization gradients.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def equiripple_gradient(y, c, exclude_lag0=True, delta=0.05):
    """Gradient that pushes toward equiripple: down at peaks, up at valleys."""
    n = len(y)
    c_work = c.copy()
    if exclude_lag0:
        c_work[0] = c_work[1:] .mean() if len(c_work) > 1 else 0
    target = c_work.mean()
    # Smooth sign: tanh((c - target) / delta)
    w = np.tanh((c_work - target) / max(delta, 1e-10))
    # Scale so positive and negative parts balance
    pos_sum = w[w > 0].sum()
    neg_sum = abs(w[w < 0].sum())
    if pos_sum > 1e-15 and neg_sum > 1e-15:
        scale = 2.0 / (pos_sum + neg_sum)
        w = w * scale
    else:
        w = np.zeros_like(w)
    return conv_gradient(y, w)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def run_adam_peak_chase(a_init, max_time, rng, mode="topk"):
    """Adam optimizer with adaptive top-k peak chasing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Adam hyperparameters
    base_lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it
        stuck = it - last_improve

        # Adaptive parameters based on stuck count
        if stuck < 100:
            k = 1
            lr = base_lr
            eq_weight = 0.0
        elif stuck < 500:
            k = 3
            lr = base_lr * 0.7
            eq_weight = 0.0
        elif stuck < 2000:
            k = 8
            lr = base_lr * 0.4
            eq_weight = 0.2
        elif stuck < 5000:
            k = 16
            lr = base_lr * 0.25
            eq_weight = 0.4
        else:
            k = min(64, 16 + (stuck - 5000) // 500)
            lr = base_lr * 0.15
            eq_weight = 0.5

        # Compute gradient
        if mode == "topk":
            grad = topk_gradient(a, c, k, power=1.0)
        elif mode == "lse":
            beta = 20.0 + 0.01 * it
            w = lse_weights(c, beta)
            grad = conv_gradient(a, w)
        elif mode == "hybrid":
            grad_top = topk_gradient(a, c, k, power=1.0)
            grad_eq = equiripple_gradient(a, c, delta=max(0.01, 0.1 * c.std()))
            grad = (1.0 - eq_weight) * grad_top + eq_weight * grad_eq
        else:
            grad = topk_gradient(a, c, k, power=1.5)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_simplex(a_new, S)

        # Escape mechanism: periodic noise + Adam reset
        if stuck > 3000 and it % 200 == 0:
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            if noise_scale > 1e-8:
                a_new = project_simplex(a_new + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

        a = a_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

        # Adaptive mode

            w = lse_weights(c, beta)

        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:

def run_lbfgs_polish(a_init, max_time, rng, n_levels=5, maxiter=500):
    """Aggressive L-BFGS-B polishing with multiple beta levels."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    levels = [(30, 10.0), (50, 20.0), (100, 50.0), (200, 100.0), (500, 200.0)]
    if n_levels < 5:
        levels = levels[:n_levels]

    for beta, pen in levels:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-15,
                     'gtol': 1e-12, 'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        print("    L-BFGS beta=%d pen=%.0f: obj=%.8f (iters=%d, t=%.1fs)" %
              (beta, pen, obj_new, result.nit, time.monotonic() - start))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj

        k = cycle[cycle_idx % len(cycle)]
        cycle_idx += 1

        grad = topk_gradient(y, c, k, power=1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.8 / gnorm
        step = min(step, 0.5)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using spectral method."""
    n = len(a)
    if new_n == n:
        return a.copy()
    if new_n < n:
        # Downsample: take FFT, keep low frequencies, IFFT
        fsize = 1 << (2 * n - 1).bit_length()
        A = fft(a, fsize)
        # Keep only new_n/2 positive frequencies
        A_reduced = np.zeros(new_n, dtype=complex)
        half_new = new_n // 2
        A_reduced[:half_new] = A[:half_new]
        A_reduced[new_n - half_new:] = A[fsize - half_new:fsize]
        a_new = np.real(ifft(A_reduced))
        a_new = np.maximum(a_new, 0.0)
        S = np.sqrt(2.0 * new_n)
        if a_new.sum() > 1e-10:
            a_new = a_new * (S / a_new.sum())
        return a_new
    else:
        # Upsample: zero-pad in frequency domain
        fsize_old = 1 << (2 * n - 1).bit_length()
        A = fft(a, fsize_old)
        fsize_new = 1 << (2 * new_n - 1).bit_length()
        A_padded = np.zeros(fsize_new, dtype=complex)
        half = fsize_new // 2
        # Copy low frequencies
        copy_pos = min(half, fsize_old // 2)
        A_padded[:copy_pos] = A[:copy_pos]
        A_padded[fsize_new - copy_pos:] = A[fsize_old - copy_pos:]
        a_new = np.real(ifft(A_padded))[:new_n]
        a_new = np.maximum(a_new, 0.0)
        S = np.sqrt(2.0 * new_n)
        if a_new.sum() > 1e-10:
            a_new = a_new * (S / a_new.sum())
        return a_new


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: Adam peak-chasing, aggressive L-BFGS, n-scaling,
    subgradient cycling, and final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None
    global_best_n = 0

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0

    # === Phase 1: Adam peak chasing (25% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.25
    print("\n=== Phase 1: Adam peak chase (n=%d, %.0fs) ===" % (global_best_n, p1))
    t1 = time.monotonic()

    a1, _ = run_adam_peak_chase(global_best_a, p1 * 0.5, rng, mode="topk")
    obj1 = compute_obj(a1)
    print("  Adam-topk: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1 - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_adam_peak_chase(global_best_a, rem1, rng, mode="hybrid")
        obj1b = compute_obj(a1b)
        print("  Adam-hybrid: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: FISTA with equiripple (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2 = remaining * 0.20
        print("\n=== Phase 2: FISTA hybrid (n=%d, %.0fs) ===" % (global_best_n, p2))
        t2 = time.monotonic()

        a2, _ = run_fista(global_best_a, p2, rng, beta_start=30.0, mode="hybrid")
        obj2 = compute_obj(a2)
        print("  FISTA: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Aggressive L-BFGS (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3 = remaining * 0.25
        print("\n=== Phase 3: L-BFGS polish (n=%d, %.0fs) ===" % (global_best_n, p3))
        t3 = time.monotonic()

        a3, obj3 = run_lbfgs_polish(global_best_a, p3, rng, n_levels=5, maxiter=500)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Try n=16384 (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p4 = remaining * 0.15
        print("\n=== Phase 4: Try n=16384 (%.0fs) ===" % p4)
        t4 = time.monotonic()

        n_large = 16384
        a_large = interpolate_sequence(global_best_a, n_large)
        obj_large = compute_obj(a_large)
        print("  Interpolated n=16384: obj=%.8f" % obj_large)

        if obj_large < global_best_obj + 0.1:  # Only worth optimizing if close
            a4, _ = run_fista(a_large, p4 * 0.5, rng, beta_start=20.0, mode="hybrid")
            obj4 = compute_obj(a4)
            print("  FISTA n=16384: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
            if obj4 < global_best_obj:
                global_best_obj = obj4
                global_best_a = a4.copy()
                global_best_n = n_large

            rem4 = p4 - (time.monotonic() - t4)
            if rem4 > 30:
                a4b, obj4b = run_lbfgs_polish(global_best_a, rem4, rng, n_levels=3, maxiter=300)
                print("  L-BFGS n=%d: obj=%.8f (t=%.0fs)" % (global_best_n, obj4b, time.monotonic() - t4))
                if obj4b < global_best_obj:
                    global_best_obj = obj4b
                    global_best_a = a4b.copy()
                    global_best_n = len(a4b)

    # === Phase 5: Subgradient cycling (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5 = remaining * 0.10
        print("\n=== Phase 5: Subgradient cycling (n=%d, %.0fs) ===" % (global_best_n, p5))
        t5 = time.monotonic()

        a5, _ = run_subgradient_cycling(global_best_a, p5, rng)
        obj5 = compute_obj(a5)
        print("  Cycling: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final squeeze (%.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng, n_levels=3, maxiter=800)
        print("  Final: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: Short Adam burst from best ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t7 = time.monotonic()
        a7, _ = run_adam_peak_chase(global_best_a, remaining * 0.7, rng, mode="topk")
        obj7 = compute_obj(a7)
        print("\nFinal Adam: obj=%.8f (t=%.0fs)" % (obj7, time.monotonic() - t7))
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: %.8f (n=%d)" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
```
