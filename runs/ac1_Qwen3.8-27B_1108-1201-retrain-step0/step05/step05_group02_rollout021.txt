    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    s = a.sum()
    if s < 1e-15:

    c = fft_conv(a)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]

    if cond.any():

    else:
        rho = n
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimization minimizing 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LSE-FISTA for coarse optimization, multi-peak coordinate descent with
exact line search for fine tuning, and peak-directed perturbation for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_val(a):
    """Compute objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - S)
    if np.any(cond):
        rho = int(np.max(np.where(cond)[0])) + 1
    else:
        rho = n
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fista_lse(a_init, time_limit, rng, beta0=50.0, beta_growth=0.008, beta_max=50000.0):
    """FISTA with log-sum-exp smoothing of the max objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a *= S / s
    y = a.copy()
    t_mom = 1.0
    best_obj = 2.0 * n * fft_conv(a).max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    last_improve = 0
    step = 0.5 / max(beta0, 1.0)

    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        beta = min(beta_max, beta0 * np.exp(beta_growth * it))

        c = fft_conv(y)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / tot

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]

        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = 2.0 * n * fft_conv(x_trial).max() / (S * S)
        cur_obj = 2.0 * n * cm / (S * S)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-15 and bt < 8:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = 2.0 * n * fft_conv(x_trial).max() / (S * S)
            bt += 1
        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.04, 2.0 / max(beta * 2.0, 1.0))
        x_new = x_trial

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y *= S / ys
        a = x_new
        t_mom = t_new

        obj_new = 2.0 * n * fft_conv(a).max() / (S * S)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 3000:
            t_mom = 1.0
            noise = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 4000.0))
            y = project_simplex(y + rng.normal(0, noise, n), S)

    return best_a, best_obj


def multi_peak_coord_descent(a_init, time_limit, rng):
    """
    Fast coordinate descent using multi-peak influence scoring.
    Transfers mass from high-influence to low-influence coordinates
    with exact line search on the transfer amount.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a *= S / a.sum()
    c = fft_conv(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30000:
                break

        # Compute multi-peak influence vector
        n_peaks = min(5, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]
        peak_weights = peak_vals / peak_vals.sum()

        infl = np.zeros(n)
        for idx in range(n_peaks):
            k = int(peak_idx[idx])
            w = peak_weights[idx]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                partners = a[k - hi : k - lo + 1][::-1]
                infl[lo:hi + 1] += w * partners

        # Source score: high infl * a[j] (lots of mass, high influence)
        # Target score: low infl (adding mass won't increase peaks)
        src_score = infl * a
        top_n = min(40, n)
        top_src = np.argsort(src_score)[::-1][:top_n]
        top_tgt = np.argsort(infl)[:top_n]

        # Generate and score candidate pairs using first-order approximation
        k0 = int(peak_idx[0])
        candidates = []
        for j in top_src[:25]:
            if a[j] < 1e-12:
                continue
            pj = k0 - j
            for l in top_tgt[:25]:
                if j == l:
                    continue
                pl = k0 - l
                if 0 <= pj < n and 0 <= pl < n:
                    improvement = a[pj] - a[pl]
                    if improvement > 0:
                        candidates.append((improvement * a[j], int(j), int(l)))

        if not candidates:
            if no_improve > 100:
                j = int(rng.integers(0, n))
                l = int(rng.integers(0, n))
                if j != l and a[j] > 1e-12:
                    delta = a[j] * 0.04 * rng.random()
                    a[j] -= delta
                    a[l] += delta
                    c = fft_conv(a)
            continue

        candidates.sort(key=lambda x: -x[0])

        # Try top candidates with line search
        best_move_obj = cur_obj
        best_move = None

        for score, j, l in candidates[:8]:
            best_frac = 0.0
            best_frac_obj = best_move_obj
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8]:
                delta = frac * a[j]
                if delta < 1e-15 or delta > a[j]:
                    continue
                a[j] -= delta
                a[l] += delta
                c_tmp = fft_conv(a)
                obj_tmp = 2.0 * n * c_tmp.max() / (S * S)
                if obj_tmp < best_frac_obj:
                    best_frac_obj = obj_tmp
                    best_frac = frac
                a[j] += delta
                a[l] -= delta

            if best_frac > 0 and best_frac_obj < best_move_obj:
                best_move_obj = best_frac_obj
                best_move = (j, l, best_frac)

        if best_move is not None:
            j, l, frac = best_move
            delta = frac * a[j]
            a[j] -= delta
            a[l] += delta
            c = fft_conv(a)
        else:
            # No improvement; apply structured perturbation
            if no_improve > 50:
                # Peak-directed perturbation
                k_star = int(np.argmax(c))
                lo = max(0, k_star - n + 1)
                hi = min(n - 1, k_star)
                infl_local = np.zeros(n)
                if hi >= lo:
                    infl_local[lo:hi + 1] = a[k_star - hi : k_star - lo + 1][::-1]
                # Remove from high infl, add to low infl
                src = int(np.argmax(infl_local * a))
                tgt = int(np.argmin(infl_local))
                if a[src] > 1e-12 and src != tgt:
                    delta = a[src] * 0.02
                    a[src] -= delta
                    a[tgt] += delta
                    c = fft_conv(a)
    
    return best_a, best_obj


def peak_directed_optimize(a_init, time_limit, rng):
    """
    Directly target convolution peaks using structured gradient steps.
    Uses exact subgradient from the active set (top-k peaks).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a *= S / a.sum()
    best_obj = 2.0 * n * fft_conv(a).max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        c = fft_conv(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        # Adaptive k for top-k subgradient
        if no_improve < 100:
            k = 1
        elif no_improve < 500:
            k = 3
        elif no_improve < 2000:
            k = 8
        else:
            k = min(32, 8 + (no_improve - 2000) // 200)

        # Compute subgradient from top-k peaks
        top_idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / k

        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 3.0 * S / gnorm)
        bt = 0
        while bt < 10:
            a_new = project_simplex(a - step * grad, S)
            obj_new = 2.0 * n * fft_conv(a_new).max() / (S * S)
            if obj_new <= cur_obj + 1e-14:
                break
            step *= 0.5
            bt += 1

        if step > 1e-15:
            a = a_new
        else:
            # Perturb
            idx = int(rng.integers(0, n))
            delta = a[idx] * 0.01 * rng.random()
            if delta > 1e-14:
                a[idx] -= delta
                a[int(rng.integers(0, n))] += delta
                a = project_simplex(a, S)

    return best_a, best_obj


def spectral_shape_optimize(a_init, time_limit, rng):
    """
    Work in frequency domain: reshape |A| to reduce time-domain peak of c.
    c = ifft(|A|^2), so we want |A| shaped to make c flat.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a *= S / a.sum()
    best_obj = 2.0 * n * fft_conv(a).max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0
    it = 0

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < time_limit:
        it += 1
        if no_improve > 10000:
            break

        c = fft_conv(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute spectrum
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

        # Peak location in time domain
        k_star = int(np.argmax(c))

        # The peak at k_star in c corresponds to interference in frequency domain.
        # To reduce c[k_star], we want to add a "notch" at frequencies
        # that constructively interfere at k_star.
        # The kernel for lag k is: h[m] = exp(-2j*pi*m*k/fsize)
        # c[k] = sum_m |A[m]|^2 * exp(-2j*pi*m*k/fsize) ... approximately
        # Actually c is real, so c[k] = sum_m |A[m]|^2 * cos(2*pi*m*k/fsize) approximately

        # Simple approach: reduce |A| at frequencies where cos(2*pi*m*k_star/fsize) > 0
        # and increase where < 0, then reconstruct and project
        freq = np.linspace(0, 1, fsize)
        kernel = np.cos(2 * np.pi * freq * k_star / n)  # Approximate kernel

        # Perturb magnitude
        alpha = 0.01 * np.exp(-no_improve / 2000.0)
        mag_new = mag * (1.0 + alpha * kernel)
        mag_new = np.maximum(mag_new, 0.0)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = project_simplex(a_new, S)

        obj_new = 2.0 * n * fft_conv(a_new).max() / (S * S)
        if obj_new < cur_obj + 1e-14:
            a = a_new
        else:
            # Try smaller step
            alpha_small = alpha * 0.1
            mag_new2 = mag * (1.0 + alpha_small * kernel)
            mag_new2 = np.maximum(mag_new2, 0.0)
            A_new2 = mag_new2 * np.exp(1j * phase)
            a_new2 = np.real(ifft(A_new2))[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 = project_simplex(a_new2, S)
            obj_new2 = 2.0 * n * fft_conv(a_new2).max() / (S * S)
            if obj_new2 < cur_obj + 1e-14:
                a = a_new2
            else:
                # Perturb randomly
                if no_improve > 200:
                    idx = int(rng.integers(0, n))
                    delta = a[idx] * 0.02 * rng.random()
                    if delta > 1e-14:
                        a[idx] -= delta
                        a[int(rng.integers(0, n))] += delta
                        a = project_simplex(a, S)

    return best_a, best_obj


def make_init(n, method, rng):
    """Create initial sequences using various heuristics."""
    S = np.sqrt(2.0 * n)
    t = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif method == "gaussian":
        a = np.exp(-3.0 * (t - 0.5) ** 2 / (0.2 ** 2))
    elif method == "chirp":
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * np.sin(2 * np.pi * 13 * t + 0.5) + 0.1 * np.sin(2 * np.pi * 23 * t + 1.2)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.02, 1.0 - 1.5 * np.abs(t - 0.5))
    elif method == "multi_bump":
        a = 0.3 + 0.4 * np.exp(-((t - 0.3) / 0.1)**2) + 0.3 * np.exp(-((t - 0.7) / 0.15)**2)
    else:
        a = np.ones(n) * 0.5
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA -> multi-peak CD -> spectral shaping -> peak-directed polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except (NameError, Exception):
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    obj0 = obj_val(a_start)
    print("Starting: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA with LSE (20% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.20
    if p1_time > 30:
        print("\n=== Phase 1: FISTA LSE (budget %.0fs) ===" % p1_time)
        # Run with two different beta schedules
        for trial, (b0, bg, bm) in enumerate([(30, 0.01, 20000), (200, 0.005, 80000)]):
            if time.monotonic() >= deadline:
                break
            rem = min(p1_time * 0.5, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = fista_lse(global_best_a, rem, rng,
                                       beta0=b0, beta_growth=bg, beta_max=bm)
            print("  FISTA trial %d: obj=%.8f" % (trial, obj_opt))
            if obj_opt < global_best_obj - 1e-12:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")

    # === Phase 2: Multi-peak coordinate descent (35% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.35
    if p2_time > 40:
        print("\n=== Phase 2: Multi-peak CD (budget %.0fs) ===" % p2_time)
        a_cd, obj_cd = multi_peak_coord_descent(global_best_a, p2_time, rng)
        print("  CD result: obj=%.8f" % obj_cd)
        if obj_cd < global_best_obj - 1e-12:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()
            print("  *** NEW BEST ***")

        # Follow-up FISTA on CD result
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_f, obj_f = fista_lse(global_best_a, min(remaining2 * 0.3, 60), rng,
                                    beta0=500, beta_growth=0.003, beta_max=100000)
            print("  FISTA after CD: obj=%.8f" % obj_f)
            if obj_f < global_best_obj - 1e-12:
                global_best_obj = obj_f
                global_best_a = a_f.copy()
                print("  *** NEW BEST ***")

    # === Phase 3: Spectral shaping (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 30:
        print("\n=== Phase 3: Spectral shaping (budget %.0fs) ===" % p3_time)
        a_spec, obj_spec = spectral_shape_optimize(global_best_a, p3_time, rng)
        print("  Spectral result: obj=%.8f" % obj_spec)
        if obj_spec < global_best_obj - 1e-12:
            global_best_obj = obj_spec
            global_best_a = a_spec.copy()
            print("  *** NEW BEST ***")

        # CD after spectral
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            a_cd2, obj_cd2 = multi_peak_coord_descent(global_best_a, remaining3 * 0.6, rng)
            print("  CD after spectral: obj=%.8f" % obj_cd2)
            if obj_cd2 < global_best_obj - 1e-12:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()
                print("  *** NEW BEST ***")

    # === Phase 4: Peak-directed optimization (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 20:
        print("\n=== Phase 4: Peak-directed (budget %.0fs) ===" % p4_time)
        a_pk, obj_pk = peak_directed_optimize(global_best_a, p4_time, rng)
        print("  Peak-directed: obj=%.8f" % obj_pk)
        if obj_pk < global_best_obj - 1e-12:
            global_best_obj = obj_pk
            global_best_a = a_pk.copy()
            print("  *** NEW BEST ***")

    # === Phase 5: Final intensive CD (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 5: Final CD (budget %.0fs) ===" % remaining)
        a_fin, obj_fin = multi_peak_coord_descent(global_best_a, remaining * 0.8, rng)
        print("  Final CD: obj=%.8f" % obj_fin)
        if obj_fin < global_best_obj - 1e-12:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()
            print("  *** NEW BEST ***")

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.8f, n=%d ===" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: Invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
