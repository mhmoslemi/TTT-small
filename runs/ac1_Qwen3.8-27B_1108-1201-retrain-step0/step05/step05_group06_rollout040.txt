"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements over previous approach:
- Backtracking FISTA for robust step size selection
- Adaptive LSE beta based on the active-set gap
- Hybrid smoothing: LSE when gap is small, exact subgradient when gap is large
- Barzilai-Borwein adaptive step for the non-smooth phase

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad_value(a, c, beta):
    """Compute LSE value and gradient efficiently via FFT."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    val = cm + (1.0 / beta) * np.log(tot + 1e-300)
    if tot < 1e-300:
        return val, np.zeros(n)
    p = e / tot
    fsize = 1 << (n + len(c) - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * P))[:n]
    return val, grad


def exact_subgrad(a, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(a)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    # grad = sum_{k in active} (1/|active|) * 2 * a[k-j] (truncated)
    w = 1.0 / len(active)
    fsize = 1 << (n + len(c) - 1).bit_length()
    w_full = np.zeros(len(c))
    w_full[active] = w
    P = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * P))[:n]
    return grad


def topk_subgrad(a, c, k, power=1.0):
    """Weighted subgradient using top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals - vals.min() + 1e-10, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (n + len(c) - 1).bit_length()
    P = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * P))[:n]
    return grad


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def active_set_gap(c):
    """Compute gap between max and second-max of c."""
    sorted_c = np.sort(c)[::-1]
    if len(sorted_c) < 2:
        return 0.0
    return sorted_c[0] - sorted_c[1]


def run_fista_backtrack(a_init, max_time, rng, beta_init=10.0, beta_max=5000.0,
                        use_hybrid=True):
    """FISTA with backtracking line search and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    L = 1.0  # Lipschitz estimate
    prev_y = None
    prev_grad = None
    prev_x = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta based on gap
        gap = active_set_gap(c)
        if gap > 0.1 * cur_max:
            # Large gap: use exact subgradient
            if use_hybrid and stuck > 100:
                grad = exact_subgrad(y, c, tol_frac=0.0)
            else:
                beta = min(beta_max, beta * 1.05)
                val, grad = lse_grad_value(y, c, beta)
        else:
            # Small gap: use LSE with moderate beta
            if stuck < 500:
                beta = max(beta_init, beta * 0.999)
            else:
                beta = min(beta_max, beta * 1.02)
            val, grad = lse_grad_value(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        # We need: f(x_new) <= f(y) + grad^T(x_new - y) + (L/2)||x_new - y||^2
        # For simplicity, use a fixed fraction rule since we're projecting
        step = 1.0 / L

        x_new = project_simplex(y - step * grad, S)

        # Check improvement (simplified backtracking)
        c_new = fft_convolve_self(x_new)
        obj_new = c_new.max()
        obj_cur = cur_max

        if obj_new > obj_cur * 1.01 and step < 1.0:
            # Step too large, reduce
            L = L * 1.5
            step = 1.0 / L
            x_new = project_simplex(y - step * grad, S)
        elif step < 0.5 / L and it % 50 == 0:
            # If we've shrunk a lot, try to grow L back slowly
            L = max(1.0, L * 0.95)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = normalize_seq(y, S)
        a = x_new
        t_mom = t_new

        # Momentum restart strategies
        if stuck > 3000:
            t_mom = 1.0
            # Add structured perturbation
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-10:
                tt = np.arange(n)
                perturb = np.sin(2 * np.pi * tt * rng.integers(1, 10) / n)
                y = normalize_seq(y + noise_scale * perturb, S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def run_mirror_descent(a_init, max_time, rng, beta_init=30.0):
    """Mirror descent (multiplicative weights) with adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    best_a = a.copy()

    last_improve = 0
    beta = beta_init
    step_mult = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta
        if stuck < 500:
            beta = beta_init
        elif stuck < 2000:
            beta = beta_init * (1.0 + stuck * 0.01)

        val, grad = lse_grad_value(y, c, beta)

def make_structured_init(n, rng, style="smooth"):
    """Create diverse structured initializations."""
    if style == "cosine":
        tt = np.linspace(0, 1, n, endpoint=False)
        return 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif style == "gaussian":
        tt = np.arange(n, dtype=np.float64)
        center = n / 2.0
        sigma = n / 4.0
        return np.exp(-0.5 * ((tt - center) / sigma) ** 2)
    elif style == "bimodal":
        tt = np.arange(n, dtype=np.float64)
        s1 = np.exp(-0.5 * ((tt - n * 0.25) / (n * 0.1)) ** 2)
        s2 = np.exp(-0.5 * ((tt - n * 0.75) / (n * 0.1)) ** 2)
        return s1 + s2
    elif style == "pow":
        tt = np.linspace(0, 1, n, endpoint=False)
        return tt ** 0.5 + (1 - tt) ** 0.5
    elif style == "chirp":
        tt = np.arange(n, dtype=np.float64)
        return 0.5 + 0.5 * np.cos(2 * np.pi * tt ** 2 / n)
    else:  # random
        return rng.random(n)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging."""
    n = len(a)
    n_new = n // factor
    a_pad = np.zeros(n_new * factor)
    a_pad[:n] = a[:n_new * factor]
    return a_pad.reshape(n_new, factor).mean(axis=1)

            global_best_obj = obj_p2
            global_best_a = a_p2.copy()

    # === Phase 3: Mirror descent ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15
    if p3_budget > 10:
        print("\n=== Phase 3: Mirror descent (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a_p3, _ = run_mirror_descent(global_best_a, p3_budget, rng, beta_init=30.0)
        obj_p3 = compute_obj(a_p3)
        print("  Phase3: obj=%.6f (t=%.1fs)" % (obj_p3, time.monotonic() - t3))
        if obj_p3 < global_best_obj:
            global_best_obj = obj_p3
            global_best_a = a_p3.copy()

    # === Phase 4: Final FISTA with high beta ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 10:
        print("\n=== Phase 4: High-beta FISTA (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a_p4, _ = run_fista_backtrack(global_best_a, p4_budget, rng,
                                       beta_init=100.0, beta_max=20000.0,
                                       use_hybrid=True)
        obj_p4 = compute_obj(a_p4)
        print("  Phase4: obj=%.6f (t=%.1fs)" % (obj_p4, time.monotonic() - t4))
        if obj_p4 < global_best_obj:
            global_best_obj = obj_p4
            global_best_a = a_p4.copy()

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 5: Multi-n exploration (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()
        best_at_base_n = global_best_a.copy()

        for dn in [-2, -1, 1, 2]:
            n_alt = n + dn
            if n_alt < 100:
                continue
            # Adjust sequence to new length
            if dn > 0:
                # Upsample by adding elements
                a_alt = upsample(global_best_a, 1)  # This won't work for +1
                # For small changes, just truncate or pad
                if dn == 1:
                    a_alt = np.concatenate([global_best_a, [global_best_a[-1]]])
                elif dn == 2:
                    a_alt = np.concatenate([global_best_a, global_best_a[-1:2]])
                else:
                    continue
            else:
                a_alt = global_best_a[:n_alt]

            S_alt = np.sqrt(2.0 * n_alt)
            a_alt = normalize_seq(a_alt, S_alt)

            # Quick optimization
            budget_this = min(20.0, remaining * 0.2)
            a_opt, _ = run_fista_backtrack(a_alt, budget_this, rng,
                                            beta_init=10.0, beta_max=1000.0)
            obj_alt = compute_obj(a_opt)
            print("  n=%d: obj=%.6f" % (n_alt, obj_alt))
            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_opt.copy()

        print("  Multi-n done (t=%.1fs)" % (time.monotonic() - t5))

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        p6_budget = remaining * 0.8

        # Hybrid polish
        a_fin1, _ = run_hybrid_polish(global_best_a, p6_budget * 0.5, rng)
        obj_fin1 = compute_obj(a_fin1)
        print("  Final1: obj=%.6f" % obj_fin1)
        if obj_fin1 < global_best_obj:
            global_best_obj = obj_fin1
            global_best_a = a_fin1.copy()

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 5:
            a_fin2, _ = run_fista_backtrack(global_best_a, rem6, rng,
                                             beta_init=200.0, beta_max=50000.0,
                                             use_hybrid=True)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final2: obj=%.6f" % obj_fin2)
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - total_start))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
