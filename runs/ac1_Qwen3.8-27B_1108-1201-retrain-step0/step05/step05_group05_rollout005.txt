def project_simplex(v, S):

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B continuation
on the scale-invariant LSE-smoothed objective, combined with 
simulated annealing in the Fourier domain for global exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_and_grad(a, beta):
    """
    Compute LSE_beta(conv(a,a)) and its gradient with respect to a.
    Returns (lse_value, gradient_vector).
    """
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    scaled = beta * (c - cmax)
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    logsumexp = max_s + np.log(exp_s.sum())
    lse_val = cmax + logsumexp / beta
    p = exp_s / exp_s.sum()
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return lse_val, grad


def scale_invariant_lse_objective(x, beta):
    """
    F(x) = LSE_beta(conv(x,x)) / (sum(x))^2 and its gradient.
    Scale-invariant, so no need for normalization penalty.
    """
    n = len(x)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.ones(n) * 1e-10
    lse_val, grad_lse = compute_lse_and_grad(x, beta)
    val = lse_val / (S * S)
    # grad F = (S * grad_lse - 2 * lse_val * ones) / S^3
    grad_F = (S * grad_lse - 2.0 * lse_val) / (S ** 3)
    return val, grad_F
