def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 400:
        beta = 400.0 / max(1e-15, (c - cm).max() - 0)
        if beta <= 0:
            return np.zeros(n)

    if tot < 1e-300:
        return np.zeros(n)

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def spectral_balance(a, alpha=0.5, S=None):
    """
    Gerchberg-Saxton style spectral balancing.
    Makes the spectrum flatter by adjusting magnitude.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Attenuate peaks in spectrum
    mag_target = mag ** alpha  # alpha < 1 compresses dynamic range
    A_new = mag_target * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Project to non-negative with correct sum
    if S is None:
        S = a.sum()
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 0:
        a_new = a_new * (S / a_new.sum())
    else:
        a_new = np.ones(n) * S / n
    return a_new

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization:
- Adaptive subgradient descent with growing top-k lag weighting
- Spectral balancing (Gerchberg-Saxton style) for diversification
- Ultra-high beta LSE polishing for final convergence

Key improvements: higher LSE beta, more lags, decreasing noise, better time allocation.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

    n = len(a)
    k = min(k, 2 * n - 1)

    c_min_top = c[top_idx].min()
    c_range = c_max - c_min_top
    if c_range < 1e-30:
        weights = np.ones(k) / k

        if j_start <= j_end:

            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]

def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    diff = c - cm
    if diff.max() <= 0:
        return np.zeros(n)
    scaled = beta * diff
    if scaled.max() > 350:
        beta = 350.0 / diff.max()
        scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def spectral_balance(a, alpha=0.6, S=None):
    """Gerchberg-Saxton style: compress spectral dynamic range, project back."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = mag ** alpha
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    if S is None:
        S = a.sum()
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new * (S / s)
    else:
        a_new = np.ones(n) * S / n
    return a_new

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 150
    spectral_used = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Adaptive k: grow from 16 to 300 lags
        k = int(16 + 284 * progress ** 0.6)
        # Adaptive gamma: sharper peak weighting over time
        gamma = gamma_init + (gamma_final - gamma_init) * progress

        grad = compute_subgradient(a, c, k, gamma=gamma)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        gap = cur_max - c.mean()
        if gap < 1e-13 * max(cur_max, 1.0):
            break

        # Step size
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        max_step = 10.0 / (gnorm + 1e-30)
        step = min(step, max_step)
        # In late stages, use smaller steps
        if progress > 0.7:
            step *= (1.0 - 0.5 * (progress - 0.7) / 0.3)

        # Line search with backtracking
        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-13):
                a = a_try
                accepted = True
                break
            step *= 0.55

        if not accepted:
            step = 0.0005 / (gnorm + 1e-30)
            a = project_simplex(a - step * grad, S)

        # Handling stagnation
        if stuck > patience:
            # Decrease noise amplitude as progress increases
            noise_base = 0.005 * max(0.1, 1.0 - progress * 0.8)
            if spectral_used < 8 and rng.random() < 0.4 and progress < 0.8:
                alpha = 0.5 + 0.3 * rng.random()
                a = spectral_balance(a, alpha=alpha, S=S)
                spectral_used += 1
            else:
                noise_scale = noise_base * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.15)

        if verbose and it % 800 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, max={best:.4f}, k={k}")

    return best_a, best


def optimize_lse(a0, S, max_time, rng, beta_init=200.0, beta_max=100000.0, verbose=False):
    """
    LSE-smoothed gradient polishing with very high beta for final convergence.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        # Exponential beta growth
        beta = min(beta_max, beta_init * np.exp(0.006 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 200:
            # Very small noise to escape
            noise_scale = 0.001 * S / np.sqrt(n) * max(0.05, 1.0 - progress)
            if noise_scale > 1e-15:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        if c.max() - c.mean() < 1e-14 * max(c.max(), 1.0):
            break

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: conservative for LSE
        step = 1.0 / (2.5 * beta * (1.0 + progress))
        step = min(step, 4.0 / (gnorm + 1e-30))

        # Line search
        accepted = False
        for _ in range(10):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-14):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - 0.2 * step * grad, S)

        if verbose and it % 500 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      LSE it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")

    return best_a, best


def fast_polish(a0, S, max_time, rng, verbose=False):
    """
    Rapid final polish: alternate tiny subgradient steps with LSE at high beta.
    Designed to squeeze out last bits of improvement.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        if cur_max - c.mean() < 1e-14 * max(cur_max, 1.0):
            break

        # Alternate: odd iterations use subgradient, even use LSE
        if it % 2 == 1:
            k = int(50 + 200 * progress)
            gamma = 8.0 + 10.0 * progress
            grad = compute_subgradient(a, c, k, gamma=gamma)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 1.0 * (cur_max - c.mean()) / (gnorm * gnorm + 1e-30)
            step = min(step, 4.0 / (gnorm + 1e-30))
        else:
            beta = 5000.0 * (1.0 + 5.0 * progress)
            grad = compute_lse_gradient(a, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = 1.0 / (3.0 * beta)
            step = min(step, 3.0 / (gnorm + 1e-30))

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-14):
            a = a_try
        else:
            a_try = project_simplex(a - 0.4 * step * grad, S)
            a = a_try

        if verbose and it % 2000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      Polish it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phases: subgradient -> LSE -> alternate lengths -> final ultra-polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0.1 and len(a_provided) >= 128:
            print(f"Provided: len={len(a_provided)}, obj={obj_value(a_provided):.6f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  ** NEW BEST: obj={obj:.6f} ({msg})")
        return obj

    # === Phase 1: Subgradient on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(700, (deadline - time.monotonic()) * 0.30)
        print(f"\nPhase 1: Subgradient n={n1}, alloc={t1:.0f}s")
        a_opt, _ = optimize_subgradient(a1, S1, t1, rng)
        update_global(a_opt, "P1 subgradient")

        # LSE polish
        remaining = deadline - time.monotonic()
        t1b = min(250, remaining * 0.18)
        if t1b > 30:
            print(f"Phase 1b: LSE polish, alloc={t1b:.0f}s")
            a_pol, _ = optimize_lse(a_opt, S1, t1b, rng, beta_init=500, beta_max=80000)
            update_global(a_pol, "P1b LSE")
            # Fast polish
            rem2 = deadline - time.monotonic()
            t1c = min(100, rem2 * 0.1)
            if t1c > 15:
                a_fp, _ = fast_polish(a_pol, S1, t1c, rng)
                update_global(a_fp, "P1c fast polish")

    # === Phase 2: Try n=16384 ===
    remaining = deadline - time.monotonic()
    if global_a is not None and len(global_a) != 16384:
        n2 = 16384
        if remaining > 200:
            S2 = np.sqrt(2.0 * n2)
            t2 = min(350, remaining * 0.25)
            print(f"\nPhase 2: n={n2}, alloc={t2:.0f}s")

            # Interpolate from best
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n2)
            init2 = np.interp(t_new, t_old, global_a)
            init2 = np.maximum(init2, 0.0)
            if init2.sum() > 0:
                init2 = init2 * (S2 / init2.sum())
            else:
                init2 = np.ones(n2) * S2 / n2

            a_opt2, _ = optimize_subgradient(init2, S2, t2 * 0.65, rng, verbose=False)
            update_global(a_opt2, "P2 n=16384 subgrad")

            rem2 = deadline - time.monotonic()
            if rem2 > 30:
                a_pol2, _ = optimize_lse(a_opt2, S2, min(150, rem2 * 0.4), rng,
                                         beta_init=500, beta_max=50000)
                update_global(a_pol2, "P2 n=16384 LSE")

    # === Phase 3: Alternate length (4096 if not tried) ===
    remaining = deadline - time.monotonic()
    tried_lengths = set()
    if global_a is not None:
        tried_lengths.add(len(global_a))
    if a_provided is not None:
        tried_lengths.add(len(a_provided))

    if 4096 not in tried_lengths and remaining > 100:
        n3 = 4096
        S3 = np.sqrt(2.0 * n3)
        t3 = min(200, remaining * 0.15)
        print(f"\nPhase 3: n={n3}, alloc={t3:.0f}s")
        if global_a is not None and len(global_a) > 1:
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n3)
            init3 = np.interp(t_new, t_old, global_a)
        else:
            tt = np.linspace(0, 1, n3, endpoint=False)
            init3 = 1.0 + 0.5 * np.cos(2 * np.pi * tt ** 2)
        init3 = np.maximum(init3, 0.0)
        if init3.sum() > 0:
            init3 = init3 * (S3 / init3.sum())
        else:
            init3 = np.ones(n3) * S3 / n3

        a_opt3, _ = optimize_subgradient(init3, S3, t3 * 0.7, rng, verbose=False)
        update_global(a_opt3, "P3 n=4096")

    # === Phase 4: Intensive final refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t4 = remaining - 15
        print(f"\nPhase 4: Final refinement n={n}, alloc={t4:.0f}s")

        a_curr = global_a.copy()
        # Split into alternating subgradient and LSE rounds
        n_rounds = 4
        t_per_round = t4 / n_rounds

        for ri in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            t_r = min(t_per_round, deadline - time.monotonic() - 5)
            if t_r < 5:
                break

            if ri % 2 == 0:
                a_curr, _ = optimize_subgradient(a_curr, S, t_r, rng, verbose=False,
                                                 gamma_init=8.0, gamma_final=15.0)
            else:
                a_curr, _ = optimize_lse(a_curr, S, t_r, rng,
                                         beta_init=5000, beta_max=200000)
            update_global(a_curr, f"P4 round {ri}")

        # Ultra-fast polish at the very end
        rem_final = deadline - time.monotonic()
        if rem_final > 15:
            a_fp, _ = fast_polish(global_a, S, min(60, rem_final - 5), rng)
            update_global(a_fp, "P4 final fast polish")

    # === Phase 5: Spectral balance restart ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print(f"\nPhase 5: Spectral restarts, remaining={remaining:.0f}s")
        for ri in range(2):
            if time.monotonic() > deadline - 10:
                break
            alpha = 0.4 + 0.3 * rng.random()
            a_spec = spectral_balance(global_a, alpha=alpha, S=S)
            t_r = min(80, (deadline - time.monotonic() - 10) / (2 - ri))
            if t_r < 5:
                break
            a_r, _ = optimize_subgradient(a_spec, S, t_r, rng, verbose=False,
                                          gamma_init=5.0, gamma_final=12.0)
            update_global(a_r, f"P5 spectral restart {ri}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No solution found, using fallback")
        global_a = np.ones(256)
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, obj={global_best:.6f}, verify={ver:.6f}")
    print(f"{'='*60}")

    return final_list
