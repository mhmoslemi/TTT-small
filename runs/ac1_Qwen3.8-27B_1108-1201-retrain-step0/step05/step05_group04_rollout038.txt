"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length 8192.
Strategy: Adaptive LSE-FISTA with pairwise coordinate descent, active-set
refinement, and multi-phase time allocation. Key innovations over previous:
1. Pairwise coordinate descent for surgical peak reduction.
2. Adaptive beta schedule based on peak flatness.
3. Dual-mode FISTA (momentum + dual averaging) with smart restarts.
4. Coarse-to-fine warm start from lower dimensions.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, n_active=50):
    """Gradient using hard active set (top n_active lags)."""
    n = len(a)
    cm = c.max()
    threshold = cm - 1.0 / max(1, n_active)
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=20.0, beta_end=10000.0, beta_steps=200,
                  momentum_restart_gap=300, use_dual_avg=False):
    """
    FISTA with adaptive LSE smoothing and optional dual averaging.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    dual_avg = a.copy()
    dual_count = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: based on peak flatness
        c_y = fft_convolve_self(y)
        cur_true_max = c_y.max()

        # Compute beta based on flatness of top lags
        if it > 50:
            top5 = np.sort(c_y)[::-1][:5]
            flatness = (top5[0] - top5[-1]) / max(1e-15, top5[0])
            # If peak is flat, use higher beta to sharpen
            frac = min(1.0, it / max(1, beta_steps))
            base_beta = beta_start * (beta_end / beta_start) ** frac
            if flatness < 0.01:
                beta = min(50000.0, base_beta * 2.0)
            else:
                beta = min(50000.0, base_beta)
            beta = max(10.0, beta)
        else:
            frac = min(1.0, it / max(1, beta_steps))
            beta = beta_start * (beta_end / beta_start) ** frac
            beta = max(10.0, min(50000.0, beta))

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        # Compute LSE gradient
        grad, lse_val = lse_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = 1.0 / beta

        # Stagnation handling with adaptive noise
        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            step *= 0.5
            last_improve_it = it

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Dual averaging
        if use_dual_avg:
            dual_avg = dual_avg + (x_new - dual_avg) / dual_count
            dual_count += 1.0

        # Periodic check
        if it % 200 == 0:
            c_check = fft_convolve_self(y)
            if c_check.max() < best:
                best = c_check.max()
                best_a = y.copy()
                last_improve_it = it
            if use_dual_avg:
                c_davg = fft_convolve_self(dual_avg)
                if c_davg.max() < best:
                    best = c_davg.max()
                    best_a = dual_avg.copy()
                    last_improve_it = it
                    y = dual_avg.copy()
                    a = dual_avg.copy()

    # Final
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def pairwise_coordinate_descent(a_init, n, S, max_time, rng,
                                 n_active_lags=10, max_rounds_per_iter=5):
    """
    Pairwise coordinate descent: at each step, find the peak lag, then
    transfer mass from high-contribution positions to low-contribution ones.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it
        elif it - last_improve > 200:
            # Decay step scale and add noise
            step_scale *= 0.5
            noise = 0.001 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise > 1e-12:
                a = project_simplex(a + rng.normal(0, noise, n), S)
            last_improve = it

        # Find top lags
        sorted_lags = np.argsort(c)[::-1][:n_active_lags]

        for _ in range(max_rounds_per_iter):
            if time.monotonic() - start > max_time:
                break

            # For each active lag, compute contribution of each a[j]
            contrib = np.zeros(n)
            for k in sorted_lags:
                # c[k] = sum_j a[j]*a[k-j]
                # ∂c[k]/∂a[m] = 2*a[k-m] if 0 <= k-m < n
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                for m in range(lo, hi + 1):
                    contrib[m] += 2.0 * a[k - m]

            # Find best pair to swap: high contrib to reduce, low contrib to increase
            # We want to decrease a[m_hi] (high contrib) and increase a[m_lo] (low contrib)
            # Subject to: a[m_lo] > 0 (so we can increase it) - actually we just need a[m_hi] > 0

            m_hi = np.argmax(contrib)
            # For m_lo, we want low contrib AND we want to be able to increase it
            # (it should be small enough that increasing it doesn't create a new peak)
            min_contrib = contrib.min()
            # Pick from among elements with low contribution
            low_mask = contrib < 0.3 * contrib.max()
            if low_mask.sum() > 0:
                candidates = np.where(low_mask)[0]
                m_lo = candidates[np.argmin(a[candidates])]  # pick the smallest one to increase
            else:
                m_lo = np.argmin(contrib)

            if m_hi == m_lo:
                continue

            # Amount to transfer: limited by a[m_hi]
            max_transfer = 0.5 * a[m_hi]  # don't reduce below 0
            # Scale by step_size
            transfer = min(max_transfer, 0.01 * S / n * step_scale)

            if transfer < 1e-15:
                continue

            a[m_hi] -= transfer
            a[m_lo] += transfer
            a = np.maximum(a, 0.0)

            # Renormalize
            s = a.sum()
            if s > 1e-15:
                a *= S / s

        # Every 50 iterations, do a full active-set gradient step
        if it % 50 == 0:
            grad = active_set_gradient(a, c, n_active=n_active_lags)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-10:
                step = 0.001 * S / (n * gnorm) * step_scale
                a = project_simplex(a - step * grad, S)

    # Final
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def active_set_refinement(a_init, n, S, max_time, rng, n_active=20):
    """
    Active-set refinement with backtracking line search.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = active_set_gradient(a, c, n_active=n_active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 0.01
        base_obj = cur_max
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            test_obj = c_test.max()
            if test_obj < base_obj * (1 - 1e-8):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            step = 0.0005 / (1 + it * 0.01)
            a = project_simplex(a - step * grad, S)

        # Periodic perturbation
        if it % 500 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def coarse_to_fine_warmstart(a_target, n_target, S_target, max_time, rng):
    """
    Try to improve by optimizing a shorter version and interpolating back.
    Only useful if the shorter version finds a better basin.
    """
    # Try n = n_target / 8
    n_short = max(256, n_target // 8)
    # Downsample
    indices = np.linspace(0, n_target - 1, n_short).astype(int)
    a_short = a_target[indices]
    S_short = np.sqrt(2.0 * n_short)
    a_short = normalize_sum(a_short, n_short)

    # Optimize briefly
    t_opt = min(30.0, max_time * 0.3)
    a_short_opt, _ = run_lse_fista(a_short, n_short, S_short, t_opt, rng,
                                    beta_start=50.0, beta_end=5000.0, beta_steps=50)

    # Interpolate back
    t_old = np.linspace(0, 1, n_short)
    t_new = np.linspace(0, 1, n_target)
    a_interpolated = np.interp(t_new, t_old, a_short_opt)
    a_interpolated = normalize_sum(a_interpolated, n_target)

    return a_interpolated
