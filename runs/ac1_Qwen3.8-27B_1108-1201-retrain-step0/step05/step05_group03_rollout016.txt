"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Key new approach: LP-based iteration (the SOTA method described in the problem).
For a fixed iterate f, solve the LP: max sum(g) s.t. f*g <= M, g >= 0.
This gives a principled descent direction that gradient-only methods lack.
Combined with full-LSE gradient polishing and multi-restart.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def full_lse_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*(a*a)_k))).
    Returns (gradient, lse_value).
    """
    n = len(a)
    c = fft_convolve_self(a)  # length 2n-1
    c_max = c.max()
    scaled = beta * (c - c_max)
    if scaled.max() > 500:
        scaled = 500.0
    e = np.exp(scaled)
    tot = e.sum()
    p = e / tot  # softmax weights, length 2n-1
    lse_val = c_max + np.log(max(tot, 1e-300)) / beta

    # Gradient: grad_j = 2 * sum_l p_{j+l} * a_l for j=0,...,n-1
    # This is the cross-correlation computed via FFT
    a_flip = a[::-1]  # length n
    # Full convolution of p (len 2n-1) and a_flip (len n) has length 3n-2
    L = 1 << (3 * n - 2 - 1).bit_length()
    P = fft(p, L)
    AF = fft(a_flip, L)
    conv = np.real(ifft(P * AF))
    # We want conv[j + n - 1] for j = 0, ..., n-1, i.e., indices n-1 to 2n-2
    q = conv[n - 1:2 * n - 1]
    grad = 2.0 * q
    return grad, lse_val


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def build_conv_matrix(f, n):
    """Build sparse matrix C where C @ g = f * g (linear convolution)."""
    n_rows = 2 * n - 1
    # Column i has nonzeros at rows i, i+1, ..., min(i+n, n_rows)-1
    # with values f[0], f[1], ..., f[min(n, n_rows-i)-1]
    col_lengths = np.minimum(n, n_rows - np.arange(n))
    total = int(col_lengths.sum())

    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    data = np.zeros(total, dtype=np.float64)

    idx = 0
    for i in range(n):
        L = int(col_lengths[i])
        if L > 0:
            rows[idx:idx + L] = np.arange(i, i + L, dtype=np.int32)
            cols[idx:idx + L] = i
            data[idx:idx + L] = f[:L]
            idx += L

    return csr_matrix((data, (rows, cols)), shape=(n_rows, n))


def solve_lp_step(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns g or None if failed.
    """
    C = build_conv_matrix(f, n)

    # linprog minimizes c^T x, so negate objective
    res = linprog(
        -np.ones(n),
        A_ub=C,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 60}
    )

    if res.success:
        g = res.x
        g = np.maximum(g, 0.0)
        if g.sum() > 1e-15:
            return g
    return None


def run_lp_iteration(a, S, max_time, rng, n_lp=None, max_lp_iters=None):
    """
    LP-based optimization loop.
    At each step: solve LP, rescale, line search on convex combination.
    """
    n = len(a)
    if n_lp is None:
        n_lp = n
    a = a.copy()
    a = normalize(a, S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    if max_lp_iters is None:
        max_lp_iters = 200

    while time.monotonic() - start < max_time and it < max_lp_iters:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (a.sum() ** 2)
        if M < best:
            best = M
            best_a = a.copy()

        # Solve LP
        t_lp_start = time.monotonic()
        g_star = solve_lp_step(a, M, n)
        t_lp = time.monotonic() - t_lp_start

        if g_star is None:
            # LP failed, use gradient step instead
            beta = 50.0 + it * 5.0
            grad = topk_lse_gradient(a, c, max(4, int(0.05 * (2 * n - 1))), beta)
            step = 1.0 / (beta * 0.1)
            a = project_simplex(a - step * grad, S)
            continue

        # Rescale g to have same sum as a
        g_sum = g_star.sum()
        if g_sum < 1e-15:
            break
        g_rescaled = g_star * (S / g_sum)

        best_obj = cur_obj

            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (s_try * s_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g_rescaled
            a = np.maximum(a, 0.0)
            a = normalize(a, S)
        else:
            # No improvement found; try a small gradient step to escape
            beta = 30.0
            grad = topk_lse_gradient(a, c, max(4, int(0.08 * (2 * n - 1))), beta)
            step = 0.5 / beta
            a = project_simplex(a - step * grad, S)

        # Print progress
        if it % 10 == 0:
            cur_m = fft_convolve_self(a).max()
            cur_o = 2.0 * n * cur_m / (a.sum() ** 2)
            print("  LP iter %d: obj=%.6f, M=%.4f, t_lp=%.1fs, elapsed=%.0fs" %
                  (it, cur_o, cur_m, t_lp, time.monotonic() - start))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_fista_lse(a_init, S, max_time, rng,
                  beta_init=20.0, beta_rate=0.003, beta_max=5000.0,
                  k_frac=0.08, use_momentum=True, use_full_lse=False):
    """FISTA with top-k or full LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    last_stuck_check = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            if it - last_stuck_check > 500:
                stuck += 1
                last_stuck_check = it

        if use_full_lse:
            grad, _ = full_lse_gradient(y, beta)
        else:
            grad = topk_lse_gradient(y, c, k, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking
        step = 1.0 / max(2.0 * beta * 0.05, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart
        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
            last_stuck_check = it

        # Spectral perturbation
        if it % 3000 == 0 and rng.random() < 0.15:
            n_sp = len(y)
            fsize = 1 << (2 * n_sp - 1).bit_length()
            A = fft(y, fsize)
            phase = rng.normal(0, 0.05, fsize)
            phase[0] = 0
            A_new = A * np.exp(1j * phase)
            y = np.real(ifft(A_new, fsize))[:n_sp]
            y = np.clip(y, 0, None)
            y = normalize(y, S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_polish_high_beta(a_init, S, max_time, rng):
    """Very high beta polishing with careful adaptive steps."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    k = max(4, int(0.02 * (2 * n - 1)))
    it = 0
    step_scale = 1.0
    consecutive_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 500.0 * (1.0 + it / 100.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        base_step = 0.8 / (1.0 + 0.005 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            # Try smaller step
            a_new2 = project_simplex(a - step * 0.1 * grad, S)
            a = a_new2

        # Restart if stuck
        if consecutive_no_improve > 5000:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            consecutive_no_improve = 0
            step_scale = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase = rng.normal(0, amount, fsize)
    phase[0] = 0
    A_new = A * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (a.sum() / s)
    return a_new


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Strategy: LP-based iteration (new) + FISTA LSE + high-beta polish.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs, seed=%d" % (budget_s, seed))

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, n):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))

    # === Phase 1: LP-based optimization ===
    # Try n=8192 first (matching previous best), fall back to 4096 if too slow
    n_primary = 8192
    S_primary = np.sqrt(2.0 * n_primary)

    if given is not None and len(given) != n_primary:
        a_start_primary = resample_sequence(given, n_primary)
        a_start_primary = normalize(a_start_primary, S_primary)
    elif given is not None:
        a_start_primary = normalize(given.copy(), S_primary)
    else:
        a_start_primary = np.ones(n_primary) * S_primary / n_primary

    print("\n=== Phase 1: LP-based optimization (n=%d) ===" % n_primary)
    t_lp_budget = 600  # seconds for LP phase
    remaining = deadline - time.monotonic()
    t_lp_use = min(t_lp_budget, remaining * 0.35)

    if t_lp_use > 30:
        a_lp, M_lp = run_lp_iteration(a_start_primary, S_primary, t_lp_use, rng,
                                       max_lp_iters=150)
        update_global(a_lp, n_primary)
        obj_lp = 2.0 * n_primary * M_lp / (a_lp.sum() ** 2)
        print("  LP phase done: obj=%.6f" % obj_lp)
        best_a = a_lp.copy()
    else:
        best_a = a_start_primary.copy()

    # === Phase 2: FISTA with LSE (top-k) ===
    remaining = deadline - time.monotonic()
    t_fista = min(remaining * 0.30, 500)
    if t_fista > 20:
        print("\n=== Phase 2: FISTA top-k LSE (n=%d) ===" % n_primary)
        a_opt, M_opt = run_fista_lse(best_a, S_primary, t_fista, rng,
                                      beta_init=20.0, beta_rate=0.003,
                                      beta_max=3000.0, k_frac=0.08)
        update_global(a_opt, n_primary)
        if M_opt < M_lp if 'M_lp' in dir() else True:
            best_a = a_opt.copy()
        obj_f = 2.0 * n_primary * M_opt / (a_opt.sum() ** 2)
        print("  FISTA done: obj=%.6f" % obj_f)

    # === Phase 3: FISTA with full LSE ===
    remaining = deadline - time.monotonic()
    t_fista2 = min(remaining * 0.20, 300)
    if t_fista2 > 20:
        print("\n=== Phase 3: FISTA full LSE (n=%d) ===" % n_primary)
        a_opt2, M_opt2 = run_fista_lse(best_a, S_primary, t_fista2, rng,
                                        beta_init=10.0, beta_rate=0.005,
                                        beta_max=2000.0, k_frac=0.05,
                                        use_full_lse=True)
        update_global(a_opt2, n_primary)
        obj_f2 = 2.0 * n_primary * M_opt2 / (a_opt2.sum() ** 2)
        print("  Full-LSE FISTA done: obj=%.6f" % obj_f2)
        if M_opt2 < (M_opt if 'M_opt' in dir() else M_lp):
            best_a = a_opt2.copy()

    # === Phase 4: High-beta polish ===
    remaining = deadline - time.monotonic()
    t_polish = min(remaining * 0.25, 400)
    if t_polish > 15:
        print("\n=== Phase 4: High-beta polish (n=%d) ===" % n_primary)
        a_pol, M_pol = run_polish_high_beta(best_a, S_primary, t_polish, rng)
        update_global(a_pol, n_primary)
        obj_p = 2.0 * n_primary * M_pol / (a_pol.sum() ** 2)
        print("  Polish done: obj=%.6f" % obj_p)
        if M_pol < (M_opt2 if 'M_opt2' in dir() else (M_opt if 'M_opt' in dir() else M_lp)):
            best_a = a_pol.copy()

    # === Phase 5: Multi-restart exploration ===
    remaining = deadline - time.monotonic()
    n_restarts = max(1, int(remaining / 80))
    n_restarts = min(n_restarts, 8)
    print("\n=== Phase 5: Multi-restart (%d restarts) ===" % n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        t_r = min(remaining * 0.8, 120)

        # Different initialization for each restart
        if r == 0 and global_a is not None:
            a_r = global_a.copy()
            # Perturb
            a_r = spectral_perturb(a_r, 0.15, rng)
            a_r = normalize(a_r, S_primary)
        elif r == 1 and given is not None:
            a_r = resample_sequence(given, n_primary) if len(given) != n_primary else given.copy()
            a_r = normalize(a_r, S_primary)
            a_r = spectral_perturb(a_r, 0.3, rng)
            a_r = normalize(a_r, S_primary)
        elif r == 2:
            # Random with structure
            freq = rng.uniform(-1, 1, n_primary)
            a_r = np.abs(fft(np.concatenate([freq, np.zeros(n_primary)])))[:n_primary]
            a_r = np.clip(a_r, 0, None)
            a_r = normalize(a_r, S_primary)
        else:
            # Perturb current best
            a_r = best_a.copy()
            noise = rng.normal(0, 0.05 / np.sqrt(n_primary), n_primary)
            a_r = a_r + S_primary * noise
            a_r = np.clip(a_r, 0, None)
            a_r = normalize(a_r, S_primary)

        # Quick FISTA
        a_opt_r, M_opt_r = run_fista_lse(a_r, S_primary, t_r, rng,
                                          beta_init=30.0, beta_rate=0.004,
                                          beta_max=2000.0, k_frac=0.06)
        update_global(a_opt_r, n_primary)
        obj_r = 2.0 * n_primary * M_opt_r / (a_opt_r.sum() ** 2)
        print("  Restart %d: obj=%.6f" % (r, obj_r))
        if global_a is not None and M_opt_r < fft_convolve_self(best_a).max():
            best_a = a_opt_r.copy()

    # === Phase 6: Try n=16384 if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_secondary = 16384
        S_secondary = np.sqrt(2.0 * n_secondary)
        a_sec = resample_sequence(global_a, n_secondary)
        a_sec = normalize(a_sec, S_secondary)
        t_sec = min(remaining * 0.4, 200)
        print("\n=== Phase 6: Refine at n=%d ===" % n_secondary)
        a_opt_s, M_opt_s = run_fista_lse(a_sec, S_secondary, t_sec, rng,
                                          beta_init=30.0, beta_rate=0.002,
                                          beta_max=2000.0, k_frac=0.05)
        update_global(a_opt_s, n_secondary)
        obj_s = 2.0 * n_secondary * M_opt_s / (a_opt_s.sum() ** 2)
        print("  n=%d: obj=%.6f" % (n_secondary, obj_s))

        # Polish at n_secondary
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_pol_s, M_pol_s = run_polish_high_beta(a_opt_s, S_secondary,
                                                     min(remaining * 0.7, 100), rng)
            update_global(a_pol_s, n_secondary)
            obj_ps = 2.0 * n_secondary * M_pol_s / (a_pol_s.sum() ** 2)
            print("  Polish n=%d: obj=%.6f" % (n_secondary, obj_ps))

    # === Final: verify and return ===
    print("\n=== Final Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        n_final = len(global_a)
        s_final = global_a.sum()
        if s_final < 0.01:
            print("  Sum too small, using fallback")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            n_final = 256
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Length: %d" % len(final_list))
        print("  Evaluate: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
        print("  Final answer: %.6f" % ver)
    else:
        print("  No valid solution found, using fallback")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Total time: %.1fs" % elapsed)
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
