import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    """Compute self-convolution via FFT."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]

def compute_conv_with_fft(a, A_cache, N):
    """Compute self-convolution using pre-allocated FFT array."""
    n = len(a)
    A_cache[:n] = a
    A_cache[n:] = 0.0
    A = fft(A_cache)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]

def lse_value_and_grad(c, beta):
    """Compute LSE value and gradient weights."""
    c_max = c.max()
    if beta < 1e-6:
        return c_max, np.ones(len(c)) / len(c)
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return lse_val, w

    """Compute LSE gradient with respect to a."""
    fsize = N
    W = fft(w, fsize)  # ifft of the weight pattern

    # Wait no: C[k] = sum_l a[l]*a[k-l]. When l=j, term is a[j]*a[k-j]. When l=k-j, term is a[k-j]*a[j].

def lse_step(a, N, n, S, beta, step_size):
    """One LSE gradient step with simplex projection."""
    c = compute_conv(a, N)
    lse_val, w = lse_value_and_grad(c, beta)
    grad = lse_gradient(a, c, w, N, n)
    a_new = project_simplex(a - step_size * grad, S)
    c_new = compute_conv(a_new, N)
    return a_new, c_new.max(), c_new

def deflation_step(a, c, n, S, top_k=10, frac=0.01):
    """One deflation step: transfer mass from high-contribution to low-contribution positions."""
    # Find top-k peaks
    peak_idx = np.argsort(c)[::-1][:top_k]
    peak_vals = c[peak_idx]
    
    # Compute contribution of each position to the peaks
    contrib = np.zeros(n)
    for p in peak_idx:
        weight = peak_vals[p] / peak_vals[0]  # normalize by largest peak
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        for j in range(lo, hi + 1):
            k = p - j
            if 0 <= k < n:
                contrib[j] += weight * a[k]
    
    if contrib.max() < 1e-20:
        return a
    
    # Transfer from high-contrib to low-contrib positions
    n_transfer = max(1, n // 100)
    top_positions = np.argsort(contrib)[::-1][:n_transfer]
    bot_positions = np.argsort(contrib)[:n_transfer]
    
    total_to_transfer = frac * a[top_positions].sum()
    a_new = a.copy()
    for idx in range(n_transfer):
        amt = total_to_transfer / n_transfer
        i = top_positions[idx]
        j = bot_positions[idx]
        transfer = min(amt, 0.5 * a_new[i])
        a_new[i] -= transfer
        a_new[j] += transfer
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    return a_new

def deflation_step_vectorized(a, c, n, S, top_k=10, frac=0.01, n_transfer_frac=0.01):

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:

def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def spectral_restart(a, N, rng, frac=1.0):
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    if frac < 1.0:
        phase = np.angle(A)
        n_rand = max(1, int(N * frac))
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    else:
        phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

def coordinate_descent_polish(a, n, S, N, max_time, rng):
    """
    Coordinate descent: for each pair (i, j), find the optimal transfer.
    """
    a = safe_normalize(a, n, S)
    c = compute_conv(a, N)
    best_a = a.copy()
    best_cmax = c.max()
    
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax - 1e-14:
            best_cmax = c_max
            best_a = a.copy()
        
        # Find the peak position
        p = int(np.argmax(c))
        
        # Compute sensitivity: how much does C[p] depend on each a[j]?
        # C[p] = sum_j a[j]*a[p-j], so dC[p]/da[j] = a[p-j] + a[p-j] = 2*a[p-j] (approx)
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        sens = np.zeros(n)
        js = np.arange(lo, hi + 1)
        sens[js] = a[p - js]
        
        # Find i with highest sensitivity and j with lowest
        if sens.max() < 1e-20:
            break
        
        i = int(np.argmax(sens + a))  # combine sensitivity with current value
        j_candidates = np.where(sens < sens.mean())[0]
        if len(j_candidates) == 0:
            j = int(np.argmin(sens))
        else:
            j = int(j_candidates[rng.integers(len(j_candidates))])
        
        if i == j:
            continue
        
        # Binary search for optimal transfer amount
        # We want to reduce C[p] by transferring from i to j
        # C[p] as function of transfer d: C[p] - 2*d*a[p-i] + 2*d*a[p-j] (approx)
        # But we also need to check other peaks
        # For now, just do a small transfer
        max_transfer = 0.05 * a[i]
        if max_transfer < 1e-15:
            continue
        
        # Try a few transfer amounts
        best_d = 0
        best_obj = c_max
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            d = frac * max_transfer
            a_trial = a.copy()
            a_trial[i] -= d
            a_trial[j] += d
            if a_trial[i] < 0:
                continue
            c_trial = compute_conv(a_trial, N)
            obj = c_trial.max()
            if obj < best_obj:
                best_obj = obj
                best_d = d
        
        if best_d > 0:
            a[i] -= best_d
            a[j] += best_d
            a = np.maximum(a, 0)
            s = a.sum()
            a = a * (S / s)
        
        # Occasionally shuffle which peak we focus on
        if it % 50 == 0:
            top_peaks = np.argsort(c)[::-1][:20]
            p = int(top_peaks[rng.integers(len(top_peaks))])
    
    return best_a, best_cmax

def propose_candidate(seed=42, budget_s=2400, **kwargs):

"""
Minimize 2*n*max_conv/sum^2 using hybrid active-set peak reduction + LSE + coordinate descent.
Key improvements over previous:
1. Active-set peak tracking with adaptive epsilon for focused gradient
2. Vectorized deflation with predictive transfer amounts

4. Adaptive time-sharing between strategies based on convergence rate
5. Coordinate descent polishing with multi-peak awareness

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    c[:2 * n - 1] = np.maximum(c[:2 * n - 1], 0.0)
    return c[:2 * n - 1]


def conv_gradient_at_peaks(a, c, peak_idx, n, N):
    """
    Compute gradient of C[peak] with respect to a, averaged over peak_idx.
    grad[j] = 2 * sum_{p in peak_idx} w_p * a[p-j] (with bounds)
    """
    n_peaks = len(peak_idx)
    w_full = np.zeros(N)
    cmax = c[peak_idx[0]] if c[peak_idx[0]] > 0 else 1.0
    for i, p in enumerate(peak_idx):
        w = c[peak_idx[i]] / cmax if cmax > 0 else 1.0
        w_full[int(p)] = w
    a_full = np.zeros(N)
    a_full[:n] = a
    W = fft(w_full)
    A = fft(a_full)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def lse_weights(c, beta):
    """Compute LSE weights."""
    c_max = c.max()
    if beta < 1e-8:
        return c_max, np.ones(len(c)) / len(c)
    shifted = beta * (c - c_max)
    exp_bc = np.exp(np.minimum(shifted, 0.0))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    return lse_val, w


def lse_gradient(a, w, n, N):
    """Compute gradient using precomputed LSE weights."""
    w_full = np.zeros(N)
    w_full[:len(w)] = w
    a_full = np.zeros(N)
    a_full[:n] = a
    W = fft(w_full)
    A = fft(a_full)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def deflation_step(a, c, n, S, top_k=20, frac=0.008, n_transfer_frac=0.02):
    """Vectorized deflation: transfer mass from high-peak-contributing to low-contributing positions."""
    peak_idx = np.argsort(c)[::-1][:top_k]
    cmax = c[peak_idx[0]]
    if cmax < 1e-20:
        return a
    
    contrib = np.zeros(n)
    for p in peak_idx:
        weight = c[p] / cmax
        lo = max(0, int(p) - n + 1)
        hi = min(n - 1, int(p))
        js = np.arange(lo, hi + 1)
        ks = int(p) - js
        contrib[js] += weight * a[ks]
    
    cmax_contrib = contrib.max()
    if cmax_contrib < 1e-20:
        return a
    
    n_transfer = max(1, int(n * n_transfer_frac))
    order = np.argsort(contrib)
    top_pos = order[::-1][:n_transfer]
    bot_pos = order[:n_transfer]
    
    total_mass = a[top_pos].sum() * frac
    amt_per = total_mass / n_transfer
    a_new = a.copy()
    transfer = np.minimum(amt_per, 0.4 * a_new[top_pos])
    a_new[top_pos] -= transfer
    a_new[bot_pos] += transfer
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    return a_new


def dct_smooth(a, n, cutoff_frac=0.1):
    """Apply DCT-based low-pass smoothing."""
    from scipy.fft import dct as scipy_dct, idct as scipy_idct
    A = scipy_dct(a, type=2, n=n)
    cutoff = max(4, int(n * cutoff_frac))
    decay = np.exp(-np.arange(n) / (cutoff * 0.5))
    A_smooth = A * decay
    a_smooth = scipy_idct(A_smooth, type=2, n=n)
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s > 1e-15:
        a_smooth = a_smooth * (a.sum() / s)
    return a_smooth

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    c = compute_conv(a, N)
    best_a = a.copy()
    best_cmax = c.max()
    
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c = compute_conv(a, N)
        cmax = c.max()
        if cmax < best_cmax - 1e-14:
            best_cmax = cmax
            best_a = a.copy()
        
        # Focus on the current maximum peak (or a random one from top-10)
        if it % 30 == 0:
            top_peaks = np.argsort(c)[::-1][:min(10, len(c))]
            p = int(top_peaks[rng.integers(len(top_peaks))])
        else:
            p = int(np.argmax(c))
        
        # Compute sensitivity at this peak
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        if hi <= lo:
            continue
        js = np.arange(lo, hi + 1)
        sens = a[p - js]
        
        if sens.max() < 1e-20:
            break
        
        # Find source (high sens) and target (low sens + low value)
        i_rel = int(np.argmax(sens))
        i = lo + i_rel
        
        # Target: minimize sens at target, prefer positions with low current value
        target_score = sens + 0.1 * a[js]
        j_rel = int(np.argmin(target_score))
        j = lo + j_rel
        
        if i == j:
            continue
        
        # Try transferring different amounts
        max_transfer = 0.08 * a[i]
        if max_transfer < 1e-14:
            continue
        
        best_d = 0.0
        best_obj = cmax
        for frac_try in [0.2, 0.4, 0.6, 0.8]:
            d = frac_try * max_transfer
            a_try = a.copy()
            a_try[i] -= d
            a_try[j] += d
            c_try = compute_conv(a_try, N)
            obj_try = c_try.max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_d = d
        
        if best_d > 0:
            a[i] -= best_d
            a[j] += best_d
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
        
        # Every 100 iterations, also do a small deflation step
        if it % 100 == 0:
            a = deflation_step(a, c, n, S, top_k=10, frac=0.003, n_transfer_frac=0.01)
    
    return best_a, best_cmax


def make_init_uniform(n):
    return np.ones(n)


def make_init_rand(n, rng):
    return rng.random(n) + 0.1


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    for freq_i in range(5):
        freq = 2 * (freq_i + 1)
        a = 0.3 + 0.1 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
        if freq_i == 0:
            base = a
        else:
            base += a
    base += 0.05 * rng.standard_normal(n)
    return np
