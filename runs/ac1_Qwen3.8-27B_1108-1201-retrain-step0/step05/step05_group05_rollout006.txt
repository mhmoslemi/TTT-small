   def toep_solve(t, c_vec, n):
       # t: Toeplitz sequence (t[0], t[1], ..., t[m], 0, ..., 0, t[-m], ..., t[-1])
       # c_vec: right-hand side (length n)
       # Returns x such that T*x = c_vec
       size = 2 * n
       # Build first row of circulant
       w = np.zeros(size)
       for k in range(len(t)):
           w[k % size] = t[k]
       # Pad c_vec
       v = np.zeros(size)
       v[:n] = c_vec
       # Solve
       W = fft(w)
       V = fft(v)
       X = ifft(V / W)
       return np.real(X[:n])
   ```

       # lambda_weights: array of length 2*m+1, representing lambda_{-m}, ..., lambda_m
       # Solve (2*T * a) = mu * 1, then normalize to sum = S

       # t_k = 2 * lambda_k
       t = 2.0 * lambda_weights  # length 2*m+1
       
       # Build circulant first row (length 2n)
       size = 2 * n
       w = np.zeros(size)
       # t[k] corresponds to lag k, where k ranges from -m to m
       # In the circulant, lag k goes to position k (if k >= 0) or size-k (if k < 0)
       for i, k in enumerate(range(-m, m+1)):
           if k >= 0:
               w[k] = t[i]
           else:
               w[size + k] = t[i]
       
       # Right-hand side: mu * 1, but we'll set mu = 1 and normalize later
       v = np.zeros(size)
       v[:n] = 1.0
       
       # Solve
       W = fft(w)
       # Avoid division by zero
       W_safe = np.where(np.abs(W) < 1e-30, 1e-30, W)
       V = fft(v)
       X = ifft(V / W_safe)
       a = np.real(X[:n])
       
       # Normalize
       a = np.maximum(a, 0)  # Project to non-negative
       s = a.sum()
       if s < 1e-15:
           return np.full(n, S / n)
       a = a * (S / s)
       
       return a
   ```
