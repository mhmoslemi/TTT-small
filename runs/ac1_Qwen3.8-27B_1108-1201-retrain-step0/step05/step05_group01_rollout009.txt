import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)

    a = np.maximum(a, 0.01)

def generate_weighted_chirp(n, rng, S, alpha=0.7):
    """Chirp with edge weighting to flatten autocorrelation."""
    t = np.arange(n)

    # Weight towards edges to compensate for fewer terms there

    a = edge_weight * (1.0 + alpha * np.cos(phase))

            pert = pert - pert.mean()
            pert = pert / (np.linalg.norm(pert) + 1e-15)
            a = project_simplex(a + perturb_scale * pert, S)
            improve_count = 0

    return best_a, best_obj


def run_multiresolution(a_init, max_time, rng):
    """
    Try different sequence lengths, optimize at each, and keep the best.
    Larger n gives more degrees of freedom.
    """
    n_base = len(a_init)
    S_base = np.sqrt(2.0 * n_base)
    a_init = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a_init = a_init * (S_base / a_init.sum())

    start = time.monotonic()
    best_obj = compute_obj(a_init)
    best_a = a_init.copy()

    n_candidates = [n_base]
    if n_base >= 4096:
        n_candidates.append(n_base // 2)
    n_candidates.append(n_base * 2)

    time_per_n = max_time / len(n_candidates)

    for n_try in n_candidates:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 30:
            break

        S_try = np.sqrt(2.0 * n_try)

        # Upsample or downsample the initial sequence
        if n_try == n_base:
            a_try = a_init.copy()
        elif n_try > n_base:
            # Upsample via interpolation
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, n_base, endpoint=False)
            x_new = np.linspace(0, 1, n_try, endpoint=False)
            f_interp = interp1d(x_old, a_init, kind='cubic', fill_value='extrapolate')
            a_try = f_interp(x_new)
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S_try / a_try.sum())
        else:
            # Downsample by averaging blocks
            block_size = n_base // n_try
            a_try = a_init[:block_size * n_try].reshape(n_try, block_size).mean(axis=1)
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S_try / a_try.sum())

        # Quick FISTA + L-BFGS-B
        a_opt, _ = run_fista_avg(a_try, time_per_n * 0.6, rng,
                                  beta_start=30.0, use_averaging=True)
        obj_opt = compute_obj(a_opt)
        if time.monotonic() - start < max_time - 15:
            a_opt2, obj_opt2 = run_lbfgs_polish(a_opt, time_per_n * 0.3, rng,
                                                 n_beta_levels=3)
            if obj_opt2 < obj_opt:
                opt_final = a_opt2
                obj_opt = obj_opt2
            else:
                opt_final = a_opt

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = opt_final.copy()
            print("  Multi-res n=%d: obj=%.8f (improved!)" % (n_try, obj_opt))
        else:
            print("  Multi-res n=%d: obj=%.8f" % (n_try, obj_opt))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    1. Load initial solution and evaluate
    2. FISTA with subgradient averaging and adaptive beta (main exploration)
    3. L-BFGS-B polishing (main exploitation)
    4. Chirp-structured restarts for diversity
    5. Multi-resolution search
    6. Final alternating polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.10f" % (n0, obj0))
        if obj0 < global_best_obj:
            global_best_obj = obj0
            global_best_a = a_start.copy()
    else:
        n0 = 8192
        S0 = np.sqrt(2.0 * n0)
        a_start = generate_weighted_chirp(n0, rng, S0, alpha=0.7)
        obj0 = compute_obj(a_start)
        print("Chirp init: n=%d, obj=%.10f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: FISTA with subgradient averaging (30% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.30
    print("\n=== Phase 1: FISTA with subgradient averaging (budget %.0fs) ===" % p1_budget)
    t1 = time.monotonic()

    # Run 1: Standard settings
    a1, _ = run_fista_avg(global_best_a, p1_budget * 0.5, rng,
                          beta_start=20.0, use_averaging=True,
                          adaptive_beta=True, step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  FISTA-avg-1: obj=%.10f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        # Run 2: Higher beta, different averaging rate
        a1b, _ = run_fista_avg(global_best_a, rem1, rng,
                               beta_start=50.0, use_averaging=True,
                               adaptive_beta=True, step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-avg-2: obj=%.10f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: L-BFGS-B polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: L-BFGS-B polishing (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, obj2 = run_lbfgs_polish(global_best_a, p2_budget, rng, n_beta_levels=6)
        print("  L-BFGS done: obj=%.10f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Chirp restarts (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Chirp restarts (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        S3 = np.sqrt(2.0 * n0)
        # Generate diverse chirp-based inits
        inits = []
        inits.append(generate_chirp_init(n0, rng, S3))
        inits.append(generate_weighted_chirp(n0, rng, S3, alpha=0.5))
        inits.append(generate_weighted_chirp(n0, rng, S3, alpha=0.9))
        inits.append(generate_asymmetric_init(n0, rng, S3))
        # Blend with current best
        blend = 0.3 * global_best_a + 0.7 * generate_chirp_init(n0, rng, S3)
        blend = project_simplex(np.maximum(blend, 0), S3)
        inits.append(blend)

        time_per_init = p3_budget / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time_per_init < 10:
                break

            a_c, _ = run_fista_avg(init, time_per_init * 0.7, rng,
                                    beta_start=30.0, use_averaging=True)
            obj_c = compute_obj(a_c)

            # Quick L-BFGS polish
            rem_c = time_per_init - (time.monotonic() - t3 - sum(
                time_per_init * 0.7 for _ in range(idx)))
            if time.monotonic() < deadline - 10:
                a_c2, obj_c2 = run_lbfgs_polish(a_c, min(10.0, time_per_init * 0.25),
                                                 rng, n_beta_levels=2)
                if obj_c2 < obj_c:
                    obj_c = obj_c2
                    a_c = a_c2

            print("  Chirp restart %d: obj=%.10f" % (idx, obj_c))
            if obj_c < global_best_obj:
                global_best_obj = obj_c
                global_best_a = a_c.copy()
                print("    -> New best!")

    # === Phase 4: Multi-resolution (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Multi-resolution (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, obj4 = run_multiresolution(global_best_a, p4_budget, rng)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()
            print("  Multi-res best: obj=%.10f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        else:
            print("  Multi-res: no improvement (best was %.10f)" % global_best_obj)

    # === Phase 5: Alternating polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.15
        print("\n=== Phase 5: Alternating FISTA+L-BFGS polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, obj5 = run_alternating_polish(global_best_a, p5_budget, rng)
        print("  Alternating polish: obj=%.10f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final aggressive L-BFGS ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final L-BFGS (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng, n_beta_levels=4)
        print("  Final L-BFGS: obj=%.10f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.10f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using uniform fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.10f" % global_best_obj)
    return final_list
