    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute LSE softmax weights for gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def topk_weights(c, k, power=2.0):
    """Compute top-k weighted weights for gradient."""
    n = len(c)
    k = min(k, n)
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(n)
    w_full[idx] = w
    return w_full


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def dct_lowpass_noise(n, rng, lowpass_frac=0.05, scale=1.0):
    """Generate smooth noise via low-pass filtering in frequency domain."""
    noise = rng.normal(0, 1.0, n)
    F = fft(noise)
    half = n // 2
    cutoff = max(2, int(n * lowpass_frac))
    F[cutoff: -cutoff] *= 0.05
    smooth = np.real(ifft(F))[:n]
    norm = np.linalg.norm(smooth)
    if norm > 1e-15:
        smooth = smooth / norm * scale
    return smooth


def run_adaptive_fista(a_init, max_time, rng, beta_start=20.0, beta_end=2000.0,
                       cycle_period=4000, mode="auto", step_init=1.0):
    """
    FISTA with adaptive cycling beta and multiple gradient modes.
    Returns (best_a, best_max_conv).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    step = step_init
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Cycling beta: ramps up, then resets
        cycle_pos = it % cycle_period
        if cycle_pos < cycle_period * 0.8:
            frac = cycle_pos / (cycle_period * 0.8)
            beta = beta_start + (beta_end - beta_start) * frac * frac
        else:
            beta = beta_start * 0.5
        beta = max(5.0, min(beta, 50000.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
            # Adaptive step: increase slightly on improvement
            step = min(step * 1.01, 2.0)
        else:
            stuck_count += 1
            if stuck_count % 50 == 0:
                step *= 0.95

        # Choose gradient mode based on stuck count
        if mode == "auto":
            if stuck_count < 500:
                # LSE mode
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)
            elif stuck_count < 2000:
                # Hybrid: mix LSE and top-k
                k = max(4, min(64, n // 16))
                w_lse = lse_weights(c, beta)
                w_topk = topk_weights(c, k, power=2.0)
                w = 0.5 * w_lse + 0.5 * w_topk
                grad = conv_gradient(y, w)
            else:
                # Top-k with increasing k
                k = min(256, 16 + stuck_count // 200)
                w = topk_weights(c, k, power=1.5)
                grad = conv_gradient(y, w)
        elif mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(128, n // 32))
            w = topk_weights(c, k, power=2.0)
            grad = conv_gradient(y, w)
        elif mode == "hybrid":
            k = max(8, min(farith64, n // 16))
            w_lse = lse_weights(c, beta)
            w_topk = topk_weights(c, k, power=1.5)
            mix = min(0.8, 0.2 + 0.002 * stuck_count)
            w = (1 - mix) * w_lse + mix * w_topk
            grad = conv_gradient(y, w)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Structured perturbation when stuck
        if stuck_count > 3000 and (it % 200 == 0):
            noise_scale = max(1e-5, 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0))
            if noise_scale > 1e-8:
                noise = dct_lowpass_noise(n, rng, lowpass_frac=0.03, scale=noise_scale)
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = max(0, stuck_count - 1000)
                step = step_init

        # FISTA update
        eff_step = step / max(beta, 1.0) * (S / gnorm)
        eff_step = min(eff_step, S / (0.5 * gnorm) * 0.1)
        x_new = project_simplex(y - eff_step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng):
    """L-BFGS-B polishing with increasing LSE beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 100.0, 300.0, 800.0]
    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += 10.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 20.0 * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 80, 'maxls': 25, 'ftol': 1e-14, 'gtol': 1e-10,
                     'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new * (S / s_new)
        obj_new = compute_obj(a_new)
        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs)" % (beta, obj_new, elapsed))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_exactive_polish(a_init, max_time, rng):
    """Active-set polishing: focus on the exact argmax of convolution."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive: start with exact, expand as we get stuck
        if stuck < 200:
            active = np.where(c >= cur_max - 1e-10)[0]
        elif stuck < 1000:
            thresh = cur_max - 0.001 * cur_max
            active = np.where(c >= thresh)[0]
        else:
            k = min(32, 4 + stuck // 200)
            active = np.argpartition(c, -k)[-k:]

        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 0.1 * S / max(gnorm, 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def generate_inits(n, rng, S, n_types=6):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Raised cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Gaussian window
    center = n / 2.0
    sigma = n / 5.0
    a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Two-lobe
    a = 0.5 + 0.5 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Smoothed random
    raw = rng.random(n)
    F = fft(raw)
    cutoff = max(3, n // 80)
    F[cutoff: -cutoff] *= 0.02
    smooth = np.real(ifft(F))[:n]
    smooth = np.maximum(smooth, 0.01)
    smooth = smooth * (S / smooth.sum())
    inits.append(smooth.copy())

    # Triangular
    tri = np.minimum(np.arange(n).astype(float), (n - 1 - np.arange(n)).astype(float))
    tri = np.maximum(tri, 1.0)
    tri = tri * (S / tri.sum())
    inits.append(tri.copy())

    # Uniform (baseline)
    inits.append(np.ones(n) * (S / n))

    return inits[:n_types]


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses adaptive LSE-FISTA, active-set polishing, and L-BFGS-B.
    Explores n=8192 (warm start) and n=16384 (fresh) with diverse inits.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a):
        nonlocal global_best_obj, global_best_a
        if a is None:
            return
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  NEW BEST: obj=%.8f (n=%d)" % (obj, len(a)))

    # === Phase 0: Initialize ===
    print("=== Phase 0: Initialization ===")
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Warm start: n=%d, obj=%.6f" % (len(a_start), obj0))
        update_best(a_start.copy())
    else:
        n0 = 8192
        S0 = np.sqrt(2.0 * n0)
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.01)
        a_start = a_start * (S0 / a_start.sum())
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        update_best(a_start.copy())

    elapsed_init = time.monotonic() - (deadline - budget_s + 60.0)
    print("Init time: %.1fs" % elapsed_init)

    # === Phase 1: Adaptive FISTA on warm start (n=8192) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p1_budget = min(remaining * 0.30, 500.0)
        print("\n=== Phase 1: Adaptive FISTA n=8192 (budget %.0fs) ===" % p1_budget)
        t1 = time.monotonic()

        a1, _ = run_adaptive_fista(
            global_best_a, p1_budget * 0.6, rng,
            beta_start=15.0, beta_end=1500.0,
            cycle_period=3000, mode="auto", step_init=1.0)
        update_best(a1)

        rem = p1_budget - (time.monotonic() - t1)
        if rem > 20:
            a1b, _ = run_adaptive_fista(
                global_best_a, rem, rng,
                beta_start=50.0, beta_end=5000.0,
                cycle_period=4000, mode="hybrid", step_init=0.8)
            update_best(a1b)

        print("  Phase 1 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t1))

    # === Phase 2: Try n=16384 with fresh starts ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        p2_budget = min(remaining * 0.30, 600.0)
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        print("\n=== Phase 2: n=16384 fresh starts (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        inits2 = generate_inits(n2, rng, S2, n_types=4)
        time_per = p2_budget / len(inits2)

        for idx, init in enumerate(inits2):
            if time.monotonic() > deadline:
                break
            per_time = min(time_per, 120.0)
            if per_time < 15:
                break
            print("  n=16384 init %d:" % idx)
            a_m, _ = run_adaptive_fista(
                init, per_time, rng,
                beta_start=15.0, beta_end=2000.0,
                cycle_period=3000, mode="auto", step_init=1.0)
            update_best(a_m)

        print("  Phase 2 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t2))

    # === Phase 3: L-BFGS-B polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.18, 300.0)
        print("\n=== Phase 3: L-BFGS-B polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a3, obj3 = run_lbfgs_polish(global_best_a, p3_budget, rng)
        update_best(a3)
        print("  Phase 3 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t3))

    # === Phase 4: Active-set polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.15, 250.0)
        print("\n=== Phase 4: Active-set polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, _ = run_exactive_polish(global_best_a, p4_budget, rng)
        update_best(a4)
        print("  Phase 4 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t4))

    # === Phase 5: Final FISTA with high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.35
        print("\n=== Phase 5: High-beta FISTA (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()
        a5, _ = run_adaptive_fista(
            global_best_a, p5_budget * 0.5, rng,
            beta_start=200.0, beta_end=10000.0,
            cycle_period=5000, mode="hybrid", step_init=0.5)
        update_best(a5)

        rem = p5_budget - (time.monotonic() - t5)
        if rem > 15:
            a5b, _ = run_adaptive_fista(
                global_best_a, rem, rng,
                beta_start=500.0, beta_end=20000.0,
                cycle_period=6000, mode="topk", step_init=0.3)
            update_best(a5b)
        print("  Phase 5 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t5))

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng)
        update_best(a6)
        print("  Phase 6 done: best=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t6))

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            S_fb = np.sqrt(2.0 * n_fb)
            global_best_a = np.ones(n_fb) * (S_fb / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_best_a = np.ones(n_fb) * (S_fb / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
