import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

    """Gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

    """Gradient of sum_i (y[i]-y[i+1])^2."""
    n = len(y)
    g = np.zeros(n)
    if n < 2:

    g[0] = 2.0 * (y[0] - y[1])
    g[-1] = 2.0 * (y[-1] - y[-2])

    s = a.sum()
    if s < 1e-15:

    c = fft_convolve_self(a)

            t_mom = 1.0  # Reset momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 5000 == 0:
            t_mom = 1.0

        # Smooth perturbation when very stuck
        if stuck > 3000 and it % 500 == 0:
            noise_scale = 0.003 * S / np.sqrt(n)
            # Generate smooth noise
            noise = rng.normal(0, 1.0, n)
            F = fft(noise)
            cutoff = max(3, n // 100)
            F[cutoff:-cutoff] *= 0.01
            noise = np.real(ifft(F))[:n]
            noise = noise / (np.linalg.norm(noise) + 1e-15) * noise_scale
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            stuck = 0  # Reset counter

    return best_a, best_obj


def run_active_refine(a_init, max_time, rng):
    """
    Active-set refinement: focus on top-k peaks with exact subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 3
        elif stuck < 500:
            k = 8
        elif stuck < 1000:
            k = 16
        else:
            k = min(64, 16 + (stuck - 1000) // 100)

        # Get top-k lags with power weighting
        idx = np.argsort(c)[::-1][:k]
        vals = c[idx]
        # Power weighting: emphasize the largest
        w_k = np.power(np.maximum(vals, 1e-30), 0.5)
        w_k = w_k / w_k.sum()

        w_full = np.zeros(len(c))
        w_full[idx] = w_k
        grad = conv_gradient(y, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with line search (simplified)
        step = S / (gnorm * np.sqrt(it + 1))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_local_search(a_init, max_time, rng):
    """
    Local search: try random smooth perturbations and coordinate optimizations.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    best_c = fft_convolve_self(a).copy()

    attempt = 0
    while time.monotonic() - start < max_time:
        attempt += 1
        elapsed = time.monotonic() - start
        # Decreasing perturbation scale
        scale = 0.01 * (1.0 - elapsed / max_time)
        scale = max(scale, 0.001)

        # Try different perturbation types
        rtype = rng.integers(3)

        if rtype == 0:
            # Smooth Gaussian perturbation
            noise = rng.normal(0, 1.0, n)
            F = fft(noise)
            cutoff = max(3, n // 50)
            F[cutoff:-cutoff] *= 0.1
            noise = np.real(ifft(F))[:n]
            noise = noise / (np.linalg.norm(noise) + 1e-15) * scale * S
            candidate = project_simplex(a + noise, S)

        elif rtype == 1:
            # Local window perturbation
            center = rng.integers(n // 4, 3 * n // 4)
            width = max(10, n // 50)
            lo = max(0, center - width)
            hi = min(n, center + width)
            window = np.exp(-0.5 * ((np.arange(lo, hi) - center) / (width / 3))**2)
            noise = np.zeros(n)
            noise[lo:hi] = window * rng.normal(0, 1.0, hi - lo)
            noise = noise / (np.linalg.norm(noise) + 1e-15) * scale * S
            candidate = project_simplex(a + noise, S)

        else:
            # Frequency-domain perturbation (modify a random frequency bin)
            F = fft(a)
            m = rng.integers(1, n // 2)
            F[m] += rng.normal(0, S / n)
            F[n - m] = np.conj(F[m])  # Keep real
            candidate = np.real(ifft(F))[:n]
            candidate = np.maximum(candidate, 0)
            if candidate.sum() > 1e-10:
                candidate = candidate * (S / candidate.sum())
            else:
                continue

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate.copy()
            best_c = fft_convolve_self(a).copy()

    return best_a, best_obj
