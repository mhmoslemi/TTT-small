```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
```

```python
import time
from scipy.fft import fft, ifft

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Compute gradient of LSE objective.
    g_k = 2 * sum_j w_j * a_{j-k}
    a has length n, w has length 2n-1 (softmax over convolution lags).
    """
    n = len(a)
    # g_k = 2 * sum_j w_j * a_{j-k}
    # = 2 * (w_rev * a)_{2n-2-k} where w_rev[j] = w[2n-2-j]

    W = fft(w_rev, fsize)
    A = fft(a, fsize)
    conv = np.real(ifft(W * A))
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value(c, beta):
    """Log-sum-exp value."""
    cm = c.max()
    return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta

            best_a = y.copy()

        c_shift -= c_shift.max()
        exp_c = np.exp(c_shift)
        w = exp_c / exp_c.sum()
        
        grad = compute_grad(y, w)
        
        # Step size with backtracking
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

            t_mom = 1.0
```
