import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    mx = scaled.max()
    if mx > 500:
        scaled = scaled * (500.0 / mx)
        beta_eff = beta * (500.0 / mx)
    else:
        beta_eff = beta
    e = np.exp(scaled - mx)  # subtract max for numerical stability
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LSE-smoothed gradient descent with momentum, adaptive beta scheduling,
LP-guided direction finding, and multi-restart strategy.
"""
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_grad_and_weights(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    smin = scaled.min()
    scaled = scaled - smin  # shift for stability
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w

def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def lse_descent(a0, max_time, rng, verbose=True):
    """LSE gradient descent with momentum and adaptive beta."""
    n = len(a0)
    S_target = 1.0
    a = project_simplex(a0.copy(), S_target)
    c = fft_convolve_self(a)
    best_cmax = c.max()
    best_a = a.copy()
    
    momentum = np.zeros(n)
    mom_rate = 0.9
    beta = 50.0
    min_beta = 10.0
    max_beta = 50000.0
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_cmax - 1e-14:
            best_cmax = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive beta: increase when peaks are concentrated
        top_std = np.std(c[np.argsort(c)[::-1][:20]])
        mean_c = c.mean()
        peak_ratio = top_std / max(mean_c, 1e-15)
        if peak_ratio > 0.5 and beta < max_beta:
            beta = min(max_beta, beta * 1.02)
        elif peak_ratio < 0.1 and beta > min_beta:
            beta = max(min_beta, beta * 0.98)
        
        grad, w = lse_grad_and_weights(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Momentum
        momentum = mom_rate * momentum - grad
        step_dir = momentum
        snorm = np.linalg.norm(step_dir)
        
        # Step size with backtracking
        step = 2.0 / (beta * snorm + 1e-30)
        step = min(step, 0.5 / snorm if snorm > 0 else 1.0)
        
        accepted = False
        for ls in range(15):
            a_try = project_simplex(a + step * step_dir, S_target)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-12) + 1e-15:
                a = a_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            # Try smaller step
            step = 0.01 / snorm if snorm > 0 else 0.01
            a = project_simplex(a + step * step_dir, S_target)
        
        # Noise when stuck

            noise_scale = min(0.05, 0.005 * (stuck / patience)) / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S_target)
            momentum *= 0.5
            stuck = 0
            patience = int(patience * 1.2)
        
        if verbose and it % 500 == 0:
            cur_obj = 2.0 * n * best_cmax / S_target**2
            print(f"    it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={cur_obj:.6f}, cmax={best_cmax:.8f}")

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)

    fsize = 1 << (size - 1).bit_length()

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    best_a = a.copy()

    momentum = np.zeros(n)
    start = time.monotonic()
    it = 0
    phase = 0  # 0 = low beta, 1 = high beta
    phase_len = 300  # iterations per phase

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Determine beta based on phase
        if phase == 0:
            beta = 30.0 + 40.0 * progress
        else:
            beta = min(50000.0, 200.0 * (1.0 + 3.0 * progress))

        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_cmax - 1e-14:
            best_cmax = cur_max
            best_a = a.copy()

        grad, w = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            phase = 1 - phase
            continue

        momentum = 0.88 * momentum - grad
        snorm = np.linalg.norm(momentum)
        if snorm < 1e-14:
            break

        lr = min(2.0 / snorm, 50.0 / beta, 1.0)
        accepted = False
        tol = max(1e-12, 1e-10 * cur_max)
        for ls in range(20):
            a_try = project_simplex(a + lr * momentum, S_target)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + tol:
                a = a_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            a = project_simplex(a + (1e-5 / (snorm + 1e-30)) * momentum, S_target)

        # Switch phase periodically
        if it % phase_len == 0:
            phase = 1 - phase
            if phase == 0:
                momentum *= 0.5

        if it % 5000 == 0:
            cur_obj = 2.0 * n * best_cmax / (S_target * S_target)
            print(f"    [{tag}] it={it}, t={elapsed:.0f}s, phase={phase}, obj={cur_obj:.6f}")

    return best_a, best_cmax


def generate_init(n, rng, base=None):
    """Generate initial sequence, optionally interpolating from a base."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a / a.sum()
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    a += rng.normal(0, 0.015, n)
    a = np.maximum(a, 0)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    - Phase 1: Accelerated LSE descent on provided sequence (n=8192)
    - Phase 1b: Two-level LSE polish (alternating low/high beta)
    - Phase 2: Explore alternative lengths (n=4096, n=16384)
    - Phase 3: Intensive multi-round refinement on global best
    - Phase 4: Targeted perturbations + re-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 80.0)

    print(f"=== Starting optimization, budget={budget_s}s ===")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Loaded provided: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception as e:
        print(f"Could not load provided sequence: {e}")

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        obj = compute_obj_normalized(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.6f} ({msg}) ***")

    time_start = time.monotonic()

    # === Phase 1: Primary optimization on provided ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(700, (deadline - time.monotonic()) * 0.30)
        print(f"\n--- Phase 1: Accelerated LSE, n={n1}, S={S1:.1f}, t={t1:.0f}s ---")
        a_opt, cmax1 = accelerated_lse_descent(a1, S1, t1, rng,
                                                beta_init=40, beta_max=40000,
                                                momentum_init=0.88,
                                                verbose=True, tag="P1")
        update_global(a_opt, "P1 accelerated LSE")
        print(f"    P1 done: cmax={cmax1:.8f}, obj={2*n1*cmax1/(S1*S1):.6f}")

        # Phase 1b: Two-level polish
        remaining = deadline - time.monotonic()
        t1b = min(350, remaining * 0.18)
        if t1b > 30:
            print(f"\n--- Phase 1b: Two-level LSE polish, t={t1b:.0f}s ---")
            a_pol, cmax2 = two_level_lse_descent(a_opt, S1, t1b,
                                                  np.random.default_rng(seed + 7),
                                                  tag="P1b")
            update_global(a_pol, "P1b two-level")
            print(f"    P1b done: cmax={cmax2:.8f}, obj={2*n1*cmax2/(S1*S1):.6f}")

    # === Phase 2: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    tried_lengths = set()
    if a_provided is not None:
        tried_lengths.add(len(a_provided))

    for n_try in [8192, 16384, 4096]:
        if time.monotonic() > deadline - 150:
            break
        if n_try in tried_lengths:
            continue
        S = np.sqrt(2.0 * n_try)
        t_alloc = min(350, (deadline - time.monotonic()) * 0.18)
        if t_alloc < 60:
            continue
        tried_lengths.add(n_try)
        print(f"\n--- Phase 2: n={n_try}, t={t_alloc:.0f}s ---")
        base_for_init = global_a if global_a is not None else None
        init = generate_init(n_try, rng, base=base_for_init)
        init = init * S
        a_opt, cmax = accelerated_lse_descent(init, S, t_alloc,
                                               np.random.default_rng(seed + n_try),
                                               beta_init=40, beta_max=40000,
                                               verbose=False, tag=f"P2-{n_try}")
        update_global(a_opt, f"P2 n={n_try}")
        print(f"    P2 n={n_try} done: obj={2*n_try*cmax/(S*S):.6f}")

    # === Phase 3: Intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 100:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a_curr = global_a * (S / global_a.sum())
        t3_total = remaining - 40

        print(f"\n--- Phase 3: Intensive refinement, n={n}, t={t3_total:.0f}s ---")

        n_rounds = 5
        for rnd in range(n_rounds):
            if time.monotonic() > deadline - 30:
                break
            t_rnd = min((deadline - time.monotonic() - 30) / (n_rounds - rnd), 250)
            if t_rnd < 20:
                break

            if rnd % 2 == 0:
                # Direct refinement with high beta
                a_curr, _ = accelerated_lse_descent(a_curr, S, t_rnd,
                                                     np.random.default_rng(seed + 1000 + rnd),
                                                     beta_init=max(100, 40 * (10 ** rnd)),
                                                     beta_max=60000,
                                                     momentum_init=0.90,
                                                     verbose=False, tag=f"P3-{rnd}")
            else:
                # Perturb then refine
                noise_scale = 0.008 * S / np.sqrt(n)
                a_pert = project_simplex(a_curr + rng.normal(0, noise_scale, n), S)
                a_curr, _ = accelerated_lse_descent(a_pert, S, t_rnd,
                                                     np.random.default_rng(seed + 2000 + rnd),
                                                     beta_init=80, beta_max=60000,
                                                     momentum_init=0.87,
                                                     verbose=False, tag=f"P3-{rnd}")
            update_global(a_curr, f"P3 round {rnd}")

    # === Phase 4: Targeted micro-perturbations ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_micro = 3
        t_micro = (remaining - 20) / n_micro

        print(f"\n--- Phase 4: {n_micro} micro-perturbations, {t_micro:.0f}s each ---")
        for mi in range(n_micro):
            if time.monotonic() > deadline - 20:
                break
            # Small structured perturbation (not pure random)
            noise_scale = 0.004 * S / np.sqrt(n)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            a_pert, _ = accelerated_lse_descent(a_pert, S, t_micro,
                                                 np.random.default_rng(seed + 5000 + mi),
                                                 beta_init=200, beta_max=80000,
                                                 momentum_init=0.92,
                                                 verbose=False, tag=f"P4-{mi}")
            update_global(a_pert, f"P4 micro {mi}")

    # === Final: Evaluate and return ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) / 256.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) / 256.0

    # Final verification
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    elapsed_total = time.monotonic() - time_start
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={len(global_a)}, internal_best={global_best:.6f}")
    print(f"evaluate_sequence() = {ver:.6f}")
    print(f"Total time: {elapsed_total:.0f}s")
    print(f"{'='*60}")

    return final_list
