"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: L-BFGS with adaptive LSE smoothing, spectral-flattening initialization,
multi-restart, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_objective_and_grad(a, beta):
    """
    Compute the LSE-smoothed objective and its gradient.
    Objective: f(a) = 2*n * LSE_beta(c) / S^2
    where c = a*a (full convolution), S = sum(a).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf, np.zeros(n)
    
    c = fft_convolve_self(a)
    # LSE: (1/beta) * log(sum(exp(beta * c)))
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = shifted - shifted.max()  # for numerical stability
    exp_vals = np.exp(shifted)
    log_sum = c_max + np.log(exp_vals.sum()) / beta
    lse = log_sum
    
    # Weights for each lag
    w = exp_vals / exp_vals.sum()
    
    # Gradient of LSE w.r.t. a:
    # d/d a_j [LSE] = sum_k w_k * dc_k / d a_j
    # dc_k / d a_j = a[k-j] + a[j-k] (valid indices only)
    # This is 2 * (w * a) circular-like

    W = fft(w, fsize)

    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the exact objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_val_and_weights(c, beta):
    """Compute LSE value and softmax weights of c with temperature 1/beta."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    tot = exp_vals.sum()
    if tot < 1e-300:
        lse = c_max
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse = c_max + np.log(tot) / beta
        w = exp_vals / tot
    return lse, w


def grad_lse_wrt_a(w, a):
    """Gradient of sum_k w_k * c_k w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def z_to_a(z, S):
    """Convert z to non-negative vector with sum S via softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) * S / len(z)
    return e / s * S


def a_to_z(a):
    """Convert non-negative vector a to z-space (log)."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def obj_z(z, beta, n, S):
    """Objective in z-space: LSE-smoothed version of f(a)."""
    a = z_to_a(z, S)
    c = fft_convolve_self(a)
    lse, _ = lse_val_and_weights(c, beta)
    return 2.0 * n * lse / (S * S)


def grad_z(z, beta, n, S):
    """Gradient of objective w.r.t. z in softmax parameterization."""
    a = z_to_a(z, S)
    c = fft_convolve_self(a)
    lse, w = lse_val_and_weights(c, beta)
    # grad of LSE w.r.t. a
    grad_a = grad_lse_wrt_a(w, a)
    # Full gradient of f = 2n*LSE/S^2 w.r.t. a (S is fixed since a sums to S)
    grad_f_a = (2.0 * n / (S * S)) * grad_a
    # Chain rule: da_j/dz_k = a_j*(delta_jk - a_k/S)
    # df/dz_k = a_k*grad_f_a[k] - (a_k/S)*sum_j(a_j*grad_f_a[j])
    dot = np.dot(a, grad_f_a)
    grad_z = a * (grad_f_a - dot / S)
    return grad_z


def spectral_flatten_init(a, n_iter=200, S=None):
    """Iterative spectral amplitude compression to flatten the convolution."""
    n = len(a)
    if S is None:
        S = a.sum()
    a = a.copy()
    alpha = 0.4
    best_a = a.copy()
    best_max = fft_convolve_self(a).max()
    
    for it in range(n_iter):
        A = fft(a, 1 << (2 * n - 1).bit_length())
        mag = np.abs(A)
        # Compress the amplitude: new_mag = mag^(1-alpha) normalized
        mag_new = mag ** (1.0 - alpha)
        # Preserve phase
        phase = A / (mag + 1e-30)
        A_new = phase * mag_new
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_max:
            best_max = c_new.max()
            best_a = a_new.copy()
            a = a_new
            # Slowly decrease alpha
            alpha *= 0.98
        else:
            alpha *= 0.85
            if alpha < 0.02:
                break
    return best_a


def make_tapered_init(n, method, rng, S):
    """Create various structured initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    if method == 0:
        a = np.sin(np.pi * t) ** rng.uniform(0.5, 4.0)
    elif method == 1:
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.35)
        a = np.exp(-3.0 * ((t - center) / width) ** 2)
    elif method == 2:
        a = 0.2 + 0.8 * np.exp(-4.0 * (t - 0.5) ** 2)
    elif method == 3:
        freq = rng.uniform(1, 6)
        a = np.abs(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif method == 4:
        a = np.maximum(0.02, 1.0 - np.abs(2 * t - 1) ** rng.uniform(0.5, 3.0))
    elif method == 5:
        # Multi-peak
        n_peaks = rng.integers(2, 5)
        a = np.zeros(n)
        for _ in range(n_peaks):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.02, 0.1)
            a += np.exp(-5.0 * ((t - c) / w) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def run_lbfgs_z(z_init, max_time, n, S, beta_start=20.0, beta_end=500.0,
                max_iter_per_run=300, n_phases=4):
    """Run L-BFGS in z-space with increasing beta."""
    start_time = time.monotonic()
    z = z_init.copy()
    best_z = z.copy()
    best_a = z_to_a(z, S)
    best_obj = compute_obj(best_a)
    
    for phase in range(n_phases):
        if time.monotonic() - start_time >= max_time:
            break
        progress = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        def fun(x):
            return obj_z(x, beta, n, S)
        
        def grad(x):
            return grad_z(x, beta, n, S)
        
        t_phase_start = time.monotonic()
        time_left = max_time - (time.monotonic() - start_time)
        
        try:
            result = scipy_minimize(
                fun, z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_run, 'maxfun': max_iter_per_run * 3,
                         'gtol': 1e-10, 'factr': 1e8, 'disp': False}
            )
            z_new = result.x
            a_new = z_to_a(z_new, S)
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                best_z = z_new.copy()
                z = z_new.copy()
                print("      Phase %d (beta=%.0f): obj=%.6f ***" % (phase, beta, obj_new))
            else:
                print("      Phase %d (beta=%.0f): obj=%.6f" % (phase, beta, obj_new))
                z = z_new.copy()  # Still update z (it's in a good region)
        except Exception as e:
            print("      Phase %d error: %s" % (phase, str(e)))
    
    return best_a, best_obj, best_z


def coordinate_descent(a_init, max_time, rng, S=None):
    """
    Coordinate descent via targeted mass transfer to reduce convolution peaks.
    """
    n = len(a_init)
    if S is None:
        S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    
    start_time = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 15000:
            break
        
        # Find top 3 peaks
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute multi-peak contribution for each coordinate
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            # dc_kk/da_j = 2*a[kk-j] if 0 <= kk-j < n
            lo_j = max(0, kk - (n - 1))
            hi_j = min(n - 1, kk)
            if hi_j >= lo_j:
                multi_contrib[lo_j:hi_j + 1] += 2.0 * a[kk - hi_j:kk - lo_j + 1]
        
        # Find best transfer: from high multi_contrib to low multi_contrib
        # Score for source: high multi_contrib * a[j] (can give up more)
        # Score for target: low multi_contrib * (1 - a[j]/(S/n)) (has room)
        src_score = multi_contrib * np.maximum(a, 1e-10)
        tgt_score = -multi_contrib * np.maximum(S / n - a, 1e-10)
        
        n_src = min(10, n)
        n_tgt = min(10, n)
        src_candidates = np.argsort(src_score)[::-1][:n_src]
        tgt_candidates = np.argsort(tgt_score)[:n_tgt]
        
        found_improvement = False
        for j in src_candidates:
            if found_improvement:
                break
            for l in tgt_candidates:
                if j == l or a[j] < 1e-10:
                    continue
                # Try transferring delta from j to l
                # Binary search for optimal delta
                delta_lo = 0.0
                delta_hi = a[j] * 0.5
                best_delta = 0.0
                best_new_max = cur_max + 1e-15
                
                for _ in range(8):
                    delta = 0.5 * (delta_lo + delta_hi)
                    if delta < 1e-12:
                        break
                    # Compute change in c
                    # c changes by: -delta * (contribution of a[j]) + delta * (contribution of a[l])
                    # Contribution of a[j] to c: c[m] gets -delta * a[m-j] for valid m
                    # Contribution of a[l] to c: c[m] gets +delta * a[m-l] for valid m
                    # For the peak lags, compute new value
                    new_max = cur_max
                    for kk in top_idx:
                        kk = int(kk)
                        # Change in c[kk]:
                        # -delta * a[kk-j] (if valid) + delta * a[kk-l] (if valid)
                        # Also: -delta * a[j-kk] (if valid) + delta * a[l-kk] (if valid)
                        dc = 0.0
                        if 0 <= kk - j < n:
                            dc -= delta * a[kk - j]
                        if 0 <= j - kk < n:
                            dc -= delta * a[j - kk]
                        if 0 <= kk - l < n:
                            dc += delta * a[kk - l]
                        if 0 <= l - kk < n:
                            dc += delta * a[l - kk]
                        new_max = max(new_max, c[kk] + dc)
                    # Also check a few other lags that might become peaks
                    check_lags = [int(k) for k in np.argsort(c)[::-1][top_k:top_k + 3]]
                    for kk in check_lags:
                        dc = 0.0
                        if 0 <= kk - j < n:
                            dc -= delta * a[kk - j]
                        if 0 <= j - kk < n:
                            dc -= delta * a[j - kk]
                        if 0 <= kk - l < n:
                            dc += delta * a[kk - l]
                        if 0 <= l - kk < n:
                            dc += delta * a[l - kk]
                        new_max = max(new_max, c[kk] + dc)
                    
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_delta = delta
                        delta_lo = delta
                    else:
                        delta_hi = delta
                
                if best_delta > 1e-12 and best_new_max < cur_max - 1e-12:
                    # Apply the transfer
                    a[j] -= best_delta
                    a[l] += best_delta
                    # Update c (full recompute for accuracy)
                    if it % 5 == 0 or best_delta > a[j] * 0.3:
                        c = fft_convolve_self(a)
                    else:
                        # Partial update
                        for m in range(max(0, n - 1), min(2 * n - 1, n)):
                            pass  # too complex, just recomput
                        c = fft_convolve_self(a)
                    found_improvement = True
                    break
        
        if not found_improvement and (it - last_improve) > 500:
            # Random perturbation
            idx = rng.integers(0, n)
            delta = a[idx] * 0.02
            if delta > 1e-12:
                a[idx] -= delta
                a[rng.integers(0, n)] += delta
                c = fft_convolve_self(a)
    
    return best_a, best_obj


def run_high_beta_polish(a_init, max_time, n, S):
    """Polish with very high beta L-BFGS."""
    z = a_to_z(a_init)
    return run_lbfgs_z(z, max_time, n, S, beta_start=200, beta_end=5000,
                       max_iter_per_run=200, n_phases=2)
