import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum_w = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum_w - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum_w[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def coordinate_descent_lagrangian(a, max_time, rng, lambda_init=0.0, n_sweeps_est=500):
    """
    Coordinate descent with Lagrangian relaxation of the sum constraint.
    Minimizes max_k (a*a)[k] + lambda * (sum(a) - S) over a >= 0.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a.copy().astype(np.float64)
    lambda_val = lambda_init
    
    # Initial convolution
    c = fft_convolve_self(a)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    
    sweep = 0
    while time.monotonic() - start < max_time:
        sweep += 1
        # Random order for coordinates
        order = rng.permutation(n)
        
        for i in order:
            # 1-D update for a[i]
            ai_old = a[i]
            
            # Partners: a[k-i] for k in [max(0,i), min(2n-2, i+n-1)]
            k_min = max(0, i)
            k_max = min(2*n-2, i+n-1)
            partner_indices = np.arange(k_min - i, k_max - i + 1)
            partners = a[partner_indices]
            
            # Skip if all partners are zero
            if partners.max() < 1e-15:
                continue
            
            # Rests: c[k] - ai_old * a[k-i]
            lags = np.arange(k_min, k_max+1)
            rests = c[lags] - ai_old * partners
            
            # 1-D minimization: min_{x>=0} max_k(partners[k]*x + rests[k]) + lambda_val * x
            # Binary search
            x_low = 0.0
            x_high = max(ai_old * 3, np.max(partners) * S / max(1, partners.sum()), 1.0)
            # Actually, a better upper bound: the value that would make a[i]*a[i] = current max
            # c[i+i] includes a[i]^2, so x_high = sqrt(c[2*i]) if valid
            # But let's keep it simple
            x_high = max(ai_old * 5, S / n * 10, 1.0)
            
            x_new = ai_old  # default
            for _ in range(40):
                x_mid = (x_low + x_high) / 2
                vals = partners * x_mid + rests
                mx = vals.max()
                # Active set with tolerance
                tol = mx * 1e-8 + 1e-12
                active = vals >= mx - tol
                subgrad = partners[active].sum() + lambda_val
                if subgrad > 1e-8:
                    x_high = x_mid
                elif subgrad < -1e-8:
                    x_low = x_mid
                else:
                    x_new = x_mid
                    break
            else:
                x_new = (x_low + x_high) / 2
            
            x_new = max(0.0, x_new)
            
            # Update a[i] and c
            delta = x_new - ai_old
            if abs(delta) > 1e-15:
                a[i] = x_new
                # Update convolution: c[k] += delta * a[k-i] for k in [k_min, k_max]
                c[lags] += delta * partners
            
            # Track best
            # (We'll check objective at end of sweep for efficiency)
        
        # Update lambda (dual ascent)
        current_sum = a.sum()
        lambda_step = S / (sweep + 1)  # decreasing step
        lambda_val += lambda_step * (current_sum - S)
        # Clip lambda to avoid explosion
        lambda_val = np.clip(lambda_val, -10000, 10000)
        
        # Periodic objective check
        if sweep % 10 == 0:

            obj = c_check.max()  # This is max_b; true objective = 2n*max_b/sum_a^2

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` over non-negative sequences. The previous approach used LSE-FISTA with active-set refinement. I'll take a fundamentally different approach: **coordinate descent with Lagrangian relaxation of the sum constraint**, where each 1-D subproblem (updating one coordinate at a time) is solved exactly via binary search on the piecewise-linear convex objective.

Key insights:
1. For a fixed coordinate `a[i]`, the objective becomes a 1-D convex piecewise-linear function (max of linear terms), solvable by binary search on the subgradient.
2. The sum constraint is handled via a Lagrange multiplier (dual ascent), removing the need for simplex projections during the main loop.
3. This naturally handles the non-smooth max objective without smoothing approximations.

"""
Minimize 2*n*max(f*f)/sum(f)^2 via coordinate descent with Lagrangian relaxation.
Each 1-D subproblem is solved exactly via binary search on the piecewise-linear
subgradient. The sum constraint is handled by dual ascent on a Lagrange multiplier.
This avoids smoothing approximations and directly targets the non-smooth max objective.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum_w = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum_w - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum_w[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=500.0, beta_end=50000.0, beta_steps=100):
    """FISTA with LSE smoothing for final polish."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta
        stagnant = it - last_improve
        if stagnant > 300:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coord_descent_update(a, c, i, lambda_val, n, binary_iters=45):
    """
    Solve the 1-D subproblem for coordinate i:
    min_{x>=0} max_k (a[k-i]*x + rest_k) + lambda_val * x
    where rest_k = c[k] - a[i]*a[k-i] (the convolution without a[i]'s contribution).
    Returns the optimal x.
    """
    ai_old = a[i]
    k_min = i
    k_max = min(2 * n - 2, i + n - 1)

    if k_min > k_max:
        return ai_old

    partner_indices = np.arange(k_min - i, k_max - i + 1)
    partners = a[partner_indices]

    if partners.max() < 1e-15:
        return ai_old

    lags = np.arange(k_min, k_max + 1)
    rests = c[lags] - ai_old * partners

    # Binary search for optimal x
    x_low = 0.0
    x_high = max(ai_old * 5.0, 2.0 * np.sqrt(max(c[lags].max(), 0.0)), S_global / n * 5.0)
    x_high = min(x_high, 1000.0)

    x_new = ai_old
    for _ in range(binary_iters):
        x_mid = (x_low + x_high) / 2.0
        vals = partners * x_mid + rests
        mx = vals.max()
        tol = max(mx * 1e-10, 1e-14)
        active = vals >= mx - tol
        subgrad = partners[active].sum() + lambda_val
        if subgrad > 1e-10:
            x_high = x_mid
        elif subgrad < -1e-10:
            x_low = x_mid
        else:
            x_new = x_mid
            break
    else:
        x_new = (x_low + x_high) / 2.0

    return max(0.0, min(x_new, 1000.0))


S_global = 0.0


def run_coordinate_descent(a_init, max_time, rng, lambda_init=0.0,
                           n_binary_iters=45, check_interval=20):
    """
    Main coordinate descent with Lagrangian relaxation.
    """
    global S_global
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    S_global = S
    a = normalize_sum(a_init, n).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = project_simplex(a, S)

    lambda_val = lambda_init
    c = fft_convolve_self(a)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()

    # Initial objective
    c_check = fft_convolve_self(a)
    s_check = c_check.max()
    sum_a = a.sum()
    if sum_a > 0.01:
        best_obj = 2.0 * n * s_check / (sum_a * sum_a)

    sweep = 0
    lambda_step = S * 0.5

    while time.monotonic() - start < max_time:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            ai_old = a[i]
            x_new = coord_descent_update(a, c, i, lambda_val, n, n_binary_iters)

            delta = x_new - ai_old
            if abs(delta) > 1e-14:
                k_min = i
                k_max = min(2 * n - 2, i + n - 1)
                if k_min <= k_max:
                    lags = np.arange(k_min, k_max + 1)
                    partner_indices = np.arange(k_min - i, k_max - i + 1)
                    partners = a[partner_indices].copy()
                    partners[i] = x_new  # a[i] itself when k=2i
                    # Actually for k=2i, partner is a[k-i] = a[i] = x_new
                    # But we're updating a[i], so the partner for k=2i is a[i] itself
                    # c[2i] includes a[i]^2, so delta * a[k-i] where k-i = i means delta * a[i]
                    # But a[i] is changing, so the update is:
                    # c[k] += delta * a[k-i] for k != 2i
                    # c[2i] += 2*delta*a[i] - delta^2 (exact quadratic update)
                    # Simplification: c[k] += delta * partners for k != 2i
                    # c[2i] += delta * (2*ai_old + delta) = delta*2*x_new - delta^2

                    # For efficiency, use linear approximation (good enough for small delta):
                    c[lags] += delta * partners

                a[i] = x_new

        # Dual ascent for lambda
        current_sum = a.sum()
        lambda_step /= (1.0 + 0.01 * sweep)
        lambda_val += lambda_step * (current_sum - S)
        lambda_val = np.clip(lambda_val, -100.0, 100.0)

        # Periodic objective check
        if sweep % check_interval == 0:
            c_chk = fft_convolve_self(a)
            sum_chk = a.sum()
            if sum_chk > 0.01:
                obj = 2.0 * n * c_chk.max() / (sum_chk * sum_chk)
            else:
                obj = np.inf
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            if sweep % (check_interval * 5) == 0:
                elapsed = time.monotonic() - start
                remain = max_time - elapsed
                print("  CD sweep %d: obj=%.8f, sum=%.6f, lam=%.4f, elapsed=%.0fs" %
                      (sweep, obj, sum_chk, lambda_val, elapsed))

    # Final: project to simplex and evaluate
    a = project_simplex(a, S)
    c_final = fft_convolve_self(a)
    obj = 2.0 * n * c_final.max() / S**2
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def raised_cosine_init(n, center_frac=0.35, width_frac=0.5):
    """Generate a raised-cosine initialization."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.zeros(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    for cf in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6]:
            inits.append(raised_cosine_init(n, cf, wf))

    for sigma in [0.1, 0.15, 0.2, 0.25]:
        for mu in [0.3, 0.4, 0.5, 0.6]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    for p in [0.5, 1.0, 1.5]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1 - t) ** p, 1e-10))

    return inits


def peak_shaving(a, n, S, n_peaks=5, n_steps=30, lr_scale=0.01):
    """Surgical peak shaving on top convolution lags."""
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.99)
            if weight < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a
