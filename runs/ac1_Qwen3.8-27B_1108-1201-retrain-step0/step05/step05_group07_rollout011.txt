"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on LSE-smoothed
objective with softmax parameterization, beta annealing, and
projected gradient polishing.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0)
    if v.sum() <= S:
        return v + (S - v.sum()) / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def softmax(w):
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()

def lse_objective_and_grad(w, beta):
    """LSE-smoothed objective and gradient for softmax parameterization."""
    a = softmax(w)
    n = len(a)
    c = fft_convolve_self(a)
    
    # LSE
    bc = beta * c
    bc_max = bc.max()
    e = np.exp(bc - bc_max)
    tot = e.sum()
    q = e / tot
    LSE = (bc_max + np.log(tot)) / beta
    
    # Gradient
    s = np.dot(q, c)
    
    # r_m = sum_k q_k * a_{k-m}
    a_flip = a[::-1]
    fsize2 = 1 << ((2 * n - 1) + n - 1).bit_length()
    Q = fft(q, fsize2)
    AF = fft(a_flip, fsize2)
    conv_q_af = np.real(ifft(Q * AF))
    r = conv_q_af[n - 1: n - 1 + n]
    
    grad = 2.0 * beta * a * (r - s)
    
    return LSE, grad

def lse_polish(w, beta, maxiter=200, tol=1e-8):
    """Run L-BFGS to minimize LSE at fixed beta."""
    def func(w_vec):
        f, g = lse_objective_and_grad(w_vec, beta)
        return f, g
    
    result = minimize(func, w, method='L-BFGS-B', 
                      options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-6,
                               'maxfun': maxiter * 2})
    return result.x

def optimize_with_annealing(a0, betas, maxiter_per_beta=100, tol=1e-7):
    """Run L-BFGS with beta annealing."""
    n = len(a0)
    a = a0.copy()
    # Initialize w from a
    w = np.log(a + 1e-15)
    # Center w (doesn't change softmax)
    w = w - w.mean()
    
    for beta in betas:
        w = lse_polish(w, beta, maxiter=maxiter_per_beta, tol=tol)
        a = softmax(w)
    
    return a

def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS on LSE-smoothed objective
with softmax parameterization, beta annealing, multiple restarts, and
projected subgradient polishing. Different from prior subgradient-only approach
by leveraging quasi-Newton methods in log-space for faster convergence.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    if v.sum() <= S:
        return v + (S - v.sum()) / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_fn(w):
    """Numerically stable softmax."""
    w_shifted = w - w.max()
    e = np.exp(w_shifted)
    return e / e.sum()


def lse_objective_and_grad(w, beta):
    """
    Compute LSE_beta(conv(softmax(w), softmax(w))) and its gradient w.r.t. w.
    Uses the identity: dLSE/dw_m = 2*beta*a_m*(r_m - s)
    where r_m = sum_k q_k * a_{k-m}, s = sum_k q_k * c_k.
    """
    a = softmax_fn(w)
    n = len(a)
    c = fft_convolve_self(a)

    # LSE via log-sum-exp
    bc = beta * c
    bc_max = bc.max()
    e = np.exp(bc - bc_max)
    tot = e.sum()
    q = e / tot
    LSE = (bc_max + np.log(tot)) / beta

    # s = sum_k q_k * c_k
    s = np.dot(q, c)

    # r_m = sum_k q_k * a_{k-m} via FFT cross-correlation
    a_flip = a[::-1]
    total_len = (2 * n - 1) + n - 1  # = 3n - 2
    fsize2 = 1 << (total_len - 1).bit_length()
    Q = fft(q, fsize2)
    AF = fft(a_flip, fsize2)
    conv_val = np.real(ifft(Q * AF))
    r = conv_val[n - 1: n - 1 + n]

    grad = 2.0 * beta * a * (r - s)

    return LSE, grad


def lse_lbfgs_single_beta(w, beta, maxiter=100, ftol=1e-8, deadline=None):
    """Run L-BFGS to minimize LSE at fixed beta with optional time deadline."""
    calls = [0]

    def func(w_vec):
        calls[0] += 1
        if deadline is not None and time.monotonic() > deadline:
            raise TimeoutError
        f, g = lse_objective_and_grad(w_vec, beta)
        return f, g

    try:
        result = minimize(func, w, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': ftol,
                                   'gtol': 1e-5, 'maxfun': maxiter * 3,
                                   'iprint': -1})
        return result.x, result.nit, result.success
    except TimeoutError:
        return w, calls[0], False
    except Exception:
        return w, calls[0], False


def optimize_annealing(a_init, beta_schedule, maxiter_per_beta=80,
                       ftol=1e-7, deadline=None):
    """Run L-BFGS with beta annealing schedule."""
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    w = np.log(a)
    w = w - w.mean()

    n_it_total = 0
    for beta in beta_schedule:
        if deadline is not None and time.monotonic() > deadline - 5:
            break
        w, n_it, _ = lse_lbfgs_single_beta(w, beta, maxiter=maxiter_per_beta,
                                           ftol=ftol, deadline=deadline)
        n_it_total += n_it

    a_final = softmax_fn(w)
    return a_final, n_it_total


def compute_obj(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def subgradient_polish(a_init, deadline, max_iters=2000):
    """Projected subgradient method for final polishing."""
    a = a_init.copy()
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 200:
            # Add noise to escape
            noise = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + np.random.default_rng(it).normal(0, noise, n), S)
            stuck = 0

        # Top-k subgradient
        k = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        median_c = np.median(c)
        weights = np.maximum(c[top_idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for ii in range(k):
            ki = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                indices = ki - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * a[indices]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * (cur_max - c.mean()) / (gnorm * gnorm + 1e-30)
        step = min(step, 3.0 / gnorm)

        accepted = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5
        if not accepted:
            a = project_simplex(a - 0.01 * step * grad, S)

    return best_a

    a = a_init.copy()

        base_for_init = global_a if global_a is not None else None

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            t_ddl = time.monotonic() + t_per_restart
            if t_ddl > deadline:
                t_ddl = deadline - 10

            # Generate diverse init
            if base_for_init is not None:
                noise_level = 0.05 + 0.1 * ri / n_restarts
                a_init = base_for_init.copy()
                a_init += rng.normal(0, noise_level * a_init.mean(), len(a_init))
                a_init = np.maximum(a_init, 1e-10)
                a_init = a_init / a_init.sum()
            else:
                a_init = generate_diverse_init(8192, rng)

            # Run annealing
            betas_r = [10, 30, 80, 200, 500, 1000]
            maxiter_r = max(20, int(80 * t_per_restart / max(1, t_per_restart)))
            a_new, iters_r = optimize_annealing(a_init, betas_r,
                                                maxiter_per_beta=min(80, maxiter_r),
                                                ftol=1e-7, deadline=t_ddl)
            try_update(a_new, f"Phase 2 restart {ri + 1}")

    # === Phase 3: Subgradient + LSE hybrid polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        print(f"\n=== Phase 3: Hybrid polish ({remaining:.0f}s) ===")
        a_curr = global_a.copy()
        n = len(a_curr)
        S = a_curr.sum()

        # Alternate between subgradient and LSE direct
        for phase_idx in range(6):
            if time.monotonic() > deadline - 15:
                break
            t_phase = min(remaining / 6, deadline - time.monotonic() - 10)
            if t_phase < 5:
                break

            if phase_idx % 2 == 0:
                a_curr = subgradient_polish(a_curr, time.monotonic() + t_phase,
                                            max_iters=int(t_phase * 50))
            else:
                a_curr = lse_polish_direct(a_curr, time.monotonic() + t_phase,
                                           beta_start=200, beta_end=2000, n_stages=5)
            try_update(a_curr, f"Phase 3 iter {phase_idx}")

    # === Phase 4: Final intensive refinement ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        print(f"\n=== Phase 4: Final refinement ({remaining:.0f}s) ===")
        a_curr = global_a / global_a.sum()
        a_curr = np.maximum(a_curr, 1e-15)

        # Very high beta LSE
        betas_final = [500, 1000, 2000, 4000, 8000, 12000]
        a_final, _ = optimize_annealing(a_curr, betas_final, maxiter_per_beta=40,
                                        ftol=1e-9, deadline=deadline - 10)
        try_update(a_final, "Phase 4 high-beta L-BFGS")

        # Quick subgradient cleanup
        if time.monotonic() < deadline - 10:
            a_final2 = subgradient_polish(a_final, deadline - 5, max_iters=500)
            try_update(a_final2, "Phase 4 subgrad cleanup")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = compute_obj(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'=' * 60}")

    return final_list
