"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization using full LSE smoothing with:
1. FISTA (forward-backward splitting with Nesterov momentum) for coarse optimization
2. Switching to high-beta subgradient for fine optimization
3. Multi-restart with spectral perturbations
4. Primary length n=16384, with n=32768 if time permits
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A, fsize))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient_fft(a, c, beta):
    """Compute gradient of LSE smoothing of max(c) where c = a*a.
    
    Returns grad such that grad = sum_k w_k * grad(c_k) where
    w_k = exp(beta*(c_k - c_max)) / sum_j exp(beta*(c_j - c_max)).
    """
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    # Compute gradient = 2 * sum_k w_k * P_k(a)
    # This equals 2 * (w * a_rev)[n-1 : 2n-2]
    a_rev = a[::-1]
    # Need FFT size >= (2n-1) + n - 1 = 3n - 2
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    g = np.real(ifft(W * AR, fsize))
    # The gradient component i is at position i + (n-1) in g
    grad = 2.0 * g[n-1 : 2*n-1]
    return grad


def lse_value(c, beta):
    """Compute LSE value (approximate max)."""
    c_max = c.max()
    scaled = np.clip(beta * (c - c_max), -500, 0)
    return c_max + np.log(max(np.exp(scaled).sum(), 1e-300)) / beta


def objective_value(a):
    """Compute the actual objective."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def run_subgradient_polish(a_init, S, max_time, rng, top_k=10):
    """Polish using subgradient of the actual max (top-k active set)."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step = 1.0 / (50.0 * n)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            step = min(step * 1.05, 1.0 / (10.0 * n))
        else:
            step = max(step * 0.7, 1.0 / (10000.0 * n))

        # Find top-k entries
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        # Weight by proximity to max
        weights = np.exp(-0.5 * (top_vals - cur_max) ** 2 / max(1e-20, (cur_max - top_vals[-1]) ** 2 + 1e-20))
        weights = weights / weights.sum()

        # Compute gradient as weighted average of gradients of top entries
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            # grad(c_k)_i = 2*a_{k-i} for valid k-i
            idx = ki - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += wi * 2.0 * a[idx[valid]]

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.1 * grad, S)
            a = a_new

        # Occasional perturbation
        if it % 500 == 0 and (time.monotonic() - start) / max_time < 0.5:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_adam(a_init, S, max_time, rng, lr=0.01, beta_start=30.0, beta_end=500.0):
    """Adam optimizer with LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.monotonic() - start < max_time:
        t += 1
        progress = (time.monotonic() - start) / max_time
        log_beta = np.log(beta_start) + progress * (np.log(beta_end) - np.log(beta_start))
        beta = np.exp(log_beta)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        grad = lse_gradient_fft(a, c, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Decay learning rate
        lr_t = lr * (1.0 - 0.5 * progress)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        # Periodic restart
        if t % 3000 == 0 and rng.random() < 0.1:
            noise_scale = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

    return best_a, best


def spectral_init(n, rng):
    """Generate a spectral initialization: random phase in frequency domain."""
    fsize = 1 << (2 * n - 2).bit_length()
    # Random magnitude (flat) and random phase
    mag = np.ones(fsize)
    phase = rng.uniform(0, 2 * np.pi, fsize)
    # Make it Hermitian for real output
    phase[0] = 0
    A = mag * np.exp(1j * phase)
    A[0] = A[0].real
    # Enforce Hermitian symmetry
    for i in range(1, fsize // 2 + 1):
        A[fsize - i] = np.conj(A[i])
    a = np.real(ifft(A, fsize))[:n]
    a = np.abs(a)  # Take absolute value to ensure non-negativity
    return a

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization using full LSE smoothing:
1. FISTA with Nesterov momentum and adaptive beta (coarse to fine)
2. Adam optimizer with LSE gradient for exploration
3. Active-set subgradient polishing for final refinement
4. Multi-restart with spectral perturbations
5. Primary n=16384, secondary n=32768 if time permits
6. Zadoff-Chu and spectral initializations for diversity

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A, fsize))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient_fft(a, c, beta):
    """Gradient of LSE smoothing of max(c) where c = a*a (self-convolution).
    
    Computes sum_k w_k * grad(c_k) where w_k = softmax(beta * c).
    Result: 2 * (w * a_rev)[n-1 : 2n-1] where * is linear convolution.
    """
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    a_rev = a[::-1]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    g = np.real(ifft(W * AR, fsize))
    grad = 2.0 * g[n - 1:2 * n - 1]
    return grad


def objective_value(a):
    """Compute the objective 2n*max(a*a)/sum(a)^2."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain (phase perturbation)."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0.0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten(a, alpha, rng):
    """Spectral flattening: damp peaks in power spectrum, boost valleys."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)
