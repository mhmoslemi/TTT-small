"""
Minimize 2*n*max(self-convolution)/sum^2 for non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, exact subgradient
with backtracking, coordinate descent, and multiplicative peak shaving.
Key insight: Asymmetry in the sequence reduces the center lag of the
convolution (which is the anti-correlation, not sum of squares), allowing
objectives well below 2. We exploit this with diverse asymmetric
initializations and aggressive peak-directed optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_a(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def obj_value(a, c):
    """Compute objective given a and its convolution c."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, c, beta):
    """LSE-approximated gradient of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()  # numerical stability
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def active_grad(a, c, top_k=3):
    """Gradient using top-k active lags (hard max approximation)."""
    n = len(a)
    cm = c.max()
    # Find lags within a small threshold of max
    threshold = cm - cm * 1e-6
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    # Use top_k most active
    if len(active) > top_k:
        idx = np.argsort(c[active])[::-1][:top_k]
        active = active[idx]
    # Uniform weights over active set
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    n_c = len(w)
    fsize = 1 << (n_c + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, max_time, rng, beta_start=50.0, beta_end=50000.0,
              beta_steps=200, momentum_gap=300):
    """FISTA with adaptive LSE smoothing."""
    S = np.sqrt(2.0 * n)
    a = normalize_a(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)

        c = fft_conv_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_val:
            best_val = cur_obj
            best_a = y.copy()
            last_improve = it

        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve
        if stagnant > momentum_gap:
            t_mom = 1.0
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stagnant / (10000.0 / max(1, n // 100)))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_conv_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_val:
        best_val = final_obj
        best_a = a.copy()
    return best_a, best_val


def run_subgradient(a_init, n, max_time, rng, top_k=5):
    """Exact subgradient method with backtracking line search."""
    S = np.sqrt(2.0 * n)
    a = normalize_a(a_init, n)
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()
            last_improve = it

        grad = active_grad(a, c, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        improved = False
        for trial in range(10):
            s_trial = step / (2 ** trial)
            a_test = project_simplex(a - s_trial * grad, S)
            c_test = fft_conv_self(a_test)
            test_obj = 2.0 * n * c_test.max() / (S * S)
            if test_obj < cur_obj * (1 - 1e-8):
                a = a_test
                step = s_trial * 1.5  # increase step if we went too small
                improved = True
                break
        if not improved:
            step *= 0.5
            a = project_simplex(a - step * grad, S)

        # Periodic perturbation
        stagnant = it - last_improve
        if stagnant > 200:
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it

    c_final = fft_conv_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_val:
        best_val = final_obj
        best_a = a.copy()
    return best_a, best_val


def run_coord_descent(a_init, n, max_time, rng, n_sweeps=100):
    """Coordinate descent: for each coordinate, greedily adjust to reduce peak."""
    S = np.sqrt(2.0 * n)
    a = normalize_a(a_init, n)
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    c = fft_conv_self(a)
    best_val = 2.0 * n * c.max() / (S * S)

    sweep = 0
    while time.monotonic() - start < max_time and sweep < n_sweeps:
        sweep += 1
        c = fft_conv_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a.copy()

        # Find peak lags
        peak_idx = np.argmax(c)
        # Also get a few secondary peaks
        top_peaks = np.argsort(c)[::-1][:5]

        # For each coordinate in random order, try to reduce its contribution
        order = rng.permutation(n)
        for i in order:
            # Time check every 1024 coordinates
            if i % 1024 == 1023:
                if time.monotonic() - start > max_time:
                    break

            # Contribution of a[i] to peak lag: a[peak_idx - i] (if valid)
            contrib = 0.0
            for k in top_peaks:
                j = k - i
                if 0 <= j < n:
                    contrib += a[j]
            
            if contrib <= 0:
                continue

            # Try reducing a[i]
            delta = min(a[i] * 0.01, contrib * 0.001)
            if delta < 1e-12:
                continue

            a_trial = a.copy()
            a_trial[i] -= delta
            # Redistribute to a random coordinate with low contribution
            j_rand = rng.integers(n)
            a_trial[j_rand] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * (S / s)

            # Quick evaluation (only check if it helps)
            # Approximate: check if the peak lag decreased
            new_peak_approx = cur_max - 2 * delta * contrib / S * 2
            if new_peak_approx < cur_max * 0.999:
                a = a_trial

    c_final = fft_conv_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_val:
        best_val = final_obj
        best_a = a.copy()
    return best_a, best_val


def peak_shave(a_in, n, S, n_peaks=10, n_iter=200, lr=0.01):
    """Multiplicative peak shaving: reduce elements that contribute to peaks."""
    a = a_in.copy()
    for _ in range(n_iter):
        c = fft_conv_self(a)
        cm = c.max()
        # Top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.999)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += contrib * weight
        
        rmax = resp.max()
        if rmax < 1e-30:
            break
        
        # Multiplicative update
        lr_eff = lr * S / (n * rmax)
        a = a * np.exp(-lr_eff * resp)
        a = project_simplex(a, S)
    
    return a


def gen_asymmetric_init(n, rng, style=0):
    """Generate diverse asymmetric initializations."""
    t = np.linspace(0, 1, n)
    if style == 0:
        # Exponential decay
        alpha = rng.uniform(2.0, 8.0)
        a = np.exp(-alpha * t)
    elif style == 1:
        # Power law decreasing
        p = rng.uniform(1.0, 4.0)
        a = (1.0 - t) ** p + 0.01
    elif style == 2:
        # Ramp up then flat
        frac = rng.uniform(0.3, 0.7)
        a = np.ones(n)
        mask = t < frac
        a[mask] = t[mask] / frac
    elif style == 3:
        # Ramp down then flat
        frac = rng.uniform(0.3, 0.7)
        a = np.ones(n)
        mask = t > frac
        a[mask] = (1.0 - t[mask]) / (1.0 - frac)
    elif style == 4:
        # Two-bump asymmetric
        mu1 = rng.uniform(0.1, 0.3)
        mu2 = rng.uniform(0.6, 0.9)
        sig = rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - mu1) / sig) ** 2) + 0.5 * np.exp(-0.5 * ((t - mu2) / sig) ** 2)
    elif style == 5:
        # Sine asymmetric
        k = rng.integers(1, 4)
        phase = rng.uniform(0, np.pi)
        a = 0.5 + 0.5 * np.sin(np.pi * k * t + phase)
        a = np.maximum(a, 0.01)
    elif style == 6:
        # Linear with noise
        slope = rng.uniform(-1, 1)
        a = 1.0 + slope * (t - 0.5)
        a = a + rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)
    elif style == 7:
        # Cosine window, asymmetric center
        center = rng.uniform(0.2, 0.4)
        width = rng.uniform(0.3, 0.6)
        x = (t - center) / (width / 2.0)
        a = np.ones(n)
        mask = np.abs(x) < 1
        a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
        a[~mask] = 0.0
        a = np.maximum(a, 1e-8)
    else:
        a = np.ones(n) / n
    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: LSE-FISTA -> subgradient -> coordinate descent -> peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_seq(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_a(a, n)
        c = fft_conv_self(a)
        val = 2.0 * n * c.max() / (S * S)
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # === Phase 1: Quick FISTA on given sequence ===
    if given is not None:
        n_g = len(given)
        t1 = min(90.0, (deadline - time.monotonic()) * 0.08)
        print("=== Phase 1: FISTA on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_fista(given, n_g, t1, rng,
                           beta_start=100.0, beta_end=20000.0, beta_steps=100)
        try_seq(a1)

    # === Phase 2: Main subgradient optimization ===
    if global_a is not None:
        n_main = len(global_a)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.35
        print("=== Phase 2: Subgradient (n=%d, %.0fs) ===" % (n_main, t2))
        a2, v2 = run_subgradient(global_a, n_main, t2, rng, top_k=5)
        try_seq(a2)

        # === Phase 3: FISTA with high beta ===
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t3 = remaining * 0.2
            print("=== Phase 3: High-beta FISTA (%.0fs) ===" % t3)
            a3, v3 = run_fista(global_a, n_main, t3, rng,
                               beta_start=5000.0, beta_end=80000.0, beta_steps=50)
            try_seq(a3)

        # === Phase 4: Peak shaving ===
        if global_a is not None:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                n_p = len(global_a)
                S_p = np.sqrt(2.0 * n_p)
                t4 = remaining * 0.15
                print("=== Phase 4: Peak shaving (%.0fs) ===" % t4)
                a_p = peak_shave(global_a, n_p, S_p, n_peaks=15,
                                 n_iter=int(200 * t4 / 60), lr=0.008)
                try_seq(a_p)

                # FISTA polish after shaving
                if a_p.sum() > 0:
                    remaining = deadline - time.monotonic()
                    if remaining > 20:
                        a_p2, _ = run_fista(a_p, n_p, remaining * 0.5, rng,
                                            beta_start=10000.0, beta_end=100000.0,
                                            beta_steps=30)
                        try_seq(a_p2)

    # === Phase 5: Try diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_try = 8192
        t5 = remaining * 0.3
        print("=== Phase 5: Diverse inits (n=%d, %.0fs) ===" % (n_try, t5))
        
        # Generate inits and score them
        inits = []
        scores = []
        for style in range(8):
            for _ in range(3):
                a_init = gen_asymmetric_init(n_try, rng, style)
                a_init = normalize_a(a_init, n_try)
                c_init = fft_conv_self(a_init)
                score = c_init.max()
                inits.append(a_init)
                scores.append(score)
        
        # Sort by score
        order = np.argsort(scores)
        per_init = max(5.0, t5 / min(8, len(order)))
        
        for idx in order[:8]:
            if time.monotonic() > deadline - 30:
                break
            init = inits[idx]
            print("  Init %d (score=%.4f)" % (idx, scores[idx]))
            a_init, _ = run_fista(init, n_try, per_init, rng,
                                  beta_start=30.0, beta_end=30000.0, beta_steps=80)
            try_seq(a_init)

            # Quick peak shave
            if global_a is not None and time.monotonic() < deadline - 30:
                # Also try shaving this one
                remaining_sub = max(5.0, deadline - time.monotonic()) * 0.05
                a_shaved = peak_shave(a_init, n_try, np.sqrt(2.0 * n_try),
                                      n_peaks=10, n_iter=30, lr=0.01)
                try_seq(a_shaved)

    # === Phase 6: Try different n values ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        # Try 2x upsampling
        n_large = len(global_a) * 2
        if n_large <= 32768:
            t6 = remaining * 0.3
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_large)
            init_large = np.interp(t_new, t_old, global_a)
            init_large = normalize_a(init_large, n_large)
            print("=== Phase 6: Upsample to n=%d (%.0fs) ===" % (n_large, t6))
            a6, _ = run_fista(init_large, n_large, t6, rng,
                              beta_start=100.0, beta_end=30000.0, beta_steps=60)
            try_seq(a6)

    # === Phase 7: Final active-set refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_f = len(global_a)
            print("=== Phase 7: Final subgradient (%.0fs) ===" % remaining)
            a_f, _ = run_subgradient(global_a, n_f, remaining * 0.6, rng, top_k=10)
            try_seq(a_f)

            # One more peak shave
            if global_a is not None:
                remaining2 = deadline - time.monotonic()
                if remaining2 > 10:
                    S_f = np.sqrt(2.0 * n_f)
                    a_ps = peak_shave(global_a, n_f, S_f, n_peaks=20,
                                      n_iter=100, lr=0.005)
                    try_seq(a_ps)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
