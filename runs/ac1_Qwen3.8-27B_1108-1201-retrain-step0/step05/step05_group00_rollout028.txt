def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. N must be >= 2*len(a)-1."""
    a_pad = np.zeros(N)
    a_pad[:len(a)] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2*len(a)-1]


def lse_grad(a, N, beta):
    """
    Compute LSE-smoothed peak of conv(a,a) and its gradient w.r.t. a.
    Returns (lse_value, gradient, conv, c_max).
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        w = exp_bc / total
        lse_val = c_max + np.log(total) / beta

    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max

"""
Minimize 2*n*max(conv)/sum^2 using FISTA on LSE + top-K subgradient + peak shaving.

Key improvements over the previous L-BFGS approach:
1. FISTA (Nesterov-accelerated projected gradient) replaces L-BFGS: no Hessian memory,
   more robust at high beta where the objective is nearly piecewise-linear.
2. Multiplicative (exponential) updates as an alternative mode for better stability.
3. Direct peak shaving via targeted coordinate moves at the end.
4. Adaptive step size with infrequent backtracking (not per-iteration).
5. Multi-scale beta schedule: coarse-to-fine exploration.
6. Crossover between diverse solutions to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def lse_grad(a, N, beta):
    """
    Compute LSE-smoothed peak of conv(a,a) and its gradient w.r.t. a.
    Returns (lse_value, gradient, c, c_max).
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = float(c.max())
    exp_bc = np.exp(beta * (c - c_max))
    total = float(exp_bc.sum())
    if total < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        w = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    W = fft(w, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def toppk_grad(a, N, top_k):
    """
    Gradient based on top-K peaks of convolution.
    Returns (topk_mean, gradient, c, c_max).
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = float(c.max())
    sort_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[sort_idx]
    total = float(top_vals.sum())
    if total < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(sort_idx[0])] = 1.0
        topk_mean = c_max
    else:
        w = np.zeros(2 * n - 1)
        w[sort_idx] = top_vals / total
        topk_mean = float(top_vals.mean())
    W = fft(w, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return topk_mean, grad, c, c_max


def fista_lse(a_init, n, S, N, budget_s, beta_init, beta_max, rng,
              use_multiplicative=False):
    """
    FISTA on LSE-smoothed peak with adaptive beta.
    Supports additive and multiplicative update modes.
    """
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    a = project_simplex(a, S)
    best_a = a.copy()
    best_cmax = float(compute_conv(a, N).max())

    beta = beta_init
    x = a.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    stagnant = 0
    it = 0

    # Initial step size
    c_init = compute_conv(x, N)
    cmax_init = float(c_init.max())
    step = 4.0 * S / (n * max(beta, 1.0) * max(cmax_init, 1e-10))
    step = min(step, 2.0 * S / n)

    while time.monotonic() - start < budget_s:
        it += 1

        # Adaptive beta
        if it % 60 == 0:
            beta = min(beta_max, beta * 1.35)
            c_tmp = compute_conv(y, N)
            cmax_tmp = float(c_tmp.max())
            step = 4.0 * S / (n * max(beta, 1.0) * max(cmax_tmp, 1e-10))
            step = min(step, 2.0 * S / n)

        # Compute gradient at y
        f_val, g, c, c_max = lse_grad(y, N, beta)

        # Gradient step
        if use_multiplicative:
            # Multiplicative update: a[i] * exp(-step * g[i] / (a[i] + eps))
            eps = S / (100.0 * n)
            update = np.exp(-step * g / (x + eps))
            x_new = x * update
            x_new = project_simplex(x_new, S)
        else:
            # Additive update with projection
            x_new = project_simplex(y - step * g, S)

        # Evaluate
        c_new = compute_conv(x_new, N)
        cmax_new = float(c_new.max())

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = x_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Step size adjustment
        if stagnant > 100 and stagnant % 100 == 0:
            step *= 0.7
        elif stagnant > 500:
            # Spectral perturbation to escape
            A = fft(x_new, N)
            mag = np.abs(A)
            phase = np.angle(A)
            n_pert = max(1, int(N * 0.2))
            pert_idx = rng.choice(N, size=n_pert, replace=False)
            phase[pert_idx] += rng.normal(0, 0.4, n_pert)
            A_new = mag * np.exp(1j * phase)
            x_new = np.real(ifft(A_new))[:n]
            x_new = project_simplex(x_new, S)
            t = 1.0
            stagnant = 0
            step = 4.0 * S / (n * max(beta, 1.0) * max(cmax_new, 1e-10))
            step = min(step, 2.0 * S / n)

        # Nesterov momentum update
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        y = x_new + ((t_old - 1.0) / t) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new

    return best_a, best_cmax


def toppk_nesterov(a_init, n, S, N, budget_s, rng, top_k_init=30, top_k_max=512):
    """Nesterov-accelerated top-K subgradient method for polishing."""
    a = project_simplex(a_init, S)
    x = a.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_cmax = float(compute_conv(a, N).max())
    start = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - start < budget_s:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        f_val, g, c, c_max = toppk_grad(y, N, top_k)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-16:
            break

        step = 3.0 * S / (n * max(top_k, 1))
        x_new = project_simplex(y - step * g / gnorm, S)

        c_new = compute_conv(x_new, N)
        cmax_new = float(c_new.max())

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = x_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 300:
            A = fft(x_new, N)
            mag = np.abs(A)
            phase = np.angle(A)
            n_pert = max(1, int(N * 0.15))
            pert_idx = rng.choice(N, size=n_pert, replace=False)
            phase[pert_idx] += rng.normal(0, 0.3, n_pert)
            A_new = mag * np.exp(1j * phase)
            x_new = np.real(ifft(A_new))[:n]
            x_new = project_simplex(x_new, S)
            t = 1.0
            stagnant = 0

        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        y = x_new + ((t_old - 1.0) / t) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new

    return best_a, best_cmax


def peak_shave(a_init, n, S, N, budget_s, rng):
    """
    Direct peak reduction: identify elements contributing to the max peak
    and perform targeted coordinate moves to reduce it.
    """
    a = project_simplex(a_init, S)
    best_a = a.copy()
    best_cmax = float(compute_conv(a, N).max())
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < budget_s:
        it += 1
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])

        if c_max < best_cmax - 1e-12:
            best_cmax = c_max
            best_a = a.copy()

        # Find contributing pairs
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

        if np.max(contrib) < 1e-15:
            break

        # Find element to reduce (high contribution to peak, low overall impact)
        # Compute "global impact": how much does reducing a[i] affect all peaks?
        # Approximate: elements near the edges have less global impact
        # (they only appear in a few c[k] terms)
        impact = np.zeros(n)
        for i in range(n):
            # a[i] appears in c[k] for k in [i, i+n-1] (as a[i]*a[k-i])
            # and for k in [i, i+n-1] (as a[k-i]*a[i])
            # So it appears in c[i..i+n-1] and c[i..i+n-1] (same range)
            # Number of c[k] it appears in: min(n, 2n-1-i) ... roughly n for middle elements
            # For edge elements (i near 0 or n-1), fewer
            count = min(i + 1, 2 * n - 1 - i)
            impact[i] = count

        # Score: high contrib / (impact + 1) means good candidate to reduce
        # But we also want the element to be relatively large
        score = contrib / (impact + 1.0) * np.maximum(a, 1e-10)
        i_reduce = int(np.argmax(score))

        # Find element to increase (low contribution to peak, can absorb mass)
        # Prefer elements with low contrib to the current peak
        i_increase = int(np.argmin(contrib))
        if i_increase == i_reduce:
            # Pick second best
            sorted_idx = np.argsort(contrib)
            i_increase = int(sorted_idx[0]) if int(sorted_idx[0]) != i_reduce else int(sorted_idx[1])

        if i_reduce == i_increase:
            break

        # Transfer mass
        transfer = 0.05 * a[i_reduce]
        if transfer < 1e-15:
            break
        a[i_reduce] -= transfer
        a[i_increase] += transfer
        a = np.maximum(a, 0.0)
        a = project_simplex(a, S)

        # Check if it helped
        c_check = compute_conv(a, N)
        if c_check.max() > best_cmax * 1.01:
            # Revert
            a = np.maximum(a - np.array([0.0] * i_reduce + [transfer] + [0.0] * (i_increase - i_reduce - 1) + [-transfer] + [0.0] * (n - i_increase - 1) if i_increase > i_reduce else [0.0] * i_increase + [-transfer] + [0.0] * (i_reduce - i_increase - 1) + [transfer] + [0.0] * (n - i_reduce - 1)), 0.0)
            a = project_simplex(a, S)

    return best_a, best_cmax


def spectral_perturb(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases for restart."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    phase[rand_idx] += rng.normal(0, 0.5, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def make_init_rand(n, rng):
    """Random initialization."""
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    """Chirp-inspired initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multi_freq(n, rng):
    """Multi-frequency sinusoidal initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    for freq in [3, 7, 13, 21, 47]:
        a += 0.15 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Random Fourier magnitude, random phase initialization."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def deflation_init(n, S, N, n_iters=1500, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = 0.02 * a[i_max]
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and project to simplex of sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    return project_simplex(a, S)


def score_from_cmax(n, cmax, S):
    """Compute the objective value from n, cmax, and S."""
    return float(2.0 * n * cmax / (S * S))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses FISTA on LSE with adaptive beta, top-K subgradient polishing,
    and direct peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine n
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def try_update(a_candidate, label):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = float(c.max())
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = score_from_cmax(n, sc, S)
            print("  [%s] peak=%.6f score=%.6f (t=%.0fs)" % (label, sc, score, time.monotonic() - (deadline - budget_s + 30)))
        return sc

    # === Phase 1: Multi-start exploration with FISTA ===
    print("=== Phase 1: Multi-start FISTA exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(500.0, (deadline - t1) * 0.20)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
            # Perturbed version of given
            inits.append(("given_pert", spectral_perturb(safe_normalize(given, n, S), N, rng, 0.2)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("fourier", make_init_fourier(n, N, N) if False else make_init_fourier(n, rng, N)))
    inits.append(("deflation", deflation_init(n, S, N, 1500, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = fista_lse(a_norm, n, S, N, min(rem, 80.0), rng,
                              beta_init=5.0, beta_max=100.0,
                              use_multiplicative=False)
        try_update(a_l, name + "/FISTA")

    print("Phase 1 done. Best score: %.6f" % score_from_cmax(n, global_best, S) if global_best < np.inf else "Phase 1: no solution")

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = float(compute_conv(global_a, N).max())

    # === Phase 2: Focused FISTA with high beta ===
    print("=== Phase 2: Focused high-beta FISTA ===")
    t2 = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2) * 0.35)

    a_curr = global_a.copy()
    stall_count = 0
    block_idx = 0

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 15:
            break

        beta_start = 10.0 + 5.0 * min(stall_count, 8)
        beta_end = min(800.0, 50.0 * (1.0 + 0.5 * stall_count))
        use_mult = (block_idx % 3 == 2)

        a_new, sc_new = fista_lse(a_curr, n, S, N, block_time, rng,
                                   beta_init=beta_start, beta_max=beta_end,
                                   use_multiplicative=use_mult)
        block_idx += 1

        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_count = 0
            score = score_from_cmax(n, sc_new, S)
            print("  [P2-%d] score=%.6f (beta %.0f-%.0f, mult=%s)" % (block_idx, score, beta_start, beta_end, use_mult))
        else:
            a_curr = a_new.copy()
            stall_count += 1
            if stall_count == 2:
                a_curr = spectral_perturb(a_curr, N, rng, 0.3)
                stall_count = 0
            elif stall_count >= 4:
                a_curr = spectral_perturb(a_curr, N, rng, 0.5)
                stall_count = 0

    # === Phase 3: Top-K Nesterov polishing ===
    print("=== Phase 3: Top-K Nesterov polishing ===")
    t3 = time.monotonic()
    remaining3 = min(400.0, (deadline - t3) * 0.20)
    if remaining3 > 20:
        a_ng, sc_ng = toppk_nesterov(global_a, n, S, N, remaining3, rng,
                                      top_k_init=20, top_k_max=300)
        try_update(a_ng, "topK_nesterov")

    # === Phase 4: Peak shaving ===
    print("=== Phase 4: Peak shaving ===")
    t4 = time.monotonic()
    remaining4 = min(200.0, (deadline - t4) * 0.10)
    if remaining4 > 15:
        a_ps, sc_ps = peak_shave(global_a, n, S, N, remaining4, rng)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Final high-beta FISTA polish ===
    print("=== Phase 5: Final high-beta FISTA ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 10
    if remaining5 > 30:
        a_fin, sc_fin = fista_lse(global_a, n, S, N, min(remaining5, 300.0), rng,
                                   beta_init=50.0, beta_max=1000.0,
                                   use_multiplicative=False)
        try_update(a_fin, "final_FISTA")

    # === Phase 6: Final top-K polish ===
    remaining6 = deadline - time.monotonic() - 5
    if remaining6 > 20:
        a_ng2, sc_ng2 = toppk_nesterov(global_a, n, S, N, min(remaining6, 150.0), rng,
                                        top_k_init=10, top_k_max=200)
        try_update(a_ng2, "final_topK")

    # === Verification and return ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = score_from_cmax(len(global_a), global_best, S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
