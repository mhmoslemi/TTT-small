import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def solve_lp_for_b(a, S, max_iter=100000):
    """
    Solve: minimize t subject to (a*b)[k] <= t for all k, b >= 0, sum(b) = S.
    Returns the optimal b and t.
    """
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t (total n+1 variables)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (a*b)[k] - t <= 0 for k = 0, ..., 2n-2
    # (a*b)[k] = sum_{j} a[k-j] * b[j] (valid j)
    # So: sum_j a[k-j]*b[j] - t <= 0
    # A_ub has 2n-1 rows, n+1 columns
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, -1] = -1.0  # -t term
    
    # Fill in the a*b part
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    # Equality constraint: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [S]
    
    # Bounds: b_j >= 0, t is free (but t >= 0 implicitly)
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        b_opt = result.x[:n]
        t_opt = result.x[-1]
        return b_opt, t_opt
    else:
        return None, None

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with subgradient warm-up and multi-length exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import time
import sys

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, c, k):
    """Compute top-k weighted subgradient."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    # Weights proportional to how close to max
    margins = c[top_idx] - c[top_idx].min()
    if margins.sum() < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = (margins ** 2) / (margins ** 2).sum()  # squared for more focus
    
    # sg[j] = 2 * sum_k weights[i] * a[k_idx - j] for valid k_idx - j
    # This is 2 * (weights_as_sequence * a)[j] restricted to [0, n)
    # Build the weight sequence (length 2n-1)
    w_seq = np.zeros(2 * n - 1)
    for ii in range(k):
        w_seq[int(top_idx[ii])] = weights[ii]
    
    # Convolve w_seq with a
    fsize = 1 << (2*n-1 + n - 1).bit_length()
    W = fft(w_seq, fsize)
    A = fft(a, fsize)
    conv = np.real(ifft(W * A))[:n]
    
    return 2.0 * conv

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def subgradient_descent(a0, S, max_time, rng, verbose=True):
    """Nesterov-accelerated subgradient descent."""
    n = len(a0)
    a = project_simplex(a0, S)
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Nesterov step
        beta = 2.0 / (it + 3.0)  # momentum parameter
        y_new = a + beta * (a - y)
        y_new = project_simplex(y_new, S)
        
        c = fft_convolve_self(y_new)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y_new.copy()
            stuck = 0
        else:
            stuck += 1
        
        k = int(max(8, 20 + 60 * progress))
        grad = compute_subgradient(y_new, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        gap = cur_max - c[n-1]  # gap to the "average" peak
        if gap < 1e-10:
            gap = cur_max - c.mean()
        
        step = 2.0 * max(gap, 1e-6) / (gnorm * gnorm)
        step = min(step, 10.0 / gnorm)
        
        a_curr = project_simplex(y_new - step * grad, S)
        
        # Noise when stuck
        if stuck > 200:
            noise_scale = 0.01 * S / np.sqrt(n)
            a_curr = project_simplex(a_curr + rng.normal(0, noise_scale, n), S)
            stuck = 0
        
        a = a_curr
        y = y_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")
    
    return best_a, best

def build_conv_matrix(a):
    """Build sparse convolution matrix Ab where (Ab @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1)
        a_vals = a[k - j_range]
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        vals_list.append(a_vals)
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_alternating(a, S, max_time, rng, alpha=0.3, verbose=True):

    Alternating LP: fix a, find b to minimize max(a*b), blend.

    a_curr = project_simplex(a, S)
    c = fft_convolve_self(a_curr)
    best_max = c.max()
    best_a = a_curr.copy()
    
    start = time.monotonic()
    it = 0
    
    try:
        import cvxpy as cp
        have_cvxpy = True
    except ImportError:
        have_cvxpy = False
    
    use_gurobi = False

        except ImportError:
            pass

        if have_cvxpy and use_gurobi:

                prob = cp.Problem(cp.Minimize(t_var), constraints)
                prob.solve(solver=cp.GUROBI, TimeLimit=60)

                    return np.asarray(b_var.value, dtype=np.float64), t_var.value
            except Exception:

        A_ub = hstack([Ab, neg_one])

            break

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with Nesterov subgradient warm-up, multi-phase refinement, and length exploration.

Key innovation over previous approach: Uses exact LP solves to find the best
"response" sequence b given fixed a, then blends. This converges much faster
than pure subgradient methods for the final 0.1-0.5% improvement needed.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix, csc_matrix, hstack
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, k):
    """Compute top-k weighted subgradient of max conv."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    margins = c[top_idx] - c[top_idx].min()
    if margins.sum() < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = (margins ** 2) / (margins ** 2).sum()

    w_seq = np.zeros(2 * n - 1)
    for ii in range(k):
        w_seq[int(top_idx[ii])] = weights[ii]

    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w_seq, fsize)
    A = fft(a, fsize)
    conv = np.real(ifft(W * A))[:n]
    return 2.0 * conv


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def subgradient_descent(a0, S, max_time, rng, verbose=True):
    """Nesterov-accelerated subgradient descent with adaptive step."""
    n = len(a0)
    a = project_simplex(a0, S)
    y = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 150

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)

        beta = 2.0 / (it + 3.0)
        y_new = a + beta * (a - y)
        y_new = project_simplex(y_new, S)

        c = fft_convolve_self(y_new)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y_new.copy()
            stuck = 0
        else:
            stuck += 1

        k = int(max(8, 16 + 80 * progress))
        grad = compute_subgradient(y_new, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = cur_max - c.mean()
        if gap < 1e-12:
            gap = 1e-6

        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 5.0 / gnorm)

        # Line search: accept if no worse
        a_try = project_simplex(y_new - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-8):
            a_curr = a_try
        else:
            # Halve step
            step *= 0.5
            a_curr = project_simplex(y_new - step * grad, S)

        # Noise when stuck
        if stuck > patience:
            noise_scale = min(0.02, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a_curr = project_simplex(a_curr + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)

        a = a_curr
        y = y_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, max={best:.6f}")

    return best_a, best


def build_conv_matrix_sparse(a):
    """Build sparse matrix where (M @ b)[k] = (a*b)[k] for k=0..2n-2."""
    n = len(a)
    rows_parts = []
    cols_parts = []
    vals_parts = []

    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_range = np.arange(js, je + 1, dtype=np.int32)
        a_vals = a[k - j_range]
        rows_parts.append(np.full(len(j_range), k, dtype=np.int32))
        cols_parts.append(j_range)
        vals_parts.append(a_vals)

    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_response(a, S, time_limit=60):
    """
    Solve: minimize t subject to (a*b)[k] <= t, b >= 0, sum(b) = S.
    Returns optimal b and t, or (None, None) on failure.
    """
    n = len(a)

    # Try cvxpy with Gurobi first
    try:
        import cvxpy as cp
        b_var = cp.Variable(n)
        t_var = cp.Variable()
        Ab = build_conv_matrix_sparse(a)
        constraints = [b_var >= 0, cp.sum(b_var) == S, Ab @ b_var <= t_var]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.GUROBI, TimeLimit=time_limit)
        if b_var.value is not None:
            b_opt = np.asarray(b_var.value, dtype=np.float64)
            b_opt = np.maximum(b_opt, 0)
            return b_opt, float(t_var.value)
    except Exception:
        pass

    # Fallback: scipy linprog with HiGHS
    try:
        from scipy.optimize import linprog
        Ab = build_conv_matrix_sparse(a)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        neg_col = csc_matrix((-np.ones(2 * n - 1), (np.arange(2 * n - 1), np.full(2 * n - 1, n))),
                             shape=(2 * n - 1, 1))
        A_ub = hstack([Ab, neg_col], format='csc')

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        bounds = [(0, None)] * (n + 1)
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[-1]
    except Exception:
        pass

    return None, None


def lp_alternating(a0, S, max_time, rng, alpha=0.25, verbose=True):
    """
    LP-based alternating optimization.
    Each iteration: fix a, solve LP for b, blend a and b.
    """
    n = len(a0)
    a_curr = project_simplex(a0, S)
    c = fft_convolve_self(a_curr)
    best_max = c.max()
    best_a = a_curr.copy()

    start = time.monotonic()
    it = 0
    consecutive_failures = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed

        # Adaptive time limit per LP based on remaining budget
        per_solve = min(60.0, max(5.0, remaining * 0.15))

        b_opt, t_opt = solve_lp_response(a_curr, S, time_limit=per_solve)

        if b_opt is None:
            consecutive_failures += 1
            if consecutive_failures >= 3:
                print(f"    LP failed 3x in a row, stopping at iter {it}")
                break
            continue

        consecutive_failures = 0
        b_opt = np.maximum(b_opt, 0.0)
        b_opt = project_simplex(b_opt, S)

        # Adaptive blend parameter: smaller when close to convergence
        progress = min(1.0, elapsed / max_time)
        alpha_eff = alpha * (1.0 - 0.5 * progress)

        a_new = (1.0 - alpha_eff) * a_curr + alpha_eff * b_opt
        a_new = project_simplex(a_new, S)

        c_new = fft_convolve_self(a_new)
        cur_max = c_new.max()

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a_new.copy()

        a_curr = a_new

        if verbose and (it % 3 == 0 or it <= 5):
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"    LP it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, "
                  f"t_opt={t_opt:.4f}, alpha={alpha_eff:.3f}")

        # Early stopping if converging very slowly
        if it > 10 and (best_max - c_new.max()) < 1e-8 * best_max:
            # Check if we've been making tiny progress
            pass

    return best_a, best_max


def generate_init(n, rng, base=None):

        a = np.maximum(a, 0.0)

            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
    return a

        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW GLOBAL BEST: obj={obj:.6f} ({msg}) ***")

    # === Phase 1: Subgradient warm-up ===
    if a_provided is not None:
        n1 = len(a_provided)
    else:
        n1 = 8192

    S1 = np.sqrt(2.0 * n1)
    if a_provided is not None:
        a1 = a_provided * (S1 / a_provided.sum())
    else:
        a1 = np.ones(n1) * S1 / n1
    a1 = project_simplex(a1, S1)

    t1 = min(240, (deadline - time.monotonic()) * 0.12)
    print(f"\nPhase 1: Subgradient warm-up, n={n1}, time={t1:.0f}s")
    a_opt, _ = subgradient_descent(a1, S1, t1, rng, verbose=True)
    update_global(a_opt, "Phase 1 subgradient")

    # === Phase 2: LP alternating optimization (main workhorse) ===
    remaining = deadline - time.monotonic()
    t2 = min(1400, remaining * 0.58)
    if t2 > 60 and global_a is not None:
        n2 = len(global_a)
        S2 = np.sqrt(2.0 * n2)
        a_start = project_simplex(global_a.copy(), S2)

        print(f"\nPhase 2: LP alternating, n={n2}, time={t2:.0f}s")
        a_lp, _ = lp_alternating(a_start, S2, t2, rng, alpha=0.25, verbose=True)
        update_global(a_lp, "Phase 2 LP alternating")

    # === Phase 3: Intensive refinement alternating subgrad + LP ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 120:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        a_curr = project_simplex(global_a.copy(), S3)

        print(f"\nPhase 3: Refinement, n={n3}, remaining={remaining-20:.0f}s")
        n_phases = 6
        for pi in range(n_phases):
            if time.monotonic() > deadline - 20:
                break
            t_ph = min(150, (deadline - time.monotonic() - 15) / max(1, n_phases - pi))
            if t_ph < 30:
                break

            if pi % 2 == 0 and t_ph > 90:
                # LP step
                a_curr, _ = lp_alternating(a_curr, S3, t_ph, rng, alpha=0.15, verbose=False)
                update_global(a_curr, f"Phase 3 LP iter {pi}")
            else:
                # Subgradient step
                a_curr, _ = subgradient_descent(a_curr, S3, t_ph, rng, verbose=False)
                update_global(a_curr, f"Phase 3 subgrad iter {pi}")

    # === Phase 4: Try alternate lengths ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 150:
        n_curr = len(global_a)
        alt_lengths = [l for l in [4096, 12288, 16384] if l != n_curr]

        for n_try in alt_lengths:
            if time.monotonic() > deadline - 60:
                break

            S_try = np.sqrt(2.0 * n_try)
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_new = np.interp(t_new, t_old, global_a)
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new, S_try)

            t_alloc = min(200, (deadline - time.monotonic() - 30) * 0.45)
            if t_alloc < 40:
                continue

            print(f"\nPhase 4: n={n_try}, time={t_alloc:.0f}s")
            a_opt_try, _ = subgradient_descent(a_new, S_try, t_alloc * 0.4, rng, verbose=False)
            update_global(a_opt_try, f"Phase 4 n={n_try} subgrad")

            rem = deadline - time.monotonic()
            if rem > 80:
                a_lp_try, _ = lp_alternating(a_opt_try, S_try, min(140, rem * 0.5),
                                             rng, alpha=0.25, verbose=False)
                update_global(a_lp_try, f"Phase 4 n={n_try} LP")

    # === Phase 5: Final multi-restart ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        n_restarts = min(3, int(remaining // 40))

        if n_restarts > 0:
            t_per = (remaining - 20) / n_restarts
            print(f"\nPhase 5: {n_restarts} perturb+refine, {t_per:.0f}s each")
            for ri in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                noise_scale = 0.015 * S5 / np.sqrt(n5)
                a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n5), S5)
                a_pert, _ = subgradient_descent(a_pert, S5, t_per, rng, verbose=False)
                update_global(a_pert, f"Phase 5 restart {ri}")

    # === Final output ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}")
    print(f"FINAL RESULT: len={len(global_a)}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence: {ver:.6f}")
    print(f"{'=' * 60}")

    return final_list
