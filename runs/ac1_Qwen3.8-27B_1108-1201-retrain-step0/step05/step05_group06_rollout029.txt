    step = 0.01
    c = fft_convolve_self(y)
    cur_lse = lse_value(c, beta)

    while time.monotonic() - start < max_time:
        it += 1

        # Compute gradient at y
        grad = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size (conservative)
        step = 1.0 / (2.0 * beta * np.max(a) + 2.0 * np.max(a) + 1e-6)
        step = max(step, 1e-10)

        # Gradient step at y
        z = y - step * grad
        x_new = project_simplex(z, S)

        # CUTRG update
        gamma_new = 0.5 * (np.sqrt(gamma * gamma + 4.0 * gamma) + gamma)
        beta_new = ((gamma_new - 1.0) * gamma + beta_mom * gamma) / (gamma_new * gamma)

        # Compute y_new = x_new + beta_new * (x_new - x) + (beta_new/2)*(x_new - x_prev) approximation
        # Simplified CUTRG: y = x + beta*(x - x_prev)
        y_new = x_new + beta_new * (x_new - x)
        y_new = np.maximum(y_new, 0.0)

        # Check improvement
        c_new = fft_convolve_self(x_new)
        lse_new = lse_value(c_new, beta)
        actual_obj = 2.0 * n * c_new.max() / (S * S)

        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = x_new.copy()
            last_improve = it

        # Backtracking: if LSE increased too much, reduce step
        if use_backtrack and lse_new > cur_lse + step * gnorm * gnorm * 0.5:
            step *= 0.5
            y_new = x  # don't extrapolate

        x = x_new
        y = y_new
        gamma = gamma_new
        beta_mom = beta_new

        # Update c for next iteration (gradient is at y)
        c = fft_convolve_self(y)
        cur_lse = lse_value(c, beta)

        # Adaptive beta
        stuck = it - last_improve
        if stuck < 200:
            pass  # don't change
        elif stuck < 2000:
            beta = min(beta_max, beta * 1.01)
        elif stuck < 10000:
            beta = min(beta_max, beta * 1.02)
        else:
            beta = min(beta_max, beta * 1.05)

        # Periodic restart
        if it % 10000 == 0 or stuck > 20000:
            gamma = 1.0
            beta_mom = 1.0
            y = x.copy()
            if stuck > 20000:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                x = y.copy()
            last_improve = it
            stuck = 0

    return best_a


def run_active_polish(a_init, max_time, rng):
    """
    Active-set polishing using exact subgradient with adaptive tolerance.
    Uses short steps and careful direction selection.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    x = a.copy()
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = x.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive subgradient selection
        if stuck < 200:
            k = 1
            tol_frac = 0.0
        elif stuck < 1000:
            k = 2
            tol_frac = 0.0
        elif stuck < 5000:
            k = min(8, 2 + stuck // 500)
            tol_frac = 0.002
        elif stuck < 20000:
            k = min(32, 8 + stuck // 1000)
            tol_frac = 0.005
        else:
            k = min(128, 32 + stuck // 2000)
            tol_frac = 0.01

        if tol_frac > 0.0:
            grad = compute_topk_grad(y, c, k=128, tol_frac=tol_frac)
        else:
            grad = compute_topk_grad(y, c, k=k, tol_frac=0.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = S / (gnorm * (1.0 + 0.005 * min(it, 5000)))
        step = min(step, 1.0 / gnorm)
        step = max(step, 1e-12)

        z = y - step * grad
        x_new = project_simplex(z, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        x = x_new
        y = y_new
        t_mom = t_new

        if stuck > 15000 and it % 500 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 50000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            x = y.copy()
            t_mom = 1.0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def run_mirror_descent(a_init, max_time, rng, beta_init=50.0):
    """
    Mirror descent (multiplicative weights) with LSE gradient.
    Good for exploring the boundary of the feasible set.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    x = a.copy()
    best_obj = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(x)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = x.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 500:
            beta = min(50000.0, beta * 1.02)

        grad = compute_lse_grad(x, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Mirror step: x_new = x * exp(-eta * grad), then normalize
        eta = min(0.5 / (beta + 1.0), 0.01)
        # Gradient clipping for stability
        grad_clipped = np.clip(grad / gnorm, -10, 10)
        x_new = x * np.exp(-eta * grad_clipped)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new * (S / s_new)
        x_new = np.maximum(x_new, 0.0)
        x_new = normalize_seq(x_new, S)

        x = x_new

        if stuck > 10000 and it % 200 == 0:
            noise = rng.normal(0, 0.01, n)
            x = normalize_seq(x * np.exp(noise * 0.01), S)
            last_improve = it

    return best_a


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_gaussian_window(n):
    tt = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    return np.exp(-0.5 * (tt - center) / sigma) ** 2


def make_two_peak(n):
    tt = np.arange(n, dtype=np.float64)
    c1 = n * 0.25
    c2 = n * 0.75
    sigma = n * 0.12
    return np.exp(-0.5 * (tt - c1) / sigma) ** 2 + np.exp(-0.5 * (tt - c2) / sigma) ** 2


def make_tapered(n):
    """Raised cosine taper (Hann window)."""
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1 + np.cos(2 * np.pi * tt / (n - 1)))


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> LSE FISTA -> active polish -> multi-scale -> final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  [NEW BEST] obj=%.6f (n=%d, t=%.0fs)" % (obj, len(a), time.monotonic() - t_start))

    # Get initial sequence if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = 8192
    if a_start is not None:
        n0 = len(a_start)

    print("Starting with n=%d" % n0)

    # Generate diverse initializations at base n
    S = np.sqrt(2.0 * n0)
    inits = []
    if a_start is not None:
        inits.append(normalize_seq(a_start, S))

    inits.append(normalize_seq(make_cosine_init(n0), S))
    inits.append(normalize_seq(make_tapered(n0), S))
    inits.append(normalize_seq(make_gaussian_window(n0), S))
    inits.append(normalize_seq(make_two_peak(n0), S))

    # Quick evaluation
    print("\n=== Initial evaluation ===")
    best_init = None
    best_init_obj = np.inf
    for a_init in inits:
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        update_best(a_init)
    print("Best init obj=%.6f" % best_init_obj)

    # === Phase 1: LSE FISTA warm-up (moderate beta) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.15
    if p1_budget > 10:
        print("\n=== Phase 1: LSE FISTA warm (budget %.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a_p1 = run_fista_lse(best_init, p1_budget, rng,
                             beta_init=5.0, beta_max=2000.0, beta_growth=1.005)
        update_best(a_p1)
        print("  Phase 1 done: %.0fs" % (time.monotonic() - t1))

    # === Phase 2: LSE FISTA main (higher beta) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.25
    if p2_budget > 10:
        print("\n=== Phase 2: LSE FISTA main (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a_p2 = run_fista_lse(global_best_a, p2_budget, rng,
                             beta_init=20.0, beta_max=10000.0, beta_growth=1.004)
        update_best(a_p2)
        print("  Phase 2 done: %.0fs" % (time.monotonic() - t2))

    # === Phase 3: Active-set polishing ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    if p3_budget > 10:
        print("\n=== Phase 3: Active-set polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a_p3 = run_active_polish(global_best_a, p3_budget, rng)
        update_best(a_p3)
        print("  Phase 3 done: %.0fs" % (time.monotonic() - t3))

    # === Phase 4: Mirror descent exploration ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10
    if p4_budget > 10:
        print("\n=== Phase 4: Mirror descent (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a_p4 = run_mirror_descent(global_best_a, p4_budget, rng, beta_init=30.0)
        update_best(a_p4)
        print("  Phase 4 done: %.0fs" % (time.monotonic() - t4))

    # === Phase 5: Multi-scale (downsample -> optimize -> upsample) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p5_budget = remaining * 0.25
        print("\n=== Phase 5: Multi-scale (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # Downsample to n/2, optimize, upsample back
        n_small = n0 // 2
        a_small = downsample(global_best_a, factor=2)
        S_small = np.sqrt(2.0 * n_small)
        a_small = normalize_seq(a_small, S_small)
        update_best(upsample(a_small, 2))

        # Optimize at smaller scale
        a_small_opt = run_fista_lse(a_small, p5_budget * 0.4, rng,
                                     beta_init=10.0, beta_max=5000.0)
        a_small_opt = normalize_seq(a_small_opt, S_small)

        # Upsample and continue
        a_up = upsample(a_small_opt, 2)
        S_full = np.sqrt(2.0 * n0)
        a_up = normalize_seq(a_up, S_full)
        update_best(a_up)

        # Quick polish at full scale
        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 10:
            a_up2 = run_fista_lse(a_up, rem5, rng,
                                   beta_init=50.0, beta_max=20000.0)
            update_best(a_up2)
        print("  Phase 5 done: %.0fs" % (time.monotonic() - t5))

    # === Phase 6: Upsample to 2n ===
    remaining = deadline - time.monotonic()
    if remaining > 180 and n0 <= 16384:
        n_big = n0 * 2
        p6_budget = remaining * 0.30
        print("\n=== Phase 6: Upsample to n=%d (budget %.0fs) ===" % (n_big, p6_budget))
        t6 = time.monotonic()

        a_big = upsample(global_best_a, factor=2)
        S_big = np.sqrt(2.0 * n_big)
        a_big = normalize_seq(a_big, S_big)
        update_best(a_big)

        a_big_opt = run_fista_lse(a_big, p6_budget * 0.7, rng,
                                   beta_init=5.0, beta_max=5000.0)
        update_best(a_big_opt)

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 10:
            a_big2 = run_active_polish(a_big_opt, rem6, rng)
            update_best(a_big2)
        print("  Phase 6 done: %.0fs" % (time.monotonic() - t6))

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p7_budget = remaining * 0.85
        print("\n=== Phase 7: Final polish (budget %.0fs) ===" % p7_budget)
        t7 = time.monotonic()

        # Alternate between LSE and active-set
        a_cur = global_best_a.copy()
        while time.monotonic() < t7 + p7_budget:
            # LSE with very high beta
            a_lse = run_fista_lse(a_cur, min(60, p7_budget * 0.3), rng,
                                   beta_init=100.0, beta_max=50000.0)
            update_best(a_lse)

            # Active polish
            a_act = run_active_polish(a_lse, min(60, p7_budget * 0.3), rng)
            update_best(a_act)

            a_cur = global_best_a.copy()
            if time.monotonic() > t7 + p7_budget * 0.8:
                break

        print("  Phase 7 done: %.0fs" % (time.monotonic() - t7))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - t_start))
            return final_list
        else:
            print("WARNING: evaluate_sequence returned inf, using fallback")

    # Fallback
    n_fb = 8192
    a_fb = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    final_list = [float(x) for x in a_fb]
    print("Using uniform fallback, obj=%.4f" % compute_obj(a_fb))
    return final_list
