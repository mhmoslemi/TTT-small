from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def spectral_shaping_step(a, S, peak_fraction=0.02, power_gamma=0.5):
    """One step of spectral shaping to flatten the power spectrum."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    total_power = power.sum()
    
    # Target: make power spectrum flatter
    # Cap the high-power components
    mean_power = total_power / fsize
    threshold = mean_power * 3.0
    power_capped = np.minimum(power, threshold)
    
    # Blend between original and capped
    new_power = power * (1 - power_gamma) + power_capped * power_gamma
    new_power = np.maximum(new_power, 1e-30)
    
    # Preserve the phase
    phase = np.angle(A)
    A_new = np.sqrt(new_power) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)

def run_intensive_polish(a_init, S, max_time, rng):
    """
    Very tight, high-beta polishing with careful step control.
    Uses k=5 (only the 5 largest peaks) and extremely high beta.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        # Very high beta for near-exact max focusing
        beta = min(100000.0, 500.0 * (1.0 + it / 100.0))
        k = 5

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient_safe(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small steps
        progress = min(1.0, (time.monotonic() - start) / max_time)
        base_step = 0.3 * (1.0 - 0.5 * progress) / (1.0 + 0.002 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max - 1e-15:
            a = a_new
            step_scale = min(1.5, step_scale * 1.01)
        else:
            step_scale = max(0.005, step_scale * 0.4)
            # Try a tiny step in the other direction (unconstrained)
            tiny = step * 0.01
            a_try = a - tiny * grad
            a_try = np.clip(a_try, 0, None)
            a_try = normalize(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max:
                a = a_try

        # Very occasional tiny shake
        if it % 3000 == 0 and progress < 0.8:
            a_shake, val_shake = shake_keep(a, S, rng, n_shakes=10, shake_scale=0.002)
            if val_shake < best:
                a = a_shake
                best = val_shake
                best_a = a.copy()

    return best_a, best


def run_simulated_annealing(a_init, S, max_time, rng,
                            T_init=0.02, T_end=0.0001, cooling=0.9995):
    """
    Simulated annealing: accept worse solutions with probability exp(-delta/T).
    This helps escape local minima that gradient methods get stuck in.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    T = T_init
    cur_val = best

    while time.monotonic() - start < max_time:
        T = max(T_end, T * cooling)
        progress = min(1.0, (time.monotonic() - start) / max_time)

        # Generate a perturbation
        scale = (0.002 + 0.01 * T / T_init) * S / np.sqrt(n)
        a_try = a + rng.normal(0, scale, n)
        a_try = np.clip(a_try, 0, None)
        a_try = normalize(a_try, S)

        # Small gradient correction toward the best
        c = fft_convolve_self(a_try)
        k = 15
        beta = 50.0
        grad = topk_lse_gradient_safe(a_try, c, k, beta)
        a_try2 = a_try - 0.1 * grad / max(np.linalg.norm(grad), 1e-10)
        a_try2 = np.clip(a_try2, 0, None)
        a_try2 = normalize(a_try2, S)

        val_try = fft_convolve_self(a_try2).max()

        # Metropolis acceptance
        delta = val_try - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try2
            cur_val = val_try
            if cur_val < best:
                best = cur_val
                best_a = a.copy()

    return best_a, best


def topk_lse_gradient_safe(a, c, k, beta):
    """Safe version of topk_lse_grad with overflow protection."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    spread = c_top - cm
    max_spread = spread.max()
    if max_spread > 500.0 / beta:
        beta = 500.0 / max(max_spread, 1e-300)
    scaled = beta * spread
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: adaptive FISTA -> simulated annealing -> intensive polish.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 25.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_and_update(a_candidate, n_val, label):
        nonlocal global_best, global_a
        obj = obj_value(a_candidate)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = np.clip(a_candidate, 0, 1000).copy()
            print("  [%s] New global best: %.6f (n=%d)" % (label, obj, n_val))
        return obj

    # --- Phase 1: Primary optimization at n=8192 (or given length) ---
    if given is not None and len(given) >= 1024:
        n_primary = len(given)
    else:
        n_primary = 8192

    S = np.sqrt(2.0 * n_primary)
    if given is not None and len(given) == n_primary:
        a_start = normalize(given.copy(), S)
    else:
        a_start = np.ones(n_primary) * S / n_primary

    print("\n=== Primary: n=%d ===" % n_primary)
    obj0 = obj_value(a_start)
    print("  Initial objective: %.6f" % obj0)
    try_and_update(a_start, n_primary, "init")

    # Phase 1a: Adaptive FISTA (50% of primary time)
    primary_budget = min(deadline - time.monotonic(), budget_s * 0.55)
    t1 = primary_budget * 0.50
    if t1 > 30:
        print("  Phase 1a: Adaptive FISTA (%.0fs)" % t1)
        a_opt, val_opt = run_adaptive_fista(a_start, S, t1, rng,
                                            k_tight=8, k_broad_frac=0.05,
                                            beta_base=25.0, beta_growth=0.003,
                                            beta_max=15000.0,
                                            lr_base=0.7, mode_period=300)
        obj_opt = 2.0 * n_primary * val_opt / (a_opt.sum() ** 2)
        print("  Phase 1a result: %.6f" % obj_opt)
        try_and_update(a_opt, n_primary, "FISTA")
        a_current = global_a if global_a is not None else a_start

    # Phase 1b: Simulated annealing (25% of primary time)
    t2 = primary_budget * 0.25
    if t2 > 20 and time.monotonic() < deadline - 60:
        a_start_sa = global_a if global_a is not None else a_start
        print("  Phase 1b: Simulated Annealing (%.0fs)" % t2)
        a_opt, val_opt = run_simulated_annealing(a_start_sa, S, t2, rng,
                                                  T_init=0.015, T_end=0.00005,
                                                  cooling=0.9997)
        obj_opt = 2.0 * n_primary * val_opt / (a_opt.sum() ** 2)
        print("  Phase 1b result: %.6f" % obj_opt)
        try_and_update(a_opt, n_primary, "SA")

    # Phase 1c: Second FISTA pass with different params (25% of primary time)
    t3 = primary_budget * 0.25
    if t3 > 20 and time.monotonic() < deadline - 40:
        a_start2 = global_a if global_a is not None else a_start
        # Perturb the starting point slightly
        a_start2 = spectral_perturb(a_start2, 0.05, rng)
        print("  Phase 1c: FISTA pass 2 (%.0fs)" % t3)
        a_opt, val_opt = run_adaptive_fista(a_start2, S, t3, rng,
                                            k_tight=5, k_broad_frac=0.03,
                                            beta_base=50.0, beta_growth=0.005,
                                            beta_max=30000.0,
                                            lr_base=0.5, mode_period=150)
        obj_opt = 2.0 * n_primary * val_opt / (a_opt.sum() ** 2)
        print("  Phase 1c result: %.6f" % obj_opt)
        try_and_update(a_opt, n_primary, "FISTA2")

    # --- Phase 2: Try n=16384 if time permits ---
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        n_secondary = 16384
        S_sec = np.sqrt(2.0 * n_secondary)
        a_sec = resample_sequence(global_a, n_secondary)
        a_sec = normalize(a_sec, S_sec)
        print("\n=== Secondary: n=%d (%.0fs) ===" % (n_secondary, remaining * 0.3))

        t_sec = remaining * 0.30
        a_opt, val_opt = run_adaptive_fista(a_sec, S_sec, t_sec, rng,
                                            k_tight=10, k_broad_frac=0.05,
                                            beta_base=30.0, beta_growth=0.003,
                                            beta_max=20000.0,
                                            lr_base=0.7, mode_period=250)
        obj_opt = 2.0 * n_secondary * val_opt / (a_opt.sum() ** 2)
        print("  Secondary result: %.6f" % obj_opt)
        try_and_update(a_opt, n_secondary, "Sec")

    # --- Phase 3: Intensive final polish on global best ---
    remaining = deadline - time.monotonic() - 10
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (%.0fs, n=%d) ===" % (remaining, n_best))

        # Polishing pass 1
        t_pol1 = remaining * 0.5
        a_pol, val_pol = run_intensive_polish(global_a, S_best, t_pol1, rng)
        obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
        print("  Polish 1: %.6f" % obj_pol)
        try_and_update(a_pol, n_best, "Pol1")

        # Polishing pass 2 with slight perturbation
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 20:
            a_pert = spectral_perturb(global_a, 0.02, rng)
            t_pol2 = remaining2 * 0.9
            a_pol2, val_pol2 = run_intensive_polish(a_pert, S_best, t_pol2, rng)
            obj_pol2 = 2.0 * n_best * val_pol2 / (a_pol2.sum() ** 2)
            print("  Polish 2: %.6f" % obj_pol2)
            try_and_update(a_pol2, n_best, "Pol2")

    # --- Final shake-and-keep with the very best ---
    remaining = deadline - time.monotonic() - 5
    if remaining > 5 and global_a is not None:
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)
        print("\n=== Final Shake (%.0fs) ===" % remaining)
        # Multiple rounds of shake-keep
        a_cur = global_a.copy()
        n_rounds = min(50, int(remaining / 0.5))
        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            a_try, val_try = shake_keep(a_cur, S_fin, rng,
                                        n_shakes=5, shake_scale=0.003)
            if val_try < fft_convolve_self(a_cur).max():
                a_cur = a_try
            obj_try = obj_value(a_cur)
            try_and_update(a_cur, n_fin, "shake%d" % r)

    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
