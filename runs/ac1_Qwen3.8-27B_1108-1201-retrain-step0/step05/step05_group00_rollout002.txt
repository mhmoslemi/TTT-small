Uses L-BFGS on LSE-smoothed convolution peak with:

- Warm start from previous best solution
- Aggressive beta schedule
- Coordinate descent polishing on peak-contributing indices
- Structured perturbation restarts targeting peak contributors

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Full self-convolution via FFT."""
    n = len(a)
    Nf = max(N, 1 << (2 * n - 1).bit_length())
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_and_grad(a, N, beta):
    """LSE-smoothed peak of conv and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total
    lse_val = c_max + np.log(max(total, 1e-300)) / beta
    # Gradient: d/d(a[i]) sum_k weights[k] * conv[k]
    # conv[k] = sum_i a[i]*a[k-i], so d/d(a[i]) conv[k] = a[k-i] + a[i-k] (symmetric)
    # Actually, grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) where valid
    # More efficiently: ifft(fft(weights) * conj(fft(a)))
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g, m_max=10):
    """L-BFGS two-loop recursion."""
    m = min(len(s_hist), m_max)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    s_hist = s_hist[-m:]
    y_hist = y_hist[-m:]
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(y_i, y_i)  # Wait, this should be np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (partially) while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    if frac >= 1.0:
        new_phase = rng.uniform(0, 2 * np.pi, N)
    else:
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        new_phase = old_phase.copy()
        new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def peak_perturbation_restart(a, c, S, rng, strength=0.3):
    """Perturb elements that contribute most to the peak lag."""
    n = len(a)
    k_star = int(np.argmax(c))
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    # Compute contribution of each index to the peak
    contrib = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] += a[j]
    max_contrib = contrib.max()
    if max_contrib < 1e-15:
        return a.copy()
    # Reduce high-contributing elements, increase low-contributing ones
    a_new = a.copy()
    total_reduce = 0
    for i in range(n):
        if contrib[i] > 0.8 * max_contrib:
            delta = strength * 0.1 * a[i]
            a_new[i] -= delta
            total_reduce += delta
        elif contrib[i] < 0.2 * max_contrib:
            pass  # will receive
    a_new = np.maximum(a_new, 0)
    # Redistribute
    low_mask = contrib < 0.2 * max_contrib
    if a_new[low_mask].sum() > 0:
        a_new[low_mask] += total_reduce * a_new[low_mask] / a_new[low_mask].sum()
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def coord_descent_polish(a, S, N, max_time, rng):
    """Coordinate descent focusing on peak-reducing updates."""
    n = len(a)
    a_curr = a.copy()
    c = compute_conv(a_curr, N)
    c_max = c.max()
    best_a = a_curr.copy()
    best_cmax = c_max
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        
        # Find the index with the highest "impact" on reducing the peak
        # Impact of reducing a[i] by delta: reduces C(k*) by delta * a[k*-i]
        # But also affects other lags
        # Simple heuristic: pick i with highest a[i] * a[k*-i] in the peak range
        if lo <= hi:
            impacts = np.zeros(hi - lo + 1)
            for idx, i in enumerate(range(lo, hi + 1)):
                j = k_star - i
                if 0 <= j < n:
                    impacts[idx] = a_curr[i] * a_curr[j]
            if impacts.max() > 0:
                i_perturb = lo + int(np.argmax(impacts))
                j_perturb = k_star - i_perturb
                # Try reducing a[i_perturb] and increasing a[j_perturb]
                # (or some other index)
                delta = 0.01 * a_curr[i_perturb]
                a_test = a_curr.copy()
                a_test[i_perturb] -= delta
                # Add to a random low-contributing index
                candidates = np.where(contrib < 0.5 * contrib.max())[0] if hasattr(contrib, 'max') else np.arange(n)
                # Simpler: add to the index that has the lowest contribution to peak
                if 0 <= j_perturb < n and j_perturb != i_perturb:
                    a_test[j_perturb] += delta
                else:
                    a_test[i_perturb // 2] += delta
                
                a_test = np.maximum(a_test, 0)
                s = a_test.sum()
                if s > 0:
                    a_test = a_test * (S / s)
                
                c_test = compute_conv(a_test, N)
                if c_test.max() < c_max - 1e-12:
                    a_curr = a_test
                    c = c_test
                    c_max = c.max()
                    if c_max < best_cmax:
                        best_cmax = c_max
                        best_a = a_curr.copy()
                else:
                    # Try the opposite: increase a[i_perturb], decrease a[j_perturb]
                    if 0 <= j_perturb < n and j_perturb != i_perturb:
                        a_test2 = a_curr.copy()
                        a_test2[j_perturb] = max(0, a_test2[j_perturb] - delta)
                        a_test2[i_perturb] += delta
                        a_test2 = np.maximum(a_test2, 0)
                        s2 = a_test2.sum()
                        if s2 > 0:
                            a_test2 = a_test2 * (S / s2)
                        c_test2 = compute_conv(a_test2, N)
                        if c_test2.max() < c_max - 1e-12:
                            a_curr = a_test2
                            c = c_test2
                            c_max = c.max()
                            if c_max < best_cmax:
                                best_cmax = c_max
                                best_a = a_curr.copy()
    
    return best_a, best_cmax


def lse_lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0, m_lbfgs=12):
    """L-BFGS on LSE-smoothed objective."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s0)
    
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    g, f_val, c, c_max = lse_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = S / (n * 0.5)
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Increase beta periodically
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.3)
            g, f_val, c, c_max = lse_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        # Trim history
        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max + 1e-12:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.5
        else:
            a_new = candidate
            cmax_new = cm_ls
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        g_new, _, c_nw, cmax_nw = lse_and_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        # Restart if stalled
        if stall_count > 50 and it % 100 == 0:
            if rng.random() < 0.5:
                a = spectral_restart(a, N, rng, frac=0.3)
            else:
                c_tmp = compute_conv(a, N)
                a = peak_perturbation_restart(a, c_tmp, S, rng, strength=0.5)
            s_hist = []
            y_hist = []
            stall_count = 0
            g, f_val, c, c_max = lse_and_grad(a, N, beta)
    
    return best_a, best_cmax


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_random(n, rng):
    return rng.random(n) + 0.01


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * (2 * tt**2)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_linear_tilt(n, rng):
    """Linearly tilted sequence - high on one side, low on the other."""
    tt = np.linspace(0, 1, n)
    a = (1.0 + tt) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_v_shape(n, rng):
    """V-shaped: high at edges, low in middle."""
    tt = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * np.abs(1.0 - 2.0 * tt) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_exp_decay(n, rng):
    """Exponential decay from one end."""
    tt = np.arange(n) / n
    a = np.exp(-2.0 * tt) + 0.3 * np.exp(-2.0 * (1.0 - tt)) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def deflation_init(n, S, N, n_iters=3000, rng=None):

        k_star = int(np.argmax(c))

        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.01 * a[i_max], 0.2 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
