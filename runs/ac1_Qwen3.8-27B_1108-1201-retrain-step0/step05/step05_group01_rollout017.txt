def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

    # We need the "valid" part

def conv_gradient(y, w):
    """
    Approximate gradient of sum_k w[k]*c[k] w.r.t. y, where c=conv(y,y).
    Uses FFT for efficiency. Accuracy is O(1/n) relative, which is fine for n>100.
    """

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    Y = fft(y, fsize)
    W = fft(w, fsize)
    # Cross-correlation: (w ⋆ y)[j] = sum_i w[i+j]*y[i]
    # = real(ifft(conj(fft(w)) * fft(y))[j])... 
    # Actually: ifft(conj(W)*Y)[k] = sum_i w[i]*y[(i+k) mod N] (circular correlation)
    # We want sum_i y[i]*w[i+j] = sum_i w[i+j]*y[i]
    # Let's use: ifft(W*conj(Y))[k] = sum_i w[(i-k) mod N]*y[i]
    # Hmm, let me just use the standard formula:
    # Linear cross-correlation: (f ⋆ g)[k] = sum_i f[i] * g[i+k]
    # = real(ifft(conj(FFT(f, N)) * FFT(g, N)))[k] for k >= 0 (with N large enough)
