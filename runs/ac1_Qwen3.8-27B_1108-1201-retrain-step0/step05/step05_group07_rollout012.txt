import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, c, k, threshold_frac=0.5):
    """
    Weighted subgradient from top-k convolution lags.
    Uses exponential weighting based on how far above the median each lag is.
    """
    n = len(a)
    k = min(k, 2 * n - 1)
    c_max = c.max()
    c_mean = c.mean()
    
    # Use lags above a threshold (between mean and max)
    threshold = c_mean + threshold_frac * (c_max - c_mean)
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    elif len(active) > k:

    weights = np.maximum(c[active] - threshold, 0.0)
    wsum = weights.sum()

        weights = weights / wsum
    
    grad = np.zeros(n)

def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 350:
        beta = 350.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses adaptive subgradient with Adam momentum, multiplicative non-negativity handling,
Fourier-domain structured perturbations, adaptive active-set sizing, and aggressive
LSE polishing. Explores multiple lengths (4096, 8192, 16384) for best result.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, c, k, threshold_frac):
    """Weighted subgradient from top-k convolution lags with exponential weighting."""
    n = len(a)
    k = min(k, 2 * n - 1)
    c_max = c.max()
    c_mean = c.mean()
    threshold = c_mean + threshold_frac * (c_max - c_mean)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    elif len(active) > k:
        order = np.argsort(c[active])[::-1][:k]
        active = active[order]
    weights = np.maximum(c[active] - threshold, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / wsum
    grad = np.zeros(n)
    for ii in range(len(active)):
        kidx = int(active[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 350:
        beta = 350.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fourier_perturb(a, rng, scale):
    """Add smooth structured noise in Fourier domain."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    noise = rng.normal(0, scale, fsize) + 1j * rng.normal(0, scale, fsize)
    noise[0] = 0
    A_new = A + noise
    a_new = np.real(ifft(A_new, fsize))[:n]
    return np.maximum(a_new, 0.0)

    block_noise = rng.normal(0, scale, n_blocks)
    noise = np.repeat(block_noise, block_size)[:n]

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 150

    while time.monotonic() - start < max_time:
        it += 1
        t_count += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        c_mean = c.mean()
        c_std = c.std()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        gap = cur_max - c_mean
        if gap < 1e-14:
            break

        # Adaptive k: wider when peak is fat
        frac_above = np.sum(c >= 0.85 * cur_max) / len(c)
        k = int(max(3, min(300, 10 + 150 * frac_above + 50 * (1 - progress))))

        threshold_frac = 0.3 + 0.3 * (1 - progress)
        grad = compute_subgradient_topk(a, c, k, threshold_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adam update
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * (grad * grad)
        m1_hat = m1 / (1 - b1 ** t_count)
        m2_hat = m2 / (1 - b2 ** t_count)

        base_lr = gap * 0.5
        denom = np.sqrt(m2_hat) + 1e-8
        lr = base_lr / denom
        lr_max = 2.0 / (gnorm + 1e-10)
        lr = np.minimum(lr, lr_max)

        # Try update with soft non-negativity
        a_try = a - lr * m1_hat
        a_try = np.maximum(a_try, 0.0)
        S_try = a_try.sum()
        if S_try > 1e-30:
            a_try = a_try * (S / S_try)
        else:
            a_try = a.copy()

        c_try = fft_convolve_self(a_try)

        # Acceptance criterion: allow slight increase early, stricter later
        tol = 0.02 * (1 - progress) + 0.001
        if c_try.max() <= cur_max * (1.0 + tol):
            a = a_try
        else:
            # Halve and retry
            a_try2 = np.maximum(a - 0.5 * lr * m1_hat, 0.0)
            S_try2 = a_try2.sum()
            if S_try2 > 1e-30:
                a_try2 = a_try2 * (S / S_try2)
            c_try2 = fft_convolve_self(a_try2)
            if c_try2.max() <= cur_max * (1.0 + tol * 0.5):
                a = a_try2
            else:
                a_try3 = np.maximum(a - 0.1 * lr * m1_hat, 0.0)
                S_try3 = a_try3.sum()
                if S_try3 > 1e-30:
                    a_try3 = a_try3 * (S / S_try3)
                a = a_try3

        # Perturbation when stuck
        if stuck > patience:
            if stuck > patience * 3:
                # Strong Fourier perturbation
                scale = 0.02 * S / np.sqrt(n)
                a_p = fourier_perturb(a, rng, scale)
                Sp = a_p.sum()
                if Sp > 1e-30:
                    a = a_p * (S / Sp)
                stuck = 0
                patience = int(patience * 1.4)
            elif stuck > patience * 1.5:
                # Block perturbation
                scale = 0.01 * S / np.sqrt(n)
                a_p = block_perturb(a, rng, scale)
                Sp = a_p.sum()
                if Sp > 1e-30:
                    a = a_p * (S / Sp)
                stuck = 0
                patience = int(patience * 1.3)
            else:
                # Mild Gaussian
                scale = min(0.015, 0.002 * (stuck / patience)) * S / np.sqrt(n)
                a_p = np.maximum(a + rng.normal(0, scale, n), 0.0)
                Sp = a_p.sum()
                if Sp > 1e-30:
                    a = a_p * (S / Sp)
                stuck = 0
                patience = int(patience * 1.15)

        if verbose and it % verbose_interval == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={obj:.6f}, cmax={best:.6f}, k={k}, stuck={stuck}")

    return best_a, best


def optimize_lse(a0, S, max_time, rng, beta_init=100.0, beta_max=50000.0,
                 beta_growth=0.004, verbose=False):
    """LSE-smoothed gradient descent with momentum for polishing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    Sa = a.sum()
    if Sa > 1e-30:
        a = a * (S / Sa)

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    m_lse = np.zeros(n)
    b1 = 0.9
    t_count = 0

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        t_count += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 200:
            scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - progress)
            if scale > 1e-12:
                a_p = fourier_perturb(a, rng, scale)
                Sp = a_p.sum()
                if Sp > 1e-30:
                    a = a_p * (S / Sp)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        m_lse = b1 * m_lse + (1 - b1) * grad
        m_hat = m_lse / (1 - b1 ** t_count)

        step = 1.0 / (1.5 * beta)
        step = min(step, 1.5 / (gnorm + 1e-10))

        a_try = np.maximum(a - step * m_hat, 0.0)
        S_try = a_try.sum()
        if S_try > 1e-30:
            a_try = a_try * (S / S_try)
        else:
            a_try = a.copy()

        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-9):
            a = a_try
        else:
            a_try = np.maximum(a - 0.3 * step * m_hat, 0.0)
            S_try = a_try.sum()
            if S_try > 1e-30:
                a = a_try * (S / S_try)
            else:
                a = a.copy()

    return best_a, best


def make_init(n, rng, base=None):
    """Generate initial sequence, optionally interpolating from base."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    return a


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2350):
    """
    Multi-phase search: adaptive subgradient with Adam, LSE polishing,
    length exploration (4096/8192/16384), structured perturbations,
    and iterative refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    print(f"=== Starting: budget={budget_s}s ===")

    # State dict to avoid closures
    state = {'best': np.inf, 'a': None}

    # Load provided sequence
    a_prov = None
    try:
        a_prov = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_prov = np.clip(a_prov, 0, 1000)
        if a_prov.sum() > 0 and len(a_prov) >= 128:
            print(f"Provided: len={len(a_prov)}, sum={a_prov.sum():.2f}")
        else:
            a_prov = None
    except Exception:
        pass

    # Helper to check and update global best (inlined to avoid closure)
    def _check_update(a, label):
        nonlocal state
        S = a.sum()
        if S < 0.01:
            return
        obj = compute_obj(a)
        if obj < state['best']:
            state['best'] = obj
            state['a'] = a.copy()
            print(f"  ** NEW BEST: obj={obj:.6f} [{label}]")

    # ========== PHASE 1: Primary optimization ==========
    if a_prov is not None:
        n1 = len(a_prov)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_prov * (S1 / a_prov.sum())

        t1 = min(800, (deadline - time.monotonic()) * 0.35)
        print(f"\n[Phase 1] Subgrad n={n1}, t={t1:.0f}s")
        a_opt, _ = optimize_subgrad_adaptive(a1, S1, t1, rng, verbose=True, verbose_interval=300)
        _check_update(a_opt, "P1 subgrad")

        # LSE polish
        rem = deadline - time.monotonic()
        t1b = min(350, rem * 0.18)
        if t1b > 25:
            print(f"[Phase 1b] LSE polish, t={t1b:.0f}s")
            a_pol, _ = optimize_lse(a_opt, S1, t1b, rng, beta_init=200, beta_max=30000, beta_growth=0.005)
            _check_update(a_pol, "P1b LSE")

        # Second subgrad pass
        rem = deadline - time.monotonic()
        t1c = min(200, rem * 0.10)
        if t1c > 20:
            print(f"[Phase 1c] 2nd subgrad, t={t1c:.0f}s")
            a_opt2, _ = optimize_subgrad_adaptive(a_pol, S1, t1c, rng, verbose=False)
            _check_update(a_opt2, "P1c subgrad2")

    # ========== PHASE 2: Alternative lengths ==========
    rem = deadline - time.monotonic()
    if rem > 120 and state['a'] is not None:
        n_base = len(state['a'])
        candidates_n = [16384, 4096]
        # Also try 12288 (not power of 2 but valid)
        tried_lengths = set()
        if a_prov is not None:
            tried_lengths.add(len(a_prov))

        for n_try in candidates_n:
            if time.monotonic() > deadline - 80:
                break
            if n_try in tried_lengths or n_try < 256 or n_try > 32768:
                continue
            tried_lengths.add(n_try)

            S_try = np.sqrt(2.0 * n_try)
            t_alloc = min(250, (deadline - time.monotonic()) * 0.18)
            if t_alloc < 30:
                continue

            print(f"\n[Phase 2] n={n_try}, t={t_alloc:.0f}s")
            init = make_init(n_try, rng, base=state['a'])
            init = np.maximum(init, 0)
            init = init * (S_try / max(init.sum(), 1e-10))

            a_o, _ = optimize_subgrad_adaptive(init, S_try, t_alloc * 0.6, rng, verbose=False)
            _check_update(a_o, f"P2 n={n_try}")

            rem2 = deadline - time.monotonic()
            if rem2 > 25:
                a_p, _ = optimize_lse(a_o, S_try, min(100, rem2 * 0.4), rng,
                                      beta_init=300, beta_max=50000, beta_growth=0.005)
                _check_update(a_p, f"P2 n={n_try} LSE")

    # ========== PHASE 3: Alternating refinement ==========
    rem = deadline - time.monotonic()
    if state['a'] is not None and rem > 50:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        a_curr = state['a'].copy()
        a_curr = a_curr * (S / max(a_curr.sum(), 1e-30))

        print(f"\n[Phase 3] Refine n={n}, rem={rem:.0f}s")

        # Cycle through methods
        methods = ['subgrad', 'lse', 'subgrad', 'lse', 'subgrad', 'lse']
        t_per = max(8, (rem - 25) / len(methods))

        for mi in range(len(methods)):
            if time.monotonic() > deadline - 15:
                break
            t_m = min(t_per, deadline - time.monotonic() - 10)
            if t_m < 5:
                break

            if methods[mi] == 'subgrad':
                a_curr, _ = optimize_subgrad_adaptive(a_curr, S, t_m, rng, verbose=False, verbose_interval=200)
            else:
                a_curr, _ = optimize_lse(a_curr, S, t_m, rng,
                                         beta_init=1000 * (mi + 1),
                                         beta_max=100000,
                                         beta_growth=0.006)
            _check_update(a_curr, f"P3 {methods[mi]}#{mi}")

    # ========== PHASE 4: Diverse restarts ==========
    rem = deadline - time.monotonic()
    if state['a'] is not None and rem > 50:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        n_r = 5
        t_per = max(8, min(60, (rem - 15) / n_r))

        print(f"\n[Phase 4] {n_r} restarts, {t_per:.0f}s each")
        for ri in range(n_r):
            if time.monotonic() > deadline - 12:
                break

            # Varied perturbation
            if ri % 3 == 0:
                scale = 0.025 * S / np.sqrt(n)
                a_p = fourier_perturb(state['a'], rng, scale)
            elif ri % 3 == 1:
                scale = 0.015 * S / np.sqrt(n)
                a_p = block_perturb(state['a'], rng, scale)
            else:
                scale = 0.02 * S / np.sqrt(n)
                a_p = np.maximum(state['a'] + rng.normal(0, scale, n), 0.0)

            Sp = a_p.sum()
            if Sp > 1e-30:
                a_p = a_p * (S / Sp)

            a_r, _ = optimize_subgrad_adaptive(a_p, S, t_per * 0.7, rng, verbose=False)
            _check_update(a_r, f"P4 r{ri}")

            rem_r = deadline - time.monotonic()
            if rem_r > 15:
                a_rp, _ = optimize_lse(a_r, S, min(25, rem_r * 0.35), rng,
                                       beta_init=2000, beta_max=80000, beta_growth=0.007)
                _check_update(a_rp, f"P4 r{ri}LSE")

    # ========== PHASE 5: Final intensive LSE ==========
    rem = deadline - time.monotonic()
    if state['a'] is not None and rem > 20:
        n = len(state['a'])
        S = np.sqrt(2.0 * n)
        t5 = rem - 10
        print(f"\n[Phase 5] Final LSE, t={t5:.0f}s")
        a_final, _ = optimize_lse(state['a'], S, t5, rng,
                                  beta_init=5000, beta_max=200000, beta_growth=0.003)
        _check_update(a_final, "P5 final LSE")

    # ========== Return ==========
    if state['a'] is None:
        print("WARNING: No solution found, using fallback")
        state['a'] = np.ones(256)
        state['best'] = compute_obj(state['a'])

    final_a = np.clip(state['a'], 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    result = [float(x) for x in final_a]
    ver = evaluate_sequence(result)
    print(f"\n{'='*60}")
    print(f"FINAL: len={len(final_a)}, internal_obj={state['best']:.6f}")
    print(f"evaluate_sequence: {ver:.6f}")
    print(f"{'='*60}")
    return result
