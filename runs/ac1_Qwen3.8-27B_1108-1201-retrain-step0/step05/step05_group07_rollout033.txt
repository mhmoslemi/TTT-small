"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences of length n.
Strategy: LSE-smoothed subgradient descent with momentum, adaptive parameters,
coordinate refinement, and multi-length exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_conv(self_a):
    """Full linear self-convolution via FFT."""
    n = len(self_a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(self_a, fsize)
    c = np.real(ifft(A * A))
    return c[:2*n-1]


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = (cs - S) / idx
    mask = vals <= w
    if not np.any(mask):
        rho = n
    else:
        rho = int(np.max(np.where(mask)[0])) + 1
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def grad_lse(a, c, k, beta):
    """LSE-weighted subgradient using top-k lags, computed directly."""
    n = len(a)
    idx = np.argpartition(c, -k)[-k:]  # top k indices
    cm = c[idx].max()
    spread = c[idx].max() - c[idx].min()
    if spread > 1e-30:
        s = beta * (c[idx] - cm) / spread
    else:
        s = np.zeros(k)
    s = np.clip(s, -50, 0)
    e = np.exp(s)
    w = e / e.sum()
    
    g = np.zeros(n)
    for ii in range(k):
        lag = int(idx[ii])
        wi = w[ii]
        j1s, j1e = max(0, lag - n + 1), min(n - 1, lag)
        if j1s <= j1e:
            g[j1s:j1e+1] += wi * a[lag - np.arange(j1s, j1e + 1)]
        j2s, j2e = max(0, lag), min(n - 1, lag + n - 1)
        if j2s <= j2e:
            g[j2s:j2e+1] += wi * a[np.arange(j2s, j2e + 1) - lag]
    return g


def run_subgrad(a_init, S, budget, rng, momentum=0.85, verbose=True):
    """Main subgradient loop."""
    n = len(a_init)
    a = proj_simplex(a_init, S)
    c = fft_conv(a)
    best_c = c.max()
    best_a = a.copy()
    vel = np.zeros(n)
    
    t0 = time.monotonic()
    it = 0
    stuck = 0
    patience = 500
    
    while time.monotonic() - t0 < budget:
        it += 1
        c = fft_conv(a)
        cur = c.max()
        
        if cur < best_c - 1e-14:
            best_c = cur
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / budget)
        
        # Adaptive hyperparameters
        k = int(4 + 256 * prog ** 0.5)
        k = min(k, 500)
        beta = 30.0 + 400.0 * prog
        
        g = grad_lse(a, c, k, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        gap = cur - c.mean()
        if gap < 1e-15:
            break
        
        step = 2.0 * gap / (gn * gn)
        step = min(step, 4.0 / gn)
        
        vel = momentum * vel - step * g
        a_new = proj_simplex(a + vel, S)
        
        # Line search (backtracking)
        ok = False
        for _ in range(10):
            c_new = fft_conv(a_new)
            if c_new.max() <= cur + 1e-12:
                a = a_new
                ok = True
                break
            vel *= 0.5
            a_new = proj_simplex(a + vel, S)
        
        if not ok:
            s_small = 0.01 * gap / max(gn, 1e-30)
            a = proj_simplex(a - s_small * g, S)
            vel = np.zeros(n)
        
        # Escape mechanism
        if stuck > patience:
            ns = min(0.08, 0.005 * stuck / patience) * S / np.sqrt(n)
            a = proj_simplex(a + rng.normal(0, ns, n), S)
            vel = np.zeros(n)
            stuck = 0
            patience = max(patience + 200, int(patience * 1.1))
