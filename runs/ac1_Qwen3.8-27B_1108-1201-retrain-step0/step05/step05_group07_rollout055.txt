Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses SCA (Successive Convex Approximation) as the main optimization engine,
exploiting the convexity of the problem. Each SCA iteration solves a linearized
subproblem via projected gradient with simplex constraints. Followed by LSE
smoothing for final polishing.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective value 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S), c.max(), c


def lse_weights(c, beta):
    """Compute LSE probability weights."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    return e / tot

    minimize max_k (2*(a_old*a)_k - c_old_k) s.t. sum(a)=S, a>=0

    w = np.full(2*n-1, 1.0/(2*n-1))  # uniform init

        W = fft(w, fsize)
        A_old = fft(a_old, fsize)

        if gnorm < 1e-14:

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key insight: The problem min max_k (a*a)_k s.t. a>=0, sum(a)=S is CONVEX
(max of convex quadratics over a convex set). We exploit this with:
1. FISTA (Nesterov-accelerated) on LSE-smoothed objective for fast convergence
2. SCA (Successive Convex Approximation) iterations for accurate refinement
3. Multi-length exploration for additional gains
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute LSE probability weights over convolution lags."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w
    return e / tot


def lse_objective(c, beta):
    """Compute LSE value: (1/beta)*log(sum(exp(beta*c_k)))."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    return cm + np.log(np.exp(scaled).sum()) / beta


def compute_gradient_lse(a, c, beta):
    """Gradient of LSE-smoothed max convolution objective."""
    n = len(a)
    p = lse_weights(c, beta)
    # grad = 2/beta * (a * p) (linear convolution, first n elements)
    fsize = 1 << (n + len(p) - 1).bit_length()
    A = fft(a, fsize)
    P = fft(p, fsize)
    conv = np.real(ifft(A * P))[:n]
    return 2.0 * conv / beta


def fista_lse(a0, S, max_time, rng, beta_start=5.0, beta_max=10000.0,
              beta_growth=1.005, verbose=True):
    """
    FISTA with LSE smoothing for the convex problem:
    minimize max_k (a*a)_k s.t. a >= 0, sum(a) = S.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    y = a.copy()
    t_mom = 1.0

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    beta = beta_start
    step = 1.0 / (2.0 * S * S)
    stuck = 0
    last_report = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Compute gradient at momentum point y
        c_y = fft_convolve_self(y)
        grad = compute_gradient_lse(y, c_y, beta)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Backtracking line search (limited)
        step_ls = step
        accepted = False
        for _bt in range(8):
            a_new = project_simplex(y - step_ls * grad, S)
            # Simple check: ensure no large increase in raw max
            if np.all(a_new >= -1e-10):
                accepted = True
                break
            step_ls *= 0.5

        if not accepted:
            a_new = project_simplex(y - (step_ls * 0.5) * grad, S)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + (t_mom - 1.0) / t_new * (a_new - a)
        a = a_new
        t_mom = t_new

        # Track best using actual max (not LSE)
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Adapt beta: grow gradually
        if it % 200 == 0:
            beta = min(beta_max, beta * beta_growth)
            step = 1.0 / (2.0 * beta * S * S + 1e-10)
            # Reset momentum on beta change
            t_mom = 1.0
            y = a.copy()

        # Noise injection when stuck for a long time
        if stuck > 500 and stuck % 250 == 0:
            noise_scale = max(0.001, 0.01 * (1.0 - stuck / 5000.0)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            y = a.copy()
            t_mom = 1.0
            stuck = 0

        # Periodic reporting
        if verbose and it - last_report > 5000:
            elapsed = time.monotonic() - start
            obj_val = 2.0 * n * best / (S * S)
            print(f"    FISTA it={it}, t={elapsed:.0f}s, beta={beta:.1f}, obj={obj_val:.6f}")
            last_report = it

    return best_a, best


def sca_inner_subgrad(a_old, c_old, S, n, n_inner=100):
    """
    Solve the SCA inner problem via projected subgradient:
    minimize max_k [2*(a_old*a)_k - c_old_k] s.t. sum(a)=S, a>=0
    """
    a = a_old.copy()
    fsize = 1 << (n + 2 * n - 1 - 1).bit_length()
    A_old = fft(a_old, fsize)

    for inner_it in range(n_inner):
        # Compute linearized objective at each lag
        # 2*(a_old * a)_k - c_old_k
        conv_ao = np.real(ifft(A_old * fft(a, fsize)))[:2 * n - 1]
        obj_lags = 2.0 * conv_ao - c_old

        # Find active lags (top portion)
        top_k = min(100, 2 * n - 1)
        top_idx = np.argsort(obj_lags)[::-1][:top_k]
        obj_top = obj_lags[top_idx]

        # Softmax weights over active lags
        beta_w = max(10.0, (obj_top[0] - obj_top[-1]) * 10.0)
        if beta_w < 1:
            beta_w = 1.0
        scaled = beta_w * (obj_top - obj_top.max())
        scaled = np.clip(scaled, -300, 300)
        e = np.exp(scaled)
        w_top = e / e.sum()

        # Build full weight vector
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top

        # Subgradient: 2 * (w * a_old)
        W = fft(w, fsize)
        grad = 2.0 * np.real(ifft(W * A_old))[:n]
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        # Step size: Polyak-type
        objective_val = obj_lags.max()
        step = min(2.0 * max(objective_val, 0.001) / (gnorm * gnorm),
                   0.5 * S / gnorm)
        step = max(step, 1e-10)

        # Gradient step + projection
        a_new = project_simplex(a - step * grad, S)
        a = a_new

    return a


def sca_optimize(a0, S, max_time, n_inner=100, verbose=True):
    """
    SCA optimization: iteratively solve linearized convex subproblems.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    outer_it = 0
    stuck = 0
    rng = np.random.default_rng(12345)
    last_report = 0

    while time.monotonic() - start < max_time - 5:
        outer_it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Solve inner problem
        a_new = sca_inner_subgrad(a, c, S, n, n_inner=n_inner)

        # Validate: only accept if it helps the original objective
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()

        if new_max < cur_max:
            a = a_new
        elif new_max < cur_max * 1.005:
            # Accept with damping
            a = 0.7 * a + 0.3 * a_new
            a = project_simplex(a, S)
        else:
            # Reject; if stuck, add perturbation
            if stuck > 30:
                noise_scale = min(0.02, 0.002 * (stuck / 10.0)) * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                stuck = 0

        if verbose and outer_it - last_report > 50:
            elapsed = time.monotonic() - start
            obj_val = 2.0 * n * best_max / (S * S)
            print(f"    SCA outer={outer_it}, t={elapsed:.0f}s, obj={obj_val:.6f}, stuck={stuck}")
            last_report = outer_it

    return best_a, best_max

            stuck = 0
        else:
            stuck += 1

            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        # Top-k subgradient
        k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        median_c = np.median(c)
        weights = np.maximum(c[top_idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:

        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for ii in range(k):
            kidx = int(top_idx[ii])

def interpolate_sequence(base, n_new):
    """Interpolate a sequence to a new length."""
    if len(base) < 2:
        return np.ones(n_new)
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n_new)
    a = np.interp(t_new, t_old, base)
    return np.maximum(a, 0.0)
