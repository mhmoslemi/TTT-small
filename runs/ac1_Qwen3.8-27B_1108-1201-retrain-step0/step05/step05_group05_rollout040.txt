- $\alpha - \beta \le t$
- $\sum w_k = 1$
- $w_k \ge 0$

Variables: $w_1, \ldots, w_m$ (active lags), $\alpha, \beta, t$
Total: $m + 3$ variables

Constraints: $2 \cdot |\text{support\_subset}| + 2$

If $m = 30$ and $|\text{support\_subset}| = 100$, that's 33 variables and 202 constraints. This is a tiny LP that CVXPY can solve in milliseconds.

But we'd need to rebuild the LP at each iteration (since $a$ changes), which means computing the matrix $A[j,k] = a[k-j]$ each time. For $m=30$ and $|\text{supp}|=100$, this is just 3000 multiplications, which is trivial.

The question is: does this actually decrease the objective? For a convex function, moving in the negative subgradient direction should decrease the function (at least for small $\eta$). But the projection onto the simplex might undo some of the progress.

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization with:
1. Diverse initialization + quick FISTA screening
2. Basin hopping with Fourier-domain mutations
3. Precision FISTA with adaptive restart, backtracking, and adaptive beta
4. Dual-fit subgradient polishing using small LP for optimal active-set weighting
5. Final ultra-high-beta polish with KKT verification
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Compute the LSE value (for objective tracking)."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        return cm + np.log(c.size) / beta
    return cm + np.log(np.exp(scaled).sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max() + 1e-30)
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def compute_subgradient(a, c, w_full):
    """Compute subgradient s = w_full * a (linear conv, truncated to len(a))."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * A))[:n]


def dual_fit_subgradient(a, c, tol_frac=1e-5, max_active=40, supp_size=150, use_lp=True):
    """
    Compute the optimal subgradient by finding dual weights on active lags
    that minimize the range of (w*a) on the support of a.
    """
    n = len(a)
    c_max = c.max()
    active = np.where(c >= c_max - tol_frac * c_max)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > max_active:
        top_idx = np.argsort(c[active])[::-1][:max_active]
        active = active[top_idx]

    support = np.where(a > 1e-6 * a.max())[0]
    if len(support) == 0:
        support = np.arange(n)
    if len(support) > supp_size:
        idx = np.linspace(0, len(support) - 1, supp_size).astype(int)
        support = support[idx]

    m = len(active)
    sp = len(support)

    # Build matrix A_mat where A_mat[i, k] = a[active[k] - support[i]] if valid
    A_mat = np.zeros((sp, m))
    for k_idx, k in enumerate(active):
        diffs = k - support
        valid = (diffs >= 0) & (diffs < n)
        A_mat[valid, k_idx] = a[diffs[valid]]

    if use_lp:
        # LP: min t s.t. A_mat @ w <= alpha, A_mat @ w >= beta, alpha - beta <= t
        #     sum(w) = 1, w >= 0
        # Variables: [w (m), alpha (1), beta (1), t (1)]
        n_vars = m + 3
        # c vector (objective): [0...0, 0, 0, 1]
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0

        # Inequalities: A_mat @ w - alpha <= 0 (sp constraints)
        #               -A_mat @ w - beta <= 0 (sp constraints)
        #               alpha - beta - t <= 0 (1 constraint)
        A_ub = np.zeros((2 * sp + 1, n_vars))
        b_ub = np.zeros(2 * sp + 1)
        # A_mat @ w - alpha <= 0
        A_ub[:sp, :m] = A_mat
        A_ub[:sp, m] = -1.0  # -alpha
        # -A_mat @ w - beta <= 0
        A_ub[sp:2*sp, :m] = -A_mat
        A_ub[sp:2*sp, m+1] = -1.0  # -beta
        # alpha - beta - t <= 0
        A_ub[2*sp, m] = 1.0
        A_ub[2*sp, m+1] = -1.0
        A_ub[2*sp, m+2] = -1.0

        # Equality: sum(w) = 1
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :m] = 1.0
        b_eq = np.array([1.0])

        # Bounds: w >= 0, alpha/beta/t unbounded
        bounds = [(0, None)] * m + [(None, None)] * 3

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                w_opt = res.x[:m]
                w_opt = np.maximum(w_opt, 0)
                w_opt = w_opt / max(w_opt.sum(), 1e-30)
            else:
                w_opt = np.ones(m) / m
        except Exception:
            w_opt = np.ones(m) / m
    else:
        # Fallback: iterative balancing
        w_opt = np.ones(m) / m
        for _ in range(30):
            s_vals = A_mat @ w_opt
            j_max = np.argmax(s_vals)
            j_min = np.argmin(s_vals)
            if s_vals[j_max] - s_vals[j_min] < 1e-12:
                break
            grad_range = A_mat[j_max, :] - A_mat[j_min, :]
            gnorm = np.linalg.norm(grad_range)
            if gnorm < 1e-15:
                break
            eta = 0.5 / gnorm
            w_new = w_opt - eta * grad_range
            w_new = np.maximum(w_new, 0)
            ws = w_new.sum()
            if ws < 1e-15:
                break
            w_new /= ws
            w_opt = w_new

    w_full = np.zeros(len(c))
    w_full[active] = w_opt
    return compute_subgradient(a, c, w_full)


def run_fista_adaptive(a_init, S, max_time, rng, beta_init=10.0, beta_max=50000.0,
                       beta_growth=0.008, use_backtrack=True, use_adaptive_restart=True,
                       use_dual_fit=False, dual_fit_period=500):
    """
    FISTA with adaptive restart, optional backtracking, and adaptive beta.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    last_obj = best

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Check for convergence/stalling
        if it - last_improve > 5000:
            # Try a small deterministic perturbation
            # Use the direction that most reduces the max peak
            peak_idx = np.argmax(c)
            # Move weight from peak-contributing positions to non-peak positions
            s_pk = compute_subgradient(y, c, np.eye(1, len(c), peak_idx)[0])
            # Projected negative step
            step_try = 0.5 * S / n
            y_try = project_simplex(y - step_try * s_pk, S)
            c_try = fft_convolve_self(y_try)
            if c_try.max() < cur_max:
                y = y_try
            else:
                # Random small perturbation
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
            t_mom = 1.0
            last_improve = it
            continue

        # Compute gradient
        if use_dual_fit and it % dual_fit_period == 0:
            grad = dual_fit_subgradient(y, c, tol_frac=1e-5, max_active=40, supp_size=150)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if use_backtrack:
            step = 1.0 / (2.0 * beta)
            # Try to take a larger step
            alpha = 1e-4
            for _ in range(5):
                x_try = project_simplex(y - step * grad, S)
                c_try = fft_convolve_self(x_try)
                # Backtracking condition (descent in LSE)
                f_y = lse_value(c, beta)
                f_try = lse_value(c_try, beta)
                if f_try <= f_y - alpha * step * np.dot(grad, y - x_try):
                    break
                step *= 0.5
            x_new = x_try
        else:
            step = 1.0 / (2.5 * beta)
            x_new = project_simplex(y - step * grad, S)

        # Adaptive restart
        if use_adaptive_restart:
            y_prev = y.copy()
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y_new = np.maximum(y_new, 0.0)
            # Check restart condition: if iterate oscillates
            if np.linalg.norm(y_new - y_prev) > np.linalg.norm(x_new - y_prev) * 1.1:
                y = x_new
                t_mom = 1.0
            else:
                y = y_new
                t_mom = t_new
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            t_mom = t_new

        a = x_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.03 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with Fourier mutations."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    population = [(best, a.copy())]

    it = 0
    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = it / max(1, max(1, int(max_time / opt_time)))

        base_scale = 0.5 * np.exp(-1.5 * progress) + 0.03
        scale = base_scale * (1.0 + 0.5 * rng.random())

        r = rng.random()
        if len(population) >= 2 and r < 0.25:
            idx1 = rng.integers(len(population))
            idx2 = rng.integers(len(population))
            if idx1 != idx2:
                a = fourier_crossover(population[idx1][1], population[idx2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, scale)
        else:
            a = fourier_phase_mutate(a, rng, scale)

        t_alloc = min(opt_time, max_time - (time.monotonic() - start))
        if t_alloc < 5:
            break

        a_opt, val = run_fista_adaptive(a, S, t_alloc, rng,
                                         beta_init=20 + 30 * min(it, 20),
                                         beta_rate=None,
                                         beta_growth=0.005,
                                         beta_max=30000.0,
                                         use_backtrack=True,
                                         use_adaptive_restart=True)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))

        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        gs = np.clip(given_seq, 0, 1000)
        if gs.sum() > 0.01:
            inits.append(gs.copy())

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25) + 0.01)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def objective_value(a, n):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def run_precision_phase(a_init, S, max_time, rng):
    """
    Long precision phase: progressively increase beta, use dual-fit
    subgradients periodically, and handle stalling with targeted perturbations.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    # Run multiple rounds with increasing beta
    # Round 1: beta 10-500 (explore)
    # Round 2: beta 500-5000 (refine)
    # Round 3: beta 5000-50000 (polish)

    rounds = [
        (10.0, 500.0, 0.01, 2.0 * max_time / 5),
        (500.0, 5000.0, 0.005, 2.0 * max_time / 5),
        (5000.0, 50000.0, 0.002, max_time / 5),
    ]

    for r_idx, (b_start, b_end, growth, t_alloc) in enumerate(rounds):
        if time.monotonic() - start > max_time - 30:
            break
        t_actual = min(t_alloc, max_time - (time.monotonic() - start) - 30)
        if t_actual < 10:
            break

        print("  Precision round %d: beta %.0f->%.0f, t=%.0fs" % (r_idx+1, b_start, b_end, t_actual))

        # Sub-phases within each round
        n_sub = 2
        sub_time = t_actual / n_sub
        for sub in range(n_sub):
            if time.monotonic() - start > max_time - 30:
                break
            t_sub = min(sub_time, max_time - (time.monotonic() - start) - 30)
            if t_sub < 5:
                break

            a_sub, val_sub = run_fista_adaptive(
                best_a, S, t_sub, rng,
                beta_init=b_start * (1 + rng.random()*0.5),
                beta_max=b_end,
                beta_growth=growth,
                use_backtrack=True,
                use_adaptive_restart=True,
                use_dual_fit=(r_idx >= 1),
                dual_fit_period=300
            )
            if val_sub < best:
                best = val_sub
                best_a = a_sub.copy()
                print("    Round %d sub %d: obj=%.8f" % (r_idx+1, sub+1, 2.0*n*best/(S*S)))

    return best_a, best


def run_final_polish(a_init, S, max_time, rng):
    """Final ultra-precise polish with dual-fit subgradients."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best - 1e-15:
            best = c.max()
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 2000:
            # Small targeted perturbation: find the lag with max c_k and
            # try to reduce it specifically
            peak = np.argmax(c)
            # Compute which a_j contribute most to c_peak
            contrib = np.zeros(n)
            for j in range(n):
                k2 = peak - j
                if 0 <= k2 < n:
                    contrib[j] = a[k2]
            # Reduce high contributors, increase low ones
            mean_c = contrib.mean()
            direction = contrib - mean_c
            step = 0.01 * S / max(np.linalg.norm(direction), 1e-10)
            a_try = project_simplex(a - step * direction, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < c.max():
                a = a_try
            else:
                a = project_simplex(a + rng.normal(0, 0.0005 * S/np.sqrt(n), n), S)
            last_improve = it
            continue

        # Dual-fit subgradient
        grad = dual_fit_subgradient(a, c, tol_frac=1e-6, max_active=30, supp_size=100)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small step with backtracking
        step = 0.5 * S / (n * gnorm + 1e-10)
        for _ in range(10):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= c.max() + 1e-12:
                break
            step *= 0.5

        a = a_new

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """

    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)

            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    def check_and_update(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2) if a.sum() > 0.01 else np.inf
        if obj < global_best:
            global_best = obj
            global_a = np.clip(a, 0, 1000).copy()
            print("  ** New global best: obj=%.8f" % global_best)

    # Phase 1: Quick screening
    print("\n=== Phase 1: Quick screening ===")
    p1_budget = min(120.0, (deadline - total_start) * 0.08)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(6, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        t_alloc = min(per_init_time, deadline - time.monotonic() - 60)
        if t_alloc < 8:
            break
        a_opt, val = run_fista_adaptive(inits[i], S, t_alloc, rng,
                                         beta_init=5.0, beta_max=2000.0,
                                         beta_growth=0.01, use_backtrack=True)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        check_and_update(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping
    print("\n=== Phase 2: Basin hopping ===")
    p2_budget = min(400.0, (deadline - total_start) * 0.22)
    p2_start = time.monotonic()

    bh_starts = min(2, len(candidates))
    for i in range(bh_starts):
        t_alloc = min(p2_budget / (bh_starts - i), deadline - time.monotonic() - 300)
        if t_alloc < 60:
            break
        print("  BH from candidate %d (obj=%.6f), t=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = run_basin_hopping(candidates[i][1], S, t_alloc, rng)
        check_and_update(a_bh, val_bh)

    # Phase 3: Precision phase (main work)
    print("\n=== Phase 3: Precision optimization ===")
    p3_budget = deadline - time.monotonic() - 200
    if p3_budget > 100 and global_a is not None:
        a_prec, val_prec = run_precision_phase(global_a, S, p3_budget, rng)
        check_and_update(a_prec, val_prec)

    # Phase 4: Final polish
    print("\n=== Phase 4: Final polish ===")
    p4_budget = max(60.0, deadline - time.monotonic() - 10)
    if global_a is not None and p4_budget > 30:
        a_fin, val_fin = run_final_polish(global_a, S, p4_budget, rng)
        check_and_update(a_fin, val_fin)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
```
