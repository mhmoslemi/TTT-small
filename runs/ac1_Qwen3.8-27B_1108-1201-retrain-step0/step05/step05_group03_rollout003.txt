from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_gradient(c, beta):
    """Compute full LSE value and weight vector over all terms."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Prevent overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w


def lse_gradient(a, c, beta):
    """Gradient of full LSE smooth max of self-convolution."""
    n = len(a)
    _, w = lse_value_and_gradient(c, beta)
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_equalize(a, alpha):
    """One step of spectral equalization to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    ratio = power / max(mean_p, 1e-20)
    # Damp high-power, boost low-power frequencies
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio + 1e-10))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / (power + 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def peak_target_correction(a, c, S, n_peaks=5, strength=0.1):
    """Identify worst peaks and apply corrective update."""
    n = len(a)
    # Find the top peaks in the convolution
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    
    # For each peak at index k, the gradient contribution is 2*a[k-j] for valid j
    # We want to reduce the peaks, so we decrease a at positions that contribute to peaks
    correction = np.zeros(n)
    for k in peak_indices:
        # (a*a)_k = sum_j a[j] * a[k-j]
        # d(a*a)_k / da[j] = a[k-j] + a_j (if both valid)
        # More precisely: if k-j is in [0,n-1], d/da[j] includes a[k-j]
        # And if j is in [0,n-1] and k-j is valid, also a[j] from the other side
        for j in range(n):
            kp = k - j
            if 0 <= kp < n:
                correction[j] += 2.0 * a[kp]  # approximate
    
    # Normalize correction
    corr_norm = np.linalg.norm(correction)
    if corr_norm > 1e-15:
        correction = correction / corr_norm
    
    a_new = a - strength * S * correction
    return project_simplex(a_new, S)


def run_fista_lse(a_init, S, max_time, rng,
                  use_full_lse=True, beta_init=30.0, beta_rate=0.005,
                  beta_max=10000.0, k_frac=0.05, use_momentum=True,
                  equalize_interval=500, perturb_interval=2000,
                  peak_target_interval=1000):
    """FISTA with LSE smoothing, spectral equalization, and peak targeting."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    if not use_full_lse:
        k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        if use_full_lse:
            grad = lse_gradient(y, c, beta)
        else:
            grad = topk_lse_gradient(y, c, k, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.05, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic spectral equalization
        if it % equalize_interval == 0 and rng.random() < 0.3:
            y = spectral_equalize(y, 0.08)
            t_val = 1.0
            stuck = 0

        # Periodic peak targeting
        if it % peak_target_interval == 0 and rng.random() < 0.25:
            c_cur = fft_convolve_self(y)
            y = peak_target_correction(y, c_cur, S, n_peaks=3, strength=0.02)
            t_val = 1.0

        # Periodic spectral perturbation for escape
        if it % perturb_interval == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.05, rng)
            t_val = 1.0
            stuck = 0

        # Restart when stuck
        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_adaptive_adam(a_init, S, max_time, rng,
                      lr_init=0.01, beta_init=30.0, beta_rate=0.008,
                      use_full_lse=True, k_frac=0.05):
    """Adam optimizer with adaptive LSE and periodic enhancements."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    if not use_full_lse:
        k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, beta_init * np.exp(beta_rate * t))
        if use_full_lse:
            grad = lse_gradient(a, c, beta)
        else:
            grad = topk_lse_gradient(a, c, k, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adaptive learning rate
        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.6 * progress)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        # Periodic enhancements
        if t % 2000 == 0:
            if rng.random() < 0.2:
                a = spectral_equalize(a, 0.06)
            elif rng.random() < 0.2:
                c_cur = fft_convolve_self(a)
                a = peak_target_correction(a, c_cur, S, n_peaks=3, strength=0.015)
            elif rng.random() < 0.1:
                noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
                if noise_scale > 1e-10:
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    m = np.zeros(n)
                    v = np.zeros(n)

    return best_a, best


def run_polish_high_beta(a_init, S, max_time, rng):
    """Very high beta polishing for fine-tuning."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 200.0 * (1.0 + it / 300.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.3 / (1.0 + 0.002 * it) * step_scale
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.03)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a_new = project_simplex(a - step * grad, S)
            a = project_simplex(a_new, S)

        # Small perturbation to escape plateaus
        if it % 1000 == 0 and (time.monotonic() - start) / max_time < 0.8:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_optimization_pipeline(a_init, S, time_limit, rng, phase_desc=""):
    """Run the full optimization pipeline with time allocation."""
    n = len(a_init)
    best_local = fft_convolve_self(a_init).max()
    best_a_local = a_init.copy()
    start = time.monotonic()
    
    remaining = time_limit - (time.monotonic() - start)
    
    # Phase 1: FISTA with full LSE (45% of time)
    t_p1 = remaining * 0.40
    if t_p1 > 30:
        a_opt, val_opt = run_fista_lse(a_init, S, t_p1, rng,
                                        use_full_lse=True,
                                        beta_init=30.0, beta_rate=0.004,
                                        beta_max=8000.0,
                                        equalize_interval=400,
                                        perturb_interval=1500,
                                        peak_target_interval=800)
        if val_opt < best_local:
            best_local = val_opt
            best_a_local = a_opt.copy()
            obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
            print("  FISTA full-LSE: obj=%.6f" % obj)
    
    remaining = time_limit - (time.monotonic() - start)
    
    # Phase 2: FISTA with top-k LSE (25% of time)
    t_p2 = remaining * 0.30
    if t_p2 > 30:
        a_opt, val_opt = run_fista_lse(best_a_local, S, t_p2, rng,
                                        use_full_lse=False,
                                        beta_init=50.0, beta_rate=0.006,
                                        beta_max=15000.0, k_frac=0.03,
                                        equalize_interval=600,
                                        perturb_interval=2000,
                                        peak_target_interval=1000)
        if val_opt < best_local:
            best_local = val_opt
            best_a_local = a_opt.copy()
            obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
            print("  FISTA top-k: obj=%.6f" % obj)
    
    remaining = time_limit - (time.monotonic() - start)
    
    # Phase 3: Adam (20% of time)
    t_p3 = remaining * 0.30
    if t_p3 > 20:
        a_opt, val_opt = run_adaptive_adam(best_a_local, S, t_p3, rng,
                                            lr_init=0.008, beta_init=40.0,
                                            beta_rate=0.006,
                                            use_full_lse=True)
        if val_opt < best_local:
            best_local = val_opt
            best_a_local = a_opt.copy()
            obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
            print("  Adam: obj=%.6f" % obj)
    
    remaining = time_limit - (time.monotonic() - start)
    
    # Phase 4: High-beta polish (15% of time)
    t_p4 = remaining * 0.25
    if t_p4 > 15:
        a_opt, val_opt = run_polish_high_beta(best_a_local, S, t_p4, rng)
        if val_opt < best_local:
            best_local = val_opt
            best_a_local = a_opt.copy()
            obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
            print("  Polish: obj=%.6f" % obj)
    
    obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
    print("  Best for %s: obj=%.6f" % (phase_desc, obj_final))
    
    return best_a_local, best_local


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase optimization with full LSE, top-k LSE, Adam, and polishing.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 10.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, budget_s - 10))

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
    except Exception as e:
        print("Could not access height_sequence_1: %s" % str(e))
        pass

    global_best = np.inf
    global_a = None

    def try_and_update(a_candidate):
        """Evaluate and update global best."""
        nonlocal global_best, global_a
        if a_candidate is None:
            return
        n = len(a_candidate)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        s = a_candidate.sum()
        if s < 0.01:
            return
        c = fft_convolve_self(a_candidate)
        obj = 2.0 * n * c.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a_candidate.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))

    # Primary run: use given sequence length (8192)
    n_primary = 8192
    if given is not None and len(given) == n_primary:
        a_start = normalize(given.copy(), np.sqrt(2.0 * n_primary))
    elif given is not None:
        a_start = resample_sequence(given, n_primary)
        a_start = normalize(a_start, np.sqrt(2.0 * n_primary))
    else:
        a_start = np.ones(n_primary) * np.sqrt(2.0 * n_primary) / n_primary

    S_primary = np.sqrt(2.0 * n_primary)
    print("\n=== Primary run: n=%d ===" % n_primary)
    print("Initial objective: %.6f" % (2.0 * n_primary * fft_convolve_self(a_start).max() / S_primary**2))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_opt, val_opt = run_optimization_pipeline(a_start, S_primary, remaining * 0.55, rng, "n=%d" % n_primary)
        try_and_update(a_opt)

    # Secondary run: try n=16384
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_secondary = 16384
        print("\n=== Secondary run: n=%d ===" % n_secondary)
        a_sec = resample_sequence(global_a, n_secondary)
        a_sec = normalize(a_sec, np.sqrt(2.0 * n_secondary))
        S_sec = np.sqrt(2.0 * n_secondary)
        
        a_opt2, val_opt2 = run_optimization_pipeline(a_sec, S_sec, remaining * 0.30, rng, "n=%d" % n_secondary)
        try_and_update(a_opt2)

    # Tertiary: try n=4096 with different initialization
    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_tert = 4096
        print("\n=== Tertiary run: n=%d ===" % n_tert)
        if given is not None:
            a_tert = resample_sequence(given, n_tert)
        else:
            a_tert = np.ones(n_tert) * np.sqrt(2.0 * n_tert) / n_tert
        a_tert = normalize(a_tert, np.sqrt(2.0 * n_tert))
        S_tert = np.sqrt(2.0 * n_tert)
        
        a_opt3, val_opt3 = run_optimization_pipeline(a_tert, S_tert, remaining * 0.30, rng, "n=%d" % n_tert)
        try_and_update(a_opt3)

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = global_a.sum()
        print("\n=== Final Polish (%.0fs remaining) ===" % remaining)

        # High beta FISTA polish
        t_fin = remaining * 0.4
        a_fin, val_fin = run_fista_lse(global_a, S_best, t_fin, rng,
                                        use_full_lse=True,
                                        beta_init=500.0, beta_rate=0.002,
                                        beta_max=30000.0,
                                        use_momentum=False,
                                        equalize_interval=800,
                                        perturb_interval=5000,
                                        peak_target_interval=2000)
        try_and_update(a_fin)

        remaining2 = deadline - time.monotonic() - 5
        if remaining2 > 20:
            # Adam polish
            a_pol, val_pol = run_adaptive_adam(global_a, S_best, remaining2 * 0.5, rng,
                                                lr_init=0.003, beta_init=200.0,
                                                beta_rate=0.003, use_full_lse=True)
            try_and_update(a_pol)

            remaining3 = deadline - time.monotonic() - 5
            if remaining3 > 15:
                # Very careful polish
                a_pol2, val_pol2 = run_polish_high_beta(global_a, S_best, remaining3, rng)
                try_and_update(a_pol2)

    # One more restart from best with different perturbation
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = global_a.sum()
        print("\n=== Diversified Restart (%.0fs) ===" % remaining)
        
        # Perturb the best solution and re-optimize
        a_div = spectral_perturb(global_a, 0.15, rng)
        a_opt_div, val_div = run_fista_lse(a_div, S_best, remaining * 0.7, rng,
                                            use_full_lse=True,
                                            beta_init=50.0, beta_rate=0.003,
                                            beta_max=10000.0)
        try_and_update(a_opt_div)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n_fb = 8192
            global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
            final_list = [float(x) for x in global_a]
    else:
        print("No valid sequence found, using fallback.")
        n_fb = 8192
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Total time: %.1fs" % elapsed)
    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
